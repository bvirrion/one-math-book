\chapter{المتتاليات}\label{ch:b1:seq}

عولجت المتتاليات في مجلّد الثانوي ومفهوم النهاية
مأخوذٌ نصفَ أخذٍ على الثقة. وهنا يُعاد بناء النظرية على
تمام $\R$ (\cref{ch:b1:reals}): فكل مبرهنة كلاسيكية
--- التقارب الرتيب، \omterm{thm:b1:seq:adjacent}{والمتتاليتان المتجاورتان}، وبولتزانو--فايرشتراس،
ومحك كوشي --- وجهٌ لتلك البديهية الوحيدة. ويُختم الفصل
بالدراسة العملية للمتتاليات المعرَّفة بالمقدار $u_{n+1} =
f(u_n)$.

\section{التقارب}

\begin{definition}[نهاية متتالية]\label{def:b1:seq:limit}
تتقارب المتتالية $(u_n)$ من الأعداد الحقيقية إلى $\ell \in \R$
إذا كان\index{نهاية!نهاية متتالية}
\[
\forall \varepsilon > 0,\ \exists N \in \N,\ \forall n \geq N,
\qquad \abs{u_n - \ell} \leq \varepsilon .
\]
ويُكتب $u_n \to \ell$ أو $\lim u_n = \ell$. وتكون المتتالية التي لا
تتقارب (إلى أيّ عدد حقيقي) \emph{متباعدة}. والتباعد \emph{إلى
$+\infty$}: $\forall M,\ \exists N,\ \forall n \geq N,\ u_n \geq M$
(وكذلك $-\infty$).
\end{definition}

\begin{example}[برهان $\varepsilon$--$N$، مكتوبًا مرة واحدة]\label{ex:b1:seq:epsilonN}
الادعاء: $u_n = \dfrac{n^2 + 1}{2n^2 - 3} \to \dfrac12$. اعزل
الخطأ أولًا:
\[
\Bigl| u_n - \frac12 \Bigr|
= \Bigl| \frac{2(n^2 + 1) - (2n^2 - 3)}{2(2n^2 - 3)} \Bigr|
= \frac{5}{2\,\abs{2n^2 - 3}}
= \frac{5}{2\,(2n^2 - 3)} \quad (n \geq 2).
\]
ثم سيطر عليه بمقدار بسيط: من أجل $n \geq 2$، $2n^2 - 3
\geq n^2$، ومنه فالخطأ هو $\leq \frac{5}{2n^2} \leq \frac 5{2n}$.
ومن أجل $\varepsilon > 0$، تعطي \omterm{thm:b1:reals:archimedes}{خاصية أرخميدس} $N \geq
\max\bigl(2, \frac{5}{2\varepsilon}\bigr)$؛ ومن أجل $n \geq N$ يكون
الخطأ $\leq \varepsilon$. وانتهى. والفكرة النافذة: لبرهان
$\varepsilon$--$N$ ثلاث حركات بالضبط --- احسب
الخطأ، وحُدّه بمقدار أوليّ متناقص، وحلّ من أجل
العتبة --- وبعد مبرهنات هذا الفصل (العمليات
والحصر) لا يكاد المرء يكتب مثل هذا البرهان مرة أخرى: إذ تعلّب
المبرهنات الحركات الثلاث مرة واحدة وإلى الأبد.
\end{example}

\begin{example}[التباعد إلى اللانهاية، مصادَقًا عليه]\label{ex:b1:seq:toinfinity}
الادعاء: $u_n = n^2 - 100n \to +\infty$. عمّل الحد المهيمن:
$u_n = n^2\bigl(1 - \frac{100}{n}\bigr) \geq \frac{n^2}{2}$ من أجل
$n \geq 200$. ومن أجل $M$ معطى، خذ $N = \max\bigl(200,
\lceil\sqrt{2M}\rceil\bigr)$: فمن أجل $n \geq N$، $u_n \geq
\frac{n^2}{2} \geq M$. وعادتان معروضتان هنا: تعميل الحد
المهيمن يحوّل تنافسًا ($n^2$ ضد $-100n$) إلى
سلّم وحيد مضروب في عامل يؤول إلى $1$؛ وقد تكون العتبة
هائلة ($u_{100} = 0$، بل المتتالية سالبة قبل
$n = 100$) --- فالتباعد إلى $+\infty$ عبارة عن الذيل،
لا تبالي بأيّ قدر منته من سوء السلوك.
\end{example}

\begin{proposition}[الخصائص الأولى]\label{prop:b1:seq:first}
\begin{enumerate}
  \item النهاية، إذا وُجدت، وحيدة.
  \item كل متتالية متقاربة محدودة.
  \item إذا كان $u_n \to \ell$، فإن أيّ تعديل لعدد منته من الحدود
        يبقي التقارب والنهاية على حالهما.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) إذا كان $u_n \to \ell$ و $u_n \to \ell'$ مع $\ell \neq \ell'$،
فخذ $\varepsilon = \frac{\abs{\ell - \ell'}}{3}$: فبعد العتبتين
يكون $\abs{\ell - \ell'} \leq \abs{\ell - u_n} + \abs{u_n -
\ell'} \leq 2\varepsilon = \frac23 \abs{\ell - \ell'}$، وهذا محال.

(2) مع $\varepsilon = 1$: بعد $N$ يكون $\abs{u_n} \leq \abs\ell +
1$؛ والحدود السابقة المنتهية العدد محدودة كذلك، ومنه $\abs{u_n}
\leq \max(\abs{u_0}, \dots, \abs{u_{N-1}}, \abs\ell + 1)$.

(3) بالتفصيل: افترض $v_n = u_n$ من أجل $n \geq n_0$ و $u_n \to
\ell$. ومن أجل $\varepsilon > 0$، خذ العتبة $N$ الخاصة بالمتتالية
$(u_n)$: فمن أجل $n \geq \max(N, n_0)$، $\abs{v_n - \ell} = \abs{u_n
- \ell} \leq \varepsilon$. ومنه $v_n \to \ell$: إذ لا يسوّر التعريف
إلا على $n \geq N$، ويمكن الكتابة فوق أيّ بادئة منتهية
بثمن تكبير العتبة. (ولهذا تكفي في كل هذا الفصل
الفرضيات من نوع «من أجل كل $n$ كبير».)
\end{proof}

\begin{theorem}[العمليات على النهايات]\label{thm:b1:seq:operations}
إذا كان $u_n \to \ell$ و $v_n \to m$، فإن
\[
u_n + v_n \to \ell + m, \qquad u_n v_n \to \ell m, \qquad
\frac{u_n}{v_n} \to \frac{\ell}{m} \ (\text{إذا كان } m \neq 0),
\qquad \abs{u_n} \to \abs\ell .
\]
\end{theorem}

\begin{proof}
\emph{المجموع:} $\abs{(u_n + v_n) - (\ell + m)} \leq \abs{u_n - \ell} +
\abs{v_n - m} \leq 2\varepsilon$ بعد العتبة الأكبر.
\emph{الجداء:} اكتب
\[
u_n v_n - \ell m = (u_n - \ell)\,v_n + \ell\,(v_n - m);
\]
والمتتالية $(v_n)$ محدودة بمقدار $B$ ما (\cref{prop:b1:seq:first})، ومنه فالطرف
الأيمن هو $\leq B\abs{u_n - \ell} + \abs{\ell}\,\abs{v_n - m}$،
وهو صغير كما نشاء. \emph{القسمة:} يكفي معالجة $\frac
1{v_n}$. مع $\varepsilon = \frac{\abs m}{2}$: بعد $N_0$ ما،
$\abs{v_n} \geq \frac{\abs m}{2}$، ومنه
\[
\Bigl| \frac{1}{v_n} - \frac 1m \Bigr|
= \frac{\abs{m - v_n}}{\abs{v_n m}}
\leq \frac{2}{m^2}\,\abs{v_n - m} \longrightarrow 0 .
\]
\emph{القيمة المطلقة:} $\bigl|\abs{u_n} - \abs\ell\bigr| \leq
\abs{u_n - \ell}$ (متراجحة المثلث المعكوسة،
\cref{prop:b1:complex:rules}).
\end{proof}

\begin{example}[العمليات مع حيلة جبرية واحدة]\label{ex:b1:seq:conjugatetrick}
احسب $\lim\,\bigl(\sqrt{n^2 + n} - n\bigr)$. يؤول الجزآن
كلٌّ على حدة إلى $+\infty$: ولا تقول مبرهنة العمليات شيئًا
عن فرقهما (وهو \emph{شكل غير محدَّد}). اضرب
في المرافق:
\[
\sqrt{n^2 + n} - n
= \frac{(n^2 + n) - n^2}{\sqrt{n^2 + n} + n}
= \frac{n}{\sqrt{n^2+n} + n}
= \frac{1}{\sqrt{1 + \frac1n} + 1} .
\]
والآن يتقارب كل شيء: $\sqrt{1 + \frac1n} \to 1$، لأن
$0 \leq \sqrt{1 + h} - 1 = \frac{h}{\sqrt{1+h} + 1} \leq h$
(بالمرافق مرة أخرى، ثم بالحصر مع $h = \frac1n$)؛ وعندئذ
تعطي مبرهنة العمليات النهاية $\frac{1}{1 + 1} =
\frac12$. والفكرة النافذة: ليست مبرهنة العمليات
حاسبةً لكل النهايات --- فالأشكال غير المحدَّدة ($\infty -
\infty$ و $\frac00$ و $0 \times \infty$ و $1^\infty$) يجب أولًا أن
\emph{تُحوَّل} بالجبر (المرافقات، وتعميل الحد
المهيمن) حتى يتقارب كل جزء؛ والآلة المنهجية للحالات
العنيدة هي النشر المقارب في
\cref{ch:b1:taylor}.
\end{example}

\begin{theorem}[النهايات والترتيب]\label{thm:b1:seq:order}
\begin{enumerate}
  \item إذا كان $u_n \leq v_n$ من أجل كل $n$ كبير، وتقاربت المتتاليتان، فإن
        $\lim u_n \leq \lim v_n$. (وأمّا المتراجحات القطعية فلا
        تمرّ إلى النهاية: إذ $\frac 1n > 0$ لكن $\lim = 0$.)
  \item (مبرهنة الحصر\index{مبرهنة الحصر}) إذا كان $u_n \leq w_n
        \leq v_n$ من أجل كل $n$ كبير و $u_n, v_n \to \ell$، فإن
        $w_n \to \ell$.
  \item إذا كان $u_n \to \ell > 0$، فإن $u_n > \frac\ell2 > 0$ من أجل كل
        $n$ كبير.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) افترض $\ell = \lim u_n > m = \lim v_n$؛ مع $\varepsilon =
\frac{\ell - m}{3}$، تحقق الحدود الكبيرة $v_n \leq m + \varepsilon <
\ell - \varepsilon \leq u_n$، وهذا يناقض $u_n \leq v_n$.

(2) بعد العتبات: $\ell - \varepsilon \leq u_n \leq w_n \leq
v_n \leq \ell + \varepsilon$.

(3) هي \cref{def:b1:seq:limit} مع $\varepsilon = \frac\ell2$.
\end{proof}

\begin{example}[حصران]\label{ex:b1:seq:squeezes}
(أ) $\dfrac{\sin n}{n} \to 0$: انطلاقًا من $-\frac1n \leq \frac{\sin
n}{n} \leq \frac1n$، وكلا الجدارين ينهار على $0$ --- دون أيّ حاجة إلى
فهم البسط المضطرب أصلًا. (ب) $(2^n +
3^n)^{1/n} \to 3$: أطّر ما بالداخل،
\[
3^n \leq 2^n + 3^n \leq 2\cdot3^n
\quad\Longrightarrow\quad
3 \leq (2^n + 3^n)^{1/n} \leq 3\cdot 2^{1/n} ,
\]
و $2^{1/n} = \eu^{\frac{\ln 2}{n}} \to 1$ (كما في $5^{1/n}$ في
\cref{exo:b1:seq:2}): فيعطي الحصر $3$. والفكرة
النافذة: مجموع أسّيات متنافسة يسلك سلوك حدّه
\emph{الأكبر} --- إذ تبتلع العواملُ الثابتة غير المؤذية
الأصغرَ منها، ثم يمحوها الجذر من الرتبة $n$.
\end{example}

\section{المتتاليات الرتيبة}

\begin{theorem}[مبرهنة النهاية الرتيبة]\label{thm:b1:seq:monotone}
كل متتالية متزايدة محدودة من أعلى تتقارب، إلى $\sup\{u_n : n \in
\N\}$؛ وكل متتالية متزايدة غير محدودة من أعلى تتباعد إلى
$+\infty$. (وعبارتان مرآتيتان من أجل المتتاليات
المتناقصة.)\index{مبرهنة النهاية الرتيبة}
\end{theorem}

\begin{proof}
ليكن $s = \sup\{u_n\}$ (\cref{thm:b1:reals:sup}). ومن أجل $\varepsilon >
0$، يعطي التمييز بالمقدار $\varepsilon$
(\cref{prop:b1:reals:epsilon}) عتبةً $N$ تحقق $u_N > s -
\varepsilon$؛ وبالرتابة، $s - \varepsilon < u_N \leq u_n \leq
s$ من أجل كل $n \geq N$: فيتحقق التقارب إلى $s$. وإذا كانت غير محدودة: فمن أجل كل
$M$ يوجد $u_N > M$، وتُبقي الرتابة كل الحدود اللاحقة فوق $M$.
\end{proof}

\begin{example}[المبرهنة الرتيبة آلةَ وجود]\label{ex:b1:seq:existencemachine}
لتكن $u_n = \prod_{k=1}^{n} \bigl(1 + \frac{1}{2^k}\bigr)$. كل
عامل يتجاوز $1$، ومنه فالمتتالية $(u_n)$ متزايدة. فهل هي محدودة من أعلى؟ خذ
اللوغاريتمات واستعمل $\ln(1 + x) \leq x$
(\cref{ex:b1:derivative:convexineq} يستبقها؛ أو استعمل
المتراجحة الفجّة $1 + x \leq \eu^x$ من مجلّد الثانوي):
\[
\ln u_n = \sum_{k=1}^{n} \ln\Bigl(1 + \frac{1}{2^k}\Bigr)
\leq \sum_{k=1}^{n} \frac{1}{2^k} < 1 ,
\]
ومنه $u_n < \eu$. فهي متزايدة ومحدودة: ومنه تتقارب $(u_n)$ إلى
$\ell \in \intoc{u_1}{\eu}$ ما --- وهو عدد حقيقي معرَّف تعريفًا تامًا
دون أيّ صورة مغلقة تلوح في الأفق ($\ell = 2.384\dots$). والفكرة
النافذة: مبرهنة النهاية الرتيبة أرخص آلة وجود
في التحليل؛ فقد سمّت $\eu$ نفسه
(\cref{ex:b1:seq:e} أدناه)، وفي \cref{ch:b1:series} سوف
تفصل في تقارب كل متسلسلة موجبة بمجرد
الحدّية.
\end{example}

\begin{theorem}[المتتاليتان المتجاورتان]\label{thm:b1:seq:adjacent}
لتكن $(a_n)$ متزايدة و $(b_n)$ متناقصة، مع $b_n - a_n \to
0$. عندئذ تتقارب المتتاليتان إلى نهاية \emph{مشتركة} $\ell$، ويكون $a_n
\leq \ell \leq b_n$ من أجل كل $n$.\index{متتاليتان متجاورتان}
\end{theorem}

\begin{proof}
أولًا، $a_n \leq b_n$ من أجل كل $n$: فالمتتالية $(b_n - a_n)$
متناقصة وتؤول إلى $0$، ومنه فهي $\geq 0$ (إذ إن حدًّا سالبًا سيجمّدها
دون $0$). ثم إن $(a_n)$ متزايدة ومحدودة من أعلى بالمقدار
$b_0$: فتتقارب إلى $\ell$ ما (\cref{thm:b1:seq:monotone})؛
وكذلك $(b_n) \to \ell'$؛ و $\ell' - \ell = \lim (b_n - a_n) =
0$. وتتبع المتراجحات $a_n \leq \ell \leq b_n$ من
الرتابة ($\ell = \sup a_k \geq a_n$ وهكذا).
\end{proof}

\begin{example}[العدد $\eu$]\label{ex:b1:seq:e}
ضع $a_n = \sum_{k=0}^{n} \frac{1}{k!}$ و $b_n = a_n + \frac{1}{n
\cdot n!}$ ($n \geq 1$). عندئذ تتزايد $(a_n)$؛ و
\[
b_{n+1} - b_n = \frac{1}{(n+1)!} + \frac{1}{(n+1)(n+1)!} -
\frac{1}{n\,n!}
= \frac{n(n+1) + n - (n+1)^2}{n(n+1)(n+1)!}
= \frac{-1}{n(n+1)(n+1)!} < 0 ,
\]
ومنه تتناقص $(b_n)$، و $b_n - a_n \to 0$: فهما متجاورتان. ونهايتهما
المشتركة هي (بالتعريف هنا) العدد $\eu \approx 2.71828$؛ والمتراجحتان
$a_n < \eu < b_n$ دقيقتان بما يكفي للبرهان على $\eu \notin
\Q$ (\cref{exo:b1:seq:9}).
\end{example}

\section{المتتاليات الجزئية وبولتزانو--فايرشتراس}

\begin{definition}[متتالية جزئية]\label{def:b1:seq:subsequence}
\emph{المتتالية الجزئية}\index{متتالية جزئية} للمتتالية $(u_n)$ هي متتالية
$(u_{\varphi(n)})$ حيث $\varphi \colon \N \to \N$ متزايدة
قطعًا (ولاحظ $\varphi(n) \geq n$، بالاستقراء).
\end{definition}

\begin{proposition}\label{prop:b1:seq:subsequences}
إذا كان $u_n \to \ell$ ($\ell \in \R$ أو $\pm\infty$)، فإن كل \omterm{def:b1:seq:subsequence}{متتالية جزئية}
تؤول إلى $\ell$. ونتيجةً لذلك، تتباعد كل متتالية لها متتاليتان جزئيتان
مختلفتا النهاية. وبالعكس، إذا تقاربت $(u_{2n})$ و
$(u_{2n+1})$ كلتاهما إلى \emph{النهاية نفسها} $\ell$، فإن $u_n \to
\ell$.
\end{proposition}

\begin{proof}
بعد العتبة $N$ الخاصة بالمتتالية $(u_n)$، تفي كل الأدلة $\varphi(n) \geq n
\geq N$ بالغرض (والمتراجحة $\varphi(n) \geq n$ هي
الاستقراء المشار إليه في \cref{def:b1:seq:subsequence}: $\varphi(0)
\geq 0$، ويفرض $\varphi(n+1) > \varphi(n) \geq n$ أن
$\varphi(n+1) \geq n + 1$). وأمّا العكس: فمن أجل
$\varepsilon$، خذ العتبتين $N_0$ (الأدلة الزوجية) و $N_1$
(الفردية)؛ فأيّ دليل $n \geq \max(2N_0, 2N_1 + 1)$ يكون
إمّا زوجيًا، $n = 2k$ مع $k \geq N_0$، وإمّا فرديًا، $n = 2k+1$ مع
$k \geq N_1$ --- وفي الحالتين $\abs{u_n - \ell} \leq
\varepsilon$: فكل دليل مغطّى بإحدى المتتاليتين
الجزئيتين، وذلك هو كل المقصود.
\end{proof}

\begin{example}[نهايات المتتاليات الجزئية]\label{ex:b1:seq:sublimits}
من أجل $u_n = (-1)^n \frac{n}{n+1}$: تؤول \omterm{def:b1:seq:subsequence}{المتتالية الجزئية} الزوجية إلى
$1$، والفردية إلى $-1$، ومنه تتباعد المتتالية --- لكنها تفعل
ذلك بكيفية منظَّمة، متكدّسة حول القيمتين $\pm 1$.
ومن أجل $u_n = \cos\frac{2\pi n}{3}$: المتتاليات الجزئية الثلاث ذات
الأدلة $3k$ و $3k + 1$ و $3k + 2$ ثابتة، تساوي $1$ و
$-\frac12$ و $-\frac12$؛ \omterm{def:b1:logic:sets}{ومجموعة} نهايات المتتاليات الجزئية هي $\{1,
-\frac12\}$. والفكرة النافذة: تتقارب متتالية \emph{محدودة}
إذا وفقط إذا كان لها نهاية جزئية وحيدة
(\cref{exo:b1:seq:8})؛ وتباعد متتالية محدودة يعني دائمًا
تكدّسين على الأقل، وتضمن بولتزانو--فايرشتراس أدناه
وجود واحد على الأقل.
\end{example}

\begin{theorem}[بولتزانو--فايرشتراس]\label{thm:b1:seq:bw}
لكل متتالية محدودة من الأعداد الحقيقية \omterm{def:b1:seq:subsequence}{متتالية جزئية}
متقاربة.\index{مبرهنة بولتزانو--فايرشتراس}
\end{theorem}

\begin{proof}
ليكن $u_n \in \intcc{a}{b}$ من أجل كل $n$. ابنِ قطعًا متداخلة
بالثنائية: ضع $\intcc{a_0}{b_0} = \intcc{a}{b}$؛ ومن أجل
$\intcc{a_k}{b_k}$ تحتوي $u_n$ من أجل عدد لا يُحصى من $n$، يبقى أحد
نصفيها محتويًا $u_n$ من أجل عدد لا يُحصى من $n$ --- ولنسمّه
$\intcc{a_{k+1}}{b_{k+1}}$. والمتتاليتان $(a_k)$ و $(b_k)$
متجاورتان ($b_k - a_k = \frac{b-a}{2^k} \to 0$)، ونهايتهما المشتركة
$\ell$ (\cref{thm:b1:seq:adjacent}).

والاستخراج: اختر $\varphi(0)$ يحقق $u_{\varphi(0)} \in
\intcc{a_0}{b_0}$، ثم، بالاستقراء، $\varphi(k+1) > \varphi(k)$
مع $u_{\varphi(k+1)} \in \intcc{a_{k+1}}{b_{k+1}}$ --- وهذا ممكن
لأن تلك القطعة تحتوي عددًا لا يُحصى من الحدود. عندئذ $a_k \leq
u_{\varphi(k)} \leq b_k$، وتعطي مبرهنة الحصر
$u_{\varphi(k)} \to \ell$.
\end{proof}

\begin{remark}[ما تقوله وما لا تقوله مبرهنة بولتزانو--فايرشتراس]
إنها \emph{تقول}: انطلاقًا من الحدّية وحدها، تتقارب \omterm{def:b1:seq:subsequence}{متتالية
جزئية} ما --- وجودٌ بلا صيغة، كما يوضّح برهان الثنائية
(فلا شيء يخبرنا \emph{أيّ} الأدلة يبقى). وهي
\emph{لا} تقول إن النهاية وحيدة: فللمتتالية $((-1)^n)$
متتاليتان جزئيتان تتقاربان إلى $1$ وإلى $-1$، بل قد تكون \omterm{def:b1:logic:sets}{مجموعة}
نهايات المتتاليات الجزئية غير منتهية
(\cref{ex:b1:seq:sublimits}، \omterm{def:b1:logic:sets}{ومجموعة} كانتور كلها في
\cref{pb:b1:topology:1}). وهي لا تصمد أمام عدم الحدّية:
فليس للمتتالية $(n)$ أيّ \omterm{def:b1:seq:subsequence}{متتالية جزئية} متقاربة --- وإن كان يمكن
دائمًا استخراج \omterm{def:b1:seq:subsequence}{متتالية جزئية} تؤول إلى $+\infty$ أو $-\infty$
من أيّ متتالية غير محدودة (اختر $\varphi(k)$ يحقق
$u_{\varphi(k)} \geq k$، مثلًا). وباستعمالها استعمالًا صحيحًا، تكون المبرهنة
\emph{مضخّة وجود}: فهي تظهر في عقدة محك كوشي
أدناه، وفي مبرهنة هاينه، وفي مبرهنة القيم
الحدّية --- ودائمًا لإنتاج نقطة لا يقدّمها أيّ بناء
صريح.
\end{remark}

\section{متتاليات كوشي والتمام}

\begin{definition}\label{def:b1:seq:cauchy}
تكون المتتالية $(u_n)$ \emph{متتالية كوشي}\index{متتالية كوشي}
إذا صارت حدودها قريبة كما نشاء \emph{بعضها من بعض}:
\[
\forall \varepsilon > 0,\ \exists N,\ \forall p, q \geq N,
\qquad \abs{u_p - u_q} \leq \varepsilon .
\]
\end{definition}

\begin{example}[التحقق من خاصية كوشي باليد]\label{ex:b1:seq:cauchyhand}
لتكن $u_n = \sum_{k=0}^{n} \frac{\cos k}{2^k}$ --- فلا رتابة،
ولا نهاية يمكن تخمينها. من أجل $p > q$:
\[
\abs{u_p - u_q}
= \Bigl| \sum_{k=q+1}^{p} \frac{\cos k}{2^k} \Bigr|
\leq \sum_{k=q+1}^{p} \frac{1}{2^k}
< \frac{1}{2^{q}} ,
\]
بمتراجحة المثلث، و $\abs{\cos k} \leq 1$، ومجموع
هندسي منته. ومن أجل $\varepsilon > 0$، اختر $N$ يحقق $2^{-N}
\leq \varepsilon$: فكل الفروق بعد $N$ هي $\leq \varepsilon$،
والمتتالية \omterm{def:b1:seq:cauchy}{متتالية كوشي}، ومنه تتقارب --- إلى نهاية لا يستطيع
أحد تسميتها في صورة مغلقة، وذلك بالضبط هو المقصود. والفكرة
النافذة: السيطرة الهندسية على التزايدات هي الطريقة المعيارية
لكسب خاصية كوشي، وسوف يعبّئ \cref{ch:b1:series}
الحجة في صورة «التقارب بإطلاق يستلزم
التقارب».
\end{example}

\begin{theorem}[تمام $\R$]\label{thm:b1:seq:complete}
تتقارب متتالية من الأعداد الحقيقية إذا وفقط إذا كانت \omterm{def:b1:seq:cauchy}{متتالية
كوشي}.\index{تمام R@تمام $\R$}
\end{theorem}

\begin{proof}
($\Rightarrow$) إذا كان $u_n \to \ell$: فبعد العتبة الخاصة بالمقدار
$\frac\varepsilon2$، $\abs{u_p - u_q} \leq \abs{u_p - \ell} +
\abs{\ell - u_q} \leq \varepsilon$.

($\Leftarrow$) لتكن $(u_n)$ \omterm{def:b1:seq:cauchy}{متتالية كوشي}. \emph{فهي محدودة}: إذ مع
$\varepsilon = 1$، تقع كل الحدود بعد $N$ على بعد $1$ من $u_N$،
والرأس منته. \emph{والاستخراج}: حسب
\cref{thm:b1:seq:bw}، تتقارب \omterm{def:b1:seq:subsequence}{متتالية جزئية} ما $u_{\varphi(n)} \to \ell$.
\emph{والاستنتاج}: من أجل $\varepsilon > 0$، خذ $N$ (من خاصية كوشي، من أجل
$\frac\varepsilon2$) و $n \geq N$ يحققان $\abs{u_{\varphi(n)} - \ell}
\leq \frac\varepsilon2$ و $\varphi(n) \geq N$؛ عندئذ من أجل كل $p
\geq N$:
\[
\abs{u_p - \ell} \leq \abs{u_p - u_{\varphi(n)}} +
\abs{u_{\varphi(n)} - \ell} \leq \varepsilon . \qedhere
\]
\end{proof}

\begin{remark}
قيمة المحك أنه يصادق على التقارب \emph{دون
تسمية النهاية}. وهو يفشل على $\Q$ (فالاقتطاعات العشرية للمقدار
$\sqrt 2$ تكوّن \omterm{def:b1:seq:cauchy}{متتالية كوشي} من الأعداد الناطقة بلا نهاية ناطقة):
فالتمام خاصية للمقدار $\R$، مكافئة لبديهية \omterm{def:b1:reals:bounds}{الحد
الأعلى}. وهو كذلك حصان العمل وراء تقارب
المتسلسلات (\cref{ch:b1:series}).
\end{remark}

\begin{example}[متتالية كوشي نهايتها غير مرئية]\label{ex:b1:seq:basel}
لتكن $S_n = \sum_{k=1}^{n} \frac{1}{k^2}$. من أجل $p > q \geq 1$:
\[
S_p - S_q = \sum_{k=q+1}^{p} \frac{1}{k^2}
\leq \sum_{k=q+1}^{p} \frac{1}{k(k-1)}
= \sum_{k=q+1}^{p} \Bigl(\frac{1}{k-1} - \frac 1k\Bigr)
= \frac 1q - \frac 1p < \frac 1q ,
\]
ومنه فبعد $N > \frac1\varepsilon$ تكون كل الفروق $\leq \varepsilon$:
فالمتتالية $(S_n)$ \omterm{def:b1:seq:cauchy}{متتالية كوشي}، ومنه تتقارب. ولاحظ ما حدث للتوّ: فقد
برهنّا على أن عددًا حقيقيًا معيّنًا موجود دون أن يكون لنا أيّ اسم
له. (وهو $\frac{\pi^2}{6}$ --- متطابقة شهيرة لأويلر،
مبرهنٌ عليها في مجلّد السنة الثانية؛ ولا شيء في هذا الفصل يستطيع
أن يخبرنا بذلك.) وتقسيم العمل هذا --- الوجود الآن،
والتعيين لاحقًا، إن كان --- هو كل مقصود محك كوشي،
ومحرّك نظرية المتسلسلات في
\cref{ch:b1:series}.
\end{example}

\section{المتتاليات التراجعية}

\begin{method}[دراسة $u_{n+1} = f(u_n)$]\label{met:b1:seq:recurrent}
من أجل $f$ ونقطة انطلاق $u_0$ معطاتين:\index{متتالية تراجعية}
\begin{enumerate}
  \item \emph{\omterm{prop:b1:reals:intervals}{فترة} مستقرة:} جد \omterm{prop:b1:reals:intervals}{فترة} $I$ تحقق $f(I)
        \subseteq I$ وتحتوي $u_0$: عندئذ يكون كل $u_n \in I$ (بالاستقراء).
  \item \emph{النهايات المرشّحة:} إذا كان $u_n \to \ell \in I$ وكانت $f$
        متصلة عند $\ell$ (\cref{ch:b1:continuity})، فإن $\ell$
        \emph{نقطة صامدة}: $f(\ell) = \ell$. حُلَّ $f(x) = x$.
  \item \emph{الرتابة:} إذا كانت $f$ متزايدة على $I$، فإن
        $(u_n)$ رتيبة (متزايدة إذا كان $u_1 \geq u_0$،
        ومتناقصة فيما عدا ذلك)؛ ومع الحدّية،
        يختم \cref{thm:b1:seq:monotone} الأمر. وإذا كانت $f$ متناقصة،
        فادرس المتتاليتين الجزئيتين $(u_{2n})$ و $(u_{2n+1})$،
        وهما رتيبتان من أجل $f \circ f$.
  \item \emph{ضبط الخطأ:} تعطي متراجحة $\abs{f(x) - \ell} \leq
        k\abs{x - \ell}$ مع $k < 1$ المقدارَ $\abs{u_n - \ell} \leq
        k^n \abs{u_0 - \ell} \to 0$ مباشرة.
\end{enumerate}
\end{method}

\begin{example}[طريقة هيرون]\label{ex:b1:seq:heron}
ليكن $u_0 = 2$ و $u_{n+1} = \dfrac12\Bigl(u_n +
\dfrac{2}{u_n}\Bigr)$: وهي الخوارزمية القديمة من أجل
$\sqrt 2$.\index{طريقة هيرون}
\begin{itemize}
  \item \emph{الاستقرار:} من أجل $x > 0$، تعطي متراجحة المتوسطين
        الحسابي والهندسي أن $\frac12(x + \frac2x) \geq \sqrt{x \cdot
        \frac 2x} = \sqrt 2$؛ ومنه \omterm{prop:b1:reals:intervals}{فالفترة} $I = \intco{\sqrt 2}{+\infty}$
        مستقرة وتحتوي $u_1$ (وفعلًا $u_1 = \frac32 \geq
        \sqrt 2$).
  \item \emph{الرتابة:} من أجل $x \geq \sqrt 2$، $\;x - f(x) =
        \frac{x^2 - 2}{2x} \geq 0$: فتتناقص المتتالية ابتداءً من
        $u_1$، وهي محدودة من أدنى بالمقدار $\sqrt 2$: ومنه تتقارب.
  \item \emph{النهاية:} تحلّ النقاط الصامدة $x = \frac12(x +
        \frac2x)$، أي $x^2 = 2$: وعلى $I$، $\ell = \sqrt 2$.
  \item \emph{السرعة:} $u_{n+1} - \sqrt 2 = \frac{(u_n -
        \sqrt2)^2}{2u_n} \leq \frac{(u_n - \sqrt 2)^2}{2\sqrt 2}$:
        فعدد الأرقام الصحيحة \emph{يتضاعف} تقريبًا في كل
        خطوة (تقارب تربيعي).
\end{itemize}
\end{example}

\begin{remark}[مزالق شائعة مع النهايات]
أربعة كلاسيكية. (أ) \emph{الخطوات الصغيرة لا تستلزم التقارب}:
فالشرط $u_{n+1} - u_n \to 0$ أضعف بكثير من خاصية كوشي ---
إذ للمجاميع التوافقية $H_n$ خطوات $\frac{1}{n+1} \to 0$ ومع ذلك
تتباعد إلى $+\infty$ (\cref{exo:b1:seq:5})؛ فشرط كوشي
يضبط $\abs{u_p - u_q}$ من أجل \emph{كل} الأزواج الكبيرة، لا
المتتالية منها. (ب) \emph{المتراجحات القطعية تموت في
النهاية}: فمن $u_n < v_n$ من أجل كل $n$ لا يُحصَّل إلا $\lim u_n
\leq \lim v_n$ (\cref{thm:b1:seq:order})؛ إذ $\frac1n > 0$ ومع ذلك
$\lim = 0$. (ج) \emph{المحدودة ليست متقاربة}: فالمتتالية $((-1)^n)$
محدودة وتتباعد؛ والحدّية مع \emph{الرتابة}
تتقارب، وأمّا الحدّية وحدها فلا تضمن إلا \omterm{def:b1:seq:subsequence}{متتالية جزئية}
متقاربة (\cref{thm:b1:seq:bw}). (د) \emph{معادلة النقطة
الصامدة تأتي ثانيةً لا أولًا}: فمن أجل $u_{n+1} = f(u_n)$،
لا يعيّن حلُّ $f(\ell) = \ell$ النهايةَ \emph{إلا بعد}
البرهان على التقارب. فللعلاقة التراجعية $u_{n+1} = 2u_n$
نقطة صامدة وحيدة $\ell = 0$، ومع ذلك تجري المتتالية من $u_0 = 1$
إلى $+\infty$: فلم تكن المعادلة $\ell = 2\ell$ مؤهلة
لنهاية قط. فترتيب العمليات دائمًا: الوجود
أولًا (\cref{met:b1:seq:recurrent}، الخطوات 1--3)، والتعيين
ثانيًا.
\end{remark}

\begin{example}[دالة $f$ متناقصة: العلاقة التراجعية الذهبية]\label{ex:b1:seq:golden}
ليكن $u_0 = 1$ و $u_{n+1} = \dfrac{1}{1 + u_n}$. وهنا $f(x) =
\frac{1}{1+x}$ \emph{متناقصة}، ومنه فالمتتالية ليست
رتيبة (بل تتناوب حول نهايتها)؛ وخطوة التقلّص
في \cref{met:b1:seq:recurrent} هي الأداة الصحيحة. الاستقرار: إذا كان
$x \in \intcc{\frac12}{1}$ فإن $1 + x \in
\intcc{\frac32}{2}$، ومنه $f(x) \in \intcc{\frac12}{\frac23}
\subseteq \intcc{\frac12}{1}$، ويضع $u_1 = \frac12$ المتتالية
كلها هناك. والنقطة الصامدة: يعطي $\ell = \frac{1}{1+\ell}$
مع $\ell > 0$ أن $\ell^2 + \ell - 1 = 0$، أي
\[
\ell = \frac{\sqrt5 - 1}{2} = 0.6180\dots
\]
(وهو مقلوب النسبة الذهبية). والتقلّص: من أجل $x, y \in
\intcc{\frac12}{1}$،
\[
\abs{f(x) - f(y)} = \frac{\abs{x - y}}{(1+x)(1+y)}
\leq \frac{\abs{x-y}}{(3/2)^2} = \frac49\,\abs{x - y} ,
\]
ومنه $\abs{u_n - \ell} \leq \bigl(\frac49\bigr)^{n-1}\abs{u_1 -
\ell} \to 0$: أي التقارب، بسرعة هندسية، دون
حاجة إلى رتابة. والفكرة النافذة: تقتسم الطرق الرتيبة وطرق
التقلّص العالمَ التراجعي بينها ---
فالدالة $f$ المتزايدة تعطي مدارات رتيبة، و $f$ المتناقصة تعطي
مدارات متناوبة يروّضها ثابت ليبشيتز $< 1$ (والنظرية
المنهجية في \cref{exo:b1:derivative:11}).
\end{example}

\begin{remark}[منظورات داخل هذا المجلّد]
المتتاليات هي أداة القياس التي يرفعها بقية المجلّد
في وجه كل غرض. ففي \cref{ch:b1:topology} هي
\emph{تميّز} الانغلاق والتراص؛ وفي
\cref{ch:b1:continuity} تنقل نهايات الدوال؛ وفي
\cref{ch:b1:integration} تكون مجاميع ريمان متتاليات متقاربة
إلى التكامل؛ و \cref{ch:b1:series} \emph{هو} نظرية صنف
خاص واحد من المتتاليات، هو المجاميع الجزئية. بل حتى فصول الجبر
تستهلكها: فتكرارات مصفوفة في
\cref{ch:b1:matrices} تكوّن متتاليات سلوكها
(تقارب $A^n$) سؤال في الجبر الخطي بمفردات هذا
الفصل. والمبرهنتان اللتان تُحملان في كل مكان:
النهاية الرتيبة (وجودٌ من الترتيب) وبولتزانو--فايرشتراس
(وجودٌ من الحدّية) --- وبينهما تولد كل نهاية
في هذا الكتاب تقريبًا.
\end{remark}

\begin{remark}[المتتاليات العقدية]
تتقارب متتالية $(z_n)$ من الأعداد العقدية إلى $\ell$ إذا كان
$\abs{z_n - \ell} \to 0$؛ وبكيفية مكافئة، إذا كان $\Re(z_n) \to \Re(\ell)$
و $\Im(z_n) \to \Im(\ell)$ (قارن $\abs{z}$ بالمقدارين $\abs{\Re z} +
\abs{\Im z}$). وأمّا المبرهنات التي لا تتدخّل فيها بنية الترتيب --- العمليات،
وبولتزانو--فايرشتراس (استخرج مرتين)، ومحك كوشي --- فتنتقل
حرفيًا.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:seq:1}
انطلاقًا من \cref{def:b1:seq:limit} مباشرة، برهن على أن
$\dfrac{2n+1}{n+3} \to 2$، وعلى أن $(u_n) = ((-1)^n)$ متباعدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:seq:2}
احسب النهايات:
\[
\frac{n^2 - 3n + 1}{2n^2 + 5},
\qquad
\sqrt{n+1} - \sqrt n,
\qquad
\frac{2^n + n^3}{3^n - n^2},
\qquad
\sqrt[n]{5}\ \Bigl(= 5^{1/n}\Bigr).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:seq:3}
برهن على المقارنة المعيارية: إذا كان $\abs{q} < 1$ فإن $q^n \to 0$
\emph{(اكتب $\frac{1}{\abs q} = 1 + h$ مع $h > 0$، واستعمل
متراجحة برنولي $(1+h)^n \geq 1 + nh$، ويُبرهن عليها
بالاستقراء)}. وما السلوكات من أجل $q = 1$ و $q = -1$ و $\abs q >
1$؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:seq:4}
لتكن $u_{n+1} = \frac{u_n + 3}{2}$ مع $u_0 = 0$. جد النقطة الصامدة
$\ell$، وبرهن على أن $v_n = u_n - \ell$ هندسية، وأعطِ صيغة
صريحة ونهاية المتتالية $(u_n)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:seq:5}
(المتسلسلة التوافقية) لتكن $H_n = \sum_{k=1}^{n} \frac 1k$. برهن على أن
$H_{2n} - H_n \geq \frac12$ من أجل كل $n \geq 1$، واستنتج أن
$(H_n)$ \emph{ليست} \omterm{def:b1:seq:cauchy}{متتالية كوشي}، ومنه تتباعد (إلى
$+\infty$، لكونها متزايدة).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:seq:6}
افترض أن $(u_{2n})$ و $(u_{2n+1})$ و $(u_{3n})$ تتقارب كلها. برهن
على أن $(u_n)$ تتقارب. \emph{(جد متتاليات جزئية مشتركة لتساوي بين
النهايات.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:seq:7}
ادرس المتتالية $u_0 = 0$، $u_{n+1} = \sqrt{2 + u_n}$:
الاستقرار والرتابة والنهاية. ثم برهن على حاصر الخطأ
$\abs{u_n - 2} \leq \dfrac{2}{3^{\,n}}$ \emph{(بيّن $2 - u_{n+1} =
\dfrac{2 - u_n}{2 + \sqrt{2 + u_n}}$ وحُدّ المقام من أدنى
بالعدد $3$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:seq:8}
لتكن $(u_n)$ محدودة، بحيث تكون لكل \omterm{def:b1:seq:subsequence}{متتالية جزئية} متقاربة من
$(u_n)$ \emph{النهاية نفسها} $\ell$. برهن على $u_n \to \ell$.
\emph{(بالخلف مع بولتزانو--فايرشتراس.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:seq:9}
بترميز \cref{ex:b1:seq:e}، افترض $\eu = \frac pq$
مع $p, q \in \N^*$. باستعمال $a_q < \eu < b_q = a_q + \frac{1}{q\,
q!}$، اضرب في $q!$ واستخرج تناقضًا بين عددين
صحيحين. واستنتج: العدد $\eu$ أصمّ.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:seq:10}
(متوسطات تشيزارو) من أجل متتالية $(u_n)_{n \geq 1}$، ضع $c_n =
\frac{u_1 + \dots + u_n}{n}$.
\begin{enumerate}
  \item برهن على أن $u_n \to \ell$ يستلزم $c_n \to \ell$ \emph{(اقطع
        المجموع عند عتبة $N$؛ وحُدّ الرأس بمقدار ثابت
        مقسوم على $n$، والذيل بالمقدار $\varepsilon$)}.
  \item بيّن بمثال أن العكس يفشل.
  \item استنتج أنه إذا كان $u_{n+1} - u_n \to \ell$، فإن $\frac{u_n}{n}
        \to \ell$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:seq:11}
لتحقق $(u_n)$ الشرط $0 \leq u_{m+n} \leq u_m + u_n$ من أجل كل $m, n$
(شبه الجمعية). برهن على أن $\bigl(\frac{u_n}{n}\bigr)$ تتقارب إلى
$\inf_{n \geq 1} \frac{u_n}{n}$. \emph{(من أجل $m$ مثبَّت، اكتب $n = qm
+ r$ وحُدّ $\frac{u_n}{n}$ مستعملًا $u_n \leq q\,u_m + u_r$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:seq:12}
باستعمال كثافة \omterm{def:b1:structures:subgroup}{الزمرة الجزئية} $\Z + 2\pi\Z$ \omterm{def:b1:structures:group}{للزمرة} $(\R, +)$
(\cref{exo:b1:reals:9})، برهن على أن المتتالية $(\sin n)_{n \in
\N}$ كثيفة في $\intcc{-1}{1}$ --- وعلى الخصوص أنها متباعدة.
\end{exercise}

\section{مسألة: تشيزارو وستولتز والسقوط البطيء للجيب}

\begin{problem}[{مسألة نهاية الأسبوع --- مبرهنة تشيزارو--ستولتز
والمقاربة $u_n \sim \sqrt{3/n}$ من أجل $u_{n+1} = \sin
u_n$}]
\label{pb:b1:seq:1}
مبرهنة تشيزارو--ستولتز هي قاعدة لوبيتال المتقطّعة: فلإيجاد
نهاية قسمة $a_n/b_n$، يكفي إيجاد
نهاية قسمة \emph{التزايدات} $(a_{n+1} -
a_n)/(b_{n+1} - b_n)$. وتبرهن هذه المسألة على المبرهنة، وتجني
بها نهايات كلاسيكية، ثم تصوّبها نحو هدف شهير:
المتتالية $u_{n+1} = \sin u_n$، التي تزحف إلى $0$ بالسرعة
القابلة للحساب بالضبط $u_n \sim \sqrt{3/n}$.\index{مبرهنة تشيزارو--ستولتز}
وتُمنح هنا واقعتان من مجلّد الثانوي
ويُعاد البرهان عليهما بأمانة لاحقًا في هذا
المجلّد: متراجحة المماس
\[
\tag{G1} \eu^{u} \geq 1 + u \quad (u \in \R),
\]
المبرهن عليها ثانيةً بالتحدّب في \cref{ch:b1:derivative}، وتأطير
الجيب
\[
\tag{G2} x - \frac{x^3}{6} \;\leq\; \sin x \;\leq\; x -
\frac{x^3}{6} + \frac{x^5}{120} \quad (0 \leq x \leq 1), \qquad
\abs{\sin x} \leq \abs{x} \quad (x \in \R),
\]
المبرهن عليه ثانيةً بصيغة تايلور في \cref{ch:b1:taylor}.

\medskip
\noindent\textbf{الجزء 1 --- مجاميع بلا صيغ مغلقة.}
\begin{enumerate}
  \item باستعمال $1 + 2 + \dots + n = \frac{n(n+1)}{2}$ و $1^2 +
        \dots + n^2 = \frac{n(n+1)(2n+1)}{6}$، احسب
        $\lim \frac{1 + 2 + \dots + n}{n^2}$ و $\lim \frac{1^2 +
        \dots + n^2}{n^3}$.
  \item لتكن $T_n = \sum_{k=1}^n \sqrt k$، ولا توجد لها أيّ صيغة
        مغلقة. برهن على التأطير
        \[
        \frac{1}{2\sqrt 2}\,n^{3/2} \;\leq\; T_n \;\leq\;
        n^{3/2}
        \]
        \emph{(احتفظ من أجل الحاصر الأدنى بالحدود $k > \frac n2$
        فقط)}. ومنه فرتبة $T_n$ هي $n^{3/2}$ --- لكن بأيّ
        ثابت؟ أرجئ السؤال حتى السؤال 8.
  \item (مبرهنة التلسكوب المساعدة) لتكن $(b_n)$ متزايدة قطعًا
        ولنفترض أنه من أجل كل $k \geq N$،
        \[
        m \;\leq\; \frac{a_{k+1} - a_k}{b_{k+1} - b_k} \;\leq\;
        M .
        \]
        برهن على أن $m \leq \dfrac{a_n - a_N}{b_n - b_N} \leq M$
        من أجل كل $n > N$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- مبرهنة تشيزارو--ستولتز.} لتكن
$(b_n)$ متزايدة قطعًا مع $b_n \to +\infty$،
ولنفترض $\dfrac{a_{n+1} - a_n}{b_{n+1} - b_n} \to \ell \in \R$.
\begin{enumerate}[resume]
  \item ثبّت $\varepsilon > 0$. بيّن أنه يوجد $N$ يحقق
        $\ell - \varepsilon \leq \dfrac{a_n - a_N}{b_n - b_N}
        \leq \ell + \varepsilon$ من أجل كل $n > N$.
  \item أثبت، من أجل $n > N$، المتطابقة
        \[
        \frac{a_n}{b_n} - \ell
        = \frac{a_N - \ell\,b_N}{b_n}
        + \Bigl(1 - \frac{b_N}{b_n}\Bigr)
        \Bigl(\frac{a_n - a_N}{b_n - b_N} - \ell\Bigr),
        \]
        واستنتج المبرهنة: $\dfrac{a_n}{b_n} \to \ell$.
  \item برهن على صيغة $+\infty$: إذا كان $\dfrac{a_{n+1} -
        a_n}{b_{n+1} - b_n} \to +\infty$ (بالفرضيات نفسها على
        $(b_n)$)، فإن $\dfrac{a_n}{b_n} \to +\infty$.
  \item خذ $b_n = n$: فتستعيد مبرهنة متوسط تشيزارو في
        \cref{exo:b1:seq:10}. ثم بيّن أن عكس
        تشيزارو--ستولتز يفشل: فمن أجل $a_n = (-1)^n$ و $b_n = n$، تتقارب
        قسمة $a_n/b_n$ بينما لا تتقارب قسمة
        التزايدات. فستولتز طريق باتجاه واحد.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- الأرباح الأولى.}
\begin{enumerate}[resume]
  \item برهن على $(1+h)^{3/2} - 1 = \dfrac{3h + 3h^2 +
        h^3}{(1+h)^{3/2} + 1}$ بالمرافقة، واستنتج
        $n\bigl((1 + \tfrac1n)^{3/2} - 1\bigr) \to \tfrac32$،
        واختم بتشيزارو--ستولتز:
        \[
        T_n = \sum_{k=1}^{n} \sqrt k \;\sim\; \tfrac23\,
        n^{3/2} ,
        \]
        فتحلّ عقدة السؤال 2.
  \item انطلاقًا من (G1) وحدها، استخرج تأطير اللوغاريتم
        \[
        \frac{t}{1 + t} \;\leq\; \ln(1 + t) \;\leq\; t
        \qquad (t > -1)
        \]
        \emph{(طبّق (G1) عند $u = \ln(1+t)$ وعند $u =
        -t/(1+t)$)}.
  \item بيّن أن $b_n = \ln n$ متزايدة قطعًا مع
        $\ln n \to +\infty$، وبرهن بتشيزارو--ستولتز و
        السؤال 9 على أن
        \[
        H_n = \sum_{k=1}^{n} \frac 1k \;\sim\; \ln n .
        \]
        (والبنية الأدقّ $H_n = \ln n + \gamma + o(1)$ هي
        مسألة نهاية الأسبوع في \cref{ch:b1:series}.)
  \item (من النسب إلى الجذور) لتكن $u_n > 0$ مع
        $\frac{u_{n+1}}{u_n} \to L > 0$. باستعمال السؤال 9، بيّن
        $\ln\frac{u_{n+1}}{u_n} \to \ln L$؛ وطبّق تشيزارو
        لتستنتج $\frac{\ln u_n}{n} \to \ln L$، ثم، مع (G1)،
        أن $u_n^{1/n} \to L$. وتطبيقًا: احسب
        $\lim\,\binom{2n}{n}^{1/n}$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- السقوط البطيء للجيب.} ليكن
$u_0 \in \R$ و $u_{n+1} = \sin u_n$.
\begin{enumerate}[resume]
  \item انطلاقًا من (G2)، بيّن $0 < \sin x < x$ من أجل $0 < x \leq 1$.
        واستنتج: $u_1 \in \intcc{-1}{1}$؛ وإذا كان $u_1 = 0$ فالمتتالية
        معدومة ابتداءً من الرتبة $1$؛ وإذا كان $u_1 > 0$ (وحالة
        $u_1 < 0$ متناظرة، لأن $\sin$ فردية)، فإن
        $(u_n)_{n \geq 1}$ متناقصة قطعًا وموجبة، و
        تتقارب إلى $0$ \emph{(عيّن النهاية عبر $\ell =
        \sin \ell$، مستعملًا $\abs{\sin a - \sin b} \leq \abs{a -
        b}$، وهي نفسها نتيجة للمقدار (G2) وصيغة تحويل الجداء
        إلى مجموع)}.
  \item افترض من الآن فصاعدًا $u_1 \in \intoc{0}{1}$. بيّن بالحصر،
        مستعملًا (G2):
        \[
        \frac{\sin u_n}{u_n} \to 1
        \qquad\text{و}\qquad
        \frac{u_n - \sin u_n}{u_n^{3}} \to \frac16 .
        \]
  \item برهن على التعميل
        \[
        w_n := \frac{1}{u_{n+1}^{2}} - \frac{1}{u_n^{2}}
        = \frac{u_n - \sin u_n}{u_n^{3}} \cdot
        \frac{u_n + \sin u_n}{u_n} \cdot
        \Bigl(\frac{u_n}{\sin u_n}\Bigr)^{2},
        \]
        واستنتج $w_n \to \frac13$.
  \item اختم بالمقدار \cref{exo:b1:seq:10} (صيغة التزايدات)
        بأن $\frac{1}{n\,u_n^{2}} \to \frac13$، ثم، بحجة
        مرافقة من أجل الجذر التربيعي، بالعنوان:
        \[
        \sqrt n\;u_n \longrightarrow \sqrt 3 ,
        \qquad\text{أي}\qquad
        u_n \sim \sqrt{\frac 3n} .
        \]
  \item كمِّم البطء: بيّن أنه في النهاية $\sqrt{2/n}
        \leq u_n \leq 2/\sqrt n$، بحيث يقتضي بلوغ $u_n \leq
        10^{-2}$ أكثر من $20\,000$ تكرار (نحو
        $30\,000$، حسب المقاربة). وقابل ذلك \omterm{ex:b1:seq:heron}{بطريقة
        هيرون} (\cref{ex:b1:seq:heron}) وفسّر السبب
        البنيوي: عند النقطة الصامدة $0$، يكون ميل
        $\sin$ هو $1$ (أي نقطة صامدة \emph{محايدة})، بينما
        تحتاج التكرارات التي تنصّف الخطأ إلى ميل مقياسه $< 1$.
  \item بيّن أنه من أجل \emph{كل} نقطة انطلاق $u_0 \in \R$،
        إمّا أن يكون $u_n = 0$ ابتداءً من الرتبة $1$، وإمّا أن يكون $\abs{u_n} \sim
        \sqrt{3/n}$ --- فالسقوط شامل، ولا تتذكّر الإشارة وحدها
        إلا $u_0$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- المبدأ العام.} الجيب
مثال واحد على آلة.
\begin{enumerate}[resume]
  \item لتكن $u_n > 0$ و $u_n \to 0$ و $\dfrac{u_n -
        u_{n+1}}{u_n^{2}} \to a > 0$. برهن تباعًا على:
        $\frac{u_{n+1}}{u_n} \to 1$؛ ثم $\frac{1}{u_{n+1}} -
        \frac{1}{u_n} \to a$؛ ثم $n\,u_n \to \frac1a$.
  \item (نموذج مضبوط) من أجل $u_{n+1} = \dfrac{u_n}{1 + u_n}$، $u_0
        > 0$: بيّن أن $\frac{1}{u_n}$ حسابية، وحُلَّ بالضبط،
        وتحقق من نتيجة السؤال 18 إزاء الصيغة
        المضبوطة.
  \item من أجل $u_{n+1} = u_n \eu^{-u_n}$ مع $u_0 > 0$: بيّن $u_n \to
        0$، واستعمل (G1) لحصر $\frac{1 - \eu^{-t}}{t}$ بين
        $\frac{1}{1+t}$ و $1$ من أجل $t > 0$، واستنتج $u_n
        \sim \frac 1n$.
  \item (تماسّ تكعيبي وتلسكوب مربَّع) لتكن $u_n > 0$ و $u_n
        \to 0$ و $\dfrac{u_n - u_{n+1}}{u_n^{3}} \to a > 0$.
        كيّف تعميل السؤال 14 لتبيّن
        $\frac{1}{u_{n+1}^2} - \frac{1}{u_n^2} \to 2a$، و
        استنتج $n\,u_n^{2} \to \frac{1}{2a}$. وتحقق من أن $a =
        \frac16$ يستعيد الجزء 4.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 6 --- حدود الطريقة، والعبر.}
\begin{enumerate}[resume]
  \item بيّن أن الفرضية $b_n \to +\infty$ لا يمكن
        إسقاطها: فمن أجل $a_n = 2 - 2^{-n}$ و $b_n = 1 - 2^{-n}$،
        تؤول قسمة التزايدات إلى $1$ بينما $\frac{a_n}
        {b_n} \to 2$. وأشِر إلى السطر الدقيق من برهان السؤال
        5 الذي ينكسر.
  \item (ستولتز مرتين) برهن على $\sum_{k=1}^{n} H_k \sim n \ln n$
        \emph{(\omterm{def:b1:logic:map}{بتطبيق} واحد لتشيزارو--ستولتز، ثم السؤال
        10؛ وحُدّ $(n+1)\ln(n+1) - n\ln n$ مستعملًا السؤال 9)}.
  \item (المتوسطات الهندسية) إذا كان $u_n > 0$ و $u_n \to \ell > 0$،
        فبيّن $(u_1 u_2 \cdots u_n)^{1/n} \to \ell$؛ وإذا كان $u_n \to
        +\infty$، فبيّن $(u_1 \cdots u_n)^{1/n} \to +\infty$.
        واستنتج $(n!)^{1/n} \to +\infty$.
  \item توليفة، جملة واحدة لكلٍّ: (أ) أين دخلت خاصية
        التمام في هذه المسألة بالضبط؛ (ب) بأيّ معنى تكون
        تشيزارو--ستولتز قاعدةَ لوبيتال متقطّعة (فتوأمها
        التفاضلي يقوم على مبرهنة التزايدات المنتهية في
        \cref{ch:b1:derivative})؛ (ج) اذكر القاعدة الاستكشافية
        الرابطة بين رتبة تماسّ $f$ عند نقطة صامدة
        محايدة وأُسّ التلاشي للمقدار $u_{n+1} = f(u_n)$؛
        (د) وتتبّع الثابت $3$ في $\sqrt{3/n}$ رجوعًا عبر
        السلسلة $\frac16 \to \frac13 \to 3$.
\end{enumerate}
\end{problem}
