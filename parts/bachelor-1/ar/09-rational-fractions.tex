\chapter{الكسور الناطقة}\label{ch:b1:fractions}

\omterm{def:b1:fractions:field}{الكسر الناطق} قسمةُ كثيرَي حدود. والمبرهنة المحورية
في هذا الفصل القصير --- تفكيك الكسر إلى عناصر
بسيطة\index{تفكيك إلى عناصر بسيطة} --- تفتّت أيّ
قسمة كهذه إلى مجموع لبنات أوّلية $\frac{c}{(X - a)^k}$.
وإلى جانب فائدته الجبرية، هو الآلة المعيارية
لمكاملة الدوال الناطقة (\cref{ch:b1:integration}) ولجمع
بعض المتسلسلات (\cref{ch:b1:series}).

\section{الحقل $K(X)$}

\begin{definition}\label{def:b1:fractions:field}
\emph{الكسر الناطق}\index{كسر ناطق} فوق $K$ ($= \R$
أو $\C$) هو قسمة $F = \frac{A}{B}$ حيث $A, B \in K[X]$ و$B \neq
0$؛ ونطابق بين القسمتين $\frac AB$ و $\frac{A'}{B'}$ عندما يكون
$AB' = A'B$. ولكل كسر \emph{صورة مختزلة} تحقق
$\gcd(A, B) = 1$، وهي وحيدة بغضّ النظر عن الثوابت. ومع العمليات الطبيعية،
تكون \omterm{def:b1:logic:sets}{مجموعة} الكسور الناطقة $K(X)$ \omterm{def:b1:structures:field}{حقلًا}.

و\emph{أقطاب}\index{قطب} $F$ (في الصورة المختزلة) هي جذور
$B$؛ و\emph{رتبة} القطب هي تضاعفه بوصفه جذرًا للمقدار $B$.
و\emph{درجة} $F$ هي $\deg F = \deg A - \deg B \in \Z \cup
\{-\infty\}$.
\end{definition}

\begin{example}[قراءة الأقطاب والرتب والدرجة]\label{ex:b1:fractions:vocab}
لتكن $F = \dfrac{X^3 - X}{X^4 - 2X^3 + X^2}$. عمّل الطبقتين:
البسط $X(X-1)(X+1)$، والمقام $X^2(X-1)^2$؛ واختصر
العامل المشترك $X(X-1)$:
\[
F = \frac{X + 1}{X(X - 1)} \quad\text{(الصورة المختزلة).}
\]
\omterm{def:b1:fractions:field}{والأقطاب}: $0$ و $1$، وكلاهما \emph{بسيط} --- فالرتب تُقرأ على
المقام المختزل، ومنه فالجذور المضاعفة الظاهرة في المقام
الأصلي لا شأن لها. والدرجة: $\deg F = 1 - 2 = -1$،
وهي مرئية مقاربيًا (إذ $xF(x) \to 1$ عندما $x \to \infty$). و
تسلك الدرجة سلوك درجة \omterm{def:b1:poly:def}{كثير الحدود} ($\deg FG = \deg F +
\deg G$ و$\deg(F + G) \leq \max$)، وهي قاعدة محاسبة تُستعمل
باستمرار في مطاردات المعاملات أدناه: فكل حجة من نوع «نهاية
$xF(x)$» عدُّ درجات متنكّر.
\end{example}

\begin{proposition}[الجزء الصحيح]\label{prop:b1:fractions:integerpart}
كل $F = \frac AB \in K(X)$ يُكتب على نحو وحيد $F = E + \frac RB$ حيث $E
\in K[X]$ (وهو \emph{الجزء الصحيح}، أو الجزء الكثيرالحدود، للمقدار $F$) و
$\deg R < \deg B$. ويكون $E \neq 0$ إذا وفقط إذا كان $\deg F \geq 0$.
\end{proposition}

\begin{proof}
القسمة الإقليدية $A = BE + R$ (\cref{thm:b1:poly:division})، مقسومةً
على $B$. وأمّا الوحدانية: فإذا كان $E + \frac RB = E' + \frac{R'}{B}$ فإن $(E -
E')B = R' - R$ مع $\deg(R' - R) < \deg B$، فيُفرض $E = E'$ ثم $R
= R'$.
\end{proof}

\begin{example}[اختزل أولًا ثم اقسم]\label{ex:b1:fractions:integerpart}
جد الجزء الصحيح للكسر $F = \dfrac{X^3 + 1}{X^2 - 1}$. فبالقسمة
العمياء: $X^3 + 1 = (X^2 - 1)X + (X + 1)$، ومنه $F = X +
\frac{X + 1}{X^2 - 1}$. لكن الكسر لم يكن مختزلًا: إذ يشترك $X^3 + 1
= (X + 1)(X^2 - X + 1)$ و $X^2 - 1 = (X+1)(X-1)$ في العامل
$X + 1$، و
\[
F = \frac{X^2 - X + 1}{X - 1} = X + \frac{1}{X - 1} :
\]
فالجزء الصحيح $X$ نفسه، لكن الجزء الكسري ينهار إلى
لبنة واحدة، ولم يكن «\omterm{def:b1:fractions:field}{القطب}» عند $-1$ قطبًا البتة.
فاختزل دائمًا إلى أبسط صورة قبل مطاردة \omterm{def:b1:fractions:field}{الأقطاب}: \omterm{def:b1:fractions:field}{فأقطاب}
$F$ هي جذور المقام \emph{المختزل}. (أمّا الجزء الصحيح
فلا يتأثر بالاختزال، كما تضمنه الوحدانية في
\cref{prop:b1:fractions:integerpart}.)
\end{example}

\section{التفكيك إلى عناصر بسيطة فوق $\C$}

\begin{theorem}[التفكيك فوق $\C$]\label{thm:b1:fractions:complex}
ليكن $F = \frac AB \in \C(X)$ في صورته المختزلة، حيث $B = c\,(X -
a_1)^{m_1} \cdots (X - a_r)^{m_r}$. عندئذ يُكتب $F$، على نحو وحيد،
\[
F = E + \sum_{i=1}^{r} \sum_{k=1}^{m_i}
\frac{c_{i,k}}{(X - a_i)^k},
\qquad E \in \C[X],\ c_{i,k} \in \C ,
\]
حيث $E$ هو الجزء الصحيح للمقدار $F$.
\end{theorem}

\begin{proof}
حسب \cref{prop:b1:fractions:integerpart} يمكننا أن نفترض $\deg A < \deg
B$ وأن نبرهن على تفكيك المجموع مع $E = 0$.

\emph{فصل \omterm{def:b1:fractions:field}{الأقطاب}.} اكتب $B = (X - a_1)^{m_1} B_1$ حيث
$B_1(a_1) \neq 0$. وكثيرا الحدود $(X-a_1)^{m_1}$ و $B_1$
\omterm{cor:b1:arith:bezout}{أوليّان فيما بينهما} (فلا جذر مشترك)، ومنه بمتطابقة بيزو في $\C[X]$ (انظر الملاحظة في
\cref{ch:b1:poly}) يوجد $U_0, V_0$ يحققان $U_0 (X-a_1)^{m_1} +
V_0 B_1 = 1$؛ وبالضرب في $A$ ووضع $U = AU_0$ و$V =
AV_0$، ثم القسمة على $B$:
\[
\frac AB = \frac{V}{(X - a_1)^{m_1}} + \frac{U}{B_1} .
\]
ويمكن فرض شروط الدرجة: اقسم $V$ على
$(X-a_1)^{m_1}$، وليكن $V = (X-a_1)^{m_1}Q + V_1$ حيث $\deg V_1 <
m_1$، وامتصّ خارج القسمة في الحدّ الثاني ($U_1 = U +
QB_1$):
\[
\frac AB = \frac{V_1}{(X - a_1)^{m_1}} + \frac{U_1}{B_1},
\qquad \deg V_1 < m_1 ;
\]
ومقارنة الدرجات ($\deg A < \deg B$ و $\deg V_1 < m_1$) تفرض
$\deg U_1 < \deg B_1$ كذلك. وبتكرار ذلك على $\frac{U_1}{B_1}$،
قطبًا بعد \omterm{def:b1:fractions:field}{قطب}، يُردّ كل شيء إلى حالة \omterm{def:b1:fractions:field}{القطب} الواحد أدناه.

\emph{\omterm{def:b1:fractions:field}{قطب} واحد.} من أجل $\frac{A}{(X-a)^m}$ حيث $\deg A < m$: انشر $A$
بقوى $(X - a)$، $A = \sum_{j=0}^{m-1} \alpha_j (X - a)^j$
(وهو نشر تايلور \omterm{def:b1:poly:def}{لكثير حدود}، كما في برهان
\cref{prop:b1:poly:multiplicity})؛ وتعطي القسمة اللبنات بالضبط
$\frac{\alpha_j}{(X-a)^{m-j}}$. وعمليًا، من أجل $\frac{X^2 +
1}{(X+2)^3}$: بتعويض $X = Y - 2$،
\[
X^2 + 1 = (Y - 2)^2 + 1 = Y^2 - 4Y + 5 ,
\qquad\text{ومنه}\qquad
\frac{X^2+1}{(X+2)^3} = \frac1{Y} - \frac4{Y^2} + \frac5{Y^3}
\]
حيث $Y = X + 2$: فتظهر اللبنات الثلاث بمجرد قسمة
النشر المنسحب --- وهو أسرع طريق كلما دخل \omterm{def:b1:fractions:field}{قطب} واحد
عالي الرتبة، وهو الطريق المنصوح به من أجل
\cref{exo:b1:fractions:3}.

\emph{الوحدانية.} نفترض تطابق تفكيكين؛ فيكون
فرقهما متطابقةً $0 = \sum_{i,k} \frac{d_{i,k}}{(X -
a_i)^k}$. اضرب الجميع في $(X - a_1)^{m_1}$: فيكتسب كل حدّ
عاملًا ينعدم عند $a_1$ \emph{عدا} الحدّ الذي فيه
$i = 1$ و $k = m_1$، الذي يصير معامله $d_{1,m_1}$ زائد
حدود تحمل عاملًا $(X - a_1)$ على الأقل. والحساب عند
$a_1$ (وهذا مشروع: إذ لم يبقَ بعد الضرب أيّ \omterm{def:b1:fractions:field}{قطب} عند
$a_1$) يعطي $d_{1,m_1} = 0$. وبزوال المعامل الأعلى،
كرّر مع $(X - a_1)^{m_1 - 1}$، وهكذا نزولًا حتى $k = 1$؛
ثم انتقل إلى \omterm{def:b1:fractions:field}{القطب} التالي. فتنعدم كل المعاملات $d_{i,k}$: أي أن
التفكيك وحيد.
\end{proof}

\begin{method}[حساب المعاملات]\label{met:b1:fractions:compute}
عمليًا، تجنّب متطابقة بيزو؛ واجمع بين ما يلي:
\begin{enumerate}
  \item \emph{التغطية من أجل القوة العليا}: معامل
        $\frac{1}{(X-a)^m}$ (حيث $m$ رتبة \omterm{def:b1:fractions:field}{القطب} $a$) هو
        \[
        c_{a,m} = \Bigl[\,(X - a)^m F\,\Bigr]_{X = a};
        \]
        ومن أجل \omterm{def:b1:fractions:field}{قطب} \emph{بسيط} للكسر $F = \frac AB$، يكون هذا
        $\frac{A(a)}{B'(a)}$؛
  \item \emph{الحسابات} عند نقاط ملائمة و\emph{نهايات}
        $xF(x)$ عندما $x \to \infty$ لجمع علاقات خطية من أجل
        المعاملات الباقية؛
  \item \emph{الزوجية} أو تناظرات المرافقة، عند وجودها، لتنصيف
        العمل.
\end{enumerate}
\end{method}

\begin{remark}[مزالق شائعة مع العناصر البسيطة]\label{rem:b1:fractions:pitfalls}
\begin{enumerate}
  \item \emph{تخطّي الجزء الصحيح.} ينطبق تفكيك اللبنات
        على الكسور ذات الدرجة $< 0$؛ فإذا كان $\deg A \geq
        \deg B$ فاقسم أولًا
        (\cref{prop:b1:fractions:integerpart})، وإلا أنتجت
        مطاردة المعاملات تناقضات.
  \item \emph{التغطية خارج نطاقها.} الضرب في
        $(X - a)^k$ والحساب عند $a$ لا يعطيان المعامل
        إلا من أجل $k = m$، أي رتبة \omterm{def:b1:fractions:field}{القطب} \emph{الكاملة}؛
        أمّا معاملات الرتب الأدنى فتقتضي علاقات أخرى (نهايات،
        وحسابات) --- انظر \cref{ex:b1:fractions:multiple}.
  \item \emph{نسيان الاختزال.} تُقرأ \omterm{def:b1:fractions:field}{الأقطاب} على
        الصورة \emph{المختزلة}؛ فعاملٌ مشترك بين البسط
        والمقام يخلق أقطابًا وهمية
        (\cref{ex:b1:fractions:integerpart}).
  \item \emph{صور اللبنات الخاطئة فوق $\R$.} فوق \omterm{def:b1:poly:def}{كثير حدود}
        من الدرجة الثانية لا يتجزأ تكون البسوط \emph{تآلفية} ($\alpha X +
        \beta$)، لا ثابتة؛ وكتابة $\frac{c}{X^2 + 1}$ وحدها
        تفقد حلولًا --- فصور اللبنات تمليها
        \cref{thm:b1:fractions:real}، ولا تُرتجل أبدًا.
\end{enumerate}
\end{remark}

\begin{proof}[برهان صيغة القطب البسيط]
قرب \omterm{def:b1:fractions:field}{قطب} بسيط $a$: $B = (X - a) Q$ حيث $Q(a) \neq 0$، و$B' =
Q + (X - a) Q'$، ومنه $B'(a) = Q(a)$. وقيمة التغطية هي
$\frac{A(a)}{Q(a)} = \frac{A(a)}{B'(a)}$.
\end{proof}

\begin{example}\label{ex:b1:fractions:simple}
فكّك $F = \dfrac{1}{X(X-1)(X-2)}$. ثلاثة \omterm{def:b1:fractions:field}{أقطاب} بسيطة؛ وبالتغطية
عند كلٍّ منها:
\[
c_0 = \frac{1}{(0-1)(0-2)} = \frac12,
\quad
c_1 = \frac{1}{1 \times (1 - 2)} = -1,
\quad
c_2 = \frac{1}{2 \times 1} = \frac12,
\]
ومنه $F = \dfrac{1/2}{X} - \dfrac{1}{X-1} + \dfrac{1/2}{X-2}$.
\emph{وللتحقق عند $X = 3$:} مباشرةً، $F(3) = \frac{1}{3\cdot2\cdot1} =
\frac16$؛ ومن التفكيك، $\frac{1/2}{3} - \frac{1}{2} +
\frac{1/2}{1} = \frac16 - \frac12 + \frac12 = \frac16$.
\end{example}

\begin{example}[قطب مضاعف]\label{ex:b1:fractions:multiple}
فكّك $F = \dfrac{X}{(X-1)^2 (X+1)}$. والصورة:
$\frac{a}{(X-1)^2} + \frac{b}{X-1} + \frac{c}{X+1}$.
\begin{itemize}
  \item التغطية عند \omterm{def:b1:fractions:field}{القطب} المضاعف: $a = \bigl[\frac{X}{X+1}\bigr]_{X=1}
        = \frac12$.
  \item التغطية عند $-1$: $c = \bigl[\frac{X}{(X-1)^2}\bigr]_{X=-1} =
        -\frac14$.
  \item نهاية $xF(x)$ عند $\infty$: $0 = b + c$، ومنه $b = \frac14$.
\end{itemize}
\[
F = \frac{1/2}{(X-1)^2} + \frac{1/4}{X-1} - \frac{1/4}{X+1} .
\]
\emph{وللتحقق عند $X = 0$:} $F(0) = 0$ و$\frac12 - \frac14 - \frac14
= 0$.
\end{example}

\section{التفكيك فوق $\R$}

\begin{theorem}[التفكيك فوق $\R$]\label{thm:b1:fractions:real}
ليكن $F \in \R(X)$ في صورته المختزلة ومقامه
\[
B = c \prod_i (X - a_i)^{m_i} \prod_j (X^2 + p_j X + q_j)^{n_j}
\qquad (p_j^2 - 4q_j < 0).
\]
عندئذ يتفكك $F$ على نحو وحيد إلى جزئه الصحيح زائد حدود
\[
\frac{c_{i,k}}{(X - a_i)^k} \quad (1 \leq k \leq m_i),
\qquad
\frac{\alpha_{j,l}\, X + \beta_{j,l}}{(X^2 + p_j X + q_j)^{l}}
\quad (1 \leq l \leq n_j),
\]
بمعاملات حقيقية.
\end{theorem}

\begin{proof}
فكّك فوق $\C$ (\cref{thm:b1:fractions:complex}). وبما أن $F$
حقيقي، يكون المعامل فوق \omterm{def:b1:fractions:field}{القطب} $\conj a$ (عند كل رتبة)
مرافقَ المعامل فوق $a$ (طبّق المرافقة على
التفكيك واستدعِ الوحدانية). وجمّع كل زوج مترافق:
\[
\frac{c}{(X - z)^l} + \frac{\conj c}{(X - \conj z)^l}
= \frac{c\,(X - \conj z)^l + \conj c\,(X - z)^l}{\bigl(X^2 - 2\Re(z)X
+ \abs z^2\bigr)^{l}},
\]
وبسطه مرافق نفسه، فهو إذن حقيقي، ودرجته $\leq l$؛
وفصل مضاعفات \omterm{def:b1:poly:def}{كثير الحدود} الحقيقي من الدرجة الثانية يخفضها إلى الدرجة
$\leq 1$ عند كل مستوى $l$ (باستقراء نزولي صغير). وتحتفظ \omterm{def:b1:fractions:field}{الأقطاب}
الحقيقية بمعاملاتها الحقيقية (فالمرافقة تصمد عندها). وتنتج الوحدانية
من الوحدانية فوق $\C$.
\end{proof}

\begin{example}[مراقبة ازدواج المرافقات]\label{ex:b1:fractions:pairing}
آلية البرهان، على أصغر حالة: فوق $\C$، تكون
\omterm{def:b1:fractions:field}{أقطاب} $\frac1{X^2+1}$ هي $\pm\iu$، ومعاملا التغطية
$\frac1{2\iu}$ عند $\iu$ و $\frac1{-2\iu}$ عند $-\iu$ ---
وهما مترافقان، كما تتنبأ به المبرهنة:
\[
\frac{1}{X^2 + 1}
= \frac{1/(2\iu)}{X - \iu} - \frac{1/(2\iu)}{X + \iu} .
\]
وبإعادة الجمع على المقام المشترك:
\[
\frac{1}{2\iu}\cdot\frac{(X + \iu) - (X - \iu)}{X^2 + 1}
= \frac{1}{2\iu}\cdot\frac{2\iu}{X^2 + 1} = \frac{1}{X^2+1} :
\]
فيختصر الجزآن التخيليان وتعود اللبنة الحقيقية سليمة.
ومن أجل المكاملات الحقيقية لا يغادر المرء $\R$ عادةً --- لكن عند
حساب \emph{مجاميع} عند نقاط عقدية (كما تفعل مسألة نهاية الأسبوع
\omterm{def:b1:complex:unity}{بجذور الوحدة})، تكون اللبنات العقدية هي العملة الطبيعية،
ويكون هذا الازدواج سعرَ الصرف بين التفكيكين.
\end{example}

\begin{example}\label{ex:b1:fractions:real}
فكّك $F = \dfrac{4}{(X^2+1)(X-1)^2}$ فوق $\R$. والصورة:
$\frac{aX + b}{X^2 + 1} + \frac{c}{(X-1)^2} + \frac{d}{X - 1}$.
التغطية عند \omterm{def:b1:fractions:field}{القطب} المضاعف: $c = \bigl[\frac{4}{X^2+1}\bigr]_{X=1} =
2$. والتغطية عند \omterm{def:b1:fractions:field}{القطب} العقدي $\iu$ (بحساب البسط فوق $X^2 + 1$
عبر التفكيك العقدي، أو مباشرةً): اضرب في
$X^2 + 1$ وضع $X = \iu$:
\[
a\iu + b = \frac{4}{(\iu - 1)^2} = \frac{4}{-2\iu} = 2\iu ,
\]
ومنه $a = 2$ و $b = 0$. ونهاية $xF(x)$ عند اللانهاية: $0 = a + d$، ومنه $d
= -2$. ومنه
\[
F = \frac{2X}{X^2+1} + \frac{2}{(X-1)^2} - \frac{2}{X-1} .
\]
\emph{وللتحقق عند $X = 0$:} $F(0) = 4$ و $0 + 2 + 2 = 4$.
\end{example}

\begin{example}[مقام من الدرجة الثالثة، من أوله إلى آخره]\label{ex:b1:fractions:cubic}
فكّك $F = \dfrac{1}{X^3 + 1}$ فوق $\R$. عمّل أولًا: $X^3 +
1 = (X + 1)(X^2 - X + 1)$، ومميّز \omterm{def:b1:poly:def}{كثير الحدود} من الدرجة الثانية $-3 <
0$. والصورة: $\frac{a}{X+1} + \frac{bX + c}{X^2 - X + 1}$. والتغطية
عند \omterm{def:b1:fractions:field}{القطب} البسيط $-1$: $a = \bigl[\frac1{X^2 - X +
1}\bigr]_{X=-1} = \frac13$. ونهاية $xF(x)$ عند اللانهاية: $0 = a +
b$، ومنه $b = -\frac13$. والحساب عند $X = 0$: $1 = a + c$، ومنه $c =
\frac23$. ومنه
\[
\frac{1}{X^3+1} = \frac13\Bigl(\frac{1}{X+1}
+ \frac{-X + 2}{X^2 - X + 1}\Bigr) ,
\]
وقد تأكد عند $X = 1$: فالطرف الأيسر $\frac12$، والطرف الأيمن
$\frac13\bigl(\frac12 + 1\bigr) = \frac12$. ولاحظ الاقتصاد:
ثلاثة مجاهيل، وثلاث وقائع خطية رخيصة (تغطية واحدة، ونهاية
واحدة، وحساب واحد)، دون نشر أيّ شيء --- وهو منهج
\cref{met:b1:fractions:compute} في صورته الخالصة.
\end{example}

\begin{remark}[فيمَ يُستعمل]
بمجرد التفكيك، تُكامل الدالة الناطقة حدًّا حدًّا: فللبنات
$\frac{1}{(x-a)^k}$ دوال أصلية أوّلية، وتُردّ اللبنات
$\frac{\alpha x + \beta}{(x^2 + px + q)^l}$ إلى $\ln$ و
$\arctan$ (\cref{ch:b1:integration}). والمجاميع التلسكوبية هي \omterm{def:b1:logic:map}{التطبيق}
المعياري الآخر (\cref{exo:b1:fractions:8}).
\end{remark}

\begin{remark}[أين يُستعمل هذا الفصل]\label{rem:b1:fractions:whereused}
العناصر البسيطة قبل كل شيء خطوة \emph{تحضير}: فيمرّر
فصل المكاملة (\cref{ch:b1:integration}) كل مكامَل ناطق
عبر \cref{thm:b1:fractions:real} قبل المكاملة،
ويقوم فصل المتسلسلات (\cref{ch:b1:series}) بتلسكوب الحدود
الناطقة تمامًا كما في \cref{exo:b1:fractions:8} وفي مسألة نهاية
الأسبوع أدناه --- التي تدفع التقنية حتى
$\sum 1/k^2 = \pi^2/6$. وتعود المشتقة اللوغاريتمية $P'/P = \sum
m_i/(X - a_i)$ (\cref{exo:b1:poly:11}) كلما دُرست مواضع
الجذور. وخارج هذا المجلد، يقوم تفكيك
$1/\chi(X)$ من أجل \omterm{def:b1:poly:def}{كثير حدود} مميِّز $\chi$ بأساس
حساب قوى المصفوفات وتحويلات لابلاس في مجلد السنة
2: فاللبنات $\frac{c}{(X - a)^k}$ هي الظلّ الجبري
للحلول $t^{k-1}\eu^{at}$ المقابَلة في \cref{ch:b1:diffeq}.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:fractions:1}
فكّك فوق $\R$: $\dfrac{1}{X^2 - 1}$؛
و$\;\dfrac{X}{X^2 - 3X + 2}$؛ و$\;\dfrac{X^2 + 1}{X(X-1)}$ \emph{(انتبه
إلى الجزء الصحيح)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:fractions:2}
فكّك فوق $\R$: $\dfrac{1}{X(X^2 + 1)}$ و
$\dfrac{X^3}{X^2 + X + 1}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:fractions:3}
فكّك $\dfrac{1}{X^2(X - 1)}$ و$\dfrac{X + 1}{(X - 1)^3}$
\emph{(ومن أجل الثاني، عوّض $Y = X - 1$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:4}
فكّك فوق $\C$، ثم فوق $\R$: $\dfrac{1}{X^4 - 1}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:5}
فكّك فوق $\R$: $\dfrac{X^2}{(X^2 + 1)^2}$، واستنتج
دالة أصلية للدالة $x \mapsto \dfrac{x^2}{(x^2+1)^2}$ مع العلم أن
$\int \frac{\dd x}{(x^2+1)^2} = \frac12\bigl(\arctan x +
\frac{x}{x^2+1}\bigr) + C$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:6}
من أجل $n \in \N^*$، فكّك $F_n = \dfrac{n!}{X(X+1)\cdots(X+n)}$
\emph{(\omterm{def:b1:fractions:field}{أقطاب} بسيطة عند $0, -1, \dots, -n$؛ استعمل صيغة التغطية
وتعرّف على المعاملات الثنائية)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:7}
باستعمال متطابقة \cref{exo:b1:poly:11} من أجل $P = X^n - 1$،
برهن على أن
\[
\sum_{k=0}^{n-1} \frac{1}{X - \omega^k} = \frac{n X^{n-1}}{X^n - 1},
\qquad \omega = \eu^{2\iu\pi/n},
\]
واحسب الطرفين عند $X = 2$ من أجل $n = 4$ للتحقق.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:8}
فكّك $\dfrac{1}{k(k+1)(k+2)}$ واحسب
\[
S_n = \sum_{k=1}^{n} \frac{1}{k(k+1)(k+2)},
\qquad\text{ثم}\qquad
\lim_{n \to \infty} S_n .
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:fractions:9}
ليكن $P \in \R[X]$ \omterm{def:b1:poly:def}{واحديًا} من الدرجة $n$ وله $n$ جذرًا حقيقيًا متمايزًا
$x_1 < \dots < x_n$. برهن على أن
\[
\sum_{i=1}^{n} \frac{1}{P'(x_i)} = 0 \quad (n \geq 2),
\qquad
\sum_{i=1}^{n} \frac{x_i^{\,n-1}}{P'(x_i)} = 1 .
\]
\emph{إرشاد: فكّك $\frac{X^m}{P}$ من أجل $m \leq n - 1$ وانظر إلى
تناقص المعاملات عند اللانهاية --- أو استعمل \omterm{thm:b1:poly:lagrange}{مقايسة لاغرانج}
(\cref{thm:b1:poly:lagrange}) للحدّانية $X^m$ عند العقد
$x_i$.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:10}
فكّك $\dfrac{1}{X(X+1)^2}$ فوق $\R$. وبقبول القيمة
$\sum_{k \geq 1} \frac1{k^2} = \frac{\pi^2}6$ (المبرهن عليها في مسألة نهاية الأسبوع
من هذا الفصل)، استنتج
\[
\sum_{n=1}^{\infty} \frac{1}{n(n+1)^2} = 2 - \frac{\pi^2}{6} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:11}
فكّك فوق $\R$: $\dfrac{1}{(X^2+1)(X^2+4)}$، ثم
$\dfrac{X^2}{(X^2+1)(X^2+4)}$. \emph{إرشاد: كلا المقامين \omterm{def:b1:poly:def}{كثير
حدود} في $X^2$: ففكّك $\frac1{(Y+1)(Y+4)}$ أولًا.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:fractions:12}
(المعادلات الدهرية) لتكن $F = \sum_{i=1}^{r} \frac{c_i}{X - p_i}$
حيث $p_1 < p_2 < \dots < p_r$ حقيقية وكل $c_i > 0$.
\begin{enumerate}
  \item بيّن أن $F$ متناقصة تمامًا على كل مجال من
        \omterm{def:b1:logic:sets}{مجموعة} تعريفها، وأعط نهاياتها عند $\pm\infty$ وعلى جانبَي
        كل \omterm{def:b1:fractions:field}{قطب}.
  \item استنتج أنه من أجل كل $\lambda > 0$، يكون للمعادلة $F(x) =
        \lambda$ بالضبط $r$ حلًّا حقيقيًا، واحد في كل
        مجال $\intoo{p_i}{p_{i+1}}$ وواحد بعد $p_r$.
        (ومعادلات كهذه تحكم اضطرابات القيم الذاتية؛ ومبرهنة
        القيم الوسطى مستعملة هنا على مستوى الثانوية
        ومبرهن عليها في \cref{ch:b1:continuity}.)
\end{enumerate}
\end{exercise}

\section{مسألة: العناصر البسيطة محرّكًا}

\begin{problem}\label{pb:b1:fractions:1}
يبدو تفكيك الكسر إلى عناصر بسيطة محاسبةً؛ وتبيّن هذه المسألة
أنه محرّك. فإذا غُذّي بالكسر
$\frac1{X(X+1)\cdots(X+k)}$، تلسكب عائلات كاملة من المجاميع
في صورة مغلقة؛ وإذا غُذّي بالمقدار $\frac1{X^n - 1}$ أنتج
متطابقات مثلثية مثل
\[
\sum_{k=1}^{n-1}\frac{1}{\sin^2\frac{k\pi}{n}} = \frac{n^2-1}{3} ;
\]
وإذا دُفع خطوة أخرى، عصرت تلك المتطابقة واحدةً من
أشهر الصيغ في الرياضيات، صيغة أويلر
\[
\sum_{k=1}^{\infty}\frac1{k^2} = \frac{\pi^2}{6}
\]
--- وقد حُصل عليها هنا بلا شيء يتجاوز جبر هذا الفصل و
مثلثات الثانوية.\index{تلسكوب}\index{مسألة بازل}
وفي كل ما يلي، $\omega = \eu^{2\iu\pi/n}$؛ ونهايات المتتاليات
مستعملة على مستوى الثانوية (ويصوغها \cref{ch:b1:seq}).

\medskip
\noindent\textbf{الجزء 1 --- التلسكوب.}
\begin{enumerate}
  \item فكّك $\frac1{X(X+1)}$ واحسب $\sum_{n=1}^{N}
        \frac1{n(n+1)}$ بالضبط؛ واستنتج أن المجموع يؤول إلى
        $1$.
  \item الأمر نفسه من أجل $\frac1{X(X+2)}$: بيّن أن $\sum_{n=1}^{N}
        \frac1{n(n+2)} = \frac12\bigl(\frac32 - \frac1{N+1} -
        \frac1{N+2}\bigr) \to \frac34$. (فمع فجوة، يبقى \emph{حدّان}
        طرفيان عند كل نهاية.)
  \item صُغ الآلية: إذا كان $F(X) = G(X) - G(X+1)$ من أجل
        \omterm{def:b1:fractions:field}{كسر ناطق} $G$ ما بلا \omterm{def:b1:fractions:field}{أقطاب} في $\intco1{+\infty}$،
        فإن $\sum_{n=1}^N F(n) = G(1) - G(N+1)$. واسترجع
        القيمة $\frac14$ في \cref{exo:b1:fractions:8} بإبراز
        الشاهد $G$ من أجل $F = \frac1{X(X+1)(X+2)}$.
  \item برهن على التلسكوب العاملي العام: من أجل $k \geq 1$،
        \[
        \frac{1}{X(X+1)\cdots(X+k)}
        = \frac1k\biggl(\frac{1}{X(X+1)\cdots(X+k-1)}
        - \frac{1}{(X+1)\cdots(X+k)}\biggr),
        \]
        واستنتج
        \[
        \sum_{n=1}^{\infty}\frac{1}{n(n+1)\cdots(n+k)}
        = \frac{1}{k \cdot k!} .
        \]
        وتحقق من الحالة $k = 2$ في مقابل السؤال 3.
  \item احسب تفكيك \cref{exo:b1:fractions:6}
        عند نقاط حسنة الاختيار للبرهان على أن
        \[
        \sum_{j=0}^{n}(-1)^j\binom nj\,\frac1{j+1}
        = \frac1{n+1},
        \qquad
        \sum_{j=0}^{n}(-1)^j\binom nj\,\frac1{j+2}
        = \frac1{(n+1)(n+2)} .
        \]
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- الكسر $1/(X^n - 1)$.}
\begin{enumerate}[resume]
  \item بيّن بصيغة التغطية أن
        \[
        \frac{1}{X^n - 1}
        = \frac1n\sum_{k=0}^{n-1}\frac{\omega^k}{X - \omega^k} .
        \]
  \item تحققان: تأكّد من الصيغة مباشرةً من أجل $n =
        2$، وبيّن أن مجموع المعاملات $n$
        ينعدم من أجل $n \geq 2$ --- واشرح لماذا يجب أن ينعدم (تأمّل
        $xF(x)$ عندما $x \to \infty$).
  \item استخرج بالتغطية من جديد متطابقة
        \cref{exo:b1:fractions:7}:
        $\frac{nX^{n-1}}{X^n-1} = \sum_k \frac1{X - \omega^k}$.
  \item جمّع \omterm{def:b1:fractions:field}{الأقطاب} المترافقة للبرهان على التفكيك الحقيقي:
        مع $\theta_k = \frac{2k\pi}n$،
        \[
        \frac{\omega^k}{X - \omega^k} +
        \frac{\omega^{n-k}}{X - \omega^{n-k}}
        = \frac{2\cos\theta_k\,X - 2}
        {X^2 - 2\cos\theta_k\,X + 1} ,
        \]
        واكتب التفكيك الحقيقي الكامل للكسر
        $\frac1{X^n-1}$ (مميّزًا بين $n$ الفرديّ و $n$ الزوجيّ).
  \item خصّص إلى $n = 4$ وتحقق في مقابل
        \cref{exo:b1:fractions:4}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- مجاميع مثلثية، وصيغة أويلر
$\pi^2/6$.} لتكن $P = 1 + X + \dots + X^{n-1}$، وجذورها
$\omega, \omega^2, \dots, \omega^{n-1}$ (وكلها بسيطة).
\begin{enumerate}[resume]
  \item باستعمال $\frac{P'}P = \sum_{k=1}^{n-1}\frac1{X - \omega^k}$
        (\cref{exo:b1:poly:11})، برهن على أن
        \[
        \sum_{k=1}^{n-1}\frac{1}{1 - \omega^k} = \frac{n-1}2 .
        \]
  \item برهن على أن $\dfrac1{1 - \eu^{\iu\theta}} = \dfrac12 +
        \dfrac\iu2\,\frac{\cos(\theta/2)}{\sin(\theta/2)}$ من أجل
        $\theta \notin 2\pi\Z$ (بتعميل نصف الزاوية،
        \cref{met:b1:complex:trig})، واستنتج من السؤال 11
        أن $\sum_{k=1}^{n-1}
        \frac{\cos(k\pi/n)}{\sin(k\pi/n)} = 0$ --- وهو مرئيّ كذلك
        بالتناظر $k \leftrightarrow n - k$.
  \item باشتقاق متطابقة السؤال 11 (أي باستعمال
        $\bigl(\frac{P'}P\bigr)' = \frac{P''}P -
        \bigl(\frac{P'}P\bigr)^2$ محسوبةً عند $X = 1$)، برهن على أن
        \[
        \sum_{k=1}^{n-1}\frac{1}{(1 - \omega^k)^2}
        = \frac{(n-1)(5-n)}{12} .
        \]
  \item بكتابة $\cot t = \frac{\cos t}{\sin t}$، استنتج من
        السؤالين 12--13 الصيغتين المغلقتين
        \[
        \sum_{k=1}^{n-1}\cot^2\frac{k\pi}n = \frac{(n-1)(n-2)}3,
        \qquad
        \sum_{k=1}^{n-1}\frac1{\sin^2\frac{k\pi}n}
        = \frac{n^2-1}3 .
        \]
  \item تحقق من الصيغتين باليد من أجل $n = 3$ ومن أجل $n = 4$.
  \item برهن على المتراجحتين $\cot t < \frac1t < \frac1{\sin t}$
        من أجل $t \in \intoo0{\frac\pi2}$ (من $\sin t < t < \tan
        t$)، واستنتج، من أجل $n = 2m + 1$ و $1 \leq k \leq m$:
        \[
        \cot^2\frac{k\pi}n \;<\; \frac{n^2}{k^2\pi^2}
        \;<\; \frac1{\sin^2\frac{k\pi}n} .
        \]
  \item اجمع هذه المتراجحات من أجل $k = 1, \dots, m$ (مستعملًا
        التناظر $k \leftrightarrow n - k$ لتنصيف صيغ
        السؤال 14) واحصر:
        \[
        \sum_{k=1}^{\infty}\frac1{k^2} = \frac{\pi^2}6 .
        \]
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- لبنات أعلى.}
\begin{enumerate}[resume]
  \item بتربيع تفكيك $\frac1{X^2-1}$ وإعادة تفكيك
        الحدّ المتقاطع، برهن على أن
        \[
        \frac1{(X^2-1)^2}
        = \frac14\Bigl(\frac1{(X-1)^2} + \frac1{(X+1)^2}\Bigr)
        - \frac14\Bigl(\frac1{X-1} - \frac1{X+1}\Bigr),
        \]
        وتحقق من ذلك عند $X = 0$.
  \item اجمع بين السؤال 18 والتلسكوب وقيمة أويلر
        (السؤال 17) للبرهان على أن
        \[
        \sum_{n=2}^{\infty}\frac1{(n^2-1)^2}
        = \frac{\pi^2}{12} - \frac{11}{16} ,
        \]
        وأكّد القيمة عدديًا بثلاثة أرقام عشرية.
  \item اشتقّ متطابقة السؤال 8 للحصول على
        صيغة مغلقة للمقدار $\sum_{k=0}^{n-1}\frac1{(X -
        \omega^k)^2}$، وتحقق منها عند $X = 2$ و $n = 2$.
  \item برهن على أنه من أجل $k \geq 2$،
        \[
        \sum_{n=k}^{\infty}\frac{1}{\binom nk} = \frac{k}{k-1} .
        \]
        \emph{(ردّ ذلك إلى السؤال 4 بكتابة $1/\binom nk$
        بالعوامل.)}
  \item من أجل $k = 3$، أعط المجموع الجزئي المضبوط
        $\sum_{n=3}^{N}\frac1{\binom n3}$ ونهايته.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- توليفة ختامية.}
\begin{enumerate}[resume]
  \item حسابًا ختاميًا، اكتب التفكيك الحقيقي الكامل
        للكسر $\dfrac1{X^6 - 1}$ وتحقق منه عند $X =
        0$.
  \item أين استعملت المسألة بالضبط: (أ) وحدانية
        التفكيك؛ (ب) \omterm{def:b1:complex:unity}{جذور الوحدة} في
        \cref{ch:b1:complex}؛ (ج) المشتقة اللوغاريتمية
        في \cref{exo:b1:poly:11}؟ جملة واحدة لكلٍّ منها.
  \item توليفة، في فقرة قصيرة: متطابقة جبرية واحدة
        --- أي تفتيت كسر إلى لبنات --- ولّدت مجاميع
        مضبوطة ومتطابقات مثلثية والقيمة $\pi^2/6$. علّق على
        توزيع العمل بين الجبر (التفكيكات
        المضبوطة، الصالحة في كل مكان) والتحليل (النهايات
        والحصر)، وأشر إلى أين يُصنَّع كل خيط:
        التلسكوب والمقارنة في
        \cref{ch:b1:series}، ومكاملة اللبنات في
        \cref{ch:b1:integration}.
\end{enumerate}
\end{problem}
