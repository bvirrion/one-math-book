\chapter{دوال المتغيرين}\label{ch:b1:multivar}

تنتهي السنة بأول تجوال في بُعد أعلى: أي الدوال
$f(x, y)$ في متغيرين حقيقيين. وكل شيء يتعمّم --- النهايات و
\omterm{def:b1:continuity:continuous}{الاتصال} والمشتقات والقيم الحدّية --- لكن كل مفهوم يكتسب التواءً:
فالنهايات يمكن الاقتراب منها على امتداد كل اتجاه دفعة واحدة، وتنقسم المشتقات
إلى جزئية، ويشير \omterm{def:b1:multivar:partial}{التدرج} إلى طريق الصعود. وأمّا النظرية
الكاملة (التفاضلات و $\R^n$ العام والمتنوعات الجزئية) فتنتمي إلى
السنة الثانية؛ ونهيّئ هنا المفردات والمبرهنات الأمينة
الأولى.

\section{المستوي $\R^2$؛ الاتصال}

\begin{definition}\label{def:b1:multivar:topology}
على $\R^2$، استعمل المعيار الإقليدي $\norm{(x,y)} = \sqrt{x^2 + y^2}$
(\cref{ch:b1:euclid}). وتُعرَّف الكرات \omterm{def:b1:topology:open}{المفتوحة} والجوارات و\emph{المجموعات
الجزئية \omterm{def:b1:topology:open}{المفتوحة}} من $\R^2$ تمامًا كما في
\cref{ch:b1:topology}، بالكرات بدل الفترات. وتكون الدالة
$f \colon U \to \R$ ($U \subseteq \R^2$ \omterm{def:b1:topology:open}{مفتوحة}) \emph{\omterm{def:b1:continuity:continuous}{متصلة}
عند} $a \in U$ إذا كان
\[
\forall\varepsilon > 0,\ \exists\delta > 0, \quad
\norm{X - a} \leq \delta \implies \abs{f(X) - f(a)} \leq
\varepsilon,
\]
بالتمييز التتالي نفسه كما في متغير واحد.
وتحفظ المجاميع والجداءات والقسمة والتراكيب مع الدوال \omterm{def:b1:continuity:continuous}{المتصلة} في
متغير واحد الاتصالَ؛ وتطبيقا الإحداثيتين
متصلان، ومنه فكثيرات الحدود في $(x,y)$ كذلك.
\end{definition}

\begin{example}[الحاصر القطبي، الكيفية النظيفة للبرهان على نهاية]
\label{ex:b1:multivar:polarbound}
بيّن أن $f(x, y) = \dfrac{x^2y^2}{x^2 + y^2}$ (مع $f(0,0) =
0$) \omterm{def:b1:continuity:continuous}{متصلة} عند المبدأ. وبالإحداثيات القطبية $x =
\rho\cos\theta$ و $y = \rho\sin\theta$:
\[
\abs{f} = \frac{\rho^4\cos^2\theta\sin^2\theta}{\rho^2}
= \rho^2\,(\cos\theta\sin\theta)^2 \leq \frac{\rho^2}{4}
\xrightarrow[\rho \to 0]{} 0 ,
\]
أي حاصر \emph{مستقل عن $\theta$}: فمهما كان اتجاه
الاقتراب، تُحصر القيم إلى $0$. وذلك الانتظام في
$\theta$ هو كل المقصود --- فحاصر مثل $\abs g =
\abs{\cos\theta\sin\theta}$ (بلا $\rho$ باقٍ) لا يبرهن على شيء،
وفعلًا فإن ذلك المقدار $g$ هو الفخّ القطري المنقطع في
المثال التالي.
\end{example}

\begin{example}[الفخّ القطري]\label{ex:b1:multivar:trap}
لتكن $f(x, y) = \dfrac{xy}{x^2 + y^2}$ من أجل $(x,y) \neq (0,0)$ و $f(0,
0) = 0$. فعلى امتداد كل محور، $f = 0 \to 0$؛ لكن على امتداد القطر $y
= x$، $f(x, x) = \frac12 \not\to 0$. \emph{ولا نهاية عند المبدأ}:
فالاقتراب على امتداد كل مستقيم، بل وإيجاد النهاية نفسها على امتداد
كلٍّ، لا يكفي (فنهايات المستقيمات هنا لا تتفق؛ وأمثلة أسوأ
تتفق على امتداد كل المستقيمات ومع ذلك تفشل على امتداد قطع مكافئ،
\cref{exo:b1:multivar:3}). \omterm{def:b1:continuity:continuous}{فالاتصال} في كل متغير على حدة
لا يستلزم \omterm{def:b1:continuity:continuous}{الاتصال}.
\end{example}

\section{المشتقات الجزئية}

\begin{definition}\label{def:b1:multivar:partial}
\emph{المشتقات الجزئية}\index{مشتق جزئي} للدالة $f$ عند
$(a, b)$ هي المشتقات في متغير واحد على امتداد المحاور:
\[
\frac{\partial f}{\partial x}(a,b)
= \lim_{h \to 0} \frac{f(a + h, b) - f(a,b)}{h},
\qquad
\frac{\partial f}{\partial y}(a,b)
= \lim_{k \to 0} \frac{f(a, b + k) - f(a,b)}{k}.
\]
وتكون $f$ من الصنف $C^1$ على $U$ إذا وُجد كلاهما وكانا متصلين على
$U$. و\emph{التدرج}\index{تدرج} هو $\nabla f(a,b) =
\bigl(\frac{\partial f}{\partial x},\, \frac{\partial f}{\partial
y}\bigr)(a,b)$.
\end{definition}

\begin{theorem}[الصنف $C^1$ يستلزم مستويًا مماسًا]\label{thm:b1:multivar:c1}
لتكن $f$ من الصنف $C^1$ على $U$ وليكن $(a,b) \in U$. عندئذ، عندما $(h, k) \to
(0,0)$:
\[
f(a + h, b + k) = f(a, b)
+ h\,\frac{\partial f}{\partial x}(a,b)
+ k\,\frac{\partial f}{\partial y}(a,b)
+ o\bigl(\norm{(h,k)}\bigr) .
\]
وعلى الخصوص تكون $f$ \omterm{def:b1:continuity:continuous}{متصلة}، ويكون للمنحني $z = f(x,y)$ عند
كل نقطة المستوي المماس المقروء من الصيغة.
\end{theorem}

\begin{example}[التقريب الخطي في العمل]
\label{ex:b1:multivar:linapprox}
قدّر $f(1.02,\ 0.99)$ من أجل $f(x, y) = x^3y^2$. وعند $(1, 1)$:
$f = 1$ و $\frac{\partial f}{\partial x} = 3x^2y^2 = 3$ و
$\frac{\partial f}{\partial y} = 2x^3y = 2$، ومنه يعطي
\cref{thm:b1:multivar:c1}
\[
f(1.02,\ 0.99) \approx 1 + 3\,(0.02) + 2\,(-0.01) = 1.04 ,
\]
إزاء القيمة الحقيقية $1.02^3 \times 0.99^2 = 1.04006\dots$ ---
والخطأ من الرتبة الثانية في التزايدات، كما تعد
$o(\norm{(h,k)})$. والمستوي المماس للمنحني عند
$(1, 1, 1)$ هو $z = 1 + 3(x - 1) + 2(y - 1)$، وهي المعادلة
الضمنية في التقدير.
\end{example}

\begin{proof}
حرّك إحداثية واحدة في كل مرة:
\[
f(a+h, b+k) - f(a,b)
= \bigl[f(a+h, b+k) - f(a, b+k)\bigr] + \bigl[f(a, b+k) -
f(a,b)\bigr].
\]
وبمبرهنة التزايدات المنتهية في متغير واحد
(\cref{thm:b1:derivative:mvt})، يكون ما بين القوسين الأول $h\,
\frac{\partial f}{\partial x}(a + \theta h,\, b + k)$ من أجل
$\theta \in \intoo{0}{1}$ ما، ويكون الثاني $k\,\frac{\partial
f}{\partial y}(a,\, b + \theta' k)$. ويتيح \omterm{def:b1:continuity:continuous}{اتصال} \omterm{def:b1:multivar:partial}{المشتقات الجزئية} عند
$(a,b)$ كتابةَ كلٍّ في صورة (القيمة عند $(a,b)$) $+$
(خطأ $\to 0$)؛ ويكون الخطأ الكلي $h\,\varepsilon_1 +
k\,\varepsilon_2 = o(\norm{(h,k)})$ لأن $\abs h, \abs k \leq
\norm{(h,k)}$.
\end{proof}

\begin{theorem}[قاعدة السلسلة]\label{thm:b1:multivar:chain}
لتكن $f$ من الصنف $C^1$ على $U$ ولتكن $t \mapsto (x(t), y(t))$ من الصنف $C^1$ من
\omterm{prop:b1:reals:intervals}{فترة} إلى $U$. عندئذ تكون $g(t) = f\bigl(x(t), y(t)\bigr)$
من الصنف $C^1$، مع\index{قاعدة السلسلة!في متغيرين}
\[
g'(t) = x'(t)\,\frac{\partial f}{\partial x}\bigl(x(t),y(t)\bigr)
+ y'(t)\,\frac{\partial f}{\partial y}\bigl(x(t),y(t)\bigr)
= \bigl\langle \nabla f,\ (x', y')\bigr\rangle .
\]
\end{theorem}

\begin{proof}
طبّق \cref{thm:b1:multivar:c1} عند $(x(t), y(t))$ مع $(h, k) =
(x(t+s) - x(t),\, y(t+s) - y(t))$: فعندما $s \to 0$، تعطي القابلية للاشتقاق في متغير
واحد أن $h = s\,x'(t) + o(s)$ و $k = s\,y'(t)
+ o(s)$، ومنه $\norm{(h, k)} = O(s)$ و
\[
g(t+s) - g(t)
= h\,\frac{\partial f}{\partial x} + k\,\frac{\partial
f}{\partial y} + o\bigl(\norm{(h,k)}\bigr)
= s\,\Bigl(x'\,\frac{\partial f}{\partial x} +
y'\,\frac{\partial f}{\partial y}\Bigr) + o(s) ,
\]
ويستوعب الخطأ النهائي المقدارين $o(s)$ في $h$ و $k$ معًا
(مضروبين في القيم المثبَّتة للمشتقات الجزئية) و
$o(O(s))$ من تقدير المستوي المماس. فاقسم على $s$ ودع
$s \to 0$. ويتبع \omterm{def:b1:continuity:continuous}{اتصال} $g'$ من \omterm{def:b1:continuity:continuous}{اتصال} كل
المكوّنات.
\end{proof}

\begin{example}[قاعدة السلسلة، متحققًا منها بكيفيتين]
\label{ex:b1:multivar:chainrun}
لتكن $f(x, y) = x^2 y$ و $g(t) = f(t, t^2)$. ومباشرةً: $g(t) =
t^2\cdot t^2 = t^4$، ومنه $g'(t) = 4t^3$. وبقاعدة السلسلة:
$\frac{\partial f}{\partial x} = 2xy$ و $\frac{\partial
f}{\partial y} = x^2$، مقوَّمتين على امتداد المنحني $(t, t^2)$:
\[
g'(t) = 1\cdot 2t\cdot t^2 + 2t\cdot t^2 = 2t^3 + 2t^3 = 4t^3 .
\]
فيتفق الحسابان، والانقسام ذو معنى: إذ يأتي $2t^3$
من النمو من التحرك \emph{يمينًا} عبر ميل
$x$، و $2t^3$ من التحرك \emph{صعودًا} عبر ميل
$y$. وعلى المنحنيات التي لا توجد فيها صورة مغلقة للمقدار $g$، لا
ينجو إلا الحساب الثاني --- وذلك هو مقصود المبرهنة.
\end{example}

\begin{remark}[قراءة التدرج]
على امتداد اتجاه واحدي $u$، تعطي قاعدة السلسلة على $t \mapsto f(a + tu)$
\emph{المشتقَ الاتجاهي} $\langle \nabla f(a),
u\rangle$: وهو أعظمي عندما يشير $u$ على امتداد $\nabla f(a)$ (حسب
كوشي--شوارتز، \cref{thm:b1:euclid:cs}). ومنه \omterm{def:b1:multivar:partial}{فالتدرج} هو
اتجاه أشدّ صعود، وهو متعامد مع منحنيات
المستوى $\{f = c\}$ (اشتق $f$ على امتداد منحن مرسوم في
\omterm{def:b1:logic:sets}{مجموعة} مستوى: فتعطي قاعدة السلسلة $\langle\nabla f,\,
\text{مماس}\rangle = 0$).
\end{remark}

\begin{example}[منحنيات المستوى والتدرجات، على دالة واحدة]
\label{ex:b1:multivar:levelcurves}
خذ $f(x, y) = x^2 - y^2$. ومجموعات مستواها: $\{f = c\}$ قطعٌ
زائد مفتوح يمينًا ويسارًا من أجل $c > 0$، وصعودًا ونزولًا من أجل $c < 0$،
وزوج المستقيمين المتقاطعين $y = \pm x$ من أجل $c = 0$ --- أي
خريطة كنتور لممرّ جبلي، \omterm{met:b1:multivar:monge}{بنقطة السرج} عند
المبدأ حيث يتقاطع مستقيما المستوى المعدوم. \omterm{def:b1:multivar:partial}{والتدرج}: $\nabla
f = (2x, -2y)$. وعند النقطة $(2, 1)$ (على المستوى $c = 3$):
$\nabla f = (4, -2)$، بينما تكون متجهة مماس منحني
المستوى، معلَّمًا بجوار تلك النقطة بالمقدار $\bigl(t,
\sqrt{t^2 - 3}\bigr)$، هي $\bigl(1, \frac{t}{\sqrt{t^2 -
3}}\bigr) = (1, 2)$ عند $t = 2$ --- وفعلًا
\[
\langle (4, -2),\ (1, 2)\rangle = 4 - 4 = 0 :
\]
\omterm{def:b1:multivar:partial}{فالتدرج} عمودي على الكنتور، مشيرًا نحو قيم
$f$ الأعلى (وهنا: بعيدًا عن محور $y$). وقراءتان
أخريان: ينعدم \omterm{def:b1:multivar:partial}{التدرج} عند السرج بالضبط، حيث
تنقبض خريطة الكنتور؛ ويكون \emph{مستقيم} المماس لمنحني
المستوى عند $(2,1)$ هو $4(x - 2) - 2(y - 1) = 0$، أي $2x -
y = 3$ --- وهي المعادلة «$\langle \nabla f,\ M - M_0\rangle =
0$» التي تعمّم مماس القطع الناقص في
\cref{exo:b1:curves:11}.
\end{example}

\begin{theorem}[شوارتز]\label{thm:b1:multivar:schwarz}
إذا كانت $f$ من الصنف $C^2$ (أي توجد \omterm{def:b1:multivar:partial}{المشتقات الجزئية} للمشتقات الجزئية
وتكون \omterm{def:b1:continuity:continuous}{متصلة})، فإن\index{مبرهنة شوارتز}
\[
\frac{\partial^2 f}{\partial x\,\partial y}
= \frac{\partial^2 f}{\partial y\,\partial x} .
\]
\end{theorem}
\admitted

\section{القيم الحدّية الموضعية}

\begin{method}[دراسات القيم الحدّية، منظَّمة]
\label{met:b1:multivar:extremamethod}
\begin{enumerate}
  \item \emph{حُلَّ $\nabla f = 0$ حلًا كاملًا.} عمّل كل
        \omterm{def:b1:derivative:def}{مشتقة} جزئية كلما أمكن (فجداءات العوامل الخطية
        تقسم الجملة إلى حالات شفافة، كما في
        \cref{ex:b1:multivar:fourpoints} أدناه)؛ فالحالة
        المنسية نقطةٌ حرجة منسية.
  \item \emph{صنّف كل نقطة} بمعطيات مونج $r, s,
        t$ --- معادةِ الحساب \emph{عند كل نقطة}، لا مرة واحدة
        وإلى الأبد.
  \item \emph{وإذا كان $rt - s^2 = 0$}، فافحص $f$ مباشرةً على امتداد
        منحنيات محسنة الاختيار عبر النقطة (المستقيمات أولًا، ثم
        القطوع المكافئة)، باحثًا إمّا عن إشارتين (فلا قيمة حدّية)
        وإمّا عن إشارة مقفلة بحجة تغطي كل
        الاتجاهات.
  \item \emph{وتراجع لترى الصورة الشاملة}: تحقق من
        السلوك عند اللانهاية (فالقيمة الصغرى الموضعية قد لا تكون
        شاملة)، وإذا لم تكن \omterm{def:b1:logic:sets}{مجموعة} التعريف \omterm{def:b1:topology:open}{مفتوحة}، فعالج حافتها
        على حدة (\cref{exo:b1:multivar:12}) --- فمبرهنة النقاط
        الحرجة لا ترى إلا النقاط الداخلية.
\end{enumerate}
\end{method}

\begin{theorem}[النقاط الحرجة]\label{thm:b1:multivar:critical}
إذا كان للدالة $f$ (من الصنف $C^1$ على \omterm{def:b1:topology:open}{المفتوحة} $U$) قيمةٌ حدّية موضعية عند $(a,b)
\in U$، فإن $\nabla f(a,b) = (0,0)$: أي إن النقطة
\emph{حرجة}\index{نقطة حرجة}.
\end{theorem}

\begin{proof}
للدالتين في متغير واحد $x \mapsto f(x, b)$ و $y \mapsto f(a,
y)$ قيمتان حدّيتان موضعيتان داخليتان عند $a$ و $b$ على الترتيب:
ومنه يقتل \cref{prop:b1:derivative:fermat} المشتقتين الجزئيتين معًا.
\end{proof}

\begin{method}[اختبار الرتبة الثانية (ترميز مونج)]\label{met:b1:multivar:monge}
عند \omterm{thm:b1:multivar:critical}{نقطة حرجة} لدالة من الصنف $C^2$، ضع
\[
r = \frac{\partial^2 f}{\partial x^2},
\qquad
s = \frac{\partial^2 f}{\partial x \partial y},
\qquad
t = \frac{\partial^2 f}{\partial y^2}
\qquad (\text{القيم عند النقطة}).
\]
\begin{itemize}
  \item إذا كان $rt - s^2 > 0$: فقيمة حدّية موضعية --- صغرى من أجل $r > 0$،
        وعظمى من أجل $r < 0$؛
  \item وإذا كان $rt - s^2 < 0$: فلا قيمة حدّية (أي \emph{نقطة
        سرج}\index{نقطة سرج})؛
  \item وإذا كان $rt - s^2 = 0$: فالاختبار صامت؛ افحص مباشرةً.
\end{itemize}
(والتسويغ --- نشر تايلور--يونغ من الرتبة $2$ ودراسة
إشارة الشكل التربيعي $r h^2 + 2shk + tk^2$ --- يُجرى
في السنة الثانية؛ ويُستعمل الاختبار هنا أداةَ عمل.)
\end{method}

\begin{example}\label{ex:b1:multivar:extrema}
$f(x,y) = x^3 + y^3 - 3xy$. والنقاط الحرجة: يعطي $\nabla f = (3x^2 -
3y,\; 3y^2 - 3x) = 0$ أن $y = x^2$ و $x = y^2$، ومنه $x = x^4$:
$x \in \{0, 1\}$: أي النقطتان $(0,0)$ و $(1,1)$.

والمشتقات الثانية: $r = 6x$ و $s = -3$ و $t = 6y$. وعند $(0,0)$: $rt -
s^2 = -9 < 0$: أي سرج. وعند $(1,1)$: $rt - s^2 = 36 - 9 > 0$ و $r = 6
> 0$: أي قيمة صغرى موضعية، $f(1,1) = -1$. (وليست شاملة: $f(x, 0) = x^3 \to
-\infty$.)
\end{example}

\begin{example}[دراسة بأربع نقاط، كاملة]
\label{ex:b1:multivar:fourpoints}
$f(x, y) = xy\,(3 - x - y) = 3xy - x^2y - xy^2$. \omterm{def:b1:multivar:partial}{والتدرج}:
\[
\frac{\partial f}{\partial x} = y\,(3 - 2x - y),
\qquad
\frac{\partial f}{\partial y} = x\,(3 - x - 2y).
\]
والنقاط الحرجة: إذا كان $y = 0$، أعطت المعادلة الثانية $x \in
\{0, 3\}$؛ وإذا كان $x = 0$، أعطت الأولى $y \in \{0, 3\}$؛ وإذا كان $xy
\neq 0$، فحُلَّ $2x + y = 3$ و $x + 2y = 3$: ومنه $x = y = 1$. أي أربع
نقاط: $(0,0)$ و $(3,0)$ و $(0,3)$ و $(1,1)$. والمشتقات الثانية:
$r = -2y$ و $s = 3 - 2x - 2y$ و $t = -2x$.
\begin{itemize}
  \item $(1,1)$: $r = -2$ و $s = -1$ و $t = -2$: ومنه $rt - s^2 = 3 >
        0$ و $r < 0$: أي قيمة عظمى موضعية، $f(1,1) = 1$.
  \item $(0,0)$: $r = t = 0$ و $s = 3$: ومنه $rt - s^2 = -9 < 0$:
        أي سرج؛ وكذلك $(3, 0)$ ($s = -3$) و $(0, 3)$: أي ثلاثة
        سروج.
\end{itemize}
والقيمة العظمى موضعية فقط: $f(-T, -T) = T^2(3 + 2T) \to
+\infty$. وتحقق التناظر: $f(x, y) = f(y, x)$، وفعلًا تكون
\omterm{def:b1:logic:sets}{المجموعة} الحرجة والتصنيف متناظرين في $x
\leftrightarrow y$. والتفسير: بين المستطيلات ذات الفسحة
$x, y \geq 0$ و $x + y \leq 3$، يكون الجداء $xy(3 - x - y)$ لِما في «أجزاء»
$3$ الثلاثة أكبرَ ما يكون عندما تتساوى الأجزاء
--- وهو ظلٌّ في متغيرين لمبدأ المتوسطين الحسابي والهندسي.
\end{example}

\begin{remark}[مزالق شائعة]
\label{rem:b1:multivar:pitfalls}
\emph{قد توجد \omterm{def:b1:multivar:partial}{المشتقات الجزئية} عند نقطة
انقطاع}: فللفخّ القطري $g(x,y) = \frac{xy}{x^2+y^2}$ في
\cref{ex:b1:multivar:trap} لدينا $\frac{\partial g}{\partial
x}(0,0) = \frac{\partial g}{\partial y}(0,0) = 0$ (لأن قصريه على
المحورين معدومان تمامًا)، ومع ذلك لا نهاية للمقدار $g$ عند
المبدأ --- \omterm{def:b1:multivar:partial}{فالمشتقات الجزئية} لا تسبر إلا اتجاهين، ويحتاج \omterm{def:b1:continuity:continuous}{الاتصال}
إلى كلها؛ ولا يعيد النظامَ إلا فرض الصنف $C^1$
(\cref{thm:b1:multivar:c1}).
\emph{نهايات المستقيمات لا تكفي أبدًا}: فلدالة \cref{exo:b1:multivar:3}
نهايةٌ $0$ على امتداد كل مستقيم ومع ذلك لا نهاية ---
فجرّب دائمًا القطوع المكافئة (أو حواصر قطبية صحيحة انتظامًا في
$\theta$).
\emph{الحرج لازم لا كافٍ}: فالسروج كثيرة
(ثلاث من أربع نقاط في \cref{ex:b1:multivar:fourpoints})؛ و
تصحّ المبرهنة على المجموعات \emph{\omterm{def:b1:topology:open}{المفتوحة}} فقط --- فعلى قرص مغلق،
قد تقع القيم الحدّية على \omterm{def:b1:topology:closure}{الحافة} \omterm{def:b1:multivar:partial}{بتدرج} غير معدوم
(\cref{exo:b1:multivar:12}).
\emph{الحالة الصامتة $rt - s^2 = 0$ صامتة حقًا}: فلكلٍّ من $x^4
+ y^4$ (بقيمة صغرى) و $x^3 + y^3$ (بلا قيمة حدّية) $r = s = t
= 0$ عند المبدأ؛ ولا تفصل إلا دراسة إشارة مباشرة
(\cref{exo:b1:multivar:6}، الدالة $h$).
\emph{\omterm{def:b1:multivar:partial}{التدرج} متعامد مع منحنيات المستوى لا موازٍ
لها}: فلتتبّع خط كنتور، تحرّك عموديًا على $\nabla
f$؛ وللصعود أسرع ما يكون، تحرّك على امتداده --- وخلط الاثنين يقلب
هندسة كل خريطة كنتور.
\end{remark}

\begin{remark}[إلى أين يقود المتغيران]
\label{rem:b1:multivar:whereused}
هذا الفصل مدخل. \omterm{def:b1:multivar:partial}{فالتدرج} وقاعدة السلسلة يمتدان
حرفيًا إلى $n$ متغيرًا في مجلّد السنة~الثانية، حيث يصير
$o(\norm{(h,k)})$ في \cref{thm:b1:multivar:c1} هو
\emph{التفاضل} ويُبرهن على اختبار مونج برهانًا كاملًا عبر
صيغة تايلور من الرتبة الثانية والأشكال التربيعية. وأمّا الحالة الخاصة
التي يمكن حسمها هذه السنة --- وهي الدوال التربيعية،
التي يكون النشر من الرتبة الثانية لها مضبوطًا --- فهي موضوع
مسألة نهاية الأسبوع، وتصادف أنها الحالة التي تدير
مقايسة معطيات العالم: انحدار \omterm{pb:b1:multivar:1}{المربعات الصغرى}. وأمّا القيم
الحدّية المقيَّدة (\cref{exo:b1:multivar:5} كان عرضًا مسبقًا) فتصير مضاعفات
لاغرانج في السنة~الثانية؛ وتعود الدوال التوافقية
(\cref{exo:b1:multivar:7}) في التحليل العقدي في مجلّد
السنة~الثالثة.
\end{remark}

\begin{remark}[منظورات داخل الكتاب 3: السنة، مغلقةً]
\label{rem:b1:multivar:perspectives}
هذا الفصل هو حيث يتصافح نصفا المجلّد. فقد وفّر
شطر التحليل أدواته مشتقةً \omterm{def:b1:derivative:def}{مشتقة}: فمبرهنة
التزايدات المنتهية تحرّك \cref{thm:b1:multivar:c1}، ونشور
تايلور تحرّك اختبارات القيم الحدّية، ومقادير $\varepsilon$ في
\cref{ch:b1:topology} عادت بكرات بدل
الفترات. ووفّر شطر الجبر الهندسة: فيُقرأ \omterm{def:b1:multivar:partial}{التدرج}
عبر \omterm{def:b1:euclid:def}{الجداء السلّمي} في \cref{ch:b1:euclid}
(وتجعله كوشي--شوارتز اتجاهَ أشدّ صعود)، وتكون معطيات
مونج $(r, s, t)$ مصفوفةً متماثلة في
\cref{ch:b1:matrices} باختبار المحدد في
\cref{ch:b1:det}، وتشغّل مسألة نهاية الأسبوع الإسقاطَ
المتعامد على متجهات المعطيات. بل حتى منحنيات
\cref{ch:b1:curves} تعود مجموعاتِ مستوى. والقارئ الذي يستطيع
إعادة بناء سبب عمل كلٍّ من هذه التسليمات الخمس يكون قد
راجع، في الواقع، السنة كلها --- وهذا هو الغرض الحقيقي
من هذا الفصل الأخير.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:multivar:1}
احسب \omterm{def:b1:multivar:partial}{المشتقات الجزئية}:
$f(x,y) = x^2 y + \eu^{xy}$؛ $\;g(x,y) = \ln(x^2 + y^2)$ (على
$\R^2\setminus\{0\}$)؛ $\;h(x,y) = \arctan\frac yx$ (على $x > 0$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:multivar:2}
ادرس \omterm{def:b1:continuity:continuous}{الاتصال} عند $(0,0)$ (بالقيمة $0$ هناك) من أجل:
\[
f(x,y) = \frac{x^2 y}{x^2 + y^2},
\qquad
g(x,y) = \frac{xy}{x^2 + y^2},
\qquad
h(x,y) = \frac{x^3 + y^3}{x^2 + y^2}.
\]
\emph{(\omterm{prop:b1:vspaces:coordinates}{والإحداثيات} القطبية $x = \rho\cos\theta$ و $y = \rho\sin\theta$
تساعد: فحُدّ بدالة في $\rho$ وحده كلما أمكن.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:3}
لتكن $f(x,y) = \dfrac{x^2 y}{x^4 + y^2}$ ($f(0,0) = 0$). برهن على أن
للمقدار $f$ نهايةً $0$ عند المبدأ \emph{على امتداد كل مستقيم}،
لكن $f\bigl(x, x^2\bigr) = \frac12$: ومنه فالدالة $f$ ليست \omterm{def:b1:continuity:continuous}{متصلة} عند
$(0,0)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:multivar:4}
تحقق من مبرهنة شوارتز باليد على $f(x, y) = x^3 y^2 + \sin(xy)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:5}
لتكن $f$ من الصنف $C^1$ على $\R^2$ ولتكن $g(t) = f(\cos t, \sin t)$. عبّر عن
$g'(t)$ عبر قاعدة السلسلة. واستنتج أن $f$ مقصورةً على الدائرة
الواحدية تبلغ قيمها الحدّية عند نقاط يكون فيها $\nabla f$ موازيًا
لمتجهة نصف القطر.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:6}
جد النقاط الحرجة وصنّفها من أجل:
\[
f(x, y) = x^2 + xy + y^2 - 3x,
\qquad
g(x, y) = x^2 - y^2 + 4y,
\qquad
h(x, y) = x^4 + y^4 - 2(x - y)^2 .
\]
\emph{(ومن أجل $h$، يكون اختبار المحدد صامتًا عند المبدأ:
فافحص $h(x, x)$ و $h(x, -x)$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:7}
تكون الدالة $f$ \emph{توافقية} إذا كان $\frac{\partial^2 f}{\partial
x^2} + \frac{\partial^2 f}{\partial y^2} = 0$. تحقق من أن $x^2 -
y^2$ و $xy$ و $\eu^x\cos y$ و $\ln(x^2 + y^2)$ (خارج المبدأ)
توافقية.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:8}
جد نقطة المستوي $\{x + 2y - z = 4\} \subseteq \R^3$
الأقرب إلى المبدأ، بكيفيتين: \omterm{thm:b1:euclid:projection}{بالإسقاط المتعامد}
(\cref{ch:b1:euclid})، وبتصغير دالة المتغيرين
$f(x, y) = x^2 + y^2 + (x + 2y - 4)^2$ المحصَّلة بإزالة $z$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:9}
(اللابلاسي \omterm{prop:b1:vspaces:coordinates}{بالإحداثيات} القطبية، لقاء أول) لتكن $f$ من الصنف $C^2$ على
$\R^2 \setminus \{0\}$ و\emph{شعاعية}: $f(x, y) =
\varphi\bigl(\sqrt{x^2+y^2}\bigr)$ مع $\varphi$ من الصنف $C^2$ على
$\intoo{0}{+\infty}$. برهن على أن
\[
\frac{\partial^2 f}{\partial x^2} + \frac{\partial^2 f}{\partial
y^2} = \varphi''(\rho) + \frac{\varphi'(\rho)}{\rho},
\qquad \rho = \sqrt{x^2 + y^2},
\]
وجد كل الدوال التوافقية الشعاعية على المستوي المثقوب.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:10}
أعطِ معادلة المستوي المماس للمنحني $z = xy$ عند
النقطة $(1, 1, 1)$. ثم جد كل نقاط منحني
$f(x,y) = x^3 + y^3 - 3xy$ التي يكون عندها المستوي المماس
\emph{أفقيًا}، واربط الجواب بالمقدار
\cref{ex:b1:multivar:extrema}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:11}
لتكن $f$ من الصنف $C^1$ على $\R^2$.
\begin{enumerate}
  \item إذا كان $\frac{\partial f}{\partial x} = 0$ في كل مكان، فبرهن
        على أن $f(x, y)$ لا تتعلق إلا بالمقدار $y$.
  \item جد كل حلول المعادلة
        $\frac{\partial f}{\partial x} = \frac{\partial
        f}{\partial y}$ من الصنف $C^1$ على $\R^2$. \emph{(ضع $g(u, v) =
        f\bigl(\frac{u+v}2, \frac{u-v}2\bigr)$ واحسب
        $\frac{\partial g}{\partial v}$ بقاعدة السلسلة.)}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:12}
جد القيمة العظمى والصغرى الشاملتين للدالة $f(x, y) = xy$ على
القرص المغلق $x^2 + y^2 \leq 1$. \emph{(اقبل مبرهنة القيم
الحدّية في متغيرين: أي إن الدالة \omterm{def:b1:continuity:continuous}{المتصلة} على القرص المغلق
تبلغ حاصريها --- وهي مبرهن عليها في مجلّد السنة~الثانية. وعالج
القرص المفتوح بالنقاط الحرجة والدائرةَ الحدّية
بتعليم \cref{exo:b1:multivar:5}.)}
\end{exercise}

\section{مسألة: المربعات الصغرى ومستقيم الانحدار}

\begin{problem}\label{pb:b1:multivar:1}
من أجل $n$ نقطة معطيات، أيّ مستقيم يمرّ «أقرب» إلى
كلها؟ أجاب لوجاندر وغاوس: المستقيم الذي يصغّر
مجموع الأخطاء العمودية \emph{المربَّعة} --- لأن ذلك
التصغير قابل للحلّ بالضبط بالجبر الخطي. وتبرهن هذه المسألة
أولًا على اختبار مونج في \cref{met:b1:multivar:monge}
برهانًا أمينًا من أجل الدوال التربيعية (وهي الحالة الوحيدة التي يكون
فيها النشر من الرتبة الثانية مضبوطًا)، ثم تبني \omterm{pb:b1:multivar:1}{المعادلات
الناظمية} \omterm{pb:b1:multivar:1}{ومستقيم الانحدار} ومعامل الارتباط
فوق الهندسة الإقليدية في
\cref{ch:b1:euclid}.\index{مربعات صغرى}\index{مستقيم الانحدار}
\index{معادلات ناظمية}\index{ارتباط}

\medskip
\noindent\textbf{الجزء 1 --- الدوال التربيعية: اختبار مونج،
مبرهنًا عليه.} ثبّت أعدادًا حقيقية $r, s, t$ ولتكن $q(h, k) = r h^2 + 2s hk +
t k^2$.
\begin{enumerate}
  \item افترض $r \neq 0$. أثبت صورة إتمام المربع
        \[
        q(h, k) = r\Bigl(h + \frac{s}{r}k\Bigr)^{\!2}
        + \frac{rt - s^2}{r}\,k^2 ,
        \]
        واستنتج: أنه إذا كان $rt - s^2 > 0$، كان للمقدار $q$ إشارة $r$
        القطعية خارج المبدأ؛ وإذا كان $rt - s^2 < 0$، أخذ $q$
        الإشارتين.
  \item احسم الحالتين الباقيتين: $r = 0$ و $t \neq 0$
        (بالتناظر)؛ و $r = t = 0$ ($q = 2shk$). واختم:
        يأخذ $q$ الإشارتين إذا وفقط إذا كان $rt - s^2 < 0$، و
        ينعدم $q$ عند المبدأ فقط إذا وفقط إذا كان $rt - s^2 > 0$.
  \item ولتكن الآن $f(x, y) = \frac12\bigl(r x^2 + 2s xy + t
        y^2\bigr) + \beta x + \gamma y + \delta$ دالةً تربيعية
        لها \omterm{thm:b1:multivar:critical}{نقطة حرجة} $X_0 = (a, b)$. برهن على النشر
        \emph{المضبوط}
        \[
        f(X_0 + (h,k)) = f(X_0) + \tfrac12\,q(h, k)
        \qquad (\text{بلا باقٍ}),
        \]
        واستنتج تصنيف مونج من أجل الدوال
        التربيعية: قيمة صغرى شاملة قطعية إذا كان $rt - s^2 > 0$ و $r >
        0$؛ وقيمة عظمى شاملة قطعية إذا كان $rt - s^2 > 0$ و $r < 0$؛
        وسرج إذا كان $rt - s^2 < 0$.
  \item افترض $rt - s^2 > 0$ و $r > 0$ (ومنه $t > 0$ كذلك).
        برهن على حاصر القسر الصريح
        \[
        q(h, k) \geq c\,(h^2 + k^2),
        \qquad
        c = \frac{rt - s^2}{r + t} > 0 .
        \]
        \emph{(بيّن أن الشكل التربيعي $q - c(h^2 + k^2)$،
        بمعاملات $r - c$ و $s$ و $t - c$، لا تزال معطيات
        اختبار مميّزه غير سالبة.)}
  \item استنتج أن الدالة التربيعية ذات الجزء التربيعي المعيَّن
        الموجب تؤول إلى $+\infty$ عندما $\norm{(x,y)} \to
        \infty$، ومنه فلها مصغِّر \emph{شامل وحيد}: هو
        نقطتها الحرجة. وقابل ذلك
        بالمقدار \cref{ex:b1:multivar:extrema}، حيث لم تكن قيمة صغرى موضعية
        لدالة غير تربيعية شاملةً.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- \omterm{pb:b1:multivar:1}{المعادلات الناظمية}.} لتكن $C_1,
C_2, b \in \R^n$ (\omterm{def:b1:euclid:def}{بالجداء السلّمي} القانوني)، و
\[
f(u, v) = \norm{u\,C_1 + v\,C_2 - b}^2 .
\]
\begin{enumerate}[resume]
  \item انشر $f$ وبيّن أنها دالة تربيعية في $(u,
        v)$ بمعاملات جزء تربيعي $r = 2\norm{C_1}^2$ و
        $s = 2\langle C_1, C_2\rangle$ و $t = 2\norm{C_2}^2$.
  \item بيّن أن $rt - s^2 > 0$ إذا وفقط إذا كانت $(C_1, C_2)$
        \omterm{def:b1:vspaces:free}{حرة} \emph{(بحالة المساواة في كوشي--شوارتز،
        \cref{thm:b1:euclid:cs})}؛ وافترض هذا من الآن فصاعدًا.
  \item بيّن أن المعادلات الحرجة $\nabla f = 0$ هي
        \emph{\omterm{pb:b1:multivar:1}{المعادلات الناظمية}}
        \[
        \begin{pmatrix}
        \norm{C_1}^2 & \langle C_1, C_2\rangle\\
        \langle C_1, C_2\rangle & \norm{C_2}^2
        \end{pmatrix}
        \begin{pmatrix} u\\ v\end{pmatrix}
        =
        \begin{pmatrix} \langle C_1, b\rangle\\ \langle C_2,
        b\rangle\end{pmatrix}
        \]
        --- بمصفوفة غرام للعائلة \cref{exo:b1:euclid:11} على
        اليسار --- وأنها تقول بالضبط: $b - (uC_1 + vC_2)
        \perp C_1, C_2$. واختم مع الجزء 1 و
        \cref{thm:b1:euclid:projection}: فيعطي المصغِّر الوحيد
        $p = uC_1 + vC_2 =$ \omterm{thm:b1:euclid:projection}{الإسقاط المتعامد} للمقدار
        $b$ على $\operatorname{Vect}(C_1, C_2)$.
  \item بيّن أن القيمة الصغرى هي $\norm{b}^2 - \langle p,
        b\rangle = \norm{b - p}^2$، وارسم الصورة
        الفيثاغورية.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- \omterm{pb:b1:multivar:1}{مستقيم الانحدار}.} نقاط المعطيات
$(x_1, y_1), \dots, (x_n, y_n)$، وليست كل $x_i$ متساوية. صغّر
\[
E(\alpha, \beta) = \sum_{i=1}^{n}\bigl(y_i - \alpha - \beta
x_i\bigr)^2 .
\]
واكتب $\bar x = \frac1n\sum x_i$ و $\bar y = \frac1n\sum y_i$ و
$v_x = \frac1n\sum x_i^2 - \bar x^2$ و $v_y = \frac1n\sum y_i^2 -
\bar y^2$ و $c_{xy} = \frac1n\sum x_i y_i - \bar x\bar y$.
\begin{enumerate}[resume]
  \item تعرّف على الجزء 2 مع $C_1 = (1, \dots, 1)$ و $C_2 =
        (x_1, \dots, x_n)$ و $b = (y_1, \dots, y_n)$، واكتب
        \omterm{pb:b1:multivar:1}{المعادلات الناظمية}
        \[
        n\alpha + \Bigl(\sum x_i\Bigr)\beta = \sum y_i,
        \qquad
        \Bigl(\sum x_i\Bigr)\alpha + \Bigl(\sum
        x_i^2\Bigr)\beta = \sum x_i y_i .
        \]
  \item حُلَّها:
        \[
        \beta = \frac{c_{xy}}{v_x},
        \qquad
        \alpha = \bar y - \beta\,\bar x ,
        \]
        ولاحظ أن \emph{\omterm{pb:b1:multivar:1}{مستقيم الانحدار}} $y = \alpha
        + \beta x$ يمرّ بنقطة المتوسط $(\bar x, \bar
        y)$.
  \item تحقق من أن $v_x > 0$ بالضبط عندما لا تتساوى كل
        $x_i$، وطابق هذا مع السؤال 7.
  \item برهن على أن الخطأ الأصغري هو
        \[
        E_{\min} = n\Bigl(v_y - \frac{c_{xy}^2}{v_x}\Bigr)
        = n\,v_y\,(1 - \rho^2),
        \qquad
        \rho = \frac{c_{xy}}{\sqrt{v_x v_y}}
        \quad (v_y \neq 0).
        \]
        واستنتج $\abs\rho \leq 1$، مع $\abs\rho = 1$ إذا وفقط
        إذا كانت المعطيات مستقيمية تمامًا.
  \item مثال معالَج: من أجل المعطيات $(0,0)$ و $(1,1)$ و $(2,3)$
        و $(3,4)$، احسب $\bar x, \bar y, v_x, c_{xy}$ و
        \omterm{pb:b1:multivar:1}{مستقيم الانحدار} والبواقي الأربعة و $E_{\min}$.
  \item برهن على أن بواقي $\varepsilon_i = y_i - \alpha -
        \beta x_i$ المستقيم الأمثل تحقق $\sum_i
        \varepsilon_i = 0$ و $\sum_i x_i\varepsilon_i = 0$،
        وفسّر كليهما عبر التعامد.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- تنويعات وتطبيقات.}
\begin{enumerate}[resume]
  \item (أفضل ثابت) بيّن أن الثابت $\alpha$
        المصغِّر للمقدار $\sum_i (y_i - \alpha)^2$ هو المتوسط $\bar
        y$، وأن القيمة الصغرى هي $n\,v_y$: أي إن التباين
        يقيس فشل المعطيات في أن تكون ثابتة.
  \item (المستقيم المارّ بالمبدأ) بيّن أن الميل المصغِّر للمقدار
        $\sum_i (y_i - \beta x_i)^2$ هو $\beta_0 = \frac{\sum
        x_iy_i}{\sum x_i^2}$، وأعطِ شرطًا على المعطيات
        ليتطابق $\beta_0$ مع الميل
        $c_{xy}/v_x$ في السؤال 11.
  \item (المسافة إلى مستقيم، مرة أخرى) من أجل نقطة $M$ والمستقيم
        $D$ المارّ بالمقدار $P_0$ الموجَّه بالمتجهة الواحدية $w$،
        صغّر $g(\tau) = \norm{P_0 + \tau w - M}^2$ و
        استنتج $d(M, D)^2 = \norm{M - P_0}^2 - \langle M - P_0,
        w\rangle^2$؛ واستعد الصيغة $d = \frac{\abs{a x_0 +
        b y_0 + c}}{\sqrt{a^2 + b^2}}$ من أجل المستقيم $ax + by + c
        = 0$ في المستوي.
  \item (تحليل التباين) برهن على التفكيك $v_y =
        \beta^2 v_x + \frac{E_{\min}}n$: أي إن تباين
        المقادير $y_i$ ينقسم إلى الجزء الذي يفسّره المستقيم زائد
        تباين البواقي.
  \item (المقايسة بقطع مكافئ) لمقايسة $y = a + bx + cx^2$، بيّن أن
        \omterm{pb:b1:multivar:1}{المعادلات الناظمية} هي الجملة $3 \times 3$ ذات
        مصفوفة العزوم
        \[
        \begin{pmatrix}
        n & \sum x_i & \sum x_i^2\\
        \sum x_i & \sum x_i^2 & \sum x_i^3\\
        \sum x_i^2 & \sum x_i^3 & \sum x_i^4
        \end{pmatrix},
        \]
        وهي مصفوفة غرام قابلة للقلب ما إن تكون ثلاث من
        المقادير $x_i$ متمايزة \emph{(بحرية $(1, X, X^2)$
        معايَنةً عند المعطيات، \cref{exo:b1:euclid:11}؛ وقارن
        مصفوفات العزوم في مسألة نهاية الأسبوع في
        \cref{ch:b1:det})}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- المتانة، والتوليفة.}
\begin{enumerate}[resume]
  \item صنّف النقاط الحرجة للدوال التربيعية الثلاث
        \[
        x^2 + xy + y^2, \qquad x^2 + 3xy + y^2, \qquad
        x^2 + 2xy + y^2 ,
        \]
        بالجزء 1، معالجًا الحالة الثالثة المنحلّة باليد
        (فأين تُبلغ القيمة الصغرى؟).
  \item (تجربة القيمة الشاذة) ألحق النقطة $(10, 0)$ بمعطيات
        السؤال 14 وأعد حساب الميل $\beta$.
        فماذا حدث، ولماذا يكون الخطأ \emph{المربَّع} شديد
        الحساسية لنقطة بعيدة واحدة؟
  \item (\omterm{pb:b1:multivar:1}{المربعات الصغرى} الموزونة) من أجل أوزان $w_i > 0$ معطاة،
        صغّر $\sum_i w_i(y_i - \alpha - \beta x_i)^2$: بيّن
        أن الحلّ تعطيه الصيغ نفسها في السؤال
        11 بمتوسطات وتباين وتغاير موزونة
        (عرّفها).
  \item بيّن أنه من أجل المستقيم الأمثل، يفرض $E_{\min} = 0$
        أن تقع كل النقاط عليه بالضبط، واربط ذلك بحالة
        المساواة $\abs\rho = 1$ في السؤال 13؛ واختبر على
        المعطيات المستقيمية $(1,1), (2,3), (3,5)$.
  \item توليفة، في أربع جمل: لماذا تكون الدوال التربيعية
        الصنفَ الوحيد الذي لا يحتاج فيه اختبار الرتبة الثانية
        في هذا الفصل إلى مبرهنة مقبولة؛ وكيف تطابق \omterm{pb:b1:multivar:1}{المعادلات الناظمية}
        التصغيرَ التحليلي مع \omterm{thm:b1:euclid:projection}{الإسقاط
        المتعامد} في \cref{ch:b1:euclid}؛ وماذا يقيس معامل
        الارتباط وأيّ متراجحة تحدّه؛
        وأيّ قطع من الفصول 18--23 (الأسس، ومصفوفات غرام و
        العزوم، والإسقاطات) عادت للظهور. وسمِّ
        الطريقة والمعادلات المدروسة في الجزأين 2--3.
\end{enumerate}
\end{problem}
