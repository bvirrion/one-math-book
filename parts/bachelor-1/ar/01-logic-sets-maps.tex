\chapter{المنطق والمجموعات والتطبيقات}\label{ch:b1:logic}

حتى الآن جرت البراهين على فكرة غير صورية، لكنها أمينة، عمّا يعنيه
«البرهان». يجعل هذا الفصل الأول من الرياضيات الجامعية قواعد اللعبة
صريحة: ما \omterm{def:b1:logic:statement}{العبارة} الرياضية، وكيف تركّب الروابط والمسوِّرات العبارات،
وأيّ الخطوات مشروعة داخل برهان --- ثم يبني على هذا الأساس لغتَي
الرياضيات الشاملتين: المجموعات والتطبيقات.

\section{العبارات والروابط}

\begin{definition}[العبارة، الروابط]\label{def:b1:logic:statement}
\emph{العبارة}\index{عبارة} (أو \emph{القضية}) جملة إمّا صادقة (ص) وإمّا
كاذبة (ك) --- إحدى الحالتين لا غير. من العبارتين $P$ و $Q$ نكوّن:
\begin{itemize}
  \item \emph{النفي} $\lnot P$ («ليس $P$»)، وهو صادق تحديدًا عندما
        تكون $P$ كاذبة؛
  \item \emph{العطف} $P \land Q$ («$P$ و $Q$»)، وهو صادق تحديدًا
        عندما تكون العبارتان صادقتين؛
  \item \emph{الفصل} $P \lor Q$ («$P$ أو $Q$»)، وهو صادق تحديدًا
        عندما تكون إحداهما على الأقل صادقة (و«أو» هنا جامعة)؛
  \item \emph{الاستلزام} $P \implies Q$، وهو كاذب تحديدًا عندما تكون
        $P$ صادقة و $Q$ كاذبة؛
  \item \emph{التكافؤ} $P \iff Q$، وهو صادق تحديدًا عندما تحمل $P$
        و $Q$ قيمة الحقيقة نفسها.
\end{itemize}
\end{definition}

\begin{remark}
يستحق جدول حقيقة $P \implies Q$ وقفة: عندما تكون $P$ كاذبة يكون
$P \implies Q$ \emph{صادقًا} مهما تكن $Q$. فالجملة «إذا كان $2 < 1$
فإن $0 = 5$» استلزام صادق. لا يقول الاستلزام شيئًا عمّا يحدث حين
يخفق فرضه.
\end{remark}

\begin{proposition}[قواعد الحساب على العبارات]\label{prop:b1:logic:rules}
لكل عبارات $P$ و $Q$ و $R$:
\begin{enumerate}
  \item $\lnot(\lnot P) \iff P$؛
  \item قانونا دي مورغان\index{قانونا دي مورغان}:
        $\lnot(P \land Q) \iff (\lnot P) \lor (\lnot Q)$ و
        $\lnot(P \lor Q) \iff (\lnot P) \land (\lnot Q)$؛
  \item $(P \implies Q) \iff \bigl((\lnot P) \lor Q\bigr)$، ومنه
        $\lnot(P \implies Q) \iff P \land (\lnot Q)$؛
  \item \omterm{prop:b1:logic:rules}{عكس النقيض}\index{عكس النقيض}:
        $(P \implies Q) \iff \bigl((\lnot Q) \implies (\lnot P)\bigr)$؛
  \item $(P \iff Q) \iff \bigl((P \implies Q) \land (Q \implies
        P)\bigr)$؛
  \item التوزيعية: $P \land (Q \lor R) \iff (P \land Q) \lor
        (P \land R)$ و $P \lor (Q \land R) \iff (P \lor Q) \land
        (P \lor R)$.
\end{enumerate}
\end{proposition}

\begin{proof}
يُتحقَّق من كل تكافؤ بمقارنة جداول الحقيقة: عبارتان مركّبتان من $P$
و $Q$ و $R$ متكافئتان تحديدًا عندما تأخذان قيمة الحقيقة نفسها في كل
حالة من الحالات (الأربع أو الثماني). ولنعرض جدولًا واحدًا كاملًا،
لقانون دي مورغان الأول:
\begin{center}
\begin{tabular}{cc|c|c|cc|c}
$P$ & $Q$ & $P \land Q$ & $\lnot(P \land Q)$ & $\lnot P$ & $\lnot Q$ &
$(\lnot P) \lor (\lnot Q)$ \\
\hline
ص & ص & ص & ك & ك & ك & ك \\
ص & ك & ك & ص & ك & ص & ص \\
ك & ص & ك & ص & ص & ك & ص \\
ك & ك & ك & ص & ص & ص & ص \\
\end{tabular}
\end{center}
يتطابق العمودان $4$ و $7$، وهذا يبرهن على القانون. أمّا في حالة \omterm{prop:b1:logic:rules}{عكس
النقيض} فالطريق اللفظي أسرع: \omterm{def:b1:logic:statement}{العبارة} $P \implies Q$ كاذبة تحديدًا في
الحالة ($P$ صادقة، $Q$ كاذبة)، و$(\lnot Q) \implies (\lnot P)$ كاذبة
تحديدًا في الحالة ($\lnot Q$ صادقة، $\lnot P$ كاذبة)، أي ($Q$
كاذبة، $P$ صادقة) --- وهي الحالة الوحيدة نفسها، فللاستلزامين إذن
الجدول نفسه. وتُتحقَّق بقية القواعد بالطريقة نفسها؛ ولاحظ أن (3)
يردّ كل استلزام إلى فصل، فتُنتج (2) آليًا قاعدة النفي
$\lnot(P \implies Q) \iff P
\land (\lnot Q)$: فلتكذيب استلزام يجب إبراز حالة يتحقق
فيها الفرض ويخفق فيها الاستنتاج.
\end{proof}

\section{المسوِّرات}

\begin{definition}[المسوِّرات]\label{def:b1:logic:quantifiers}
لتكن $P(x)$ خاصية لعنصر $x$ من \omterm{def:b1:logic:sets}{مجموعة} $E$.
\begin{itemize}
  \item $\forall x \in E,\ P(x)$ («لكل $x$ في $E$، $P(x)$») صادقة
        عندما يحقق كل عنصر من $E$ الخاصية $P$؛
  \item $\exists x \in E,\ P(x)$ («يوجد $x$ في $E$ بحيث
        $P(x)$») صادقة عندما يحقق عنصر واحد على الأقل من $E$
        الخاصية $P$.
\end{itemize}
ونكتب $\exists!$ للدلالة على «يوجد عنصر وحيد».
\end{definition}

\begin{proposition}[نفي المسوِّرات]\label{prop:b1:logic:negquant}
\[
\lnot\bigl(\forall x \in E,\ P(x)\bigr) \iff
\exists x \in E,\ \lnot P(x),
\qquad
\lnot\bigl(\exists x \in E,\ P(x)\bigr) \iff
\forall x \in E,\ \lnot P(x).
\]
\end{proposition}

\begin{proof}
لنبرهن على التكافؤ الأول في الاتجاهين؛ والثاني مماثل. إذا كانت
$\forall x \in E,\ P(x)$ كاذبة فليس كل عنصر يحقق $P$: أي أن
\omterm{def:b1:logic:sets}{المجموعة} $A = \{x \in E : \lnot P(x)\}$ لا يمكن أن تكون خالية، وأيّ
عنصر منها يشهد على $\exists x \in
E,\ \lnot P(x)$. وبالعكس، إذا حقق عنصر $x_0 \in E$ الخاصية $\lnot
P(x_0)$ فإن $x_0$ مثال مضاد وتخفق \omterm{def:b1:logic:statement}{العبارة} الكلية. أمّا
القاعدة الثانية فتقول: «لا يوجد $x$ يحقق $P$» تعني أن \omterm{def:b1:logic:sets}{المجموعة}
$\{x : P(x)\}$ خالية، أي أن كل $x$ ينتمي إلى متمّمتها
$A$. وبتطبيق القاعدتين تباعًا على بادئة متداخلة من المسوِّرات نحصل
على الإجراء الآلي في \cref{ex:b1:logic:limit}:
يسير النفي من اليسار إلى اليمين، فيقلب كل $\forall$ إلى
$\exists$ وكل $\exists$ إلى $\forall$، ثم ينفي أخيرًا
المحمول الأعمق.
\end{proof}

\begin{example}[نفي جمل رياضية مألوفة]\label{ex:b1:logic:negpractice}
لتكن $f \colon \R \to \R$. تُكتب الجملة «$f$ متزايدة» هكذا
\[
\forall x \in \R,\ \forall y \in \R,\quad
x \leq y \implies f(x) \leq f(y) ,
\]
ونفيها، حسب \cref{prop:b1:logic:negquant} وبالقاعدة
$\lnot(P \implies Q) \iff P \land \lnot Q$:
\[
\exists x \in \R,\ \exists y \in \R,\quad
x \leq y \ \text{ و}\ f(x) > f(y) :
\]
فيكفي زوج شاهد واحد. وبالمثل تُكتب «$f$ محدودة»
$\exists M \in \R,\ \forall x \in \R,\ \abs{f(x)} \leq M$، ونفيها
\[
\forall M \in \R,\ \exists x \in \R,\quad \abs{f(x)} > M :
\]
\emph{مهما} يكن الحاصر المقترح فثمّة نقطة تتجاوزه. والفكرة
النافذة: لا يحتوي نفي صحيح قط على «ليس» مطبَّقة على كتلة
مسوَّرة --- بل هو \omterm{def:b1:logic:statement}{عبارة} إيجابية جديدة تتبادل فيها الأدوار: صار على
المرء أن ينتج الشواهد التي كان يتلقّاها.
\end{example}

\begin{example}[ترتيب المسوِّرات]\label{ex:b1:logic:order}
ترتيب المسوِّرات المختلفة مهمّ:
\[
\forall x \in \R,\ \exists y \in \R,\ y > x
\quad\text{صادقة (خذ } y = x+1\text{)،}
\]
\[
\exists y \in \R,\ \forall x \in \R,\ y > x
\quad\text{كاذبة (لا يفوق أيّ عدد حقيقي جميع الأعداد الحقيقية).}
\]
في \omterm{def:b1:logic:statement}{العبارة} الأولى يجوز أن يتعلق $y$ بالعنصر $x$؛ وفي الثانية يجب أن يفي
$y$ واحد بعينه بالغرض لكل $x$. أمّا المسوِّران المتماثلان فيتبادلان
موقعيهما دائمًا.
\end{example}

\begin{example}[قراءة تعريف بثلاثة مسوِّرات]\label{ex:b1:logic:limit}
الجملة «المتتالية $(u_n)$ تتقارب نحو $\ell$» ستُكتب في
\cref{ch:b1:seq} هكذا
\[
\forall \varepsilon > 0,\ \exists N \in \N,\ \forall n \geq N,\quad
\abs{u_n - \ell} \leq \varepsilon .
\]
ونفيها، \omterm{def:b1:logic:map}{بتطبيق} \cref{prop:b1:logic:negquant} ثلاث مرات، هو
\[
\exists \varepsilon > 0,\ \forall N \in \N,\ \exists n \geq N,\quad
\abs{u_n - \ell} > \varepsilon .
\]
والقدرة على نفي جمل كهذه آليًا، دون التفكير في معناها، مهارة
حقيقية: فهي تفصل العمل المنطقي عن العمل الرياضي.
\end{example}

\section{طرائق البرهان}

\begin{method}[أنماط البرهان المعيارية]\label{met:b1:logic:proofs}
للبرهان على\dots
\begin{enumerate}
  \item \emph{استلزام $P \implies Q$ مباشرةً}: نفترض $P$ ونستنتج
        $Q$؛
  \item \emph{\omterm{prop:b1:logic:rules}{بعكس النقيض}}: نفترض $\lnot Q$ ونستنتج $\lnot P$
        --- وهذا مشروع حسب \cref{prop:b1:logic:rules}~(4)؛
  \item \emph{بالخلف}: نفترض أن \omterm{def:b1:logic:statement}{العبارة} كاذبة ونستخرج تناقضًا؛
  \item \emph{تكافؤ}: نبرهن على الاستلزامين كلٍّ على حدة (أو نصل
        سلسلة من التكافؤات المعروفة)؛
  \item \emph{\omterm{def:b1:logic:statement}{عبارة} «لكل»}: نأخذ $x$ \emph{كيفيًا} في $E$ («ليكن
        $x \in E$») ونبرهن على $P(x)$؛
  \item \emph{\omterm{def:b1:logic:statement}{عبارة} «يوجد»}: نبرز شاهدًا، أو نبرهن على الوجود
        بطريقة غير مباشرة؛
  \item \emph{بالاستقراء}: انظر \cref{thm:b1:logic:induction}.
\end{enumerate}
وعند البرهان على \omterm{def:b1:logic:statement}{عبارة} تخصّ عنصرًا كيفيًا حسن الاختيار، لا تمنح
العنصر خصائص زائدة أبدًا: فقولك «ليكن $x \in \R$» متبوعًا بالقول «بما أن
$x > 0$\dots» لا يبرهن على شيء يخصّ قيم $x$ السالبة.
\end{method}

\begin{remark}[مزالق شائعة في البراهين]\label{rem:b1:logic:pitfalls}
أربعة أفخاخ كلاسيكية، يستحق كلٌّ منها أن يُسمّى مرة واحدة.
\begin{enumerate}
  \item \emph{العكس بدل \omterm{prop:b1:logic:rules}{عكس النقيض}.} \omterm{def:b1:logic:statement}{العبارة} $Q \implies P$
        \emph{ليست} مكافئة \omterm{def:b1:logic:statement}{للعبارة} $P \implies Q$؛ المكافئ لها هو
        $\lnot Q
        \implies \lnot P$ وحده. فجملة «إذا أمطرت السماء
        ابتلّ الشارع» لا تخوّل استنتاج المطر من شارع مبتلّ.
  \item \emph{البرهان على تكافؤ باستلزام واحد.} دعوى «إذا وفقط
        إذا» مبرهنتان؛ فصرّح بالاتجاه الذي تبرهن عليه، وبرهن على
        الاتجاهين معًا. ولا تكون سلاسل $\iff$ مشروعة إلا إذا كانت
        \emph{كل} حلقة فيها قابلة للعكس حقًّا --- وتربيع معادلة،
        مثلًا، ليس كذلك.
  \item \emph{البراهين المقلوبة.} الانطلاق من الاستنتاج المطلوب
        واستخراج \omterm{def:b1:logic:statement}{عبارة} صادقة منه لا يبرهن على شيء (فمن $-1 = 1$
        نستخرج $1 = 1$ الصادقة بالتربيع). قد \emph{يُكتشف} حساب
        ما بالمقلوب، لكن يجب أن \emph{يُكتب} بالاتجاه الصحيح، أو
        بتكافؤات صريحة.
  \item \emph{الشاهد المثبَّت في مقابل العنصر الكيفي.} للبرهان
        على $\exists x,\ P(x)$ يكفي إبراز $x$ واحد مختار بذكاء؛
        وللبرهان على $\forall x,\ P(x)$ يجب أن يبقى $x$ المختار
        كيفيًا. والخلط بين الأمرين --- أي التحقق من دعوى كلية
        على مثال --- أشيع الأخطاء في كرارير المبتدئين.
\end{enumerate}
\end{remark}

\begin{example}[عكس النقيض والخلف في العمل]\label{ex:b1:logic:sqrt2}
\emph{ليكن $n \in \N$: إذا كان $n^2$ زوجيًا فإن $n$ زوجي.} \omterm{prop:b1:logic:rules}{بعكس
النقيض}: إذا كان $n$ فرديًا، $n = 2k+1$، فإن
$n^2 = 4k^2 + 4k + 1$ فردي.

\emph{العدد $\sqrt 2$ أصمّ.} بالخلف: نفترض أن
$\sqrt 2 = p/q$ حيث $p, q \in \N^*$ والكسر في أبسط صورة.
عندئذ $p^2 = 2q^2$ زوجي، فالعدد $p$ زوجي (بالنقطة السابقة)، $p = 2r$؛
ومنه $q^2 = 2r^2$ زوجي، فالعدد $q$ زوجي --- وهذا يناقض كون الكسر في
أبسط صورة.
\end{example}

\begin{theorem}[الاستقراء]\label{thm:b1:logic:induction}
لتكن $P(n)$ خاصية للعدد الصحيح الطبيعي $n$. إذا كان
\begin{enumerate}
  \item $P(0)$ صادقة، و
  \item لكل $n \in \N$، $P(n) \implies P(n+1)$،
\end{enumerate}
فإن $P(n)$ صادقة لكل $n \in \N$.

\emph{الاستقراء القوي:} يبقى الاستنتاج نفسه إذا استُبدلت (2) بما يلي:
لكل $n$، $\bigl(P(0) \land \dots \land P(n)\bigr) \implies
P(n+1)$.
\end{theorem}

\begin{proof}
هذه خاصية \omterm{def:b1:logic:sets}{للمجموعة} $\N$ نفسها، مكافئة لما يلي: \emph{كل جزء غير خالٍ
من $\N$ له عنصر أصغر} (ونأخذه معلومًا). وبالفعل، نفترض تحقق (1)
و (2) ولتكن $A = \{n \in \N : P(n)
\text{ كاذبة}\}$. إذا كانت $A \neq \emptyset$ فلها عنصر أصغر $m$؛
و $m \neq 0$ حسب (1)؛ عندئذ $m - 1 \notin A$، فتتحقق $P(m-1)$،
وتعطي (2) تحقق $P(m)$ --- وهذا تناقض. إذن $A = \emptyset$. وأمّا
الاستقراء القوي فنطبّق عليه الحجة نفسها: تتحقق $P(0), \dots, P(m-1)$
جميعًا لأن $m$ أصغر عناصر $A$.
\end{proof}

\begin{example}[البرهان على الوجود الوحيد]\label{ex:b1:logic:uniqueexistence}
\omterm{def:b1:logic:statement}{العبارة} $\exists!\,x,\ P(x)$ عبارتان \emph{اثنتان}، يُبرهن على كلٍّ
منهما على حدة: الوجود (بإبراز أو إنشاء $x_0$ يحقق
$P(x_0)$) والوحدانية (نفترض $P(x)$ و $P(x')$ ونستنتج $x =
x'$). مثال على ذلك: \emph{يوجد عدد حقيقي وحيد $x$ يحقق $x^3 + x =
2$.} الوجود: $x_0 = 1$ يفي بالغرض لأن $1 + 1 = 2$. الوحدانية:
إذا كان $x^3 + x = x'^3 + x'$ فإن
\[
0 = (x^3 - x'^3) + (x - x')
= (x - x')\,\bigl(x^2 + xx' + x'^2 + 1\bigr),
\]
والعامل الثاني موجب (فهو يساوي $\bigl(x +
\tfrac{x'}2\bigr)^2 + \tfrac34 x'^2 + 1 \geq 1$)، ومنه $x = x'$.
ولاحظ توزيع العمل: استعمل الوجود تخمينًا موفقًا، واستعملت
الوحدانية جبرًا صالحًا لحلول \emph{كيفية} --- فلا تنوب إحدى
الحجتين عن الأخرى، ونسيان النصف الثاني إغراء دائم بمجرد العثور
على حلّ.
\end{example}

\begin{example}\label{ex:b1:logic:induction}
لكل $n \in \N^*$: $\;\sum_{k=1}^n k = \frac{n(n+1)}{2}$. حالة البداية
$n = 1$: الطرفان يساويان $1$. خطوة الانتقال: بافتراض الصيغة من أجل $n$،
\[
\sum_{k=1}^{n+1} k = \frac{n(n+1)}{2} + (n+1)
= (n+1)\Bigl(\frac n2 + 1\Bigr) = \frac{(n+1)(n+2)}{2}. \qedhere
\]
\end{example}

\begin{example}[الاستقراء القوي في العمل]\label{ex:b1:logic:strongind}
\emph{كل عدد صحيح $n \geq 2$ جداءُ أعداد أولية} (والعدد الأولي عدد
صحيح $\geq 2$ قواسمه الوحيدة التي $\geq 1$ هي $1$ وهو نفسه؛ وتُدرس
الأعداد الأولية لذاتها في \cref{ch:b1:arith}).
والاستقراء العادي عاجز هنا: فمعرفة أن $95 = 5 \times 19$ لا تقول
شيئًا عن $96$. أمّا الاستقراء القوي فيلائم المقام تمامًا. حالة
البداية: $2$ أوليّ، فهو إذن جداء أعداد أولية (بعامل واحد). خطوة
الانتقال: ليكن $n \geq 2$ ولنفترض أن كل عدد صحيح $m$ يحقق
$2 \leq m \leq n$ جداءُ أعداد أولية. إذا كان $n + 1$ أوليًا فقد
انتهينا. وإلا فإن $n + 1 = ab$ مع $2 \leq a, b \leq n$؛ وبالفرض
القوي يكون كلٌّ من $a$ و $b$ جداء أعداد أولية، فكذلك $n + 1$.
والفكرة النافذة: الاستقراء القوي هو الأداة الصحيحة كلما كان «سبب»
$P(n+1)$ قابعًا في رتبة سابقة لا يمكن التنبؤ بها، لا في الرتبة $n$.
\end{example}

\section{المجموعات}

\begin{definition}[العمليات على المجموعات]\label{def:b1:logic:sets}
نأخذ مفهوم \emph{المجموعة}\index{مجموعة} وعلاقة الانتماء
$x \in E$ مفهومين أوليين. من أجل مجموعتين $A, B$ داخل مجموعة محيطة $E$:
\begin{itemize}
  \item \emph{الاحتواء}: $A \subseteq B$ عندما $\forall x,\ x \in A
        \implies x \in B$؛ والتساوي $A = B$ عندما $A \subseteq B$ و
        $B \subseteq A$؛
  \item \emph{الاتحاد} $A \cup B$، و\emph{التقاطع} $A \cap B$،
        و\emph{الفرق} $A \setminus B = \{x \in A : x \notin B\}$،
        و\emph{المتمّمة} $\overline{A} = E \setminus A$؛
  \item \emph{المجموعة الخالية} $\emptyset$، وهي محتواة في كل
        مجموعة؛
  \item \emph{مجموعة الأجزاء}\index{مجموعة الأجزاء}
        $\mathcal{P}(E)$: مجموعة كل أجزاء $E$؛
  \item \emph{الجداء} $E \times F$: مجموعة الثنائيات المرتبة
        $(x, y)$ حيث $x \in E$ و $y \in F$.
\end{itemize}
\end{definition}

\begin{example}[التآلف مع مجموعة الأجزاء]\label{ex:b1:logic:powerset}
من أجل $E = \{a, b\}$:
\[
\mathcal P(E) = \bigl\{\, \emptyset,\ \{a\},\ \{b\},\ \{a, b\}
\,\bigr\},
\]
أربعة عناصر --- ولاحظ انضباط الأنماط: $a \in E$ لكن
$\{a\} \in \mathcal P(E)$؛ والعبارتان $a \in \mathcal P(E)$
و$\{a\} \subseteq \mathcal P(E)$ كاذبتان كما كُتبتا
(إذ تقتضي الثانية أن يكون $a$ \emph{جزءًا} من $E$).
وبالانطلاق من لا شيء: $\mathcal P(\emptyset) = \{\emptyset\}$
له عنصر واحد، و$\mathcal P(\mathcal P(\emptyset)) = \{\emptyset,
\{\emptyset\}\}$ له عنصران، والتالي له أربعة --- فمجموعات
المجموعات مجموعات عادية، وسيؤكد \cref{ch:b1:counting} نمط
التضاعف: $\abs{\mathcal P(E)} = 2^{\abs E}$. والحفاظ على تمييز
المستويات ($x$ و $\{x\}$ و $\{\{x\}\}$) نصف المعركة في تمارين
مثل \cref{exo:b1:logic:11,exo:b1:logic:12}.
\end{example}

\begin{proposition}[جبر المجموعات]\label{prop:b1:logic:setalgebra}
من أجل أجزاء $A, B, C$ من $E$:
\begin{enumerate}
  \item $A \cap (B \cup C) = (A \cap B) \cup (A \cap C)$ و
        $A \cup (B \cap C) = (A \cup B) \cap (A \cup C)$؛
  \item دي مورغان: $\overline{A \cup B} = \overline{A} \cap
        \overline{B}$ و $\overline{A \cap B} = \overline{A} \cup
        \overline{B}$؛
  \item $A \subseteq B \iff \overline{B} \subseteq \overline{A}$.
\end{enumerate}
\end{proposition}

\begin{proof}
تترجم كل متطابقة قاعدةً من قواعد \cref{prop:b1:logic:rules} عبر
القاموس ($\in A$ أو لا) $\leftrightarrow$ (صدق \omterm{def:b1:logic:statement}{العبارة} أو
كذبها): مثلًا $x \in \overline{A \cup B} \iff \lnot(x \in A \lor x \in
B) \iff (x \notin A) \land (x \notin B) \iff x \in \overline{A} \cap
\overline{B}$. والنقطة (3) هي \omterm{prop:b1:logic:rules}{عكس النقيض}. وكمثال ثانٍ، إليك قانون
التوزيعية الأول كاملًا:
\[
x \in A \cap (B \cup C)
\iff (x \in A) \land \bigl(x \in B \lor x \in C\bigr)
\iff \bigl(x \in A \land x \in B\bigr) \lor
\bigl(x \in A \land x \in C\bigr),
\]
حسب توزيعية \cref{prop:b1:logic:rules}~(6)، وتقرأ \omterm{def:b1:logic:statement}{العبارة} الأخيرة
$x \in (A \cap B) \cup (A \cap C)$. وكل متطابقة مجموعات من
هذا النوع قابلة للبرهان بهذه الترجمة الآلية الواحدة --- ولهذا لا
تحتاج أيٌّ منها إلى حفظ.
\end{proof}

\begin{method}[البرهان على تساوي مجموعتين]\label{met:b1:logic:doubleinc}
للبرهان على $A = B$، نبرهن على الاحتوائين: ليكن $x \in A$، ونبيّن
أن $x \in
B$؛ ثم ليكن $x \in B$، ونبيّن أن $x \in A$. وبدلًا من ذلك
نصل التكافؤات $x \in A \iff \dots \iff x \in B$ عندما تكون كل خطوة
تكافؤًا حقًّا.
\end{method}

\section{التطبيقات}

\begin{definition}[التطبيق، الصورة، الصورة العكسية]\label{def:b1:logic:map}
\emph{التطبيق}\index{تطبيق} (أو \emph{الدالة}) $f \colon E \to F$
يقرن بكل عنصر $x$ من \omterm{def:b1:logic:sets}{المجموعة} $E$ (\emph{\omterm{def:b1:logic:sets}{مجموعة} التعريف}) عنصرًا
واحدًا $f(x)$ من \omterm{def:b1:logic:sets}{المجموعة} $F$ (\emph{\omterm{def:b1:logic:sets}{مجموعة} الوصول}). من أجل $A
\subseteq E$ و $B \subseteq F$:
\[
f(A) = \{f(x) : x \in A\} \subseteq F,
\qquad
f^{-1}(B) = \{x \in E : f(x) \in B\} \subseteq E
\]
هما \emph{الصورة المباشرة} \omterm{def:b1:logic:sets}{للمجموعة} $A$ و\emph{الصورة العكسية}
\index{صورة عكسية} \omterm{def:b1:logic:sets}{للمجموعة} $B$. و\emph{تركيب} $f \colon E \to F$
مع $g \colon F \to G$ هو $g \circ f \colon E \to G$، $x \mapsto
g(f(x))$.
\end{definition}

\begin{remark}
الترميز $f^{-1}(B)$ \emph{لا} يفترض وجود \omterm{def:b1:logic:map}{تطبيق} عكسي: \omterm{def:b1:logic:sets}{فالمجموعة} $f^{-1}(B)$ معرَّفة من أجل كل $f$. والصور العكسية أحسن سلوكًا من
الصور المباشرة: إذ يحافظ $f^{-1}$ على الاتحادات والتقاطعات
والمتمّمات، بينما قد يكون $f(A \cap A') \subseteq f(A) \cap f(A')$ احتواءً تامًّا
(\cref{exo:b1:logic:8}).
\end{remark}

\begin{example}[حساب الصور والصور العكسية]\label{ex:b1:logic:images}
لتكن $f \colon \R \to \R$، $x \mapsto x^2$. عندئذ:
\[
f\bigl(\intcc{-1}{2}\bigr) = \intcc04, \qquad
f^{-1}\bigl(\intcc14\bigr) = \intcc{-2}{-1} \cup \intcc12, \qquad
f^{-1}(\{-1\}) = \emptyset .
\]
من أجل الأولى: كل $x \in \intcc{-1}2$ يحقق $x^2 \in \intcc04$، وكل
$y \in \intcc04$ يُبلغ على الصورة $y = (\sqrt y)^2$ حيث $\sqrt y
\in \intcc02 \subseteq \intcc{-1}2$ --- ولاحظ أن الصورة
\emph{ليست} $\intcc14 = \{(-1)^2, 2^2\}$: فصور المجالات لا
تُحسب من الطرفين وحدهما. ومن أجل الثانية: $1 \leq x^2 \leq 4 \iff
1 \leq \abs x \leq 2$، وهذا ينقسم إلى قطعتين. وتوضح الثالثة
أن \omterm{def:b1:logic:map}{الصورة العكسية} قد تكون خالية --- \omterm{def:b1:logic:sets}{فالمجموعة} $f^{-1}(B)$ ذات معنى دائمًا،
مهما صغر تقاطع $B$ مع الصورة. وأخيرًا لاحظ على هذا المثال ظاهرة
الاحتواء التام المذكورة في الملاحظة أعلاه: بأخذ $A = \intcc{-1}0$
و $A' = \intcc01$ نجد $f(A \cap A') =
f(\{0\}) = \{0\}$، بينما $f(A) \cap f(A') = \intcc01$.
\end{example}

\begin{definition}[متباين، شامل، تقابلي]\label{def:b1:logic:inj}
\omterm{def:b1:logic:map}{التطبيق} $f \colon E \to F$ يكون:
\begin{itemize}
  \item \emph{متباينًا}\index{تطبيق متباين} عندما تكون صور العناصر
        المتمايزة متمايزة: $\forall x, x' \in E,\ f(x) = f(x') \implies
        x = x'$؛
  \item \emph{شاملًا}\index{تطبيق شامل} عندما يُبلغ كل عنصر من $F$:
        $\forall y \in F,\ \exists x \in E,\ f(x) = y$؛
  \item \emph{تقابليًا}\index{تطبيق تقابلي} عندما يكون متباينًا
        وشاملًا معًا، أي عندما يكون لكل $y \in F$ سابقةٌ واحدة تمامًا.
\end{itemize}
\end{definition}

\begin{theorem}[التطبيق العكسي]\label{thm:b1:logic:inverse}
\omterm{def:b1:logic:map}{التطبيق} $f \colon E \to F$ تقابليّ إذا وفقط إذا وُجد \omterm{def:b1:logic:map}{تطبيق}
$g \colon F \to E$ يحقق $g \circ f = \mathrm{id}_E$ و $f \circ g =
\mathrm{id}_F$. وعندئذ يكون $g$ وحيدًا؛ ويُكتب $f^{-1}$ ويُسمّى
\emph{مقلوب} $f$، ويكون $f^{-1}$ نفسه \omterm{def:b1:logic:inj}{تقابليًا} ومقلوبه
$(f^{-1})^{-1} = f$.
\end{theorem}

\begin{proof}
($\Rightarrow$) إذا كان $f$ \omterm{def:b1:logic:inj}{تقابليًا} فلكل $y \in F$ سابقةٌ وحيدة؛
ونعرّف $g(y)$ بأنه تلك السابقة. عندئذ $f(g(y)) = y$ بحكم
الإنشاء، و $g(f(x)) = x$ لأن $x$ هو سابقة $f(x)$ \emph{الوحيدة}.

($\Leftarrow$) نفترض وجود $g$ كهذا. إذا كان $f(x) = f(x')$ فبتطبيق
$g$ نجد $x = x'$: أي أن $f$ متباين. ومن أجل $y \in F$، فإن $x = g(y)$
يحقق $f(x) = y$: أي أن $f$ شامل.

الوحدانية: إذا وفى كلٌّ من $g$ و $h$ بالغرض فإن $g = g \circ \mathrm{id}_F
= g \circ (f \circ h) = (g \circ f) \circ h = h$. وأخيرًا فإن زوج
المتطابقتين متناظر في $f$ و $g$، فيكون $g = f^{-1}$ \omterm{def:b1:logic:inj}{تقابليًا}
ومقلوبه $f$.
\end{proof}

\begin{example}[حساب المقلوب عمليًا]\label{ex:b1:logic:inversecomp}
لتكن $f \colon \R \to \intoo0{+\infty}$، $f(x) = \eu^{2x+1}$. لقلبه
نحلّ المعادلة $y = f(x)$ من أجل $y > 0$ معطى:
\[
y = \eu^{2x+1} \iff \ln y = 2x + 1 \iff x = \frac{\ln y - 1}2 ,
\]
وكل خطوة قابلة للعكس على المجموعتين المعلنتين. ويعطي هذا الحساب
كل شيء دفعة واحدة: فلكل $y$ من \omterm{def:b1:logic:sets}{مجموعة} الوصول حلٌّ واحد $x$ لا غير، \omterm{def:b1:logic:map}{فالتطبيق} $f$ تقابليّ، و
\[
f^{-1} \colon \intoo0{+\infty} \to \R,
\qquad
f^{-1}(y) = \frac{\ln y - 1}2 .
\]
والتحقق السريع من التركيبين ($f^{-1}(f(x)) = \frac{(2x+1) -
1}2 = x$ و $f(f^{-1}(y)) = \eu^{\ln y} = y$) يؤكد
معيار \cref{thm:b1:logic:inverse}. والفكرة النافذة: «حُلَّ من أجل
$x$ وراقب التكافؤات» هو في آن واحد برهان الوجود وبرهان الوحدانية
والصيغة --- لكنه لا يفلح إلا إذا أُعلنت \omterm{def:b1:logic:sets}{مجموعة} الوصول إعلانًا
صحيحًا (\omterm{def:b1:logic:map}{فالتطبيق} $f$ \emph{ليس} \omterm{def:b1:logic:inj}{شاملًا} على $\R$).
\end{example}

\begin{proposition}[التركيب والخصائص الثلاث]\label{prop:b1:logic:comp}
لتكن $f \colon E \to F$ و $g \colon F \to G$.
\begin{enumerate}
  \item إذا كان $f$ و $g$ متباينين (أو شاملين، أو تقابليين على
        التوالي) فكذلك $g \circ f$؛ وعندئذ $(g \circ f)^{-1} = f^{-1} \circ
        g^{-1}$ في الحالة التقابلية.
  \item إذا كان $g \circ f$ \omterm{def:b1:logic:inj}{متباينًا} فإن $f$ متباين. وإذا كان
        $g \circ f$ \omterm{def:b1:logic:inj}{شاملًا} فإن $g$ شامل.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) إذا كان $g(f(x)) = g(f(x'))$ فتباين $g$ يعطي $f(x) = f(x')$،
ثم تباين $f$ يعطي $x = x'$. وإذا كان $z \in G$ فشمول $g$ يعطي $y$
يحقق $g(y) = z$، ثم شمول $f$ يعطي $x$ يحقق $f(x) = y$، ومنه
$g(f(x)) = z$. وفي الحالة التقابلية نتحقق مباشرة من أن
$f^{-1} \circ g^{-1}$ مقلوب ثنائي الجانب للتركيب $g \circ f$، وتُتمّ
الوحدانية في \cref{thm:b1:logic:inverse} البرهان.

(2) إذا كان $f(x) = f(x')$ فإن $g(f(x)) = g(f(x'))$، ويعطي تباين
$g \circ f$ أن $x = x'$. وإذا كان $z \in G$ فشمول $g \circ f$
يعطي $x$ يحقق $g(f(x)) = z$: عندئذ $y = f(x)$ يحقق $g(y) = z$.
\end{proof}

\begin{example}[النقطة (2) دقيقة]\label{ex:b1:logic:compsharp}
في \cref{prop:b1:logic:comp}~(2) لا يمكن تقوية
الاستنتاجين: فكون $g \circ f$ \omterm{def:b1:logic:inj}{تقابليًا} \emph{لا} يفرض شمول $f$
ولا تباين $g$. خذ $E = G = \{1\}$، و$F = \{1,
2\}$، مع $f(1) = 1$ و $g(1) = g(2) = 1$: عندئذ $g \circ f =
\mathrm{id}_E$ تقابليّ، ومع ذلك يفوت $f$ العنصر $2$
ويطوي $g$ العنصرين في واحد. والعبرة قاعدةُ محاسبة دقيقة:
تنتقل معلومة التركيب إلى \omterm{def:b1:logic:map}{التطبيق} \emph{الداخلي} في مسألة
التباين، وإلى \omterm{def:b1:logic:map}{التطبيق} \emph{الخارجي} في مسألة الشمول، ولا
تنتقل قط في الاتجاه المعاكس. (ويبني \cref{exo:b1:logic:9} الظاهرة
نفسها على مجموعات لا نهائية، حيث تكون المحرك وراء المقلوبات
أحادية الجانب.)
\end{example}

\begin{example}\label{ex:b1:logic:injexamples}
\omterm{def:b1:logic:map}{التطبيق} $f \colon \R \to \R$، $x \mapsto x^2$ ليس \omterm{def:b1:logic:inj}{متباينًا} ($f(-1) =
f(1)$) ولا \omterm{def:b1:logic:inj}{شاملًا} (فالعدد $-1$ ليست له سابقة). وبتقييد \omterm{def:b1:logic:sets}{مجموعة} التعريف
\omterm{def:b1:logic:sets}{ومجموعة} الوصول يصير $f \colon \R_+ \to \R_+$، $x \mapsto x^2$ \omterm{def:b1:logic:inj}{تقابليًا}،
ومقلوبه $y \mapsto \sqrt y$. فتباين \omterm{def:b1:logic:map}{تطبيق} أو شموله يتعلق بمجموعتي
التعريف والوصول المعلنتين، لا بالصيغة وحدها.
\end{example}

\section{العلاقات}

\begin{definition}[علاقة التكافؤ]\label{def:b1:logic:equiv}
\emph{العلاقة الثنائية} $\mathcal{R}$ على \omterm{def:b1:logic:sets}{مجموعة} $E$ تكون
\emph{علاقة تكافؤ}\index{علاقة تكافؤ} عندما تكون:
\emph{انعكاسية} ($x \mathbin{\mathcal{R}} x$ لكل $x$)،
و\emph{تماثلية} ($x \mathbin{\mathcal{R}} y \implies y
\mathbin{\mathcal{R}} x$) و\emph{متعدّية} ($x
\mathbin{\mathcal{R}} y$ و $y \mathbin{\mathcal{R}} z$ يستلزمان $x
\mathbin{\mathcal{R}} z$). و\emph{صف التكافؤ} للعنصر $x$ هو
$\mathrm{cl}(x) = \{y \in E : x \mathbin{\mathcal{R}} y\}$.
\end{definition}

\begin{example}[التحقق من البديهيات الثلاث]\label{ex:b1:logic:fracpart}
على $\R$، نضع $x \mathbin{\mathcal{R}} y$ عندما يكون $x - y \in \Z$.
\emph{انعكاسية:} $x - x = 0 \in \Z$. \emph{تماثلية:} إذا كان $x - y
\in \Z$ فإن $y - x = -(x - y) \in \Z$. \emph{متعدّية:} إذا كان $x -
y \in \Z$ و $y - z \in \Z$ فإن $x - z = (x - y) + (y - z) \in
\Z$ (مجموع عددين صحيحين). إذن $\mathcal R$ \omterm{def:b1:logic:equiv}{علاقة
تكافؤ}، و$\mathrm{cl}(x) = x + \Z = \{x + k : k \in \Z\}$:
ويحتوي كل صف على ممثّل واحد بالضبط في $\intco01$، هو
\emph{جزؤه الكسري}. وعلى النقيض من ذلك فالعلاقة «$\abs{x - y}
\leq 1$» على $\R$ انعكاسية وتماثلية لكنها \emph{غير}
متعدّية ($0 \mathbin{\mathcal R} 1$ و $1 \mathbin{\mathcal R}
2$، ومع ذلك $\abs{0 - 2} > 1$): فالقرب لا ينتشر، ولا وجود
لأيّ تجزئة إلى صفوف --- وهو مثال مضاد يحسن استحضاره حين يبدو
التحقق من البديهيات أمرًا روتينيًا.
\end{example}

\begin{theorem}[الصفوف تكوّن تجزئة]\label{thm:b1:logic:partition}
لتكن $\mathcal{R}$ \omterm{def:b1:logic:equiv}{علاقة تكافؤ} على $E$. عندئذ تكون صفوف التكافؤ
غير خالية، ومتقاطعتين عند التساوي أو منفصلتين مثنى مثنى، واتحادها
هو $E$: أي أنها تكوّن \emph{تجزئة}\index{تجزئة} \omterm{def:b1:logic:sets}{للمجموعة} $E$.
وبالعكس، كل تجزئة \omterm{def:b1:logic:sets}{للمجموعة} $E$ تنشأ على هذا النحو من \omterm{def:b1:logic:equiv}{علاقة تكافؤ} واحدة
لا غير (هي «الانتماء إلى القطعة نفسها»).
\end{theorem}

\begin{proof}
لدينا $x \in \mathrm{cl}(x)$ بالانعكاسية، فالصفوف غير خالية
واتحادها $E$. ولنفترض أن $\mathrm{cl}(x) \cap \mathrm{cl}(y) \neq \emptyset$،
وليكن $z$ في كليهما. عندئذ $x \mathbin{\mathcal{R}} z$ و $y
\mathbin{\mathcal{R}} z$، فبالتماثل والتعدّي نجد $x
\mathbin{\mathcal{R}} y$. والآن من أجل $t \in \mathrm{cl}(y)$ كيفيّ
يعطي التعدّي أن $t \in \mathrm{cl}(x)$، وبالتناظر كذلك: فالصفّان
متساويان. وأمّا العكس فلتكن $(E_i)_{i \in I}$ تجزئةً \omterm{def:b1:logic:sets}{للمجموعة} $E$ ولنعرّف $x \mathbin{\mathcal S} y$ بأنها تعني
«توجد قطعة تحتوي $x$ و $y$ معًا». \emph{انعكاسية:} ينتمي $x$
إلى قطعة ما، وهي تحتوي $x$ عندئذ مرتين. \emph{تماثلية:} الشرط
المعرِّف متناظر في $x$ و $y$. \emph{متعدّية:} إذا كان
$x, y \in E_i$ و $y, z \in E_j$ فإن $y \in E_i \cap E_j$، ومنه
$E_i = E_j$ (فالقطع المتمايزة منفصلة) ويشترك $x, z$ في قطعة.
وصفّ $x$ بالنسبة إلى $\mathcal S$ هو بالضبط القطعة التي تحتوي
$x$، فالصفوف هي القطع المعطاة. وأخيرًا فالعلاقة محدَّدة بصفوفها:
إذ إن علاقتَي تكافؤ لهما الصفوف نفسها تربطان الأزواج نفسها، لأن
كلًّا منهما تربط بين $x$ و $y$ تحديدًا عندما ينتمي $y$ إلى صف $x$
--- ومن هنا دعوى الوحدانية.
\end{proof}

\begin{example}\label{ex:b1:logic:congruence}
على $\Z$، التوافق بترديد $n$ (أي $x \equiv y \pmod n$ عندما يقسم $n$
الفرق $x - y$) علاقةُ تكافؤ؛ وصفوفها هي المجموعات $n$ المكوَّنة من
الأعداد الصحيحة ذات باقٍ معطى عند القسمة على $n$. ويصير هذا المثال
الحلقة $\Z/n\Z$ في \cref{ch:b1:structures}.
\end{example}

\begin{definition}[علاقة الترتيب]\label{def:b1:logic:order}
العلاقة $\preceq$ على $E$ تكون \emph{ترتيبًا}\index{علاقة ترتيب}
عندما تكون انعكاسية و\emph{ضدّ تماثلية} ($x \preceq y$ و$y
\preceq x$ يستلزمان $x = y$) ومتعدّية. ويكون الترتيب \emph{كليًا}
عندما يكون أيّ عنصرين قابلين للمقارنة، و\emph{جزئيًا} فيما عدا ذلك.
والعنصر $M \in A \subseteq E$ يكون \emph{عنصرًا أكبر} في $A$ عندما
يكون $a \preceq M$ لكل $a \in A$؛ والعناصر الكبرى (والصغرى) وحيدة
عند وجودها.
\end{definition}

\begin{example}\label{ex:b1:logic:orders}
\omterm{def:b1:logic:sets}{المجموعة} $(\R, \leq)$ مرتّبة \omterm{def:b1:logic:order}{ترتيبًا} كليًا. أمّا $(\mathcal{P}(E), \subseteq)$ فهي
مرتّبة \omterm{def:b1:logic:order}{ترتيبًا} جزئيًا بمجرد أن يكون \omterm{def:b1:logic:sets}{للمجموعة} $E$ عنصران: إذ إن $\{a\}$
و $\{b\}$ غير قابلين للمقارنة. والجزء $A = \{\{a\}, \{b\}\}$ من
$\mathcal{P}(\{a,b\})$ ليس له عنصر أكبر، ومع ذلك له حاصر أعلى
$\{a, b\}$: والتمييز بين العناصر الكبرى والحواصر العليا يعود، من
أجل $\R$، في \cref{ch:b1:reals}.
\end{example}

\begin{example}[ترتيبان على الشبكة $\N^2$]\label{ex:b1:logic:lexorder}
على ثنائيات الأعداد الطبيعية نقارن مركّبةً مركّبةً: $(a, b) \preceq (a',
b')$ عندما يكون $a \leq a'$ \emph{و} $b \leq b'$ (\emph{ترتيب
الجداء}). وهذا ترتيب --- إذ تُورَث كل بديهية إحداثيةً إحداثيةً
--- لكنه جزئيّ: فالثنائيتان $(1, 3)$ و $(2, 0)$ غير قابلين للمقارنة.
والآن لنقارن كما يقارن المعجم: $(a, b)
\preceq_{\mathrm{lex}} (a', b')$ عندما يكون $a < a'$، أو $a = a'$ و$b
\leq b'$ (\emph{الترتيب المعجمي}). ويقتضي التعدّي تحققًا من
حالتين لكنه يصمد، وصار أيّ ثنائيتين قابلتين للمقارنة: فالترتيب
كليّ. والترتيبان يرتّبان \omterm{def:b1:logic:sets}{المجموعة} نفسها ترتيبين مختلفين --- فلدينا $(0, 100) \preceq_{\mathrm{lex}} (1, 0)$ في حين لا يقول ترتيب
الجداء شيئًا --- وهو تذكير بأن الترتيب بنيةٌ \emph{نختارها}، لا
خاصيةٌ \omterm{def:b1:logic:sets}{للمجموعة}. والمقارنة المعجمية هي كذلك الحيلة المعيارية
لردّ عدة معايير للترتيب إلى معيار واحد.
\end{example}

\begin{remark}[استراحة: الحجم بوصفه تقابلًا]\label{rem:b1:logic:interlude}
ثمة موضوع صامت في هذا الفصل يستحق أن يُسلَّط عليه الضوء:
التقابلات هي مفهوم الرياضياتي عن «الحجم نفسه». وفي حالة
المجموعات المنتهية يصير هذا حسابَ العدّ في \cref{ch:b1:counting}،
حيث تخفي كل صيغة تقابلًا؛ وفي حالة المجموعات اللانهائية يصير
مسألة نهاية الأسبوع أدناه، حيث يتبيّن أن $\N$ و $\Q$ و $\R$ ذات
أحجام مختلفة حقًّا. ويعود القاموس نفسه مرتين أخريين في هذا
المجلد في صور مصقولة: فالمتتاليات (\cref{ch:b1:seq}) ليست
إلا تطبيقات $\N \to \R$، فالعبارات عن المتتاليات عبارات عن
\omterm{def:b1:logic:sets}{مجموعة} تطبيقات؛ وسيقيس الجبر الخطي الفضاءات المتجهية لا
بالتقابلات بل بالتقابلات \emph{الخطية}، التي يحكم وجودَها عددٌ
واحد هو البُعد (\cref{ch:b1:findim}). وكلما ظهر ضربٌ جديد
من «التماثل» --- تساوي القوة، وتماثل الزمر
(\cref{ch:b1:structures})، والتماثل الخطي --- تكرّر نمط
\cref{thm:b1:logic:inverse}: التماثل \omterm{def:b1:logic:map}{تطبيق} قابل للقلب يحترم البنية.
\end{remark}

\begin{remark}[أين يُستعمل هذا الفصل]\label{rem:b1:logic:whereused}
في كل مكان --- لكنّ مواضع قليلة تستحق التنويه. فرياضة المسوِّرات
الثلاثة في \cref{ex:b1:logic:limit} هي الخبز اليومي في
\cref{ch:b1:seq,ch:b1:continuity}: إذ إن كل برهان على نهاية لعبةٌ تُلعب
ضد $\varepsilon$ كيفيّ. وتعود صفوف التكافؤ في صورة صفوف التوافق
في $\Z/n\Z$ في \cref{ch:b1:structures}، حيث تكتسب تجزئة
\cref{thm:b1:logic:partition} بنيةً جبرية خاصة بها. وتصير علاقات
الترتيب والحواصر العليا والحدود العليا القلبَ البديهي \omterm{def:b1:logic:sets}{للمجموعة} $\R$ في
\cref{ch:b1:reals}.
وتعود التطبيقات المتباينة والشاملة والتقابلية في صورة التطبيقات
الخطية في \cref{ch:b1:linmaps}، حيث يمكن اختبار التباين على
متجهة واحدة (النواة)؛ وتحوّل مسألة نهاية الأسبوع أدناه مفهومَ
التقابل المجرّد إلى نظرية في \emph{أحجام المجموعات اللانهائية}،
تعود استنتاجاتها (قابلية $\Q$ للعدّ، وعدم قابلية $\R$ للعدّ) إلى
الظهور في \cref{ch:b1:reals,ch:b1:topology}.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:logic:1}
اكتب نفي كل \omterm{def:b1:logic:statement}{عبارة}، دون استعمال كلمة «ليس»:
\begin{enumerate}
  \item $\forall x \in \R,\ \exists y \in \R,\ x + y > 0$؛
  \item $\exists x \in \R,\ \forall y \in \R,\ xy = 0$؛
  \item $\forall \varepsilon > 0,\ \exists \delta > 0,\ \forall x \in
        \R,\ \abs{x} \leq \delta \implies \abs{f(x)} \leq \varepsilon$
        (من أجل \omterm{def:b1:logic:map}{تطبيق} $f \colon \R \to \R$ مثبَّت).
\end{enumerate}
ثم حدّد أصادقةٌ العبارتان (1) و (2) أم كاذبتان.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:logic:2}
لتكن $P, Q$ عبارتين. برهن، باستعمال جداول الحقيقة، على أن
$\lnot(P \implies Q) \iff P \land (\lnot Q)$، واستنتج نفي \omterm{def:b1:logic:statement}{العبارة}:
«إذا كانت دالة قابلة للاشتقاق فهي متصلة».
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:logic:3}
برهن \omterm{prop:b1:logic:rules}{بعكس النقيض}: من أجل $x \in \R$، إذا كان $x^3 + x \geq 2$ فإن
$x \geq 1$. ثم برهن بالخلف على أنه لا يوجد أصغر عدد حقيقي موجب
تمامًا.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:logic:4}
برهن بالاستقراء على أنه، لكل $n \in \N$:
\begin{enumerate}
  \item $\sum_{k=0}^{n} 2^k = 2^{n+1} - 1$؛
  \item العدد $4^n + 5$ يقبل القسمة على $3$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:logic:5}
جد الخلل في «البرهان» التالي على أن جميع الأقلام لها اللون نفسه.
\emph{لتكن $P(n)$: «في كل \omterm{def:b1:logic:sets}{مجموعة} مكوَّنة من $n$ قلمًا، تحمل جميع
الأقلام اللون نفسه». \omterm{def:b1:logic:statement}{العبارة} $P(1)$ بيّنة. نفترض $P(n)$ ونأخذ
$n+1$ قلمًا؛ بحذف الأخير تشترك الأقلام $n$ الأولى في لونها؛ وبحذف
الأول تشترك الأقلام $n$ الأخيرة في لونها؛ ومنه تشترك الأقلام
$n+1$ جميعها في لونها.}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:logic:6}
لتكن $A, B, C$ أجزاءً من $E$. برهن على أن:
\begin{enumerate}
  \item $A \setminus B = A \cap \overline{B}$؛
  \item $(A \cup B) \setminus C = (A \setminus C) \cup (B \setminus
        C)$؛
  \item $A \subseteq B \iff A \cup B = B \iff A \cap B = A$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:logic:7}
من أجل كل \omterm{def:b1:logic:map}{تطبيق} ممّا يلي، حدّد (مع البرهان) أهو متباين أم شامل أم
تقابليّ:
\begin{enumerate}
  \item $f \colon \N \to \N$، $n \mapsto n + 1$؛
  \item $g \colon \Z \to \Z$، $n \mapsto n + 1$؛
  \item $h \colon \R \setminus \{1\} \to \R$، $x \mapsto
        \frac{x+1}{x-1}$.
\end{enumerate}
ومن أجل $h$، عدّل \omterm{def:b1:logic:sets}{مجموعة} الوصول لجعله \omterm{def:b1:logic:inj}{تقابليًا} واحسب مقلوبه.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:logic:8}
لتكن $f \colon E \to F$، ولتكن $A, A' \subseteq E$ و$B, B' \subseteq
F$.
\begin{enumerate}
  \item برهن على أن $f^{-1}(B \cap B') = f^{-1}(B) \cap f^{-1}(B')$ وأن
        $f(A \cup A') = f(A) \cup f(A')$.
  \item برهن على أن $f(A \cap A') \subseteq f(A) \cap f(A')$ وأعط
        مثالًا يكون فيه الاحتواء تامًّا.
  \item برهن على أن: $f$ متباين إذا وفقط إذا كان $f(A \cap A') = f(A)
        \cap f(A')$ لكل $A, A'$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:logic:9}
لتكن $f \colon E \to F$ و $g \colon F \to E$ يحققان $g \circ f =
\mathrm{id}_E$. برهن على أن $f$ متباين وأن $g$ شامل.
وأعط مثالًا لا يكون فيه $f$ ولا $g$ \omterm{def:b1:logic:inj}{تقابليًا}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:logic:10}
على $\R$، نعرّف $x \mathbin{\mathcal{R}} y \iff x^2 - y^2 = x - y$.
برهن على أن $\mathcal{R}$ \omterm{def:b1:logic:equiv}{علاقة تكافؤ}، وصِف \omterm{def:b1:logic:equiv}{صف التكافؤ} لكل عدد
حقيقي $x$. وأيّ الصفوف يحتوي على عنصر واحد بالضبط؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:logic:11}
(كانتور) لتكن $E$ \omterm{def:b1:logic:sets}{مجموعة}. برهن على أنه لا يوجد \omterm{def:b1:logic:inj}{تطبيق شامل} من $E$
على $\mathcal{P}(E)$. \emph{إرشاد: من أجل $f \colon E \to
\mathcal{P}(E)$ معطى، تأمّل $D = \{x \in E : x \notin f(x)\}$.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:logic:12}
ليكن $f \colon E \to F$ تطبيقًا. نعرّف $\Phi \colon \mathcal{P}(F)
\to \mathcal{P}(E)$ بالعلاقة $\Phi(B) = f^{-1}(B)$.
\begin{enumerate}
  \item برهن على أن $f$ شامل إذا وفقط إذا كان $\Phi$ \omterm{def:b1:logic:inj}{متباينًا}.
  \item برهن على أن $f$ متباين إذا وفقط إذا كان $\Phi$ \omterm{def:b1:logic:inj}{شاملًا}.
\end{enumerate}
\end{exercise}

\section{مسألة: مقارنة اللانهايات}

\begin{problem}\label{pb:b1:logic:1}
متى تكون لمجموعتين «العدد نفسه من العناصر»؟ جواب كانتور --- عندما
يوجد تقابل بينهما --- يتبيّن أنه صالح للاستعمال حتى مع المجموعات
اللانهائية، وهو يشقّ اللانهاية إلى أحجام مختلفة حقًّا. وتبني هذه
المسألة صندوق العدّة كاملًا انطلاقًا من تعاريف هذا الفصل المجردة:
مبرهنة كانتور--شرودر--برنشتاين (حَقنان يصنعان تقابلًا)، وقابلية
$\Q$ للعدّ، وعدم قابلية $\R$ للعدّ بالحجة القطرية، واستنتاج كانتور
المذهل سنة 1874: \emph{الأعداد المتسامية موجودة، وبكثرة هائلة}،
دون إبراز عدد واحد منها.
\index{مجموعة قابلة للعدّ}\index{مبرهنة كانتور--شرودر--برنشتاين}
وفي كل ما يلي، من أجل مجموعتين $E$ و $F$، نكتب $E \preceq F$ عندما
يوجد \omterm{def:b1:logic:inj}{تطبيق متباين} من $E$ في $F$، ونكتب $E \approx F$ («المجموعتان
$E$ و $F$ \emph{متساويتا القوة}») عندما يوجد تقابل من $E$ على $F$.

\medskip
\noindent\textbf{الجزء 1 --- مفردات المقارنة.}
\begin{enumerate}
  \item بيّن أن $\approx$ تسلك سلوك \omterm{def:b1:logic:equiv}{علاقة تكافؤ}:
        $E \approx E$؛ وإذا كان $E \approx F$ فإن $F \approx E$؛
        وإذا كان $E \approx F$ و $F \approx G$ فإن $E \approx G$.
        (اذكر بدقة \cref{thm:b1:logic:inverse} و
        \cref{prop:b1:logic:comp}.)
  \item بيّن أن $\preceq$ متعدّية، وأن كل \omterm{def:b1:logic:inj}{تطبيق متباين}
        $f \colon E \to F$ يستدعي $E \approx f(E)$.
  \item لتكن $E \neq \emptyset$. بيّن أن $E \preceq F$ إذا وفقط
        إذا وُجد \omterm{def:b1:logic:inj}{تطبيق شامل} من $F$ على $E$.
  \item تحقق من أن $n \mapsto n + 1$ تقابل من $\N$ على
        $\N^* = \N \setminus \{0\}$، وأن
        \[
        \sigma(n) = \frac n2 \ \ (n \text{ زوجي}), \qquad
        \sigma(n) = -\frac{n+1}2 \ \ (n \text{ فردي})
        \]
        تقابل من $\N$ على $\Z$. ومنه فحذف نقطة، أو المضاعفة نحو
        الأعداد السالبة، لا يغيّر حجم $\N$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- مبرهنة
كانتور--شرودر--برنشتاين.} ليكن $f \colon E \to F$ و $g \colon F \to E$
تطبيقين متباينين. نعرّف
\[
C_0 = E \setminus g(F), \qquad C_{n+1} = g\bigl(f(C_n)\bigr)
\ \ (n \in \N), \qquad C = \bigcup_{n \in \N} C_n,
\]
ونجعل $h \colon E \to F$ يرسل $x \in C$ إلى $f(x)$، ويرسل
$x \notin C$ إلى العنصر الوحيد $y \in F$ الذي يحقق $g(y) = x$.
\begin{enumerate}[resume]
  \item تحقق من أن $h$ معرَّف تعريفًا سليمًا: إذا كان $x \notin C$
        فإن $x \in
        g(F)$، والعنصر $y$ الذي يحقق $g(y) = x$ وحيد.
  \item بيّن أن $g\bigl(f(C)\bigr) = \bigcup_{n \geq 1} C_n
        \subseteq C$. (الصور المباشرة تتبادل مع الاتحادات:
        \cref{exo:b1:logic:8}.)
  \item بيّن أن $h$ متباين. (ثلاث حالات؛ وفي الحالة المختلطة
        $x \in C$ و $x' \notin C$، بيّن أن $h(x) = h(x')$ يفرض
        $x' \in g(f(C)) \subseteq C$.)
  \item بيّن أن $h$ شامل: من أجل $y \in F$ معطى، ميّز بين
        الحالتين $g(y) \notin C$ و $g(y) \in C_n$ من أجل $n
        \geq 1$ ما (ولماذا يستحيل أن يكون $g(y) \in C_0$؟)، وأبرز
        سابقةً للعنصر $y$ في كل حالة.
  \item استنتج \emph{مبرهنة كانتور--شرودر--برنشتاين}: إذا كان
        $E \preceq F$ و $F \preceq E$ فإن $E
        \approx F$. وعلّق في جملة واحدة على ما يجعل هذه \omterm{def:b1:logic:statement}{العبارة}
        غير بديهية.
  \item تطبيقان. (أ) بيّن أن $\intcc01 \approx \intoo01$.
        (ب) بيّن أن $\varphi(p, q) = 2^p(2q + 1) - 1$ يعرّف
        تقابلًا من $\N \times \N$ على $\N$ --- التباين بحجة
        الزوجية، والشمول بالاستقراء القوي
        (\cref{thm:b1:logic:induction}). ومنه $\N \times \N \approx
        \N$: فمستوي النقاط الصحيحة ليس أكبر من
        المستقيم.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- المجموعات القابلة للعدّ.} نقول عن
\omterm{def:b1:logic:sets}{مجموعة} $E$ إنها \emph{قابلة للعدّ على الأكثر} عندما يكون
$E \preceq \N$، وإنها \emph{قابلة للعدّ} عندما يكون $E \approx \N$.
\begin{enumerate}[resume]
  \item بيّن أن كل جزء لا نهائي $A \subseteq \N$ قابل للعدّ.
        (عرّف $\varphi(n)$ تراجعيًا بأنه أصغر عنصر في
        $A \setminus \{\varphi(0), \dots,
        \varphi(n-1)\}$؛ وبيّن أن $\varphi$ متزايد تمامًا، وأنه
        يحقق $\varphi(n) \geq n$، وأنه يبلغ كل عنصر من $A$.)
  \item استنتج أن \omterm{def:b1:logic:sets}{مجموعة} تكون قابلة للعدّ على الأكثر إذا وفقط إذا
        كانت منتهية أو قابلة للعدّ، ولاحظ أن السؤال 9 يعطي
        الاختصار التالي: إذا كان $E \preceq \N$ و $\N \preceq E$
        فإن $E$ قابلة للعدّ.
  \item بيّن أنه إذا كانت $E$ و $F$ قابلتين للعدّ على الأكثر
        فكذلك $E
        \times F$. واستنتج أن $\Z \times \N^*$ قابلة للعدّ.
  \item بيّن أن $\Q$ قابلة للعدّ. (احقن $\Q$ في $\Z \times
        \N^*$ بكتابة كل عدد ناطق في أبسط صورة بمقام موجب --- ووحدانية
        هذا التمثيل مبرهن عليها في \cref{ch:b1:arith}؛ ثم طبّق السؤال 12.)
  \item بيّن أن اتحادًا قابلًا للعدّ من مجموعات قابلة للعدّ على
        الأكثر هو قابل للعدّ على الأكثر: أي إذا كانت كل $E_n$
        ($n \in \N$) قابلة للعدّ على الأكثر فكذلك $\bigcup_{n \in \N} E_n$. (أرسل $x$ إلى
        الثنائية $(n, f_n(x))$ حيث $n$ هو \emph{أصغر} دليل يحقق
        $x \in E_n$.)
  \item بيّن أن \omterm{def:b1:logic:sets}{مجموعة} أجزاء $\N$ \emph{المنتهية} قابلة للعدّ.
        (أرسل جزءًا منتهيًا $F$ إلى $\sum_{i \in F} 2^i$؛ وبرهن
        على التباين بمقارنة أكبر عنصر تختلف عنده مجموعتان
        منتهيتان، مستعملًا $\sum_{k=0}^{m-1} 2^k = 2^m - 1$
        من \cref{exo:b1:logic:4}.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- القطرنة.} لتكن $\{0,1\}^{\N}$
\omterm{def:b1:logic:sets}{مجموعة} كل التطبيقات $u \colon \N \to \{0, 1\}$، أي \omterm{def:b1:logic:sets}{مجموعة}
المتتاليات الثنائية.
\begin{enumerate}[resume]
  \item أنشئ تقابلًا بين $\mathcal{P}(\N)$ و
        $\{0,1\}^{\N}$ (الدوال المميِّزة).
  \item (الحجة القطرية) ليكن $\Phi \colon \N \to
        \{0,1\}^{\N}$ تطبيقًا كيفيًا. تأمّل المتتالية $d$ المعرَّفة بالعلاقة $d(n) = 1 - \Phi(n)(n)$. بيّن أن $d$ ليست في
        صورة $\Phi$، واستنتج أن $\{0,1\}^{\N}$ \emph{غير} قابلة
        للعدّ على الأكثر. واشرح في جملة واحدة لماذا يكون هذا،
        عبر السؤال 17، هو بالضبط مبرهنة كانتور
        (\cref{exo:b1:logic:11}) من أجل $E = \N$.
  \item اقبل --- بوصفه مألوفًا من المدرسة، ومبرهنًا عليه بدقة في
        \cref{ch:b1:reals} --- أن لكل $x \in
        \intco01$ نشرًا عشريًا \emph{سويًا} وحيدًا $x =
        0.d_1 d_2 d_3\dots$ (أي لا ينتهي بسلسلة لا نهائية من
        الأرقام $9$). ومن أجل متتالية كيفية $(x_n)_{n \geq 1}$ من
        عناصر $\intco01$، أنشئ $x \in \intco01$ يحقق $x \neq x_n$
        لكل $n$: اختر رقمه من الرتبة $n$ مساويًا $5$ إذا كان رقم
        $x_n$ من الرتبة $n$ مخالفًا $5$، و $6$ فيما عدا ذلك. وبرّر
        بعناية أن $x$ سويّ وأنه يتفادى كل $x_n$، واستنتج أن
        $\intco01$ غير قابلة للعدّ على الأكثر.
  \item استنتج أن $\R$ غير قابلة للعدّ، وأن \omterm{def:b1:logic:sets}{مجموعة} الأعداد الصمّاء
        $\R
        \setminus \Q$ غير قابلة للعدّ كذلك.
        وبأيّ معنى دقيق تكون «أغلب» الأعداد الحقيقية صمّاء؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- مبرهنة كانتور لسنة 1874: وجود الأعداد
المتسامية.} نقول عن عدد حقيقي $x$ إنه \emph{جبريّ}
\index{عدد جبري} عندما يكون $P(x) = 0$ من أجل كثير حدود غير معدوم
$P$ ذي معاملات صحيحة، وإنه \emph{متسامٍ}
\index{عدد متسام} فيما عدا ذلك. واقبل في هذا الجزء --- فهو مبرهن
عليه في \cref{ch:b1:poly} --- أن كثير حدود غير معدوم من
الدرجة $n$ له $n$ جذور حقيقية على الأكثر.
\begin{enumerate}[resume]
  \item بيّن أن كل عدد ناطق جبريّ، وجد كثيرات حدود صريحة ذات
        معاملات صحيحة تُعدم $\sqrt 2$ و $\sqrt 2 + \sqrt 3$.
  \item من أجل $n \in \N$ مثبَّت، بيّن أن \omterm{def:b1:logic:sets}{مجموعة} كثيرات الحدود ذات
        الدرجة $n$ على الأكثر والمعاملات الصحيحة قابلة للعدّ.
        (احقنها في $\Z^{n+1}$ واستقرِ على $n$ بالسؤال 13.)
  \item استنتج أن \omterm{def:b1:logic:sets}{مجموعة} \emph{كل} كثيرات الحدود ذات المعاملات
        الصحيحة قابلة للعدّ.
  \item برهن على \emph{مبرهنة كانتور في الأعداد الجبرية}: \omterm{def:b1:logic:sets}{مجموعة}
        الأعداد الحقيقية الجبرية $\mathcal{A}$ قابلة للعدّ.
  \item استنتج: توجد أعداد حقيقية متسامية، \omterm{def:b1:logic:sets}{ومجموعة} الأعداد
        المتسامية غير قابلة للعدّ. ثم لخّص المسألة كلها في بضع
        جمل: السلسلة $\N \approx
        \Z \approx \Q \approx \mathcal{A}$، والقفزة التامة إلى $\R
        \approx$ (أساسًا) $\mathcal{P}(\N)$، وأين كانت كل أداة
        (كانتور--شرودر--برنشتاين، والاتحادات القابلة للعدّ،
        والقطرنة) حاسمة --- ثم الأثر الفلسفي للبرهان على وجود
        عدد غير قابل للعدّ من الأعداد المتسامية دون تسمية واحد
        منها. (أمّا البرهان على تسامي عدد \emph{بعينه} مثل
        $\pi$ فأصعب بكثير ويتجاوز هذا المجلد.)
\end{enumerate}
\end{problem}
