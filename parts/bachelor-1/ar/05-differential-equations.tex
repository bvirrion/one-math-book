\chapter{المعادلات التفاضلية الخطية}\label{ch:b1:diffeq}

قُوبلت المعادلات التفاضلية أول مرة في مجلد الثانوية، وتُعالَج
هنا ببراهين كاملة وبعمومية أكبر: المعادلات الخطية من الرتبة الأولى
ذات المعاملات المتغيرة (المحلولة حلًّا كاملًا بطريقة \omterm{thm:b1:diffeq:voc}{تغيير
الثابت})، والمعادلات الخطية من الرتبة الثانية ذات المعاملات
الثابتة، وهي نموذج التذبذبات. وتُظهر الحالتان البنية نفسها:
\emph{الحل العام $=$ حلّ خاص واحد $+$ الحل العام للمعادلة
المتجانسة}.

\section{المعادلات الخطية من الرتبة الأولى}

\begin{definition}\label{def:b1:diffeq:linear1}
ليكن $I$ مجالًا ولتكن $a, b \colon I \to \R$ (أو $\C$)
متصلتين. المعادلة
\[
(E)\colon\quad y' + a(x)\,y = b(x),
\]
ذات المجهول الدالة القابلة للاشتقاق $y \colon I \to \R$ (أو $\C$)،
تُسمّى \emph{معادلة تفاضلية خطية من الرتبة
الأولى}\index{معادلة تفاضلية!خطية من الرتبة الأولى}. والمعادلة
$(H)\colon y' + a(x) y = 0$ هي معادلتها \emph{المتجانسة}.
\end{definition}

\begin{theorem}[حلّ المعادلة المتجانسة]\label{thm:b1:diffeq:homogeneous1}
لتكن $A$ دالة أصلية للدالة $a$ على $I$ (وهي موجودة:
\cref{ch:b1:integration}). حلول $(H)$ على $I$ هي بالضبط
الدوال
\[
y(x) = \lambda\, \eu^{-A(x)}, \qquad \lambda \in \R \text{ (أو } \C).
\]
\end{theorem}

\begin{proof}
هذه الدوال حلول: $y' = -\lambda A' \eu^{-A} = -a y$.
وبالعكس، ليكن $y$ حلًّا للمعادلة $(H)$ ولنضع $z(x) = y(x)\, \eu^{A(x)}$.
عندئذ
\[
z' = y' \eu^{A} + y\, a\, \eu^{A} = (y' + ay)\, \eu^{A} = 0 ,
\]
فتكون $z$ ثابتة على المجال $I$، وليكن $z = \lambda$: أي $y =
\lambda \eu^{-A}$. (ولاحظ المنطق: لا يضيع أيّ حلّ، لأن
\emph{كل} حلّ كُتب في الصورة المعلنة.)
\end{proof}

\begin{example}[معادلة متجانسة بمعامل متغير]\label{ex:b1:diffeq:varcoeff}
حُلَّ $y' + (\cos x)\,y = 0$ على $\R$. الدالة الأصلية للدالة $a(x) =
\cos x$ هي $A(x) = \sin x$، ومنه فالحلول هي
\[
y(x) = \lambda\,\eu^{-\sin x}, \qquad \lambda \in \R .
\]
وقراءتان. كل حلّ دوريّ (ودوره $2\pi$) ولا ينعدم أبدًا إلا إذا كان
$\lambda = 0$ --- فإشارة $\lambda$ هي إشارة $y$ إلى الأبد، لأن
الدالة الأسّية لا يمكن أن تقطع الصفر. والحلّ الذي يحقق $y(0) = y_0$ هو $y_0\eu^{-\sin x}$: أي منحنٍ واحد بالضبط
من العائلة يمرّ بكل نقطة ابتدائية، وهي الصورة
الأحادية البُعد في \cref{thm:b1:diffeq:voc}~(2).
\end{example}

\begin{theorem}[تغيير الثابت؛ مسألة كوشي]\label{thm:b1:diffeq:voc}
بالترميز أعلاه:
\begin{enumerate}
  \item حلول $(E)$ على $I$ هي بالضبط
        \[
        y(x) = \Bigl(\lambda + \int_{x_0}^{x} b(t)\, \eu^{A(t)} \dd t
        \Bigr)\, \eu^{-A(x)},
        \qquad \lambda \in \R,
        \]
        حيث $x_0 \in I$ مثبَّت. وعلى نحو مكافئ: الحل العام
        للمعادلة $(H)$ زائد حلّ خاص واحد للمعادلة $(E)$.
  \item من أجل كل $x_0 \in I$ وكل $y_0$، تكون \emph{لمسألة كوشي}\index{مسألة كوشي} «$(E)$ مع $y(x_0) = y_0$»
        حلٌّ واحد بالضبط على $I$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) باتباع الطريقة المسمّاة \emph{تغيير
الثابت}\index{تغيير الثابت}، نبحث عن حلول على
الصورة $y = \mu(x)\, \eu^{-A(x)}$ حيث $\mu$ قابلة للاشتقاق --- ولا تضيع
أيّ عمومية، لأن كل دالة على $I$ يمكن كتابتها هكذا
($\mu = y\,\eu^{A}$). وبالتعويض،
\[
y' + ay = \mu' \eu^{-A} - \mu a \eu^{-A} + a\mu\eu^{-A}
= \mu'\, \eu^{-A},
\]
فيكون $y$ حلًّا للمعادلة $(E)$ إذا وفقط إذا كان $\mu'(x) = b(x)\,\eu^{A(x)}$، أي إذا
وفقط إذا كان $\mu(x) = \lambda + \int_{x_0}^x b(t)\eu^{A(t)}\dd t$ من أجل
ثابت $\lambda$ ما (فدالتان أصليتان للدالة المتصلة نفسها
على مجال تختلفان بثابت).

(2) في الصيغة، $y(x_0) = \lambda\,\eu^{-A(x_0)}$: فالشرط
$y(x_0) = y_0$ يحدّد $\lambda = y_0 \eu^{A(x_0)}$ تحديدًا وحيدًا.
\end{proof}

\begin{example}\label{ex:b1:diffeq:order1}
حُلَّ $y' + \dfrac{y}{x} = x^2$ على $I = \intoo{0}{+\infty}$. هنا
$a(x) = \frac 1x$ و $A(x) = \ln x$ و$\eu^{-A(x)} = \frac 1x$.
والحلول المتجانسة: $\frac{\lambda}{x}$. وبتغيير الثابت:
$\mu'(x) = x^2 \cdot x = x^3$، ومنه $\mu = \frac{x^4}{4} + \lambda$، و
\[
y(x) = \frac{x^3}{4} + \frac{\lambda}{x}, \qquad \lambda \in \R .
\]
ومع الشرط الابتدائي $y(1) = 0$: يكون $\lambda = -\frac14$.
\emph{وللتحقق:} $y' + \frac yx = \frac{3x^2}{4} - \frac{\lambda}{x^2} +
\frac{x^2}{4} + \frac{\lambda}{x^2} = x^2$.
\end{example}

\begin{example}[التخمين يسبق المكاملة]\label{ex:b1:diffeq:guess}
حُلَّ $y' + 2x\,y = x$ على $\R$. \omterm{thm:b1:diffeq:voc}{تغيير الثابت} يفي بالغرض ($A =
x^2$ و $\mu' = x\,\eu^{x^2}$ و$\mu = \frac12\eu^{x^2} + \lambda$)،
لكن ملاحظة أن \emph{الثابت} $y_p = \frac12$ حلٌّ
للمعادلة ($0 + 2x\cdot\frac12 = x$) أسرع. ومع
الحلول المتجانسة $\lambda\,\eu^{-x^2}$:
\[
y(x) = \frac12 + \lambda\,\eu^{-x^2}, \qquad \lambda \in \R .
\]
وكل حلّ يتقارب نحو $\frac12$ بسرعة بالغة عندما $x \to
\pm\infty$: فالحل الخاص الثابت \emph{توازن}
تلتحق به كل الحلول. والفكرة النافذة: قبل إطلاق
الطريقة العامة، أنفق عشر ثوانٍ في البحث عن حلّ خاص
بيّن (ثابت، أو حدّانيّ، أو مضاعف للطرف الأيمن)؛ ثم تُتمّ
مبرهنة البنية العمل.
\end{example}

\begin{remark}[المجالات مهمّة]
تحيا المبرهنة على \emph{مجال} تكون فيه $a$ و $b$
متصلتين. فمن أجل $y' + \frac yx = 0$ على $\R^*$، تكون الحلول
$\frac{\lambda}{x}$ على $\intoo{0}{+\infty}$ و
$\frac{\mu}{x}$ على $\intoo{-\infty}{0}$ بثابتين
\emph{مستقلين}: فلا سبب يجعل صيغة واحدة تلتحم عبر
الشذوذ عند $0$.
\end{remark}

\begin{example}[طرف أيمن عقديّ، وجوابان حقيقيان]\label{ex:b1:diffeq:complexrhs}
حُلَّ $y' - y = \cos x$ و $y' - y = \sin x$ دفعة واحدة. اعمل
في $\C$ مع الطرف الأيمن $\eu^{\iu x}$: فتجريب $y_p =
c\,\eu^{\iu x}$ يعطي $c(\iu - 1)\eu^{\iu x} = \eu^{\iu x}$، ومنه
\[
c = \frac1{\iu - 1} = \frac{-1 - \iu}2,
\qquad
y_p = -\frac{(1 + \iu)(\cos x + \iu\sin x)}2
= \frac{\sin x - \cos x}2 + \iu\,\frac{-\sin x - \cos x}2 .
\]
وبما أن للمعادلة معاملات حقيقية، ينفصل الجزآن الحقيقي
والتخيلي: فالدالة $\frac{\sin x - \cos x}2$ حلٌّ للمعادلة $y' - y = \cos x$،
والدالة $-\frac{\sin x + \cos x}2$ حلٌّ للمعادلة $y' - y = \sin x$ (وتحقق
من الأولى: المشتقة $\frac{\cos x + \sin x}2$، ناقصَ
الدالة، تعطي $\cos x$). فقد عوّض سطر عقديّ واحد عن جولتين من
\omterm{thm:b1:diffeq:voc}{تغيير الثابت} --- وهو الاقتصاد نفسه الذي يمنهجه
\cref{met:b1:diffeq:particular} من أجل الرتبة الثانية،
وهو فائدة متكررة من \cref{ch:b1:complex}.
\end{example}

\section{المعادلات الخطية من الرتبة الثانية ذات المعاملات الثابتة}

\begin{definition}\label{def:b1:diffeq:linear2}
ليكن $a, b \in \R$ ولتكن $f \colon I \to \R$ متصلة. المعادلة
\[
(E)\colon\quad y'' + a\,y' + b\,y = f(x)
\]
تُسمّى \emph{معادلة خطية من الرتبة الثانية ذات معاملات
ثابتة}\index{معادلة تفاضلية!خطية من الرتبة الثانية}؛
و$(H)\colon y'' + ay' + by = 0$ هي معادلتها المتجانسة، و
$\chi(r) = r^2 + ar + b$ هو \emph{كثير حدودها
المميِّز}\index{كثير الحدود المميز}.
\end{definition}

\begin{theorem}[الحلول المتجانسة]\label{thm:b1:diffeq:homogeneous2}
ليكن $\Delta = a^2 - 4b$ مميّز $\chi$. الحلول الحقيقية
للمعادلة $(H)$ على $\R$ هي:
\begin{enumerate}
  \item إذا كان $\Delta > 0$، مع $r_1 \neq r_2$ الجذرين الحقيقيين:
        $\;y = \lambda\, \eu^{r_1 x} + \mu\, \eu^{r_2 x}$؛
  \item وإذا كان $\Delta = 0$، مع $r_0$ الجذر المضاعف:
        $\;y = (\lambda + \mu x)\, \eu^{r_0 x}$؛
  \item وإذا كان $\Delta < 0$، والجذران $\alpha \pm \iu\omega$
        (حيث $\omega > 0$):
        $\;y = \eu^{\alpha x} (\lambda \cos\omega x + \mu
        \sin\omega x)$؛
\end{enumerate}
وفي كل حالة يجري $(\lambda, \mu)$ على $\R^2$.
\end{theorem}

\begin{proof}
لاحظ أولًا أنه من أجل $r \in \C$، تكون $x \mapsto \eu^{rx}$ حلًّا للمعادلة $(H)$
إذا وفقط إذا كان $\chi(r) = 0$ (عوّض: $(r^2 + ar + b)\eu^{rx} =
0$). وهذا هو سبب كون الأسّيات التخمين الأول الطبيعي:
فالاشتقاق يؤثر في $\eu^{rx}$ كالضرب في العدد
$r$، فتصير المعادلة التفاضلية المعادلةَ العددية
$\chi(r) = 0$ --- وتنضغط المسألة التحليلية كلها في
إيجاد جذور معادلة واحدة من الدرجة الثانية.

والخطوة المفتاحية تغييرُ مجهول يخفض الرتبة. ليكن $r$
جذرًا (عقديًا ربما) لكثير الحدود $\chi$ ولنكتب $y = z\, \eu^{rx}$،
وهذا لا يفقد أيّ عمومية. عندئذ
\[
y'' + ay' + by
= \bigl(z'' + (2r + a) z' + \chi(r) z\bigr)\eu^{rx}
= \bigl(z'' + (2r + a) z'\bigr)\eu^{rx},
\]
فتصير $(H)$ المعادلةَ \emph{من الرتبة الأولى} $u' + (2r + a) u = 0$
من أجل $u = z'$.

\emph{الحالة $\Delta \neq 0$:} نختار $r = r_1$؛ عندئذ $2r_1 + a = r_1 -
r_2$ (لأن $r_1 + r_2 = -a$). وحسب
\cref{thm:b1:diffeq:homogeneous1}، يكون $z' = c\,\eu^{(r_2 - r_1)x}$ من أجل
ثابت $c$ ما؛ وبالمكاملة على $\R$، يكون $z = \mu\, \eu^{(r_2 - r_1)x}
+ \lambda$ حيث $\mu = \frac{c}{r_2 - r_1}$، ومنه $y = z\,
\eu^{r_1 x} = \lambda\,\eu^{r_1x} + \mu\,\eu^{r_2x}$.
وعندما $\Delta < 0$، يكون الجذران $\alpha \pm \iu\omega$ وتكون
الحلول العقدية $y = c_1\eu^{(\alpha+\iu\omega)x} +
c_2\eu^{(\alpha-\iu\omega)x}$ حيث $c_1, c_2 \in \C$. فأيّ
هذه الحلول حقيقية القيم؟ بما أن $\conj{\eu^{(\alpha+\iu\omega)x}} =
\eu^{(\alpha-\iu\omega)x}$، يكون مرافق $y$ هو $\conj{c_2}\,
\eu^{(\alpha+\iu\omega)x} + \conj{c_1}\,\eu^{(\alpha-\iu\omega)x}$،
ويفرض $y = \conj y$ من أجل كل $x$ أن $c_2 = \conj{c_1}$ (فالأسّيتان
مستقلتان خطيًا: احسب عند نقطتين، أو
قارن عند $x=0$ بعد القسمة على $\eu^{\alpha x}$). وبكتابة $c_1
= \frac{\lambda - \iu\mu}2$ حيث $\lambda, \mu$ حقيقيان:
\[
y = 2\,\Re\Bigl(c_1\,\eu^{\alpha x}(\cos\omega x +
\iu\sin\omega x)\Bigr)
= \eu^{\alpha x}\bigl(\lambda\cos\omega x + \mu\sin\omega
x\bigr),
\]
وبالعكس فإن كل دالة كهذه حلّ (فهي جزء حقيقي لحلّ
عقديّ لمعادلة حقيقية): فيكون فضاء الحلول الحقيقية
كما أُعلن.

\emph{الحالة $\Delta = 0$:} $r = r_0$ و $2r_0 + a = 0$، ومنه $z'' = 0$:
أي $z = \lambda + \mu x$ و$y = (\lambda + \mu x)\eu^{r_0 x}$.
\end{proof}

\begin{example}[مسألة كوشي من أولها إلى آخرها]\label{ex:b1:diffeq:cauchy2}
حُلَّ $y'' - 3y' + 2y = 0$ مع $y(0) = 0$ و $y'(0) = 1$. لكثير
الحدود المميِّز $r^2 - 3r + 2 = (r - 1)(r - 2)$ الجذران
الحقيقيان $1$ و $2$: فالحل العام $y = \lambda\eu^{x} +
\mu\eu^{2x}$. ويعطي الشرطان الجملة الخطية
\[
\lambda + \mu = 0, \qquad \lambda + 2\mu = 1 ,
\]
ومنه $\mu = 1$ و $\lambda = -1$:
\[
y(x) = \eu^{2x} - \eu^{x} .
\]
وللتحقق: $y(0) = 0$؛ و$y' = 2\eu^{2x} - \eu^x$ يحقق $y'(0) = 1$؛ و
$y'' - 3y' + 2y = (4 - 6 + 2)\eu^{2x} + (-1 + 3 - 2)\eu^x = 0$.
ولاحظ شكل الجواب: فقرب $-\infty$ يهيمن النمط البطيء
$-\eu^x$؛ وقرب $+\infty$ يهيمن النمط السريع $\eu^{2x}$.
وقراءة الحلول بوصفها تراكبًا لأنماط ذات معدلات تناقص أو
نموّ مختلفة عادةٌ مربحة --- فبها يُنظَّم
فصل العابر عن المستقرّ في مسألة نهاية الأسبوع.
\end{example}

\begin{theorem}[البنية ومسألة كوشي]\label{thm:b1:diffeq:structure2}
\begin{enumerate}
  \item إذا كان $y_p$ حلًّا خاصًّا للمعادلة $(E)$، فإن حلول
        $(E)$ هي بالضبط $y_p + y_h$، حيث يجري $y_h$ على
        حلول $(H)$.
  \item (التراكب) إذا كان $y_1$ حلًّا للمعادلة $y'' + ay' + by = f_1$ وكان
        $y_2$ حلًّا للمعادلة $y'' + ay' + by = f_2$، فإن $y_1 + y_2$ حلٌّ
        للمعادلة ذات الطرف الأيمن $f_1 + f_2$.
  \item من أجل كل $x_0 \in I$ وكل $(y_0, y_0')$، يكون \omterm{thm:b1:diffeq:voc}{لمسألة كوشي}
        «$(E)$ مع $y(x_0) = y_0$ و $y'(x_0) = y_0'$» حلٌّ واحد
        بالضبط على $I$. \emph{(والوجود ممنوح بوجود حلّ
        خاص؛ والوحدانية كاملة.)}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) يكون $y$ حلًّا للمعادلة $(E)$ إذا وفقط إذا كان $y - y_p$ حلًّا للمعادلة $(H)$، بخطية $y
\mapsto y'' + ay' + by$. و (2) هي الخطية نفسها.

(3) حسب (1) يكفي أن نبرهن على أن الثابتين $(\lambda,
\mu)$ يمكن دائمًا ضبطهما، ضبطًا وحيدًا، على أيّ معطيات $(y_0, y_0')$.
وبانسحاب المتغير، لنفترض $x_0 = 0$. في الحالة (1) من
\cref{thm:b1:diffeq:homogeneous2}، يعطي $y = \lambda\eu^{r_1x} +
\mu\eu^{r_2x}$
\[
y(0) = \lambda + \mu, \qquad y'(0) = r_1\lambda + r_2\mu :
\]
وهي جملة خطية في $(\lambda, \mu)$ محدّدها $r_2 -
r_1 \neq 0$؛ وبحلّها صراحةً نجد $\mu = \frac{y_0' -
r_1y_0}{r_2 - r_1}$ و $\lambda = y_0 - \mu$: أي حلّ واحد
بالضبط. وفي الحالة (2)، $y(0) = \lambda$ و$y'(0) = r_0\lambda +
\mu$: فالجملة مثلثية ومحدّدها $1$، وحلّها
$\lambda = y_0$ و $\mu = y_0' - r_0y_0$. وفي الحالة (3)، $y(0) =
\lambda$ و$y'(0) = \alpha\lambda + \omega\mu$: فالمحدّد
$\omega \neq 0$، والحلّ $\lambda = y_0$ و$\mu = \frac{y_0' -
\alpha y_0}\omega$. وفي كل حالة يكون \omterm{def:b1:logic:map}{التطبيق} $(\lambda, \mu) \mapsto
(y(x_0), y'(x_0))$ تقابلًا خطيًا --- وستضغط لغة
\cref{ch:b1:linmaps} فحصَ الحالات هذا في جملة
واحدة.
\end{proof}

\begin{method}[حلّ خاص من أجل $f(x) = P(x)\,\eu^{\gamma x}$]\label{met:b1:diffeq:particular}
عندما يكون الطرف الأيمن $P(x)\,\eu^{\gamma x}$ حيث $P$ كثير
حدود و $\gamma \in \R$ (وهذا يشمل كثيرات الحدود والأسّيات،
وعبر $\gamma$ عقديّ أو التراكب، الدالتين $\cos$ و $\sin$): ابحث
عن حلّ خاص على الصورة
\[
y_p(x) = x^{m}\, Q(x)\, \eu^{\gamma x},
\qquad
m = \text{تضاعف } \gamma \text{ بوصفه جذرًا لكثير الحدود } \chi
\ (m = 0, 1 \text{ أو } 2),
\]
حيث $Q$ كثير حدود من درجة $P$ نفسها، وتُوجد معاملاته
بالتعويض والمطابقة. ومن أجل $f = K\cos\omega x$
(أو $\sin$)، حُلَّ مع الطرف الأيمن $K\eu^{\iu\omega x}$ وخذ
الجزء الحقيقي (أو التخيلي على التوالي).
\end{method}

\begin{example}[التراكب في العمل]\label{ex:b1:diffeq:superposition}
حُلَّ $y'' - y = \eu^{x} + 4$ على $\R$. والمتجانسة: $\chi(r) = r^2
- 1$، وجذراها $\pm1$، ومنه $y_h = \lambda\eu^x + \mu\eu^{-x}$. وافصل
الطرف الأيمن وعالج كل قطعة بصندوق الطريقة.
\emph{القطعة $\eu^x$:} هنا $\gamma = 1$ جذر بسيط لكثير الحدود $\chi$، فجرّب $y_1 = c\,x\,\eu^x$: عندئذ $y_1'' - y_1 = c(x +
2)\eu^x - cx\eu^x = 2c\,\eu^x$، ومنه $c = \frac12$.
\emph{القطعة $4$:} العدد $\gamma = 0$ ليس جذرًا؛ فالثابت $y_2 =
-4$ يفي بالغرض. وبالتراكب
(\cref{thm:b1:diffeq:structure2}~(2)):
\[
y = \frac{x\,\eu^x}2 - 4 + \lambda\,\eu^x + \mu\,\eu^{-x},
\qquad (\lambda, \mu) \in \R^2 .
\]
ولاحظ كيف اقتضت القطعتان صورتين \emph{مختلفتين} ($m = 1$
في مقابل $m = 0$): فاختبار التضاعف يُطبَّق على كل
أسّ على حدة، وهذا هو كل مغزى فصل
الطرف الأيمن قبل التخمين.
\end{example}

\begin{example}[قاعدة التضاعف في العمل]\label{ex:b1:diffeq:multiplicity}
حُلَّ $y'' + y' = x$ على $\R$. الطرف الأيمن هو $P(x)\eu^{0
\cdot x}$ مع $P(x) = x$، والعدد $\gamma = 0$ جذر \emph{بسيط} لكثير الحدود $\chi(r) = r^2 + r = r(r + 1)$: ومنه $m = 1$، ويكون التخمين
الصحيح $y_p = x\,(\alpha x + \beta) = \alpha x^2 + \beta x$، أي بدرجة
أعلى بواحد من درجة $P$. وبالتعويض:
\[
y_p'' + y_p' = 2\alpha + (2\alpha x + \beta)
= 2\alpha x + (2\alpha + \beta) ,
\]
والمطابقة مع $x$ تعطي $\alpha = \frac12$ و $\beta = -1$:
$y_p = \frac{x^2}2 - x$. والحل العام: $y = \frac{x^2}2 - x +
\lambda + \mu\,\eu^{-x}$. ولو خمّنّا $y_p = \alpha x + \beta$
(بتجاهل التضاعف) لأعطى التعويض $y_p'' + y_p'
= \alpha$، وهو ثابت --- فلا اختيار للثابتين $\alpha, \beta$ يطابق
$x$، والإخفاق بنيويّ: فالثوابت تحلّ المعادلة
المتجانسة أصلًا، فهي غير مرئية للطرف الأيسر.
والعامل $x^m$ موجود بالضبط للتسلّق خارج فضاء الحلول
المتجانسة.
\end{example}

\begin{remark}[بوليصة التأمين ذات الثلاثين ثانية]\label{rem:b1:diffeq:check}
تنتهي كل معادلة محلولة في هذا الفصل بتحقق بالتعويض،
وليس هذا زخرفًا. فحساب معادلة تفاضلية
يصل خطوات صغيرة كثيرة (دالة أصلية، وقاعدة جداء،
وثابتان)، وخطأ إشارة واحد ينتشر دون أن يُرى؛
وتعويض الصيغة النهائية في المعادلة يمسك بها كلها
تقريبًا بكلفة اشتقاق واحد.
فنمِّ هذا المنعكس في ثلاث طبقات: تحقق من \emph{الحل
الخاص} وحده (فالجزء المتجانس يختصر على أيّ حال)، وتحقق من
\emph{الشروط الابتدائية} على الحل الكامل، وإذا وُجد
وسيط فتحقق من \emph{قيمة منحلّة} (هل تعيد
صيغة $\Omega$ العامة الجوابَ المعروف عند
$\Omega = 0$؟). وتكلّف هذه العادة نصف دقيقة؛ وهي تحوّل
«صحيح غالبًا» إلى «متحقَّق منه».
\end{remark}

\begin{remark}[مزالق شائعة]\label{rem:b1:diffeq:pitfalls}
\begin{enumerate}
  \item \emph{سوِّ أولًا.} تفترض الصيغ أن المعادلة
        تُقرأ $y' + a(x)y = b(x)$ --- أي بمعامل $1$ للمقدار $y'$.
        فمن أجل $xy' - 2y = x^3$، اقسم على $x$ (على مجال
        يتفادى $0$) قبل تعيين $a$ و $b$، كما في
        \cref{exo:b1:diffeq:2}.
  \item \emph{ثابت واحد لكل بُعد، يُثبَّت في النهاية.} يحمل
        الحل العام من الرتبة الأولى ثابتًا واحدًا، ويحمل
        الذي من الرتبة الثانية ثابتين؛ وتُفرض الشروط الابتدائية على
        الحل \emph{الكامل} $y_p + y_h$، لا على $y_h$
        وحده --- وفرضها قبل إضافة $y_p$ أشيع
        الأخطاء البنيوية.
  \item \emph{انتبه إلى التضاعف.} تخمين حلّ خاص
        يحلّ المعادلة المتجانسة غير مرئيّ للطرف
        الأيسر؛ والعامل $x^m$ في
        \cref{met:b1:diffeq:particular} ليس اختياريًا
        (\cref{ex:b1:diffeq:multiplicity}).
  \item \emph{المجالات جزء من الجواب.} تحيا الحلول
        على مجالات تكون فيها المعاملات متصلة؛
        وقد يخلق الالتحام عبر شذوذ ثوابت زائفة
        (\cref{exo:b1:diffeq:12}) أو يهدم الوحدانية. \omterm{def:b1:logic:statement}{فعبارة} «حُلَّ
        على $\R^*$» تعني مسألتين مستقلتين.
\end{enumerate}
\end{remark}

\begin{example}[إثارة بعيدة عن الرنين]\label{ex:b1:diffeq:offresonance}
حُلَّ $y'' + 4y = \sin x$ على $\R$. التواتر الذاتي $2$، وتواتر
الإثارة $1$: وبما أن $\iu$ \emph{ليس} جذرًا لكثير الحدود $\chi(r) = r^2
+ 4$، يكون التضاعف $m = 0$ ويكفي جيب بسيط.
وبتجريب $y_p = \alpha\sin x$ (ولا حاجة إلى جيب تمام: فالمعادلة
بلا حدّ $y'$، والدالة $\sin$ تولّد $\sin$):
\[
y_p'' + 4y_p = -\alpha\sin x + 4\alpha\sin x = 3\alpha\sin x ,
\]
ومنه $\alpha = \frac13$ ويكون الحل العام
\[
y = \frac{\sin x}3 + \lambda\cos 2x + \mu\sin 2x .
\]
وكل حلّ يبقى محدودًا: فهو تراكب لتذبذبين
عند التواترين $1$ (المفروض) و $2$ (الذاتي). وقارن ذلك
بالمثال التالي، حيث تغيّر الإثارة \emph{عند} التواتر الذاتي
شكلَ الجواب نفسه.
\end{example}

\begin{example}[تذبذب مُثار]\label{ex:b1:diffeq:oscillation}
حُلَّ $y'' + y = \cos x$ مع $y(0) = 0$ و $y'(0) = 0$.

\emph{المتجانسة:} $\chi(r) = r^2 + 1$، وجذراها $\pm\iu$: أي $y_h =
\lambda\cos x + \mu \sin x$.

\emph{الحل الخاص:} الطرف الأيمن $\Re(\eu^{\iu x})$ مع $\gamma =
\iu$ جذرًا بسيطًا لكثير الحدود $\chi$: فجرّب $z_p = c\, x\, \eu^{\iu x}$
(حيث $c \in \C$). عندئذ $z_p'' + z_p = c\,(2\iu)\eu^{\iu x}$، وهو يساوي
$\eu^{\iu x}$ من أجل $c = \frac{1}{2\iu} = -\frac\iu2$. ومنه $z_p =
-\frac{\iu}{2} x (\cos x + \iu \sin x)$ و$y_p = \Re(z_p) =
\frac{x \sin x}{2}$.

\emph{الحل العام:} $y = \frac{x\sin x}{2} + \lambda\cos x +
\mu\sin x$. والشروط: $y(0) = \lambda = 0$؛ و$y' = \frac{\sin x +
x\cos x}{2} + \mu\cos x$، ومنه $y'(0) = \mu = 0$. والجواب: $y =
\frac{x\sin x}{2}$ --- وهو تذبذب تنمو سعته خطيًا:
أي ظاهرة \emph{الرنين}\index{رنين}، الناتجة عن إثارة
الجملة عند تواترها الذاتي.
\end{example}

\begin{remark}[استراحة: الخطية هندسة]\label{rem:b1:diffeq:interlude}
انظر إلى شكل كل \omterm{def:b1:logic:sets}{مجموعة} حلول في هذا الفصل: حلّ
خاص زائد فضاء من الحلول المتجانسة فيه ثابت
حرّ واحد (الرتبة الأولى) أو اثنان (الرتبة الثانية). وستوفّر فصول
الجبر الخطي (\cref{ch:b1:vspaces,ch:b1:findim,ch:b1:linmaps})
المفردات المضبوطة: \omterm{def:b1:logic:map}{فالتطبيق} $L(y) = y'' + ay' + by$ \emph{خطيّ}،
وحلوله المتجانسة تكوّن \emph{نواة} $L$، وهي فضاء
متجهي \emph{بُعده} يساوي رتبة المعادلة ---
وهذا هو المضمون الأمين \omterm{def:b1:logic:statement}{لعبارة} «ثابت واحد لكل رتبة» ---
\omterm{def:b1:logic:sets}{ومجموعة} حلول $L(y) = f$ \emph{فضاء جزئي تآلفي}، أي
منسحب النواة. بل إن \omterm{def:b1:logic:map}{تطبيق} كوشي $(\lambda, \mu)
\mapsto (y(x_0), y'(x_0))$ في \cref{thm:b1:diffeq:structure2}
تقابل خطيّ بين مستويين، أي جملة $2
\times 2$ قابلة للقلب (\cref{ch:b1:matrices}). ولا شيء في هذا الفصل
يحتاج إلى إعادة --- بل يحتاج إلى إعادة تسمية لا غير،
وإعادة التسمية أفضل تسخين ممكن للجبر الخطي:
فكل تعريف مجرد هناك كسب رزقه هنا سلفًا.
\end{remark}

\begin{remark}[أين يُستعمل هذا الفصل]\label{rem:b1:diffeq:whereused}
مبرهنة البنية --- أي أن حلول $(E)$ تكوّن «حلًّا
خاصًّا زائد حلول $(H)$» --- هي الظهور الأول
لنمط سيسمّيه \cref{ch:b1:vspaces,ch:b1:linmaps}: \omterm{def:b1:logic:sets}{فمجموعة} حلول
$(H)$ هي \emph{نواة} \omterm{def:b1:logic:map}{التطبيق} الخطي $y
\mapsto y'' + ay' + by$، \omterm{def:b1:logic:sets}{ومجموعة} حلول $(E)$ منسحبٌ
تآلفيّ لها. ويعود كثير الحدود المميِّز في صورة
كثير الحدود المميِّز لمصفوفة في \cref{ch:b1:matrices}:
فالمعادلة من الرتبة الثانية جملةٌ $2 \times 2$ من الرتبة الأولى
متنكرة، وهو منظور يمنهجه مجلد السنة 2. و
التكاملات التي يقتضيها \omterm{thm:b1:diffeq:voc}{تغيير الثابت} يوفّرها
\cref{ch:b1:integration}، ومسألة نهاية الأسبوع أدناه --- المتذبذب
المخمَد المُثار --- هي الحالة النموذج لكل سؤال عن
التذبذب في العلوم، من الدارات إلى الجسور المعلقة.
\end{remark}

\begin{exercise}[$\star\star\star$]\label{exo:b1:diffeq:9}
(المتذبذب المخمَد) من أجل $\varepsilon \geq 0$، تأمّل
$y'' + 2\varepsilon y' + y = 0$.
\begin{enumerate}
  \item حُلَّ من أجل $\varepsilon \in \intco{0}{1}$ ومن أجل $\varepsilon = 1$
        ومن أجل $\varepsilon > 1$.
  \item بيّن أنه من أجل $\varepsilon > 0$ يؤول كل حلّ إلى $0$
        عند $+\infty$، وأنه من أجل $\varepsilon = 0$ لا تؤول
        الحلول غير المعدومة إلى ذلك.
  \item من أجل $\varepsilon \in \intoo{0}{1}$، بيّن أن أصفار حلّ غير
        معدوم متباعدة بانتظام، وأن الفجوة
        $\frac{\pi}{\sqrt{1 - \varepsilon^2}}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:diffeq:10}
جد كل الدوال $f \colon \R \to \R$ القابلة للاشتقاق مرتين والتي
تحقق
\[
\forall x, y \in \R, \qquad f(x + y) + f(x - y) = 2 f(x) f(y),
\]
مع $f(0) \ne 0$ و $f$ غير ثابتة.
\emph{إرشاد: ثبّت $y$، واشتقّ مرتين بالنسبة إلى $x$ عند $0$؛
وبيّن أن $f(0) = 1$ وأن $f'' = c f$ من أجل ثابت $c$ ما؛ ثم حُلَّ
بحسب إشارة $c$ وتحقق من أيّ الحلول يحقق المعادلة
الدالية.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:diffeq:11}
(معادلة أويلر) حُلَّ $x^2 y'' - x y' + y = 0$ على
$\intoo{0}{+\infty}$. \emph{إرشاد: ضع $z(t) = y(\eu^t)$، أي
عوّض $x = \eu^t$، وبيّن أن $z$ تحقق معادلة خطية
ذات معاملات ثابتة.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:diffeq:12}
تأمّل المعادلة $x\,y' = 2y$ على المستقيم الحقيقي كله، ذات
المجهول الدالة القابلة للاشتقاق $y \colon \R \to \R$.
\begin{enumerate}
  \item حُلَّ على $\intoo{0}{+\infty}$ وعلى $\intoo{-\infty}{0}$.
  \item بيّن أنه من أجل \emph{أيّ} ثابتين $a, b \in \R$، تكون
        الدالة التي تساوي $ax^2$ من أجل $x \geq 0$ و $bx^2$ من أجل
        $x < 0$ قابلة للاشتقاق على $\R$ وتحلّ المعادلة
        في كل مكان.
  \item استنتج أن \omterm{def:b1:logic:sets}{مجموعة} الحلول على $\R$ عائلة ذات
        وسيطين، واشرح لماذا لا يناقض هذا
        الوحدانية في \cref{thm:b1:diffeq:voc}.
\end{enumerate}
\end{exercise}

\section{مسألة: المتذبذب المخمَد المُثار}

\begin{problem}\label{pb:b1:diffeq:1}
معادلة واحدة تحكم كتلةً على نابض في وسط لزج، والشحنةَ
في دارة RLC، ومبنى يتمايل في الريح:
\[
(E_\Omega)\colon\quad
x'' + 2\lambda x' + \omega_0^2\,x = A\cos(\Omega t),
\]
حيث $\lambda \geq 0$ التخميد و $\omega_0 > 0$ التواتر
الذاتي و $A > 0$ و $\Omega > 0$ سعة الإثارة وتواترها.
وتستخرج هذه المسألة سلوكها الكامل: تناقص
العابرات، والنظام الدوري المستقرّ الوحيد، ومنحني \omterm{ex:b1:diffeq:oscillation}{الرنين}
وحدّته (وهي \emph{\omterm{pb:b1:diffeq:1}{معامل الجودة}})، وضربات الحالة غير
المخمَدة، وميزان الطاقة الذي يديم
التذبذب.\index{رنين}\index{متذبذب}
وما لم يُذكر خلاف ذلك، $0 < \lambda < \omega_0$ (النظام دون
التخميد الحرج) ونكتب $\omega_d = \sqrt{\omega_0^2 - \lambda^2}$.

\medskip
\noindent\textbf{الجزء 1 --- المتذبذب الحرّ.} هنا $A = 0$.
\begin{enumerate}
  \item حُلَّ المعادلة المتجانسة $(H)$ من أجل $0 < \lambda <
        \omega_0$ ومن أجل $\lambda = 0$. (والنظامان $\lambda
        \geq \omega_0$ عولجا في \cref{exo:b1:diffeq:9}؛
        فاذكرهما.)
  \item بيّن أنه من أجل كل $\lambda > 0$، تؤول كل حلول $(H)$
        إلى $0$ عند $+\infty$ --- في الأنظمة الثلاثة كلها.
  \item عرّف الطاقة $\mathcal E(t) = \frac12 x'(t)^2 +
        \frac12\omega_0^2\,x(t)^2$ على طول حلّ للمعادلة $(H)$. بيّن أن
        $\mathcal E'(t) = -2\lambda\,x'(t)^2 \leq 0$، واستنتج
        (دون حلّ أيّ شيء) أن مسألة كوشي
        «$(H)$ مع $x(t_0) = x'(t_0) = 0$» ليس لها إلا الحل
        المعدوم، من أجل كل $\lambda \geq 0$.
  \item من أجل $0 < \lambda < \omega_0$، اكتب الحل غير المعدوم على الصورة
        $x(t) = R\,\eu^{-\lambda t}\cos(\omega_d t - \varphi)$ وليكن
        $T_d = \frac{2\pi}{\omega_d}$ شبه الدور.
        بيّن أن $x(t + T_d) = \eu^{-\lambda T_d}\,x(t)$: أي أن كل
        تأرجحة هي السابقة مصغَّرةً بالعامل الثابت
        $\eu^{-\delta}$، حيث $\delta = \frac{2\pi\lambda}{\omega_d}$
        (وهو \emph{التناقص اللوغاريتمي}). واحسب $\delta$ من أجل
        $\omega_0 = 1$ و $\lambda = 0.1$.
  \item عرّف \emph{\omterm{pb:b1:diffeq:1}{معامل الجودة}} $Q =
        \dfrac{\omega_0}{2\lambda}$\index{معامل الجودة}. بيّن
        أنه بعد الزمن $\frac1\lambda$ (أي تناقص السعة بالعامل
        $\eu$ مرة واحدة)، يكون المتذبذب قد أتمّ
        $\frac{\omega_d}{2\pi\lambda}$ من أشباه الأدوار، وهو من أجل
        التخميد الضعيف ($\lambda \ll \omega_0$) قريب من
        $\frac Q\pi$: \omterm{pb:b1:diffeq:1}{فمعامل الجودة} يعدّ، بمعامل $\pi$،
        التذبذبات التي تنجو قبل أن تتناقص السعة بالعامل
        $\eu$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- النظام المستقرّ.} الآن $A > 0$ و
$\lambda > 0$.
\begin{enumerate}[resume]
  \item ابحث عن حلّ خاص بوصفه الجزء الحقيقي للمقدار
        $z\,\eu^{\iu\Omega t}$ حيث $z \in \C$
        (\cref{met:b1:diffeq:particular}). بيّن أن ذلك يفي بالغرض
        مع
        \[
        z = \frac{A}{\omega_0^2 - \Omega^2 + 2\iu\lambda\Omega} .
        \]
  \item استنتج النظام المستقرّ في صورة السعة والطور:
        $x_p(t) = R(\Omega)\cos\bigl(\Omega t -
        \varphi(\Omega)\bigr)$ حيث
        \[
        R(\Omega) = \frac{A}{\sqrt{(\omega_0^2 - \Omega^2)^2 +
        4\lambda^2\Omega^2}},
        \qquad
        \tan\varphi = \frac{2\lambda\Omega}{\omega_0^2 -
        \Omega^2},
        \quad \varphi \in \intoo0\pi .
        \]
  \item فسّر النظامين الطرفيين: احسب نهايتَي
        $R$ و $\varphi$ عندما $\Omega \to 0^+$ (أي الاستجابة
        شبه الساكنة $A/\omega_0^2$، والطور $0$) وعندما $\Omega \to
        +\infty$ (أي $R \sim A/\Omega^2 \to 0$، والطور $\to \pi$: فتتحرك
        الكتلة عكس إثارة أسرع مما ينبغي).
  \item بيّن أن \emph{كل} حلّ للمعادلة $(E_\Omega)$ هو $x_p$
        زائد حلٍّ للمعادلة $(H)$، ومنه فهو يتقارب نحو النظام
        المستقرّ $x_p$ عندما $t \to +\infty$، مهما تكن الشروط
        الابتدائية: فبعد فناء العابر لا تبقى للمتذبذب
        أيّ ذاكرة عن كيفية بدئه.
  \item بيّن أن $x_p$ هو الحل الدوري \emph{الوحيد} للمعادلة
        $(E_\Omega)$.
  \item أنجز مسألة كوشي واحدة إلى نهايتها: من أجل $x'' + 2x' + 2x =
        \cos t$ مع $x(0) = x'(0) = 0$، بيّن أن الحل هو
        \[
        x(t) = \frac{\cos t + 2\sin t}5
        - \eu^{-t}\,\frac{\cos t + 3\sin t}5 ,
        \]
        وعيّن الجزأين العابر والمستقرّ.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- منحني \omterm{ex:b1:diffeq:oscillation}{الرنين}.} دراسة
$\Omega \mapsto R(\Omega)$ على $\intoo0{+\infty}$.
\begin{enumerate}[resume]
  \item بوضع $u = \Omega^2$ و$g(u) = (\omega_0^2 - u)^2 +
        4\lambda^2 u$، بيّن: أنه إذا كان $2\lambda^2 < \omega_0^2$ بلغت $R$
        قيمة قصوى تامة عند \emph{تواتر \omterm{ex:b1:diffeq:oscillation}{الرنين}}
        $\Omega_r = \sqrt{\omega_0^2 - 2\lambda^2}$، مع
        \[
        R_{\max} = R(\Omega_r)
        = \frac{A}{2\lambda\sqrt{\omega_0^2 - \lambda^2}} .
        \]
  \item بيّن أن $\dfrac{R_{\max}}{R(0)} = Q\,\bigl(1 -
        \tfrac{\lambda^2}{\omega_0^2}\bigr)^{-1/2} \geq Q$: أي أنه عند
        \omterm{ex:b1:diffeq:oscillation}{الرنين} تتضخّم الإثارة \omterm{pb:b1:diffeq:1}{بمعامل الجودة} (أساسًا).
  \item برهن على أن سعة \emph{السرعة} $V(\Omega) =
        \Omega\,R(\Omega)$ تبلغ قيمتها القصوى عند $\Omega =
        \omega_0$ بالضبط (لا عند $\Omega_r$)، وأن الطور هناك
        $\varphi(\omega_0) = \frac\pi2$: أي أنه عند $\Omega = \omega_0$
        تكون السرعة في طور القوة بالضبط.
  \item (عرض النطاق) حُلَّ $g(u) = 2\,g(u_r)$ حلًّا مضبوطًا، حيث $u_r =
        \omega_0^2 - 2\lambda^2$، واستنتج أن التواترين
        $\Omega_\pm$ اللذين عندهما $R = R_{\max}/\sqrt2$
        يحققان $\Omega_+^2 - \Omega_-^2 =
        4\lambda\sqrt{\omega_0^2 - \lambda^2}$؛ واستنتج أنه من أجل
        التخميد الضعيف يكون عرض النطاق $\Omega_+ - \Omega_-
        \approx 2\lambda$، أي $Q \approx
        \frac{\omega_0}{\Omega_+ - \Omega_-}$: فقمم \omterm{ex:b1:diffeq:oscillation}{الرنين}
        الحادّة هي جمل ذات $Q$ عالٍ.
  \item صورة عددية من أجل $\omega_0 = 1$ و $\lambda = 0.05$
        (أي $Q = 10$) و $A = 1$: احسب $\Omega_r$ و $R_{\max}$ و
        الاستجابة الساكنة $R(0)$ وعرض النطاق التقريبي.
  \item بيّن أنه إذا كان $2\lambda^2 \geq \omega_0^2$ فإن $R$
        متناقصة تمامًا على $\intoo0{+\infty}$: أي أن الجمل الشديدة
        التخميد بلا قمة رنين البتة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- بلا تخميد: \omterm{pb:b1:diffeq:1}{الضربات} \omterm{ex:b1:diffeq:oscillation}{والرنين}.}
هنا $\lambda = 0$.
\begin{enumerate}[resume]
  \item من أجل $\Omega \neq \omega_0$، جد الحل العام للمعادلة
        $x'' + \omega_0^2 x = A\cos(\Omega t)$.
  \item حُلَّ مسألة كوشي $x(0) = x'(0) = 0$ وحوّل
        الجواب إلى الصورة الجدائية
        \[
        x(t) = \frac{2A}{\omega_0^2 - \Omega^2}\,
        \sin\Bigl(\frac{(\omega_0 - \Omega)t}2\Bigr)
        \sin\Bigl(\frac{(\omega_0 + \Omega)t}2\Bigr) .
        \]
  \item من أجل $\Omega$ قريبة من $\omega_0$، اقرأ الجداء بوصفه
        تذبذبًا سريعًا عند التواتر $\frac{\omega_0 + \Omega}2$
        مضمَّنًا بغلاف بطيء عند التواتر
        $\frac{\abs{\omega_0 - \Omega}}2$: وهي \emph{الضربات}
        \index{ضربات}. أعط دور الغلاف و
        السعة القصوى، ولاحظ كيف ينفجران معًا عندما $\Omega
        \to \omega_0$.
  \item ثبّت $t$ واجعل $\Omega \to \omega_0$ في صيغة السؤال 19:
        بيّن أن النهاية هي
        \[
        x_\infty(t) = \frac{A\,t\,\sin(\omega_0 t)}{2\omega_0} ,
        \]
        وتحقق مباشرة من أن $x_\infty$ يحلّ المعادلة
        الرنينية $x'' + \omega_0^2 x = A\cos(\omega_0 t)$ مع
        $x(0) = x'(0) = 0$ (وقارن
        \cref{ex:b1:diffeq:oscillation}): \omterm{ex:b1:diffeq:oscillation}{فالرنين} هو نهاية
        ضربات تزداد بطئًا واتساعًا أبدًا.
  \item قابل بين مصيرَي \omterm{ex:b1:diffeq:oscillation}{الرنين}: النموّ الخطي
        $\frac{At}{2\omega_0}$ بلا تخميد، في مقابل التشبّع
        عند $R_{\max} \approx Q\,\frac{A}{\omega_0^2}$ مع تخميد
        ضعيف. وفي جملة واحدة: ما الآلية الفيزيائية التي تحوّل
        الأول إلى الثاني؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- ميزان الطاقة وتوليفة ختامية.}
\begin{enumerate}[resume]
  \item في النظام المستقرّ من الجزء 2، احسب متوسط
        كلٍّ ممّا يلي على دور واحد $\frac{2\pi}\Omega$: (أ) القدرة
        التي تحقنها الإثارة، $P_{\mathrm{in}}(t) = A\cos(\Omega t)
        \cdot x_p'(t)$؛ و (ب) القدرة التي يبدّدها
        التخميد، $P_{\mathrm{diss}}(t) = 2\lambda\,x_p'(t)^2$.
        وبيّن أن المتوسطين يساويان $\lambda\,R^2\Omega^2$: فتغذّي
        الإثارة بالضبط ما يحرقه التخميد --- ولهذا
        يكون النظام المستقرّ مستقرًّا.
  \item أين استعملت المسألة بالضبط: (أ) مبرهنة البنية
        \cref{thm:b1:diffeq:structure2}؛ (ب) طريقة الأسّي
        العقدي؛ (ج) دراسة دالة لمتغير حقيقي
        على نهج \cref{ch:b1:functions}؟ جملة واحدة
        لكلٍّ منها.
  \item توليفة: صِف خريطة السلوك الكاملة للمعادلة $(E_\Omega)$
        --- الحرّ في مقابل المُثار، والمخمَد في مقابل غير المخمَد،
        ودور $Q$ بوصفه القرص اللابعدي الوحيد الذي يضبط ارتفاع
        القمة وعرض النطاق وعمر العابر --- واذكر
        أين تستمر القصة: الجمل $2 \times 2$ من الرتبة الأولى
        (\cref{ch:b1:matrices} ومجلد السنة 2) و
        تفكيك إثارة دورية عامة إلى
        جيوب (متسلسلات فورييه، في مجلد السنة 3)، والحالة
        الجيبية في هذه المسألة هي لبنتها الأساسية.
\end{enumerate}
\end{problem}
