\chapter{حساب الأعداد الصحيحة}\label{ch:b1:arith}

بدأ الحساب --- أي دراسة \omterm{def:b1:arith:divides}{قابلية القسمة} في $\Z$ --- في
مجلد الثانوية. ويعيد هذا الفصل بناءه كاملًا انطلاقًا من
القسمة الإقليدية، ببراهين تامة: \omterm{thm:b1:arith:gcd}{القاسم المشترك الأكبر}
\omterm{met:b1:arith:euclid}{وخوارزمية إقليدس}، ومتطابقة بيزو ومبرهنة غاوس المساعدة، \omterm{thm:b1:arith:fta}{والتفكيك
إلى عوامل أولية}، وحساب التوافقات حتى مبرهنة فيرما
الصغرى. وإلى جانب فتنتها الخاصة، هذه المادة هي النموذج الذي
يحاكيه \cref{ch:b1:poly} من أجل كثيرات الحدود.

\section{قابلية القسمة والقسمة الإقليدية}

\begin{definition}[قابلية القسمة]\label{def:b1:arith:divides}
من أجل $a, b \in \Z$، نقول إن $b$ \emph{يقسم} $a$ (ويُكتب $b \mid
a$)\index{قابلية القسمة} عندما يكون $a = bq$ من أجل $q \in \Z$ ما. والنتائج
الأساسية: إذا كان $b \mid a$ و $b \mid a'$ فإن $b \mid (ua + va')$
لكل $u, v \in \Z$؛ وإذا كان $b \mid a$ و $a \neq 0$ فإن $\abs b \leq
\abs a$؛ و $a \mid b$ مع $b \mid a$ يفرضان $b = \pm a$.
\end{definition}

\begin{theorem}[القسمة الإقليدية]\label{thm:b1:arith:division}
لكل $a \in \Z$ وكل $b \in \N^*$، يوجد زوج واحد بالضبط $(q, r)
\in \Z \times \N$ يحقق
\[
a = bq + r, \qquad 0 \leq r < b .
\]
\end{theorem}

\begin{proof}
\emph{الوجود.} \omterm{def:b1:logic:sets}{المجموعة} $A = \{a - bk : k \in \Z\} \cap \N$ جزء
غير خالٍ من $\N$ (خذ $k = -\abs a$: $a + b\abs a \geq a +
\abs a \geq 0$). وليكن $r = a - bq$ أصغر عناصرها. فإذا كان $r \geq b$
لكان $r - b = a - b(q+1)$ عنصرًا أصغر من $A$:
وهذا تناقض. إذن $0 \leq r < b$.

\emph{الوحدانية.} إذا كان $bq + r = bq' + r'$ مع $0 \leq r, r' < b$
فإن $b(q - q') = r' - r$ و $\abs{r' - r} < b$: فمضاعف $b$
في الطرف الأيسر يجب أن يكون $0$، ومنه $q = q'$ و $r = r'$.
\end{proof}

\begin{example}[الترقيم الموضعي بالقسمة المتكررة]\label{ex:b1:arith:base7}
اكتب $2026$ في الأساس $7$. اقسم مرارًا على $7$، محتفظًا
بالبواقي:
\[
2026 = 7 \times 289 + 3, \quad
289 = 7 \times 41 + 2, \quad
41 = 7 \times 5 + 6, \quad
5 = 7 \times 0 + 5 .
\]
وبقراءة البواقي من الأخير إلى الأول: $2026 =
(5\,6\,2\,3)_7$. وللتحقق: $5 \times 343 + 6 \times 49 + 2 \times 7
+ 3 = 1715 + 294 + 14 + 3 = 2026$. ووحدانية القسمة
الإقليدية هي بالضبط ما يجعل كل رقم \emph{مفروضًا}: ففي كل
خطوة يكون الباقي هو العدد الصحيح الوحيد في $\intint06$ الموافق
للقيمة الحالية بترديد $7$، فتكون الكتابة في الأساس $7$ وحيدة ---
وهي الواقعة المستعملة ضمنًا كلما تلاعبت مسألة نهاية الأسبوع \omterm{def:b1:logic:statement}{بعبارة} «أرقام $n$ في الأساس $p$».
\end{example}

\section{القاسم المشترك الأكبر}

\begin{theorem}[الزمر الجزئية للمجموعة $\Z$؛ وجود القاسم المشترك الأكبر]\label{thm:b1:arith:gcd}
\begin{enumerate}
  \item كل زمرة جزئية من $(\Z, +)$ على الصورة $n\Z = \{nk : k
        \in \Z\}$ من أجل $n \in \N$ وحيد.
  \item من أجل $a, b \in \Z$ غير معدومين معًا، تكون \omterm{def:b1:logic:sets}{المجموعة} $a\Z + b\Z = \{au +
        bv : u, v \in \Z\}$ زمرةً جزئية من $\Z$، ومنه فهي تساوي
        $d\,\Z$ من أجل $d \in \N^*$ وحيد. وهذا العدد $d$ هو
        \emph{القاسم المشترك الأكبر}\index{القاسم المشترك الأكبر}
        $\gcd(a, b)$: فهو \omterm{def:b1:arith:divides}{يقسم} $a$ و $b$، وكل قاسم مشترك
        للعددين $a$ و $b$ \omterm{def:b1:arith:divides}{يقسم} $d$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) لتكن $H \subseteq \Z$ زمرةً جزئية (غير خالية ومستقرة
بالطرح؛ والتعريف الصوري في \cref{ch:b1:structures}،
ولا تُستعمل إلا هاتان الخاصيتان). إذا كانت $H = \{0\}$ فخذ $n = 0$.
وإلا احتوت $H$ عنصرًا غير معدوم ومقابله، ومنه
أصغر عنصر موجب تمامًا $n$. عندئذ $n\Z \subseteq H$. ومن أجل $x
\in H$، اكتب $x = nq + r$ مع $0 \leq r < n$
(\cref{thm:b1:arith:division})؛ عندئذ $r = x - nq \in H$، وتفرض أصغرية
$n$ أن $r = 0$: أي $x \in n\Z$. والوحدانية: $n$ هو أصغر عنصر موجب
في $n\Z$.

(2) تحتوي $a\Z + b\Z$ العنصرَ $0$ وهي مستقرة بالطرح، فهي إذن
$d\Z$ حيث $d \geq 1$ (إذ تحتوي $a$ أو $b$ غير المعدوم). وبما أن $a, b
\in d\Z$، \omterm{def:b1:arith:divides}{يقسم} $d$ كليهما. وإذا قسم $c$ كلًّا من $a$ و $b$ فإن $c$
\omterm{def:b1:arith:divides}{يقسم} كل $au + bv$ --- وبوجه خاص $c \mid d$، لأن $d \in a\Z
+ b\Z$. وهذه هي الخاصية المعلنة (وهي تستلزم $\abs c \leq
d$، فيستحق $d$ اسم القاسم المشترك \emph{الأكبر}).
\end{proof}

\begin{corollary}[متطابقة بيزو]\label{cor:b1:arith:bezout}
من أجل $a, b$ غير معدومين معًا، يوجد $u, v \in \Z$ يحققان
\[
au + bv = \gcd(a, b) .
\]
وبوجه خاص (الحالة $\gcd(a,b) = 1$، حالة \emph{الأوليّين فيما
بينهما}\index{أوليان فيما بينهما}): يكون $a$ و $b$ \omterm{cor:b1:arith:bezout}{أوليّين فيما بينهما} إذا وفقط إذا كان للمعادلة $au + bv = 1$ حلّ.
\end{corollary}

\begin{proof}
$\gcd(a,b) = d \in d\Z = a\Z + b\Z$. وأمّا التكافؤ: فإذا كان
$\gcd(a,b) = 1$ وفّرت متطابقة بيزو الحلّ؛ وبالعكس فإن $au + bv =
1$ يفرض على كل قاسم مشترك للعددين $a, b$ أن \omterm{def:b1:arith:divides}{يقسم} $1$.
\end{proof}

\begin{method}[خوارزمية إقليدس، الممدَّدة]\label{met:b1:arith:euclid}
لحساب $\gcd(a, b)$ (حيث $a > b > 0$): اقسم $a = bq + r$؛ عندئذ
$\gcd(a, b) = \gcd(b, r)$ (فالقواسم المشتركة للزوج $(a,b)$ وللزوج $(b,r)$ تتطابق، لأن $r = a - bq$)؛ وكرّر حتى يصير الباقي $0$؛ ويكون
آخر باقٍ غير معدوم هو \omterm{thm:b1:arith:gcd}{القاسم المشترك الأكبر}\index{خوارزمية إقليدس}. وبإجراء
القسمات بالمقلوب (أو بحفظ المعاملات أثناء
النزول) نحصل على زوج بيزو $(u, v)$.
\end{method}

\begin{example}\label{ex:b1:arith:euclid}
$\gcd(120, 23)$: $120 = 5 \times 23 + 5$؛ و $23 = 4 \times 5 + 3$؛ و$5 =
1\times 3 + 2$؛ و $3 = 1 \times 2 + 1$؛ و $2 = 2 \times 1 + 0$. ومنه
$\gcd = 1$. وبالمقلوب:
\begin{align*}
1 &= 3 - 2 = 3 - (5 - 3) = 2\times 3 - 5 = 2(23 - 4\times 5) - 5 \\
&= 2 \times 23 - 9 \times 5 = 2\times 23 - 9(120 - 5\times 23)
= 47 \times 23 - 9 \times 120 .
\end{align*}
وللتحقق: $47 \times 23 = 1081$ و $9 \times 120 = 1080$.
\end{example}

\begin{theorem}[مبرهنة غاوس المساعدة ونتائجها]\label{thm:b1:arith:gauss}
ليكن $a, b, c \in \Z$.
\begin{enumerate}
  \item (مبرهنة غاوس المساعدة\index{مبرهنة غاوس المساعدة}) إذا كان $a \mid bc$ و
        $\gcd(a, b) = 1$ فإن $a \mid c$.
  \item إذا كان $a \mid c$ و $b \mid c$ و $\gcd(a,b) = 1$ فإن $ab \mid
        c$.
  \item إذا كان $\gcd(a, b) = \gcd(a, c) = 1$ فإن $\gcd(a, bc) = 1$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) بمتطابقة بيزو: $au + bv = 1$. اضرب في $c$: $acu + bcv = c$. والحدّان
كلاهما يقبل القسمة على $a$ (والثاني لأن $a \mid bc$)، ومنه $a
\mid c$.

(2) اكتب $c = aq$؛ ومن $b \mid aq$ و $\gcd(a, b) = 1$، تعطي النقطة (1)
أن $b \mid q$، ومنه $ab \mid aq = c$.

(3) لدينا $au + bv = 1$ و $au' + cv' = 1$. اضرب العلاقتين:
\[
1 = (au + bv)(au' + cv')
= a\,\bigl(auu' + ucv' + u'bv\bigr) + bc\,(vv') ,
\]
وهي علاقة بيزو بين $a$ و $bc$: ومنه حسب
\cref{cor:b1:arith:bezout} يكون $\gcd(a, bc) = 1$.
\end{proof}

\begin{example}[حلّ معادلة ديوفانتية خطية]\label{ex:b1:arith:diophantine}
جد كل $(x, y) \in \Z^2$ يحقق $6x + 10y = 4$. أولًا
\emph{اختبار الوجود}: القاسم $\gcd(6, 10) = 2$ \omterm{def:b1:arith:divides}{يقسم} $4$، فتوجد
حلول (فلو لم \omterm{def:b1:arith:divides}{يقسم} \omterm{thm:b1:arith:gcd}{القاسم المشترك الأكبر} الطرفَ الأيمن لكان الطرف الأيسر
دائمًا مضاعفًا له ولما وُجد أيّ حلّ). واقسم
الجميع: $3x + 5y = 2$. والحل الخاص مرئيّ: $(x_0,
y_0) = (-1, 1)$. وأمّا العام فاطرح: $3(x + 1) = -5(y -
1)$، ومنه $3 \mid 5(y-1)$، وتعطي مبرهنة غاوس المساعدة (إذ $\gcd(3,5) = 1$)
أن $3 \mid y - 1$: أي $y = 1 - 3k$، ثم $x = -1 + 5k$. وبالعكس فكل
زوج كهذا يفي بالغرض:
\[
(x, y) = (-1 + 5k,\ 1 - 3k), \qquad k \in \Z .
\]
والنمط عام: حلّ خاص واحد زائد المضاعفات الصحيحة
للمقدار $\bigl(\frac b{\gcd}, -\frac a{\gcd}\bigr)$ --- وهي بنية «الخاص زائد
المتجانس» نفسها التي في
\cref{ch:b1:diffeq}، مع قيام مبرهنة غاوس المساعدة بدور
الوحدانية.
\end{example}

\begin{definition}[المضاعف المشترك الأصغر]\label{def:b1:arith:lcm}
المقدار $\operatorname{lcm}(a, b)$\index{المضاعف المشترك الأصغر} هو
المولّد في $\N$ للزمرة الجزئية $a\Z \cap b\Z$: فهو مضاعف
مشترك للعددين $a$ و $b$ \omterm{def:b1:arith:divides}{يقسم} كل مضاعف مشترك، ومن أجل
$a, b \in \N^*$،
\[
\gcd(a,b) \times \operatorname{lcm}(a,b) = ab
\qquad (\text{والبرهان في } \cref{exo:b1:arith:5}).
\]

\end{definition}

\begin{example}[مسائل التوافق مسائلُ مضاعف مشترك أصغر]\label{ex:b1:arith:gears}
لترسين متعاشقين $84$ و $36$ سنًّا. فبعد كم سنٍّ من
الحركة المشتركة يعودان معًا إلى وضعهما الابتدائي؟
يتكرر التشكيل عندما يكون عدد الأسنان المنقضية
مضاعفًا مشتركًا للعددين $84$ و $36$؛ وأول مرة هي عند
\[
\operatorname{lcm}(84, 36) = \frac{84 \times 36}{\gcd(84, 36)}
= \frac{3024}{12} = 252
\]
سنًّا --- أي $3$ دورات للترس الكبير و $7$ للترس
الصغير ($252/84$ و $252/36$). ولاحظ الطريق العملي:
\emph{احسب \omterm{thm:b1:arith:gcd}{القاسم المشترك الأكبر} أولًا} (بإقليدس: $84 = 2\times36 + 12$ و$36
= 3\times12$)، ثم اقسم --- ولا تبنِ \omterm{def:b1:arith:lcm}{المضاعف المشترك الأصغر} أبدًا بسرد
المضاعفات. فكل سؤال عن \omterm{def:b1:arith:congruence}{توافق} دوريّ (تروس، واصطفافات
كوكبية، والتقاء أعداد عشرية دورية) يُردّ إلى هذا الحساب
الواحد.
\end{example}

\section{الأعداد الأولية}

\begin{definition}\label{def:b1:arith:prime}
العدد الصحيح $p \geq 2$ يكون \emph{أوليًا}\index{عدد أولي} عندما تكون
قواسمه الموجبة الوحيدة هي $1$ و $p$. ومن أجل $p$ أوليّ و $a \in \Z$:
إمّا $p \mid a$ وإمّا $\gcd(p, a) = 1$. ومنه
(\cref{thm:b1:arith:gauss}) تصحّ \emph{مبرهنة إقليدس المساعدة}: إذا كان $p \mid
ab$ فإن $p \mid a$ أو $p \mid b$.
\end{definition}

\begin{remark}[اختبار الأولية بالقسمة التجريبية]\label{rem:b1:arith:trialdivision}
إذا كان $n = ab$ مع $2 \leq a \leq b$ فإن $a^2 \leq ab = n$، ومنه $a
\leq \sqrt n$: أي أن للعدد المركّب $n$ دائمًا قاسمًا \omterm{def:b1:arith:prime}{أوليًا} $\leq
\sqrt n$. ومنه فلاختبار أولية $n$ يكفي تجريب
الأعداد الأولية حتى $\sqrt n$. ومن أجل $n = 271$: لدينا $\sqrt{271} < 17$، و
$271$ لا يقبل القسمة على أيّ من $2, 3, 5, 7, 11, 13$ (فهو فرديّ، ومجموع
أرقامه $10$، ولا ينتهي بالرقم $0$ ولا بالرقم $5$، و$271 = 7\cdot38 + 5 =
11\cdot24 + 7 = 13\cdot20 + 11$): فهو أوليّ، بعد ستّ قسمات
بدل مئتين. والحاجز $\sqrt n$ عتبة حقيقية:
فتجاوزه بكفاءة من أجل أعداد ذات مئة رقم
يقتضي اختبارات الأولية الحديثة النابتة من
\cref{thm:b1:arith:fermat}.
\end{remark}

\begin{theorem}[إقليدس]\label{thm:b1:arith:euclidprimes}
الأعداد الأولية لا نهائية العدد.
\end{theorem}

\begin{proof}
لكل عدد صحيح $n \geq 2$ قاسم أوليّ: فأصغر قواسمه
التي $\geq 2$ أوليّ (إذ إن تعميلًا فعليًا له يُنتج
قاسمًا أصغر للعدد $n$). ولنفترض الآن أن $p_1, \dots, p_k$ هي كل
الأعداد الأولية، ولنضع $N = p_1 p_2 \cdots p_k + 1 \geq 2$. عندئذ \omterm{def:b1:arith:divides}{يقسم} عدد أوليّ $p_i$ ما
العددَ $N$؛ لكن $p_i$ \omterm{def:b1:arith:divides}{يقسم} كذلك $N - 1 = p_1\cdots p_k$، ومنه $p_i
\mid 1$ --- وهذا محال.
\end{proof}

\begin{theorem}[المبرهنة الأساسية في الحساب]\label{thm:b1:arith:fta}
كل عدد صحيح $n \geq 2$ جداءُ أعداد أولية، والتفكيك
\[
n = p_1^{\alpha_1} p_2^{\alpha_2} \cdots p_k^{\alpha_k}
\qquad (p_1 < p_2 < \dots < p_k \text{ أوليّة},\ \alpha_i \in \N^*)
\]
وحيد.\index{التفكيك إلى عوامل أولية}
\end{theorem}

\begin{proof}
\emph{الوجود} بالاستقراء القوي (\cref{thm:b1:logic:induction}):
فالعدد $n = 2$ أوليّ؛ ومن أجل $n > 2$، إمّا أن يكون $n$ \omterm{def:b1:arith:prime}{أوليًا} وإمّا $n = ab$ مع
$2 \leq a, b < n$، ويفكّك فرض الاستقراء كلًّا من $a$ و $b$.

\emph{الوحدانية.} نفترض $p_1 \cdots p_r = q_1 \cdots q_s$
(والأعداد الأولية مسرودة بتكرار، وليكن $r \leq s$)، ونستقرئ على
$r$. فإذا كان $r = 0$ كان الطرف الأيسر $1$، فيُفرض $s = 0$ (لأن جداءً
غير خالٍ من أعداد أولية يفوق $1$). ومن أجل $r \geq 1$: \omterm{def:b1:arith:divides}{يقسم} العدد الأوليّ $p_1$
المقدارَ $q_1(q_2\cdots q_s)$، ومنه بمبرهنة إقليدس المساعدة إمّا $p_1
\mid q_1$ وإمّا $p_1 \mid q_2\cdots q_s$؛ وبالتكرار \omterm{def:b1:arith:divides}{يقسم} $p_1$
عددًا $q_j$ ما. لكن $q_j$ أوليّ و $p_1 \geq 2$: فبالضرورة $p_1
= q_j$. واختصر هذا العامل المشترك (وهذا مشروع: فالحلقة $\Z$ تامة) لتحصل على
\[
p_2 \cdots p_r = q_1 \cdots \widehat{q_j} \cdots q_s
\]
(والقبعة تشير إلى الحذف)، وهو تساوٍ بين جداءين أقصر؛ ويقول
فرض الاستقراء إن القائمتين $p_2, \dots, p_r$ و
$q_1, \dots, \widehat{q_j}, \dots, q_s$ تتطابقان بغضّ النظر عن الترتيب،
ومنه كذلك القائمتان الأصليتان. وتجمع صورة الأسس الأعدادَ
الأولية المتساوية.
\end{proof}

\begin{proposition}[التقييمات]\label{prop:b1:arith:valuation}
من أجل $p$ أوليّ و $n \in \N^*$، نكتب $v_p(n)$\index{تقييم p@التقييم $p$-اديّ} للدلالة على أس $p$ في
تفكيك $n$ (مع $v_p(n) = 0$ إذا كان $p \nmid n$). عندئذ
\[
v_p(mn) = v_p(m) + v_p(n),
\qquad
m \mid n \iff \forall p,\ v_p(m) \leq v_p(n),
\]
\[
v_p\bigl(\gcd(m,n)\bigr) = \min\bigl(v_p(m), v_p(n)\bigr),
\qquad
v_p\bigl(\operatorname{lcm}(m,n)\bigr) = \max\bigl(v_p(m),
v_p(n)\bigr).
\]
\end{proposition}

\begin{proof}
تصحّ المتطابقة الأولى لأن التفكيكات تتضارب ولأن
تفكيك $mn$ وحيد. وإذا كان $m \mid n$ فاكتب $n = mq$ و
طبّقها. وبالعكس، إذا كان $v_p(m) \leq v_p(n)$ دائمًا، فإن العدد الصحيح $q =
\prod_p p^{\,v_p(n) - v_p(m)}$ يحقق $mq = n$. وأمّا صيغة \omterm{thm:b1:arith:gcd}{القاسم المشترك الأكبر}:
فالعدد الصحيح $d = \prod p^{\min}$ \omterm{def:b1:arith:divides}{يقسم} كليهما بالمعيار، و
كل قاسم مشترك $c$ يحقق $v_p(c) \leq \min$ لكل $p$، ومنه $c
\mid d$؛ والاستدلال نفسه من أجل \omterm{def:b1:arith:lcm}{المضاعف المشترك الأصغر} مع $\max$.
\end{proof}

\begin{example}[المربعات والمكعبات عبر التقييمات]\label{ex:b1:arith:squarecube}
العدد الصحيح $n \geq 1$ مربع تامّ إذا وفقط إذا كان كل
$v_p(n)$ زوجيًا (فإذا كان $n = m^2$ فإن $v_p(n) = 2v_p(m)$؛
وبالعكس نصّف كل أسّ). وبالمثل من أجل المكعبات مع
مضاعفات $3$. ومنه فإن $21168 = 2^4 \times 3^3 \times 7^2$ ليس
مربعًا (إذ $v_3 = 3$ فرديّ) ولا مكعبًا (إذ $v_2 = 4$)؛ و
أصغر عدد صحيح موجب $m$ يجعل $21168\,m$ مكعبًا يُوجد
برفع كل أسّ إلى مضاعف $3$ التالي:
\[
m = 2^{6-4} \times 3^{3-3} \times 7^{3-2} = 2^2 \times 7 = 28,
\qquad
21168 \times 28 = 2^6\,3^3\,7^3 = (2^2 \times 3 \times 7)^3
= 84^3 .
\]
والفكرة النافذة: تصير الأسئلة الجدائية (المربعات والمكعبات والقواسم
\omterm{thm:b1:arith:gcd}{والقاسم المشترك الأكبر} \omterm{def:b1:arith:lcm}{والمضاعف المشترك الأصغر}) أسئلةً \emph{إحداثيةً إحداثية} على متجهات
الأسس $(v_2, v_3, v_5, \dots)$ --- ووحدانية التفكيك هي
\omterm{def:b1:logic:statement}{العبارة} القائلة إن هذه الإحداثيات موجودة ومعرَّفة تعريفًا سليمًا.
\end{example}

\section{التوافقات}

\begin{definition}\label{def:b1:arith:congruence}
من أجل $n \in \N^*$: نكتب $a \equiv b \pmod n$\index{توافق} عندما يكون $n \mid
a - b$. وهذه \omterm{def:b1:logic:equiv}{علاقة تكافؤ} متوافقة مع الجمع و
الضرب: فإذا كان $a \equiv b$ و $a' \equiv b'$ (بترديد $n$) فإن
$a + a' \equiv b + b'$ و $aa' \equiv bb'$ و $a^k \equiv b^k$ من أجل
$k \in \N$.
\end{definition}

\begin{example}[التحقق بالتسعة]\label{ex:b1:arith:castingnines}
\omterm{def:b1:arith:congruence}{التوافق} مع $+$ و $\times$ أداة تحقق قديمة قدم
التجارة. وبما أن $10 \equiv 1 \pmod 9$، يكون كل عدد صحيح
موافقًا بترديد $9$ لمجموع أرقامه (المبرهن عليه في
\cref{exo:b1:arith:2}). وللتحقق من الدعوى $1234 \times 567 =
699\,678$: يعطي مجموعا الأرقام $1234 \equiv 1$ و$567 \equiv 18
\equiv 0 \pmod 9$، ومنه يجب أن يكون الجداء $\equiv 1 \times 0 =
0$؛ وبالفعل $6 + 9 + 9 + 6 + 7 + 8 = 45 \equiv 0$. فالتحقق
يمرّ (والجداء صحيح في الواقع). ولو أفاد أحدهم بالقيمة
$699\,478$ لأدانه مجموع الأرقام $43 \equiv 7 \not\equiv 0$
فورًا. والاختبار أحاديّ الجانب --- فهو يمسك الخطأ
إلا إذا كان الخطأ نفسه مضاعفًا للعدد $9$ --- وهذا بالضبط
درس شبه الأولية في
\cref{ex:b1:arith:pseudoprime} مصغَّرًا: فتحققات \omterm{def:b1:arith:congruence}{التوافق}
تدحض ولا تشهد.
\end{example}

\begin{proposition}[قابلية القلب بترديد $n$]\label{prop:b1:arith:invmod}
العدد $a$ \emph{قابل للقلب بترديد $n$} (أي $ab \equiv 1 \pmod n$ من أجل
$b$ ما) إذا وفقط إذا كان $\gcd(a, n) = 1$. ويكون المقلوب عندئذ وحيدًا بترديد
$n$ ويُحسب \omterm{met:b1:arith:euclid}{بخوارزمية إقليدس} الممدَّدة.
\end{proposition}

\begin{proof}
تعني $ab \equiv 1 \pmod n$ أن $ab + nk = 1$ من أجل $k$ ما: وهي
علاقة بيزو، وهي موجودة إذا وفقط إذا كان $\gcd(a,n) = 1$
(\cref{cor:b1:arith:bezout}). والوحدانية: إذا كان $ab \equiv ab' \equiv 1$
فإن $b \equiv b(ab') = (ab)b' \equiv b' \pmod n$.
\end{proof}

\begin{example}[قلب العدد $7$ بترديد $26$]\label{ex:b1:arith:inv7mod26}
بما أن $\gcd(7, 26) = 1$، يكون صف $7$ قابلًا للقلب بترديد $26$.
وبإقليدس الممدَّدة:
\[
26 = 3 \times 7 + 5, \qquad
7 = 1 \times 5 + 2, \qquad
5 = 2 \times 2 + 1 ,
\]
ثم بالمقلوب:
\[
1 = 5 - 2 \times 2 = 5 - 2(7 - 5) = 3 \times 5 - 2 \times 7
= 3(26 - 3 \times 7) - 2 \times 7 = 3 \times 26 - 11 \times 7 .
\]
ومنه $7 \times (-11) \equiv 1 \pmod{26}$، أي $7^{-1} \equiv
-11 \equiv 15 \pmod{26}$؛ وللتحقق: $7 \times 15 = 105 = 4 \times 26
+ 1$. وبامتلاك المقلوب، يُحلّ أيّ \omterm{def:b1:arith:congruence}{توافق} $7x \equiv c
\pmod{26}$ بعملية ضرب واحدة: $x \equiv 15c$. وهذا
القلب الآليّ هو حصان العمل في الحساب الترديدي ---
وفي بروتوكولات المفتاح العام المذكورة في
\cref{rem:b1:arith:whereused}، حيث تكون الترديدات ذات مئات
الأرقام لكن الخوارزمية هي هذه بالضبط.
\end{example}

\begin{example}[عندما لا يكون المعامل قابلًا للقلب]\label{ex:b1:arith:noninvertible}
حُلَّ $12x \equiv 8 \pmod{20}$. هنا $\gcd(12, 20) = 4$، فلا يكون $12$
قابلًا للقلب بترديد $20$ --- لكن المعادلة تبقى
قابلة للمعالجة. يقول \omterm{def:b1:arith:congruence}{التوافق} إن $20 \mid 12x - 8$؛ وبقسمة
العلاقة كلها على $4$ (وهو قاسم للمكوّنات الثلاثة)، تكافئ
$5 \mid 3x - 2$، أي
\[
3x \equiv 2 \pmod 5 .
\]
والآن $\gcd(3, 5) = 1$ و$3^{-1} \equiv 2 \pmod 5$ ($3 \times 2 =
6 \equiv 1$)، ومنه $x \equiv 4 \pmod 5$: فالحلول هي $x
\equiv 4, 9, 14, 19 \pmod{20}$ --- أي \emph{أربعة} صفوف بترديد $20$،
مطابقةً للقاسم المشترك الأكبر. (ولو لم يقبل الطرف الأيمن القسمة على $4$،
مثل $12x \equiv 6 \pmod{20}$، لما وُجد أيّ حلّ البتة:
فالطرف الأيسر دائمًا $\equiv 0 \pmod 4$.) والشكل العام: $ax
\equiv b \pmod n$ قابل للحلّ إذا وفقط إذا كان $\gcd(a, n) \mid b$، ويكون له عندئذ
بالضبط $\gcd(a, n)$ صفًّا من الحلول --- فاقسم كل شيء
على \omterm{thm:b1:arith:gcd}{القاسم المشترك الأكبر} ثم اقلب.
\end{example}

\begin{theorem}[مبرهنة فيرما الصغرى]\label{thm:b1:arith:fermat}
ليكن $p$ عددًا \omterm{def:b1:arith:prime}{أوليًا}. لكل $a \in \Z$:
\[
a^p \equiv a \pmod p,
\]
وإذا كان $p \nmid a$ فإن $a^{p-1} \equiv 1 \pmod
p$.\index{مبرهنة فيرما الصغرى}
\end{theorem}

\begin{proof}
أولًا، من أجل $1 \leq k \leq p - 1$، يقبل \omterm{def:b1:counting:objects}{المعامل الثنائي} $\binom pk
= \frac{p!}{k!(p-k)!}$ القسمة على $p$: إذ إن $k!\,(p-k)!\,
\binom pk = p!$ و $p$ \omterm{def:b1:arith:divides}{يقسم} $p!$ لكنه أوليّ مع $k!(p-k)!$
(فكل العوامل $< p$)، ومنه تعطي مبرهنة غاوس المساعدة أن $p \mid \binom pk$.

ولنبرهن الآن على $a^p \equiv a$ من أجل $a \in \N$ بالاستقراء. وهي صحيحة من أجل $a =
0$. وإذا كان $a^p \equiv a$ فإن مبرهنة ثنائي الحدّ تعطي
\[
(a+1)^p = \sum_{k=0}^{p} \binom pk a^k
\equiv a^p + 1 \equiv a + 1 \pmod p,
\]
إذ تنعدم كل الحدود الوسطى بترديد $p$. ومن أجل $a < 0$، طبّق النتيجة على
$-a$ وافصل $p = 2$ (حيث $x \equiv -x$) عن $p$ الفرديّ (حيث
$(-a)^p = -a^p$). وأخيرًا، إذا كان $p \nmid a$، فاضرب $a^p \equiv a$ في
مقلوب للعدد $a$ بترديد $p$ (\cref{prop:b1:arith:invmod}).
\end{proof}

\begin{example}[عكس مبرهنة فيرما يخفق: العدد $341$]\label{ex:b1:arith:pseudoprime}
تعطي مبرهنة فيرما الصغرى اختبارَ \emph{تركيبٍ} رخيصًا:
فإذا كان $a^{n-1} \not\equiv 1 \pmod n$ من أجل $a$ ما أوليّ مع $n$
فإن $n$ ليس \omterm{def:b1:arith:prime}{أوليًا}. فهل يمكن للاختبار أن يشهد بالأولية كذلك؟ لا:
خذ $n = 341 = 11 \times 31$، وهو مركّب، و $a = 2$. وبما أن
$2^{10} = 1024 = 3 \times 341 + 1$، يكون
\[
2^{10} \equiv 1 \pmod{341}
\qquad\Longrightarrow\qquad
2^{340} = \bigl(2^{10}\bigr)^{34} \equiv 1 \pmod{341} :
\]
فيجتاز العدد المركّب $341$ اختبار فيرما من أجل الأساس $2$ (وهو
أصغر \emph{شبه أوليّ} كهذا). ويكشفه الأساس $3$
(إذ $3^{340} \not\equiv 1$)، ومنه فاختبار الأولية العملي
يُجري الاختبار على عدة أسس، مع تحسينات ---
والنسخ الصناعية من هذه الفكرة هي التي تشهد بأولية
الأعداد الكبيرة في \cref{rem:b1:arith:whereused}. والعبرة: أن الاستلزام
وعكسه يحيا كلٌّ منهما حياته
(\cref{rem:b1:logic:pitfalls})، حتى في المبرهنات.
\end{example}

\begin{example}[حسابات توافق عملية]\label{ex:b1:arith:congruences}
ما باقي $7^{2026}$ بترديد $11$؟ بمبرهنة فيرما، $7^{10}
\equiv 1 \pmod{11}$. وبما أن $2026 = 10 \times 202 + 6$:
\[
7^{2026} \equiv 7^6 = (7^2)^3 = 49^3 \equiv 5^3 = 125 \equiv 4
\pmod{11}.
\]
فالباقي هو $4$. والاستراتيجية: أرجِع الأسّ بترديد
الرتبة التي توفّرها مبرهنة فيرما، ثم أرجِع القوى الوسيطة في كل
خطوة.
\end{example}

\begin{remark}[مزالق شائعة في الحساب]\label{rem:b1:arith:pitfalls}
\begin{enumerate}
  \item \emph{قسمة \omterm{def:b1:arith:congruence}{توافق}.} من $ac \equiv bc \pmod n$
        \emph{لا} يجوز استنتاج $a \equiv b$ إلا إذا كان $\gcd(c,
        n) = 1$: إذ $6 \equiv 2 \pmod 4$ لكن $3 \not\equiv 1 \pmod
        4$. والقاعدة العامة الصحيحة تقسم الترديد كذلك: $ac
        \equiv bc \pmod n \iff a \equiv b \pmod{n/\gcd(c,n)}$.
  \item \emph{إساءة استعمال مبرهنة إقليدس المساعدة.} يستلزم $a \mid bc$ أن $a
        \mid b$ أو $a \mid c$ من أجل $a$ \emph{الأوليّ} وحده (أو
        الأوليّ مع أحد العاملين): إذ $6 \mid 4 \times 9$ ومع ذلك لا \omterm{def:b1:arith:divides}{يقسم} $6$
        أيًّا من العاملين.
  \item \emph{الأولية فيما بين عددين علاقة لا خاصية.} فقولنا «العددان $8$ و
        $9$ \omterm{cor:b1:arith:bezout}{أوليّان فيما بينهما}» صحيح وإن لم يكن أيٌّ منهما \omterm{def:b1:arith:prime}{أوليًا}؛
        وقولنا «أوليّة مثنى مثنى» أقوى من «أوليّة
        إجمالًا» (إذ $\gcd(6, 10, 15) = 1$ لكن لا زوج منها
        أوليّ فيما بين عنصريه).
  \item \emph{الأسس لا تحيا بترديد $n$.} في $a^k \bmod n$،
        لا يجوز إرجاع الأسّ إلا بترديد \emph{رتبة}
        $a$ (وهي مثلًا $p - 1$ عندما تنطبق مبرهنة فيرما)، ولا
        بترديد $n$ أبدًا: إذ $2^{10} \bmod 11$ يساوي $1$، لا $2^{10 \bmod
        11} = 2^{10}$ --- والإرجاع الذي يفلح هو الذي
        يجريه \cref{ex:b1:arith:congruences}.
\end{enumerate}
\end{remark}

\begin{remark}[أين يُستعمل هذا الفصل]\label{rem:b1:arith:whereused}
هذا الفصل قالبٌ بقدر ما هو صندوق عدّة. فالسلسلة كلها
--- القسمة الإقليدية، \omterm{thm:b1:arith:gcd}{والقاسم المشترك الأكبر}، وبيزو، وغاوس، ووحدانية التفكيك
--- تُعاد حرفيًا من أجل كثيرات الحدود في \cref{ch:b1:poly}،
حيث تلعب «الدرجة» دور القيمة المطلقة؛ ومقارنة
الفصلين جنبًا إلى جنب أحسن سبيل لفهمهما معًا. ويصير
حساب التوافقات الحلقةَ $\Z/n\Z$ في
\cref{ch:b1:structures}، وتكوّن عناصرها القابلة للقلب
(\cref{prop:b1:arith:invmod}) أول مثال غير بديهي على
زمرة العناصر القابلة للقلب. وتعود التقييمات في مسألة نهاية الأسبوع أدناه
(صيغة لوجاندر) وتشغّل براهين الصمم في
\cref{ch:b1:reals}. وخارج هذا المجلد، يكون قلب بيزو بترديد $n$
محركَ التعمية ذات المفتاح العام، وتكون مبرهنة فيرما
الصغرى جدَّ اختبارات الأولية التي تشهد بأولية
الأعداد الكبيرة المستعملة هناك.
\end{remark}

\begin{remark}[استراحة: $\Z$ بوصفها قالبًا]\label{rem:b1:arith:interlude}
تراجع خطوة عن المبرهنات المفردة ولاحظ
معمار الفصل: أداة واحدة (القسمة الإقليدية)
أنتجت تصنيفًا (الزمر الجزئية $n\Z$)، الذي أنتج
مبرهنة وجود (\omterm{thm:b1:arith:gcd}{القاسم المشترك الأكبر} وبيزو)، التي أنتجت حساب
\omterm{def:b1:arith:divides}{قابلية القسمة} (غاوس)، الذي أنتج وحدانية التفكيك --- وكل
طابق يستند إلى الطابق الذي تحته لا غير. وسيُشيَّد المبنى
نفسه مرتين أخريين في هذا المجلد بطوابق أرضية مختلفة:
في \cref{ch:b1:poly}، حيث تحلّ القسمة بالدرجة محلّ القسمة
بالحجم ويتكرر كل ما فوقها \emph{حرفيًا}؛ وفي
صورة مصغَّرة داخل كل $\Z/n\Z$ في \cref{ch:b1:structures}،
حيث تصير أسئلة قابلية القلب (وهي
\cref{prop:b1:arith:invmod} من هذا الفصل) عباراتٍ بنيوية عن
الحلقات والحقول. والتعرف على حجة بوصفها «حجة $\Z$
منقولة» أسرع طريق لتعلّم تلك الفصول --- وهو
أول مذاق لعادة الجبر الأساسية، أي البرهان على مبرهنات تخصّ
\emph{بديهيات} لا أغراضًا.
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:6}
لتكن $a = 2^{10} \times 3^4 \times 5^2$ و$b = 2^6 \times 3^7 \times
7$. احسب $\gcd(a, b)$ و$\operatorname{lcm}(a,b)$ وعدد
القواسم الموجبة للعدد $a$. \emph{(وبرهن على صيغة عدّ القواسم
$\prod_i (\alpha_i + 1)$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:7}
برهن على أن $\sqrt p$ أصمّ من أجل كل عدد أوليّ $p$، باستعمال
التقييمات: قارن $v_p$ لطرفَي $p q^2 = r^2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:8}
(مسألة البواقي الصينية) جد كل الأعداد الصحيحة $x$ التي تحقق
\[
x \equiv 2 \pmod 7, \qquad x \equiv 5 \pmod{11}.
\]
وبرهن في الطريق على أنه من أجل $m, n$ \omterm{cor:b1:arith:bezout}{أوليّين فيما بينهما}، يكون لزوج التوافقين
$x \equiv a \ (m)$ و $x \equiv b\ (n)$ حلٌّ دائمًا، وحيدٌ
بترديد $mn$.\index{مبرهنة البواقي الصينية}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:9}
احسب $3^{1000}$ بترديد $7$، والرقمين العشريين الأخيرين من
$7^{100}$ \emph{(بترديد $100 = 4 \times 25$: استعمل
\cref{exo:b1:arith:8})}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:arith:10}
من أجل $m, n \in \N^*$، برهن على أن $\gcd(2^m - 1,\, 2^n - 1) =
2^{\gcd(m,n)} - 1$. \emph{إرشاد: بيّن أولًا أن باقي $2^m
- 1$ بترديد $2^n - 1$ هو $2^r - 1$ حيث $r$ باقي $m$ بترديد
$n$؛ ثم اتبع \omterm{met:b1:arith:euclid}{خوارزمية إقليدس}.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:arith:11}
(مبرهنة ويلسون) ليكن $p$ عددًا \omterm{def:b1:arith:prime}{أوليًا}. برهن على أن
\[
(p-1)! \equiv -1 \pmod p ,
\]
بازدواج كل عامل من $(p-1)!$ مع مقلوبه بترديد $p$ و
تعيين العوامل المزدوجة مع نفسها (وحُلَّ $x^2 \equiv 1 \pmod p$
أولًا). وتحقق من العكس: إذا كان $n \geq 2$ غير أوليّ فإن $(n-1)!
\not\equiv -1 \pmod n$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:arith:12}
(أعداد فيرما) من أجل $n \in \N$، لتكن $F_n = 2^{2^n} + 1$.
\begin{enumerate}
  \item برهن على أن $F_0 F_1 \cdots F_{n-1} = F_n - 2$ من أجل $n \geq
        1$ (بالاستقراء).
  \item استنتج أن أعداد فيرما أوليّة مثنى مثنى.
  \item استنتج برهانًا ثانيًا، مستقلًا عن
        \cref{thm:b1:arith:euclidprimes}، على أن الأعداد الأولية
        لا نهائية العدد.
\end{enumerate}
\end{exercise}

\section{مسألة: صيغة لوجاندر واحتفاظات كومر}

\begin{problem}\label{pb:b1:arith:1}
كم صفرًا ينتهي به التمثيل العشري للعدد $1000!$ --- وأعمق من ذلك،
ما القوة المضبوطة لعدد أوليّ $p$ التي تقسم $n!$، أو تقسم
معاملًا ثنائيًا؟ الجوابان الكاملان جوهرتان من
الحساب الابتدائي: \emph{صيغة لوجاندر}
\index{صيغة لوجاندر} $v_p(n!) = \sum_{k\geq1}
\lfloor n/p^k \rfloor$، وصورتها الرقمية $v_p(n!) = \frac{n
- s_p(n)}{p-1}$، و\emph{مبرهنة كومر}\index{مبرهنة
كومر}: إذ يعدّ $v_p\binom{m+n}m$ عددَ \emph{الاحتفاظات} عند جمع
$m$ و $n$ في الأساس $p$. وتبرهن هذه المسألة على الاثنتين، وتتحقق منهما
إحداهما بالأخرى عدديًا، وتجني النتائج الكلاسيكية
--- الأصفار الختامية، وزوجية مثلث باسكال،
وحاصرًا أول في اتجاه مبرهنة الأعداد الأولية.
وفي كل ما يلي، $p$ عدد أوليّ، و $\floor{x}$ الجزء الصحيح، و
$s_p(n)$ مجموع أرقام $n$ مكتوبًا في الأساس $p$.

\medskip
\noindent\textbf{الجزء 1 --- الأجزاء الصحيحة والتقييمات وصيغة
لوجاندر.}
\begin{enumerate}
  \item تسخين: احسب $10!$ واقرأ عدد أصفاره الختامية؛
        واحسب $v_2(10!)$ و $v_5(10!)$ مباشرةً من
        تفكيك كل عامل من $1, 2, \dots, 10$.
  \item برهن على أنه من أجل $x \in \R$ و $n \in \N^*$،
        $\bigl\lfloor \lfloor x \rfloor / n \bigr\rfloor =
        \lfloor x/n \rfloor$.
  \item برهن على أن $v_p(a + b) \geq \min\bigl(v_p(a),
        v_p(b)\bigr)$ لكل $a, b \in \N^*$، مع التساوي
        كلما كان $v_p(a) \neq v_p(b)$.
  \item بيّن أن عدد مضاعفات $m$ في $\intint1n$ هو
        $\lfloor n/m \rfloor$.
  \item برهن على \emph{صيغة لوجاندر}: من أجل كل $n \in \N^*$،
        \[
        v_p(n!) = \sum_{k=1}^{\infty}
        \Bigl\lfloor \frac{n}{p^k} \Bigr\rfloor
        \]
        (وهو مجموع منتهٍ: إذ تنعدم الحدود بمجرد أن يكون $p^k > n$).
        \emph{عُدَّ، من أجل كل $k$، عوامل $\intint1n$ التي
        تقبل القسمة على $p^k$: فيسهم كلٌّ منها بوحدة واحدة بالضبط عن كل
        مستوى يبلغه.}
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- الصورة الرقمية والأصفار
الختامية.}
\begin{enumerate}[resume]
  \item احسب $v_5(1000!)$ و $v_2(1000!)$، واستنتج: كم
        صفرًا ينتهي به $1000!$؟
  \item برهن على الصورة الرقمية لصيغة لوجاندر: بكتابة $n =
        \sum_i a_i p^i$ في الأساس $p$،
        \[
        v_p(n!) = \frac{n - s_p(n)}{p - 1} .
        \]
  \item نتيجتان من أجل $p = 2$: بيّن أن $2^n$ لا \omterm{def:b1:arith:divides}{يقسم} $n!$
        أبدًا، وأن $2^{n-1}$ \omterm{def:b1:arith:divides}{يقسم} $n!$ تحديدًا عندما يكون
        $n$ قوةً للعدد $2$.
  \item حاصر النقص: بيّن أن $\frac n{p-1} - \log_p(n) - 1 \leq
        v_p(n!) < \frac n{p-1}$، بحيث يكون $\frac{v_p(n!)}{n} \to
        \frac1{p-1}$: أي أنه على المدى الطويل تتراكم نسبة
        $\frac1{p-1}$ من عامل $p$ واحد عن كل وحدة.
  \item لتكن $Z(n) = v_5(n!)$ عدد الأصفار الختامية للعدد
        $n!$. بيّن أن $Z(n) - Z(n-1) = v_5(n)$، واستنتج أن $Z$ تتخطى
        القيمة $5$ كليًا (احسب $Z(24)$ و $Z(25)$)، و
        برهن على أنه لا يوجد عاملي ينتهي بخمسة أصفار بالضبط.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- مبرهنة كومر.}
\begin{enumerate}[resume]
  \item برهن على أن $\lfloor x + y \rfloor - \lfloor x \rfloor -
        \lfloor y \rfloor \in \{0, 1\}$ لكل $x, y \in \R$، و
        استنتج من صيغة لوجاندر أن
        \[
        v_p\binom{m+n}m
        = \sum_{k\geq1}\Bigl(
        \Bigl\lfloor\frac{m+n}{p^k}\Bigr\rfloor
        - \Bigl\lfloor\frac{m}{p^k}\Bigr\rfloor
        - \Bigl\lfloor\frac{n}{p^k}\Bigr\rfloor\Bigr),
        \]
        وهو مجموع حدود يساوي كلٌّ منها $0$ أو $1$.
  \item برهن على \emph{مبرهنة كومر}: أن الحدّ ذا الرتبة $k$ من ذلك المجموع
        يساوي $1$ تحديدًا عندما يُنتج جمع $m$ و $n$ في
        الأساس $p$ احتفاظًا نحو الموضع $k$؛ ومنه فإن
        $v_p\binom{m+n}m$ هو العدد الكلي للاحتفاظات.
        \emph{(اكتب $m = p^km_1 + m_0$ و $n = p^kn_1 + n_0$
        مع $0 \leq m_0, n_0 < p^k$ وافحص $\lfloor (m_0 +
        n_0)/p^k \rfloor$.)}
  \item استنتج أنه من أجل $0 < j < p^k$:
        \[
        v_p\binom{p^k}{j} = k - v_p(j) ,
        \]
        بعدّ الاحتفاظات في الجمع $j + (p^k - j)$.
        (وبوجه خاص $p \mid \binom p j$ من أجل $0 < j < p$: وهي
        الخطوة المفتاحية في \cref{thm:b1:arith:fermat}، مستعادةً.)
  \item برهن على أن $v_2\binom{2n}n = s_2(n)$. واستنتج أن
        \omterm{def:b1:counting:objects}{المعامل الثنائي} المركزي زوجيّ دائمًا، وأن
        $\binom{2n}n \equiv 2 \pmod 4$ تحديدًا عندما يكون $n$ قوةً
        للعدد $2$.
  \item بيّن، باستعمال متطابقة فاندرموند
        (\cref{exo:b1:counting:7}) والسؤال 13، أن
        $\binom{2p}p \equiv 2 \pmod p$ من أجل كل عدد أوليّ $p$.
  \item احسب $v_3\binom{1000}{500}$ مرتين: مرةً بمبرهنة كومر
        (اكتب $500$ في الأساس $3$ وعُدَّ الاحتفاظات في $500 +
        500$)، ومرةً بالصورة الرقمية لصيغة لوجاندر (احسب $s_3(500)$
        و $s_3(1000)$)؛ وتأكّد من أن الاثنتين تعطيان القيمة نفسها.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- زوجية مثلث باسكال، وحاصر
على كثافة الأعداد الأولية.}
\begin{enumerate}[resume]
  \item برهن على المعيار الرقمي: يكون $\binom nk$ \emph{فرديًا} إذا
        وفقط إذا كان كل رقم ثنائي من $k$ أصغر من الرقم
        المقابل من $n$ أو مساويًا له. وصُغ وبرهن على المعيار المماثل \omterm{def:b1:logic:statement}{للعبارة} $p \nmid \binom nk$ في الأساس $p$.
  \item استنتج أن السطر $n$ من مثلث باسكال يحوي بالضبط
        $2^{s_2(n)}$ مدخلة فردية؛ وتحقق من ذلك على السطرين $4$ و $5$.
  \item استنتج أن كل المدخلات الداخلية $\binom nk$ (حيث $0 < k < n$)
        زوجية إذا وفقط إذا كان $n$ قوةً للعدد $2$.
  \item برهن على أن كل قوة أولية تقسم $\binom{m+n}m$ لا تفوق
        $m + n$: أي إذا كان $p^a \mid \binom{m+n}m$ فإن $p^a \leq
        m + n$. \emph{(كم حدًّا غير معدوم يمكن أن يحوي مجموع
        السؤال 11؟)}
  \item استنتج أن $\binom{2n}n$ \omterm{def:b1:arith:divides}{يقسم}
        $\operatorname{lcm}(1, 2, \dots, 2n)$، واجمع ذلك مع
        الحاصر الأدنى $\binom{2n}n \geq \frac{4^n}{2n+1}$
        (وستبرهن عليه: فالمدخلة المركزية أكبر
        مدخلات السطر $2n$ التي عددها $2n + 1$) للحصول على
        \[
        \operatorname{lcm}(1, \dots, 2n) \geq \frac{4^n}{2n+1} :
        \]
        فالمضاعفات المشتركة للأعداد الصحيحة الأولى تنمو
        \emph{أسّيًا} --- وهي لمحة كمّية أولى عن
        وفرة الأعداد الأولية.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- توليفة ختامية.}
\begin{enumerate}[resume]
  \item جد أصغر $n$ يجعل $n!$ ينتهي بعدد $2026$ صفرًا على
        الأقل. \emph{(قدّر $Z(n) \approx n/4$، ثم
        عدّل بالصيغة المضبوطة.)}
  \item تحقق متقاطع أخير: بيّن أن $7$ \emph{لا} \omterm{def:b1:arith:divides}{يقسم}
        $\binom{100}{50}$، أولًا بكتابة $50$ في الأساس $7$ و
        التأكد من أن الجمع $50 + 50$ بلا احتفاظ، ثم
        بحساب $v_7(100!)$ و $v_7(50!)$ بصيغة
        لوجاندر.
  \item أين استعملت المسألة بالضبط: (أ) وحدانية
        التفكيك؛ (ب) تفكيك القسمة الإقليدية
        $n = p^k n_1 + n_0$؛ (ج) حجة عدّ من
        \cref{ch:b1:counting}؟ جملة واحدة لكلٍّ منها.
  \item توليفة، في فقرة قصيرة: تحوّل صيغة لوجاندر
        سؤالًا في \omterm{def:b1:arith:divides}{قابلية القسمة} إلى حساب أرقام، و
        تقرأ مبرهنة كومر الجوابَ من احتفاظات عملية جمع
        واحدة --- علّق على هذه الترجمة، وعلى تحققات
        السؤال 16، وعلى ما يوحي به حاصر السؤال 21
        عن الأعداد الأولية (\omterm{def:b1:logic:statement}{والعبارة} الكاملة، وهي مبرهنة
        الأعداد الأولية، تتجاوز هذا المجلد بكثير؛ والنظير
        الكثيرحدودي لصندوق عدّة هذا الفصل هو
        \cref{ch:b1:poly}).
\end{enumerate}
\end{problem}
