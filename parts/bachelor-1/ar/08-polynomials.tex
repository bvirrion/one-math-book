\chapter{كثيرات الحدود}\label{ch:b1:poly}

كثيرات الحدود هي دوال الجبريّ المفضلة --- إلا أنها
لا تُعامَل هنا بوصفها دوالّ، بل بوصفها عبارات صورية في
مجهول $X$، تُجمع وتُضرب بقواعد \omterm{def:b1:structures:ring}{حلقة}
\omterm{def:b1:structures:group}{تبديلية}. وتجري النظرية موازيةً على نحو لافت لما في \cref{ch:b1:arith}: قسمة إقليدية، وقاسم مشترك أكبر وعلاقات بيزو،
وعناصر لا تتجزأ ووحدانية التفكيك. وفي كل ما يلي، يرمز $K$
إلى $\Q$ أو $\R$ أو $\C$.

\section{الحلقة $K[X]$}

\begin{definition}[كثير الحدود، الدرجة]\label{def:b1:poly:def}
\emph{كثير الحدود}\index{كثير حدود} ذو المعاملات في $K$ مجموعٌ
صوريّ
\[
P = a_0 + a_1 X + a_2 X^2 + \dots + a_n X^n
= \sum_{k} a_k X^k,
\]
حيث $a_k \in K$ معدومة كلها ابتداءً من دليل ما. ومع الجمع
الطبيعي والجداء
\[
\Bigl(\sum_i a_i X^i\Bigr)\Bigl(\sum_j b_j X^j\Bigr)
= \sum_k \Bigl(\sum_{i+j=k} a_i b_j\Bigr) X^k,
\]
تكون \omterm{def:b1:logic:sets}{المجموعة} $K[X]$ حلقةً \omterm{def:b1:structures:group}{تبديلية}. و\emph{درجة} $\deg P$ لكثير الحدود
$P \neq 0$ هي أكبر $n$ يحقق $a_n \neq 0$؛ ويكون $a_n$
\emph{المعامل المهيمن} (ويكون $P$ \emph{واحديًا}\index{كثير حدود
واحدي} عندما $a_n = 1$)، ونصطلح على $\deg 0 = -\infty$.
ويعرّف كل كثير حدود دالةً $x \mapsto P(x)$ على $K$ بالتعويض.
\end{definition}

\begin{proposition}[قواعد الدرجة؛ الحلقة التامة]\label{prop:b1:poly:degree}
من أجل $P, Q \in K[X]$:
\[
\deg(P + Q) \leq \max(\deg P, \deg Q),
\qquad
\deg(PQ) = \deg P + \deg Q .
\]
ومنه فإن $K[X]$ \omterm{def:b1:structures:field}{حلقة تامة}، وعناصرها القابلة للقلب هي
الثوابت غير المعدومة.
\end{proposition}

\begin{proof}
قاعدة المجموع بيّنة (فالمعاملات بعد الأكبر تنعدم). وأمّا
الجداء، فليكن $a_m$ و $b_n$ المعاملين المهيمنين: يكون
معامل $X^{m+n}$ في $PQ$ هو $a_m b_n \neq 0$ (فإن $K$
\omterm{def:b1:structures:field}{حقل}، فهو إذن \omterm{def:b1:structures:field}{حلقة تامة})، وتنعدم كل المعاملات الأعلى.
وإذا كان $P, Q \neq 0$ فإن $\deg PQ = \deg P + \deg Q \geq 0$، ومنه $PQ \neq
0$: أي \omterm{def:b1:structures:field}{حلقة تامة}. وإذا كان $PQ = 1$ فإن $\deg P + \deg Q = 0$ يفرض
$\deg P = \deg Q = 0$: فالعناصر القابلة للقلب هي الثوابت
القابلة للقلب، أي كل $K^*$.
\end{proof}

\begin{theorem}[القسمة الإقليدية]\label{thm:b1:poly:division}
ليكن $A, B \in K[X]$ حيث $B \neq 0$. يوجد زوج واحد بالضبط $(Q,
R)$ من كثيرات الحدود يحقق
\[
A = BQ + R, \qquad \deg R < \deg B .
\]
\end{theorem}

\begin{proof}
\emph{الوجود}، بالاستقراء القوي على $\deg A$. إذا كان $\deg A < \deg
B$ فخذ $(Q, R) = (0, A)$. وإلا فاكتب $A = a X^m + \dots$ و$B =
b X^n + \dots$ مع $m \geq n$؛ فيكون \omterm{def:b1:poly:def}{لكثير الحدود} $A_1 = A - \frac ab
X^{m-n} B$ درجةٌ $< m$ (إذ تختصر الحدود المهيمنة)، ومنه
بالاستقراء $A_1 = BQ_1 + R$ مع $\deg R < \deg B$، و$A = B(Q_1 +
\frac ab X^{m-n}) + R$.

\emph{الوحدانية}: إذا كان $BQ + R = BQ' + R'$ فإن $B(Q - Q') = R' - R$
مع $\deg(R' - R) < \deg B$؛ وبقاعدة الدرجة يفرض هذا $Q - Q'
= 0$، ثم $R = R'$.
\end{proof}

\begin{example}\label{ex:b1:poly:division}
اقسم $A = X^4 + X^3 - 2X + 1$ على $B = X^2 + 1$:
\[
X^4 + X^3 - 2X + 1 = (X^2 + 1)(X^2 + X - 1) + (-3X + 2).
\]
(والحساب: اطرح $X^2 B$، ثم $X B$، ثم $-B$؛ فالباقي
$-3X + 2$ درجته $1 < 2$.)
\end{example}

\begin{method}[خوارزمية هورنر]\label{met:b1:poly:horner}
لحساب قيمة $P = a_nX^n + \dots + a_0$ عند $x$، أو لقسمة $P$ على
$X - x$، تجنّب حساب القوى: اقرأ المعاملات من اليسار إلى
اليمين، وكرّر \emph{اضرب في $x$ ثم أضف المعامل
التالي}:
\[
b_n = a_n, \qquad b_{k} = a_{k} + x\,b_{k+1}
\quad (k = n-1, \dots, 0) .
\]
عندئذ $b_0 = P(x)$، وتكون $b_k$ السابقة معاملات
خارج القسمة: $P = (X - x)(b_nX^{n-1} + \dots + b_1) + b_0$
(انشر وقارن). مثال: $P = X^4 - 5X^3 + 6X^2 + 4X - 8$
عند $x = 2$: تكون قيم $b$ هي $1, -3, 0, 4, 0$، ومنه $P(2) = 0$ و$P
= (X-2)(X^3 - 3X^2 + 4)$ --- أي سطر واحد بدل قسمة
طويلة، و $n$ عملية ضرب بدل $\approx n^2/2$
في الحساب الساذج. وتكرار الخوارزمية عند النقطة نفسها
يستخرج التضاعفات (قارن
\cref{ex:b1:poly:multexample}).
\end{method}

\begin{remark}[{حساب الحلقة $K[X]$}]
بامتلاك القسمة الإقليدية، ينتقل حساب
\cref{ch:b1:arith} كله إلى $K[X]$، بالبراهين نفسها، مع قيام الدرجة
بدور القيمة المطلقة: \omterm{thm:b1:arith:gcd}{القاسم المشترك الأكبر} (مسوًّى ليكون \omterm{def:b1:poly:def}{واحديًا})،
\omterm{met:b1:arith:euclid}{وخوارزمية إقليدس} الممدَّدة، ومتطابقة بيزو، ومبرهنة غاوس المساعدة،
وكثيرات الحدود التي لا تتجزأ ووحدانية التفكيك. ونستعمل هذه
النتائج المنقولة بحرية، ويعيد \cref{exo:b1:poly:6} إجراء إحداها.
\end{remark}

\section{الجذور}

\begin{theorem}[مبرهنة العامل]\label{thm:b1:poly:factor}
ليكن $P \in K[X]$ و $a \in K$. باقي قسمة $P$ على
$X - a$ هو الثابت $P(a)$. وبوجه خاص
\[
P(a) = 0 \iff (X - a) \mid P .
\]
وبعمومية أكبر، تعطي الجذور المتمايزة $a_1, \dots, a_r$ للعدد $P$
التعميلَ $P = (X - a_1)\cdots(X - a_r)\, Q$.
\end{theorem}

\begin{proof}
اقسم: $P = (X - a) Q + R$ مع $\deg R < 1$، فيكون $R$ ثابتًا
$c$؛ وبتعويض $X = a$ (فالتعويض يحترم المجاميع والجداءات)
نجد $P(a) = c$. ومنه ينتج التكافؤ. وأمّا من أجل عدة جذور،
فاستقرِ على $r$: فالحالة $r = 1$ هي التكافؤ المبرهن عليه للتوّ.
ولنفترض \omterm{def:b1:logic:statement}{العبارة} صحيحة من أجل $r - 1$ جذرًا وليكن $a_1, \dots, a_r$
جذورًا متمايزة للعدد $P$. اكتب $P = (X - a_1)Q_1$؛ ومن أجل كل $i
\geq 2$، بتعويض $a_i$:
\[
0 = P(a_i) = (a_i - a_1)\,Q_1(a_i),
\qquad a_i - a_1 \neq 0 ,
\]
وبما أن $K$ بلا قواسم للصفر، يكون $Q_1(a_i) = 0$: فالنقاط
المتمايزة $a_2, \dots, a_r$ التي عددها $r - 1$ جذورٌ للعدد $Q_1$. ويعمّل
فرض الاستقراء $Q_1 = (X - a_2)\cdots(X - a_r)\,Q$،
وبالتعويض من جديد نحصل على الدعوى.
\end{proof}

\begin{corollary}[لكثير حدود من الدرجة $n$ عدد جذور لا يفوق $n$]\label{cor:b1:poly:nroots}
\omterm{def:b1:poly:def}{لكثير الحدود} غير المعدوم $P \in K[X]$ من الدرجة $n$ عددُ جذور متمايزة
في $K$ لا يفوق $n$. ومنه فإن \omterm{def:b1:poly:def}{كثير الحدود} (من الدرجة $\leq n$) الذي ينعدم عند
$n + 1$ نقطة متمايزة هو \omterm{def:b1:poly:def}{كثير الحدود} المعدوم، وكثيرَي حدود
من الدرجة $\leq n$ يتوافقان عند $n+1$ نقطة متساويان.
\end{corollary}

\begin{proof}
إذا كانت $a_1, \dots, a_r$ جذورًا متمايزة فإن \cref{thm:b1:poly:factor}
يعطي $P = (X-a_1)\cdots(X-a_r) Q$، ومنه $n = \deg P \geq r$. وتنتج النتيجتان
بالخلف وبأخذ الفرق.
\end{proof}

\begin{example}[حيلة كثير الحدود المساعد]\label{ex:b1:poly:auxiliary}
ليكن $P$ \omterm{def:b1:poly:def}{كثير الحدود} من الدرجة $\leq n$ الذي يحقق
\[
P(k) = \frac{k}{k+1} \qquad (k = 0, 1, \dots, n) ;
\]
وهو موجود ووحيد \omterm{thm:b1:poly:lagrange}{بمقايسة لاغرانج} أدناه. فما
$P(n+1)$؟ أزل المقامات: \omterm{def:b1:poly:def}{فكثير الحدود} $Q = (X+1)P - X$
درجته $\leq n + 1$ وينعدم عند النقاط $n + 1$ التي هي $0, 1,
\dots, n$، ومنه حسب \cref{thm:b1:poly:factor}
\[
Q = c\,X(X-1)(X-2)\cdots(X-n)
\]
من أجل ثابت $c$ ما. واحسب حيث يكون $Q$ معلومًا باستقلال:
عند $X = -1$، $Q(-1) = 0 \cdot P(-1) + 1 = 1$، بينما يساوي الجداء
$(-1)(-2)\cdots(-1-n) = (-1)^{n+1}(n+1)!$؛ ومنه $c =
\frac{(-1)^{n+1}}{(n+1)!}$. والآن احسب عند $X = n + 1$:
\[
(n+2)\,P(n+1) - (n+1) = Q(n+1) = c\,(n+1)! = (-1)^{n+1} ,
\]
ومنه $P(n+1) = \dfrac{(n+1) + (-1)^{n+1}}{n+2}$: أي يساوي $1$ من أجل
$n$ الفرديّ، و $\frac{n}{n+2}$ من أجل $n$ الزوجيّ --- \omterm{def:b1:poly:def}{فكثير الحدود}
المقايس \emph{لا} يواصل النمط
$\frac{n+1}{n+2}$. والحيلة الجديرة بالتذكر: رمّز المعطيات في صورة
جذور \omterm{def:b1:poly:def}{لكثير حدود} مساعد، وعيّن الثابت المجهول
عند نقطة خارج المعطيات، ثم اجنِ الثمار.
\end{example}

\begin{definition}[المشتقة، التضاعف]\label{def:b1:poly:derivative}
\emph{المشتقة الصورية} \omterm{def:b1:poly:def}{لكثير الحدود} $P = \sum a_k X^k$ هي $P' = \sum_{k
\geq 1} k\,a_k X^{k-1}$؛ وهي تحقق القاعدتين المعتادتين $(P+Q)' = P' +
Q'$ و $(PQ)' = P'Q + PQ'$ (ويُتحقَّق منهما على الحدّانيات ثم تُمدَّدان
بالخطية). ويكون للجذر $a$ \omterm{def:b1:poly:def}{لكثير الحدود} $P$
\emph{تضاعف}\index{تضاعف جذر} $m \geq 1$ عندما يكون
$(X-a)^m \mid P$ و $(X-a)^{m+1} \nmid P$؛ ويكون الجذر \emph{بسيطًا}
إذا كان $m = 1$، و\emph{مضاعفًا} إذا كان $m \geq 2$.
\end{definition}

\begin{proposition}[التضاعف عبر المشتقات]\label{prop:b1:poly:multiplicity}
يكون $a$ جذرًا \omterm{def:b1:poly:def}{لكثير الحدود} $P$ تضاعفه $\geq m$ إذا وفقط إذا كان
\[
P(a) = P'(a) = \dots = P^{(m-1)}(a) = 0 .
\]
وبوجه خاص، يكون $a$ جذرًا مضاعفًا \omterm{def:b1:poly:def}{لكثير الحدود} $P$ إذا وفقط إذا كان $P(a) =
P'(a) = 0$.
\end{proposition}

\begin{proof}
اكتب $P = (X - a)^m Q + R$ حيث $R$ باقي القسمة
على $(X-a)^m$ و $\deg R < m$. وباشتقاق $k \leq m - 1$ مرة ثم
الحساب عند $a$: يسهم الحدّ الأول بالمقدار $0$ (فكل مشتقة
تحتفظ بعامل $(X-a)$)، ومنه $P^{(k)}(a) = R^{(k)}(a)$.

والآن، \omterm{def:b1:poly:def}{كثير الحدود} $R$ من الدرجة $< m$ محدَّد بالقيم $R(a), R'(a),
\dots, R^{(m-1)}(a)$: فبكتابة $R = \sum_{k < m} c_k (X - a)^k$
(وهذا ممكن: انشر قوى $X = (X - a) + a$)، نجد
$R^{(k)}(a) = k!\, c_k$. ومنه: تكون كل $P^{(k)}(a) = 0$ من أجل $k < m$
$\iff$ تكون كل $c_k = 0$ $\iff$ يكون $R = 0$ $\iff$ يكون $(X-a)^m \mid P$.
\end{proof}

\begin{example}[حساب تضاعف]\label{ex:b1:poly:multexample}
ما \omterm{def:b1:poly:derivative}{تضاعف الجذر} $2$ في $P = X^4 - 5X^3 + 6X^2
+ 4X - 8$؟ احسب المشتقات المتتالية عند $2$:
\[
P(2) = 16 - 40 + 24 + 8 - 8 = 0, \qquad
P'(2) = 32 - 60 + 24 + 4 = 0,
\]
\[
P''(2) = 48 - 60 + 12 = 0, \qquad
P'''(2) = 48 - 30 = 18 \neq 0
\]
(مع $P' = 4X^3 - 15X^2 + 12X + 4$ و$P'' = 12X^2 - 30X + 12$ و
$P''' = 24X - 30$). ثلاث قيم معدومة ثم قيمة غير معدومة:
أي \omterm{def:b1:poly:derivative}{تضاعف} يساوي $3$ بالضبط. وبالقسمة، $P = (X - 2)^3(X + 1)$ ---
ويُتحقَّق من ذلك بنشر $(X-2)^3 = X^3 - 6X^2 + 12X - 8$ و
الضرب في $X + 1$. والفكرة النافذة: تُقرأ التضاعفات من
\emph{الحسابات}، دون حاجة إلى تعميل --- وهذا بالضبط
كيف نكشفها عندما يكون التعميل بعيد المنال.
\end{example}

\begin{example}[كشف الجذور المضاعفة بقاسم مشترك أكبر]\label{ex:b1:poly:gcdmultiple}
عندما لا يُعرف أيّ جذر، تظلّ \cref{prop:b1:poly:multiplicity}
تعطي كاشفًا \emph{إجماليًا} للجذور المضاعفة: فيكون $a$ جذرًا
مضاعفًا للمقدار $P$ إذا وفقط إذا كان جذرًا مشتركًا للمقدار $P$ و $P'$، ومنه فلكثير الحدود $P$
جذر مضاعف (في $\C$) إذا وفقط إذا كان $\gcd(P, P') \neq 1$ --- وهو قابل للحساب
\omterm{met:b1:arith:euclid}{بخوارزمية إقليدس} دون حلّ أيّ شيء. مثال: $P =
X^3 - 3X + 2$ و$P' = 3X^2 - 3 = 3(X - 1)(X + 1)$. وبتجريب
جذرَي $P'$ اللذين هما $\pm1$ داخل $P$: $P(1) = 0$ لكن $P(-1) = 4$، ومنه
\[
\gcd(P, P') = X - 1 :
\]
فالجذر $1$ مضاعف؛ وبالقسمة مرتين، $P = (X - 1)^2(X + 2)$.
بل إن \omterm{thm:b1:arith:gcd}{القاسم المشترك الأكبر} يفيد \omterm{def:b1:logic:sets}{بمجموعة} الجذور المضاعفة كاملةً، كلٌّ منها
\omterm{def:b1:poly:derivative}{بتضاعف} منقوص بواحد --- وهي الواقعة التي يستثمرها كل نظام جبر
حاسوبي «للتعميل الخالي من المربعات» قبل أيّ
مطاردة للجذور، وهي التوأم الكثيرحدودي لحجج
\cref{exo:b1:poly:9} في نفي الجذور المضاعفة.
\end{example}

\begin{theorem}[المبرهنة الأساسية في الجبر]\label{thm:b1:poly:fta}
لكل \omterm{def:b1:poly:def}{كثير حدود} غير ثابت من $\C[X]$ جذرٌ في
$\C$.\index{المبرهنة الأساسية في الجبر}
\end{theorem}
\admitted

\begin{remark}
رغم اسمها، المبرهنة عبارةُ \emph{تحليل}:
فكل برهان معروف لها يستعمل تمام $\R$ في صورة ما، ولا
يكون أيٌّ منها جبريًا خالصًا --- والبرهان الأمين يُعطى في
مجلد السنة 3، بمجرد أن يتوفر التكامل العقدي أو حجج
التراص. أمّا ما يبرهن عليه هذا الفصل حقًّا فهو
\emph{الردّ}: فبمنح جذر واحد لكل \omterm{def:b1:poly:def}{كثير حدود}
غير ثابت، ينتج التعميلان الكاملان فوق $\C$ و $\R$ أدناه
بالجبر الخالص.
\end{remark}

\begin{corollary}[التعميل فوق $\C$ وفوق $\R$]\label{cor:b1:poly:factorization}
\begin{enumerate}
  \item كل $P \in \C[X]$ غير معدوم يُعمَّل هكذا
        \[
        P = c\, (X - a_1)^{m_1} \cdots (X - a_r)^{m_r},
        \]
        حيث $c$ المعامل المهيمن و $a_i$ الجذور العقدية المتمايزة
        و $\sum m_i = \deg P$: \emph{أي أن \omterm{def:b1:poly:def}{لكثير حدود} من الدرجة $n$،
        بالعدّ مع \omterm{def:b1:poly:derivative}{التضاعف}، $n$ جذرًا عقديًا بالضبط}.
  \item كل $P \in \R[X]$ غير معدوم يُعمَّل فوق $\R$ هكذا
        \[
        P = c \prod_i (X - a_i)^{m_i} \prod_j (X^2 + p_j X +
        q_j)^{n_j},
        \]
        وتكون العوامل من الدرجة الثانية متمايزة مع $p_j^2 - 4q_j < 0$
        (أي بلا جذور حقيقية).
\end{enumerate}
\end{corollary}

\begin{proof}
(1) بالاستقراء على الدرجة، بفصل جذر في كل مرة حسب
\cref{thm:b1:poly:factor}؛ ويتطابق عدّ الدرجات في كل خطوة.

(2) ليكن \omterm{def:b1:poly:def}{لكثير الحدود} $P$ معاملات حقيقية. إذا كان $z$ جذرًا عقديًا
تضاعفه $m$، فكذلك $\conj z$: إذ إن مرافقة $P(z) = 0$ تعطي
$P(\conj z) = \conj{P(z)} = 0$ (فالمعاملات مرافقات أنفسها)،
والأمر نفسه ينطبق على المشتقات
(\cref{prop:b1:poly:multiplicity}). وبتجميع الجذور غير الحقيقية في
أزواج مترافقة، يسهم كل زوج بالمقدار \[
(X - z)(X - \conj z) = X^2 - 2\Re(z)\, X + \abs z^2 ,
\]
وهو \omterm{def:b1:poly:def}{كثير حدود} حقيقي من الدرجة الثانية بمميّز سالب. وتسهم الجذور الحقيقية
بالعوامل من الدرجة الأولى.
\end{proof}

\begin{example}\label{ex:b1:poly:factorexample}
عُمِّل $X^4 + 4$ فوق $\R$ في \cref{exo:b1:complex:5} بازدواج
الجذور العقدية الأربعة $\pm 1 \pm \iu$:
$X^4 + 4 = (X^2 - 2X + 2)(X^2 + 2X + 2)$. ولا ينشقّ أيٌّ من كثيرَي الحدود من الدرجة الثانية
فوق $\R$ (فمميّزاهما $-4$). ولاحظ: أن \omterm{def:b1:poly:def}{كثير الحدود} الحقيقي
\emph{الذي لا يتجزأ} درجته $1$ أو $2$ --- وهذا بالضبط ما تقوله
مبرهنة التعميل. وبإجراء ازدواج المرافقات نفسه على
$X^4 + 1$، وجذوره $\eu^{\pm\iu\pi/4}$ و
$\eu^{\pm3\iu\pi/4}$: يسهم كل زوج بالمقدار $X^2 -
2\cos\theta\,X + 1$، ومنه
\[
X^4 + 1 = \bigl(X^2 - \sqrt2\,X + 1\bigr)
\bigl(X^2 + \sqrt2\,X + 1\bigr) ,
\]
وهي متطابقة غير مرئية لمحاولات التعميل الساذجة فوق $\Q$ ---
وهي ثمن الإصرار على معاملات حقيقية (بل صمّاء هنا)،
ومدخل معياريّ لمكاملة $\frac1{x^4 +
1}$ في \cref{ch:b1:integration}.
\end{example}

\begin{remark}[مزالق شائعة مع كثيرات الحدود]\label{rem:b1:poly:pitfalls}
\begin{enumerate}
  \item \emph{\omterm{def:b1:poly:def}{كثير الحدود} في مقابل الدالة.} فوق $K = \Q, \R, \C$
        يتطابق المفهومان (فالدالتان المتساويتان لهما المعاملات
        نفسها، حسب \cref{cor:b1:poly:nroots} ولكون $K$
        لا نهائية)، لكن \omterm{def:b1:poly:def}{كثير الحدود} مفهوميًا هو
        قائمة معاملاته: ففوق \omterm{def:b1:structures:field}{الحقل} ذي العنصرين
        $\Z/2\Z$ في \cref{ch:b1:structures}، ينعدم $X^2 + X$
        عند النقطتين، ومع ذلك ليس \omterm{def:b1:poly:def}{كثير الحدود} المعدوم.
  \item \emph{الدرجات عند الجمع.} قد تهبط $\deg(P + Q)$
        دون $\max(\deg P, \deg Q)$ عندما تختصر الحدود المهيمنة؛
        فكتابة «$\deg(P + Q) = \max(\dots)$» لا تسلم إلا من أجل
        درجتين متمايزتين.
  \item \emph{عدّ الجذور عدًّا صحيحًا.} \omterm{def:b1:logic:statement}{عبارة} «$n$ جذرًا» في
        \cref{cor:b1:poly:factorization} تعني \emph{مع
        \omterm{def:b1:poly:derivative}{التضاعف}، في $\C$}: فللمقدار $X^2 + 1$ لا جذور حقيقية، و
        للمقدار $(X-1)^2$ جذر متمايز واحد لكن جذران مع
        \omterm{def:b1:poly:derivative}{التضاعف}. والعبارات التي تخلط بين العدّات الثلاث أشيع
        مصادر البراهين الخاطئة.
  \item \emph{عدم التجزؤ يتعلق \omterm{def:b1:structures:field}{بالحقل}.} فإن $X^2 - 2$
        لا يتجزأ فوق $\Q$، وينشقّ فوق $\R$؛ و $X^2 + 1$
        لا يتجزأ فوق $\R$، وينشقّ فوق $\C$. فكلمة
        «لا يتجزأ» وحدها لا تعني شيئًا حتى يُسمّى \omterm{def:b1:structures:field}{حقل}
        المعاملات.
\end{enumerate}
\end{remark}

\section{المعاملات والجذور}

\begin{theorem}[صيغ فييت]\label{thm:b1:poly:vieta}
ليكن $P = X^n + c_{n-1} X^{n-1} + \dots + c_0$ \omterm{def:b1:poly:def}{واحديًا} وجذوره
$a_1, \dots, a_n \in \C$ (مع \omterm{def:b1:poly:derivative}{التضاعف}). عندئذ\index{صيغ
فييت}
\[
\sum_i a_i = -c_{n-1},
\qquad
\sum_{i < j} a_i a_j = c_{n-2},
\qquad \dots, \qquad
a_1 a_2 \cdots a_n = (-1)^n c_0 ,
\]
وتكون الدالة المتناظرة ذات الرتبة $k$ للجذور مساويةً $(-1)^k c_{n-k}$.
\end{theorem}

\begin{proof}
حسب \cref{cor:b1:poly:factorization}، $P = (X - a_1)\cdots(X -
a_n)$ (فهو واحديّ وكل الجذور مسرودة). ونشر الجداء
توزيعيًا يُنتج حدًّا واحدًا لكل طريقة لاختيار، في كل
عامل، إمّا $X$ وإمّا حدّ الجذر $-a_i$: فاختيار الجذور في
العوامل المفهرسة بالمقدار $i_1 < \dots < i_k$ واختيار $X$ في العوامل
$n - k$ الأخرى يسهم بالمقدار $(-a_{i_1})\cdots(-a_{i_k})\,X^{n-k}$. وبالتجميع
بحسب قوة $X$:
\[
P = \sum_{k=0}^{n} (-1)^k
\Bigl(\sum_{i_1 < \dots < i_k} a_{i_1}\cdots a_{i_k}\Bigr)
X^{n-k} ,
\]
وبالمطابقة مع $P = \sum_k c_{n-k}X^{n-k}$ (فالمعاملات
وحيدة، \cref{def:b1:poly:def}) نجد $c_{n-k} = (-1)^k
\sigma_k$، أي $\sigma_k = (-1)^kc_{n-k}$، حيث يرمز $\sigma_k$
إلى الدالة المتناظرة ذات الرتبة $k$ المعروضة أعلاه. والحالات
الثلاث المعروضة هي $k = 1$ و $k = 2$ و $k = n$.
\end{proof}

\begin{example}\label{ex:b1:poly:vieta}
من أجل المعادلة من الدرجة الثانية $X^2 - sX + p$: مجموع الجذرين $s$ وجداؤهما $p$ ---
وقد استُعمل ذلك مرارًا (\cref{exo:b1:complex:8}). ومن أجل \omterm{def:b1:poly:def}{كثير حدود} واحديّ من الدرجة الثالثة
$X^3 + aX^2 + bX + c$ جذوره $\alpha, \beta, \gamma$:
\[
\alpha + \beta + \gamma = -a,
\quad
\alpha\beta + \beta\gamma + \gamma\alpha = b,
\quad
\alpha\beta\gamma = -c ,
\]
وهذا يتيح حساب مقادير متناظرة مثل $\alpha^2 + \beta^2
+ \gamma^2 = a^2 - 2b$ دون حلّ.
\end{example}

\begin{example}[تحويل الجذور دون إيجادها]\label{ex:b1:poly:transformroots}
ليكن $\alpha, \beta$ جذرَي $X^2 - 3X + 1$. فأيّ \omterm{def:b1:poly:def}{كثير حدود} واحديّ
من الدرجة الثانية جذراه $\alpha^2, \beta^2$؟ حسب صيغ فييت، $\alpha +
\beta = 3$ و $\alpha\beta = 1$، ومنه
\[
\alpha^2 + \beta^2 = (\alpha+\beta)^2 - 2\alpha\beta = 7,
\qquad
\alpha^2\beta^2 = (\alpha\beta)^2 = 1 :
\]
فالجواب هو $X^2 - 7X + 1$ --- وقد حُصل عليه دون حساب
$\alpha = \frac{3 + \sqrt5}2$. (وللتحقق: $\alpha^2 = \frac{7 +
3\sqrt5}2$ وبالفعل $\alpha^2 + \beta^2 = 7$.) والاستراتيجية
نفسها تعالج المقلوبات (التحويلات من نمط $X^2 - \frac ba X + \frac
ca$) والانسحابات وأيّ معطيات متناظرة: فصيغ فييت
تحوّل الأسئلة عن الجذور \emph{المجهولة} إلى جبر على
المعاملات \emph{المعلومة}. وستفيد باستمرار عندما تكون
الجذور قيمًا ذاتية (\cref{ch:b1:det}).
\end{example}

\begin{example}[المعادلات الطردية العكسية]\label{ex:b1:poly:palindrome}
حُلَّ $X^4 + X^3 - 4X^2 + X + 1 = 0$. المعاملات تُقرأ
كما هي في الاتجاهين، ومنه فإن $0$ ليس جذرًا والقسمة على
$X^2$ لا تفقد أيّ حلّ:
\[
X^2 + X - 4 + \frac1X + \frac1{X^2} = 0 .
\]
ضع $y = X + \frac1X$: عندئذ $X^2 + \frac1{X^2} = y^2 - 2$، و
تنهار المعادلة إلى
\[
y^2 + y - 6 = 0 \iff (y + 3)(y - 2) = 0 .
\]
وافتح كل قيمة عبر $X^2 - yX + 1 = 0$: فمن أجل $y = 2$، تعطي $X^2 -
2X + 1 = (X - 1)^2$ الجذرَ المضاعف $1$؛ ومن أجل $y = -3$،
تعطي $X^2 + 3X + 1 = 0$ الجذرين $X = \frac{-3 \pm \sqrt5}2$. أي أربعة جذور
مع \omterm{def:b1:poly:derivative}{التضاعف} من أجل معادلة من الدرجة الرابعة، كما تقتضيه
\cref{cor:b1:poly:factorization} --- وقد حُصل عليها بحلّ
معادلتين من الدرجة الثانية. والحيلة تشمل كل \omterm{def:b1:poly:def}{كثير حدود}
\emph{طرديّ عكسيّ}: فجذوره تأتي في أزواج مقلوبة $\{x, 1/x\}$
(عوّض $X$ بالمقدار $1/X$ وأزل المقامات)، والمقدار $y = X +
\frac1X$ هو بالضبط المقدار الثابت على أزواج كهذه،
فينصّف الدرجة.
\end{example}

\begin{theorem}[مقايسة لاغرانج]\label{thm:b1:poly:lagrange}
لتكن $x_0, \dots, x_n$ نقاطًا متمايزة من $K$ ولتكن $y_0, \dots, y_n
\in K$. يوجد $P \in K[X]$ واحد بالضبط من الدرجة $\leq n$ يحقق
$P(x_i) = y_i$ لكل $i$، وهو\index{مقايسة لاغرانج}
\[
P = \sum_{i=0}^{n} y_i\, L_i,
\qquad
L_i = \prod_{j \neq i} \frac{X - x_j}{x_i - x_j} .
\]
\end{theorem}

\begin{proof}
لكل $L_i$ درجةٌ $n$ ويحقق $L_i(x_i) = 1$ و$L_i(x_j) =
0$ من أجل $j \neq i$ (إذ ينعدم كل عامل عند $x_j$ المقابل).
ومنه فإن $P$ المعروض درجته $\leq n$ ويقايس. وأمّا الوحدانية:
فكثيرا حدود مقايسين من الدرجة $\leq n$ يتوافقان عند النقاط $n+1$
التي هي $x_i$، فهما إذن متساويان (\cref{cor:b1:poly:nroots}).
\end{proof}

\begin{remark}[استراحة: كثيرات الحدود متجهات كذلك]\label{rem:b1:poly:interlude}
تغييرُ منظور سيجعله \cref{ch:b1:vspaces} رسميًا:
كثيرات الحدود من الدرجة $\leq n$ تكوّن فضاءً يسلك فيه
الجمع والضرب السلّمي سلوك الإحداثيات تمامًا ---
\omterm{def:b1:poly:def}{فكثير الحدود} \emph{هو} قائمة معاملاته $n + 1$. وثلاث
عبارات من هذا الفصل هي جبر خطيّ مستتر.
\omterm{thm:b1:poly:lagrange}{فمقايسة لاغرانج} (\cref{thm:b1:poly:lagrange}) تقول إن
معطيات الحساب $(P(x_0), \dots, P(x_n))$ تحدّد $P$
تحديدًا وحيدًا: فالحساب عند $n + 1$ نقطة تقابلٌ خطيّ،
وتكون $L_i$ الأساس المكيَّف له. والنشر $R =
\sum c_k(X - a)^k$ في برهان
\cref{prop:b1:poly:multiplicity} يقول إن قوى
$(X - a)$ تكوّن نسق إحداثيات آخر، مع $c_k =
R^{(k)}(a)/k!$ إحداثيات. و
\cref{cor:b1:poly:nroots} --- أي أن جذورًا أكثر من الدرجة تفرض
\omterm{def:b1:poly:def}{كثير الحدود} المعدوم --- هي محرك كل وحدانية: وستصير
«\omterm{def:b1:logic:map}{تطبيق} خطيّ متباين على فضاء بُعده $n + 1$»
في \cref{ch:b1:findim}. وعندما تأتي تلك الفصول، سيكون الفضاء
$K_n[X]$ مثالها المفضل؛ ويجدر بلوغها متقنًا له سلفًا.
\end{remark}

\begin{remark}[أين يُستعمل هذا الفصل]\label{rem:b1:poly:whereused}
التعميل فوق $\R$ و $\C$
(\cref{cor:b1:poly:factorization}) هو محرك الكسور
الجزئية في \cref{ch:b1:fractions}، ومنه محرك صنف واسع من
التكاملات في \cref{ch:b1:integration}. ونشر \omterm{def:b1:poly:def}{كثير
الحدود} بقوى $(X - a)$، الذي قوبل في برهان
\cref{prop:b1:poly:multiplicity}، هو الظلّ الجبري
لصيغ تايلور في \cref{ch:b1:taylor}. وقد ظهرت كثيرات الحدود المميِّزة
سلفًا من أجل المعادلات التفاضلية (\cref{ch:b1:diffeq})
وتعود من أجل المصفوفات في \cref{ch:b1:det}؛ \omterm{thm:b1:poly:lagrange}{ومقايسة
لاغرانج} هي أول مبرهنة وجود ووحدانية في
التحليل العددي، وتقول \omterm{pb:b1:poly:1}{كثيرات حدود تشيبيشيف} في
\cref{exo:b1:poly:10} --- التي ترسي مسألة نهاية الأسبوع
أدناه أمثليتها --- لذلك الاختصاص \emph{أين} يقايس.
وأخيرًا فإن حساب $K[X]$ كله، المنسوخ من
\cref{ch:b1:arith}، يغذّي دراسة مثاليات $K[X]$ وحلقات
القسمة في مجلد السنة 2.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b1:poly:1}
أجرِ القسمتين الإقليديتين: $X^5 - 1$ على $X^2 + X + 1$؛ ثم

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:poly:2}
من أجل أيّ $n \in \N$ \omterm{def:b1:arith:divides}{يقسم} $X^2 + X + 1$ المقدارَ $X^{2n} + X^n + 1$؟
\emph{إرشاد: جذرا $X^2 + X + 1$ هما $j$ و $j^2$ حيث $j =

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:poly:3}
عيّن العددين الحقيقيين $a, b$ بحيث \omterm{def:b1:arith:divides}{يقسم} $(X-1)^2$ المقدارَ $P = X^4 + aX^3 +
bX^2 + 1$، ثم عمّل $P$ فوق $\R$ من أجل هاتين القيمتين.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:poly:4}
عمّل فوق $\C$ وفوق $\R$: $X^3 - 1$؛ و $\;X^4 + X^2 + 1$؛
و $\;X^6 - 1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:5}
لتكن $P = X^3 - 6X^2 + 11X - 6$.
\begin{enumerate}
  \item جد الجذور الناطقة \emph{(فالجذر الناطق $p/q$ في
        أبسط صورة \omterm{def:b1:poly:def}{لكثير حدود} صحيح واحديّ عددٌ صحيح
        \omterm{def:b1:arith:divides}{يقسم} الحدّ الثابت --- فبرهن على ذلك)}، وعمّل $P$.
  \item دون حلّ، احسب مجموع مربعات الجذور ومجموع
        مقلوباتها بصيغ فييت، وتحقق من ذلك على
        التعميل.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:6}
احسب $\gcd(X^4 - 1,\; X^3 - X^2 + X - 1)$ \omterm{met:b1:arith:euclid}{بخوارزمية

الحدود.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:7}
ليكن $P \in \R[X]$ حيث $P(x) \geq 0$ لكل $x \in \R$. برهن على أن
$P$ مجموع مربعَي كثيرَي حدود حقيقيين: $P = A^2 + B^2$.
\emph{إرشاد: في التعميل الحقيقي، تكون تضاعفات الجذور الحقيقية
زوجية؛ فاكتب العوامل من الدرجة الثانية على الصورة $(X - z)(X - \conj z)$
واستعمل $\abs{\,\cdot\,}^2 = (\Re)^2 + (\Im)^2$ على جداء
المقادير $(X - z)$.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:8}
جد \omterm{def:b1:poly:def}{كثير الحدود} $P$ من الدرجة $\leq 2$ الذي يحقق $P(0) = 1$ و$P(1) =
3$ و $P(2) = 2$، أولًا بصيغة لاغرانج، ثم بحلّ
الجملة الخطية على المعاملات. وتحقق من \omterm{def:b1:arith:congruence}{توافق} الجوابين.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:9}
برهن على أن \omterm{def:b1:poly:def}{لكثير الحدود} $P = X^{2n+1} - 1$ جذرًا حقيقيًا واحدًا بالضبط، وعلى أنه
من أجل كل $n \geq 1$ لا يكون \omterm{def:b1:poly:def}{لكثير الحدود} $1 + X + \frac{X^2}{2!} + \dots +
\frac{X^n}{n!}$ جذر مضاعف \emph{(قارن $P$ مع
$P'$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:poly:10}
(\omterm{pb:b1:poly:1}{كثيرات حدود تشيبيشيف}) عرّف $T_0 = 1$ و $T_1 = X$ و$T_{n+1} =
2X\,T_n - T_{n-1}$.
\begin{enumerate}
  \item برهن بالاستقراء على أن $T_n(\cos\theta) = \cos n\theta$ من أجل
        كل $\theta$.
  \item استنتج جذور $T_n$ التي عددها $n$ ومعامله المهيمن.
  \item برهن على أن $\sup_{x \in \intcc{-1}{1}} \abs{T_n(x)} = 1$،
        وأنه يُبلغ عند $n + 1$ نقطة من $\intcc{-1}{1}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:poly:11}
ليكن $P \in \C[X]$ غير ثابت وجذوره المتمايزة $a_1, \dots,
a_r$ (بتضاعفات $m_1, \dots, m_r$). برهن على متطابقة
الكسور الناطقة
\[
\frac{P'(X)}{P(X)} = \sum_{i=1}^{r} \frac{m_i}{X - a_i},
\]
واستنتج مبرهنة غاوس--لوكاس: أن كل جذر للمقدار $P'$ يقع في
المحدَّب المولَّد بجذور $P$ \emph{(احسب المتطابقة عند
جذر $w$ للمقدار $P'$ ليس جذرًا للمقدار $P$، وخذ المرافقات، ثم
اقرأ النتيجة بوصف $w$ متوسطًا موزونًا للأعداد $a_i$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:12}
(مرشِّح \omterm{def:b1:complex:unity}{جذور الوحدة}) ليكن $n \in \N^*$ و $j = \eu^{2\iu\pi/3}$.
بحساب $(1 + X)^n$ عند $1$ و $j$ و $j^2$، برهن على أن
\[
\sum_{k \geq 0} \binom{n}{3k}
= \frac{2^n + 2\cos\frac{n\pi}{3}}{3} ,
\]
وتحقق من الصيغة من أجل $n = 3$ ومن أجل $n = 6$. \emph{إرشاد: $1 +
j^m + j^{2m}$ يساوي $3$ إذا كان $3 \mid m$ و $0$ فيما عدا ذلك؛ و$1 +
j = \eu^{\iu\pi/3}$.}
\end{exercise}

\section{مسألة: كثيرات حدود تشيبيشيف وأكثر كثيرات الحدود انبساطًا}

\begin{problem}\label{pb:b1:poly:1}
من بين كل كثيرات الحدود \emph{الواحدية} من الدرجة $n$، أيّها يبقى
أقرب إلى الصفر على $\intcc{-1}1$؟ الجواب --- مبرهنة تشيبيشيف،
وهي شهادة ميلاد نظرية التقريب --- هو
$2^{1-n}T_n$، حيث $T_n$ \omterm{def:b1:poly:def}{كثير حدود} تشيبيشيف في
\cref{exo:b1:poly:10}، ولا يمكن لأيّ منافس واحديّ أن يتفوق على
انحرافه $2^{1-n}$.\index{كثيرات حدود تشيبيشيف} وتطوّر هذه المسألة
جبر العائلة $(T_n)$ (\omterm{def:b1:structures:law}{قانون التركيب}، والمعاملات
الصريحة، والعائلة من النوع الثاني $U_n$، ومعادلة
تفاضلية)، وتبرهن على مبرهنة الأمثلية مع حالة
تساويها، وتجمع التطبيقات: عقد المقايسة المثلى،
والقيمة المضبوطة للمقدار $\cos 36^\circ$، \omterm{def:b1:arith:congruence}{وتوافق}
$T_p \equiv X^p \pmod p$. وفي كل ما يلي، $T_0 = 1$ و $T_1 = X$ و
$T_{n+1} = 2X\,T_n - T_{n-1}$، ونستعمل بحرية
$T_n(\cos\theta) = \cos n\theta$ من \cref{exo:b1:poly:10}.

\medskip
\noindent\textbf{الجزء 1 --- العائلة $(T_n)$.}
\begin{enumerate}
  \item احسب $T_2, T_3, T_4, T_5$ من التراجع. (وقارن
        $T_3$ بالمتطابقة $\cos3\theta = 4\cos^3\theta -
        3\cos\theta$ في \cref{ex:b1:complex:cos3}.)
  \item برهن بالاستقراء: أن $\deg T_n = n$ ومعامله المهيمن
        $2^{n-1}$ من أجل $n \geq 1$، وأن للمقدار $T_n$ زوجية
        $n$ (فلا تظهر إلا القوى الزوجية أو الفردية وحدها).
  \item برهن على مبدأ الوحدانية: أن $T_n$ هو \omterm{def:b1:poly:def}{كثير الحدود}
        \emph{الوحيد} الذي يحقق $P(\cos\theta) = \cos n\theta$ من أجل
        كل $\theta$. (فكثيرا حدود يتوافقان على $\intcc{-1}1$
        يتوافقان في كل مكان: \cref{cor:b1:poly:nroots}.)
  \item استنتج قانونَي التركيب والجداء:
        \[
        T_m \circ T_n = T_{mn},
        \qquad
        2\,T_m T_n = T_{m+n} + T_{\abs{m-n}} .
        \]
  \item استرجع من \cref{exo:b1:poly:10} الجذور $x_k =
        \cos\frac{(2k+1)\pi}{2n}$ ونقاط التذبذب المتساوي
        $y_k = \cos\frac{k\pi}n$ حيث $T_n(y_k) = (-1)^k$. واكتب
        التعميل الكامل للمقدار $T_n$ فوق $\R$، وبرّر
        أن النقاط $y_k$ تتشابك: $y_n < x_{n-1} < y_{n-1}
        < \dots < x_0 < y_0$.
  \item برهن على أن $T_n(\cosh t) = \cosh(nt)$ لكل $t \in \R$
        (بالاستقراء نفسه، باستعمال
        \cref{prop:b1:functions:hyprules})، واستنتج من أجل $x
        \geq 1$ الصيغةَ المغلقة
        \[
        T_n(x) = \frac{\bigl(x + \sqrt{x^2 - 1}\bigr)^n +
        \bigl(x - \sqrt{x^2 - 1}\bigr)^n}{2} ,
        \]
        ومنه $T_n(x) > 1$ من أجل $x > 1$: أي أن \omterm{def:b1:poly:def}{كثير الحدود} يهرب
        فورًا خارج $\intcc{-1}1$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- المعاملات، والعائلة $U_n$، ومعادلة
تفاضلية.}
\begin{enumerate}[resume]
  \item انطلاقًا من صيغة دي موافر
        (\cref{cor:b1:complex:demoivre})، برهن على \omterm{def:b1:logic:statement}{العبارة}
        الصريحة
        \[
        T_n(x) = \sum_{0 \leq 2j \leq n} \binom{n}{2j}\,
        x^{\,n-2j}\,(x^2 - 1)^j ,
        \]
        وتحقق منها من أجل $n = 3$.
  \item احسب $T_n(1)$ و $T_n(-1)$ و $T_n(0)$ من أجل كل $n$.
  \item عرّف $U_n$ (\emph{من النوع الثاني}) بالمقدار $U_0 = 1$ و$U_1 =
        2X$ و$U_{n+1} = 2X\,U_n - U_{n-1}$. برهن على أن
        $U_n(\cos\theta) = \frac{\sin(n+1)\theta}{\sin\theta}$
        من أجل $\theta \notin \pi\Z$، وعلى أن $T_n' = n\,U_{n-1}$
        من أجل $n \geq 1$.
  \item برهن على أن $\abs{\sin n\theta} \leq n\,\abs{\sin\theta}$
        لكل $\theta$ (بالاستقراء)، واستنتج الحاصر من نمط
        ماركوف
        \[
        \abs{T_n'(x)} \leq n^2
        \quad\text{على } \intcc{-1}1,
        \qquad\text{حيث } T_n'(\pm1) = (\pm1)^{n-1}\,n^2 .
        \]
  \item بيّن أن $y = T_n$ يحقق المعادلة التفاضلية
        \[
        (1 - x^2)\,y'' - x\,y' + n^2\,y = 0 ,
        \]
        وذلك باشتقاق المتطابقة $\sin\theta\,
        T_n'(\cos\theta) = n\sin n\theta$ بالنسبة إلى
        $\theta$؛ وتحقق من ذلك مباشرةً من أجل $T_2$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- مبرهنة تشيبيشيف في الأمثلية.}
لتكن $\widetilde T_n = 2^{1-n}\,T_n$ (وهي واحدية حسب السؤال 2) ولنكتب
$\norm{P}_\infty = \sup_{x \in \intcc{-1}1}\abs{P(x)}$.
\begin{enumerate}[resume]
  \item برّر أن $\norm{\widetilde T_n}_\infty = 2^{1-n}$، وأنه يُبلغ
        بإشارات متناوبة عند النقاط $n + 1$ التي هي $y_n < \dots
        < y_0$.
  \item نفترض أن \omterm{def:b1:poly:def}{كثير حدود} \omterm{def:b1:poly:def}{واحديًا} $P$ من الدرجة $n$ يحقق
        $\norm P_\infty < 2^{1-n}$، ونضع $D = \widetilde T_n -
        P$. بيّن أن $\deg D \leq n - 1$، وأن $D(y_k)$ له
        إشارة $(-1)^k$ التامة من أجل كل $k = 0, \dots, n$.
  \item استنتج أن للمقدار $D$ على الأقل $n$ جذرًا حقيقيًا متمايزًا (واحدًا
        في كل فجوة، بخاصية القيم الوسطى، المستعملة
        هنا على مستوى الثانوية والمبرهن عليها في
        \cref{ch:b1:continuity})، واستنتج \emph{مبرهنة
        تشيبيشيف}: أن كل \omterm{def:b1:poly:def}{كثير حدود} واحديّ $P$ من الدرجة $n$ يحقق
        \[
        \norm{P}_\infty \geq 2^{1-n} .
        \]
  \item (حالة التساوي، الخطوة الأولى) نفترض الآن $\norm P_\infty =
        2^{1-n}$ بالضبط، مع $P$ واحديّ من الدرجة $n$، ولتكن $D =
        \widetilde T_n - P$. بيّن أن $(-1)^kD(y_k) \geq 0$ من أجل
        كل $k$، وأنه إذا كان $D(y_k) = 0$ عند نقطة \emph{داخلية}
        $y_k$ (حيث $0 < k < n$) فإن $D'(y_k) = 0$ كذلك.
        \emph{(فعند $y_k$ داخلية، يبلغ كلٌّ من $\widetilde T_n$ و
        $P$ قيمة حدّية قيمتها المطلقة $\norm{\cdot}
        _\infty$؛ ومشتقة الدالة القابلة للاشتقاق تنعدم
        عند قيمة حدّية داخلية --- وهذا مستعمل على مستوى الثانوية،
        ومبرهن عليه في \cref{ch:b1:derivative}.)}
  \item (حالة التساوي، الخاتمة) عُدَّ جذور $D$ مع
        \omterm{def:b1:poly:derivative}{التضاعف} لتبيّن أن $D = 0$: أي أن مبلغ الصغرى
        \emph{وحيد}، $P = \widetilde T_n$.
  \item انقل ذلك إلى قطعة كيفية $\intcc ab$: بيّن أن
        أصغر معيار أعلى \omterm{def:b1:poly:def}{لكثير حدود} واحديّ من الدرجة $n$ على
        $\intcc ab$ هو $2\bigl(\frac{b-a}4\bigr)^n$، ويُبلغ \omterm{def:b1:poly:def}{بكثير حدود}
        تشيبيشيف بعد إعادة السلّم. \emph{(عوّض $x =
        \frac{a+b}2 + \frac{b-a}2\,t$ وتتبّع المعامل
        المهيمن.)}
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- تطبيقات.}
\begin{enumerate}[resume]
  \item أنجز الحالة $n = 3$ باليد: حدّد القيم الحدّية للمقدار $\widetilde T_3 = X^3 - \frac34X$ على $\intcc{-1}1$،
        وتحقق من التذبذب المتساوي الرباعي بالقيمة
        $\frac14$، واستنتج أن لا \omterm{def:b1:poly:def}{كثير حدود} واحديّ من الدرجة الثالثة يفعل أفضل.
  \item (عقد المقايسة المثلى) من أجل $n + 1$ عقدة $x_0,
        \dots, x_n \in \intcc{-1}1$، يحكم خطأَ المقايسة المقدارُ $\omega(X) = \prod_i (X - x_i)$ (كما سيكمّمه
        \cref{ch:b1:taylor}). برهن على أن الاختيار
        الذي يصغّر $\norm\omega_\infty$ هو \omterm{def:b1:logic:sets}{مجموعة} جذور $T_{n+1}$
        التي عددها $n + 1$، مع $\norm\omega_\infty = 2^{-n}$:
        فعقد تشيبيشيف هي المواضع الصحيحة للمقايسة.
  \item باستعمال $T_5$، برهن على أن $c = \cos 36^\circ$ يحقق
        $16c^5 - 20c^3 + 5c + 1 = 0$، وعمّل \omterm{def:b1:poly:def}{كثير الحدود} هذا هكذا
        $(x + 1)(4x^2 - 2x - 1)^2$، واستنتج
        \[
        \cos 36^\circ = \frac{1 + \sqrt5}4 .
        \]
        وتحقق من الاتساق مع $\cos 72^\circ =
        \frac{\sqrt5 - 1}4$ في \cref{exo:b1:complex:8}.
  \item قدّر $T_{10}(1.1)$ بالصيغة المغلقة من السؤال 6
        (ويكفي رقمان معنويان)، وفسّر: \omterm{def:b1:poly:def}{فكثير حدود}
        محدود بالمقدار $1$ على $\intcc{-1}1$ يمكن أن يتجاوز
        $40$ أصلًا عند $x = 1.1$. (وكون $T_n$ ينمو
        \emph{أسرع} من أمثاله خاصية أمثلية أخرى
        للعائلة، وهي خارج هذه المسألة.)
  \item برهن على \omterm{def:b1:arith:congruence}{التوافق} التالي: من أجل كل عدد أوليّ فرديّ $p$، تقبل كل
        معاملات $T_p - X^p$ القسمة على $p$.
        \emph{(استعمل السؤال 7 و $p \mid \binom p{2j}$ من أجل $0 <
        2j < p$، من برهان
        \cref{thm:b1:arith:fermat}.)} وتحقق من ذلك على $T_3$ و $T_5$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- توليفة ختامية.}
\begin{enumerate}[resume]
  \item احسب صراحةً \omterm{def:b1:poly:def}{كثير الحدود} الواحديّ من الدرجة الثانية ذا المعيار
        الأعلى الأصغر على $\intcc01$ وانحرافه. (السؤال 17
        مع $n = 2$.)
  \item أين استعملت المسألة بالضبط: (أ) صلابة
        كثيرات الحدود (\cref{cor:b1:poly:nroots})؛ (ب) مثلثات
        \cref{ch:b1:complex} و\cref{ch:b1:functions}؛ (ج) حساب المعاملات
        الثنائية في \cref{ch:b1:arith}؟ جملة واحدة لكلٍّ منها.
  \item توليفة، في فقرة قصيرة: تقول المبرهنة إن
        أكثر كثيرات الحدود الواحدية انبساطًا هو الذي
        \emph{يتذبذب تذبذبًا متساويًا}، ويحوّل البرهان الأمثلية
        إلى عدّ جذور. علّق على هذه الآلية، وعلى
        دور التعويض $x = \cos\theta$ جسرًا
        بين الجبر والمثلثات، وسمِّ الموضعين
        اللذين احتاجت فيهما المسألة إلى وقائع تحليلية (خاصية القيم الوسطى،
        والقيمة الحدّية الداخلية) تبرهن عليها فصول لاحقة.
\end{enumerate}
\end{problem}
