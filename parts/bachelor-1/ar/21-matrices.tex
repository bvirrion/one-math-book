\chapter{المصفوفات}\label{ch:b1:matrices}

المصفوفة تطبيقٌ خطي مكتوبٌ \omterm{prop:b1:vspaces:coordinates}{بالإحداثيات}. ويهيّئ هذا الفصل
القاموس --- فيصير التركيب جداءَ مصفوفات، وتصير التقابلية
قابليةً للقلب، ويصير \omterm{def:b1:matrices:changeofbasis}{تغيير الأساس} اقترانًا --- و
الجانب الخوارزمي: \omterm{met:b1:matrices:gauss}{العمليات على السطور}، وحساب الرتب و
المعكوسات. وقد لُقيت المصفوفات أول مرة في مجلّد الثانوي، وهي الآن
مؤسَّسة على نظرية
\cref{ch:b1:vspaces,ch:b1:findim,ch:b1:linmaps}.

\section{المصفوفات والتطبيقات الخطية}

\begin{definition}\label{def:b1:matrices:def}
$\mathcal{M}_{n,p}(K)$ هو \omterm{def:b1:vspaces:def}{الفضاء المتجهي} للجداول $n \times p$ $A
= (a_{ij})$ من السلالم ($i$: السطر، $j$: العمود)، وبُعده $np$
(\omterm{def:b1:vspaces:free}{بالأساس}: المصفوفات $E_{ij}$ ذات $1$ واحد). ومن أجل أساسين
$\mathcal{B} = (e_1, \dots, e_p)$ للمقدار $E$ و $\mathcal{C}$ للمقدار $F$
($\dim F = n$)، تكون \emph{مصفوفة $u \in \mathcal{L}(E, F)$} هي
الجدول الذي يسرد عموده $j$ \omterm{prop:b1:vspaces:coordinates}{إحداثيات} $u(e_j)$ في
$\mathcal{C}$:
\[
\operatorname{Mat}_{\mathcal{B},\mathcal{C}}(u) = (a_{ij}),
\qquad u(e_j) = \sum_{i=1}^{n} a_{ij}\, f_i .
\]
\omterm{def:b1:logic:map}{والتطبيق} $u \mapsto \operatorname{Mat}_{\mathcal{B},\mathcal{C}}(u)$
تشاكل تقابلي من $\mathcal{L}(E, F)$ على
$\mathcal{M}_{n,p}(K)$ (\cref{prop:b1:linmaps:basis}: \omterm{def:b1:linmaps:def}{فالتطبيق الخطي}
هو بالضبط اختيار صور المقادير $e_j$).
\end{definition}

\begin{example}[المشتقة، مصفوفةً]
\label{ex:b1:matrices:derivative}
لتكن $D(P) = P'$ على $\R_3[X]$. وفي \omterm{def:b1:vspaces:free}{الأساس} الوحيد الحدّ $(1, X, X^2,
X^3)$: $D(1) = 0$ و $D(X) = 1$ و $D(X^2) = 2X$ و $D(X^3) = 3X^2$، ومنه
\[
\operatorname{Mat}(D) =
\begin{pmatrix}
0 & 1 & 0 & 0\\
0 & 0 & 2 & 0\\
0 & 0 & 0 & 3\\
0 & 0 & 0 & 0
\end{pmatrix}.
\]
وفي \omterm{def:b1:vspaces:free}{الأساس} \emph{المقسوم} $\bigl(1,\ X,\ \frac{X^2}2,\
\frac{X^3}6\bigr)$، يُرسل كل متجهة \omterm{def:b1:vspaces:free}{أساس} إلى السابقة
($D\bigl(\frac{X^k}{k!}\bigr) = \frac{X^{k-1}}{(k-1)!}$)، و
تصير المصفوفة إزاحةً محضة: آحادٌ على القطر الفوقي، وأصفار
فيما عداه. ودرسان: المصفوفة تنتمي إلى \emph{الزوج}
(\omterm{def:b1:logic:map}{التطبيق}، \omterm{def:b1:vspaces:free}{الأساس})، لا إلى \omterm{def:b1:logic:map}{التطبيق} وحده؛ \omterm{def:b1:vspaces:free}{والأساس} الجيد يجعل
البنية مرئية بنظرة --- فصورة الإزاحة تبيّن فورًا
أن $D^4 = 0$ على $\R_3[X]$، إذ تدفع كل قوة للمصفوفة
قطرَ آحادها خطوةً أبعد.
\end{example}

\begin{definition}[الجداء]\label{def:b1:matrices:product}
من أجل $A \in \mathcal{M}_{n,p}$ و $B \in \mathcal{M}_{p,q}$:
\[
(AB)_{ik} = \sum_{j=1}^{p} a_{ij}\, b_{jk}
\qquad (1 \leq i \leq n,\ 1 \leq k \leq q).
\]
وهذا بالضبط مصفوفة التركيب:
$\operatorname{Mat}(v \circ u) = \operatorname{Mat}(v)\,
\operatorname{Mat}(u)$ (بتطابق الأسس في الوسط). وكذلك،
إذا كان $X$ عمود \omterm{prop:b1:vspaces:coordinates}{إحداثيات} $x$، فإن عمود $u(x)$ هو
$AX$.
\end{definition}

\begin{proof}[برهان صيغة التركيب]
\[
v(u(e_k)) = v\Bigl(\sum_j b_{jk} f_j\Bigr) = \sum_j b_{jk}\, v(f_j)
= \sum_j b_{jk} \sum_i a_{ij}\, g_i
= \sum_i \Bigl(\sum_j a_{ij} b_{jk}\Bigr) g_i . \qedhere
\]
\end{proof}

\begin{proposition}[الجبر $\mathcal{M}_n(K)$]\label{prop:b1:matrices:ring}
تكوّن المصفوفات المربعة $\mathcal{M}_n(K)$ حلقةً (غير \omterm{def:b1:structures:group}{تبديلية} من أجل $n
\geq 2$) بالعنصر المحايد $I_n$؛ \omterm{def:b1:structures:group}{وزمرة} عناصرها القابلة للقلب هي
\emph{الزمرة الخطية العامة}\index{الزمرة الخطية العامة}
$GL_n(K)$، وهي تقابل التشاكلات الذاتية التقابلية. ومن أجل $A, B \in
\mathcal{M}_n(K)$:
\[
AB = I_n \implies A \in GL_n(K) \text{ و } B = A^{-1}
\]
(والمعكوسات من جهة واحدة ثنائية الجهة، حسب
\cref{cor:b1:linmaps:samedim}).
\end{proposition}

\begin{proof}
تنتقل بديهيات \omterm{def:b1:structures:ring}{الحلقة} من $\mathcal{L}(E)$ عبر
التشاكل التقابلي في \cref{def:b1:matrices:def}: فهو يحوّل
التركيب إلى جداء والمجموع إلى مجموع، ومنه تُورَّث التجميعية
والتوزيعية ودور $I_n$ من الوقائع المقابلة عن
التطبيقات، دون أيّ تحقق مركّبةً مركّبة.
وعدم \omterm{def:b1:structures:group}{التبديلية}: $E_{12}E_{21} = E_{11} \neq E_{22}
= E_{21}E_{12}$. وإذا كان $AB = I_n$: فإن التشاكل الذاتي $a$ للمقدار $A$
يحقق $a \circ b = \mathrm{id}$، ومنه \omterm{def:b1:logic:map}{فالتطبيق} $a$ شامل (لأن $x =
a(b(x))$ يظهر \omterm{def:b1:logic:map}{صورة عكسية} لكل $x$)، ومنه تقابلي في
البُعد المنتهي (\cref{cor:b1:linmaps:samedim})؛ وبتركيب $a
\circ b = \mathrm{id}$ مع $a^{-1}$ من اليسار نجد $b =
a^{-1}$، وعندئذ $b\circ a = \mathrm{id}$ كذلك: أي إن المعكوس
من جهة واحدة كان ثنائي الجهة طوال الوقت --- وهي منّة
منتهية البُعد قطعًا.
\end{proof}

\begin{definition}[المنقولة؛ الأثر]\label{def:b1:matrices:transpose}
\emph{منقولة} $A = (a_{ij}) \in \mathcal{M}_{n,p}$ هي $A^{\mathsf T}
= (a_{ji}) \in \mathcal{M}_{p,n}$؛ وهي تحقق $(AB)^{\mathsf T} = B^{\mathsf T}
A^{\mathsf T}$ و $(A^{\mathsf T})^{\mathsf T} = A$. و
\emph{أثر}\index{أثر} مصفوفة مربعة هو $\operatorname{tr} A
= \sum_i a_{ii}$؛ وهو خطي، و
\[
\operatorname{tr}(AB) = \operatorname{tr}(BA)
\qquad (A \in \mathcal{M}_{n,p},\ B \in \mathcal{M}_{p,n}).
\]
\end{definition}

\begin{proof}[برهان متطابقة الأثر]
$\operatorname{tr}(AB) = \sum_i \sum_j a_{ij} b_{ji}$ و
$\operatorname{tr}(BA) = \sum_j \sum_i b_{ji} a_{ij}$: أي المجموع
المزدوج نفسه.
\end{proof}

\begin{example}[الأثر في العمل]
\label{ex:b1:matrices:tracework}
\omterm{def:b1:linmaps:projection}{للإسقاط} من \cref{ch:b1:linmaps} على
$\operatorname{Vect}(1,1)$ على امتداد $\operatorname{Vect}(0,1)$،
$p(x, y) = (x, x)$، المصفوفةُ $A = \begin{pmatrix} 1 & 0\\ 1 &
0\end{pmatrix}$ في \omterm{def:b1:vspaces:free}{الأساس} القانوني: فعلًا $A^2 = A$، و
\[
\operatorname{tr} A = 1 = \operatorname{rk} A ,
\]
موضّحًا \cref{exo:b1:matrices:8}: فمن أجل الجوامد \emph{يعدّ} \omterm{def:b1:matrices:transpose}{الأثر}
بُعد الصورة، مهما كان \omterm{def:b1:vspaces:free}{الأساس} المائل
الذي كُتبت فيه المصفوفة. وآلية الحفظ هي
المتطابقة $\operatorname{tr}(AB) = \operatorname{tr}(BA)$:
\[
\operatorname{tr}\bigl(P^{-1}(AP)\bigr) =
\operatorname{tr}\bigl((AP)P^{-1}\bigr) = \operatorname{tr} A ,
\]
ومنه فكل المصفوفات المشابهة للمقدار $A$ تتقاسم أثره --- وهو أول
\emph{قرينة عددية} لتشاكل ذاتي، وسينضم إليها
المحدد في \cref{ch:b1:det} (والزوج $(s, p)$ في
مسألة نهاية الأسبوع أدناه).
\end{example}

\begin{example}[متماثلة زائد متخالفة التماثل]
\label{ex:b1:matrices:symsplit}
سمِّ $A$ \emph{متماثلة} إذا كان $A^{\mathsf T} = A$، و
\emph{متخالفة التماثل} إذا كان $A^{\mathsf T} = -A$. وكل مصفوفة
مربعة تنقسم انقسامًا وحيدًا إلى واحدة زائد الأخرى:
\[
A = \underbrace{\frac{A + A^{\mathsf T}}{2}}_{\text{متماثلة}}
+ \underbrace{\frac{A - A^{\mathsf T}}{2}}_{\text{متخالفة التماثل}},
\]
والمصفوفة التي هي الاثنتان معًا معدومة (لأن $A = -A$): ومنه فالمجموعتان
فضاءان جزئيان متتامّان من $\mathcal{M}_n(K)$ --- وهو النظير
المضبوط لانقسام الدوال إلى زوجية وفردية
(\cref{ex:b1:vspaces:evenodd})، مع النقل يلعب
دور $x \mapsto -x$. والأبعاد: فالمصفوفة المتماثلة \omterm{def:b1:vspaces:free}{حرة}
على القطر وفوقه، والمتخالفة التماثل فوقه قطعًا
(بقطر معدوم):
\[
\frac{n(n+1)}{2} + \frac{n(n-1)}{2} = n^2 ,
\]
وتوازن العدّ هو تأكيد غراسمان
للانفصال. ومن أجل $n = 2$: $\begin{pmatrix} 1 & 5\\ 1 &
2\end{pmatrix} = \begin{pmatrix} 1 & 3\\ 3 & 2\end{pmatrix} +
\begin{pmatrix} 0 & 2\\ -2 & 0\end{pmatrix}$. وتعود المصفوفات
المتماثلة معطياتِ المشتقات الثانية في
\cref{ch:b1:multivar} (ثلاثي مونج $r, s, t$)، وتُصنَّف
المتماثلة المتعامدة في
\cref{exo:b1:euclid:12}.
\end{example}

\section{تغيير الأساس}

\begin{definition}\label{def:b1:matrices:changeofbasis}
ليكن $\mathcal{B}, \mathcal{B}'$ أساسين للمقدار $E$. \emph{مصفوفة تغيير
الأساس}\index{تغيير الأساس} $P =
P_{\mathcal{B}\to\mathcal{B}'}$ أعمدتها \omterm{prop:b1:vspaces:coordinates}{إحداثيات}
متجهات \omterm{def:b1:vspaces:free}{الأساس} \emph{الجديد} في \omterm{def:b1:vspaces:free}{الأساس} \emph{القديم}. وهي
قابلة للقلب، و $P^{-1} = P_{\mathcal{B}'\to\mathcal{B}}$، و
تتحول \omterm{prop:b1:vspaces:coordinates}{الإحداثيات} بالمقدار $X = PX'$ (قديم $=$ $P\,\cdot$ جديد).
\end{definition}

\begin{example}[قراءة مصفوفة تغيير الأساس]
\label{ex:b1:matrices:changerun}
في $\R^2$، من القانوني $\mathcal B$ إلى $\mathcal B' =
\bigl((1,1), (1,-1)\bigr)$:
\[
P = P_{\mathcal B\to\mathcal B'} =
\begin{pmatrix} 1 & 1\\ 1 & -1 \end{pmatrix}
\]
(بكتابة المتجهات الجديدة \omterm{prop:b1:vspaces:coordinates}{بالإحداثيات} القديمة، عمودًا عمودًا). و
لمتجهة \omterm{prop:b1:vspaces:coordinates}{الإحداثيات} القديمة $X = (3, 1)^{\mathsf T}$ إحداثياتٌ
جديدة $X' = P^{-1}X = \frac12(3 + 1,\ 3 - 1)^{\mathsf T} =
(2, 1)^{\mathsf T}$: فعلًا $2(1,1) + 1(1,-1) = (3,1)$. وانتبه
للاتجاه --- فالمصفوفة $P$ مبنية من \omterm{def:b1:vspaces:free}{الأساس} \emph{الجديد}
لكنها تحوّل \emph{من الجديد إلى القديم} ($X = PX'$)؛ وأمّا الانتقال
من القديم إلى الجديد فيكلّف المعكوس. وكتابة تحقق السلامة
$2(1,1) + (1,-1) = (3,1)$ بعد كل تحويل تمسك خطأ
قلب $P$، وهو أشيع أخطاء الفصل.
\end{example}

\begin{theorem}[تغيير الأساس من أجل تطبيق]\label{thm:b1:matrices:conjugation}
ليكن $u \in \mathcal{L}(E)$ بمصفوفة $A$ في $\mathcal{B}$ و $A'$
في $\mathcal{B}'$، ولتكن $P = P_{\mathcal{B}\to\mathcal{B}'}$. عندئذ
\[
A' = P^{-1} A\, P .
\]
والمصفوفتان المرتبطتان هكذا \emph{متشابهتان}\index{مصفوفات
متشابهة}. (ومن أجل $u \colon E \to F$ بزوجَي أسس، تكون
الصيغة $A' = Q^{-1} A P$ --- وهما مصفوفتان
\emph{متكافئتان}\index{مصفوفات متكافئة}.)
\end{theorem}

\begin{proof}
من أجل أيّ $x$: $X = PX'$ وتحقق الصورة $Y = AX$ و $Y = PY'$.
ومنه $PY' = APX'$، أي $Y' = (P^{-1}AP)X'$ من أجل كل $X'$: ومنه فمصفوفة
$u$ في \omterm{def:b1:vspaces:free}{الأساس} الجديد هي $P^{-1}AP$ (بأخذ الأعمدة القانونية
من أجل $X'$).
\end{proof}

\begin{example}[أساس جيد يجعل التطبيق شفافًا]
\label{ex:b1:matrices:conjugationrun}
لتكن $u(x, y) = (y, x)$ (وهي مبادلة)، بمصفوفة $A = \begin{pmatrix} 0
& 1\\ 1 & 0\end{pmatrix}$ في \omterm{def:b1:vspaces:free}{الأساس} القانوني. وفي \omterm{def:b1:vspaces:free}{الأساس}
$\mathcal B' = \bigl((1,1), (1,-1)\bigr)$:
\[
P = \begin{pmatrix} 1 & 1\\ 1 & -1 \end{pmatrix},
\qquad
P^{-1} = \frac12\begin{pmatrix} 1 & 1\\ 1 & -1 \end{pmatrix},
\qquad
P^{-1} A P = \begin{pmatrix} 1 & 0\\ 0 & -1 \end{pmatrix}.
\]
ولم يُحتَج فعلًا إلى أيّ جداء مصفوفات: \omterm{def:b1:logic:map}{فالتطبيق} $u$ يثبّت $(1,1)$ و
يعكس $(1,-1)$، ومنه \emph{يجب} أن تكون مصفوفته في $\mathcal B'$
$\operatorname{diag}(1, -1)$ --- فالمبادلة هي الانعكاس
عبر المستقيم $y = x$. وإيجاد \omterm{def:b1:vspaces:free}{أساس}، من أجل تشاكل ذاتي معطى،
تصير فيه مصفوفته قطرية هو المسألة المركزية
في مجلّد السنة~الثانية (نظرية التقطير)؛ وتبيّن مسألة نهاية الأسبوع
أدناه إلى أيّ مدى تذهب المتطابقات الكثيرةالحدود وحدها أصلًا.
\end{example}

\begin{example}[تغيير الأساس، مشغَّلًا بالمقلوب]
\label{ex:b1:matrices:reverserun}
\omterm{def:b1:linmaps:projection}{للإسقاط} على $F = \operatorname{Vect}(1,1)$ على امتداد $G =
\operatorname{Vect}(1,-1)$، في \omterm{def:b1:vspaces:free}{الأساس} المكيَّف $\mathcal
B' = \bigl((1,1),(1,-1)\bigr)$، المصفوفةُ الشفافة $A' =
\operatorname{diag}(1, 0)$. وللحصول على مصفوفته في \omterm{def:b1:vspaces:free}{الأساس} القانوني،
شغّل \cref{thm:b1:matrices:conjugation} بالمقلوب، $A = P A'
P^{-1}$:
\[
P = \begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix},
\quad
P^{-1} = \frac12\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix},
\quad
A = P\begin{pmatrix} 1 & 0\\ 0 & 0\end{pmatrix}P^{-1}
= \frac12\begin{pmatrix} 1 & 1\\ 1 & 1\end{pmatrix}.
\]
وللتحقق: $A^2 = A$ (فهي جامدة)، و $\operatorname{tr} A = 1 =
\operatorname{rk} A$، و $A\binom{1}{1} = \binom11$ و
$A\binom{1}{-1} = 0$، كما فُرض. وهذا الاتجاه العكسي ---
صمّم المصفوفة في \omterm{def:b1:vspaces:free}{الأساس} الجيد، ثم اقترن رجوعًا --- هو
كيف تُنتج فعلًا مصفوفات الدوران والانعكاس \omterm{def:b1:linmaps:projection}{والإسقاط} في الممارسة.
\end{example}

\begin{theorem}[الصورة الناظمية للرتبة]\label{thm:b1:matrices:rank}
\emph{رتبة} مصفوفة (وهي رتبة أعمدتها، وبكيفية مكافئة
رتبة \omterm{def:b1:linmaps:def}{التطبيق الخطي} المرتبط بها) هي القرينة الوحيدة للتكافؤ:
فكل $A \in \mathcal{M}_{n,p}$ رتبتها $r$ تكافئ
\[
J_r = \begin{pmatrix} I_r & 0 \\ 0 & 0 \end{pmatrix},
\]
و $\operatorname{rk}(A^{\mathsf T}) = \operatorname{rk}(A)$: أي إن رتبة السطور
تساوي رتبة الأعمدة.
\end{theorem}

\begin{proof}
ليكن $u \colon E \to F$ ذا رتبة $r$. اختر متتامًّا $S$ للمقدار
$\ker u$ ($\dim S = r$، \cref{thm:b1:linmaps:ranknullity}) \omterm{def:b1:vspaces:free}{بأساس}
$(e_1, \dots, e_r)$، متمَّمًا \omterm{def:b1:vspaces:free}{بأساس} للمقدار $\ker u$ ليعطي
\omterm{def:b1:vspaces:free}{أساسًا} للمقدار $E$؛ فتكوّن الصور $f_i = u(e_i)$ مع $i \leq r$ \omterm{def:b1:vspaces:free}{أساسًا} للمقدار
$\operatorname{im} u$ (لأن القصر تشاكل تقابلي)، متمَّمًا
ليعطي \omterm{def:b1:vspaces:free}{أساسًا} للمقدار $F$. وفي هذين الأساسين تكون مصفوفة $u$ هي بالضبط
$J_r$. ومنه $A = Q J_r P^{-1}$ من أجل $P, Q$ قابلتين للقلب.

وبالنقل: $A^{\mathsf T} = (P^{-1})^{\mathsf T} J_r^{\mathsf T} Q^{\mathsf T}$
مع $J_r^{\mathsf T}$ من الشكل نفسه (بالرتبة $r$) والعاملان
الخارجيان قابلان للقلب (لأن \omterm{def:b1:matrices:transpose}{منقولة} القابلة للقلب قابلة للقلب، من
$(AB)^{\mathsf T} = B^{\mathsf T}A^{\mathsf T}$ مطبَّقًا على $AA^{-1} = I$):
$\operatorname{rk} A^{\mathsf T} = r$.
\end{proof}

\section{العمليات على السطور}

\begin{method}[إزاحة غاوس على المصفوفات]\label{met:b1:matrices:gauss}
\emph{العمليات الأولية على السطور}\index{عمليات على السطور} الثلاث ---
مبادلة سطرين، وضرب سطر في $\lambda \neq 0$، وإضافة مضاعف
لسطر إلى آخر --- لا تغيّر الرتبة (لأن كلًّا منها ضربٌ
من اليسار في مصفوفة قابلة للقلب). والخوارزمية: أنشئ محورًا
(وهو أقصى مركّبة غير معدومة إلى اليسار)، ونظّف عموده تحته، وانتقل إلى السطر
والعمود التاليين؛ ويكون عدد محاور الصورة الدرجية الناتجة
هو الرتبة.

\emph{وحساب المعكوس:} شغّل الخوارزمية على الكتلة $(A \mid
I_n)$ حتى تصير الكتلة اليسرى $I_n$ (وهذا ممكن إذا وفقط إذا كانت $A$
قابلة للقلب)؛ فتكون الكتلة اليمنى عندئذ $A^{-1}$ --- إذ إن جداء
المصفوفات الأولية المستعملة يساوي $A^{-1}$.
\end{method}

\begin{example}\label{ex:b1:matrices:inverse}
$A = \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}$: قلّص $(A \mid
I_2)$:
\[
\begin{pmatrix} 1 & 2 & 1 & 0\\ 3 & 4 & 0 & 1 \end{pmatrix}
\to
\begin{pmatrix} 1 & 2 & 1 & 0\\ 0 & -2 & -3 & 1 \end{pmatrix}
\to
\begin{pmatrix} 1 & 0 & -2 & 1\\ 0 & 1 & \tfrac32 & -\tfrac12
\end{pmatrix},
\]
(والعمليات: $L_2 \leftarrow L_2 - 3L_1$؛ ثم $L_1 \leftarrow L_1
+ L_2$ و $L_2 \leftarrow -\frac12 L_2$). ومنه $A^{-1} =
\begin{pmatrix} -2 & 1 \\ \tfrac32 & -\tfrac12\end{pmatrix}$.
\emph{وللتحقق:} $AA^{-1} = I_2$.
\end{example}

\begin{example}[رتبة بوسيط، بالسطور وحدها]
\label{ex:b1:matrices:rankparameter}
من أجل $m \in \R$، رتبة
$M_m = \begin{pmatrix} 1 & 1 & m\\ 1 & m & 1\\ m & 1 &
1\end{pmatrix}$. قلّص: يعطي $L_2 \leftarrow L_2 - L_1$ و $L_3
\leftarrow L_3 - mL_1$ السطرين
\[
(1,\ 1,\ m), \qquad (0,\ m - 1,\ 1 - m), \qquad
(0,\ 1 - m,\ 1 - m^2).
\]
\emph{الحالة $m = 1$}: ينعدم السطران الأخيران --- فمحور واحد و
$\operatorname{rk} M_1 = 1$ (لأن السطور الأصلية الثلاثة كانت
متساوية). \emph{والحالة $m \neq 1$}: اضرب $L_2$ في $\frac1{m-1}$ و
$L_3$ في $\frac1{1-m}$ لتحصل على $(0, 1, -1)$ و $(0, 1, 1 + m)$،
ثم $L_3 \leftarrow L_3 - L_2 = (0, 0, m + 2)$. وإذا كان $m = -2$:
فمحوران، والرتبة $2$؛ وإلا فثلاثة محاور، والرتبة $3$. وتلخيصًا:
\[
\operatorname{rk} M_m =
\begin{cases}
1 & m = 1,\\
2 & m = -2,\\
3 & \text{فيما عدا ذلك}.
\end{cases}
\]
وسوف تسقط العتبات نفسها من حساب محدد واحد
في \cref{ch:b1:det} (وهو \omterm{def:b1:poly:def}{كثير الحدود} $-(m+2)(m-1)^2$ في
\cref{exo:b1:det:7}) --- لكن لاحظ ما تعطيه الإزاحة ولا يعطيه
المحدد: أي \emph{قيمة} الرتبة في الحالات
المنحلّة، لا مجرد كونها هبطت.
\end{example}

\begin{example}[حساب القوى]\label{ex:b1:matrices:powers}
$A = \begin{pmatrix} 1 & 1 \\ 0 & 1\end{pmatrix} = I + N$ مع $N =
E_{12}$ و $N^2 = 0$. ولأن $I$ و $N$ يتبادلان، تُقطع مبرهنة ثنائي الحدّ
(\cref{prop:b1:structures:binomial}):
\[
A^k = I + kN = \begin{pmatrix} 1 & k \\ 0 & 1 \end{pmatrix}
\qquad (k \in \N, \text{ و } k \in \Z \text{ باستعمال } A^{-1} = I -
N).
\]
\end{example}

\begin{method}[حساب $A^n$: الطرق الثلاثة]
\label{met:b1:matrices:powers}
\begin{enumerate}
  \item \emph{الطريق الثنائي}: إذا كان $A = \lambda I + N$ مع $N$
        معدومة القوى، تُقطع مبرهنة ثنائي الحدّ
        (\cref{ex:b1:matrices:powers} و \cref{exo:b1:matrices:5})؛
        وهو ينطبق لأن $\lambda I$ يتبادل مع كل شيء.
  \item \emph{الطريق الكثيرالحدود}: جد متطابقة كثيرةالحدود
        يحققها $A$ (وفي البُعد $2$، دائمًا $A^2 = sA -
        pI$) وردّ $X^n$ بترديدها؛ وتبني مسألة نهاية الأسبوع
        أدناه هذا الطريق كاملًا.
  \item \emph{طريق التشابه}: جد $P$ قابلة للقلب تحقق
        $P^{-1}AP = D$ بسيطة (قطرية أو إزاحة)، واحسب
        $D^n$، وافكك: $A^n = P D^n P^{-1}$
        (\cref{thm:b1:matrices:conjugation} و
        \cref{ex:b1:matrices:conjugationrun})؛ والبحث المنهجي
        عن مثل هذه $P$ هو نظرية التقطير في السنة~الثانية.
\end{enumerate}
ومهما كان الطريق، تحقق من النتيجة عند $n = 0, 1, 2$: أي ثلاثة
اختبارات رخيصة تمسك كل زلّة تقريبًا.
\end{method}

\begin{remark}[مزالق شائعة: ثمن عدم التبديلية]
\label{rem:b1:matrices:pitfalls}
كل متطابقة في الجبر السلّمي يعيد برهانها ترتيب العوامل
تموت في $\mathcal{M}_n(K)$ من أجل $n \geq 2$.
\emph{المربعات}: $(A + B)^2 = A^2 + AB + BA + B^2$، ولا ينهار
الوسط إلى $2AB$ إلا إذا كان $AB = BA$
(\cref{exo:b1:matrices:1}).
\emph{قوى الجداءات}: $(AB)^k$ هو $ABAB\cdots$، لا
$A^kB^k$.
\emph{قواسم الصفر}: $E_{12}E_{12} = 0$ مع $E_{12} \neq 0$؛
ومنه \emph{لا اختصار}: فالمقدار $AB = AC$ يستلزم $B = C$
فقط عندما تكون $A$ قابلة للقلب (بالضرب في $A^{-1}$ --- من
الجهة الصحيحة).
\emph{الآثار}: $\operatorname{tr}(AB) = \operatorname{tr}(BA)$
دائمًا، لكن $\operatorname{tr}(AB) \neq
\operatorname{tr}A\operatorname{tr}B$ عمومًا (خذ $A = B =
I_2$: $2 \neq 4$)، و $\operatorname{tr}(ABC) =
\operatorname{tr}(BCA)$ (دوريًا) بينما قد يختلف $\operatorname{tr}(ACB)$.
\emph{المنقولات تعكس}: $(AB)^{\mathsf T} = B^{\mathsf
T}A^{\mathsf T}$ --- ونسيان العكس أشيع
خطأ في حسابات التعامد (\cref{ch:b1:euclid}).
وعند الشكّ، اختبر أيّ متطابقة مدَّعاة على $E_{12}$ و
$E_{21}$: فأصغر زوج غير متبادل ينقض أكثر
الصيغ الخاطئة في سطر واحد.
\end{remark}

\begin{remark}[إلى أين يذهب القاموس]
\label{rem:b1:matrices:whereused}
يُستعمل قاموس المصفوفات في كل صفحة باقية من هذا
المجلّد: فالفصل \cref{ch:b1:det} يلحق بكل مصفوفة مربعة عددًا
واحدًا يفصل في القابلية للقلب، ويحلّ $AX = B$
منهجيًا؛ و \cref{ch:b1:euclid} يفرز المصفوفات
التي تحفظ الأطوال (المصفوفات المتعامدة)؛ وفي
\cref{ch:b1:multivar}، يكون السلوك من الرتبة الثانية لدالة في
متغيرين مصفوفةً متماثلة $2 \times 2$. وأمّا \omterm{def:b1:matrices:transpose}{الأثر}،
المقدَّم أعلاه عرضًا تقريبًا، فيصير قرينة قوية:
\cref{exo:b1:matrices:6,exo:b1:matrices:8} يعطيان مذاقًا أولًا،
ويبني مجلّد السنة~الثانية نظرية القيم الذاتية عليه. وتطوّر مسألة
نهاية الأسبوع حصانَ العمل الآخر: المتطابقات الكثيرةالحدود
التي تحققها مصفوفة، والتي تحوّل حساب $A^n$ إلى
علاقة تراجعية خطية ذات حدين.
\end{remark}

\begin{remark}[منظورات داخل الكتاب 3]
\label{rem:b1:matrices:perspectives}
لثلاث عائلات مصفوفات مقدَّمة هنا مواعيد لاحقة في
هذا المجلّد. إن \emph{المصفوفات المتماثلة}
(\cref{ex:b1:matrices:symsplit}) تحمل معطيات الرتبة الثانية لدوال
المتغيرين: فاختبار مونج في
\cref{ch:b1:multivar} عبارةٌ عن سلوك إشارة
مصفوفة متماثلة $2\times2$، ويُحسب محددها $rt - s^2$
بآلة \cref{ch:b1:det}. و\emph{المصفوفات
المتعامدة} ($A^{\mathsf T}A = I$) هي تقايسات
\cref{ch:b1:euclid}، حيث تكتسب \omterm{def:b1:matrices:transpose}{المنقولة} أخيرًا معناها
الهندسي: فهي الظلّ الجبري للجداء
السلّمي. و\emph{المصفوفات القابلة للقلب} تلقى اختبارها العملي
في \cref{ch:b1:det} --- عدد واحد، $\det A \neq 0$ --- فيُغلق
البحث الذي بدأه هذا الفصل بتقليص السطور. ثم يسافر \omterm{def:b1:matrices:transpose}{الأثر} و
المحدد زوجَ القرائن $(s, p)$ في
مسألة نهاية الأسبوع، حتى نظرية القيم الذاتية في
السنة~الثانية.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:matrices:1}
لتكن $A = \begin{pmatrix} 1 & 2 \\ 0 & 1 \end{pmatrix}$ و $B =
\begin{pmatrix} 0 & 1 \\ 1 & 0\end{pmatrix}$. احسب $AB$ و $BA$ و
$A^2 - B^2$ و $(A+B)(A-B)$؛ وفسّر لماذا يختلف الأخيران.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:matrices:2}
احسب رتبة
\[
M = \begin{pmatrix}
1 & 2 & 3\\
2 & 4 & 6\\
1 & 1 & 1
\end{pmatrix},
\qquad
N = \begin{pmatrix}
1 & 1 & 0 & 2\\
0 & 1 & 1 & 1\\
1 & 2 & 1 & 3
\end{pmatrix}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:matrices:3}
اقلب، بتقليص السطور، $A = \begin{pmatrix} 1 & 0 & 1\\ 2 & 1 &
1\\ 1 & 1 & 1 \end{pmatrix}$، وتحقق على جداء واحد.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:matrices:4}
اكتب مصفوفة التشاكل الذاتي $u(P) = P(X + 1)$، في \omterm{def:b1:vspaces:free}{الأساس} القانوني
للمقدار $\R_2[X]$. وفسّر، دون حساب، لماذا
هي قابلة للقلب، وأعطِ مصفوفة $u^{-1}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:5}
لتكن $A = \begin{pmatrix} 2 & 1 \\ 0 & 2\end{pmatrix}$. اكتب $A =
2I + N$، واحسب $N^2$، واستنتج $A^k$ من أجل كل $k \in \N$ بمبرهنة
ثنائي الحدّ.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:6}
برهن على أنه لا توجد مصفوفتان $A, B \in \mathcal{M}_n(K)$ (مع
$K = \R$ أو $\C$) تحققان $AB - BA = I_n$. \emph{(خذ الآثار.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:7}
تكون المصفوفة $A$ \emph{معدومة القوى} إذا كان $A^m = 0$ من أجل $m$ ما. برهن
على أن $I - A$ عندئذ قابلة للقلب، مع
\[
(I - A)^{-1} = I + A + A^2 + \dots + A^{m-1} .
\]
وتطبيقًا: اقلب $\begin{pmatrix} 1 & 2 & 3\\ 0 & 1 & 2\\ 0 & 0 &
1\end{pmatrix}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:8}
لتحقق $A \in \mathcal{M}_n(\R)$ الشرط $A^2 = A$ (فهي جامدة). برهن
على أن $\operatorname{tr} A = \operatorname{rk} A$. \emph{(فسّر
$A$ إسقاطًا واختر \omterm{def:b1:vspaces:free}{أساسًا} مكيَّفًا؛ ويقول
\cref{thm:b1:matrices:conjugation} إن \omterm{def:b1:matrices:transpose}{الأثر}
لا يتعلق \omterm{def:b1:vspaces:free}{بالأساس} لأن $\operatorname{tr}(P^{-1}MP) =
\operatorname{tr} M$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:matrices:9}
لتكن $J \in \mathcal{M}_n(\R)$ مصفوفة كل مركّباتها آحاد. احسب $J^2$،
واستنتج، من أجل $a, b \in \R$، شرط قابلية قلب $M
= aI + bJ$ مع $M^{-1}$ \emph{(ابحث عن معكوس من
الشكل نفسه $\alpha I + \beta J$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:matrices:10}
(متراجحات الرتبة) من أجل $A, B \in \mathcal{M}_n(K)$، برهن على
\[
\operatorname{rk}(A + B) \leq \operatorname{rk} A +
\operatorname{rk} B,
\qquad
\operatorname{rk}(AB) \geq \operatorname{rk} A + \operatorname{rk}
B - n .
\]
\emph{(ومن أجل الثانية --- وهي متراجحة سيلفستر --- طبّق
مبرهنة الرتبة على قصر \omterm{def:b1:logic:map}{تطبيق} $A$ على
$\operatorname{im} B$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:11}
لتكن $D = \operatorname{diag}(d_1, \dots, d_n)$ مع المقادير $d_i$
\emph{متمايزة مثنى مثنى}.
\begin{enumerate}
  \item برهن على أن المصفوفة $A$ تتبادل مع $D$ إذا وفقط إذا كانت
        $A$ قطرية. \emph{(قارن مركّبتَي $(i,j)$ في
        $AD$ و $DA$.)}
  \item استنتج \emph{مركز} $\mathcal{M}_n(K)$: فالمصفوفات
        المتبادلة مع \emph{كل} مصفوفة هي بالضبط
        المصفوفات السلّمية $\lambda I_n$. \emph{(اختبر إزاء
        $D$، ثم إزاء المصفوفات $E_{ij}$.)}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:matrices:12}
(مصفوفات الرتبة واحد) لتكن $A \in \mathcal{M}_n(K)$ مع $A \neq 0$.
\begin{enumerate}
  \item برهن على أن $\operatorname{rk} A = 1$ إذا وفقط إذا كان $A =
        CL$ من أجل عمود غير معدوم $C \in \mathcal{M}_{n,1}$ وسطر
        غير معدوم $L \in \mathcal{M}_{1,n}$.
  \item ومن أجل مثل هذه $A$، برهن على $A^2 = (\operatorname{tr} A)\,A$؛
        واستنتج أن مصفوفة الرتبة واحد تكون معدومة القوى إذا وفقط إذا
        كان أثرها معدومًا.
  \item إذا كان $\operatorname{tr} A \neq -1$، فبرهن على أن $I_n + A$
        قابلة للقلب مع
        \[
        (I_n + A)^{-1} = I_n - \frac{1}{1 + \operatorname{tr}
        A}\,A ,
        \]
        وعلى أن $I_n + A$ \emph{ليست} قابلة للقلب عندما
        $\operatorname{tr} A = -1$. \emph{(جد متجهة يقتلها
        $I_n + A$.)}
\end{enumerate}
\end{exercise}

\section{مسألة: قوى مصفوفة بالقسمة الكثيرةالحدود}

\begin{problem}\label{pb:b1:matrices:1}
حساب $A^{100}$ مركّبةً مركّبة ميؤوس منه؛ وحسابه
عبر متطابقة كثيرةالحدود يحققها $A$ ثلاثة أسطر.
وتبني هذه المسألة الطريقة من الصفر: القسمة الإقليدية
للمقدار $X^n$، والمتطابقة $A^2 - sA + pI = 0$ التي تحققها كل مصفوفة $2
\times 2$ (وهي مبرهنة كايلي--هاملتون في البُعد
$2$)، والقاموس بين قوى المصفوفات والعلاقات
التراجعية الخطية --- مع \omterm{pb:b1:matrices:1}{أعداد فيبوناتشي} مثالًا جاريًا.
\index{مبرهنة كايلي--هاملتون (البعد 2)}
\index{أعداد فيبوناتشي}\index{علاقة تراجعية خطية}

\medskip
\noindent\textbf{الجزء 1 --- حساب البواقي.} ثبّت $s, p
\in K$ و $D = X^2 - sX + p$.
\begin{enumerate}
  \item سوّغ أنه من أجل كل $n \in \N$ يوجد $Q_n
        \in K[X]$ و $(a_n, b_n) \in K^2$ وحيدة تحقق
        \[
        X^n = Q_n\,D + a_n X + b_n ,
        \]
        واحسب $(a_0, b_0)$ و $(a_1, b_1)$.
  \item بالضرب في $X$ والقسمة مرة أخرى، أثبت
        العلاقتين التراجعيتين
        \[
        a_{n+1} = s\,a_n + b_n,
        \qquad
        b_{n+1} = -p\,a_n ,
        \]
        واستنتج $a_{n+2} = s\,a_{n+1} - p\,a_n$: أي إن
        متتالية المعاملات تخضع للعلاقة التراجعية الخطية المرتبطة
        بالمقدار $D$.
  \item افترض أن للمقدار $D$ جذرين متمايزين $\lambda \neq \mu$.
        وبتقويم متطابقة القسمة، برهن على
        \[
        a_n = \frac{\lambda^n - \mu^n}{\lambda - \mu},
        \qquad
        b_n = \frac{\lambda\mu^n - \mu\lambda^n}{\lambda - \mu} .
        \]
  \item افترض $D = (X - \lambda)^2$. وباستعمال \omterm{def:b1:derivative:def}{مشتقة}
        متطابقة القسمة، برهن على $a_n = n\lambda^{n-1}$ و
        $b_n = (1 - n)\lambda^{n}$.
  \item بيّن أن تعويض مصفوفة مثبَّتة $M \in
        \mathcal{M}_k(K)$ في كثيرات الحدود يحترم المجاميع و
        الجداءات: $(PQ)(M) = P(M)\,Q(M)$. واستنتج أنه إذا كان $D(M) =
        0$، فإن
        \[
        M^n = a_n\,M + b_n\,I \qquad (n \in \N).
        \]
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- البُعد 2: \omterm{def:b1:matrices:transpose}{الأثر} وعدد
المحدد وكايلي--هاملتون.} من أجل $A = \begin{pmatrix} a & b\\ c &
d\end{pmatrix}$ ضع $s = a + d = \operatorname{tr} A$ و $p = ad
- bc$ (وهو العدد الذي سيسمّيه \cref{ch:b1:det}
المحدد).
\begin{enumerate}[resume]
  \item تحقق بحساب مباشر من \emph{متطابقة كايلي--هاملتون
        في البُعد $2$}:
        \[
        A^2 - s\,A + p\,I_2 = 0 .
        \]
  \item برهن بالنشر المباشر على أن $p$ ضربي:
        بالترميز البديهي، $p(AB) = p(A)\,p(B)$. ثم بيّن:
        أن $A$ قابلة للقلب إذا وفقط إذا كان $p \neq 0$، وعندئذ
        \[
        A^{-1} = \frac1p\,\bigl(s\,I_2 - A\bigr).
        \]
  \item لتكن $A = \begin{pmatrix} 1 & 1\\ 0 & 2\end{pmatrix}$.
        احسب $s$ و $p$ وجذور $D$، واستنتج صيغةً
        \omterm{def:b1:topology:closed}{مغلقة} من أجل $A^n$؛ وتحقق منها إزاء حساب مباشر
        للمقدار $A^2$.
  \item لتكن $A = \begin{pmatrix} 3 & 1\\ -1 & 1\end{pmatrix}$.
        بيّن أن للمقدار $D$ جذرًا مضاعفًا واحسب $A^n$؛ وتحقق
        عند $n = 2$.
  \item لتكن $F = \begin{pmatrix} 1 & 1\\ 1 & 0\end{pmatrix}$ و
        عرّف \omterm{pb:b1:matrices:1}{أعداد فيبوناتشي} بالمقدار $F_0 = 0$ و $F_1 = 1$ و
        $F_{n+2} = F_{n+1} + F_n$. برهن على
        \[
        F^n = \begin{pmatrix} F_{n+1} & F_n\\ F_n &
        F_{n-1}\end{pmatrix} \quad (n \geq 1),
        \]
        واستنتج صيغة بيني $F_n = \dfrac{\varphi^n -
        \psi^n}{\sqrt5}$ حيث $\varphi = \frac{1 + \sqrt5}2$ و
        $\psi = \frac{1 - \sqrt5}2$، واستنتج، باستعمال السؤال 7،
        متطابقة كاسيني $F_{n+1}F_{n-1} - F_n^2 = (-1)^n$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- العلاقات التراجعية الخطية، بنيويًا.}
ثبّت $s, p \in K$ مع $p \neq 0$، ولتكن $E_D$ \omterm{def:b1:logic:sets}{مجموعة}
المتتاليات التي تحقق $u_{n+2} = s\,u_{n+1} - p\,u_n$ من أجل كل $n$.
\begin{enumerate}[resume]
  \item بيّن أن $E_D$ \omterm{def:b1:vspaces:def}{فضاء متجهي} بُعده $2$
        (كيّف \cref{exo:b1:findim:10}).
  \item بيّن أن المتتالية $(a_n)$ في الجزء 1 هي عنصر
        $E_D$ ذو القيمتين الابتدائيتين $0, 1$، وأن كل $u
        \in E_D$ يحقق
        \[
        u_n = u_1\,a_n + u_0\,b_n \qquad (n \in \N),
        \]
        مع $(b_n)$ كما في الجزء 1: أي إن بواقي القسمة تحلّ
        \emph{كل} العلاقات التراجعية دفعة واحدة.
  \item إذا كان $\lambda \neq \mu$ جذرَي $D$، فبيّن أن
        $\bigl((\lambda^n), (\mu^n)\bigr)$ \omterm{def:b1:vspaces:free}{أساس} للمقدار $E_D$؛
        وإذا كان $D = (X-\lambda)^2$ مع $\lambda \neq 0$، فبيّن أن
        $\bigl((\lambda^n), (n\lambda^n)\bigr)$ \omterm{def:b1:vspaces:free}{أساس}.
  \item حُلَّ حلًا كاملًا: $u_{n+2} = u_{n+1} + 6u_n$ مع $u_0 = 1$ و
        $u_1 = 8$؛ وتحقق من الجواب على $u_2$ و $u_3$.
  \item لتكن $C = \begin{pmatrix} 0 & 1\\ -p & s\end{pmatrix}$
        (وهي \emph{المصفوفة المرافقة} للمقدار $D$). بيّن أن
        \[
        \begin{pmatrix} u_{n}\\ u_{n+1}\end{pmatrix}
        = C^n \begin{pmatrix} u_0\\ u_1\end{pmatrix}
        \quad (u \in E_D),
        \]
        وأن $\operatorname{tr} C = s$ و $p(C) = p$: أي إن
        العلاقة التراجعية والمصفوفة تحملان \omterm{def:b1:poly:def}{كثير الحدود} $D$ نفسه.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- الدرجة الثالثة.} لتكن $D_3 = X^3 -
\alpha X^2 - \beta X - \gamma$ و
\[
C_3 = \begin{pmatrix} 0 & 1 & 0\\ 0 & 0 & 1\\ \gamma & \beta &
\alpha \end{pmatrix}.
\]
\begin{enumerate}[resume]
  \item بيّن أن $D_3(C_3) = 0$. \emph{(احسب صور
        متجهات \omterm{def:b1:vspaces:free}{الأساس} القانوني تحت قوى $C_3$: فيرسل
        \omterm{def:b1:logic:map}{تطبيق} $C_3$ المقدارَ $e_1 \mapsto \dots \mapsto$ تركيبةً
        يفرضها السطر الأخير.)}
  \item بيّن أنه إذا كان للمقدار $D_3$ ثلاثة جذور متمايزة $\lambda_1,
        \lambda_2, \lambda_3$، فإن باقي $R_n$ الناتج عن قسمة $X^n$
        على $D_3$ هو \emph{مقايِس لاغرانج}
        للقيم $\lambda_i^n$ عند العقد $\lambda_i$
        (\cref{thm:b1:poly:lagrange})؛ واستنتج أن كل مركّبة
        من $C_3^{\,n}$ تركيبةٌ خطية مثبَّتة من
        $\lambda_1^n, \lambda_2^n, \lambda_3^n$.
  \item حُلَّ: $u_{n+3} = 2u_{n+2} + u_{n+1} - 2u_n$ مع $u_0 =
        0$ و $u_1 = 1$ و $u_2 = 1$. \emph{(عمّل $D_3 = (X - 1)(X
        + 1)(X - 2)$.)} وتحقق على $u_3$.
  \item احسب باقي $X^n$ بترديد $(X -
        \lambda)^3$ \emph{(بنشر تايلور للمقدار $X^n$ عند
        $\lambda$)}، واستنتج صيغةً من أجل $(\lambda I +
        N)^n$ عندما يكون $N^3 = 0$ ويتبادل $N$ مع كل ما في
        الأفق؛ وتحقق منها إزاء مبرهنة ثنائي الحدّ.
  \item بيّن أنه من أجل $D_3$ ذي الجذور المتمايزة، يكون الحلّ
        العام للعلاقة التراجعية من الرتبة $3$ هو $u_n = c_1
        \lambda_1^n + c_2\lambda_2^n + c_3\lambda_3^n$: وبرهن
        على أن المتتاليات الهندسية الثلاث تكوّن \omterm{def:b1:vspaces:free}{أساسًا} لفضاء
        الحلول. \emph{(ومن أجل الحرية، قوّم تركيبةً
        معدومة عند $n = 0, 1, 2$ وتعرّف على جملة
        مقايسة عند العقد المتمايزة
        $\lambda_i$.)}
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- أرباح فيبوناتشي، وتوليفة.}
\begin{enumerate}[resume]
  \item برهن على $F_1 + F_2 + \dots + F_n = F_{n+2} - 1$.
  \item من $F^{m+n} = F^m F^n$، استخرج صيغة الجمع
        \[
        F_{m+n} = F_{m+1}F_n + F_m F_{n-1},
        \]
        واستنتج $F_{2n} = F_n(F_{n+1} + F_{n-1})$.
  \item برهن على أن $F_n$ هو أقرب عدد صحيح إلى
        $\varphi^n/\sqrt5$ من أجل كل $n \geq 0$.
  \item لتكن $t_n = \operatorname{tr}(F^n) = F_{n+1} + F_{n-1}$
        (وهي \emph{أعداد لوكاس} $L_n$). بيّن $t_{n+2} =
        t_{n+1} + t_n$ و $t_1 = 1$ و $t_2 = 3$، وأن $L_n =
        \varphi^n + \psi^n$، واستعد $F_{2n} = F_n L_n$.
  \item توليفة، في أربع جمل: لماذا تعيش قوى مصفوفة $2
        \times 2$ في المستوي
        $\operatorname{Vect}(I, A)$ من $\mathcal{M}_2(K)$ (وأيّ
        حجة بُعد تضمن متطابقة من الدرجة الثانية، و
        أيّ متطابقة صريحة أنتجها الجزء 2)؛ وكيف تحوّل القسمة
        الإقليدية الأسّ إلى علاقة تراجعية ذات
        حدين؛ وأيّ \omterm{def:b1:logic:statement}{عبارة} في هذه المسألة هي الحالة $n =
        2$ من مبرهنة صحيحة في كل الأبعاد (سمِّها،
        وقل أين يُبرهن عليها في هذه السلسلة)؛ وماذا
        يضيف بناء المصفوفة المرافقة إلى الصورة.
\end{enumerate}
\end{problem}
