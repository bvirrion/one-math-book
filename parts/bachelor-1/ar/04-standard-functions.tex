\chapter{الدوال المعيارية}\label{ch:b1:functions}

لا يفيد التحليل إلا بقدر مخزون الدوال التي يتقنها المرء. وإلى
\omterm{def:b1:logic:sets}{المجموعة} الموروثة عن الثانوية --- القوى والدالة الأسّية
واللوغاريتم والدوال المثلثية --- يضيف هذا الفصل دوالَّها العكسية
($\arcsin$ و $\arccos$ و $\arctan$) والعائلة الزائدية.
وتُستعمل المشتقات بحرية على مستوى الثانوية؛ أمّا النظرية
التي وراء الدوال العكسية فتُستكمل في
\cref{ch:b1:continuity,ch:b1:derivative}.

\section{الدالة الأسّية واللوغاريتم والقوى}

\begin{proposition}[تذكير وتوصيف]\label{prop:b1:functions:expln}
التطبيقان $\exp \colon \R \to \intoo{0}{+\infty}$ و$\ln \colon
\intoo{0}{+\infty} \to \R$ تقابلان أحدهما مقلوب الآخر، متزايدان
تمامًا، مع
\[
\exp(x + y) = \exp x \exp y, \qquad \ln(xy) = \ln x + \ln y,
\qquad \exp' = \exp, \qquad \ln'(x) = \frac 1x .
\]
وزيادةً على ذلك، فإن $\exp$ هي الدالة القابلة للاشتقاق
\emph{الوحيدة} $f \colon
\R \to \R$ التي تحقق $f' = f$ و $f(0) = 1$.
\end{proposition}

\begin{proof}
التذكير مادة ثانوية. وأمّا الوحدانية، فلتكن $f' = f$ و
$f(0) = 1$، ولنضع $g(x) = f(x)\,\eu^{-x}$. عندئذ $g' = f'\eu^{-x} -
f\eu^{-x} = 0$، فتكون $g$ ثابتة تساوي $g(0) = 1$: أي $f = \exp$.
\end{proof}

\begin{definition}[القوى العامة]\label{def:b1:functions:powers}
من أجل $x > 0$ و $\alpha \in \R$:
$\;x^\alpha = \eu^{\alpha \ln x}$\index{دالة القوة}. ومن أجل $a > 0$
حيث $a \neq 1$، يكون \emph{اللوغاريتم ذو الأساس $a$} هو $\log_a x =
\frac{\ln x}{\ln a}$، وهو مقلوب $x \mapsto a^x$.
\end{definition}

\begin{example}[حلّ المعادلات الأسّية]\label{ex:b1:functions:expequations}
حُلَّ $2^x = 5^{\,x-1}$ في $\R$. الطرفان موجبان، فخذ
اللوغاريتمات --- وهي خطوة قابلة للعكس:
\[
x\ln2 = (x - 1)\ln5
\iff x(\ln2 - \ln5) = -\ln5
\iff x = \frac{\ln5}{\ln5 - \ln2} = \frac{\ln 5}{\ln\frac52}
\approx 1.756 .
\]
ثم حُلَّ $x^{\sqrt2} = 3$ من أجل $x > 0$: ارفع إلى القوة
$\frac1{\sqrt2}$ (أي طبّق التقابل المقلوب):
$x = 3^{1/\sqrt2} = \eu^{(\ln 3)/\sqrt2} \approx 2.175$. والفكرة
النافذة: كل معادلة تخلط القوى تنفكّ عبر $\ln$ و
$\exp$، لأن التعريف $x^\alpha = \eu^{\alpha\ln x}$
يردّ كل تلاعب بالقوى إلى حساب الأسس ---
لكن على \omterm{def:b1:logic:sets}{مجموعة} التعريف $x > 0$ وحدها حيث يحيا ذلك التعريف.
\end{example}

\begin{example}[أزمنة التضاعف]\label{ex:b1:functions:doubling}
مقدار ينمو بنسبة $3\%$ في كل خطوة: فبعد $n$ خطوة يكون قد ضُرب
في $(1.03)^n$. فمتى يتضاعف؟ حُلَّ $(1.03)^n
\geq 2$:
\[
n\ln(1.03) \geq \ln2
\iff n \geq \frac{\ln 2}{\ln 1.03}
= \frac{0.6931}{0.02956} \approx 23.4 ,
\]
فيقع التضاعف الأول عند الخطوة $24$. (و«قاعدة $72$» عند
الماليين، التي تقدّر زمن التضاعف بقسمة
$72$ على النسبة المئوية، هي هذا الحساب مع
التقريب $\ln(1 + x) \approx x$، المكمَّم في
\cref{ch:b1:taylor}.) وأحسن ما يُعقل به عن
السيرورات الأسّية هو لوغاريتماتها: فعلى ذلك السلّم يكون
النموّ خطيًا وتصير الأسئلة قسمات.

\end{example}

\begin{example}[كم طول العدد $2^{2026}$؟]\label{ex:b1:functions:digits}
عدد الأرقام العشرية لعدد صحيح $N \geq 1$ هو
$\floor{\log_{10} N} + 1$ (فللعدد $N$ بالفعل $d$ رقمًا تحديدًا عندما
$10^{d-1} \leq N < 10^d$، أي $d - 1 \leq \log_{10}N < d$). ومن أجل
$N = 2^{2026}$:
\[
\log_{10} 2^{2026} = 2026\,\log_{10}2
= 2026 \times 0.301030 = 609.887 ,
\]
ومنه فللعدد $2^{2026}$ من الأرقام $610$. ويحمل الجزء الكسري
مكافأةً: $10^{0.887} \approx 7.7$، فالعدد \emph{يبدأ} بالرقم
$7$. فقد أجابت عملية ضرب واحدة عن سؤال يخصّ عددًا لن يكتبه
أحد كاملًا --- فاللوغاريتمات تضغط الحجم الجدائي
في حجم جمعيّ، وهذا كل مغزاها التاريخي
(و\cref{ex:b1:structures:expmorphism} يجعل هذه الجملة دقيقة).
\end{example}

\begin{proposition}[قواعد القوى ومقارنة النموّ]\label{prop:b1:functions:powerrules}
من أجل $x, y > 0$ و $\alpha, \beta \in \R$:
\[
x^{\alpha+\beta} = x^\alpha x^\beta, \quad
(x^\alpha)^\beta = x^{\alpha\beta}, \quad
(xy)^\alpha = x^\alpha y^\alpha, \quad
(x^\alpha)' = \alpha\, x^{\alpha - 1}.
\]
وسلّم النموّ\index{مقارنة النمو} عندما $x \to +\infty$، من أجل كل
$\alpha > 0$ و $\beta > 0$:
\[
\frac{(\ln x)^\beta}{x^\alpha} \longrightarrow 0,
\qquad
\frac{x^\alpha}{\eu^{\beta x}} \longrightarrow 0 .
\]
\end{proposition}

\begin{proof}
تنسخ المتطابقات نظيراتها الخاصة بالدالتين $\exp$ و $\ln$ عبر
التعريف. وبالتفصيل من أجل الثانية (وهي أقلّها وضوحًا): $x^\alpha
> 0$ و$\ln(x^\alpha) = \alpha\ln x$ (طبّق $\ln$ على
التعريف)، ومنه
\[
(x^\alpha)^\beta = \eu^{\beta\ln(x^\alpha)}
= \eu^{\beta\alpha\ln x} = x^{\alpha\beta} ;
\]
والأولى والثالثة هما الفكّ نفسه في سطرين عبر
$\exp(u + v) = \exp u\exp v$ و$\ln(xy) = \ln x + \ln y$. والمشتقة
تأتي من قاعدة السلسلة: $(\eu^{\alpha\ln x})' =
\frac{\alpha}{x} \eu^{\alpha\ln x}$.

وأمّا المقارنات: فمن $\frac{\ln t}{t} \to 0$ (المبرهن عليها في
مجلد الثانوية)، عوّض $t = x^{\alpha/\beta}$:
$\frac{\ln x}{x^{\alpha/\beta}} \to 0$، ثم ارفع إلى القوة
$\beta$. وأمّا الثانية فعوّض $x = \ln u$ في الأولى.
\end{proof}

\begin{example}[نهايتان ينبغي لكل قارئ أن يمتلكهما]\label{ex:b1:functions:xx}
ما $\lim_{x \to 0^+} x^x$ و$\lim_{x \to +\infty}
x^{1/x}$؟ كلتا القوتين معرَّفة عبر الدالة الأسّية، فاحسم
الأس أولًا:
\[
x^x = \eu^{x\ln x}, \qquad
x\ln x = -\frac{\ln(1/x)}{1/x} \longrightarrow 0
\quad (x \to 0^+),
\qquad\text{ومنه } x^x \longrightarrow \eu^0 = 1 ;
\]
و$x^{1/x} = \eu^{(\ln x)/x} \to \eu^0 = 1$ عندما $x \to +\infty$،
مباشرةً بمقارنة النموّ. والفكرة النافذة: القوة غير المعيَّنة
(من الشكل $0^0$ أو $\infty^0$) تُعالَج \emph{دائمًا} بإعادة
كتابة $u^v = \eu^{v\ln u}$ وتحليل الجداء $v \ln u$
--- لا بالتخمين من الأساس والأس كلٍّ على حدة. والدالة
$x \mapsto \frac{\ln x}{x}$ التي حسمت النهايتين
تُدرس دراسةً مستوفاة في مسألة نهاية الأسبوع من هذا الفصل.
\end{example}

\begin{remark}[مزالق شائعة مع القوى واللوغاريتمات]\label{rem:b1:functions:pitfalls}
\begin{enumerate}
  \item \emph{مجموعات التعريف.} المقدار $x^\alpha$ من أجل $\alpha$
        أصمّ يقتضي $x > 0$؛ ويحسن تجنّب $(-8)^{1/3}$ لصالح «الجذر
        التكعيبي الحقيقي للعدد $-8$»، لأن القاعدة
        $(x^\alpha)^\beta = x^{\alpha\beta}$ تخفق ضمنًا على
        الأعداد السالبة: $\bigl((-8)^2\bigr)^{1/6} = 2 \neq -2$.
  \item \emph{$\sqrt{x^2} = \abs x$}، لا $x$: ونسيان
        القيمة المطلقة هو المصدر الكلاسيكي لضياع الحلول السالبة.
  \item \emph{القاعدة $\ln(xy) = \ln x + \ln y$ تقتضي $x, y > 0$.} ففي حالة
        الوسائط السالبة قد يكون $\ln(xy)$ معرَّفًا بينما لا يكون
        الطرف الأيمن كذلك.
  \item \emph{أيّ الدالتين تنمو؟} في $x^\alpha$ يتغير
        \emph{الأساس}؛ وفي $a^x$ يتغير \emph{الأس}.
        ونموّاهما مختلفان اختلافًا هائلًا
        (\cref{prop:b1:functions:powerrules})، ويجب إعادة كتابة
        العبارات المهجّنة مثل $x^{\ln x}$ أو $x^{1/x}$ في صورة
        $\eu^{(\cdot)}$ قبل أيّ استدلال ---
        كما في \cref{ex:b1:functions:xx}.
\end{enumerate}
\end{remark}

\section{الدوال المثلثية العكسية}

\begin{definition}[$\arcsin$ و $\arccos$ و $\arctan$]\label{def:b1:functions:arc}
القصور
\[
\sin \colon \intcc{-\tfrac\pi2}{\tfrac\pi2} \to \intcc{-1}{1},
\qquad
\cos \colon \intcc{0}{\pi} \to \intcc{-1}{1},
\qquad
\tan \colon \intoo{-\tfrac\pi2}{\tfrac\pi2} \to \R
\]
تقابلات رتيبة تمامًا. وتُكتب تطبيقاتها العكسية
$\arcsin$\index{قوس الجيب} و $\arccos$\index{قوس جيب التمام} و
$\arctan$\index{قوس الظل}. وهكذا فإن $y = \arcsin x$
(حيث $x \in \intcc{-1}{1}$) هو مثلًا \emph{الزاوية} الوحيدة في
$\intcc{-\frac\pi2}{\frac\pi2}$ التي جيبها $x$.
\end{definition}

\begin{proposition}[المشتقات]\label{prop:b1:functions:arcderiv}
على داخل مجموعات تعريفها:
\[
\arcsin' x = \frac{1}{\sqrt{1 - x^2}},
\qquad
\arccos' x = \frac{-1}{\sqrt{1 - x^2}},
\qquad
\arctan' x = \frac{1}{1 + x^2}.
\]
\end{proposition}

\begin{proof}
باستباق قاعدة الدالة العكسية المبرهن عليها في
\cref{ch:b1:derivative}: إذا كان $f$ تقابلًا قابلًا للاشتقاق و
$f' \neq 0$ فإن $(f^{-1})'(x) = \frac{1}{f'(f^{-1}(x))}$. ومن أجل
$\arcsin$: مع $y = \arcsin x$،
\[
\arcsin' x = \frac{1}{\cos y}
= \frac{1}{\sqrt{1 - \sin^2 y}} = \frac{1}{\sqrt{1 - x^2}},
\]
حيث $\cos y = +\sqrt{1 - \sin^2 y}$ لأن $y \in
\intoo{-\frac\pi2}{\frac\pi2}$ يفرض $\cos y > 0$. وبالمثل
$\arccos' x = \frac{1}{-\sin y}$ مع $\sin y > 0$ على
$\intoo{0}{\pi}$، و$\arctan' x = \frac{1}{1 + \tan^2 y} =
\frac{1}{1 + x^2}$ باستعمال $\tan' = 1 + \tan^2$.
\end{proof}

\begin{example}[التعرف على ثابت مستتر]\label{ex:b1:functions:hiddenconst}
ادرس $g(x) = \arctan\dfrac{1 - x}{1 + x}$ على $\intoo{-1}
{+\infty}$. قاعدة السلسلة وحساب قصير يعطيان:
\[
g'(x) = \frac{1}{1 + \bigl(\frac{1-x}{1+x}\bigr)^2}\cdot
\frac{-(1+x) - (1-x)}{(1+x)^2}
= \frac{-2}{(1+x)^2 + (1-x)^2}
= \frac{-1}{1 + x^2} ,
\]
لأن $(1+x)^2 + (1-x)^2 = 2 + 2x^2$. إذن $g' = -\arctan'$: أي أن
الدالة $g + \arctan$ لها مشتقة معدومة على \emph{المجال}
$\intoo{-1}{+\infty}$، فهي ثابتة هناك؛ وقيمتها عند $x =
0$ هي $\arctan 1 + \arctan 0 = \frac\pi4$. والخلاصة:
\[
\arctan\frac{1 - x}{1 + x} = \frac\pi4 - \arctan x
\qquad (x > -1) .
\]
وعلى $\intoo{-\infty}{-1}$ يبقى حساب المشتقة نفسه صالحًا لكن
الثابت مختلف ($-\frac{3\pi}4$: احسب النهاية عندما
$x \to -\infty$). والفكرة النافذة: \omterm{def:b1:logic:statement}{عبارة} «المشتقة المعدومة تستلزم
الثبات» عبارةٌ تصحّ مجالًا مجالًا --- وهي بالضبط
الدقة المستثمَرة في \cref{exo:b1:functions:6} و
\cref{exo:b1:functions:10}.
\end{example}

\begin{proposition}[متطابقات معيارية]\label{prop:b1:functions:arcidentities}
\begin{enumerate}
  \item من أجل $x \in \intcc{-1}{1}$: $\arcsin x + \arccos x =
        \dfrac{\pi}{2}$.
  \item من أجل $x > 0$: $\arctan x + \arctan\dfrac 1x = \dfrac{\pi}{2}$
        (والقيمة $-\frac\pi2$ من أجل $x < 0$).
  \item $\sin(\arccos x) = \cos(\arcsin x) = \sqrt{1 - x^2}$؛
        و$\;\tan(\arcsin x) = \dfrac{x}{\sqrt{1 - x^2}}$ من أجل
        $\abs x < 1$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) مشتقة $x \mapsto \arcsin x + \arccos x$ معدومة على
$\intoo{-1}{1}$ (\cref{prop:b1:functions:arcderiv})، فالدالة
ثابتة هناك، وتساوي قيمتها $\frac\pi2$ عند $0$؛ وتُتحقَّق قيمتا
الطرفين $\pm 1$ مباشرةً ($\frac\pi2 + 0$ و$-\frac\pi2 +
\pi$).

(2) الطريقة نفسها على $\intoo{0}{+\infty}$: المشتقة هي
$\frac{1}{1+x^2} + \frac{-1/x^2}{1 + 1/x^2} = \frac{1}{1+x^2} -
\frac{1}{x^2 + 1} = 0$، وعند $x = 1$ يكون المجموع $2\arctan 1 =
\frac\pi2$. ومن أجل $x < 0$ استعمل فردية $\arctan$.

(3) مع $y = \arccos x \in \intcc{0}{\pi}$: لدينا $\sin y \geq 0$، ومنه
$\sin y = \sqrt{1 - \cos^2 y} = \sqrt{1 - x^2}$؛ وكذلك من أجل
$\cos(\arcsin x)$، وصيغة الظل هي القسمة.
\end{proof}

\begin{remark}
تصحّ $\arcsin(\sin\theta) = \theta$ \emph{وحدها} من أجل $\theta \in
\intcc{-\frac\pi2}{\frac\pi2}$: فمثلًا
$\arcsin(\sin\pi) = 0 \neq \pi$. أمّا التركيب في الاتجاه
الآخر، $\sin(\arcsin x) = x$، فصالح على $\intcc{-1}{1}$ كلها.
\end{remark}

\begin{example}[العودة إلى المجال الرئيس]
\label{ex:b1:functions:wrap}
احسب
\[
\arctan\Bigl(\tan\frac{3\pi}4\Bigr)
\qquad\text{و}\qquad
\arctan\Bigl(\tan\frac{17\pi}5\Bigr).
\]
والوصفة: استبدل بالزاوية الزاويةَ الوحيدة من $\intoo{-\frac\pi2}{\frac\pi2}$ التي لها الظل نفسه، أي اطرح المضاعف المناسب للعدد $\pi$
(وهو دور $\tan$). أولًا: $\frac{3\pi}4 - \pi =
-\frac\pi4$، فيكون الجواب $-\frac\pi4$. وثانيًا: $\frac{17\pi}
5 - 3\pi = \frac{2\pi}5 \in \intoo{-\frac\pi2}{\frac\pi2}$، فيكون
الجواب $\frac{2\pi}5$. والحساب قسمةٌ إقليدية للزاوية
على $\pi$ متنكرةً --- والوصفتان المماثلتان للدالة $\arcsin$ (بالانعكاس
إلى $\intcc{-\frac\pi2}{\frac\pi2}$، والدور $2\pi$) وللدالة $\arccos$
(بالانعكاس إلى $\intcc0\pi$) تحرّكان \cref{exo:b1:functions:1} و
الجوابَ المتشعّب في \cref{exo:b1:functions:10}.
\end{example}

\begin{example}[جمع أقواس الظل بأمان]\label{ex:b1:functions:arctansum}
لنبرهن على أن
\[
\arctan\frac12 + \arctan\frac15 + \arctan\frac18 = \frac\pi4 .
\]
والمكوّنان اثنان: صيغة جمع الظل، و --- وهي الخطوة
التي ينساها المبتدئون --- \emph{تحديد موضع} المجموع. أولًا،
تعطي $\tan(u + v) = \frac{\tan u + \tan v}{1 - \tan u\tan v}$ مع
$u = \arctan\frac12$ و $v = \arctan\frac15$
\[
\tan(u + v) = \frac{\frac12 + \frac15}{1 - \frac1{10}}
= \frac{7/10}{9/10} = \frac79,
\qquad\text{ثم}\qquad
\tan\Bigl(u + v + \arctan\frac18\Bigr)
= \frac{\frac79 + \frac18}{1 - \frac7{72}}
= \frac{65/72}{65/72} = 1 .
\]
وثانيًا، تقع كل زاوية من الزوايا الثلاث في $\intoo0{\frac\pi4}$
(فوسائطها أصغر من $1$)، فيقع المجموع في
$\intoo0{\frac{3\pi}4}$؛ والزاوية الوحيدة هناك التي ظلها $1$ هي
$\frac\pi4$. ودون تحديد الموضع لكان الاستنتاج
«$\frac\pi4$ بمضاعف للعدد $\pi$» --- أي نصف برهان. وهذا
الانضباط ذو الخطوتين نفسه يشغّل \cref{exo:b1:functions:8} و
\cref{exo:b1:functions:12}.
\end{example}

\begin{example}[معادلة بقوس الظل]\label{ex:b1:functions:arctaneq}
حُلَّ $\arctan x + \arctan 2x = \dfrac\pi4$. أولًا حدّد الموضع: إشارة
الطرف الأيسر هي إشارة $x$ (فكلا الحدّين كذلك)، فأيّ حلّ يحقق
$x > 0$، وعندئذ يقع المجموع في $\intoo0\pi$. خذ الظلال
(وهي غير متباينة على أيّ مجال طوله $\pi$، لكنها مقرونةً
بتحديد الموضع ستكفي): تعطي صيغة الجمع
\[
\tan\bigl(\arctan x + \arctan 2x\bigr)
= \frac{3x}{1 - 2x^2} = 1
\iff 2x^2 + 3x - 1 = 0
\iff x = \frac{-3 \pm \sqrt{17}}4 .
\]
ويُستبعد الجذر السالب بتحديد الموضع؛ وأمّا المرشح
الموجب $x_0 = \frac{\sqrt{17} - 3}4 \approx 0.28$،
فيقع المجموع في $\intoo0\pi$ وظله $1$، والزاوية الوحيدة
كذلك هي $\frac\pi4$ (ففي $\intoo{\frac\pi2}\pi$ يكون الظل
سالبًا): إذن $x_0$ هو الحل الوحيد. ولاحظ شكل
الحجة: أخذ الظلال قد يخلق حلولًا ولا يفقدها أبدًا،
فيُحلّ المعادلةَ الكثيرةَ الحدود \emph{ثم} يُصفّى الحلول
بتحديد الموضع --- وهو انضباط التحقق الأمامي نفسه المتبع عند
تربيع معادلة.
\end{example}

\section{الدوال الزائدية}

\begin{definition}[$\cosh$ و $\sinh$ و $\tanh$]\label{def:b1:functions:hyperbolic}
من أجل $x \in \R$:
\[
\cosh x = \frac{\eu^x + \eu^{-x}}{2},
\qquad
\sinh x = \frac{\eu^x - \eu^{-x}}{2},
\qquad
\tanh x = \frac{\sinh x}{\cosh x}
\]
(\emph{جيب التمام الزائدي والجيب الزائدي والظل الزائدي}\index{دوال
زائدية}). والدالة $\cosh$ زوجية، والدالتان $\sinh$ و $\tanh$
فرديتان.
\end{definition}

\begin{proposition}[خصائص أساسية]\label{prop:b1:functions:hyprules}
لكل $x, y \in \R$:
\begin{enumerate}
  \item $\cosh^2 x - \sinh^2 x = 1$؛
  \item $\cosh' = \sinh$ و $\;\sinh' = \cosh$ و$\;\tanh' = 1 - \tanh^2
        = \dfrac{1}{\cosh^2}$؛
  \item $\cosh(x+y) = \cosh x\cosh y + \sinh x \sinh y$ و
        $\sinh(x+y) = \sinh x \cosh y + \cosh x \sinh y$؛
  \item الدالة $\sinh$ تقابل متزايد تمامًا من $\R$ على $\R$؛
        والدالة $\cosh$ مقصورةً على $\R_+$ تقابل متزايد تمامًا
        على $\intco{1}{+\infty}$؛ والدالة $\tanh$ تقابل متزايد
        تمامًا من $\R$ على $\intoo{-1}{1}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) $\bigl(\frac{\eu^x + \eu^{-x}}{2}\bigr)^2 - \bigl(\frac{\eu^x -
\eu^{-x}}{2}\bigr)^2 = \frac{(\eu^{2x} + 2 + \eu^{-2x}) - (\eu^{2x} -
2 + \eu^{-2x})}{4} = 1$.

(2) اشتقّ الصيغ المعرِّفة؛ وأمّا $\tanh$ فتعطي قاعدة القسمة
$\frac{\cosh^2 - \sinh^2}{\cosh^2}$، وهو $1 - \tanh^2$ و
$\frac{1}{\cosh^2}$ معًا حسب (1).

(3) انشر الطرفين الأيمنين باستعمال التعاريف. وبالتفصيل
من أجل الصيغة الأولى:
\[
\cosh x\cosh y + \sinh x\sinh y
= \frac{(\eu^x + \eu^{-x})(\eu^y + \eu^{-y})
+ (\eu^x - \eu^{-x})(\eu^y - \eu^{-y})}{4} ;
\]
فمن الجداءات الثمانية، تختصر الأربعة «المختلطة»
($\eu^{x-y}$ و $\eu^{y-x}$) مثنى مثنى، بينما يظهر كلٌّ من
$\eu^{x+y}$ و $\eu^{-(x+y)}$ مرتين:
والمجموع هو $\frac{2\eu^{x+y} + 2\eu^{-(x+y)}}{4} = \cosh(x+y)$.
وصيغة الجيب مماثلة، لكن مع بقاء الحدود المختلطة بدل ذلك.

(4) $\sinh' = \cosh \geq 1 > 0$، فتكون $\sinh$ متزايدة تمامًا،
ولها النهايتان $\pm\infty$ (والحدّ المهيمن $\pm\frac{\eu^{\abs
x}}{2}$)؛ ثم تنتج \omterm{def:b1:logic:statement}{عبارة} التقابل من مبرهنة القيم
الوسطى (المستعملة على مستوى الثانوية؛ والمعالجة المنهجية في
\cref{ch:b1:continuity}). وعلى $\R_+$ لدينا $\cosh' = \sinh \geq 0$
ولا ينعدم إلا عند $0$: فتتزايد تمامًا من $\cosh 0 = 1$ إلى
$+\infty$. ولدينا $\tanh' > 0$ مع النهايتين $\pm 1$ عند $\pm\infty$:
وبالتفصيل،
\[
\tanh x = \frac{\eu^x - \eu^{-x}}{\eu^x + \eu^{-x}}
= \frac{1 - \eu^{-2x}}{1 + \eu^{-2x}}
\xrightarrow[x \to +\infty]{} 1 ,
\]
بعد قسمة البسط والمقام على $\eu^x$، وتعطي الفردية
النهاية $-1$ عند $-\infty$؛ فترسل الدالة $\tanh$ المتزايدة
تمامًا المجموعةَ $\R$ إذن على $\intoo{-1}1$.
\end{proof}

\begin{remark}[لماذا «زائدية»؟]
تجري النقطة $(\cosh t, \sinh t)$ على الفرع $x > 0$ من
القطع الزائد $x^2 - y^2 = 1$ (حسب
\cref{prop:b1:functions:hyprules}~(1))، تمامًا كما تجري $(\cos t, \sin
t)$ على الدائرة $x^2 + y^2 = 1$. ولكل متطابقة دائرية
شقيقةٌ زائدية، وتحكم تغيّرات الإشارة فيها
$\sinh^2 \leftrightarrow -\sin^2$.
\end{remark}

\begin{example}[صيغة الجمع للدالة $\tanh$]\label{ex:b1:functions:tanhadd}
بقسمة صيغتَي الجمع في
\cref{prop:b1:functions:hyprules}~(3) على $\cosh x\cosh y$:
\[
\tanh(x + y)
= \frac{\sinh x\cosh y + \cosh x\sinh y}
{\cosh x\cosh y + \sinh x\sinh y}
= \frac{\tanh x + \tanh y}{1 + \tanh x\,\tanh y} ,
\]
وهي شقيقة صيغة جمع الظل الزائدية --- مع إشارة
$+$ حيث تضع المثلثات إشارة $-$. ومن فوائدها: بما أن $\abs{\tanh}
< 1$، يكون الطرف الأيمن قاعدةَ «جمع سرعات» لا
تغادر $\intoo{-1}1$ أبدًا: فإذا كان $u, v \in \intoo{-1}1$ فإن
$\frac{u + v}{1 + uv} \in \intoo{-1}1$ كذلك (اكتب $u = \tanh
a$ و $v = \tanh b$، وهذا ممكن بالتقابلية، واقرأ الصيغة
بالمقلوب). والتحقق من ذلك جبريًا، دون \omterm{def:b1:functions:hyperbolic}{دوال زائدية}،
تمرين مؤلم بعض الشيء؛ أمّا الوسْط بالدالة $\tanh$
فيجعله سطرًا واحدًا --- وهي الاستراتيجية نفسها التي توفّرها
الدوال الدائرية لدائرة الوحدة.
\end{example}

\begin{proposition}[الدوال الزائدية العكسية]\label{prop:b1:functions:invhyp}
للتقابلات العكسية التي يمنحها \cref{prop:b1:functions:hyprules}~(4)
صيغ مغلقة:
\[
\operatorname{arsinh} x = \ln\bigl(x + \sqrt{x^2 + 1}\bigr)
\ (x \in \R),
\qquad
\operatorname{arcosh} x = \ln\bigl(x + \sqrt{x^2 - 1}\bigr)
\ (x \geq 1),
\]
\[
\operatorname{artanh} x = \frac 12 \ln\frac{1 + x}{1 - x}
\ (\abs x < 1),
\]
ومشتقاتها $\frac{1}{\sqrt{x^2+1}}$ و$\frac{1}{\sqrt{x^2-1}}$
(من أجل $x > 1$) و $\frac{1}{1 - x^2}$ على الترتيب.
\end{proposition}

\begin{proof}
من أجل $\operatorname{arsinh}$: حُلَّ $x = \sinh y = \frac{\eu^y -
\eu^{-y}}{2}$. وبوضع $u = \eu^y > 0$: $u^2 - 2xu - 1 = 0$، ومنه $u = x
+ \sqrt{x^2 + 1}$ (فالجذر $x - \sqrt{x^2+1}$ سالب)، و$y =
\ln(x + \sqrt{x^2+1})$. والاثنان الآخران حسابان مماثلان
($u^2 - 2xu + 1 = 0$ من أجل $\operatorname{arcosh}$، مع الاحتفاظ بالجذر
$\geq 1$؛ وإعادة ترتيب في سطرين من أجل $\operatorname{artanh}$).
وأمّا المشتقات: فاشتقّ العبارات اللوغاريتمية، مثلًا
\[
\bigl(\ln(x + \sqrt{x^2+1})\bigr)'
= \frac{1 + \frac{x}{\sqrt{x^2+1}}}{x + \sqrt{x^2+1}}
= \frac{1}{\sqrt{x^2 + 1}} . \qedhere
\]
\end{proof}

\begin{example}[الصيغ المغلقة في العمل]\label{ex:b1:functions:arcoshtwo}
حلّ المعادلة $\cosh t = 2$ حيث $t \geq 0$ هو، بالصيغة
المغلقة، $t = \operatorname{arcosh} 2 = \ln(2 + \sqrt3) \approx
1.317$؛ والحل الآخر هو $-t$ بالزوجية --- وبالفعل
$\ln(2 - \sqrt3) = \ln\frac1{2 + \sqrt3} = -\ln(2 + \sqrt3)$: أي أن
جذرَي المعادلة من الدرجة الثانية $u^2 - 4u + 1 = 0$، وهما $u = \eu^t$،
مقلوبان أحدهما للآخر، كما يقتضيه جداؤهما $1$ (فييت). ويُظهر هذا
الحساب الصغير النمط العام: فالمعادلات الزائدية
تتحول إلى معادلات من الدرجة الثانية في $\eu^t$، ويظهر التناظر $t \mapsto -t$
في صورة التناظر $u \mapsto 1/u$ للمعادلة من الدرجة الثانية ---
ويجدر تذكّر ذلك عند حلّ \cref{exo:b1:functions:7}.
\end{example}

\begin{method}[اختيار صيغة الدالة الأصلية المناسبة]\label{met:b1:functions:primitive}
أنماط المشتقات الثلاثة التي ينبغي حفظها من أجل المكاملة
(\cref{ch:b1:integration}):
\[
\int \frac{\dd x}{1 + x^2} = \arctan x + C,
\qquad
\int \frac{\dd x}{\sqrt{1 - x^2}} = \arcsin x + C,
\qquad
\int \frac{\dd x}{\sqrt{x^2 + 1}} = \operatorname{arsinh} x + C,
\]
وأمّا في حالة كثير حدود عام من الدرجة الثانية فنردّه إلى هذه
بإكمال المربع وإعادة السلّم.
\end{method}

\begin{remark}[أين يُستعمل هذا الفصل]\label{rem:b1:functions:whereused}
هذا الفصل هو المفردات العاملة لكل التحليل الآتي.
فسلّم النموّ في \cref{prop:b1:functions:powerrules} يحسم
مسائل التقارب في كل \cref{ch:b1:seq,ch:b1:series}؛
وصيغ المشتقات في \cref{prop:b1:functions:arcderiv} و
\cref{prop:b1:functions:invhyp} هي الدوال الأصلية الأكثر
لزومًا في \cref{ch:b1:integration}، عبر
\cref{met:b1:functions:primitive}؛ وتوسِّط \omterm{def:b1:functions:hyperbolic}{الدوال الزائدية}
حلولَ المعادلة $y'' = y$ في
\cref{ch:b1:diffeq} تمامًا كما توسّط الدوال الدائرية
حلولَ $y'' = -y$. وتوصيف $\exp$ بالعلاقتين $f' = f$ و
$f(0) = 1$ (\cref{prop:b1:functions:expln}) هو البذرة الأحادية
البُعد لنظرية المعادلات التفاضلية الخطية، ويعود
منحنى السلسلة $y = \cosh x$ بين المنحنيات المستوية في
\cref{ch:b1:curves}.
\end{remark}

\begin{remark}[استراحة: برنامج الدالة العكسية]\label{rem:b1:functions:interlude}
شغّل هذا الفصل برنامجًا واحدًا أربع مرات: اقصر دالةً
حتى تصير تقابلًا رتيبًا تمامًا، وسمِّ مقلوبها،
وانقل كل صيغة عبره. أمّا ما \emph{استُعمل} في كل
خطوة --- أي أن الدالة المتصلة الرتيبة تمامًا على
مجال تكون تقابلًا على مجال، وأن مقلوبها
متصل ثم قابل للاشتقاق بعيدًا عن النقاط الحرجة --- فقد
استُلف على حساب الثانوية. ويُسدَّد الدين داخل هذا
المجلد: إذ يبرهن \cref{ch:b1:continuity} على \omterm{def:b1:logic:statement}{عبارة} التقابل
(وهي مبرهنة المقلوب الرتيب، القائمة على مبرهنة القيم
الوسطى)، ويبرهن \cref{ch:b1:derivative} على قاعدة المشتقة
$(f^{-1})' = 1/(f' \circ f^{-1})$ التي أنتجت ضمنًا كل
صيغة في \cref{prop:b1:functions:arcderiv} و
\cref{prop:b1:functions:invhyp}. وقراءة تلك الفصول مع استحضار
$\arcsin$ و$\operatorname{arsinh}$ --- بوصفها الأمثلة المشغولة
التي بُنيت النظرية لتبريرها --- هي الطريق المقصود
للخروج من الدائرية الظاهرة.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b1:functions:2}

(وهي صيغة دي موافر الزائدية).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:functions:4}
بسّط $\cos(2\arcsin x)$ و $\sin(2\arctan x)$ إلى عبارتين
جبريتين في $x$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:functions:5}
رتّب، من أجل $x$ كبيرة، الدوال
$x^{100}$ و $\;\eu^{\sqrt x}$ و $\;(\ln x)^{1000}$ و $\;\eu^{x/100}$
و $\;x^{\ln x}$، من الأبطأ نموًّا إلى الأسرع، مع تبريرات
مبنيّة على \cref{prop:b1:functions:powerrules}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:functions:6}
ادرس الدالة $f(x) = \arctan\dfrac{2x}{1 - x^2}$ على \omterm{def:b1:logic:sets}{مجموعة} تعريفها:
احسب $f'$، وقارنها بالمشتقة $(2\arctan x)'$، وعبّر عن $f(x)$
بدلالة $\arctan x$ على كل مجال من المجالات الثلاثة \omterm{def:b1:logic:sets}{لمجموعة} التعريف.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:functions:7}
حُلَّ في $\R$: $\;\cosh x = 2$؛ ثم $5\cosh x - 4 \sinh x = 3$
(عبّر عن الحلول باللوغاريتمات). \emph{إرشاد للثانية:
اكتب كل شيء بوضع $u = \eu^x$.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:functions:8}
برهن على المتطابقة $\arctan\frac{1}{2} + \arctan\frac{1}{3} =

\[
4\arctan\frac 15 - \arctan\frac{1}{239} = \frac{\pi}{4}.
\]
\emph{إرشاد: احسب ظلّ الطرفين باستعمال صيغة الجمع، وتحكّم
في المجال الذي يقع فيه كل طرف.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:functions:9}
برهن على أنه لكل $x \geq 0$: $\;x - \dfrac{x^3}{6} \leq \sin x \leq
x$ \emph{(ادرس المشتقات المتتالية للفروق)}، واستنتج
أن $\lim_{x \to 0^+} \frac{x - \sin x}{x^3} = \frac 16$ مقبولة
--- أمّا النهاية نفسها فتُرسى في
\cref{ch:b1:taylor}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:functions:10}
من أجل $x \in \intcc{-1}{1}$، ضع
$h(x) = \arcsin\bigl(2x\sqrt{1 - x^2}\,\bigr) - 2\arcsin x$. وقد يُتوقَّع
أن $h = 0$ من المتطابقة $\sin 2y = 2\sin y\cos y$ --- لكن
$h$ \emph{ليست} معدومة تطابقيًا. احسب $h'$ على المجالات المفتوحة
التي توجد فيها، واحسب $h$ عند نقاط حسنة الاختيار، وأعط الوصف الكامل للدالة $h$ بوصفها ثابتة متشعّبة على $\intcc{-1}{1}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:functions:11}
(دالة غودرمان) لتكن $g(x) = \arctan(\sinh x)$ من أجل $x \in \R$. برهن
على أن $g$ تقابل فرديّ متزايد تمامًا من $\R$ على
$\intoo{-\frac\pi2}{\frac\pi2}$، وأن $g'(x) = \frac{1}{\cosh x}$،
وأن $\tan g(x) = \sinh x$ و$\;\sin g(x) = \tanh x$ و
$\;\cos g(x) = \frac{1}{\cosh x}$: أي أن الدالة $g$ تربط المثلثات
الدائرية بالمثلثات الزائدية دون أعداد عقدية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:functions:12}
برهن على أن
\[
\arctan 1 + \arctan 2 + \arctan 3 = \pi .
\]
\emph{إرشاد: احسب $\arctan 2 + \arctan 3$ أولًا، مع تحديد موضع
المجموع كما في \cref{ex:b1:functions:arctansum}؛ وانتبه إلى أن
مقام صيغة الجمع سالب هنا.}
\end{exercise}

\section{مسألة: المعادلة $x^y = y^x$}

\begin{problem}\label{pb:b1:functions:1}
أيّ أزواج الأعداد الموجبة يحقق $x^y = y^x$؟ يعرف الجميع
مثالًا واحدًا يبدو عرضيًا، $2^4 = 4^2 = 16$؛ وتبيّن هذه المسألة
أن لا شيء فيه عرضيّ. فالمعادلة كلها تحكمها
تغيرات الدالة الواحدة
\[
f(t) = \frac{\ln t}{t} \qquad (t > 0),
\]
التي تعطي دراستها: \omterm{def:b1:logic:sets}{مجموعة} الحلول كاملةً (قطر مع فرع منحنٍ واحد يمرّ بالنقطة $(\eu, \eu)$)، وتوسيطًا ناطقًا للفرع،
وواقعة أن $(2, 4)$ هي نقطته الصحيحة \emph{الوحيدة}،
وتصنيف كل نقاطه الناطقة --- ومن الفوائد الجانبية،
مقارنة $\eu^\pi$ مع $\pi^\eu$ والتقارب الرتيب للمتتالية
$\bigl(1 + \frac1n\bigr)^n$ نحو $\eu$.
\index{دالة القوة}

\medskip
\noindent\textbf{الجزء 1 --- الدالة $f(t) = \ln t / t$.}
\begin{enumerate}
  \item برّر أن $f$ قابلة للاشتقاق على $\intoo0{+\infty}$،
        واحسب $f'$، وضع جدول التغيرات: تتزايد $f$
        تمامًا على $\intoc0\eu$، وتتناقص تمامًا على
        $\intco\eu{+\infty}$، وقيمتها القصوى $f(\eu) = \frac1\eu$.
  \item عيّن نهايتَي $f$ عند $0^+$ وعند $+\infty$
        (\cref{prop:b1:functions:powerrules})، وإشارة $f$
        (سالبة على $\intoo01$، ومعدومة عند $1$، وموجبة بعده)،
        وارسم المنحني تخطيطيًا.
  \item تحقق بحساب مباشر من أن $f(2) = f(4)$. (واحتفظ بهذا
        التساوي في ذهنك: فالمسألة كلها تنبت منه.)
  \item ليكن $c \in \R$. ناقش، بحسب قيمة $c$، عدد
        حلول $f(t) = c$: حلٌّ واحد بالضبط من أجل $c \leq
        0$؛ وحلّان بالضبط (أحدهما في $\intoo1\eu$ والآخر في
        $\intoo\eu{+\infty}$) من أجل $0 < c < \frac1\eu$؛ وحلٌّ واحد
        بالضبط من أجل $c = \frac1\eu$؛ ولا حلّ من أجل $c > \frac1\eu$.
  \item استنتج أن $\eu^x > x^\eu$ من أجل كل $x > 0$ حيث $x \neq
        \eu$، وبوجه خاص احسم أيّ العددين $\eu^\pi$ و
        $\pi^\eu$ أكبر.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- المعادلة ومنحنيها.} في هذا
الجزء $x, y > 0$.
\begin{enumerate}[resume]
  \item بيّن أن $x^y = y^x \iff f(x) = f(y)$.
  \item استنتج بنية \omterm{def:b1:logic:sets}{مجموعة} الحلول: كل أزواج القطر
        $(x, x)$؛ والأزواج \emph{غير البديهية} (حيث $x \neq
        y$)، وهي تحقق: أن كلتا الإحداثيتين $> 1$، وأن إحداهما
        تقع في $\intoo1\eu$ بينما تقع الأخرى في
        $\intoo\eu{+\infty}$.
  \item وسّط الأزواج غير البديهية: بكتابة $y = tx$ حيث $t
        > 0$ و $t \neq 1$، بيّن أن $x^y = y^x$ تفرض
        \[
        x(t) = t^{\frac1{t-1}},
        \qquad
        y(t) = t\,x(t) = t^{\frac{t}{t-1}},
        \]
        وأن كل زوج كهذا حلٌّ بالعكس.
  \item تحقق من أن $t = 2$ يعطي $(2, 4)$، وبرهن على التناظر
        $x(1/t) = y(t)$ و $y(1/t) = x(t)$: أي أن قلب الوسيط
        يبادل بين الإحداثيتين.
  \item عيّن نهايتَي $x(t)$ و $y(t)$ عندما $t \to 1$
        (وكلتاهما تؤول إلى $\eu$)، وعندما $t \to +\infty$ ($x \to 1$ و$y
        \to +\infty$) وعندما $t \to 0^+$ ($x \to +\infty$ و$y \to
        1$). وصِف الفرع الناتج: منحنٍ مقارب
        للمستقيمين $x = 1$ و $y = 1$، ويقطع القطر عند
        $(\eu, \eu)$.
  \item بيّن أن $t \mapsto x(t)$ متناقص تمامًا على
        $\intoo1{+\infty}$. \emph{(ادرس $h(t) = 1 - \frac1t - \ln
        t$، فإشارتها تتحكم في مشتقة $\ln x(t) =
        \frac{\ln t}{t - 1}$.)}
  \item استنتج أن الحلول غير البديهية تعرّف تقابلًا متناقصًا
        تمامًا $\varphi \colon \intoo1\eu \to
        \intoo\eu{+\infty}$ حيث $x^{\varphi(x)} = \varphi(x)^x$،
        وأن $\varphi$ انطواء على الفرع:
        $\varphi(\varphi(x)) = x$ حيثما كان الطرفان معرَّفين.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- النقاط الصحيحة والناطقة.}
\begin{enumerate}[resume]
  \item برهن على أن $(2, 4)$ و $(4, 2)$ هما الحلان
        \emph{الصحيحان} غير البديهيين الوحيدان للمعادلة $x^y = y^x$.
  \item من أجل $n \in \N^*$، طبّق التوسيط مع $t = 1 +
        \frac1n$ وبيّن أن
        \[
        x_n = \Bigl(1 + \frac1n\Bigr)^{n},
        \qquad
        y_n = \Bigl(1 + \frac1n\Bigr)^{n+1}
        \]
        حلٌّ \emph{ناطق} غير بديهي من أجل كل $n$،
        مع $(x_1, y_1) = (2, 4)$.
  \item وبالعكس، ليكن $(x, y)$ حلًّا ناطقًا غير بديهي
        حيث $y > x$، واكتب $t = y/x = 1 + \frac rs$ في أبسط
        صورة (حيث $r, s \in \N^*$). باستعمال $x = t^{s/r}$ وبقبول
        وحدانية التفكيك إلى عوامل أولية (وهي مألوفة من
        المدرسة؛ ومبرهن عليها في \cref{ch:b1:arith})، بيّن أن نطقية $x$
        تفرض أن يكون كلٌّ من $s$ و $s + r$ قوةً من الرتبة $r$
        لعدد صحيح.
  \item بيّن بمبرهنة ثنائي الحدّ أن $a^r - b^r = r$ ليس له
        حلول صحيحة $a > b \geq 1$ عندما $r \geq 2$، و
        استنتج: أن الحلول الناطقة للمعادلة $x^y = y^x$ هي
        بالضبط الأزواج $(x_n, y_n)$ من السؤال 14 (مع
        مبادلاتها).
  \item تحقق من الحالة $n = 2$ عدديًا: احسب $f(9/4)$ و
        $f(27/8)$ بأربعة أرقام عشرية وتأكّد من تطابقهما.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- فوائد جانبية.}
\begin{enumerate}[resume]
  \item استعمل السؤالين 7 و 11 للبرهان، دون أيّ حساب
        إضافي، على أن المتتالية $x_n = \bigl(1 +
        \frac1n\bigr)^n$ متزايدة تمامًا مع $x_n < \eu$،
        وأن $y_n = \bigl(1 + \frac1n\bigr)^{n+1}$ متناقصة
        تمامًا مع $y_n > \eu$، وأن كلتيهما تتقارب نحو
        $\eu$. (فالوسيط $t_n = 1 + \frac1n$ يتناقص نحو
        $1$.)
  \item أرسِ قاعدة المقارنة العامة من أجل $1 < a < b$:
        إذا كان $\eu \leq a$ فإن $a^b > b^a$؛ وإذا كان $b \leq \eu$ فإن
        $a^b < b^a$؛ وبيّن بالمثالين $(2, 3)$ و
        $(2, 5)$ أنه في الحالة المختلطة $a < \eu < b$ تقع
        النتيجتان معًا فعلًا.
  \item افترض أن الانطواء $\varphi$ من السؤال 12 قابل
        للاشتقاق عند $\eu$ (وهو كذلك). اشتقّ
        المتطابقة $\varphi(\varphi(x)) = x$ عند $x = \eu$ واستنتج أن
        $\varphi'(\eu) = -1$: أي أن الفرع يقطع القطر
        عموديًا عليه.
  \item جد زوج الحلول الوحيد الذي يحقق $y = 3x$، في صورة
        مغلقة، وتحقق منه عدديًا بأربعة أرقام عشرية عبر $f$.
  \item من بين الأعداد $\sqrt2$ و $\sqrt[3]3$ و $\sqrt[4]4$ و
        $\sqrt[5]5$، عيّن الأكبر وحدّد العددين
        المتساويين. \emph{(قارن $n^{1/n} =
        \eu^{f(n)}$.)}
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- توليفة ختامية.}
\begin{enumerate}[resume]
  \item صِف \omterm{def:b1:logic:sets}{مجموعة} حلول $x^y = y^x$ كاملةً في
        الربع $x, y > 0$ --- القطر مع الفرع، وتقاطعهما
        $(\eu, \eu)$، والمقاربات، والنقطة الصحيحة
        $(2, 4)$، والنقاط الناطقة المتراكمة عند
        $(\eu, \eu)$ --- في صورة تستطيع رسمها من الذاكرة.
  \item أين استعملت المسألة بالضبط: (أ) مقارنات النموّ في
        \cref{prop:b1:functions:powerrules}؛ (ب)
        مبرهنة القيم الوسطى (عبر عبارات
        التقابل)؛ (ج) وحدانية التفكيك إلى عوامل
        أولية المقبولة؟ جملة واحدة لكلٍّ منها.
  \item العبرة، في فقرة قصيرة: جدول تغيرات واحد
        حلّ معادلةً في مجهولين، وصنّف نقاطها
        الناطقة، وبرهن على التقارب الرتيب للمتتالية
        $\bigl(1 + \frac1n\bigr)^n$ --- علّق على هذا الاقتصاد،
        وسمِّ أين يُصنَّع كل خيط لاحقًا في
        المجلد (\cref{ch:b1:seq} من أجل المتتالية،
        و\cref{ch:b1:derivative} من أجل جداول التغيرات،
        و\cref{ch:b1:taylor} من أجل الدقة التي يفتقر إليها الجدول).
\end{enumerate}
\end{problem}
