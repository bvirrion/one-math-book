\chapter{البُعد المنتهي}\label{ch:b1:findim}

يحمل \omterm{def:b1:vspaces:span}{الفضاء المولَّد} بعدد منته من المتجهات \emph{بُعدًا} معرَّفًا
جيدًا --- وهو الحجم المشترك لكل أسسه. وتتدفق البراهين
أدناه كلها من محرّك توفيقي واحد، هو مبرهنة التبادل المساعدة:
فلا يمكن \omterm{def:b1:vspaces:free}{لعائلة حرة} أن تفوق عددًا عائلةً مولِّدة. ومع البُعد
تأتي الأدوات المستعملة في كل مكان بعده: مبرهنة \omterm{def:b1:vspaces:free}{الأساس}
غير التام، \omterm{def:b1:findim:rank}{ورتبة} عائلة، وصيغة غراسمان.

\section{وجود الأسس}

\begin{definition}\label{def:b1:findim:def}
يكون $E$ \emph{ذا بُعد منته}\index{بُعد منته} إذا كانت له
عائلة مولِّدة منتهية. (وإلا فهو \emph{ذو بُعد غير منته}:
وكذلك $K[X]$، الذي لا تولّد عائلاته المنتهية إلا كثيرات حدود محدودة
الدرجة.)
\end{definition}

\begin{theorem}[مبرهنة التبادل المساعدة]\label{thm:b1:findim:exchange}
لتولّد $(g_1, \dots, g_n)$ الفضاءَ $E$ ولتكن $(f_1, \dots, f_p)$ \omterm{def:b1:vspaces:free}{حرة}
في $E$. عندئذ $p \leq n$.
\end{theorem}

\begin{proof}
نبرهن بالاستقراء على $k \leq p$: \emph{بعد إعادة ترقيم
المقادير $g_j$، تولّد العائلة $(f_1, \dots, f_k, g_{k+1}, \dots, g_n)$ الفضاءَ
$E$} --- وهذا يفرض $k \leq n$ عند كل مرحلة، و $p \leq n$ في
النهاية.

$k = 0$: وهو الفرض. والخطوة: افترضها من أجل $k - 1$؛ عندئذ يكون $f_k$
تركيبةً من $(f_1, \dots, f_{k-1},$ $g_k, \dots, g_n)$. وفي
هذه التركيبة يكون لبعض $g_j$ ($j \geq k$) معامل غير معدوم
---
وإلا لكان $f_k$ تركيبةً من $f_1, \dots, f_{k-1}$،
مناقضًا الحرية. فأعد الترقيم بحيث يكون $j = k$، وحُلَّ من أجل
$g_k$: فيكون $g_k$ تركيبةً من $(f_1, \dots, f_k, g_{k+1}, \dots,
g_n)$. وعندئذ يمكن إعادة التعبير عن كل متجهة من $E$، معبَّرًا عنها بالعائلة $(k-1)$،
بالعائلة ذات $k$: فهي مولِّدة.
(وإذا كان $k - 1 = n$، فلا يبقى أيّ $g$ ويكون $f_k$ تركيبةً من
المقادير $f_i$ وحدها: وهذا محال؛ ومنه $k \leq n$.)
\end{proof}

\begin{example}[التبادل، مراقَبًا مرة واحدة]
\label{ex:b1:findim:exchangerun}
في $\R^2$، خذ العائلة المولِّدة $\bigl(g_1, g_2\bigr) =
\bigl((1,0), (0,1)\bigr)$ \omterm{def:b1:vspaces:free}{والعائلة الحرة} $\bigl(f_1,
f_2\bigr) = \bigl((1,2), (3,4)\bigr)$. الخطوة $1$: $f_1 = 1\cdot
g_1 + 2\cdot g_2$؛ ومعامل $g_2$ غير معدوم، ومنه
بادِل $g_2$ بالمقدار $f_1$: فتبقى العائلة $\bigl(f_1, g_1\bigr)$
مولِّدة ($g_2 = \frac12(f_1 - g_1)$). والخطوة $2$: $f_2 =
(3,4) = 2\,f_1 + 1\cdot g_1$؛ ومعامل $g_1$ الباقي
غير معدوم (وهذا لازم: لأن $f_2$ ليس مضاعفًا للمقدار $f_1$)،
فبادِل مرة أخرى: فتولّد $\bigl(f_1, f_2\bigr)$ الفضاءَ $\R^2$. ولو
كانت هناك متجهة \omterm{def:b1:vspaces:free}{حرة} ثالثة $f_3$، لما بقي أيّ $g$ ليمتصّها
--- وهذا بالضبط كيف تمنع المبرهنة المساعدة $3$ متجهات \omterm{def:b1:vspaces:free}{حرة}
في $\R^2$. والبرهان أعلاه هو مسك الدفاتر هذا معمَّمًا.
\end{example}

\begin{theorem}[الأسس في البُعد المنتهي]\label{thm:b1:findim:bases}
ليكن $E \neq \{0\}$ \omterm{def:b1:findim:def}{ذا بُعد منته}.
\begin{enumerate}
  \item يمكن استخراج \omterm{def:b1:vspaces:free}{أساس} من كل عائلة مولِّدة منتهية.
  \item (مبرهنة \omterm{def:b1:vspaces:free}{الأساس} غير التام\index{مبرهنة الأساس غير التام})
        كل \omterm{def:b1:vspaces:free}{عائلة حرة} تمتدّ إلى \omterm{def:b1:vspaces:free}{أساس}، باستعمال متجهات من أيّ
        عائلة مولِّدة مختارة.
  \item كل أسس $E$ منتهية، وبالعدد نفسه من
        العناصر: وهو \emph{البُعد} $\dim E$. (والاصطلاح:
        $\dim\{0\} = 0$.)
\end{enumerate}
\end{theorem}

\begin{proof}
(1) أسقط، واحدةً في كل مرة، أيّ متجهة تكون تركيبةً من
البقية؛ فتبقى العائلة مولِّدة، لأنه يمكن في أيّ \omterm{def:b1:logic:statement}{عبارة}
تستعمل المتجهة المسقَطة تعويضُ تركيبتها من
الناجيات. وتنتهي العملية --- إذ تقلّص كل خطوة عائلةً
منتهية بواحد --- وتتوقف بالضبط عندما لا تكون أيّ متجهة
باقية تركيبةً من البقية. وتبقى العائلة النهائية
مولِّدة، وهي \omterm{def:b1:vspaces:free}{حرة}: إذ ستحمل تركيبة معدومة غير بديهية
معاملًا غير معدوم، والقسمة عليه ستحلّ
من أجل المتجهة المقابلة بدلالة البقية، فتجعلها
قابلة للإسقاط بعد كل شيء --- مناقضًا توقّف العملية.

(2) لتكن $(f_1, \dots, f_p)$ \omterm{def:b1:vspaces:free}{حرة} و $(g_1, \dots, g_n)$ مولِّدة.
اجرِ على $g_1, \dots, g_n$، ملحقًا $g_j$ بالعائلة الحالية
كلما لم يكن في فضائها المولَّد أصلًا
(\cref{prop:b1:vspaces:freecriteria}~(2) يبقي \omterm{def:b1:vspaces:free}{العائلة حرة}).
والعائلة النهائية \omterm{def:b1:vspaces:free}{حرة}، ومولِّدة: لأن كل $g_j$ يقع في
فضائها المولَّد --- فإمّا أُلحق وإمّا كان تركيبةً أصلًا.

(3) الأساسان كلٌّ منهما حرّ ومولِّد: فتعطي مبرهنة التبادل المساعدة
المتراجحتين بين قوّتيهما العدديتين. (والتناهي:
\omterm{def:b1:vspaces:free}{الأساس} حرّ، ومنه فهو بالتبادل ليس أكبر من عائلة مولِّدة
منتهية.)
\end{proof}

\begin{example}\label{ex:b1:findim:dims}
$\dim K^n = n$ (\omterm{def:b1:vspaces:free}{بالأساس} القانوني)؛ و $\dim K_n[X] = n + 1$ (بالوحيدات الحدّ)؛
و $\dim_\R \C = 2$؛ ولفضاء حلول $y'' + ay' + by = 0$
بُعدٌ $2$ (\cref{thm:b1:diffeq:homogeneous2}: فالحلول
معلَّمة تعليمًا \omterm{def:b1:logic:inj}{تقابليًا} وخطيًا بالمقدار $(\lambda, \mu) \in K^2$).
\end{example}

\begin{example}[المجموعة نفسها ببُعدين]
\label{ex:b1:findim:twodimensions}
\omterm{def:b1:logic:sets}{المجموعة} $\C^2$ من أزواج الأعداد العقدية \omterm{def:b1:vspaces:def}{فضاء متجهي} على $\C$
بُعده $2$ (\omterm{def:b1:vspaces:free}{بالأساس} القانوني $e_1, e_2$) --- \omterm{def:b1:vspaces:def}{وفضاء متجهي}
على $\R$ بُعده $4$، \omterm{def:b1:vspaces:free}{بالأساس}
\[
(1, 0),\quad (\iu, 0),\quad (0, 1),\quad (0, \iu):
\]
فلكل $(z, w) = (a + \iu b,\ c + \iu d)$ \omterm{prop:b1:vspaces:coordinates}{إحداثيات} حقيقية
$(a, b, c, d)$، وحيدة. فالبُعد ليس خاصية \omterm{def:b1:logic:sets}{لمجموعة}
المتجهات وحدها: بل يعدّ درجات الحرية \emph{بالنسبة
إلى السلالم المسموح بها}، وتنصيفُ مخزون السلالم من $\C$
إلى $\R$ يضاعف العدّ. (وتستغلّ مسألة نهاية الأسبوع الحالة
القصوى من هذه الحساسية، بسلالم مقلَّصة إلى
$\Q$.)
\end{example}

\begin{example}[تشغيل خوارزمية الإتمام]
\label{ex:b1:findim:completion}
أتمّ \omterm{def:b1:vspaces:free}{العائلة الحرة} $\bigl((1,1,1)\bigr)$ إلى \omterm{def:b1:vspaces:free}{أساس} للمقدار
$\R^3$ باستعمال المتجهات القانونية. اجرِ برهان
\cref{thm:b1:findim:bases}~(2) على العائلة المولِّدة $(e_1,
e_2, e_3)$: هل $e_1 \in \operatorname{Vect}(1,1,1)$؟ لا (لأن مضاعفات
$(1,1,1)$ لها \omterm{prop:b1:vspaces:coordinates}{إحداثيات} متساوية) --- فألحقه. وهل $e_2 \in
\operatorname{Vect}\bigl((1,1,1), e_1\bigr)$؟ للتركيبة
$\alpha(1,1,1) + \beta e_1$ إحداثيتان ثانية وثالثة متساويتان،
وليس ذلك حال $e_2$ --- فألحقه. والعائلة
$\bigl((1,1,1), e_1, e_2\bigr)$ \omterm{def:b1:vspaces:free}{حرة} بعدد $3$ متجهات: فتوقّف،
فهي \omterm{def:b1:vspaces:free}{أساس} (\cref{prop:b1:findim:twoofthree} سيجعل هذا
المنعكس رسميًا). ولاحظ أن النتيجة تتعلق بالترتيب الذي تُمسح فيه
المقادير $g_j$: فالإتمام خوارزمية لا صيغة.
\end{example}

\begin{proposition}[قاعدة اثنين من ثلاثة]\label{prop:b1:findim:twoofthree}
ليكن $\dim E = n$ ولتكن $\mathcal{F}$ عائلةً من $n$ متجهة بالضبط
من $E$. عندئذ
\[
\mathcal{F} \text{ حرة} \iff \mathcal{F} \text{ مولِّدة} \iff
\mathcal{F} \text{ أساس}.
\]
وفوق ذلك لكل \omterm{def:b1:vspaces:free}{عائلة حرة} $\leq n$ متجهة، ولكل عائلة مولِّدة
$\geq n$.
\end{proposition}

\begin{proof}
حواصر القوة العددية هي مبرهنة التبادل المساعدة إزاء \omterm{def:b1:vspaces:free}{أساس}. فإذا كانت
$\mathcal{F}$ (بالحجم $n$) \omterm{def:b1:vspaces:free}{حرة} وغير مولِّدة، فإن $x$ ما يقع
خارج فضائها المولَّد؛ وإلحاق $x$ يعطي \omterm{def:b1:vspaces:free}{عائلة حرة} من $n + 1$
متجهة: وهذا محال. وإذا كانت $\mathcal{F}$ مولِّدة وغير \omterm{def:b1:vspaces:free}{حرة}، فإن
استخراج \omterm{def:b1:vspaces:free}{أساس} (\cref{thm:b1:findim:bases}~(1)) يعطي \omterm{def:b1:vspaces:free}{أساسًا} من
$< n$ متجهة: وهذا محال.
\end{proof}

\begin{example}[اثنان من ثلاثة، بتوفير نصف العمل]
\label{ex:b1:findim:twoofthreerun}
هل $\bigl((1,1,0), (0,1,1), (1,0,1)\bigr)$ \omterm{def:b1:vspaces:free}{أساس} للمقدار $\R^3$؟
عُدّ: ثلاث متجهات وبُعد ثلاثة --- ومنه تفصل الحرية
وحدها. وتعطي تركيبة معدومة $a + c = 0$ و $a + b = 0$ و $b +
c = 0$؛ وبجمع الثلاث، $2(a + b + c) = 0$، وبطرح
كل معادلة أصلية من $a + b + c = 0$ يبقى $b = c = a =
0$: فهي \omterm{def:b1:vspaces:free}{حرة}، ومنه \omterm{def:b1:vspaces:free}{أساس}، والنصف التوليدي من
التحقق تعطيه المبرهنة بالمجّان. وقارن
\cref{ex:b1:vspaces:testbasis}، حيث كان يجب إجراء التحقق
المزدوج نفسه باليد --- فيتحوّل فصل واحد من النظرية
إلى ذلك التوفير بالضبط، في كل تحقق من \omterm{def:b1:vspaces:free}{أساس} في
بقية الكتاب.
\end{example}

\begin{method}[حساب بُعد]
\label{met:b1:findim:computedim}
ثلاثة طرق معيارية، بترتيب تناقص التواتر.
\begin{enumerate}
  \item \emph{علّم، ثم اقرأ \omterm{def:b1:vspaces:free}{أساسًا}.} حُلَّ القيود
        المعرِّفة، وعبّر عن العنصر العام
        تعبيرًا خطيًا بالوسائط الناجية، وتحقق من أن
        المتجهات التي تضرب الوسائط \omterm{def:b1:vspaces:free}{حرة}: فيكون
        البُعد عدد الوسائط. (ويجري هذا أدناه على \omterm{def:b1:vspaces:subspace}{فضاء
        جزئي} ملموس من $\R^4$.)
  \item \emph{أظهر تعليمًا خطيًا \omterm{def:b1:logic:inj}{تقابليًا}.} حين
        تتحدد العناصر بعدد منته من القيم ---
        الشروط الابتدائية لعلاقة تراجعية
        (\cref{exo:b1:findim:10})، ومعاملات صيغة حلّ
        (\cref{ex:b1:findim:dims}) --- يكون البُعد
        عدد تلك القيم.
  \item \emph{استعمل الصيغ.} غراسمان من أجل التقاطعات و
        المجاميع، \omterm{def:b1:findim:rank}{والرتبة} من أجل الفضاءات المولَّدة، ولاحقًا مبرهنة \omterm{def:b1:findim:rank}{الرتبة}
        من أجل النوى والصور: فالأبعاد \emph{تُحسب}
        عادةً، لا تُخمَّن.
\end{enumerate}
وفي الطرق الثلاثة كلها، تكون قاعدة اثنين من ثلاثة هي الخاتمة:
فما إن يتطابق العدّ، تختم الحرية أو التوليد وحدها.
\end{method}

\begin{example}[الطريق 1، كاملًا]
\label{ex:b1:findim:parametrize}
بُعد $H = \{(x, y, z, t) \in \R^4 : x + y + z + t = 0
\text{ و } x = t\}$. حُلَّ: $t = x$ و $y + z = -2x$، ومنه $z =
-2x - y$ مع $x, y$ حرين:
\[
(x,\ y,\ -2x - y,\ x) = x\,(1, 0, -2, 1) + y\,(0, 1, -1, 0) .
\]
والمتجهتان حرتان (انظر في الإحداثيتين الأوليين:
$(x, y) = (0,0)$)، ومنه تكوّنان \omterm{def:b1:vspaces:free}{أساسًا} للمقدار $H$ و $\dim H = 2$.
وكان العدّ متوقعًا --- إذ إن قيدين خطيين مستقلين
في $\R^4$ ينبغي أن يأكل كلٌّ منهما بُعدًا --- لكن
التعليم يبرهن على ذلك \emph{و}يسلّم \omterm{def:b1:vspaces:free}{أساسًا}، وهذا ما لا
يفعله التنبؤ وحده أبدًا؛ ويجعل \cref{exo:b1:findim:9}
شعار «كل معادلة تأكل بُعدًا واحدًا على الأكثر»
مبرهنةً.
\end{example}

\begin{example}[الطريق 2، كاملًا]
\label{ex:b1:findim:route2}
بُعد $W = \{P \in \R_n[X] : P(1) = P(2) = 0\}$ (من أجل $n
\geq 2$). فبمبرهنة العامل مطبَّقةً مرتين
(\cref{thm:b1:poly:factor}؛ والجذران $1$ و $2$ متمايزان)،
يكون $P \in W$ بالضبط عندما يكون $P = (X - 1)(X - 2)\,Q$ مع $\deg Q \leq
n - 2$. والتقابل $Q \mapsto (X-1)(X-2)Q$ خطي،
ويبلغ $W$ كله، وهو متباين (لأن الجداء لا ينعدم إلا إذا كان
$Q = 0$): ومنه فالفضاء $W$ معلَّم تعليمًا \omterm{def:b1:logic:inj}{تقابليًا} وخطيًا بالمقدار
$\R_{n-2}[X]$، ومنه
\[
\dim W = \dim \R_{n-2}[X] = n - 1 .
\]
ويأتي \omterm{def:b1:vspaces:free}{أساس} مع التعليم: أي صور
الوحيدات الحدّ، $\bigl((X-1)(X-2),\ (X-1)(X-2)X,\ \dots,\
(X-1)(X-2)X^{n-2}\bigr)$. وكل قيد تقويم جديد عند
نقطة جديدة يكلّف بُعدًا واحدًا بالضبط --- وهو العمود
الفقري العدّي \omterm{thm:b1:poly:lagrange}{لمقايسة لاغرانج}
(\cref{thm:b1:poly:lagrange}).
\end{example}

\begin{example}[القيد التكاملي يكلّف بُعدًا واحدًا كذلك]
\label{ex:b1:findim:integralconstraint}
بُعد $H = \{P \in \R_2[X] : \int_0^1 P = 0\}$. فبكتابة
$P = a + bX + cX^2$، يُقرأ القيد $a + \frac b2 + \frac
c3 = 0$؛ فحُلَّ من أجل $a$ وعلّم:
\[
P = b\Bigl(X - \frac12\Bigr) + c\Bigl(X^2 - \frac13\Bigr),
\]
ومنه $H = \operatorname{Vect}\bigl(X - \frac12,\ X^2 -
\frac13\bigr)$، والبُعد $2$ (لأن لكثيرَي الحدود درجتين
متمايزتين: فهما حران). فالشرط الخطي الواحد --- سواء كان تقويمًا
أو تكاملًا أو أيّ وصفة خطية أخرى --- يزيل بُعدًا واحدًا على الأكثر،
وبُعدًا واحدًا بالضبط ما إن لا يكون الشرط معدومًا تمامًا.
وسوف يسمّي \cref{ch:b1:linmaps} مثل هذه الوصفات
\emph{أشكالًا خطية} ومجموعاتِ حلولها \emph{مستويات فائقة}؛
وتعود متجهات \omterm{def:b1:vspaces:free}{الأساس} الموجودة هنا في
\cref{ch:b1:euclid} بداية عائلة لوجاندر.
\end{example}

\section{الفضاءات الجزئية والرتبة وغراسمان}

\begin{theorem}[الفضاءات الجزئية]\label{thm:b1:findim:subspaces}
ليكن $E$ \omterm{def:b1:findim:def}{ذا بُعد منته} ولتكن $F$ \omterm{def:b1:vspaces:subspace}{فضاءً جزئيًا}. عندئذ يكون $F$
\omterm{def:b1:findim:def}{ذا بُعد منته}، و $\dim F \leq \dim E$، مع المساواة إذا وفقط
إذا كان $F = E$. وفوق ذلك لكل \omterm{def:b1:vspaces:subspace}{فضاء جزئي} \omterm{def:b1:vspaces:subspace}{فضاء جزئي} متتامّ.
\end{theorem}

\begin{proof}
للعائلات \omterm{def:b1:vspaces:free}{الحرة} من $F$ عدد $\dim E$ متجهة على الأكثر (بمبرهنة التبادل المساعدة
في $E$: \omterm{def:b1:vspaces:free}{فالعائلة الحرة} من $F$ \omterm{def:b1:vspaces:free}{حرة} في $E$ على الخصوص، و $E$
له عائلة مولِّدة منتهية). وبين العائلات \omterm{def:b1:vspaces:free}{الحرة} من $F$
اختر واحدة ذات حجم \emph{أكبري} $p$ --- وهذا ممكن لأن الأحجام
أعداد صحيحة محدودة بالمقدار $\dim E$. وهي تولّد $F$: وإلا لوقع
$x \in F$ ما خارج فضائها المولَّد، ولأعطى إلحاق $x$
عائلةً \omterm{def:b1:vspaces:free}{حرة} من $F$ بحجم $p + 1$
(\cref{prop:b1:vspaces:freecriteria}~(2))، مناقضًا
الأكبرية. ولكونها \omterm{def:b1:vspaces:free}{حرة} ومولِّدة، تكون \omterm{def:b1:vspaces:free}{أساسًا} للمقدار $F$، و
$\dim F = p \leq \dim E$. وإذا كان $p = \dim E = n$: \omterm{def:b1:vspaces:free}{فالعائلة الحرة} من $n$ متجهة من
$E$ أساسٌ للمقدار $E$ (\cref{prop:b1:findim:twoofthree})، ومنه $F
\supseteq \operatorname{span} = E$. والمتتامّ: أتمّ \omterm{def:b1:vspaces:free}{أساسًا}
$(f_1, \dots, f_p)$ للمقدار $F$ إلى \omterm{def:b1:vspaces:free}{أساس} $(f_1, \dots, f_p, g_{p+1},
\dots, g_n)$ للمقدار $E$ (بمبرهنة \omterm{def:b1:vspaces:free}{الأساس} غير التام)؛ عندئذ يحقق $G =
\operatorname{Vect}(g_{p+1}, \dots, g_n)$ أن $E = F \oplus G$
(فوجود التفكيكات ووحدانيتها = \omterm{prop:b1:vspaces:coordinates}{الإحداثيات} في
\omterm{def:b1:vspaces:free}{الأساس} الكبير).
\end{proof}

\begin{definition}[رتبة عائلة]\label{def:b1:findim:rank}
\emph{رتبة}\index{رتبة!رتبة عائلة} عائلة منتهية من
المتجهات هي بُعد فضائها المولَّد:
$\operatorname{rk}(x_1, \dots, x_p) = \dim
\operatorname{Vect}(x_1, \dots, x_p) \leq \min(p, \dim E)$،
مع المساواة بالمقدار $p$ إذا وفقط إذا كانت \omterm{def:b1:vspaces:free}{العائلة حرة}.
\end{definition}

\begin{example}[حساب رتبة بالإزاحة]
\label{ex:b1:findim:rankcomputation}
\omterm{def:b1:findim:rank}{رتبة} $\bigl((1,2,3), (2,3,4), (3,4,5), (1,1,1)\bigr)$ في
$\R^3$. لا يتغير \omterm{def:b1:vspaces:span}{الفضاء المولَّد} حين يُطرح من متجهة
تركيبةٌ من البقية (لأن العائلتين تولّدان التركيبات
نفسها): فاستبدل بالمقدار $(2,3,4)$ المقدارَ $(2,3,4) - (1,2,3) = (1,1,1)$
وبالمقدار $(3,4,5)$ المقدارَ $(3,4,5) - (1,2,3) = (2,2,2)$. \omterm{def:b1:vspaces:span}{والفضاء المولَّد} الآن هو
$\operatorname{Vect}\bigl((1,2,3), (1,1,1), (2,2,2),
(1,1,1)\bigr) = \operatorname{Vect}\bigl((1,2,3), (1,1,1)\bigr)$،
وهاتان المتجهتان غير متناسبتين: \omterm{def:b1:findim:rank}{فالرتبة} $2$. وهذا
الإجراء بالطرح والإسقاط منظَّم في صورة إزاحة غاوس
في \cref{ch:b1:det}.
\end{example}

\begin{example}[الجمع بالوصل]
\label{ex:b1:findim:sumconcat}
خذ، في $\R^3$،
\[
F = \operatorname{Vect}\bigl((1,2,3),\ (1,1,1)\bigr),
\qquad
G = \operatorname{Vect}\bigl((2,3,4)\bigr) .
\]
المجموع $F + G$ مولَّد بالعائلة الموصولة لكل
المولّدات الثلاثة، و
\[
(2, 3, 4) = (1, 2, 3) + (1, 1, 1)
\]
تبيّن أن الثالث زائد: ومنه $F + G = F$، وبُعده $2$ ---
وبكيفية مكافئة $G \subseteq F$، وهو ما تعرضه العلاقة.
ويؤكد غراسمان: $\dim(F \cap G) = 2 + 1 - 2 = 1 = \dim G$.
فالمجاميع تُحسب بوصل المولّدات ثم تقليص
الكومة بخوارزمية \omterm{def:b1:findim:rank}{الرتبة}؛ ولا يلزم أيّ تقنية جديدة أبدًا.
\end{example}

\begin{theorem}[صيغة غراسمان]\label{thm:b1:findim:grassmann}
من أجل فضاءين جزئيين $F, G$ من $E$ ذوَي \omterm{def:b1:findim:def}{بُعد
منته}:\index{صيغة غراسمان}
\[
\dim(F + G) = \dim F + \dim G - \dim(F \cap G) .
\]
وعلى الخصوص يكون $F + G$ مباشرًا إذا وفقط إذا كان $\dim(F + G) = \dim F + \dim G$.
\end{theorem}

\begin{proof}
ابدأ من \omterm{def:b1:vspaces:free}{أساس} $(e_1, \dots, e_r)$ للمقدار $F \cap G$؛ وأتمّه
إلى \omterm{def:b1:vspaces:free}{أساس} $(e_1, \dots, e_r, f_1, \dots, f_s)$ للمقدار $F$ وإلى
\omterm{def:b1:vspaces:free}{أساس} $(e_1, \dots, e_r, g_1, \dots, g_t)$ للمقدار $G$
(\cref{thm:b1:findim:bases}~(2)). ونحن ندّعي أن
\[
\mathcal{B} = (e_1, \dots, e_r, f_1, \dots, f_s, g_1, \dots, g_t)
\]
أساسٌ للمقدار $F + G$؛ وتتبع الصيغة بالعدّ: $(r + s) +
(r + t) - r = r + s + t$.

والعائلة $\mathcal{B}$ تولّد $F + G$: فأيّ $u + v$ ($u \in F$ و $v \in G$)
ينتشر بها. والحرية: افترض $\sum \alpha_i e_i + \sum
\beta_j f_j + \sum \gamma_k g_k = 0$. فالمتجهة $w = \sum \gamma_k
g_k = -\sum\alpha_i e_i - \sum\beta_j f_j$ تقع في $G \cap F$، ومنه
تنتشر بالمقادير $(e_i)$ وحدها؛ لكن $w$ ينتشر كذلك بالمقادير $(g_k)$، وفي
أساس $G$ يجب أن يتطابق هذان التعبيران: ومنه كل
$\gamma_k = 0$ (وتتطابق \omterm{prop:b1:vspaces:coordinates}{إحداثيات} $e$). وتُردّ العلاقة
إلى $\sum\alpha_i e_i + \sum\beta_j f_j = 0$، وهي علاقة في أساس
$F$: فتنعدم كل المعاملات الباقية.
\end{proof}

\begin{example}\label{ex:b1:findim:planes}
لمستويين متمايزين $F, G$ (بُعدهما $2$) من $\R^3$ يكون $F + G
= \R^3$ (لأن مجموعهما يحتوي مستويًا احتواءً قطعيًا)، ومنه $\dim(F \cap G) =
2 + 2 - 3 = 1$: فهما يتقاطعان دائمًا على امتداد مستقيم --- فلا
«مستويات متوازية» مارّة بالمبدأ.
\end{example}

\begin{example}[غراسمان في العمل، في $\R^4$]
\label{ex:b1:findim:grassmannrun}
لتكن $F = \operatorname{Vect}\bigl(e_1,\ e_2,\ (1,1,1,0)\bigr)$
و $G = \operatorname{Vect}(e_3, e_4)$ في $\R^4$. والأبعاد:
$\dim F = 3$ (لأن للمولّد الثالث إحداثية ثالثة غير معدومة،
خارج $\operatorname{Vect}(e_1, e_2)$) و $\dim G = 2$. والمجموع:
يحتوي $F + G$ المقادير $e_1, e_2, e_4$ و $e_3 = (1,1,1,0) - e_1 -
e_2$: ومنه فهو $\R^4$ كله. وعندئذ \emph{يحسب} غراسمان
حجم التقاطع دون أيّ إزاحة أصلًا:
\[
\dim(F \cap G) = 3 + 2 - 4 = 1 .
\]
ولتعيين المستقيم، انظر داخل $G$: تقع المتجهة $(0, 0, c, d)$
في $F$ بالضبط عندما تكون $a e_1 + b e_2 + \lambda(1,1,1,0)$،
فيُفرض $\lambda = -a = -b$ و $d = 0$: ومنه فالتقاطع هو
$\operatorname{Vect}(e_3)$ --- وهذا متسق، لأن $e_3$ أُظهر
في $F$ أعلاه ويقع في $G$ بحكم التعريف. وهذا تقسيم عمل
نموذجي: فغراسمان يتنبّأ \emph{بكم} يجب البحث عنه،
ثم تجد الجملة الخطية \emph{ماذا}.
\end{example}

\begin{example}[التقاطع بالمعادلات]
\label{ex:b1:findim:intersectequations}
حين يأتي الفضاءان الجزئيان مجموعتَي حلول، يكون التقاطع مجرد
تكديس للمعادلات. ففي $\R^3$: يعطي $F = \{x + y + z = 0\}$ و
$G = \{x = y\}$
\[
F \cap G = \{x = y,\ 2x + z = 0\}
= \{(x,\ x,\ -2x)\} = \operatorname{Vect}(1, 1, -2),
\]
أي مستقيمًا. وللتحقق المتقاطع بغراسمان: $F + G = \R^3$ (لأن المستويين
متمايزان، ومنه فمجموعهما يحتوي مستويًا احتواءً قطعيًا)، ومنه
$\dim(F\cap G) = 2 + 2 - 3 = 1$. والوصفان \omterm{def:b1:vspaces:subspace}{لفضاء
جزئي} --- بالمعادلات، وبالمولّدات --- يجعل كلٌّ منهما عملية
واحدة بديهية: فالمعادلات تتقاطع بالتكديس، والمولّدات
تُجمع بالوصل؛ والتحويل بينهما هو بالضبط ما تعنيه
حلّ جملة خطية (\cref{ch:b1:det}).
\end{example}

\begin{remark}[مزالق شائعة]
\label{rem:b1:findim:pitfalls}
\emph{الأبعاد لا تُجمع على امتداد المجاميع} ما لم يكن المجموع مباشرًا:
فلمستويين من $\R^3$ يكون $\dim(F + G) = 3$، لا $4$؛ فصحّح
دائمًا بحدّ التقاطع (غراسمان).
\emph{الاحتواء يحتاج إلى البُعد \emph{و}الاحتواء}:
فالمقدار $\dim F = \dim G$ وحده لا يعطي $F = G$ أبدًا (فمستقيمان متمايزان
من $\R^2$)؛ وتقتضي حالة المساواة في
\cref{thm:b1:findim:subspaces} أن يكون $F \subseteq G$ أولًا.
\emph{عدّ الوسائط ليس برهانًا بعد}: فالقول «معادلتان في
$\R^4$، ومنه البُعد $2$» يفشل عندما تكون المعادلتان
مرتبطتين (فالمعادلتان $x + y = 0$ و $2x + 2y = 0$ تتركان البُعد $3$)؛
ولا يفصل إلا تعليم فعلي أو حساب \omterm{def:b1:findim:rank}{رتبة}.
\emph{لا تتكلم عن بُعد ما ليس \omterm{def:b1:vspaces:subspace}{فضاءً جزئيًا}}: فمجموعات
حلول الجمل \emph{غير المتجانسة} تفوّت $0$؛ و«بُعدها»
هو بُعد فضاء الحلول المتجانس المرتبط بها
(\cref{ch:b1:det} يجعل هذا دقيقًا).
\emph{البُعد غير المنتهي موجود}: فالمقدار $K[X]$ يحتوي عائلات \omterm{def:b1:vspaces:free}{حرة}
من كل حجم (الوحيدات الحدّ)، ومنه فلا يمكن وجود عائلة مولِّدة
منتهية --- فعبارات مثل قاعدة اثنين من ثلاثة
منتهية البُعد قطعًا وتفشل فشلًا ذريعًا على $K[X]$
(\cref{cor:b1:linmaps:samedim} سيبيّن الشيء نفسه من أجل
التباين والشمول).
\emph{\omterm{def:b1:findim:rank}{الرتبة} عن \omterm{def:b1:vspaces:span}{الفضاء المولَّد} لا عن القائمة}: فتكرار متجهة
أو إعادة الترتيب أو الضرب في ثوابت غير معدومة يبقي \omterm{def:b1:findim:rank}{الرتبة}
على حالها، و $\operatorname{rk} = p$ (وهو عدد المتجهات)
\emph{خاصية يجب البرهان عليها} --- وهي بالضبط الحرية. و
العائلة من $5$ متجهات ورتبتها $2$ تحمل ما قيمته ثلاث متجهات
من الزيادة، وتحدّد الإزاحة
(\cref{ex:b1:findim:rankcomputation}) موضعها صراحةً.
\end{remark}

\begin{remark}[أين يذهب البُعد للعمل]
\label{rem:b1:findim:whereused}
البُعد هو حجة العدّ المفضلة عند الكتاب من الآن فصاعدًا.
فالفصل \cref{ch:b1:linmaps} يبرهن على مبرهنة \omterm{def:b1:findim:rank}{الرتبة}، وهي الصيغة
الدالية لصيغة غراسمان؛ و \cref{ch:b1:matrices}
يحسب الرتب بالتقليص على السطور؛ و \cref{ch:b1:det} يحوّل «$n$
متجهة من $K^n$ تكوّن \omterm{def:b1:vspaces:free}{أساسًا}» إلى عدد واحد غير معدوم. و
تبيّن مسألة نهاية الأسبوع أدناه البُعدَ يقوم ب«\emph{حساب أعداد}»:
فعدّ الأبعاد على \omterm{def:b1:structures:field}{الحقل} $\Q$ يبرهن على عبارات صمم
تبدو منيعة على اليد. وفي مجلّد السنة~الثالثة تفصل
حسابات الأبعاد نفسها، ملطَّفةً بنظرية الزمر، في أيّ مسائل
البناء الكلاسيكية قابل للحلّ --- وتلك القصة نظرية غالوا.
\end{remark}

\begin{remark}[منظورات داخل الكتاب 3]
\label{rem:b1:findim:perspectives}
البُعد هو المقدار المحفوظ في بقية هذا المجلّد،
ويجدر تسمية قوانين الحفظ مسبقًا.
فالفصل \cref{ch:b1:linmaps} يبرهن على $\dim E = \dim\ker u +
\operatorname{rk} u$: أي إن ما يسحقه \omterm{def:b1:logic:map}{التطبيق} الخطي مضافًا إليه ما
يحفظه يساوي المصدر دائمًا. و \cref{ch:b1:det} يلطّف هذا
إلى بنية مجموعات الحلول: فإن $p$ مجهولًا ناقص
$\operatorname{rk} A$ محورًا يترك بُعد
فضاء الحلول، وتظهره إزاحة غاوس وسائطَ
\omterm{def:b1:vspaces:free}{حرة}. و \cref{ch:b1:euclid} \omterm{def:b1:arith:divides}{يقسم} $\dim E = \dim F + \dim
F^\perp$ قسمةً متعامدة، وتنفق مسألة نهاية الأسبوع في
\cref{ch:b1:multivar} هذه الميزانية بالضبط: $n$ نقطة
معطيات، و $2$ وسيطًا مقايَسين، و $n - 2$ بُعدًا للباقي.
وكلما رفض عدٌّ أن يتوازن في فصل لاحق، كان
الخطأ نواةً منسية أو مجموعًا غير مباشر --- فعُد إلى
صيغة غراسمان أولًا.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b1:findim:3}

\begin{exercise}[$\star\star$]\label{exo:b1:findim:5}

$E$ مع $\dim E = 7$. ما القيم الممكنة للمقدار $\dim(F \cap

\begin{exercise}[$\star\star$]\label{exo:b1:findim:6}
لتكن $H = \{P \in \R_n[X] : P(1) = 0\}$. برهن على أن $H$
مستوٍ فائق في $\R_n[X]$ (أي \omterm{def:b1:vspaces:subspace}{فضاء جزئي} بُعده $n$)، وأظهر

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:findim:7}
لتكن $u_1, \dots, u_p$ متجهات رتبتها $r$. برهن على أن حذف
متجهة واحدة يعطي عائلة رتبتها $r$ أو $r - 1$، وأن
إلحاق متجهة يعطي \omterm{def:b1:findim:rank}{رتبة} $r$ أو $r + 1$. واستنتج أن \omterm{def:b1:findim:rank}{الرتبة}
تتغير بمقدار $1$ على الأكثر تحت أيّ إدخال أو حذف مفرد.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:findim:8}
لتكن $F_1 \subseteq F_2 \subseteq \dots \subseteq F_k$ فضاءات جزئية
من $E$ ($\dim E = n$) مع $F_i \neq F_{i+1}$ من أجل كل $i$. برهن على $k
\leq n + 1$. واستنتج أن السلسلة المتزايدة قطعًا من الفضاءات الجزئية
من $\R^n$ طولها $n + 1$ على الأكثر، وأظهر واحدة ذات طول
أكبري.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:findim:9}
ليكن $E$ ذا بُعد $n$ ولتكن $F$ و $G$ مستويين فائقين (بُعدهما
$n - 1$) مع $F \neq G$. احسب $\dim(F \cap G)$. وعمّم: تقاطع
$k$ مستوٍ فائق بُعده $\geq n - k$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:findim:10}
ليكن $E$ \omterm{def:b1:logic:sets}{مجموعة} المتتاليات الحقيقية التي تحقق $u_{n+2} =
3u_{n+1} - 2u_n$ من أجل كل $n$.
\begin{enumerate}
  \item بيّن أن $E$ \omterm{def:b1:vspaces:subspace}{فضاء جزئي} من فضاء المتتاليات، وأن
        متتالية من $E$ تتحدد تمامًا، تحدّدًا خطيًا،
        بالزوج $(u_0, u_1)$؛ واستنتج $\dim E = 2$.
  \item تحقق من أن المتتالية الثابتة $(1)$ والمتتالية الهندسية
        $(2^n)$ تقعان في $E$ وتكوّنان \omterm{def:b1:vspaces:free}{أساسًا} للمقدار $E$.
  \item جد متتالية $E$ ذات $u_0 = 0$ و $u_1 = 1$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:findim:11}
ليكن $E$ ذا بُعد $n$.
\begin{enumerate}
  \item إذا كانت $F, G$ فضاءين جزئيين مع $\dim F + \dim G > n$، فبرهن على
        $F \cap G \neq \{0\}$. وأوضح: فالفضاءان الجزئيان ذوا
        البُعدين $51$ و $50$ من $\R^{100}$ يتقاسمان دائمًا
        متجهةً غير معدومة.
  \item إذا كان $H$ مستويًا فائقًا و $F$ \omterm{def:b1:vspaces:subspace}{فضاءً جزئيًا} مع $F \cap H =
        \{0\}$، فبرهن على $\dim F \leq 1$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:findim:12}
(متتامّ مشترك) لتكن $F, G$ فضاءين جزئيين من $E$ (ذي \omterm{def:b1:findim:def}{بُعد
منته}) مع $\dim F = \dim G$. برهن على أن للمقدارين $F$ و $G$
متتامًّا \emph{مشتركًا}: أي إنه يوجد \omterm{def:b1:vspaces:subspace}{فضاء جزئي} $S$ يحقق $E = F
\oplus S = G \oplus S$. \emph{(استقرِ نزولًا على $\dim F$: فإذا كان $F
\neq G \neq E$، فاختر $x \notin F \cup G$ ---
\cref{exo:b1:vspaces:12} يسمح بذلك --- وانظر في $F \oplus
\operatorname{Vect}(x)$ و $G \oplus \operatorname{Vect}(x)$.)}
\end{exercise}

\section{مسألة: قانون البرج لديدكيند}

\begin{problem}\label{pb:b1:findim:1}
لم يستعمل أيّ شيء في الفصلين 18--19 عن السلالم فوق
بديهيات \omterm{def:b1:structures:field}{الحقل} (\cref{def:b1:structures:field}): ومنه يمكن
أخذ $K = \Q$ وقياس مجموعات من \emph{الأعداد الحقيقية}
بمسطرة البُعد على $\Q$. وتحسب هذه المسألة
بُعد $\Q(\sqrt2, \sqrt3)$، وتبرهن على \emph{\omterm{pb:b1:findim:1}{قانون البرج}
لديدكيند} $\dim_\Q M = \dim_\Q K \cdot \dim_K M$، وتحصد
مبرهنات صمم بعدّ الأبعاد المحض --- بلا
$\varepsilon$ وبلا أرقام عشرية، بالأسس فحسب.\index{قانون البرج}
\index{حقل!حقل جزئي من $\R$}\index{عدد أصمّ}

\medskip
\noindent\textbf{الجزء 1 --- السلالم الناطقة.}
\begin{enumerate}
  \item تحقق من أن $\Q$ \omterm{def:b1:structures:field}{حقل} ومن أن كل تعريف وكل
        برهان في الفصلين 18--19 لا يستعمل إلا بديهيات \omterm{def:b1:structures:field}{حقل}
        السلالم؛ واستنتج أن $\R$ \omterm{def:b1:vspaces:def}{فضاء متجهي} على $\Q$ وأن
        مبرهنة التبادل المساعدة ومبرهنات الأسس و
        صيغة غراسمان تصحّ على $\Q$. وأشِر إلى الخطوة الوحيدة
        من برهان \cref{thm:b1:findim:exchange} التي تُجرى فيها
        قسمة على سلّم غير معدوم.
  \item بيّن أن $(1, \sqrt2)$ \omterm{def:b1:vspaces:free}{حرة} على $\Q$ لكنها مرتبطة على
        $\R$. ضع $\Q(\sqrt2) = \operatorname{Vect}_\Q(1,
        \sqrt2) = \{a + b\sqrt2 : a, b \in \Q\}$؛ فما هو
        $\dim_\Q \Q(\sqrt2)$؟
  \item بيّن أن $\Q(\sqrt2)$ مستقر بالضرب.
        ومن أجل $u = a + b\sqrt2$ ضع $\sigma(u) = a - b\sqrt2$ و
        $N(u) = u\,\sigma(u) = a^2 - 2b^2$. بيّن $\sigma(uv) =
        \sigma(u)\sigma(v)$، واستنتج $N(uv) = N(u)N(v)$، وبيّن أن
        $N(u) \neq 0$ كلما كان $u \neq 0$.
  \item استنتج أن لكل $u \in \Q(\sqrt2)$ غير معدوم مقلوبًا
        في $\Q(\sqrt2)$، وهو $u^{-1} =
        \sigma(u)/N(u)$: أي إن $\Q(\sqrt2)$ \omterm{def:b1:structures:field}{حقل} جزئي من $\R$.
        واحسب $\dfrac1{3 + 2\sqrt2}$ و $\dfrac1{1 + \sqrt2}$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- ضمّ $\sqrt3$.}
\begin{enumerate}[resume]
  \item برهن على أن $\sqrt6 \notin \Q$ \emph{(قارن أُسّ
        $2$ على طرفَي $6q^2 = p^2$، كما في
        \cref{exo:b1:arith:7})}، ثم على أن $\sqrt3 \notin
        \Q(\sqrt2)$ \emph{(ربّع $\sqrt3 = a + b\sqrt2$ و
        ناقش الحالات $ab \neq 0$ و $b = 0$ و $a = 0$)}.
  \item لتكن $M = \operatorname{Vect}_\Q(1, \sqrt2, \sqrt3,
        \sqrt6)$. بيّن أن $M$ مستقر بالضرب
        \emph{(ويكفي جدول لجداءات متجهات \omterm{def:b1:vspaces:free}{الأساس})}.
  \item اكتب $K = \Q(\sqrt2)$. بيّن أن $(1, \sqrt3)$ \omterm{def:b1:vspaces:free}{حرة}
        \emph{على \omterm{def:b1:structures:field}{الحقل} $K$}، واستنتج أن $M = K +
        K\sqrt3$ \omterm{def:b1:vspaces:def}{فضاء متجهي} على $K$ بُعده $2$
        \omterm{def:b1:vspaces:free}{بالأساس} $(1, \sqrt3)$.
  \item برهن على أن $(1, \sqrt2, \sqrt3, \sqrt6)$ \omterm{def:b1:vspaces:free}{حرة} على
        $\Q$، ومنه $\dim_\Q M = 4$. \emph{(جمّع علاقة معدومة
        في صورة $(a + b\sqrt2) + (c + d\sqrt2)\sqrt3 = 0$ وطبّق
        السؤال 7 ثم 2.)}
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- \omterm{pb:b1:findim:1}{قانون البرج}.} ليكن $\Q \subseteq K
\subseteq M \subseteq \R$ حيث $K$ و $M$ حقلان جزئيان، ولتكن $(e_1,
\dots, e_m)$ \omterm{def:b1:vspaces:free}{أساسًا} للمقدار $K$ فضاءً متجهيًا على $\Q$، ولتكن $(f_1,
\dots, f_n)$ \omterm{def:b1:vspaces:free}{أساسًا} للمقدار $M$ فضاءً متجهيًا على $K$.
\begin{enumerate}[resume]
  \item بيّن أن الجداءات $mn$ الممثَّلة بالمقادير $(e_i f_j)$ تولّد $M$ على
        $\Q$.
  \item بيّن أن العائلة $(e_i f_j)$ \omterm{def:b1:vspaces:free}{حرة} على $\Q$.
        \emph{(أعد تنظيم تركيبة معدومة على $\Q$ في صورة $\sum_j
        \bigl(\sum_i \lambda_{ij} e_i\bigr) f_j$، ومعاملاتها
        الداخلية تعيش في $K$.)}
  \item اختم بما يسمّى \emph{\omterm{pb:b1:findim:1}{قانون البرج} لديدكيند}:
        \[
        \dim_\Q M \;=\; \dim_\Q K \,\cdot\, \dim_K M ,
        \]
        وتحقق منه على $\Q \subseteq \Q(\sqrt2) \subseteq M$
        إزاء السؤالين 7 و 8.
  \item (حقول بالمجّان) لتكن $A \subseteq \R$ \omterm{def:b1:vspaces:subspace}{فضاءً جزئيًا}
        على $\Q$ \omterm{def:b1:findim:def}{ذا بُعد منته} يحتوي $1$ و
        مستقرًا بالضرب، وليكن $u \in A$ مع $u \neq
        0$. بيّن أنه إذا كانت $(a_1, \dots, a_d)$ \omterm{def:b1:vspaces:free}{أساسًا} للمقدار $A$،
        فإن $(u a_1, \dots, u a_d)$ أساسٌ للمقدار $A$ مرة أخرى؛
        واستنتج أن للمقدار $u$ مقلوبًا \emph{في $A$}: أي إن $A$
        \omterm{def:b1:structures:field}{حقل} جزئي من $\R$. وأيّ الأسئلة السابقة يستعيدها هذا؟
  \item بيّن أنه من أجل كل $x \in A$ (كما في السؤال 12، $\dim_\Q
        A = d$) تكون العائلة $(1, x, x^2, \dots, x^{d})$ مرتبطة:
        أي إن كل عنصر من $A$ جذرٌ \omterm{def:b1:poly:def}{لكثير حدود} غير معدوم
        بمعاملات ناطقة، من درجة $d$ على الأكثر.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- عدد واحد يولّد كل شيء.}
ضع $s = \sqrt2 + \sqrt3$.
\begin{enumerate}[resume]
  \item احسب \omterm{prop:b1:vspaces:coordinates}{إحداثيات} $s^2$ و $s^3$ و $s^4$ في
        \omterm{def:b1:vspaces:free}{الأساس} $(1, \sqrt2, \sqrt3, \sqrt6)$ للمقدار $M$.
  \item بيّن أن $(1, s, s^2, s^3)$ \omterm{def:b1:vspaces:free}{حرة} على $\Q$، واستنتج
        $\operatorname{Vect}_\Q(1, s, s^2, s^3) = M$: أي إن كل
        عنصر من $M$ كثيرُ حدود ناطق في $s$. وعبّر عن
        $\sqrt2$ و $\sqrt3$ بمثل هذه كثيرات الحدود.
  \item تحقق من $s^4 - 10s^2 + 1 = 0$، وبيّن أن $X^4 - 10X^2
        + 1$ هو \emph{\omterm{def:b1:poly:def}{كثير الحدود} الواحديّ الأصغر درجةً}
        المنعدم عند $s$. وحدّد جذوره الحقيقية الأربعة.
  \item استنتج: $1/s = 10s - s^3$؛ وعيّن هذا العدد. وبيّن
        أن $a + b\sqrt2 + c\sqrt3 + d\sqrt6$ ($a,b,c,d \in \Q$)
        يكون ناطقًا إذا وفقط إذا كان $b = c = d = 0$؛ وعلى
        الخصوص يكون $\sqrt2 + \sqrt3 + \sqrt6$ أصمّ
        (وهذا تقوية للمقدار \cref{exo:b1:reals:7}).
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- الجذر التكعيبي يبقى خارجًا.} ضع $t
= 2^{1/3}$.
\begin{enumerate}[resume]
  \item بيّن أن $X^3 - 2$ ليس له جذر ناطق \emph{(بمحك
        الجذر الناطق في \cref{exo:b1:poly:5}، أو بعدّ
        تقييمات)}؛ واستنتج أن $(1, t)$ \omterm{def:b1:vspaces:free}{حرة} على
        $\Q$.
  \item بيّن أن $(1, t, t^2)$ \omterm{def:b1:vspaces:free}{حرة} على $\Q$. \emph{(إذا قتل
        كثيرُ حدود غير معدوم $P \in \Q[X]$ من درجة $\leq 2$ المقدارَ $t$، فخذ
        واحدًا من أصغر درجة واقسم $X^3 - 2$ عليه،
        \cref{thm:b1:poly:division}؛ واستنتج أن $X^3 - 2$
        سيكون له جذر ناطق.)} واستنتج أن $A =
        \operatorname{Vect}_\Q(1, t, t^2)$ بُعده $3$، و
        مستقر بالضرب، \omterm{def:b1:structures:field}{وحقل} جزئي من $\R$.
  \item برهن على أن $t \notin M$: وإلا لكان $M$ فضاءً متجهيًا
        على \omterm{def:b1:structures:field}{الحقل} $A$، ولفرض \omterm{pb:b1:findim:1}{قانون البرج} أن $3 \mid 4$.
        ومنه فالمقدار $2^{1/3}$ ليس تركيبةً ناطقة
        من $1, \sqrt2, \sqrt3, \sqrt6$.
  \item استنتج أن $t \notin \Q(\sqrt2)$، وأن $t + \sqrt2$
        أصمّ، وأنه لا يوجد عددان ناطقان $a, b$ يحققان
        $2^{1/3} = a + b\sqrt3$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 6 --- كل الحقول الجزئية، وتوليفة.}
\begin{enumerate}[resume]
  \item برهن على قابلية قسمة الدرجات: إذا كان $\Q \subseteq A
        \subseteq B \subseteq \R$ حقولًا جزئية مع $\dim_\Q B$
        منتهيًا، فإن $\dim_\Q A$ \omterm{def:b1:arith:divides}{يقسم} $\dim_\Q B$. وما
        الأبعاد الممكنة للحقول الجزئية من $M$؟
  \item حدّد \emph{كل} الحقول الجزئية من $M$ ذات البُعد $2$
        على $\Q$. \emph{(ردّ الأمر إلى إيجاد $w = a + b\sqrt2
        + c\sqrt3 + d\sqrt6 \in M \setminus \Q$ التي تحقق $w^2 \in
        \Q$: انشر $w^2$ وأعدِم \omterm{prop:b1:vspaces:coordinates}{الإحداثيات} الصمّاء
        الثلاث.)}
  \item عبّر عن $\dfrac1{1 + \sqrt2 + \sqrt3}$ في \omterm{def:b1:vspaces:free}{الأساس} $(1,
        \sqrt2, \sqrt3, \sqrt6)$ \emph{(بالضرب في مرافقات
        محسنة الاختيار)}.
  \item توليفة، في أربع جمل: لماذا يكون البُعد على $\Q$
        قرينةً \emph{حسابية} \omterm{def:b1:logic:sets}{لمجموعة} من الأعداد الحقيقية؛ وماذا
        يفرض تناهي البُعد عن كل عنصر
        (السؤال 13)؛ وكيف أنتجت قاعدة اثنين من ثلاثة
        مقلوبات من العدم (السؤال 12)؛ وماذا يمنع
        \omterm{pb:b1:findim:1}{قانون البرج} (السؤال 20). وسمِّ المبرهنة المبرهن عليها
        في الجزء 3.
\end{enumerate}
\end{problem}
