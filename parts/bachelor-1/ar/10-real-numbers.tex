\chapter{الأعداد الحقيقية}\label{ch:b1:reals}

يقوم التحليل كله على خاصية واحدة تميّز $\R$ من
$\Q$: لكل \omterm{def:b1:logic:sets}{مجموعة} غير خالية محدودة من أعلى \emph{أصغرُ} \omterm{def:b1:reals:bounds}{حاصر
أعلى}. يذكر هذا الفصل هذه الخاصية بدقة، ويستخرج نتائجها
الأولى --- \omterm{thm:b1:reals:archimedes}{خاصية أرخميدس}، \omterm{thm:b1:reals:floor}{ودالة الجزء الصحيح}،
وكثافة الأعداد الناطقة والأعداد الصماء --- ويهيّئ
المفردات (\omterm{def:b1:reals:bounds}{الحد الأعلى}، \omterm{def:b1:reals:bounds}{والحد الأدنى}، وأكبر عنصر، وأصغر عنصر) المستعملة باستمرار ابتداءً من
\cref{ch:b1:seq}.

\section{خاصية الحد الأعلى}

\begin{definition}[الحواصر والحد الأعلى والحد الأدنى]\label{def:b1:reals:bounds}
لتكن $A \subseteq \R$ غير خالية. يكون العدد الحقيقي $M$ \emph{حاصرًا
أعلى}\index{حاصر أعلى} \omterm{def:b1:logic:sets}{للمجموعة} $A$ إذا كان $a \leq M$ من أجل كل $a \in A$؛
وتكون $A$ \emph{محدودة من أعلى} إذا كان لها حاصر أعلى (وكذلك
من أدنى، مع الحواصر الدنيا؛ و\emph{محدودة} تعني الأمرين معًا). و
\emph{أكبر عنصر} في $A$ حاصرٌ أعلى ينتمي إلى $A$.

و\emph{الحد الأعلى} $\sup A$\index{حد أعلى} أصغرُ الحواصر العليا
\omterm{def:b1:logic:sets}{للمجموعة} $A$، إذا وُجد؛ و\emph{الحد الأدنى} $\inf A$\index{حد أدنى} هو
أكبرُ الحواصر الدنيا.
\end{definition}

\begin{theorem}[بديهية تمام $\R$]\label{thm:b1:reals:sup}
$\R$ حقلٌ مرتّب يحتوي $\Q$، وفيه \emph{لكل \omterm{def:b1:logic:sets}{مجموعة} جزئية
غير خالية محدودة من أعلى حدٌّ أعلى}.\index{تمام R@تمام $\R$}
\end{theorem}
\admitted

\begin{remark}
نأخذ هذه الخاصية بديهيةً مؤسِّسة للمقدار $\R$؛ أمّا بناء نموذج (بقطوع
ديدكيند أو بمتتاليات كوشي من الأعداد الناطقة) والبرهان على وحدانيته
فأمينان لكنهما طويلان، ونتركهما لدراسة لاحقة. ولاحظ
أن $\Q$ لا يحقق الخاصية: \omterm{def:b1:logic:sets}{فالمجموعة} $\{x \in \Q : x^2 < 2\}$ محدودة
من أعلى لكن ليس لها أصغر \omterm{def:b1:reals:bounds}{حاصر أعلى} \emph{في $\Q$} --- إذ مرشّحها،
$\sqrt 2$، مفقود (\cref{ex:b1:logic:sqrt2}). وبالانتقال إلى
المقابلات ($\sup(-A) = -\inf A$)، يكون لكل \omterm{def:b1:logic:sets}{مجموعة} غير خالية محدودة من أدنى
حدٌّ أدنى.
\end{remark}

\begin{proposition}[التمييز بالمقدار $\varepsilon$]\label{prop:b1:reals:epsilon}
لتكن $A \neq \emptyset$ محدودة من أعلى وليكن $s \in \R$. عندئذ $s =
\sup A$ إذا وفقط إذا تحقق:
\begin{enumerate}
  \item $s$ \omterm{def:b1:reals:bounds}{حاصر أعلى}: $\forall a \in A$، $a \leq s$؛ و
  \item لا شيء أصغر منه كذلك: $\forall \varepsilon > 0$، $\exists a \in
        A$، $a > s - \varepsilon$.
\end{enumerate}
\end{proposition}

\begin{proof}
إذا كان $s = \sup A$: تحقق (1) بحكم التعريف، وأمّا (2) فلأن $s -
\varepsilon < s$ ليس \omterm{def:b1:reals:bounds}{حاصرًا أعلى}، وهذا بالضبط
وجود $a > s - \varepsilon$. وبالعكس، تقول (1) إن $s$ \omterm{def:b1:reals:bounds}{حاصر
أعلى}؛ وتقول (2) إن لا $t < s$ حاصرٌ أعلى (خُذ $\varepsilon
= s - t$): فيكون $s$ أصغرها.
\end{proof}

\begin{example}\label{ex:b1:reals:supexamples}
$\sup \intoo{0}{1} = 1$، وهو غير مبلوغ (فلا أكبر عنصر)؛
$\sup \intcc{0}{1} = 1 = \max$. ومن أجل $A = \{1 - \frac 1n : n \in
\N^*\}$: يكون $\sup A = 1$ غير مبلوغ؛ و $\inf A = \min A = 0$. وأكبر عنصر،
إذا وُجد، هو \omterm{def:b1:reals:bounds}{الحد الأعلى}؛ وكل فائدة $\sup$ أن يكون
بديلًا حين لا يوجد أكبر عنصر.
\end{example}

\begin{example}[حساب الحدود العليا عمليًا]\label{ex:b1:reals:supcompute}
تمرينان كاملان على \cref{prop:b1:reals:epsilon}.

\emph{\omterm{def:b1:logic:sets}{المجموعة} $A = \{x + \frac1x : x > 0\}$.} من أجل كل $x > 0$،
$x + \frac1x - 2 = \frac{(\,\sqrt x - 1/\sqrt x\,)^2}{1} \geq 0$،
ومنه فالعدد $2$ حاصر أدنى؛ و $2 = 1 + \frac11 \in A$: ولذلك
$\inf A = \min A = 2$، مبلوغًا عند $x = 1$. وأمّا من أعلى \omterm{def:b1:logic:sets}{فالمجموعة} $A$
غير محدودة (إذ يمكن للمقدار $x + \frac1x > x$ أن يتجاوز أيّ $M$ بأخذ
\cref{thm:b1:reals:archimedes}): \omterm{def:b1:reals:bounds}{فالحد الأعلى} $\sup A$ غير موجود في $\R$
(وهو $+\infty$ في $\overline\R$).

\emph{\omterm{def:b1:logic:sets}{المجموعة} $B = \bigl\{\frac{m}{m + n} : m, n \in
\N^*\bigr\}$.} كل عنصر يقع في $\intoo{0}{1}$، ومنه فالعددان $0$ و
$1$ حاصران. وليس أيّ منهما مبلوغًا: إذ $\frac{m}{m+n} = 1$ يفرض
$n = 0$. وأمّا \omterm{def:b1:reals:bounds}{الحد الأعلى}، فثبّت $n = 1$ ودع $m$ يكبر:
$\frac{m}{m+1} = 1 - \frac{1}{m+1} > 1 - \varepsilon$ ما إن يكون
$m + 1 > \frac1\varepsilon$ (أرخميدس): فيكون $\sup B = 1$.
وبالتناظر ($m = 1$ و $n$ كبير)، يكون $\inf B = 0$. والفكرة
النافذة: لتثبيت حدٍّ أعلى يكفي \emph{مسار وحيد الوسيط}
مُحسَن الاختيار داخل \omterm{def:b1:logic:sets}{المجموعة} --- وهو هنا المسار $n = 1$ ---
ولا يطلب التمييز بالمقدار $\varepsilon$ أكثر من ذلك.
\end{example}

\begin{example}[مرآة الحد الأدنى]\label{ex:b1:reals:infmirror}
\omterm{def:b1:reals:bounds}{للحد الأدنى} تمييزُه بالمقدار $\varepsilon$، ويُستخرج
من \cref{prop:b1:reals:epsilon} عبر $\inf A = -\sup(-A)$:
فيكون $i = \inf A$ إذا وفقط إذا حصر $i$ \omterm{def:b1:logic:sets}{المجموعة} $A$ من أدنى وكان، من أجل كل
$\varepsilon > 0$، عنصرٌ $a \in A$ يحقق $a < i + \varepsilon$. وإليك
تمرينًا على الحاصرين معًا: لتكن
\[
A = \Bigl\{(-1)^n + \frac1n : n \in \N^*\Bigr\}
= \Bigl\{0,\ \tfrac32,\ -\tfrac23,\ \tfrac54,\ -\tfrac45,\
\dots\Bigr\} .
\]
تعطي الأدلة الزوجية $1 + \frac1n \leq \frac32$، مع المساواة عند
$n = 2$: ولأن القيم ذات الأدلة الفردية هي كذلك $\leq 0 <
\frac32$، نحصل على $\sup A = \max A = \frac32$. وتعطي الأدلة الفردية
$-1 + \frac1n > -1$، متناقصةً نحو $-1$: فكل عنصر من
$A$ هو $> -1$، ويُتجاوز $-1 + \varepsilon$ بالمقدار $-1 +
\frac1n$ من أجل $n > \frac1\varepsilon$ فردي: فيكون $\inf A = -1$ غير
مبلوغ. \omterm{def:b1:logic:sets}{مجموعة} واحدة وأربعة سلوكات معروضة: حدٌّ أعلى
هو أكبر عنصر، وحدٌّ أدنى ليس أصغر عنصر.
\end{example}

\begin{remark}[مزالق شائعة مع الحد الأعلى والحد الأدنى]
أربعة أخطاء تستأثر بأكثر النقاط الضائعة. (أ) \emph{الخلط بين
$\sup$ و $\max$}: فليس على $\sup A$ أن ينتمي إلى $A$؛ ولا تكتب $\max$
إلا بعد إظهار عنصر من $A$ يكون \omterm{def:b1:reals:bounds}{حاصرًا أعلى}.
(ب) \emph{تمرير المتراجحات القطعية إلى \omterm{def:b1:reals:bounds}{الحد الأعلى}}: فإذا كان $a <
b$ من أجل كل $a \in A$، فلا يجوز إلا استنتاج $\sup A \leq b$ ---
وشاهدُه $A = \intoo{0}{1}$ و $b = 1$. (ج) \emph{كتابة $\sup A$
قبل التحقق من المشروعية}: فالرمز يقتضي أن تكون $A$ غير خالية
ومحدودة من أعلى (\cref{met:b1:reals:supproofs})؛ وأمّا $\sup \emptyset$
و $\sup \N$ فغير معرَّفين في $\R$ (واصطلاحات
$\overline\R$ فعلٌ منفصل صريح). (د) \emph{عمليات
المجموعات}: $\sup(A \cup B) = \max(\sup A, \sup B)$ دائمًا، لكن
لا شيء عامًّا يصحّ من أجل $A \cap B$ --- فقد يكون خاليًا، بل
حتى إن لم يكن كذلك فقد يقع $\sup(A \cap B)$ دون
$\min(\sup A, \sup B)$ بكثير: خذ $A = \{0, 2\}$ و $B = \{0, 3\}$،
حيث $\sup(A \cap B) = 0$.
\end{remark}

\begin{example}[للمجموعات المنتهية أكبر عنصر --- مبرهنة مساعدة تُستعمل صامتة]\label{ex:b1:reals:finitemax}
لكل \omterm{def:b1:counting:card}{مجموعة منتهية} غير خالية $F \subseteq \R$ أكبرُ عنصر (وأصغرُ
عنصر). وذلك بالاستقراء على \omterm{def:b1:counting:card}{عدد العناصر}: \omterm{def:b1:logic:sets}{فالمجموعة} الأحادية
$\{a\}$ لها $\max = a$؛ وإذا صحّ الادعاء من أجل المجموعات ذات $n$ عنصرًا
وكان \omterm{def:b1:logic:sets}{للمجموعة} $F$ عددُ $n + 1$ عنصرًا، فاختر أيّ $a \in F$: \omterm{def:b1:logic:sets}{للمجموعة} $F
\setminus \{a\}$ أكبرُ عنصر $m$، ويكون $\max F$ هو $m$ إذا كان $a
\leq m$، وإلا فهو $a$. ولا تدخل هنا خاصية التمام --- فهذا ترتيب
محض مع استقراء، صحيح في $\Q$ أصلًا --- ومع ذلك تستحق المبرهنة المساعدة
صياغةً أمينة واحدة لأن البراهين الآتية تستدعيها
صامتة: بناء الجزء الصحيح أدناه («\omterm{def:b1:logic:sets}{لمجموعة} أعداد صحيحة
محبوسة في مجال منته أكبرُ عنصر»)، وكل حاصر
$\max(\abs{u_0}, \dots, \abs{u_{N-1}}, \dots)$ في
\cref{ch:b1:seq}، وكل «خذ أكبر الأعداد $\delta$ المنتهية العدد» في \cref{ch:b1:continuity}. وأمّا المجموعات غير المنتهية فهي حيث
يموت أكبر عنصر ويحلّ \omterm{def:b1:reals:bounds}{الحد الأعلى} محله: ولأجل الحالة غير المنتهية
وُجد هذا الفصل.
\end{example}

\begin{theorem}[خاصية أرخميدس]\label{thm:b1:reals:archimedes}
من أجل كل $x \in \R$ يوجد $n \in \N$ يحقق $n > x$. وبكيفية مكافئة:
من أجل كل $\varepsilon > 0$ و $y > 0$، يتجاوز مضاعفٌ ما $n\varepsilon$
العددَ $y$.\index{خاصية أرخميدس}
\end{theorem}

\begin{proof}
افترض العكس: فيكون عددٌ $x$ \omterm{def:b1:reals:bounds}{حاصرًا أعلى} \omterm{def:b1:logic:sets}{للمجموعة} $\N$. عندئذ يوجد $s = \sup \N$
(\cref{thm:b1:reals:sup}). وحسب
\cref{prop:b1:reals:epsilon}~(2) مع $\varepsilon = 1$، يوجد $n
\in \N$ يحقق $n > s - 1$؛ لكن عندئذ $n + 1 \in \N$ و $n + 1 > s$،
وهذا يناقض كون $s$ \omterm{def:b1:reals:bounds}{حاصرًا أعلى}. وأمّا الصورة الثانية،
فليكن $\varepsilon > 0$ و $y > 0$: تعطي الصورة الأولى مطبَّقةً على $x
= \frac{y}{\varepsilon}$ عددًا $n \in \N$ يحقق $n >
\frac{y}{\varepsilon}$، وبالضرب في $\varepsilon > 0$
(وهو يحفظ المتراجحات القطعية) نجد $n\varepsilon > y$.
وبالعكس، تستعيد الصورة الثانية مع $\varepsilon = 1$ و $y = x$
الصورةَ الأولى من أجل $x > 0$، ويعالج $n = 1$ حالة $x \leq 0$:
فالعبارتان متكافئتان تمامًا.
\end{proof}

\begin{example}[أرخميدس في العمل]\label{ex:b1:reals:archimedeswork}
ثلاثة استعمالات مباشرة، تُحتاج باستمرار فيما بعد. (أ) \emph{لا
عدد حقيقي موجب يقع دون كل $\frac1n$}: فإذا كان $0 < \varepsilon$،
فاختر $n > \frac1\varepsilon$؛ عندئذ $\frac1n < \varepsilon$. وبعبارة
أخرى، لا يحوي $\R$ لا متناهيات في الصغر --- فالقول غير الصوري
«يصير $\frac1n$ صغيرًا كما نشاء» هو هذه المبرهنة بالضبط.
(ب) \emph{عتبات صريحة}: كم يجب أن يكبر $n$ حتى يكون
$\frac{1}{n^2} \leq 10^{-6}$؟ يكفي أن يكون $n \geq 10^3$
--- فأرخميدس يضمن وجود مثل هذا $n$، والجبر
يحدّد موضعه. (ج) \emph{القوى تغلب أيّ حاصر}: $2^n \geq n +
1$ (بالاستقراء)، ومنه فمن أجل كل $M$ تتجاوز قوةٌ للعدد $2$ المقدارَ $M$:
وهذا النمو الهندسي هو المستعمل من أجل الأعداد الثنائية في
\cref{exo:b1:reals:8}. والفكرة النافذة: \omterm{thm:b1:reals:archimedes}{خاصية
أرخميدس} هي الرخصة وراء كل \omterm{def:b1:logic:statement}{عبارة} من نوع «خذ
$n$ كبيرًا بما يكفي» --- ومن الآن فصاعدًا نستعمل هذه \omterm{def:b1:logic:statement}{العبارة} بحرية،
وهذا المثال تسويغها لمرة واحدة.
\end{example}

\begin{theorem}[دالة الجزء الصحيح]\label{thm:b1:reals:floor}
من أجل كل $x \in \R$ يوجد عدد صحيح واحد بالضبط، هو
\emph{الجزء الصحيح}\index{دالة الجزء الصحيح} $\lfloor x \rfloor$، يحقق
\[
\lfloor x \rfloor \leq x < \lfloor x \rfloor + 1 .
\]
\end{theorem}

\begin{proof}
\emph{الوجود.} \omterm{def:b1:logic:sets}{المجموعة} $E = \{k \in \Z : k \leq x\}$ غير خالية:
فحسب \cref{thm:b1:reals:archimedes} يوجد $m \in \N$ يحقق $m > -x$،
وعندئذ $-m < x$، ومنه $-m \in E$. وهي محدودة من أعلى (بأيّ عدد صحيح
$n > x$، وهو موجود للسبب نفسه)، ومنه، لكونها \omterm{def:b1:logic:sets}{مجموعة} أعداد
صحيحة محبوسة في المجال المنتهي $\intint{-m}{n}$، لها
أكبرُ عنصر $k = \max E$. عندئذ $k \leq x$، ويعني $k + 1 \notin E$
أن $x < k + 1$.

\emph{الوحدانية.} إذا حقق $k$ و $k'$ كلاهما المتراجحتين،
فإن $k \leq x < k' + 1$ يعطي $k \leq k'$، وبالتناظر $k'
\leq k$.
\end{proof}

\begin{example}[الجزء الصحيح عمليًا]\label{ex:b1:reals:floorpractice}
$\lfloor 3.7 \rfloor = 3$ و $\lfloor 5 \rfloor = 5$ و $\lfloor
-3.7 \rfloor = -4$: فالجزء الصحيح ينزل \emph{إلى أسفل}، لا نحو $0$.
ونتيجتان للوحدانية في \cref{thm:b1:reals:floor} سوف
نستعملهما صامتين. الأولى، من أجل $n \in \Z$،
\[
\lfloor x + n \rfloor = \lfloor x \rfloor + n ,
\]
لأن $\lfloor x \rfloor + n$ عدد صحيح يحقق المتراجحتين
المعرِّفتين للمقدار $x + n$ --- ولا يحققهما إلا عدد صحيح واحد.
والثانية، $\lfloor \, \cdot \, \rfloor$ غير متناقصة: فإذا كان $x \leq
y$ فإن $\lfloor x \rfloor \leq x \leq y < \lfloor y \rfloor + 1$،
وكل عدد صحيح $< \lfloor y \rfloor + 1$ هو $\leq \lfloor y
\rfloor$. لكن احذر أن $\lfloor 2x \rfloor \neq 2\lfloor
x \rfloor$ عمومًا: فالعدد $x = 0.6$ يعطي $\lfloor 1.2 \rfloor = 1
\neq 0 = 2\lfloor 0.6 \rfloor$.

وأمّا ما \emph{يصحّ} فمتطابقة محسوبة تستحق الحفظ (وهي متطابقة إرميت،
في أبسط حالاتها): من أجل كل عدد حقيقي $x$،
\[
\lfloor x \rfloor + \Bigl\lfloor x + \frac12 \Bigr\rfloor
= \lfloor 2x \rfloor .
\]
اكتب $x = \lfloor x\rfloor + u$ حيث $u \in \intco{0}{1}$ وافصل
حالتين. إذا كان $u < \frac12$: فالطرف الأيسر هو $\lfloor
x\rfloor + \lfloor x\rfloor = 2\lfloor x\rfloor$، و $2x =
2\lfloor x\rfloor + 2u$ مع $2u \in \intco{0}{1}$، ومنه فالطرف
الأيمن هو $2\lfloor x\rfloor$ كذلك. وإذا كان $u \geq \frac12$: فالطرف
الأيسر هو $\lfloor x\rfloor + (\lfloor x\rfloor + 1)$، ويجعل $2u \in
\intco{1}{2}$ الطرفَ الأيمن $2\lfloor x\rfloor + 1$. والفكرة
النافذة: $\lfloor x + \frac12\rfloor$ هو \emph{تدوير}
$x$ إلى أقرب عدد صحيح، ومنه فالمتطابقة تقول
إن الجزء الصحيح مضافًا إليه التدوير يساوي الجزء الصحيح للضِّعف ---
وفصل الحالات على الجزء الكسري $u$ هو التقنية المعيارية
وراء كل متطابقة للجزء الصحيح
(وتجري عليه \cref{exo:b1:reals:2,exo:b1:reals:3} كذلك).
\end{example}

\begin{theorem}[كثافة $\Q$ و $\R \setminus \Q$]\label{thm:b1:reals:density}
بين كل عددين حقيقيين $x < y$ يقع عدد ناطق وعدد
أصمّ.\index{كثافة}
\end{theorem}

\begin{proof}
\emph{عدد ناطق.} حسب \cref{thm:b1:reals:archimedes}، اختر $n \in
\N^*$ يحقق $n > \frac{1}{y - x}$، فيكون $ny - nx > 1$. وليكن $m = \lfloor
nx \rfloor + 1$. من جهة، $nx < \lfloor nx \rfloor + 1 = m$
(\cref{thm:b1:reals:floor})؛ ومن جهة أخرى، $m = \lfloor nx \rfloor +
1 \leq nx + 1 < ny$. وبالقسمة على $n$: $x < \frac mn < y$.

\emph{عدد أصمّ.} طبّق النقطة السابقة على الزوج $x -
\sqrt 2 < y - \sqrt 2$: فيقع بينهما عدد ناطق $q$، وعندئذ
$q + \sqrt 2 \in \intoo{x}{y}$ عدد أصمّ (إذ لو كان $q + \sqrt 2$
ناطقًا لكان $\sqrt 2$ كذلك).
\end{proof}

\begin{example}[تنفيذ برهان الكثافة]\label{ex:b1:reals:densityrun}
البرهان خوارزمية؛ فلننفّذها على $x = 1.414$ و
$y = \sqrt 2$. ولأن $1.4142^2 = 1.99996164 < 2$، لدينا $\sqrt 2
> 1.4142$، ومنه $y - x > 0.0002$ و $\frac{1}{y - x} < 5000$: فاختيار
$n = 5000$ مشروع. عندئذ $nx = 7070$، ومنه $m = \lfloor
7070 \rfloor + 1 = 7071$، والعدد الناطق الناتج هو
\[
\frac{m}{n} = \frac{7071}{5000} = 1.4142,
\qquad
1.414 < 1.4142 < \sqrt 2 .
\]
والفكرة النافذة: لا يحتاج البرهان إلا إلى $n$ أكبر قليلًا من
$\frac{1}{y-x}$، ويعيد أولَ مضاعف للمقدار $\frac 1n$
بعد $x$. فالكثافة ليست معجزة مجرّدة --- بل هي قسمةٌ
مطوّلة متنكّرة، وهو موضوع يُطوَّر مطوّلًا في مسألة نهاية
الأسبوع (\cref{pb:b1:reals:1}).
\end{example}

\begin{remark}[أين تُستعمل خاصية التمام تاليًا]
\Cref{thm:b1:reals:sup} هي البديهية الوحيدة غير الجبرية في هذا
الكتاب، وكل مبرهنة وجود في التحليل هي تلك البديهية
في ثياب مختلفة: مبرهنة التقارب الرتيب
(\cref{ch:b1:seq})، ومبرهنة بولتزانو--فايرشتراس
(\cref{ch:b1:topology})، ومبرهنتا القيم الوسطى والقيم
الحدّية (\cref{ch:b1:continuity})، وتعريف التكامل نفسه
حدًّا أعلى للمجاميع الدنيا (\cref{ch:b1:integration}).
ويبني مجلّد السنة الثالثة نظرية القياس وفضاءات هيلبرت على
البديهية الوحيدة نفسها. وحين ينتج برهانٌ في الفصول الآتية
عددًا حقيقيًا من العدم، فابحث عن \omterm{def:b1:reals:bounds}{الحد الأعلى} المستتر.
\end{remark}

\begin{remark}[بين التقطّع والكثافة]
تجلس $\Z$ و $\Q$ على طرفي نقيض داخل $\R$: فحول كل
عدد صحيح فجوة طولها $1$ لا تحوي عددًا صحيحًا آخر
(التقطّع --- وهو ما يجعل الجزء الصحيح معرَّفًا جيدًا)،
بينما بين كل عددين حقيقيين أعدادٌ ناطقة لا تُحصى
(الكثافة). وممّا يلفت أن لا شيء بين الحالين من أجل
\emph{الزمر الجزئية الجمعية} للمقدار $\R$: إذ يبرهن \cref{exo:b1:reals:9} على أن
مثل هذه \omterm{def:b1:structures:subgroup}{الزمرة الجزئية} إمّا من الشكل $\alpha\Z$ (متقطّعة) وإمّا
كثيفة --- وهي ثنائية تحرّك كثافة $\{\sin n\}$ في
\cref{ch:b1:seq} والوحش البنّاء في
\cref{pb:b1:continuity:1}. وأمّا المجموعات العامة فتخلط
السلوكين بحرية بالطبع: \omterm{def:b1:logic:sets}{فالمجموعة} $\Z \cup \Q\cap\intcc{0}{1}$ متقطّعة بعيدًا
وكثيفة في الوسط.
\end{remark}

\begin{method}[البرهان على المساواة بالحد الأعلى والحد الأدنى]\label{met:b1:reals:supproofs}
للبرهان على $\sup A = s$: تحقق من أن $s$ يحصر $A$ من أعلى، ثم أنتج،
من أجل كل $\varepsilon > 0$ (أو من أجل متتالية $\varepsilon = \frac
1n$)، عنصرًا من $A$ فوق $s - \varepsilon$. ولمقارنة الحدود العليا،
استعمل: $A \subseteq B \implies \sup A \leq \sup B$؛ ومن أجل كل $a, b$:
$\sup(A + B) = \sup A + \sup B$، حيث $A + B = \{a + b\}$
(\cref{exo:b1:reals:5}). ولا تكتب أبدًا $\sup A$ قبل أن تعلم أن $A$
غير خالية ومحدودة من أعلى.
\end{method}

\section{الفترات}

\begin{proposition}[تمييز الفترات]\label{prop:b1:reals:intervals}
تكون \omterm{def:b1:logic:sets}{المجموعة} الجزئية $I \subseteq \R$ \emph{فترة}\index{فترة} (من
الأنواع المألوفة $\intoo{a}{b}$ و $\intcc{a}{b}$ و $\intco{a}{b}$
و $\intoc{a}{b}$، وأنصاف المستقيمات، و $\R$، و $\emptyset$، والمجموعات الأحادية) إذا
وفقط إذا كانت \emph{محدَّبة}:
\[
\forall x, y \in I,\ \forall z \in \R, \quad
x \leq z \leq y \implies z \in I .
\]
\end{proposition}

\begin{proof}
كل نوع من الأنواع المذكورة محدَّب بجلاء. وبالعكس، لتكن $I$ محدَّبة
وغير خالية. ضع $a = \inf I$ إذا كانت $I$ محدودة من أدنى، وإلا فضع $a =
-\infty$؛ وكذلك $b = \sup I$ أو $+\infty$. ونحن ندّعي $\intoo{a}{b}
\subseteq I \subseteq \intcc{a}{b}$ (مع الاصطلاحات البديهية عند
$\pm\infty$). والاحتواء الثاني هو تعريف الحواصر. وأمّا
الأول، فليكن $z \in \intoo{a}{b}$: بما أن $z > a$، فليس $z$ حاصرًا
أدنى (أو أن $a = -\infty$)، ومنه فعنصر ما $x \in I$ يحقق $x < z$؛
وكذلك عنصر ما $y \in I$ يحقق $y > z$؛ ويضع التحدّب $z \in I$.

ويبقى أن نقرأ النوع من الاحتواء المزدوج
$\intoo{a}{b} \subseteq I \subseteq \intcc{a}{b}$: فالمجموعات
المحشورة بين \omterm{prop:b1:reals:intervals}{فترة} مفتوحة وغلقها لا تختلف عن
$\intoo{a}{b}$ إلا بحضور الطرفين (المنتهيين)
أو غيابهما. وصراحةً: إذا كان $a, b \in \R$، فالإمكانات الأربع
من أجل $(a \in I,\ b \in I)$ تعطي $\intoo{a}{b}$ و $\intco{a}{b}$
و $\intoc{a}{b}$ و $\intcc{a}{b}$ (بما فيها الحالتان المنحلّتان
$a = b$: \omterm{def:b1:logic:sets}{مجموعة} أحادية إذا كان $a \in I$)؛ وإذا كان $a = -\infty$ و $b \in
\R$، فنحصل على $\intoo{-\infty}{b}$ أو $\intoc{-\infty}{b}$؛
وبالتناظر من أجل $a \in \R$ و $b = +\infty$؛ ويعطي $a = -\infty$ و
$b = +\infty$ أن $I = \R$. فكل حالة مذكورة في القائمة: وانتهى.
\end{proof}

\begin{remark}[لماذا التحدّب هو الاختبار الصحيح]
تحوّل القضية تعريفًا \emph{هندسيًا} (قائمةً من عشرة
أشكال) إلى اختبار \emph{منطقي في سطر واحد}، والاختبار
هو ما يُستعمل فعلًا: فللبرهان على أن \omterm{def:b1:logic:sets}{مجموعة} ما \omterm{prop:b1:reals:intervals}{فترة}،
لا تطارد أبدًا أيَّ الأشكال العشرة هي --- بل تحقق من التحدّب
ودع القضية تفرز النوع. وسوف تُصاغ مبرهنة القيم
الوسطى في \cref{ch:b1:continuity} على هذه الكيفية بالضبط
(«صورة \omterm{prop:b1:reals:intervals}{فترة} بدالة متصلة فترةٌ»)، وينتج برهانها
التحدّبَ لا الشكل.
\end{remark}

\begin{remark}[المستقيم الحقيقي الموسَّع]
من المريح أن نضمّ رمزين ونعمل في $\overline\R = \R
\cup \{-\infty, +\infty\}$، مع الاصطلاحين $\sup A = +\infty$
حين لا تكون $A$ محدودة من أعلى و $\sup \emptyset = -\infty$. عندئذ يكون
\emph{لكل} \omterm{def:b1:logic:sets}{مجموعة} جزئية من $\R$ حدٌّ أعلى في $\overline\R$ --- وهي
راحة ترميزية تُستعمل بحرية من أجل النهايات في \cref{ch:b1:seq}.
\end{remark}

\begin{example}[الحساب في $\overline\R$]\label{ex:b1:reals:extendedcompute}
مع سريان الاصطلاحات: $\sup \Z = +\infty$ و $\inf \Z =
-\infty$؛ ومن أجل $A = \{n + (-1)^n n : n \in \N\} = \{0, 4, 0, 8,
\dots\} \cup \{0\}$، يكون $\sup A = +\infty$ (فالحدود الزوجية $2n$ غير
محدودة) و $\inf A = \min A = 0$؛ و $\sup\emptyset =
-\infty \leq \inf\emptyset = +\infty$ --- وهي \omterm{def:b1:logic:sets}{المجموعة} الوحيدة التي
حدّها الأعلى \emph{أصغر} من حدّها الأدنى، وهو تذكير بأن
الاصطلاحات مختارة لتجعل $\sup$ متزايدًا و $\inf$
متناقصًا بالنسبة إلى الاحتواء:
\[
A \subseteq B \implies \sup A \leq \sup B
\quad\text{و}\quad \inf A \geq \inf B ,
\]
فيصحّان الآن دون شرط عدم الخلوّ. وأمّا ما \emph{لا}
توفره الاصطلاحات فهو الحساب: إذ يبقى $+\infty + (-\infty)$ و $0
\times (+\infty)$ غير معرَّفين، وعلى كل معالجة
جبرية للحدود العليا أن تتحقق أولًا من أنها لا تكوّنهما أبدًا.
فالمستقيم الموسَّع مسك دفاتر، لا نظام أعداد.
\end{example}

\begin{example}[الحد الأعلى الذي أفلت من $\Q$]\label{ex:b1:reals:supescape}
عُد إلى \omterm{def:b1:logic:sets}{مجموعة} الملاحظة الافتتاحية، $A = \{x \in \Q : x^2 <
2\}$، واحسب حدّها الأعلى \emph{في $\R$}. فهي غير خالية
($1 \in A$) ومحدودة من أعلى بالعدد $1.5$ (إذ إن $x > 1.5$ يعطي $x^2 >
2.25 > 2$)، ومنه يوجد $s = \sup A$. وندّعي أن $s = \sqrt 2$ (وهو العدد
الحقيقي المبنيّ في \cref{exo:b1:reals:12}). الحصر من أعلى: كل
$a \in A$ يحقق $a < \sqrt2$ --- فهذا جليّ من أجل $a \leq 0$،
وأمّا من أجل $a > 0$ فإن $a \geq \sqrt2$ يعطي $a^2 \geq 2$.
ولا شيء أصغر يفي بالغرض: فمن أجل $t < \sqrt2$، تعطي الكثافة
(\cref{thm:b1:reals:density}) عددًا ناطقًا $q$ يحقق
$\max(1, t) < q < \sqrt 2$، وعندئذ $q^2 < 2$، ومنه $q \in A$
يتجاوز $t$. وحسب \cref{prop:b1:reals:epsilon}، $s = \sqrt2
\notin \Q$. والفكرة النافذة: ليس على \omterm{def:b1:reals:bounds}{الحد الأعلى} \omterm{def:b1:logic:sets}{لمجموعة} من
الأعداد الناطقة أن يكون ناطقًا --- فخاصية التمام هي بالضبط
الوعد بأن $\R$، خلافًا للمقدار $\Q$، لا يدع حدًّا أعلى يفلت أبدًا؛
وهذا المثال هو ملاحظة الفصل الافتتاحية، مبرهنًا عليها الآن
بدل الإشارة إليها.
\end{example}

\begin{remark}[منظورات داخل هذا المجلّد]
لأدوات الفصل الثلاث مسارات متمايزة فيما يأتي.
\omterm{def:b1:reals:bounds}{فالحد الأعلى} يحكم شطر التحليل: النهايات الرتيبة
(\cref{ch:b1:seq})، وتعريف التكامل نفسه
(\cref{ch:b1:integration})، وفي هندسة
\cref{ch:b1:euclid} المسافةَ من نقطة إلى فضاء جزئي ---
وهي حدٌّ أدنى يحوّله الإسقاط المتعامد إلى أصغر عنصر. وأمّا
\omterm{thm:b1:reals:floor}{دالة الجزء الصحيح} فتعود حيثما التقى المتقطّع
بالمتصل: نشر الأرقام (مسألة نهاية الأسبوع في هذا الفصل)،
ومقاربة ديريكليه بمبدأ الأدراج
(\cref{pb:b1:derivative:1})، ومقارنات المجاميع بالتكاملات
(\cref{ch:b1:series}). وأمّا حجج الكثافة فترتقي إلى طريقة
في \cref{ch:b1:continuity}: إذ لا تحتاج متطابقةٌ بين دالتين متصلتين
إلا إلى التحقق على $\Q$ --- ونصف معادلة كوشي
الدالية (\cref{pb:b1:continuity:1}) هو هذه الحركة بالضبط. وحين
تشكّ في مصدر عبارات الوجود في برهان من هذا المجلّد،
فالجواب دائمًا تقريبًا: هذا الفصل.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b1:reals:1}
حدّد (مع البراهين) \omterm{def:b1:reals:bounds}{الحد الأعلى} \omterm{def:b1:reals:bounds}{والحد الأدنى} وأكبر عنصر وأصغر عنصر --- حين توجد --- من أجل:
\[
A = \Bigl\{\frac{1}{n} : n \in \N^*\Bigr\},
\qquad
B = \Bigl\{\frac{(-1)^n n}{n+1} : n \in \N\Bigr\},
\qquad
C = \{x \in \R : x^2 < 3\}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:reals:2}
برهن على أن من أجل كل $x, y \in \R$: $\lfloor x \rfloor + \lfloor y
\rfloor \leq \lfloor x + y \rfloor \leq \lfloor x \rfloor + \lfloor y
\rfloor + 1$، وعلى أن الحاصرين مبلوغان.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:reals:3}
برهن على أن من أجل كل $x \in \R$ و $n \in \N^*$:
$\Bigl\lfloor \frac{\lfloor nx \rfloor}{n} \Bigr\rfloor = \lfloor x
\rfloor$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:reals:4}
لتكن $A \subseteq B$ مجموعتين جزئيتين غير خاليتين من $\R$، مع $B$ محدودة. برهن على
$\inf B \leq \inf A \leq \sup A \leq \sup B$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:5}
من أجل $A, B \subseteq \R$ غير خاليتين ومحدودتين، عرّف $A + B = \{a + b : a
\in A,\ b \in B\}$ و $-A = \{-a : a \in A\}$. برهن على:
\[
\sup(A + B) = \sup A + \sup B,
\qquad
\sup(-A) = -\inf A .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:6}
لتكن $f, g \colon E \to \R$ دالتين محدودتين. برهن على
\[
\sup_{x \in E}\, \bigl(f(x) + g(x)\bigr) \leq \sup_{x \in E} f(x) +
\sup_{x \in E} g(x),
\]
وأعطِ مثالًا تكون فيه المتراجحة قطعية. ولماذا لا يناقض هذا
\cref{exo:b1:reals:5}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:7}
برهن على أن $\sqrt 2 + \sqrt 3$ عدد أصمّ. \emph{(ربّعه واستعمل
كون $\sqrt 6$ أصمّ، ويُبرهَن على ذلك عبر
\cref{exo:b1:arith:7}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:8}
برهن على أن \omterm{def:b1:logic:sets}{مجموعة} الأعداد الناطقة الثنائية $D = \bigl\{\frac{m}{2^n} : m \in \Z,\ n \in
\N\bigr\}$ كثيفة في $\R$: أي أن بين كل عددين
حقيقيين يقع عدد ناطق ثنائي.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:reals:9}
لتكن $G$ \omterm{def:b1:structures:subgroup}{زمرة جزئية} من $(\R, +)$ مع $G \neq \{0\}$. ضع $\alpha =
\inf\,(G \cap \intoo{0}{+\infty})$. برهن على:
\begin{enumerate}
  \item إذا كان $\alpha > 0$، فإن $G = \alpha\Z$؛
  \item إذا كان $\alpha = 0$، فإن $G$ كثيفة في $\R$.
\end{enumerate}
واستنتج أن $\Z + \sqrt 2\,\Z$ كثيفة في $\R$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:reals:10}
من أجل مجموعتين غير خاليتين $A, B$ من الأعداد الحقيقية الموجبة، ليكن $AB = \{ab : a \in
A, b \in B\}$. برهن على $\sup(AB) = \sup A \cdot \sup B$ (في الحالة المحدودة)،
وبيّن بمثال أن الإيجابية جوهرية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:reals:11}
من أجل \omterm{def:b1:logic:sets}{مجموعة} غير خالية ومحدودة $A \subseteq \R$، عرّف
\emph{القطر}
\[
\operatorname{diam} A = \sup\,\{\abs{a - a'} : a, a' \in A\} .
\]
برهن على أن $\operatorname{diam} A = \sup A - \inf A$، وعلى أن
$\intcc{\inf A}{\sup A}$ أصغرُ \omterm{prop:b1:reals:intervals}{فترة} مغلقة
تحتوي $A$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:reals:12}
ليكن $y > 0$ ولتكن $E = \{x \geq 0 : x^2 \leq y\}$. برهن على أن $E$
غير خالية ومحدودة من أعلى، وعلى أن $s = \sup E$ يحقق $s^2 =
y$ \emph{(استبعد $s^2 < y$ و $s^2 > y$ بإظهار، في كل
حالة، عدد صغير $h > 0$ يناقض تعريف
\omterm{def:b1:reals:bounds}{الحد الأعلى})}. واستنتج أن لكل $y > 0$ جذرًا تربيعيًا وحيدًا
$\sqrt y > 0$ وأن $y \mapsto \sqrt y$ متزايدة على
$\intoo{0}{+\infty}$.
\end{exercise}

\section{مسألة: نشر الأرقام وإيقاع الأعداد الناطقة}

\begin{problem}[{مسألة نهاية الأسبوع --- النشر في الأساس $b$:
الوجود والوحدانية، والدورية تميّز $\Q$}]
\label{pb:b1:reals:1}
لكل عدد حقيقي في $\intco{0}{1}$ نشرٌ بالأرقام في كل أساس
$b \geq 2$؛ والنشر وحيد ما إن نمنع السلاسل النهائية من
الرقم $b - 1$؛ ويكون دوريًا في النهاية إذا وفقط إذا كان
العدد ناطقًا. وتبرهن هذه المسألة على الوقائع الثلاث كلها
انطلاقًا من بديهية التمام وحدها --- لا متتاليات ولا متسلسلات:
بل \omterm{def:b1:reals:bounds}{الحد الأعلى} \omterm{thm:b1:reals:archimedes}{وخاصية أرخميدس} \omterm{thm:b1:reals:floor}{ودالة الجزء الصحيح} فقط
--- وتُختم بحجة كانتور القطرية في صورة الأرقام.
وفي كل ما يأتي، $b \geq 2$ عدد صحيح مثبَّت (وهو \emph{الأساس})،
و\emph{الرقم} عنصر من $\intint{0}{b-1}$، وتكون سلسلة
الأرقام $(d_n)_{n \geq 1}$ \emph{سليمة} إذا لم تكن
مساوية في النهاية للمقدار $b - 1$ (أي: من أجل كل $N$ يوجد $n >
N$ يحقق $d_n \leq b - 2$).\index{نشر في الأساس b@النشر في الأساس $b$}\index{نشر عشري}

\medskip
\noindent\textbf{الجزء 1 --- الأرقام باليد.} القسمة المطوّلة للمقدار $p$
على $q$ في الأساس $b$: اضرب الباقي الحالي في $b$، واقسم
على $q$، وسجّل الحاصل رقمًا تاليًا، واحتفظ بالباقي.
\begin{enumerate}
  \item في الأساس $10$، نفّذ الخوارزمية على $\frac 18$ وعلى
        $\frac 17$، مسجّلًا في كل خطوة الرقم \emph{و}
        الباقي. وتحقق من أن بواقي $\frac 17$
        تدور على $1, 3, 2, 6, 4, 5$ وأن الأرقام
        $142857$ تتكرر بعدئذ إلى الأبد.
  \item احسب النشرين في الأساس $2$ للمقدارين $\frac 13$ و
        $\frac{5}{16}$، والنشر في الأساس $3$ للمقدار $\frac 12$.
        ولاحظ: عدد واحد ينتهي، والآخران يتكرران ---
        و $\frac 12$، وهو وديع جدًا في الأساس $10$، يتكرر إلى الأبد في
        الأساس $3$.
  \item من أجل $x = \frac pq \in \intco{0}{1}$ في الصورة المختزلة، بيّن
        أن الأرقام التي تنتجها الخوارزمية تكون في النهاية
        كلها $0$ إذا وفقط إذا انعدم الباقي $b^N p \bmod q$
        من أجل $N$ ما، إذا وفقط إذا قسم $q$ قوةً ما
        $b^N$، إذا وفقط إذا قسم كلُّ عامل أوليّ للمقدار $q$
        العددَ $b$. وتحقق: $\frac{1}{20}$ ينتهي في الأساس
        $10$، لا في الأساس $3$.
  \item عرّف \emph{الاقتطاع} $s_n = \lfloor b^n x \rfloor
        / b^n$. من أجل $x = \sqrt 2$ و $b = 10$، احسب $s_0,
        \dots, s_4$ بالتحقق في كل خطوة من أن مربعين
        متتاليين يحيطان بالعدد $2$ (مثلًا $1.4142^2
        = 1.99996164 < 2 < 2.00024449 = 1.4143^2$)، وتحقق من
        $s_n \leq \sqrt 2 < s_n + 10^{-n}$ في كل مرة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- الوجود، انطلاقًا من \omterm{def:b1:reals:bounds}{الحد الأعلى}.} ثبّت
$x \in \intco{0}{1}$ وضع $A_n = \lfloor b^n x \rfloor$ و
$d_n = A_n - b\,A_{n-1}$ من أجل $n \geq 1$.
\begin{enumerate}[resume]
  \item بيّن أن $A_0 = 0$ وأن $b\,A_{n-1} \leq A_n \leq b\,A_{n-1} +
        b - 1$؛ واستنتج أن كل $d_n$ رقم.
  \item بيّن أن $s_n := A_n b^{-n}$ يحقق
        \[
        s_n = \sum_{k=1}^{n} d_k\,b^{-k}
        \qquad\text{و}\qquad
        s_n \leq x < s_n + b^{-n} .
        \]
  \item برهن على $b^n \geq n + 1$ بالاستقراء، ثم بيّن أن
        $(s_n)$ غير متناقصة وأن $x = \sup_n s_n$
        \emph{(استعمل \cref{prop:b1:reals:epsilon} و
        \cref{thm:b1:reals:archimedes})}.
  \item بيّن أن السلسلة $(d_n)$ سليمة: فإذا كان $d_k = b - 1$
        من أجل كل $k > N$، فاحسب $s_n$ من أجل $n > N$ بمجموع
        هندسي منته وناقض السؤال 6.
  \item وبالعكس، لتكن $(e_n)_{n \geq 1}$ أيَّ سلسلة أرقام
        سليمة ولتكن $t_n = \sum_{k=1}^n e_k b^{-k}$. بيّن أن $y
        = \sup_n t_n$ موجود ويقع في $\intco{0}{1}$،
        وأنه يحقق $t_n \leq y < t_n + b^{-n}$ من أجل كل $n$
        \emph{(ومن أجل المتراجحة القطعية، استعمل رقمًا $e_m \leq b
        - 2$ مع $m > n$)}. واستنتج $\lfloor b^n y \rfloor = b^n
        t_n$، ثم أن أرقام $y$، بمعنى
        السؤال 5، هي بالضبط الأعداد $e_n$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- الوحدانية والترتيب والإزاحة.}
\begin{enumerate}[resume]
  \item اجمع الأسئلة 5--9 في \emph{مبرهنة النشر في الأساس
        $b$}: التطبيقان $x \mapsto (d_n)$ و $(e_n) \mapsto
        \sup_n t_n$ تقابلان متعاكسان بين
        $\intco{0}{1}$ \omterm{def:b1:logic:sets}{ومجموعة} سلاسل الأرقام السليمة. وعلى
        الخصوص، لا تأخذ سلسلتان سليمتان متمايزتان القيمة نفسها.
  \item اسمح الآن بالسلاسل غير السليمة. بيّن أن سلسلة تحقق $e_n
        = b - 1$ من أجل كل $n > M$ (مع $M \geq 0$ أصغريًا) قيمتُها
        $t_M + b^{-M}$؛ واستنتج أن $0.999\dots = 1$ في
        الأساس $10$، وأن الأعداد الحقيقية ذات التمثيلين
        بالأرقام هي بالضبط الكسور في الأساس $b$
        $m/b^N \in \intoo{0}{1}$ --- ولكل عدد حقيقي آخر تمثيل
        واحد فقط، حتى بين السلاسل غير السليمة.
  \item برهن على أن تقابل السؤال 10 يحفظ
        الترتيب من أجل الترتيب المعجمي: أي إذا اختلفت
        السلسلتان السليمتان للمقدارين $x$ و $y$ أول ما اختلفتا عند الدليل $m$،
        فإن $x < y$ إذا وفقط إذا كان $d_m < e_m$.
  \item (مبرهنة الإزاحة المساعدة) ليكن $x \in \intco{0}{1}$ ذا الأرقام
        $(d_n)$. بيّن أن الجزء الكسري للمقدار $bx$ أرقامه
        $(d_{n+1})_{n \geq 1}$ \emph{(احسب $\lfloor b^n(bx -
        A_1)\rfloor$ مستعملًا $\lfloor u - K \rfloor = \lfloor u
        \rfloor - K$ من أجل عدد صحيح $K$)}، واستنتج بالاستقراء
        أن الجزء الكسري للمقدار $b^m x$ أرقامه
        $(d_{n+m})_{n \geq 1}$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- النطقية هي الدورية.} ليكن $x
= \frac pq \in \intco{0}{1}$ في الصورة المختزلة وليكن $r_n = b^n p
\bmod q$ باقي القسمة الإقليدية للمقدار $b^n p$ على
$q$.
\begin{enumerate}[resume]
  \item بيّن $A_n = \dfrac{b^n p - r_n}{q}$ و $r_n =
        (b\,r_{n-1}) \bmod q$.
  \item بيّن $d_n = \Bigl\lfloor \dfrac{b\,r_{n-1}}{q}
        \Bigr\rfloor$: أي أن كل رقم دالةٌ في الباقي
        السابق وحده. وهذه بالضبط هي القسمة المطوّلة في
        الجزء 1.
  \item طبّق مبدأ الأدراج
        (\cref{cor:b1:counting:pigeonhole}) على $r_0, \dots, r_q$
        واستنتج: نشر كل عدد ناطق دوريٌّ
        في النهاية، وما قبل دوره ودوره على الأكثر
        $q$.
  \item وبالعكس، افترض أن أرقام $y \in \intco{0}{1}$
        دورية \emph{محضًا}: $d_{n+T} = d_n$ من أجل كل $n
        \geq 1$. باستعمال مبرهنة الإزاحة المساعدة ووحدانية
        السؤال 10، بيّن أن الجزء الكسري للمقدار $b^T y$
        يساوي $y$، واستنتج $(b^T - 1)\,y \in \N$: فيكون $y$
        ناطقًا ومقامه \omterm{def:b1:arith:divides}{يقسم} $b^T - 1$. وتحقق من
        الآلية على $0.(142857)$: $142857 \times 7 = 999999$.
  \item عالج الحالة الدورية في النهاية بالإزاحة، واذكر
        \emph{محك الدورية}: يكون $x \in \intco{0}{1}$
        ناطقًا إذا وفقط إذا كان نشره السليم في الأساس $b$
        دوريًا في النهاية --- في أساس واحد إذا وفقط إذا كان كذلك في
        كل الأسس.
  \item من أجل $x = \frac 1q$ مع $\gcd(q, b) = 1$، بيّن أن
        النشر دوري محضًا وأن أصغر دور له هو
        أصغر $T \geq 1$ يحقق $b^T \equiv 1 \pmod q$ (وهو
        الرتبة الضربية للمقدار $b$ بترديد $q$). وتحقق من أنه من أجل
        $q = 7$ و $b = 10$ تجري قوى $10$ بترديد $7$
        على $3, 2, 6, 4, 5, 1$: فالرتبة $6$، مطابقةً للسؤال
        1.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- الأرباح والقطر.}
\begin{enumerate}[resume]
  \item ليكن $x^*$ العدد الحقيقي من $\intco{0}{1}$ الذي أرقامه
        في الأساس $10$ هي $1$ في المواضع المثلثية $\frac{j(j +
        1)}{2}$ ($j \geq 1$) و $0$ فيما عداها: $x^* =
        0.101001000100001\dots$ بيّن أن سلسلة أرقامه
        سليمة لكنها ليست دورية في النهاية \emph{(إذ إن دورًا $T$
        سيفرض آحادًا بفجوات $T$ على الأكثر، لكن الفجوات تنمو)}،
        واستنتج أن $x^*$ عدد أصمّ: عددٌ بُرهن على صممه
        بالإيقاع المحض.
  \item بيّن أن من أجل كل أساس $b$ تكون \omterm{def:b1:logic:sets}{المجموعة} $\{m/b^n : m \in \Z,
        n \in \N\}$ كثيفة في $\R$ (تعميمًا للمقدار
        \cref{exo:b1:reals:8})، وأن لكل عدد ناطق $\frac pq
        \in \intoo{0}{1}$ نشرًا \emph{منتهيًا} في
        الأساس $q$. والعبرة: الانتهاء خاصية للزوج
        (العدد، الأساس)؛ وأمّا الدورية --- أي النطقية --- فذاتية.
  \item (قطر كانتور) ليكن $k \mapsto x_k$ أيَّ \omterm{def:b1:logic:map}{تطبيق} من
        $\N^*$ إلى $\intco{0}{1}$. عرّف سلسلة الأرقام $e_k =
        1$ إذا اختلف الرقم $k$ للمقدار $x_k$ عن $1$، و
        $e_k = 2$ فيما عدا ذلك. بيّن أن $(e_k)$ سليمة، وأن
        قيمتها $y$ تقع في $\intco{0}{1}$، وأن $y \neq
        x_k$ من أجل كل $k$. واستنتج: لا \omterm{def:b1:logic:map}{تطبيق} $\N^* \to
        \intco{0}{1}$ شامل. (ومفردات
        قابلية العدّ، وموطن هذه المبرهنة الصحيح، في
        \cref{ch:b1:topology}.)
  \item بيّن أنه إذا تطابق النشران السليمان للمقدارين $x$ و $y$
        حتى الدليل $n$ فإن $\abs{x - y} < b^{-n}$، وانقض
        العكس بالمقدارين $x = 0.1$ و $y = 0.0999$ في الأساس $10$:
        فتقارب الأعداد لا يفرض تطابق الأرقام.
        فمن المسؤول من الأعداد الحقيقية عن ذلك؟
  \item نفّذ الجزء 4 على $x = \frac{1}{10}$ في الأساس $b = 2$:
        احسب البواقي والأرقام حتى تدور، و
        استنتج $\frac{1}{10} = (0.0\overline{0011})_2$، مع
        ما قبل دور $1$ ودور $4$. وفسّر عبر السؤال 3 لماذا
        لن تساوي أيُّ سلسلة ثنائية منتهية $\frac{1}{10}$ أبدًا
        --- وهو سبب أن $0.1 + 0.2$ بالفاصلة العائمة في الحاسوب
        ليس $0.3$ بالضبط.
  \item توليفة. في جملة واحدة لكلٍّ: أين استعمل البرهان
        (أ) خاصية التمام، (ب) \omterm{thm:b1:reals:archimedes}{خاصية أرخميدس}، (ج)
        بند الوحدانية في الجزء الصحيح، (د) مبدأ الأدراج؟
        والعبرة: تُرمَّز $\intco{0}{1}$ ترميزًا أمينًا
        بسلاسل الأرقام السليمة، وتُقرأ النطقية دوريةً
        --- ومع ذلك يفضّل التحليل \omterm{def:b1:reals:bounds}{الحد الأعلى} على
        الأرقام. لماذا؟ (فكّر في جمع سلسلتَي أرقام.)
\end{enumerate}
\end{problem}
