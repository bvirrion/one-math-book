\chapter{طوبولوجيا المستقيم الحقيقي}\label{ch:b1:topology}

تحيل النهايات باستمرار إلى المفردات الهندسية نفسها: نقاط
«قريبة من» \omterm{def:b1:logic:sets}{مجموعة}، ومجموعات «بلا تسرّب عند \omterm{def:b1:topology:closure}{الحافة}»، وفترات
لا تستطيع المتتاليات الفرار منها. ويثبّت هذا الفصل تلك المفردات ---
المجموعات \omterm{def:b1:topology:open}{المفتوحة} \omterm{def:b1:topology:closed}{والمغلقة}، والداخل \omterm{def:b1:topology:closure}{والغلق}، \omterm{def:b1:topology:dense}{والكثافة} --- على المستقيم
الحقيقي، ويبرهن على تراص القطع في صورته التتالية.
والمفاهيم نفسها، في الفضاءات المعيارية، مادة السنة الثانية؛
وأمّا على $\R$ فهي في المتناول ونافعة فورًا من أجل
\cref{ch:b1:continuity}.

\section{المجموعات المفتوحة والمجموعات المغلقة}

\begin{definition}[الجوار والمجموعة المفتوحة]\label{def:b1:topology:open}
تكون \omterm{def:b1:logic:sets}{المجموعة} $V \subseteq \R$ \emph{جوارًا}\index{جوار}
للنقطة $x \in \R$ إذا احتوت \omterm{prop:b1:reals:intervals}{فترة} $\intoo{x - r}{x + r}$ من أجل
$r > 0$ ما. وتكون \omterm{def:b1:logic:sets}{المجموعة} $U \subseteq \R$ \emph{مفتوحة}\index{مجموعة مفتوحة}
إذا كانت جوارًا لكل نقطة من نقطها:
\[
\forall x \in U,\ \exists r > 0, \quad \intoo{x - r}{x + r}
\subseteq U .
\]
\end{definition}

\begin{example}\label{ex:b1:topology:open}
الفترات \omterm{def:b1:topology:open}{المفتوحة} \omterm{def:b1:topology:open}{مفتوحة}: فمن أجل $x \in \intoo{a}{b}$، خذ $r = \min(x -
a,\, b - x) > 0$. وأنصاف المستقيمات $\intoo{a}{+\infty}$ \omterm{def:b1:topology:open}{مفتوحة}؛ و $\R$ و
$\emptyset$ مفتوحتان (والثانية بالمجّان). وأمّا $\intcc{0}{1}$ فهي
\emph{ليست} \omterm{def:b1:topology:open}{مفتوحة}: إذ لا \omterm{prop:b1:reals:intervals}{فترة} حول $0$ تبقى بالداخل.
\end{example}

\begin{proposition}[استقرار المجموعات المفتوحة]\label{prop:b1:topology:openstable}
كل اتحاد لمجموعات \omterm{def:b1:topology:open}{مفتوحة} مفتوح؛ وكل تقاطع \emph{منته} لمجموعات
\omterm{def:b1:topology:open}{مفتوحة} مفتوح. وأمّا التقاطعات غير المنتهية فقد تفشل:
$\bigcap_{n \geq 1} \intoo{-\frac 1n}{\frac 1n} = \{0\}$، وهي ليست \omterm{def:b1:topology:open}{مفتوحة}.
\end{proposition}

\begin{proof}
الاتحاد: إذا كان $x \in \bigcup_i U_i$، فإن $x \in U_{i_0}$ من أجل $i_0$ ما،
وتقع \omterm{prop:b1:reals:intervals}{الفترة} التي تعطيها $U_{i_0}$ داخل الاتحاد. والتقاطع
المنتهي: إذا كان $x \in U_1 \cap \dots \cap U_k$، فخذ $r = \min(r_1,
\dots, r_k) > 0$ لأنصاف الأقطار التي تعطيها كل $U_j$. وأمّا
المثال المضاد: فإن أيّ \omterm{prop:b1:reals:intervals}{فترة} حول $0$ تحتوي $\frac{1}{n}$ ما
(أرخميدس)، ومنه تخرج من التقاطع.
\end{proof}

\begin{example}[المصادقة على الانفتاح بأنصاف أقطار صريحة]\label{ex:b1:topology:radii}
هل $U = \{x \in \R : x^2 > 2\}$ \omterm{def:b1:topology:open}{مفتوحة}؟ نعم، ويمكن كتابة
الشهادة: $U = \intoo{-\infty}{-\sqrt2} \cup
\intoo{\sqrt2}{+\infty}$، وهي اتحاد نصفَي مستقيم مفتوحين، \omterm{def:b1:topology:open}{مفتوحة}
حسب \cref{prop:b1:topology:openstable}. وبدلًا من ذلك، حاجج
نقطةً نقطة: فمن أجل $x \in U$ مع $x > \sqrt 2$، خذ $r = x -
\sqrt2 > 0$: فكل $y \in \intoo{x - r}{x + r}$ يحقق $y >
\sqrt 2$، ومنه $y^2 > 2$؛ وبالتناظر على اليسار. والأسلوبان
كلاهما مهمّ --- فالبنيوي (ابنِ من مجموعات \omterm{def:b1:topology:open}{مفتوحة}
معلومة باتحادات وتقاطعات منتهية) أقدر على التوسّع، والأسلوب
بالمقدار $\varepsilon$ ينفع حين لا تُرى أيّ بنية؛ و
سوف يضيف \cref{ch:b1:continuity} ثالثًا، هو الأقوى:
\omterm{def:b1:logic:sets}{فالمجموعة} $U$ هي \omterm{def:b1:logic:map}{الصورة العكسية} \omterm{def:b1:topology:open}{للمجموعة المفتوحة} $\intoo{2}{+\infty}$ \omterm{def:b1:logic:map}{بالتطبيق}
المتصل $x \mapsto x^2$.
\end{example}

\begin{definition}[المجموعة المغلقة]\label{def:b1:topology:closed}
تكون \omterm{def:b1:logic:sets}{المجموعة} $F \subseteq \R$ \emph{مغلقة}\index{مجموعة مغلقة} إذا كانت
متممتها $\R \setminus F$ \omterm{def:b1:topology:open}{مفتوحة}. وحسب دي مورغان و
\cref{prop:b1:topology:openstable}: يكون كل تقاطع لمجموعات مغلقة
مغلقًا، وتكون الاتحادات المنتهية للمجموعات المغلقة مغلقة.
\end{definition}

\begin{theorem}[التمييز التتالي للمجموعات المغلقة]\label{thm:b1:topology:seqclosed}
تكون $F$ \omterm{def:b1:topology:closed}{مغلقة} إذا وفقط إذا: كانت النهاية $\ell$ تنتمي إلى
$F$ من أجل كل متتالية $(u_n)$ من نقاط
$F$ تتقارب إلى $\ell \in \R$ ما. («\omterm{def:b1:topology:closed}{مغلقة}» $=$ «مستقرة بالنهايات».)
\end{theorem}

\begin{proof}
($\Rightarrow$) لتكن $F$ \omterm{def:b1:topology:closed}{مغلقة} و $u_n \in F$ و $u_n \to \ell$، و
لنفترض $\ell \notin F$. فالمتممة \omterm{def:b1:topology:open}{مفتوحة}: ومنه توجد
$\intoo{\ell - r}{\ell + r}$ تتجنّب $F$. لكن التقارب يضع $u_n$
في تلك \omterm{prop:b1:reals:intervals}{الفترة} من أجل $n$ كبير: وهذا تناقض مع $u_n \in F$.

($\Leftarrow$) افترض أن $F$ ليست \omterm{def:b1:topology:closed}{مغلقة}: فالمتممة ليست \omterm{def:b1:topology:open}{مفتوحة}،
ومنه توجد $x \notin F$ لا \omterm{prop:b1:reals:intervals}{فترة} $\intoo{x - r}{x + r}$ لها داخل
المتممة؛ وبأخذ $r = \frac{1}{n+1}$، اختر $u_n \in F$ يحقق
$\abs{u_n - x} < \frac{1}{n+1}$. عندئذ $u_n \in F$ و $u_n \to x \notin
F$: فتفشل الخاصية التتالية.
\end{proof}

\begin{example}[الاختبار التتالي في الاتجاهين]\label{ex:b1:topology:seqtest}
\emph{\omterm{def:b1:topology:closed}{مغلقة}:} $F = \Z \cup \bigl\{n + \frac1n : n \geq 2\bigr\}$.
لتكن $u_k \in F$ مع $u_k \to \ell$. النافذة $\intcc{\ell -
1}{\ell + 1}$ لا تحتوي إلا عددًا منتهيًا من نقاط $F$ (عددًا منتهيًا
من الأعداد الصحيحة، وعددًا منتهيًا من المقادير $n + \frac1n$)، وبعد رتبة ما
تقع كل $u_k$ فيها: فتأخذ المتتالية عندئذ عددًا منتهيًا من
القيم، ولكونها متقاربة تصير ثابتة في النهاية (كما في
\cref{exo:b1:topology:3}): ومنه $\ell \in F$. فهي \omterm{def:b1:topology:closed}{مغلقة} --- وإن كانت
$F$ تحتوي أزواج نقاط على بعد $\frac1n$، أي قريبة
كما نشاء.

\emph{ليست \omterm{def:b1:topology:closed}{مغلقة}:} $G = \bigl\{\frac1m + \frac1n : m, n \in
\N^*\bigr\}$. المتتالية $\frac1n + \frac1n \in G$ تؤول إلى
$0$، و $0 \notin G$ (لأنه مجموع حدين موجبين): فيفشل
الاختبار التتالي، ومنه \omterm{def:b1:logic:sets}{فالمجموعة} $G$ ليست \omterm{def:b1:topology:closed}{مغلقة}. ومن الطريف أن كل
$\frac1m$ \emph{ينتمي} فعلًا إلى $\overline G \cap G$: إذ
$\frac1m = \frac{1}{m+1} + \frac{1}{m(m+1)} \in G$. والفكرة
النافذة: للبرهان على الانغلاق، اضبط \emph{كل} المتتاليات
المتقاربة دفعة واحدة (عادةً عبر تناهٍ موضعي أو حجة
صيغة \omterm{def:b1:topology:closed}{مغلقة})؛ ولنقضه تكفي متتالية هاربة واحدة
مُحسَنة الاختيار --- فعدم التناظر يجعل الاتجاه السالب
هو السهل، ولكل الأمثلة المضادة في هذا الفصل هذا الشكل ذو
السطر الواحد.
\end{example}

\begin{example}\label{ex:b1:topology:closed}
القطع $\intcc{a}{b}$، وأنصاف المستقيمات $\intco{a}{+\infty}$، والمجموعات
المنتهية، و $\Z$ (لأن متتالية متقاربة من الأعداد الصحيحة ثابتة في النهاية)
كلها \omterm{def:b1:topology:closed}{مغلقة}. وأمّا $\intoc{0}{1}$ فليست \omterm{def:b1:topology:open}{مفتوحة} (إذ تفشل عند $1$) ولا \omterm{def:b1:topology:closed}{مغلقة}
($\frac 1n \to 0 \notin$ \omterm{def:b1:logic:sets}{المجموعة}): فأكثر المجموعات ليست هذه ولا تلك. و $\R$ و
$\emptyset$ مفتوحتان ومغلقتان معًا --- وهما المجموعتان الجزئيتان الوحيدتان
من $\R$ كذلك (\cref{exo:b1:topology:9}).
\end{example}

\begin{example}[مجموعة مفتوحة مركّبة من قطع لا تُحصى]\label{ex:b1:topology:rminusz}
$\R \setminus \Z = \bigcup_{n \in \Z} \intoo{n}{n+1}$: أي
اتحاد غير منته لفترات \omterm{def:b1:topology:open}{مفتوحة}، وهو مفتوح حسب
\cref{prop:b1:topology:openstable} --- ومنه \omterm{def:b1:logic:sets}{فالمجموعة} $\Z$ \omterm{def:b1:topology:closed}{مغلقة} دون أيّ حاجة إلى
حجة تتالية. ولاحظ تقسيم العمل في
قواعد الاستقرار: قد تكون \emph{اتحادات} المجموعات \omterm{def:b1:topology:open}{المفتوحة} كيفية
(فكل نقطة لا تحتاج إلا إلى شهادتها الخاصة، تعطيها \omterm{def:b1:logic:sets}{المجموعة} الوحيدة
التي تحتويها)، بينما يجب أن تبقى \emph{التقاطعات} منتهية
(إذ يجب تقاطع الشهادات، وقد تتقلص أنصاف أقطار لا تُحصى
إلى العدم). وسوف يبيّن \Cref{exo:b1:topology:10} أن هذا
المثال هو الشكل العام: فكل \omterm{def:b1:logic:sets}{مجموعة} جزئية \omterm{def:b1:topology:open}{مفتوحة} من $\R$
اتحادٌ قابل للعدّ ومنفصل لفترات \omterm{def:b1:topology:open}{مفتوحة}.
\end{example}

\section{الداخل والغلق والكثافة}

\begin{definition}[الداخل والغلق والحافة]\label{def:b1:topology:closure}
لتكن $A \subseteq \R$.
\begin{itemize}
  \item تكون النقطة $x$ \emph{داخلية} \omterm{def:b1:logic:sets}{للمجموعة} $A$ إذا كانت $A$
        \omterm{def:b1:topology:open}{جوارًا} للنقطة $x$؛ و\emph{الداخل}
        $\mathring{A}$\index{داخل} هو \omterm{def:b1:logic:sets}{مجموعة} النقاط الداخلية.
  \item تكون النقطة $x$ \emph{ملاصقة} \omterm{def:b1:logic:sets}{للمجموعة} $A$ إذا كان كل \omterm{def:b1:topology:open}{جوار}
        للنقطة $x$ يقاطع $A$؛ و\emph{الغلق}
        $\overline{A}$\index{غلق} هو \omterm{def:b1:logic:sets}{مجموعة} النقاط الملاصقة.
  \item و\emph{الحافة} هي $\partial A = \overline A \setminus
        \mathring A$.\index{حافة}
\end{itemize}
وعندئذ $\mathring A \subseteq A \subseteq \overline A$.
\end{definition}

\begin{proposition}[الخصائص الرئيسة]\label{prop:b1:topology:closureprops}
\begin{enumerate}
  \item $\mathring A$ أكبر \omterm{def:b1:topology:open}{مجموعة مفتوحة} محتواة في $A$؛ وتكون $A$
        \omterm{def:b1:topology:open}{مفتوحة} إذا وفقط إذا كان $A = \mathring A$.
  \item $\overline A$ أصغر \omterm{def:b1:topology:closed}{مجموعة مغلقة} تحتوي $A$؛ وتكون $A$
        \omterm{def:b1:topology:closed}{مغلقة} إذا وفقط إذا كان $A = \overline A$.
  \item (التمييز التتالي للملاصقة) $x \in \overline
        A$ إذا وفقط إذا كانت $x$ نهاية متتالية من نقاط
        $A$.
  \item التتميم يبادل بين المفهومين: $\R \setminus
        \overline A = \bigl(\R \setminus A\bigr)^{\!\circ}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(4) $x \notin \overline A$ $\iff$ يتجنّب جوارٌ ما للنقطة $x$ المجموعةَ $A$
$\iff$ تقع \omterm{prop:b1:reals:intervals}{فترة} ما حول $x$ في $\R \setminus A$ $\iff$ تكون $x$
داخلية \omterm{def:b1:logic:sets}{للمجموعة} $\R \setminus A$.

(1) \omterm{def:b1:logic:sets}{المجموعة} $\mathring A$ \omterm{def:b1:topology:open}{مفتوحة}: فإذا كان $x \in \mathring A$، فتوجد
$\intoo{x-r}{x+r} \subseteq A$؛ ولكل نقطة $y$ من تلك \omterm{prop:b1:reals:intervals}{الفترة}
\omterm{prop:b1:reals:intervals}{فترة} أصغر حولها داخلها، ومنه داخل $A$: \omterm{prop:b1:reals:intervals}{فالفترة} كلها
في $\mathring A$. وكل \omterm{def:b1:topology:open}{مفتوحة} $U \subseteq A$ تتألف من
نقاط داخلية \omterm{def:b1:logic:sets}{للمجموعة} $A$، ومنه $U \subseteq \mathring A$: فهي الأكبر. ويتبع
تمييز الانفتاح.

(2) بالتفصيل. حسب (4)، تكون $\R \setminus \overline A$ داخلَ
$\R \setminus A$، وهي \omterm{def:b1:topology:open}{مفتوحة} حسب (1): ومنه \omterm{def:b1:logic:sets}{فالمجموعة} $\overline A$
\omterm{def:b1:topology:closed}{مغلقة}، وهي تحتوي $A$. والأصغرية: لتكن $F \supseteq A$
\omterm{def:b1:topology:closed}{مغلقة}. عندئذ $\R \setminus F$ \omterm{def:b1:topology:open}{مفتوحة} ومحتواة في $\R
\setminus A$، ومنه، حسب الأكبرية في (1)،
\[
\R \setminus F \subseteq \bigl(\R \setminus A\bigr)^{\!\circ}
= \R \setminus \overline A ,
\]
وبأخذ المتممات مرة أخرى: $\overline A \subseteq F$. ومنه \omterm{def:b1:logic:sets}{فالمجموعة}
$\overline A$ أصغر حاوية \omterm{def:b1:topology:closed}{مغلقة}. وأمّا التمييز:
فإذا كان $A = \overline A$ فإن $A$ \omterm{def:b1:topology:closed}{مغلقة} (وقد بُيّن ذلك للتوّ)؛ وإذا كانت $A$
\omterm{def:b1:topology:closed}{مغلقة}، فهي نفسها \omterm{def:b1:topology:closed}{مجموعة مغلقة} تحتوي $A$، ومنه تفرض الأصغرية
أن $\overline A \subseteq A$، ومنه المساواة.

(3) إذا كان $u_n \in A$ و $u_n \to x$: فكل \omterm{def:b1:topology:open}{جوار} للنقطة $x$ يحتوي
عنصرًا $u_n \in A$، ومنه $x \in \overline A$. وبالعكس، إذا كان $x \in
\overline A$: فكل \omterm{prop:b1:reals:intervals}{فترة} $\intoo{x - \frac{1}{n+1}}{x +
\frac{1}{n+1}}$ تقاطع $A$ عند $u_n$ ما، و $u_n \to x$.
\end{proof}

\begin{example}\label{ex:b1:topology:closureexamples}
$\overline{\intoo{0}{1}} = \intcc{0}{1}$؛
$\mathring{\intcc{0}{1}} = \intoo{0}{1}$؛ $\partial\intoo{0}{1} =
\{0, 1\}$. ومن أجل $A = \{\frac 1n : n \in \N^*\}$: $\overline A = A
\cup \{0\}$ و $\mathring A = \emptyset$ و $\partial A = A \cup \{0\}$.
ومن أجل $\Q$: \omterm{def:b1:topology:dense}{بالكثافة} (\cref{thm:b1:reals:density}) يكون كل عدد حقيقي
ملاصقًا \omterm{def:b1:logic:sets}{للمجموعة} $\Q$، ومنه $\overline{\Q} = \R$ بينما $\mathring{\Q} =
\emptyset$ (لأن كل \omterm{prop:b1:reals:intervals}{فترة} تحتوي أعدادًا صماء): ومنه \omterm{def:b1:topology:closure}{فحافة}
$\Q$ هي $\R$ كلها.
\end{example}

\begin{example}[تشريح كامل]\label{ex:b1:topology:anatomy}
لتكن $A = \intoc{0}{1} \,\cup\, \bigl(\Q \cap \intoo{2}{3}\bigr)
\,\cup\, \{4\}$. نحسب المجموعات الثلاث في
\cref{def:b1:topology:closure}، قطعةً قطعة.

\emph{الداخل.} لكل نقطة من $\intoo{0}{1}$ \omterm{prop:b1:reals:intervals}{فترة} كاملة
داخل $A$: فهي داخلية. وأمّا النقطة $1$: فكل \omterm{prop:b1:reals:intervals}{فترة} حولها
تتسرّب يمين $1$، حيث لا شيء في $A$ حتى $2$: فهي ليست
داخلية. ولا نقطة من $\Q \cap \intoo{2}{3}$ داخلية (لأن كل
\omterm{prop:b1:reals:intervals}{فترة} تحتوي أعدادًا صماء، \cref{thm:b1:reals:density})؛
وكذلك النقطة المعزولة $4$. ومنه $\mathring A = \intoo{0}{1}$.

\emph{\omterm{def:b1:topology:closure}{الغلق}.} نهايات نقاط $A$: كل $\intcc{0}{1}$
($0 = \lim \frac1n$ مع $\frac 1n \in A$)؛ وكل
$\intcc{2}{3}$ (لأن كل عدد حقيقي هناك نهاية أعداد ناطقة من
\omterm{prop:b1:reals:intervals}{الفترة}، \omterm{def:b1:topology:dense}{بالكثافة} مرة أخرى)؛ والنقطة $4$. ولا شيء آخر: فلأيّ نقطة
خارج $\intcc{0}{1} \cup \intcc{2}{3} \cup \{4\}$ مسافة موجبة
إلى تلك \omterm{def:b1:topology:closed}{المجموعة المغلقة}. ومنه $\overline A = \intcc{0}{1} \cup
\intcc{2}{3} \cup \{4\}$.

\emph{\omterm{def:b1:topology:closure}{الحافة}.} $\partial A = \overline A \setminus \mathring A
= \{0, 1\} \cup \intcc{2}{3} \cup \{4\}$.

والفكرة النافذة: تعمل العمليات الثلاث \emph{موضعيًا} ---
فكل قطعة من $A$ تسهم حسب طبيعتها هي (\omterm{prop:b1:reals:intervals}{فالفترة}
الصلبة تحتفظ بداخلها، والقطعة \omterm{def:b1:topology:dense}{الكثيفة} المسامية تتحول
كلها إلى \omterm{def:b1:topology:closure}{حافة}، والنقطة المعزولة \omterm{def:b1:topology:closure}{حافة} محضة)،
ورسمٌ من سطرين \omterm{def:b1:logic:sets}{للمجموعة} $A$ يتنبّأ بكل جواب قبل كتابة أيّ
برهان.
\end{example}

\begin{remark}[مزالق شائعة في التفكير النقطي-المجموعي]
(أ) \emph{«ليست \omterm{def:b1:topology:open}{مفتوحة}» لا تعني «\omterm{def:b1:topology:closed}{مغلقة}»}: فأكثر المجموعات ليست
هذه ولا تلك ($\intoc{0}{1}$)، ومجموعتان هما الاثنتان معًا
($\emptyset$ و $\R$) --- فالمفتوح والمغلق ليسا نقيضين بل
مزدوجان بالتتميم. (ب) \emph{الداخل \omterm{def:b1:topology:closure}{والغلق}
لا يتبادلان}: فمن أجل $A = \Q$،
\[
\overline{\mathring A} = \overline\emptyset = \emptyset
\qquad\text{بينما}\qquad
\bigl(\,\overline A\,\bigr)^{\!\circ} = \mathring \R = \R :
\]
فيختلف المؤثران المتكرران بأقصى ما يمكن أن تختلف به مجموعتان. (ج)
\emph{الاتحادات غير المنتهية للمجموعات \omterm{def:b1:topology:closed}{المغلقة} قد تفشل في أن تكون \omterm{def:b1:topology:closed}{مغلقة}}:
$\bigcup_{n\geq1} \intcc{\frac1n}{1} = \intoc{0}{1}$ --- وهي
مرآة المثال المضاد للتقاطع في
\cref{prop:b1:topology:openstable}. (د) \emph{الكثيف لا
يعني الكبير}: \omterm{def:b1:logic:sets}{فالمجموعة} $\Q$ \omterm{def:b1:topology:dense}{كثيفة} وقابلة للعدّ وداخلها خالٍ، و
متممتها \omterm{def:b1:topology:dense}{كثيفة} كذلك؛ \omterm{def:b1:topology:dense}{فالكثافة} تقول «قريبة كما نشاء من
كل شيء»، لا «كل شيء تقريبًا» --- وتقول \omterm{pb:b1:topology:1}{مجموعة كانتور} في مسألة
نهاية الأسبوع (\cref{pb:b1:topology:1}) العكس،
فهي \omterm{def:b1:logic:sets}{مجموعة} صغيرة طوبولوجيًا وكبيرة كبرًا غير قابل للعدّ.
\end{remark}

\begin{example}[غلق محسوب بالضبط]\label{ex:b1:topology:closurecomputed}
لتكن $G = \bigl\{\frac1m + \frac1n : m, n \in \N^*\bigr\}$ (من
\cref{ex:b1:topology:seqtest}). الادعاء:
\[
\overline G = G \,\cup\, \Bigl\{\frac1m : m \in \N^*\Bigr\}
\,\cup\, \{0\} .
\]
($\supseteq$) $\frac1m = \lim_n \bigl(\frac1m + \frac1n\bigr)$
و $0 = \lim_n \frac2n$: فهي ملاصقة حسب التمييز
التتالي. ($\subseteq$) ليكن $x = \lim_k \bigl(
\frac{1}{m_k} + \frac{1}{n_k}\bigr)$؛ رتّب كل زوج بحيث يكون
$m_k \leq n_k$. فإذا كانت $(m_k)$ غير محدودة، فلمتتالية جزئية $m_k
\to \infty$، ومنه $n_k \to \infty$ كذلك و $x = 0$. وإلا فإن
$(m_k)$ تأخذ عددًا منتهيًا من القيم، إحداها، ولنقل $m$،
عددًا لا يُحصى من المرات؛ وعلى تلك \omterm{def:b1:seq:subsequence}{المتتالية الجزئية} $\frac{1}{n_k} \to x -
\frac1m$: فإذا كانت $(n_k)$ محدودة أخذت قيمة $n$ ما
عددًا لا يُحصى من المرات وكان $x = \frac1m + \frac1n \in G$؛ وإلا فإن $x =
\frac1m$. فكل حالة تحطّ في \omterm{def:b1:logic:sets}{المجموعة} المعلنة. والفكرة
النافذة: حساب \omterm{def:b1:topology:closure}{غلق} تحليلُ حالات على الأدلة بأسلوب
التراص --- فالدليل المحدود يعني عددًا منتهيًا من
القيم (مبدأ الأدراج)، والدليل غير المحدود يعني نهايةً تفلت ---
ويعرض الجوابُ البنيةَ النموذجية ذات الطبقتين لنقاط
التراكم: \omterm{def:b1:logic:sets}{المجموعة}، ونهاياتها من الجيل الأول، ونهايتها
$0$.
\end{example}

\begin{definition}[الكثافة في صورتها الطوبولوجية]\label{def:b1:topology:dense}
تكون $A$ \emph{كثيفة}\index{كثافة} في $\R$ إذا كان $\overline A = \R$
--- وبكيفية مكافئة، إذا قاطعت كلُّ \omterm{prop:b1:reals:intervals}{فترة} \omterm{def:b1:topology:open}{مفتوحة} غير خالية المجموعةَ $A$؛
وبكيفية مكافئة (حسب \cref{prop:b1:topology:closureprops}~(3))، إذا كان كل
عدد حقيقي نهايةَ عناصر من $A$. أمثلة: $\Q$، و $\R \setminus
\Q$، والأعداد الثنائية (\cref{exo:b1:reals:8})، والزمر الجزئية الكثيفة
(\cref{exo:b1:reals:9}).
\end{definition}

\begin{example}[الكثافة نسبية]\label{ex:b1:topology:relativedensity}
تعني «\omterm{def:b1:topology:dense}{كثيفة}» بالمعنى المعرَّف هنا كثيفةً \emph{في $\R$}؛ وقد تكون
\omterm{def:b1:logic:sets}{مجموعة} بدل ذلك \omterm{def:b1:topology:dense}{كثيفة} في جزء من المستقيم فقط. فالأعداد الثنائية في
$\intcc{0}{1}$، أي $D \cap \intcc{0}{1}$
(\cref{exo:b1:reals:8})، تقاطع كل \omterm{prop:b1:reals:intervals}{فترة} \omterm{def:b1:topology:open}{مفتوحة} محتواة في
$\intcc{0}{1}$ لكنها بالطبع تفوّت $\intoo{2}{3}$ كليًا: فهي
\omterm{def:b1:topology:dense}{كثيفة} \emph{في} $\intcc{0}{1}$، بمعنى $\overline{D \cap
\intcc{0}{1}} = \intcc{0}{1}$. \omterm{def:b1:logic:statement}{والعبارة} العامة «$A$ \omterm{def:b1:topology:dense}{كثيفة}
في $B$» اختصارٌ للمقدار $B \subseteq \overline A$ --- وسمِّ دائمًا
\omterm{def:b1:logic:sets}{المجموعة} المحيطة، لأن طرفَي مسألة نهاية الأسبوع كثيفان
\emph{في \omterm{pb:b1:topology:1}{مجموعة كانتور}} بينما ليسا كثيفين في أيّ موضع \emph{في
$\R$}: فهي \omterm{def:b1:logic:sets}{المجموعة} نفسها، ووصفان صادقان يبدوان متعاكسين.
\end{example}

\begin{example}[التعامل مع الكثافة]\label{ex:b1:topology:densitytricks}
ثلاث حركات سريعة تتكرر باستمرار. \emph{التكبير}: إذا كانت
$A$ \omterm{def:b1:topology:dense}{كثيفة} و $A \subseteq B$، فإن $B$ \omterm{def:b1:topology:dense}{كثيفة} (لأن كل
\omterm{prop:b1:reals:intervals}{فترة} تقاطع $A$ أصلًا). \emph{النقل}: إذا كانت $A$
\omterm{def:b1:topology:dense}{كثيفة}، فكذلك $\lambda A + \mu$ من أجل $\lambda \neq 0$ --- \omterm{prop:b1:reals:intervals}{فالفترة}
$I$ تقاطع $\lambda A + \mu$ إذا وفقط إذا قاطعت \omterm{prop:b1:reals:intervals}{الفترة}
$\frac{I - \mu}{\lambda}$ المجموعةَ $A$؛ ومنه فمضاعفات
$10^{-9}$ الفردية، مثلًا، \omterm{def:b1:topology:dense}{كثيفة}. \emph{والتقاطع يفشل}: فقد تفوّت مجموعتان
كثيفتان إحداهما الأخرى كليًا ($\Q$ و $\R \setminus \Q$):
\omterm{def:b1:topology:dense}{فالكثافة} تصمد أمام الاتحادات والتطبيقات الأفينية، لا أمام التقاطعات أبدًا.
\end{example}

\section{تراص القطع}

\begin{theorem}[القطع متراصة تتاليًا]\label{thm:b1:topology:compact}
ليكن $a \leq b$. لكل متتالية من نقاط $\intcc{a}{b}$
\omterm{def:b1:seq:subsequence}{متتالية جزئية} تتقارب إلى نقطة \emph{من}
$\intcc{a}{b}$.\index{تراص}

وأعمّ من ذلك، المجموعات الجزئية من $\R$ التي لها هذه الخاصية (أي أن لكل
متتالية متتاليةً جزئية تتقارب داخل \omterm{def:b1:logic:sets}{المجموعة}) هي بالضبط
المجموعات \emph{\omterm{def:b1:topology:closed}{المغلقة} والمحدودة}.
\end{theorem}

\begin{proof}
كل متتالية في $\intcc{a}{b}$ محدودة، ومنه تستخرج بولتزانو--فايرشتراس
(\cref{thm:b1:seq:bw}) \omterm{def:b1:seq:subsequence}{متتالية جزئية} متقاربة؛ وتبقى نهايتها
في $\intcc{a}{b}$ لأن القطع \omterm{def:b1:topology:closed}{مغلقة}
(\cref{thm:b1:topology:seqclosed}).

الحالة العامة. \emph{(\omterm{def:b1:topology:closed}{مغلقة} ومحدودة $\Rightarrow$ متراصة)}: لتكن
$F$ \omterm{def:b1:topology:closed}{مغلقة} ومحدودة، ولتكن $(u_n)$ متتالية في $F$.
تحدّ حدّيةُ $F$ المتتاليةَ، ومنه تستخرج بولتزانو--فايرشتراس
$u_{\varphi(n)} \to \ell$؛ و $\ell \in F$ لأن $F$
\omterm{def:b1:topology:closed}{مغلقة} \omterm{def:b1:seq:subsequence}{والمتتالية الجزئية} متتالية متقاربة من نقاط
$F$ (\cref{thm:b1:topology:seqclosed}): فتُستهلك الفرضيتان
واحدةً لكلٍّ، الحدّية من أجل وجود النهاية،
والانغلاق من أجل انتمائها. \emph{(متراصة $\Rightarrow$ \omterm{def:b1:topology:closed}{مغلقة} و
محدودة)}: إذا كانت $F$ غير محدودة، فاختر $u_n \in F$ يحقق $\abs{u_n} \geq
n$؛ فتكون كل \omterm{def:b1:seq:subsequence}{متتالية جزئية} غير محدودة، ومنه متباعدة
(\cref{prop:b1:seq:first}): فلا \omterm{def:b1:seq:subsequence}{متتالية جزئية} متقاربة أصلًا. وإذا كانت $F$
غير \omterm{def:b1:topology:closed}{مغلقة}، فخذ $u_n \in F$ مع $u_n \to \ell \notin F$
(\cref{thm:b1:topology:seqclosed}): فتتقارب كل \omterm{def:b1:seq:subsequence}{متتالية جزئية} إلى
$\ell \notin F$، ومنه لا \omterm{def:b1:seq:subsequence}{متتالية جزئية} تتقارب \emph{في} $F$.
\end{proof}

\begin{example}[مجموعات متراصة متداخلة]\label{ex:b1:topology:nested}
تمرين أول على المبرهنة. لتكن $K_0 \supseteq K_1 \supseteq
K_2 \supseteq \dots$ مجموعات جزئية غير خالية متراصة (\omterm{def:b1:topology:closed}{مغلقة} ومحدودة)
من $\R$. عندئذ $\bigcap_n K_n \neq \emptyset$. فعلًا، اختر $x_n
\in K_n$ من أجل كل $n$: فتعيش المتتالية في المتراصة $K_0$،
ومنه تتقارب \omterm{def:b1:seq:subsequence}{متتالية جزئية} $x_{\varphi(n)}$ إلى $x$ ما
(\cref{thm:b1:topology:compact}). ومن أجل كل $m$ مثبَّت، تقع كل الحدود
$x_{\varphi(n)}$ ذات $\varphi(n) \geq m$ في \omterm{def:b1:topology:closed}{المجموعة المغلقة}
$K_m$، ومنه تقع النهاية $x$ في $K_m$
(\cref{thm:b1:topology:seqclosed})؛ ولأن $m$ كان كيفيًا، $x \in
\bigcap_n K_n$. ورفيقٌ نافع: إذا احتوت \omterm{def:b1:topology:open}{مجموعة مفتوحة} $U$ المجموعةَ
$\bigcap_n K_n$، فإن $U \supseteq K_n$ من أجل $n$ ما --- طبّق
الحجة نفسها على نقاط $x_n \in K_n \setminus U$؛ فستقع النهاية
$x$ في $\bigcap K_n \subseteq U$، ومع ذلك يفرض انفتاح $U$ أن
$x_{\varphi(n)} \in U$ في النهاية، وهذا تناقض. والعبارتان
كلتاهما تفشلان دون التراص: $\bigcap_n \intoo{0}{\frac
1n} = \emptyset$ و $\bigcap_n \intco{n}{+\infty} = \emptyset$.
والفكرة النافذة: يحوّل التراص سلسلةً غير منتهية من
تأكيدات عدم الخلوّ إلى نقطة نهاية واحدة --- فهو
الأداة التي تنجو من الانتقال إلى التقاطع غير المنتهي، وسوف
تتّكئ عليه مسألة نهاية الأسبوع (\cref{pb:b1:topology:1}) مرتين.
\end{example}

\begin{example}[الحواف تحت الاتحادات]\label{ex:b1:topology:boundaryunion}
دائمًا $\partial(A \cup B) \subseteq \partial A \cup \partial
B$: فلكل نقطة من $\partial(A \cup B)$ كل \omterm{def:b1:topology:open}{جوار}
يقاطع $A \cup B$ (ومنه $A$ أو $B$، وإحداهما عددًا لا يُحصى
من المرات) ويقاطع متممة $A \cup B$، وهي تقع في
المتممتين معًا --- ثم يضع تحقّق قصير النقطةَ في
$\partial A$ أو $\partial B$. وقد يكون الاحتواء قطعيًا
بصورة مذهلة: فمع $A = \Q$ و $B = \R\setminus\Q$،
\[
\partial(A \cup B) = \partial \R = \emptyset ,
\qquad
\partial A \cup \partial B = \R \cup \R = \R :
\]
فيمكن لمجموعتين ممزّقتين أن تلتصقا في \omterm{def:b1:logic:sets}{مجموعة} بلا خياطة، وحافتاهما
تفني إحداهما الأخرى. والفكرة النافذة: يسلك الداخل و
\omterm{def:b1:topology:closure}{الغلق} سلوكًا \emph{رتيبًا} تحت الاتحادات و
التقاطعات، وأمّا الحواف فلا --- فعامل $\partial$ معاملة
مقدار مشتقّ ($\overline A \setminus \mathring A$)، لا معاملة
مؤثر له جبرُه الخاص.
\end{example}

\begin{remark}[منظورات داخل هذا المجلّد]
تُستهلك المفردات المبنيّة هنا مرتين أخريين في هذا الكتاب.
ففي \cref{ch:b1:continuity}، كل مبرهنة \omterm{def:b1:logic:statement}{عبارة} طوبولوجية
متنكّرة: فمبرهنة القيم الوسطى تقول إن التطبيقات
المتصلة تحفظ خاصية \omterm{prop:b1:reals:intervals}{الفترة}، ومبرهنة القيم
الحدّية تقول إنها تحفظ التراص --- وتستدعي البراهين
\cref{thm:b1:topology:seqclosed,thm:b1:topology:compact}
بالاسم. وفي \cref{ch:b1:multivar}، تُقرأ التعاريف نفسها
ثانيةً في $\R^2$ بالأقراص بدل الفترات: فتنتقل المجموعات \omterm{def:b1:topology:open}{المفتوحة}
\omterm{def:b1:topology:closure}{والغلق} والتراص كلمةً كلمة، وتركب مبرهنة القيم الحدّية
ذات المتغيرين مرة أخرى على
بولتزانو--فايرشتراس (استخرج على كل إحداثية). والمفهوم الوحيد الذي
\emph{لا} يعمَّم بلا ألم هو \omterm{prop:b1:reals:intervals}{الفترة} نفسها --- ففي المستوي،
يحلّ الترابط محلّ التحدّب، وهي قصة تبدأ مع
\omterm{def:b1:logic:statement}{عبارة} \cref{exo:b1:topology:9} «لا \omterm{def:b1:topology:open}{مجموعة مفتوحة} \omterm{def:b1:topology:closed}{ومغلقة} إلا $\emptyset$
و $\R$».
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:topology:1}
من أجل كل \omterm{def:b1:logic:sets}{مجموعة}، قل هل هي \omterm{def:b1:topology:open}{مفتوحة} أم \omterm{def:b1:topology:closed}{مغلقة} أم الاثنتان معًا أم لا هذه ولا تلك (مع
التسويغ): $\intoo{0}{1} \cup \intoo{2}{3}$؛ $\;\intco{0}{1}$؛
$\;\{0\} \cup \intcc{1}{2}$؛ $\;\R \setminus \Z$؛ $\;\Q \cap
\intoo{0}{1}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:topology:2}
حدّد $\mathring A$ و $\overline A$ و $\partial A$ من أجل:
$A = \intoc{0}{1} \cup \{2\}$؛ $\;A = \R \setminus \Q$؛
$\;A = \bigl\{\frac{(-1)^n n}{n+1} : n \in \N\bigr\}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:topology:3}
برهن على أن كل \omterm{def:b1:counting:card}{مجموعة منتهية} \omterm{def:b1:topology:closed}{مغلقة}، أولًا عبر المتممات، ثم عبر
التمييز التتالي.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:topology:4}
برهن على أنه من أجل كل $A, B \subseteq \R$: $\overline{A \cup B} =
\overline A \cup \overline B$. وبيّن بمثال أن
$\overline{A \cap B}$ قد تختلف عن $\overline A \cap \overline
B$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:5}
ليكن $u_n \to \ell$ في $\R$. برهن على أن \omterm{def:b1:logic:sets}{المجموعة} $\{u_n : n \in \N\}
\cup \{\ell\}$ \omterm{def:b1:topology:closed}{مغلقة} (ومنه متراصة إذا أضفنا أنها
محدودة --- وهي كذلك).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:6}
لتكن $U$ \omterm{def:b1:topology:open}{مفتوحة} و $A$ كيفية. برهن على أن $U + A = \{u + a\}$
\omterm{def:b1:topology:open}{مفتوحة}. واستنتج أن مجموع \omterm{def:b1:topology:open}{مجموعة مفتوحة} وأيّ \omterm{def:b1:logic:sets}{مجموعة} مفتوحٌ، و
قابل: أظهر مجموعتين \emph{مغلقتين} مجموعهما ليس مغلقًا.
\emph{(جرّب $\Z$ و $\sqrt 2\,\Z$، مع \cref{exo:b1:reals:9}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:7}
تكون النقطة $x \in A$ \emph{معزولة} في $A$ إذا كان \omterm{def:b1:topology:open}{جوار} ما
للنقطة $x$ لا يقاطع $A$ إلا عند $x$. برهن على أن كل نقطة من $\Z$ معزولة
في $\Z$، وعلى أن كل نقاط $A = \{\frac1n\}$ معزولة ومع ذلك
$\overline A \neq A$، وعلى أن كل \omterm{def:b1:logic:sets}{مجموعة} كل نقاطها معزولة
داخلها خالٍ.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:8}
برهن على أن \omterm{def:b1:topology:closure}{غلق} \omterm{def:b1:logic:sets}{مجموعة} محدودة محدود، وعلى أن
$\sup \overline A = \sup A$ من أجل $A$ غير خالية ومحدودة من أعلى. واستنتج
أن $\sup A \in \overline A$: \omterm{def:b1:reals:bounds}{فالحد الأعلى} ملاصق دائمًا.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:topology:9}
برهن على أن المجموعتين الجزئيتين الوحيدتين من $\R$ المفتوحتين والمغلقتين معًا هما
$\emptyset$ و $\R$. \emph{إرشاد: افترض أن $A$ \omterm{def:b1:topology:open}{مفتوحة} \omterm{def:b1:topology:closed}{ومغلقة}، مع
$A \neq \emptyset$ و $\R \setminus A \neq \emptyset$؛ واختر $a \in
A$ و $b \notin A$، وليكن $a < b$، وانظر في $s = \sup\,(A \cap
\intcc{a}{b})$؛ وقرّر هل يمكن أن ينتمي $s$ إلى $A$ أو إلى
متممتها.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:topology:10}
(بنية المجموعات \omterm{def:b1:topology:open}{المفتوحة}) لتكن $U \subseteq \R$ \omterm{def:b1:topology:open}{مفتوحة} وغير خالية. من أجل
$x \in U$، لتكن $I_x$ اتحاد كل الفترات \omterm{def:b1:topology:open}{المفتوحة} التي تحتوي
$x$ وتُحتوى في $U$. برهن على أن $I_x$ \omterm{prop:b1:reals:intervals}{فترة} \omterm{def:b1:topology:open}{مفتوحة}، وعلى أن
المجموعتين $I_x$ و $I_y$ متساويتان أو منفصلتان، وعلى أن $U$ اتحاد
\emph{عدد قابل للعدّ} من الفترات \omterm{def:b1:topology:open}{المفتوحة} المنفصلة مثنى مثنى \emph{(اختر
عددًا ناطقًا في كل واحدة)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:topology:11}
تكون النقطة $x \in \R$ \emph{نقطة تراكم} \omterm{def:b1:logic:sets}{للمجموعة} $A$ إذا كان
كل \omterm{def:b1:topology:open}{جوار} للنقطة $x$ يقاطع $A \setminus \{x\}$؛ ومجموعتها هي
\emph{\omterm{def:b1:logic:sets}{المجموعة} المشتقة} $A'$. برهن على أن $\overline A = A \cup A'$،
وعلى أن $A$ \omterm{def:b1:topology:closed}{مغلقة} إذا وفقط إذا كان $A' \subseteq A$. وحدّد
$A'$ من أجل $A = \{\frac 1n : n \in \N^*\}$، ومن أجل $A = \Z$، ومن أجل
$A = \Q$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:topology:12}
من أجل $A \subseteq \R$ غير خالية، عرّف $d_A(x) = \inf\{\abs{x - a}
: a \in A\}$. برهن على:
\begin{enumerate}
  \item $\abs{d_A(x) - d_A(y)} \leq \abs{x - y}$ من أجل كل $x, y$
        (أي إن $d_A$ ليبشيتزية بالثابت $1$)؛
  \item $d_A(x) = 0$ إذا وفقط إذا كان $x \in \overline A$؛ وعلى
        الخصوص، إذا كانت $F$ \omterm{def:b1:topology:closed}{مغلقة} و $x \notin F$، فإن
        $d_F(x) > 0$؛
  \item من أجل كل $\varepsilon > 0$، تكون \omterm{def:b1:logic:sets}{المجموعة} $V_\varepsilon =
        \{x : d_F(x) < \varepsilon\}$ \omterm{def:b1:topology:open}{مفتوحة} وتحتوي $F$، و
        $\bigcap_{\varepsilon > 0} V_\varepsilon = F$ من أجل $F$
        \omterm{def:b1:topology:closed}{مغلقة}: فكل \omterm{def:b1:topology:closed}{مجموعة مغلقة} تقاطعٌ قابل للعدّ لمجموعات
        \omterm{def:b1:topology:open}{مفتوحة}.
\end{enumerate}
\end{exercise}

\section{مسألة: مجموعة كانتور، صغيرة وهائلة في آن}

\begin{problem}[{مسألة نهاية الأسبوع --- \omterm{pb:b1:topology:1}{مجموعة كانتور} ذات الأثلاث
الوسطى: طولها معدوم، وغير قابلة للعدّ، وتامّة، و $C + C =
\intcc{0}{2}$}]
\label{pb:b1:topology:1}
احذف من $\intcc{0}{1}$ ثلثها الأوسط المفتوح، ثم الثلث
الأوسط من كل قطعة باقية، وكرّر ذلك إلى الأبد: فما
ينجو هو \emph{\omterm{pb:b1:topology:1}{مجموعة كانتور}} $C$، مصنعُ الأمثلة المضادة
الأساسي في التحليل. وتبني هذه المسألة \omterm{def:b1:logic:sets}{المجموعة}،
وتقرؤها عبر آلة الأساس $3$ في \cref{pb:b1:reals:1}،
وتؤسّس صورتها المفارِقة: طولٌ كلي معدوم، ومع ذلك
غير قابلة للعدّ؛ وداخلٌ خالٍ، ومع ذلك لا نقطة معزولة؛
ومفكّكة كليًا، ومع ذلك يملأ $C + C$ القطعة كلها
$\intcc{0}{2}$.\index{مجموعة كانتور} وصوريًا: $C_0 = \intcc{0}{1}$،
ونحصّل $C_{n+1}$ من $C_n$ بحذف الثلث الأوسط
المفتوح من كل قطعة من $C_n$؛ وأخيرًا $C = \bigcap_{n \geq 0}
C_n$. وفي كل ما يأتي، \emph{الترميز الثلاثي} للمقدار $x \in \intcc{0}{1}$
هو أيّ سلسلة أرقام $(d_k)_{k\geq1}$ مع $d_k \in \{0, 1, 2\}$
تساوي قيمتُها $\sup_n \sum_{k=1}^n d_k 3^{-k}$ العددَ $x$ ---
والترميزات غير السليمة (المساوية $2$ في النهاية) مسموحة؛ وحسب
\cref{pb:b1:reals:1} (الأسئلة 9--11)، لكل $x \in
\intcc{0}{1}$ ترميز أو ترميزان، واثنان بالضبط عندما $x = m/3^N
\in \intoo{0}{1}$.

\medskip
\noindent\textbf{الجزء 1 --- البناء.}
\begin{enumerate}
  \item صف $C_1$ و $C_2$ صراحةً اتحاداتٍ لقطع،
        وبرهن بالاستقراء: $C_n$ اتحاد منفصل
        لعدد $2^n$ من القطع \omterm{def:b1:topology:closed}{المغلقة}، طول كلٍّ منها $3^{-n}$.
  \item بيّن أن $C$ \omterm{def:b1:topology:closed}{مغلقة} ومحدودة --- ومنه متراصة
        (\cref{thm:b1:topology:compact}) --- وغير خالية، وأن
        كل طرف لكل قطعة من كل $C_n$ ينتمي
        إلى $C$.
  \item الطول الكلي للمقدار $C_n$ هو $\bigl(\frac23\bigr)^n$.
        استنتج أنه من أجل كل $\varepsilon > 0$، يمكن تغطية \omterm{def:b1:logic:sets}{المجموعة} $C$
        بعدد منته من القطع طولها الكلي
        $\leq \varepsilon$: فطول \omterm{pb:b1:topology:1}{مجموعة كانتور} \emph{معدوم}.
  \item بيّن أن كل \omterm{prop:b1:reals:intervals}{فترة} محتواة في $C$ طولها $\leq
        3^{-n}$ من أجل كل $n$، ومنه فهي أحادية أو خالية:
        $\mathring C = \emptyset$. ولكونها \omterm{def:b1:topology:closed}{مغلقة} وداخلها
        خالٍ، تكون $C$ \emph{غير \omterm{def:b1:topology:dense}{كثيفة} في أيّ موضع}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- الترميز الثلاثي.}
\begin{enumerate}[resume]
  \item برهن على العلاقة التراجعية للتشابه الذاتي
        \[
        C_{n+1} = \tfrac13 C_n \,\cup\,
        \bigl(\tfrac23 + \tfrac13 C_n\bigr),
        \qquad\text{ومنه}\qquad
        C = \tfrac13 C \,\cup\, \bigl(\tfrac23 + \tfrac13
        C\bigr),
        \]
        والقطعتان منفصلتان: \omterm{def:b1:logic:sets}{فالمجموعة} $C$ نسختان من
        نفسها بالسلّم $\frac13$.
  \item برهن بالاستقراء على $n$: يكون $x \in C_n$ إذا وفقط إذا
        كان للمقدار $x$ ترميز ثلاثي أرقامه $n$ الأولى في
        $\{0, 2\}$. واستنتج، مستعملًا كون $x$ له ترميزان
        على الأكثر: يكون $x \in C$ إذا وفقط إذا كان للمقدار $x$ ترميز
        \emph{لا رقم فيه يساوي $1$} (ترميز \emph{خالٍ من $1$}).
  \item الترميزات في العمل: أعطِ ترميزات خالية من $1$ من أجل $0$ و $1$ و
        $\frac13$ و $\frac23$؛ وبيّن $\frac14 = (0.\overline{02})_3$
        و $\frac34 = (0.\overline{20})_3$، ومنه ينتمي كلاهما إلى
        $C$؛ وتحقق من أن $\frac14$ \emph{ليس} طرفًا
        لأيّ $C_n$ (فللأطراف الشكل $m/3^n$).
  \item بيّن أن لكل $x \in C$ ترميزًا \emph{واحدًا بالضبط} خاليًا من $1$
        \emph{(فعندما يكون للمقدار $x$ ترميزان، برهن على أن واحدًا بالضبط
        من الزوج يحتوي الرقم $1$)}. واستنتج: \omterm{def:b1:logic:map}{تطبيق}
        القيمة تقابلٌ من سلاسل $\{0,2\}$ على
        $C$.
  \item (القطر) ليكن $k \mapsto x_k$ أيَّ \omterm{def:b1:logic:map}{تطبيق} $\N^* \to C$.
        ابنِ سلسلة $\{0, 2\}$ تختلف عند الدليل $k$ عن
        ترميز $x_k$، واستنتج أن $C$ غير قابلة للعدّ
        --- بينما، على النقيض، يقول السؤال 3 إنها
        مهملة متريًا.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- الصورة الطوبولوجية.}
\begin{enumerate}[resume]
  \item اجمع السجل حتى الآن: $C$ متراصة وغير قابلة للعدّ،
        وطولها معدوم، وغير \omterm{def:b1:topology:dense}{كثيفة} في أيّ موضع. وأيّ احتواء واحد
        $C \subseteq C_n$ يحمل كلًّا من هذه الخصائص؟
  \item ($C$ تامّة) ليكن $x \in C$ ذا الترميز الخالي من $1$
        $(d_k)$. قلبُ الرقم $d_n$ ($0 \leftrightarrow
        2$) ينتج $x_n \in C$ مع $\abs{x_n - x} = 2 \cdot
        3^{-n}$. استنتج أن لا نقطة معزولة في $C$: فكل
        نقطة من $C$ نهايةٌ لنقاط \emph{أخرى} من $C$.
  \item بيّن أن أطراف السؤال 2 تكوّن \omterm{def:b1:logic:sets}{مجموعة} جزئية قابلة للعدّ
        \omterm{def:b1:topology:dense}{وكثيفة} في $C$ \emph{(اقتطع الترميز بعد $n$
        رقمًا وواصل بالأرقام $0$؛ وقابلية العدّ كما في
        \cref{exo:b1:topology:10})}. واستنتج: النقطة النموذجية
        من $C$ --- مثل $\frac14$ --- \emph{ليست} طرفًا:
        فالأطراف هيكل قابل للعدّ داخل جسم غير قابل
        للعدّ.
  \item (مفكّكة كليًا) ليكن $x < y$ في $C$. اختر $n$
        يحقق $3^{-n} < y - x$ وأنتج نقطة $z \in
        \intoo{x}{y}$ مع $z \notin C$. استنتج أن الفترات
        الوحيدة غير الخالية المحتواة في $C$ هي المجموعات الأحادية.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- حساب $C$.}
\begin{enumerate}[resume]
  \item بيّن $1 - C = C$ \emph{(ماذا يفعل $x \mapsto 1 - x$
        بترميز خالٍ من $1$؟ وتذكّر $1 = (0.\overline{2})_3$)}.
  \item (جمع الترميزات) بيّن أنه إذا كان للمقدارين $x$ و $x'$ ترميزان
        $(a_k)$ و $(b_k)$، فإن $x + x' = \lim_n\,(t_n + t'_n)$
        حيث $t_n, t'_n$ هما المجموعان الجزئيان. واستنتج: كل $y
        \in \intcc{0}{1}$ هو \emph{منتصف} نقطتين
        من $C$ --- فمن أجل ترميز $(e_k)$ للمقدار $y$، اختر أرقامًا
        $a_k, b_k \in \{0, 2\}$ تحقق $\frac{a_k + b_k}{2} =
        e_k$.
  \item استنتج $C + C = \intcc{0}{2}$، ومع السؤال 14،
        $C - C = \intcc{-1}{1}$. ومثال ملموس: اكتب $1$
        مجموعًا للنقطتين غير الطرفيتين في السؤال 7.
  \item تأمّل: \omterm{def:b1:logic:sets}{مجموعة} طولها معدوم \omterm{def:b1:logic:sets}{ومجموعة} فروقها تملأ
        $\intcc{-1}{1}$. فلماذا لا تناقض بين
        «$C$ مهملة متريًا» و«$C + C$ ذات طول
        كامل»؟ (جملة واحدة؛ فكّر فيما يضبطه الطول
        وما لا يضبطه.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- الأعضاء، الناطقون والصمّ.}
\begin{enumerate}[resume]
  \item اجمع بين السؤال 8 ومحك الدورية في
        \cref{pb:b1:reals:1}: تكون نقطة من $C$ ناطقة إذا و
        فقط إذا كان ترميزها الخالي من $1$ دوريًا في النهاية. ونفّذ
        القسمة المطوّلة في الأساس $3$ للتحقق من $\frac1{13} =
        (0.\overline{002})_3 \in C$.
  \item أنتج عضوًا \emph{أصمّ} صراحةً في $C$:
        قيمةَ الترميز ذي $d_k = 2$ في المواضع المثلثية
        $k = \frac{j(j+1)}{2}$ و $d_k = 0$
        فيما عداها. وسوّغ الصمم بحجة تنامي الفجوات في
        \cref{pb:b1:reals:1} (السؤال 20).
  \item (على قطعة كاملة) انظر في $h$ الذي يرسل نقطة
        $C$ ذات الترميز الخالي من $1$ $(d_k)$ إلى قيمة
        السلسلة \emph{الثنائية} $\bigl(\frac{d_k}2\bigr)$، أي
        $h(x) = \sup_n \sum_{k=1}^n \frac{d_k}{2}\,2^{-k}$.
        بيّن أن $h$ يرسل $C$ \emph{على} $\intcc{0}{1}$. ومنه
        \omterm{def:b1:logic:sets}{فالمجموعة} المهملة $C$ تشمل قطعة ذات طول كامل
        --- وهو برهان ثانٍ على أن $C$ غير قابلة للعدّ.
  \item (طول ذاتي التشابه) افترض أن مفهومًا ما للطول $L$
        عُرّف من أجل $C$ ونسخها المصغَّرة، محترمًا
        السلّمة ($L(\lambda A) = \lambda L(A)$) والحفظ
        بالانسحاب والجمعية على التفكيك المنفصل
        في السؤال 5. بيّن أن ذلك يعطي $L(C) = \frac23\,L(C)$،
        فيفرض $L(C) = 0$: فالتشابه الذاتي وحده يحكم
        على $C$ أصلًا بطول معدوم.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 6 --- ابن عمّ سمين، والعبرة.}
\begin{enumerate}[resume]
  \item (\omterm{pb:b1:topology:1}{مجموعة كانتور} السمينة) أعد البناء، لكن عند المرحلة
        $n$ ($n = 0, 1, 2, \dots$) احذف من كل قطعة من القطع $2^n$
        الحالية فترةً \omterm{def:b1:topology:open}{مفتوحة} مركزية طولها
        $4^{-(n+1)}$ فقط. بيّن أن أطوال القطع $l_n$
        تحقق $l_{n+1} = \frac{l_n - 4^{-(n+1)}}{2}$ و $l_n =
        \frac{2^n + 1}{2\cdot 4^n} > 0$، وأن
        $K = \bigcap K_n$ الناتجة متراصة وداخلها خالٍ، وأن
        الطول الكلي المحذوف هو $\sum_{n\geq0} 2^n
        4^{-(n+1)} = \frac12$. وبقبول جمعية الطول (البديهية، من السنة الثالثة)
        من أجل الاتحادات المنتهية للفترات، و
        باستعمال عبارتَي \cref{ex:b1:topology:nested} معًا،
        بيّن أن أيّ عائلة منتهية من الفترات \omterm{def:b1:topology:open}{المفتوحة} تغطي
        $K$ طولها الكلي $\geq \frac12$: \omterm{def:b1:logic:sets}{فالمجموعة} $K$ غير \omterm{def:b1:topology:dense}{كثيفة} في
        أيّ موضع لكنها \emph{ليست} مهملة. فللصغر معانٍ عدة
        غير متكافئة.
  \item (المسافات) بيّن أنه من أجل \omterm{def:b1:topology:closed}{مجموعة مغلقة} غير خالية $F \subseteq
        \R$ و $x \in \R$، يكون \omterm{def:b1:reals:bounds}{الحد الأدنى} $d(x, F)$
        \emph{مبلوغًا} \emph{(بمتتالية تصغيرية مع
        بولتزانو--فايرشتراس)}. ثم احسب
        \[
        \max_{y \in \intcc{0}{1}} d(y, C) = \frac16 ,
        \]
        مبلوغًا بالضبط عند المركز $y = \frac12$ \emph{(فكل
        نقطة من فجوة أُنشئت عند المرحلة $n$ تقع على بعد
        $\frac{3^{-n}}{2}$ من طرفَي الفجوة، وهما في
        $C$)}.
  \item (كل نقطة نهايةُ \omterm{def:b1:seq:subsequence}{متتالية جزئية}) باستعمال السؤالين 12
        و 2، أنتج متتالية واحدة في $C$ مجموعةُ
        نهايات متتالياتها الجزئية هي \emph{كل} $C$. (وقارن:
        فمن أجل متتالية متقاربة تكون تلك \omterm{def:b1:logic:sets}{المجموعة} نقطة واحدة ---
        وتحقق $C$ الطرف المعاكس بين المتراصات.)
  \item توليفة، جملة واحدة لكلٍّ: (أ) أيّ مبرهنات من هذا
        الفصل استهلكها البناء فعلًا (استقرار
        المجموعات \omterm{def:b1:topology:closed}{المغلقة}، والتراص، والتمييزات
        التتالية)؟ (ب) اذكر الأزواج المفارِقة الأربعة
        في الصورة (طول معدوم/غير قابلة للعدّ،
        \omterm{def:b1:topology:closed}{مغلقة}/داخل خالٍ، تامّة/مفكّكة كليًا،
        مهملة/$C+C$ كامل)؛ (ج) وأين تعود $C$ إلى الظهور
        لاحقًا (سلّم الشيطان المبنيّ على $h$ في نظرية
        الاتصال، ونظرية القياس في مجلّد السنة الثالثة،
        حيث تفصل $C$ بين «قابلة للعدّ» و
        «مهملة»)؟
\end{enumerate}
\end{problem}
