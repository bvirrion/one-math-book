\chapter{الأعداد العقدية}\label{ch:b1:complex}

قابلنا الأعداد العقدية في مجلد الثانوية بوصفها أداة حسابية
للمعادلات من الدرجة الثانية. ويعاملها هذا الفصل بوصفها غرضًا
محوريًا: الشكل الأسّي ونتائجه (دي موافر، والجذور من الرتبة $n$،
\omterm{def:b1:complex:unity}{وجذور الوحدة})، والترجمة المنهجية بين $\C$ وهندسة المستوي، واستعمال
$\eu^{\iu\theta}$ آلةً للبرهان على المتطابقات المثلثية.

\section{الحقل $\C$ والمقياس والمرافق}

\begin{definition}[حقل الأعداد العقدية]\label{def:b1:complex:field}
$\C = \{a + \iu b : a, b \in \R\}$ مع الجمع المعتاد والضرب المحدَّد
بالعلاقة $\iu^2 = -1$. ولكل $z = a + \iu
b$ غير معدوم مقلوب: $z^{-1} = \frac{a - \iu b}{a^2 + b^2}$ --- وبلغة
\cref{ch:b1:structures}، تكون $\C$ حقلًا. ونكتب $a =
\Re(z)$ و $b = \Im(z)$ و $\conj{z} = a - \iu b$ (وهو
\emph{المرافق}\index{مرافق}) و$\abs{z} = \sqrt{a^2 + b^2}$
(وهو \emph{المقياس}\index{مقياس}).
\end{definition}

\begin{proposition}[قواعد المرافق والمقياس]\label{prop:b1:complex:rules}
من أجل $z, w \in \C$:
\begin{enumerate}
  \item $\conj{z + w} = \conj z + \conj w$ و$\conj{zw} = \conj z\,
        \conj w$ و $\conj{\conj z} = z$؛
  \item $z \conj z = \abs z^2$؛ و$\;\Re(z) = \frac{z + \conj z}{2}$ و
        $\Im(z) = \frac{z - \conj z}{2\iu}$؛
  \item $\abs{zw} = \abs z\, \abs w$، و$\abs{z^{-1}} =
        \abs{z}^{-1}$ من أجل $z \neq 0$؛
  \item (المتراجحة المثلثية\index{متراجحة مثلثية}) $\abs{z + w}
        \leq \abs z + \abs w$، مع التساوي إذا وفقط إذا وقع $z$ و
        $w$ على نصف مستقيم واحد مبدؤه $0$ (أي $w = \lambda z$ أو
        $z = \lambda w$ مع $\lambda \geq 0$)؛
  \item (المتراجحة المثلثية المعكوسة) $\bigl|\abs z - \abs w\bigr|
        \leq \abs{z - w}$.
\end{enumerate}
\end{proposition}

\begin{proof}
النقطتان (1) و (2) حسابان مباشران على الجزأين الحقيقي والتخيلي.
و (3): $\abs{zw}^2 = zw\,\conj{zw} = z\conj z\, w \conj w = \abs z^2 \abs
w^2$، ثم خذ الجذر التربيعي؛ وطبّق ذلك على $z \cdot z^{-1} = 1$ من أجل
المقلوب.

(4) الطرفان موجبان، فلنقارن مربعيهما:
\[
\abs{z+w}^2 = (z+w)(\conj z + \conj w)
= \abs z^2 + \abs w^2 + 2\,\Re(z \conj w),
\]
وتعطي $\Re(z\conj w) \leq \abs{z \conj w} = \abs z \abs w$ أن
$\abs{z+w}^2 \leq (\abs z + \abs w)^2$. ويفرض التساوي أن $\Re(z \conj
w) = \abs{z \conj w}$، أي $z \conj w \in \R_+$؛ فإذا كان $w \neq 0$ أعطى ذلك
$z = \frac{z\conj w}{\abs w^2}\, w = \lambda w$ مع $\lambda
\geq 0$ (وحالة $w = 0$ بديهية).

(5) تعطي $\abs z = \abs{(z - w) + w} \leq \abs{z-w} + \abs w$ أن $\abs z
- \abs w \leq \abs{z - w}$؛ وبادل بين $z$ و $w$ من أجل الإشارة الأخرى.
\end{proof}

\begin{example}[تمرين كامل على المقياس والعمدة]\label{ex:b1:complex:workout}
اكتب $w = \dfrac{3 + 4\iu}{1 - 2\iu}$ في الشكل الجبري واحسب
مقياسه بطريقتين. بالضرب في \omterm{def:b1:complex:field}{مرافق} المقام:
\[
w = \frac{(3 + 4\iu)(1 + 2\iu)}{(1 - 2\iu)(1 + 2\iu)}
= \frac{3 + 6\iu + 4\iu - 8}{1 + 4}
= \frac{-5 + 10\iu}{5} = -1 + 2\iu .
\]
ومباشرةً: $\abs w = \sqrt{1 + 4} = \sqrt5$. وبقاعدة القسمة
(\cref{prop:b1:complex:rules}~(3)): $\abs w = \frac{\abs{3 +
4\iu}}{\abs{1 - 2\iu}} = \frac5{\sqrt5} = \sqrt5$ --- الجواب نفسه،
دون حاجة إلى الشكل الجبري. والدرس يتعمّم: تنتقل المقاييس
والعمد انتقالًا حسنًا عبر الجداءات والقسمات، وينتقل الجزآن
الحقيقي والتخيلي انتقالًا حسنًا عبر المجاميع. فاختر التمثيل
الذي يلائم العمليات المطروحة، ولا تحوّل إلا مضطرًّا.
\end{example}

\begin{example}[معادلات يدخلها المرافق]\label{ex:b1:complex:conjeq}
حُلَّ في $\C$: $\;z + 2\conj z = 6 + 2\iu$. المعادلة التي تخلط $z$ مع \omterm{def:b1:complex:field}{المرافق} $\conj z$ \emph{ليست} كثيرة حدود في $z$؛ والحركة
الموثوقة هي التفكيك إلى إحداثيتين حقيقيتين. فمع $z = x + \iu y$:
\[
z + 2\conj z = 3x - \iu y ,
\]
فتصير المعادلة $3x = 6$ و $-y = 2$: والحل الوحيد
هو $z = 2 - 2\iu$. (وللتحقق: $(2 - 2\iu) + 2(2 + 2\iu) = 6 +
2\iu$.) وبطريقة أخرى، رافق المعادلة كلها لتحصل على
$\conj z + 2z = 6 - 2\iu$ وحُلَّ الجملة الخطية في
المجهولين $z, \conj z$ --- الجواب نفسه، وهي حيلة مفيدة عندما
تكون المعاملات عقدية. فالمعادلات في $z$ و $\conj z$ هي في الحقيقة
جمل من معادلتين حقيقيتين؛ وتوقّع «الدرجة $1$، حلّ واحد» سليم
هنا، لكن $z\conj z = -1$ (وهي بلا حلّ) يُظهر إخفاق الحدس
الكثيرحدودي بمجرد ظهور الجداءات.
\end{example}

\section{الشكل الأسّي}

\begin{definition}[الأسّي العقدي لعمدة تخيلية]\label{def:b1:complex:expi}
من أجل $\theta \in \R$ نعرّف
\[
\eu^{\iu\theta} = \cos\theta + \iu \sin\theta .
\]
ويمكن كتابة كل $z \neq 0$ على الصورة $z = r\,\eu^{\iu\theta}$ حيث $r =
\abs z > 0$؛ ويكون $\theta$ \emph{عمدةً}\index{عمدة} للعدد $z$،
محدَّدةً بإضافة مضاعف للعدد $2\pi$. والقيمة الواقعة في
$\intoc{-\pi}{\pi}$ هي \emph{العمدة الرئيسة}، وتُكتب
$\arg z$.
\end{definition}

\begin{example}[القيم الأولى، ومتطابقة شهيرة]\label{ex:b1:complex:eulerid}
بقراءة التعريف عند الزوايا الأصلية:
\[
\eu^{\iu\pi/2} = \iu, \qquad
\eu^{\iu\pi} = -1, \qquad
\eu^{2\iu\pi} = 1, \qquad
\eu^{\iu\pi/4} = \frac{\sqrt2}2\,(1 + \iu) .
\]
والثانية، إذا أُعيدت كتابتها $\eu^{\iu\pi} + 1 = 0$، هي متطابقة
أويلر الشهيرة التي تربط $\eu$ و $\iu$ و $\pi$ و $1$ و $0$؛ وهي في
هذه المرحلة من الكتاب فكُّ تعريف لا مبرهنة، أمّا مضمونها الحقيقي
--- أي لماذا تستحق الدالة الأسّية التحليلية في
\cref{ch:b1:functions}، ممدَّدةً إلى العمد التخيلية، الاسم نفسه ---
فتحسمه المتسلسلة الصحيحة في \cref{ch:b1:series}. وفي هذه
الأثناء، يستحق العرض أعلاه أن يُحفظ بوصفه جدول تحويل: فهو
يُستعمل ضمنًا كلما قُرئت عمدةٌ من رسم.
\end{example}

\begin{theorem}[المعادلة الدالية]\label{thm:b1:complex:funceq}
لكل $\theta, \varphi \in \R$:
\[
\eu^{\iu\theta}\, \eu^{\iu\varphi} = \eu^{\iu(\theta + \varphi)},
\qquad
\abs{\eu^{\iu\theta}} = 1,
\qquad
\conj{\eu^{\iu\theta}} = \eu^{-\iu\theta} = (\eu^{\iu\theta})^{-1}.
\]
ومنه فإن $\abs{zw} = \abs z \abs w$ يحمل العمد كذلك:
$\arg(zw) \equiv \arg z + \arg w \pmod{2\pi}$.
\end{theorem}

\begin{proof}
انشر الجداء واستعمل صيغتَي جمع الزاويتين:
\[
(\cos\theta + \iu\sin\theta)(\cos\varphi + \iu\sin\varphi)
= (\cos\theta\cos\varphi - \sin\theta\sin\varphi)
+ \iu\,(\sin\theta\cos\varphi + \cos\theta\sin\varphi),
\]
وهو $\cos(\theta+\varphi) + \iu\sin(\theta+\varphi)$. \omterm{def:b1:complex:field}{والمقياس}
هو $\sqrt{\cos^2\theta + \sin^2\theta} = 1$، وصيغة \omterm{def:b1:complex:field}{المرافق} هي
زوجية جيب التمام وفردية الجيب؛ وهي تقلب $\eu^{\iu\theta}$ لأن
$\eu^{\iu\theta}\eu^{-\iu\theta} = \eu^0 = 1$.
\end{proof}

\begin{corollary}[صيغة دي موافر]\label{cor:b1:complex:demoivre}
من أجل $\theta \in \R$ و $n \in \Z$:
$\;(\cos\theta + \iu\sin\theta)^n = \cos(n\theta) +
\iu\sin(n\theta)$.\index{صيغة دي موافر}
\end{corollary}

\begin{proof}
من أجل $n \geq 0$ نستقرئ. الحالة $n = 0$ تعطي $1 = 1$. وبافتراض
الصيغة من أجل $n$، تعطي المعادلة الدالية
(\cref{thm:b1:complex:funceq})
\[
\bigl(\eu^{\iu\theta}\bigr)^{n+1}
= \bigl(\eu^{\iu\theta}\bigr)^{n}\,\eu^{\iu\theta}
= \eu^{\iu n\theta}\,\eu^{\iu\theta}
= \eu^{\iu(n+1)\theta} ,
\]
وهو الصيغة عند الرتبة $n + 1$. ومن أجل $n < 0$ نكتب $n = -m$
مع $m > 0$: وبما أن مقلوب $\eu^{\iu m\theta}$ هو
$\eu^{-\iu m\theta}$ (حسب \cref{thm:b1:complex:funceq} من جديد)، فإن
\[
\bigl(\eu^{\iu\theta}\bigr)^{-m}
= \bigl(\eu^{\iu m\theta}\bigr)^{-1}
= \eu^{\iu(-m)\theta} . \qedhere
\]
\end{proof}

\begin{example}[نشر $\cos 3\theta$ بصيغة دي موافر]\label{ex:b1:complex:cos3}
نكتب $c = \cos\theta$ و $s = \sin\theta$. تعطي صيغة دي موافر ومبرهنة
ثنائي الحدّ
\[
\cos 3\theta + \iu \sin 3\theta = (c + \iu s)^3
= c^3 - 3cs^2 + \iu\,(3c^2 s - s^3),
\]
وبمطابقة الجزأين الحقيقيين، ثم بتعويض $s^2 = 1 - c^2$:
\[
\cos 3\theta = c^3 - 3c(1 - c^2) = 4\cos^3\theta - 3\cos\theta .
\]
ويعطي الجزء التخيلي $\sin 3\theta = 3\sin\theta -
4\sin^3\theta$ مجانًا: فكل متطابقة عقدية تحمل \emph{متطابقتين}
حقيقيتين دائمًا. والمتطابقة المؤطَّرة، مقروءةً بالمقلوب، هي
مفتاح معادلة التثليث الكلاسيكية: فإنشاء
$\cos(\theta/3)$ انطلاقًا من $\cos\theta$ يعني حلّ المعادلة
التكعيبية $4x^3 -
3x = \cos\theta$، وهنا يتسلّم الأمر جبرُ
\cref{ch:b1:poly}.
\end{example}

\begin{proposition}[صيغتا أويلر]\label{prop:b1:complex:euler}
\[
\cos\theta = \frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2},
\qquad
\sin\theta = \frac{\eu^{\iu\theta} - \eu^{-\iu\theta}}{2\iu}.
\]
\end{proposition}

\begin{proof}
اجمع $\eu^{\iu\theta} = \cos\theta +
\iu\sin\theta$ و$\eu^{-\iu\theta} = \cos\theta - \iu\sin\theta$، ثم اطرحهما.
\end{proof}

\begin{method}[المثلثات عبر الأسّيات]\label{met:b1:complex:trig}
\begin{enumerate}
  \item \emph{خطِّط} المقدار $\cos^p\theta \sin^q\theta$ (أي حوّل القوى إلى
        مجموع من $\cos k\theta$ و $\sin k\theta$): عوّض بصيغتَي
        أويلر، وانشر بمبرهنة ثنائي الحدّ
        (\cref{thm:b1:counting:binomial})، ثم أعد تجميع الحدود المترافقة.
  \item \emph{انشر} $\cos n\theta$ في صورة كثير حدود في $\cos\theta$:
        اكتب $\cos n\theta = \Re\bigl((\cos\theta +
        \iu\sin\theta)^n\bigr)$، وانشر، وحوّل القوى الزوجية
للدالة $\sin$ بالعلاقة $\sin^2 = 1 - \cos^2$.
  \item \emph{اجمع المتسلسلات المثلثية} من نوع $\sum_k \cos k\theta$:
        بأن تتعرف على الجزء الحقيقي لمجموع هندسي $\sum_k
        (\eu^{\iu\theta})^k$.
  \item \emph{تعميل نصف الزاوية}: لكل $p, q$،
        \[
        \eu^{\iu p} + \eu^{\iu q}
        = 2 \cos\tfrac{p - q}{2}\; \eu^{\iu \frac{p+q}{2}},
        \qquad
        \eu^{\iu p} - \eu^{\iu q}
        = 2\iu \sin\tfrac{p - q}{2}\; \eu^{\iu \frac{p+q}{2}} .
        \]
\end{enumerate}
\end{method}

\begin{example}[التخطيط]\label{ex:b1:complex:linearize}
\[
\cos^3\theta
= \Bigl(\frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2}\Bigr)^{\!3}
= \frac{\eu^{3\iu\theta} + 3\eu^{\iu\theta} + 3\eu^{-\iu\theta} +
\eu^{-3\iu\theta}}{8}
= \frac{\cos 3\theta + 3\cos\theta}{4}.
\]
وهذه الصورة تُكامل فورًا --- وهو بالضبط سبب أهمية التخطيط في
\cref{ch:b1:integration}. والجداء المختلط يعمل بالطريقة
نفسها، لكن بصيغتَي أويلر معًا:
\[
\sin^2\theta\cos^2\theta
= \Bigl(\frac{\sin2\theta}2\Bigr)^{\!2}
= \frac{1}{4}\cdot
\Bigl(\frac{\eu^{2\iu\theta} - \eu^{-2\iu\theta}}{2\iu}
\Bigr)^{\!2}
= \frac{2 - \eu^{4\iu\theta} - \eu^{-4\iu\theta}}{16}
= \frac{1 - \cos4\theta}{8} ,
\]
حيث وفّر اختصار الزاوية المضاعفة في الخطوة الأولى نشرًا ثنائيًا
--- ويجدر دائمًا البحث عنه قبل الحوسبة الآلية.
\end{example}

\begin{example}[مجموع مثلثي ثنائي]\label{ex:b1:complex:binomsum}
من أجل $n \in \N$ و $\theta \in \R$، احسب $S =
\sum_{k=0}^{n}\binom nk \cos k\theta$. تعرّف على الجزء الحقيقي
لنشر ثنائي:
\[
S = \Re\sum_{k=0}^n \binom nk \bigl(\eu^{\iu\theta}\bigr)^k
= \Re\bigl(1 + \eu^{\iu\theta}\bigr)^n ,
\]
ثم عمّل نصف الزاوية (\cref{met:b1:complex:trig}~(4)): $1 +
\eu^{\iu\theta} = 2\cos\frac\theta2\,\eu^{\iu\theta/2}$، ومنه
\[
S = \Re\Bigl(2^n\cos^n\frac\theta2\;\eu^{\iu n\theta/2}\Bigr)
= 2^n \cos^n\frac\theta2\,\cos\frac{n\theta}2 .
\]
ويعطي الجزء التخيلي $\sum_k\binom nk\sin k\theta =
2^n\cos^n\frac\theta2\sin\frac{n\theta}2$ مجانًا. وللتحقق:
$\theta = 0$ يستعيد $\sum\binom nk = 2^n$، و $\theta = \pi$
يعطي $S = 0$ من أجل $n \geq 1$ (إذ ينعدم كل عامل $\cos\frac\pi2$)،
أي المجموع المتناوب لسطر
\cref{ex:b1:counting:binomial}. والطريقة --- «انظر إلى مجموع
جيوب التمام بوصفه ظلًّا لقوة عقدية، ثم عمّل أنصاف الزوايا»
--- هي بالضبط طريقة \cref{exo:b1:complex:6}، مع إحلال مبرهنة
ثنائي الحدّ محلّ المتسلسلة الهندسية.
\end{example}

\section{جذور الأعداد العقدية}

\begin{theorem}[الجذور من الرتبة $n$]\label{thm:b1:complex:roots}
لتكن $a = r\,\eu^{\iu\alpha} \neq 0$ و $n \in \N^*$. للمعادلة
$z^n = a$ بالضبط $n$ حلًّا:
\[
z_k = r^{1/n}\, \eu^{\iu\left(\frac{\alpha}{n} +
\frac{2k\pi}{n}\right)},
\qquad k = 0, 1, \dots, n - 1 .
\]
\end{theorem}

\begin{proof}
نكتب $z = \rho\,\eu^{\iu\theta}$ (حيث $\rho > 0$). عندئذ يكون $z^n =
\rho^n \eu^{\iu n\theta} = r \eu^{\iu\alpha}$ إذا وفقط إذا كان $\rho^n
= r$ (المقاييس) و$n\theta \equiv \alpha \pmod{2\pi}$ (العمد)،
أي $\rho = r^{1/n}$ و$\theta = \frac{\alpha}{n} +
\frac{2k\pi}{n}$ من أجل $k \in \Z$ ما. ويبقى أن نرى متى يعطي عددان
صحيحان $k, k'$ العدد نفسه: وهذا يحدث تحديدًا عندما تختلف
الزاويتان بمضاعف للعدد $2\pi$،
\[
\frac{2k\pi}n - \frac{2k'\pi}n \in 2\pi\Z
\iff \frac{k - k'}n \in \Z
\iff n \mid k - k' .
\]
وبالقسمة الإقليدية، يكون كل $k \in \Z$ موافقًا بترديد $n$ لعنصر
واحد بالضبط من $\{0, 1, \dots, n-1\}$، فيسرد هذا المجال
كل حلّ مرة واحدة ويكون العدد $n$ بالضبط. (والحجة نفسها،
مطبَّقةً داخل $\mathbb U_n$، تبيّن أن الجذور تكوّن مضلعًا منتظمًا
ذا $n$ ضلعًا: إذ تدور القيم المتتالية للدليل $k$ بالزاوية الثابتة
$\frac{2\pi}n$.)
\end{proof}

\begin{example}[الجذور التكعيبية للعدد $-27$]\label{ex:b1:complex:cuberoots}
حُلَّ $z^3 = -27$. الشكل الأسّي للطرف الأيمن: $-27 =
27\,\eu^{\iu\pi}$، ومنه فالجذور الثلاثة هي
\[
z_k = 3\,\eu^{\iu(\frac\pi3 + \frac{2k\pi}3)}, \quad k = 0, 1, 2 :
\qquad
z_0 = 3\eu^{\iu\pi/3} = \frac32 + \frac{3\sqrt3}2\,\iu,
\quad
z_1 = -3,
\quad
z_2 = \conj{z_0} .
\]
وتحققان. أولًا: الجذر الحقيقي $-3$ هو البيّن، والآخران دورانان
له بالزاويتين $\pm\frac{2\pi}3$ --- أي، على نحو مكافئ،
$-3j$ و $-3j^2$. وثانيًا يؤكد الجبر ذلك: $z^3 + 27 = (z +
3)(z^2 - 3z + 9)$، ومميّز المعادلة من الدرجة الثانية هو $9 - 36 =
-27 < 0$ وجذراها $\frac{3 \pm 3\iu\sqrt3}2 = z_0, \conj{z_0}$.
والفكرة النافذة: من أجل الأطراف اليمنى الحقيقية، تأتي الجذور غير
الحقيقية دائمًا في أزواج مترافقة، فيكون رسم \omterm{def:b1:logic:sets}{مجموعة} الحلول
متناظرًا بالنسبة إلى المحور الحقيقي --- وهو تمهيد لمبرهنة
التعميل الحقيقي في \cref{ch:b1:poly}.
\end{example}

\begin{example}[طرف أيمن غير حقيقي]\label{ex:b1:complex:fourthroots}
حُلَّ $z^4 = -8 + 8\iu\sqrt3$. الشكل الأسّي للطرف الأيمن:
\omterm{def:b1:complex:field}{المقياس} $\sqrt{64 + 192} = 16$، \omterm{def:b1:complex:expi}{والعمدة} $\theta$ حيث
$\cos\theta = -\frac12$ و$\sin\theta = \frac{\sqrt3}2$، أي
$\theta = \frac{2\pi}3$. والجذور الأربعة هي
\[
z_k = 2\,\eu^{\iu(\frac\pi6 + \frac{k\pi}2)}, \quad k = 0, 1, 2,
3 :
\qquad
z_0 = \sqrt3 + \iu,\quad z_1 = \iu z_0 = -1 + \iu\sqrt3,
\]
\[
z_2 = -z_0 = -\sqrt3 - \iu,\qquad z_3 = -\iu z_0 = 1 - \iu\sqrt3 .
\]
(وللتحقق: $z_0^2 = 2 + 2\iu\sqrt3$، ومنه $z_0^4 = (2 +
2\iu\sqrt3)^2 = 4 - 12 + 8\iu\sqrt3 = -8 + 8\iu\sqrt3$.) وبمجرد
العثور على \emph{جذر واحد} تأتي الثلاثة الأخرى مجانًا: فهي
دوراناته المتتالية بالزاوية $\frac\pi2$، أي جداءاته بالجذور
الرابعة للوحدة --- وهو البناء العام وراء
\cref{thm:b1:complex:roots}، ويجدر استغلاله قبل إعادة حساب
كل جذر من الصفر. ولا تناظر مترافق هذه المرة: فالطرف الأيمن ليس
حقيقيًا.
\end{example}

\begin{definition}[جذور الوحدة]\label{def:b1:complex:unity}
\emph{الجذور من الرتبة $n$ للوحدة}\index{جذور الوحدة} هي
حلول المعادلة $z^n = 1$:
\[
\mathbb{U}_n = \bigl\{\, \omega^k : k = 0, \dots, n-1 \,\bigr\},
\qquad \omega = \eu^{2\iu\pi/n}.
\]
وهي تكوّن زمرة بالنسبة إلى الضرب (\cref{ch:b1:structures}) وتقع
عند رؤوس مضلع منتظم ذي $n$ ضلعًا مرسوم داخل دائرة الوحدة.
\end{definition}

\begin{proposition}[مجموع جذور الوحدة]\label{prop:b1:complex:sumroots}
من أجل $n \geq 2$، يكون مجموع الجذور من الرتبة $n$ للوحدة معدومًا:
$\sum_{k=0}^{n-1} \omega^k = 0$.
\end{proposition}

\begin{proof}
مجموع هندسي أساسه $\omega \neq 1$:
$\sum_{k=0}^{n-1} \omega^k = \frac{\omega^n - 1}{\omega - 1} = 0$
لأن $\omega^n = 1$.
\end{proof}

\begin{example}[قراءة الجزأين الحقيقي والتخيلي]\label{ex:b1:complex:sumrootsparts}
تفكيك $\sum_{k=0}^{n-1}\omega^k = 0$ إلى جزأين حقيقي وتخيلي
يعطي متطابقتين مثلثيتين مجانًا:
\[
\sum_{k=0}^{n-1}\cos\frac{2k\pi}n = 0,
\qquad
\sum_{k=0}^{n-1}\sin\frac{2k\pi}n = 0
\qquad (n \geq 2).
\]
وهندسيًا: مركز ثقل مضلع منتظم ذي $n$ ضلعًا مرسوم داخل دائرة
الوحدة هو مركزها --- فالرؤوس تتوازن تمامًا.
ومن أجل $n = 5$ تعطي المتطابقة الأولى $1 + 2\cos\frac{2\pi}5 +
2\cos\frac{4\pi}5 = 0$ (بازدواج $k$ مع $n - k$)، وهي نقطة انطلاق
حساب $\cos\frac{2\pi}5$ في \cref{exo:b1:complex:8}.
\end{example}

\begin{example}[الجذور التربيعية في الشكل الجبري]\label{ex:b1:complex:sqrt}
لحلّ $z^2 = 3 + 4\iu$ دون مثلثات، ضع $z = x + \iu y$:
\[
x^2 - y^2 = 3, \qquad 2xy = 4, \qquad x^2 + y^2 = \abs{3 + 4\iu} = 5 .
\]
وبجمع الأولى والأخيرة: $x^2 = 4$، ومنه $x = \pm 2$، ثم $y = 2/x =
\pm 1$ مع اقتران الإشارتين \emph{نفسه} (إذ $xy = 2 > 0$): $z = \pm(2 +
\iu)$. وبضمّ ذلك إلى الصيغة المعتادة، يُحلّ كل معادلة من الدرجة
الثانية ذات معاملات عقدية
(\cref{exo:b1:complex:7}).
\end{example}

\section{الأعداد العقدية وهندسة المستوي}

\begin{proposition}[القاموس الهندسي]\label{prop:b1:complex:geometry}
نطابق النقطة $M(x, y)$ من المستوي مع \emph{لاحقتها} $z = x
+ \iu y$\index{لاحقة}. من أجل نقاط متمايزة $A, B, C$ لواحقها $a, b,
c$:
\begin{enumerate}
  \item المقدار $\abs{b - a}$ هو المسافة $AB$؛
  \item والمقدار $\arg\dfrac{c - a}{b - a}$ هو الزاوية بين المتجهتين
        $\vect{AB}$ و $\vect{AC}$ (بترديد $2\pi$)؛
  \item والنقاط $A, B, C$ في استقامية إذا وفقط إذا كان $\dfrac{c - a}{b - a} \in
        \R$؛ والمستقيمان $AB$ و $AC$ متعامدان إذا وفقط إذا كان
        $\dfrac{c - a}{b - a} \in \iu\R$.
\end{enumerate}
\end{proposition}

\begin{proof}
النقطة (1) هي تعريف \omterm{def:b1:complex:field}{المقياس} مطبَّقًا على $b - a$، وهو لاحقة
$\vect{AB}$. و (2): اكتب $b - a = r\eu^{\iu\theta}$ و$c - a =
s\eu^{\iu\varphi}$؛ عندئذ تكون \omterm{def:b1:complex:expi}{عمدة} $\frac{c-a}{b-a} = \frac sr
\eu^{\iu(\varphi - \theta)}$ هي $\varphi - \theta$، وهي
الزاوية من $\vect{AB}$ إلى $\vect{AC}$. و (3): الاستقامية تعني
زاويةً $0$ أو $\pi$، أي \omterm{def:b1:complex:expi}{عمدةً} في $\pi\Z$، أي أن القسمة حقيقية؛
والتعامد يعني زاوية $\pm\frac\pi2$، أي أن القسمة تخيلية بحتة.
(والقسمة غير معدومة لأن $C \neq A$.)
\end{proof}

\begin{remark}[استراحة: $\C$ هو المستوي مزوَّدًا بضرب]\label{rem:b1:complex:interlude}
يجدر الوقوف عند ما يجعل هذا الفصل ممكنًا أصلًا:
المستوي $\R^2$ يحمل جمعًا في كل اتجاه، لكن لا ضرب
مُنزَلًا فيه --- و $\C$ \emph{هو} المستوي
مزوَّدًا بواحد، حيث يدير الضرب في عدد مثبَّت ويكبّر أو يصغّر.
وستُستثمر هذه البنية الواحدة ثلاث مرات أخرى في هذا المجلد. ففي
\cref{ch:b1:matrices}، يعود الضرب في
$a + \iu b$ في صورة المصفوفة $2 \times 2$ ذات السطرين $(a,
-b)$ و $(b, a)$: فحساب الأعداد العقدية عائلة أولى ملموسة كل
الملموسية من جداءات المصفوفات. وفي \cref{ch:b1:euclid}، يتبيّن
أن الصيغة $\Re(\conj z\,w)$ هي الجداء السلّمي، وأن
$\abs z$ هو المعيار الإقليدي: فالمتراجحة المثلثية المبرهن عليها هنا
هي نموذج قصة كوشي--شوارتز هناك. وفي
\cref{ch:b1:curves}، أحسنُ كتابةٍ لنقطة متحركة هي $t \mapsto
z(t)$، فتصير السرعة والتسارع مشتقات عقدية القيم
--- والحركة الدائرية، مثلًا، ليست إلا $z(t) =
R\,\eu^{\iu\omega t}$. ضربٌ واحد جيد، وأربعة فصول من
الأرباح.
\end{remark}

\begin{proposition}[التطبيقات $z \mapsto az + b$]\label{prop:b1:complex:similitude}
ليكن $a \in \C^*$ و $b \in \C$. التحويل $f(z) = az + b$ للمستوي:
\begin{itemize}
  \item يكون انسحابًا إذا كان $a = 1$؛
  \item وإلا كانت له نقطة صامدة وحيدة $\zeta = \frac{b}{1-a}$،
        ولدينا $f(z) - \zeta = a\,(z - \zeta)$: أي أن $f$ هو دوران
        مركزه $\zeta$ وزاويته $\arg a$، مركَّبًا مع التحاكي
        الذي مركزه $\zeta$ ونسبته $\abs a$.
\end{itemize}
وبوجه خاص فإن $z \mapsto \eu^{\iu\theta} z$ هو الدوران بالزاوية
$\theta$ حول المبدأ، و $\conj z$ هو التناظر بالنسبة إلى المحور
الحقيقي.
\end{proposition}

\begin{proof}
إذا كان $a = 1$ فإن $f(z) = z + b$ ينسحب بالمتجهة ذات اللاحقة $b$.
وإذا كان $a \neq 1$ فلمعادلة النقطة الصامدة $z = az + b$ الحلُّ
الوحيد $\zeta = \frac{b}{1-a}$، وعندئذ $f(z) - \zeta = az + b -
(a\zeta + b) = a(z - \zeta)$. وبكتابة $a = \abs a\, \eu^{\iu\arg a}$،
يضرب الضربُ في $a$ المسافاتِ إلى $\zeta$ في $\abs a$ ويضيف
$\arg a$ إلى الزوايا عند $\zeta$، وهو التركيب المعلن.
\end{proof}

\begin{example}[تصنيف تطبيق $z \mapsto az+b$]\label{ex:b1:complex:classify}
خذ $f(z) = \iu z + 1$. هنا $a = \iu \neq 1$: فالنقطة الصامدة هي
\[
\zeta = \frac{b}{1 - a} = \frac1{1 - \iu}
= \frac{1 + \iu}{2},
\]
وبما أن $\abs a = 1$ فلا تحاكي البتة: أي أن $f$ هو الدوران
الخالص الذي مركزه $\frac{1+\iu}2$ وزاويته $\arg \iu = \frac\pi2$.
وللتحقق: $f(\zeta) = \iu\,\frac{1+\iu}2 + 1 = \frac{\iu - 1}2 + 1 =
\frac{1 + \iu}2 = \zeta$ و $f(0) = 1$ و $f(1) = 1 + \iu$ و$f(1 +
\iu) = \iu\,(1 + \iu) + 1 = \iu$: فالنقاط الأربع $0, 1, 1+\iu, \iu$
من المربع الوحدوي تدور حول مركزها $\zeta$، ربع دورة في كل
مرة --- وهو بالضبط ما يجب أن يفعله دوران بالزاوية $\frac\pi2$ حول
مركز المربع.
\end{example}

\begin{remark}[مزالق شائعة مع المقاييس والعمد]\label{rem:b1:complex:pitfalls}
\begin{enumerate}
  \item \emph{لا متراجحات في $\C$.} كتابة $z \leq w$ من أجل
        أعداد غير حقيقية بلا معنى؛ فلا يقارَن إلا المقاييس
        والأجزاء الحقيقية والأجزاء التخيلية.
  \item \emph{الرمز $\sqrt{\phantom z}$ محجوز \omterm{def:b1:logic:sets}{للمجموعة} $\R_+$.} لكل
        عدد عقدي غير معدوم جذران تربيعيان \emph{اثنان} ولا
        أفضلية لأحدهما: فاكتب «ليكن $\delta$ جذرًا تربيعيًا
        للمقدار $\Delta$» (يُحسب كما في
        \cref{ex:b1:complex:sqrt})، ولا تكتب $\sqrt\Delta$ أبدًا ---
        فالقاعدة $\sqrt{ab} = \sqrt a\sqrt b$ تخفق أصلًا عند
        $a = b = -1$.
  \item \emph{العمد تعيش بترديد $2\pi$.} من
        $\eu^{\iu\alpha} = \eu^{\iu\beta}$ استنتج $\alpha
        \equiv \beta \pmod{2\pi}$، لا $\alpha = \beta$؛
        فنسيان المقدار $2k\pi$ هو ما يجعل مجموعات حلول $z^n = a$
        تفقد $n - 1$ من عناصرها $n$.
  \item \emph{المقدار $\abs{z + w}$ ليس $\abs z + \abs w$.} التساوي
        في المتراجحة المثلثية هو الحالة الاستثنائية التي تقع فيها
        النقاط في استقامية (\cref{prop:b1:complex:rules}~(4))؛ وفي الحالة العامة
        يجب تقدير \omterm{def:b1:complex:field}{مقياس} المجموع، لا حسابه.
\end{enumerate}
\end{remark}

\begin{example}\label{ex:b1:complex:equilateral}
النقاط $A, B, C$ (ذات اللواحق $a, b, c$، المتمايزة مثنى مثنى) تكوّن
مثلثًا متساوي الأضلاع رؤوسه بالترتيب المباشر (عكس عقارب الساعة)
إذا وفقط إذا كان $\frac{c - a}{b - a} = \eu^{\iu\pi/3}$: إذ إن الدوران
الذي مركزه $A$ وزاويته $\frac\pi3$ يرسل $B$ إلى $C$. وتلتقط
المعادلةُ المتناظرة $a^2 + b^2 + c^2 = ab
+ bc + ca$ التوجيهين معًا (\cref{exo:b1:complex:10}).
\end{example}

\begin{remark}[أين يُستعمل هذا الفصل]\label{rem:b1:complex:whereused}
الشكل الأسّي هو الأداة الحسابية الأكثر إعادةَ استعمال في المجلد.
\omterm{def:b1:complex:unity}{فجذور الوحدة} تصير المثال المعياري للزمرة الدائرية في
\cref{ch:b1:structures} وتحرّك تعميل $X^n - 1$ في
\cref{ch:b1:poly}؛ ويُنتج تجميع الجذور المترافقة هناك
التعميلات الحقيقية التي تستعملها الكسور الجزئية في
\cref{ch:b1:fractions}. والتخطيط (\cref{met:b1:complex:trig})
هو التحضير المعياري لمكاملة القوى المثلثية في
\cref{ch:b1:integration}، وللمعادلات المميِّزة في
\cref{ch:b1:diffeq} جذور عقدية تُنتج أجزاؤها الحقيقية
والتخيلية الحلول المتذبذبة $\eu^{\lambda t}\cos\omega
t$. ويعود القاموس الهندسي في ثوب مصفوفي: فتصير الدورانات
والتشابهات المصفوفات المتعامدة في
\cref{ch:b1:matrices,ch:b1:euclid}، وأحسنُ كتابةٍ للمنحنيات ذات الوسيط في
\cref{ch:b1:curves} هي في الغالب صورة التطبيقات $t \mapsto z(t)
\in \C$.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b1:complex:1}
اكتب في الشكل الأسّي: $1 + \iu$؛ و $\;\sqrt 3 - \iu$؛ و $\;-5$؛

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:2}
احسب $(1 + \iu)^{20}$. ومن أجل أيّ $n \in \N$ يكون $(1 + \iu)^n$
عددًا حقيقيًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:3}
صِف هندسيًا \omterm{def:b1:logic:sets}{مجموعة} الأعداد $z \in \C$ التي تحقق:
$\abs{z - 2} = \abs{z + \iu}$؛ و $\;\abs{z - 1} = 2$؛
و$\;\dfrac{z - 1}{z + 1} \in \iu\R$ (من أجل $z \neq -1$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:4}
خطِّط $\sin^4\theta$، وانشر $\cos 4\theta$ في صورة كثير حدود في
$\cos\theta$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:5}
حُلَّ $z^3 = 8\iu$ وضع الحلول على رسم. وحُلَّ
$z^4 = -4$ وعمّل $X^4 + 4$ إلى كثيرَي حدود حقيقيين من الدرجة الثانية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:6}
من أجل $\theta \in \R$ حيث $\eu^{\iu\theta} \neq 1$ ومن أجل $n \in \N$،
احسب
\[
C_n = \sum_{k=0}^{n} \cos k\theta
\qquad\text{و}\qquad
S_n = \sum_{k=0}^{n} \sin k\theta
\]
بجمع المتسلسلة الهندسية $\sum_k \eu^{\iu k\theta}$ وباستعمال
تعميل نصف الزاوية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:7}
حُلَّ في $\C$: $z^2 - (3 + 4\iu) z + (-1 + 5\iu) = 0$. \emph{(احسب
المميِّز، واستخرج جذريه التربيعيين كما في
\cref{ex:b1:complex:sqrt}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:8}
لتكن $\omega = \eu^{2\iu\pi/5}$.
\begin{enumerate}
  \item برّر أن $1 + \omega + \omega^2 + \omega^3 + \omega^4 = 0$.
  \item ضع $u = \omega + \omega^4$ و$v = \omega^2 + \omega^3$.
        احسب $u + v$ و $uv$، واستنتج أن $u$ و $v$ جذرا
        كثير الحدود $X^2 + X - 1$.
  \item استنتج أن $\cos\frac{2\pi}{5} = \frac{\sqrt 5 - 1}{4}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:9}
برهن على أنه لكل $z, w \in \C$ (متطابقة متوازي الأضلاع):
\[
\abs{z + w}^2 + \abs{z - w}^2 = 2\abs z^2 + 2\abs w^2 ,
\]
وفسّرها هندسيًا في متوازي الأضلاع ذي الرؤوس $0,
z, w, z + w$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:complex:10}
برهن على أن ثلاث نقاط متمايزة مثنى مثنى لواحقها $a, b, c$ تكوّن
مثلثًا متساوي الأضلاع (بأيّ توجيه) إذا وفقط إذا كان
\[
a^2 + b^2 + c^2 = ab + bc + ca .
\]
\emph{إرشاد: الحالة المباشرة هي $\frac{c-a}{b-a} = -j^2$ والحالة
غير المباشرة $\frac{c-a}{b-a} = -j$، حيث يحقق $j = \eu^{2\iu\pi/3}$
العلاقة $j^2 + j + 1 = 0$؛ أو عمّل $a + jb + j^2c$ و$a + j^2 b
+ jc$.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:complex:11}
من أجل $n \in \N^*$، احسب $P = \prod_{k=1}^{n-1}
\bigl(1 - \omega^k\bigr)$ حيث $\omega = \eu^{2\iu\pi/n}$.
\emph{إرشاد: $X^n - 1 = \prod_{k=0}^{n-1} (X - \omega^k)$؛ اقسم على
$X - 1$ وعوّض $X = 1$.} واستنتج
$\prod_{k=1}^{n-1} \sin\frac{k\pi}{n} = \dfrac{n}{2^{n-1}}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:12}
ليكن $n \geq 2$. حُلَّ المعادلة $(z + 1)^n = (z - 1)^n$ في $\C$:
وبيّن أن لها $n - 1$ حلًّا بالضبط، وكلها تخيلية بحتة، وهي
\[
z_k = -\iu\,\frac{\cos\frac{k\pi}{n}}{\sin\frac{k\pi}{n}},
\qquad k = 1, \dots, n - 1 .
\]
\emph{إرشاد: العدد $z = 1$ ليس حلًّا، فاقسم واستعمل \omterm{def:b1:complex:unity}{جذور
الوحدة}؛ ثم طبّق تعميل نصف الزاوية من
\cref{met:b1:complex:trig}.}
\end{exercise}

\section{مسألة: مبرهنة نابليون}

\begin{problem}\label{pb:b1:complex:1}
ارفع على كل ضلع من أضلاع مثلث \emph{كيفيّ} مثلثًا متساوي
الأضلاع نحو الخارج: تكوّن مراكز تلك المثلثات الثلاثة دائمًا
مثلثًا متساوي الأضلاع. هذه هي مبرهنة نابليون
\index{مبرهنة نابليون} --- وهي \omterm{def:b1:logic:statement}{عبارة} لا سبب ظاهرًا لصحتها،
يبرهن عليها جبرُ $j = \eu^{2\iu\pi/3}$ في ثلاثة أسطر من الحساب.
وتبني هذه المسألة صندوق العدّة كاملًا
(الدورانات، والتشابهات المباشرة، ومعيار $j$ في
\cref{exo:b1:complex:10})، وتبرهن على مبرهنة نابليون الداخلية
والخارجية، وتحدّد الحالة المنحلّة، وتختم بجوهرة ثانية
من المدرسة نفسها: مبرهنة فان سخوتن في المثلثات المتساوية الأضلاع
ومتراجحة بطليموس. وفي كل ما يلي، نطابق نقاط المستوي
بلواحقها.

\medskip
\noindent\textbf{الجزء 1 --- العدد $j$ والدورانات.}
\begin{enumerate}
  \item احسب الشكل الجبري للعدد $j$، ثم $j^2$ و $j^3$ و$1 + j
        + j^2$ و $\conj j$ و $j^{-1}$. وضع $1$ و $j$ و $j^2$ على
        رسم تخطيطي لدائرة الوحدة.
  \item بيّن أن الدوران الذي مركزه $a$ وزاويته $\theta$ هو
        $r(z) = a + \eu^{\iu\theta}(z - a)$، وأن
        تركيب دورانين زاويتاهما $\theta$ و
        $\theta'$ هو دوران زاويته $\theta + \theta'$ إذا كان
        $\theta + \theta' \notin 2\pi\Z$، وانسحابٌ فيما عدا
        ذلك. (استعمل \cref{prop:b1:complex:similitude}.)
  \item ليكن $b \neq c$. بيّن أنه توجد بالضبط نقطتان $p$
        تجعلان $(b, c, p)$ متساوي الأضلاع، وهما $p = b +
        \eu^{\pm\iu\pi/3}(c - b)$. ومن أجل مثلث \emph{مباشر}
        $(a, b, c)$، تحقق على المثال $a = 0$ و $b = 1$ و$c =
        \iu$ من أن الاختيار $\eu^{-\iu\pi/3}$ هو الواقع على
        الجهة البعيدة من المستقيم $BC$ عن $a$ --- وهو الرأس
        \emph{الخارجي}.
  \item انطلاقًا من حلّ \cref{exo:b1:complex:10}، يكون $(a, b, c)$
        مثلثًا مباشرًا متساوي الأضلاع إذا وفقط إذا كان
        $a + jb + j^2c = 0$. برهن على متطابقتَي الدوران
        \[
        b + jc + j^2a = j^2\,(a + jb + j^2c),
        \qquad
        c + ja + j^2b = j\,(a + jb + j^2c),
        \]
        واستنتج أن المعيار لا يتغير بالتبديل الدائري
        للثلاثية $(a, b, c)$.
  \item بيّن أنه إذا كان $a + jb + j^2c = 0$ وانطبقت نقطتان من
        النقاط الثلاث فإن الثلاث تنطبق. (فالمعيار يميّز إذن
        تمييزًا تامًّا: مثلث مباشر متساوي الأضلاع، أو نقطة
        واحدة.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- التشابهات المباشرة.}
\begin{enumerate}[resume]
  \item بيّن أن التطبيقات $f(z) = az + b$ حيث $a \in \C^*$
        (وتُسمّى \emph{التشابهات المباشرة}) مستقرة بالتركيب
        وبالقلب: أي أن تركيب اثنين منها، ومقلوب أيٍّ منها،
        على هذه الصورة كذلك. (وبلغة
        \cref{ch:b1:structures}: تكوّن زمرة.)
  \item من أجل $z_1 \neq z_2$ و $w_1 \neq w_2$ معطاة، بيّن أنه يوجد
        تشابه مباشر \emph{وحيد} $f$ يحقق $f(z_1) = w_1$ و
        $f(z_2) = w_2$، وأعط $a$ و $b$ صراحةً.
  \item بيّن أن التشابه المباشر $f(z) = az + b$ يضرب
        جميع المسافات في $\abs a$ ويحافظ على \emph{شكل}
        $\frac{c - a'}{b' - a'}$ أيّ مثلث $(a', b', c')$
        \emph{(إذ يُضرب كلٌّ من البسط والمقام في
        $a$)}. واستنتج أن مثلثين يتشابهان تشابهًا مباشرًا
        تحديدًا عندما يتساوى شكلاهما.
  \item عيّن تعيينًا كاملًا التشابهَ المباشر الذي يحقق $f(0) = 1$
        و $f(1) = \iu$: أعط $a$ و $b$ والنقطة الصامدة والنسبة
        والزاوية.
  \item ليكن $\alpha + \beta = 1$. بيّن أن «النقطة الموزونة»
        $g(a', b') = \alpha a' + \beta b'$ تتبادل مع كل
        تشابه مباشر: $f(\alpha a' + \beta b') = \alpha f(a')
        + \beta f(b')$. واستنتج أن مراكز الثقل والأوساط ومراكز
        نابليون أدناه تنتقل جميعها بالتشابهات
        --- وهي الرخصة الجبرية وراء كل حجة من نوع «دون فقدان
        العمومية، ضع الدائرة المحيطة على دائرة الوحدة».
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- مبرهنة نابليون.} ليكن $(a, b, c)$
مثلثًا مباشرًا. ارفع على كل ضلع المثلثَ المتساوي الأضلاع
الخارجي (السؤال 3) ولتكن $n_a$ و $n_b$ و $n_c$ مراكز
(مراكز ثقل) المثلثات المرفوعة على $[b, c]$ و $[c, a]$ و
$[a, b]$ على الترتيب.
\begin{enumerate}[resume]
  \item بيّن أن
        \[
        n_a = \alpha b + \beta c, \qquad
        n_b = \alpha c + \beta a, \qquad
        n_c = \alpha a + \beta b,
        \qquad\text{حيث}\quad
        \alpha = \frac{3 + \iu\sqrt3}{6},\ \beta = \conj\alpha .
        \]
  \item تحقق من أن $\alpha + \beta = 1$، واستنتج أن للمثلث
        $(n_a, n_b, n_c)$ \emph{مركز الثقل نفسه} الذي للمثلث
        $(a, b, c)$.
  \item باستعمال متطابقات السؤال 4، بيّن أن
        \[
        n_a + j\,n_b + j^2 n_c
        = (\alpha j^2 + \beta j)\,(a + jb + j^2c) .
        \]
  \item احسب $\alpha j + \beta$ واستنتج: $\alpha j^2 +
        \beta j = j(\alpha j + \beta) = 0$، ومنه $n_a + j n_b + j^2
        n_c = 0$ من أجل \emph{كل} مثلث: أي أن مثلث نابليون
        الخارجي مباشر ومتساوي الأضلاع (أو نقطة). وبهذا تكون
        مبرهنة نابليون مبرهنًا عليها.
  \item ارفع المثلثات المتساوية الأضلاع نحو \emph{الداخل}
        بدلًا من ذلك (بالاختيار $\eu^{+\iu\pi/3}$ في السؤال 3)
        ولتكن $m_a, m_b,
        m_c$ مراكزها. بيّن أن $m_a = \beta b + \alpha c$ (وكذلك
        بالتبديل الدائري)، ثم برهن على أن $m_a + j^2 m_b + j\,m_c = 0$: أي أن
        مثلث نابليون الداخلي متساوي الأضلاع كذلك، لكن
        بالتوجيه المعاكس.
  \item حدّد الانحلال: بيّن أن $n_a = n_b = n_c$ يحدث
        تحديدًا عندما يكون $a + j^2 b + jc = 0$، أي عندما يكون
        $(a, b, c)$ مثلثًا \emph{غير مباشر} متساوي الأضلاع.
        \emph{(تنطبق نقطتان من $n_a, n_b, n_c$ إذا وفقط إذا
        انطبقت الثلاث، حسب السؤال 5؛ ثم استعمل $n_a + j^2 n_b + j n_c = j(\alpha +
        \beta j)(a + j^2b + jc)$ وتحقق من أن $\alpha + \beta j \neq
        0$.)}
  \item أجرِ الحساب كله على المثلث $a = 0$ و
        $b = 1$ و $c = \iu$: أعط $n_a, n_b, n_c$ بالضبط، وتحقق
        بحساب مباشر لمربعات أطوال الأضلاع الثلاثة من أن المثلث
        متساوي الأضلاع، ومربع ضلعه
        $\frac{2 + \sqrt3}{3}$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- بطليموس وفان سخوتن.}
\begin{enumerate}[resume]
  \item برهن على المتطابقة التالية، الصالحة لكل الأعداد العقدية
        $a, b, c, d$:
        \[
        (a - b)(c - d) + (a - d)(b - c) = (a - c)(b - d) .
        \]
  \item استنتج \emph{متراجحة بطليموس}\index{متراجحة
        بطليموس}: من أجل أيّ أربع نقاط $A, B, C, D$،
        \[
        AC \cdot BD \;\leq\; AB \cdot CD + AD \cdot BC ,
        \]
        مع التساوي إذا وفقط إذا وقع $(a-b)(c-d)$ و
        $(a-d)(b-c)$ على نصف مستقيم واحد مبدؤه $0$.
  \item بيّن أنه من أجل $\theta, \varphi \in \R$،
        $\abs{\eu^{\iu\theta} - \eu^{\iu\varphi}} =
        2\,\abs{\sin\frac{\theta - \varphi}2}$.
  \item (مبرهنة فان سخوتن\index{مبرهنة فان سخوتن}) لتكن
        $(a, b, c) = (1, j, j^2)$ --- وهذا لا يفقد أيّ
        عمومية بين المثلثات المباشرة المتساوية الأضلاع حسب الجزء 2 ---
        وليكن $p = \eu^{\iu\theta}$ مع $\theta \in
        \intoo{2\pi/3}{4\pi/3}$، وهي نقطة من الدائرة المحيطة على
        القوس $BC$ الذي لا يحتوي $A$. برهن على أن
        \[
        PA = PB + PC .
        \]
        \emph{(عبّر عن المسافات الثلاث بالسؤال 20 واستعمل
        صيغة تحويل المجموع إلى جداء.)}
  \item تحقق عند $p = -1$ من أن هذه هي بالضبط حالة التساوي
        في متراجحة بطليموس من أجل الترتيب الدائري $A, B, P, C$:
        احسب $(a - b)(p - c)$ و $(a - c)(b - p)$ وتأكّد
        من أن نسبتهما عدد حقيقي موجب.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- توليفة ختامية.}
\begin{enumerate}[resume]
  \item احسب مثلث نابليون الخارجي للمثلث المتساوي الأضلاع
        $(1, j, j^2)$ نفسه، وصِف النتيجة
        هندسيًا.
  \item أين استعملت المسألة بالضبط: (أ) الضرب بوصفه
        دورانًا؛ (ب) بنية الزمرة للتشابهات و
        الشكل بوصفه ثابتًا؛ (ج) تعميل نصف الزاوية في
        \cref{met:b1:complex:trig}؟ جملة واحدة لكلٍّ منها.
  \item صُغ في فقرة قصيرة عبرة المسألة:
        ماذا يربح القاموس بين هندسة المستوي وجبر
        $\C$، وماذا يكلّف، وأيّ الخطوتين البرهانيتين --- المتطابقة
        $\alpha j + \beta = 0$ أم الرسم الكلاسيكي ---
        \emph{يشرح} مبرهنة نابليون شرحًا أفضل في رأيك؟ واذكر
        موضعًا واحدًا سيعود فيه القاموس في صورة مصفوفية لاحقًا في
        هذا المجلد.
\end{enumerate}
\end{problem}
