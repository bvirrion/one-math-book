\chapter{صيغ تايلور والنشور المقاربة}\label{ch:b1:taylor}

بجوار نقطة، تكون الدالة الملساء بجودة \omterm{def:b1:poly:def}{كثير حدود} --- مع خطأ
قابل للضبط. وتجعل صيغ تايلور هذا مضبوطًا في ثلاث
نكهات (الباقي التكاملي، وباقي لاغرانج، وباقي يونغ)،
وتصير \emph{النشور المقاربة} الناتجة، معالَجةً
جبريًا، أحدَّ أدوات التحليل الأوّلي:
النهايات والمكافئات والسلوك الموضعي والمقاربات.

\section{ترميز المقارنة}

\begin{definition}[ترميز لانداو]\label{def:b1:taylor:landau}
لتكن $f, g$ معرَّفتين بجوار $x_0$ ($x_0 \in \R$ أو $\pm\infty$). يُكتب،
عندما $x \to x_0$:\index{ترميز لانداو}
\begin{itemize}
  \item $f = o(g)$ («o صغير») إذا كان $f = \varepsilon g$ مع
        $\varepsilon(x) \to 0$؛
  \item $f = O(g)$ («O كبير») إذا كان $f = u g$ مع $u$ محدودة بجوار
        $x_0$؛
  \item $f \sim g$ («مكافئة»\index{دوال متكافئة}) إذا كان
        $f = (1 + \varepsilon) g$ مع $\varepsilon \to 0$ ---
        وبكيفية مكافئة $f - g = o(g)$.
\end{itemize}
وينطبق الترميز نفسه على المتتاليات ($n \to \infty$).
\end{definition}

\begin{proposition}[القواعد]\label{prop:b1:taylor:landaurules}
عندما $x \to x_0$:
\begin{enumerate}
  \item العلاقة $\sim$ \omterm{def:b1:logic:equiv}{علاقة تكافؤ}؛ ويستلزم $f \sim g$ أن تتقاسم
        $f$ و $g$ النهايات والإشارات (بجوار $x_0$) وانعدام
        الجذور؛
  \item المكافئات \emph{تُضرب وتُقسم}: يستلزم $f_1 \sim g_1$ و $f_2
        \sim g_2$ أن $f_1 f_2 \sim g_1 g_2$ و $\frac{f_1}{f_2}
        \sim \frac{g_1}{g_2}$؛
  \item المكافئات \emph{لا} تُجمع: فالمقدار $x + 1 \sim x$ و $-x \sim
        -x + 2$ عند $+\infty$، ومع ذلك فالمجموعان $1$ و $2$ غير
        متكافئين. وللجمع، عُد إلى النشور بحدود
        $o(\cdot)$ صريحة؛
  \item $o(g) + o(g) = o(g)$ و $\;u \cdot o(g) = o(ug)$ و $\;o(o(g)) =
        o(g)$ و $f \sim g \iff f = g + o(g)$.
\end{enumerate}
\end{proposition}

\begin{proof}
كلٌّ منها معالجة قصيرة للتعاريف؛ فمثلًا
$f_1 f_2 = (1+\varepsilon_1)(1+\varepsilon_2) g_1 g_2$ و $(1 +
\varepsilon_1)(1+\varepsilon_2) \to 1$. ويستحق بندان من (4)
سطرهما. $u\cdot o(g) = o(ug)$: فإذا كان $f = \varepsilon g$ مع
$\varepsilon \to 0$، فإن $uf = \varepsilon\,(ug)$ بالمقدار
$\varepsilon$ نفسه. و $o(o(g)) = o(g)$: فإذا كان $f = \varepsilon_1 h$ و
$h = \varepsilon_2 g$ مع $\varepsilon_i \to 0$ كليهما، فإن $f =
(\varepsilon_1\varepsilon_2) g$ وجداء اللامتناهيَين في
الصغر واحدٌ منهما. والتكافؤ $f \sim g \iff f = g +
o(g)$ هو التعريف مقروءًا مرتين: $f - g = \varepsilon g$.
والمثال المضاد في (3) هو برهان (3).
\end{proof}

\begin{example}[سلّم المقارنة]\label{ex:b1:taylor:scale}
عندما $x \to +\infty$، يُقرأ السلّم المعياري، بترتيب
قوة متزايدة:
\[
1 \;=\; o(\ln x), \quad
\ln x = o(x^{0.01}), \quad
x^{0.01} = o(\sqrt x), \quad
\sqrt x = o(x^{10}), \quad
x^{10} = o(\eu^{x}), \quad
\eu^x = o(\eu^{2x}) ,
\]
وكل خطوة مثالٌ على مقارنات النمو في
\cref{prop:b1:functions:powerrules} (فالقوى تغلب اللوغاريتمات،
والأسّيات تغلب القوى، وداخل العائلة الواحدة يفصل
الأُسّ). وعادتان جديرتان بالتكوّن: أولًا، أن $O(\cdot)$ يحطّ
في صنف أصغر يرتقي صامتًا (فالمقدار $O(\ln x)$ هو كذلك
$o(x^{0.01})$)؛ وثانيًا، أنه عند $x \to 0^+$ ينقلب السلّم كله
بالتعويض $x \mapsto \frac1x$ ---
فهناك $\ln x = o(x^{-0.01})$، ومنه فالقول «$x^\alpha \ln x \to 0$»
يصحّ من أجل كل $\alpha > 0$. وضبط السلّم
نصف كل حجة مقاربة في
\cref{ch:b1:series}.
\end{example}

\begin{example}[وحدانية النشور، وربح من الزوجية]\label{ex:b1:taylor:uniqueness}
إذا قبلت دالةٌ نشرين عند $0$ من الرتبة نفسها،
\[
a_0 + a_1 x + \dots + a_n x^n + o(x^n)
= b_0 + b_1 x + \dots + b_n x^n + o(x^n),
\]
فإن $a_k = b_k$ من أجل كل $k$: فبالطرح ووضع $c_k =
a_k - b_k$، قوّم المتطابقة $c_0 + c_1 x + \dots + c_n x^n
= o(x^n)$ عند $x \to 0$ لتجد $c_0 = 0$؛ ثم اقسم على $x$ و
كرّر --- وكل قسمة مشروعة لأن المقدار الباقي
هو مرة أخرى $o(x^{n-k})$. ومنه فالمعاملات ذاتية،
ويمكن حسابها عبر \emph{أيّ} طريق (مشتقات تايلور،
أو جبر على نشور معلومة، أو مكاملة): وعلى كل
الطرق أن تتفق. والربح: للدالة \emph{الزوجية} قوى
زوجية فقط في نشرها --- استبدل بالمقدار $x$ المقدارَ $-x$ واستدعِ
الوحدانية؛ وكذلك للدوال الفردية قوى فردية. ولهذا
يحمل $\cos$ المقدارَ $o(x^{2p+1})$ بدل $o(x^{2p})$ في
الجدول أدناه: فالحدّ الفردي الغائب معلومة مجانية، أي رتبة
دقة واحدة بلا ثمن.
\end{example}

\section{صيغ تايلور الثلاث}

\begin{theorem}[تايلور بالباقي التكاملي]\label{thm:b1:taylor:integral}
لتكن $f$ من الصنف $C^{n+1}$ على \omterm{prop:b1:reals:intervals}{فترة} تحتوي $a$ و $x$.
عندئذ\index{صيغة تايلور}
\[
f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}\,(x - a)^k
+ \int_a^x \frac{(x - t)^n}{n!}\, f^{(n+1)}(t)\, \dd t .
\]
\end{theorem}

\begin{proof}
بالاستقراء على $n$. من أجل $n = 0$: المقدار $f(x) = f(a) + \int_a^x f'(t)\dd t$ هو
المبرهنة الأساسية (\cref{thm:b1:integration:ftc}). والخطوة:
كامل الباقي بالتجزئة،
\[
\int_a^x \frac{(x-t)^n}{n!} f^{(n+1)}(t)\,\dd t
= \Bigl[-\frac{(x-t)^{n+1}}{(n+1)!} f^{(n+1)}(t)\Bigr]_a^x
+ \int_a^x \frac{(x-t)^{n+1}}{(n+1)!} f^{(n+2)}(t)\,\dd t ,
\]
ويسهم ما بين القوسين بالحدّ $\frac{f^{(n+1)}(a)}{(n+1)!}(x -
a)^{n+1}$.
\end{proof}

\begin{example}[نشر مضبوط مع باقيه]\label{ex:b1:taylor:lnexact}
من أجل $\ln(1 + x)$ يمكن جعل الباقي التكاملي صريحًا
تمامًا دون اشتقاق أيّ شيء $n$ مرة: كامل
المتطابقة الهندسية المنتهية $\frac{1}{1+t} =
\sum_{k=0}^{n-1}(-t)^k + \frac{(-t)^n}{1+t}$ من $0$ إلى $x$:
\[
\ln(1 + x) = \sum_{k=1}^{n} \frac{(-1)^{k-1}x^k}{k}
+ (-1)^n \int_0^x \frac{t^n}{1 + t}\,\dd t ,
\]
ومن أجل $0 \leq x \leq 1$ يكون الباقي محدودًا بالمقدار $\int_0^x
t^n\,\dd t = \frac{x^{n+1}}{n+1}$. وهذا أقوى من
تايلور--يونغ بكيفيتين: فهو \emph{متطابقة} صحيحة من أجل
$x$ مثبَّت (لا من أجل $x \to 0$ فقط)، وحاصر الخطأ عددي.
وتعيش مسألة نهاية الأسبوع (\cref{pb:b1:taylor:1}) على مثل هذه
الصور المضبوطة؛ وأمّا تايلور--يونغ أدناه فهو الأداة الأخفّ من أجل النهايات، حيث
لا يهمّ إلا شكل الخطأ.
\end{example}

\begin{theorem}[متراجحة تايلور--لاغرانج]\label{thm:b1:taylor:lagrange}
لتكن $f$ من الصنف $C^{n+1}$ مع $\abs{f^{(n+1)}} \leq M$ بين $a$ و
$x$. عندئذ
\[
\Bigl| f(x) - \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k \Bigr|
\leq M\, \frac{\abs{x - a}^{n+1}}{(n+1)!} .
\]
\end{theorem}

\begin{proof}
حُدّ الباقي التكاملي: $\bigl|\int_a^x \frac{(x-t)^n}{n!}
f^{(n+1)}(t)\,\dd t\bigr| \leq M \bigl|\int_a^x
\frac{\abs{x-t}^n}{n!}\dd t\bigr| = M\frac{\abs{x-a}^{n+1}}{(n+1)!}$.
\end{proof}

\begin{example}[أعداد مصادَق عليها]\label{ex:b1:taylor:certified}
كم يبلغ $\sqrt{1.02}$؟ طبّق تايلور--لاغرانج على $f(t) =
\sqrt{1 + t}$ عند $a = 0$، من الرتبة $2$، مع $x = 0.02$:
\[
\sqrt{1.02} \approx 1 + \frac{0.02}{2} - \frac{0.02^2}{8}
= 1.00995 ,
\qquad
\abs{f'''(t)} = \frac{3}{8}(1+t)^{-5/2} \leq \frac 38 ,
\]
ومنه فالخطأ على الأكثر $\frac38 \cdot \frac{0.02^3}{6} =
5\cdot10^{-7}$: أي $\sqrt{1.02} = 1.00995$ بستة أرقام عشرية
مصادَق عليها (والقيمة الحقيقية $1.0099504938\dots$ --- فالحاصر يكاد
يكون حادًّا). والفكرة النافذة: لا يقول تايلور--يونغ إلا \emph{كم
يسرع} تلاشي الخطأ؛ ويحوّل تايلور--لاغرانج كثيرَ الحدود نفسه
إلى \emph{شهادة}، أي عدد مع عمود خطأ مبرهن عليه. وكلما
ادُّعي رقم عشري في هذا الكتاب، وقف وراءه
حاصرٌ من نوع لاغرانج؛ وتصنّع مسألة نهاية الأسبوع
(\cref{pb:b1:taylor:1}) هذه الفكرة.
\end{example}

\begin{theorem}[تايلور--يونغ]\label{thm:b1:taylor:young}
لتكن $f$ \omterm{def:b1:derivative:def}{قابلة للاشتقاق} $n$ مرة عند $a$. عندئذ، عندما $x \to a$:
\[
f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}\,(x-a)^k
+ o\bigl((x-a)^n\bigr) .
\]
\end{theorem}

\begin{proof}
بالاستقراء على $n$. من أجل $n = 1$ هذا هو تعريف
\omterm{def:b1:derivative:def}{المشتقة} (\cref{def:b1:derivative:def}). افترض العبارة عند
الرتبة $n - 1$، ولتكن $f$ \omterm{def:b1:derivative:def}{قابلة للاشتقاق} $n$ مرة عند $a$. طبّق
فرض الاستقراء على $f'$ (وهي \omterm{def:b1:derivative:def}{قابلة للاشتقاق} $n-1$ مرة
عند $a$):
\[
f'(t) = \sum_{k=0}^{n-1} \frac{f^{(k+1)}(a)}{k!}(t-a)^k + r(t),
\qquad r(t) = o\bigl((t-a)^{n-1}\bigr).
\]
لتكن $g(x) = f(x) - \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k$؛ عندئذ
$g' = r$ و $g(a) = 0$. ومن أجل $\varepsilon > 0$ معطى، اختر $\delta$
يحقق $\abs{r(t)} \leq \varepsilon\abs{t - a}^{n-1}$ من أجل $\abs{t-a}
\leq \delta$؛ وتعطي متراجحة التزايدات المنتهية
(\cref{thm:b1:derivative:mvt}) مطبَّقةً على القطعة من $a$ إلى
$x$ (حيث $\abs{g'} \leq \varepsilon\abs{x-a}^{n-1}$) أن
$\abs{g(x)} \leq \varepsilon\abs{x - a}^n$: أي $g(x) =
o((x-a)^n)$ بالضبط.
\end{proof}

\begin{remark}[ثلاث صيغ وثلاثة أثمان وثلاثة منتجات]
تتدرّج الفرضيات مع النتائج بالضبط تدرّجًا.
فتايلور--يونغ يطلب الأقل ($n$ \omterm{def:b1:derivative:def}{مشتقة} \emph{عند
النقطة} فقط) ويعطي الأقل: أي $o((x-a)^n)$ كيفيًا،
مثاليًا من أجل النهايات، عديم النفع من أجل الأرقام المصادَق عليها. وتطلب
متراجحة لاغرانج الصنف $C^{n+1}$ \emph{على \omterm{prop:b1:reals:intervals}{الفترة}} وحاصرًا
$M$ هناك، وتعيد عمود خطأ عدديًا. وتطلب الصيغة
التكاملية الانتظام نفسه وتعيد الأكثر: أي الخطأ
غرضًا صريحًا يمكن تحويله (\omterm{thm:b1:integration:parts}{المكاملة بالتجزئة}، والحصر
قطعةً قطعة، وتغيير المتغيرات) --- وهي الصيغة التي حرّكت
آلة الصمم في \cref{pb:b1:integration:1}. واختيار
أضعف صيغة تدعم الهدف ليس تنطّعًا: فالدالة
المنبسطة في \cref{pb:b1:taylor:1} تحقق تايلور--يونغ
عند كل رتبة بينما كل نتيجة أقوى عنها
خاطئة بعيدًا عن $0$.
\end{remark}

\begin{proposition}[النشور المعيارية عند $0$]
عندما $x \to 0$، من أجل كل رتبة مثبَّتة $n$:
\index{نشر تايلور!معياري}
\begin{align*}
\eu^x &= 1 + x + \frac{x^2}{2!} + \dots + \frac{x^n}{n!} + o(x^n),\\
\cos x &= 1 - \frac{x^2}{2!} + \frac{x^4}{4!} - \dots
+ \frac{(-1)^p x^{2p}}{(2p)!} + o(x^{2p+1}),\\
\sin x &= x - \frac{x^3}{3!} + \dots + \frac{(-1)^p
x^{2p+1}}{(2p+1)!} + o(x^{2p+2}),\\
\frac{1}{1 - x} &= 1 + x + x^2 + \dots + x^n + o(x^n),\\
\ln(1 + x) &= x - \frac{x^2}{2} + \frac{x^3}{3} - \dots +
\frac{(-1)^{n-1} x^n}{n} + o(x^n),\\
(1 + x)^\alpha &= 1 + \alpha x + \frac{\alpha(\alpha-1)}{2!}x^2 +
\dots + \binom{\alpha}{n} x^n + o(x^n),
\end{align*}
حيث $\binom{\alpha}{n} = \frac{\alpha(\alpha - 1)\cdots(\alpha - n
+ 1)}{n!}$ من أجل $\alpha$ حقيقي. (و $\cosh$ و $\sinh$: كما $\cos$ و
$\sin$ دون الإشارات المتناوبة.)
\label{prop:b1:taylor:standard}
\end{proposition}

\begin{proof}
كل دالة ملساء بجوار $0$ ومشتقاتها سهلة التقويم:
$(\eu^x)^{(k)} = \eu^x$؛ ومشتقات $\sin$ و $\cos$ تدور
بالدور $4$؛ و $\bigl((1+x)^\alpha\bigr)^{(k)} = \alpha(\alpha -
1)\cdots(\alpha - k + 1)(1+x)^{\alpha - k}$؛
و $\bigl(\ln(1+x)\bigr)^{(k)} = \frac{(-1)^{k-1}(k-1)!}{(1+x)^k}$.
فطبّق تايلور--يونغ عند $a = 0$. (والهندسية مضبوطة:
$\frac{1}{1-x} - \sum_0^n x^k = \frac{x^{n+1}}{1 - x} = o(x^n)$.)
\end{proof}

\begin{method}[الحساب بالنشور]\label{met:b1:taylor:compute}
\begin{enumerate}
  \item \emph{ثبّت الرتبة الهدف} $n$ أولًا، واقطع كل نتيجة
        وسيطة عندها --- فحمل حدود أعلى عملٌ ضائع،
        وإسقاط حدود أدنى خطأ.
  \item \emph{المجاميع والجداءات}: انشر كل عامل حتى الرتبة $n$ و
        اضرب، مسقطًا ما بعد $x^n$.
  \item \emph{التركيب} $f(u(x))$ مع $u(x) \to 0$: عوّض
        نشر $u$ في نشر $f$، رتبةً رتبة.
  \item \emph{القسمة}: اكتب $\frac{1}{1 + v}$ مع $v \to 0$ و
        استعمل النشر الهندسي.
  \item \emph{كامل} نشرًا حدًّا حدًّا (وأمّا الاشتقاق
        فيقتضي عناية أكبر --- والتسويغ: تكامل $o(t^n)$
        من $0$ إلى $x$ هو $o(x^{n+1})$، بالحصر المباشر).
\end{enumerate}
\end{method}

\begin{example}[التركيب، مع إظهار مسك الدفاتر]\label{ex:b1:taylor:esin}
انشر $\eu^{\sin x}$ حتى الرتبة $3$. والنشر الداخلي: $u = \sin x
= x - \frac{x^3}{6} + o(x^3)$، وهو فعلًا يؤول إلى $0$. والخارجي:
$\eu^u = 1 + u + \frac{u^2}{2} + \frac{u^3}{6} + o(u^3)$، و
$o(u^3) = o(x^3)$ لأن $u \sim x$. وقوى $u$، مقطوعةً عند
$x^3$:
\[
u^2 = x^2 + o(x^3), \qquad u^3 = x^3 + o(x^3)
\]
(فالحدّ المتقاطع $2x\cdot(-\frac{x^3}{6})$ هو $x^4$ أصلًا).
ورَكِّب:
\[
\eu^{\sin x} = 1 + \Bigl(x - \frac{x^3}{6}\Bigr) + \frac{x^2}{2}
+ \frac{x^3}{6} + o(x^3)
= 1 + x + \frac{x^2}{2} + o(x^3) :
\]
فتتلاشى إسهاما $x^3$ تلاشيًا تامًا. والفكرة النافذة:
تتطابق $\eu^{\sin x}$ و $\eu^x$ حتى الرتبة $3$ --- لا
لأن $\sin x \approx x$ فجًّا، بل لأن أول
اختلاف بين الأُسّين ($-\frac{x^3}{6}$) يدخل
مضروبًا في $\eu^0 = 1$ ثم يقابله الحدّ التكعيبي في
الأسّي الخارجي؛ ويكشف مسك الدفاتر رتبةً رتبة مثل هذه
المؤامرات، ولا تكشفها النظرة أبدًا. (والحدّ التالي هو
$-\frac{x^4}{8}$: فتنتهي الهدنة عند الرتبة $4$.)
\end{example}

\begin{example}\label{ex:b1:taylor:tan}
نشر $\tan$ من الرتبة $5$. اكتب $\tan x =
\sin x \cdot \frac{1}{\cos x}$:
\[
\frac{1}{\cos x} = \frac{1}{1 - \bigl(\frac{x^2}{2} - \frac{x^4}{24}
+ o(x^5)\bigr)}
= 1 + \Bigl(\frac{x^2}{2} - \frac{x^4}{24}\Bigr)
+ \Bigl(\frac{x^2}{2}\Bigr)^{\!2} + o(x^5)
= 1 + \frac{x^2}{2} + \frac{5x^4}{24} + o(x^5),
\]
ثم
\[
\tan x = \Bigl(x - \frac{x^3}{6} + \frac{x^5}{120}\Bigr)
\Bigl(1 + \frac{x^2}{2} + \frac{5x^4}{24}\Bigr) + o(x^5)
= x + \frac{x^3}{3} + \frac{2x^5}{15} + o(x^5) .
\]
\end{example}

\begin{example}[النهايات]\label{ex:b1:taylor:limits}
\[
\lim_{x \to 0} \frac{x - \sin x}{x^3}:
\qquad
x - \sin x = \frac{x^3}{6} + o(x^3) \sim \frac{x^3}{6},
\qquad\text{ومنه فالنهاية هي } \frac16
\]
--- فيُحسم السؤال المطروح في \cref{exo:b1:functions:9}.
وكذلك $\displaystyle\lim_{x\to0}\Bigl(\frac{\sin
x}{x}\Bigr)^{1/x^2}$: فاللوغاريتم هو
\[
\frac{1}{x^2}\ln\Bigl(1 - \frac{x^2}{6} + o(x^2)\Bigr)
= \frac{1}{x^2}\Bigl(-\frac{x^2}{6} + o(x^2)\Bigr)
\longrightarrow -\frac16,
\qquad\text{والنهاية } \eu^{-1/6}.
\]
\end{example}

\begin{remark}[مزالق شائعة مع النشور]
(أ) \emph{لا تجمع المكافئات ولا تطرحها أبدًا}: فمن $\tan x \sim
x$ و $\sin x \sim x$ \emph{لا} يجوز استنتاج $\tan x -
\sin x \sim 0$ (وهو بلا معنى) --- والطريق الأمين هو
النشور:
\[
\tan x - \sin x
= \Bigl(x + \frac{x^3}{3}\Bigr) - \Bigl(x - \frac{x^3}{6}\Bigr)
+ o(x^3) = \frac{x^3}{2} + o(x^3) \sim \frac{x^3}{2} .
\]
(ب) \emph{انشر بعد المجزرة}: ففي الحساب نفسه،
لا ترى الرتبة $1$ إلا $x - x = 0$؛ وكلما تلاشت الحدود المهيمنة،
فارفع الرتبة حتى ينجو معامل غير معدوم، وعندئذ
فقط حوّل رجوعًا إلى مكافئ. (ج) \emph{المكافئات لا
تمرّ عبر الأسّيات}: فالمقدار $n^2 + n \sim n^2$، ومع ذلك
$\eu^{n^2+n} = \eu^{n}\,\eu^{n^2}$ \emph{ليس} مكافئًا للمقدار
$\eu^{n^2}$ --- فلا تأخذ الأسّي إلا لِما في \emph{نشور الأُسّ}
التي يؤول خطؤها إلى $0$، لا لمكافئات
الأُسّ أبدًا. (وأمّا اللوغاريتمات فأأمن: فإذا كان $u_n \sim v_n \to \ell \neq
1$ مع $\ell > 0$، فإن $\ln u_n \sim \ln v_n$.) (د) \emph{حساب
$o(\cdot)$ وحيد الاتجاه}: $o(x^2) + o(x^2) =
o(x^2)$ و $5\,o(x^2) = o(x^2)$ و $x\cdot o(x^2) = o(x^3)$ --- لكن
$o(x^2)$ ليس دالة معيّنة، فلا تختصر أبدًا اثنين
منها أحدهما بالآخر: فالمقدار $o(x^2) - o(x^2)$ هو $o(x^2)$، لا
$0$.
\end{remark}

\begin{proposition}[السلوك الموضعي]\label{prop:b1:taylor:local}
افترض $f(x) = f(a) + c\,(x - a)^p + o\bigl((x-a)^p\bigr)$ مع $c
\neq 0$ (أي أول حدّ غير معدوم بعد الثابت؛ و $p \geq 2$ عند
نقطة حرجة).
\begin{itemize}
  \item إذا كان $p$ زوجيًا: فللمقدار $f$ قيمة صغرى موضعية عند $a$ إذا كان $c > 0$، و
        قيمة عظمى موضعية إذا كان $c < 0$.
  \item وإذا كان $p$ فرديًا: فلا قيمة حدّية (لأن $f - f(a)$ تغيّر إشارتها)؛
        وإذا بدأ النشر فوق ذلك بعد حدّ خطي $f'(a)(x
        - a)$، عبر المنحنى مماسَه: أي \emph{انعطاف}.
\end{itemize}
\end{proposition}

\begin{proof}
بجوار $a$، تكون للمقدار $f(x) - f(a) = (x-a)^p\bigl(c + o(1)\bigr)$ إشارةُ
$c\,(x-a)^p$: فهي ثابتة من أجل $p$ زوجي، ومتغيرة من أجل $p$ فردي.
\end{proof}

\begin{example}[يجب نشر الأُسس حتى $o(1)$]\label{ex:b1:taylor:expdiscipline}
جد مكافئًا للمقدار $u_n = \bigl(1 + \frac1n\bigr)^{n^2}$.
انشر \emph{الأُسّ} حتى يؤول خطؤه إلى $0$:
\[
n^2 \ln\Bigl(1 + \frac1n\Bigr)
= n^2\Bigl(\frac1n - \frac{1}{2n^2} +
O\Bigl(\frac{1}{n^3}\Bigr)\Bigr)
= n - \frac12 + O\Bigl(\frac1n\Bigr),
\]
ومنه $u_n = \eu^{\,n - 1/2}\,\eu^{O(1/n)}$ مع $\eu^{O(1/n)} \to
1$:
\[
u_n \;\sim\; \eu^{\,n - \frac12} .
\]
ولاحظ ما كان سيسوء بعناية أقلّ: فوقف
الأُسّ عند $n^2\cdot\frac1n = n + O(1)$ يترك عاملًا
$\eu^{O(1)}$ --- محدودًا لكنه لا يؤول إلى $1$ --- و
لا يمكن تأكيد أيّ مكافئ. وقاعدة
المزالق أعلاه، في صورتها الموجبة: يقتضي مكافئُ
$\eu^{a_n}$ نشرَ $a_n$ \emph{حتى حدّ
يؤول إلى الصفر}، مع الاحتفاظ بكل معامل قبله
احتفاظًا مضبوطًا.
\end{example}

\begin{example}[تصنيف نقطة حرجة منبسطة]\label{ex:b1:taylor:flatmin}
ادرس $f(x) = \cos x + \frac{x^2}{2}$ بجوار $0$. فكلٌّ من $f'(0) =
0$ و $f''(0) = -\cos 0 + 1 = 0$: ومنه فاختبار \omterm{def:b1:derivative:def}{المشتقة} الثانية
أخرس. وانشر بدل ذلك:
\[
f(x) = \Bigl(1 - \frac{x^2}{2} + \frac{x^4}{24} +
o(x^4)\Bigr) + \frac{x^2}{2}
= 1 + \frac{x^4}{24} + o(x^4) :
\]
فأول حدّ غير معدوم $c\,x^p$ مع $p = 4$ زوجي و $c =
\frac{1}{24} > 0$: أي قيمة صغرى موضعية، بانبساط غير معتاد (إذ يغادر
المنحنى قيمته الصغرى مثل $x^4$، لا مثل $x^2$). والفكرة
النافذة: يرى النشر في سطر واحد ما يحجبه
الاشتقاق المكرَّر --- و \cref{prop:b1:taylor:local} هو
القاموس المنهجي من «أول حدّ ناجٍ» إلى
«الشكل الموضعي».
\end{example}

\begin{example}[النشور عند اللانهاية]\label{ex:b1:taylor:atinfinity}
حسابان يجري فيهما المتغير إلى $+\infty$ ويستورد
التعويض $h = \frac1x \to 0^+$ العدّةَ كلها.
أولًا،
\[
x - x^2\ln\Bigl(1 + \frac1x\Bigr)
= x - x^2\Bigl(\frac1x - \frac{1}{2x^2} +
O\Bigl(\frac{1}{x^3}\Bigr)\Bigr)
= \frac12 + O\Bigl(\frac1x\Bigr) \longrightarrow \frac12 .
\]
وثانيًا، ظلّ الزاوية العكسي عند اللانهاية: انطلاقًا من $\arctan x +
\arctan\frac1x = \frac\pi2$ من أجل $x > 0$
(\cref{prop:b1:functions:arcidentities}) ومن نشر
$\arctan$ عند $0$ (\cref{exo:b1:taylor:3})،
\[
\arctan x = \frac\pi2 - \arctan\frac1x
= \frac\pi2 - \frac1x + \frac{1}{3x^3} +
o\Bigl(\frac{1}{x^3}\Bigr) :
\]
فيقترب المنحنى من مقاربه $y = \frac\pi2$ من أسفل بسرعة
$\frac1x$. والفكرة النافذة: لا توجد نظرية منفصلة
للنشور عند اللانهاية --- بل يردّها تعويض مقلوب واحد
إلى نشور عند $0$، شرط أن يُحمل كل $O$ و $o$
وسيط حملًا أمينًا.
\end{example}

\begin{example}[مقارب بالنشر]\label{ex:b1:taylor:asymptote}
عندما $x \to +\infty$،
\[
\sqrt{x^2 + x} = x\sqrt{1 + \tfrac1x}
= x\Bigl(1 + \frac{1}{2x} - \frac{1}{8x^2} + o\bigl(\tfrac{1}{x^2}\bigr)\Bigr)
= x + \frac12 - \frac{1}{8x} + o\bigl(\tfrac 1x\bigr):
\]
فالمستقيم $y = x + \frac12$ مقارب، ويُقترب منه \emph{من
أسفل} (لأن الحدّ التالي $-\frac{1}{8x}$ سالب).
\end{example}

\begin{remark}[أين تعمل النشور تاليًا]
النشور المقاربة هي اللغة الدائمة لبقية
الكتاب: ففي \cref{ch:b1:series} تفصل في التقارب
(إذ تغذّي المكافئات محكات المقارنة، ودراسة $\sum
\frac{1}{n^\alpha}$ نشرٌ متنكّر)؛ وفي مجلّد السنة الثانية
تصير \emph{متسلسلات صحيحة}، حيث يكتسب \omterm{def:b1:poly:def}{كثير حدود}
تايلور حدودًا لا تُحصى ونصفَ قطر تقارب؛ وكل تخطيط
في الفيزياء --- البندول، والاضطراب من الرتبة الأولى --- عبارةُ
تايلور--يونغ مع إسقاط $o(\cdot)$ صامتًا. والتحذير الوحيد الجدير بالحفر:
النشر لا يصف الدالة إلا \emph{بجوار نقطة واحدة} ---
انظر الدالة المنبسطة في مسألة نهاية الأسبوع، التي نشرها عند
$0$ معدوم تمامًا دون أن تكون الدالة كذلك.
\end{remark}

\begin{remark}[منظورات داخل هذا المجلّد]
النشور هي لغة العمل في بقية التحليل
وفي الهندسة الآتية. فالفصل \Cref{ch:b1:series} يحوّلها
إلى أحكام تقارب: فمكافئ الحدّ العام
نشرٌ مقطوع عند حدّه الأول، وتستهلك المحكات الأدقّ
(المتناوبة مع ضبط الخطأ) الحدَّ الثاني كذلك.
و \Cref{ch:b1:curves} يقرأ الهندسة الموضعية من نشور
دالتَي الإحداثيات: فأن يعبر منحن معلَّمي أو يقبّل أو
يرتدّ عند نقطة أمرٌ تفصل فيه أيُّ قوى $t$
تنجو في $x(t)$ و $y(t)$ --- وهي صيغة المستوي للمقدار
\cref{prop:b1:taylor:local}. و \cref{ch:b1:multivar} يقف عند
الرتبة واحد قصدًا: فالمستوي المماس عبارةُ
تايلور--يونغ في متغيرين، مع تأجيل النظرية الكاملة من الرتبة الثانية
(الهسّيات ونقاط السرج) إلى مجلّد السنة الثانية. و
الخيط المشترك: كل سؤال «موضعي» في هذا الكتاب يُجاب عنه
بكتابة أول حدّ ناجٍ من نشر.
\end{remark}

\begin{exercise}[$\star\star\star$]\label{exo:b1:taylor:9}
لتكن $f$ من الصنف $C^2$ على $\R$. برهن على أنه من أجل كل $x$ و $h > 0$:
\[
\abs{f'(x)} \leq \frac{\abs{f(x+h) - f(x-h)}}{2h} +
\frac{h}{2}\sup_{\intcc{x-h}{x+h}}\abs{f''} ,
\]
واستنتج المتراجحة من نوع لانداو--كولموغوروف: إذا كان $\abs f \leq
M_0$ و $\abs{f''} \leq M_2$ على $\R$، فإن $\abs{f'} \leq
\sqrt{2 M_0 M_2}$ في كل مكان. \emph{(أمثِل على $h$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:taylor:10}

  \item ومع تشيزارو (\cref{exo:b1:seq:10})، استنتج $\frac{v_n}{n}
        \to \frac13$، ثم المكافئ $u_n \sim
        \sqrt{\dfrac{3}{n}}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:11}

\begin{exercise}[$\star\star\star$]\label{exo:b1:taylor:12}

\end{exercise}

\section{مسألة: المجاميع المتناوبة والأرقام المصادَق عليها و
صمم $\cos 1$}

\begin{problem}[{مسألة نهاية الأسبوع --- التقدير المتناوب
$\abs{S - S_n} \leq a_{n+1}$: $\ln 2$ و $\pi$ بأرقام عشرية
مبرهن عليها، وصيغة ماشان، و $\cos 1 \notin \Q$}]
\label{pb:b1:taylor:1}
المجموع المتناوب ذو الحدود المتناقصة ألطفُ
غرض في التحليل العددي: فخطؤه محدود بأول
حدّ محذوف، \emph{بإشارة معلومة}. وتبرهن هذه المسألة على ذلك
المبدأ بمبرهنة المتتاليتين المتجاورتين، ثم تنفقه
ثلاث كيفيات: أرقام عشرية مصادَق عليها للمقدار $\ln 2$ (بثلاثة طرق متنافسة)
وللمقدار $\pi$ (لايبنتز، ثم صيغة ماشان لسنة 1706، وهي لا تزال
الفكرة وراء حسابات قياسية طوال قرون)،
وصمم $\cos 1$ و $\sin 1$ و $\cosh 1$، و، بوصفه
ثقلًا مقابلًا، \emph{مساواة} تايلور--لاغرانج والدالة
المنبسطة التي يكذب نشر تايلور لها.\index{تقدير المتسلسلة
المتناوبة}\index{صيغة ماشان}\index{صمم!صمم $\cos 1$} وفي كل ما يأتي، لغة
«المتسلسلة» غير صورية: فكل مجموع
هنا \emph{متتالية مجاميع جزئية}، كما في
\cref{ex:b1:seq:e}؛ والنظرية بمعناها الدقيق تُفتتح في
\cref{ch:b1:series}.

\medskip
\noindent\textbf{الجزء 1 --- التقدير المتناوب.} لتكن
$(a_k)_{k \geq 0}$ متناقصة إلى $0$ ولتكن $S_n = \sum_{k=0}^{n}
(-1)^k a_k$.
\begin{enumerate}
  \item بيّن أن $(S_{2n+1})$ غير متناقصة و $(S_{2n})$
        غير متزايدة، وأنهما متجاورتان
        (\cref{thm:b1:seq:adjacent}): فتتقارب كلتاهما إلى
        $S$ مشترك مع، من أجل كل $n$،
        \[
        S_{2n+1} \leq S \leq S_{2n},
        \qquad
        \abs{S - S_n} \leq a_{n+1} ,
        \]
        وللخطأ إشارة أول حدّ محذوف.
        وبيّن فوق ذلك أنه إذا كان التناقص \emph{قطعيًا}، فكل
        هذه المتراجحات قطعية.
  \item الربح الأول: من أجل $x = 1$ في المتسلسلة الأسّية،
        قارن بالمقدار \cref{thm:b1:taylor:lagrange} عند $a = 0$:
        بيّن أن $T_n = \sum_{k=0}^{n} \frac{(-1)^k}{k!}$
        تتقارب إلى $\eu^{-1}$ مع $\abs{\eu^{-1} - T_n} <
        \frac{1}{(n+1)!}$.
  \item (لايبنتز، 1674) انطلاقًا من المتطابقة المنتهية \emph{المضبوطة}
        \[
        \frac{1}{1 + t^2} = \sum_{k=0}^{n} (-1)^k t^{2k}
        + \frac{(-1)^{n+1} t^{2n+2}}{1 + t^2} ,
        \]
        مكامَلةً على $\intcc{0}{1}$، برهن على
        \[
        \frac\pi4 = \sum_{k=0}^{n} \frac{(-1)^k}{2k+1} + \rho_n,
        \qquad \abs{\rho_n} \leq \frac{1}{2n+3} .
        \]
  \item البطء: كم حدًّا من لايبنتز يضمن ستة أرقام عشرية
        مضبوطة للمقدار $\pi$؟ (نحو مليونين.) وقوّم $4 S_4
        = 4\bigl(1 - \frac13 + \frac15 - \frac17 +
        \frac19\bigr)$ ومسافته إلى $\pi$، لتشعر
        بالألم.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 2 --- $\ln 2$ بثلاث طرق.}
\begin{enumerate}[resume]
  \item (الطريق 1: التوافقية المتناوبة) من $\frac{1}{1+t} =
        \sum_{k=0}^{n-1}(-1)^k t^k + \frac{(-1)^n t^n}{1+t}$
        مكامَلًا على $\intcc{0}{1}$:
        \[
        \ln 2 = \sum_{k=1}^{n} \frac{(-1)^{k-1}}{k} + (-1)^n
        R_n, \qquad \frac{1}{2(n+1)} \leq R_n \leq
        \frac{1}{n+1} :
        \]
        والخطأ من الرتبة المضبوطة $\frac 1n$ --- أي مليون
        حدّ من أجل ستة أرقام عشرية.
  \item (الطريق 2: المتسلسلة السريعة) كامل $\frac{1}{1 - t^2} =
        \sum_{k=0}^{n} t^{2k} + \frac{t^{2n+2}}{1-t^2}$ من $0$
        إلى $x \in \intoo{0}{1}$ وقوّم عند $x = \frac13$
        (ولاحظ $\frac{1 + 1/3}{1 - 1/3} = 2$):
        \[
        \ln 2 = 2\sum_{k=0}^{n} \frac{(1/3)^{2k+1}}{2k+1} +
        \tilde\rho_n,
        \qquad
        0 < \tilde\rho_n \leq \frac{9}{4}\cdot
        \frac{(1/3)^{2n+3}}{2n+3} :
        \]
        فالتقارب هندسي، بنحو رقم واحد لكل حدّ.
  \item (الطريق 3: \omterm{thm:b1:integration:riemann}{مجاميع ريمان} ومتطابقة مستترة) برهن
        بالاستقراء على المتطابقة
        \[
        \sum_{k=1}^{2n} \frac{(-1)^{k-1}}{k} = H_{2n} - H_n =
        \sum_{k=1}^{n} \frac{1}{n + k} ,
        \]
        واستعد $\ln 2$ نهايةً \omterm{thm:b1:integration:riemann}{لمجاميع ريمان} للمقدار
        \cref{ex:b1:integration:riemannexample}: فالطريقان 1 و 3
        هما سرًّا العدد نفسه منظورًا إليه مرتين.
  \item مبارزة عند ستة حدود: قارن $\sum_{k=1}^{6}
        \frac{(-1)^{k-1}}{k} = 0.6167$ بالطريق 2 عند $n = 5$،
        الذي يعطي أصلًا $\ln 2 = 0.693147$ بخطأ $\leq
        1.1\cdot10^{-7}$. وفسّر السبب البنيوي
        (نقطة تقويم عميقة داخل \omterm{prop:b1:reals:intervals}{فترة}
        التقارب إزاء نقطة على حدّها).
  \item كم حدًّا من الطريق 2 يصادق على \emph{عشرة} أرقام عشرية للمقدار
        $\ln 2$؟ بيّن أن $n = 10$ يكفي.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 3 --- صيغة ماشان.}
\begin{enumerate}[resume]
  \item احسب $(5 + \iu)^4$ وتحقق من المتطابقة العقدية
        \[
        (5 + \iu)^4 = 2\,(1 + \iu)\,(239 + \iu) .
        \]
        وبأخذ العمد (باصطلاحات \cref{ch:b1:complex})،
        استنتج صيغة ماشان
        \[
        \frac\pi4 = 4\arctan\frac15 - \arctan\frac{1}{239} .
        \]
        (وتحقق من أن لا \omterm{def:b1:complex:expi}{عمدة} تغادر
        $\intoo{-\frac\pi2}{\frac\pi2}$.)
  \item كما في السؤال 3، أثبت من أجل $0 < x < 1$:
        \[
        \arctan x = \sum_{k=0}^{n} \frac{(-1)^k
        x^{2k+1}}{2k+1} + r_n(x), \qquad \abs{r_n(x)} \leq
        \frac{x^{2n+3}}{2n+3} .
        \]
  \item صادِق على $\pi$ حتى سبعة أرقام عشرية بستة حدود: حُدّ
        الخطأ الكلي للمقدار
        \[
        \pi \approx 16\sum_{k=0}^{4}
        \frac{(-1)^k (1/5)^{2k+1}}{2k+1}
        - 4\Bigl(\frac{1}{239} - \frac{1}{3\cdot239^3}\Bigr)
        \]
        بالمقدار $16\,\frac{(1/5)^{11}}{11} +
        4\,\frac{(1/239)^5}{5} < 5\cdot10^{-8}$، وأعطِ
        القيمة الناتجة $3.1415926\dots$
  \item قارن الطرق الثلاثة إلى $\pi$ المتوفرة الآن ---
        لايبنتز (السؤال 4)، وتكاملات دالزل في
        \cref{pb:b1:integration:1} (بخطأ $4^{1-5m}$)، وماشان
        (بخطأ $\approx 16\cdot 5^{-(2n+3)}$) --- بعدد الأرقام لكل
        حدّ، وفسّر لماذا يغلب تصغيرُ نقطة التقويم
        كلَّ شيء.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 4 --- فخّ العدد الصحيح، الطبعة
المتناوبة.}
\begin{enumerate}[resume]
  \item افترض $\cos 1 = \frac pq$. اضرب التأطير المتناوب
        القطعي للمقدار $\sum_{k}
        \frac{(-1)^k}{(2k)!}$ (السؤال 1) في $(2n)!$ مع $2n
        \geq q$، واستخرج تناقضًا: فيكون $\cos 1$
        أصمّ.
  \item كيّف ذلك على $\sin 1 = \sum_k \frac{(-1)^k}{(2k+1)!}$
        (بالضرب في $(2n+1)!$): فيكون $\sin 1 \notin \Q$. وكلاهما
        أصمّ، ومع ذلك $\cos^2 1 + \sin^2 1 = 1$: فالصمم
        ليس مستقرًا تحت الجبر.
  \item وابن العمّ غير المتناوب: $\cosh 1 = \sum_k
        \frac{1}{(2k)!}$ (بمعنى المجاميع الجزئية، مع
        حاصر الذيل ذي الطرفين $0 < \cosh 1 - \sum_{k \leq n}
        \frac{1}{(2k)!} < \frac{2}{(2n+2)!}$، ويُبرهن عليه).
        واستنتج $\cosh 1 \notin \Q$ بالفخّ نفسه.
  \item ادفع ذلك إلى $\cos\frac 1m$ من أجل كل عدد صحيح $m \geq 1$:
        اضرب في $m^{2n}(2n)!$ واستنتج $\cos\frac1m
        \notin \Q$. وأين تنكسر المحاولة نفسها من أجل
        $\cos\frac ab$ مع $b > 1$ عام؟ (حدّد
        المقام الذي لم يعد يُزال.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء 5 --- أحدّ وأقتم: صيغة
المساواة، ودالة تخدع تايلور.}
\begin{enumerate}[resume]
  \item (تايلور--لاغرانج، صيغة المساواة) لتكن $f$ \omterm{def:b1:derivative:def}{قابلة للاشتقاق}
        $n + 1$ مرة بين $a$ و $x$. عرّف $g(t) =
        f(x) - \sum_{k=0}^{n} \frac{f^{(k)}(t)}{k!}(x - t)^k -
        A\,\frac{(x-t)^{n+1}}{(n+1)!}$ مع اختيار الثابت $A$
        بحيث يكون $g(a) = 0$. احسب $g'$ (فالمجموع
        يتلسكب)، وطبّق رول على $\intcc{a}{x}$، واستنتج
        أنه يوجد $c$ قطعًا بين $a$ و $x$ يحقق
        \[
        f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k
        + \frac{f^{(n+1)}(c)}{(n+1)!}\,(x-a)^{n+1} .
        \]
  \item ربح المساواة: من أجل $x > 0$ بيّن
        \[
        \eu^x > 1 + x + \frac{x^2}{2!} + \dots +
        \frac{x^n}{n!}
        \]
        (قطعًا، من أجل كل $n$)، وحدّد أين تنقلب
        المتراجحة من أجل $x < 0$ تبعًا لزوجية
        $n$.
  \item قارن البواقي على $\sin(0.5)$ من الرتبة $3$:
        فيعطي يونغ $o(x^3)$ فقط (بلا عدد)؛ ويعطي لاغرانج
        $\abs{\sin 0.5 - (0.5 - \frac{0.5^3}{6})} \leq
        \frac{0.5^5}{120} = 2.61\cdot10^{-4}$؛ ويعطي التقدير
        المتناوب الحاصرَ نفسه \emph{مع} معلومة
        الإشارة $\sin 0.5 > 0.5 - \frac{0.5^3}{6}$. وقارن
        بالخطأ الحقيقي $2.59\cdot10^{-4}$: فالحاصر يكاد
        يُبلغ. وأيّ أداة ستمدّ يدك إليها، ومتى؟
  \item (الدالة المنبسطة) لتكن $f(x) = \eu^{-1/x^2}$ من أجل $x
        \neq 0$ و $f(0) = 0$. بيّن أن $f$ \omterm{def:b1:continuity:continuous}{متصلة} عند $0$، وأن
        $f'(0) = 0$، وبعبارة أعمّ --- بالبرهان على أن
        لكل \omterm{def:b1:derivative:def}{مشتقة} الشكلَ $f^{(k)}(x) =
        P_k\bigl(\frac1x\bigr)\eu^{-1/x^2}$ من أجل \omterm{def:b1:poly:def}{كثير حدود}
        $P_k$ (بالاستقراء) --- أن $f^{(k)}(0) = 0$ من أجل كل $k$
        \emph{(\omterm{prop:b1:functions:powerrules}{بمقارنة النمو}
        \cref{prop:b1:functions:powerrules})}. واستنتج: تنعدم كل
        كثيرات حدود تايلور للمقدار $f$ عند $0$، ومع ذلك $f(x) > 0$
        من أجل $x \neq 0$: فيصحّ تايلور--يونغ عند كل رتبة، و
        لا يقول شيئًا عن $f$ بعيدًا عن $0$. فالنشور
        تصف \emph{جراثيم}، لا دوالّ.
\end{enumerate}

\medskip
\noindent\textbf{الجزء 6 --- توليفة.}
\begin{enumerate}[resume]
  \item شغّل الفخّ مرة أخرى، على $\eu^{-1} = \sum_k
        \frac{(-1)^k}{k!}$: اضرب التأطير المتناوب القطعي
        في $n!$ واستنتج $\eu^{-1} \notin \Q$،
        ومنه $\eu \notin \Q$ --- وهو البرهان الثالث على هذه الواقعة
        في المجلّد. واذكر الثلاثة (\omterm{thm:b1:seq:adjacent}{المتتاليتان المتجاورتان}،
        \cref{exo:b1:seq:9}؛ والتكاملات،
        \cref{pb:b1:integration:1}؛ والمجاميع المتناوبة، هنا) و
        ما احتاج إليه كلٌّ منها.
  \item والتفاصيل الدقيقة: الرتابة ليست زينة. لتكن
        $b_k = \frac1k$ من أجل $k$ فردي و $b_k = \frac{1}{k^2}$
        من أجل $k$ زوجي: فالمقادير $b_k$ موجبة وتؤول إلى $0$،
        ومع ذلك تتباعد المجاميع الجزئية للمقدار $\sum (-1)^k b_k$ إلى
        $-\infty$. برهن على ذلك \emph{(اقسم المجموع الجزئي إلى
        الجزء الزوجي، المحدود عبر
        \cref{ex:b1:seq:basel}، والجزء الفردي، الذي
        يهيمن على نصف المتسلسلة التوافقية،
        \cref{exo:b1:seq:5})}، وقل بالضبط أيّ خطوة من
        السؤال 1 استعملت الرتابة.
  \item تحقق من متطابقة أويلر الأبسط $\arctan\frac12 +
        \arctan\frac13 = \frac\pi4$ عبر $(2 + \iu)(3 + \iu) =
        5(1 + \iu)$، وقدّر الحدود اللازمة من أجل ستة أرقام عشرية
        للمقدار $\pi$ بهذا الطريق ($n = 10$ يكفي)، وضعه
        بين لايبنتز وماشان في ترتيب السؤال
        13.
  \item توليفة، جملة واحدة لكلٍّ: (أ) اذكر التقدير
        المتناوب ومخرجيه (الحاصر والإشارة)؛ (ب) ولماذا
        تغلب المتطابقات المنتهية المضبوطة ذات البواقي الصريحة
        عباراتِ النهايات من أجل الأعداد المصادَق عليها؛ (ج) وجرد
        المسألة ($\pi$ حتى $10^{-7}$ باليد، و $\ln 2$ حتى
        عشرة أرقام عشرية، وأربعة براهين صمم، ومبرهنة مساواة
        واحدة، ومثال تحذيري واحد)؛ (د) وأيّ هذه
        الخيوط سيلتقطه \cref{ch:b1:series} (محك المتسلسلات
        المتناوبة، والتقارب بإطلاق إزاء التقارب الشرطي،
        ودراما إعادة الترتيب في مسألة نهاية الأسبوع فيه).
\end{enumerate}
\end{problem}
