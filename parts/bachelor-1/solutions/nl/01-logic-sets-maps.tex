\section*{Hoofdstuk \ref{ch:b1:logic} --- Logica, verzamelingen en afbeeldingen}

\begin{solution}{exo:b1:logic:1}
Negaties, waarbij $\lnot$ door elke kwantor geduwd wordt
(\cref{prop:b1:logic:negquant}) en de regel $\lnot(P \implies Q) \iff
P \land \lnot Q$ gebruikt wordt:
\begin{enumerate}
  \item $\exists x \in \R,\ \forall y \in \R,\ x + y \leq 0$;
  \item $\forall x \in \R,\ \exists y \in \R,\ xy \neq 0$;
  \item $\exists \varepsilon > 0,\ \forall \delta > 0,\ \exists x \in
        \R,\ \abs{x} \leq \delta \text{ en } \abs{f(x)} >
        \varepsilon$.
\end{enumerate}
\omterm{def:b1:logic:statement}{Uitspraak} (1) is waar: neem bij gegeven $x$ het getal $y = -x + 1$; dan
is $x + y = 1 > 0$. \omterm{def:b1:logic:statement}{Uitspraak} (2) is waar: $x = 0$ voldoet aan $xy = 0$
voor alle $y$.
\end{solution}

\begin{solution}{exo:b1:logic:2}
Waarheidstabel, met W/O voor de vier gevallen $(P, Q)$:
\begin{center}
\begin{tabular}{cc|c|c|c|c}
$P$ & $Q$ & $P \implies Q$ & $\lnot(P \implies Q)$ & $\lnot Q$ &
$P \land \lnot Q$ \\
\hline
W & W & W & O & O & O \\
W & O & O & W & W & W \\
O & W & W & O & O & O \\
O & O & W & O & W & O \\
\end{tabular}
\end{center}
Kolom $4$ en kolom $6$ vallen samen, wat de equivalentie bewijst. De
negatie van ``als een functie afleidbaar is, dan is ze continu'' luidt
dus: ``er bestaat een functie die afleidbaar is en niet continu'' (een
onware \omterm{def:b1:logic:statement}{uitspraak} overigens: de oorspronkelijke implicatie is waar, zie
\cref{ch:b1:derivative}).
\end{solution}

\begin{solution}{exo:b1:logic:3}
\emph{Contrapositie.} Neem aan dat $x < 1$. Dan is $x^3 < 1$ (de
derdemachtsfunctie is stijgend) en $x < 1$, dus $x^3 + x < 2$. Daarmee
is de contrapositie bewezen, en dus de \omterm{def:b1:logic:statement}{uitspraak}.

\emph{Ongerijmde.} Stel dat $a > 0$ het kleinste strikt positieve reële
getal is. Dan is $a/2$ strikt positief en $a/2 < a$ (want $a > 0$), in
tegenspraak met de minimaliteit. Zo'n $a$ bestaat dus niet.
\end{solution}

\begin{solution}{exo:b1:logic:4}
\begin{enumerate}
  \item Basisgeval $n = 0$: $2^0 = 1 = 2^1 - 1$. Stap: neem de
        identiteit aan voor $n$, dan is
        \[
        \sum_{k=0}^{n+1} 2^k = (2^{n+1} - 1) + 2^{n+1}
        = 2 \cdot 2^{n+1} - 1 = 2^{n+2} - 1 .
        \]
  \item Basisgeval $n = 0$: $4^0 + 5 = 6 = 3 \times 2$. Stap: is
        $4^n + 5 = 3m$, dan is
        \[
        4^{n+1} + 5 = 4(4^n + 5) - 15 = 3(4m - 5),
        \]
        deelbaar door $3$.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:logic:5}
De inductiestap veronderstelt stilzwijgend dat de twee groepen (``de
eerste $n$'' en ``de laatste $n$'') overlappen, zodat de gedeelde
potloden de kleur van de ene groep naar de andere overdragen. Voor
$n + 1 = 2$ zijn die groepen $\{$eerste potlood$\}$ en $\{$tweede
potlood$\}$: ze zijn disjunct, en het argument stort in. $P(1) \implies
P(2)$ is dus nooit bewezen, en daarmee valt de hele inductie weg --- ook
al is $P(n) \implies P(n+1)$ voor elke $n \geq 2$ wel geldig.
\end{solution}

\begin{solution}{exo:b1:logic:6}
\begin{enumerate}
  \item $x \in A \setminus B \iff x \in A \land x \notin B \iff x \in
        A \land x \in \overline{B} \iff x \in A \cap \overline{B}$.
  \item Met (1) en de distributiviteit
        (\cref{prop:b1:logic:setalgebra}):
        $(A \cup B) \cap \overline{C} = (A \cap \overline{C}) \cup
        (B \cap \overline{C})$.
  \item Stel $A \subseteq B$. Dan is $A \cup B \subseteq B$ (beide
        stukken liggen in $B$) en steeds $B \subseteq A \cup B$, dus
        $A \cup B = B$. Stel $A \cup B = B$: steeds is $A \cap B
        \subseteq A$, en uit $A \subseteq A \cup B = B$ volgt
        $A \subseteq A \cap B$, dus $A \cap B = A$. Stel $A \cap B =
        A$: dan is $A = A \cap B \subseteq B$. De drie voorwaarden zijn
        dus equivalent (we bewezen een kring van implicaties).
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:logic:7}
\begin{enumerate}
  \item $f$ is \omterm{def:b1:logic:inj}{injectief} ($n + 1 = m + 1 \implies n = m$) maar niet
        \omterm{def:b1:logic:inj}{surjectief}: $0$ heeft geen \omterm{def:b1:logic:map}{origineel} in $\N$.
  \item $g$ is \omterm{def:b1:logic:inj}{bijectief}: $n \mapsto n - 1$ is een tweezijdige inverse
        op $\Z$.
  \item $h$ is \omterm{def:b1:logic:inj}{injectief}: uit $\frac{x+1}{x-1} = \frac{x'+1}{x'-1}$
        volgt $(x+1)(x'-1) = (x'+1)(x-1)$, dat wil zeggen $xx' - x + x'
        - 1 = xx' - x' + x - 1$, dus $2x' = 2x$. \omterm{def:b1:logic:inj}{Surjectief} op $\R$ is
        ze niet: $y = \frac{x+1}{x-1}$ oplossen geeft $x(y - 1) = y +
        1$, wat voor $y = 1$ geen oplossing heeft (de vergelijking luidt
        dan $0 = 2$). Met codomein $\R \setminus \{1\}$ levert dezelfde
        berekening het unieke \omterm{def:b1:logic:map}{origineel} $x = \frac{y+1}{y-1}$, zodat
        $h \colon \R \setminus \{1\} \to \R \setminus \{1\}$ \omterm{def:b1:logic:inj}{bijectief} is
        met $h^{-1}(y) = \frac{y+1}{y-1} = h(y)$: $h$ is haar eigen
        inverse.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:logic:8}
\begin{enumerate}
  \item $x \in f^{-1}(B \cap B') \iff f(x) \in B \cap B' \iff f(x) \in
        B \land f(x) \in B' \iff x \in f^{-1}(B) \cap f^{-1}(B')$.
        Voor beelden: $y \in f(A \cup A')$ precies wanneer $y = f(x)$
        voor een zekere $x$ in $A$ of in $A'$, dus precies wanneer $y
        \in f(A)$ of $y \in f(A')$.
  \item Is $y \in f(A \cap A')$, dan is $y = f(x)$ met $x \in A$ en $x
        \in A'$, dus $y \in f(A)$ en $y \in f(A')$. Striktheid: neem $f
        \colon \R \to \R$, $x \mapsto x^2$, $A = \{-1\}$, $A' = \{1\}$:
        dan is $f(A \cap A') = f(\emptyset) = \emptyset$, terwijl $f(A)
        \cap f(A') = \{1\}$.
  \item ($\Leftarrow$) Neem $A = \{x\}$, $A' = \{x'\}$ met $x \neq x'$:
        was $f(x) = f(x')$, dan zou $f(A) \cap f(A') = \{f(x)\}$ zijn
        terwijl $f(A \cap A') = \emptyset$, in tegenspraak met de
        veronderstelde gelijkheid; dus is $f$ \omterm{def:b1:logic:inj}{injectief}.
        ($\Rightarrow$) Zij $f$ \omterm{def:b1:logic:inj}{injectief} en $y \in f(A) \cap f(A')$:
        $y = f(x) = f(x')$ met $x \in A$, $x' \in A'$; de injectiviteit
        geeft $x = x' \in A \cap A'$, dus $y \in f(A \cap A')$. Samen
        met (2) is dat de gelijkheid.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:logic:9}
$g \circ f = \mathrm{id}_E$ is \omterm{def:b1:logic:inj}{injectief} en \omterm{def:b1:logic:inj}{surjectief}, dus is volgens
\cref{prop:b1:logic:comp}~(2) $f$ \omterm{def:b1:logic:inj}{injectief} en $g$ \omterm{def:b1:logic:inj}{surjectief}. Voorbeeld:
$E = \N$, $F = \Z$, $f$ de inclusie $n \mapsto n$, en $g \colon \Z \to
\N$ met $g(n) = n$ voor $n \geq 0$ en $g(n) = 0$ voor $n < 0$. Dan is
$g(f(n)) = n$ voor alle $n \in \N$, maar $f$ is niet \omterm{def:b1:logic:inj}{surjectief} en $g$
niet \omterm{def:b1:logic:inj}{injectief}.
\end{solution}

\begin{solution}{exo:b1:logic:10}
$x^2 - y^2 = x - y \iff (x - y)(x + y) = x - y \iff (x - y)(x + y - 1)
= 0 \iff y = x$ of $y = 1 - x$. \emph{Reflexief:} $y = x$ voldoet.
\emph{Symmetrisch:} de voorwaarde ``$y = x$ of $y = 1 - x$'' is
symmetrisch in $x$ en $y$ (uit $y = 1 - x$ volgt $x = 1 - y$).
\emph{Transitief:} stel $x \mathbin{\mathcal{R}} y$ en $y
\mathbin{\mathcal{R}} z$; loop de vier gevallen af, en telkens is $z$
gelijk aan $x$ of aan $1 - x$ (uit $y = 1 - x$ en $z = 1 - y$ volgt
bijvoorbeeld $z = x$). Dus is $\mathcal{R}$ een \omterm{def:b1:logic:equiv}{equivalentierelatie} en
$\mathrm{cl}(x) = \{x,\, 1 - x\}$. Die klasse heeft precies één element
wanneer $x = 1 - x$, dat wil zeggen voor $x = \frac12$.
\end{solution}

\begin{solution}{exo:b1:logic:11}
Zij $f \colon E \to \mathcal{P}(E)$ een willekeurige \omterm{def:b1:logic:map}{afbeelding} en zet
$D = \{x \in E : x \notin f(x)\} \in \mathcal{P}(E)$. Stel $D = f(a)$
voor zekere $a \in E$. Is $a \in D$, dan is per definitie van $D$ juist
$a \notin f(a) = D$: tegenspraak. Is $a \notin D$, dan is $a \notin
f(a)$, dus per definitie van $D$ juist $a \in D$: tegenspraak. Dus ligt
$D$ niet in het beeld van $f$, en $f$ is niet \omterm{def:b1:logic:inj}{surjectief}. (In het
bijzonder staat geen enkele \omterm{def:b1:logic:sets}{verzameling} in bijectie met haar
\omterm{def:b1:logic:sets}{machtsverzameling}: er zijn ``meer'' deelverzamelingen van $\N$ dan gehele
getallen.)
\end{solution}

\begin{solution}{exo:b1:logic:12}
\begin{enumerate}
  \item ($\Rightarrow$) Zij $f$ \omterm{def:b1:logic:inj}{surjectief} en $\Phi(B) = \Phi(B')$.
        Kies bij $y \in B$ een $x$ met $f(x) = y$; dan is $x \in
        f^{-1}(B) = f^{-1}(B')$, dus $y = f(x) \in B'$. Bijgevolg is $B
        \subseteq B'$, en symmetrisch $B' \subseteq B$: $\Phi$ is
        \omterm{def:b1:logic:inj}{injectief}. ($\Leftarrow$) Is $f$ niet \omterm{def:b1:logic:inj}{surjectief}, kies dan een
        $y_0 \in F$ buiten het beeld; dan is $f^{-1}(\{y_0\}) =
        \emptyset = f^{-1}(\emptyset)$ terwijl $\{y_0\} \neq \emptyset$,
        dus $\Phi$ is niet \omterm{def:b1:logic:inj}{injectief}.
  \item ($\Rightarrow$) Zij $f$ \omterm{def:b1:logic:inj}{injectief} en $A \subseteq E$. Zet $B =
        f(A)$; dan is $f^{-1}(B) = \{x : f(x) \in f(A)\}$, en de
        injectiviteit geeft $f(x) \in f(A) \iff x \in A$, dus $\Phi(B) =
        A$: $\Phi$ is \omterm{def:b1:logic:inj}{surjectief}. ($\Leftarrow$) Is $f$ niet \omterm{def:b1:logic:inj}{injectief},
        neem dan $x \neq x'$ met $f(x) = f(x')$. Elk \omterm{def:b1:logic:map}{origineel}
        $f^{-1}(B)$ bevat $x$ dan en slechts dan als het $x'$ bevat;
        bijgevolg is $\{x\}$ niet van de vorm $\Phi(B)$, en $\Phi$ is
        niet \omterm{def:b1:logic:inj}{surjectief}.
\end{enumerate}
\end{solution}

\begin{solution}{pb:b1:logic:1}
\textbf{1.} \emph{Reflexief:} $\mathrm{id}_E$ is een bijectie van $E$ op
zichzelf. \emph{Symmetrisch:} is $f \colon E \to F$ \omterm{def:b1:logic:inj}{bijectief}, dan levert
\cref{thm:b1:logic:inverse} de eveneens \omterm{def:b1:logic:inj}{bijectieve} $f^{-1} \colon F \to
E$. \emph{Transitief:} zijn $f \colon E \to F$ en $g \colon F \to G$
bijecties, dan zegt \cref{prop:b1:logic:comp}~(1) dat $g \circ f \colon E
\to G$ een bijectie is. (Het gaat hier slechts om iets dat zich ``als''
een \omterm{def:b1:logic:equiv}{equivalentierelatie} gedraagt: de collectie van alle \omterm{def:b1:logic:sets}{verzamelingen} is
zelf geen \omterm{def:b1:logic:sets}{verzameling}, wegens de paradoxen waarnaar
\cref{exo:b1:logic:11} verwijst; het zijn de drie eigenschappen die
tellen.)

\textbf{2.} Zijn $f \colon E \to F$ en $g \colon F \to G$ \omterm{def:b1:logic:inj}{injectief}, dan
is $g \circ f$ \omterm{def:b1:logic:inj}{injectief} volgens \cref{prop:b1:logic:comp}~(1): $E
\preceq G$. Voor het tweede punt beperk je het codomein van $f$ tot haar
beeld: de \omterm{def:b1:logic:map}{afbeelding} $\tilde f \colon E \to f(E)$, $x \mapsto f(x)$, is
per constructie van $f(E)$ \omterm{def:b1:logic:inj}{surjectief} en \omterm{def:b1:logic:inj}{injectief} omdat $f$ dat is, dus
\omterm{def:b1:logic:inj}{bijectief}: $E \approx f(E)$.

\textbf{3.} ($\Rightarrow$) Zij $f \colon E \to F$ \omterm{def:b1:logic:inj}{injectief} en kies een
vaste $a \in E$ ($E \neq \emptyset$). Definieer $s \colon F \to E$ door:
$s(y)$ is de unieke $x$ met $f(x) = y$ wanneer $y \in f(E)$ (uniek wegens
de injectiviteit), en $s(y) = a$ in de andere gevallen. Voor elke $x \in
E$ is $s(f(x)) = x$, dus wordt elke $x$ bereikt: $s$ is \omterm{def:b1:logic:inj}{surjectief}.
($\Leftarrow$) Zij $s \colon F \to E$ \omterm{def:b1:logic:inj}{surjectief}. Kies bij elke $x \in E$
één $y_x \in F$ met $s(y_x) = x$ en zet $u(x) = y_x$. Uit $u(x) = u(x')$
volgt $x = s(u(x)) = s(u(x')) = x'$: $u \colon E \to F$ is \omterm{def:b1:logic:inj}{injectief}.

\textbf{4.} $n \mapsto n + 1$ beeldt $\N$ af in $\N^*$, is \omterm{def:b1:logic:inj}{injectief}
($n + 1 = m + 1 \implies n = m$) en \omterm{def:b1:logic:inj}{surjectief} (elke $m \geq 1$ is
$(m - 1) + 1$ met $m - 1 \in \N$). Voor $\sigma$: ze stuurt de even
getallen $0, 2, 4, \dots$ naar $0, 1, 2, \dots$ en de oneven getallen
$1, 3, 5, \dots$ naar $-1, -2, -3, \dots$ Injectiviteit: de even
argumenten landen in $\N$ ($\sigma(n) = n/2 \geq 0$) en de oneven in de
strikt negatieve gehele getallen ($\sigma(n) = -(n+1)/2 \leq -1$), zodat
een botsing binnen één pariteitsklasse moet plaatsvinden, waar $\sigma$
strikt monotoon is (uit $n/2 = m/2$ of $(n+1)/2 = (m+1)/2$ volgt $n =
m$). Surjectiviteit: $k \geq 0$ is $\sigma(2k)$; $k \leq -1$ is
$\sigma(-2k - 1)$ met $-2k - 1 \geq 1$ oneven. Dus $\N \approx \N^*$ en
$\N \approx \Z$.

\textbf{5.} Er is $C_0 = E \setminus g(F) \subseteq C$, dus uit $x \notin
C$ volgt $x \notin C_0$, oftewel $x \in g(F)$: een zekere $y \in F$
voldoet aan $g(y) = x$. Geldt ook $g(y') = x$, dan geeft de injectiviteit
van $g$ dat $y' = y$. De tweede clausule in de definitie van $h$ wijst
dus een uniek, welbepaald element $g^{-1}(x)$ aan.

\textbf{6.} Beelden verwisselen met verenigingen
(\cref{exo:b1:logic:8}~(1), toegepast op $f$ en dan op $g$):
\[
g\bigl(f(C)\bigr)
= g\Bigl(f\Bigl(\bigcup_{n \in \N} C_n\Bigr)\Bigr)
= \bigcup_{n \in \N} g\bigl(f(C_n)\bigr)
= \bigcup_{n \in \N} C_{n+1}
= \bigcup_{n \geq 1} C_n \subseteq C .
\]

\textbf{7.} Zij $x \neq x'$ in $E$. Liggen beide in $C$, dan is $h(x) =
f(x) \neq f(x') = h(x')$ wegens de injectiviteit van $f$. Ligt geen van
beide in $C$, dan is $g(h(x)) = x \neq x' = g(h(x'))$, dus $h(x) \neq
h(x')$. Is $x \in C$ en $x' \notin C$ (het gemengde geval, op verwisseling
van namen na): stel $h(x) = h(x')$, dat wil zeggen $f(x) = g^{-1}(x')$.
Toepassing van $g$ geeft $x' = g(f(x)) \in g(f(C))$, en vraag 6 levert
$x' \in C$ --- tegenspraak. In alle gevallen is dus $h(x) \neq h(x')$:
$h$ is \omterm{def:b1:logic:inj}{injectief}.

\textbf{8.} Zij $y \in F$. \emph{Geval 1: $g(y) \notin C$.} Dan is
$h(g(y)) = g^{-1}(g(y)) = y$: het element $g(y)$ is een \omterm{def:b1:logic:map}{origineel}.
\emph{Geval 2: $g(y) \in C$}, zeg $g(y) \in C_n$. Omdat $g(y) \in g(F)$,
is $g(y) \notin C_0 = E \setminus g(F)$, zodat $n \geq 1$ en $g(y) \in
C_n = g(f(C_{n-1}))$: er is een $x \in C_{n-1}$ met $g(y) = g(f(x))$. De
injectiviteit van $g$ geeft $y = f(x)$, en $x \in C_{n-1} \subseteq C$,
dus $h(x) = f(x) = y$. In beide gevallen wordt $y$ bereikt: $h$ is
\omterm{def:b1:logic:inj}{surjectief}, en dus \omterm{def:b1:logic:inj}{bijectief}.

\textbf{9.} Zijn $E \preceq F$ en $F \preceq E$, kies dan injecties $f
\colon E \to F$ en $g \colon F \to E$; de vragen 5--8 bouwen een bijectie
$h \colon E \to F$, dus $E \approx F$. De \omterm{def:b1:logic:statement}{uitspraak} is niet-triviaal
omdat de twee gegeven injecties niets met elkaar te maken hebben ---
geen van beide hoeft \omterm{def:b1:logic:inj}{surjectief} te zijn, en geen enkele naïeve formule in
$f$ en $g$ definieert een \omterm{def:b1:logic:map}{afbeelding}: de hele inhoud zit in de \omterm{thm:b1:logic:partition}{partitie}
van $E$ in het gebied $C$ (waar je $f$ overneemt) en zijn complement
(waar je $g$ achterwaarts doorloopt).

\textbf{10.} (a) De inclusie $\intoo01 \to \intcc01$ is \omterm{def:b1:logic:inj}{injectief}; en $x
\mapsto \frac{x + 1}3$ beeldt $\intcc01$ \omterm{def:b1:logic:inj}{injectief} af in
$\intcc{\frac13}{\frac23} \subseteq \intoo01$ (ze is affien met
richtingscoëfficiënt ongelijk aan nul). Met vraag 9 volgt $\intcc01
\approx \intoo01$ --- een bijectie die bijzonder onaangenaam expliciet op
te schrijven is. (b) \emph{Injectiviteit.} Stel $2^p(2q + 1) =
2^{p'}(2q' + 1)$ met bijvoorbeeld $p \leq p'$. Deling door $2^p$ geeft
$2q + 1 = 2^{p' - p}(2q' + 1)$. Was $p' > p$, dan zou het rechterlid even
en het linkerlid oneven zijn --- onmogelijk; dus $p = p'$, waarna $2q + 1
= 2q' + 1$ en $q = q'$. \emph{Surjectiviteit.} We tonen met sterke
inductie aan dat elk geheel getal $m \geq 1$ van de vorm $2^p(2q + 1)$
is. Voor $m = 1$: $p = q = 0$. Zij $m \geq 1$ en neem de bewering aan
voor alle getallen uit $\intint1m$. Is $m + 1$ oneven, dan is $m + 1 =
2q + 1$ met $p = 0$. Is $m + 1$ even, dan is $m + 1 = 2m'$ met $1 \leq m'
\leq m$; volgens de hypothese is $m' = 2^p(2q + 1)$, dus $m + 1 =
2^{p+1}(2q + 1)$. Bijgevolg bereikt $\varphi(p, q) = 2^p(2q + 1) - 1$
elke $n \in \N$, en is $\varphi$ een bijectie $\N \times \N \to \N$.

\textbf{11.} Omdat $A$ oneindig is, is $A \setminus \{\varphi(0), \dots,
\varphi(n - 1)\}$ nooit leeg, en de eigenschap van het kleinste element
van $\N$ (gebruikt in het bewijs van \cref{thm:b1:logic:induction}) maakt
de recursieve definitie geoorloofd. \emph{Strikt stijgend:} $\varphi(n +
1)$ behoort tot $A \setminus \{\varphi(0), \dots, \varphi(n)\} \subseteq
A \setminus \{\varphi(0), \dots, \varphi(n - 1)\}$, waarvan het minimum
$\varphi(n)$ is; dus $\varphi(n + 1) \geq \varphi(n)$, en gelijkheid is
uitgesloten, zodat $\varphi(n+1) > \varphi(n)$. \emph{$\varphi(n) \geq
n$:} met inductie, want $\varphi(0) \geq 0$ en $\varphi(n + 1) \geq
\varphi(n) + 1 \geq n + 1$. \emph{Injectiviteit} volgt uit de strikte
monotonie. \emph{Surjectiviteit op $A$:} stel dat een zekere $a \in A$
nooit bereikt wordt. Omdat $\varphi(a + 1) \geq a + 1 > a$, is de
\omterm{def:b1:logic:sets}{verzameling} van de $n$ met $\varphi(n) > a$ niet leeg; zij $n$ haar
kleinste element. Voor elke $k < n$ is $\varphi(k) \leq a$, en dus
$\varphi(k) < a$ ($a$ wordt niet bereikt). Dan ligt $a$ in $A \setminus
\{\varphi(0), \dots, \varphi(n - 1)\}$ met $a < \varphi(n)$, in
tegenspraak met de minimaliteit die $\varphi(n)$ definieert. Dus is
$\varphi$ een bijectie $\N \to A$ en is $A$ aftelbaar.

\textbf{12.} Zij $E \preceq \N$ via een injectie $f$; dan is $E \approx
f(E)$ (vraag 2). Is $f(E)$ eindig, dan is $E$ eindig; is $f(E)$ oneindig,
dan geeft vraag 11 dat $f(E) \approx \N$, dus $E \approx \N$ wegens de
transitiviteit (vraag 1). Omgekeerd injecteren eindige en \omterm{pb:b1:logic:1}{aftelbare
verzamelingen} uiteraard in $\N$. De kortere weg: uit $E \preceq \N$ en
$\N \preceq E$ volgt $E \approx \N$ rechtstreeks met
Cantor--Schröder--Bernstein --- zonder enig opsommingsargument.

\textbf{13.} Zij $f \colon E \to \N$ en $g \colon F \to \N$ injecties.
Dan is $(x, y) \mapsto \varphi\bigl(f(x), g(y)\bigr)$ een injectie $E
\times F \to \N$: vallen de beelden samen, dan geeft de injectiviteit van
$\varphi$ (vraag 10) dat $f(x) = f(x')$ en $g(y) = g(y')$, en dus $x =
x'$, $y = y'$. Voor $\Z \times \N^*$: beide factoren zijn aftelbaar
(vraag 4), dus $\Z \times \N^* \preceq \N$; de \omterm{def:b1:logic:sets}{verzameling} is oneindig
(ze bevat $\{0\} \times \N^*$), en dus aftelbaar volgens vraag 12.

\textbf{14.} Elk rationaal getal $r$ heeft precies één schrijfwijze $r =
p/q$ met $p \in \Z$, $q \in \N^*$ en de breuk onvereenvoudigbaar (de
uniciteit wordt bewezen in \cref{ch:b1:arith}; voor $r = 0$ neem je
$0/1$). De \omterm{def:b1:logic:map}{afbeelding} $r \mapsto (p, q)$ is dan \omterm{def:b1:logic:inj}{injectief}: het paar legt
$r = p/q$ vast. Bijgevolg is $\Q \preceq \Z \times \N^* \preceq \N$
volgens vraag 13. Omdat $\N \subseteq \Q$ ook $\N \preceq \Q$ geeft,
toont vraag 12 (of rechtstreeks Cantor--Schröder--Bernstein) dat $\Q
\approx \N$: de rationale getallen zijn aftelbaar.

\textbf{15.} Kies bij elke $n$ een injectie $f_n \colon E_n \to \N$. Zij
voor $x \in \bigcup_n E_n$ het getal $n(x)$ de \emph{kleinste} $n$ met $x
\in E_n$, en zet $u(x) = \varphi\bigl(n(x), f_{n(x)}(x)\bigr) \in \N$.
Uit $u(x) = u(x')$ geeft de injectiviteit van $\varphi$ dat $n(x) = n(x')
= n$ en $f_n(x) = f_n(x')$, dus $x = x'$ wegens de injectiviteit van
$f_n$. De vereniging injecteert dus in $\N$: ze is hoogstens aftelbaar.

\textbf{16.} Zet $\Psi(F) = \sum_{i \in F} 2^i$ voor eindige $F \subseteq
\N$ (met $\Psi(\emptyset) = 0$). Stel $F \neq F'$ en zij $m$ het grootste
element waarop ze verschillen, zeg $m \in F \setminus F'$ (verwissel zo
nodig de namen). Elementen $> m$ behoren tot beide of tot geen van beide
en dragen dus even veel bij aan beide sommen; vergelijken we de bijdragen
van de elementen $\leq m$:
\[
\sum_{i \in F,\, i \leq m} 2^i \geq 2^m
> 2^m - 1 = \sum_{k=0}^{m-1} 2^k
\geq \sum_{i \in F',\, i \leq m} 2^i ,
\]
met de meetkundige som uit \cref{exo:b1:logic:4}. Bijgevolg is $\Psi(F)
\neq \Psi(F')$: $\Psi$ is \omterm{def:b1:logic:inj}{injectief} en de \omterm{def:b1:logic:sets}{verzameling} van eindige
deelverzamelingen van $\N$ is hoogstens aftelbaar; ze is oneindig (ze
bevat alle singletons) en dus aftelbaar.

\textbf{17.} Stuur $A \subseteq \N$ naar haar indicator $\mathbf 1_A
\colon \N \to \{0,1\}$, met $\mathbf 1_A(n) = 1$ als $n \in A$ en $0$ in
het andere geval; stuur $u \in \{0,1\}^{\N}$ naar $A_u = \{n \in \N :
u(n) = 1\}$. Beide \omterm{def:b1:logic:map}{afbeeldingen} zijn elkaars inverse: $A_{\mathbf 1_A} =
A$ en $\mathbf 1_{A_u} = u$ (ga de waarde in elke $n$ na). Volgens
\cref{thm:b1:logic:inverse} is elk van beide een bijectie:
$\mathcal{P}(\N) \approx \{0,1\}^{\N}$.

\textbf{18.} Voor elke $n$ is $d(n) = 1 - \Phi(n)(n) \neq \Phi(n)(n)$,
zodat de rijen $d$ en $\Phi(n)$ op de index $n$ verschillen: $d \neq
\Phi(n)$. Geen enkele $\Phi$ is dus \omterm{def:b1:logic:inj}{surjectief}, en volgens vraag 3
bestaat er evenmin een injectie $\{0,1\}^{\N} \to \N$: $\{0,1\}^{\N}$ is
niet hoogstens aftelbaar. Via het woordenboek van vraag 17 is een
\omterm{def:b1:logic:map}{afbeelding} $\Phi \colon \N \to \{0,1\}^{\N}$ een \omterm{def:b1:logic:map}{afbeelding} $f \colon \N
\to \mathcal{P}(\N)$, en $d$ komt overeen met de \omterm{def:b1:logic:sets}{verzameling} $D = \{n : n
\notin f(n)\}$ (immers $d(n) = 1 \iff \Phi(n)(n) = 0 \iff n \notin
f(n)$): het diagonaalargument \emph{is} Cantors bewijs van
\cref{exo:b1:logic:11} voor $E = \N$.

\textbf{19.} Schrijf $x_n = 0.d_1(n)\,d_2(n)\,d_3(n)\dots$ in eigenlijke
vorm en definieer $\delta_n = 5$ als $d_n(n) \neq 5$ en $\delta_n = 6$
als $d_n(n) = 5$, en zet $x = 0.\delta_1\delta_2\delta_3\dots$ Deze
ontwikkeling gebruikt alleen de cijfers $5$ en $6$ en eindigt dus niet op
louter negens: ze is de eigenlijke ontwikkeling van een reëel getal $x
\in \intco01$. Voor elke $n$ verschillen de $n$-de cijfers van $x$ en
$x_n$ ($\delta_n \neq d_n(n)$ per constructie); omdat eigenlijke
ontwikkelingen uniek zijn, is $x \neq x_n$. Geen enkele rij put $\intco01$
dus uit: opnieuw met vraag 3 is $\intco01$ niet hoogstens aftelbaar.

\textbf{20.} Er geldt $\intco01 \subseteq \R$, dus een injectie $\R \to
\N$ zou tot een injectie op $\intco01$ beperken, in tegenspraak met vraag
19: $\R$ is overaftelbaar. Was $\R \setminus \Q$ hoogstens aftelbaar, dan
zou $\R = \Q \cup (\R \setminus \Q)$ een vereniging van twee hoogstens
\omterm{pb:b1:logic:1}{aftelbare verzamelingen} zijn en dus hoogstens aftelbaar volgens vraag 15
(neem $E_0 = \Q$ en $E_n = \R \setminus \Q$ voor $n \geq 1$) ---
tegenspraak. De irrationale getallen zijn dus overaftelbaar. Preciezer:
binnen $\R$ vormen de rationale getallen een \omterm{pb:b1:logic:1}{aftelbare verzameling}
terwijl hun complement overaftelbaar is; geen enkele bijectie kan $\R
\setminus \Q$ ooit met $\Q$ laten samenvallen --- er zijn strikt ``meer''
irrationale dan rationale getallen, ook al zijn beide oneindig en beide
dicht.

\textbf{21.} $p/q$ (met $q \neq 0$) is een nulpunt van $qX - p$, een
veelterm met gehele coëfficiënten die niet nul is. Verder is $\sqrt 2$
een nulpunt van $X^2 - 2$. Voor $x = \sqrt 2 + \sqrt 3$: $x^2 = 5 +
2\sqrt 6$, dus $x^2 - 5 = 2\sqrt 6$ en $(x^2 - 5)^2 = 24$, oftewel
\[
x^4 - 10x^2 + 1 = 0 :
\]
$\sqrt 2 + \sqrt 3$ is een nulpunt van $X^4 - 10X^2 + 1$.

\textbf{22.} Beeld $P = a_0 + a_1X + \dots + a_nX^n$ (graad $\leq n$,
gehele coëfficiënten) af op $(a_0, \dots, a_n) \in \Z^{n+1}$: dit is
\omterm{def:b1:logic:inj}{injectief}, want een veelterm ligt vast door zijn coëfficiënten. Met
inductie naar $n$: $\Z^1 = \Z$ is aftelbaar (vraag 4), en $\Z^{n+2}
\approx \Z^{n+1} \times \Z$ is hoogstens aftelbaar volgens vraag 13. Elke
\omterm{def:b1:logic:sets}{verzameling} van gehele veeltermen van begrensde graad is dus hoogstens
aftelbaar; ze is oneindig (ze bevat de constanten) en dus aftelbaar
volgens vraag 12.

\textbf{23.} De \omterm{def:b1:logic:sets}{verzameling} van alle gehele veeltermen is
\[
\bigcup_{n \in \N} \{P : \deg P \leq n,\ P \text{ heeft gehele
coëfficiënten}\},
\]
een aftelbare vereniging van \omterm{pb:b1:logic:1}{aftelbare verzamelingen}: hoogstens
aftelbaar volgens vraag 15, oneindig, en dus aftelbaar.

\textbf{24.} Voor elke gehele veelterm $P$ die niet nul is, is de
nulpuntenverzameling $R_P = \{x \in \R : P(x) = 0\}$ eindig (hoogstens
$\deg P$ elementen, aangenomen). Volgens vraag 23 kunnen de gehele
veeltermen die niet nul zijn opgesomd worden als $P_0, P_1, P_2, \dots$;
dan is $\mathcal{A} = \bigcup_{n \in \N} R_{P_n}$ een aftelbare
vereniging van eindige (dus hoogstens aftelbare) \omterm{def:b1:logic:sets}{verzamelingen}: hoogstens
aftelbaar volgens vraag 15. Ze bevat $\Q$ (vraag 21) en is dus oneindig:
$\mathcal{A}$ is aftelbaar.

\textbf{25.} Was $\R \setminus \mathcal{A}$ hoogstens aftelbaar, dan zou
$\R = \mathcal{A} \cup (\R \setminus \mathcal{A})$ hoogstens aftelbaar
zijn (vraag 15), in tegenspraak met vraag 20. Er bestaan dus
\omterm{pb:b1:logic:1}{transcendente} getallen, en ze vormen zelfs een overaftelbare
\omterm{def:b1:logic:sets}{verzameling}, terwijl de algebraïsche getallen --- waaronder elk getal dat
uit gehele getallen met worteltrekkingen gebouwd is --- slechts een
aftelbaar skelet binnen $\R$ vormen. Overzicht van de architectuur: de
vragen 1--3 leggen de vergelijkingstaal vast;
Cantor--Schröder--Bernstein (vragen 5--9) laat toe gelijkmachtigheid te
bewijzen met twee makkelijke injecties in plaats van één slimme bijectie,
en werd gebruikt voor $\intcc01 \approx \intoo01$, voor $\Q$ en overal in
Deel V; de koppelingsbijectie (vraag 10) dreef de producten en de
aftelbare verenigingen aan (vragen 13 en 15), die op hun beurt $\Q$, de
gehele veeltermen en $\mathcal{A}$ aandreven; het diagonaalargument
(vragen 18--19) leverde de ene strikte ongelijkheid $\N \prec \R$ die het
hele verhaal niet-triviaal maakt. Cantors conclusie is filosofisch
opzienbarend: het bewijs wijst geen enkel \omterm{pb:b1:logic:1}{transcendent getal} aan en toont
toch dat in de zin van gelijkmachtigheid \emph{bijna elk} reëel getal
\omterm{pb:b1:logic:1}{transcendent} is. Van één specifiek getal --- $\pi$ of $\eu$ --- de
transcendentie aantonen vergde volstrekt andere wiskunde en nog decennia
werk.
\end{solution}
