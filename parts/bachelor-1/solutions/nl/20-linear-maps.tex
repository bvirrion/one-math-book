\section*{Hoofdstuk \ref{ch:b1:linmaps} --- Lineaire afbeeldingen}

\begin{solution}{exo:b1:linmaps:1}
(1) \omterm{def:b1:linmaps:def}{Lineair}: \omterm{prop:b1:vspaces:coordinates}{coördinaten} zijn \omterm{def:b1:linmaps:def}{lineaire} uitdrukkingen. (2) Niet
\omterm{def:b1:linmaps:def}{lineair}: $u(2(1,1)) = 4 \neq 2 = 2u(1,1)$. (3) \omterm{def:b1:linmaps:def}{Lineair}: afleiden en
vermenigvuldigen met $X$ zijn dat, en sommen van \omterm{def:b1:linmaps:def}{lineaire
afbeeldingen} ook. (4) \omterm{def:b1:linmaps:def}{Lineair}: de evaluatie respecteert de
puntsgewijze bewerkingen.
\end{solution}

\begin{solution}{exo:b1:linmaps:2}
\omterm{def:b1:linmaps:kerim}{Kern}: los $x + y - z = 0$, $2x + y + z = 0$, $3x + 2y = 0$ op. Uit
de derde volgt $y = -\frac{3x}{2}$; de eerste geeft $z = x + y =
-\frac x2$; controle in de tweede: $2x - \frac{3x}{2} - \frac x2 =
0$: voldaan. Dus $\ker u = \operatorname{Vect}\bigl((2, -3,
-1)\bigr)$ (met $x = 2$), van dimensie $1$.

Dimensiestelling: $\operatorname{rk} u = 3 - 1 = 2$. \omterm{def:b1:linmaps:kerim}{Beeld}:
opgespannen door de \omterm{def:b1:linmaps:kerim}{beelden} van de canonieke \omterm{def:b1:vspaces:free}{basis}, $u(e_1) =
(1,2,3)$, $u(e_2) = (1,1,2)$, $u(e_3) = (-1,1,0)$; de eerste twee
zijn \omterm{def:b1:vspaces:free}{vrij}, en de \omterm{def:b1:linmaps:rank}{rang} is $2$: \omterm{def:b1:vspaces:free}{basis} $\bigl((1,2,3),
(1,1,2)\bigr)$.

Niet \omterm{def:b1:logic:inj}{injectief} ($\ker \neq \{0\}$), niet \omterm{def:b1:logic:inj}{surjectief} (\omterm{def:b1:linmaps:rank}{rang} $2 < 3$):
in overeenstemming met \cref{cor:b1:linmaps:samedim}.
\end{solution}

\begin{solution}{exo:b1:linmaps:3}
\emph{\omterm{def:b1:linmaps:kerim}{Kern}:} $P = P'$ dwingt $\deg P = \deg P'$ af tenzij $P = 0$;
maar $\deg P' < \deg P$ voor $P \neq 0$: dus $\ker u = \{0\}$, en
$u$, een \omterm{def:b1:logic:inj}{injectief} endomorfisme van het \omterm{def:b1:findim:def}{eindigdimensionale}
$\R_n[X]$, is een isomorfisme (\cref{cor:b1:linmaps:samedim}).

\emph{Inverse:} zij $v(P) = P + P' + P'' + \dots + P^{(n)}$ (een
eindige som op $\R_n[X]$). Dan is
\[
v\bigl(u(P)\bigr) = \sum_{k=0}^{n} (P - P')^{(k)}
= \sum_{k=0}^{n} P^{(k)} - \sum_{k=0}^{n} P^{(k+1)}
= P - P^{(n+1)} = P ,
\]
telescoperend, want $P^{(n+1)} = 0$. Dus $v = u^{-1}$.
\end{solution}

\begin{solution}{exo:b1:linmaps:4}
$\operatorname{im}(v \circ u) = v(\operatorname{im} u) \subseteq
\operatorname{im} v$: de \omterm{def:b1:linmaps:rank}{rang} is $\leq \operatorname{rk} v$. En $v$
beperkt tot $\operatorname{im} u$ heeft als \omterm{def:b1:linmaps:kerim}{beeld}
$\operatorname{im}(vu)$, waarop de dimensiestelling binnen
$\operatorname{im} u$ geeft: $\operatorname{rk}(vu) \leq
\dim\operatorname{im} u = \operatorname{rk} u$.
\end{solution}

\begin{solution}{exo:b1:linmaps:5}
$u^2 = 0$ betekent $u(u(x)) = 0$ voor alle $x$: elke $u(x)$ ligt in
$\ker u$, dat wil zeggen $\operatorname{im} u \subseteq \ker u$.
Dan geeft de dimensiestelling
\[
\dim E = \dim\ker u + \operatorname{rk} u \geq 2\operatorname{rk} u .
\]
Voorbeeld met gelijkheid in $\R^2$: $u(x, y) = (y, 0)$: $u^2 = 0$,
$\operatorname{rk} u = 1 = \frac{\dim E}{2}$.
\end{solution}

\begin{solution}{exo:b1:linmaps:6}
$(pq)^2 = pqpq = ppqq = pq$ (commutatie): een \omterm{def:b1:linmaps:projection}{projectie}
(\cref{thm:b1:linmaps:projchar}).

\omterm{def:b1:linmaps:kerim}{Beeld}: $\operatorname{im}(pq) \subseteq \operatorname{im} p$, en $=
\operatorname{im}(qp) \subseteq \operatorname{im} q$: bevat in de
doorsnede. Omgekeerd, is $x \in \operatorname{im} p \cap
\operatorname{im} q$, dan is $p(x) = x$ en $q(x) = x$ (de vaste
punten karakteriseren het \omterm{def:b1:linmaps:kerim}{beeld} van een \omterm{def:b1:linmaps:projection}{projectie}), dus $pq(x) =
x$: $x \in \operatorname{im}(pq)$.

\omterm{def:b1:linmaps:kerim}{Kern}: $\ker p \subseteq \ker(qp) = \ker(pq)$ en evenzo $\ker q
\subseteq \ker(pq)$: de som is erin bevat. Omgekeerd, zij $pq(x) =
0$, en schrijf
\[
x = \underbrace{q(x)}_{\in\, \ker p} +
\underbrace{(x - q(x))}_{\in\, \ker q} :
\]
de eerste term voldoet aan $p(q(x)) = 0$ en ligt dus in $\ker p$; de
tweede ligt in $\ker q$ omdat $q(x - q(x)) = q(x) - q^2(x) = 0$.
Bijgevolg $x \in \ker p + \ker q$.
\end{solution}

\begin{solution}{exo:b1:linmaps:7}
Merk eerst de algemene insluitingen $\ker u \subseteq \ker u^2$ en
$\operatorname{im} u^2 \subseteq \operatorname{im} u$ op, en dat
volgens de dimensiestelling (2) $\iff$ (3) (gelijke \omterm{def:b1:linmaps:kerim}{kernen} $\iff$
gelijke \omterm{def:b1:linmaps:rank}{rangen} $\iff$ gelijke \omterm{def:b1:linmaps:kerim}{beelden}, gegeven de insluitingen).

(1 $\Rightarrow$ 2): zij $u^2(x) = 0$; dan is $u(x) \in \ker u \cap
\operatorname{im} u = \{0\}$, dus $x \in \ker u$.

(2 $\Rightarrow$ 1): volgens Grassmann en de dimensiestelling is
$\dim(\ker u + \operatorname{im} u) = \dim\ker u + \operatorname{rk}
u - \dim(\ker u \cap \operatorname{im} u) = \dim E - \dim(\ker u
\cap \operatorname{im} u)$: de som is $E$ dan en slechts dan als de
doorsnede $\{0\}$ is. Zij $y \in \ker u \cap \operatorname{im} u$:
$y = u(x)$ en $u(y) = 0$, dus $u^2(x) = 0$, dus (wegens (2)) $u(x) =
0$: $y = 0$. Bijgevolg $E = \ker u \oplus \operatorname{im} u$.
\end{solution}

\begin{solution}{exo:b1:linmaps:8}
Is $\varphi = 0$, dan dwingt $\ker\varphi = E \subseteq \ker\psi$
af dat $\psi = 0 = 0\cdot\varphi$. Is $\varphi \neq 0$, dan is
$\ker\varphi$ een \omterm{def:b1:linmaps:forms}{hypervlak}; kies $a \notin \ker\varphi$. Stel
$\lambda = \frac{\psi(a)}{\varphi(a)}$. De vorm $\psi -
\lambda\varphi$ wordt nul op $\ker\varphi$ (beide worden dat, wegens
de insluiting) en in $a$: zij wordt nul op $\ker\varphi \oplus Ka =
E$. Dus $\psi = \lambda\varphi$.
\end{solution}

\begin{solution}{exo:b1:linmaps:9}
Stel $\lambda_0 x + \lambda_1 u(x) + \dots + \lambda_{n-1}
u^{n-1}(x) = 0$. Pas $u^{n-1}$ toe: alle termen met een factor
$u^{\geq n}$ sterven, zodat $\lambda_0 u^{n-1}(x) = 0$ overblijft,
dus $\lambda_0 = 0$. Pas $u^{n-2}$ toe op de overgebleven
betrekking: $\lambda_1 u^{n-1}(x) = 0$, dus $\lambda_1 = 0$;
enzovoort. De familie is \omterm{def:b1:vspaces:free}{vrij}; met $n = \dim E$ elementen is zij een
\omterm{def:b1:vspaces:free}{basis} (\cref{prop:b1:findim:twoofthree}). (In die \omterm{def:b1:vspaces:free}{basis} werkt $u$
als een verschuiving --- het model van de maximale nilpotentie.)
\end{solution}

\begin{solution}{exo:b1:linmaps:10}
\begin{enumerate}
  \item $f \circ f = -\mathrm{id}$ is \omterm{def:b1:logic:inj}{bijectief}, dus $f$ ook
        (\cref{prop:b1:logic:comp}, aangepast: $f$ heeft de
        tweezijdige inverse $-f$). Is $f(x) = \lambda x$ met $x
        \neq 0$, dan geeft toepassing van $f$ dat $-x = \lambda^2
        x$, dus $\lambda^2 = -1$: onmogelijk in $\R$.
  \item Bouw de familie gulzig op. Neem $x_1 \neq 0$: $(x_1,
        f(x_1))$ is \omterm{def:b1:vspaces:free}{vrij} volgens (1). Is het \omterm{def:b1:vspaces:span}{opspansel} van de
        huidige familie $\bigl(x_1, f(x_1), \dots, x_k,
        f(x_k)\bigr)$, noem het $V_k$ --- een \omterm{def:b1:vspaces:subspace}{deelruimte} die onder
        $f$ stabiel is (elke voortbrenger wordt op een andere
        voortbrenger of diens tegengestelde afgebeeld: $f(f(x_i)) =
        -x_i$) --- niet heel $E$, kies dan $x_{k+1} \notin V_k$.
        \emph{Bewering: de vergrote familie is \omterm{def:b1:vspaces:free}{vrij}.} Stel $\alpha
        x_{k+1} + \beta f(x_{k+1}) + v = 0$ met $v \in V_k$ en
        $(\alpha, \beta) \neq (0,0)$. Pas $f$ toe: $\alpha
        f(x_{k+1}) - \beta x_{k+1} + f(v) = 0$ met $f(v) \in V_k$.
        Elimineer $f(x_{k+1})$ tussen beide betrekkingen
        (vermenigvuldig de eerste met $\alpha$, de tweede met
        $-\beta$, en tel op):
        \[
        (\alpha^2 + \beta^2)\, x_{k+1} \in V_k ,
        \]
        en $\alpha^2 + \beta^2 \neq 0$ dwingt $x_{k+1} \in V_k$ af:
        tegenspraak. De constructie gaat dus door en voegt telkens
        \emph{twee} vectoren toe, totdat $V_k = E$: de uiteindelijke
        familie is een \omterm{def:b1:vspaces:free}{basis} van even omvang, en $n$ is even.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:linmaps:11}
\omterm{def:b1:reals:bounds}{Bovengrens}: voor elke $x$ is $(u + v)(x) = u(x) + v(x) \in
\operatorname{im} u + \operatorname{im} v$, dus
\[
\operatorname{rk}(u + v)
\leq \dim(\operatorname{im} u + \operatorname{im} v)
\leq \operatorname{rk} u + \operatorname{rk} v
\]
(Grassmann, \cref{thm:b1:findim:grassmann}). Ondergrens: pas de
\omterm{def:b1:reals:bounds}{bovengrens} toe op het paar $(u + v, -v)$, waarvan de som $u$ is:
\[
\operatorname{rk} u \leq \operatorname{rk}(u + v) +
\operatorname{rk}(-v) = \operatorname{rk}(u + v) +
\operatorname{rk} v,
\]
dus $\operatorname{rk} u - \operatorname{rk} v \leq
\operatorname{rk}(u+v)$; verwisseling van $u$ en $v$ geeft de
absolute waarde.
\end{solution}

\begin{solution}{exo:b1:linmaps:12}
\emph{Exacte formule.} Zij $v'$ de beperking van $v$ tot de
\omterm{def:b1:vspaces:subspace}{deelruimte} $\operatorname{im} w$. Haar \omterm{def:b1:linmaps:kerim}{beeld} is $v(w(F)) =
\operatorname{im}(v \circ w)$, en haar \omterm{def:b1:linmaps:kerim}{kern} is $\ker v \cap
\operatorname{im} w$. De dimensiestelling voor $v'$ op de ruimte
$\operatorname{im} w$ geeft
\[
\operatorname{rk} w = \dim\operatorname{im} w
= \operatorname{rk}(v \circ w) + \dim(\ker v \cap
\operatorname{im} w) .
\]

\emph{Frobenius.} Pas de exacte formule twee keer toe, op $w$ en op
$w \circ u$:
\[
\operatorname{rk} w - \operatorname{rk}(vw)
= \dim\bigl(\ker v \cap \operatorname{im} w\bigr),
\qquad
\operatorname{rk}(wu) - \operatorname{rk}(vwu)
= \dim\bigl(\ker v \cap \operatorname{im}(wu)\bigr) .
\]
Omdat $\operatorname{im}(w \circ u) \subseteq \operatorname{im} w$,
is de tweede doorsnede in de eerste bevat en is haar dimensie niet
groter:
\[
\operatorname{rk}(wu) - \operatorname{rk}(vwu)
\;\leq\; \operatorname{rk} w - \operatorname{rk}(vw) ,
\]
wat zich herschikt tot de ongelijkheid van Frobenius. Met $w =
\mathrm{id}_F$ (\omterm{def:b1:linmaps:rank}{rang} $\dim F$, en $\operatorname{im}\,
\mathrm{id}_F = F$): $\operatorname{rk} v + \operatorname{rk} u
\leq \dim F + \operatorname{rk}(vu)$, de ongelijkheid van Sylvester
--- opnieuw bewezen, in matrixvorm, in \cref{exo:b1:matrices:10}.
\end{solution}

\begin{solution}{pb:b1:linmaps:1}
\textbf{1.} $(\mathrm{id} - p)^2 = \mathrm{id} - 2p + p^2 =
\mathrm{id} - p$: een projector. Is $y = x - p(x)$, dan is $p(y) =
p(x) - p^2(x) = 0$, en omgekeerd geeft $x \in \ker p$ dat $x =
(\mathrm{id} - p)(x)$: $\operatorname{im}(\mathrm{id} - p) = \ker
p$. En $(\mathrm{id} - p)(x) = 0 \iff p(x) = x \iff x \in
\operatorname{im} p$ (vaste punten,
\cref{thm:b1:linmaps:projchar}): $\ker(\mathrm{id} - p) =
\operatorname{im} p$.

\textbf{2.} $(\lambda\,\mathrm{id} + \mu p)^2 =
\lambda^2\,\mathrm{id} + (2\lambda\mu + \mu^2)\,p$. Het paar
$(\mathrm{id}, p)$ is \omterm{def:b1:vspaces:free}{vrij} in $\mathcal{L}(E)$: uit $p = c\,
\mathrm{id}$ zou $c^2 = c$ volgen, dus $p = 0$ of $\mathrm{id}$,
wat is uitgesloten. Door de coëfficiënten te identificeren is de
\omterm{def:b1:logic:map}{afbeelding} een projector dan en slechts dan als $\lambda^2 =
\lambda$ en $2\lambda\mu + \mu^2 = \mu$. Voor $\lambda = 0$: $\mu
\in \{0, 1\}$. Voor $\lambda = 1$: $\mu^2 + \mu = 0$, dus $\mu \in
\{0, -1\}$. Precies vier projectoren in het vlak: $0$, $p$,
$\mathrm{id}$, $\mathrm{id} - p$.

\textbf{3.} $(\lambda\,\mathrm{id} + \mu p)(\lambda'\,\mathrm{id}
+ \mu' p) = \lambda\lambda'\,\mathrm{id} + (\lambda\mu' +
\mu\lambda' + \mu\mu')\,p$: het vlak is stabiel onder
samenstelling. Omdat $p^k = p$ voor elke $k \geq 1$, is voor $Q =
\sum_k a_k X^k$
\[
Q(p) = a_0\,\mathrm{id} + \Bigl(\sum_{k \geq 1} a_k\Bigr) p
= Q(0)\,\mathrm{id} + \bigl(Q(1) - Q(0)\bigr)\,p .
\]

\textbf{4.} Op $\operatorname{im} p$ (waar $p$ als de identiteit
werkt) vermenigvuldigt $\lambda\,\mathrm{id} + \mu p$ met $\lambda
+ \mu$; op $\ker p$ met $\lambda$. Omdat $E = \operatorname{im} p
\oplus \ker p$, is de \omterm{def:b1:logic:map}{afbeelding} \omterm{def:b1:logic:inj}{bijectief} dan en slechts dan als
$\lambda \neq 0$ en $\lambda + \mu \neq 0$. Oplossen van
$\lambda\alpha = 1$ en $\lambda\beta + \mu\alpha + \mu\beta = 0$ in
de samenstellingsregel van vraag 3 geeft
\[
(\lambda\,\mathrm{id} + \mu p)^{-1}
= \frac1\lambda\,\mathrm{id} -
\frac{\mu}{\lambda(\lambda + \mu)}\,p ,
\]
waarvan de werking $1/\lambda$ is op $\ker p$ en $1/(\lambda +
\mu)$ op $\operatorname{im} p$, zoals het moet.

\textbf{5.} Schrijf $F = \operatorname{im} p = \operatorname{im}
p'$. Voor elke $x$ is $p'(x) \in F$ en houdt $p$ de ruimte $F$
puntsgewijs vast: $p(p'(x)) = p'(x)$, dus $p\,p' = p'$;
symmetrisch $p'\,p = p$. Delen twee \omterm{def:b1:linmaps:projection}{projecties} hun \omterm{def:b1:linmaps:kerim}{beeld}, dan
beslist degene die \emph{eerst} wordt toegepast: haar uitvoer ligt
al in $F$, waar de buitenste \omterm{def:b1:linmaps:projection}{projectie} als de identiteit werkt en
niets verandert.

\textbf{6.} $(p + q)^2 = p^2 + pq + qp + q^2 = (p + q) + pq + qp$,
dus dwingt ``$p + q$ is een projector'' af dat $pq + qp = 0$. Stel
links met $p$ samen: $pq + pqp = 0$; rechts met $p$: $pqp + qp =
0$. Aftrekken geeft $pq = qp$; dan is $pq + qp = 2pq = 0$ en is de
karakteristiek niet $2$: $pq = qp = 0$.

\textbf{7.} Met $pq = qp = 0$ geeft dezelfde uitwerking $(p + q)^2
= p + q$. \omterm{def:b1:linmaps:kerim}{Beeld}: $\operatorname{im}(p + q) \subseteq
\operatorname{im} p + \operatorname{im} q$ geldt altijd. Omgekeerd
is voor $x \in \operatorname{im} p$: $q(x) = q(p(x)) = 0$, dus $(p
+ q)(x) = p(x) = x$ en $x \in \operatorname{im}(p+q)$; net zo voor
$\operatorname{im} q$. Directheid: $x \in \operatorname{im} p \cap
\operatorname{im} q$ geeft $x = p(x) = p(q(x)) = 0$. \omterm{def:b1:linmaps:kerim}{Kern}: is $p(x)
+ q(x) = 0$, dan geeft toepassing van $p$ dat $p(x) + p(q(x)) =
p(x) = 0$, en toepassing van $q$ dat $q(x) = 0$: $\ker(p + q) =
\ker p \cap \ker q$ (de omgekeerde insluiting is duidelijk).

\textbf{8.} $p - q$ is een projector dan en slechts dan als
$\mathrm{id} - (p - q) = (\mathrm{id} - p) + q$ er een is (twee
keer vraag 1). Volgens de vragen 6--7, toegepast op de projectoren
$\mathrm{id} - p$ en $q$, geldt dit dan en slechts dan als
$(\mathrm{id} - p)q = q(\mathrm{id} - p) = 0$, dat wil zeggen dan
en slechts dan als $pq = q$ en $qp = q$.

\textbf{9.} $pq = q$ betekent dat $p$ elke $q(x)$ vasthoudt, dat
wil zeggen $\operatorname{im} q \subseteq \ker(p - \mathrm{id}) =
\operatorname{im} p$. En $qp = q$ betekent dat
$q\bigl((\mathrm{id} - p)(x)\bigr) = 0$ voor alle $x$, dat wil
zeggen dat $q$ nul wordt op $\operatorname{im}(\mathrm{id} - p) =
\ker p$: $\ker p \subseteq \ker q$. Beide stappen zijn
equivalenties: de ordening $q \leq p$ zegt dat $q$ op een kleiner
\omterm{def:b1:linmaps:kerim}{beeld} projecteert, langs een grotere \omterm{def:b1:linmaps:kerim}{kern}.

\textbf{10.} Uitwerken geeft $(\mathrm{id} - p)(\mathrm{id} - q) =
\mathrm{id} - p - q + pq = \mathrm{id} - r$. De projectoren
$\mathrm{id} - p$ en $\mathrm{id} - q$ commuteren, dus is hun
product $\mathrm{id} - r$ volgens \cref{exo:b1:linmaps:6} de
projector op $\operatorname{im}(\mathrm{id} - p) \cap
\operatorname{im}(\mathrm{id} - q) = \ker p \cap \ker q$ langs
$\ker(\mathrm{id} - p) + \ker(\mathrm{id} - q) = \operatorname{im}
p + \operatorname{im} q$. Volgens vraag 1 is $r = \mathrm{id} -
(\mathrm{id} - r)$ dan de projector met $\operatorname{im} r =
\operatorname{im} p + \operatorname{im} q$ en $\ker r = \ker p \cap
\ker q$.

\textbf{11.} $p_i$ is welgedefinieerd (eenduidigheid van de
ontbinding) en \omterm{def:b1:linmaps:def}{lineair} (de ontbinding van $x + \lambda y$ is de som
van de ontbindingen, opnieuw wegens de eenduidigheid). Voor $x_i
\in F_i$ is de ontbinding $x_i$ zelf, dus $p_i(x_i) = x_i$: $p_i^2
= p_i$, en $p_j(x_i) = 0$ voor $j \neq i$: $p_i p_j = 0$ (want
$p_j(x) \in F_j$). Optelling van de componenten geeft $\sum_i p_i =
\mathrm{id}$. Ten slotte is $\operatorname{im} p_i = F_i$ en $\ker
p_i = \bigoplus_{j \neq i} F_j$.

\textbf{12.} $p_i = p_i \circ \mathrm{id} = p_i\sum_j p_j = p_i^2 +
\sum_{j \neq i} p_i p_j = p_i^2$: elke $p_i$ is een projector. Elke
$x = \mathrm{id}(x) = \sum_i p_i(x)$ ligt in $\sum_i
\operatorname{im} p_i$: de \omterm{def:b1:linmaps:kerim}{beelden} sommeren tot $E$. Directheid:
stel $y_1 + \dots + y_k = 0$ met $y_i \in \operatorname{im} p_i$,
zodat $p_i(y_i) = y_i$. Pas $p_j$ toe: $p_j(y_i) = p_j p_i (y_i) =
0$ voor $i \neq j$, dus $0 = p_j\bigl(\sum y_i\bigr) = y_j$, voor
elke $j$. Bijgevolg $E = \bigoplus_i \operatorname{im} p_i$.

\textbf{13.} $q = \mathrm{id} - p$, en vraag 1 geeft rechtstreeks
$pq = p - p^2 = 0 = qp$: voor twee projectoren dwingt het sommeren
tot de identiteit de orthogonaliteit van het paar al af.

\textbf{14.} $p + q = \mathrm{id} - r$ met $r$ een projector, en
$(\mathrm{id} - r)$ is een projector (vraag 1): dus is $p + q$ een
projector, en geeft vraag 6 dat $pq = qp = 0$. Uit symmetrie ($q +
r = \mathrm{id} - p$ en $p + r = \mathrm{id} - q$) worden alle
paarsgewijze producten nul, en besluit vraag 12: $E =
\operatorname{im} p \oplus \operatorname{im} q \oplus
\operatorname{im} r$, automatisch.

\textbf{15.} \emph{Lemma.} Met inductie en Grassmann
(\cref{thm:b1:findim:grassmann}):
\[
\dim(F_1 + \dots + F_k) \leq \dim(F_1 + \dots + F_{k-1}) + \dim
F_k \leq \dots \leq \sum_i \dim F_i .
\]
Is het totaal een gelijkheid, dan is elke stap dat: $(F_1 + \dots +
F_{j-1}) \cap F_j = \{0\}$ voor elke $j$, en een betrekking $y_1 +
\dots + y_k = 0$ ($y_i \in F_i$) stort van rechts in: $y_k \in (F_1
+ \dots + F_{k-1}) \cap F_k = \{0\}$, dan $y_{k-1} = 0$,
enzovoort: de som is direct. Omgekeerd heeft een \omterm{def:b1:vspaces:sum}{directe som}
optelbare dimensies (schakel de \omterm{def:b1:vspaces:free}{bases} aaneen). \emph{Toepassing:}
$x = \sum_i p_i(x)$ toont $E = \sum_i \operatorname{im} p_i$, dus
$n \leq \sum_i \operatorname{rk} p_i$; de hypothese geeft
gelijkheid, en dus directheid. Producten: leg $j$ vast en zij $y
\in \operatorname{im} p_j$. Dan is $y = \sum_i p_i(y)$ met $p_i(y)
\in \operatorname{im} p_i$, terwijl $y = y$ ook een ontbinding is
(component $j$ alleen); de eenduidigheid dwingt $p_i(y) = 0$ af
voor $i \neq j$. Toegepast op $y = p_j(x)$: $p_i p_j = 0$.

\textbf{16.} Is $u^k(x) = 0$, dan is $u^{k+1}(x) = u(0) = 0$. En
$\operatorname{im} u^{k+1} = u^k\bigl(u(E)\bigr) \subseteq u^k(E) =
\operatorname{im} u^k$.

\textbf{17.} Neem aan dat $\ker u^{r} = \ker u^{r+1}$ en zij $x \in
\ker u^{r+2}$: dan is $u(x) \in \ker u^{r+1} = \ker u^{r}$, dus
$u^{r+1}(x) = 0$: $x \in \ker u^{r+1}$. Met vraag 16 volgt $\ker
u^{r+1} = \ker u^{r+2}$, en met inductie vallen alle latere \omterm{def:b1:linmaps:kerim}{kernen}
samen met $\ker u^{r}$. Voor de \omterm{def:b1:linmaps:kerim}{beelden}: de dimensiestelling geeft
$\dim\operatorname{im} u^k = n - \dim\ker u^k$, dus bevriezen de
dimensies van de \omterm{def:b1:linmaps:kerim}{beelden} precies wanneer die van de \omterm{def:b1:linmaps:kerim}{kernen} dat
doen, en met de insluitingen van vraag 16 betekenen gelijke
dimensies gelijke \omterm{def:b1:vspaces:subspace}{deelruimten}
(\cref{thm:b1:findim:subspaces}).

\textbf{18.} De rij $\bigl(\dim\ker u^k\bigr)_k$ is stijgend met
waarden in $\intint{0}{n}$; zij kan niet $n + 1$ keer strikt
stijgen, dus is voor zekere $r \leq n$ $\dim\ker u^{r} = \dim\ker
u^{r+1}$, en dus $\ker u^{r} = \ker u^{r+1}$ (insluiting plus
gelijke dimensie). Neem $r$ minimaal; vraag 17 bevriest alles vanaf
$r$, de \omterm{def:b1:linmaps:kerim}{beelden} inbegrepen.

\textbf{19.} Doorsnede: zij $x \in \ker u^{r} \cap \operatorname{im}
u^{r}$, zeg $x = u^{r}(y)$ met $u^{r}(x) = 0$. Dan is $u^{2r}(y) =
0$, en $\ker u^{2r} = \ker u^{r}$ (vraag 17), dus $x = u^{r}(y) =
0$. Dimensies: de dimensiestelling voor $u^{r}$ geeft $\dim\ker
u^{r} + \dim\operatorname{im} u^{r} = n$; met de triviale doorsnede
maakt Grassmann de som tot een \omterm{def:b1:vspaces:subspace}{deelruimte} van dimensie $n$: $E =
\ker u^{r} \oplus \operatorname{im} u^{r}$.

\textbf{20.} Stabiliteit: $u^{r}(u(x)) = u(u^{r}(x)) = 0$ voor $x
\in \ker u^{r}$; en $u(u^{r}(y)) = u^{r}(u(y)) \in
\operatorname{im} u^{r}$. Op $N = \ker u^{r}$ is $(u|_N)^{r} = 0$
per definitie van $N$: nilpotent. Op $I = \operatorname{im} u^{r}$
is $\ker(u|_I) = \ker u \cap I \subseteq \ker u^{r} \cap I =
\{0\}$, dus is $u|_I$ een \omterm{def:b1:logic:inj}{injectief} endomorfisme van het
\omterm{def:b1:findim:def}{eindigdimensionale} $I$, en bijgevolg \omterm{def:b1:logic:inj}{bijectief}
(\cref{cor:b1:linmaps:samedim}).

\textbf{21.} Zij $x = a + b$ met $a \in N$ en $b \in I$. Dan is
$u(x) = u(a) + u(b)$ met $u(a) \in N$ en $u(b) \in I$ (vraag 20):
dat \emph{is} de ontbinding van $u(x)$, dus $\pi(u(x)) = u(a) =
u(\pi(x))$: $\pi u = u\pi$.

\textbf{22.} $u^2(x,y,z) = u(y, 0, z) = (0, 0, z)$ en
$u^3(x,y,z) = u(0,0,z) = (0,0,z) = u^2(x,y,z)$. \omterm{def:b1:linmaps:kerim}{Kernen}: $\ker u =
\{y = z = 0\} = \operatorname{Vect}(e_1)$, $\ker u^2 = \{z = 0\} =
\operatorname{Vect}(e_1, e_2)$, $\ker u^3 = \ker u^2$: stabilisatie
bij $r = 2$. \omterm{def:b1:linmaps:kerim}{Beelden}: $\operatorname{im} u =
\operatorname{Vect}(e_1, e_3)$, $\operatorname{im} u^2 =
\operatorname{Vect}(e_3)$. Fitting: $\R^3 =
\operatorname{Vect}(e_1, e_2) \oplus \operatorname{Vect}(e_3)$, en
$\pi(x, y, z) = (x, y, 0)$. Controle: $\pi u(x,y,z) = \pi(y, 0, z)
= (y, 0, 0)$ en $u\pi(x,y,z) = u(x, y, 0) = (y, 0, 0)$: gelijk. Op
de eerste factor is $u(x, y, 0) = (y, 0, 0)$, waarvan het kwadraat
$0$ is: nilpotent; op de tweede is $u(0,0,z) = (0,0,z)$: de
identiteit, \omterm{def:b1:logic:inj}{bijectief}.

\textbf{23.} Is $u^m = 0$, dan is $\ker u^m = E$; omdat de \omterm{def:b1:linmaps:kerim}{kernen}
vanaf $r$ bevroren zijn, is $\ker u^{r} = \ker u^{\max(m, r)} = E$.
Omgekeerd betekent $\ker u^{r} = E$ dat $u^{r} = 0$. En $\ker u^{r}
= E \iff$ de Fitting-projector projecteert op $E$ langs $\{0\}$,
dat wil zeggen $\pi = \mathrm{id}$. Ten slotte geeft $r \leq n$
(vraag 18): elk nilpotent endomorfisme voldoet aan $u^{n} = 0$ ---
de nilpotentie-index overtreft de dimensie nooit.

\textbf{24.} Zij $m$ een nilpotentie-index van $u|_A$: $A \subseteq
\ker u^{m} \subseteq \ker u^{\max(m,r)} = \ker u^{r}$. Omdat $u|_B$
\omterm{def:b1:logic:inj}{bijectief} is, is $B = u(B) = u^{k}(B) \subseteq \operatorname{im}
u^{k}$ voor elke $k$, in het bijzonder $B \subseteq
\operatorname{im} u^{r}$. Dan is
\[
n = \dim A + \dim B \leq \dim\ker u^{r} +
\dim\operatorname{im} u^{r} = n :
\]
beide insluitingen zijn gelijkheden van dimensies, en dus van
\omterm{def:b1:vspaces:subspace}{deelruimten}: $A = \ker u^{r}$ en $B = \operatorname{im} u^{r}$.

\textbf{25.} (i) Deel III is een woordenboek: splitsingen $E = F_1
\oplus \dots \oplus F_k$ corresponderen precies met families
projectoren met $\sum p_i = \mathrm{id}$ en $p_i p_j = 0$, waarbij
de $F_i$ de \omterm{def:b1:linmaps:kerim}{beelden} zijn. (ii) Voor $k = 3$ zijn de complementen
$\mathrm{id} - p_i$ zelf projectoren, wat het argument zonder extra
hypothese sloot; voor algemene $k$ heeft men $\sum_i
\operatorname{rk} p_i \leq n$ nodig, een ongelijkheid die het spoor
uit bachelorjaar 2 gratis levert ($\operatorname{tr} p =
\operatorname{rk} p$ voor een projector, en sporen tellen op tot
$\operatorname{tr} \mathrm{id} = n$). (iii)
\cref{exo:b1:linmaps:7} is het lemma van Fitting in het reeds
gestabiliseerde geval $r \leq 1$; in het algemeen laat men de
ketens van \omterm{def:b1:linmaps:kerim}{kernen} en \omterm{def:b1:linmaps:kerim}{beelden} bevriezen, wat hoogstens $n$ stappen
kost. (iv) In het volume van bachelorjaar 2 wordt, toegepast op $u
- \lambda\,\mathrm{id}$, de nilpotente factor de veralgemeende
eigenruimte bij $\lambda$, en worden de projectoren van deel III de
spectrale projectoren van de reductietheorie. De stelling van deel
IV is het \emph{lemma van Fitting}.
\end{solution}
