\section*{Hoofdstuk \ref{ch:b1:reals} --- Reële getallen}

\begin{solution}{exo:b1:reals:1}
$A$: elk element is $\leq 1$ en $1 \in A$, dus $\sup A = \max A = 1$.
\omterm{def:b1:reals:bounds}{Ondergrenzen}: $0$ begrenst naar onderen; voor $\varepsilon > 0$ levert
Archimedes een $n$ met $\frac 1n < \varepsilon$, zodat geen positief
getal $A$ naar onderen begrenst: $\inf A = 0$, niet bereikt (geen min).

$B$: de termen zijn $0, -\frac12, \frac23, -\frac34, \frac45, \dots$ De
termen met even index $\frac{n}{n+1}$ stijgen naar $1$ zonder het te
bereiken; die met oneven index $-\frac{n}{n+1}$ dalen naar $-1$. Dus is
$\sup B = 1$ en $\inf B = -1$, geen van beide bereikt: geen max, geen
min. (Grenzen: $\abs{b} < 1$ voor alle $b \in B$; en $\frac{n}{n+1} = 1 -
\frac{1}{n+1} > 1 - \varepsilon$ voor grote $n$, analoog naar onderen.)

$C = \intoo{-\sqrt 3}{\sqrt 3}$: $\sup C = \sqrt 3$ en $\inf C = -\sqrt
3$, geen van beide bereikt.
\end{solution}

\begin{solution}{exo:b1:reals:2}
Schrijf $x = \lfloor x \rfloor + u$ en $y = \lfloor y \rfloor + v$ met
$u, v \in \intco{0}{1}$. Dan is $x + y = \lfloor x \rfloor + \lfloor y
\rfloor + (u + v)$ met $u + v \in \intco{0}{2}$. Is $u + v < 1$, dan is
$\lfloor x + y\rfloor = \lfloor x\rfloor + \lfloor y \rfloor$; is $1 \leq
u + v < 2$, dan is $\lfloor x+y \rfloor = \lfloor x \rfloor + \lfloor y
\rfloor + 1$. Beide gevallen komen voor: $(x, y) = (0.2,\, 0.3)$ geeft de
linkergelijkheid, $(0.7,\, 0.8)$ de rechter.
\end{solution}

\begin{solution}{exo:b1:reals:3}
Zij $k = \lfloor x \rfloor$, zodat $k \leq x < k + 1$. Vermenigvuldigen
met $n$ geeft $nk \leq nx < nk + n$, en \omterm{thm:b1:reals:floor}{vloeren} nemen (een stijgende
bewerking aan de kant van de gehele getallen) geeft $nk \leq \lfloor nx
\rfloor \leq nk + n - 1$. Deling door $n$: $k \leq \frac{\lfloor nx
\rfloor}{n} < k + 1$, zodat de buitenste \omterm{thm:b1:reals:floor}{vloer} $k$ is.
\end{solution}

\begin{solution}{exo:b1:reals:4}
Elk element van $A$ ligt in $B$, dus begrenst $\sup B$ ook $A$ naar
boven: bijgevolg is $\sup A \leq \sup B$ ($\sup A$ is de
\emph{kleinste} \omterm{def:b1:reals:bounds}{bovengrens}). Symmetrisch is $\inf B \leq \inf A$. Ten
slotte is $\inf A \leq \sup A$ omdat $A$ niet-leeg is: elke $a \in A$
ligt ertussen.
\end{solution}

\begin{solution}{exo:b1:reals:5}
Zij $s = \sup A$ en $t = \sup B$. Elke $a + b \leq s + t$: een
\omterm{def:b1:reals:bounds}{bovengrens}. Kies voor $\varepsilon > 0$ elementen $a > s -
\frac\varepsilon2$ en $b > t - \frac\varepsilon2$
(\cref{prop:b1:reals:epsilon}): dan is $a + b > s + t - \varepsilon$.
Volgens de $\varepsilon$-karakterisering is $\sup(A+B) = s + t$.

Voor $-A$: $m$ begrenst $-A$ naar boven $\iff$ $-m$ begrenst $A$ naar
onderen; de kleinste \omterm{def:b1:reals:bounds}{bovengrens} van $-A$ hoort dus bij de grootste
ondergrens van $A$: $\sup(-A) = -\inf A$.
\end{solution}

\begin{solution}{exo:b1:reals:6}
Voor elke $x$ is $f(x) + g(x) \leq \sup f + \sup g$; het \omterm{def:b1:reals:bounds}{supremum} van het
linkerlid nemen geeft de ongelijkheid. Strikt voorbeeld: $E = \{0, 1\}$,
$f = \mathbf{1}_{\{0\}}$ (waarde $1$ in $0$, elders $0$) en $g =
\mathbf{1}_{\{1\}}$: $\sup(f + g) = 1 < 2 = \sup f + \sup g$.

Geen tegenspraak met \cref{exo:b1:reals:5}: daar variëren $a \in A$ en $b
\in B$ \emph{onafhankelijk}; hier voedt dezelfde $x$ zowel $f$ als $g$
--- de \omterm{def:b1:logic:sets}{verzameling} $\{f(x) + g(x) : x \in E\}$ is kleiner dan de
\omterm{def:b1:logic:sets}{verzameling} $\{f(x) + g(y) : x, y \in E\}$.
\end{solution}

\begin{solution}{exo:b1:reals:7}
$\sqrt 6$ is irrationaal: $6 = 2 \times 3$ is geen volkomen kwadraat, en
$v_2(6q^2) = 1 + 2v_2(q)$ is oneven, wat $6q^2 = r^2$ verhindert (als in
\cref{exo:b1:arith:7}). Stel nu $x = \sqrt 2 + \sqrt 3 \in \Q$. Dan is
$x^2 = 5 + 2\sqrt 6 \in \Q$, dus $\sqrt 6 = \frac{x^2 - 5}{2} \in \Q$:
tegenspraak. Bijgevolg is $\sqrt 2 + \sqrt 3 \notin \Q$.
\end{solution}

\begin{solution}{exo:b1:reals:8}
Zij $x < y$. Kies $n \in \N$ met $2^n > \frac{1}{y - x}$ (Archimedes:
$2^n \geq n + 1$ met een eenvoudige inductie, zodat een zekere macht van
$2$ elk reëel getal overtreft). Dan geldt, net als in het bewijs van
\cref{thm:b1:reals:density} met $2^n$ in de plaats van $n$: $m = \lfloor
2^n x \rfloor + 1$ voldoet aan $x < \frac{m}{2^n} < y$. Dus ligt $D$
dicht.
\end{solution}

\begin{solution}{exo:b1:reals:9}
\begin{enumerate}
  \item Stel $\alpha > 0$. Ten eerste is $\alpha \in G$. Stel van niet:
        volgens de $\varepsilon$-karakterisering van het \omterm{def:b1:reals:bounds}{infimum} met
        $\varepsilon = \alpha$ is er een $g \in G$ met $\alpha < g <
        2\alpha$ (links strikt, want $\alpha \notin G$); vervolgens is
        er met $\varepsilon = g - \alpha$ een $h \in G$ met $\alpha < h
        < g$. Nu is $g - h \in G$ met $0 < g - h < g - \alpha < \alpha$:
        een element van $G \cap \intoo{0}{+\infty}$ onder zijn \omterm{def:b1:reals:bounds}{infimum},
        ongerijmd. Dus $\alpha \in G$, en $\alpha\Z \subseteq G$ ($G$ is
        een \omterm{def:b1:structures:group}{groep}). Omgekeerd, zij $x \in G$ en $k = \lfloor x/\alpha
        \rfloor$: dan is $x - k\alpha \in G$ met $0 \leq x - k\alpha <
        \alpha$, en de definitie van $\alpha$ dwingt $x - k\alpha = 0$
        af. Bijgevolg is $G = \alpha\Z$.
  \item Stel $\alpha = 0$ en zij $x < y$. Er is een $g \in G$ met $0 < g
        < y - x$. Het veelvoud $kg$ met $k = \lfloor x/g \rfloor + 1$
        voldoet aan $x < kg \leq x + g < y$, en $kg \in G$: dichtheid.
\end{enumerate}
$G = \Z + \sqrt 2\,\Z$ is een \omterm{def:b1:structures:subgroup}{deelgroep} van $(\R, +)$. Ze is niet van de
vorm $\alpha\Z$: anders zouden $1 = p\alpha$ en $\sqrt 2 = q\alpha$ (met
$p, q \in \Z$) geven dat $\sqrt 2 = \frac qp \in \Q$, een tegenspraak.
Wegens de dichotomie ligt $G$ dus dicht in $\R$.
\end{solution}

\begin{solution}{exo:b1:reals:10}
Zij $s = \sup A > 0$ en $t = \sup B > 0$. Voor $a \in A$ en $b \in B$ is
$ab \leq st$ (ongelijkheden tussen \emph{positieve} getallen
vermenigvuldigen). Kies voor $0 < \varepsilon < \min(s, t)$ elementen $a
> s - \varepsilon$ en $b > t - \varepsilon$; dan is
\[
ab > (s - \varepsilon)(t - \varepsilon)
= st - \varepsilon(s + t) + \varepsilon^2
> st - \varepsilon (s + t),
\]
en $\varepsilon(s+t)$ kan willekeurig klein gemaakt worden: volgens de
$\varepsilon$-karakterisering (in de vorm: geen getal $< st$ begrenst
$AB$ naar boven) is $\sup AB = st$.

De positiviteit is essentieel: $A = B = \{-1, 0\}$ geeft $AB = \{0, 1\}$
met $\sup AB = 1$, terwijl $\sup A \cdot \sup B = 0 \times 0 = 0$.
\end{solution}

\begin{solution}{exo:b1:reals:11}
Schrijf $s = \sup A$ en $i = \inf A$. Voor $a, a' \in A$ geven $a \leq s$
en $a' \geq i$ dat $a - a' \leq s - i$; wegens de symmetrie is $\abs{a -
a'} \leq s - i$, zodat $s - i$ de \omterm{def:b1:logic:sets}{verzameling} verschillen naar boven
begrenst. Kies voor $\varepsilon > 0$ elementen $a > s -
\frac\varepsilon2$ en $a' < i + \frac\varepsilon2$
(\cref{prop:b1:reals:epsilon} en haar spiegelbeeld voor het \omterm{def:b1:reals:bounds}{infimum}): dan
is $\abs{a - a'} \geq a - a' > s - i - \varepsilon$. Volgens de
$\varepsilon$-karakterisering is $\operatorname{diam} A = s - i$.

Elke $a \in A$ voldoet aan $i \leq a \leq s$, dus $A \subseteq
\intcc{i}{s}$, een gesloten \omterm{prop:b1:reals:intervals}{interval} van lengte $\operatorname{diam} A$.
Bevat een gesloten \omterm{prop:b1:reals:intervals}{interval} $\intcc{u}{v}$ de \omterm{def:b1:logic:sets}{verzameling} $A$, dan is $v$
een \omterm{def:b1:reals:bounds}{bovengrens} en $u$ een ondergrens van $A$, dus $u \leq i$ en $v \geq
s$: $\intcc{i}{s} \subseteq \intcc{u}{v}$. Bijgevolg is $\intcc{i}{s}$
het kleinste.
\end{solution}

\begin{solution}{exo:b1:reals:12}
Er is $0 \in E$, dus $E \neq \emptyset$. Is $x > \max(1, y)$, dan is $x^2
> x > y$, zodat $E$ naar boven begrensd is door $\max(1, y)$: $s = \sup
E$ bestaat (\cref{thm:b1:reals:sup}), en $s \geq \min(1, y) > 0$ omdat
$\min(1, y) \in E$: is immers $y \geq 1$, dan is $1^2 = 1 \leq y$, en is
$y < 1$, dan is $y^2 < y$.

\emph{$s^2 < y$ is onmogelijk.} Kies $0 < h < 1$ met $h < \frac{y -
s^2}{2s + 1}$. Dan is
\[
(s + h)^2 = s^2 + 2sh + h^2 \leq s^2 + (2s + 1)h < y ,
\]
dus $s + h \in E$, in tegenspraak met het feit dat $s$ de \omterm{def:b1:logic:sets}{verzameling} $E$
naar boven begrenst.

\emph{$s^2 > y$ is onmogelijk.} Kies $0 < h < s$ met $h < \frac{s^2 -
y}{2s}$. Dan is $(s - h)^2 = s^2 - 2sh + h^2 > s^2 - 2sh > y$; elke $x
\in E$ voldoet aan $x^2 \leq y < (s - h)^2$, dus $x < s - h$ (beide zijn
$\geq 0$): $s - h$ is een \omterm{def:b1:reals:bounds}{bovengrens} van $E$ die kleiner is dan $s$, in
tegenspraak met de minimaliteit.

Bijgevolg is $s^2 = y$. Uniciteit: is $0 < s < s'$, dan is $s^2 < s'^2$,
zodat twee verschillende positieve wortels niet beide $y$ als kwadraat
kunnen hebben. Monotonie: is $0 < y < y'$, dan is $\sqrt y \neq
\sqrt{y'}$, en $\sqrt y > \sqrt{y'}$ zou na kwadrateren $y > y'$ geven:
dus $\sqrt y < \sqrt{y'}$.
\end{solution}

\begin{solution}{pb:b1:reals:1}
\textbf{1.} Voor $\frac 18$: $10 = 8 \cdot 1 + 2$, $20 = 8 \cdot 2 + 4$,
$40 = 8 \cdot 5 + 0$; cijfers $1, 2, 5$, rest $0$, daarna alleen nullen:
$\frac 18 = 0.125$. Voor $\frac 17$: $10 = 7 \cdot 1 + 3$, $30 = 7 \cdot
4 + 2$, $20 = 7 \cdot 2 + 6$, $60 = 7 \cdot 8 + 4$, $40 = 7 \cdot 5 + 5$,
$50 = 7 \cdot 7 + 1$: cijfers $1, 4, 2, 8, 5, 7$ en resten $3, 2, 6, 4,
5, 1$. De rest is naar $r = 1$ teruggekeerd, dus herhalen de zes stappen
zich letterlijk voor altijd: $\frac 17 = 0.(142857)$, met de resten in de
cyclus $1, 3, 2, 6, 4, 5$.

\textbf{2.} $\frac 13$ in grondtal $2$ ($r_0 = 1$): $2 = 3 \cdot 0 + 2$,
$4 = 3 \cdot 1 + 1$, en $r = 1$ keert terug: $\frac 13 =
(0.\overline{01})_2$. $\frac{5}{16}$ in grondtal $2$: $10 = 16 \cdot 0 +
10$, $20 = 16 \cdot 1 + 4$, $8 = 16 \cdot 0 + 8$, $16 = 16 \cdot 1 + 0$:
$\frac{5}{16} = (0.0101)_2$, afbrekend. $\frac 12$ in grondtal $3$: $3 =
2 \cdot 1 + 1$, en $r = 1$ keert meteen terug: $\frac 12 =
(0.\overline{1})_3$.

\textbf{3.} De rest van het algoritme na $N$ stappen is $r_N = b^N p
\bmod q$ (formeel bewezen in vraag 14; hier is het de vaststelling dat
elke stap de rest met $b$ vermenigvuldigt en modulo $q$ reduceert). Alle
latere cijfers zijn $0$ dan en slechts dan als een zekere $r_N = 0$, dus
$q \mid b^N p$; omdat $\gcd(p, q) = 1$, geeft het lemma van Gauss dat $q
\mid b^N$. \omterm{def:b1:arith:divides}{Deelt} $q$ het getal $b^N$, dan \omterm{def:b1:arith:divides}{deelt} elke priemfactor van $q$
het getal $b^N$ en dus $b$ (wegens de primaliteit). Omgekeerd, \omterm{def:b1:arith:divides}{deelt} elk
\omterm{def:b1:arith:prime}{priemgetal} van $q = p_1^{a_1} \cdots p_r^{a_r}$ het getal $b$, dan \omterm{def:b1:arith:divides}{deelt}
met $A = \max_i a_i$ elke $p_i^{a_i}$ het getal $b^A$, en omdat de
$p_i^{a_i}$ twee aan twee \omterm{cor:b1:arith:bezout}{relatief priem} zijn, is $q \mid b^A$. Voor $q =
20 = 2^2 \cdot 5$ delen beide \omterm{def:b1:arith:prime}{priemgetallen} $10$ ($\frac{1}{20} = 0.05$),
maar $2 \nmid 3$, zodat $\frac{1}{20}$ zich in grondtal $3$ eeuwig
herhaalt.

\textbf{4.} $1^2 = 1 < 2 < 4 = 2^2$ geeft $s_0 = 1$. Vervolgens is
$1.4^2 = 1.96 < 2 < 2.25 = 1.5^2$: $\lfloor 10\sqrt 2 \rfloor = 14$ en
$s_1 = 1.4$. Dan $1.41^2 = 1.9881 < 2 < 2.0164 = 1.42^2$: $s_2 = 1.41$;
$1.414^2 = 1.999396 < 2 < 2.002225 = 1.415^2$: $s_3 = 1.414$; en
$1.4142^2 = 1.99996164 < 2 < 2.00024449 = 1.4143^2$: $s_4 = 1.4142$. In
elk geval zeggen de getoonde ongelijkheden precies $s_n \leq \sqrt 2 <
s_n + 10^{-n}$, en dat is de definitie van de \omterm{thm:b1:reals:floor}{vloer} van $10^n \sqrt 2$.

\textbf{5.} $A_0 = \lfloor x \rfloor = 0$ omdat $0 \leq x < 1$.
Vermenigvuldig $A_{n-1} \leq b^{n-1} x < A_{n-1} + 1$ met $b$:
\[
b\,A_{n-1} \leq b^n x < b\,A_{n-1} + b .
\]
Het gehele getal $b\,A_{n-1}$ is $\leq b^n x$, dus $b\,A_{n-1} \leq A_n$;
en $b^n x < b\,A_{n-1} + b$ met $b\,A_{n-1} + b$ geheel dwingt $A_n \leq
b\,A_{n-1} + b - 1$ af. Bijgevolg is $0 \leq d_n = A_n - b\,A_{n-1} \leq
b - 1$: een cijfer.

\textbf{6.} Telescoperen: $d_k b^{-k} = A_k b^{-k} - A_{k-1}
b^{-(k-1)}$, dus
\[
\sum_{k=1}^n d_k b^{-k} = A_n b^{-n} - A_0 = s_n .
\]
Deling van $A_n \leq b^n x < A_n + 1$ door $b^n$ geeft $s_n \leq x < s_n
+ b^{-n}$.

\textbf{7.} Inductie: $b^0 = 1 \geq 1$, en $b^{n+1} = b \cdot b^n \geq
2(n + 1) \geq n + 2$. Monotonie: $s_n - s_{n-1} = d_n b^{-n} \geq 0$.
Elke $s_n \leq x$ (vraag 6): $x$ is een \omterm{def:b1:reals:bounds}{bovengrens} van $\{s_n\}$. Voor
$\varepsilon > 0$ levert de \omterm{thm:b1:reals:archimedes}{archimedische eigenschap} een $n$ met $n + 1 >
\frac1\varepsilon$, dus $b^{-n} < \varepsilon$, en dan is $s_n > x -
b^{-n} > x - \varepsilon$ wegens vraag 6. Volgens
\cref{prop:b1:reals:epsilon} is $x = \sup_n s_n$.

\textbf{8.} Stel $d_k = b - 1$ voor alle $k > N$. Voor $n > N$ geeft de
eindige meetkundige som
\[
s_n = s_N + (b - 1)\sum_{k=N+1}^{n} b^{-k}
= s_N + b^{-N} - b^{-n} .
\]
Dus is $x \geq s_n = s_N + b^{-N} - b^{-n}$ voor elke $n$; laat je de
laatste term onder elke $\varepsilon$ zakken (vraag 7), dan is $x \geq
s_N + b^{-N}$. Maar vraag 6 op rang $N$ zegt $x < s_N + b^{-N}$:
tegenspraak. De rij $(d_n)$ is eigenlijk.

\textbf{9.} Begrensd: $t_n \leq (b-1)\sum_{k=1}^n b^{-k} = 1 - b^{-n} <
1$, en $(t_n)$ is niet-dalend, dus bestaat $y = \sup t_n$ met $0 \leq y
\leq 1$. Leg $n$ vast. Voor de tweezijdige afschatting: $t_n \leq y$ is
duidelijk. Wegens de eigenlijkheid kies je een $m > n$ met $e_m \leq b -
2$. Voor $p \geq m$ is
\[
t_p - t_n = \sum_{k=n+1}^{p} e_k b^{-k}
\leq (b^{-n} - b^{-p}) - b^{-m} < b^{-n} - b^{-m},
\]
waarbij de middelste som minstens $b^{-m}$ inboet tegenover het maximum
met louter $(b-1)$'s; voor $p \leq m$ is eveneens $t_p \leq t_m \leq t_n
+ b^{-n} - b^{-m}$ (monotonie plus het geval $p = m$). Bijgevolg is elke
$t_p \leq t_n + b^{-n} - b^{-m}$, zodat $y \leq t_n + b^{-n} - b^{-m} <
t_n + b^{-n}$. (Met $n = 0$: $y < 1$, dus $y \in \intco{0}{1}$.) Nu is
$b^n t_n = \sum_{k \leq n} e_k b^{n-k}$ geheel, en $b^n t_n \leq b^n y <
b^n t_n + 1$: dus $\lfloor b^n y \rfloor = b^n t_n$. Ten slotte de
cijfers van $y$: $d_n(y) = b^n t_n - b \cdot b^{n-1} t_{n-1} = b^n(t_n -
t_{n-1}) = e_n$.

\textbf{10.} Vraag 9 zegt: (de waarde van een rij) heeft als cijfers (die
rij); de vragen 5--8 zeggen: (de cijfers van $x$) vormen een eigenlijke
rij waarvan de afknottingen $x$ als \omterm{def:b1:reals:bounds}{supremum} hebben (vraag 7). De twee
\omterm{def:b1:logic:map}{afbeeldingen} zijn in beide volgordes dus de identiteit: het zijn elkaars
inverse bijecties tussen $\intco{0}{1}$ en de eigenlijke rijen. Hadden
twee eigenlijke rijen dezelfde waarde, dan zou de cijferafbeelding erop
toepassen ze gelijk maken: uniciteit. Dat is de \emph{stelling over de
$b$-adische ontwikkeling}.

\textbf{11.} Zij $e_n = b - 1$ voor $n > M$, met $M \geq 0$ minimaal. Als
in vraag 8 is $t_n = t_M + b^{-M} - b^{-n}$ voor $n \geq M$, zodat de
waarde $\sup t_n = t_M + b^{-M}$ is. Is $M = 0$, dan is de waarde $0 + 1
= 1$: in grondtal $10$ is $0.999\dots = 1$ exact --- niet bij benadering.
Is $M \geq 1$, dan geeft de minimaliteit $e_M \leq b - 2$, en is de
waarde
\[
t_M + b^{-M} = \frac{b^M t_M + 1}{b^M} \in \intoo{0}{1},
\]
een $b$-adische breuk, waarvan de \emph{eigenlijke} ontwikkeling $e_1
\dots e_{M-1}\,(e_M + 1)\,000\dots$ is (een afbrekende rij is eigenlijk,
en haar waarde is hetzelfde getal). Omgekeerd moet een reëel getal met
twee voorstellingen er één oneigenlijke hebben (de eigenlijkheid legt de
voorstelling vast, vraag 10) en dus van deze vorm zijn. En elke $m/b^N
\in \intoo{0}{1}$, geschreven met laatste cijfer $d_N \neq 0$, heeft
inderdaad de oneigenlijke tweelingvorm $d_1 \dots d_{N-1}(d_N -
1)(b-1)(b-1)\dots$: precies de $b$-adische breuken dragen twee namen,
alle andere reële getallen één.

\textbf{12.} Stel dat de rijen tot en met $m - 1$ overeenstemmen, met
gemeenschappelijke afknotting $P = s_{m-1}$, en dat $d_m < e_m$. Volgens
vraag 9 (de strikte bovenafschatting op rang $m$) is $x < P + d_m b^{-m}
+ b^{-m} = P + (d_m + 1)b^{-m} \leq P + e_m b^{-m} \leq y$, waarbij de
laatste stap geldt omdat $P + e_m b^{-m}$ de afknotting $t_m \leq y$ van
$y$ is. Dus $d_m < e_m \implies x < y$; met omgekeerde rollen $e_m < d_m
\implies y < x$; en omdat de rijen op $m$ verschillen, geldt een van
beide. Beide richtingen volgen.

\textbf{13.} Zij $z = bx - A_1 \in \intco{0}{1}$ (immers $A_1 \leq bx <
A_1 + 1$). Voor $n \geq 0$ is $b^n z = b^{n+1} x - b^n A_1$ met $b^n A_1
\in \Z$, dus met $\lfloor u - K \rfloor = \lfloor u \rfloor - K$ ($K$
geheel):
\[
A_n(z) = A_{n+1}(x) - b^n A_1(x) .
\]
Bijgevolg is $d_n(z) = A_n(z) - b\,A_{n-1}(z) = A_{n+1} - b^n A_1 -
b\,A_n + b^n A_1 = d_{n+1}(x)$. Het decimale deel van $bx$ draagt dus de
verschoven cijfers; na $m$ iteraties heeft het decimale deel van $b^m x$
de cijfers $(d_{n+m})_{n \geq 1}$.

\textbf{14.} Euclidische deling: $b^n p = q\,Q_n + r_n$ met $0 \leq r_n <
q$. Deling door $q$ geeft $b^n x = Q_n + \frac{r_n}{q}$ met $0 \leq
\frac{r_n}{q} < 1$, dus $Q_n = \lfloor b^n x \rfloor = A_n$, waaruit $A_n
= \frac{b^n p - r_n}{q}$. Voor de recursie: $b^n p = b(q\,A_{n-1} +
r_{n-1}) = q\,(b\,A_{n-1}) + b\,r_{n-1}$, zodat $b^n p$ en $b\,r_{n-1}$
een veelvoud van $q$ verschillen: $r_n = (b\,r_{n-1}) \bmod q$.

\textbf{15.} Deel $b\,r_{n-1}$ door $q$: $b\,r_{n-1} = q\,c + r_n$ met $c
= \lfloor b\,r_{n-1}/q \rfloor$. Substitutie in de formule van vraag 14
geeft $b^n p = q(b\,A_{n-1} + c) + r_n$, en de uniciteit van de
euclidische deling identificeert $A_n = b\,A_{n-1} + c$, dat wil zeggen
$d_n = c = \lfloor b\,r_{n-1}/q \rfloor$. Cijfer $n$ hangt alleen van
$r_{n-1}$ af --- de lus van de staartdeling uit Deel I, nu gewaarborgd.

\textbf{16.} De $q + 1$ resten $r_0, \dots, r_q$ nemen waarden in de
\omterm{def:b1:logic:sets}{verzameling} $\intint{0}{q-1}$ met $q$ elementen: volgens het
duivenhokprincipe (\cref{cor:b1:counting:pigeonhole}) vallen er twee
samen, zeg $r_N = r_{N+T}$ met $0 \leq N < N + T \leq q$. Omdat $r_n$ de
rest $r_{n+1}$ vastlegt (vraag 14), geeft inductie $r_{n+T} = r_n$ voor
alle $n \geq N$; en omdat $r_{n-1}$ het cijfer $d_n$ vastlegt (vraag 15),
is $d_{n+T} = d_n$ voor alle $n \geq N + 1$. De ontwikkeling van elk
rationaal getal is dus uiteindelijk periodiek, met voorperiode $\leq q$
en periode $\leq q$.

\textbf{17.} De cijfers van het decimale deel van $b^T y$ zijn
$(d_{n+T}) = (d_n)$ (verschuivingslemma, dan de zuivere periodiciteit):
dezelfde eigenlijke rij als die van $y$. Volgens vraag 10 zijn de waarden
gelijk: $b^T y - \lfloor b^T y \rfloor = y$, dus $(b^T - 1)\,y = \lfloor
b^T y \rfloor = A_T \in \N$ en
\[
y = \frac{A_T}{b^T - 1} ,
\]
rationaal met een noemer die $b^T - 1$ \omterm{def:b1:arith:divides}{deelt}; de teller $A_T$ is het
gehele getal met cijfers $d_1 \dots d_T$ in grondtal $b$. Controle:
$0.(142857) = \frac{142857}{999999}$, en $142857 \times 7 = 999999$, dus
dat is $\frac 17$.

\textbf{18.} Is $d_{n+T} = d_n$ voor $n > N$, dan heeft het decimale deel
$z$ van $b^N x$ de cijfers $(d_{N+n})_{n\geq1}$ (verschuivingslemma), die
zuiver periodiek zijn; volgens vraag 17 is $z \in \Q$. Uit $b^N x = A_N +
z$ volgt dan $x = (A_N + z)/b^N \in \Q$. Samen met vraag 16: $x$
rationaal $\iff$ de ontwikkeling uiteindelijk periodiek. Het rechterlid
noemt het grondtal, het linkerlid niet: periodiciteit in één grondtal is
equivalent met rationaliteit, en dus met periodiciteit in elk grondtal.

\textbf{19.} Voor $x = \frac 1q$ is $r_n = b^n \bmod q$. Is $\gcd(b, q) =
1$, dan is $r_T = r_0 = 1$ dan en slechts dan als $b^T \equiv 1 \pmod q$;
zo'n $T$ bestaat (het duivenhokprincipe geeft $b^i \equiv b^j$ met $i <
j$, en $b$ is inverteerbaar modulo $q$, dus $b^{j-i} \equiv 1$), en de
kleinste --- de multiplicatieve orde --- maakt de resten, en dus de
cijfers, zuiver periodiek met periode $T$. Een kleinere periode is
onmogelijk: een periode $T'$ zou $(b^{T'} - 1)\frac1q \in \N$ geven
(vraag 17), dus $q \mid b^{T'} - 1$. Voor $q = 7$ en $b = 10$: $10 \equiv
3$, $10^2 \equiv 2$, $10^3 \equiv 6$, $10^4 \equiv 4$, $10^5 \equiv 5$,
$10^6 \equiv 1 \pmod 7$: orde $6$, en inderdaad heeft $\frac 17$ periode
zes.

\textbf{20.} De rij bevat oneindig veel nullen (de cijfers van $x^*$ zijn
grotendeels nul), dus is ze eigenlijk en is $x^*$ welgedefinieerd (vraag
9). Stel dat de cijfers voorbij $N$ periodiek zijn met periode $T$. Er
zijn oneindig veel cijfers gelijk aan $1$ (één per driehoeksgetal), dus
staat er een $1$ op een positie $j > N$; de periodiciteit zet dan een $1$
op elke positie $j + kT$: vanaf $j$ zijn de gaten tussen opeenvolgende
enen hoogstens $T$. Maar de enen staan precies op de driehoeksgetallen,
waarvan de opeenvolgende gaten $\frac{(j+1)(j+2)}{2} - \frac{j(j+1)}{2} =
j + 1$ uiteindelijk $T$ overtreffen: tegenspraak. Niet uiteindelijk
periodiek, dus volgens vraag 18 is $x^* \notin \Q$ --- irrationaliteit,
enkel afgelezen aan het ritme van de cijfers.

\textbf{21.} Bij gegeven $x < y$ levert vraag 7 een $n$ met $b^{-n} < y -
x$; zet $m = \lfloor b^n x \rfloor + 1$. Dan is $b^n x < m \leq b^n x + 1
< b^n y$, dus $x < \frac{m}{b^n} < y$: dichtheid, voor elk grondtal
tegelijk ($b = 2$ geeft \cref{exo:b1:reals:8} terug). Voor $\frac pq \in
\intoo{0}{1}$ in grondtal $b = q$: het eerste cijfer is $\lfloor q \cdot
\frac pq \rfloor = p$, en het decimale deel van $q \cdot \frac pq = p$ is
$0$: alle latere cijfers verdwijnen, een afbrekende ontwikkeling $\frac
pq = (0.p)_q$. Afbreken hangt van het grondtal af; periodiciteit ---
rationaliteit --- niet (vraag 18).

\textbf{22.} Elke $e_k \in \{1, 2\}$ is een cijfer in grondtal $10$, en
de rij eindigt nooit op louter negens: eigenlijk. Haar waarde $y$ ligt in
$\intco{0}{1}$ en heeft precies de cijfers $(e_k)$ (vraag 9). Leg $k$
vast: het $k$-de cijfer van $y$ is $e_k$, gekozen $\neq$ het $k$-de
cijfer van $x_k$, zodat de eigenlijke rijen van $y$ en $x_k$ verschillen
en dus $y \neq x_k$ (vraag 10: de codering is \omterm{def:b1:logic:inj}{injectief}). Bijgevolg staat
$y$ op geen enkele lijst: geen enkele \omterm{def:b1:logic:map}{afbeelding} $\N^* \to \intco{0}{1}$
is \omterm{def:b1:logic:inj}{surjectief}. De reële getallen laten zich, anders dan de rationale,
niet opsommen --- de overaftelbaarheid, waarvan \cref{ch:b1:topology} de
theorie ontwikkelt.

\textbf{23.} Stemmen de ontwikkelingen tot en met $n$ overeen, dan hebben
$x$ en $y$ dezelfde afknotting $s_n$, en vraag 6 legt beide in
$\intco{s_n}{s_n + b^{-n}}$, een \omterm{prop:b1:reals:intervals}{interval} van lengte $b^{-n}$: $\abs{x -
y} < b^{-n}$. Omkering: $x = 0.1$ en $y = 0.0999$ (afbrekend, dus
eigenlijk) voldoen aan $\abs{x - y} = 10^{-4} < 10^{-3}$, en toch
verschillen hun ontwikkelingen al bij het allereerste cijfer. De
schuldigen zijn de $b$-adische breuken van vraag 11: in hun buurt keert
een minieme verschuiving elk getoond cijfer om ($0.0999 \to 0.1000$),
juist omdat zij de reële getallen zijn waar de oneigenlijke tweeling op
de loer ligt.

\textbf{24.} $p = 1$, $q = 10$, $b = 2$, $r_0 = 1$: $2 = 10 \cdot 0 + 2$,
$4 = 10 \cdot 0 + 4$, $8 = 10 \cdot 0 + 8$, $16 = 10 \cdot 1 + 6$, $12 =
10 \cdot 1 + 2$ --- en $r_5 = 2 = r_1$: de resten doorlopen vanaf index
$1$ de cyclus $(2, 4, 8, 6)$. Cijfers: $d_1 = 0$, daarna het herhalende
blok $d_2 d_3 d_4 d_5 = 0, 0, 1, 1$:
\[
\tfrac{1}{10} = (0.0\overline{0011})_2 ,
\]
met voorperiode $1$ en periode $4$. Volgens vraag 3 zou een afbrekende
ontwikkeling in grondtal $2$ vergen dat elk \omterm{def:b1:arith:prime}{priemgetal} van $10$ het getal
$2$ \omterm{def:b1:arith:divides}{deelt}; het \omterm{def:b1:arith:prime}{priemgetal} $5$ weigert. Dus is $0.1$ door \emph{geen}
enkele eindige binaire rij voor te stellen --- een computer die eindig
veel bits bewaart houdt slechts een afknotting over, en de opgestapelde
afknottingsfouten zijn de reden waarom $0.1 + 0.2$ in de
drijvendekommarekening in de laatste bits van $0.3$ verschilt.

\textbf{25.} (i) De volledigheid bracht de waarden voort: $x = \sup s_n$
en $y = \sup t_n$ (vragen 7 en 9) --- binnen $\Q$ alleen zou de
eigenlijke rij van $\sqrt 2$ niets benoemen. (ii) De \omterm{thm:b1:reals:archimedes}{archimedische
eigenschap} maakte $b^{-n}$ uiteindelijk kleiner dan elke $\varepsilon$ en
dwong de afknottingen zo naar hun \omterm{def:b1:reals:bounds}{supremum} toe (vragen 7 en 21). (iii) De
uniciteitsclausule van de \omterm{thm:b1:reals:floor}{vloer} identificeerde $Q_n = A_n$ in vraag 14 en
wettigde elke cijferextractie $\lfloor u - K \rfloor = \lfloor u \rfloor
- K$ (vraag 13). (iv) Het duivenhokprincipe, toegepast op eindig veel
resten, is de enige motor van de periodiciteit (vraag 16). Moraal:
eigenlijke rijen coderen $\intco{0}{1}$ getrouw en maken van rationaliteit
een zichtbaar ritme; maar het optellen van cijferrijen vergt overdrachten
die van oneindig ver rechts komen, zodat geen enkele regel in eindig veel
stappen zelfs het eerste cijfer van een som berekent --- terwijl de
supremuminterface van \cref{thm:b1:reals:sup} de hele analyse met één
axioma bedient. Cijfers zijn een schitterend \emph{beeld} van $\R$; het
\omterm{def:b1:reals:bounds}{supremum} is haar \emph{motor}.
\end{solution}
