\begin{solution}{exo:b1:euclid:6}

\[
\Bigl(\int_0^1 f \cdot 1\Bigr)^2 \leq \int_0^1 f^2 \cdot \int_0^1
1^2 = \int_0^1 f^2 ,

\begin{solution}{exo:b1:euclid:7}
حسب \cref{thm:b1:euclid:projection}، $E = F \oplus F^\perp$، ومنه
تُجمع الأبعاد: $\dim F^\perp = \dim E - \dim F$. والاحتواء $F

\omterm{def:b1:euclid:orthogonal}{متعامدة} مع كل شيء في $F^\perp$). والأبعاد:
$\dim(F^\perp)^\perp = \dim E - \dim F^\perp = \dim F$؛ والاحتواء
بأبعاد (منتهية) متساوية مساواةٌ
(\cref{thm:b1:findim:subspaces}).
\end{solution}

\begin{solution}{exo:b1:euclid:10}
ليكن $y \in \ker(u - \mathrm{id})$ (أي $u(y) = y$) وليكن $x \in E$.
عندئذ، باستعمال حفظ \omterm{def:b1:euclid:def}{الجداء السلّمي} ($\langle u(a),
u(b)\rangle = \langle a, b\rangle$):
\[
\langle x - u(x),\, y\rangle
= \langle x, y\rangle - \langle u(x), y\rangle
= \langle x, y\rangle - \langle u(x), u(y)\rangle
= \langle x, y\rangle - \langle x, y\rangle = 0 :
\]
ومنه فكل متجهة من $\operatorname{im}(u - \mathrm{id})$ \omterm{def:b1:euclid:orthogonal}{متعامدة}
مع كل متجهة صامدة.

ومنه $\operatorname{im}(u - \mathrm{id}) \subseteq \ker(u -
\mathrm{id})^{\perp}$، ومنه، بمبرهنة الرتبة مع
\cref{exo:b1:euclid:7}، يكون لكليهما البُعد $\dim E - \dim\ker(u -
\mathrm{id})$: فهما متساويان. وعندئذ $E = \ker(u - \mathrm{id})
\oplus \ker(u - \mathrm{id})^\perp = \ker(u - \mathrm{id}) \oplus
\operatorname{im}(u - \mathrm{id})$.
\end{solution}

\begin{solution}{pb:b1:euclid:1}
\textbf{1.} $\norm{t_a(x) - t_a(y)} = \norm{x - y}$؛ \omterm{def:b1:euclid:isometry}{والتقايس}
الخطي يحفظ المعايير، ومنه المسافات ($\norm{g(x) - g(y)} =
\norm{g(x - y)} = \norm{x-y}$)؛ وتركيب
التطبيقات الحافظة للمسافات يحفظ المسافات. وكل \omterm{def:b1:euclid:isometry}{تقايس}
متباين (لأن النقاط المتمايزة تبقى على مسافة موجبة) و، حسب
التصنيف أدناه، تقابلي؛ \omterm{def:b1:logic:map}{والتطبيق} المطابق والمعكوسات
تقايسات: أي \omterm{def:b1:structures:group}{زمرة}.

\textbf{2.} $\norm{f(x)} = \norm{f(x) - f(0)} = \norm{x - 0} =
\norm x$. وبالاستقطاب:
\[
\langle f(x), f(y)\rangle
= \frac{\norm{f(x)}^2 + \norm{f(y)}^2 - \norm{f(x) -
f(y)}^2}{2}
= \frac{\norm x^2 + \norm y^2 - \norm{x - y}^2}{2}
= \langle x, y\rangle .
\]

\textbf{3.} بالنشر مع السؤال 2 (فكل جداء لصور $f$
يساوي جداء المتغيرين):
\begin{align*}
\norm{f(x+y) - f(x) - f(y)}^2
&= \norm{x+y}^2 + \norm x^2 + \norm y^2\\
&\quad - 2\langle x+y, x\rangle - 2\langle x+y, y\rangle
+ 2\langle x, y\rangle = 0 ,
\end{align*}
كما يبيّن تحقّق مباشر؛ وكذلك $\norm{f(\lambda x) -
\lambda f(x)}^2 = \norm{\lambda x}^2 - 2\lambda\langle \lambda
x, x\rangle + \lambda^2\norm x^2 = 0$. ومنه $f(x + y) = f(x) +
f(y)$ و $f(\lambda x) = \lambda f(x)$: أي إن $f$ خطي، و
حافظ للمعيار: ومنه $f \in O(\R^2)$.

\textbf{4.} ضع $a = f(0)$ و $g = t_{-a}\circ f$: فهو \omterm{def:b1:euclid:isometry}{تقايس}
يثبّت $0$، ومنه \omterm{def:b1:euclid:isometry}{تقايس} خطي (السؤال 3)، و $f = t_a
\circ g$. والوحدانية: يعطي $t_a \circ g = t_{a'}\circ g'$ مقوَّمًا
عند $0$ أن $a = a'$، ثم $g = g'$.

\textbf{5.} $(t_a\circ g)\circ(t_{a'}\circ g') = t_{a +
g(a')}\circ(g\circ g')$، لأن $g\,t_{a'} = t_{g(a')}\,g$ من أجل
$g$ خطي. ومنه فالجزء الخطي لتركيب هو $g \circ g'$،
و $\det(gg') = \det g \det g'$: أي مباشر$\circ$مباشر $=$
غير مباشر$\circ$غير مباشر $=$ مباشر، و
مباشر$\circ$غير مباشر $=$ غير مباشر --- وهي قاعدة إشارة $\pm1$.

\textbf{6.} يرسل $z \mapsto \eu^{\iu\theta}z$ المقدارَ $1$ إلى
$(\cos\theta, \sin\theta)$ و $\iu$ إلى
$\iu\eu^{\iu\theta} = (-\sin\theta, \cos\theta)$: أي أعمدة
$R_\theta$. ويرسل $z \mapsto \eu^{\iu\theta}\conj z$ المقدارَ $1$ إلى
$(\cos\theta, \sin\theta)$ و $\iu$ إلى $-\iu\eu^{\iu\theta} =
(\sin\theta, -\cos\theta)$: أي أعمدة $S_\theta$.

\textbf{7.} $a = 1$: $f = t_b$. و $a \neq 1$: لمعادلة
النقطة الصامدة $az_0 + b = z_0$ الحلُّ الوحيد $z_0 = b/(1 -
a)$، وعندئذ
\[
f(z) - z_0 = az + b - (az_0 + b) = a(z - z_0):
\]
وفي المعلم المتمركز عند $z_0$، يكون $f$ الضربَ في $a
= \eu^{\iu\arg a}$: أي الدوران بمركز $z_0$ وزاوية
$\arg a$.

\textbf{8.} $f(f(z)) = a\conj{(a\conj z + b)} + b = a\conj a\,z
+ a\conj b + b = z + v$: أي $f^2 = t_v$. والتبادل: $f(z + v) =
a\conj z + a\conj v + b$ و $a\conj v = a\conj{(a\conj b + b)}
= b + a\conj b = v$، ومنه $f\circ t_v = t_v\circ f$.

\emph{الحالة $v = 0$}: $f^2 = \mathrm{id}$. ومن أجل أيّ $z$، يحقق
المنتصف $m = \frac{z + f(z)}2$ (ولأن التقايسات
أفينية، الجزء 1) $f(m) = \frac{f(z) + f^2(z)}{2} = m$: ومنه توجد
نقاط صامدة. وبالاقتران بالانسحاب إلى نقطة صامدة،
يصير $f$ \omterm{def:b1:euclid:isometry}{تقايسًا} \omterm{def:b1:linmaps:def}{خطيًا} غير مباشر، أي $S_\theta$ ما
(\cref{thm:b1:euclid:plane}): أي انعكاس في مستقيم.

\emph{والحالة $v \neq 0$}: يكون $r = t_{-v/2}\circ f$ غير مباشر و
\[
r^2 = t_{-v/2}\,f\,t_{-v/2}\,f = t_{-v/2}\,t_{-v/2}\,f^2
= t_{-v}\,t_v = \mathrm{id}
\]
(باستعمال التبادل)، ومنه \omterm{def:b1:logic:map}{فالتطبيق} $r$ انعكاس، و $f =
t_{v/2}\circ r$. ومحوره موازٍ للمقدار $v$: لأن $r$ يتبادل مع
$t_v$ (فكلٌّ من $f$ و $t_{v/2}$ يفعل)، ومنه يرسل $t_v$ المحورَ (وهو
المستقيم الصامد للمقدار $r$) إلى نفسه، فيُفرض على $v$ أن يوجّهه. ومنه
\omterm{def:b1:logic:map}{فالتطبيق} $f$ \omterm{pb:b1:euclid:1}{انعكاس منزلق} بمتجهة انزلاق $v/2$. وقد صُنّف كل
\omterm{def:b1:euclid:isometry}{تقايس} الآن: انسحاب أو دوران (مباشر)،
أو انعكاس أو \omterm{pb:b1:euclid:1}{انعكاس منزلق} (غير مباشر).

\textbf{9.} الدورانان $f(z) = az + b$ و $g(z) = a'z + b'$ مع
$a = \eu^{\iu\alpha}$ و $a' = \eu^{\iu\beta}$: فللتركيب $g\circ f(z) =
a'az + (a'b + b')$ معاملٌ خطي
$\eu^{\iu(\alpha+\beta)}$: أي دوران بالزاوية $\alpha + \beta$
حسب السؤال 7، أو انسحاب عندما $\eu^{\iu(\alpha+\beta)} =
1$. والانعكاسان $r_i(z) = a_i\conj z + b_i$ مع $a_i =
\eu^{\iu\theta_i}$ (بمحور بالزاوية $\theta_i/2$):
\[
r_2\circ r_1(z) = a_2\conj{a_1}\,z + (a_2\conj{b_1} + b_2),
\]
وهو مباشر بالزاوية $\theta_2 - \theta_1 = 2\bigl(\tfrac{\theta_2}2
- \tfrac{\theta_1}2\bigr)$: أي ضِعف الزاوية بين المحورين؛
والمحوران المتوازيان ($\theta_1 = \theta_2$) يعطيان انسحابًا.

\textbf{10.} الدوران بمركز $c$ وزاوية $\theta$ هو
جداء انعكاسين في مستقيمين مارّين بالمقدار $c$ يصنعان الزاوية
$\theta/2$ (السؤال 9، مقروءًا بالمقلوب)؛ والانسحاب $t_v$ هو
جداء انعكاسين في مستقيمين متوازيين عموديين على
$v$ على بعد $\norm v/2$؛ والانعكاس انعكاسٌ واحد؛ و
\omterm{pb:b1:euclid:1}{الانعكاس المنزلق} انعكاسٌ مركَّب مع انسحاب،
ومنه ثلاثة. والحدّ الأقصى: ثلاثة.

\textbf{11.} $a = \iu$ و $b = 1 + \iu$: $v = a\conj b + b =
\iu(1 - \iu) + 1 + \iu = (1 + \iu) + (1 + \iu) = 2 + 2\iu \neq
0$: أي \omterm{pb:b1:euclid:1}{انعكاس منزلق} بمتجهة انزلاق $v/2 = 1 + \iu$. و
الانعكاس $r = t_{-(1+\iu)}\circ f$ هو $r(z) = \iu\conj z$،
ومحوره المستقيم بالزاوية $\frac12\arg\iu = \frac\pi4$:
أي المستقيم $y = x$. ومنه \omterm{def:b1:logic:map}{فالتطبيق} $f$ \omterm{pb:b1:euclid:1}{انعكاس منزلق} محوره $y =
x$ ومتجهته $(1, 1)$.

\textbf{12.} $f(z) = \iu z$ و $g(z) = 1 + \iu(z - 1) = \iu z
+ 1 - \iu$، ومنه
\[
g\circ f(z) = \iu(\iu z) + 1 - \iu = -z + 1 - \iu :
\]
فالمعامل الخطي $-1 = \eu^{\iu\pi}$، أي دوران بالزاوية
$\pi$ (نصف دورة)، ومركزه $z_0 = \frac{1 - \iu}{1 - (-1)}
= \frac{1 - \iu}{2}$.

\textbf{13.} $r_2\circ r_1(z) = \iu\,\conj{\conj z} = \iu z$:
أي الدوران بالزاوية $\frac\pi2$ حول $0$. ويتلاقى المحوران (المحور
الحقيقي بالزاوية $0$؛ والمستقيم $y = x$ بالزاوية $\frac\pi4$) بالزاوية
$\frac\pi4$، وضِعفها $\frac\pi2$: ومنه يتأكد السؤال 9.

\textbf{14.} مع $f = t_a\circ g$ و $g$ خطي و $\sum_i
\lambda_i = 1$:
\[
f\Bigl(\sum_i \lambda_i x_i\Bigr)
= a + \sum_i \lambda_i\,g(x_i)
= \sum_i \lambda_i\bigl(a + g(x_i)\bigr)
= \sum_i \lambda_i f(x_i) .
\]

\textbf{15.} من أجل $h \in G$، باستعمال السؤال 14 (فالمعاملات
$\frac1{\abs G}$ مجموعها $1$):
\[
h(c) = \frac{1}{\abs G}\sum_{g \in G} h\bigl(g(x_0)\bigr)
= \frac{1}{\abs G}\sum_{g' \in G} g'(x_0) = c ,
\]
لأن $g \mapsto hg$ تقابلٌ من $G$ على نفسها.

\textbf{16.} تكوّن المقترنات $t_{-c}\,g\,t_{c}$ ($g \in G$)
زمرةً من التقايسات التي تثبّت $0$، ومنه من التقايسات
\emph{الخطية} (السؤال 3): أي \omterm{def:b1:structures:subgroup}{زمرة جزئية} منتهية من $O(\R^2)$. وإذا كان
$s \in G$ ما غير مباشر، فإن \omterm{def:b1:logic:map}{التطبيق} $g \mapsto sg$ يرسل $G^+$
إرسالًا \omterm{def:b1:logic:inj}{متباينًا} داخل $G\setminus G^+$ ويكون $h \mapsto s^{-1}h$
معكوسه (بقاعدة الإشارة في السؤال 5): ومنه $\abs G = 2\,\abs{G^+}$؛
وإلا فإن $G = G^+$.

\textbf{17.} إذا كان $G^{+} = \{\mathrm{id}\}$، فهي $C_1$.
وإلا فاكتب عناصرها $R_\theta$ مع $\theta \in
\intco{0}{2\pi}$، ولتكن $\theta_0$ أصغر زاوية
موجبة تظهر. ومن أجل $R_\theta \in G^{+}$، تعطي القسمة الإقليدية
$\theta = k\theta_0 + \rho$ مع $0 \leq \rho < \theta_0$ أن
$R_\rho = R_\theta R_{\theta_0}^{-k} \in G^{+}$، ومنه $\rho = 0$
بالأصغرية: ومنه $G^{+} = \langle R_{\theta_0}\rangle$. وبقسمة
$2\pi$ على $\theta_0$ بالكيفية نفسها يتبيّن $2\pi = n\theta_0$:
أي $G^{+} \cong C_n$، مولَّدة بالدوران بالزاوية
$\frac{2\pi}n$.

\textbf{18.} $G = G^{+} \cup sG^{+}$ بالعدّ في
السؤال 16. ومع $s(z) = a\conj z$ و $r(z) = \omega z$
(في الصورتين الخطيتين، بعد السؤال 16؛ $\abs a = \abs\omega = 1$):
\[
s r s(z) = a\,\conj{\omega\,a\conj z}
= a\conj\omega\conj a\,z = \conj\omega\,z = r^{-1}(z) .
\]
وكل عنصر $sr^k$ من $sG^{+}$ غير مباشر ويحقق
$(sr^k)^2 = (sr^ks)\,r^k = r^{-k}r^k = \mathrm{id}$: أي \omterm{def:b1:euclid:isometry}{تقايس} غير مباشر
تقابضي يثبّت $c$، أي انعكاس. ومنه
تتألف $G$ من $n$ دورانًا و $n$ انعكاسًا بالعلاقات
$r^n = s^2 = \mathrm{id}$ و $srs = r^{-1}$: أي \omterm{pb:b1:euclid:1}{الزمرة
ثنائية السطح} $D_n$.

\textbf{19.} وبالجمع: تثبّت \omterm{def:b1:structures:group}{الزمرة} المنتهية من تقايسات المستوي
نقطةً $c$ (السؤال 15)، وتُردّ إلى \omterm{def:b1:structures:subgroup}{زمرة جزئية} منتهية
من $O(2)$ (السؤال 16)، وتكون $C_n$ إذا احتوت
دورانات فقط (السؤال 17)، و $D_n$ فيما عدا ذلك (السؤال 18):
وهي \emph{مبرهنة ليوناردو}.

\textbf{20.} للانسحاب $t_v \neq \mathrm{id}$ قوًى
$t_{kv}$، وكلها متمايزة (لأن $kv$ متمايزة مثنى مثنى من أجل $v \neq 0$):
أي رتبة غير منتهية. وللانعكاس المنزلق $f$ لدينا $f^2 = t_v$ مع $v
\neq 0$: أي رتبة غير منتهية مرة أخرى. فلا واحد منهما يسع في \omterm{def:b1:structures:group}{زمرة} منتهية
--- وهذا متسق مع الأسئلة 15--19، التي لم تنتج إلا
دورانات وانعكاسات.

\textbf{21.} التطبيقات $2n$ تحفظ \omterm{def:b1:logic:sets}{مجموعة} الرؤوس: $\omega^k
\omega^j = \omega^{k+j}$ و $\omega^k\conj{\omega^j} =
\omega^{k-j}$؛ ولكونها تقايسات، فهي تحفظ المضلع
(لأن القطع تذهب إلى قطع). وبالعكس، يحفظ التماثل
مرجّح الرؤوس $0$ (السؤال 14)، ومنه فهو خطي،
ويبدّل الرؤوس (لأنها نقاط المضلع
الأبعد مسافةً عن $0$). \omterm{def:b1:euclid:isometry}{والتقايس} الخطي الذي يرسل
الرأس $1$ إلى $\omega^k$ هو $z \mapsto \omega^k z$ إذا كان مباشرًا، و
$z \mapsto \omega^k\conj z$ إذا كان غير مباشر (لأن مصفوفته
تتحدد بعمود واحد وبالإشارة): ومنه $2n$ تماثلًا على
الأكثر، أي بالضبط $D_n$ أعلاه.

\textbf{22.} $C_1$: مثلث مختلف الأضلاع (بلا تماثل
غير بديهي). و $D_1$: مثلث متساوي الساقين غير متساوي الأضلاع (بانعكاس
واحد). و $D_2$: مستطيل غير مربع (\omterm{def:b1:logic:map}{التطبيق} المطابق، ونصف الدورة
حول المركز، وانعكاسا المحورين). و $C_3$: شكل
ثلاثي الأذرع --- ثلاثة أذرع منحنية متطابقة مثبَّتة على فترات
$120$ درجة؛ فيقتل الانحناء كل انعكاس ويبقي
الدورانات من الرتبة $3$.

\textbf{23.} مع $\omega = \iu$: الدورانات $z \mapsto z$ و
$\iu z$ و $-z$ و $-\iu z$ (بالزوايا $0, \frac\pi2, \pi,
\frac{3\pi}2$)، والانعكاسات
\[
z \mapsto \conj z \ (\text{محور } y = 0), \quad
\iu\conj z \ (y = x), \quad
-\conj z \ (x = 0), \quad
-\iu\conj z \ (y = -x):
\]
أي القطران والمتوسطان في المربع --- وهي
\omterm{pb:b1:euclid:1}{الزمرة ثنائية السطح} $D_4$ من الرتبة $8$.

\textbf{24.} اكتب $a = \eu^{\iu\theta} \neq 1$: $f(z) = az +
c_1(1 - a)$ و $g(z) = az + c_2(1 - a)$. عندئذ
\[
f\circ g(z) - g\circ f(z)
= (1-a)\bigl(ac_2 + c_1 - ac_1 - c_2\bigr)
= -(1 - a)^2(c_2 - c_1) \neq 0 ,
\]
ومنه $f g \neq g f$. ولكلا التركيبين معامل خطي
$a^2$، ومنه فللمقدار $(fg)\circ(gf)^{-1}$ معاملٌ خطي $1$: أي إنه
الانسحاب بالمقدار $fg(z) - gf(z)$، وهو الثابت غير المعدوم أعلاه.
\omterm{def:b1:structures:group}{والزمرة} التي تحتوي $f$ و $g$ تحتوي هذا الانسحاب و
كل قواه: ومنه فهي غير منتهية. فمركزا دوران أكثر مما
تحتمل \omterm{def:b1:structures:group}{زمرة} منتهية --- وهو القلب الهندسي لمبرهنة
ليوناردو.

\textbf{25.} (أ) يبيّن السؤالان 3--4 أن كل \omterm{def:b1:logic:map}{تطبيق} حافظ
للمسافات أفينيٌّ بجزء خطي متعامد، ويثبّت السؤال 15
كل \omterm{def:b1:structures:group}{زمرة} منتهية عند نقطة صامدة: فتحوّل النتيجتان
الهندسةَ المترية إلى جبر خطي عند المبدأ. (ب) و
القائمة الكاملة هي الانسحاب والدوران والانعكاس \omterm{pb:b1:euclid:1}{والانعكاس
المنزلق}؛ ويفصل محدد الجزء الخطي المباشرَ
عن غير المباشر، ويفصل وجود النقاط الصامدة بين
النوعين داخل كل زوجية. (ج) وحسب السؤال 9، يتركّب
انعكاسان ليعطيا أيّ دوران أو انسحاب، ومنه تولّد
الانعكاسات الزمرةَ كلها --- ويكفي ثلاثة على الأكثر
من أجل أيّ \omterm{def:b1:euclid:isometry}{تقايس}. (د) وعلى السلّم المنتهي لا تنجو إلا عائلتان،
الزمر الدورية والزمر ثنائية السطح، ولهذا تأتي زخارف
الورديات في نوعين بالضبط (بمحاور مرآة أو بدونها).
وقد برهن الجزء 2 على \emph{تصنيف تقايسات
المستوي}؛ وبرهن الجزء 4 على \emph{مبرهنة ليوناردو}.
\end{solution}
