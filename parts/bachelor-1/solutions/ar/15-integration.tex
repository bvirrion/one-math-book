\section*{الفصل \ref{ch:b1:integration} --- التكامل على قطعة}

\begin{solution}{exo:b1:integration:1}
$\dfrac{1}{x^2-4} = \dfrac{1/4}{x - 2} - \dfrac{1/4}{x+2}$
(بالتغطية)، ومنه
\[
\int_0^1 \frac{\dd x}{x^2 - 4}
= \frac14\Bigl[\ln\abs{x-2} - \ln\abs{x+2}\Bigr]_0^1
= \frac14\Bigl(\ln\frac{1}{3} - \ln 1\Bigr) = -\frac{\ln 3}{4}.
\]

وبالتجزئة ($u' = 1$ و $v = \ln t$): $\int_1^{\eu} \ln t\,\dd t =
\bigl[t\ln t\bigr]_1^{\eu} - \int_1^{\eu} \dd t = \eu - (\eu - 1) =
1$.

وبالتجزئة ($u' = \sin t$ و $v = t$): $\int_0^\pi t\sin t\,\dd t =
\bigl[-t\cos t\bigr]_0^\pi + \int_0^\pi \cos t\,\dd t = \pi + 0 =
\pi$.
\end{solution}

\begin{solution}{exo:b1:integration:2}
بالتعويض $u = t^2 + 1$ و $\dd u = 2t\,\dd t$:
\[
\int_0^1 \frac{t\,\dd t}{(t^2+1)^2}
= \frac12 \int_1^2 \frac{\dd u}{u^2}
= \frac12\Bigl[-\frac1u\Bigr]_1^2 = \frac14 .
\]
ومع $u = \sin t$: $\int_0^{\pi/2} \cos^3 t\,\dd t = \int_0^{\pi/2}
(1 - \sin^2 t)\cos t\,\dd t = \bigl[\sin t - \frac{\sin^3
t}{3}\bigr]_0^{\pi/2} = 1 - \frac13 = \frac23$.
\end{solution}

\begin{solution}{exo:b1:integration:3}
$u_n = \dfrac1n \sum_{k=1}^{n} \dfrac{1}{1 + (k/n)^2}$: وهو مجموع ريمان
للمقدار $x \mapsto \frac{1}{1+x^2}$ على $\intcc{0}{1}$، ومنه $u_n \to
\int_0^1 \frac{\dd x}{1+x^2} = \arctan 1 = \dfrac\pi4$.

$\ln v_n = \dfrac1n \sum_{k=1}^{n} \ln\dfrac{n+k}{n} = \dfrac 1n
\sum_{k=1}^n \ln\Bigl(1 + \dfrac kn\Bigr) \to \int_0^1 \ln(1+x)\,\dd
x = \bigl[(1+x)\ln(1+x) - x\bigr]_0^1 = 2\ln 2 - 1$. ومنه $v_n \to
\eu^{2\ln 2 - 1} = \dfrac 4\eu$. (وللتحقق من التعرّف:
$\frac{(2n)!}{n!\,n^n} = \prod_{k=1}^{n} \frac{n+k}{n}$.)
\end{solution}

\begin{solution}{exo:b1:integration:4}
النهاية هي $0$. ليكن $M = \sup \abs f$ و $\varepsilon \in
\intoo{0}{1}$. اقطع عند $1 - \varepsilon$:
\[
\Bigl| \int_0^1 x^n f \Bigr|
\leq \int_0^{1 - \varepsilon} x^n \abs f + \int_{1-\varepsilon}^1
x^n \abs f
\leq M\,(1-\varepsilon)^n + M\varepsilon .
\]
ولأن $(1 - \varepsilon)^n \to 0$ (\cref{exo:b1:seq:3})، تكون النهاية العليا
للطرف الأيسر $\leq M\varepsilon$ من أجل كل $\varepsilon$:
ومنه يؤول \omterm{thm:b1:integration:def}{التكامل} إلى $0$.
\end{solution}

\begin{solution}{exo:b1:integration:5}
$Q(\lambda) = \int_a^b (f + \lambda g)^2 = \int f^2 + 2\lambda \int
fg + \lambda^2 \int g^2 \geq 0$ من أجل كل $\lambda$. فإذا كان $\int g^2 =
0$، فإن $g = 0$ (بالإيجابية القطعية،
\cref{thm:b1:integration:props}~(4)) وتكون المتراجحة $0 \leq
0$. وإلا فإن $Q$ \omterm{def:b1:poly:def}{كثير حدود} حقيقي من الدرجة الثانية، وهو $\geq 0$ في كل مكان: ومنه
فمميّزه $\leq 0$، أي $\bigl(\int fg\bigr)^2 \leq \int
f^2 \int g^2$.

والمساواة إذا وفقط إذا انعدم المميّز إذا وفقط إذا كان $Q(\lambda_0) = 0$ من أجل
$\lambda_0$ ما، أي $\int (f + \lambda_0 g)^2 = 0$، أي (بالإيجابية
القطعية مرة أخرى) $f = -\lambda_0 g$: ومنه تتحقق المساواة بالضبط عندما
يكون $f$ و $g$ متناسبين.
\end{solution}

\begin{solution}{exo:b1:integration:6}
لتكن $\Phi(a) = \int_a^{a+T} f$. وحسب المبرهنة الأساسية
(\cref{thm:b1:integration:ftc})، تكون $\Phi$ \omterm{def:b1:derivative:def}{قابلة للاشتقاق} مع
$\Phi'(a) = f(a + T) - f(a) = 0$: ومنه فهي ثابتة.

ومن أجل $x > 0$، اكتب $x = nT + r$ مع $0 \leq r < T$ ($n = \lfloor x/T
\rfloor$). وبشال:
\[
\int_0^x f = n \int_0^T f + \int_{nT}^{nT + r} f,
\qquad
\Bigl| \int_{nT}^{nT+r} f \Bigr| \leq T \sup_{\intcc{0}{T}} \abs f
= C .
\]
عندئذ $\frac 1x \int_0^x f = \frac{nT}{x}\cdot\frac 1T \int_0^T f +
O\bigl(\frac 1x\bigr)$، و $\frac{nT}{x} \to 1$: فالنهاية هي
$\frac1T \int_0^T f$.
\end{solution}

\begin{solution}{exo:b1:integration:7}
لتكن $g(x) = f(x) - x$: فهي \omterm{def:b1:continuity:continuous}{متصلة}، مع
\[
\int_0^1 g = \int_0^1 f - \frac12 = 0 .
\]
ولو لم تنعدم $g$ أبدًا، لفرضت مبرهنة القيم الوسطى
إشارةً ثابتة (لأن الدالة \omterm{def:b1:continuity:continuous}{المتصلة} على \omterm{prop:b1:reals:intervals}{فترة} تأخذ الإشارتين
تنعدم)؛ ولنقل $g > 0$. عندئذ، بالإيجابية القطعية
(\cref{thm:b1:integration:props}~(4) مطبَّقةً على $g > 0$، فتعطي
$\int g > 0$): يقع تناقض مع $\int g = 0$. ومنه $g(c) = 0$ من أجل
$c$ ما: أي $f(c) = c$.
\end{solution}

\begin{solution}{exo:b1:integration:8}
\begin{enumerate}
  \item بالتجزئة مع $u' = \sin t$ و $v = \sin^{n-1} t$:
        \[
        W_n = \bigl[-\cos t\sin^{n-1}t\bigr]_0^{\pi/2}
        + (n-1)\int_0^{\pi/2} \cos^2 t\,\sin^{n-2} t\,\dd t
        = (n-1)(W_{n-2} - W_n),
        \]
        ومنه $nW_n = (n-1)W_{n-2}$. وانطلاقًا من $W_0 = \frac\pi2$ و $W_1 = 1$:
        \[
        W_{2p} = \frac{(2p-1)(2p-3)\cdots 1}{(2p)(2p-2)\cdots 2}\,
        \frac{\pi}{2} = \frac{(2p)!}{4^p (p!)^2}\,\frac\pi2,
        \qquad
        W_{2p+1} = \frac{(2p)(2p-2)\cdots 2}{(2p+1)(2p-1)\cdots 3}
        = \frac{4^p (p!)^2}{(2p+1)!} .
        \]
  \item على $\intoo{0}{\frac\pi2}$، $0 < \sin t < 1$، ومنه $\sin^{n+1} <
        \sin^n$ وتكون $(W_n)$ متناقصة (قطعًا) وموجبة.
        وبالحصر مع العلاقة التراجعية:
        \[
        \frac{n}{n+1} = \frac{W_{n+1}}{W_{n-1}}
        \leq \frac{W_{n+1}}{W_n} \leq 1
        \quad\implies\quad \frac{W_{n+1}}{W_n} \to 1 .
        \]
        والثابت: يحقق $a_n = (n+1)W_{n+1}W_n$ الشرط $a_n = a_{n-1}$
        بالعلاقة التراجعية $(n+1)W_{n+1} = nW_{n-1}$، ومنه $a_n = a_0 =
        1 \cdot W_1 W_0 = \frac\pi2$. عندئذ
        \[
        n W_n^2 \sim (n+1) W_{n+1} W_n = \frac\pi2
        \quad\implies\quad
        W_n \sim \sqrt{\frac{\pi}{2n}} .
        \]
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:integration:9}
\begin{enumerate}
  \item على $\intoo{0}{\pi}$: $x > 0$ و $a - bx = b(\frac ab - x) =
        b(\pi - x) > 0$ و $\sin x > 0$، ومنه فالمكامَل $> 0$
        و $I_n > 0$ (بالإيجابية القطعية). والحاصر: على
        $\intcc{0}{\pi}$، $x \leq \pi$ و $a - bx \leq a$، ومنه $P
        \leq \frac{\pi^n a^n}{n!}$ و $I_n \leq
        \pi\,\frac{(\pi a)^n}{n!}$، وهو يؤول إلى $0$ (لأن
        العاملي يغلب الحدّ الهندسي: فهو الحدّ العام
        للمتسلسلة الأسّية المتقاربة، قارن
        \cref{ex:b1:seq:e})؛ وعلى الخصوص $I_n < 1$ من أجل $n$ كبير.
  \item انشر $x^n(a - bx)^n = \sum_{j=0}^{n} \binom nj a^{n-j}
        (-b)^j x^{n+j}$: ومنه $P = \frac{1}{n!}\sum_j c_j x^{n+j}$
        بمعاملات صحيحة $c_j$. وعندئذ $P^{(k)}(0) = 0$ من أجل $k < n$
        (بالتقييم) و، من أجل $n \leq k \leq 2n$، $P^{(k)}(0) =
        \frac{k!}{n!} c_{k-n}$، وهو عدد صحيح لأن $n! \mid k!$.
        وفوق ذلك $P(\pi - x) = P(x)$ (بالتعويض: إذ يبادل $\pi - x$
        بين العاملين، باستعمال $a - b(\pi - x) = bx$)، ومنه $P^{(k)}(\pi)
        = \pm P^{(k)}(0)$: فهي أعداد صحيحة كذلك.
  \item مع $Q = P - P'' + P^{(4)} - \dots$ (وهو منته: فدرجة $P$ هي
        $2n$): $Q + Q'' = P$، و
        \[
        \bigl(Q'\sin x - Q\cos x\bigr)' = (Q + Q'')\sin x = P\sin x .
        \]
        ومنه $I_n = \bigl[Q'\sin x - Q\cos x\bigr]_0^{\pi} = Q(\pi)
        + Q(0)$، وهو مجموع قيم $P^{(2k)}$ عند $0$ وعند $\pi$: أي
        عدد صحيح حسب (2).
  \item ومن أجل $n$ كبير، يكون $I_n$ عددًا صحيحًا مع $0 < I_n < 1$:
        وهذا محال. ومنه يفشل الفرض $\pi = \frac ab$: أي إن $\pi$
        أصمّ.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:integration:10}
حالة الصنف $C^1$: بالتجزئة،
\[
\int_a^b f(t)\sin\lambda t\,\dd t
= \Bigl[-f(t)\frac{\cos\lambda t}{\lambda}\Bigr]_a^b
+ \frac{1}{\lambda}\int_a^b f'(t)\cos\lambda t\,\dd t,
\]
وهو محدود قيمةً مطلقة بالمقدار $\frac{2\sup\abs f + (b - a)\sup\abs{f'}}
{\lambda} \to 0$.

وحالة \omterm{def:b1:continuity:continuous}{الاتصال} فحسب: ليكن $\varepsilon > 0$ ولنختر \omterm{def:b1:integration:step}{دالة درجية}
$\varphi$ تحقق $\abs{f - \varphi} \leq \varepsilon$
(\cref{thm:b1:integration:approx} يعطي $\varphi \leq f \leq
\psi$ بفجوة $\leq\varepsilon$؛ فخذ $\varphi$). عندئذ
\[
\Bigl|\int_a^b f\sin\lambda t\Bigr|
\leq \int_a^b \abs{f - \varphi}
+ \Bigl|\int_a^b \varphi \sin\lambda t\Bigr|
\leq (b-a)\varepsilon
+ \sum_i \abs{c_i}\,\Bigl|\int_{x_{i-1}}^{x_i} \sin\lambda t\,\dd
t\Bigr| ,
\]
وكل $\bigl|\int \sin \lambda t\bigr| = \bigl|\frac{\cos\lambda
x_{i-1} - \cos\lambda x_i}{\lambda}\bigr| \leq \frac{2}{\lambda}$:
ومنه يؤول الحدّ الثاني إلى $0$. ومنه فالنهاية العليا هي $\leq
(b-a)\varepsilon$ من أجل كل $\varepsilon$: فالنهاية هي $0$.
\end{solution}

\begin{solution}{exo:b1:integration:11}
ليكن $m = \min f$ و $M = \max f$، وهما مبلوغان حسب مبرهنة القيم
الحدّية. ولأن $g \geq 0$: $m\,g \leq fg \leq M\,g$، ومنه بالرتابة
\[
m \int_a^b g \;\leq\; \int_a^b fg \;\leq\; M \int_a^b g .
\]
فإذا كان $\int_a^b g = 0$: فرضت الإيجابية القطعية
(\cref{thm:b1:integration:props}~(4)) أن $g \equiv 0$، فينعدم
الطرفان، ويفي أيّ $c$ بالغرض. وإلا فإن $t = \frac{\int
fg}{\int g}$ يقع في $\intcc{m}{M} = f(\intcc{a}{b})$
(\cref{thm:b1:continuity:evt,thm:b1:continuity:ivt})، ومنه $t =
f(c)$ من أجل $c$ ما.

والإشارة مهمة: فعلى $\intcc{-1}{1}$ مع $f(t) = g(t) = t$: $\int
fg = \int_{-1}^1 t^2 = \frac23$، بينما $f(c)\int_{-1}^1 t\,\dd t
= 0$ من أجل كل $c$.
\end{solution}

\begin{solution}{exo:b1:integration:12}
إذا كان $f \equiv 0$ فالادعاء فارغ (فكل نقطة جذر). ومنه
افترض $f \not\equiv 0$ ولنفترض أن لها $n$ جذرًا متمايزًا على الأكثر
في $\intoo{a}{b}$؛ وليكن $z_1 < \dots < z_m$ ($m \leq n$)
تلك الجذور التي \emph{تغيّر} $f$ إشارتها عندها (وقد لا توجد). ضع
$P(t) = \prod_{i=1}^{m}(t - z_i)$ (والجداء الخالي $= 1$)، من
الدرجة $m \leq n$. وعلى كل \omterm{prop:b1:reals:intervals}{فترة} جزئية تقطعها $z_i$، تكون لكلٍّ من
$f$ و $P$ إشارة ثابتة، وتنقلب إشارتهما معًا عند عبور
$z_i$ ما: ومنه للجداء $fP$ إشارة ثابتة واحدة على كل
$\intoo{a}{b}$. ولكونه متصلًا وغير معدوم تمامًا وذا
إشارة ثابتة، يكون له $\bigl|\int_a^b fP\bigr| > 0$ (بالإيجابية
القطعية مطبَّقةً على $\abs{fP}$). لكن $\int f P$ تركيبة
خطية من العزوم $\int f\,t^k$ مع $k \leq n$، وكلها معدومة:
وهذا تناقض. ومنه فللمقدار $f$ عدد $n + 1$ جذرًا متمايزًا على الأقل في
$\intoo{a}{b}$.
\end{solution}

\begin{solution}{pb:b1:integration:1}
\textbf{1.} ليكن $N = \lceil 2c \rceil$. من أجل $n \geq N$:
$\frac{c^{n+1}/(n+1)!}{c^n/n!} = \frac{c}{n+1} \leq \frac12$، ومنه
$0 < \frac{c^n}{n!} \leq \frac{c^N}{N!}\,2^{-(n - N)} \to 0$:
وهو الحصر.

\textbf{2.} ثبّت $k$؛ وبالاستقراء على $m$. من أجل $m = 0$: $\int_0^1
x^k = \frac{1}{k+1} = \frac{k!\,0!}{(k+1)!}$. والخطوة، بالتجزئة
($u = (1-x)^m$ و $v' = x^k$):
\[
\int_0^1 x^k (1-x)^m \dd x
= \Bigl[\frac{x^{k+1}}{k+1}(1-x)^m\Bigr]_0^1
+ \frac{m}{k+1}\int_0^1 x^{k+1}(1-x)^{m-1}\dd x
= \frac{m}{k+1}\cdot\frac{(k+1)!\,(m-1)!}{(k+m+1)!} ,
\]
وهو $\frac{k!\,m!}{(k+m+1)!}$.

\textbf{3.} $k = m = n$: $\int_0^1 (x(1-x))^n =
\frac{(n!)^2}{(2n+1)!} = \frac{1}{(2n+1)\binom{2n}{n}}$. ولأن
$x(1-x) \leq \frac14$ على $\intcc{0}{1}$، يكون \omterm{thm:b1:integration:def}{التكامل} $\leq
4^{-n}$، ومنه $\binom{2n}{n} \geq \frac{4^n}{2n+1}$ ---
وهذا متسق مع $\binom{2n}{n}^{1/n} \to 4$
(\cref{pb:b1:seq:1}).

\textbf{4.} المكامَل متصل و $\geq 0$ وموجب
على $\intoo{0}{1}$، ومنه فهو غير معدوم تمامًا: ومنه تكامله
$> 0$ (\cref{thm:b1:integration:props}~(4)). \omterm{def:b1:reals:bounds}{والحاصر الأعلى}:
$(x(1-x))^n \leq 4^{-n}$ و $g \leq \sup\abs g$، ثم
الرتابة.

\textbf{5.} $A_0 = \eu - 1$؛ $A_1 = [x\eu^x]_0^1 - \int_0^1
\eu^x = \eu - (\eu - 1) = 1$. وبالتجزئة: $A_n = [x^n \eu^x]_0^1 -
n\int_0^1 x^{n-1}\eu^x = \eu - n A_{n-1}$. والحواصر: المكامَل
موجب، ومنه $A_n > 0$؛ ويعطي $\eu^x \leq \eu$ أن $A_n \leq
\eu\int_0^1 x^n = \frac{\eu}{n+1}$.

\textbf{6.} $A_0 = -1 + 1\cdot\eu$. فإذا كان $A_{n-1} = \alpha_{n-1} +
\beta_{n-1}\eu$ بمركبات صحيحة، فإن
\[
A_n = \eu - n\alpha_{n-1} - n\beta_{n-1}\eu
= \underbrace{(-n\,\alpha_{n-1})}_{\alpha_n}
+ \underbrace{(1 - n\,\beta_{n-1})}_{\beta_n}\,\eu ,
\]
وكلاهما عدد صحيح.

\textbf{7.} إذا كان $\eu = \frac pq$: $q A_n = q\alpha_n + p\beta_n
\in \Z$، و $0 < qA_n \leq \frac{q\eu}{n+1} < 1$ من أجل $n$ كبير:
أي عدد صحيح قطعًا بين $0$ و $1$ --- وهذا محال. ومنه $\eu
\notin \Q$. وفي \cref{exo:b1:seq:9} كان العدد الصحيح المحبوس هو
$q!\,\frac pq - q!\,a_q$؛ وهنا هو $qA_n$: أي الفخّ نفسه،
بوقود تكاملي.

\textbf{8.} $n = 0$: $R_0 = \int_0^1 \eu^t = \eu - 1$، ومنه $\eu =
1 + R_0$. وبالتجزئة ($u = \eu^t$ و $v = -\frac{(1-t)^{n+1}}{n+1}$):
\[
R_n = \frac{1}{n!}\Bigl(\Bigl[-\frac{(1-t)^{n+1}}{n+1}
\eu^t\Bigr]_0^1 + \frac{1}{n+1}\int_0^1 (1-t)^{n+1}\eu^t\Bigr)
= \frac{1}{(n+1)!} + R_{n+1} ,
\]
ومنه تنتشر الصيغة من $n$ إلى $n + 1$. والحواصر: يعطي $1 \leq
\eu^t \leq \eu$ على $\intcc{0}{1}$ و $\int_0^1 (1-t)^n =
\frac{1}{n+1}$ أن $\frac{1}{(n+1)!} \leq R_n \leq
\frac{\eu}{(n+1)!}$.

\textbf{9.} $\sum_{k=0}^{9} \frac{1}{k!} = \frac{986410}{362880}
= 2.71828152\dots$، و
\[
\frac{1}{10!} = 2.76\cdot10^{-7} \leq R_9 \leq
\frac{2.75}{10!} = 7.58\cdot10^{-7} ,
\]
ومنه $2.7182818 \leq \eu \leq 2.7182823$: أي، مع برهان، $\eu =
2.718282$ إلى ستة أرقام عشرية (والقيمة الحقيقية $2.7182818\dots$).

\textbf{10.} $f(1 - x) = \frac{(1-x)^n x^n}{n!} = f(x)$. وعلى
$\intoo{0}{1}$: $0 < x(1-x) \leq \frac14$، ومنه $0 < f \leq
\frac{1}{4^n\,n!}$.

\textbf{11.} $x^n(1-x)^n = \sum_{j=0}^{n} (-1)^j\binom
nj\,x^{n+j}$، ومنه $f = \frac{1}{n!}\sum_j c_j\,x^{n+j}$ مع
$c_j \in \Z$. ومنه $f^{(k)}(0) = 0$ من أجل $k < n$ أو $k > 2n$،
ومن أجل $n \leq k \leq 2n$: $f^{(k)}(0) = \frac{k!}{n!}\,
c_{k-n}$، وهو عدد صحيح لأن $n! \mid k!$. ويعطي التناظر
$f^{(k)}(1) = (-1)^k f^{(k)}(0) \in \Z$.

\textbf{12.} $b^n \pi^{2n-2k} = b^n\bigl(\frac
ab\bigr)^{\,n-k} = a^{\,n-k}\,b^{\,k} \in \Z$، ومنه $G(0) =
\sum_k (-1)^k a^{n-k} b^k f^{(2k)}(0)$ وكذلك $G(1)$ عددان
صحيحان حسب السؤال 11.

\textbf{13.} في $\pi^2 G + G''$، يحمل الحدّ $k$ من $\pi^2 G$
المقدارَ $\pi^{2n-2k+2}f^{(2k)}$ ويحمل الحدّ $j = k - 1$ من
$G''$ المقدارَ $(-1)^{k-1}\pi^{2n-2k+2}f^{(2k)}$: فيتلاشى كل شيء
إلا $k = 0$ في المجموع الأول و $j = n$ في
الثاني، أي
\[
G'' + \pi^2 G = b^n\bigl(\pi^{2n+2} f + (-1)^n f^{(2n+2)}\bigr)
= b^n \pi^{2n+2} f = \pi^2 a^n f
\]
(فدرجة $f$ هي $2n$، ومنه $f^{(2n+2)} = 0$؛ و $b^n\pi^{2n} =
a^n$). عندئذ
\[
\bigl(G'\sin\pi x - \pi G\cos\pi x\bigr)'
= (G'' + \pi^2 G)\sin \pi x = \pi^2 a^n f\sin\pi x .
\]

\textbf{14.} بالمكاملة على $\intcc{0}{1}$:
\[
\pi^2 a^n \int_0^1 f\sin(\pi x)\,\dd x
= \bigl[G'\sin\pi x - \pi G\cos\pi x\bigr]_0^1
= \pi\bigl(G(1) + G(0)\bigr) ,
\]
ومنه $\pi a^n \int_0^1 f\sin\pi x = G(0) + G(1) \in \Z$. وعلى
$\intoo{0}{1}$، $f > 0$ و $\sin\pi x > 0$: ومنه فالطرف الأيسر
موجب، ومنه $G(0) + G(1) \geq 1$؛ ويحدّه $\sin \leq 1$ مع
السؤال 10 بالمقدار $\frac{\pi a^n}{4^n\,n!}$.

\textbf{15.} حسب السؤال 1 (مع $c = \frac a4$)،
$\frac{\pi a^n}{4^n n!} \to 0$: فمن أجل $n$ كبير يكون $< 1$،
مناقضًا $G(0) + G(1) \geq 1$. ومنه فلا كسر $\frac ab$
يساوي $\pi^2$: وهي مبرهنة لوجاندر. ولو كان $\pi$ ناطقًا،
لكان $\pi^2$ كذلك: ومنه $\pi \notin \Q$ --- ولا يجري الاستلزام
إلا في هذا الاتجاه (فالعدد $\sqrt2$ أصمّ ومربعه
ناطق)، ولهذا يكون $\pi^2 \notin \Q$ أقوى قطعًا
من \cref{exo:b1:integration:9}.

\textbf{16.} الصحيحية: السؤالان 11--12 (مشتقات
الطرفين)؛ والصغر: السؤالان 10 و 1؛ والجسر: السؤالان
13--14 (المكاملة المزدوجة المتلسكبة بالتجزئة). وبدون
$\frac{1}{n!}$، تبقى معطيات الطرفين أعدادًا صحيحة (بل بسهولة
أكبر)، لكن الحاصر يصير $\frac{\pi a^n}{4^n}$، وهو
لا يؤول إلى $0$ إلا عندما $a < 4$ --- ولكل مرشّح $a =
b\,\pi^2 > 9$. فالعاملي هو بالضبط ما يسبق
النمو الهندسي $a^n$: فلا عاملي، لا مبرهنة.

\textbf{17.} من أجل $a \leq 10$، يكون العدد الصحيح $G(0) + G(1)$
موجبًا و $\pi\,(10/4)^n/n!$ على الأكثر. وعند $n = 7$:
$2.5^7 = 610.35\dots$، ومنه فالحاصر هو $\frac{\pi \times
610.35}{5040} \approx 0.38 < 1$ (وعند $n = 6$ لا يزال
$1.07$): فيحطّ التناقض عند المرحلة السابعة،
صراحةً.

\textbf{18.} بمكاملتين بالتجزئة:
\[
\int_0^1 x(1-x)\sin\pi x\,\dd x
= \frac1\pi\int_0^1 (1 - 2x)\cos\pi x\,\dd x
= \frac{2}{\pi^2}\int_0^1 \sin\pi x\,\dd x
= \frac{2}{\pi^2}\cdot\frac{2}{\pi} = \frac{4}{\pi^3}
\]
(وتنعدم الحدود بين القوسين: $x(1-x)$ عند $0, 1$، و $\sin\pi x$
عند $0, 1$). ورمزيًا، يُقرأ تلسكوب $n = 1$ (دون أيّ فرض
على $\pi$) $\pi^3\int_0^1 f_1\sin\pi x = -(f_1''(0) +
f_1''(1))$ مع $f_1 = x(1 - x)$ و $f_1'' = -2$: فالطرف الأيمن $4$
--- ومنه يتوافق الحسابان.

\textbf{19.} يحلّ $\eu^x$ المعادلةَ $y' = y$ ويحلّ $\sin\pi x$ المعادلةَ
$y'' = -\pi^2 y$: أي معادلتان خطيتان بمعاملات ثابتة،
ومنه تدير \omterm{thm:b1:integration:parts}{المكاملة بالتجزئة} النواةَ عائدةً إلى نفسها و
تُبقي كل معطيات الحدّ داخل $\Z + \Z\eu$ (أو كثيرات
حدود صحيحة في $\pi^2$). وخاصية الانغلاق تلك هي ما تحتاج إليه
الآلة. وبتلطيفها بنوى مكيَّفة مع عدة نقاط
دفعة واحدة، تعطي الآلية نفسها مبرهنة إرميت (تسامي $\eu$،
1873) ومبرهنة لينديمان (تسامي $\pi$،
1882) --- وهما خارج هذا المجلّد.

\textbf{20.} بالقسمة الكثيرحدودية (أو اضرب رجوعًا وتحقق):
\[
x^4(1-x)^4 = (x^6 - 4x^5 + 5x^4 - 4x^2 + 4)(1 + x^2) - 4 .
\]
وبمكاملة المتطابقة المعروضة مقسومةً على $1 + x^2$:
\[
\int_0^1 \frac{x^4(1-x)^4}{1+x^2}\dd x
= \Bigl(\frac17 - \frac46 + 1 - \frac43 + 4\Bigr)
- 4\arctan 1
= \frac{22}{7} - \pi ,
\]
باستعمال $\frac17 - \frac23 + 1 - \frac43 + 4 = 3 + \frac17$ و
$\arctan 1 = \frac\pi4$.

\textbf{21.} المكامَل متصل وموجب على
$\intoo{0}{1}$: ومنه \omterm{thm:b1:integration:def}{فالتكامل} $> 0$، ومنه $\pi < \frac{22}{7}$.
وفوق ذلك $\frac12 \leq \frac{1}{1+x^2} \leq 1$ على
$\intcc{0}{1}$ و $\int_0^1 (x(1-x))^4 = \frac{(4!)^2}{9!} =
\frac{1}{630}$ (السؤال 3):
\[
\frac{1}{1260} \leq \frac{22}{7} - \pi \leq \frac{1}{630}
\quad\Longrightarrow\quad
3.14126 < \frac{22}{7} - \frac{1}{630} \leq \pi \leq
\frac{22}{7} - \frac{1}{1260} < 3.14207 .
\]

\textbf{22.} بترديد $x^2 + 1$: $x^2 \equiv -1$، ومنه $x^{4m} =
(x^2)^{2m} \equiv 1$ و $(1 - x)^2 = 1 - 2x + x^2 \equiv -2x$،
ومنه $(1-x)^{4m} \equiv (-2x)^{2m} = 4^m (x^2)^m \equiv
(-4)^m$: فالباقي هو الثابت $(-4)^m$، و
للحاصل $Q_m$ معاملات صحيحة (لأنها قسمة على \omterm{def:b1:poly:def}{كثير حدود}
صحيح واحديّ). وبقسمة المتطابقة على $1 + x^2$ و
المكاملة:
\[
J_m := \int_0^1 \frac{(x(1-x))^{4m}}{1+x^2}\dd x
= s_m + (-4)^m\,\frac{\pi}{4},
\qquad s_m = \int_0^1 Q_m \in \Q .
\]
وبالحلّ من أجل $\pi$: مع $r_m = (-1)^{m+1}\,4^{\,1-m} s_m \in
\Q$، $\;\abs{\pi - r_m} = 4^{\,1-m} J_m \leq
4^{\,1-m}\cdot4^{-4m} = 4^{\,1-5m}$. ومن أجل $m = 1$: $s_1 =
\frac{22}{7}$ و $r_1 = \frac{22}{7}$ والحاصر $4^{-4} =
\frac{1}{256}$ --- أي السؤالان 20--21 مرة أخرى.

\textbf{23.} $\bigl|\pi - \frac{22}{7}\bigr| = \frac{22}{7} -
\pi \approx 1.26\cdot10^{-3}$، أي أفضل ستة عشر مرة من
المرجع من الرتبة $2$ $\frac{1}{7^2} \approx 2.0\cdot10^{-2}$
الذي يضمنه ديريكليه (\cref{pb:b1:derivative:1}، السؤال 4)؛
و $\bigl|\pi - \frac{355}{113}\bigr| \approx 2.7\cdot10^{-7}$
يغلب $\frac{1}{113^2} \approx 7.8\cdot10^{-5}$ بعامل
$\approx 300$. ولا تناقض مع أيّ شيء مبرهن عليه: فمتراجحات
ليوفيل \emph{تحدّ من أدنى} أخطاء التقريب من أجل
الأعداد الجبرية وحدها، ولا يتوفر مثل هذا الحاصر من أجل $\pi$ في
هذا المستوى --- فالعدد $\pi$ حرّ في أن يُقرَّب تقريبًا رائعًا.

\textbf{24.} المبرهنة المساعدة: افترض $x = \frac pq$ و $0 < \abs{a_n +
b_n x} \to 0$. عندئذ $\abs{a_n + b_n x} = \frac{\abs{q a_n + p
b_n}}{q}$، مع كون $q a_n + p b_n$ عددًا صحيحًا غير معدوم (وهو غير معدوم
لأن القيمة المطلقة $> 0$): ومنه $\abs{a_n + b_n x}
\geq \frac1q$ من أجل كل $n$، وهذا يناقض التقارب إلى
$0$. والأمثلة: السؤال 7 ($x = \eu$ و $a_n = \alpha_n$ و $b_n =
\beta_n$)؛ و \cref{exo:b1:seq:9} ($x = \eu$ مرة أخرى، مع $a_q =
-q!\sum_{k \leq q}\frac{1}{k!}$ و $b_q = q!$)؛ و
\cref{pb:b1:derivative:1}، السؤال 1، هو صيغته الهندسية.
وفي الجزء 3 وفي \cref{exo:b1:integration:9} يجري الفخّ
\emph{داخل} التناقض: إذ يحوّل فرض النطقية
مقدارًا إلى عدد صحيح، ثم يحصره التحليل
في $\intoo{0}{1}$ --- أي المبدأ نفسه، منقولًا.

\textbf{25.} (أ) تعيش الصحيحية في حساب كثيرات الحدود
عند الطرفين (الأسئلة 6 و 11--12)، ويعيش الصغر في الحواصر
التي تعطيها رتابة $\sup$ (السؤالان 4 و 10)، ويعيش الجسر في
\omterm{thm:b1:integration:parts}{المكاملة بالتجزئة} (الأسئلة 8 و 13--14) --- وثلاثتها
مبرهنات في هذا الفصل. (ب) ويوفّر \omterm{thm:b1:integration:def}{التكامل} ما لم تستطع
مبرهنة التزايدات المنتهية توفيره: أي متطابقة \emph{مضبوطة} بين
الغرض التحليلي والمعطيات الحسابية (مساواة، لا مجرد
متراجحة بمجهول $c$)، ولهذا تبلغ الآلة
$\pi^2$ بينما لم يبلغ \cref{pb:b1:derivative:1} إلا
أُسس التقريب. (ج) والمستخرج: $\eu \notin \Q$ و
$\pi^2 \notin \Q$ (ومنه $\pi \notin \Q$)، و $\eu = 2.718282$
مصادَقًا عليه، و $\frac{22}{7} - \frac{1}{630} \leq \pi \leq
\frac{22}{7} - \frac{1}{1260}$، و $\binom{2n}{n} \geq
\frac{4^n}{2n+1}$. (د) والحدّ الأمامي: يدفع إرميت ولينديمان
الآلةَ نفسها إلى التسامي؛ وشغّل آبيري (1979)
فخَّ العدد الصحيح على $\zeta(3)$ --- فالآلة لا تزال تنتج
رياضيات القرن العشرين.
