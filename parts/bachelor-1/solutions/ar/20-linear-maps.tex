\begin{solution}{exo:b1:linmaps:2}
\omterm{def:b1:linmaps:kerim}{النواة}: حُلَّ $x + y - z = 0$ و $2x + y + z = 0$ و $3x + 2y = 0$.
ومن الثالثة، $y = -\frac{3x}{2}$؛ وتعطي الأولى $z = x + y =
-\frac x2$؛ وتحقق في الثانية: $2x - \frac{3x}{2} - \frac x2 = 0$:
فهي محققة. ومنه $\ker u = \operatorname{Vect}\bigl((2, -3, -1)\bigr)$
(بأخذ $x = 2$)، والبُعد $1$.

ومبرهنة \omterm{def:b1:linmaps:rank}{الرتبة}: $\operatorname{rk} u = 3 - 1 = 2$. \omterm{def:b1:linmaps:kerim}{والصورة}: مولَّدة
بصور \omterm{def:b1:vspaces:free}{الأساس} القانوني، $u(e_1) = (1,2,3)$ و $u(e_2) =
(1,1,2)$ و $u(e_3) = (-1,1,0)$؛ والأوليان حرتان، \omterm{def:b1:linmaps:rank}{والرتبة}
$2$: \omterm{def:b1:vspaces:free}{فالأساس} $\bigl((1,2,3), (1,1,2)\bigr)$.

وهو ليس \omterm{def:b1:logic:inj}{متباينًا} (لأن $\ker \neq \{0\}$)، ولا \omterm{def:b1:logic:inj}{شاملًا} (لأن \omterm{def:b1:linmaps:rank}{الرتبة} $2 < 3$):
وهذا متسق مع \cref{cor:b1:linmaps:samedim}.
\end{solution}

\begin{solution}{exo:b1:linmaps:3}
\emph{\omterm{def:b1:linmaps:kerim}{النواة}:} يفرض $P = P'$ أن $\deg P = \deg P'$ ما لم يكن $P = 0$؛
لكن $\deg P' < \deg P$ من أجل $P \neq 0$: ومنه $\ker u = \{0\}$، ويكون $u$،
وهو تشاكل ذاتي متباين للفضاء $\R_n[X]$ ذي البُعد المنتهي،
\omterm{def:b1:structures:morphism}{تشاكلًا} \omterm{def:b1:logic:inj}{تقابليًا} (\cref{cor:b1:linmaps:samedim}).

\emph{والمعكوس:} لتكن $v(P) = P + P' + P'' + \dots + P^{(n)}$ (وهو
مجموع منته على $\R_n[X]$). عندئذ
\[
v\bigl(u(P)\bigr) = \sum_{k=0}^{n} (P - P')^{(k)}
= \sum_{k=0}^{n} P^{(k)} - \sum_{k=0}^{n} P^{(k+1)}
= P - P^{(n+1)} = P ,
\]
بالتلسكوب، لأن $P^{(n+1)} = 0$. ومنه $v = u^{-1}$.
\end{solution}

\begin{solution}{exo:b1:linmaps:4}
$\operatorname{im}(v \circ u) = v(\operatorname{im} u) \subseteq
\operatorname{im} v$: \omterm{def:b1:linmaps:rank}{فالرتبة} $\leq \operatorname{rk} v$. و $v$
مقصورًا على $\operatorname{im} u$ صورته
$\operatorname{im}(vu)$ ومع مبرهنة \omterm{def:b1:linmaps:rank}{الرتبة} داخل
$\operatorname{im} u$: $\operatorname{rk}(vu) \leq
\dim\operatorname{im} u = \operatorname{rk} u$.
\end{solution}

\begin{solution}{exo:b1:linmaps:5}
$u^2 = 0$ يعني $u(u(x)) = 0$ من أجل كل $x$: أي إن كل $u(x)$ يقع في
$\ker u$، أي $\operatorname{im} u \subseteq \ker u$. وعندئذ
مبرهنة \omterm{def:b1:linmaps:rank}{الرتبة}:
\[
\dim E = \dim\ker u + \operatorname{rk} u \geq 2\operatorname{rk} u .
\]
ومثال المساواة في $\R^2$: $u(x, y) = (y, 0)$: $u^2 = 0$، و
$\operatorname{rk} u = 1 = \frac{\dim E}{2}$.
\end{solution}

\begin{solution}{exo:b1:linmaps:6}
$(pq)^2 = pqpq = ppqq = pq$ (بالتبادل): فهو \omterm{def:b1:linmaps:projection}{إسقاط}
(\cref{thm:b1:linmaps:projchar}).

\omterm{def:b1:linmaps:kerim}{والصورة}: $\operatorname{im}(pq) \subseteq \operatorname{im} p$، و
$= \operatorname{im}(qp) \subseteq \operatorname{im} q$: فهي محتواة
في التقاطع. وبالعكس، إذا كان $x \in \operatorname{im} p \cap
\operatorname{im} q$، فإن $p(x) = x$ و $q(x) = x$ (لأن النقاط الصامدة
تميّز \omterm{def:b1:linmaps:kerim}{صورة} \omterm{def:b1:linmaps:projection}{إسقاط})، ومنه $pq(x) = x$: أي $x \in
\operatorname{im}(pq)$.

\omterm{def:b1:linmaps:kerim}{والنواة}: $\ker p \subseteq \ker(qp) = \ker(pq)$ وكذلك $\ker q
\subseteq \ker(pq)$: فالمجموع محتوى. وبالعكس، ليكن $pq(x) =
0$، واكتب
\[
x = \underbrace{q(x)}_{\in\, \ker p} +
\underbrace{(x - q(x))}_{\in\, \ker q} :
\]
فالحدّ الأول يحقق $p(q(x)) = 0$، ومنه يقع في $\ker p$؛ و
الثاني في $\ker q$ لأن $q(x - q(x)) = q(x) - q^2(x) = 0$. ومنه
$x \in \ker p + \ker q$.
\end{solution}

\begin{solution}{exo:b1:linmaps:7}
لاحظ أولًا الاحتواءين العامين $\ker u \subseteq \ker u^2$ و
$\operatorname{im} u^2 \subseteq \operatorname{im} u$، ومنه، بمبرهنة
\omterm{def:b1:linmaps:rank}{الرتبة}، (2) $\iff$ (3) (فتساوي النوى $\iff$ تساوي الرتب
$\iff$ تساوي الصور، بحكم الاحتواءين).

(1 $\Rightarrow$ 2): ليكن $u^2(x) = 0$؛ عندئذ $u(x) \in \ker u \cap
\operatorname{im} u = \{0\}$، ومنه $x \in \ker u$.

(2 $\Rightarrow$ 1): بغراسمان ومبرهنة \omterm{def:b1:linmaps:rank}{الرتبة}، $\dim(\ker u +
\operatorname{im} u) = \dim\ker u + \operatorname{rk} u -
\dim(\ker u \cap \operatorname{im} u) = \dim E - \dim(\ker u \cap
\operatorname{im} u)$: فيكون المجموع $E$ إذا وفقط إذا كان التقاطع
$\{0\}$. وليكن $y \in \ker u \cap \operatorname{im} u$: فإن $y = u(x)$ و
$u(y) = 0$، ومنه $u^2(x) = 0$، ومنه (حسب (2)) $u(x) = 0$: أي $y = 0$. ومنه
$E = \ker u \oplus \operatorname{im} u$.
\end{solution}

\begin{solution}{exo:b1:linmaps:8}
إذا كان $\varphi = 0$: فإن $\ker\varphi = E \subseteq \ker\psi$ يفرض
$\psi = 0 = 0\cdot\varphi$. وإذا كان $\varphi \neq 0$: فإن $\ker\varphi$
\omterm{def:b1:linmaps:forms}{مستوٍ فائق}؛ فاختر $a \notin \ker\varphi$. ضع $\lambda =
\frac{\psi(a)}{\varphi(a)}$. والشكل $\psi - \lambda\varphi$
ينعدم على $\ker\varphi$ (لأن كليهما كذلك، بحكم الاحتواء) وعند $a$:
ومنه ينعدم على $\ker\varphi \oplus Ka = E$. ومنه $\psi =
\lambda\varphi$.
\end{solution}

\begin{solution}{exo:b1:linmaps:9}
افترض $\lambda_0 x + \lambda_1 u(x) + \dots + \lambda_{n-1}
u^{n-1}(x) = 0$. وطبّق $u^{n-1}$: فتموت كل الحدود ذات العامل $u^{\geq
n}$، ويبقى $\lambda_0 u^{n-1}(x) = 0$، ومنه $\lambda_0 = 0$.
وطبّق $u^{n-2}$ على العلاقة الباقية: $\lambda_1 u^{n-1}(x) =
0$، ومنه $\lambda_1 = 0$؛ وهكذا. \omterm{def:b1:vspaces:free}{فالعائلة حرة}؛ وبحجم
$n = \dim E$، تكون \omterm{def:b1:vspaces:free}{أساسًا} (\cref{prop:b1:findim:twoofthree}). (وفي
هذا \omterm{def:b1:vspaces:free}{الأساس}، يفعل $u$ فعلَ إزاحة --- وهو نموذج انعدام
القوى الأكبري.)
\end{solution}

\begin{solution}{exo:b1:linmaps:10}
\begin{enumerate}
  \item $f \circ f = -\mathrm{id}$ تقابلي، ومنه \omterm{def:b1:logic:map}{فالتطبيق} $f$ كذلك
        (\cref{prop:b1:logic:comp} مكيَّفًا: لأن للمقدار $f$ المعكوسَ
        الثنائي الجهة $-f$). وإذا كان $f(x) = \lambda x$ مع $x \neq 0$:
        فبتطبيق $f$، $-x = \lambda^2 x$، ومنه $\lambda^2 = -1$:
        وهذا محال في $\R$.
  \item ابنِ العائلة بجشع. خذ $x_1 \neq 0$: فالعائلة $(x_1,
        f(x_1))$ \omterm{def:b1:vspaces:free}{حرة} حسب (1). وإذا كان $\operatorname{Vect}$ للعائلة
        الحالية $\bigl(x_1, f(x_1), \dots, x_k,
        f(x_k)\bigr)$، ولنسمّه $V_k$ --- وهو \omterm{def:b1:vspaces:subspace}{فضاء جزئي} مستقر تحت $f$
        (لأن كل مولّد يُرسل إلى مولّد آخر أو مقابله:
        $f(f(x_i)) = -x_i$) --- ليس كل $E$، فاختر $x_{k+1}
        \notin V_k$. \emph{والادعاء: العائلة الموسَّعة \omterm{def:b1:vspaces:free}{حرة}.}
        افترض $\alpha x_{k+1} + \beta f(x_{k+1}) + v = 0$ مع $v
        \in V_k$ و $(\alpha, \beta) \neq (0,0)$. وطبّق $f$:
        $\alpha f(x_{k+1}) - \beta x_{k+1} + f(v) = 0$ مع $f(v)
        \in V_k$. وأزل $f(x_{k+1})$ بين العلاقتين
        (اضرب الأولى في $\alpha$، والثانية في $-\beta$، و
        اجمع):
        \[
        (\alpha^2 + \beta^2)\, x_{k+1} \in V_k ,
        \]
        ويفرض $\alpha^2 + \beta^2 \neq 0$ أن $x_{k+1} \in V_k$:
        وهذا تناقض. ومنه يستمر البناء، مضيفًا متجهات
        \emph{اثنتين في كل مرة}، حتى $V_k = E$: فتكون العائلة النهائية
        \omterm{def:b1:vspaces:free}{أساسًا} بحجم زوجي، ويكون $n$ زوجيًا.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:linmaps:11}
\omterm{def:b1:reals:bounds}{الحاصر الأعلى}: من أجل كل $x$، $(u + v)(x) = u(x) + v(x) \in
\operatorname{im} u + \operatorname{im} v$، ومنه
\[
\operatorname{rk}(u + v)
\leq \dim(\operatorname{im} u + \operatorname{im} v)
\leq \operatorname{rk} u + \operatorname{rk} v
\]
(غراسمان، \cref{thm:b1:findim:grassmann}). والحاصر الأدنى: طبّق
\omterm{def:b1:reals:bounds}{الحاصر الأعلى} على الزوج $(u + v, -v)$، ومجموعه $u$:
\[
\operatorname{rk} u \leq \operatorname{rk}(u + v) +
\operatorname{rk}(-v) = \operatorname{rk}(u + v) +
\operatorname{rk} v,
\]
ومنه $\operatorname{rk} u - \operatorname{rk} v \leq
\operatorname{rk}(u+v)$؛ وبتبادل $u$ و $v$ نجد القيمة
المطلقة.
\end{solution}

\begin{solution}{exo:b1:linmaps:12}
\emph{الصيغة المضبوطة.} ليكن $v'$ قصر $v$ على
\omterm{def:b1:vspaces:subspace}{الفضاء الجزئي} $\operatorname{im} w$. فصورته هي $v(w(F)) =
\operatorname{im}(v \circ w)$، ونواته هي $\ker v \cap
\operatorname{im} w$. ومبرهنة \omterm{def:b1:linmaps:rank}{الرتبة} من أجل $v'$ على الفضاء
$\operatorname{im} w$:
\[
\operatorname{rk} w = \dim\operatorname{im} w
= \operatorname{rk}(v \circ w) + \dim(\ker v \cap
\operatorname{im} w) .
\]

\emph{وفروبينيوس.} طبّق الصيغة المضبوطة مرتين، على $w$ وعلى
$w \circ u$:
\[
\operatorname{rk} w - \operatorname{rk}(vw)
= \dim\bigl(\ker v \cap \operatorname{im} w\bigr),
\qquad
\operatorname{rk}(wu) - \operatorname{rk}(vwu)
= \dim\bigl(\ker v \cap \operatorname{im}(wu)\bigr) .
\]
ولأن $\operatorname{im}(w \circ u) \subseteq \operatorname{im}
w$، يكون التقاطع الثاني محتوًى في الأول، وبُعده
ليس أكبر:
\[
\operatorname{rk}(wu) - \operatorname{rk}(vwu)
\;\leq\; \operatorname{rk} w - \operatorname{rk}(vw) ,
\]
وهذا يُعاد ترتيبه ليعطي متراجحة فروبينيوس. ومع $w =
\mathrm{id}_F$ (\omterm{def:b1:linmaps:rank}{والرتبة} $\dim F$، و $\operatorname{im}\,
\mathrm{id}_F = F$): $\operatorname{rk} v + \operatorname{rk} u
\leq \dim F + \operatorname{rk}(vu)$، وهي متراجحة سيلفستر ---
مبرهنًا عليها مرة أخرى، في صورة مصفوفات، في \cref{exo:b1:matrices:10}.
\end{solution}

\begin{solution}{pb:b1:linmaps:1}
\textbf{1.} $(\mathrm{id} - p)^2 = \mathrm{id} - 2p + p^2 =
\mathrm{id} - p$: أي مسقط. وإذا كان $y = x - p(x)$، فإن $p(y) =
p(x) - p^2(x) = 0$، وبالعكس يعطي $x \in \ker p$ أن $x =
(\mathrm{id} - p)(x)$: ومنه $\operatorname{im}(\mathrm{id} - p) = \ker
p$. و $(\mathrm{id} - p)(x) = 0 \iff p(x) = x \iff x \in
\operatorname{im} p$ (بالنقاط الصامدة،
\cref{thm:b1:linmaps:projchar}): $\ker(\mathrm{id} - p) =
\operatorname{im} p$.

\textbf{2.} $(\lambda\,\mathrm{id} + \mu p)^2 =
\lambda^2\,\mathrm{id} + (2\lambda\mu + \mu^2)\,p$. والزوج
$(\mathrm{id}, p)$ حرّ في $\mathcal{L}(E)$: إذ إن $p = c\,
\mathrm{id}$ سيعطي $c^2 = c$، ومنه $p = 0$ أو $\mathrm{id}$،
وهما مستبعدان. وبمطابقة المعاملات، يكون \omterm{def:b1:logic:map}{التطبيق} مسقطًا إذا وفقط إذا كان
$\lambda^2 = \lambda$ و $2\lambda\mu + \mu^2 = \mu$. ومن أجل
$\lambda = 0$: $\mu \in \{0, 1\}$. ومن أجل $\lambda = 1$: $\mu^2 +
\mu = 0$ و $\mu \in \{0, -1\}$. أي أربعة مساقط بالضبط في
المستوي: $0$ و $p$ و $\mathrm{id}$ و $\mathrm{id} - p$.

\textbf{3.} $(\lambda\,\mathrm{id} + \mu p)(\lambda'\,\mathrm{id}
+ \mu' p) = \lambda\lambda'\,\mathrm{id} + (\lambda\mu' +
\mu\lambda' + \mu\mu')\,p$: ومنه فالمستوي مستقر بالتركيب.
ولأن $p^k = p$ من أجل كل $k \geq 1$، فإنه من أجل $Q =
\sum_k a_k X^k$:
\[
Q(p) = a_0\,\mathrm{id} + \Bigl(\sum_{k \geq 1} a_k\Bigr) p
= Q(0)\,\mathrm{id} + \bigl(Q(1) - Q(0)\bigr)\,p .
\]

\textbf{4.} على $\operatorname{im} p$ (حيث يفعل $p$ فعل
\omterm{def:b1:logic:map}{التطبيق} المطابق)، يضرب $\lambda\,\mathrm{id} + \mu p$ في $\lambda
+ \mu$؛ وعلى $\ker p$، في $\lambda$. ولأن $E = \operatorname{im} p
\oplus \ker p$، يكون \omterm{def:b1:logic:map}{التطبيق} \omterm{def:b1:logic:inj}{تقابليًا} إذا وفقط إذا كان $\lambda \neq 0$ و
$\lambda + \mu \neq 0$. وبحلّ $\lambda\alpha = 1$ و
$\lambda\beta + \mu\alpha + \mu\beta = 0$ في قاعدة التركيب
في السؤال 3:
\[
(\lambda\,\mathrm{id} + \mu p)^{-1}
= \frac1\lambda\,\mathrm{id} -
\frac{\mu}{\lambda(\lambda + \mu)}\,p ,
\]
وفعله بالمقدار $1/\lambda$ على $\ker p$ وبالمقدار $1/(\lambda +
\mu)$ على $\operatorname{im} p$، كما يجب.

\textbf{5.} اكتب $F = \operatorname{im} p = \operatorname{im}
p'$. فمن أجل كل $x$، يكون $p'(x) \in F$ ويثبّت $p$ الفضاءَ $F$ نقطةً نقطة:
ومنه $p(p'(x)) = p'(x)$، أي $p\,p' = p'$؛ وبالتناظر $p'\,p =
p$. وحين يتقاسم إسقاطان صورتهما، يفصل المطبَّق
\emph{أولًا}: لأن مخرجه يقع أصلًا في $F$، حيث يفعل \omterm{def:b1:linmaps:projection}{الإسقاط}
الخارجي فعل \omterm{def:b1:logic:map}{التطبيق} المطابق ولا يغيّر شيئًا.

\textbf{6.} $(p + q)^2 = p^2 + pq + qp + q^2 = (p + q) + pq +
qp$، ومنه يفرض كون $p + q$ مسقطًا أن $pq + qp = 0$. وركّب من
اليسار مع $p$: $pq + pqp = 0$؛ ومن اليمين مع $p$: $pqp + qp
= 0$. وبالطرح، $pq = qp$؛ ثم $pq + qp = 2pq = 0$ و
المميّز ليس $2$: ومنه $pq = qp = 0$.

\textbf{7.} مع $pq = qp = 0$، يعطي النشر نفسه $(p +
q)^2 = p + q$. \omterm{def:b1:linmaps:kerim}{والصورة}: $\operatorname{im}(p + q) \subseteq
\operatorname{im} p + \operatorname{im} q$ دائمًا. وبالعكس،
من أجل $x \in \operatorname{im} p$: $q(x) = q(p(x)) = 0$، ومنه $(p +
q)(x) = p(x) = x$ و $x \in \operatorname{im}(p+q)$؛ وكذلك من أجل
$\operatorname{im} q$. والانفصال: يعطي $x \in \operatorname{im} p
\cap \operatorname{im} q$ أن $x = p(x) = p(q(x)) = 0$.
\omterm{def:b1:linmaps:kerim}{والنواة}: إذا كان $p(x) + q(x) = 0$، فإن \omterm{def:b1:logic:map}{تطبيق} $p$ يعطي $p(x) +
p(q(x)) = p(x) = 0$، \omterm{def:b1:logic:map}{وتطبيق} $q$ يعطي $q(x) = 0$: ومنه $\ker(p
+ q) = \ker p \cap \ker q$ (والاحتواء العكسي جليّ).

\textbf{8.} يكون $p - q$ مسقطًا إذا وفقط إذا كان $\mathrm{id} - (p - q) =
(\mathrm{id} - p) + q$ مسقطًا (بالسؤال 1 مرتين). وحسب السؤالين
6--7 مطبَّقين على المسقطين $\mathrm{id} - p$ و $q$، يصحّ هذا
إذا وفقط إذا كان $(\mathrm{id} - p)q = q(\mathrm{id} - p) = 0$، أي
إذا وفقط إذا كان $pq = q$ و $qp = q$.

\textbf{9.} $pq = q$ يعني أن $p$ يثبّت كل $q(x)$، أي
$\operatorname{im} q \subseteq \ker(p - \mathrm{id}) =
\operatorname{im} p$. و $qp = q$ يعني $q\bigl((\mathrm{id} -
p)(x)\bigr) = 0$ من أجل كل $x$، أي أن $q$ ينعدم على
$\operatorname{im}(\mathrm{id} - p) = \ker p$: ومنه $\ker p \subseteq
\ker q$. والخطوتان تكافؤان: فالترتيب $q \leq p$ يقول إن
$q$ يسقط على \omterm{def:b1:linmaps:kerim}{صورة} أصغر، على امتداد \omterm{def:b1:linmaps:kerim}{نواة} أكبر.

\textbf{10.} بالنشر، $(\mathrm{id} - p)(\mathrm{id} - q) =
\mathrm{id} - p - q + pq = \mathrm{id} - r$. والمسقطان
$\mathrm{id} - p$ و $\mathrm{id} - q$ يتبادلان، ومنه، حسب
\cref{exo:b1:linmaps:6}، يكون جداؤهما $\mathrm{id} - r$ هو
المسقط على $\operatorname{im}(\mathrm{id} - p) \cap
\operatorname{im}(\mathrm{id} - q) = \ker p \cap \ker q$ على امتداد
$\ker(\mathrm{id} - p) + \ker(\mathrm{id} - q) =
\operatorname{im} p + \operatorname{im} q$. وحسب السؤال 1، يكون $r =
\mathrm{id} - (\mathrm{id} - r)$ عندئذ المسقط مع
$\operatorname{im} r = \operatorname{im} p + \operatorname{im}
q$ و $\ker r = \ker p \cap \ker q$.

\textbf{11.} المقدار $p_i$ معرَّف جيدًا (بوحدانية
التفكيك) وخطي (لأن تفكيك $x + \lambda y$
هو مجموع التفكيكين، بالوحدانية مرة أخرى). ومن أجل $x_i
\in F_i$ يكون التفكيك $x_i$ نفسه، ومنه $p_i(x_i) = x_i$:
أي $p_i^2 = p_i$، و $p_j(x_i) = 0$ من أجل $j \neq i$: ومنه $p_i p_j = 0$
(لأن $p_j(x) \in F_j$). وبجمع المركّبات، $\sum_i p_i =
\mathrm{id}$. وأخيرًا $\operatorname{im} p_i = F_i$ و $\ker
p_i = \bigoplus_{j \neq i} F_j$.

\textbf{12.} $p_i = p_i \circ \mathrm{id} = p_i\sum_j p_j =
p_i^2 + \sum_{j \neq i} p_i p_j = p_i^2$: فكل $p_i$
مسقط. وكل $x = \mathrm{id}(x) = \sum_i p_i(x)$ يقع في
$\sum_i \operatorname{im} p_i$: ومنه تُجمع الصور إلى $E$.
والانفصال: افترض $y_1 + \dots + y_k = 0$ مع $y_i \in
\operatorname{im} p_i$، ومنه $p_i(y_i) = y_i$. وطبّق $p_j$: $p_j
(y_i) = p_j p_i (y_i) = 0$ من أجل $i \neq j$، ومنه $0 = p_j\bigl(\sum
y_i\bigr) = y_j$ من أجل كل $j$. ومنه $E = \bigoplus_i
\operatorname{im} p_i$.

\textbf{13.} $q = \mathrm{id} - p$، ويعطي السؤال 1 $pq =
p - p^2 = 0 = qp$ مباشرة: فمن أجل مسقطين، يفرض كون مجموعهما
التطبيقَ المطابق تعامدَ الزوج أصلًا.

\textbf{14.} $p + q = \mathrm{id} - r$ مع $r$ مسقطًا، و
$(\mathrm{id} - r)$ مسقط (السؤال 1): ومنه فالمقدار $p + q$
مسقط، ويعطي السؤال 6 أن $pq = qp = 0$. وبالتناظر ($q +
r = \mathrm{id} - p$ و $p + r = \mathrm{id} - q$)، تنعدم كل
الجداءات مثنى مثنى، ويختم السؤال 12: $E =
\operatorname{im} p \oplus \operatorname{im} q \oplus
\operatorname{im} r$، تلقائيًا.

\textbf{15.} \emph{مبرهنة مساعدة.} بالاستقراء مع غراسمان
(\cref{thm:b1:findim:grassmann}):
\[
\dim(F_1 + \dots + F_k) \leq \dim(F_1 + \dots + F_{k-1}) + \dim
F_k \leq \dots \leq \sum_i \dim F_i .
\]
وإذا كان المجموع الكلي مساواةً، فكل خطوة كذلك: $(F_1 + \dots +
F_{j-1}) \cap F_j = \{0\}$ من أجل كل $j$، وتنهار علاقة $y_1 +
\dots + y_k = 0$ ($y_i \in F_i$) من اليمين: $y_k
\in (F_1 + \dots + F_{k-1}) \cap F_k = \{0\}$، ثم $y_{k-1} =
0$، وهكذا: \omterm{def:b1:vspaces:sum}{فالمجموع مباشر}. وبالعكس فللمجموع المباشر
أبعاد جمعية (بوصل الأسس). \emph{\omterm{def:b1:logic:map}{والتطبيق}:}
يبيّن $x = \sum_i p_i(x)$ أن $E = \sum_i \operatorname{im} p_i$،
ومنه $n \leq \sum_i \operatorname{rk} p_i$؛ ويعطي الفرض
المساواة، ومنه الانفصال. والجداءات: ثبّت $j$ و $y \in
\operatorname{im} p_j$. عندئذ $y = \sum_i p_i(y)$ مع $p_i(y)
\in \operatorname{im} p_i$، بينما $y = y$ تفكيكٌ
كذلك (بالمركّبة $j$ وحدها)؛ وتفرض الوحدانية أن $p_i(y)
= 0$ من أجل $i \neq j$. وبتطبيق ذلك على $y = p_j(x)$: $p_i p_j = 0$.

\textbf{16.} إذا كان $u^k(x) = 0$ فإن $u^{k+1}(x) = u(0) = 0$. و
$\operatorname{im} u^{k+1} = u^k\bigl(u(E)\bigr) \subseteq
u^k(E) = \operatorname{im} u^k$.

\textbf{17.} افترض $\ker u^{r} = \ker u^{r+1}$ وليكن $x \in
\ker u^{r+2}$: عندئذ $u(x) \in \ker u^{r+1} = \ker u^{r}$، ومنه
$u^{r+1}(x) = 0$: أي $x \in \ker u^{r+1}$. ومع السؤال 16،
$\ker u^{r+1} = \ker u^{r+2}$، وبالاستقراء تتطابق كل
النوى اللاحقة مع $\ker u^{r}$. وأمّا الصور: فتعطي مبرهنة \omterm{def:b1:linmaps:rank}{الرتبة}
$\dim\operatorname{im} u^k = n - \dim\ker u^k$، ومنه تتجمّد أبعاد
الصور بالضبط حين تتجمّد أبعاد النوى،
ومع احتواءات السؤال 16، يعني تساوي الأبعاد
تساوي الفضاءات الجزئية (\cref{thm:b1:findim:subspaces}).

\textbf{18.} المتتالية $\bigl(\dim\ker u^k\bigr)_k$
غير متناقصة وقيمها في $\intint{0}{n}$؛ ولا يمكنها أن تتزايد
تزايدًا قطعيًا $n + 1$ مرة، ومنه فإن $\dim\ker u^{r} = \dim\ker
u^{r+1}$ ما مع $r \leq n$، ومنه $\ker u^{r} = \ker u^{r+1}$
(بالاحتواء مع تساوي البُعد). وخذ $r$ أصغريًا؛ ويجمّد السؤال 17
كل شيء ابتداءً من $r$، بما فيه الصور.

\textbf{19.} التقاطع: ليكن $x \in \ker u^{r} \cap
\operatorname{im} u^{r}$، ولنقل $x = u^{r}(y)$ مع $u^{r}(x) =
0$. عندئذ $u^{2r}(y) = 0$، و $\ker u^{2r} = \ker u^{r}$
(السؤال 17)، ومنه $x = u^{r}(y) = 0$. والأبعاد: تعطي مبرهنة \omterm{def:b1:linmaps:rank}{الرتبة}
من أجل $u^{r}$ أن $\dim\ker u^{r} + \dim\operatorname{im} u^{r}
= n$؛ ومع التقاطع البديهي، يجعل غراسمان المجموعَ
\omterm{def:b1:vspaces:subspace}{فضاءً جزئيًا} بُعده $n$: ومنه $E = \ker u^{r} \oplus
\operatorname{im} u^{r}$.

\textbf{20.} الاستقرار: $u^{r}(u(x)) = u(u^{r}(x)) = 0$ من أجل $x
\in \ker u^{r}$؛ و $u(u^{r}(y)) = u^{r}(u(y)) \in
\operatorname{im} u^{r}$. وعلى $N = \ker u^{r}$: $(u|_N)^{r} = 0$
بحكم تعريف $N$: فهو \omterm{pb:b1:linmaps:1}{معدوم القوى}. وعلى $I = \operatorname{im}
u^{r}$: $\ker(u|_I) = \ker u \cap I \subseteq \ker u^{r} \cap I
= \{0\}$، ومنه فالمقدار $u|_I$ تشاكل ذاتي متباين للفضاء
$I$ ذي البُعد المنتهي، ومنه تقابلي
(\cref{cor:b1:linmaps:samedim}).

\textbf{21.} ليكن $x = a + b$ مع $a \in N$ و $b \in I$. عندئذ
$u(x) = u(a) + u(b)$ مع $u(a) \in N$ و $u(b) \in I$
(السؤال 20): وهذا \emph{هو} تفكيك $u(x)$، ومنه
$\pi(u(x)) = u(a) = u(\pi(x))$: أي $\pi u = u\pi$.

\textbf{22.} $u^2(x,y,z) = u(y, 0, z) = (0, 0, z)$ و
$u^3(x,y,z) = u(0,0,z) = (0,0,z) = u^2(x,y,z)$. والنوى: $\ker u
= \{y = z = 0\} = \operatorname{Vect}(e_1)$ و $\ker u^2 = \{z =
0\} = \operatorname{Vect}(e_1, e_2)$ و $\ker u^3 = \ker u^2$:
فالاستقرار عند $r = 2$. والصور: $\operatorname{im} u =
\operatorname{Vect}(e_1, e_3)$ و $\operatorname{im} u^2 =
\operatorname{Vect}(e_3)$. وفيتينغ: $\R^3 =
\operatorname{Vect}(e_1, e_2) \oplus \operatorname{Vect}(e_3)$،
و $\pi(x, y, z) = (x, y, 0)$. وللتحقق: $\pi u(x,y,z) = \pi(y, 0,
z) = (y, 0, 0)$ و $u\pi(x,y,z) = u(x, y, 0) = (y, 0, 0)$:
فهما متساويان. وعلى العامل الأول $u(x, y, 0) = (y, 0, 0)$، ومربعه
$0$: فهو \omterm{pb:b1:linmaps:1}{معدوم القوى}؛ وعلى الثاني $u(0,0,z) = (0,0,z)$:
أي \omterm{def:b1:logic:map}{التطبيق} المطابق، وهو تقابلي.

\textbf{23.} إذا كان $u^m = 0$ فإن $\ker u^m = E$؛ ولأن النوى
مجمَّدة ابتداءً من $r$، $\ker u^{r} = \ker u^{\max(m, r)} = E$.
وبالعكس يعني $\ker u^{r} = E$ أن $u^{r} = 0$. و $\ker u^{r} =
E \iff$ يكون مسقط فيتينغ على $E$ على امتداد $\{0\}$، أي
$\pi = \mathrm{id}$. وأخيرًا يعطي $r \leq n$ (السؤال 18):
أن كل تشاكل ذاتي \omterm{pb:b1:linmaps:1}{معدوم القوى} يحقق $u^{n} = 0$ --- فدليل
انعدام القوى لا يتجاوز البُعد أبدًا.

\textbf{24.} ليكن $m$ دليل انعدام قوى للمقدار $u|_A$: $A
\subseteq \ker u^{m} \subseteq \ker u^{\max(m,r)} = \ker u^{r}$.
ولأن $u|_B$ تقابلي، $B = u(B) = u^{k}(B) \subseteq
\operatorname{im} u^{k}$ من أجل كل $k$، وعلى الخصوص $B
\subseteq \operatorname{im} u^{r}$. عندئذ
\[
n = \dim A + \dim B \leq \dim\ker u^{r} +
\dim\operatorname{im} u^{r} = n :
\]
والاحتواءان مساواتان في الأبعاد، ومنه في
الفضاءات الجزئية: $A = \ker u^{r}$ و $B = \operatorname{im} u^{r}$.

\textbf{25.} (أ) الجزء 3 قاموس: فالانقسامات $E = F_1
\oplus \dots \oplus F_k$ تقابل بالضبط عائلات
المساقط ذات $\sum p_i = \mathrm{id}$ و $p_i p_j = 0$، وتكون
المقادير $F_i$ هي الصور. (ب) ومن أجل $k = 3$ تكون المتممات
$\mathrm{id} - p_i$ نفسها مساقط، وهذا أغلق
الحجة دون أيّ فرض إضافي؛ وأمّا من أجل $k$ عام فيلزم
$\sum_i \operatorname{rk} p_i \leq n$، وهي متراجحة يعطيها
أثر السنة~الثانية بالمجّان ($\operatorname{tr} p = \operatorname{rk}
p$ من أجل مسقط، والآثار تُجمع إلى $\operatorname{tr}
\mathrm{id} = n$). (ج) و \cref{exo:b1:linmaps:7} هي مبرهنة
فيتينغ المساعدة في الحالة المستقرة أصلًا $r \leq 1$؛ وفي العموم
يُترك تجمّدُ سلسلتَي \omterm{def:b1:linmaps:kerim}{النواة} \omterm{def:b1:linmaps:kerim}{والصورة}، وهو يستغرق $n$ خطوة
على الأكثر. (د) وفي مجلّد السنة~الثانية، مطبَّقةً على $u - \lambda\,
\mathrm{id}$، يصير العامل \omterm{pb:b1:linmaps:1}{المعدوم القوى} الفضاءَ
الذاتي المعمَّم عند $\lambda$ وتصير مساقط الجزء 3
المساقطَ الطيفية في نظرية التقطير. ومبرهنة
الجزء 4 هي \emph{مبرهنة فيتينغ المساعدة}.
\end{solution}
