\section*{Capítulo \ref{ch:b1:findim} --- Dimensión finita}

\begin{solution}{exo:b1:findim:1}
\begin{enumerate}
  \item $z = -x - y$: $F = \{(x, y, -x-y)\} =
        \operatorname{Vect}\bigl((1,0,-1), (0,1,-1)\bigr)$; los dos
        vectores son \omterm{def:b1:vspaces:free}{libres} (\omterm{prop:b1:vspaces:coordinates}{coordenadas}): $\dim F = 2$.
  \item $G = \{(x, x, 2t, t)\} =
        \operatorname{Vect}\bigl((1,1,0,0), (0,0,2,1)\bigr)$:
        \omterm{def:b1:vspaces:free}{libre}, $\dim G = 2$.
  \item $P(1) = P'(1) = 0$ significa $(X-1)^2 \mid P$
        (\cref{prop:b1:poly:multiplicity}): $P = (X-1)^2(aX + b)$.
        \omterm{def:b1:vspaces:free}{Base} $\bigl((X-1)^2, X(X-1)^2\bigr)$, dimensión $2$.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:findim:2}

\end{solution}

\begin{solution}{exo:b1:findim:3}
Pruébese a añadir $e_1 = (1,0,0,0)$ y $e_3 = (0,0,1,0)$. La familia
$\bigl((1,1,0,0), (0,0,1,1), e_1, e_3\bigr)$ es \omterm{def:b1:vspaces:free}{libre}: una
combinación nula $\alpha(1,1,0,0) + \beta(0,0,1,1) + \gamma e_1 +
\delta e_3 = 0$ dice $(\alpha + \gamma, \alpha, \beta + \delta,
\beta) = 0$, luego $\alpha = \beta = 0$ y después
$\gamma = \delta = 0$. Cuatro vectores \omterm{def:b1:vspaces:free}{libres} en dimensión $4$: una
\omterm{def:b1:vspaces:free}{base} (\cref{prop:b1:findim:twoofthree}).
\end{solution}

\begin{solution}{exo:b1:findim:4}
Grados $0, 1, 2, 3$ distintos dos a dos: \omterm{def:b1:vspaces:free}{libres}
(\cref{prop:b1:vspaces:freecriteria}), y son cuatro vectores en
dimensión $4$: \omterm{def:b1:vspaces:free}{base}. Para $X^3$, desarróllese hacia abajo,
\[
X(X-1)(X-2) = X^3 - 3X^2 + 2X,
\qquad
X(X-1) = X^2 - X,
\]
luego $X^3 - X(X-1)(X-2) = 3X^2 - 2X$; y
$3X^2 - 2X = 3(X^2 - X) + X = 3\,X(X-1) + X$. Por tanto,
\[
X^3 = X(X-1)(X-2) + 3\,X(X-1) + 1\cdot X + 0\cdot 1 :
\]
\omterm{prop:b1:vspaces:coordinates}{coordenadas} $(0, 1, 3, 1)$ sobre $\bigl(1, X, X(X-1),
X(X-1)(X-2)\bigr)$. (Son números de Stirling disfrazados.)
\end{solution}

\begin{solution}{exo:b1:findim:5}
Grassmann: $\dim(F \cap G) = 4 + 5 - \dim(F + G)$, y $F + G$ es un
\omterm{def:b1:vspaces:subspace}{subespacio} de $E$ que contiene a $G$: $5 \leq \dim(F+G) \leq 7$. Por
tanto, $\dim(F \cap G) \in \{2, 3, 4\}$. Realizaciones en $\R^7$ con
la \omterm{def:b1:vspaces:free}{base} canónica $(e_1, \dots, e_7)$, tomando
$G = \operatorname{Vect}(e_1, \dots, e_5)$:
\begin{itemize}
  \item $F = \operatorname{Vect}(e_1, e_2, e_6, e_7)$: $F + G = \R^7$,
        intersección $\operatorname{Vect}(e_1, e_2)$, dimensión $2$;
  \item $F = \operatorname{Vect}(e_1, e_2, e_3, e_6)$: intersección
        de dimensión $3$;
  \item $F = \operatorname{Vect}(e_1, e_2, e_3, e_4) \subseteq G$:
        dimensión $4$.
\end{itemize}
\end{solution}

\begin{solution}{exo:b1:findim:6}
$H$ es un \omterm{def:b1:vspaces:subspace}{subespacio} (el~\cref{exo:b1:vspaces:1}~(4) al pie de la
letra). Por el teorema del factor (\cref{thm:b1:poly:factor}),
$P \in H \iff P = (X-1)Q$ con $\deg Q \leq n - 1$: la \omterm{def:b1:logic:map}{aplicación}
$Q \mapsto (X-1)Q$ es una \omterm{def:b1:logic:inj}{biyección} lineal de $\R_{n-1}[X]$ sobre
$H$, de modo que una \omterm{def:b1:vspaces:free}{base} de $H$ es
\[
\bigl((X-1),\ (X-1)X,\ (X-1)X^2,\ \dots,\ (X-1)X^{n-1}\bigr),
\qquad \dim H = n .
\]
Una recta suplementaria: $\operatorname{Vect}(1)$ (las constantes).
En efecto, $H \cap \operatorname{Vect}(1) = \{0\}$ (una constante no
nula no se anula en $1$) y las dimensiones suman $n + 1$: por
Grassmann, $H \oplus \operatorname{Vect}(1) = \R_n[X]$.
\end{solution}

\begin{solution}{exo:b1:findim:7}
Supresión: al quitar $u_p$, el \omterm{def:b1:vspaces:span}{subespacio generado} solo puede
encoger; y encoge a lo sumo una dimensión, ya que devolver $u_p$ a
una \omterm{def:b1:vspaces:free}{base} del \omterm{def:b1:vspaces:span}{subespacio generado} menor da una familia generadora
del mayor con a lo sumo un vector extra. Simétricamente, al añadir un
vector $v$: el nuevo \omterm{def:b1:vspaces:span}{subespacio generado} contiene al antiguo con a lo
sumo un generador extra, de modo que su dimensión es $r$ (si $v$ ya
estaba en el \omterm{def:b1:vspaces:span}{subespacio generado}) o $r + 1$ (si no, por
la~\cref{prop:b1:vspaces:freecriteria}~(2) una \omterm{def:b1:vspaces:free}{base} se extiende). Los
dos \omterm{def:b1:logic:statement}{enunciados} juntos dan la conclusión «el \omterm{def:b1:findim:rank}{rango} es
$1$-lipschitziano».
\end{solution}

\begin{solution}{exo:b1:findim:8}
A lo largo de una cadena estrictamente creciente, las dimensiones
crecen estrictamente ($F_i \subseteq F_{i+1}$, $F_i \neq F_{i+1}$ y
el~\cref{thm:b1:findim:subspaces}: la igualdad de dimensiones
forzaría la igualdad de los espacios). Así,
$\dim F_1 < \dim F_2 < \dots < \dim F_k$ es una sucesión
estrictamente creciente de enteros de $\intint{0}{n}$: a lo sumo
$n + 1$ valores, $k \leq n + 1$. Cadena máxima en $\R^n$:
\[
\{0\} \subsetneq \operatorname{Vect}(e_1) \subsetneq
\operatorname{Vect}(e_1, e_2) \subsetneq \dots \subsetneq \R^n ,
\]
de longitud exactamente $n + 1$.
\end{solution}

\begin{solution}{exo:b1:findim:9}
$F + G$ contiene estrictamente a $F$ (ya que $G \not\subseteq F$),
luego $\dim(F + G) = n$ y Grassmann da
$\dim(F \cap G) = (n-1) + (n-1) - n = n - 2$.

Afirmación general, por inducción sobre $k$: la intersección $I_k$ de
$k$ hiperplanos cumple $\dim I_k \geq n - k$. Cierto para $k = 1$.
Paso: $I_{k+1} = I_k \cap H_{k+1}$, y Grassmann dentro de $E$:
\[
\dim(I_k \cap H_{k+1}) = \dim I_k + (n - 1) - \dim(I_k + H_{k+1})
\geq \dim I_k + (n-1) - n \geq n - k - 1 . \qedhere
\]
\end{solution}

\begin{solution}{exo:b1:findim:10}
\begin{enumerate}
  \item La condición $u_{n+2} - 3u_{n+1} + 2u_n = 0$ es lineal y la
        cumple la sucesión nula: $E$ es un \omterm{def:b1:vspaces:subspace}{subespacio}. Por inducción,
        $u_0$ y $u_1$ determinan todos los $u_n$, y la dependencia es
        lineal (cada paso es una combinación lineal de los dos
        valores anteriores); recíprocamente, cada pareja $(a, b)$
        proviene de exactamente una sucesión de $E$ (defínase $u_n$
        por la recurrencia). Como en el~\cref{ex:b1:findim:dims}, $E$
        está parametrizado \omterm{def:b1:logic:inj}{biyectiva} y linealmente por
        $(u_0, u_1) \in \R^2$: $\dim E = 2$.
  \item Constantes: $3\cdot1 - 2\cdot1 = 1$. Geométrica:
        $3\cdot 2^{n+1} - 2\cdot 2^n = (6 - 2)2^n = 2^{n+2}$. Las dos
        están en $E$. Libertad: $a\cdot 1 + b\cdot 2^n = 0$ para todo
        $n$ da, en $n = 0$ y $n = 1$: $a + b = 0$, $a + 2b = 0$,
        luego $a = b = 0$. Dos vectores \omterm{def:b1:vspaces:free}{libres} en dimensión $2$: una
        \omterm{def:b1:vspaces:free}{base} (\cref{prop:b1:findim:twoofthree}).
  \item Resuélvase $a + b = 0$, $a + 2b = 1$: $b = 1$, $a = -1$,
        luego $u_n = 2^n - 1$ (la sucesión de Mersenne).
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:findim:11}
\begin{enumerate}
  \item Grassmann: $\dim(F \cap G) = \dim F + \dim G - \dim(F +
        G) \geq \dim F + \dim G - n > 0$, luego
        $F \cap G \neq \{0\}$. Con $n = 100$: $51 + 50 - 100 = 1 >
        0$, la intersección contiene una recta.
  \item Si $F \cap H = \{0\}$, la suma es directa y
        $\dim F + (n - 1) = \dim(F \oplus H) \leq n$, luego
        $\dim F \leq 1$. (Recíprocamente, una recta no contenida en
        $H$ sí lo cumple: los hiperplanos no se pierden casi nada.)
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:findim:12}
Inducción descendente sobre $d = \dim F = \dim G$, desde $d = n$
hasta $d = 0$. Si $d = n$: $F = G = E$ y sirve $S = \{0\}$.
Supóngase el \omterm{def:b1:logic:statement}{enunciado} para parejas de \omterm{def:b1:vspaces:subspace}{subespacios} de dimensión
$d + 1 \leq n$, y sea $\dim F = \dim G = d < n$.

Si $F = G$: tómese como $S$ un \omterm{def:b1:vspaces:sum}{suplementario} cualquiera de $F$
(\cref{thm:b1:findim:subspaces}). Si $F \neq G$: los dos son propios,
luego por el~\cref{exo:b1:vspaces:12} existe $x \notin F \cup G$. Las
sumas $F' = F \oplus \operatorname{Vect}(x)$ y
$G' = G \oplus \operatorname{Vect}(x)$ son directas ($x \notin F$,
$x \notin G$) y tienen dimensión $d + 1$; por la hipótesis de
inducción admiten un \omterm{def:b1:vspaces:sum}{suplementario} común $S'$:
$E = F' \oplus S' = G' \oplus S'$. Póngase
$S = \operatorname{Vect}(x) \oplus S'$ --- directa, porque
$S' \cap \operatorname{Vect}(x) \subseteq S' \cap F' = \{0\}$.

Entonces $F + S = F + \operatorname{Vect}(x) + S' = F' + S' = E$, y
$F \cap S = \{0\}$: si $f = \lambda x + s'$ con $f \in F$,
$s' \in S'$, entonces $s' = f - \lambda x \in F' \cap S' = \{0\}$,
luego $f = \lambda x$, lo que fuerza $\lambda = 0$ ($x \notin F$) y
$f = 0$. Por tanto, $E = F \oplus S$ y, simétricamente,
$E = G \oplus S$.
\end{solution}

\begin{solution}{pb:b1:findim:1}
\textbf{1.} $\Q$ contiene $0 \neq 1$, es estable por la suma, el
producto y los opuestos, y todo racional no nulo tiene inverso
racional: es un \omterm{def:b1:structures:field}{cuerpo} (\cref{def:b1:structures:field}). Las
definiciones de \omterm{def:b1:vspaces:span}{subespacio generado}, de libertad y de \omterm{def:b1:vspaces:free}{base}, y las
demostraciones de los capítulos 18--19, usan solo la suma de
vectores, la distributividad y la aritmética de escalares en un
\omterm{def:b1:structures:field}{cuerpo} --- nunca un valor absoluto, un orden ni un límite. Así que
$\R$, con su propia suma y con la multiplicación
$\Q \times \R \to \R$, es un $\Q$-espacio vectorial, y todos los
teoremas se aplican. La división por un escalar aparece una vez en la
demostración del~\cref{thm:b1:findim:exchange}: para «despejar $g_k$»
se \omterm{def:b1:arith:divides}{divide} por su coeficiente no nulo.

\textbf{2.} Si $a + b\sqrt2 = 0$ con $a, b \in \Q$ no ambos nulos:
$b \neq 0$ daría $\sqrt2 = -a/b \in \Q$, en contradicción con la
irracionalidad de $\sqrt2$ (\cref{exo:b1:arith:7} con $p = 2$); luego
$b = 0$ y después $a = 0$: \omterm{def:b1:vspaces:free}{libre} sobre $\Q$. Sobre $\R$, la relación
$\sqrt2\cdot 1 + (-1)\cdot\sqrt2 = 0$ es no trivial: ligada. Por
tanto, $\dim_\Q \Q(\sqrt2) = 2$, con \omterm{prop:b1:vspaces:coordinates}{coordenadas} únicas $(a, b)$.

\textbf{3.} $(a + b\sqrt2)(c + d\sqrt2) = (ac + 2bd) + (ad +
bc)\sqrt2 \in \Q(\sqrt2)$. Entonces
\[
\sigma(uv) = (ac + 2bd) - (ad + bc)\sqrt2
= (a - b\sqrt2)(c - d\sqrt2) = \sigma(u)\sigma(v),
\]
luego $N(uv) = uv\,\sigma(uv) = u\sigma(u)\,v\sigma(v) = N(u)N(v)$.
Si $N(u) = a^2 - 2b^2 = 0$ con $u \neq 0$: $b \neq 0$ daría
$(a/b)^2 = 2$, una raíz cuadrada racional de $2$; luego $b = 0$, y
entonces $a = 0$ y $u = 0$: contradicción. Por tanto,
$N(u) \neq 0$ para $u \neq 0$.

\textbf{4.} $u \cdot \dfrac{\sigma(u)}{N(u)} = \dfrac{N(u)}{N(u)}
= 1$, y $\sigma(u)/N(u) \in \Q(\sqrt2)$: todo elemento no nulo es
invertible dentro de $\Q(\sqrt2)$, que es, por tanto, un subcuerpo
de $\R$. Ejemplos: $N(3 + 2\sqrt2) = 9 - 8 = 1$, luego
\[
\frac1{3 + 2\sqrt2} = 3 - 2\sqrt2 ;
\qquad
N(1 + \sqrt2) = -1, \quad
\frac1{1 + \sqrt2} = \sqrt2 - 1 .
\]

\textbf{5.} Si $\sqrt6 = p/q$, entonces $6q^2 = p^2$; el exponente de
$2$ es $1 + 2v_2(q)$, impar, a la izquierda, y $2v_2(p)$, par, a la
derecha: imposible. Supóngase ahora $\sqrt3 = a + b\sqrt2$ con
$a, b \in \Q$. Elevando al cuadrado:
$3 = a^2 + 2b^2 + 2ab\sqrt2$. Si $ab \neq 0$, entonces
$\sqrt2 = (3 - a^2 - 2b^2)/(2ab) \in \Q$: imposible. Si $b = 0$:
$\sqrt3 = a \in \Q$, en contradicción con
el~\cref{exo:b1:arith:7} ($p = 3$). Y si $a = 0$: $\sqrt3 = b\sqrt2$
y, multiplicando por $\sqrt2$: $\sqrt6 = 2b \in \Q$: imposible. Así
que $\sqrt3 \notin \Q(\sqrt2)$.

\textbf{6.} Los productos de los vectores de la \omterm{def:b1:vspaces:free}{base} son
\[
\sqrt2\,\sqrt3 = \sqrt6,\quad
\sqrt2\,\sqrt6 = 2\sqrt3,\quad
\sqrt3\,\sqrt6 = 3\sqrt2,\quad
(\sqrt2)^2 = 2,\ (\sqrt3)^2 = 3,\ (\sqrt6)^2 = 6,
\]
todos en $M$. Un producto de dos elementos de $M$ se desarrolla por
bilinealidad en combinaciones racionales de estos: $M$ es estable por
la multiplicación.

\textbf{7.} Sea $x + y\sqrt3 = 0$ con $x, y \in K = \Q(\sqrt2)$. Si
$y \neq 0$, entonces $\sqrt3 = -x/y \in K$ (pregunta 4: $K$ es un
\omterm{def:b1:structures:field}{cuerpo}), en contradicción con la pregunta 5. Luego $y = 0$ y
después $x = 0$: $(1, \sqrt3)$ es \omterm{def:b1:vspaces:free}{libre} sobre $K$. Y genera:
$K + K\sqrt3 = \{(a + b\sqrt2) + (c + d\sqrt2)\sqrt3\} =
\operatorname{Vect}_\Q (1, \sqrt2, \sqrt3, \sqrt6) = M$. Por tanto,
$\dim_K M = 2$.

\textbf{8.} Una relación $a + b\sqrt2 + c\sqrt3 + d\sqrt6 = 0$ se
reagrupa como $(a + b\sqrt2) + (c + d\sqrt2)\sqrt3 = 0$ con
coeficientes en $K$; por la pregunta 7 los dos se anulan y, por la
pregunta 2, $a = b = 0$ y $c = d = 0$. Así que la familia es \omterm{def:b1:vspaces:free}{libre} y
$\dim_\Q M = 4$.

\textbf{9.} Todo $x \in M$ se escribe $x = \sum_j y_j f_j$ con
$y_j \in K$ (\omterm{def:b1:vspaces:free}{base} de $M$ sobre $K$), y cada
$y_j = \sum_i \lambda_{ij} e_i$ con $\lambda_{ij} \in \Q$ (\omterm{def:b1:vspaces:free}{base} de
$K$ sobre $\Q$); sustituyendo,
$x = \sum_{i,j} \lambda_{ij}\, e_i f_j$: los productos generan $M$
sobre $\Q$.

\textbf{10.} Supóngase $\sum_{i,j} \lambda_{ij}\, e_i f_j = 0$ con
$\lambda_{ij} \in \Q$. Reagrúpese:
$\sum_j \bigl(\sum_i \lambda_{ij} e_i\bigr) f_j = 0$, y las sumas
\omterm{def:b1:topology:closure}{interiores} están en $K$. Que $(f_j)$ sea \omterm{def:b1:vspaces:free}{libre} sobre $K$ da
$\sum_i \lambda_{ij} e_i = 0$ para cada $j$; y que $(e_i)$ sea \omterm{def:b1:vspaces:free}{libre}
sobre $\Q$ da entonces $\lambda_{ij} = 0$ para todos $i, j$.

\textbf{11.} Por las preguntas 9 y 10,
$(e_i f_j)_{i \leq m,\, j \leq n}$ es una \omterm{def:b1:vspaces:free}{base} de $M$ sobre $\Q$ con
$mn$ elementos:
\[
\dim_\Q M = m\,n = \dim_\Q K \cdot \dim_K M .
\]
Comprobación: $\dim_\Q K = 2$ y $\dim_K M = 2$ (preguntas 2 y 7) dan
$\dim_\Q M = 4$, que es la pregunta 8.

\textbf{12.} Los vectores $u a_1, \dots, u a_d$ están en $A$ (por la
estabilidad). Son \omterm{def:b1:vspaces:free}{libres}: si $\sum_i \lambda_i\, u a_i = 0$, entonces
$u \sum_i \lambda_i a_i = 0$ en $\R$, y $u \neq 0$ fuerza
$\sum_i \lambda_i a_i = 0$, luego $\lambda_i = 0$ (los $a_i$ forman
una \omterm{def:b1:vspaces:free}{base}). Así, $(u a_1, \dots, u a_d)$ es una \omterm{def:b1:vspaces:free}{familia libre} de $d$
vectores en $A$, con $\dim_\Q A = d$: una \omterm{def:b1:vspaces:free}{base}
(\cref{prop:b1:findim:twoofthree}). En particular, $1 \in A$ se
descompone como $1 = \sum_i \mu_i\, u a_i = u\,v$ con
$v = \sum_i \mu_i a_i \in A$: el inverso de $u$ está en $A$. Esto
recupera la pregunta 4 ($A = \Q(\sqrt2)$) y demuestra de golpe que
$M$ es un subcuerpo de $\R$ (con la pregunta 6).

\textbf{13.} Los $d + 1$ vectores $1, x, x^2, \dots, x^{d}$ están
todos en $A$ (estabilidad por productos); y una \omterm{def:b1:vspaces:free}{familia libre} de $A$
tiene a lo sumo $d$ vectores (\cref{prop:b1:findim:twoofthree}), así
que son ligados: hay racionales $\lambda_0, \dots, \lambda_d$, no
todos nulos, con $\sum_k \lambda_k x^k = 0$. El \omterm{def:b1:poly:def}{polinomio}
$P = \sum_k \lambda_k X^k$ es no nulo, tiene coeficientes
racionales, grado $\leq d$ y $P(x) = 0$.

\textbf{14.} $s^2 = 2 + 2\sqrt6 + 3 = 5 + 2\sqrt6$: \omterm{prop:b1:vspaces:coordinates}{coordenadas}
$(5, 0, 0, 2)$. Después,
\[
s^3 = s\,s^2 = (\sqrt2 + \sqrt3)(5 + 2\sqrt6)
= 5\sqrt2 + 2\sqrt{12} + 5\sqrt3 + 2\sqrt{18}
= 11\sqrt2 + 9\sqrt3 ,
\]
\omterm{prop:b1:vspaces:coordinates}{coordenadas} $(0, 11, 9, 0)$ (usando $\sqrt{12} = 2\sqrt3$,
$\sqrt{18} = 3\sqrt2$). Por último,
$s^4 = (s^2)^2 = 25 + 20\sqrt6 + 24 = 49 + 20\sqrt6$: \omterm{prop:b1:vspaces:coordinates}{coordenadas}
$(49, 0, 0, 20)$.

\textbf{15.} Supóngase $a + bs + cs^2 + ds^3 = 0$. Leyendo las cuatro
\omterm{prop:b1:vspaces:coordinates}{coordenadas} sobre $(1, \sqrt2, \sqrt3, \sqrt6)$:
\[
a + 5c = 0,\qquad b + 11d = 0,\qquad b + 9d = 0,\qquad 2c = 0 .
\]
Luego $c = 0$ y después $a = 0$; restando las ecuaciones centrales,
$2d = 0$ y después $b = 0$: $(1, s, s^2, s^3)$ es \omterm{def:b1:vspaces:free}{libre}. Cuatro
vectores \omterm{def:b1:vspaces:free}{libres} en $M$, de dimensión $4$: una \omterm{def:b1:vspaces:free}{base}, luego
$\operatorname{Vect}_\Q(1, s, s^2, s^3) = M$. De la pregunta 14,
$s^3 - 9s = 2\sqrt2$ y $11s - s^3 = 2\sqrt3$:
\[
\sqrt2 = \frac{s^3 - 9s}{2},
\qquad
\sqrt3 = \frac{11s - s^3}{2} .
\]

\textbf{16.} $s^4 - 10s^2 + 1 = (49 + 20\sqrt6) - 10(5 + 2\sqrt6) +
1 = 0$. Un \omterm{def:b1:poly:def}{polinomio mónico} de grado $\leq 3$ que se anulara en $s$
produciría una combinación nula no trivial de $(1, s, s^2, s^3)$, en
contradicción con la pregunta 15: $X^4 - 10X^2 + 1$ es de grado
mínimo. Sus raíces: $X^2 = 5 \pm 2\sqrt6 = (\sqrt3 \pm \sqrt2)^2$,
así que las cuatro raíces reales son $\pm(\sqrt3 + \sqrt2)$ y
$\pm(\sqrt3 - \sqrt2)$, es decir, $\pm\sqrt2 \pm \sqrt3$.

\textbf{17.} De $s^4 - 10s^2 + 1 = 0$: $s\,(10s - s^3) = 1$, luego
\[
\frac1s = 10s - s^3 = 10(\sqrt2 + \sqrt3) - (11\sqrt2 + 9\sqrt3)
= \sqrt3 - \sqrt2 ,
\]
en consonancia con $(\sqrt3 + \sqrt2)(\sqrt3 - \sqrt2) = 1$. Si
$a + b\sqrt2 + c\sqrt3 + d\sqrt6 = r \in \Q$, entonces
$(a - r) + b\sqrt2 + c\sqrt3 + d\sqrt6 = 0$ y la libertad
(pregunta 8) fuerza $b = c = d = 0$ (y $a = r$). Para
$\sqrt2 + \sqrt3 + \sqrt6$, las \omterm{prop:b1:vspaces:coordinates}{coordenadas} $(0, 1, 1, 1)$ no son de
esa forma: irracional --- un \omterm{def:b1:logic:statement}{enunciado} más fuerte que
el~\cref{exo:b1:reals:7}, obtenido sin ningún truco de elevar al
cuadrado.

\textbf{18.} Una raíz racional $p/q$ (irreducible) de $X^3 - 2$ cumple
$p^3 = 2q^3$, y el criterio del~\cref{exo:b1:poly:5} da $p \mid 2$,
$q \mid 1$: candidatas $\pm1, \pm2$, cuyos cubos son
$\pm1, \pm8 \neq 2$. No hay raíz racional; en particular,
$t = 2^{1/3} \notin \Q$, luego $(1, t)$ es \omterm{def:b1:vspaces:free}{libre} sobre $\Q$ (como en
la pregunta 2).

\textbf{19.} Supóngase $(1, t, t^2)$ ligada: algún $P \in \Q[X]$ no
nulo con $\deg P \leq 2$ cumple $P(t) = 0$; elíjase un $P$ así de grado
\emph{mínimo} $d \geq 1$. Por la pregunta 18, $d \neq 1$, luego
$d = 2$. División euclídea (\cref{thm:b1:poly:division}):
$X^3 - 2 = PQ + R$ con $Q, R \in \Q[X]$, $\deg R < 2$. Evaluando en
$t$: $0 = P(t)Q(t) + R(t) = R(t)$, luego $R$ se anula en $t$; y la
minimalidad de $d$ fuerza $R = 0$. Entonces $X^3 - 2 = PQ$ con
$\deg Q = 1$: la raíz racional de $Q$ es una raíz racional de
$X^3 - 2$, en contradicción con la pregunta 18. Por tanto,
$(1, t, t^2)$ es \omterm{def:b1:vspaces:free}{libre} y $\dim_\Q A = 3$. Estabilidad: $t^3 = 2$
reduce todo producto de $1, t, t^2$ a una combinación de ellos
($t\cdot t^2 = 2$, $t^2\cdot t^2 = 2t$); y $A$ contiene a $1$: por la
pregunta 12, $A$ es un subcuerpo de $\R$.

\textbf{20.} Supóngase $t \in M$. Entonces $t^2 \in M$ (estabilidad),
luego $A \subseteq M$, y $M$ es un \omterm{def:b1:vspaces:def}{espacio vectorial} sobre el \omterm{def:b1:structures:field}{cuerpo}
$A$: los axiomas son los de la aritmética de $\R$, restringidos. Y es
\omterm{def:b1:findim:def}{de dimensión finita} sobre $A$ (una familia $\Q$-generadora finita
genera a fortiori sobre $A \supseteq \Q$). La \omterm{pb:b1:findim:1}{ley de la torre} para
$\Q \subseteq A \subseteq M$ da
\[
4 = \dim_\Q M = \dim_\Q A \cdot \dim_A M = 3\,\dim_A M ,
\]
imposible: $3$ no \omterm{def:b1:arith:divides}{divide} a $4$. Luego $2^{1/3} \notin M$: ninguna
combinación racional de $1, \sqrt2, \sqrt3, \sqrt6$ es igual a
$2^{1/3}$.

\textbf{21.} $\Q(\sqrt2) \subseteq M$, luego $t \notin \Q(\sqrt2)$.
Si $t + \sqrt2 = r \in \Q$, entonces $t = r - \sqrt2 \in \Q(\sqrt2)$:
contradicción --- $2^{1/3} + \sqrt2$ es irracional. Y si
$t = a + b\sqrt3$, entonces
$t \in \operatorname{Vect}_\Q(1, \sqrt3) \subseteq M$: otra
contradicción.

\textbf{22.} $B$ es un \omterm{def:b1:vspaces:def}{espacio vectorial} sobre el \omterm{def:b1:structures:field}{cuerpo} $A$
(restricción de escalares), \omterm{def:b1:findim:def}{de dimensión finita} ya que $\dim_\Q B$
es finita. La \omterm{pb:b1:findim:1}{ley de la torre} para $\Q \subseteq A \subseteq B$ da
$\dim_\Q B = \dim_\Q A \cdot \dim_A B$: el factor de la izquierda
\omterm{def:b1:arith:divides}{divide}. Los subcuerpos de $M$ tienen, por tanto, $\Q$-dimensión
$1$, $2$ o $4$: la dimensión $1$ es $\Q$ mismo y la $4$ es $M$.

\textbf{23.} Sea $F \subseteq M$ un subcuerpo con $\dim_\Q F = 2$ y
sea $w \in F \setminus \Q$: $(1, w)$ es \omterm{def:b1:vspaces:free}{libre} y, por tanto, una
\omterm{def:b1:vspaces:free}{base} de $F$. Por la pregunta 13 ($d = 2$), $w^2 = \alpha + \beta w$
para ciertos racionales $\alpha, \beta$. Poniendo
$v = w - \beta/2 \in F \setminus \Q$:
\[
v^2 = w^2 - \beta w + \frac{\beta^2}4 = \alpha + \frac{\beta^2}4
\;\in\; \Q,
\]
y $F = \operatorname{Vect}(1, v)$. Escríbase ahora
$v = a + b\sqrt2 + c\sqrt3 + d\sqrt6$ y desarróllese:
\[
v^2 = (a^2 + 2b^2 + 3c^2 + 6d^2)
+ (2ab + 6cd)\sqrt2 + (2ac + 4bd)\sqrt3 + (2ad + 2bc)\sqrt6 .
\]
Las tres \omterm{prop:b1:vspaces:coordinates}{coordenadas} irracionales se anulan:
$ab + 3cd = ac + 2bd = ad + bc = 0$. Si $a \neq 0$: $b = -3cd/a$, y
la segunda condición se convierte en $c(a^2 - 6d^2) = 0$; y como
$a^2 = 6d^2$ con $d \neq 0$ haría racional $\sqrt6 = \abs{a/d}$, o
bien $c = 0$ o bien $d = 0$, y en los dos casos las condiciones
restantes fuerzan $b = c = d = 0$, es decir, $v \in \Q$: excluido.
Luego $a = 0$, y las condiciones dicen $3cd = 2bd = bc = 0$: a lo
sumo uno de $b, c, d$ es no nulo. Por tanto, $v$ es un múltiplo
racional de $\sqrt2$, $\sqrt3$ o $\sqrt6$, y
\[
F \in \bigl\{\Q(\sqrt2),\ \Q(\sqrt3),\ \Q(\sqrt6)\bigr\} ,
\]
cada uno de los cuales es, en efecto, un subcuerpo de dimensión $2$
(la estabilidad como en la pregunta 6, los inversos por la
pregunta 12): exactamente tres subcuerpos cuadráticos.

\textbf{24.} $(1 + \sqrt2 + \sqrt3)(1 + \sqrt2 - \sqrt3) =
(1 + \sqrt2)^2 - 3 = 2\sqrt2$, luego
\[
\frac1{1 + \sqrt2 + \sqrt3}
= \frac{1 + \sqrt2 - \sqrt3}{2\sqrt2}
= \frac{\sqrt2 + 2 - \sqrt6}{4}
= \frac12 + \frac14\sqrt2 + 0\cdot\sqrt3 - \frac14\sqrt6 .
\]
(Compruébese: $(1 + \sqrt2 + \sqrt3)(2 + \sqrt2 - \sqrt6) = 4$ tras
desarrollar.)

\textbf{25.} (i) La $\Q$-dimensión asocia un \emph{entero} a cada
subcuerpo de $\R$, y la \omterm{pb:b1:findim:1}{ley de la torre} hace que esos enteros se
multipliquen a lo largo de las inclusiones: la dimensión se comporta
como un invariante aritmético, y las restricciones de
\omterm{def:b1:arith:divides}{divisibilidad} se convierten en demostraciones de imposibilidad.
(ii) La \omterm{def:b1:findim:def}{dimensión finita} obliga a que todo elemento cumpla una
ecuación polinómica racional no nula de grado a lo sumo la
dimensión: finitud significa algebraicidad. (iii) La regla de dos de
tres convirtió «multiplicar por $u$ lleva una \omterm{def:b1:vspaces:free}{base} a una
\omterm{def:b1:vspaces:free}{familia libre} de tamaño máximo» en \omterm{def:b1:logic:inj}{sobreyectividad}, produciendo
$u^{-1}$ sin ninguna fórmula: los inversos salieron de contar.
(iv) La \omterm{pb:b1:findim:1}{ley de la torre} prohíbe un \omterm{def:b1:structures:field}{cuerpo} de dimensión $3$ dentro de
uno de dimensión $4$, y por eso $2^{1/3}$ no se puede alcanzar desde
$\sqrt2$ y $\sqrt3$. El teorema de la parte III es la \emph{\omterm{pb:b1:findim:1}{ley de
la torre} de Dedekind}, el movimiento de apertura de la teoría de
Galois, desarrollada en el volumen del tercer año.
\end{solution}
