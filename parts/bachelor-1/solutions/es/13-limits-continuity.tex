\section*{Capítulo \ref{ch:b1:continuity} --- Límites y continuidad}

\begin{solution}{exo:b1:continuity:1}
Tómense $u_n = \frac{1}{2\pi n + \pi/2}$ y $v_n = \frac{1}{2\pi n}$:
los dos tienden a $0^+$ y, sin embargo, $\sin\frac{1}{u_n} = 1$ y
$\sin\frac{1}{v_n} = 0$. Dos sucesiones, dos límites distintos de las
imágenes: por el~\cref{thm:b1:continuity:seqchar}, no hay límite en
$0^+$.

$x \sin\frac1x$: emparedado por $\abs{x\sin\frac1x} \leq \abs x \to 0$,
luego el límite en $0$ existe y vale $0$.
\end{solution}

\begin{solution}{exo:b1:continuity:2}
$f = \lfloor\cdot\rfloor$ es \omterm{def:b1:continuity:continuous}{continua} en $\R \setminus \Z$ (localmente
constante) y discontinua en cada $n \in \Z$: límite por la izquierda
$n - 1$, valor $n$.

$g(x) = x - \lfloor x\rfloor$: los mismos puntos de discontinuidad (la
identidad es \omterm{def:b1:continuity:continuous}{continua}, luego $g$ hereda los saltos de $f$); en
$n \in \Z$, el límite por la izquierda es $1 \neq 0 = g(n)$.

$h(x) = \lfloor x\rfloor + (x - \lfloor x\rfloor)^2$: en
$\intco{n}{n + 1}$ se tiene $h(x) = n + (x - n)^2$, \omterm{def:b1:continuity:continuous}{continua} ahí; y en
$x = n$ el límite por la izquierda es $(n-1) + 1 = n = h(n)$: los
saltos se cancelan. $h$ es \omterm{def:b1:continuity:continuous}{continua} en $\R$ (y estrictamente creciente).
\end{solution}

\begin{solution}{exo:b1:continuity:3}
$P(x) = x^5 - 3x + 1$: $P(-2) = -32 + 6 + 1 = -25 < 0$; $P(0) = 1 > 0$;
$P(1) = -1 < 0$; $P(2) = 27 > 0$. Tres cambios de signo en los
segmentos disjuntos $\intcc{-2}{0}$, $\intcc{0}{1}$, $\intcc{1}{2}$:
por el~\cref{thm:b1:continuity:ivt}, al menos tres raíces. (Siendo de
grado $5$, $P$ tiene a lo sumo cinco; un estudio de las variaciones
mostraría que son exactamente tres.)
\end{solution}

\begin{solution}{exo:b1:continuity:4}
Sea $P = a_{2m+1}X^{2m+1} + \dots$ con $a_{2m+1} > 0$ (si no,
sustitúyase $P$ por $-P$). Sacando factor común el término dominante,
$P(x) = a_{2m+1} x^{2m+1}\bigl(1 + o(1)\bigr)$ cuando
$x \to \pm\infty$: luego $P(x) \to +\infty$ en $+\infty$ y a $-\infty$
en $-\infty$. Tómense $a$ con $P(a) < 0$ y $b$ con $P(b) > 0$: el
teorema del valor intermedio en $\intcc{a}{b}$ proporciona una raíz.
\end{solution}

\begin{solution}{exo:b1:continuity:5}
Sea $g(x) = f(x) - x$, \omterm{def:b1:continuity:continuous}{continua} en $\intcc{0}{1}$. Como $f$ aplica
dentro de $\intcc{0}{1}$: $g(0) = f(0) \geq 0$ y
$g(1) = f(1) - 1 \leq 0$. Por el~\cref{thm:b1:continuity:ivt},
$g(c) = 0$ para algún $c$: un punto fijo.

Necesidad de las hipótesis: en $\intcc{0}{1}$, la \omterm{def:b1:logic:map}{aplicación}
discontinua $f(x) = 1$ para $x \leq \frac12$ y $f(x) = 0$ para
$x > \frac12$ no tiene punto fijo; en el \omterm{prop:b1:reals:intervals}{intervalo} $\intoo{0}{1}$ (que
no es un segmento), $f(x) = \frac x2$ es \omterm{def:b1:continuity:continuous}{continua} con valores en
$\intoo{0}{1}$ y sin punto fijo (falta el candidato $0$); y en $\R$,
$f(x) = x + 1$.
\end{solution}

\begin{solution}{exo:b1:continuity:6}
$g(x) = \cos x - x$ es \omterm{def:b1:continuity:continuous}{continua}, $g(0) = 1 > 0$ y
$g(1) = \cos 1 - 1 < 0$: existe una solución en $\intoo{0}{1}$
(\cref{thm:b1:continuity:ivt}). Unicidad: $g$ es estrictamente
decreciente en $\R$ --- para $x \leq 0$, $g(x) \geq 1 - x > 0$ no
tiene ningún cero de todos modos; y $g'(x) = -\sin x - 1 \leq 0$ con
igualdad solo en puntos aislados ($x \equiv -\frac\pi2 \bmod 2\pi$),
luego $g$ es estrictamente decreciente (\cref{ch:b1:derivative};
alternativamente: en $\intcc{0}{1}$, $\cos$ es estrictamente
decreciente y $-x$ también, luego $g$ lo es). Una función
estrictamente monótona se anula a lo sumo una vez.
\end{solution}

\begin{solution}{exo:b1:continuity:7}
Fíjese $M = f(0) + 1$. Existe $A > 0$ con $f(x) \geq M$ para
$\abs x \geq A$ (definición de los dos límites infinitos; tómese el
umbral mayor). En el segmento $\intcc{-A}{A}$, el teorema de los
valores extremos (\cref{thm:b1:continuity:evt}) proporciona $c$ con
$f(c) = \inf_{\intcc{-A}{A}} f \leq f(0)$. Y para $\abs x \geq A$:
$f(x) \geq M > f(0) \geq f(c)$. Luego $f(c)$ es el mínimo global.
\end{solution}

\begin{solution}{exo:b1:continuity:8}
En el segmento $\intcc{0}{T}$, $f$ está acotada y alcanza sus cotas
(\cref{thm:b1:continuity:evt}); por periodicidad, esas son las cotas
en todo $\R$, y se siguen alcanzando.

Sea $g(x) = f(x + \frac T2) - f(x)$, \omterm{def:b1:continuity:continuous}{continua}. Entonces
\[
g(0) + g\bigl(\tfrac T2\bigr)
= \bigl(f(\tfrac T2) - f(0)\bigr) + \bigl(f(T) - f(\tfrac T2)\bigr)
= f(T) - f(0) = 0 :
\]
$g(0)$ y $g(\frac T2)$ tienen signos opuestos (o uno de ellos se
anula), luego el teorema del valor intermedio en $\intcc{0}{T/2}$ da
$c$ con $g(c) = 0$, es decir, $f(c + \frac T2) = f(c)$.
\end{solution}

\begin{solution}{exo:b1:continuity:9}
Primero la desigualdad: para $0 \leq y \leq x$,
\[
\bigl(\sqrt y + \sqrt{x - y}\bigr)^2 = x + 2\sqrt{y(x-y)} \geq x,
\]
luego $\sqrt x \leq \sqrt y + \sqrt{x - y}$, es decir,
$\sqrt x - \sqrt y \leq \sqrt{x - y}$. Por tanto,
$\abs{\sqrt x - \sqrt y} \leq \sqrt{\abs{x - y}}$ para todos
$x, y \geq 0$.

\omterm{def:b1:continuity:uniform}{Continuidad uniforme}: dado $\varepsilon > 0$, tómese
$\delta = \varepsilon^2$; entonces $\abs{x - y} \leq \delta$ implica
$\abs{\sqrt x - \sqrt y} \leq \sqrt\delta = \varepsilon$.

No lipschitziana cerca de $0$:
$\frac{\sqrt x - \sqrt 0}{x - 0} = \frac{1}{\sqrt x} \to +\infty$
cuando $x \to 0^+$, luego ninguna constante $k$ puede dominar todos
los cocientes incrementales.
\end{solution}

\begin{solution}{exo:b1:continuity:10}
Supóngase $f$ \omterm{def:b1:logic:inj}{inyectiva}, \omterm{def:b1:continuity:continuous}{continua} y no estrictamente monótona.
Entonces hay $a < b < c$ en $I$ con $f(b)$ no comprendido entre $f(a)$
y $f(c)$ --- en efecto, si en \emph{todas} las ternas el valor central
estuviera entre los extremos, $f$ sería monótona (compárense dos
parejas cualesquiera; comprobación breve por casos). Digamos que
$f(b) > \max(f(a), f(c))$ (el otro caso es simétrico: sustitúyase $f$
por $-f$). Elíjase $v$ con $\max(f(a), f(c)) < v < f(b)$. Por el
teorema del valor intermedio aplicado en $\intcc{a}{b}$ y en
$\intcc{b}{c}$, hay $u_1 \in \intoo{a}{b}$ y $u_2 \in \intoo{b}{c}$
con $f(u_1) = v = f(u_2)$: dos puntos distintos con imágenes iguales,
en contradicción con la \omterm{def:b1:logic:inj}{inyectividad}.
\end{solution}

\begin{solution}{exo:b1:continuity:11}
$f(0) = f(0+0) = 2f(0)$ da $f(0) = 0$; y $f(-x) = -f(x)$ sale de
$0 = f(x - x)$. Póngase $\alpha = f(1)$. Inducción: $f(n) = n\alpha$
para $n \in \N$, y después para $n \in \Z$ por imparidad. Para
$q \in \N^*$: $q\,f(\frac pq) = f(p) = p\alpha$ (súmese $\frac pq$
consigo mismo $q$ veces), luego $f(\frac pq) = \alpha\frac pq$: se
tiene $f = \alpha\,\mathrm{id}$ en $\Q$.

Sea ahora $x \in \R$ y $(r_n)$ una sucesión de racionales con
$r_n \to x$ (\omterm{def:b1:topology:dense}{densidad}, \cref{thm:b1:reals:density}, aplicada en
\omterm{prop:b1:reals:intervals}{intervalos} encajados; o bien $r_n = \frac{\lfloor nx\rfloor}{n}$).
Por \omterm{def:b1:continuity:continuous}{continuidad}: $f(x) = \lim f(r_n) = \lim \alpha r_n = \alpha x$.
\end{solution}

\begin{solution}{exo:b1:continuity:12}
Sea $\varepsilon > 0$. Por el límite en $+\infty$, hay $A$ con
$\abs{f(x) - \ell} \leq \frac\varepsilon2$ para $x \geq A$; luego,
para $x, y \geq A$: $\abs{f(x) - f(y)} \leq \varepsilon$ (sin
necesidad de proximidad alguna).

En el segmento $\intcc{0}{A + 1}$, el teorema de Heine
(\cref{thm:b1:continuity:heine}) da $\delta_0 > 0$ para este
$\varepsilon$; tómese $\delta = \min(\delta_0, 1)$.

Tómense ahora $x, y \geq 0$ cualesquiera con $\abs{x - y} \leq \delta$,
digamos $x \leq y$. Si $y \leq A + 1$, los dos están en el segmento y
se aplica el $\delta_0$ de Heine. Si no, $y > A + 1$ y entonces
$x \geq y - 1 > A$: los dos están en $\intco{A}{+\infty}$, donde se
aplica el argumento del límite. En ambos casos,
$\abs{f(x) - f(y)} \leq \varepsilon$: \omterm{def:b1:continuity:uniform}{continuidad uniforme}.
\end{solution}

\begin{solution}{pb:b1:continuity:1}
\textbf{1.} Para $n \in \N$: $f(nx) = n f(x)$ por inducción
($f((n+1)x) = f(nx) + f(x)$). Además, $f(0) = 2f(0)$ da $f(0) = 0$,
y $0 = f(x - x) = f(x) + f(-x)$ da la imparidad, de modo que
$f(nx) = nf(x)$ para $n \in \Z$. Para $r = \frac pq$:
$q\,f\bigl(\tfrac pq x\bigr) = f(px) = p\,f(x)$, luego
$f(rx) = r f(x)$: $f$ es $\Q$-lineal.

\textbf{2.} La aditividad da, para todos $x$ y $h$:
\[
f(x + h) - f(x) = f(h) = f(x_0 + h) - f(x_0) .
\]
Cuando $h \to 0$, el miembro derecho tiende a $0$ por la
\omterm{def:b1:continuity:continuous}{continuidad} en $x_0$; luego $f(x + h) \to f(x)$: \omterm{def:b1:continuity:continuous}{continuidad} en todo
$x$. Y entonces el~\cref{exo:b1:continuity:11} da $f(x) = f(1)\,x$.

\textbf{3.} Dos funciones aditivas \omterm{def:b1:continuity:continuous}{continuas} son de la forma $cx$ y
$c'x$ (pregunta 2); si coinciden en el \omterm{def:b1:structures:subgroup}{subgrupo} \omterm{def:b1:topology:dense}{denso}
$\Z + \sqrt2\,\Z$ (\cref{exo:b1:reals:9}), entonces $c\,g = c'g$ para
algún $g \neq 0$ de él: $c = c'$ y las funciones son iguales. Sin
\omterm{def:b1:continuity:continuous}{continuidad}: la $\Q$-linealidad solo liga los valores en las
$\Q$-combinaciones, y $1, \sqrt2$ son $\Q$-independientes
($\sqrt2 \notin \Q$), de modo que $f(1) = 0$, $f(\sqrt2) = 1$ es
compatible y fuerza, sobre el \omterm{def:b1:structures:subgroup}{subgrupo},
\[
f(m + n\sqrt2) = m\,f(1) + n\,f(\sqrt2) = n .
\]

\textbf{4.} Para $t \in \intcc{0}{\ell}$: $a + t \in \intcc{a}{b}$,
luego $f(t) = f(a + t) - f(a) \leq M - f(a) =: M'$.

\textbf{5.} Para $t \in \intcc{0}{\ell}$, también
$\ell - t \in \intcc{0}{\ell}$, y la aditividad da
$f(t) = f(\ell) - f(\ell - t) \geq f(\ell) - M'$. Por tanto,
$\abs{f} \leq C$ en $\intcc{0}{\ell}$ con
$C = \max\bigl(\abs{M'}, \abs{f(\ell) - M'}\bigr)$.

\textbf{6.} Para $t \in \intcc{0}{\ell/n}$: $nt \in \intcc{0}{\ell}$
y $f(t) = \frac{f(nt)}{n}$ (pregunta 1), luego
$\abs{f(t)} \leq \frac Cn$. Dado $\varepsilon > 0$, elíjase
$n > \frac C\varepsilon$: para $0 \leq h \leq \frac\ell n$ se tiene
$\abs{f(h)} \leq \varepsilon$, y para $h$ negativo úsese
$f(h) = -f(-h)$. Así, $f(h) \to 0 = f(0)$ cuando $h \to 0$:
\omterm{def:b1:continuity:continuous}{continuidad} en $0$ y, por tanto, en todas partes (pregunta 2),
luego $f(x) = f(1)x$.

\textbf{7.} Si $f$ es no decreciente en $\intcc{a}{b}$, entonces
$f(a) \leq f(x) \leq f(b)$ ahí: está acotada, luego es lineal por la
pregunta 6, $f(x) = cx$; y $c(b - a) = f(b) - f(a) \geq 0$ obliga a
$c \geq 0$.

\textbf{8.} (a)$\Rightarrow$(b)$\Rightarrow$(c): trivial.
(c)$\Rightarrow$(a): pregunta 2. (a)$\Rightarrow$(d): una función
lineal es monótona en todas partes. (d)$\Rightarrow$(e): una función
monótona en $\intcc{a}{b}$ está acotada ahí por sus valores en los
extremos. (e)$\Rightarrow$(a): preguntas 4--6. Los cinco
\omterm{def:b1:logic:statement}{enunciados} son equivalentes --- la escalera de regularidad.

\textbf{9.} La pregunta 4 solo usó la \omterm{def:b1:reals:bounds}{cota superior} $M$; la
pregunta 5 \emph{dedujo} la cota inferior de la superior mediante la
reflexión $f(t) = f(\ell) - f(\ell - t)$; y la pregunta 6 corrió
sobre $\abs f \leq C$. Así pues: aditiva y acotada
\emph{superiormente} en un \omterm{prop:b1:reals:intervals}{intervalo} no degenerado ya implica lineal.
Para una cota inferior, aplíquese esto a $-f$ (aditiva y acotada
superiormente). Peldaño (e) más fuerte: basta una cota unilateral en
un \omterm{prop:b1:reals:intervals}{intervalo}.

\textbf{10.} Si $\frac{f(x)}{x}$ fuera una misma constante $c$ para
todo $x \neq 0$, $f$ sería lineal; luego hay $u, v$ no nulos con
$\frac{f(u)}u \neq \frac{f(v)}v$, es decir,
$u f(v) - v f(u) \neq 0$: el determinante de los vectores
$(u, f(u))$, $(v, f(v))$ es no nulo, y generan $\R^2$.

\textbf{11.} Para $r, s \in \Q$: $f(ru + sv) = r f(u) + s f(v)$
(pregunta 1 dos veces, más la aditividad), luego la gráfica contiene
\[
\bigl(ru + sv,\; r f(u) + s f(v)\bigr)
= r\,(u, f(u)) + s\,(v, f(v)) .
\]
Dados $(x_0, y_0)$ y $\varepsilon > 0$: el sistema $2 \times 2$
$a(u, f(u)) + b(v, f(v)) = (x_0, y_0)$ tiene una (única) solución
real $(a, b)$, pues el determinante es no nulo. Tómense racionales
$r_n \to a$, $s_n \to b$: entonces
$r_n(u, f(u)) + s_n(v, f(v)) \to (x_0, y_0)$ coordenada a coordenada,
y todos esos puntos están en la gráfica: la gráfica es \omterm{def:b1:topology:dense}{densa} en $\R^2$.

\textbf{12.} Sea $I$ un \omterm{prop:b1:reals:intervals}{intervalo} no degenerado, $x_0$ su punto
medio y $M$ arbitrario: la \omterm{def:b1:topology:dense}{densidad} proporciona un punto de la gráfica
a distancia menor que $\min\bigl(\frac{\abs I}{2}, 1\bigr)$ de
$(x_0, M + 1)$, es decir, un $x \in I$ con $f(x) > M$: no acotada en
$I$ y, por tanto (pregunta 8), discontinua en todo punto y monótona
en ningún \omterm{prop:b1:reals:intervals}{intervalo}; y, para cualquier objetivo $y_0$, los puntos de
la gráfica próximos a $(x_0, y_0)$ dan valores $f(x)$ arbitrariamente
cercanos a $y_0$ con $x \in I$: $f(I)$ es \omterm{def:b1:topology:dense}{denso} en $\R$. Esto es la
pregunta 8 leída al revés: como todos los peldaños son equivalentes,
una función aditiva no lineal tiene que fallarlos \emph{todos}, en
todas partes.

\textbf{13.} Está bien definida: $m + n\sqrt2 = m' + n'\sqrt2$ obliga
a $(n - n')\sqrt2 = m' - m \in \Z$, luego $n = n'$ (si no,
$\sqrt2 \in \Q$) y $m = m'$. La aditividad en $G$ es entonces clara
coordenada a coordenada. No acotación cerca de $0^+$: fíjense
$\varepsilon \in \intoo{0}{1}$ y $N \in \N$. Para cada $n$ fijo con
$\abs n \leq N$, la condición $m + n\sqrt2 \in \intoo{0}{1}$ encierra
$m$ en un \omterm{prop:b1:reals:intervals}{intervalo} de longitud $1$: a lo sumo un entero $m$ por cada
$n$, de modo que a lo sumo $2N + 1$ elementos de
$G \cap \intoo{0}{1}$ cumplen $\abs{f} \leq N$. Pero
$G \cap \intoo{0}{\varepsilon}$ es infinito ($G$ es \omterm{def:b1:topology:dense}{denso},
\cref{exo:b1:reals:9}); luego contiene algún $g$ con
$\abs{f(g)} > N$: $f$ no está acotada en ningún \omterm{def:b1:topology:open}{entorno} por la
derecha de $0$. Extender $f$ a una función aditiva en $\R$ significa
elegir valores de forma coherente sobre una familia de reales que sea
$\Q$-linealmente independiente y genere $\R$ sobre $\Q$ --- una base
de Hamel; producir una exige el axioma de elección, y ninguna fórmula
explícita puede hacerlo: constructivamente solo poseemos el monstruo
sobre $G$.

\textbf{14.} $f(x) = f(\frac x2 + \frac x2) = f(\frac x2)^2 \geq 0$.
Si $f(x_0) = 0$, entonces $f(x) = f(x - x_0)f(x_0) = 0$ para todo
$x$: excluido. Luego $f > 0$ y $g = \ln \circ f$ es \omterm{def:b1:continuity:continuous}{continua}
(\cref{prop:b1:continuity:algebra}) con $g(x + y) = g(x) + g(y)$: por
el~\cref{exo:b1:continuity:11}, $g(x) = cx$, luego $f(x) = \eu^{cx}$.
Recíprocamente, cada $\eu^{cx}$ sirve: los \omterm{def:b1:structures:morphism}{morfismos} \omterm{def:b1:continuity:continuous}{continuos}
$(\R, +) \to (\R^*, \times)$ son exactamente las exponenciales.

\textbf{15.} $g(u) = f(\eu^u)$ es \omterm{def:b1:continuity:continuous}{continua} y
$g(u + v) = f(\eu^u \eu^v) = g(u) + g(v)$: $g(u) = cu$, y todo
$x > 0$ se escribe $x = \eu^u$ con $u = \ln x$: $f(x) = c\ln x$.

\textbf{16.} $h = \ln \circ f$ es \omterm{def:b1:continuity:continuous}{continua} en $\intoo{0}{+\infty}$
con $h(xy) = h(x) + h(y)$: por la pregunta 15, $h(x) = c\ln x$, luego
$f(x) = \eu^{c\ln x} = x^c$.

\textbf{17.} $x = y = 0$: $f(0) = 2f(0) + f(0)^2$, luego
$f(0)(1 + f(0)) = 0$. Si $f(0) = -1$: haciendo $y = 0$,
$f(x) = f(x) + f(0) + f(x)f(0) = -1$ para todo $x$: la constante
$f \equiv -1$ (que, en efecto, cumple la ecuación). En caso
contrario, $f(0) = 0$; $h = 1 + f$ es \omterm{def:b1:continuity:continuous}{continua}, $h(0) = 1$, y
\[
h(x + y) = 1 + f(x) + f(y) + f(x)f(y) = h(x)\,h(y) :
\]
por la pregunta 14, $h(x) = \eu^{cx}$, es decir,
$f(x) = \eu^{cx} - 1$ (el caso $c = 0$ da $f \equiv 0$). Lista
completa: $f \equiv -1$ y $f(x) = \eu^{cx} - 1$, $c \in \R$.

\textbf{18.} $x = y = 0$: $2f(0) = 4f(0)$, luego $f(0) = 0$. $x = 0$:
$f(y) + f(-y) = 2f(y)$, luego $f$ es par. $y = x$: $f(2x) = 4f(x)$.
Inducción usando $(x, y) \to (nx, x)$:
\[
f((n{+}1)x) = 2f(nx) + 2f(x) - f((n{-}1)x)
= (2n^2 + 2 - (n-1)^2) f(x) = (n+1)^2 f(x) .
\]
Después, $f(x) = f\bigl(q\cdot\frac xq\bigr) = q^2 f\bigl(\frac
xq\bigr)$ da $f\bigl(\frac pq x\bigr) = \frac{p^2}{q^2}f(x)$:
$f(r) = r^2 f(1)$ en $\Q$ (la paridad se ocupa de los signos). Las dos
funciones \omterm{def:b1:continuity:continuous}{continuas} $f$ y $x \mapsto f(1)x^2$ coinciden en el
\omterm{def:b1:logic:sets}{conjunto} \omterm{def:b1:topology:dense}{denso} $\Q$ y, por tanto, en todas partes (pregunta 24):
$f(x) = f(1)\,x^2$; y todo $cx^2$ cumple la ecuación.

\textbf{19.} $g = f - f(0)$ es \omterm{def:b1:continuity:continuous}{continua}, $g(0) = 0$ y cumple la
ecuación de Jensen (las constantes se cancelan). Haciendo $y = 0$:
$g\bigl(\frac x2\bigr) = \frac{g(x)}{2}$. Entonces, para todos
$x, y$:
\[
\frac{g(x + y)}{2} = g\Bigl(\frac{x+y}{2}\Bigr)
= \frac{g(x) + g(y)}{2} ,
\]
luego $g$ es aditiva y \omterm{def:b1:continuity:continuous}{continua}: $g(x) = cx$ (pregunta 2), y
$f(x) = cx + d$ con $d = f(0)$. Todas las funciones afines cumplen la
de Jensen: la lista es completa.

\textbf{20.} Inducción sobre $n$. Para $n = 0$: $\lambda \in \{0, 1\}$,
trivial. Supóngase la desigualdad para todos los pesos
$\frac{k}{2^n}$. Un peso $\lambda = \frac{k}{2^{n+1}}$ con $k$ par se
reduce al nivel $n$; y para $k = 2j + 1$, $\lambda$ es el punto medio
de $\lambda_1 = \frac{j}{2^n}$ y $\lambda_2 = \frac{j+1}{2^n}$. Con
$z_i = \lambda_i x + (1 - \lambda_i)y$:
$\lambda x + (1-\lambda)y = \frac{z_1 + z_2}{2}$, luego
\[
f\bigl(\lambda x + (1{-}\lambda)y\bigr)
\leq \frac{f(z_1) + f(z_2)}{2}
\leq \frac{(\lambda_1 + \lambda_2)f(x) + (2 - \lambda_1 -
\lambda_2)f(y)}{2}
= \lambda f(x) + (1 - \lambda)f(y) .
\]

\textbf{21.} Fíjense $x, y$. Las aplicaciones
$\lambda \mapsto f(\lambda x + (1 - \lambda)y)$ y
$\lambda \mapsto \lambda f(x) + (1 - \lambda)f(y)$ son \omterm{def:b1:continuity:continuous}{continuas} en
$\intcc{0}{1}$ (composición y álgebra,
\cref{prop:b1:continuity:algebra}). La desigualdad se cumple en los
pesos diádicos, que son \omterm{def:b1:topology:dense}{densos} en $\intcc{0}{1}$
(\cref{exo:b1:reals:8}); y para $\lambda$ arbitrario tómense diádicos
$\lambda_n \to \lambda$ y pásese al límite
(\cref{thm:b1:continuity:seqchar} y \cref{thm:b1:seq:order}): la
desigualdad de convexidad vale para todo
$\lambda \in \intcc{0}{1}$ --- convexidad en el punto medio más
\omterm{def:b1:continuity:continuous}{continuidad} es igual a convexidad (la noción
del~\cref{ch:b1:derivative}).

\textbf{22.} Una $f$ aditiva no lineal cumple
$f\bigl(\frac{x+y}{2}\bigr) = \frac{f(x) + f(y)}{2}$ con igualdad
(pregunta 1 con $r = \frac12$, más la aditividad): es convexa en el
punto medio, e incluso afín en el punto medio. Si cumpliera la
desigualdad de convexidad completa en algún \omterm{prop:b1:reals:intervals}{intervalo}
$\intcc{x}{y}$, entonces, para $\lambda \in \intcc{0}{1}$:
$f(\lambda x + (1-\lambda)y) \leq \max(f(x), f(y))$ --- acotada
superiormente en un \omterm{prop:b1:reals:intervals}{intervalo} no degenerado y, por tanto, lineal por
la pregunta 9: contradicción. Así pues, la \omterm{def:b1:continuity:continuous}{continuidad} de la
pregunta 21 no es un lujo: sin ella, la convexidad en el punto medio
solo controla el esqueleto diádico numerable, y el \omterm{def:b1:continuity:continuous}{continuo} que queda
en medio se desboca.

\textbf{23.} $p(x) = \frac{x^2}{2}$ cumple
$p(x+y) = p(x) + p(y) + xy$. Si $f$ es una solución \omterm{def:b1:continuity:continuous}{continua}
cualquiera, $g = f - p$ es \omterm{def:b1:continuity:continuous}{continua} y aditiva, luego $g(x) = cx$:
\[
f(x) = \frac{x^2}{2} + cx , \qquad c \in \R ,
\]
y cada una de ellas es solución: la lista es completa.

\textbf{24.} Principio general: si $u, v$ son \omterm{def:b1:continuity:continuous}{continuas} y coinciden
en un $D \subseteq \R$ \omterm{def:b1:topology:dense}{denso}, entonces, para $x \in \R$, elíjanse
$d_n \in D$ con $d_n \to x$ (\cref{prop:b1:topology:closureprops});
$u(x) = \lim u(d_n) = \lim v(d_n) = v(x)$. Sea ahora $f$ \omterm{def:b1:continuity:continuous}{continua} y
aditiva en el \omterm{def:b1:structures:subgroup}{subgrupo} \omterm{def:b1:topology:dense}{denso} $G$. Fíjense $x, y \in \R$ y tómense
$g_n \to x$, $g'_n \to y$ con $g_n, g'_n \in G$; entonces
$g_n + g'_n \to x + y$ y, por \omterm{def:b1:continuity:continuous}{continuidad} secuencial en $x + y$, en
$x$ y en $y$:
\[
f(x + y) = \lim f(g_n + g'_n) = \lim\bigl(f(g_n) +
f(g'_n)\bigr) = f(x) + f(y) :
\]
$f$ es aditiva en todo $\R$ (y, por tanto, lineal, por la pregunta 2).

\textbf{25.} (i) Para $f$ aditiva: lineal $\iff$ \omterm{def:b1:continuity:continuous}{continua} $\iff$
\omterm{def:b1:continuity:continuous}{continua} en un punto $\iff$ monótona en algún \omterm{prop:b1:reals:intervals}{intervalo} $\iff$
acotada (incluso por un solo lado) en algún \omterm{prop:b1:reals:intervals}{intervalo}. (ii) La
\omterm{def:b1:topology:dense}{densidad} de la gráfica en el plano muestra que el fallo no es un
defecto local, sino una explosión global: sobre cada subintervalo los
valores se untan por todo $\R$, de modo que toda propiedad de
regularidad falla en todas partes a la vez. (iii) El botín: $cx$,
$\eu^{cx}$, $c\ln x$, $x^c$, $cx^2$ y $cx + d$ --- seis
caracterizaciones, un solo método: transportar la ecuación a la de
Cauchy, demostrar el esqueleto sobre $\Q$ por inducción y ascender a
$\R$ por \omterm{def:b1:topology:dense}{densidad} más \omterm{def:b1:continuity:continuous}{continuidad}. (iv) La \omterm{def:b1:topology:dense}{densidad} de $\Q$ (o de
los diádicos) sostuvo los ascensos del~\cref{exo:b1:continuity:11} y
de la pregunta 21; y no sostuvo nada en la pregunta 13, porque sin
\omterm{def:b1:continuity:continuous}{continuidad} los valores no se propagan de un \omterm{def:b1:logic:sets}{conjunto} \omterm{def:b1:topology:dense}{denso} a su
\omterm{def:b1:topology:closure}{clausura} --- la \omterm{def:b1:topology:dense}{densidad} solo transfiere información a lo largo de
la \omterm{def:b1:continuity:continuous}{continuidad}.
\end{solution}
