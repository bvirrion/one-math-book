\section*{Capítulo \ref{ch:b1:topology} --- Topología de la recta real}

\begin{solution}{exo:b1:topology:1}
$\intoo{0}{1} \cup \intoo{2}{3}$: \omterm{def:b1:topology:open}{abierto} (unión de \omterm{def:b1:topology:open}{abiertos}), no
\omterm{def:b1:topology:closed}{cerrado} ($\frac 1n \to 0$, que queda fuera).

$\intco{0}{1}$: ninguna de las dos cosas. No \omterm{def:b1:topology:open}{abierto} (ningún
\omterm{prop:b1:reals:intervals}{intervalo} en torno a $0$ está dentro); no \omterm{def:b1:topology:closed}{cerrado}
($1 - \frac1n \to 1 \notin$ el \omterm{def:b1:logic:sets}{conjunto}).

$\{0\} \cup \intcc{1}{2}$: \omterm{def:b1:topology:closed}{cerrado} (unión finita de \omterm{def:b1:topology:closed}{cerrados}), no
\omterm{def:b1:topology:open}{abierto} (falla en $0$).

$\R \setminus \Z$: \omterm{def:b1:topology:open}{abierto} ($\Z$ es \omterm{def:b1:topology:closed}{cerrado}), no \omterm{def:b1:topology:closed}{cerrado}: la sucesión
$\bigl(\frac 1n\bigr)$ está en él, pero su límite $0$ pertenece a $\Z$,
es decir, se escapa del \omterm{def:b1:logic:sets}{conjunto}.

$\Q \cap \intoo{0}{1}$: ninguna de las dos cosas. No \omterm{def:b1:topology:open}{abierto}: todo
\omterm{prop:b1:reals:intervals}{intervalo} en torno a un racional contiene irracionales. No \omterm{def:b1:topology:closed}{cerrado}:
contiene sucesiones que tienden al irracional $\frac{\sqrt 2}{2}$
(\omterm{def:b1:topology:dense}{densidad}).
\end{solution}

\begin{solution}{exo:b1:topology:2}
$A = \intoc{0}{1} \cup \{2\}$: $\mathring A = \intoo{0}{1}$,
$\overline A = \intcc{0}{1} \cup \{2\}$, $\partial A = \{0, 1, 2\}$.

$A = \R \setminus \Q$: $\mathring A = \emptyset$ (todo \omterm{prop:b1:reals:intervals}{intervalo}
contiene racionales), $\overline A = \R$ (\omterm{def:b1:topology:dense}{densidad} de los
irracionales), $\partial A = \R$.

$A = \bigl\{\frac{(-1)^n n}{n+1}\bigr\}$: los términos pares tienden a
$1$ y los impares a $-1$, y ninguno de los dos límites está en $A$.
$\mathring A = \emptyset$ (puntos aislados),
$\overline A = A \cup \{-1, 1\}$, $\partial A = \overline A$.
\end{solution}

\begin{solution}{exo:b1:topology:3}
\emph{Complementarios.} $F = \{a_1 < a_2 < \dots < a_k\}$: el
complementario es la unión de los \omterm{prop:b1:reals:intervals}{intervalos} \omterm{def:b1:topology:open}{abiertos}
$\intoo{-\infty}{a_1}$, $\intoo{a_i}{a_{i+1}}$,
$\intoo{a_k}{+\infty}$ --- \omterm{def:b1:topology:open}{abierto} por
la~\cref{prop:b1:topology:openstable}.

\emph{Sucesiones.} Sean $u_n \in F$ y $u_n \to \ell$. Con
$\varepsilon = \min\{\abs{a_i - a_j} : i \neq j\}/2 > 0$ (o cualquier
$\varepsilon$ si $F$ es unitario): más allá de cierto rango, todos los
términos distan menos de $\varepsilon$ de $\ell$ y, por tanto, menos
de $2\varepsilon$ entre sí, lo que los obliga a ser un único $a_i$ a
partir de ese rango; entonces $\ell = a_i \in F$.
\end{solution}

\begin{solution}{exo:b1:topology:4}
$\subseteq$: $\overline A \cup \overline B$ es \omterm{def:b1:topology:closed}{cerrado} (unión finita)
y contiene a $A \cup B$, luego contiene al \emph{menor} \omterm{def:b1:topology:closed}{cerrado} que
lo contiene, $\overline{A \cup B}$. $\supseteq$: de
$A \subseteq A \cup B$ se sigue $\overline A \subseteq
\overline{A \cup B}$ (la \omterm{def:b1:topology:closure}{clausura} es monótona: los puntos adherentes
a $A$ lo son al \omterm{def:b1:logic:sets}{conjunto} mayor), e igual para $B$.

Contraejemplo para las intersecciones: $A = \intoo{0}{1}$,
$B = \intoo{1}{2}$: $\overline{A \cap B} = \overline\emptyset =
\emptyset$, mientras que $\overline A \cap \overline B = \{1\}$.
\end{solution}

\begin{solution}{exo:b1:topology:5}
Úsese la caracterización secuencial
(\cref{thm:b1:topology:seqclosed}). Sean $S = \{u_n\} \cup \{\ell\}$ y
$(v_k)$ una sucesión de $S$ con $v_k \to m$; veamos que $m \in S$. Dos
casos. Si hay un valor $v \in S$ que $(v_k)$ toma infinitas veces, una
\omterm{def:b1:seq:subsequence}{subsucesión} constante da $m = v \in S$. En caso contrario, todo
valor se toma finitas veces; en particular, para cada $n$ el término
$u_n$ aparece finitas veces, y $\ell$ también. Entonces, para cada
$N$, los índices $k$ con $v_k \in \{u_0, \dots, u_N, \ell\}$ son
finitos: los $v_k$ restantes son términos $u_n$ con $n > N$. Dado
$\varepsilon > 0$, elíjase $N$ con
$\abs{u_n - \ell} \leq \varepsilon$ para $n > N$: todos los $v_k$
salvo finitos cumplen $\abs{v_k - \ell} \leq \varepsilon$. Luego
$v_k \to \ell$ y, por tanto, $m = \ell \in S$.
\end{solution}

\begin{solution}{exo:b1:topology:6}
$U + A = \bigcup_{a \in A} (U + a)$, y cada trasladado $U + a$ es
\omterm{def:b1:topology:open}{abierto} (basta trasladar los certificados de \omterm{prop:b1:reals:intervals}{intervalo}). Una unión
de \omterm{def:b1:topology:open}{abiertos} es \omterm{def:b1:topology:open}{abierta} (\cref{prop:b1:topology:openstable}).

\omterm{def:b1:topology:closed}{Cerrados}: $\Z$ y $\sqrt 2\,\Z$ son \omterm{def:b1:topology:closed}{cerrados} (como todo $\alpha\Z$:
las sucesiones convergentes acaban siendo constantes, cf.\
\cref{ex:b1:topology:closed}). Su suma $\Z + \sqrt 2\,\Z$ es \omterm{def:b1:topology:dense}{densa} en
$\R$ (\cref{exo:b1:reals:9}) pero no es $\R$ (es numerable o,
sencillamente: $\frac{\sqrt 2}{2} \notin \Z + \sqrt2\Z$, pues si no
$\sqrt 2$ sería racional --- escribir
$\frac{\sqrt2}{2} = m + n\sqrt 2$ obliga a $(2n - 1)\sqrt 2 = -2m$,
luego $\sqrt 2 \in \Q$ salvo que $n = \frac12$, lo cual es
imposible). Un subconjunto propio y \omterm{def:b1:topology:dense}{denso} no es \omterm{def:b1:topology:closed}{cerrado}: su
\omterm{def:b1:topology:closure}{clausura} es $\R \neq$ él mismo.
\end{solution}

\begin{solution}{exo:b1:topology:7}
$\Z$: el \omterm{def:b1:topology:open}{entorno} $\intoo{n - \frac12}{n + \frac12}$ de $n$ corta a
$\Z$ solo en $n$.

$A = \{\frac1n : n \in \N^*\}$: en torno a $\frac 1n$, el \omterm{prop:b1:reals:intervals}{intervalo}
de radio $\frac{1}{n} - \frac{1}{n+1} = \frac{1}{n(n+1)}$ (dividido
por dos, digamos) lo aísla de sus vecinos --- todos los puntos son
aislados. Y, sin embargo, $0 \in \overline A \setminus A$: que los
puntos sean aislados no impide que haya adherentes de fuera.

Si todos los puntos de $A$ son aislados, ningún punto de $A$ es
\omterm{def:b1:topology:closure}{interior}, pues un punto \omterm{def:b1:topology:closure}{interior} tiene a su alrededor todo un
\omterm{prop:b1:reals:intervals}{intervalo} de vecinos de $A$ (un \omterm{prop:b1:reals:intervals}{intervalo} es infinito), en
contradicción con el aislamiento. Luego $\mathring A = \emptyset$.
\end{solution}

\begin{solution}{exo:b1:topology:8}
Si $A \subseteq \intcc{-M}{M}$, el \omterm{def:b1:topology:closed}{cerrado} $\intcc{-M}{M}$ contiene a
$A$ y, por tanto, contiene a $\overline A$ (el menor \omterm{def:b1:topology:closed}{cerrado} que lo
contiene): $\overline A$ está acotado.

Sea $s = \sup A$ (finito). Como $A \subseteq \overline A$, se tiene
$\sup \overline A \geq s$. Recíprocamente,
$\overline A \subseteq \intoc{-\infty}{s}$: la semirrecta es \omterm{def:b1:topology:closed}{cerrada}
y contiene a $A$; así, todo elemento de $\overline A$ es $\leq s$, lo
que da $\sup\overline A \leq s$. Hay igualdad.

$s \in \overline A$: por la caracterización con $\varepsilon$
(\cref{prop:b1:reals:epsilon}), todo \omterm{prop:b1:reals:intervals}{intervalo}
$\intoo{s - \varepsilon}{s + \varepsilon}$ contiene un elemento de
$A$: $s$ es adherente.
\end{solution}

\begin{solution}{exo:b1:topology:9}
Supóngase $A$ \omterm{def:b1:topology:open}{abierto} y \omterm{def:b1:topology:closed}{cerrado}, con $a \in A$ y
$b \in \R \setminus A$; sin pérdida de generalidad, $a < b$. El
\omterm{def:b1:logic:sets}{conjunto} $B = A \cap \intcc{a}{b}$ es no vacío ($a$) y acotado: sea
$s = \sup B$. Por el~\cref{exo:b1:topology:8},
$s \in \overline B \subseteq \overline A = A$ ($A$ \omterm{def:b1:topology:closed}{cerrado}).
Obsérvese que $s \leq b$ y, como $b \notin A$, resulta $s < b$. Ahora
bien, $A$ es \omterm{def:b1:topology:open}{abierto}: algún \omterm{prop:b1:reals:intervals}{intervalo} $\intoo{s - r}{s + r}$ está en
$A$, y podemos tomar $r < b - s$. Entonces $s + \frac{r}{2}$ está en
$A \cap \intcc{a}{b} = B$ y supera a $s$ --- en contradicción con
$s = \sup B$. Luego no existe tal par $(a, b)$: uno de los dos
\omterm{def:b1:logic:sets}{conjuntos} $A$, $\R \setminus A$ es vacío.
\end{solution}

\begin{solution}{exo:b1:topology:10}
$I_x$ es una unión de \omterm{prop:b1:reals:intervals}{intervalos} \omterm{def:b1:topology:open}{abiertos} que contienen todos a $x$:
es \omterm{def:b1:topology:open}{abierto}, y es un \omterm{prop:b1:reals:intervals}{intervalo} por ser convexo --- si $u < z < v$ con
$u, v \in I_x$, entonces $u$ y $v$ están en subintervalos \omterm{def:b1:topology:open}{abiertos}
$J_u \ni x$, $J_v \ni x$ de $U$, y $J_u \cup J_v$ es un \omterm{prop:b1:reals:intervals}{intervalo}
(los dos contienen a $x$) dentro de $U$ que contiene a $z$; luego
$z \in I_x$ (\cref{prop:b1:reals:intervals}).

Si $I_x \cap I_y \neq \emptyset$, entonces $I_x \cup I_y$ es un
\omterm{prop:b1:reals:intervals}{intervalo} \omterm{def:b1:topology:open}{abierto} (convexo: dos \omterm{prop:b1:reals:intervals}{intervalos} que se solapan)
contenido en $U$ y que contiene a $x$ y a $y$, de modo que
$I_x \cup I_y \subseteq I_x$ y $\subseteq I_y$ por la maximalidad de
cada uno: $I_x = I_y$.

Así pues, $U$ es la unión disjunta de los \omterm{def:b1:logic:sets}{conjuntos} $I_x$ distintos
(cada $x \in U$ está en su propio $I_x$). Numerabilidad: cada
\omterm{prop:b1:reals:intervals}{intervalo} \omterm{def:b1:topology:open}{abierto} no vacío $I$ de la familia contiene un racional
$q_I$ (\cref{thm:b1:reals:density}), y a \omterm{prop:b1:reals:intervals}{intervalos} disjuntos
distintos les corresponden racionales distintos: la familia se
inyecta en $\Q$, que es numerable (está indexado por pares de
enteros). Luego hay a lo sumo numerablemente muchos \omterm{prop:b1:reals:intervals}{intervalos}.
\end{solution}

\begin{solution}{exo:b1:topology:11}
$\overline A = A \cup A'$. ($\supseteq$) Siempre
$A \subseteq \overline A$; y si $x \in A'$, todo \omterm{def:b1:topology:open}{entorno} de $x$
corta a $A \setminus \{x\} \subseteq A$, luego $x$ es adherente.
($\subseteq$) Sea $x \in \overline A$. Si $x \in A$, listo. Si
$x \notin A$, todo \omterm{def:b1:topology:open}{entorno} de $x$ corta a $A = A \setminus \{x\}$:
$x \in A'$.

En consecuencia, $A$ \omterm{def:b1:topology:closed}{cerrado} $\iff$ $A = \overline A = A \cup A'$
$\iff$ $A' \subseteq A$.

$A = \{\frac 1n\}$: $0$ es punto de acumulación ($\frac 1n \to 0$,
con términos $\neq 0$); cada $\frac 1n$ es aislado
(\cref{exo:b1:topology:7}), luego no está en $A'$; y un punto
$x \notin A \cup \{0\}$ tiene todo un \omterm{prop:b1:reals:intervals}{intervalo} que evita $A$ (entre
los dos vecinos de $x$ en $A \cup \{0\}$, o más allá de $1$). Luego
$A' = \{0\}$.

$\Z' = \emptyset$: todo entero es aislado, y todo no entero tiene un
\omterm{def:b1:topology:open}{entorno} dentro de $\R \setminus \Z$.

$\Q' = \R$: todo \omterm{prop:b1:reals:intervals}{intervalo} en torno a un real cualquiera contiene
infinitos racionales (\cref{thm:b1:reals:density}), en particular uno
distinto del centro.
\end{solution}

\begin{solution}{exo:b1:topology:12}
\begin{enumerate}
  \item Para todo $a \in A$: $\abs{x - a} \leq \abs{x - y} +
        \abs{y - a}$, luego $d_A(x) \leq \abs{x - y} + \abs{y - a}$;
        tomando el \omterm{def:b1:reals:bounds}{ínfimo} sobre $a$: $d_A(x) \leq \abs{x - y} +
        d_A(y)$. Intercambiando $x$ e $y$ se obtiene la otra
        desigualdad: $\abs{d_A(x) - d_A(y)} \leq \abs{x - y}$.
  \item $d_A(x) = 0$ $\iff$ para todo $\varepsilon > 0$ existe
        $a \in A$ con $\abs{x - a} < \varepsilon$ $\iff$ todo
        \omterm{prop:b1:reals:intervals}{intervalo} en torno a $x$ corta a $A$ $\iff$
        $x \in \overline A$. Si $F$ es \omterm{def:b1:topology:closed}{cerrado} y
        $x \notin F = \overline F$, entonces $d_F(x) \neq 0$, es
        decir, $d_F(x) > 0$.
  \item Si $d_F(x) < \varepsilon$, tómese $r = \varepsilon -
        d_F(x) > 0$: para $\abs{y - x} < r$, el apartado (1) da
        $d_F(y) \leq d_F(x) + \abs{x - y} < \varepsilon$: el
        \omterm{prop:b1:reals:intervals}{intervalo} $\intoo{x - r}{x + r}$ está en $V_\varepsilon$,
        que es por tanto \omterm{def:b1:topology:open}{abierto}; y contiene a $F$, pues ahí
        $d_F = 0$. Por último, $x \in \bigcap_{\varepsilon>0}
        V_\varepsilon$ $\iff$ $d_F(x) < \varepsilon$ para todo
        $\varepsilon$ $\iff$ $d_F(x) = 0$ $\iff$ $x \in \overline
        F = F$. Como $\bigcap_{\varepsilon > 0} V_\varepsilon =
        \bigcap_{n \geq 1} V_{1/n}$, todo \omterm{def:b1:topology:closed}{cerrado} es una
        intersección numerable de \omterm{def:b1:topology:open}{abiertos}.
\end{enumerate}
\end{solution}

\begin{solution}{pb:b1:topology:1}
\textbf{1.} $C_1 = \intcc{0}{\frac13} \cup \intcc{\frac23}{1}$
y
\[
C_2 = \intcc{0}{\tfrac19} \cup \intcc{\tfrac29}{\tfrac13}
\cup \intcc{\tfrac23}{\tfrac79} \cup \intcc{\tfrac89}{1} .
\]
Inducción: si $C_n$ es unión disjunta de $2^n$ segmentos \omterm{def:b1:topology:closed}{cerrados} de
longitud $3^{-n}$, suprimir el tercio central \omterm{def:b1:topology:open}{abierto} de cada uno deja
dos segmentos \omterm{def:b1:topology:closed}{cerrados} de longitud $3^{-n-1}$ por padre: $2^{n+1}$
segmentos, disjuntos dos a dos (los hijos de padres distintos están
separados porque lo estaban los padres; los hijos de un mismo padre,
por el hueco suprimido).

\textbf{2.} Cada $C_n$ es unión finita de segmentos y, por tanto,
\omterm{def:b1:topology:closed}{cerrado}; $C = \bigcap C_n$ es una intersección de \omterm{def:b1:topology:closed}{cerrados}:
\omterm{def:b1:topology:closed}{cerrado} (\cref{def:b1:topology:closed}); acotado
($\subseteq \intcc{0}{1}$): compacto por
el~\cref{thm:b1:topology:compact}. No vacío: $0$ está en el segmento
más a la izquierda de todo $C_n$. Sea $a$ un extremo de un segmento
$S$ de $C_n$. Para $m \leq n$, $a \in C_n \subseteq C_m$. Para las
etapas posteriores: la supresión del tercio central nunca quita un
extremo, y $a$ vuelve a ser extremo de uno de los dos hijos de $S$
(el hijo que toca a $a$); por inducción, $a \in C_m$ para todo
$m \geq n$: $a \in C$.

\textbf{3.} Longitud total de $C_n$: $2^n \cdot 3^{-n} =
(\frac23)^n \to 0$. Dado $\varepsilon > 0$, elíjase $n$ con
$(\frac23)^n \leq \varepsilon$: entonces $C \subseteq C_n$, unión de
finitos segmentos de longitud total $\leq \varepsilon$.

\textbf{4.} Sea $I \subseteq C$ un \omterm{prop:b1:reals:intervals}{intervalo} con dos puntos
distintos. Para todo $n$: $I \subseteq C_n$ y, siendo $I$ convexo,
debe estar dentro de un \emph{único} segmento de $C_n$ (cortar a dos
segmentos obligaría a $I$ a contener un punto del hueco intermedio,
que está fuera de $C_n$). Luego la longitud de $I$ es $\leq 3^{-n}$
para todo $n$: contradicción. Así pues, los únicos \omterm{prop:b1:reals:intervals}{intervalos} dentro
de $C$ son el vacío y los unitarios; en particular, ningún
$\intoo{x-r}{x+r}$ cabe dentro de $C$: $\mathring C = \emptyset$.
Como $C$ es \omterm{def:b1:topology:closed}{cerrado}, $\overline C = C$ tiene \omterm{def:b1:topology:closure}{interior} vacío: $C$ es
\omterm{def:b1:topology:dense}{denso} en ninguna parte.

\textbf{5.} Escríbanse $\varphi_0(x) = \frac x3$ y
$\varphi_2(x) = \frac{2 + x}{3}$, biyecciones afines crecientes de
$\intcc{0}{1}$ sobre $\intcc{0}{\frac13}$ y $\intcc{\frac23}{1}$.
Afirmación: $C_{n+1} = \varphi_0(C_n) \cup \varphi_2(C_n)$. Para
$n = 0$, esto es la pregunta 1. Inducción: una \omterm{def:b1:logic:map}{aplicación} afín
creciente envía el tercio central de un segmento al tercio central
del segmento imagen, luego suprimir tercios centrales conmuta con
$\varphi_0$ y $\varphi_2$; aplicando la etapa de supresión a
$C_{n+1} = \varphi_0(C_n) \cup \varphi_2(C_n)$ se obtiene
$C_{n+2} = \varphi_0(C_{n+1}) \cup \varphi_2(C_{n+1})$. Intersecando
sobre $n$: para $x \leq \frac13$, $x \in C \iff x \in \varphi_0(C_n)$
para todo $n$ $\iff 3x \in \bigcap C_n = C$; igual sobre
$\intcc{\frac23}{1}$; y ningún punto de $\intoo{\frac13}{\frac23}$
está en $C_1$. Luego $C = \varphi_0(C) \cup \varphi_2(C)$, de forma
disjunta.

\textbf{6.} Inducción sobre $n$; el caso $n = 0$ dice que todo
$x \in \intcc{0}{1}$ tiene un código, que es el~\cref{pb:b1:reals:1}
(pregunta 9 para $x < 1$; $1 = (0.\overline 2)_3$). Supóngase la
equivalencia en el rango $n$. Si $x \in C_{n+1}$: por la pregunta 5,
$x = \varphi_i(z)$ con $z \in C_n$ e $i \in \{0, 2\}$; si $(e_k)$ es
un código de $z$ cuyas primeras $n$ cifras son libres de $1$,
entonces $(i, e_1, e_2, \dots)$ tiene sumas parciales
$\frac i3 + \frac13\sum_{k\leq m} e_k 3^{-k} \to \varphi_i(z) = x$:
un código de $x$ cuyas primeras $n + 1$ cifras son libres de $1$.
Recíprocamente, si $x$ tiene un código $(d_k)$ con
$d_1, \dots, d_{n+1} \in \{0, 2\}$: la cadena desplazada
$(d_2, d_3, \dots)$ tiene cierto valor $z \in \intcc{0}{1}$, sus
primeras $n$ cifras son libres de $1$, y el cálculo de las sumas
parciales leído al revés da $x = \varphi_{d_1}(z)$; por inducción
$z \in C_n$, luego $x \in C_{n+1}$ por la pregunta 5. Por último: un
código enteramente libre de $1$ mete a $x$ en todo $C_n$ y, por
tanto, en $C$; recíprocamente, si $x \in C$, para cada $n$ uno de los
a lo sumo dos códigos de $x$ (\cref{pb:b1:reals:1}, pregunta 11)
tiene sus primeras $n$ cifras libres de $1$; un mismo código fijo
debe servir para $n$ arbitrariamente grande (palomar entre dos
códigos), y un código cuyas primeras $n$ cifras son libres de $1$
para $n$ arbitrariamente grande es libre de $1$ sin más.

\textbf{7.} $0 = (0.\overline 0)_3$, $1 = (0.\overline 2)_3$,
$\frac13 = (0.0\overline{2})_3$ (el gemelo impropio de $(0.1)_3$),
$\frac23 = (0.2\overline{0})_3$. Sumas geométricas:
\[
(0.\overline{02})_3 = \sum_{j\geq1} \frac{2}{9^{\,j}}
= \frac{2/9}{1 - 1/9} = \frac14 ,
\qquad
(0.\overline{20})_3 = \sum_{j\geq1} \frac{2}{3\cdot 9^{\,j-1}}
= \frac{2/3}{1 - 1/9} = \frac34 ,
\]
los dos libres de $1$: $\frac14, \frac34 \in C$. (Las sumas infinitas
abrevian \omterm{def:b1:reals:bounds}{supremos} de sumas parciales, como en \cref{pb:b1:reals:1}.)
Los extremos de los segmentos de $C_n$ son de la forma $m/3^n$
(inducción: los extremos de los hijos son extremos del padre o
difieren de uno de ellos en un múltiplo de $3^{-n-1}$). Si
$\frac14 = \frac{m}{3^n}$, entonces $3^n = 4m$, y $4 \nmid 3^n$:
imposible. Luego $\frac14 \in C$ sin ser nunca un extremo.

\textbf{8.} Supóngase que $x$ tuviera dos códigos libres de $1$
distintos. Tener dos códigos significa (\cref{pb:b1:reals:1},
pregunta 11, base $3$) que $x = m/3^N \in \intoo{0}{1}$ y que los dos
códigos son: el finito, con última cifra no nula $d_N \in \{1, 2\}$
seguida de $0$, y su gemelo, con $d_N - 1$ en la posición $N$
seguido de $2$. Si $d_N = 1$, el primero contiene un $1$; si
$d_N = 2$, el gemelo lleva $d_N - 1 = 1$. En cualquier caso, a lo
sumo uno de los dos es libre de $1$: contradicción. Luego cada
$x \in C$ tiene exactamente un código libre de $1$ (la existencia,
por la pregunta 6), y cadenas de $\{0,2\}$ distintas tienen valores
distintos. Toda cadena de $\{0,2\}$ tiene valor en $\intcc{0}{1}$
(sumas parciales $\leq 1$) con todos sus prefijos libres de $1$, y
por tanto valor en todo $C_n$, es decir, en $C$: la \omterm{def:b1:logic:map}{aplicación} valor
es una \omterm{def:b1:logic:inj}{biyección} de las cadenas de $\{0,2\}$ sobre $C$.

\textbf{9.} Sea $(d^{(k)})$ el código libre de $1$ de $x_k$ y
tómese $e_k = 2 - d^{(k)}_k \in \{0, 2\}$: una cadena de $\{0,2\}$
cuyo valor $y$ está en $C$ y tiene $(e_k)$ como único código libre de
$1$ (pregunta 8). Para cada $k$, los códigos de $y$ y de $x_k$
difieren en la posición $k$, luego $y \neq x_k$: ninguna \omterm{def:b1:logic:map}{aplicación}
$\N^* \to C$ es \omterm{def:b1:logic:inj}{sobreyectiva}. Un \omterm{def:b1:logic:sets}{conjunto} no numerable de longitud
cero: grande en \omterm{def:b1:counting:card}{cardinal}, pequeño en medida --- a la vez.

\textbf{10.} Compacidad: que la intersección infinita sea \omterm{def:b1:topology:closed}{cerrada},
más la acotación (pregunta 2) --- la única propiedad que no sostiene
una sola contención. Longitud cero: $C \subseteq C_n$, de longitud
total $(\frac23)^n$ (pregunta 3). \omterm{def:b1:topology:dense}{Densidad} en ninguna parte:
$C \subseteq C_n$ obliga a que los \omterm{prop:b1:reals:intervals}{intervalos} dentro de $C$ tengan
longitud $\leq 3^{-n}$ (pregunta 4). La no numerabilidad no cabalga
sobre ninguna contención: necesita toda la estructura de la
intersección, codificada en la \omterm{def:b1:logic:inj}{biyección} de la pregunta 8.

\textbf{11.} Cámbiese $d_n$ por $2 - d_n$: la nueva cadena sigue
siendo una cadena de $\{0,2\}$, luego su valor $x_n$ está en $C$; y
las sumas parciales más allá del rango $n$ difieren exactamente en
$2\cdot3^{-n}$, de modo que $\abs{x_n - x} = 2\cdot3^{-n}$. Así,
$x_n \neq x$ y $x_n \to x$: todo punto de $C$ es límite de otros
puntos de $C$ --- $C$ es \emph{perfecto}, sin ningún punto aislado.

\textbf{12.} Por la inducción de la pregunta 5, los segmentos de
$C_n$ son exactamente los $\intcc{t}{t + 3^{-n}}$ donde $t$ recorre
los valores de las cadenas de longitud $n$ formadas con $\{0,2\}$. Dado
$x \in C$ de código $(d_k)$, la truncatura $t_n$ (cifras
$d_1 \dots d_n$ y después $0$) es, por tanto, un extremo izquierdo,
y $0 \leq x - t_n \leq 3^{-n}$: los extremos son \omterm{def:b1:topology:dense}{densos} en $C$.
Forman un subconjunto de $\{m/3^n : m, n\}$, un \omterm{def:b1:logic:sets}{conjunto} indexado
por pares de enteros y, por tanto, numerable (como $\Q$ en
el~\cref{exo:b1:topology:10}). Como $C$ no es numerable (pregunta 9),
todos los puntos de $C$ salvo numerablemente muchos no son extremos
--- $\frac14$ (pregunta 7) es la punta visible de ese iceberg.

\textbf{13.} Tómese $n$ con $3^{-n} < y - x$. Los dos, $x, y \in
C_n$, y no pueden estar en el mismo segmento (de longitud
$3^{-n} < y - x$): el hueco suprimido entre sus segmentos
proporciona $z$ con $x < z < y$ y $z \notin C_n \supseteq C$. Así
pues, dos puntos cualesquiera de $C$ están separados por el
complementario: los únicos subconjuntos convexos de $C$ son los
unitarios --- $C$ es totalmente disconexo.

\textbf{14.} Si $(d_k)$ es el código libre de $1$ de $x$, la cadena
$(2 - d_k)$ es de nuevo una cadena de $\{0,2\}$, con sumas parciales
\[
\sum_{k=1}^{n} (2 - d_k)3^{-k} = (1 - 3^{-n}) - \sum_{k=1}^n
d_k 3^{-k} \longrightarrow 1 - x :
\]
luego $1 - x \in C$. Así, $1 - C \subseteq C$ y, aplicando la
\omterm{def:b1:logic:map}{aplicación} dos veces, $1 - C = C$: el \omterm{pb:b1:topology:1}{conjunto de Cantor} es simétrico
respecto de $\frac12$.

\textbf{15.} Las sumas parciales $t_n \to x$ y $t'_n \to x'$ (las
sucesiones crecientes convergen a su \omterm{def:b1:reals:bounds}{supremo}, es decir, al valor),
luego $t_n + t'_n \to x + x'$ por
el~\cref{thm:b1:seq:operations}. Dado $y \in \intcc{0}{1}$ de código
$(e_k)$, elíjase $(a_k, b_k) = (0,0), (0,2), (2,2)$ según sea
$e_k = 0, 1, 2$: entonces $a_k + b_k = 2e_k$, las cadenas $(a_k)$,
$(b_k)$ son cadenas de $\{0,2\}$ con valores $x, x' \in C$, y
\[
x + x' = \lim_n\,(t_n + t'_n) = \lim_n 2\sum_{k=1}^n e_k 3^{-k}
= 2y :
\]
todo $y \in \intcc{0}{1}$ es el punto medio de dos puntos de $C$.

\textbf{16.} La pregunta 15 da $\intcc{0}{2} = 2\,\intcc{0}{1}
\subseteq C + C$, y $C + C \subseteq \intcc{0}{1} + \intcc{0}{1} =
\intcc{0}{2}$: igualdad. Después, usando $1 - C = C$:
\[
C - C = C + (C - 1) = (C + C) - 1 = \intcc{-1}{1} .
\]
Instancia concreta: $1 = \frac14 + \frac34$, suma de dos miembros
de $C$ que no son extremos.

\textbf{17.} La longitud mide cuánto de la recta ocupa el propio
\omterm{def:b1:logic:sets}{conjunto}; no dice nada del \omterm{def:b1:logic:sets}{conjunto} de \emph{sumas}, que es la
imagen de la familia de dos parámetros $C \times C$ por
$(x, x') \mapsto x + x'$ --- las dos cadenas de cifras se eligen de
forma independiente, y esa libertad es exactamente lo que llena
$\intcc{0}{2}$. Ningún teorema acota la longitud de un \omterm{def:b1:logic:sets}{conjunto} suma
por las longitudes de los sumandos, y $C$ es la prueba de que ninguno
puede hacerlo.

\textbf{18.} Por el~\cref{pb:b1:reals:1} (pregunta 18), $x$ es
racional si y solo si su desarrollo propio es finalmente periódico.
El código libre de $1$ de $x \in C$ es o bien ese desarrollo propio,
o bien el gemelo impropio de uno finito; y una cadena finita y su
gemelo (finalmente constante igual a $2$) son ambos finalmente
periódicos, de modo que la periodicidad del código libre de $1$
equivale a la racionalidad de $x$. División larga de $\frac1{13}$ en
base $3$ ($r_0 = 1$): $3 = 13\cdot0 + 3$, $9 = 13\cdot0 + 9$,
$27 = 13\cdot2 + 1$, y el resto vuelve a $1$: cifras
$\overline{002}$, luego $\frac1{13} = (0.\overline{002})_3$, libre de
$1$ y periódico: un miembro racional de $C$. (Comprobación:
$\frac{2/27}{1 - 1/27} = \frac{2}{26} = \frac1{13}$.)

\textbf{19.} La cadena con $d_k = 2$ en las posiciones triangulares
$k = \frac{j(j+1)}{2}$ y $0$ en las demás es una cadena de
$\{0,2\}$, luego su valor $x^*$ pertenece a $C$ (pregunta 8). Tiene
infinitos $2$ con huecos $j + 1 \to \infty$ entre dos consecutivos,
así que no es finalmente periódica (un período $T$ acabaría
forzando $2$ a distancias $\leq T$: el argumento de los huecos
crecientes del~\cref{pb:b1:reals:1}, pregunta 20); por la
pregunta 18, $x^* \notin \Q$. Y por la pregunta 9 más la
numerabilidad de $\Q$, todos los miembros de $C$ salvo
numerablemente muchos son irracionales: $x^*$ es la norma, no la
excepción.

\textbf{20.} Sea $y \in \intcc{0}{1}$: tiene un código binario
$(c_k)$ con $c_k \in \{0, 1\}$ (\cref{pb:b1:reals:1}, pregunta 9,
base $2$; para $y = 1$ se toma la cadena de $1$). Entonces $(2c_k)$
es una cadena de $\{0,2\}$, su valor $x$ está en $C$, y $h(x)$ es el
valor de $(c_k)$, es decir, $y$: $h$ aplica $C$ sobre
$\intcc{0}{1}$. Si $C$ fuese la imagen de una \omterm{def:b1:logic:map}{aplicación} desde
$\N^*$, componer con $h$ enumeraría todo $\intco{0}{1}$, en
contradicción con el teorema diagonal del~\cref{pb:b1:reals:1}
(pregunta 22): $C$ no es numerable, otra vez. Un \omterm{def:b1:logic:sets}{conjunto} de
longitud cero que se aplica sobreyectivamente sobre un segmento
completo.

\textbf{21.} Por la pregunta 5, $C$ es la unión disjunta de
$\varphi_0(C)$ y $\varphi_2(C)$, cada uno un trasladado de la copia
a escala $\frac13 C$. La aditividad, el escalado y la invariancia por
traslación dan
\[
L(C) = L(\varphi_0(C)) + L(\varphi_2(C))
= \tfrac13 L(C) + \tfrac13 L(C) = \tfrac23\,L(C),
\]
luego $\frac13 L(C) = 0$: $L(C) = 0$. La autosemejanza sola condena
a $C$ a longitud cero --- la pregunta 3 se limitó a ejecutar la
sentencia.

\textbf{22.} Un segmento de longitud $l_n$ pierde un \omterm{prop:b1:reals:intervals}{intervalo}
central de longitud $4^{-(n+1)}$, y quedan dos segmentos de longitud
$l_{n+1} = \frac{l_n - 4^{-(n+1)}}{2}$; a partir de $l_0 = 1$, la
inducción confirma $l_n = \frac{2^n + 1}{2\cdot4^n}$: en efecto,
$\frac12\Bigl(\frac{2^n+1}{2\cdot4^n} - \frac{1}{4^{n+1}}\Bigr)
= \frac{2(2^n + 1) - 1}{2\cdot4^{n+1}} = \frac{2^{n+1} +
1}{2\cdot4^{n+1}}$, y $l_n > 0$ siempre: la construcción nunca se
queda sin material. $K = \bigcap K_n$ es \omterm{def:b1:topology:closed}{cerrado} y acotado y, por
tanto, compacto; y un \omterm{prop:b1:reals:intervals}{intervalo} dentro de $K$ está en un segmento de
$K_n$, de longitud $l_n \to 0$: \omterm{def:b1:topology:closure}{interior} vacío. Longitud suprimida:
$\sum_{n\geq0} 2^n \cdot 4^{-(n+1)} =
\frac14\sum_{n\geq0}\bigl(\frac12\bigr)^n = \frac12$, y cada $K_n$
tiene longitud total $2^n l_n = \frac{2^n + 1}{2^{n+1}} > \frac12$.
Sea ahora una familia finita de \omterm{prop:b1:reals:intervals}{intervalos} \omterm{def:b1:topology:open}{abiertos} de unión
$U \supseteq K$. Por el \omterm{def:b1:logic:statement}{enunciado} compañero
del~\cref{ex:b1:topology:nested}, $U \supseteq K_n$ para algún $n$;
admitiendo la aditividad de la longitud sobre uniones finitas de
\omterm{prop:b1:reals:intervals}{intervalos}, la longitud total de los \omterm{prop:b1:reals:intervals}{intervalos} del recubrimiento es
al menos la de $K_n$, que supera $\frac12$. Luego $K$ es \omterm{def:b1:topology:dense}{denso} en
ninguna parte y, sin embargo, no hay recubrimiento barato: la
pequeñez topológica (\omterm{def:b1:topology:dense}{densidad} en ninguna parte) y la pequeñez métrica
(longitud cero) son nociones genuinamente distintas, y $K$ las separa.

\textbf{23.} Que se alcanza: sea $d = d(x, F)$ y tómense $a_k \in F$
con $\abs{x - a_k} \leq d + \frac1k$: los $a_k$ están acotados, luego
Bolzano--Weierstrass (\cref{thm:b1:seq:bw}) extrae
$a_{\varphi(k)} \to a$, con $a \in F$ ($F$ \omterm{def:b1:topology:closed}{cerrado},
\cref{thm:b1:topology:seqclosed}) y
$\abs{x - a} = \lim \abs{x - a_{\varphi(k)}} = d$. Ahora el máximo:
si $y \in C$, entonces $d(y, C) = 0$; en caso contrario, $y$ está en
un hueco suprimido en cierta etapa $n \geq 1$, un \omterm{prop:b1:reals:intervals}{intervalo}
\omterm{def:b1:topology:open}{abierto} de longitud $3^{-n}$ cuyos dos extremos pertenecen a $C$
(pregunta 2), luego $d(y, C) \leq \frac{3^{-n}}{2} \leq \frac16$,
con igualdad solo si $n = 1$ e $y$ es el centro del hueco
$\intoo{\frac13}{\frac23}$, es decir, $y = \frac12$; y, en efecto,
$d\bigl(\frac12, C\bigr) = \frac16$, pues
$C \cap \intoo{\frac13}{\frac23} = \emptyset$ y
$\frac13, \frac23 \in C$. Luego
$\max_{y\in\intcc{0}{1}} d(y, C) = \frac16$, alcanzado exactamente
en $\frac12$.

\textbf{24.} Los extremos forman un subconjunto numerable y \omterm{def:b1:topology:dense}{denso} de
$C$ (pregunta 12): enumérense como una única sucesión
$(e_j)_{j\geq1}$, que es una sucesión de $C$. Todos sus límites de
subsucesiones están en $C$ ($C$ \omterm{def:b1:topology:closed}{cerrado}). Recíprocamente, fíjese
$x \in C$: para cada $n$, los segmentos de los $C_m$ que contienen a
$x$ ($m \geq n$) tienen sus extremos a distancia
$3^{-m} \leq 3^{-n}$ de $x$, luego infinitos extremos distintos
están a menos de $3^{-n}$ de $x$; elíjanse índices
$j_1 < j_2 < \dots$ con $\abs{e_{j_n} - x} \leq 3^{-n}$: una
\omterm{def:b1:seq:subsequence}{subsucesión} que converge a $x$. Así pues, el \omterm{def:b1:logic:sets}{conjunto} de límites de
subsucesiones de $(e_j)$ es exactamente $C$ --- una sola sucesión que
se acumula en no numerablemente muchos puntos, el extremo opuesto a
una sucesión convergente, cuyo \omterm{def:b1:logic:sets}{conjunto} de puntos de acumulación es
unitario.

\textbf{25.} (i) La construcción consumió: la estabilidad de los
\omterm{def:b1:topology:closed}{cerrados} por intersección arbitraria (la existencia de $C$ como
\omterm{def:b1:topology:closed}{cerrado}), el teorema de compacidad
\cref{thm:b1:topology:compact} (preguntas 2, 22, 23) y las
caracterizaciones secuenciales de la \omterm{def:b1:topology:closure}{clausura} y la adherencia (el
argumento de compactos encajados del~\cref{ex:b1:topology:nested} y
la pregunta 23). (ii) Las cuatro parejas: longitud cero y, aun así,
no numerable (preguntas 3, 9); \omterm{def:b1:topology:closed}{cerrado} y, aun así, de \omterm{def:b1:topology:closure}{interior}
vacío (pregunta 4); perfecto ---sin ningún punto aislado--- y, aun
así, totalmente disconexo (preguntas 11, 13); despreciable y, aun
así, con $C + C = \intcc{0}{2}$ (pregunta 16). (iii) La \omterm{def:b1:logic:inj}{sobreyección}
$h$ de la pregunta 20, hecha continua y no decreciente, se convierte
en la escalera del diablo en la teoría de las funciones continuas; y
en la teoría de la medida del volumen del tercer año, $C$ es el
testigo estándar de que «despreciable» no significa «numerable», con
su \omterm{def:b1:arith:prime}{primo} gordo (pregunta 22) separando «\omterm{def:b1:topology:dense}{denso} en ninguna parte» de
«despreciable».
\end{solution}
