\begin{solution}{exo:b1:vspaces:1}
\begin{enumerate}
  \item Sí: contiene a $0$ y la ecuación que lo define es lineal
        (estable por $x + \lambda y$).
  \item No: no contiene a $(0,0,0)$.
  \item No: $(1, 0)$ y $(0, -1)$ pertenecen ($xy = 0$) y su suma

\begin{solution}{exo:b1:vspaces:2}
$(1,2,1) = a(1,0,1) + b(1,1,0)$ exige $a + b = 1$, $b = 2$, $a = 1$:
incompatible ($a + b = 3 \neq 1$): no está en el \omterm{def:b1:vspaces:span}{subespacio
generado}. $(2,1,1) = a(1,0,1) + b(1,1,0)$: $b = 1$, $a = 1$,
$a + b = 2$: compatible, luego $(2,1,1) = (1,0,1) + (1,1,0)$, sí está
en el \omterm{def:b1:vspaces:span}{subespacio generado}.

Ecuación: $(x, y, z) = (a + b, b, a)$ significa $x = y + z$: el
\omterm{def:b1:vspaces:span}{subespacio generado} es el plano $\{x - y - z = 0\}$.
\end{solution}

\begin{solution}{exo:b1:vspaces:3}
Primera familia: $\lambda(1,1,0) + \mu(1,0,1) + \nu(0,1,1) = 0$ da
$\lambda + \mu = 0$, $\lambda + \nu = 0$, $\mu + \nu = 0$: sumando,
$2(\lambda + \mu + \nu) = 0$ y, restando cada ecuación,
$\lambda = \mu = \nu = 0$: \omterm{def:b1:vspaces:free}{libre}.

Segunda: $(2,4,6) = 2(1,2,3)$: ligada.

Tercera: cuatro vectores en $\R^3$ --- necesariamente ligados en
cuanto se disponga de la dimensión (\cref{ch:b1:findim});
directamente: $(0,1,1) = -(1,0,0) + 0\cdot(1,1,0) + (1,1,1)$, pues en
efecto $(-1,0,0) + (1,1,1) = (0,1,1)$: una relación no trivial.
\end{solution}

\begin{solution}{exo:b1:vspaces:4}
Los \omterm{def:b1:poly:def}{polinomios} $1, (X-1), (X-1)^2, (X-1)^3$ tienen grados distintos
$0, 1, 2, 3$: \omterm{def:b1:vspaces:free}{libres} (\cref{prop:b1:vspaces:freecriteria}~(1)); y son
cuatro vectores \omterm{def:b1:vspaces:free}{libres} y generadores (todo $P \in \R_3[X]$ se
desarrolla en potencias de $X - 1$, por ejemplo por Taylor para
\omterm{def:b1:poly:def}{polinomios}, cf.\ la demostración de
la~\cref{prop:b1:poly:multiplicity}): una \omterm{def:b1:vspaces:free}{base}. Para $X^3$, Taylor en
$1$: $P = X^3$, $P(1) = 1$, $P'(1) = 3$, $P''(1) = 6$,
$P'''(1) = 6$:
\[
X^3 = 1 + 3(X - 1) + 3(X-1)^2 + (X-1)^3 ,
\]
\omterm{prop:b1:vspaces:coordinates}{coordenadas} $(1, 3, 3, 1)$ (la fila de Pascal, como cabía esperar de
$X^3 = ((X-1)+1)^3$).
\end{solution}

\begin{solution}{exo:b1:vspaces:5}
$F \cap G$: las condiciones $x = y = z$ y $x = t = 0$ juntas dan
$x = 0$, luego $y = z = 0$, y $t = 0$: la intersección es $\{0\}$.
Suma: dado $(x,y,z,t)$, búsquense $(a,a,a,b) \in F$ y
$(0,c,d,0) \in G$ que sumen ese vector: $a = x$, $b = t$,
$c = y - x$, $d = z - x$: siempre es posible. Luego
$\R^4 = F \oplus G$, y
\[
(1,2,3,4) = (1,1,1,4) + (0,1,2,0) .
\]
\end{solution}

\begin{solution}{exo:b1:vspaces:6}
Un elemento de $G$ es $\lambda u$; si converge, entonces (dado que
$\lambda u_n = \lambda(-1)^n$ tiene los dos límites de subsucesiones
$\pm \lambda$) necesariamente $\lambda = 0$: $F \cap G = \{0\}$.

$F + G$ no lo es todo: consta de las sucesiones de la forma
$c_n + \lambda(-1)^n$ con $(c_n)$ convergente. La sucesión $v_n = n$
no es de esa forma ($v_n - \lambda(-1)^n$ no está acotada, luego
nunca converge). Así, $F \oplus G \subsetneq$ (espacio de todas las
sucesiones).
\end{solution}

\begin{solution}{exo:b1:vspaces:7}
($\supseteq$) Tanto $F \cap G$ como $F \cap H$ están en $F$, y su
suma está en $G + (F \cap H)$: la inclusión se sigue porque el
miembro izquierdo es un \omterm{def:b1:vspaces:subspace}{subespacio} que contiene las dos piezas ---
concretamente, un elemento $g + h$ con $g \in F\cap G$ y
$h \in F \cap H$ está en $F$ (suma de dos elementos de $F$) y en
$G + (F \cap H)$.

($\subseteq$) Sea $x \in F$ con $x = g + h$, $g \in G$,
$h \in F \cap H$. Entonces $g = x - h \in F$ (diferencia de elementos
de $F$), luego $g \in F \cap G$ y $x = g + h \in (F \cap G) + (F \cap H)$.

Contraejemplo para la distributividad completa en $\R^2$:
$F = \operatorname{Vect}(1,1)$, $G = \operatorname{Vect}(1,0)$,
$H = \operatorname{Vect}(0,1)$. Entonces $G + H = \R^2$, luego
$F \cap (G+H) = F$, mientras que $F \cap G = F \cap H = \{0\}$: el
miembro derecho es $\{0\} \neq F$.
\end{solution}

\begin{solution}{exo:b1:vspaces:8}
\begin{enumerate}
  \item Supóngase $\sum_{i} \lambda_i \eu^{a_i x} = 0$ para todo
        $x$. Multiplíquese por $\eu^{-a_p x}$: $\lambda_p +
        \sum_{i < p} \lambda_i \eu^{(a_i - a_p)x} \to \lambda_p$
        cuando $x \to +\infty$ (cada exponente $a_i - a_p < 0$). El
        miembro izquierdo es idénticamente $0$, luego
        $\lambda_p = 0$; e itérese hacia abajo.
  \item Sea $a\cos x + b \sin x + c \cos 2x + d\sin 2x = 0$ para
        todo $x$. Evalúese en $x = 0$: $a + c = 0$; en $x = \pi$:
        $-a + c = 0$; luego $a = c = 0$, y la relación se reduce a
        $b\sin x + d \sin 2x = 0$. Evalúese en $x = \frac\pi2$:
        $b = 0$; y después en $x = \frac\pi4$: $d = 0$.
  \item Supóngase $\sum \lambda_i \abs{x - a_i} = 0$ para todo $x$.
        La función $\sum_{i \neq j} \lambda_i\abs{x - a_i}$ es
        \omterm{def:b1:derivative:def}{derivable} en $a_j$ (cada término lo es, lejos de su propio
        pico), luego $-\lambda_j \abs{x - a_j}$, que es su
        diferencia, también tiene que ser \omterm{def:b1:derivative:def}{derivable} en $a_j$ --- lo
        que fuerza $\lambda_j = 0$ ($\abs{\,\cdot\,}$ tiene un pico).
        Y esto vale para todo $j$.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:vspaces:9}
Sea $h \in H$. Como $h \in H \subseteq F + H = F + G$, escríbase
$h = f + g$ con $f \in F$, $g \in G$. Entonces $f = h - g \in H$ (los
dos términos están en $H$, usando $G \subseteq H$), luego
$f \in F \cap H = F \cap G \subseteq G$, y $h = f + g \in G$. Por
tanto, $H \subseteq G$ y, con la hipótesis $G \subseteq H$: igualdad.

Contraejemplo sin $G \subseteq H$: en $\R^2$, tómense
$F = \operatorname{Vect}(1,0)$, $G = \operatorname{Vect}(0,1)$,
$H = \operatorname{Vect}(1,1)$: entonces $F + G = F + H = \R^2$ y
$F \cap G = F \cap H = \{0\}$ y, sin embargo, $G \neq H$.
\end{solution}

\begin{solution}{exo:b1:vspaces:10}
Los dos \omterm{def:b1:logic:sets}{conjuntos} son \omterm{def:b1:vspaces:subspace}{subespacios} (las condiciones que los
definen son lineales y valen para $0$). Descomposición: para
$P \in \R[X]$,
\[
P(X) = \underbrace{\frac{P(X) + P(-X)}{2}}_{\in\,\mathcal P}
+ \underbrace{\frac{P(X) - P(-X)}{2}}_{\in\,\mathcal I},
\]
y un \omterm{def:b1:poly:def}{polinomio} a la vez par e impar cumple $P = -P$, luego $P = 0$:
la suma es directa y vale $\R[X]$.

Sea ahora $P = \sum_k a_k X^k$ par. Entonces
$P(X) - P(-X) = 2 \sum_{k \text{ impar}} a_k X^k$ es el \omterm{def:b1:poly:def}{polinomio}
nulo, de modo que todos los coeficientes de grado impar se anulan
(\cref{def:b1:poly:def}): $P \in \operatorname{Vect}(1, X^2, X^4,
\dots)$, es decir, $P = Q(X^2)$ para cierto \omterm{def:b1:poly:def}{polinomio} $Q$.
Recíprocamente, todo \omterm{def:b1:poly:def}{polinomio} en $X^2$ es par.
\end{solution}

\begin{solution}{exo:b1:vspaces:11}
Supóngase $a(x_1 + x_2) + b(x_2 + x_3) + c(x_3 + x_1) = 0$.
Reagrupando sobre la \omterm{def:b1:vspaces:free}{familia libre} $(x_1, x_2, x_3)$:
\[
(a + c)\,x_1 + (a + b)\,x_2 + (b + c)\,x_3 = 0
\implies a + c = a + b = b + c = 0 .
\]
Restando las dos primeras ecuaciones se obtiene $c = b$; la tercera
da entonces $2b = 0$, luego $b = c = 0$ y después $a = 0$: la familia
es \omterm{def:b1:vspaces:free}{libre}.

Para cuatro vectores, la familia análoga es \emph{siempre} ligada:
\[
(x_1 + x_2) - (x_2 + x_3) + (x_3 + x_4) - (x_4 + x_1) = 0
\]
es una combinación nula no trivial (coeficientes $1, -1, 1, -1$), sea
cual sea $(x_1, x_2, x_3, x_4)$. Lo decide la paridad de la longitud
del ciclo.
\end{solution}

\begin{solution}{exo:b1:vspaces:12}
\begin{enumerate}
  \item Si $F_1 \subseteq F_2$ o $F_2 \subseteq F_1$, la unión es
        uno de los dos y, por tanto, es propia. En caso contrario,
        tómense $x \in F_1 \setminus F_2$ e $y \in F_2 \setminus F_1$
        y considérese $x + y$. Si $x + y \in F_1$, entonces
        $y = (x + y) - x \in F_1$: contradicción. Si $x + y \in F_2$,
        entonces $x \in F_2$: contradicción. Luego
        $x + y \notin F_1 \cup F_2$ y $E \neq F_1 \cup F_2$.
  \item Supóngase por reducción al absurdo que
        $E = F_1 \cup \dots \cup F_k$, con $k$ elegido \emph{mínimo}
        entre todos esos recubrimientos. La minimalidad prohíbe
        $F_1 \subseteq F_2 \cup \dots \cup F_k$ (si no, quítese
        $F_1$), luego hay $x \in F_1$ con $x \notin F_i$ para todo
        $i \geq 2$. Como $F_1$ es propio, tómese $y \notin F_1$.
        Para cada escalar $t$, el vector $y + t x$ está en algún
        $F_i$. Y nunca está en $F_1$: si no,
        $y = (y + tx) - tx \in F_1$ (pues $x \in F_1$). El \omterm{def:b1:structures:field}{cuerpo}
        es infinito, así que elíjanse $k$ escalares distintos
        $t_1, \dots, t_k$: los $k$ vectores $y + t_j x$ caen en los
        $k - 1$ \omterm{def:b1:vspaces:subspace}{subespacios} $F_2, \dots, F_k$, y dos de ellos,
        digamos $y + t x$ e $y + t' x$ con $t \neq t'$, están en el
        mismo $F_i$ ($i \geq 2$). Entonces su diferencia
        $(t - t')x \in F_i$, luego $x \in F_i$: contradicción. Por
        tanto, no existe ningún recubrimiento finito por
        \omterm{def:b1:vspaces:subspace}{subespacios} propios.
\end{enumerate}
\end{solution}

\begin{solution}{pb:b1:vspaces:1}
\textbf{1.} $L_i$ es un producto de $n$ factores lineales dividido
por una constante no nula (los $x_i$ son distintos), luego
$\deg L_i = n$. Evaluando en $x_j$ con $j \neq i$: el factor
$X - x_j$ del numerador se anula, luego $L_i(x_j) = 0$. Y en $x_i$,
numerador y denominador coinciden: $L_i(x_i) = 1$.

\textbf{2.} Supóngase $\sum_i \lambda_i L_i = 0$. Evalúese en $x_j$:
todos los términos mueren salvo $\lambda_j L_j(x_j) = \lambda_j$,
luego $\lambda_j = 0$ para todo $j$: la familia es \omterm{def:b1:vspaces:free}{libre}.

\textbf{3.} Sea $D = P - \sum_i P(x_i) L_i$. Entonces
$\deg D \leq n$ y, por la pregunta 1,
$D(x_j) = P(x_j) - P(x_j) = 0$ en los $n + 1$ puntos distintos
$x_0, \dots, x_n$. Un \omterm{def:b1:poly:def}{polinomio} no nulo de grado $\leq n$ tiene a lo
sumo $n$ raíces (\cref{cor:b1:poly:nroots}), luego $D = 0$. Así, todo
$P \in \R_n[X]$ es combinación de los $L_i$: la familia es generadora
y, con la pregunta 2, una \omterm{def:b1:vspaces:free}{base}.

\textbf{4.} Dados $y_0, \dots, y_n$, el \omterm{def:b1:poly:def}{polinomio}
$P = \sum_i y_i L_i$ tiene grado $\leq n$ e interpola. Unicidad: un
$P$ interpolador tiene, por la pregunta 3, \omterm{prop:b1:vspaces:coordinates}{coordenadas}
$(P(x_0), \dots, P(x_n)) = (y_0, \dots, y_n)$ en la base $(L_i)$, y
las \omterm{prop:b1:vspaces:coordinates}{coordenadas} en una \omterm{def:b1:vspaces:free}{base} son únicas
(\cref{prop:b1:vspaces:coordinates}). Las \omterm{prop:b1:vspaces:coordinates}{coordenadas} de $P$ en la
\omterm{def:b1:vspaces:free}{base} de Lagrange son sus \emph{valores en los nodos} --- ese es todo
el sentido de esa \omterm{def:b1:vspaces:free}{base}.

\textbf{5.} Aplíquese la pregunta 3 a $P = X^k$ ($0 \leq k \leq n$):
\[
X^k = \sum_{i=0}^{n} x_i^{k} L_i ,
\]
y $k = 0$ da $\sum_i L_i = 1$.

\textbf{6.} $\deg N_k = k$ exactamente: la familia
$(N_0, \dots, N_n)$ es una escalera de grados en $\R_n[X]$ y, por
tanto, una \omterm{def:b1:vspaces:free}{base} por el~\cref{ex:b1:vspaces:staircase} (que sea \omterm{def:b1:vspaces:free}{libre},
por la~\cref{prop:b1:vspaces:freecriteria}~(1); que sea generadora,
por descenso finito sobre el grado).

\textbf{7.} De la recurrencia,
\[
f[x_0, x_1] = \frac{f(x_1) - f(x_0)}{x_1 - x_0},
\qquad
f[x_0, x_1, x_2] = \frac{f[x_1, x_2] - f[x_0, x_1]}{x_2 - x_0}.
\]
Para $f(x) = x^2$:
\[
f[x_0, x_1] = \frac{x_1^2 - x_0^2}{x_1 - x_0} = x_0 + x_1,
\]
y entonces
\[
f[x_0, x_1, x_2]
= \frac{(x_1 + x_2) - (x_0 + x_1)}{x_2 - x_0}
= \frac{x_2 - x_0}{x_2 - x_0} = 1 .
\]

\textbf{8.} $\deg S \leq n$, ya que $Q, R$ tienen grado $\leq n - 1$.
En $x_0$: $S(x_0) = \frac{-(x_0 - x_n) R(x_0)}{x_n - x_0} =
R(x_0) = f(x_0)$. En $x_n$:
$S(x_n) = \frac{(x_n - x_0) Q(x_n)}{x_n - x_0} = Q(x_n) = f(x_n)$. Y
en un nodo \omterm{def:b1:topology:closure}{interior} $x_i$ ($1 \leq i \leq n-1$), tanto $Q$ como $R$
toman el valor $f(x_i)$, luego
\[
S(x_i) = \frac{(x_i - x_0) - (x_i - x_n)}{x_n - x_0}\, f(x_i)
= f(x_i) .
\]

\textbf{9.} Inducción sobre el número de nodos. Con un nodo: el
interpolador es la constante $f(x_0) = f[x_0]$. Supóngase la
afirmación para $k$ nodos y sea $S$ el interpolador en
$x_0, \dots, x_k$; por unicidad (pregunta 4), $S$ viene dado por el
lema de Aitken a partir de $R$ (nodos $x_0, \dots, x_{k-1}$) y de $Q$
(nodos $x_1, \dots, x_k$). El coeficiente de $X^{k}$ en $S$ es
\[
\frac{[X^{k-1}]\,Q - [X^{k-1}]\,R}{x_k - x_0}
= \frac{f[x_1, \dots, x_k] - f[x_0, \dots, x_{k-1}]}{x_k - x_0}
= f[x_0, \dots, x_k]
\]
por la hipótesis de inducción y por la recurrencia que las define.

\textbf{10.} Sea $P_k$ el interpolador de $f$ en $x_0, \dots, x_k$.
La diferencia $P_k - P_{k-1}$ tiene grado $\leq k$ y se anula en
$x_0, \dots, x_{k-1}$, de modo que, por el teorema del factor
aplicado $k$ veces (\cref{thm:b1:poly:factor}), vale $c\,N_k$ para
cierta constante $c$; comparando los coeficientes de $X^{k}$ y
usando la pregunta 9, $c = f[x_0, \dots, x_k]$. Telescopando desde
$P_0 = f(x_0) N_0$ se obtiene la fórmula de Newton. Para la forma
\omterm{def:b1:topology:closed}{cerrada}, escríbase $P_k = \sum_{i \leq k} f(x_i) L_i$ (Lagrange,
sobre los nodos $x_0, \dots, x_k$) y léase el coeficiente de
$X^{k}$: cada $L_i$ aporta $\frac{1}{\prod_{j \neq i}(x_i - x_j)}$,
de donde
\[
f[x_0, \dots, x_k] = \sum_{i=0}^{k}
\frac{f(x_i)}{\prod_{j \neq i,\, j \leq k}(x_i - x_j)} .
\]
El miembro derecho es invariante por cualquier \omterm{def:b1:counting:objects}{permutación} de los
nodos, así que la diferencia dividida no depende de su orden.

\textbf{11.} Si $P = a X^m + (\text{grados inferiores})$, el teorema
del binomio da
\[
\Delta P = a\bigl((X+1)^m - X^m\bigr) + \dots
= a\,m\,X^{m-1} + (\text{lower degrees}),
\]
porque $(X+1)^m - X^m = m X^{m-1} + \dots$ y la parte de grado
inferior de $P$ aporta grado $\leq m - 2$ tras $\Delta$ (o términos
de grado $\leq m-2$). Luego $\deg \Delta P = m - 1$ con coeficiente
director $m a$. Y una constante $c$ da $\Delta c = c - c = 0$.

\textbf{12.} $\deg B_k = k$: escalera y, por tanto, \omterm{def:b1:vspaces:free}{base} de
$\R_n[X]$ (\cref{ex:b1:vspaces:staircase}). Para $\Delta B_k$
($k \geq 1$), sáquese factor común el producto:
\begin{align*}
k!\,\Delta B_k
&= (X+1)X\cdots(X-k+2) - X(X-1)\cdots(X-k+1) \\
&= X(X-1)\cdots(X-k+2)\,\bigl[(X+1) - (X-k+1)\bigr] \\
&= k\,X(X-1)\cdots(X-k+2),
\end{align*}
luego $\Delta B_k = \frac{X(X-1)\cdots(X-k+2)}{(k-1)!} = B_{k-1}$.

\textbf{13.} Escríbase $P = \sum_{k=0}^{n} c_k B_k$ (\omterm{def:b1:vspaces:free}{base},
pregunta 12). Aplíquese $\Delta^{j}$: por la pregunta 12,
$\Delta^{j} P = \sum_{k \geq j} c_k B_{k-j}$. Evalúese en $0$:
$B_0(0) = 1$ y $B_m(0) = 0$ para $m \geq 1$ (el factor $X$ se
anula), luego $\bigl(\Delta^{j}P\bigr)(0) = c_j$. Esta es la fórmula
de las diferencias progresivas.

\textbf{14.} Inducción sobre $k$. Para $k = 0$, la identidad dice
$P(0) = P(0)$. Supóngase para $k$ y aplíquese a $\Delta P$:
\[
\bigl(\Delta^{k+1} P\bigr)(0)
= \sum_{j=0}^{k} (-1)^{k-j}\binom kj \bigl(P(j+1) - P(j)\bigr).
\]
Recójase el coeficiente de $P(i)$: es
$(-1)^{k-i+1}\binom k{i-1}\cdot(-1)^{0}$ de la primera suma
(desplazada) y $-(-1)^{k-i}\binom ki$ de la segunda --- juntos,
\[
(-1)^{k+1-i}\Bigl(\binom k{i-1} + \binom ki\Bigr)
= (-1)^{k+1-i}\binom{k+1}i
\]
por la regla de Pascal, que es la identidad en el rango $k + 1$.

\textbf{15.} Iterando la pregunta 11 desde el grado $n$ con
coeficiente director $a_n$: tras un $\Delta$, grado $n-1$ y
coeficiente director $n a_n$; tras dos, $n(n-1)a_n$; y tras $n$
pasos, grado $0$ y valor $n(n-1)\cdots 1\, a_n = n!\,a_n$, una
constante. Un $\Delta$ más la mata: $\Delta^{n+1}P = 0$.

\textbf{16.} Si $m \geq k$: $B_k(m) = \binom mk \in \N$. Si
$0 \leq m < k$: uno de los factores de $m(m-1)\cdots(m-k+1)$ es
cero, luego $B_k(m) = 0$. Y si $m = -q$ con $q \geq 1$:
\[
B_k(-q) = \frac{(-q)(-q-1)\cdots(-q-k+1)}{k!}
= (-1)^k\,\frac{q(q+1)\cdots(q+k-1)}{k!}
= (-1)^k \binom{q+k-1}{k},
\]
un entero. Así, todo $B_k$ lleva $\Z$ dentro de $\Z$.

\textbf{17.} ($\Leftarrow$) Si $P = \sum_k c_k B_k$ con
$c_k \in \Z$, entonces, para $m \in \Z$,
$P(m) = \sum_k c_k B_k(m) \in \Z$ por la pregunta 16.
($\Rightarrow$) Si $P$ es de valores enteros, sus \omterm{prop:b1:vspaces:coordinates}{coordenadas} son
$c_k = \bigl(\Delta^k P\bigr)(0) = \sum_{j=0}^k (-1)^{k-j}\binom kj
P(j)$ (preguntas 13 y 14), una combinación entera de los enteros
$P(0), \dots, P(k)$. Esta es la caracterización de Pólya de los
\omterm{pb:b1:vspaces:1}{polinomios de valores enteros}.

\textbf{18.} Póngase $Q(X) = P(X + a)$, un \omterm{def:b1:poly:def}{polinomio} de grado
$\leq n$ con $Q(0), Q(1), \dots, Q(n) \in \Z$. Sus \omterm{prop:b1:vspaces:coordinates}{coordenadas} en
$(B_k)_{k \leq n}$ son
$c_k = \sum_{j \leq k}(-1)^{k-j}\binom kj Q(j) \in \Z$ (la
pregunta 14 solo usa los valores en $0, \dots, k \leq n$). Por la
pregunta 17 ($\Leftarrow$), $Q$ es de valores enteros en todo $\Z$, y
por tanto también lo es $P(X) = Q(X - a)$.

\textbf{19.} Un producto de $k$ enteros consecutivos es
$m(m-1) \cdots(m-k+1) = k!\,B_k(m)$ para cierto $m \in \Z$, y
$B_k(m) \in \Z$ por la pregunta 16: el producto es divisible por $k!$.

\textbf{20.} Valores de $P = \frac{X(X+1)(2X+1)}{6}$ en $0,1,2,3$:
$0, 1, 5, 14$. Tabla de diferencias: fila $\Delta$: $1, 4, 9$; fila
$\Delta^2$: $3, 5$; fila $\Delta^3$: $2$. Por tanto, por la
pregunta 13,
\[
P = 0\cdot B_0 + 1\cdot B_1 + 3\,B_2 + 2\,B_3 ,
\]
con \omterm{prop:b1:vspaces:coordinates}{coordenadas} enteras: $P$ es de valores enteros (pregunta 17),
mientras que sus coeficientes monomiales $\frac13, \frac12,
\frac16$ no son enteros. Cálculo directo:
\begin{align*}
\Delta P &= \frac{(X+1)(X+2)(2X+3) - X(X+1)(2X+1)}{6} \\
&= \frac{(X+1)\bigl[(X+2)(2X+3) - X(2X+1)\bigr]}{6}
= \frac{(X+1)(6X+6)}{6} = (X+1)^2 .
\end{align*}
Telescopando, $P(m) = \sum_{j=0}^{m-1}\Delta P(j) = \sum_{j=1}^{m}
j^2$ (con $P(0) = 0$): la fórmula de la suma de cuadrados.

\textbf{21.} Los valores $2^i$ en $i = 0, \dots, n$ tienen una tabla
de diferencias constantemente igual a $1$ en el borde izquierdo:
$\Delta^k$ de la sucesión $(2^i)$ es otra vez $(2^i)$ (ya que
$2^{i+1} - 2^i = 2^i$), luego $\bigl(\Delta^k P\bigr)(0) = 2^0 = 1$
para todo $k \leq n$, y $P = B_0 + B_1 + \dots + B_n$ por la
pregunta 13. Entonces
\[
P(n+1) = \sum_{k=0}^{n}\binom{n+1}{k}
= 2^{n+1} - \binom{n+1}{n+1} = 2^{n+1} - 1 \neq 2^{n+1}:
\]
el patrón se rompe en el primer punto no controlado.

\textbf{22.} Por la pregunta 12, $B_k = \Delta B_{k+1}$, luego
\[
\sum_{j=0}^{m-1} B_k(j)
= \sum_{j=0}^{m-1}\bigl(B_{k+1}(j+1) - B_{k+1}(j)\bigr)
= B_{k+1}(m) - B_{k+1}(0) = B_{k+1}(m).
\]
Para $j < k$ los términos $B_k(j)$ se anulan, así que la suma empieza
en realidad en $j = k$: $\sum_{j=k}^{m-1}\binom jk = \binom m{k+1}$,
la identidad del palo de hockey.

\textbf{23.} Las tablas de diferencias (o el desarrollo directo) dan
\[
X^2 = B_1 + 2 B_2, \qquad X^3 = B_1 + 6 B_2 + 6 B_3
\]
(compruébese: $B_1 + 2B_2 = X + X(X-1) = X^2$; y en $X = 1, 2, 3$ la
segunda da $1, 8, 27$). La pregunta 22 da entonces
\[
\sum_{j=0}^{m-1} j^2 = B_2(m) + 2B_3(m)
= \binom m2 + 2\binom m3 = \frac{m(m-1)(2m-1)}{6},
\]
\[
\sum_{j=0}^{m-1} j^3 = B_2(m) + 6B_3(m) + 6B_4(m)
= \binom m2 + 6\binom m3 + 6\binom m4 .
\]
Desarrollando la última expresión:
$\binom m2 + 6\binom m3 + 6\binom m4 = \frac{m(m-1)}{2}\bigl[1 +
2(m-2) + \frac{(m-2)(m-3)}{2}\bigr] = \frac{m^2(m-1)^2}{4} =
\binom m2^2$. Sustituyendo $m$ por $m + 1$:
$1^3 + \dots + m^3 = \bigl(\frac{m(m+1)}2\bigr)^2 =
(1 + \dots + m)^2$, la identidad de Nicómaco.

\textbf{24.} Nodos $0, 1, 2$ y \omterm{def:b1:poly:def}{polinomio} $X^2$. \omterm{def:b1:vspaces:free}{Base} monomial
$(1, X, X^2)$: \omterm{prop:b1:vspaces:coordinates}{coordenadas} $(0, 0, 1)$. \omterm{def:b1:vspaces:free}{Base} de Lagrange: las
\omterm{prop:b1:vspaces:coordinates}{coordenadas} son los valores $(0, 1, 4)$ (pregunta 4). \omterm{def:b1:vspaces:free}{Base} de
Newton $(1, X, X(X-1))$: \omterm{pb:b1:vspaces:1}{diferencias divididas} $f[0] = 0$,
$f[0,1] = 1$, $f[0,1,2] = \frac{3 - 1}{2} = 1$ (pregunta 9), luego
\omterm{prop:b1:vspaces:coordinates}{coordenadas} $(0, 1, 1)$ --- y, en efecto, $X + X(X-1) = X^2$. Tres
\omterm{def:b1:vspaces:free}{bases}, tres vectores de \omterm{prop:b1:vspaces:coordinates}{coordenadas}, un solo \omterm{def:b1:poly:def}{polinomio}.

\textbf{25.} (i) La pregunta 4 es automática porque $(L_i)$ es una
\emph{\omterm{def:b1:vspaces:free}{base}}: la existencia y la unicidad de la interpolación son
exactamente la existencia y la unicidad de las \omterm{prop:b1:vspaces:coordinates}{coordenadas}. (ii) La
\omterm{def:b1:vspaces:free}{base} de Newton es una escalera, de modo que las \omterm{prop:b1:vspaces:coordinates}{coordenadas} se
calculan por divisiones sucesivas ---cada nodo nuevo añade un término
sin perturbar los anteriores---, mientras que las \omterm{prop:b1:vspaces:coordinates}{coordenadas} de $P$
en la de Lagrange son los valores $P(x_i)$, disponibles sin ningún
cálculo. (iii) Las dos \omterm{def:b1:vspaces:free}{bases} son \omterm{def:b1:vspaces:free}{libres} por los dos mismos criterios
de la~\cref{prop:b1:vspaces:freecriteria}: grados distintos para
Newton y evaluación en los nodos para Lagrange. (iv) El teorema de
Pólya dice que «$P(\Z) \subseteq \Z$», una propiedad de los valores,
equivale a que sean enteras las \omterm{prop:b1:vspaces:coordinates}{coordenadas} en la base $(B_k)$ ---
la aritmética de un \omterm{def:b1:poly:def}{polinomio} solo se vuelve visible en la \omterm{def:b1:vspaces:free}{base}
adaptada a la pregunta.
\end{solution}
