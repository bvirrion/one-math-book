\section*{अध्याय \ref{ch:b1:multivar} --- दो चरों के फलन}

\begin{solution}{exo:b1:multivar:1}
$\dfrac{\partial f}{\partial x} = 2xy + y\,\eu^{xy}$,
$\dfrac{\partial f}{\partial y} = x^2 + x\,\eu^{xy}$।

$\dfrac{\partial g}{\partial x} = \dfrac{2x}{x^2+y^2}$,
$\dfrac{\partial g}{\partial y} = \dfrac{2y}{x^2+y^2}$।

$\dfrac{\partial h}{\partial x} = \dfrac{-y/x^2}{1 + y^2/x^2} =
\dfrac{-y}{x^2+y^2}$,
$\dfrac{\partial h}{\partial y} = \dfrac{x}{x^2+y^2}$।
\end{solution}

\begin{solution}{exo:b1:multivar:2}
ध्रुवीय निर्देशांकों में ($\rho \to 0$):

$\abs{f} = \dfrac{\rho^3\abs{\cos^2\theta\sin\theta}}{\rho^2} \leq
\rho \to 0$: \omterm{def:b1:continuity:continuous}{संतत}।

$g = \cos\theta\sin\theta$: $\rho$ से स्वतंत्र, और अलग-अलग किरणों पर
अलग-अलग मान लेता है (तुलना कीजिए \cref{ex:b1:multivar:trap}): कोई सीमा
नहीं, अतः \omterm{def:b1:continuity:continuous}{संतत} नहीं।

$\abs h \leq \dfrac{\rho^3(\abs{\cos^3\theta} +
\abs{\sin^3\theta})}{\rho^2} \leq 2\rho \to 0$: \omterm{def:b1:continuity:continuous}{संतत}।
\end{solution}

\begin{solution}{exo:b1:multivar:3}
$y = mx$ के सहारे: $f(x, mx) = \dfrac{m x^3}{x^4 + m^2 x^2} =
\dfrac{mx}{x^2 + m^2} \to 0$ ($m \neq 0$ के लिए; $y = 0$ के सहारे और
$y$-अक्ष पर $f = 0$)। अतः हर सरल रेखा वाली सीमा $0$ है। पर परवलय $y = x^2$
पर:
\[
f(x, x^2) = \frac{x^4}{x^4 + x^4} = \frac12 :
\]
अनुक्रम $\bigl(\frac1n, \frac{1}{n^2}\bigr) \to (0,0)$ के लिए $f \to
\frac12 \neq 0$। \omterm{def:b1:continuity:continuous}{संतत} नहीं: दो चरों वाली सीमाएँ जाँचने के लिए रेखाएँ
पर्याप्त नहीं हैं।
\end{solution}

\begin{solution}{exo:b1:multivar:4}
$\frac{\partial f}{\partial x} = 3x^2y^2 + y\cos(xy)$; अतः
\[
\frac{\partial^2 f}{\partial y\,\partial x}
= 6x^2 y + \cos(xy) - xy\sin(xy) .
\]
$\frac{\partial f}{\partial y} = 2x^3 y + x\cos(xy)$; अतः
\[
\frac{\partial^2 f}{\partial x\,\partial y}
= 6x^2 y + \cos(xy) - xy\sin(xy) :
\]
दोनों बराबर, जैसा श्वार्ट्ज़ का वचन है।
\end{solution}

\begin{solution}{exo:b1:multivar:5}
$(x, y) = (\cos t, \sin t)$ के साथ \cref{thm:b1:multivar:chain} से:
\[
g'(t) = -\sin t\,\frac{\partial f}{\partial x}(\cos t, \sin t)
+ \cos t\,\frac{\partial f}{\partial y}(\cos t, \sin t)
= \bigl\langle \nabla f,\ (-\sin t, \cos t)\bigr\rangle .
\]
$g$ के किसी चरम पर $g'(t) = 0$: अर्थात् $\nabla f$ वृत्त के स्पर्श सदिश
$(-\sin t, \cos t)$ पर \omterm{def:b1:euclid:orthogonal}{लांबिक} है, अतः त्रिज्या सदिश $(\cos t, \sin t)$ के
समांतर (समतल में किसी एकक सदिश का \omterm{def:b1:euclid:orthogonal}{लांबिक पूरक} वही रेखा है जिसे वह जनित
करता है, लंब दिशा में ली गई)। यह लाग्रांज गुणक का सरलतम उदाहरण है।
\end{solution}

\begin{solution}{exo:b1:multivar:6}
$f$: $\nabla f = (2x + y - 3,\; x + 2y) = 0$: $y = -\frac x2$ और $2x -
\frac x2 = 3$: $x = 2$, $y = -1$। द्वितीय कोटि: $r = 2$, $s = 1$, $t = 2$:
$rt - s^2 = 3 > 0$, $r > 0$: $(2, -1)$ पर स्थानीय (और द्विघात होने के कारण
वस्तुतः समग्र) न्यूनतम, मान $f(2,-1) = -3$।

$g$: $\nabla g = (2x,\; -2y + 4) = 0$: बिंदु $(0, 2)$; $r = 2$, $s = 0$,
$t = -2$: $rt - s^2 = -4 < 0$: काठी।

$h$: $\nabla h = \bigl(4x^3 - 4(x - y),\; 4y^3 + 4(x-y)\bigr) = 0$। जोड़ने
पर: $x^3 + y^3 = 0$, अतः $y = -x$; प्रतिस्थापित करने पर: $4x^3 - 8x = 0$:
$x \in \{0, \pm\sqrt2\}$। \omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु}: $(0,0)$, $(\sqrt2, -\sqrt2)$,
$(-\sqrt2, \sqrt2)$।
$(\pm\sqrt2, \mp\sqrt2)$ पर: $r = 12x^2 - 4 = 20$, $s = 4$, $t = 20$: $rt
- s^2 > 0$, $r > 0$: स्थानीय न्यूनतम (मान $h = 4 + 4 - 16 = -8$)।
$(0,0)$ पर: $r = t = -4$, $s = 4$: $rt - s^2 = 0$: मौन। सीधी जाँच: छोटे $x
\neq 0$ के लिए $h(x, x) = 2x^4 > 0$ और $h(x, -x) = 2x^4 - 8x^2 < 0$: हर
\omterm{def:b1:topology:open}{प्रतिवेश} में दोनों चिह्न --- अतः मूल बिंदु पर कोई चरम नहीं।
\end{solution}

\begin{solution}{exo:b1:multivar:7}
$x^2 - y^2$: द्वितीय \omterm{def:b1:multivar:partial}{आंशिक अवकलज} $2$ और $-2$: योग $0$। $xy$: दोनों शुद्ध
द्वितीय \omterm{def:b1:multivar:partial}{आंशिक अवकलज} लुप्त हैं। $\eu^x\cos y$: $\frac{\partial^2}{\partial
x^2} = \eu^x\cos y$, $\frac{\partial^2}{\partial y^2} = -\eu^x\cos y$: योग
$0$। $\ln(x^2+y^2)$: \cref{exo:b1:multivar:1} से,
\[
\frac{\partial^2}{\partial x^2}\ln(x^2+y^2)
= \frac{2(x^2+y^2) - 4x^2}{(x^2+y^2)^2}
= \frac{2(y^2 - x^2)}{(x^2+y^2)^2},
\]
और $y$ वाला रूप उसका विपरीत है: योग $0$।
\end{solution}

\begin{solution}{exo:b1:multivar:8}
\emph{\omterm{def:b1:linmaps:projection}{प्रक्षेप} से:} समतल $P$ का अभिलंब $n = (1, 2, -1)$ है और वह $A = (4,
0, 0)$ से होकर जाता है। मूल बिंदु के सबसे निकट का बिंदु $O$ का \omterm{def:b1:linmaps:projection}{प्रक्षेप}
है: $p = O + \frac{\langle A - O, n\rangle}{\norm n^2}\,n =
\frac{4}{6}(1,2,-1) = \bigl(\frac23, \frac43, -\frac23\bigr)$, जिसकी दूरी
$\frac{4}{\sqrt 6}$ है। (सूत्र: मूल बिंदु से समतल $\langle X, n \rangle =
c$ तक की दूरी $\frac{\abs c}{\norm n}$ है, यहाँ $c = 4$।)

\emph{न्यूनीकरण से:} $f(x,y) = x^2 + y^2 + (x + 2y - 4)^2$। अंतिम गुणनखंड
के लिए $w = x + 2y - 4$ लिखने पर:
\[
\nabla f = \bigl(2x + 2w,\; 2y + 4w\bigr) = 0
\iff x = -w \text{ तथा } y = -2w .
\]
$w$ की परिभाषा में प्रतिस्थापित करने पर: $w = -w - 4w - 4$, अतः $w =
-\frac23$, जिससे $x = \frac23$, $y = \frac43$ और $z = w = -\frac23$।
\omterm{def:b1:linmaps:projection}{प्रक्षेप} वाला ही बिंदु, दूरी $\sqrt{\frac49 + \frac{16}{9} + \frac49} =
\frac{2\sqrt 6}{3} = \frac{4}{\sqrt6}$; और यह न्यूनतम है, क्योंकि अनंत पर
$f \to \infty$ (धनात्मक निश्चित द्विघात के साथ \omterm{def:b1:linmaps:def}{रैखिक} पद)।
\end{solution}

\begin{solution}{exo:b1:multivar:9}
$\rho = \sqrt{x^2+y^2}$ के साथ: $\frac{\partial \rho}{\partial x} =
\frac{x}{\rho}$, अतः
\[
\frac{\partial f}{\partial x} = \varphi'(\rho)\,\frac{x}{\rho},
\qquad
\frac{\partial^2 f}{\partial x^2}
= \varphi''(\rho)\,\frac{x^2}{\rho^2}
+ \varphi'(\rho)\,\Bigl(\frac{1}{\rho} -
\frac{x^2}{\rho^3}\Bigr),
\]
(भागफल और शृंखला नियम)। सममित $y$-व्यंजक जोड़ने पर: $\varphi''$ वाले पद
$\frac{x^2 + y^2}{\rho^2} = 1$ बटोरते हैं, और $\varphi'$ वाले पद
$\frac{2}{\rho} - \frac{x^2 + y^2}{\rho^3} = \frac{1}{\rho}$:
\[
\Delta f = \varphi'' + \frac{\varphi'}{\rho} .
\]
त्रिज्य प्रसंवादी फलन: $\varphi'' + \frac{\varphi'}{\rho} = 0$ हल कीजिए:
$u = \varphi'$ के लिए प्रथम-कोटि समीकरण $u' + \frac u\rho = 0$ $u =
\frac{c}{\rho}$ देता है (\cref{thm:b1:diffeq:homogeneous1}), और फिर
$\varphi = c\ln\rho + d$। अतः त्रिज्य प्रसंवादी फलन $c\,\ln\sqrt{x^2 +
y^2} + d$ हैं --- \cref{exo:b1:multivar:7} का लघुगणकीय विभव और अचर, इनके
सिवा कुछ नहीं।
\end{solution}

\begin{solution}{exo:b1:multivar:10}
$(1,1)$ पर $z = xy$ के लिए: \omterm{def:b1:multivar:partial}{आंशिक अवकलज} $y = 1$ और $x = 1$, अतः स्पर्श
समतल $z = 1 + (x - 1) + (y - 1) = x + y - 1$। स्पर्श समतल का क्षैतिज होना
दोनों आंशिक अवकलजों के लुप्त होने के बराबर है, अर्थात् $\nabla f = 0$:
\cref{ex:b1:multivar:extrema} से ठीक \omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु} $(0, 0)$ और $(1, 1)$,
जहाँ क्षैतिज समतल $z = 0$ और $z = -1$ हैं। ``क्षैतिज स्पर्श समतल'' और
``\omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु}'' एक ही धारणा है --- एक आलेख पर देखी गई, दूसरी सूत्र में।
\end{solution}

\begin{solution}{exo:b1:multivar:11}
\begin{enumerate}
  \item $y$ को स्थिर रखने पर एक चर वाले फलन $x \mapsto f(x,y)$ का
        \omterm{def:b1:derivative:def}{अवकलज} $\R$ पर शून्य है, अतः वह अचर है
        (\cref{thm:b1:derivative:mvt}): सभी $x$ के लिए $f(x, y) = f(0,
        y)$, अर्थात् $f$ केवल $y$ पर निर्भर करता है।
  \item मान लीजिए $g(u, v) = f\bigl(\frac{u+v}2, \frac{u-v}2\bigr)$।
        शृंखला नियम से (\cref{thm:b1:multivar:chain}, जिसे $u$ जमाकर
        चर $v$ में लगाया गया):
        \[
        \frac{\partial g}{\partial v}
        = \frac12\,\frac{\partial f}{\partial x}
        - \frac12\,\frac{\partial f}{\partial y} = 0 .
        \]
        (1) से $g$ केवल $u$ पर निर्भर करता है: $g(u, v) = \varphi(u)$,
        जहाँ $\varphi$ $C^1$ वर्ग का है, और चर-परिवर्तन उलटने पर ($u =
        x + y$, $v = x - y$),
        \[
        f(x, y) = \varphi(x + y).
        \]
        विलोमतः ऐसा हर $f$ $f_x = f_y = \varphi'$ को संतुष्ट करता है:
        अतः हल ठीक $x + y$ के $C^1$ फलन हैं।
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:multivar:12}
\omterm{def:b1:topology:open}{विवृत} चक्रिका पर कोई चरम क्रांतिक होगा: $\nabla f = (y, x) = 0$ केवल
$(0,0)$ पर, जहाँ $f = 0$; और वह काठी है ($f(\pm \varepsilon,
\pm\varepsilon) = \varepsilon^2 > 0 > -\varepsilon^2 = f(\pm\varepsilon,
\mp\varepsilon)$), अतः वहाँ कोई चरम नहीं। मानी हुई चरम मान प्रमेय से
परिबंध प्राप्त होते हैं, और अनिवार्यतः परिसीमा-वृत्त पर। वहाँ,
\cref{exo:b1:multivar:5} के साथ,
\[
f(\cos t, \sin t) = \cos t\sin t = \frac{\sin 2t}{2}
\in \intcc{-\tfrac12}{\tfrac12},
\]
जिसमें अधिकतम $\frac12$ $t = \frac\pi4, \frac{5\pi}4$ पर (बिंदु

\frac{7\pi}4$ पर (बिंदु $\pm\frac{1}{\sqrt2}(1,-1)$)। समग्र अधिकतम
$\frac12$, समग्र न्यूनतम $-\frac12$।
\end{solution}

\begin{solution}{pb:b1:multivar:1}
\textbf{1.} दाएँ पक्ष का प्रसार करने पर:
\[
r\Bigl(h + \frac srk\Bigr)^{\!2} + \frac{rt - s^2}{r}k^2
= rh^2 + 2shk + \frac{s^2}{r}k^2 + \frac{rt - s^2}{r}k^2
= q(h,k) .
\]
यदि $rt - s^2 > 0$: दोनों वर्ग $r$ के चिह्न का गुणक ढोते हैं, और $q(h,k)
= 0$ पहले $k = 0$ और फिर $h = 0$ बाध्य कर देता है: अतः मूल बिंदु के बाहर
चिह्न कठोरतः $r$ का। यदि $rt - s^2 < 0$: $q(1, 0) = r$ है, जबकि
$q\bigl(-\frac sr, 1\bigr) = \frac{rt - s^2}{r}$ का चिह्न विपरीत है: दोनों
चिह्न आते हैं।

\textbf{2.} यदि $r = 0 \neq t$, तो $h$ और $k$ की भूमिकाएँ बदल दीजिए ($k$
में पूर्ण वर्ग), निष्कर्ष वही रहते हैं; ध्यान दीजिए कि $s \neq 0$ होते ही
$rt - s^2 = -s^2 < 0$, और सचमुच तब $q = 2shk + tk^2$ दोनों चिह्न लेता है
($k$ को $h$ के मुक़ाबले छोटा लीजिए)। यदि $r = t = 0$: $q = 2shk$, जो दोनों
चिह्न लेता है यदि और केवल यदि $s \neq 0$, अर्थात् यदि और केवल यदि $rt -
s^2 = -s^2 < 0$। सारांश: $q$ दोनों चिह्न लेता है $\iff rt - s^2 < 0$; और
$q$ केवल मूल बिंदु पर लुप्त होता है (निश्चित) $\iff rt - s^2 > 0$ (जो $r
\neq 0$ बाध्य कर देता है, क्योंकि $r = 0$ पर $q(1,0) = 0$)।

\textbf{3.} $f_x = rx + sy + \beta$ तथा $f_y = sx + ty + \gamma$ के साथ
सीधा प्रसार देता है
\[
f(a + h, b + k) = f(a, b) + h\,f_x(a,b) + k\,f_y(a,b)
+ \tfrac12 q(h, k),
\]
और इससे ऊँचे पद नहीं हैं (फलन घात $2$ का \omterm{def:b1:poly:def}{बहुपद} है)। किसी \omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु} पर
\omterm{def:b1:linmaps:def}{रैखिक} भाग लुप्त हो जाता है: $f(X_0 + H) - f(X_0) = \frac12 q(H)$
\emph{यथार्थतः}, अतः प्रश्न 1--2 का चिह्न-अध्ययन वर्गीकरण कर देता है: कठोर
समग्र न्यूनतम ($rt - s^2 > 0$, $r > 0$), कठोर समग्र अधिकतम ($rt - s^2 >
0$, $r < 0$), और $rt - s^2 < 0$ पर काठी --- हर \omterm{def:b1:topology:open}{प्रतिवेश} में दोनों चिह्न।
इससे \cref{met:b1:multivar:monge} द्विघात फलनों के लिए सिद्ध हो जाती है।

\textbf{4.} रूप $q - c(h^2 + k^2)$ की राशियाँ $r - c$, $s$, $t - c$ हैं।
$c = \frac{rt - s^2}{r + t}$ के साथ (ध्यान दीजिए $t > 0$, क्योंकि $rt >
s^2 \geq 0$ और $r > 0$):
\[
(r - c)(t - c) - s^2 = rt - s^2 - c(r + t) + c^2 = c^2 \geq 0,
\]
और $r - c \geq 0$ ($c \leq r \iff rt - s^2 \leq r^2 + rt$, जो सत्य है)।
यदि $r - c > 0$, तो प्रश्न 1 का पूर्ण वर्ग $q - c(h^2 + k^2) \geq 0$ दिखा
देता है; और यदि $r - c = 0$, तो $s = 0$ (प्रदर्शित राशि $\geq 0$ रहने के
लिए, जबकि पहला गुणनखंड $0$ है) और रूप $(t - c)k^2 \geq 0$ रह जाता है।
दोनों स्थितियों में $q(h,k) \geq c(h^2 + k^2)$।

\textbf{5.} प्रश्न 3--4 से $f(X) = f(X_0) + \frac12 q(X - X_0) \geq f(X_0)
+ \frac c2\norm{X - X_0}^2 \to +\infty$ जब $\norm X \to \infty$: अतः $f$
प्रबल है, और $X \neq X_0$ के लिए असमिका कठोर है: \omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु} ही
अद्वितीय समग्र न्यूनकारी है। \cref{ex:b1:multivar:extrema} के त्रिघात $f =
x^3 + y^3 - 3xy$ के लिए ऐसा कोई निष्कर्ष नहीं टिकता: $f(x,0) = x^3 \to
-\infty$, और $(1,1)$ पर स्थानीय न्यूनतम समग्र नहीं है --- जो चूका, वह
द्विघात प्रसार की यथार्थता थी।

\textbf{6.} वर्ग मानक का प्रसार करने पर:
\[
f(u,v) = u^2\norm{C_1}^2 + 2uv\,\langle C_1, C_2\rangle +
v^2\norm{C_2}^2 - 2u\,\langle C_1, b\rangle - 2v\,\langle C_2,
b\rangle + \norm b^2 ,
\]
जो $(u, v)$ का द्विघात फलन है और जिसका द्विघात भाग $\frac12(ru^2 + 2suv +
tv^2)$ है, जहाँ $r = 2\norm{C_1}^2$, $s = 2\langle C_1, C_2\rangle$, $t =
2\norm{C_2}^2$।

\textbf{7.} कोशी--श्वार्ट्ज़ (\cref{thm:b1:euclid:cs}) से $rt - s^2 =
4\bigl(\norm{C_1}^2\norm{C_2}^2 - \langle C_1, C_2\rangle^2\bigr) \geq 0$,
और समता ठीक तब है जब $C_1, C_2$ समानुपाती हों (या उनमें से एक शून्य हो),
अर्थात् जब युग्म परतंत्र हो। अतः $rt - s^2 > 0 \iff (C_1, C_2)$ \omterm{def:b1:vspaces:free}{स्वतंत्र}।

\textbf{8.} $\nabla f = 0$ का अर्थ है
\[
\norm{C_1}^2 u + \langle C_1, C_2\rangle v = \langle C_1,
b\rangle,
\qquad
\langle C_1, C_2\rangle u + \norm{C_2}^2 v = \langle C_2,
b\rangle,
\]
अर्थात् बाईं ओर ग्राम आव्यूह वाले \omterm{pb:b1:multivar:1}{प्रसामान्य समीकरण}; और वे कहते हैं कि $i
= 1, 2$ के लिए $\langle uC_1 + vC_2 - b,\ C_i\rangle = 0$, अर्थात् $p =
uC_1 + vC_2$ के साथ $b - p \perp \operatorname{Vect}(C_1, C_2)$। भाग I
(प्रश्न 3, 5) से अद्वितीय \omterm{thm:b1:multivar:critical}{क्रांतिक बिंदु} ही अद्वितीय समग्र न्यूनकारी है,
और \cref{thm:b1:euclid:projection} से अभिलक्षणता ``$p \in F$, $b - p \perp
F$'' $p$ को $b$ का $F = \operatorname{Vect}(C_1, C_2)$ पर \omterm{thm:b1:euclid:projection}{लांबिक प्रक्षेप}
ठहरा देती है।

\textbf{9.} $\norm{b - p}^2 = \norm b^2 - 2\langle b, p\rangle + \norm
p^2$, और $\langle p, b - p\rangle = 0$ से $\norm p^2 = \langle p,
b\rangle$: अतः न्यूनतम $\norm b^2 - \langle p, b\rangle$ के बराबर है।
पाइथागोरस: $\norm b^2 = \norm p^2 + \norm{b - p}^2$ --- आँकड़ा-सदिश अपने
व्याख्यायित भाग $p$ और अपने अवशिष्ट भाग $b - p$ में बँट जाता है, जो परस्पर
\omterm{def:b1:euclid:orthogonal}{लांबिक} हैं।

\textbf{10.} दिए गए $C_1, C_2, b$ के साथ $E(\alpha, \beta) = \norm{\alpha
C_1 + \beta C_2 - b}^2$; और प्रश्न 8 के \omterm{pb:b1:multivar:1}{प्रसामान्य समीकरण}, प्रविष्टिवार,
ये हैं:
\[
n\alpha + \Bigl(\sum x_i\Bigr)\beta = \sum y_i,
\qquad
\Bigl(\sum x_i\Bigr)\alpha + \Bigl(\sum x_i^2\Bigr)\beta = \sum
x_i y_i .
\]

\textbf{11.} $n$ से भाग देने पर: $\alpha + \beta\bar x = \bar y$ और
$\alpha\bar x + \beta\bigl(v_x + \bar x^2\bigr) = c_{xy} + \bar x\bar y$।
दूसरे में $\alpha = \bar y - \beta\bar x$ प्रतिस्थापित करने पर: $\beta v_x
= c_{xy}$, अतः
\[
\beta = \frac{c_{xy}}{v_x}, \qquad \alpha = \bar y - \beta\bar
x ,
\]
और पहला समीकरण ठीक यही कहता है कि $(\bar x, \bar y)$ रेखा पर स्थित है।

\textbf{12.} $v_x = \frac1n\sum(x_i - \bar x)^2 \geq 0$, जो शून्य है यदि
और केवल यदि हर $x_i$ $\bar x$ के बराबर हो। और $rt - s^2 = 4\bigl(n\sum
x_i^2 - (\sum x_i)^2\bigr) = 4n^2 v_x$: अतः प्रश्न 7 की स्वतंत्रता-शर्त
$v_x > 0$ है, अर्थात् सभी $x_i$ बराबर न हों।

\textbf{13.} $\alpha = \bar y - \beta\bar x$ के साथ आँकड़ों को केंद्रित
कीजिए ($\tilde x_i = x_i - \bar x$, $\tilde y_i = y_i - \bar y$):
\[
E(\alpha, \beta) = \sum_i(\tilde y_i - \beta\tilde x_i)^2
= n\,v_y - 2\beta\,n\,c_{xy} + \beta^2 n\,v_x ,
\]
जो $\beta = c_{xy}/v_x$ पर न्यूनतम है, मान $E_{\min} = n\bigl(v_y -
\frac{c_{xy}^2}{v_x}\bigr) = n v_y(1 - \rho^2)$। चूँकि $E_{\min} \geq 0$
और $v_y > 0$: $\rho^2 \leq 1$, और $\abs\rho = 1$ यदि और केवल यदि $E_{\min}
= 0$, अर्थात् यदि और केवल यदि हर अवशेष लुप्त हो: बिंदु ठीक रेखा पर स्थित
हैं।

\textbf{14.} $n = 4$: $\bar x = \frac64 = 1.5$, $\bar y = 2$, $\sum x_i^2
= 14$ अतः $v_x = 3.5 - 2.25 = 1.25$, $\sum x_iy_i = 0 + 1 + 6 + 12 = 19$
अतः $c_{xy} = 4.75 - 3 = 1.75$। इससे
\[
\beta = \frac{1.75}{1.25} = 1.4,
\qquad
\alpha = 2 - 1.4\times1.5 = -0.1 :
\qquad y = 1.4\,x - 0.1 .
\]
फिट किए गए मान $-0.1,\ 1.3,\ 2.7,\ 4.1$; अवशेष $0.1,\ -0.3,\ 0.3,\ -0.1$;
$E_{\min} = 0.01 + 0.09 + 0.09 + 0.01 = 0.2$ (जाँच: $v_y = \frac{26}4 - 4
= 2.5$ और $4(2.5 - \frac{1.75^2}{1.25}) = 4(2.5 - 2.45) = 0.2$)।

\textbf{15.} प्रश्न 10 के दोनों \omterm{pb:b1:multivar:1}{प्रसामान्य समीकरण} ठीक $\sum_i\varepsilon_i
= 0$ और $\sum_i x_i\varepsilon_i = 0$ हैं: अवशेष-सदिश $C_1 = (1, \dots,
1)$ पर और $C_2 = (x_i)$ पर --- अर्थात् पूरी प्रतिरूप-समष्टि पर --- \omterm{def:b1:euclid:orthogonal}{लांबिक}
है। विशेष रूप से इष्टतम रेखा अपनी त्रुटियाँ सदा संतुलित रखती है: उनका योग
शून्य होता है।

\textbf{16.} $\frac{\dd}{\dd\alpha}\sum(y_i - \alpha)^2 = -2\sum(y_i -
\alpha) = 0$ से $\alpha = \bar y$ (और द्वितीय \omterm{def:b1:derivative:def}{अवकलज} $2n > 0$ इसे समग्र
न्यूनतम बना देता है, क्योंकि फलन एक चर का प्रबल द्विघात है)। न्यूनतम मान
$\sum(y_i - \bar y)^2 = n v_y$: अर्थात् प्रसरण किसी अचर प्रतिरूप की
अनिवार्य वर्ग-त्रुटि है।

\textbf{17.} $\frac{\dd}{\dd\beta}\sum(y_i - \beta x_i)^2 = -2\sum x_i(y_i
- \beta x_i) = 0$ से $\beta_0 = \frac{\sum x_iy_i}{\sum x_i^2}$। दोनों
प्रवणताओं को केंद्रित राशियों में लिखने पर: $\beta_0 = \frac{c_{xy} + \bar
x\bar y}{v_x + \bar x^2}$ $\frac{c_{xy}}{v_x}$ के बराबर है यदि और केवल यदि
$c_{xy}\bar x^2 = \bar x\bar y\,v_x$, अर्थात् यदि और केवल यदि $\bar x = 0$
(केंद्रित भुज) या $\bar y = \beta\bar x$ --- और दूसरे का अर्थ है $\alpha =
0$: पूरी \omterm{pb:b1:multivar:1}{प्रतिगमन रेखा} पहले से ही मूल बिंदु से होकर जाती है।

\textbf{18.} $g(\tau) = \norm{M - P_0}^2 - 2\tau\langle M - P_0, w\rangle
+ \tau^2$ $\tau^* = \langle M - P_0, w\rangle$ पर न्यूनतम है, जिसका मान
$d(M,D)^2 = \norm{M - P_0}^2 - \langle M - P_0, w\rangle^2$ है। समतल में
$w$ को $n = \frac{(a, b)}{\sqrt{a^2+b^2}}$ ($D$ का एकक अभिलंब) के साथ किसी
\omterm{def:b1:euclid:orthogonal}{लांबिक-प्रसामान्य} \omterm{def:b1:vspaces:free}{आधार} $(w, n)$ में पूरा कीजिए: तब $\norm{M - P_0}^2 =
\langle M - P_0, w\rangle^2 + \langle M - P_0, n\rangle^2$, अतः $d(M, D) =
\abs{\langle M - P_0, n\rangle}$, और $aP_{0x} + bP_{0y} = -c$ के साथ:
\[
d = \frac{\abs{a x_0 + b y_0 + c}}{\sqrt{a^2 + b^2}} .
\]

\textbf{19.} $E_{\min}/n = v_y - \frac{c_{xy}^2}{v_x}$ और $\beta^2 v_x =
\frac{c_{xy}^2}{v_x}$ से:
\[
v_y = \beta^2 v_x + \frac{E_{\min}}n :
\]
कुल प्रसरण $=$ फिट की गई रेखा के सहारे का प्रसरण $+$ अवशिष्ट प्रसरण।
($v_y$ से भाग देने पर: $1 = \rho^2 + (1 - \rho^2)$, अर्थात् रेखा से
``व्याख्यायित'' प्रसरण का हिस्सा $\rho^2$ है।)

\textbf{20.} प्रतिरूप-समष्टि $\operatorname{Vect}\bigl(C_1, C_2,
C_3\bigr)$ है, जहाँ $C_1 = (1)$, $C_2 = (x_i)$, $C_3 = (x_i^2)$;
$\norm{aC_1 + bC_2 + cC_3 - b}^2$ का न्यूनीकरण ठीक प्रश्न 8 की तरह
$3\times3$ ग्राम निकाय तक ले जाता है, जिसके आव्यूह की प्रविष्टियाँ $\langle
C_i, C_j\rangle = \sum_k x_k^{\,i+j-2}$ हैं: वही प्रदर्शित आघूर्ण आव्यूह।
वह व्युत्क्रमणीय है यदि और केवल यदि $(C_1, C_2, C_3)$ \omterm{def:b1:vspaces:free}{स्वतंत्र} हो
(\cref{exo:b1:euclid:11}); क्योंकि सभी $k$ के लिए संबंध $a + bx_k + cx_k^2
= 0$ हर $x_k$ को घात $\leq 2$ के एक ही \omterm{def:b1:poly:def}{बहुपद} का मूल बना देता है, जो तीन
भिन्न मानों के साथ असंभव है जब तक $a = b = c = 0$ न हो। \cref{ch:b1:det}
की सप्ताहांत समस्या के आघूर्ण आव्यूहों से तुलना कीजिए, जहाँ उनके सारणिक
वांडरमोंड मानों के वर्ग निकले थे।

\textbf{21.} पहला: $r = 2, s = 1, t = 2$, $rt - s^2 = 3 > 0$, $r > 0$:
मूल बिंदु पर कठोर समग्र न्यूनतम। दूसरा: $r = 2, s = 3, t = 2$, $rt - s^2
= -5 < 0$: काठी। तीसरा: $r = s = t = 2$, $rt - s^2 = 0$: अपह्रासी --- पर
$x^2 + 2xy + y^2 = (x + y)^2 \geq 0$ पूरी रेखा $y = -x$ पर लुप्त है: एक
(अकठोर) समग्र न्यूनतम, जो पूरी रेखा पर प्राप्त होता है और सारणिक-कसौटी को
दिखाई नहीं देता।

\textbf{22.} नए योग ($n = 5$): $\bar x = 3.2$, $\bar y = 1.6$, $\sum
x_iy_i = 19$ अतः $c_{xy} = 3.8 - 5.12 = -1.32$, $\sum x_i^2 = 114$ अतः
$v_x = 22.8 - 10.24 = 12.56$। नई \omterm{def:b1:multivar:partial}{प्रवणता}:
\[
\beta = \frac{-1.32}{12.56} \approx -0.105 :
\]
एक ही बिंदु ने स्पष्ट रूप से बढ़ती प्रवृत्ति ($\beta = 1.4$) को हलकी घटती
प्रवृत्ति में बदल दिया। वर्ग-त्रुटि हर अवशेष पर $\varepsilon^2$ वसूलती है,
अतः एक ही दूर बैठा बिंदु --- बड़े $\abs{x_{i} - \bar x}$ \emph{और} बड़े
अवशेष वाला --- $c_{xy}$ तथा $v_x$ दोनों पर हावी हो जाता है: \omterm{pb:b1:multivar:1}{न्यूनतम वर्ग}
दक्ष तो हैं, सुदृढ़ नहीं।

\textbf{23.} $W = \sum w_i$ रखिए और $\bar x_w = \frac1W\sum w_ix_i$, इसी
प्रकार $\bar y_w$, तथा $v_x^w = \frac1W\sum w_ix_i^2 - \bar x_w^2$, $c^w_{xy}
= \frac1W\sum w_ix_iy_i - \bar x_w\bar y_w$ परिभाषित कीजिए। \omterm{def:b1:logic:map}{प्रतिचित्रण}
$\langle u, v\rangle_w = \sum_i w_iu_iv_i$ $\R^n$ पर एक \omterm{def:b1:euclid:def}{अंतर्गुणन} है
($w_i > 0$ निश्चितता देता है), अतः भाग II अक्षरशः लागू होता है और
\omterm{pb:b1:multivar:1}{प्रसामान्य समीकरण} $W$ से भाग देने पर
\[
\alpha + \beta\bar x_w = \bar y_w,
\qquad
\alpha\bar x_w + \beta(v^w_x + \bar x_w^2) = c^w_{xy} + \bar
x_w\bar y_w
\]
बन जाते हैं, जिनसे $\beta = c^w_{xy}/v^w_x$ और $\alpha = \bar y_w -
\beta\bar x_w$: वही सूत्र, बस हर औसत भारित।

\textbf{24.} $E_{\min} = \sum\varepsilon_i^2 = 0$ हर $\varepsilon_i = 0$
बाध्य कर देता है: सभी बिंदु ठीक रेखा पर; और प्रश्न 13 से यही स्थिति
$\abs\rho = 1$ है। जाँच: $(1,1), (2,3), (3,5)$ के लिए $\bar x = 2$, $\bar
y = 3$, $v_x = \frac{14}3 - 4 = \frac23$, $c_{xy} = \frac{22}3 - 6 =
\frac43$: अतः $\beta = 2$, $\alpha = -1$, और सचमुच तीनों बिंदुओं के लिए
$y_i = 2x_i - 1$; साथ ही $v_y = \frac{35}3 - 9 = \frac83$ और $\rho^2 =
\frac{(4/3)^2}{(2/3)(8/3)} = 1$।

\textbf{25.} (क) द्विघात फलनों के लिए कोटि-दो प्रसार एक \emph{सर्वसमिका}
है, अतः द्विघात रूप का चिह्न-अध्ययन --- शुद्ध बीजगणित, प्रश्न 1--2 ---
क्रांतिक बिंदुओं का वर्गीकरण बिना किसी स्वीकृत टेलर प्रमेय के कर देता है।
(ख) \omterm{pb:b1:multivar:1}{प्रसामान्य समीकरण} कहते हैं ``अवशेष प्रतिरूप-समष्टि पर \omterm{def:b1:euclid:orthogonal}{लांबिक}'', अतः
विश्लेषणी न्यूनतम आँकड़ा-सदिश का \omterm{thm:b1:euclid:projection}{लांबिक प्रक्षेप} \emph{ही} है: कलन और
यूक्लिडीय ज्यामिति एक ही वस्तु संगणित करते हैं। (ग) सहसंबंध $\rho =
c_{xy}/\sqrt{v_xv_y}$ रेखा से व्याख्यायित प्रसरण का हिस्सा $\rho^2$ नापता
है, और कोशी--श्वार्ट्ज़ उसे परिबद्ध करती है: $\abs\rho \leq 1$, जिसमें समता
केवल संरेख आँकड़ों के लिए। (घ) \omterm{def:b1:vspaces:free}{आधार} और स्वतंत्रता (अध्याय 18), ग्राम तथा
आघूर्ण आव्यूह और उनके सारणिक (अध्याय 22--23), और \omterm{thm:b1:euclid:projection}{लांबिक प्रक्षेप} (अध्याय
23) --- सब एक ही कलनविधि के काम करते पुर्जों के रूप में फिर से प्रकट हुए।
भाग II और III \emph{\omterm{pb:b1:multivar:1}{न्यूनतम वर्ग} की विधि} (लजांद्र, गाउस) को उसके
\emph{प्रसामान्य समीकरणों} के रास्ते विकसित करते हैं।
