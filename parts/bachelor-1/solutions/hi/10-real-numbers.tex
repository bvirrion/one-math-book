\section*{अध्याय \ref{ch:b1:reals} --- वास्तविक संख्याएँ}

\begin{solution}{exo:b1:reals:1}
$A$: प्रत्येक अवयव $\leq 1$ है और $1 \in A$: $\sup A = \max A = 1$। निचले
परिबंध: $0$ नीचे से परिबद्ध करता है; $\varepsilon > 0$ के लिए आर्किमिडीज़
ऐसा $n$ देता है कि $\frac 1n < \varepsilon$, अतः कोई धनात्मक संख्या $A$ को
नीचे से परिबद्ध नहीं करती: $\inf A = 0$, जो प्राप्त नहीं होता (कोई न्यूनतम
नहीं)।

$B$: पद $0, -\frac12, \frac23, -\frac34, \frac45, \dots$ सम पद
$\frac{n}{n+1}$ ($n$ सम) $1$ तक पहुँचे बिना बढ़ते हैं; विषम पद
$-\frac{n}{n+1}$ $-1$ तक घटते हैं। अतः $\sup B = 1$ और $\inf B = -1$, जिनमें
से कोई प्राप्त नहीं होता: न अधिकतम, न न्यूनतम। (परिबंध: सभी $b \in B$ के लिए
$\abs{b} < 1$; और बड़े $n$ के लिए $\frac{n}{n+1} = 1 - \frac{1}{n+1} > 1 -
\varepsilon$, और नीचे भी इसी प्रकार।)

$C = \intoo{-\sqrt 3}{\sqrt 3}$: $\sup C = \sqrt 3$, $\inf C = -\sqrt 3$,
जिनमें से कोई प्राप्त नहीं होता।
\end{solution}

\begin{solution}{exo:b1:reals:2}
$u, v \in \intco{0}{1}$ के साथ $x = \lfloor x \rfloor + u$, $y = \lfloor y
\rfloor + v$ लिखिए। तब $u + v \in \intco{0}{2}$ के साथ $x + y = \lfloor x
\rfloor + \lfloor y \rfloor + (u + v)$। यदि $u + v < 1$, तो $\lfloor x +
y\rfloor = \lfloor x\rfloor + \lfloor y \rfloor$; यदि $1 \leq u + v < 2$, तो
$\lfloor x+y \rfloor = \lfloor x \rfloor + \lfloor y \rfloor + 1$। दोनों
स्थितियाँ होती हैं: $(x, y) = (0.2,\, 0.3)$ बाईं समता देता है, $(0.7,\,
0.8)$ दाईं।
\end{solution}

\begin{solution}{exo:b1:reals:3}
मान लीजिए $k = \lfloor x \rfloor$, अतः $k \leq x < k + 1$। $n$ से गुणा करने
पर: $nk \leq nx < nk + n$, और \omterm{thm:b1:reals:floor}{फ़र्श} लेने पर (जो पूर्णांकों की ओर वर्धमान
संक्रिया है): $nk \leq \lfloor nx \rfloor \leq nk + n - 1$। $n$ से भाग देने
पर: $k \leq \frac{\lfloor nx \rfloor}{n} < k + 1$, अतः बाहरी \omterm{thm:b1:reals:floor}{फ़र्श} $k$ है।
\end{solution}

\begin{solution}{exo:b1:reals:4}
$A$ का प्रत्येक अवयव $B$ में है, अतः $\sup B$ $A$ को ऊपर से परिबद्ध करता है:
इसलिए $\sup A \leq \sup B$ ($\sup A$ \emph{लघुतम} \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} है)। सममित रूप
से $\inf B \leq \inf A$। अंत में $\inf A \leq \sup A$, क्योंकि $A$ अरिक्त
है: कोई भी $a \in A$ उनके बीच बैठता है।
\end{solution}

\begin{solution}{exo:b1:reals:5}
मान लीजिए $s = \sup A$, $t = \sup B$। प्रत्येक $a + b \leq s + t$: \omterm{def:b1:reals:bounds}{ऊपरी
परिबंध}। $\varepsilon > 0$ के लिए $a > s - \frac\varepsilon2$ और $b > t -
\frac\varepsilon2$ (\cref{prop:b1:reals:epsilon}) चुनिए: तब $a + b > s + t -
\varepsilon$। $\varepsilon$-अभिलक्षण से $\sup(A+B) = s + t$।

$-A$ के लिए: $m$ $-A$ को ऊपर से परिबद्ध करता है $\iff$ $-m$ $A$ को नीचे से
परिबद्ध करता है; अतः $-A$ का लघुतम \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} $A$ के महत्तम निचले परिबंध के
अनुरूप है: $\sup(-A) = -\inf A$।
\end{solution}

\begin{solution}{exo:b1:reals:6}
प्रत्येक $x$ के लिए: $f(x) + g(x) \leq \sup f + \sup g$; बाएँ पक्ष का \omterm{def:b1:reals:bounds}{उच्चतम}
लेने पर असमिका मिल जाती है। कठोर उदाहरण: $E = \{0, 1\}$, $f =
\mathbf{1}_{\{0\}}$ ($0$ पर मान $1$, अन्यथा $0$), $g = \mathbf{1}_{\{1\}}$:
$\sup(f + g) = 1 < 2 = \sup f + \sup g$।

\cref{exo:b1:reals:5} से कोई विरोध नहीं: वहाँ $a \in A$ और $b \in B$
\emph{स्वतंत्र रूप से} बदलते हैं; यहाँ वही $x$ $f$ और $g$ दोनों को खिलाता है
--- \omterm{def:b1:logic:sets}{समुच्चय} $\{f(x) + g(x) : x \in E\}$ \omterm{def:b1:logic:sets}{समुच्चय} $\{f(x) + g(y) : x, y \in
E\}$ से छोटा है।
\end{solution}

\begin{solution}{exo:b1:reals:7}
$\sqrt 6$ अपरिमेय है: $6 = 2 \times 3$ पूर्ण वर्ग नहीं है, और $v_2(6q^2) = 1
+ 2v_2(q)$ का विषम होना $6q^2 = r^2$ को रोक देता है (जैसा

विरोधाभास। अतः $\sqrt 2 + \sqrt 3 \notin \Q$।
\end{solution}

\begin{solution}{exo:b1:reals:8}
मान लीजिए $x < y$। ऐसा $n \in \N$ चुनिए कि $2^n > \frac{1}{y - x}$
(आर्किमिडीज़: सरल आगमन से $2^n \geq n + 1$, अतः $2$ की कोई घात किसी भी
वास्तविक संख्या को पार कर जाती है)। फिर, $n$ के स्थान पर $2^n$ रखकर
\cref{thm:b1:reals:density} की उपपत्ति की भाँति: $m = \lfloor 2^n x \rfloor
+ 1$ $x < \frac{m}{2^n} < y$ संतुष्ट करता है। अतः $D$ सघन है।
\end{solution}

\begin{solution}{exo:b1:reals:9}
\begin{enumerate}
  \item मान लीजिए $\alpha > 0$। पहले, $\alpha \in G$। मान लीजिए ऐसा नहीं है:
  $\varepsilon = \alpha$ के साथ \omterm{def:b1:reals:bounds}{निम्नतम} के $\varepsilon$-अभिलक्षण से ऐसा $g
  \in G$ है कि $\alpha < g < 2\alpha$ (बाईं ओर कठोर, क्योंकि $\alpha \notin
  G$); फिर $\varepsilon = g - \alpha$ के साथ ऐसा $h \in G$ है कि $\alpha < h
  < g$। अब $g - h \in G$ और $0 < g - h < g - \alpha < \alpha$: अर्थात् $G
  \cap \intoo{0}{+\infty}$ का कोई अवयव अपने \omterm{def:b1:reals:bounds}{निम्नतम} से नीचे, जो असंगत है।
  अतः $\alpha \in G$, और $\alpha\Z \subseteq G$ ($G$ \omterm{def:b1:structures:group}{समूह} है)। विलोमतः, $x
  \in G$ के लिए मान लीजिए $k = \lfloor x/\alpha \rfloor$: तब $x - k\alpha
  \in G$ और $0 \leq x - k\alpha < \alpha$, और $\alpha$ की परिभाषा $x -
  k\alpha = 0$ पर बाध्य कर देती है। अतः $G = \alpha\Z$।
  \item मान लीजिए $\alpha = 0$, और मान लीजिए $x < y$। ऐसा $g \in G$ है कि $0
  < g < y - x$। $k = \lfloor x/g \rfloor + 1$ वाला गुणज $kg$ $x < kg \leq x
  + g < y$ संतुष्ट करता है, और $kg \in G$: अर्थात् सघनता।
\end{enumerate}
$G = \Z + \sqrt 2\,\Z$ $(\R, +)$ का \omterm{def:b1:structures:subgroup}{उपसमूह} है। वह $\alpha\Z$ रूप का नहीं है:
अन्यथा $1 = p\alpha$ और $\sqrt 2 = q\alpha$ ($p, q \in \Z$) से $\sqrt 2 =
\frac qp \in \Q$ मिलता, जो विरोधाभास है। द्विभाजन से $G$ $\R$ में सघन है।
\end{solution}

\begin{solution}{exo:b1:reals:10}
मान लीजिए $s = \sup A > 0$, $t = \sup B > 0$। $a \in A$ के लिए $b \in B$:
$ab \leq st$ (\emph{धनात्मक} संख्याओं के बीच की असमिकाओं को गुणा करके)। $0 <
\varepsilon < \min(s, t)$ के लिए: $a > s - \varepsilon$ और $b > t -
\varepsilon$ चुनिए; तब
\[
ab > (s - \varepsilon)(t - \varepsilon)
= st - \varepsilon(s + t) + \varepsilon^2
> st - \varepsilon (s + t),
\]
और $\varepsilon(s+t)$ को मनचाहा छोटा किया जा सकता है: $\varepsilon$-अभिलक्षण
से (इस रूप में: कोई संख्या $< st$ $AB$ को ऊपर से परिबद्ध नहीं करती) $\sup AB
= st$।

धनात्मकता अनिवार्य है: $A = B = \{-1, 0\}$ $AB = \{0, 1\}$, $\sup AB = 1$
देता है, जबकि $\sup A \cdot \sup B = 0 \times 0 = 0$।
\end{solution}

\begin{solution}{exo:b1:reals:11}
$s = \sup A$, $i = \inf A$ लिखिए। $a, a' \in A$ के लिए: $a \leq s$ और $a'
\geq i$ $a - a' \leq s - i$ देते हैं; सममिति से $\abs{a - a'} \leq s - i$,
अतः $s - i$ अंतरों के \omterm{def:b1:logic:sets}{समुच्चय} को ऊपर से परिबद्ध करता है। $\varepsilon > 0$
के लिए $a > s - \frac\varepsilon2$ और $a' < i + \frac\varepsilon2$ चुनिए
(\cref{prop:b1:reals:epsilon} और \omterm{def:b1:reals:bounds}{निम्नतम} के लिए उसका दर्पण): तब $\abs{a -
a'} \geq a - a' > s - i - \varepsilon$। $\varepsilon$-अभिलक्षण से
$\operatorname{diam} A = s - i$।

प्रत्येक $a \in A$ $i \leq a \leq s$ संतुष्ट करता है, अतः $A \subseteq
\intcc{i}{s}$, जो लंबाई $\operatorname{diam} A$ का संवृत \omterm{prop:b1:reals:intervals}{अंतराल} है। यदि कोई
संवृत \omterm{prop:b1:reals:intervals}{अंतराल} $\intcc{u}{v}$ $A$ को समेटता है, तो $v$ $A$ का \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} है
और $u$ निचला, अतः $u \leq i$ और $v \geq s$: $\intcc{i}{s} \subseteq
\intcc{u}{v}$। अतः $\intcc{i}{s}$ सबसे छोटा है।
\end{solution}

\begin{solution}{exo:b1:reals:12}
$0 \in E$, अतः $E \neq \emptyset$। यदि $x > \max(1, y)$, तो $x^2 > x > y$,
अतः $E$ $\max(1, y)$ से ऊपर परिबद्ध है: $s = \sup E$ विद्यमान है
(\cref{thm:b1:reals:sup}), और $s \geq \min(1, y) > 0$, क्योंकि $\min(1, y)
\in E$: वस्तुतः यदि $y \geq 1$, तो $1^2 = 1 \leq y$, और यदि $y < 1$, तो $y^2
< y$।

\emph{$s^2 < y$ असंभव है।} ऐसा $0 < h < 1$ चुनिए कि $h < \frac{y - s^2}{2s +
1}$। तब
\[
(s + h)^2 = s^2 + 2sh + h^2 \leq s^2 + (2s + 1)h < y ,
\]
अतः $s + h \in E$, जो $s$ के $E$ को ऊपर से परिबद्ध करने का विरोध करता है।

\emph{$s^2 > y$ असंभव है।} ऐसा $0 < h < s$ चुनिए कि $h < \frac{s^2 -
y}{2s}$। तब $(s - h)^2 = s^2 - 2sh + h^2 > s^2 - 2sh > y$; प्रत्येक $x \in
E$ $x^2 \leq y < (s - h)^2$ संतुष्ट करता है, अतः $x < s - h$ (दोनों $\geq 0$
हैं): अर्थात् $s - h$ $E$ का ऐसा \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} है जो $s$ से छोटा है, जो
लघुतमता का विरोध करता है।

अतः $s^2 = y$। अद्वितीयता: यदि $0 < s < s'$, तो $s^2 < s'^2$, अतः दो भिन्न
धनात्मक मूलों का वर्ग $y$ नहीं हो सकता। एकदिष्टता: यदि $0 < y < y'$, तो
$\sqrt y \neq \sqrt{y'}$, और $\sqrt y > \sqrt{y'}$ वर्ग करने पर $y > y'$
देता: अतः $\sqrt y < \sqrt{y'}$।
\end{solution}

\begin{solution}{pb:b1:reals:1}
\textbf{1.} $\frac 18$ के लिए: $10 = 8 \cdot 1 + 2$, $20 = 8 \cdot 2 + 4$,
$40 = 8 \cdot 5 + 0$; अंक $1, 2, 5$, शेषफल $0$, और फिर केवल शून्य: $\frac 18
= 0.125$। $\frac 17$ के लिए: $10 = 7 \cdot 1 + 3$, $30 = 7 \cdot 4 + 2$, $20
= 7 \cdot 2 + 6$, $60 = 7 \cdot 8 + 4$, $40 = 7 \cdot 5 + 5$, $50 = 7 \cdot
7 + 1$: अंक $1, 4, 2, 8, 5, 7$, शेषफल $3, 2, 6, 4, 5, 1$। शेषफल $r = 1$ पर
लौट आया है, अतः वही छह पद सदा के लिए अक्षरशः दोहराते हैं: $\frac 17 =
0.(142857)$, और शेषफल $1, 3, 2, 6, 4, 5$ में चक्कर लगाते हैं।

\textbf{2.} आधार $2$ ($r_0 = 1$) में $\frac 13$: $2 = 3 \cdot 0 + 2$, $4 = 3
\cdot 1 + 1$, और $r = 1$ फिर लौट आता है: $\frac 13 = (0.\overline{01})_2$।
आधार $2$ में $\frac{5}{16}$: $10 = 16 \cdot 0 + 10$, $20 = 16 \cdot 1 + 4$,
$8 = 16 \cdot 0 + 8$, $16 = 16 \cdot 1 + 0$: $\frac{5}{16} = (0.0101)_2$, जो
समाप्त हो जाता है। आधार $3$ में $\frac 12$: $3 = 2 \cdot 1 + 1$, और $r = 1$
तुरंत लौट आता है: $\frac 12 = (0.\overline{1})_3$।

\textbf{3.} $N$ पदों के बाद कलनविधि का शेषफल $r_N = b^N p \bmod q$ है
(औपचारिक उपपत्ति प्रश्न 14 में; यहाँ यह केवल यह देखना है कि हर पद शेषफल को
$b$ से गुणा करता है और $q$ के सापेक्ष घटा देता है)। आगे के सभी अंक $0$ हैं
यदि और केवल यदि किसी $r_N = 0$, अर्थात् $q \mid b^N p$; और चूँकि $\gcd(p, q)
= 1$, गाउस की प्रमेयिका $q \mid b^N$ देती है। यदि $q \mid b^N$, तो $q$ का
प्रत्येक \omterm{def:b1:arith:prime}{अभाज्य} गुणनखंड $b^N$ को \omterm{def:b1:arith:divides}{विभाजित करता है}, अतः (अभाज्यता से) $b$ को
भी। विलोमतः, यदि $q = p_1^{a_1} \cdots p_r^{a_r}$ का प्रत्येक \omterm{def:b1:arith:prime}{अभाज्य} $b$ को
\omterm{def:b1:arith:divides}{विभाजित करता है}, तो $A = \max_i a_i$ के साथ प्रत्येक $p_i^{a_i}$ $b^A$ को
\omterm{def:b1:arith:divides}{विभाजित करता है}, और $p_i^{a_i}$ जोड़ों में \omterm{cor:b1:arith:bezout}{सहअभाज्य} हैं, अतः $q \mid b^A$।
$q = 20 = 2^2 \cdot 5$ के लिए: दोनों \omterm{def:b1:arith:prime}{अभाज्य} $10$ को विभाजित करते हैं
($\frac{1}{20} = 0.05$), पर $2 \nmid 3$, अतः आधार $3$ में $\frac{1}{20}$ सदा
दोहराता रहता है।

\textbf{4.} $1^2 = 1 < 2 < 4 = 2^2$ $s_0 = 1$ देता है। फिर $1.4^2 = 1.96 < 2
< 2.25 = 1.5^2$: $\lfloor 10\sqrt 2 \rfloor = 14$, $s_1 = 1.4$। इसके बाद
$1.41^2 = 1.9881 < 2 < 2.0164 = 1.42^2$: $s_2 = 1.41$; $1.414^2 = 1.999396 <
2 < 2.002225 = 1.415^2$: $s_3 = 1.414$; $1.4142^2 = 1.99996164 < 2 <
2.00024449 = 1.4143^2$: $s_4 = 1.4142$। हर स्थिति में दिखाई गई असमिकाएँ ठीक
$s_n \leq \sqrt 2 < s_n + 10^{-n}$ कहती हैं, जो $10^n \sqrt 2$ के \omterm{thm:b1:reals:floor}{फ़र्श} की
परिभाषा है।

\textbf{5.} $A_0 = \lfloor x \rfloor = 0$, क्योंकि $0 \leq x < 1$। $A_{n-1}
\leq b^{n-1} x < A_{n-1} + 1$ से, $b$ से गुणा कीजिए:
\[
b\,A_{n-1} \leq b^n x < b\,A_{n-1} + b .
\]
पूर्णांक $b\,A_{n-1}$ $\leq b^n x$ है, अतः $b\,A_{n-1} \leq A_n$; और
$b\,A_{n-1} + b$ के पूर्णांक होने के साथ $b^n x < b\,A_{n-1} + b$ $A_n \leq
b\,A_{n-1} + b - 1$ पर बाध्य कर देता है। अतः $0 \leq d_n = A_n - b\,A_{n-1}
\leq b - 1$: अर्थात् एक अंक।

\textbf{6.} दूरबीनी: $d_k b^{-k} = A_k b^{-k} - A_{k-1} b^{-(k-1)}$, अतः
\[
\sum_{k=1}^n d_k b^{-k} = A_n b^{-n} - A_0 = s_n .
\]
$A_n \leq b^n x < A_n + 1$ को $b^n$ से भाग देने पर $s_n \leq x < s_n +
b^{-n}$ मिलता है।

\textbf{7.} आगमन: $b^0 = 1 \geq 1$, और $b^{n+1} = b \cdot b^n \geq 2(n + 1)
\geq n + 2$। एकदिष्टता: $s_n - s_{n-1} = d_n b^{-n} \geq 0$। प्रत्येक $s_n
\leq x$ (प्रश्न 6): अतः $x$ $\{s_n\}$ का \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} है। $\varepsilon > 0$
के लिए \omterm{thm:b1:reals:archimedes}{आर्किमिडीज़ गुणधर्म} ऐसा $n$ देता है कि $n + 1 > \frac1\varepsilon$,
अतः $b^{-n} < \varepsilon$, और तब प्रश्न 6 से $s_n > x - b^{-n} > x -
\varepsilon$। \cref{prop:b1:reals:epsilon} से $x = \sup_n s_n$।

\textbf{8.} मान लीजिए सभी $k > N$ के लिए $d_k = b - 1$। $n > N$ के लिए
परिमित गुणोत्तर योग देता है
\[
s_n = s_N + (b - 1)\sum_{k=N+1}^{n} b^{-k}
= s_N + b^{-N} - b^{-n} .
\]
अतः प्रत्येक $n$ के लिए $x \geq s_n = s_N + b^{-N} - b^{-n}$; अंतिम पद को
किसी भी $\varepsilon$ से नीचे सिकोड़ने पर (प्रश्न 7) $x \geq s_N + b^{-N}$।
पर कोटि $N$ पर प्रश्न 6 कहता है $x < s_N + b^{-N}$: विरोधाभास। अतः माला
$(d_n)$ उचित है।

\textbf{9.} परिबद्धता: $t_n \leq (b-1)\sum_{k=1}^n b^{-k} = 1 - b^{-n} < 1$,
और $(t_n)$ अह्रासमान है, अतः $y = \sup t_n$ विद्यमान है और $0 \leq y \leq
1$। $n$ नियत कीजिए। दुतरफ़ा आकलन के लिए: $t_n \leq y$ स्पष्ट है। उचितता से
ऐसा $m > n$ चुनिए कि $e_m \leq b - 2$। $p \geq m$ के लिए:
\[
t_p - t_n = \sum_{k=n+1}^{p} e_k b^{-k}
\leq (b^{-n} - b^{-p}) - b^{-m} < b^{-n} - b^{-m},
\]
जहाँ बीच का योग सर्व-$(b-1)$ अधिकतम के सामने कम से कम $b^{-m}$ खो देता है;
और $p \leq m$ के लिए भी $t_p \leq t_m \leq t_n + b^{-n} - b^{-m}$ (एकदिष्टता
और स्थिति $p = m$ से)। अतः प्रत्येक $t_p \leq t_n + b^{-n} - b^{-m}$, इसलिए
$y \leq t_n + b^{-n} - b^{-m} < t_n + b^{-n}$। ($n = 0$ के साथ: $y < 1$, अतः
$y \in \intco{0}{1}$।) अब $b^n t_n = \sum_{k \leq n} e_k b^{n-k}$ पूर्णांक
है, और $b^n t_n \leq b^n y < b^n t_n + 1$: अतः $\lfloor b^n y \rfloor = b^n
t_n$। अंत में $y$ के अंक: $d_n(y) = b^n t_n - b \cdot b^{n-1} t_{n-1} =
b^n(t_n - t_{n-1}) = e_n$।

\textbf{10.} प्रश्न 9 कहता है: (माला का मान) के अंक (वही माला) हैं; प्रश्न
5--8 कहते हैं: ($x$ के अंक) ऐसी उचित माला बनाते हैं जिसकी कटाइयों का \omterm{def:b1:reals:bounds}{उच्चतम}
$x$ है (प्रश्न 7)। अतः दोनों \omterm{def:b1:logic:map}{प्रतिचित्रण} दोनों क्रमों में तत्समक बनते हैं:
वे $\intco{0}{1}$ और उचित मालाओं के बीच परस्पर प्रतिलोम एकैकी आच्छादन हैं।
यदि दो उचित मालाओं का मान बराबर होता, तो अंक-प्रतिचित्रण लगाने पर वे बराबर
हो जातीं: अर्थात् अद्वितीयता। यही \emph{$b$-आदिक प्रसार प्रमेय} है।

\textbf{11.} मान लीजिए $n > M$ के लिए $e_n = b - 1$, जहाँ $M \geq 0$ न्यूनतम
है। प्रश्न 8 की भाँति $n \geq M$ के लिए $t_n = t_M + b^{-M} - b^{-n}$, अतः
मान $\sup t_n = t_M + b^{-M}$ है। यदि $M = 0$, तो मान $0 + 1 = 1$ है: आधार
$10$ में ठीक-ठीक $0.999\dots = 1$ --- सन्निकट नहीं। यदि $M \geq 1$, तो
न्यूनतमता $e_M \leq b - 2$ देती है और मान है
\[
t_M + b^{-M} = \frac{b^M t_M + 1}{b^M} \in \intoo{0}{1},
\]
अर्थात् एक $b$-आदिक भिन्न, जिसका \emph{उचित} प्रसार $e_1 \dots e_{M-1}\,(e_M
+ 1)\,000\dots$ है (समाप्त होने वाली माला उचित होती है, और उसका मान वही
संख्या है)। विलोमतः, जिस वास्तविक संख्या के दो निरूपण हों उनमें से एक अनुचित
होना चाहिए (उचितता निरूपण को कील देती है, प्रश्न 10), अतः वह इसी रूप की
होगी। और अंतिम अशून्य अंक $d_N$ के साथ लिखी प्रत्येक $m/b^N \in
\intoo{0}{1}$ का अनुचित जुड़वाँ $d_1 \dots d_{N-1}(d_N - 1)(b-1)(b-1)\dots$
होता ही है: अर्थात् ठीक $b$-आदिक भिन्नों के दो नाम हैं, शेष सभी वास्तविक
संख्याओं का एक।

\textbf{12.} मान लीजिए मालाएँ $m - 1$ तक मेल खाती हैं, उभयनिष्ठ कटाई $P =
s_{m-1}$ है, और $d_m < e_m$। प्रश्न 9 से (कोटि $m$ पर कठोर ऊपरी आकलन) $x < P
+ d_m b^{-m} + b^{-m} = P + (d_m + 1)b^{-m} \leq P + e_m b^{-m} \leq y$,
जहाँ अंतिम पद इसलिए कि $P + e_m b^{-m}$ $y$ की कटाई $t_m \leq y$ है। अतः
$d_m < e_m \implies x < y$; भूमिकाएँ बदलने पर $e_m < d_m \implies y < x$; और
चूँकि मालाएँ $m$ पर भिन्न हैं, इनमें से एक सत्य है। दोनों दिशाएँ निकल आती
हैं।

\textbf{13.} मान लीजिए $z = bx - A_1 \in \intco{0}{1}$ (वस्तुतः $A_1 \leq bx
< A_1 + 1$)। $n \geq 0$ के लिए: $b^n A_1 \in \Z$ के साथ $b^n z = b^{n+1} x -
b^n A_1$, अतः $\lfloor u - K \rfloor = \lfloor u \rfloor - K$ से ($K$
पूर्णांक),
\[
A_n(z) = A_{n+1}(x) - b^n A_1(x) .
\]
अतः $d_n(z) = A_n(z) - b\,A_{n-1}(z) = A_{n+1} - b^n A_1 - b\,A_n + b^n A_1
= d_{n+1}(x)$। अर्थात् $bx$ का भिन्नात्मक भाग विस्थापित अंक धारण करता है;
$m$ बार दोहराने पर $b^m x$ के भिन्नात्मक भाग के अंक $(d_{n+m})_{n \geq 1}$
होते हैं।

\textbf{14.} यूक्लिडीय भाग: $0 \leq r_n < q$ के साथ $b^n p = q\,Q_n + r_n$।
$q$ से भाग दीजिए: $0 \leq \frac{r_n}{q} < 1$ के साथ $b^n x = Q_n +
\frac{r_n}{q}$, अतः $Q_n = \lfloor b^n x \rfloor = A_n$, जिससे $A_n =
\frac{b^n p - r_n}{q}$ मिलता है। पुनरावृत्ति के लिए: $b^n p = b(q\,A_{n-1} +
r_{n-1}) = q\,(b\,A_{n-1}) + b\,r_{n-1}$, अतः $b^n p$ और $b\,r_{n-1}$ $q$ के
गुणज से भिन्न हैं: $r_n = (b\,r_{n-1}) \bmod q$।

\textbf{15.} $b\,r_{n-1}$ को $q$ से भाग दीजिए: $c = \lfloor b\,r_{n-1}/q
\rfloor$ के साथ $b\,r_{n-1} = q\,c + r_n$। प्रश्न 14 के प्रदर्शन में
प्रतिस्थापित करने पर $b^n p = q(b\,A_{n-1} + c) + r_n$, और यूक्लिडीय भाग की
अद्वितीयता $A_n = b\,A_{n-1} + c$ की पहचान करा देती है, अर्थात् $d_n = c =
\lfloor b\,r_{n-1}/q \rfloor$। अंक $n$ केवल $r_{n-1}$ पर निर्भर करता है ---
अर्थात् भाग I का लंबा-भाग चक्र, अब प्रमाणित।

\textbf{16.} $q + 1$ शेषफल $r_0, \dots, r_q$ $q$-अवयवी \omterm{def:b1:logic:sets}{समुच्चय}
$\intint{0}{q-1}$ में मान लेते हैं: अतः कबूतरखाना सिद्धांत
(\cref{cor:b1:counting:pigeonhole}) से दो संपाती हो जाते हैं, मान लीजिए $0
\leq N < N + T \leq q$ के साथ $r_N = r_{N+T}$। चूँकि $r_n$ $r_{n+1}$ को
निर्धारित करता है (प्रश्न 14), आगमन से सभी $n \geq N$ के लिए $r_{n+T} =
r_n$; और चूँकि $r_{n-1}$ $d_n$ को निर्धारित करता है (प्रश्न 15), सभी $n \geq
N + 1$ के लिए $d_{n+T} = d_n$। अतः प्रत्येक परिमेय संख्या का प्रसार अंततः
आवर्ती है, जिसका पूर्वावर्त $\leq q$ और आवर्त $\leq q$ है।

\textbf{17.} $b^T y$ के भिन्नात्मक भाग के अंक $(d_{n+T}) = (d_n)$ हैं
(विस्थापन प्रमेयिका, फिर शुद्ध आवर्तिता): अर्थात् वही उचित माला जो $y$ की
है। प्रश्न 10 से मान बराबर हैं: $b^T y - \lfloor b^T y \rfloor = y$, अतः
$(b^T - 1)\,y = \lfloor b^T y \rfloor = A_T \in \N$ और
\[
y = \frac{A_T}{b^T - 1} ,
\]
जो परिमेय है और जिसका हर $b^T - 1$ को \omterm{def:b1:arith:divides}{विभाजित करता है}; अंश $A_T$ वह पूर्णांक
है जिसके आधार-$b$ अंक $d_1 \dots d_T$ हैं। जाँच: $0.(142857) =
\frac{142857}{999999}$, और $142857 \times 7 = 999999$, अतः यह $\frac 17$ है।

\textbf{18.} यदि $n > N$ के लिए $d_{n+T} = d_n$, तो $b^N x$ के भिन्नात्मक
भाग $z$ के अंक $(d_{N+n})_{n\geq1}$ हैं (विस्थापन प्रमेयिका), जो शुद्ध रूप
से आवर्ती हैं; प्रश्न 17 से $z \in \Q$। तब $b^N x = A_N + z$ $x = (A_N +
z)/b^N \in \Q$ देता है। प्रश्न 16 के साथ: $x$ परिमेय है $\iff$ प्रसार अंततः
आवर्ती है। दायाँ पक्ष आधार का उल्लेख करता है, बायाँ नहीं: अतः किसी एक आधार
में आवर्तिता परिमेयता के तुल्य है, और इसलिए हर आधार में आवर्तिता के भी।

\textbf{19.} $x = \frac 1q$ के लिए $r_n = b^n \bmod q$। यदि $\gcd(b, q) =
1$, तो $r_T = r_0 = 1$ यदि और केवल यदि $b^T \equiv 1 \pmod q$; ऐसा $T$
विद्यमान है (कबूतरखाना $b^i \equiv b^j$, $i < j$ देता है, और $b$ $q$ के
सापेक्ष व्युत्क्रमणीय है, अतः $b^{j-i} \equiv 1$), और उनमें लघुतम ---
अर्थात् गुणात्मक कोटि --- शेषफलों को, और इसलिए अंकों को, आवर्त $T$ के साथ
शुद्ध रूप से आवर्ती बना देती है। इससे छोटा कोई आवर्त संभव नहीं: आवर्त $T'$
$(b^{T'} - 1)\frac1q \in \N$ देता (प्रश्न 17), अर्थात् $q \mid b^{T'} - 1$।
$q = 7$ के लिए $b = 10$: $10 \equiv 3$, $10^2 \equiv 2$, $10^3 \equiv 6$,
$10^4 \equiv 4$, $10^5 \equiv 5$, $10^6 \equiv 1 \pmod 7$: कोटि $6$, और
सचमुच $\frac 17$ का आवर्त छह है।

\textbf{20.} माला में अनंत $0$ हैं ($x^*$ के अंक अधिकतर शून्य हैं), अतः वह
उचित है, और $x^*$ सुपरिभाषित है (प्रश्न 9)। मान लीजिए अंक $N$ के आगे आवर्त
$T$ के साथ अंततः आवर्ती हैं। अनंत अंक $1$ के बराबर हैं (प्रत्येक त्रिभुजीय
संख्या पर एक), अतः कोई $1$ किसी स्थान $j > N$ पर बैठता है; तब आवर्तिता
प्रत्येक स्थान $j + kT$ पर $1$ रख देती है: अर्थात् $j$ से आगे क्रमागत $1$ के
बीच के \omterm{prop:b1:reals:intervals}{अंतराल} अधिक से अधिक $T$ हैं। पर $1$ ठीक त्रिभुजीय संख्याओं पर बैठते
हैं, जिनके क्रमागत \omterm{prop:b1:reals:intervals}{अंतराल} $\frac{(j+1)(j+2)}{2} - \frac{j(j+1)}{2} = j + 1$
अंततः $T$ को पार कर जाते हैं: विरोधाभास। अतः वह अंततः आवर्ती नहीं है, और
प्रश्न 18 से $x^* \notin \Q$ --- अर्थात् अपरिमेयता केवल अंकों की लय से पढ़
ली गई।

\textbf{21.} $x < y$ दिया हो, तो प्रश्न 7 ऐसा $n$ देता है कि $b^{-n} < y -
x$; $m = \lfloor b^n x \rfloor + 1$ रखिए। तब $b^n x < m \leq b^n x + 1 < b^n
y$, अतः $x < \frac{m}{b^n} < y$: अर्थात् सघनता, और वह भी एक साथ हर आधार के
लिए ($b = 2$ \cref{exo:b1:reals:8} लौटा देता है)। आधार $b = q$ में $\frac pq
\in \intoo{0}{1}$ के लिए: पहला अंक $\lfloor q \cdot \frac pq \rfloor = p$ है
और $q \cdot \frac pq = p$ का भिन्नात्मक भाग $0$ है: आगे के सभी अंक लुप्त हो
जाते हैं, अर्थात् समाप्त होने वाला प्रसार $\frac pq = (0.p)_q$। समाप्त होना
आधार पर निर्भर करता है; आवर्तिता --- अर्थात् परिमेयता --- नहीं (प्रश्न 18)।

\textbf{22.} प्रत्येक $e_k \in \{1, 2\}$ आधार $10$ का अंक है, और माला कभी
सर्व-$9$ पर समाप्त नहीं होती: अतः उचित। उसका मान $y$ $\intco{0}{1}$ में है
और उसके अंक ठीक $(e_k)$ हैं (प्रश्न 9)। $k$ नियत कीजिए: $y$ का $k$-वाँ अंक
$e_k$ है, जो $x_k$ के $k$-वें अंक से $\neq$ चुना गया है, अतः $y$ और $x_k$ की
उचित मालाएँ भिन्न हैं, इसलिए $y \neq x_k$ (प्रश्न 10: कूटन \omterm{def:b1:logic:inj}{एकैकी} है)। अतः
$y$ किसी भी सूची में नहीं है: कोई भी \omterm{def:b1:logic:map}{प्रतिचित्रण} $\N^* \to \intco{0}{1}$
\omterm{def:b1:logic:inj}{आच्छादक} नहीं है। परिमेय संख्याओं के विपरीत वास्तविक संख्याओं की गणना नहीं की
जा सकती --- यही अगणनीयता है, जिसका सिद्धांत \cref{ch:b1:topology} विकसित
करता है।

\textbf{23.} यदि प्रसार $n$ तक मेल खाते हैं, तो $x$ और $y$ की कटाई एक ही
$s_n$ है, और प्रश्न 6 दोनों को $\intco{s_n}{s_n + b^{-n}}$ में रख देता है,
जो लंबाई $b^{-n}$ का \omterm{prop:b1:reals:intervals}{अंतराल} है: $\abs{x - y} < b^{-n}$। विलोम: $x = 0.1$ और
$y = 0.0999$ (जो समाप्त होने वाला, अतः उचित है) $\abs{x - y} = 10^{-4} <
10^{-3}$ संतुष्ट करते हैं, फिर भी उनके प्रसार पहले ही अंक पर भिन्न हैं। दोषी
प्रश्न 11 की $b$-आदिक भिन्नें हैं: उनके पास ज़रा-सी हलचल हर दिखाई देने वाला
अंक पलट देती है ($0.0999 \to 0.1000$), क्योंकि ठीक वहीं अनुचित जुड़वाँ घात
लगाए बैठा है।

\textbf{24.} $p = 1$, $q = 10$, $b = 2$, $r_0 = 1$: $2 = 10 \cdot 0 + 2$, $4
= 10 \cdot 0 + 4$, $8 = 10 \cdot 0 + 8$, $16 = 10 \cdot 1 + 6$, $12 = 10
\cdot 1 + 2$ --- और $r_5 = 2 = r_1$: शेषफल सूचकांक $1$ से $(2, 4, 8, 6)$ में
चक्कर लगाते हैं। अंक: $d_1 = 0$, फिर दोहराता खंड $d_2 d_3 d_4 d_5 = 0, 0, 1,
1$:
\[
\tfrac{1}{10} = (0.0\overline{0011})_2 ,
\]
पूर्वावर्त $1$, आवर्त $4$। प्रश्न 3 से, समाप्त होने वाले आधार-$2$ प्रसार के
लिए $10$ के प्रत्येक \omterm{def:b1:arith:prime}{अभाज्य} को $2$ विभाजित करना पड़ता; \omterm{def:b1:arith:prime}{अभाज्य} $5$ यह नहीं
करता। अतः $0.1$ किसी भी परिमित द्विआधारी माला से निरूपित \emph{नहीं} हो सकता
--- परिमित बिट रखने वाला संगणक केवल एक कटाई रखता है, और संचित कटाई-त्रुटियाँ
ही वह कारण हैं जिससे प्लवन-बिंदु $0.1 + 0.2$ अंतिम बिटों में $0.3$ से भिन्न
होता है।

\textbf{25.} (क) पूर्णता ने मान उत्पन्न किए: $x = \sup s_n$ और $y = \sup
t_n$ (प्रश्न 7 और 9) --- अकेले $\Q$ पर $\sqrt 2$ की उचित माला किसी का नाम न
लेती। (ख) \omterm{thm:b1:reals:archimedes}{आर्किमिडीज़ गुणधर्म} ने $b^{-n}$ को अंततः किसी भी $\varepsilon$ से
छोटा बना दिया, जिससे कटाइयाँ अपने \omterm{def:b1:reals:bounds}{उच्चतम} के पास सिमटने को बाध्य हुईं (प्रश्न
7, 21)। (ग) \omterm{thm:b1:reals:floor}{फ़र्श} के अद्वितीयता वाले उपवाक्य ने प्रश्न 14 में $Q_n = A_n$ की
पहचान की और हर अंक-निष्कर्षण $\lfloor u - K \rfloor = \lfloor u \rfloor - K$
को वैध ठहराया (प्रश्न 13)। (घ) परिमित कितने शेषफलों पर लगाया गया कबूतरखाना
सिद्धांत आवर्तिता का एकमात्र इंजन है (प्रश्न 16)। उपदेश: उचित मालाएँ
$\intco{0}{1}$ को पूरी निष्ठा से कूटित करती हैं और परिमेयता को दृश्य लय में
बदल देती हैं; पर अंक-मालाओं के योग के लिए अनंत दूर दाईं ओर से हासिल आगे
बढ़ते हैं, अतः कोई परिमित-चरण नियम योग का पहला अंक भी नहीं निकाल सकता ---
जबकि \cref{thm:b1:reals:sup} का \omterm{def:b1:reals:bounds}{उच्चतम} वाला इंटरफ़ेस एक ही अभिगृहीत से पूरे
विश्लेषण को सँभाल लेता है। अंक $\R$ का एक भव्य \emph{चित्र} हैं; \omterm{def:b1:reals:bounds}{उच्चतम} उसका
\emph{इंजन} है।
\end{solution}
