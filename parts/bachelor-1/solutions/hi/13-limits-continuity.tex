\begin{solution}{exo:b1:continuity:2}
$f = \lfloor\cdot\rfloor$ $\R \setminus \Z$ पर \omterm{def:b1:continuity:continuous}{संतत} है (स्थानीय रूप से अचर)
और प्रत्येक $n \in \Z$ पर असंतत: बाईं सीमा $n - 1$, मान $n$।

$g(x) = x - \lfloor x\rfloor$: वही असांतत्य-बिंदु (तत्समक फलन \omterm{def:b1:continuity:continuous}{संतत} है, अतः
$g$ $f$ की कूदें विरासत में पाता है); $n \in \Z$ पर बाईं सीमा $1 \neq 0 =
g(n)$।

$h(x) = \lfloor x\rfloor + (x - \lfloor x\rfloor)^2$: $\intco{n}{n + 1}$ पर
$h(x) = n + (x - n)^2$, अतः वहाँ \omterm{def:b1:continuity:continuous}{संतत}; $x = n$ पर बाईं सीमा $(n-1) + 1 = n =
h(n)$ है: कूदें कट जाती हैं। $h$ $\R$ पर \omterm{def:b1:continuity:continuous}{संतत} है (और कठोरतः वर्धमान)।
\end{solution}

\begin{solution}{exo:b1:continuity:4}

\begin{solution}{exo:b1:continuity:5}
मान लीजिए $g(x) = f(x) - x$, जो $\intcc{0}{1}$ पर \omterm{def:b1:continuity:continuous}{संतत} है। चूँकि $f$
$\intcc{0}{1}$ में भेजता है: $g(0) = f(0) \geq 0$ और $g(1) = f(1) - 1 \leq
0$। \cref{thm:b1:continuity:ivt} से किसी $c$ के लिए $g(c) = 0$: अर्थात् एक
अचल बिंदु।

परिकल्पनाओं की आवश्यकता: $\intcc{0}{1}$ पर असंतत \omterm{def:b1:logic:map}{प्रतिचित्रण}, जो $x \leq
\frac12$ के लिए $f(x) = 1$ और $x > \frac12$ के लिए $f(x) = 0$ है, का कोई अचल
बिंदु नहीं है; \omterm{prop:b1:reals:intervals}{अंतराल} $\intoo{0}{1}$ (जो खंड नहीं है) पर $f(x) = \frac x2$
$\intoo{0}{1}$ में \omterm{def:b1:continuity:continuous}{संतत} है और उसका कोई अचल बिंदु नहीं (उम्मीदवार $0$
अनुपस्थित है); और $\R$ पर $f(x) = x + 1$।
\end{solution}

\begin{solution}{exo:b1:continuity:6}
$g(x) = \cos x - x$ \omterm{def:b1:continuity:continuous}{संतत} है, $g(0) = 1 > 0$, $g(1) = \cos 1 - 1 < 0$: अतः
$\intoo{0}{1}$ में कोई हल विद्यमान है (\cref{thm:b1:continuity:ivt})।
अद्वितीयता: $g$ $\R$ पर कठोरतः ह्रासमान है --- $x \leq 0$ के लिए $g(x) \geq
1 - x > 0$ का वैसे भी कोई शून्य नहीं है; और $g'(x) = -\sin x - 1 \leq 0$,
जहाँ समता केवल वियुक्त बिंदुओं पर ($x \equiv -\frac\pi2 \bmod 2\pi$), अतः
$g$ कठोरतः ह्रासमान है (\cref{ch:b1:derivative}; वैकल्पिक रूप से:
$\intcc{0}{1}$ पर $\cos$ कठोरतः ह्रासमान है और $-x$ भी, अतः $g$ भी)। कठोरतः
एकदिष्ट फलन अधिक से अधिक एक बार लुप्त होता है।
\end{solution}

\begin{solution}{exo:b1:continuity:7}
$M = f(0) + 1$ नियत कीजिए। ऐसा $A > 0$ है कि $\abs x \geq A$ के लिए $f(x)
\geq M$ (दोनों अनंत सीमाओं की परिभाषा; बड़ी देहली लीजिए)। खंड
$\intcc{-A}{A}$ पर चरम मान प्रमेय (\cref{thm:b1:continuity:evt}) ऐसा $c$
देती है कि $f(c) = \inf_{\intcc{-A}{A}} f \leq f(0)$। $\abs x \geq A$ के
लिए: $f(x) \geq M > f(0) \geq f(c)$। अतः $f(c)$ समग्र न्यूनतम है।
\end{solution}

\begin{solution}{exo:b1:continuity:8}
खंड $\intcc{0}{T}$ पर $f$ परिबद्ध है और अपने परिबंध प्राप्त करता है
(\cref{thm:b1:continuity:evt}); और आवर्तिता से वही परिबंध पूरे $\R$ पर हैं,
और वहीं प्राप्त भी होते हैं।

मान लीजिए $g(x) = f(x + \frac T2) - f(x)$, जो \omterm{def:b1:continuity:continuous}{संतत} है। तब
\[
g(0) + g\bigl(\tfrac T2\bigr)
= \bigl(f(\tfrac T2) - f(0)\bigr) + \bigl(f(T) - f(\tfrac T2)\bigr)
= f(T) - f(0) = 0 :
\]
$g(0)$ और $g(\frac T2)$ के चिह्न विपरीत हैं (या कोई एक लुप्त है), अतः
$\intcc{0}{T/2}$ पर मध्यवर्ती मान प्रमेय ऐसा $c$ देती है कि $g(c) = 0$,
अर्थात् $f(c + \frac T2) = f(c)$।
\end{solution}

\begin{solution}{exo:b1:continuity:9}
पहले असमिका: $0 \leq y \leq x$ के लिए,
\[
\bigl(\sqrt y + \sqrt{x - y}\bigr)^2 = x + 2\sqrt{y(x-y)} \geq x,
\]
अतः $\sqrt x \leq \sqrt y + \sqrt{x - y}$, अर्थात् $\sqrt x - \sqrt y \leq
\sqrt{x - y}$। इसलिए सभी $x, y \geq 0$ के लिए $\abs{\sqrt x - \sqrt y} \leq
\sqrt{\abs{x - y}}$।

\omterm{def:b1:continuity:uniform}{एकसमान सांतत्य}: $\varepsilon > 0$ दिया हो, तो $\delta = \varepsilon^2$
लीजिए; तब $\abs{x - y} \leq \delta$ से $\abs{\sqrt x - \sqrt y} \leq
\sqrt\delta = \varepsilon$ आ जाता है।

$0$ के पास लिप्शिट्ज़ नहीं: $x \to 0^+$ होने पर $\frac{\sqrt x - \sqrt 0}{x
- 0} = \frac{1}{\sqrt x} \to +\infty$, अतः कोई अचर $k$ सभी अंतर-भागफलों पर
हावी नहीं हो सकता।
\end{solution}

\begin{solution}{exo:b1:continuity:10}
मान लीजिए $f$ \omterm{def:b1:logic:inj}{एकैकी} और \omterm{def:b1:continuity:continuous}{संतत} है पर कठोरतः एकदिष्ट नहीं। तब $I$ में ऐसे $a < b
< c$ हैं कि $f(b)$ $f(a)$ और $f(c)$ के बीच नहीं है --- वस्तुतः, यदि
\emph{सभी} त्रिकियों के लिए बीच का मान बाहरी दोनों के बीच होता, तो $f$
एकदिष्ट होता (किन्हीं दो युग्मों की तुलना कीजिए; एक छोटी स्थिति-जाँच)। मान
लीजिए $f(b) > \max(f(a), f(c))$ (दूसरी स्थिति सममित है, $f$ के स्थान पर $-f$
रखिए)। ऐसा $v$ चुनिए कि $\max(f(a), f(c)) < v < f(b)$। $\intcc{a}{b}$ पर और
$\intcc{b}{c}$ पर मध्यवर्ती मान प्रमेय लगाने से ऐसे $u_1 \in \intoo{a}{b}$
और $u_2 \in \intoo{b}{c}$ हैं कि $f(u_1) = v = f(u_2)$: अर्थात् बराबर
प्रतिबिंब वाले दो भिन्न बिंदु, जो एकैकीयता का विरोध करता है।
\end{solution}

\begin{solution}{exo:b1:continuity:11}
$f(0) = f(0+0) = 2f(0)$ $f(0) = 0$ देता है; $0 = f(x - x)$ से $f(-x) =
-f(x)$। $\alpha = f(1)$ रखिए। आगमन: $n \in \N$ के लिए $f(n) = n\alpha$, फिर
विषमता से $n \in \Z$ के लिए भी। $q \in \N^*$ के लिए: $q\,f(\frac pq) = f(p)
= p\alpha$ ($\frac pq$ को $q$ बार अपने में जोड़िए), अतः $f(\frac pq) =
\alpha\frac pq$: $\Q$ पर $f = \alpha\, \mathrm{id}$।

अब मान लीजिए $x \in \R$ और $(r_n)$ $r_n \to x$ वाला परिमेय संख्याओं का
अनुक्रम है (\omterm{def:b1:topology:dense}{सघनता}, \cref{thm:b1:reals:density}, नेस्टेड अंतरालों में लगाकर;
अथवा $r_n = \frac{\lfloor nx\rfloor}{n}$)। \omterm{def:b1:continuity:continuous}{सांतत्य}: $f(x) = \lim f(r_n) =
\lim \alpha r_n = \alpha x$।
\end{solution}

\begin{solution}{exo:b1:continuity:12}
मान लीजिए $\varepsilon > 0$। $+\infty$ पर सीमा से ऐसा $A$ है कि $x \geq A$
के लिए $\abs{f(x) - \ell} \leq \frac\varepsilon2$; अतः $x, y \geq A$ के लिए
$\abs{f(x) - f(y)} \leq \varepsilon$ (किसी निकटता की आवश्यकता नहीं)।

खंड $\intcc{0}{A + 1}$ पर हाइने की प्रमेय (\cref{thm:b1:continuity:heine})
इस $\varepsilon$ के लिए $\delta_0 > 0$ देती है; $\delta = \min(\delta_0, 1)$
रखिए।

अब $\abs{x - y} \leq \delta$ वाला कोई भी $x, y \geq 0$ लीजिए, मान लीजिए $x
\leq y$। यदि $y \leq A + 1$: दोनों उस खंड में हैं, और हाइने वाला $\delta_0$
लागू हो जाता है। अन्यथा $y > A + 1$, और तब $x \geq y - 1 > A$: दोनों
$\intco{A}{+\infty}$ में हैं, जहाँ सीमा वाला तर्क लागू होता है। दोनों
स्थितियों में $\abs{f(x) - f(y)} \leq \varepsilon$: अर्थात् \omterm{def:b1:continuity:uniform}{एकसमान सांतत्य}।
\end{solution}

\begin{solution}{pb:b1:continuity:1}
\textbf{1.} $n \in \N$ के लिए: आगमन से $f(nx) = n f(x)$ ($f((n+1)x) = f(nx)
+ f(x)$)। साथ ही $f(0) = 2f(0)$ $f(0) = 0$ देता है, और $0 = f(x - x) = f(x)
+ f(-x)$ विषमता देता है, अतः $n \in \Z$ के लिए $f(nx) = nf(x)$। $r = \frac
pq$ के लिए: $q\,f\bigl(\tfrac pq x\bigr) = f(px) = p\,f(x)$, अतः $f(rx) = r
f(x)$: अर्थात् $f$ $\Q$-रैखिक है।

\textbf{2.} योज्यता सभी $x$ और $h$ के लिए देती है:
\[
f(x + h) - f(x) = f(h) = f(x_0 + h) - f(x_0) .
\]
जब $h \to 0$, तब $x_0$ पर \omterm{def:b1:continuity:continuous}{सांतत्य} से दायाँ पक्ष $0$ की ओर जाता है; अतः $f(x
+ h) \to f(x)$: अर्थात् प्रत्येक $x$ पर \omterm{def:b1:continuity:continuous}{सांतत्य}। फिर
\cref{exo:b1:continuity:11} $f(x) = f(1)\,x$ दे देता है।

\textbf{3.} दो \omterm{def:b1:continuity:continuous}{संतत} योज्य फलन $cx$ और $c'x$ रूप के हैं (प्रश्न 2); यदि वे
\omterm{def:b1:topology:dense}{सघन} \omterm{def:b1:structures:subgroup}{उपसमूह} $\Z + \sqrt2\,\Z$ (\cref{exo:b1:reals:9}) पर मेल खाते हैं, तो
उसके किसी $g \neq 0$ के लिए $c\,g = c'g$: $c = c'$, अर्थात् दोनों फलन बराबर
हैं। \omterm{def:b1:continuity:continuous}{सांतत्य} के बिना: $\Q$-रैखिकता केवल $\Q$-संयोजनों पर मान बाँधती है, और
$1, \sqrt2$ $\Q$-स्वतंत्र हैं ($\sqrt2 \notin \Q$), अतः $f(1) = 0$,
$f(\sqrt2) = 1$ संगत है और \omterm{def:b1:structures:subgroup}{उपसमूह} पर बाध्य कर देता है
\[
f(m + n\sqrt2) = m\,f(1) + n\,f(\sqrt2) = n .
\]

\textbf{4.} $t \in \intcc{0}{\ell}$ के लिए: $a + t \in \intcc{a}{b}$, अतः
$f(t) = f(a + t) - f(a) \leq M - f(a) =: M'$।

\textbf{5.} $t \in \intcc{0}{\ell}$ के लिए $\ell - t \in \intcc{0}{\ell}$
भी, और योज्यता $f(t) = f(\ell) - f(\ell - t) \geq f(\ell) - M'$ देती है। अतः
$C = \max\bigl(\abs{M'}, \abs{f(\ell) - M'}\bigr)$ के साथ $\intcc{0}{\ell}$
पर $\abs{f} \leq C$।

\textbf{6.} $t \in \intcc{0}{\ell/n}$ के लिए: $nt \in \intcc{0}{\ell}$ और
$f(t) = \frac{f(nt)}{n}$ (प्रश्न 1), अतः $\abs{f(t)} \leq \frac Cn$।
$\varepsilon > 0$ दिया हो, तो $n > \frac C\varepsilon$ चुनिए: $0 \leq h \leq
\frac\ell n$ के लिए $\abs{f(h)} \leq \varepsilon$, और ऋणात्मक $h$ के लिए
$f(h) = -f(-h)$ का प्रयोग कीजिए। अतः $h \to 0$ होने पर $f(h) \to 0 = f(0)$:
$0$ पर \omterm{def:b1:continuity:continuous}{सांतत्य}, इसलिए सर्वत्र (प्रश्न 2), और इसलिए $f(x) = f(1)x$।

\textbf{7.} यदि $f$ $\intcc{a}{b}$ पर अह्रासमान है, तो वहाँ $f(a) \leq f(x)
\leq f(b)$: अर्थात् परिबद्ध, अतः प्रश्न 6 से रैखिक, $f(x) = cx$; और $c(b -
a) = f(b) - f(a) \geq 0$ $c \geq 0$ पर बाध्य कर देता है।

\textbf{8.} (क)$\Rightarrow$(ख)$\Rightarrow$(ग): तुच्छ। (ग)$\Rightarrow$(क):
प्रश्न 2। (क)$\Rightarrow$(घ): रैखिक फलन सर्वत्र एकदिष्ट है।
(घ)$\Rightarrow$(ङ): $\intcc{a}{b}$ पर एकदिष्ट फलन वहाँ अपने सिरों के मानों
से परिबद्ध है। (ङ)$\Rightarrow$(क): प्रश्न 4--6। पाँचों \omterm{def:b1:logic:statement}{कथन} तुल्य हैं ---
अर्थात् नियमितता की सीढ़ी।

\textbf{9.} प्रश्न 4 ने केवल \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} $M$ प्रयुक्त किया; प्रश्न 5 ने
परावर्तन $f(t) = f(\ell) - f(\ell - t)$ के द्वारा निचला परिबंध ऊपरी से
\emph{व्युत्पन्न} किया; और फिर प्रश्न 6 $\abs f \leq C$ पर चला। अतः: किसी एक
अनपकर्षी \omterm{prop:b1:reals:intervals}{अंतराल} पर योज्य और \emph{ऊपर से} परिबद्ध होना ही रैखिकता दे देता
है। निचले परिबंध के लिए यही $-f$ (जो योज्य और ऊपर से परिबद्ध है) पर लगाइए।
प्रबलतम डंडा (ङ): किसी एक \omterm{prop:b1:reals:intervals}{अंतराल} पर एकपक्षीय परिबंध पर्याप्त है।

\textbf{10.} यदि सभी $x \neq 0$ के लिए $\frac{f(x)}{x}$ कोई एक ही अचर $c$
होता, तो $f$ रैखिक होता; अतः ऐसे अशून्य $u, v$ हैं कि $\frac{f(u)}u \neq
\frac{f(v)}v$, अर्थात् $u f(v) - v f(u) \neq 0$: सदिशों $(u, f(u))$, $(v,
f(v))$ का सारणिक अशून्य है, और वे $\R^2$ को फैलाते हैं।

\textbf{11.} $r, s \in \Q$ के लिए: $f(ru + sv) = r f(u) + s f(v)$ (प्रश्न 1
दो बार और योज्यता), अतः आलेख में ये हैं
\[
\bigl(ru + sv,\; r f(u) + s f(v)\bigr)
= r\,(u, f(u)) + s\,(v, f(v)) .
\]
$(x_0, y_0)$ और $\varepsilon > 0$ दिए हों: $2 \times 2$ निकाय $a(u, f(u)) +
b(v, f(v)) = (x_0, y_0)$ का (अद्वितीय) वास्तविक हल $(a, b)$ है, क्योंकि
सारणिक अशून्य है। परिमेय $r_n \to a$, $s_n \to b$ चुनिए: तब
निर्देशांक-दर-निर्देशांक $r_n(u, f(u)) + s_n(v, f(v)) \to (x_0, y_0)$, और
इनमें से हर बिंदु आलेख पर है: अतः आलेख $\R^2$ में \omterm{def:b1:topology:dense}{सघन} है।

\textbf{12.} मान लीजिए $I$ कोई अनपकर्षी \omterm{prop:b1:reals:intervals}{अंतराल} है, $x_0$ उसका मध्यबिंदु, और
$M$ स्वेच्छ: \omterm{def:b1:topology:dense}{सघनता} $(x_0, M + 1)$ के $\min\bigl(\frac{\abs I}{2}, 1\bigr)$
के भीतर कोई आलेख-बिंदु देती है, अर्थात् $f(x) > M$ वाला $x \in I$: अतः $I$
पर अपरिबद्ध, इसलिए (प्रश्न 8) प्रत्येक बिंदु पर असंतत और किसी भी \omterm{prop:b1:reals:intervals}{अंतराल} पर
एकदिष्ट नहीं; और किसी भी लक्ष्य $y_0$ के लिए $(x_0, y_0)$ के पास के
आलेख-बिंदु $x \in I$ वाला $f(x)$ $y_0$ के मनचाहे निकट दे देते हैं: अर्थात्
$f(I)$ $\R$ में \omterm{def:b1:topology:dense}{सघन} है। यह प्रश्न 8 को उलटी दिशा में पढ़ना है: चूँकि सभी
डंडे तुल्य हैं, अतः अरैखिक योज्य फलन को उन \emph{सबमें}, और सर्वत्र, विफल
होना ही पड़ता है।

\textbf{13.} सुपरिभाषित: $m + n\sqrt2 = m' + n'\sqrt2$ $(n - n')\sqrt2 = m'
- m \in \Z$ पर बाध्य करता है, अतः $n = n'$ (अन्यथा $\sqrt2 \in \Q$) और $m =
m'$। $G$ पर योज्यता तब निर्देशांक-दर-निर्देशांक स्पष्ट है। $0^+$ के पास
अपरिबद्धता: $\varepsilon \in \intoo{0}{1}$ और $N \in \N$ नियत कीजिए। $\abs n
\leq N$ वाले प्रत्येक नियत $n$ के लिए शर्त $m + n\sqrt2 \in \intoo{0}{1}$
$m$ को $1$ लंबाई के \omterm{prop:b1:reals:intervals}{अंतराल} में कील देती है: अतः प्रति $n$ अधिक से अधिक एक
पूर्णांक $m$, इसलिए $G \cap \intoo{0}{1}$ के अधिक से अधिक $2N + 1$ अवयवों के
लिए $\abs{f} \leq N$। पर $G \cap \intoo{0}{\varepsilon}$ अनंत है ($G$ \omterm{def:b1:topology:dense}{सघन}
है, \cref{exo:b1:reals:9}); अतः उसमें ऐसा कोई $g$ है कि $\abs{f(g)} > N$:
अर्थात् $f$ $0$ के प्रत्येक दक्षिण \omterm{def:b1:topology:open}{प्रतिवेश} पर अपरिबद्ध है। $f$ को $\R$ पर
योज्य फलन तक बढ़ाने का अर्थ है वास्तविक संख्याओं के ऐसे परिवार पर संगत रूप
से मान चुनना जो $\Q$-रैखिकतः स्वतंत्र हो और $\Q$ पर $\R$ को फैलाता हो ---
अर्थात् एक हामेल आधार; और ऐसा एक उत्पन्न करने के लिए चयन-अभिगृहीत चाहिए, तथा
कोई स्पष्ट सूत्र यह नहीं कर सकता: रचनात्मक रूप से यह राक्षस हमारे पास केवल
$G$ पर है।

\textbf{14.} $f(x) = f(\frac x2 + \frac x2) = f(\frac x2)^2 \geq 0$। यदि
$f(x_0) = 0$, तो सभी $x$ के लिए $f(x) = f(x - x_0)f(x_0) = 0$: वर्जित। अतः
$f > 0$ और $g = \ln \circ f$ \omterm{def:b1:continuity:continuous}{संतत} है (\cref{prop:b1:continuity:algebra}) और
$g(x + y) = g(x) + g(y)$: \cref{exo:b1:continuity:11} से $g(x) = cx$, अतः
$f(x) = \eu^{cx}$। विलोमतः प्रत्येक $\eu^{cx}$ काम करता है: अर्थात् \omterm{def:b1:continuity:continuous}{संतत}
समाकारिताएँ $(\R, +) \to (\R^*, \times)$ ठीक चरघातांकी हैं।

\textbf{15.} $g(u) = f(\eu^u)$ \omterm{def:b1:continuity:continuous}{संतत} है और $g(u + v) = f(\eu^u \eu^v) = g(u)
+ g(v)$: $g(u) = cu$, और प्रत्येक $x > 0$ $u = \ln x$ के साथ $x = \eu^u$
लिखा जाता है: $f(x) = c\ln x$।

\textbf{16.} $h = \ln \circ f$ $\intoo{0}{+\infty}$ पर \omterm{def:b1:continuity:continuous}{संतत} है और $h(xy) =
h(x) + h(y)$: अतः प्रश्न 15 से $h(x) = c\ln x$, इसलिए $f(x) = \eu^{c\ln x} =
x^c$।

\textbf{17.} $x = y = 0$: $f(0) = 2f(0) + f(0)^2$, अतः $f(0)(1 + f(0)) = 0$।
यदि $f(0) = -1$: $y = 0$ रखने पर सभी $x$ के लिए $f(x) = f(x) + f(0) +
f(x)f(0) = -1$: अर्थात् अचर $f \equiv -1$ (जो सचमुच समीकरण संतुष्ट करता है)।
अन्यथा $f(0) = 0$; $h = 1 + f$ \omterm{def:b1:continuity:continuous}{संतत} है, $h(0) = 1$, और
\[
h(x + y) = 1 + f(x) + f(y) + f(x)f(y) = h(x)\,h(y) :
\]
प्रश्न 14 से $h(x) = \eu^{cx}$, अर्थात् $f(x) = \eu^{cx} - 1$ (और स्थिति $c
= 0$ $f \equiv 0$ देती है)। पूरी सूची: $f \equiv -1$ और $f(x) = \eu^{cx} -
1$, $c \in \R$।

\textbf{18.} $x = y = 0$: $2f(0) = 4f(0)$, अतः $f(0) = 0$। $x = 0$: $f(y) +
f(-y) = 2f(y)$, अतः $f$ सम है। $y = x$: $f(2x) = 4f(x)$। $(x, y) \to (nx,
x)$ का प्रयोग करते हुए आगमन:
\[
f((n{+}1)x) = 2f(nx) + 2f(x) - f((n{-}1)x)
= (2n^2 + 2 - (n-1)^2) f(x) = (n+1)^2 f(x) .
\]
फिर $f(x) = f\bigl(q\cdot\frac xq\bigr) = q^2 f\bigl(\frac xq\bigr)$
$f\bigl(\frac pq x\bigr) = \frac{p^2}{q^2}f(x)$ देता है: $\Q$ पर $f(r) = r^2
f(1)$ (समता चिह्न सँभाल लेती है)। दोनों \omterm{def:b1:continuity:continuous}{संतत} फलन $f$ और $x \mapsto f(1)x^2$
\omterm{def:b1:topology:dense}{सघन} \omterm{def:b1:logic:sets}{समुच्चय} $\Q$ पर मेल खाते हैं, अतः सर्वत्र (प्रश्न 24): $f(x) =
f(1)\,x^2$; और प्रत्येक $cx^2$ समीकरण संतुष्ट करता है।

\textbf{19.} $g = f - f(0)$ \omterm{def:b1:continuity:continuous}{संतत} है, $g(0) = 0$, और येंसन का समीकरण संतुष्ट
करता है (अचर कट जाते हैं)। $y = 0$ लेने पर: $g\bigl(\frac x2\bigr) =
\frac{g(x)}{2}$। फिर सभी $x, y$ के लिए:
\[
\frac{g(x + y)}{2} = g\Bigl(\frac{x+y}{2}\Bigr)
= \frac{g(x) + g(y)}{2} ,
\]
अतः $g$ योज्य और \omterm{def:b1:continuity:continuous}{संतत} है: $g(x) = cx$ (प्रश्न 2), और $d = f(0)$ के साथ $f(x)
= cx + d$। सभी ऐफ़ीन फलन येंसन को संतुष्ट करते हैं: सूची पूरी है।

\textbf{20.} $n$ पर आगमन। $n = 0$ के लिए: $\lambda \in \{0, 1\}$, जो तुच्छ
है। असमिका को सभी भार $\frac{k}{2^n}$ के लिए मान लीजिए। सम $k$ वाला भार
$\lambda = \frac{k}{2^{n+1}}$ स्तर $n$ पर आ जाता है; और $k = 2j + 1$ के लिए
$\lambda$ $\lambda_1 = \frac{j}{2^n}$ तथा $\lambda_2 = \frac{j+1}{2^n}$ का
मध्यबिंदु है। $z_i = \lambda_i x + (1 - \lambda_i)y$ के साथ: $\lambda x +
(1-\lambda)y = \frac{z_1 + z_2}{2}$, अतः
\[
f\bigl(\lambda x + (1{-}\lambda)y\bigr)
\leq \frac{f(z_1) + f(z_2)}{2}
\leq \frac{(\lambda_1 + \lambda_2)f(x) + (2 - \lambda_1 -
\lambda_2)f(y)}{2}
= \lambda f(x) + (1 - \lambda)f(y) .
\]

\textbf{21.} $x, y$ नियत कीजिए। \omterm{def:b1:logic:map}{प्रतिचित्रण} $\lambda \mapsto f(\lambda x +
(1 - \lambda)y)$ और $\lambda \mapsto \lambda f(x) + (1 - \lambda)f(y)$
$\intcc{0}{1}$ पर \omterm{def:b1:continuity:continuous}{संतत} हैं (संयोजन और बीजगणित,
\cref{prop:b1:continuity:algebra})। असमिका द्विआधारी भारों पर सत्य है, जो
$\intcc{0}{1}$ में \omterm{def:b1:topology:dense}{सघन} हैं (\cref{exo:b1:reals:8}); स्वेच्छ $\lambda$ के लिए
द्विआधारी $\lambda_n \to \lambda$ लीजिए और सीमा तक जाइए
(\cref{thm:b1:continuity:seqchar} और \cref{thm:b1:seq:order}): अतः उत्तलता
असमिका प्रत्येक $\lambda \in \intcc{0}{1}$ के लिए सत्य है --- अर्थात्
मध्यबिंदु उत्तलता और \omterm{def:b1:continuity:continuous}{सांतत्य} मिलकर उत्तलता हैं (यही धारणा
\cref{ch:b1:derivative} की है)।

\textbf{22.} कोई अरैखिक योज्य $f$ $f\bigl(\frac{x+y}{2}\bigr) = \frac{f(x) +
f(y)}{2}$ को ठीक-ठीक संतुष्ट करता है ($r = \frac12$ के साथ प्रश्न 1, फिर
योज्यता): अर्थात् वह मध्यबिंदु-उत्तल, बल्कि मध्यबिंदु-ऐफ़ीन है। यदि वह किसी
\omterm{prop:b1:reals:intervals}{अंतराल} $\intcc{x}{y}$ पर पूरी उत्तलता असमिका संतुष्ट करता, तो $\lambda \in
\intcc{0}{1}$ के लिए $f(\lambda x + (1-\lambda)y) \leq \max(f(x), f(y))$ ---
अर्थात् किसी अनपकर्षी \omterm{prop:b1:reals:intervals}{अंतराल} पर ऊपर से परिबद्ध, इसलिए प्रश्न 9 से रैखिक:
विरोधाभास। अतः प्रश्न 21 में \omterm{def:b1:continuity:continuous}{सांतत्य} कोई विलासिता नहीं है: उसके बिना
मध्यबिंदु उत्तलता केवल गणनीय द्विआधारी ढाँचे को नियंत्रित करती है, और बीच का
सातत्य जंगली भागता रहता है।

\textbf{23.} $p(x) = \frac{x^2}{2}$ $p(x+y) = p(x) + p(y) + xy$ संतुष्ट करता
है। यदि $f$ कोई भी \omterm{def:b1:continuity:continuous}{संतत} हल है, तो $g = f - p$ \omterm{def:b1:continuity:continuous}{संतत} और योज्य है, अतः $g(x) =
cx$:
\[
f(x) = \frac{x^2}{2} + cx , \qquad c \in \R ,
\]
और इनमें से हर एक हल है: सूची पूरी है।

\textbf{24.} व्यापक सिद्धांत: यदि $u, v$ \omterm{def:b1:continuity:continuous}{संतत} हैं और किसी \omterm{def:b1:topology:dense}{सघन} $D \subseteq
\R$ पर मेल खाते हैं, तो $x \in \R$ के लिए ऐसे $d_n \in D$ चुनिए कि $d_n \to
x$ (\cref{prop:b1:topology:closureprops}); $u(x) = \lim u(d_n) = \lim v(d_n)
= v(x)$। अब मान लीजिए $f$ \omterm{def:b1:continuity:continuous}{संतत} है और \omterm{def:b1:topology:dense}{सघन} \omterm{def:b1:structures:subgroup}{उपसमूह} $G$ पर योज्य है। $x, y \in
\R$ नियत कीजिए और ऐसे $g_n \to x$, $g'_n \to y$ लीजिए कि $g_n, g'_n \in G$;
तब $g_n + g'_n \to x + y$, और $x + y$ पर अनुक्रमीय \omterm{def:b1:continuity:continuous}{सांतत्य} से $x$ तथा $y$:
\[
f(x + y) = \lim f(g_n + g'_n) = \lim\bigl(f(g_n) +
f(g'_n)\bigr) = f(x) + f(y) :
\]
अतः $f$ पूरे $\R$ पर योज्य है (और इसलिए प्रश्न 2 से रैखिक)।

\textbf{25.} (क) योज्य $f$ के लिए: रैखिक $\iff$ \omterm{def:b1:continuity:continuous}{संतत} $\iff$ किसी एक बिंदु पर
\omterm{def:b1:continuity:continuous}{संतत} $\iff$ किसी \omterm{prop:b1:reals:intervals}{अंतराल} पर एकदिष्ट $\iff$ किसी \omterm{prop:b1:reals:intervals}{अंतराल} पर (एकपक्षीय तक)
परिबद्ध। (ख) आलेख का समतल में \omterm{def:b1:topology:dense}{सघन} होना दिखाता है कि विफलता कोई स्थानीय दोष
नहीं, बल्कि समग्र विस्फोट है: हर उपअंतराल के ऊपर मान पूरे $\R$ पर बिखर जाते
हैं, अतः नियमितता का हर गुणधर्म एक साथ सर्वत्र विफल हो जाता है। (ग) पकड़:
$cx$, $\eu^{cx}$, $c\ln x$, $x^c$, $cx^2$ और $cx + d$ --- छह अभिलक्षण, और एक
ही विधि: समीकरण को कोशी वाले पर ले जाइए, $\Q$-ढाँचा आगमन से सिद्ध कीजिए, और
\omterm{def:b1:topology:dense}{सघनता} तथा \omterm{def:b1:continuity:continuous}{सांतत्य} से $\R$ तक उठाइए। (घ) $\Q$ (या द्विआधारी संख्याओं) की
\omterm{def:b1:topology:dense}{सघनता} ने \cref{exo:b1:continuity:11} में और प्रश्न 21 में उन्नयन ढोए; प्रश्न
13 में उसने कुछ नहीं ढोया, क्योंकि \omterm{def:b1:continuity:continuous}{सांतत्य} के बिना मान किसी \omterm{def:b1:topology:dense}{सघन} \omterm{def:b1:logic:sets}{समुच्चय} से
उसके \omterm{def:b1:topology:closure}{संवरक} तक नहीं फैलते --- \omterm{def:b1:topology:dense}{सघनता} सूचना केवल \omterm{def:b1:continuity:continuous}{सांतत्य} के अनुदिश ले जाती है।
\end{solution}
