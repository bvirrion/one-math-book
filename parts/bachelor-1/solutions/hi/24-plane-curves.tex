\section*{अध्याय \ref{ch:b1:curves} --- समतल वक्र}

\begin{solution}{exo:b1:curves:1}
$M(-t) = (t^2, -t^3)$: $x$-अक्ष में परावर्तन; अतः $t \geq 0$ का अध्ययन
कीजिए। $x' = 2t$ और $y' = 3t^2$ दोनों $\geq 0$ हैं: अतः शाखा दाईं और ऊपर की
ओर बढ़ती है, $(0,0)$ से अनंत तक। $t = 0$ पर वेग शून्य हो जाता है; और प्रसार
$x = t^2$, $y = t^3$ दिखाते हैं कि स्पर्शरेखा $x$-अक्ष है ($y/x = t \to 0$),
जहाँ $y$ चिह्न बदलता है जबकि $x \geq 0$: अर्थात् बाईं ओर इशारा करता एक
\emph{\omterm{ex:b1:curves:astroid}{उभयाग्र}}। यह वक्र अर्ध-घनीय परवलय $y^2 = x^3$ है।
\end{solution}

\begin{solution}{exo:b1:curves:2}
$M(t + 2\pi) = M(t) + (2\pi, 0)$: वक्र स्वयं को दोहराता है, बस एक
पहिया-परिधि जितना सरका हुआ; अतः एक आवर्त का अध्ययन कीजिए। $x'(t) = 1 - \cos
t \geq 0$, $y'(t) = \sin t$: मेहराब $\intoo{0}{\pi}$ पर चढ़ता है और
$\intoo{\pi}{2\pi}$ पर उतरता है, तथा उसका शिखर $(\pi, 2)$ है। विचित्र बिंदु
वहाँ जहाँ $x' = y' = 0$: $t \in 2\pi\Z$, अर्थात् ज़मीन पर। $t = 0$ के निकट:
\[
x(t) = \frac{t^3}{6} + o(t^3),
\qquad
y(t) = \frac{t^2}{2} + o(t^2):
\]
$x$ चिह्न बदलता है, $y \geq 0$, और $\frac{x}{y^{3/2}}$ परिबद्ध है: अतः
स्पर्शरेखा \emph{ऊर्ध्वाधर} है (दिशा $(0,1)$), और यह एक \omterm{ex:b1:curves:astroid}{उभयाग्र} है जहाँ
अनुगत बिंदु की चाल क्षण भर के लिए शून्य होती है --- बिना फिसले लुढ़कने का
भौतिक हस्ताक्षर।
\end{solution}

\begin{solution}{exo:b1:curves:3}
$t = \frac\pi4$ पर: $M = \bigl(\frac{\sqrt2}{4}, \frac{\sqrt2}{4}\bigr)$,
स्पर्श-दिशा $(-\cos t, \sin t) = \frac{1}{\sqrt2}(-1, 1)$
(\cref{ex:b1:curves:astroid})। स्पर्शरेखा: $y - \frac{\sqrt2}{4} = -(x -
\frac{\sqrt2}{4})$, अर्थात् $x + y = \frac{\sqrt2}{2}$। अंतःखंड:
$\bigl(\frac{\sqrt2}{2}, 0\bigr)$ और $\bigl(0, \frac{\sqrt2}{2}\bigr)$; और
उनके बीच के खंड की \omterm{def:b1:curves:arclength}{लंबाई} $\sqrt{\frac12 + \frac12} = 1$ है।

(व्यापक बिंदु $t$: $(\cos^3 t, \sin^3 t)$ पर की स्पर्शरेखा अक्षों को $(\cos
t, 0)$ और $(0, \sin t)$ पर काटती है --- जाँचिए कि इन बिंदुओं से जाती रेखा की
दिशा $(-\cos t, \sin t)$ है और वह $M(t)$ से होकर जाती है --- और कटे हुए खंड
की \omterm{def:b1:curves:arclength}{लंबाई} $\sqrt{\cos^2 t + \sin^2 t} = 1$ है।)
\end{solution}

\begin{solution}{exo:b1:curves:4}
$r = \cos 2\theta$: $\theta$ में आवर्त $\pi$, और दोनों अक्षों में सममित; $r$
$\theta = \pm\frac\pi4$ पर शून्य होता है (मूल बिंदु पर स्पर्शरेखाएँ विकर्णों
के अनुदिश) और $\theta \in \intoo{\frac\pi4}{\frac{3\pi}{4}}$ के लिए ऋणात्मक

0, \frac\pi2, \pi, \frac{3\pi}{2}$ के अनुदिश चार पंखुड़ियाँ बनती हैं, और हर
एक की अधिकतम त्रिज्या $1$ है।

$r\cos\theta = 1$ वही समीकरण $x = 1$ है: अतः \omterm{def:b1:curves:polar}{ध्रुवीय वक्र} $r =
\frac{1}{\cos\theta}$ ऊर्ध्वाधर रेखा $x = 1$ है ($\theta \in
\intoo{-\frac\pi2}{\frac\pi2}$ के लिए एक बार बुहारी हुई)।
\end{solution}

\begin{solution}{exo:b1:curves:5}
$r(\theta) = 1 + 2\cos\theta$ $\cos\theta = -\frac12$ के लिए शून्य होता है:
$\theta = \pm\frac{2\pi}{3}$। $x$-अक्ष में सममिति; अतः $\theta \in
\intcc{0}{\pi}$ का अध्ययन कीजिए। $\theta \in \intco{0}{\frac{2\pi}{3}}$ के
लिए: $r > 0$, जो $3$ से घटकर $0$ तक आता है: बाहरी चाप, जो किरण $\theta =
\frac{2\pi}{3}$ के स्पर्शतः मूल बिंदु में प्रवेश करता है। $\theta \in
\intoc{\frac{2\pi}{3}}{\pi}$ के लिए: $r < 0$: बिंदु $r\vec u(\theta)$ विपरीत
किरण पर बैठते हैं (कोण $\theta - \pi \in \intoc{-\frac\pi3}{0}$), और दूरी
$\abs r$ $0$ से बढ़कर $1$ हो जाती है: इससे मूल बिंदु से होकर एक छोटा
\emph{भीतरी फंदा} खिंचता है। $\theta = \pi$ पर $r = -1$, और बिंदु $-\vec
u(\pi) = (1, 0)$ है: अतः फंदा धनात्मक $x$-अक्ष पर बंद हो जाता है। रेखाचित्र:
एक बड़ा हृदय-जैसा बाहरी वक्र, जिसकी अधिकतम पहुँच $\theta = 0$ पर $3$ है, और
उसके भीतर $O$ तथा $(1,0)$ से होकर जाता एक फंदा।
\end{solution}

\begin{solution}{exo:b1:curves:6}
वर्ग पूरे कीजिए:
\[
9(x^2 - 4x) + 25(y^2 - 2y) = 164
\iff 9(x-2)^2 + 25(y-1)^2 = 164 + 36 + 25 = 225 .
\]
$225$ से भाग देने पर: $\frac{(x-2)^2}{25} + \frac{(y-1)^2}{9} = 1$: अर्थात्
केंद्र $(2, 1)$ वाला दीर्घवृत्त, जिसके अर्ध-अक्ष $a = 5$ (क्षैतिज), $b = 3$
हैं; $c = \sqrt{25 - 9} = 4$: नाभियाँ $(2 \pm 4,\, 1) = (-2, 1)$ और $(6,
1)$; उत्केंद्रता $e = \frac45$।
\end{solution}

\begin{solution}{exo:b1:curves:7}
नाभि मूल बिंदु पर, नियता $D: x = d$, जहाँ $d = \frac pe > 0$। $r > 0$ वाले
किसी बिंदु $M = (r\cos\theta, r\sin\theta)$ के लिए:
\[
MF = r,
\qquad
d(M, D) = \abs{d - r\cos\theta},
\]
और शांकव-शर्त $MF = e\,d(M, D)$, उस स्थिति में जब $M$ नियता की नाभि वाली ओर
है ($r\cos\theta < d$), $r = e(d - r\cos\theta)$ कहती है, अर्थात्
\[
r(1 + e\cos\theta) = ed = p,
\qquad
r = \frac{p}{1 + e\cos\theta} .
\]
$e < 1$ के लिए हर कभी शून्य नहीं होता: अतः सभी $\theta$ अनुमत (दीर्घवृत्त)।
$e = 1$ के लिए: $\theta \neq \pi$ (परवलय, जो नियता की दूर वाली ओर खुलता है)।
$e > 1$ के लिए: $1 + e\cos\theta > 0$ चाहिए, अर्थात् $\theta_0 =
\arccos\bigl(-\frac1e\bigr)$ के साथ $\theta \in
\intoo{-\theta_0}{\theta_0}$: अर्थात् अतिपरवलय की एक \emph{शाखा} (दूसरी शाखा
चिह्न-चुनाव $r < 0$ के, अथवा दूसरी नाभि के अनुरूप है)।
\end{solution}

\begin{solution}{exo:b1:curves:8}
\emph{माली:} $2a$ \omterm{def:b1:curves:arclength}{लंबाई} की एक डोरी दो खूँटों $F, F'$ से बाँधिए और उसे
अनुरेखी बिंदु $M$ से तनी रखिए: तब बाध्यता ठीक $MF + MF' = 2a$ है, अतः खिंचा
हुआ वक्र वही दीर्घवृत्त है।

\emph{परावर्तन गुण:} मान लीजिए $t \mapsto M(t)$ कोई नियमित प्राचलन है और
$u(t) = \frac{M(t) - F}{\norm{M(t) - F}}$, तथा $u'(t)$ $F'$ के लिए तदनुरूप
इकाई सदिश। $\norm{M - F} + \norm{M - F'} = 2a$ का अवकलन करने पर,
$\frac{\dd}{\dd t}\norm{M - F} = \bigl\langle \frac{M - F}{\norm{M-F}},\,
M'\bigr\rangle$ का प्रयोग करते हुए ($\sqrt{\langle\cdot,\cdot\rangle}$ पर
शृंखला नियम):
\[
\bigl\langle u + u',\, M'\bigr\rangle = 0 .
\]
अतः स्पर्श-दिशा $M'$ दोनों नाभीय किरणों की समद्विभाजक दिशा $u + u'$ के
\omterm{def:b1:euclid:orthogonal}{लांबिक} है: इसलिए स्पर्शरेखा $MF$ और $MF'$ के साथ बराबर कोण बनाती है। (एक
नाभि से चली प्रकाश-किरण दीर्घवृत्त से परावर्तित होकर दूसरी नाभि पर पहुँचती
है।)
\end{solution}

\begin{solution}{exo:b1:curves:9}
\begin{enumerate}
  \item प्रांत: $\cos 2\theta \geq 0$: $\theta \in
  \intcc{-\frac\pi4}{\frac\pi4} \cup
  \intcc{\frac{3\pi}{4}}{\frac{5\pi}{4}}$। सममितियाँ: $\theta \mapsto
  -\theta$ ($x$-अक्ष) और $\theta \mapsto \pi - \theta$ ($y$-अक्ष): अतः
  $\intcc{0}{\frac\pi4}$ का अध्ययन कीजिए; $r$ $1$ से घटकर $0$ तक आता है।
  $\theta = \pm\frac\pi4$ पर: $r = 0$, और मूल बिंदु पर स्पर्शरेखाएँ विकर्णों
  के अनुदिश। वक्र $\infty$ चिह्न जैसा है: $O$ पर मिलते दो सममित फंदे, जिनकी
  पहुँच $(\pm 1, 0)$ है।
  \item $F, F' = (\pm c, 0)$, $c = \frac{1}{\sqrt 2}$, और $M = (r\cos\theta,
  r\sin\theta)$ के साथ:
        \[
        MF^2\, MF'^2
        = \bigl((r\cos\theta - c)^2 + r^2\sin^2\theta\bigr)
        \bigl((r\cos\theta + c)^2 + r^2\sin^2\theta\bigr)
        = (r^2 + c^2)^2 - (2rc\cos\theta)^2 ,
        \]
        यह $A = r^2 + c^2$, $B = 2rc\cos\theta$ के साथ सर्वसमिका $(A - B)(A
        + B) = A^2 - B^2$ से मिलता है। $c^2 = \frac12$ के साथ:
        \[
        MF^2 MF'^2 = \Bigl(r^2 + \frac12\Bigr)^2 - 2r^2\cos^2\theta
        = r^4 + r^2\bigl(1 - 2\cos^2\theta\bigr) + \frac14
        = r^4 - r^2\cos 2\theta + \frac14 .
        \]
        लेम्निस्केट पर $r^2 = \cos 2\theta$: अतः पहले दो पद कट जाते हैं और
        $MF^2MF'^2 = \frac14$ बचता है, अर्थात् $MF \cdot MF' = \frac12$।
        विलोमतः, इसी संगणना को उलटा पढ़ने पर दिखता है कि बिंदुपथ-समीकरण
        $MF\,MF' = \frac12$ $r^2 = \cos 2\theta$ है ($r \neq 0$ के लिए; और
        $O$ दोनों को संतुष्ट करता है)।
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:curves:10}
$r = 1 + \cos\theta$, $r' = -\sin\theta$:
\[
r^2 + r'^2 = 1 + 2\cos\theta + \cos^2\theta + \sin^2\theta
= 2 + 2\cos\theta = 4\cos^2\frac\theta2 ,
\]
अतः $\sqrt{r^2 + r'^2} = 2\abs{\cos\frac\theta2}$, और $x$-अक्ष में सममिति
से,
\[
L = \int_0^{2\pi} 2\Bigl|\cos\frac\theta2\Bigr|\,\dd\theta
= 2\int_0^{\pi} 2\cos\frac\theta2\,\dd\theta
= 2\Bigl[4\sin\frac\theta2\Bigr]_0^{\pi} = 8 .
\]
एक और बीजीय, $\pi$-मुक्त परिमाप।
\end{solution}

\begin{solution}{exo:b1:curves:11}
बिंदु $(a\cos t, b\sin t)$ समीकरण को संतुष्ट करता है: $\cos^2 t + \sin^2 t =
1$। रेखा का अभिलंब सदिश $\bigl(\frac{\cos t}a, \frac{\sin t}b\bigr)$ है, और
वेग $(-a\sin t, b\cos t)$ के साथ उसका गुणन $-\sin t\cos t + \sin t\cos t =
0$ है: अतः वह रेखा उस बिंदु से होकर स्पर्श-दिशा में जाती है --- यानी वही
स्पर्शरेखा है। दीर्घवृत्त पर $(x_0, y_0)$ के लिए $\cos t = \frac{x_0}a$,
$\sin t = \frac{y_0}b$ लिखिए और प्रतिस्थापित कीजिए:
\[
\frac{x\,x_0}{a^2} + \frac{y\,y_0}{b^2} = 1 ,
\]
जो दीर्घवृत्त-समीकरण से $x^2 \mapsto x\,x_0$ और $y^2 \mapsto y\,y_0$ को
``बाँट देने'' से मिलता है।
\end{solution}

\begin{solution}{exo:b1:curves:12}
\begin{enumerate}
  \item $r' = k\eu^{k\theta}$, अतः $\tan V = \frac{r}{r'} = \frac1k$: जो अचर
  है। सर्पिल मूल बिंदु से जाती हर किरण को एक ही कोण $V = \arctan\frac1k$ पर
  काटता है।
  \item सम्मिश्र संकेतन में सर्पिल $\{\eu^{k\theta} \eu^{\iu\theta} : \theta
  \in \R\}$ है। $c$ से घुमाने पर वह $\eu^{\iu c}$ से गुणित हो जाता है:
        \[
        \eu^{k\theta}\eu^{\iu(\theta + c)}
        = \eu^{-kc}\,\eu^{k(\theta + c)}\eu^{\iu(\theta+c)} ,
        \]
        और जैसे-जैसे $\theta + c$ $\R$ पर चलता है, यह सर्पिल का $\eu^{-kc}$
        गुना बताता है: अर्थात् घूर्णन $=$ मापन। रेखाओं और वृत्तों के
        अतिरिक्त कोई अन्य चिकना वक्र यह गुण नहीं रखता।
  \item $\sqrt{r^2 + r'^2} = \sqrt{1 + k^2}\,\eu^{k\theta}$, अतः
  $\intcc{A}{\theta_0}$ पर \omterm{def:b1:curves:arclength}{लंबाई}
  $\frac{\sqrt{1+k^2}}{k}\bigl(\eu^{k\theta_0} - \eu^{kA}\bigr)$ है, और $A
  \to -\infty$ होने पर:
        \[
        L = \frac{\sqrt{1 + k^2}}{k}\,\eu^{k\theta_0} < \infty :
        \]
        अर्थात् मूल बिंदु के चारों ओर अनंत चक्कर, फिर भी परिमित कुल \omterm{def:b1:curves:arclength}{लंबाई}
        (चक्कर गुणोत्तर रूप से सिकुड़ते जाते हैं)।
\end{enumerate}
\end{solution}

\begin{solution}{pb:b1:curves:1}
\textbf{1.} बिना फिसले लुढ़कने का अर्थ है कि स्पर्श-बिंदु उतनी ही दूरी चला
है जितना पहिये का चाप खुला: अतः $t$ घूमने के बाद केंद्र $(Rt, R)$ पर है।
चिह्नित बिंदु परिधि पर, नीचे की ऊर्ध्वाधर से $t$ कोण \emph{पीछे} बैठा है
(पहिया आगे बढ़ते हुए दक्षिणावर्त घूमता है):
\[
M(t) = \Omega(t) + R(-\sin t, -\cos t)
= \bigl(R(t - \sin t),\ R(1 - \cos t)\bigr),
\]
जो $t = 0$ पर ($M = (0,0)$) और $t = \pi$ पर ($M = (\pi R, 2R)$, शीर्ष) सही
है।

\textbf{2.} $f'(t) = R(1 - \cos t,\ \sin t)$ और
\[
\norm{f'}^2 = R^2\bigl((1-\cos t)^2 + \sin^2 t\bigr)
= 2R^2(1 - \cos t) = 4R^2\sin^2\frac t2 ,
\]
अतः $\norm{f'} = 2R\abs{\sin\frac t2}$। विचित्र बिंदु $t \in 2\pi\Z$ पर:
ज़मीन पर के \omterm{ex:b1:curves:astroid}{उभयाग्र} (\cref{exo:b1:curves:2}); और मेहराब का शीर्ष $t = \pi$
है, जहाँ चाल $2R$ अधिकतम है।

\textbf{3.} अर्ध-कोण गुणनखंडन:
\[
f'(t) = 2R\sin\frac t2\,\Bigl(\sin\frac t2,\ \cos\frac t2\Bigr),
\quad
M - C = 2R\sin\frac t2\,\Bigl(-\cos\frac t2,\ \sin\frac t2\Bigr),
\]
\[
T - M = \bigl(R\sin t,\ R(1 + \cos t)\bigr)
= 2R\cos\frac t2\,\Bigl(\sin\frac t2,\ \cos\frac t2\Bigr).
\]
$0 < t < 2\pi$ के लिए $\sin\frac t2 \neq 0$: $M - C$ स्पर्श-दिशा
$\bigl(\sin\frac t2, \cos\frac t2\bigr)$ के \omterm{def:b1:euclid:orthogonal}{लांबिक} है (उनका गुणन $-\sin\frac
t2\cos\frac t2 + \sin\frac t2\cos\frac t2 = 0$ है), और $T - M$ उसके समांतर।
पहिये के शीर्ष तक की जीवा स्पर्शरेखा \emph{है}; और स्पर्श-बिंदु तक की जीवा
अभिलंब।

\textbf{4.} हर क्षण पहिया अपने स्पर्श-बिंदु के परितः घूमता है (उस बिंदु का
वेग शून्य है: बिना फिसले लुढ़कना), अतः पहिये का हर दृढ़ बिंदु उसे $C$ से
जोड़ती रेखा के \omterm{def:b1:euclid:orthogonal}{लांबिक} चलता है, और उसकी चाल (कोणीय वेग $1$) उसी दूरी के बराबर
होती है। जाँच: $\norm{M - C} = 2R\abs{\sin\frac t2} = \norm{f'(t)}$।

\textbf{5.} $2R\,y(t) = 2R\cdot 2R\sin^2\frac t2 = 4R^2\sin^2\frac t2 =
\norm{f'(t)}^2$। ऊँचाई $y$ पर चाल $\sqrt{2R\,y}$ है --- अर्थात् औपचारिक रूप
से मुक्त-पतन का नियम $v = \sqrt{2gh}$, जहाँ ज़मीन छत की भूमिका निभा रही है;
और भाग IV इसी प्रेक्षण को घड़ी में बदल देता है।

\textbf{6.} \cref{def:b1:curves:arclength} और प्रश्न 2 से ($\intcc{0}{2\pi}$
पर $\sin\frac t2 \geq 0$):
\[
L = \int_0^{2\pi} 2R\sin\frac t2\,\dd t
= 2R\Bigl[-2\cos\frac t2\Bigr]_0^{2\pi} = 2R(2 + 2) = 8R .
\]
रेन की प्रमेय: ठीक चार व्यास।

\textbf{7.} $s(t) = \int_0^t 2R\sin\frac u2\,\dd u = 4R\bigl(1 - \cos\frac
t2\bigr)$; $s(2\pi) = 4R(1+1) = 8R$, जो संगत है।

\textbf{8.} $\sigma(t) = \abs{s(t) - s(\pi)} = \abs{4R(1 - \cos\frac t2) -
4R} = 4R\abs{\cos\frac t2}$, अतः $\sigma^2 = 16R^2\cos^2\frac t2$; और $2R -
y = R(1 + \cos t) = 2R\cos^2\frac t2$, जिससे $8R(2R - y) = 16R^2\cos^2\frac
t2 = \sigma^2$।

\textbf{9.} \omterm{ex:b1:curves:astroid}{उभयाग्र} $t = 0$ पर (अथवा $2\pi$): $\sigma = 4R$, जो $8R$ का आधा
है: अर्थात् शीर्ष मेहराब को आधा कर देता है। दोनों संगणनाओं में चाल
$\abs{\text{त्रिकोणमितीय बहुपद, चर } t/2}$ है, जिसका प्रतिअवकलज फिर
त्रिकोणमितीय है: अतः \omterm{def:b1:curves:arclength}{लंबाई} कोज्याओं के \emph{मानों} का अंतर है --- यानी
परिमेय संख्याएँ गुणा $R$ --- और नापने के लिए कोई वृत्त-चाप है ही नहीं, इसलिए
कोई $\pi$ भी नहीं। वृत्त की अपनी चाल अचर है, अतः उसका लंबाई-समाकल पूरा
अंतराल-दैर्घ्य $2\pi$ पैदा करता है; जबकि साइक्लॉयड की चाल सिरों पर शून्य हो
जाती है और बीजीय रूप से समाकलित होती है।

\textbf{10.} मेहराब $x$ के $0$ से $2\pi R$ तक बढ़ते हुए बुहारा जाता है ($x'
= R(1 - \cos t) \geq 0$, जो केवल छिटपुट बिंदुओं पर शून्य होता है)।
क्षेत्रफल-समाकल $\int_0^{2\pi R} y\,\dd x$ (\cref{thm:b1:integration:parts})
में $x = x(t)$ प्रतिस्थापित करने पर $A = \int_0^{2\pi} y(t)\,x'(t)\,\dd t$
मिलता है।

\textbf{11.}
\[
A = \int_0^{2\pi} R(1 - \cos t)\cdot R(1 - \cos t)\,\dd t
= R^2\int_0^{2\pi}\bigl(1 - 2\cos t + \cos^2 t\bigr)\dd t
= R^2\Bigl(2\pi - 0 + \pi\Bigr) = 3\pi R^2 ,
\]
$\int_0^{2\pi}\cos^2 = \pi$ का प्रयोग करते हुए। ठीक तीन पहिया-क्षेत्रफल:
गैलीलियो की तराज़ू ने कहा था ``लगभग $3$''; रोबेरवाल की संगणना कहती है
``ठीक-ठीक''।

\textbf{12.} $x = \cos^3 t$, $y = \sin^3 t$ के साथ: $y\,x' = -3\sin^4
t\cos^2 t$। \omterm{def:b1:linmaps:def}{रैखिक} कीजिए:
\[
\sin^4 t\cos^2 t = \frac{1 - \cos 2t}{2}\cdot\frac{\sin^2
2t}{4} = \frac{\sin^2 2t}{8} - \frac{\sin^2 2t\cos 2t}{8} ,
\]
और $\intcc{0}{2\pi}$ पर: $\int\sin^2 2t = \pi$, $\int \sin^2 2t\cos 2t =
\bigl[\frac{\sin^3 2t}{6}\bigr] = 0$। अतः $\oint y\,\dd x =
-3\cdot\frac\pi8$, और घिरा हुआ क्षेत्रफल $\frac{3\pi}8$ है (चिह्न वामावर्त
दिक्विन्यास दर्ज करता है)।

\textbf{13.} $\pi$ से $0$ तक $t$ के साथ $(\cos t, \sin t)$ के लिए $x$ $-1$
से बढ़कर $1$ हो जाता है और
\[
\int_\pi^0 \sin t\cdot(-\sin t)\,\dd t
= \int_0^\pi \sin^2 t\,\dd t = \frac\pi2 :
\]
अतः सूत्र ऊपरी अर्ध-चक्र का क्षेत्रफल लौटा देता है, जैसा होना ही चाहिए।

\textbf{14.} $x' = R(1 + \cos t) = 2R\cos^2\frac t2$, $y' = R\sin t =
2R\sin\frac t2\cos\frac t2$, अतः $\norm{f'} = 2R\cos\frac t2$
($\intcc{-\pi}{\pi}$ पर ऋणेतर)। तल से चाप: $s(t) = \int_0^t 2R\cos\frac
u2\,\dd u = 4R\sin\frac t2$। तब
\[
y = R(1 - \cos t) = 2R\sin^2\frac t2
= 2R\Bigl(\frac{s}{4R}\Bigr)^{\!2} = \frac{s^2}{8R} .
\]

\textbf{15.} $v = \frac{\dd s}{\dd\tau}$ और $y = \frac{s^2}{8R}$, $y_0 =
\frac{s_0^2}{8R}$ के साथ ऊर्जा-संरक्षण:
\[
\Bigl(\frac{\dd s}{\dd\tau}\Bigr)^{\!2} = 2g(y_0 - y)
= \frac{2g}{8R}\bigl(s_0^2 - s^2\bigr)
= \frac{g}{4R}\bigl(s_0^2 - s^2\bigr).
\]

\textbf{16.} $\tau$ में अवकलन करने पर: $2s's'' = -\frac{g}{4R}\,2ss'$, अतः
जहाँ भी $s' \neq 0$ (और संततता से सर्वत्र), $s'' = -\frac{g}{4R}s$: अर्थात्
सरल आवर्त दोलक। \cref{thm:b1:diffeq:homogeneous2} से $\omega =
\sqrt{g/(4R)}$ के साथ $s(\tau) = A\cos\omega\tau + B\sin\omega\tau$; और
आरंभिक शर्तें $s(0) = s_0$, $s'(0) = 0$ $s(\tau) = s_0\cos\omega\tau$ देती
हैं।

\textbf{17.} मनका तल पर तब पहुँचता है जब $s = 0$, अर्थात् $\omega\tau =
\frac\pi2$ पर:
\[
T_{\downarrow} = \frac{\pi}{2\omega}
= \frac\pi2\sqrt{\frac{4R}{g}} = \pi\sqrt{\frac Rg}\,,
\]
जो $s_0$ से स्वतंत्र है: कटोरे में कहीं से भी छोड़े जाएँ, सारे मनके एक ही
क्षण पहुँचते हैं --- यही समकाल गुण है।

\textbf{18.} $s = s_0\cos\omega\tau$ प्रतिस्थापित करने पर: बायाँ पक्ष
$s_0^2\omega^2\sin^2\omega\tau$ है और दायाँ पक्ष $\frac{g}{4R}s_0^2(1 -
\cos^2\omega\tau) = s_0^2\omega^2\sin^2\omega\tau$: अर्थात् यथार्थ समानता,
अतः कोई मिथ्या हल नहीं आया। किसी वृत्तीय चाप के लिए $y$ $s^2$ का समानुपाती
नहीं होता ($y = \ell(1 - \cos\frac s\ell) = \frac{s^2}{2\ell} -
\frac{s^4}{24\ell^3} + \dots$): प्रत्यानयन पद केवल \emph{लगभग} \omterm{def:b1:linmaps:def}{रैखिक} है, अतः
वृत्तीय लोलक का आवर्त आयाम के साथ खिसकता है, जबकि साइक्लॉयड का कठोरता से अचर
रहता है।

\textbf{19.} $\pi\sqrt{R/g} = 1$ से $R = \frac{g}{\pi^2} \approx
\frac{9.81}{9.87} \approx 0.994$ मीटर मिलता है --- लगभग ठीक एक मीटर। पूरा
दोलन $\frac{2\pi}\omega = 2\pi\sqrt{4R/g}$ लेता है, जो \omterm{def:b1:curves:arclength}{लंबाई} $\ell = 4R$
वाले लोलक के लिए ठीक लघु-कोण सूत्र $2\pi\sqrt{\ell/g}$ है: हाइगेंस ने अपना
लोलक ठीक उसी अनुपात के साइक्लॉयडी गालों से लटकाया था।

\textbf{20.} $t = \frac\pi2$ पर, $R = 1$: $M = \bigl(\frac\pi2 - 1,\
1\bigr)$, $T = \bigl(\frac\pi2,\ 2\bigr)$, अतः $T - M = (1, 1)$। और
$f'(\frac\pi2) = (1 - 0,\ 1) = (1, 1)$: अर्थात् शीर्ष तक की जीवा ही वेग है,
जैसा वादा था।

\textbf{21.} ट्रोकॉयड के लिए $x'(t) = R - d\cos t$ और $y'(t) = d\sin t$। यदि
$d < R$: तो $x' \geq R - d > 0$, अतः बिंदु सदा आगे बढ़ता है; वेग कभी शून्य
नहीं होता (उसका पहला घटक धनात्मक है): कोई विचित्र बिंदु नहीं, बस एक नियमित
लहर। यदि $d > R$: तो $x'(0) = R - d < 0 < R + d = x'(\pi)$, अतः बिंदु
स्पर्श-क्षणों के निकट पीछे और अन्यत्र आगे चलता है: और वक्र फंदों में स्वयं
को काटता है। रेल के पहिये के किनारे पर का कोई बिंदु (पटरी के शीर्ष से नीचे,
$d > R$) हर चक्कर में पीछे की ओर चलता है।

\textbf{22.} \omterm{ex:b1:curves:astroid}{उभयाग्र} $(\pi R, 2R)$ से मूल बिंदु तक की जीवा की \omterm{def:b1:curves:arclength}{लंबाई} $L =
R\sqrt{\pi^2 + 4}$ है और उसका ढाल-कोण $\alpha$ है, जहाँ $\sin\alpha =
\frac{2R}{L}$। विराम से अचर त्वरण $g\sin\alpha$ के साथ फिसलने पर: $L =
\frac12 g\sin\alpha\,T^2$, अतः
\[
T = \sqrt{\frac{2L}{g\sin\alpha}} = \sqrt{\frac{2L^2}{2Rg}}
= \frac{L}{\sqrt{Rg}} = \sqrt{\pi^2 + 4}\,\sqrt{\frac Rg}
\approx 3.72\sqrt{\frac Rg}\,,
\]
जबकि साइक्लॉयड के लिए $\pi\sqrt{R/g} \approx 3.14\sqrt{R/g}$: अर्थात् वक्र
वाला पथ तेज़ है। वस्तुतः वही सबसे तेज़ संभव पथ है --- लघुतमकाल वक्र --- और
यह विचरण-कलन की प्रमेय है, जो इस खंड से आगे की बात है।

\textbf{23.} $\dfrac{\dd y}{\dd x} = \dfrac{y'}{x'} = \dfrac{\sin t}{1 -
\cos t} = \cot\dfrac t2$ (अर्ध-कोण सूत्र)। तब
\[
\frac{\dd^2y}{\dd x^2}
= \frac{\dd}{\dd t}\Bigl(\cot\frac t2\Bigr)\cdot\frac{1}{x'(t)}
= -\frac{1}{2\sin^2\frac t2}\cdot\frac{1}{2R\sin^2\frac t2}
= -\frac{1}{4R\sin^4\frac t2} < 0 :
\]
अर्थात् पूरे मेहराब पर अवतल; और $t \to 0^+$ अथवा $2\pi^-$ होने पर ढाल
$\cot\frac t2 \to \pm\infty$: यानी उभयाग्रों पर ऊर्ध्वाधर स्पर्शरेखाएँ।

\textbf{24.} जीवा की दिशा $T - M \parallel \bigl(\sin\frac t2, \cos\frac
t2\bigr)$ है, जो $t \to 0^{+}$ होने पर $(0, 1)$ की ओर जाती है: अतः \omterm{ex:b1:curves:astroid}{उभयाग्र}
पर स्पर्शरेखा ऊर्ध्वाधर है --- और यह शुद्ध ज्यामिति से, बिना किसी टेलर
प्रसार के मिल गया।

\textbf{25.} (क) रेन की प्रमेय, $L = 8R$; रोबेरवाल का क्षेत्रफल, $A = 3\pi
R^2$ (गैलीलियो का अनुमानित अनुपात $3$); हाइगेंस का समकाल गुण, $T_\downarrow
= \pi\sqrt{R/g}$। (ख) हर संगणना अर्ध-कोण गुणनखंडनों $f' = 2R\sin\frac
t2(\sin\frac t2, \cos\frac t2)$ और कटोरे वाले उनके समकक्ष पर चली --- एक ही
सर्वसमिका, जो स्पर्शरेखा, \omterm{def:b1:curves:arclength}{लंबाई} और घड़ी, तीनों को चलाती रही। (ग) आंतरिक
संबंध $y = s^2/8R$ ने ऊर्जा-समीकरण को $s'' = -\frac{g}{4R}s$ में बदल दिया,
जो अचर गुणांकों वाला \omterm{def:b1:linmaps:def}{रैखिक} समीकरण है और जिसके हल ठीक-ठीक समकालिक हैं। (घ)
भाग I ने \cref{ch:b1:derivative} का अवकल कलन प्रयुक्त किया, भाग II तथा III
ने \cref{ch:b1:integration} का \omterm{thm:b1:integration:def}{समाकल}, और भाग IV ने \cref{ch:b1:diffeq} के
अवकल समीकरण --- अर्थात् साइक्लॉयड इस पुस्तक का पूरा पाठ्यक्रम एक ही वक्र में
लपेटा हुआ है।
\end{solution}
