\section*{अध्याय \ref{ch:b1:arith} --- पूर्णांक अंकगणित}

\begin{solution}{exo:b1:arith:1}
$1001 = 1 \times 777 + 224$; $777 = 3 \times 224 + 105$; $224 = 2 \times 105
+ 14$; $105 = 7 \times 14 + 7$; $14 = 2 \times 7 + 0$। अतः $\gcd(1001, 777)
= 7$। उलटी दिशा में:
\[
7 = 105 - 7 \times 14
= 105 - 7(224 - 2\times 105) = 15 \times 105 - 7 \times 224
\]
\[
= 15(777 - 3\times 224) - 7\times 224 = 15 \times 777 - 52 \times 224
= 15 \times 777 - 52(1001 - 777) = 67 \times 777 - 52 \times 1001 .
\]
जाँच: $67 \times 777 = 52\,059$ और $52 \times 1001 = 52\,052$; अंतर $7$।
बेज़ू युग्म: $1001u + 777v = 7$ के लिए $(u, v) = (-52, 67)$।
\end{solution}

\begin{solution}{exo:b1:arith:2}
चूँकि $10 \equiv 1 \pmod 9$: $10^k \equiv 1$, अतः $\sum_k d_k 10^k \equiv
\sum_k d_k \pmod 9$। चूँकि $10 \equiv -1 \pmod{11}$: $10^k \equiv (-1)^k$,
अतः पूर्णांक $11$ के सापेक्ष एकांतरित योग $\sum_k (-1)^k d_k$ के सर्वांगसम
है (\emph{इकाई} अंक से चिह्न $+$ के साथ आरंभ करते हुए)।

$123\,456\,789$: अंकों का योग $45 \equiv 0 \pmod 9$। इकाई से एकांतरित योग:
$9 - 8 + 7 - 6 + 5 - 4 + 3 - 2 + 1 = 5$, अतः संख्या $\equiv 5 \pmod{11}$।
\end{solution}

\begin{solution}{exo:b1:arith:3}
यूक्लिड: $237 = 2 \times 91 + 55$; $91 = 1 \times 55 + 36$; $55 = 1 \times
36 + 19$; $36 = 1 \times 19 + 17$; $19 = 1 \times 17 + 2$; $17 = 8 \times 2
+ 1$। उलटी दिशा में:
\[
1 = 17 - 8\times 2 = 17 - 8(19 - 17) = 9\times 17 - 8\times 19
= 9(36 - 19) - 8\times 19 = 9\times 36 - 17\times 19
\]
\[
= 9\times 36 - 17(55 - 36) = 26\times 36 - 17\times 55
= 26(91 - 55) - 17\times 55 = 26\times 91 - 43\times 55
\]
\[
= 26\times 91 - 43(237 - 2\times 91) = 112 \times 91 - 43 \times 237.
\]
अतः $91 \times 112 \equiv 1 \pmod{237}$: हल $x \equiv 112 \pmod{237}$ हैं।
(जाँच: $91 \times 112 = 10\,192 = 43 \times 237 + 1$।)
\end{solution}

\begin{solution}{exo:b1:arith:4}
$\gcd(17, 39) = 1$: यूक्लिड $39 = 2\times 17 + 5$, $17 = 3\times 5 + 2$, $5
= 2\times 2 + 1$ देता है, और उलटी दिशा में
\[
1 = 5 - 2\times 2 = 5 - 2(17 - 3\times 5) = 7\times 5 - 2\times 17
= 7(39 - 2\times 17) - 2\times 17 = 7\times 39 - 16\times 17 .
\]
विशिष्ट हल $(x_0, y_0) = (-16, 7)$। समांगी समीकरण $17x + 39y = 0$ का व्यापक
हल: $x = 39k$, $y = -17k$ (क्योंकि $17 \mid 39y$ और $\gcd(17,39) = 1$ $17
\mid y$ पर बाध्य कर देते हैं --- गाउस की प्रमेयिका)। अतः
\[
(x, y) = (-16 + 39k,\; 7 - 17k), \qquad k \in \Z .
\]
दाएँ पक्ष $5$ के लिए विशिष्ट हल को $5$ से गुणा कीजिए: $(x, y) = (-80 +
39k,\; 35 - 17k)$, $k \in \Z$।
\end{solution}

\begin{solution}{exo:b1:arith:5}
प्रत्येक \omterm{def:b1:arith:prime}{अभाज्य} $p$ के लिए, $\alpha = v_p(a)$ और $\beta = v_p(b)$ के साथ:
\[
v_p\bigl(\gcd(a,b)\bigr) + v_p\bigl(\operatorname{lcm}(a,b)\bigr)
= \min(\alpha, \beta) + \max(\alpha, \beta)
= \alpha + \beta = v_p(ab) .
\]
जिन दो धनात्मक पूर्णांकों का प्रत्येक \omterm{def:b1:arith:prime}{अभाज्य} पर मूल्यांकन समान हो, वे बराबर
होते हैं (\cref{prop:b1:arith:valuation}), अतः
$\gcd(a,b)\operatorname{lcm}(a,b) = ab$।
\end{solution}

\begin{solution}{exo:b1:arith:6}
मूल्यांकन: $\gcd(a, b) = 2^{\min(10,6)} 3^{\min(4,7)} 5^{\min(2,0)}
7^{\min(0,1)} = 2^6\, 3^4 = 5184$; $\operatorname{lcm}(a,b) = 2^{10}\, 3^7\,
5^2\, 7$।

भाजकों की गणना: $n = \prod p_i^{\alpha_i}$ का धनात्मक भाजक ठीक ऐसा चयन
$\prod p_i^{\beta_i}$ है कि $0 \leq \beta_i \leq \alpha_i$
(\cref{prop:b1:arith:valuation}); चयन स्वतंत्र हैं, अतः $\prod_i (\alpha_i +
1)$ भाजक हैं। $a$ के लिए: $(10+1)(4+1)(2+1) = 165$।
\end{solution}

\begin{solution}{exo:b1:arith:7}
मान लीजिए $r, q \in \N^*$ के साथ $\sqrt p = \frac rq$, अर्थात् $p q^2 =
r^2$। $v_p$ लगाइए: $v_p(pq^2) = 1 + 2v_p(q)$ विषम है, जबकि $v_p(r^2) = 2
v_p(r)$ सम। कोई पूर्णांक एक ही साथ विषम और सम $p$-मूल्यांकन नहीं रख सकता:
विरोधाभास। अतः $\sqrt p \notin \Q$।
\end{solution}

\begin{solution}{exo:b1:arith:8}
\emph{व्यापक तथ्य।} $\gcd(m,n) = 1$ के साथ बेज़ू $mu + nv = 1$ देता है। $x_0
= b\,mu + a\,nv$ रखिए। तब $x_0 \equiv a\,nv \equiv a(1 - mu) \equiv a \pmod
m$ और इसी प्रकार $x_0 \equiv b \pmod n$: अस्तित्व। यदि $x$ और $x'$ दो हल
हैं, तो $m$ और $n$ $x - x'$ को विभाजित करते हैं, अतः $mn \mid x - x'$
(\cref{thm:b1:arith:gauss}~(2)): $mn$ के सापेक्ष अद्वितीयता।

\emph{संख्यात्मक रूप से:} $m = 7$, $n = 11$: $7 \times (-3) + 11 \times 2 =
1$। अतः $x_0 = 5 \times 7 \times (-3) + 2 \times 11 \times 2 = -105 + 44 =
-61 \equiv 16 \pmod{77}$। जाँच: $16 = 2\times 7 + 2 \equiv 2 \pmod 7$; $16 =
11 + 5 \equiv 5 \pmod{11}$। हल: $x \equiv 16 \pmod{77}$।
\end{solution}

\begin{solution}{exo:b1:arith:9}
$7$ के सापेक्ष: फर्मा $3^6 \equiv 1$ देता है, और $1000 = 6 \times 166 + 4$,
अतः $3^{1000} \equiv 3^4 = 81 \equiv 4 \pmod 7$।

$7^{100}$ के अंतिम दो अंक: $4$ के सापेक्ष और $25$ के सापेक्ष काम कीजिए। $4$
के सापेक्ष: $7 \equiv -1$, अतः $7^{100} \equiv 1$। $25$ के सापेक्ष: $7^2 =
49 \equiv -1$, अतः $7^4 \equiv 1$ और $7^{100} = (7^4)^{25} \equiv 1$। चीनी
शेषफल प्रमेय (\cref{exo:b1:arith:8}) से $7^{100} \equiv 1 \pmod{100}$: अंतिम
दो अंक $01$ हैं।
\end{solution}

\begin{solution}{exo:b1:arith:10}
$m = nq + r$, $0 \leq r < n$ लिखिए। तब
\[
2^m - 1 = 2^r\bigl(2^{nq} - 1\bigr) + 2^r - 1,
\]
और $2^n - 1$ $2^{nq} - 1 = (2^n - 1)(2^{n(q-1)} + \dots + 1)$ को \omterm{def:b1:arith:divides}{विभाजित
करता है}। अतः $2^n - 1$ के सापेक्ष $\;2^m - 1 \equiv 2^r - 1$, और चूँकि $0
\leq 2^r - 1 < 2^n - 1$, यही यूक्लिडीय शेषफल \emph{है}।

अतः युग्म $(2^m - 1, 2^n - 1)$ पर \omterm{met:b1:arith:euclid}{यूक्लिडीय कलनविधि}, घातांक-दर-घातांक, $(m,
n)$ पर की कलनविधि का दर्पण है: हर भाग-पद ऊपर $(m, n)$ के स्थान पर $(n, r)$
और नीचे $(2^m - 1, 2^n - 1)$ के स्थान पर $(2^n - 1, 2^r - 1)$ रख देता है।
ऊपर की कलनविधि $\gcd(m,n)$ पर समाप्त होती है, अतः नीचे वह $2^{\gcd(m,n)} -
1$ पर समाप्त होती है।
\end{solution}

\begin{solution}{exo:b1:arith:11}
पहले $x^2 \equiv 1 \pmod p$ हल कीजिए: $p \mid (x-1)(x+1)$, अतः यूक्लिड की
प्रमेयिका से $x \equiv 1$ या $x \equiv -1 \pmod p$।

गुणनफल $(p-1)! = 1 \times 2 \times \dots \times (p-1)$ में प्रत्येक गुणनखंड
$a$ $p$ के सापेक्ष व्युत्क्रमणीय है, और उसका प्रतिलोम $a^{-1}$ फिर उन्हीं
गुणनखंडों में से एक है (\cref{prop:b1:arith:invmod})। प्रत्येक $a$ को
$a^{-1}$ के साथ युग्मित कीजिए: युग्मों का गुणनफल $1$ होता है, केवल स्वयं से
युग्मित होने वाले गुणनखंड ($a = a^{-1}$, अर्थात् $a^2 \equiv 1$) अकेले खड़े
रहते हैं --- और वे ठीक $1$ तथा $p - 1$ हैं। अतः
\[
(p-1)! \equiv 1 \times (p - 1) \equiv -1 \pmod p .
\]
($p = 2$ के लिए: $1! = 1 \equiv -1 \pmod 2$; युग्मन का तर्क अपकर्षित हो जाता
है पर परिणाम सत्य रहता है।)

\emph{विलोम।} मान लीजिए $n \geq 2$ भाज्य है, $1 < a \leq b < n$ के साथ $n =
ab$। यदि $a < b$, तो दोनों $(n-1)!$ के भिन्न गुणनखंडों के रूप में आते हैं,
अतः $n \mid (n-1)!$ और $(n-1)! \equiv 0 \not\equiv -1$। यदि $a = b$ (अर्थात्
$n = a^2$): $a \geq 3$ के लिए $a$ और $2a$ दोनों $< n$ हैं, अतः $n = a^2 \mid
a \times 2a \mid (n-1)!$, वही निष्कर्ष; और $n = 4$ के लिए $(n-1)! = 6 \equiv
2 \not\equiv -1 \pmod 4$।
\end{solution}

\begin{solution}{exo:b1:arith:12}
\begin{enumerate}
  \item आगमन। $n = 1$ के लिए: $F_0 = 3 = F_1 - 2 = 5 - 2$। $F_0\cdots
  F_{n-1} = F_n - 2$ मानने पर:
        \[
        F_0 \cdots F_n = (F_n - 2)F_n
        = \bigl(2^{2^n} - 1\bigr)\bigl(2^{2^n} + 1\bigr)
        = 2^{2^{n+1}} - 1 = F_{n+1} - 2 .
        \]
  \item मान लीजिए $m < n$ और $d = \gcd(F_m, F_n)$। (1) से $F_m$ $F_n - 2$ को
  \omterm{def:b1:arith:divides}{विभाजित करता है}, अतः $d$ $F_n$ और $F_n - 2$ दोनों को \omterm{def:b1:arith:divides}{विभाजित करता है},
  इसलिए $2$ को भी। पर हर फर्मा संख्या विषम है, अतः $d = 1$।
  \item प्रत्येक $F_n \geq 3$ का कोई \omterm{def:b1:arith:prime}{अभाज्य} भाजक $p_n$ है
  (\cref{thm:b1:arith:euclidprimes} का पहला पद)। यदि $m \neq n$, तो $p_m
  \neq p_n$, क्योंकि कोई उभयनिष्ठ \omterm{def:b1:arith:prime}{अभाज्य} $\gcd(F_m, F_n) = 1$ को विभाजित
  करता। अतः \omterm{def:b1:logic:map}{प्रतिचित्रण} $n \mapsto p_n$ $\N$ से अभाज्यों में \omterm{def:b1:logic:inj}{एकैकी} है:
  \omterm{def:b1:arith:prime}{अभाज्य} संख्याएँ अनंत हैं।
\end{enumerate}
\end{solution}

\begin{solution}{pb:b1:arith:1}
\textbf{1.} $10! = 3\,628\,800$: अंत में दो शून्य। गुणनखंड-दर-गुणनखंड
मूल्यांकन: $2$ की घातें $2, 4 = 2^2, 6, 8 = 2^3, 10$ से आती हैं, कुल
$v_2(10!) = 1 + 2 + 1 + 3 + 1 = 8$; $5$ की घातें $5$ और $10$ से: $v_5(10!) =
2$। अंत के शून्य $= \min(v_2, v_5) = 2$, जो संगत है।

\textbf{2.} यूक्लिडीय भाग $\lfloor x\rfloor = nq + r$, $0 \leq r \leq n - 1$
लिखिए। तब $0 \leq r + \{x\} < n$ के साथ $x = nq + r + \{x\}$, अतः $\lfloor
x/n \rfloor = q = \bigl\lfloor \lfloor x \rfloor / n \bigr\rfloor$।

\textbf{3.} मान लीजिए $\alpha = v_p(a) \leq \beta = v_p(b)$ (आवश्यकता हो तो
अदला-बदली कीजिए) और $p \nmid a', b'$ के साथ $a = p^\alpha a'$, $b = p^\beta
b'$ लिखिए। तब $a + b = p^\alpha\bigl(a' + p^{\beta - \alpha}b'\bigr)$, अतः
$v_p(a + b) \geq \alpha = \min$। यदि $\alpha < \beta$, तो कोष्ठक $a' +
p^{\beta-\alpha}b' \equiv a' \not\equiv 0 \pmod p$ है: मूल्यांकन ठीक
$\alpha$ है।

\textbf{4.} $\intint1n$ में $m$ के गुणज $m, 2m, \dots, qm$ हैं, जहाँ $q$ वह
बृहत्तम पूर्णांक है जिसके लिए $qm \leq n$, अर्थात् $q = \lfloor n/m
\rfloor$।

\textbf{5.} अद्वितीय गुणनखंडन से $v_p(n!) = \sum_{j=1}^{n} v_p(j)$। दूसरी
तरह गिनिए: प्रत्येक $j$ $v_p(j) = \#\{k \geq 1 : p^k \mid j\}$ का योगदान
करता है, अतः
\[
v_p(n!) = \sum_{j=1}^n \#\{k : p^k \mid j\}
= \sum_{k\geq1} \#\{j \leq n : p^k \mid j\}
= \sum_{k\geq1} \Bigl\lfloor \frac n{p^k} \Bigr\rfloor
\]
प्रश्न 4 से --- यही लजांद्र का सूत्र है। योग परिमित है: $p^k > n$ वाले पद
लुप्त हो जाते हैं।

\textbf{6.} $v_5(1000!) = 200 + 40 + 8 + 1 = 249$ ($5, 25, 125, 625$ से
भाग); $v_2(1000!) = 500 + 250 + 125 + 62 + 31 + 15 + 7 + 3 + 1 = 994$।
$1000!$ के अंत के शून्य: हर शून्य एक $2$ और एक $5$ खाता है, अतः उनकी संख्या
$\min(994, 249) = 249$ है।

\textbf{7.} $n = \sum_i a_ip^i$ के साथ प्रश्न 2 $\lfloor n/p^k \rfloor =
\sum_{i \geq k} a_ip^{i-k}$ देता है (आधार-$p$ प्रसार को काट दीजिए)। $k \geq
1$ पर योग करके और दोनों परिमित योगों की अदला-बदली करके:
\[
v_p(n!) = \sum_{i\geq1} a_i \sum_{k=1}^{i} p^{i-k}
= \sum_{i\geq0} a_i\,\frac{p^i - 1}{p - 1}
= \frac{n - s_p(n)}{p - 1} .
\]

\textbf{8.} $p = 2$ के लिए: $v_2(n!) = n - s_2(n)$। चूँकि $n \geq 1$ में
$s_2(n) \geq 1$ है, सदा $v_2(n!) \leq n - 1 < n$: $2^n \nmid n!$। और
$v_2(n!) = n - 1$ तभी जब $s_2(n) = 1$, अर्थात् तभी जब $n$ $2$ की घात हो।

\textbf{9.} $n$ के $\lfloor \log_p n \rfloor + 1$ आधार-$p$ अंक हैं, जिनमें
से हर अधिक से अधिक $p - 1$ है, अतः $1 \leq s_p(n) \leq (p-1)\bigl(\log_p(n)
+ 1\bigr)$। प्रश्न 7 में प्रतिस्थापित करने पर:
\[
\frac n{p-1} - \log_p(n) - 1 \;\leq\; v_p(n!) \;<\; \frac n{p-1},
\]
और $n$ से भाग देने पर: $\frac{v_p(n!)}n \to \frac1{p-1}$।

\textbf{10.} $Z(n) - Z(n-1) = v_5(n!/(n-1)!) = v_5(n)$: अंत के शून्यों की
गणना $5$ के प्रत्येक गुणज पर $v_5(n)$ से कूदती है और बीच में अचर रहती है।
$Z(24) = \lfloor24/5\rfloor = 4$ और $Z(25) = 5 + 1 = 6$: $n = 25$ पर गणना
$4$ से सीधे $6$ पर कूद जाती है ($v_5(25) = 2$), और चूँकि $Z$ अह्रासमान है और
पहले $Z \leq 4$ तथा बाद में $Z \geq 6$ है, मान $5$ कभी प्राप्त नहीं होता:
किसी क्रमगुणित के अंत में ठीक पाँच शून्य नहीं होते।

\textbf{11.} $x = \lfloor x\rfloor + \{x\}$ लिखिए: $\lfloor x + y\rfloor =
\lfloor x\rfloor + \lfloor y\rfloor + \lfloor \{x\} + \{y\}\rfloor$, और $0
\leq \{x\} + \{y\} < 2$ अंतिम फ़र्श को $0$ या $1$ बना देता है। फिर लजांद्र
को तीन बार लगाने पर,
\[
v_p\binom{m+n}m = v_p\bigl((m{+}n)!\bigr) - v_p(m!) - v_p(n!)
= \sum_{k\geq1}\Bigl(
\Bigl\lfloor\frac{m+n}{p^k}\Bigr\rfloor
- \Bigl\lfloor\frac{m}{p^k}\Bigr\rfloor
- \Bigl\lfloor\frac{n}{p^k}\Bigr\rfloor\Bigr),
\]
जो $0$ और $1$ का परिमित योग है (पहला दावा $x = m/p^k$, $y = n/p^k$ पर
लगाइए)।

\textbf{12.} $k \geq 1$ नियत कीजिए और $0 \leq m_0, n_0 < p^k$ के साथ $m =
p^km_1 + m_0$, $n = p^kn_1 + n_0$ लिखिए (यूक्लिडीय भाग: $m_0$ $m$ के $k$
निचले अंकों से बनी संख्या है)। तब
\[
\Bigl\lfloor\frac{m+n}{p^k}\Bigr\rfloor
- \Bigl\lfloor\frac m{p^k}\Bigr\rfloor
- \Bigl\lfloor\frac n{p^k}\Bigr\rfloor
= \Bigl\lfloor\frac{m_0 + n_0}{p^k}\Bigr\rfloor ,
\]
जो $m_0 + n_0 \geq p^k$ होने पर $1$ है और अन्यथा $0$। पर $m_0 + n_0 \geq
p^k$ ठीक यही कहता है कि $m$ और $n$ के $k$ निचले अंकों को जोड़ने पर स्थान $k$
में उफान आता है --- अर्थात् विद्यालय की जोड़-विधि में स्थान $k$ में एक
हासिल। $k$ पर योग करने पर: $v_p\binom{m+n}m$ आधार-$p$ जोड़ $m + n$ में
हासिलों की संख्या है। (कुमर, 1852।)

\textbf{13.} कुमर को $m = j$, $n = p^k - j$, योग $p^k =
(1\underbrace{0\cdots0}_{k})_p$ पर लगाइए। मान लीजिए $a = v_p(j)$, अतः $j$ के
स्थानों $0, \dots, a-1$ पर आधार-$p$ अंक $0$ हैं और स्थान $a$ पर अंक अशून्य
है। $p^k - j$ के स्थान $a$ से नीचे के अंक भी $0$ हैं ($p^k - j = p^a(p^{k-a}
- j/p^a)$)। स्थान $a$ पर दोनों अशून्य अंकों का योग $p$ होना ही चाहिए
(परिणामी अंक $0$): एक हासिल; और $a+1, \dots, k-1$ के प्रत्येक स्थान पर अंक
तथा आता हुआ हासिल मिलकर $p$ बनते हैं (परिणामी अंक फिर $0$): हासिल आगे बढ़ता
जाता है। कुल: $k - a$ हासिल, अतः $v_p\binom{p^k}j = k - v_p(j)$। $k = 1$ के
लिए: $0 < j < p$ के लिए $v_p\binom pj = 1$, अर्थात् वही \omterm{def:b1:arith:divides}{विभाज्यता} जो
\cref{thm:b1:arith:fermat} में प्रयुक्त हुई।

\textbf{14.} अंकीय रूप (प्रश्न 7) से, $s_2(2n) = s_2(n)$ का प्रयोग करते हुए
(एक शून्य अंक जोड़कर):
\[
v_2\binom{2n}n = \bigl(2n - s_2(2n)\bigr) - 2\bigl(n -
s_2(n)\bigr) = 2s_2(n) - s_2(2n) = s_2(n) \geq 1 :
\]
$\binom{2n}n$ सदा सम है, और $v_2 = 1$ (अर्थात् $\binom{2n}n \equiv 2 \pmod
4$) ठीक तब जब $s_2(n) = 1$, अर्थात् जब $n$ $2$ की घात हो।

\textbf{15.} $m = n = k = p$ के साथ वांडरमोंड: $\binom{2p}p = \sum_{j=0}^p
\binom pj\binom p{p-j} = \sum_{j=0}^p \binom pj^2$। $0 < j < p$ के लिए $p
\mid \binom pj$ (प्रश्न 13), अतः $\binom pj^2 \equiv 0 \pmod p$; और सिरों के
पद $1 + 1$ देते हैं: $\binom{2p}p \equiv 2 \pmod p$।

\textbf{16.} आधार $3$: $500 = 486 + 9 + 3 + 2$, अंक (निम्न से उच्च) $(2, 1,
1, 0, 0, 2)$, अतः $s_3(500) = 6$; और $1000 = 729 + 243 + 27 + 1$, अंक $(1,
0, 0, 1, 0, 1, 1)$, अतः $s_3(1000) = 4$। \emph{कुमर:} आधार $3$ में $500 +
500$ जोड़िए: स्थान $0$: $2 + 2 = 4$, अंक $1$ हासिल $1$; स्थान $1$: $1 + 1 +
1 = 3$, अंक $0$ हासिल $1$; स्थान $2$: $1 + 1 + 1 = 3$, अंक $0$ हासिल $1$;
स्थान $3$: $0 + 0 + 1 = 1$, कोई हासिल नहीं; स्थान $4$: $0$; स्थान $5$: $2 +
2 = 4$, अंक $1$ हासिल $1$; स्थान $6$: हासिल आ गिरता है: अंक $1$। चार हासिल:
$v_3\binom{1000}{500} = 4$। \emph{लजांद्र:} $v_3(1000!) = \frac{1000 - 4}2 =
498$ और $v_3(500!) = \frac{500 - 6}2 = 247$, अतः $v_3\binom{1000}{500} = 498
- 2\times247 = 4$। दोनों संगणनाएँ मेल खाती हैं --- और जोड़ के अंक $(1, 0, 0,
1, 0, 1, 1)$ $1000$ को फिर से बना देते हैं, जैसा होना ही चाहिए।

\textbf{17.} कुमर ($p = 2$, $m = k$, $n' = n - k$) से: $\binom nk$ विषम है
तभी जब आधार $2$ में जोड़ $k + (n - k)$ में कोई हासिल न हो, अर्थात् तभी जब हर
स्थान पर अंक $k_i + (n - k)_i = n_i$ को संतुष्ट करें; उस स्थिति में सभी $i$
के लिए $k_i \leq n_i$। विलोमतः, यदि सभी $i$ के लिए $k_i \leq n_i$, तो अंकों
$n_i - k_i$ वाली संख्या $n - k$ है और जोड़ हासिल-रहित है। आधार $p$ में वही
उपपत्ति: $p \nmid \binom nk$ तभी जब $k$ का प्रत्येक आधार-$p$ अंक $n$ के
तदनुरूप अंक से अधिक न हो।

\textbf{18.} ऐसे $k \in \intint0n$ गिनिए जिनके अंक $k_i \leq n_i$ का पालन
करते हों: $k$ का हर अंक स्वतंत्र रूप से $n_i + 1$ मानों में से चुना जाता है,
जिससे $\prod_i (n_i + 1)$ चयन मिलते हैं; आधार $2$ में यह $2^{\#\{i : n_i =
1\}} = 2^{s_2(n)}$ है। पंक्ति $4 = (100)_2$: $2^1 = 2$ विषम प्रविष्टियाँ ---
और सचमुच $1, 4, 6, 4, 1$ की विषम प्रविष्टियाँ केवल सिरों पर हैं। पंक्ति $5 =
(101)_2$: $2^2 = 4$ --- और सचमुच $1, 5, 10, 10, 5, 1$।

\textbf{19.} सभी भीतरी प्रविष्टियाँ सम हैं $\iff$ पंक्ति में ठीक $2$ विषम
प्रविष्टियाँ हैं (दोनों सिरे सदा विषम होते हैं) $\iff 2^{s_2(n)} = 2 \iff
s_2(n) = 1 \iff n$ $2$ की घात है।

\textbf{20.} प्रश्न 11 के योग में $k$-वाँ पद तभी लुप्त हो जाता है जब $p^k >
m + n$ (तब तीनों फ़र्श बराबर हैं; वस्तुतः $p^k > m+n$ होने पर पहला $0$ है;
और सरलता से, हर पद $0$ है)। अतः अधिक से अधिक $\lfloor \log_p(m+n)\rfloor$ पद
अशून्य हैं, और हर एक $1$ का है: $a = v_p\binom{m+n}m \leq \log_p(m+n)$,
अर्थात् $p^a \leq m + n$।

\textbf{21.} प्रत्येक \omterm{def:b1:arith:prime}{अभाज्य} $p$ के लिए $v_p\bigl(\operatorname{lcm}(1,
\dots, 2n)\bigr) = \lfloor\log_p(2n)\rfloor$ ($2n$ से अधिक न होने वाली $p$
की बृहत्तम घात $1, \dots, 2n$ में आती है)। $m = n$ के साथ प्रश्न 20 प्रत्येक
$p$ के लिए $v_p\binom{2n}n \leq \lfloor\log_p(2n)\rfloor$ देता है:
\cref{prop:b1:arith:valuation} से $\binom{2n}n \mid \operatorname{lcm}(1,
\dots, 2n)$। आकार के लिए: अनुपात $\binom{2n}{k+1}/\binom{2n}k =
\frac{2n-k}{k+1} \geq 1$ ठीक $k < n$ के लिए है, अतः पंक्ति $2n$ की $2n + 1$
प्रविष्टियों में केंद्रीय सबसे बड़ी है, जिससे $4^n = \sum_k \binom{2n}k \leq
(2n+1)\binom{2n}n$। मिलाकर:
\[
\operatorname{lcm}(1, \dots, 2n) \geq \binom{2n}n \geq
\frac{4^n}{2n + 1} .
\]
यदि $2n$ से नीचे \omterm{def:b1:arith:prime}{अभाज्य} कम होते, तो लघुत्तम समापवर्त्य इतना बड़ा नहीं हो
सकता था: लघुत्तम समापवर्त्य की चरघातांकी वृद्धि अभाज्यों की प्रचुरता का
परिमाणात्मक चिह्न है।

\textbf{22.} $Z(n) = \sum_k\lfloor n/5^k\rfloor \approx \frac n4$, अतः $n =
4 \times 2026 = 8104$ के आसपास निशाना लगाइए: $Z(8104) = 1620 + 324 + 64 + 12
+ 2 = 2022$। $5$ के गुणजों से आगे बढ़िए: $Z(8110) = 2024$, $Z(8115) = 2025$,
और
\[
Z(8120) = 1624 + 324 + 64 + 12 + 2 = 2026 .
\]
चूँकि $Z$ $5$ और $Z(8119) = Z(8115) = 2025$ के गुणजों के बीच अचर है, कम से
कम $2026$ अंत के शून्यों वाला लघुतम $n$ $n = 8120$ है।

\textbf{23.} आधार $7$: $50 = 49 + 1$, अंक (निम्न से उच्च) $(1, 0, 1)$। $50 +
50$ जोड़ने पर: स्थान $0$: $1 + 1 = 2 < 7$, कोई हासिल नहीं; स्थान $1$: $0 + 0
= 0$; स्थान $2$: $1 + 1 = 2 < 7$, कोई हासिल नहीं। हासिल-रहित, अतः कुमर से
$v_7\binom{100}{50} = 0$: $7 \nmid \binom{100}{50}$। लजांद्र भी यही कहता है:
$v_7(100!) = \lfloor 100/7 \rfloor + \lfloor 100/49 \rfloor = 14 + 2 = 16$
और $v_7(50!) = 7 + 1 = 8$, अतः $v_7\binom{100}{50} = 16 - 2\times8 = 0$।

\textbf{24.} (क) अद्वितीय गुणनखंडन $v_p$ की परिभाषा और उसकी योगात्मकता के
मूल में है, अतः लजांद्र के सूत्र और हर विभाज्यता-निष्कर्ष के भी
(\cref{prop:b1:arith:valuation})। (ख) यूक्लिडीय भाग ने प्रश्न 2 की
कटाई-सर्वसमिका दी और वह \omterm{thm:b1:logic:partition}{विभाजन} $m = p^km_1 + m_0$ दिया जो हासिल को अलग कर
देता है (प्रश्न 12)। (ग) गणना: $m$ के गुणजों की गणना (प्रश्न 4), अंक-चयन का
गुणनफल (प्रश्न 18), और पंक्ति-योग का परिबंध $4^n \leq (2n+1)\binom{2n}n$
(प्रश्न 21) --- ये सब \cref{ch:b1:counting} की शैली के तर्क हैं।

\textbf{25.} लजांद्र ``$n!$ को $p$ की कौन-सी घात विभाजित करती है'' को
आधार-$p$ के अंक-अंकगणित में बदल देता है; कुमर द्विपद गुणांकों के लिए उत्तर
को एक ही जोड़ के हासिलों में सिकोड़ देता है --- \omterm{def:b1:arith:divides}{विभाज्यता}, जो विशाल संख्याओं
का समग्र गुणधर्म लगती है, स्थानीय रूप से, अंक-दर-अंक, पढ़ ली जाती है। प्रश्न
16 इसका आदर्श है: हाथ से गिने हुए चार हासिल सैकड़ों अंकों वाली संख्या में
$3$ की ठीक-ठीक घात निर्धारित कर देते हैं। और प्रश्न 21 दिखाता है कि यही
विचार-मंडल गहरे जल को छू रहा है: $\operatorname{lcm}(1, \dots, 2n)$ के लिए
चरघातांकी निम्न परिबंध \omterm{def:b1:arith:prime}{अभाज्य संख्या} प्रमेय की दिशा में पहला, पूर्णतः
प्रारंभिक पद है, जिसकी उपपत्ति इस खंड से बहुत परे है। पूरा औज़ार-संदूक ---
भाग, \omterm{thm:b1:arith:gcd}{महत्तम समापवर्तक}, मूल्यांकन --- \cref{ch:b1:poly} में बहुपदों के लिए
फिर चलाया जाता है, जहाँ अंक-प्रसार का अनुरूप $(X - a)$ की घातों में प्रसार
है।
