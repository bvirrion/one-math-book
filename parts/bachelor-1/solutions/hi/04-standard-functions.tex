\begin{solution}{exo:b1:functions:1}
$\arcsin\frac{\sqrt 3}{2} = \frac\pi3$;
$\;\arccos\bigl(-\frac12\bigr) = \frac{2\pi}{3}$;
$\;\arctan(-1) = -\frac\pi4$.

\begin{solution}{exo:b1:functions:2}
$f(x) = \arcsin(2x - 1)$ के लिए $-1 \leq 2x - 1 \leq 1$ आवश्यक है, अर्थात्
$x \in \intcc{0}{1}$। $\intoo{0}{1}$ पर शृंखला नियम और
\cref{prop:b1:functions:arcderiv} देते हैं
\[
f'(x) = \frac{2}{\sqrt{1 - (2x-1)^2}} = \frac{2}{\sqrt{4x - 4x^2}}
= \frac{1}{\sqrt{x(1 - x)}} .
\]
$g(x) = \arctan\sqrt x$ $x \geq 0$ के लिए परिभाषित है, और $x > 0$ के लिए:
\[
g'(x) = \frac{1}{1 + x} \cdot \frac{1}{2\sqrt x}
= \frac{1}{2\sqrt x\,(1 + x)} .
\]
\end{solution}

\begin{solution}{exo:b1:functions:3}
$\cosh x + \sinh x = \frac{\eu^x + \eu^{-x}}{2} + \frac{\eu^x - \eu^{-x}}{2}
= \eu^x$। अतः $n \in \Z$ के लिए:
\[
(\cosh x + \sinh x)^n = (\eu^x)^n = \eu^{nx}
= \cosh nx + \sinh nx .
\]
\end{solution}

\begin{solution}{exo:b1:functions:4}
$y = \arcsin x$ के साथ: $\cos 2y = 1 - 2\sin^2 y = 1 - 2x^2$, अतः
$\cos(2\arcsin x) = 1 - 2x^2$।

$y = \arctan x$ के साथ: $\sin 2y = 2 \sin y \cos y = 2 \tan y \cos^2 y =
\frac{2\tan y}{1 + \tan^2 y}$, अतः $\sin(2\arctan x) = \dfrac{2x}{1 + x^2}$।
\end{solution}

\begin{solution}{exo:b1:functions:5}
सबसे धीमी से सबसे तेज़ तक:
\[
(\ln x)^{1000} \;\ll\; x^{100} \;\ll\; x^{\ln x} \;\ll\;
\eu^{\sqrt x} \;\ll\; \eu^{x/100} .
\]
न्यायसंगति: \cref{prop:b1:functions:powerrules} से $(\ln x)^{1000}/x^{100}
\to 0$ (लघुगणक घातों से हार जाते हैं)। $x^{100} = \eu^{100\ln x}$ और $x^{\ln
x} = \eu^{(\ln x)^2}$: चूँकि $(\ln x)^2 - 100\ln x \to +\infty$, दूसरा जीतता
है। $x^{\ln x} = \eu^{(\ln x)^2} \ll \eu^{\sqrt x}$, क्योंकि $(\ln
x)^2/\sqrt x \to 0$ (लघुगणक घात $x^{1/4}$ से हारते हैं, और वह वर्गित भी है)।
अंत में $\sqrt x - x/100 \to -\infty$, अतः $\eu^{\sqrt x} \ll \eu^{x/100}$।
\end{solution}

\begin{solution}{exo:b1:functions:6}
प्रांत: $x \neq \pm 1$, तीन अंतराल। प्रत्येक पर,
\[
f'(x) = \frac{\bigl(\frac{2x}{1-x^2}\bigr)'}{1 +
\frac{4x^2}{(1-x^2)^2}}
= \frac{\frac{2(1-x^2) + 4x^2}{(1-x^2)^2}}
{\frac{(1-x^2)^2 + 4x^2}{(1-x^2)^2}}
= \frac{2(1 + x^2)}{(1 + x^2)^2} = \frac{2}{1 + x^2}
= (2\arctan x)' .
\]
अतः $f(x) - 2\arctan x$ प्रत्येक अंतराल पर अचर है। मान: $x = 0$ पर $f(0) =
0$: अचर $\intoo{-1}{1}$ पर $0$ है। जब $x \to +\infty$, तब $\frac{2x}{1-x^2}
\to 0^-$, अतः $f \to 0$, जबकि $2\arctan x \to \pi$: अचर $\intoo{1}{+\infty}$
पर $-\pi$ है। विषमता से वह $\intoo{-\infty}{-1}$ पर $+\pi$ है। सारांश:
$\intoo{-1}{1}$ पर $f = 2\arctan x$, $x > 1$ के लिए $f = 2\arctan x - \pi$,
$x < -1$ के लिए $f = 2\arctan x + \pi$। (यह स्पर्शज्या का द्विगुण-कोण सूत्र
ही है, $\arctan$ के द्वारा पढ़ा हुआ।)
\end{solution}

\begin{solution}{exo:b1:functions:7}
$\cosh x = 2$: $u = \eu^x > 0$, $u + u^{-1} = 4$ के साथ $u^2 - 4u + 1 = 0$,
$u = 2 \pm \sqrt 3$: $x = \ln(2 + \sqrt 3)$ या $x = \ln(2 - \sqrt 3) =
-\ln(2 + \sqrt 3)$ (दोनों हल परस्पर विपरीत हैं, क्योंकि $\cosh$ सम है; दोनों
वैध हैं)। समतुल्य रूप से $x = \pm \operatorname{arcosh} 2$।

$5\cosh x - 4\sinh x = 3$: चरघातांकी परिभाषाएँ प्रतिस्थापित करने पर
$\frac{5(u + u^{-1}) - 4(u - u^{-1})}{2} = 3$, अर्थात् $u + 9u^{-1} = 6$,
अर्थात् $u^2 - 6u + 9 = (u - 3)^2 = 0$: $u = 3$, एक हल $x = \ln 3$।
\end{solution}

\begin{solution}{exo:b1:functions:8}
मान लीजिए $\alpha = \arctan\frac12 + \arctan\frac13$। योग-सूत्र:
\[
\tan\alpha = \frac{\frac12 + \frac13}{1 - \frac12\cdot\frac13}
= \frac{5/6}{5/6} = 1 .
\]
दोनों प्रतिलोम स्पर्शज्याएँ $\intoo{0}{\frac\pi4}$ में हैं (उनके प्रांतिक
$\intoo{0}{1}$ में हैं), अतः $\alpha \in \intoo{0}{\frac\pi2}$; वहाँ
स्पर्शज्या $1$ वाला एकमात्र कोण $\frac\pi4$ है।

मैकिन: मान लीजिए $\beta = \arctan\frac15$। द्विगुण कोण दो बार:
\[
\tan 2\beta = \frac{2/5}{1 - 1/25} = \frac{5}{12},
\qquad
\tan 4\beta = \frac{2 \cdot 5/12}{1 - 25/144} = \frac{120}{119}.
\]
फिर
\[
\tan\Bigl(4\beta - \frac\pi4\Bigr)
= \frac{\frac{120}{119} - 1}{1 + \frac{120}{119}}
= \frac{1/119}{239/119} = \frac{1}{239}.
\]
स्थान: $\beta < \arctan 1 = \frac\pi4$, और वस्तुतः $4\beta \in
\intoo{0}{\frac\pi2}$ के साथ $\tan 4\beta = \frac{120}{119}$ $1$ के निकट है
(क्योंकि $\beta < \frac\pi8$, जबकि $\tan\frac\pi8 = \sqrt 2 - 1 > \frac15$);
अतः $4\beta - \frac\pi4 \in \intoo{-\frac\pi4}{\frac\pi4}$, जहाँ $\arctan$
$\tan$ का प्रतिलोमन करता है: $4\beta - \frac\pi4 = \arctan\frac{1}{239}$, जो
मैकिन का सूत्र है।
\end{solution}

\begin{solution}{exo:b1:functions:9}
मान लीजिए $f(x) = x - \sin x$: $f(0) = 0$ और $f'(x) = 1 - \cos x \geq 0$,
अतः $\R_+$ पर $f \geq 0$: $\sin x \leq x$।

मान लीजिए $g(x) = \sin x - x + \frac{x^3}{6}$: $\R_+$ पर $g(0) = 0$, $g'(x)
= \cos x - 1 + \frac{x^2}{2}$, $g'(0) = 0$, $g''(x) = -\sin x + x = f(x)
\geq 0$। अतः $g'$ वर्धमान है और $g'(0) = 0$, इसलिए $g' \geq 0$, इसलिए $g$
वर्धमान है और $g(0) = 0$: $g \geq 0$, अर्थात् $\R_+$ पर $x - \frac{x^3}{6}
\leq \sin x$।

फलस्वरूप $x > 0$ के लिए $0 \leq \frac{x - \sin x}{x^3} \leq \frac 16$:
अनुपात $\frac16$ के पैमाने के पास फँसा हुआ है, और \cref{ch:b1:taylor} दिखाता
है कि उसकी सीमा ठीक $\frac 16$ है।
\end{solution}

\begin{solution}{exo:b1:functions:10}
$y = \arcsin x$ लिखिए, अतः $2x\sqrt{1 - x^2} = 2\sin y\cos y = \sin 2y$ और
$h(x) = \arcsin(\sin 2y) - 2y$।

$\abs x \leq \frac{\sqrt 2}{2}$ के लिए: $2y \in
\intcc{-\frac\pi2}{\frac\pi2}$, अतः $\arcsin(\sin 2y) = 2y$ और $h = 0$।

$x \in \intoc{\frac{\sqrt 2}{2}}{1}$ के लिए: $2y \in
\intoc{\frac\pi2}{\pi}$, और $\intcc{-\frac\pi2}{\frac\pi2}$ का वह कोण जिसकी
ज्या $\sin 2y$ है, $\pi - 2y$ है: अतः $h(x) = \pi - 4\arcsin x$। (अवकलज से
जाँच: वहाँ $h'(x) = \frac{2(1 - 2x^2)}{\abs{1 - 2x^2}\sqrt{1 - x^2}} -
\frac{2}{\sqrt{1-x^2}} = \frac{-4}{\sqrt{1 - x^2}}$, जो $-4\arcsin x$ का
अवकलज है; और $x = 1$ पर $h(1) = \arcsin 0 - 2\cdot\frac\pi2 = -\pi = \pi -
4\cdot\frac\pi2$।)

$x \in \intco{-1}{-\frac{\sqrt 2}{2}}$ के लिए, $h$ की विषमता से: $h(x) =
-\pi - 4\arcsin x$।
\end{solution}

\begin{solution}{exo:b1:functions:11}
$g = \arctan \circ \sinh$ विषम, कठोरतः वर्धमान फलनों का संयोजन है, अतः वह
विषम और कठोरतः वर्धमान है; जब $x \to +\infty$, तब $\sinh x \to +\infty$, अतः
$g(x) \to \frac\pi2$, और $g$ $\intoo{-\frac\pi2}{\frac\pi2}$ पर एकैकी
आच्छादन है (सांतत्य और मध्यवर्ती मान प्रमेय)।
\cref{prop:b1:functions:hyprules}~(1) के साथ शृंखला नियम:
\[
g'(x) = \frac{\cosh x}{1 + \sinh^2 x} = \frac{\cosh x}{\cosh^2 x}
= \frac{1}{\cosh x} .
\]
रचना से $\tan g(x) = \sinh x$। फिर, चूँकि $g(x) \in
\intoo{-\frac\pi2}{\frac\pi2}$ की कोज्या धनात्मक है,
\begin{align*}
\cos g(x) &= \frac{1}{\sqrt{1 + \tan^2 g(x)}}
= \frac{1}{\sqrt{1 + \sinh^2 x}} = \frac{1}{\cosh x},\\
\sin g(x) &= \tan g(x) \cos g(x) = \frac{\sinh x}{\cosh x} = \tanh x .
\end{align*}
\end{solution}

\begin{solution}{exo:b1:functions:12}
$u = \arctan 2$ और $v = \arctan 3$ रखिए; दोनों
$\intoo{\frac\pi4}{\frac\pi2}$ में हैं (उनके प्रांतिक $1$ से अधिक हैं), अतः
$u + v \in \intoo{\frac\pi2}{\pi}$। योग-सूत्र देता है
\[
\tan(u + v) = \frac{2 + 3}{1 - 6} = -1 ,
\]
और $\intoo{\frac\pi2}{\pi}$ का स्पर्शज्या $-1$ वाला अद्वितीय कोण
$\frac{3\pi}4$ है: अतः $\arctan 2 + \arctan 3 = \frac{3\pi}4$। (स्पर्शज्या
पर आँख मूँदकर $\arctan$ लगाने से $-\frac\pi4$ मिलता, जो $\pi$ से चूक जाता
--- स्थानीकरण ही इस संगणना को बचाता है, और ऋणात्मक हर $1 - 6$ ठीक वही संकेत
है कि योग $\intoo{-\frac\pi2}{\frac\pi2}$ से बाहर निकल चुका है।) $\arctan 1
= \frac\pi4$ जोड़ने पर:
\[
\arctan 1 + \arctan 2 + \arctan 3 = \frac\pi4 + \frac{3\pi}4 = \pi .
\]
\end{solution}

\begin{solution}{pb:b1:functions:1}
\textbf{1.} $f$ अवकलनीय फलनों का भागफल है जिसका हर $\intoo0{+\infty}$ पर
अशून्य है, और
\[
f'(t) = \frac{\frac1t \cdot t - \ln t}{t^2} = \frac{1 - \ln
t}{t^2} ,
\]
जो $t < \eu$ के लिए धनात्मक, $t = \eu$ पर शून्य और $t > \eu$ के लिए ऋणात्मक
है: $f$ $\intoc0\eu$ पर कठोरतः बढ़ता है, $\intco\eu{+\infty}$ पर कठोरतः घटता
है, और उसका अधिकतम $f(\eu) = \frac1\eu$ है।

\textbf{2.} जब $t \to 0^+$: $\ln t \to -\infty$ और $\frac1t \to +\infty$,
अतः $f(t) \to -\infty$। जब $t \to +\infty$:
\cref{prop:b1:functions:powerrules} की \omterm{prop:b1:functions:powerrules}{वृद्धि तुलना} से $f(t) \to 0$ ($\beta
= \alpha = 1$)। चिह्न: वही जो $\ln t$ का है, अतः $\intoo01$ पर $f < 0$,
$f(1) = 0$, और $\intoo1{+\infty}$ पर $f > 0$। आलेख $-\infty$ से चढ़ता है,
$1$ पर शून्य काटता है, $(\eu, \frac1\eu)$ पर शिखर छूता है, फिर $0^+$ तक क्षय
होता है।

\textbf{3.} $f(4) = \frac{\ln 4}4 = \frac{2\ln 2}4 = \frac{\ln 2}2 = f(2)$।

\textbf{4.} विचरण सारणी और मध्यवर्ती मान प्रमेय से (जो उच्चतर माध्यमिक स्तर
पर प्रयुक्त है; \cref{ch:b1:continuity} में औपचारिक)। $c < 0$ के लिए: हल
केवल वहीं हैं जहाँ $f < 0$, अर्थात् $\intoo01$ में, जहाँ $f$
$\intoo{-\infty}0$ पर कठोरतः वर्धमान एकैकी आच्छादन है: ठीक एक। $c = 0$ के
लिए: केवल $t = 1$। $0 < c < \frac1\eu$ के लिए: $\intoo1\eu$ पर $f$ $0$ से
$\frac1\eu$ तक बढ़ता है: एक हल; $\intoo\eu{+\infty}$ पर $f$ $\frac1\eu$ से
$0$ तक घटता है: एक और; कुल दो। $c = \frac1\eu$ के लिए: केवल अधिकतम बिंदु $t
= \eu$। $c > \frac1\eu$ के लिए: कोई नहीं।

\textbf{5.} $x \neq \eu$ के लिए कठोर अधिकतमता $f(x) < f(\eu)$ देती है,
अर्थात् $\frac{\ln x}x < \frac1\eu$, अर्थात् $\eu\ln x < x$, अर्थात्
$\ln(x^\eu) < \ln(\eu^x)$: $x^\eu < \eu^x$। $x = \pi$ के साथ: $\eu^\pi >
\pi^\eu$ (संख्यात्मक रूप से $23.14 > 22.46$)।

\textbf{6.} $x, y > 0$ के लिए दोनों पक्ष धनात्मक हैं, अतः
\[
x^y = y^x \iff y\ln x = x\ln y \iff \frac{\ln x}x = \frac{\ln y}y
\iff f(x) = f(y) ,
\]
$xy > 0$ से भाग देकर।

\textbf{7.} मान लीजिए $f(x) = f(y) = c$, जहाँ $x \neq y$। यदि $c \leq 0$ या
$c = \frac1\eu$, तो प्रश्न 4 कहता है कि समीकरण $f = c$ का एक ही हल है:
असंभव। अतः $0 < c < \frac1\eu$, और फिर प्रश्न 4 से दोनों हलों में एक
$\intoo1\eu$ का बिंदु है और एक $\intoo\eu{+\infty}$ का: दोनों निर्देशांक $1$
से बड़े हैं और वे $\eu$ के दोनों ओर पड़ते हैं।

\textbf{8.} $y\ln x = x\ln y$ में $y = tx$ प्रतिस्थापित करने पर:
\[
tx\ln x = x(\ln t + \ln x)
\iff (t - 1)\ln x = \ln t
\iff \ln x = \frac{\ln t}{t - 1} ,
\]
अतः $x = t^{1/(t-1)}$ और $y = tx = t^{1 + \frac1{t-1}} = t^{t/(t-1)}$।
विलोमतः, इन मानों के लिए $\ln y = \frac{t\ln t}{t-1} = t\ln x$ और $y = tx$,
अतः $\frac{\ln y}y = \frac{t\ln x}{tx} = \frac{\ln x}x$: एक हल। प्रत्येक
अतुच्छ हल का कोई अनुपात $t = y/x \in \intoo0{+\infty} \setminus \{1\}$ होता
है, अतः प्राचलन पूरा है।

\textbf{9.} $t = 2$: $x = 2^{1/1} = 2$, $y = 2^{2/1} = 4$। और
\[
x(1/t) = (1/t)^{\frac1{\frac1t - 1}}
= (t^{-1})^{\frac{t}{1 - t}}
= t^{\frac{t}{t - 1}} = y(t) ,
\]
फिर $y(1/t) = \frac1t\,x(1/t) = \frac{y(t)}t = x(t)$: प्राचल-परिवर्तन $t
\mapsto 1/t$ निर्देशांकों की अदला-बदली कर देता है, जैसा समीकरण की सममिति
माँगती है।

\textbf{10.} जब $t \to 1$: $\frac{\ln t}{t-1} \to 1$ (वह $1$ पर $\ln$ का
अंतर-भागफल है), अतः $x \to \eu^1 = \eu$ और $y = tx \to \eu$। जब $t \to
+\infty$: $\ln x = \frac{\ln t}{t-1} \to 0$, अतः $x \to 1$, जबकि $\ln y =
\frac{t\ln t}{t-1} \to +\infty$, अतः $y \to +\infty$। जब $t \to 0^+$: $\ln x
= \frac{\ln t}{t-1} \to \frac{-\infty}{-1} = +\infty$, अतः $x \to +\infty$,
जबकि $\ln y = \frac{t\ln t}{t-1} \to \frac{0}{-1} = 0$, अतः $y \to 1$ ($t\ln
t \to 0$, \cref{ex:b1:functions:xx} का प्रयोग करते हुए)। अतः शाखा
अनंतस्पर्शी $y = 1$ (दूर दाएँ) से चलकर विकर्ण पर $(\eu, \eu)$ से होती हुई
ऊपर जाती है और अनंतस्पर्शी $x = 1$ (दूर ऊपर) के साथ निकल जाती है --- प्रश्न
9 से यह विकर्ण के सापेक्ष सममित है।

\textbf{11.} $\ln x(t) = g(t) = \frac{\ln t}{t-1}$, और
\[
g'(t) = \frac{\frac{t-1}t - \ln t}{(t-1)^2}
= \frac{h(t)}{(t-1)^2},
\qquad h(t) = 1 - \frac1t - \ln t .
\]
$t > 1$ के लिए $h(1) = 0$ और $h'(t) = \frac1{t^2} - \frac1t = \frac{1 -
t}{t^2} < 0$: $\intoo1{+\infty}$ पर $h < 0$, अतः $g' < 0$ और $x = \eu^g$
वहाँ कठोरतः ह्रासमान है (प्रश्न 10 से $\eu$ से $1$ तक)।

\textbf{12.} प्रश्न 11 से $t \mapsto x(t)$ $\intoo1{+\infty}$ का
$\intoo1\eu$ पर कठोरतः ह्रासमान एकैकी आच्छादन है; उसके प्रतिलोम को $t(x)$
लिखिए (वह भी कठोरतः ह्रासमान है) और $\varphi(x) = y(t(x))$ रखिए। ध्यान दीजिए
कि $\intoo01$ पर भी $h < 0$ (वहाँ $h' > 0$ और $h(1) = 0$), अतः $\intoo01$ पर
भी $x(\cdot)$ ह्रासमान है; इसलिए $y(t) = x(1/t)$ (प्रश्न 9)
$\intoo1{+\infty}$ पर $t$ में \emph{वर्धमान} है, $\eu$ से $+\infty$ तक।
संयोजन करने पर: $\varphi = y \circ t(\cdot)$ $\intoo1\eu$ से
$\intoo\eu{+\infty}$ पर कठोरतः ह्रासमान है, और प्रश्न 8 से $x^{\varphi(x)} =
\varphi(x)^x$। अंत में, किसी उभयनिष्ठ मान $c = f(x) \in \intoo0{\frac1\eu}$
के लिए युग्म $\{x, \varphi(x)\}$ $f = c$ का \emph{वही} दो-अवयवी \omterm{def:b1:logic:sets}{हल-समुच्चय}
है (प्रश्न 4); $\varphi$ को प्रतिलोम \omterm{def:b1:logic:map}{प्रतिचित्रण} के रूप में
$\intoo\eu{+\infty}$ तक बढ़ाने पर $\varphi(\varphi(x))$ दूसरे (अर्थात् मूल)
अवयव पर लौट आता है: $\varphi \circ \varphi = \mathrm{id}$।

\textbf{13.} यदि $(x, y)$ $x < y$ वाला अतुच्छ पूर्णांक हल है, तो प्रश्न 7 $x
\in \intoo1\eu$ रखता है: वहाँ एकमात्र पूर्णांक $x = 2$ है। तब $y > \eu$ के
साथ $f(y) = f(2) = \frac{\ln2}2$; प्रश्न 4 से समीकरण $f = \frac{\ln2}2$ का
$\eu$ से आगे ठीक एक हल है, और प्रश्न 3 उसे प्रस्तुत करता है: $y = 4$। अतः
$(2, 4)$ और उसकी अदला-बदली ही एकमात्र हैं।

\textbf{14.} $t = 1 + \frac1n = \frac{n+1}n$ से $\frac1{t - 1} = n$ और
$\frac t{t-1} = n + 1$ मिलते हैं, अतः
\[
x_n = \Bigl(\frac{n+1}n\Bigr)^{n},
\qquad
y_n = \Bigl(\frac{n+1}n\Bigr)^{n+1},
\]
जो स्पष्टतः परिमेय और भिन्न है ($y_n = t\,x_n \neq x_n$), और $n = 1$ से $x_1
= 2$, $y_1 = 4$ मिलता है।

\textbf{15.} $t = y/x$ परिमेय है और $> 1$; $t - 1 = \frac rs$ को न्यूनतम
पदों में लिखिए, अतः $\gcd(s + r, s) = \gcd(r, s) = 1$ के साथ $t = \frac{s +
r}s$। प्रश्न 8 से $x = t^{s/r}$, इसलिए
\[
x^r = t^s = \frac{(s+r)^s}{s^s} ,
\]
जो न्यूनतम पदों में भिन्न है (कोई अभाज्य $s + r$ और $s$ दोनों को विभाजित
नहीं करता)। $x = \frac pq$ को न्यूनतम पदों में लिखने पर $x^r = \frac{p^r}
{q^r}$ भी न्यूनतम पदों में है, और लघुकृत निरूपण की अद्वितीयता से: $p^r =
(s+r)^s$ और $q^r = s^s$। $q^r = s^s$ में किसी भी अभाज्य $\ell$ का घातांक
तुलनाइए: $r \cdot v_\ell(q) = s \cdot v_\ell(s)$, अतः $r$ $s\,v_\ell(s)$ को
विभाजित करता है; चूँकि $\gcd(r, s) = 1$, अतः प्रत्येक अभाज्य $\ell$ के लिए
$r$ $v_\ell(s)$ को विभाजित करता है (अद्वितीय गुणनखंडन, स्वीकृत;
\cref{ch:b1:arith} में सिद्ध), अतः किसी पूर्णांक $b$ के लिए $s = b^r$। $p^r
= (s+r)^s$ पर वही तर्क $s + r = a^r$ देता है।

\textbf{16.} मान लीजिए $a^r - b^r = r$, जहाँ $a > b \geq 1$ पूर्णांक हैं और
$r \geq 2$। तब $a \geq b + 1$, अतः द्विपद प्रमेय से
\[
a^r - b^r \geq (b+1)^r - b^r
= \sum_{k=0}^{r-1}\binom rk b^k
\geq \binom r0 + \binom r{r-1} b^{r-1}
\geq 1 + r > r ,
\]
जो विरोधाभास है। प्रश्न 15 से $a^r - b^r = (s + r) - s = r$ $r = 1$ पर बाध्य
कर देता है: $t = 1 + \frac1s$, और हल प्रश्न 14 का $(x_s, y_s)$ है। अदला-बदली
किए हुए युग्मों के साथ मिलाकर वर्गीकरण पूरा हो जाता है।

\textbf{17.} $f\bigl(\tfrac94\bigr) = \frac{\ln 2.25}{2.25} =
\frac{0.81093}{2.25} = 0.3604$ और $f\bigl(\tfrac{27}8\bigr) = \frac{\ln
3.375}{3.375} = \frac{1.21640}{3.375} = 0.3604$ (चार दशमलव): बराबर, जैसा
रचना का वादा है --- अतः $\bigl(\tfrac94\bigr)^{27/8} =
\bigl(\tfrac{27}8\bigr)^{9/4}$।

\textbf{18.} प्राचल $t_n = 1 + \frac1n$ कठोरतः घटकर $1$ तक जाते हैं। चूँकि
$x(\cdot)$ $\intoo1{+\infty}$ पर कठोरतः ह्रासमान है (प्रश्न 11), $x_n =
x(t_n)$ कठोरतः \emph{वर्धमान} है; और चूँकि वहाँ $y(\cdot)$ कठोरतः वर्धमान है
(प्रश्न 12), $y_n = y(t_n)$ कठोरतः \emph{ह्रासमान} है। प्रश्न 7 से $x_n \in
\intoo1\eu$ और $y_n \in \intoo\eu{+\infty}$: अतः प्रत्येक $n$ के लिए $x_n <
\eu < y_n$। अंत में $t_n \to 1$ और प्रश्न 10 से $x_n \to \eu$ और $y_n \to
\eu$ मिलते हैं। $\bigl(1 + \frac1n\bigr)^n$ का चिरपरिचित एकदिष्ट अभिसरण शाखा
की ज्यामिति से ही निकल आता है, बिना किसी नई असमिका के।

\textbf{19.} यदि $\eu \leq a < b$: $\intco\eu{+\infty}$ पर कठोरतः ह्रासमान
$f$ $f(a) > f(b)$ देता है, अर्थात् $b\ln a > a\ln b$, अर्थात् $a^b > b^a$।
यदि $1 < a < b \leq \eu$: कठोरतः वर्धमान $f$ $f(a) < f(b)$ देता है, इसलिए
$a^b < b^a$। मिश्रित स्थिति: $2 < \eu < 3$ और $2^3 = 8 < 9 = 3^2$; पर $2 <
\eu < 5$ और $2^5 = 32 > 25 = 5^2$: दोनों परिणाम होते हैं, और यह इस पर निर्भर
करता है कि बिंदु $(a, b)$ शाखा के किस ओर गिरता है।

\textbf{20.} शाखा विकर्ण को $\eu$ पर मिलती है और $\varphi$ वहाँ
$\varphi(\eu) = \eu$ के साथ संतत रूप से विस्तारित होता है। शृंखला नियम से $x
= \eu$ पर $\varphi(\varphi(x)) = x$ का अवकलन करने पर:
$\varphi'(\varphi(\eu))\,\varphi'(\eu) = \varphi'(\eu)^2 = 1$, अतः
$\varphi'(\eu) = \pm1$; $\varphi$ ह्रासमान है, अतः $\varphi'(\eu) = -1$।
शाखा विकर्ण को प्रवणता $-1$ के साथ काटती है: अर्थात् लंबवत्।

\textbf{21.} $y = 3x$ स्थिति $t = 3$ है: $x = 3^{1/2} = \sqrt3$ और $y =
3^{3/2} = 3\sqrt3$। जाँच: $f(\sqrt3) = \frac{0.54931}{1.73205} = 0.3171$ और
$f(3\sqrt3) = \frac{\ln 5.19615}{5.19615} = \frac{1.64792}{5.19615} =
0.3171$: चार दशमलव तक बराबर, अतः $(\sqrt3)^{3\sqrt3} = (3\sqrt3)^{\sqrt3}$।

\textbf{22.} $n^{1/n} = \eu^{f(n)}$, और $\exp$ वर्धमान है, अतः क्रम वही है
जो $f(n)$ का है: $f(3) = \frac{\ln3}3 \approx 0.3662$ $f(2) = f(4) =
\frac{\ln2}2 \approx 0.3466$ से आगे है (प्रश्न 3) और $f(5) \approx 0.3219$।
सबसे बड़ी $\sqrt[3]3$ है, और बराबर युग्म $\sqrt2 = \sqrt[4]4$ है --- अर्थात्
पूर्णांक युग्म $(2, 4)$ ही एक और वेश में।

\textbf{23.} \omterm{def:b1:logic:sets}{हल-समुच्चय} विकर्ण $\{(x, x) : x > 0\}$ और एक ही शाखा का सम्मिलन
है; शाखा विकर्ण के सापेक्ष सममित है, अनंतस्पर्शी $x = 1$ (जब $y \to
+\infty$) से अनंतस्पर्शी $y = 1$ (जब $x \to +\infty$) तक कठोरतः ह्रासमान है,
विकर्ण को ठीक एक बार $(\eu, \eu)$ पर काटती है और वहाँ उसकी प्रवणता $-1$ है।
शाखा पर पूर्णांक बिंदु $(2, 4)$ और $(4, 2)$ बैठते हैं --- केवल यही --- तथा
परिमेय बिंदु $(x_n, y_n)$, जो शाखा के साथ एकदिष्ट रूप से $(\eu, \eu)$ की ओर
बढ़ते जाते हैं पर उस तक कभी नहीं पहुँचते ($\eu$ अपरिमेय है; शाखा के परिमेय
बिंदु एक अपरिमेय कोने पर संचित होते हैं)।

\textbf{24.} (क) वृद्धि तुलनाओं ने $+\infty$ (प्रश्न 2) और $t\ln t \to 0$
(प्रश्न 10) पर $f$ की सीमाएँ दीं, जिनसे विचरण सारणी और अनंतस्पर्शी रेखाएँ
दोनों बनीं। (ख) मध्यवर्ती मान प्रमेय ने, एकैकी आच्छादन वाले कथनों के द्वारा,
विचरण सारणी को यथार्थ हल-संख्याओं (प्रश्न 4) में और प्रतिलोम \omterm{def:b1:logic:map}{प्रतिचित्रण}
$t(x)$ के अस्तित्व (प्रश्न 12) में बदल दिया। (ग) अद्वितीय गुणनखंडन ने प्रश्न
15 की दोनों न्यूनतम-पद पहचानों को शक्ति दी, जो परिमेय-बिंदु वर्गीकरण का
अंकगणितीय हृदय हैं।

\textbf{25.} एक ही अवकलज संगणना --- $1 - \ln t$ का चिह्न --- ने सब कुछ
उत्पन्न किया: प्रत्येक स्तर $c$ के लिए हलों की संख्या, शाखा का आकार और
अनंतस्पर्शी रेखाएँ, चरम असमिका $\eu^x \geq x^\eu$, पूर्णांक और परिमेय हलों
का वर्गीकरण, तथा $\bigl(1 + \frac1n\bigr)^n$ का एकदिष्ट अभिसरण। यही विचरण
सारणियों के साथ सोचने की मितव्ययिता है: एकविमीय अध्ययन दो चरों वाला समीकरण
हल कर देता है, क्योंकि समीकरण एक ही फलन से होकर गुजरता है। \cref{ch:b1:seq}
$(x_n)$ के अभिसरण को एकदिष्ट अनुक्रमों के सामान्य सिद्धांत से फिर करेगा;
\cref{ch:b1:derivative} विचरण सारणियों को माध्य मान प्रमेय पर कठोरता से खड़ा
करता है; और \cref{ch:b1:taylor} वह देता है जो सारणी नहीं दे सकती --- अभिसरण
$x_n \to \eu$ की \emph{गति} (वह कोटि $1/n$ की है)।
