\begin{solution}{exo:b1:linmaps:3}
\emph{\omterm{def:b1:linmaps:kerim}{अष्टि}:} $P = P'$ $\deg P = \deg P'$ बाध्य कर देता है, जब तक $P = 0$ न
हो; पर $P \neq 0$ के लिए $\deg P' < \deg P$: अतः $\ker u = \{0\}$, और $u$,
जो \omterm{def:b1:findim:def}{परिमित-विमीय} $\R_n[X]$ की \omterm{def:b1:logic:inj}{एकैकी} अंतःसमाकारिता है, एक तुल्याकारिता है
(\cref{cor:b1:linmaps:samedim})।

\emph{प्रतिलोम:} मान लीजिए $v(P) = P + P' + P'' + \dots + P^{(n)}$
($\R_n[X]$ पर एक परिमित योग)। तब
\[
v\bigl(u(P)\bigr) = \sum_{k=0}^{n} (P - P')^{(k)}
= \sum_{k=0}^{n} P^{(k)} - \sum_{k=0}^{n} P^{(k+1)}
= P - P^{(n+1)} = P ,
\]
जो दूरबीनी हो जाता है, क्योंकि $P^{(n+1)} = 0$। अतः $v = u^{-1}$।
\end{solution}

\begin{solution}{exo:b1:linmaps:5}
$u^2 = 0$ का अर्थ है कि सभी $x$ के लिए $u(u(x)) = 0$: अर्थात् हर $u(x)$

\begin{solution}{exo:b1:linmaps:6}
$(pq)^2 = pqpq = ppqq = pq$ (क्रमविनिमय): अतः एक \omterm{def:b1:linmaps:projection}{प्रक्षेप}
(\cref{thm:b1:linmaps:projchar})।

\omterm{def:b1:linmaps:kerim}{प्रतिबिंब}: $\operatorname{im}(pq) \subseteq \operatorname{im} p$, और $=
\operatorname{im}(qp) \subseteq \operatorname{im} q$: अतः वह सर्वनिष्ठ में
समाया है। विलोमतः यदि $x \in \operatorname{im} p \cap \operatorname{im} q$,
तो $p(x) = x$ और $q(x) = x$ (अचल बिंदु किसी \omterm{def:b1:linmaps:projection}{प्रक्षेप} के \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} का
अभिलक्षण करते हैं), अतः $pq(x) = x$: $x \in \operatorname{im}(pq)$।

\omterm{def:b1:linmaps:kerim}{अष्टि}: $\ker p \subseteq \ker(qp) = \ker(pq)$ और इसी प्रकार $\ker q
\subseteq \ker(pq)$: अतः योग उसमें समाया है। विलोमतः $pq(x) = 0$ लीजिए, और
लिखिए
\[
x = \underbrace{q(x)}_{\in\, \ker p} +
\underbrace{(x - q(x))}_{\in\, \ker q} :
\]
पहला पद $p(q(x)) = 0$ संतुष्ट करता है, अतः वह $\ker p$ में है; और दूसरा
$\ker q$ में है, क्योंकि $q(x - q(x)) = q(x) - q^2(x) = 0$। इस प्रकार $x \in
\ker p + \ker q$।
\end{solution}

\begin{solution}{exo:b1:linmaps:7}
पहले व्यापक अंतर्विष्टताएँ $\ker u \subseteq \ker u^2$ और $\operatorname{im}
u^2 \subseteq \operatorname{im} u$ देख लीजिए, और कोटि--शून्यता से (2) $\iff$
(3) (बराबर अष्टियाँ $\iff$ बराबर कोटियाँ $\iff$ बराबर \omterm{def:b1:linmaps:kerim}{प्रतिबिंब}, क्योंकि
अंतर्विष्टताएँ दी हुई हैं)।

(1 $\Rightarrow$ 2): मान लीजिए $u^2(x) = 0$; तब $u(x) \in \ker u \cap
\operatorname{im} u = \{0\}$, अतः $x \in \ker u$।

(2 $\Rightarrow$ 1): ग्रासमान और कोटि--शून्यता से $\dim(\ker u +
\operatorname{im} u) = \dim\ker u + \operatorname{rk} u - \dim(\ker u \cap
\operatorname{im} u) = \dim E - \dim(\ker u \cap \operatorname{im} u)$: अतः
योग $E$ है तभी जब सर्वनिष्ठ $\{0\}$ हो। मान लीजिए $y \in \ker u \cap
\operatorname{im} u$: $y = u(x)$ और $u(y) = 0$, अतः $u^2(x) = 0$, इसलिए ((2)
से) $u(x) = 0$: $y = 0$। अतः $E = \ker u \oplus \operatorname{im} u$।
\end{solution}

\begin{solution}{exo:b1:linmaps:10}
\begin{enumerate}
  \item $f \circ f = -\mathrm{id}$ \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है, अतः $f$ भी है
  (\cref{prop:b1:logic:comp} अनुकूलित: $f$ का दुतरफ़ा प्रतिलोम $-f$ है)। यदि
  $x \neq 0$ के साथ $f(x) = \lambda x$: तो $f$ लगाने पर $-x = \lambda^2 x$,
  अतः $\lambda^2 = -1$: जो $\R$ में असंभव है।
  \item कुल को एक-एक चरण में बढ़ाते हुए बनाइए। $x_1 \neq 0$ लीजिए: (1) से
  $(x_1, f(x_1))$ \omterm{def:b1:vspaces:free}{स्वतंत्र} है। अब मान लीजिए वर्तमान कुल
        \[
        \bigl(x_1, f(x_1), \dots, x_k, f(x_k)\bigr)
        \]
  है और उसका $\operatorname{Vect}$ $V_k$ है --- यह $f$ के अंतर्गत स्थायी
  \omterm{def:b1:vspaces:subspace}{उपसमष्टि} है (हर जनक किसी अन्य जनक या उसके ऋण पर जाता है: $f(f(x_i)) =
  -x_i$)। यदि $V_k$ पूरा $E$ न हो, तो $x_{k+1} \notin V_k$ चुनिए।
  \emph{दावा: बढ़ा हुआ कुल \omterm{def:b1:vspaces:free}{स्वतंत्र} है।} मान लीजिए $v \in V_k$ और $(\alpha, \beta) \neq
  (0,0)$ के साथ $\alpha x_{k+1} + \beta f(x_{k+1}) + v = 0$। $f$ लगाइए:
  $f(v) \in V_k$ के साथ $\alpha f(x_{k+1}) - \beta x_{k+1} + f(v) = 0$।
  दोनों संबंधों में से $f(x_{k+1})$ विलुप्त कीजिए (पहले को $\alpha$ से,
  दूसरे को $-\beta$ से गुणा कीजिए और जोड़िए):
        \[
        (\alpha^2 + \beta^2)\, x_{k+1} \in V_k ,
        \]
        और $\alpha^2 + \beta^2 \neq 0$ $x_{k+1} \in V_k$ बाध्य कर देता है:
        विरोधाभास। अतः रचना चलती रहती है, और \emph{एक बार में दो} सदिश
        जोड़ती जाती है, जब तक $V_k = E$: अंतिम कुल सम आकार का \omterm{def:b1:vspaces:free}{आधार} है, और
        $n$ सम है।
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:linmaps:11}
\omterm{def:b1:reals:bounds}{ऊपरी परिबंध}: हर $x$ के लिए $(u + v)(x) = u(x) + v(x) \in \operatorname{im} u
+ \operatorname{im} v$, अतः
\[
\operatorname{rk}(u + v)
\leq \dim(\operatorname{im} u + \operatorname{im} v)
\leq \operatorname{rk} u + \operatorname{rk} v
\]
(ग्रासमान, \cref{thm:b1:findim:grassmann})। निचला परिबंध: \omterm{def:b1:reals:bounds}{ऊपरी परिबंध} को
जोड़ी $(u + v, -v)$ पर लगाइए, जिसका योग $u$ है:
\[
\operatorname{rk} u \leq \operatorname{rk}(u + v) +
\operatorname{rk}(-v) = \operatorname{rk}(u + v) +
\operatorname{rk} v,
\]
अतः $\operatorname{rk} u - \operatorname{rk} v \leq \operatorname{rk}(u+v)$;
और $u$ तथा $v$ की अदला-बदली करने पर निरपेक्ष मान मिल जाता है।
\end{solution}

\begin{solution}{exo:b1:linmaps:12}
\emph{यथार्थ सूत्र।} मान लीजिए $v'$ \omterm{def:b1:vspaces:subspace}{उपसमष्टि} $\operatorname{im} w$ पर $v$ का
सीमन है। उसका \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} $v(w(F)) = \operatorname{im}(v \circ w)$ है, और उसकी
\omterm{def:b1:linmaps:kerim}{अष्टि} $\ker v \cap \operatorname{im} w$। समष्टि $\operatorname{im} w$ पर
$v'$ के लिए कोटि--शून्यता:
\[
\operatorname{rk} w = \dim\operatorname{im} w
= \operatorname{rk}(v \circ w) + \dim(\ker v \cap
\operatorname{im} w) .
\]

\emph{फ्रोबेनियस।} यथार्थ सूत्र दो बार लगाइए, $w$ पर और $w \circ u$ पर:
\[
\operatorname{rk} w - \operatorname{rk}(vw)
= \dim\bigl(\ker v \cap \operatorname{im} w\bigr),
\qquad
\operatorname{rk}(wu) - \operatorname{rk}(vwu)
= \dim\bigl(\ker v \cap \operatorname{im}(wu)\bigr) .
\]
चूँकि $\operatorname{im}(w \circ u) \subseteq \operatorname{im} w$, अतः
दूसरा सर्वनिष्ठ पहले में समाया है, और उसकी विमा उससे बड़ी नहीं:
\[
\operatorname{rk}(wu) - \operatorname{rk}(vwu)
\;\leq\; \operatorname{rk} w - \operatorname{rk}(vw) ,
\]
जिसे पुनर्व्यवस्थित करने पर फ्रोबेनियस की असमिका मिल जाती है। $w =
\mathrm{id}_F$ के साथ (\omterm{def:b1:linmaps:rank}{कोटि} $\dim F$, और $\operatorname{im}\, \mathrm{id}_F
= F$): $\operatorname{rk} v + \operatorname{rk} u \leq \dim F +
\operatorname{rk}(vu)$, अर्थात् सिल्वेस्टर की असमिका --- जो
\cref{exo:b1:matrices:10} में आव्यूह-रूप में फिर से सिद्ध की गई है।
\end{solution}

\begin{solution}{pb:b1:linmaps:1}
\textbf{1.} $(\mathrm{id} - p)^2 = \mathrm{id} - 2p + p^2 = \mathrm{id} -
p$: अतः एक प्रक्षेपक। यदि $y = x - p(x)$, तो $p(y) = p(x) - p^2(x) = 0$, और
विलोमतः $x \in \ker p$ $x = (\mathrm{id} - p)(x)$ देता है:
$\operatorname{im}(\mathrm{id} - p) = \ker p$। और $(\mathrm{id} - p)(x) = 0
\iff p(x) = x \iff x \in \operatorname{im} p$ (अचल बिंदु,
\cref{thm:b1:linmaps:projchar}): $\ker(\mathrm{id} - p) = \operatorname{im}
p$।

\textbf{2.} $(\lambda\,\mathrm{id} + \mu p)^2 = \lambda^2\,\mathrm{id} +
(2\lambda\mu + \mu^2)\,p$। जोड़ी $(\mathrm{id}, p)$ $\mathcal{L}(E)$ में
\omterm{def:b1:vspaces:free}{स्वतंत्र} है: $p = c\, \mathrm{id}$ $c^2 = c$ देता, अतः $p = 0$ या
$\mathrm{id}$, जो निषिद्ध हैं। गुणांकों की पहचान करने पर \omterm{def:b1:logic:map}{प्रतिचित्रण}
प्रक्षेपक है तभी जब $\lambda^2 = \lambda$ और $2\lambda\mu + \mu^2 = \mu$।
$\lambda = 0$ के लिए: $\mu \in \{0, 1\}$। $\lambda = 1$ के लिए: $\mu^2 + \mu
= 0$, $\mu \in \{0, -1\}$। अतः समतल में ठीक चार प्रक्षेपक: $0$, $p$,
$\mathrm{id}$, $\mathrm{id} - p$।

\textbf{3.} $(\lambda\,\mathrm{id} + \mu p)(\lambda'\,\mathrm{id} + \mu' p)
= \lambda\lambda'\,\mathrm{id} + (\lambda\mu' + \mu\lambda' + \mu\mu')\,p$:
अतः समतल संयुक्त \omterm{def:b1:logic:map}{प्रतिचित्रण} के अंतर्गत स्थायी है। चूँकि हर $k \geq 1$ के
लिए $p^k = p$, अतः $Q = \sum_k a_k X^k$ के लिए:
\[
Q(p) = a_0\,\mathrm{id} + \Bigl(\sum_{k \geq 1} a_k\Bigr) p
= Q(0)\,\mathrm{id} + \bigl(Q(1) - Q(0)\bigr)\,p .
\]

\textbf{4.} $\operatorname{im} p$ पर (जहाँ $p$ तत्समक की तरह काम करता है)
$\lambda\,\mathrm{id} + \mu p$ $\lambda + \mu$ से गुणा करता है; और $\ker p$
पर $\lambda$ से। चूँकि $E = \operatorname{im} p \oplus \ker p$, अतः
\omterm{def:b1:logic:map}{प्रतिचित्रण} \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है तभी जब $\lambda \neq 0$ और $\lambda + \mu \neq
0$। प्रश्न 3 के संयोजन-नियम में $\lambda\alpha = 1$, $\lambda\beta +
\mu\alpha + \mu\beta = 0$ हल करने पर:
\[
(\lambda\,\mathrm{id} + \mu p)^{-1}
= \frac1\lambda\,\mathrm{id} -
\frac{\mu}{\lambda(\lambda + \mu)}\,p ,
\]
जिसकी क्रिया $\ker p$ पर $1/\lambda$ से और $\operatorname{im} p$ पर
$1/(\lambda + \mu)$ से है, जैसा होना ही चाहिए।

\textbf{5.} $F = \operatorname{im} p = \operatorname{im} p'$ लिखिए। हर $x$
के लिए $p'(x) \in F$, और $p$ $F$ को बिंदुवार अचल रखता है: $p(p'(x)) =
p'(x)$, अर्थात् $p\,p' = p'$; सममित रूप से $p'\,p = p$। जब दो प्रक्षेपों का
\omterm{def:b1:linmaps:kerim}{प्रतिबिंब} एक ही हो, तो निर्णय वही करता है जो \emph{पहले} लगाया जाए: उसका
निर्गम पहले ही $F$ में होता है, जहाँ बाहरी \omterm{def:b1:linmaps:projection}{प्रक्षेप} तत्समक की तरह काम करता
है और कुछ नहीं बदलता।

\textbf{6.} $(p + q)^2 = p^2 + pq + qp + q^2 = (p + q) + pq + qp$, अतः $p +
q$ के प्रक्षेपक होने से $pq + qp = 0$ बाध्य हो जाता है। बाईं ओर से $p$ के
साथ संयुक्त कीजिए: $pq + pqp = 0$; दाईं ओर से $p$ के साथ: $pqp + qp = 0$।
घटाने पर $pq = qp$; फिर $pq + qp = 2pq = 0$, और अभिलक्षण $2$ नहीं है: अतः
$pq = qp = 0$।

\textbf{7.} $pq = qp = 0$ के साथ वही प्रसार $(p + q)^2 = p + q$ देता है।
\omterm{def:b1:linmaps:kerim}{प्रतिबिंब}: $\operatorname{im}(p + q) \subseteq \operatorname{im} p +
\operatorname{im} q$ सदा। विलोमतः $x \in \operatorname{im} p$ के लिए: $q(x)
= q(p(x)) = 0$, अतः $(p + q)(x) = p(x) = x$ और $x \in
\operatorname{im}(p+q)$; $\operatorname{im} q$ के लिए भी वही। प्रत्यक्षता:
$x \in \operatorname{im} p \cap \operatorname{im} q$ $x = p(x) = p(q(x)) =
0$ देता है। \omterm{def:b1:linmaps:kerim}{अष्टि}: यदि $p(x) + q(x) = 0$, तो $p$ लगाने पर $p(x) + p(q(x)) =
p(x) = 0$ मिलता है, और $q$ लगाने पर $q(x) = 0$: अतः $\ker(p + q) = \ker p
\cap \ker q$ (उलटी अंतर्विष्टता स्पष्ट है)।

\textbf{8.} $p - q$ प्रक्षेपक है तभी जब $\mathrm{id} - (p - q) =
(\mathrm{id} - p) + q$ प्रक्षेपक हो (प्रश्न 1 दो बार)। प्रक्षेपकों
$\mathrm{id} - p$ और $q$ पर प्रश्न 6--7 लगाने से यह तभी सत्य है जब
$(\mathrm{id} - p)q = q(\mathrm{id} - p) = 0$, अर्थात् तभी जब $pq = q$ और
$qp = q$।

\textbf{9.} $pq = q$ का अर्थ है कि $p$ हर $q(x)$ को अचल रखता है, अर्थात्
$\operatorname{im} q \subseteq \ker(p - \mathrm{id}) = \operatorname{im} p$।
और $qp = q$ का अर्थ है कि सभी $x$ के लिए $q\bigl((\mathrm{id} - p)(x)\bigr)
= 0$, अर्थात् $q$ $\operatorname{im}(\mathrm{id} - p) = \ker p$ पर शून्य
होता है: $\ker p \subseteq \ker q$। दोनों चरण तुल्यताएँ हैं: क्रम $q \leq p$
कहता है कि $q$ छोटे \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} पर, और बड़ी \omterm{def:b1:linmaps:kerim}{अष्टि} के अनुदिश \omterm{def:b1:linmaps:projection}{प्रक्षेप} करता है।

\textbf{10.} प्रसार करने पर $(\mathrm{id} - p)(\mathrm{id} - q) =
\mathrm{id} - p - q + pq = \mathrm{id} - r$। प्रक्षेपक $\mathrm{id} - p$ और
$\mathrm{id} - q$ क्रमविनिमेय हैं, अतः \cref{exo:b1:linmaps:6} से उनका
गुणनफल $\mathrm{id} - r$ $\ker(\mathrm{id} - p) + \ker(\mathrm{id} - q) =
\operatorname{im} p + \operatorname{im} q$ के अनुदिश
$\operatorname{im}(\mathrm{id} - p) \cap \operatorname{im}(\mathrm{id} - q)
= \ker p \cap \ker q$ पर प्रक्षेपक है। प्रश्न 1 से $r = \mathrm{id} -
(\mathrm{id} - r)$ तब वह प्रक्षेपक है जिसका $\operatorname{im} r =
\operatorname{im} p + \operatorname{im} q$ और $\ker r = \ker p \cap \ker q$।

\textbf{11.} $p_i$ सुपरिभाषित है (विघटन की अद्वितीयता) और \omterm{def:b1:linmaps:def}{रैखिक} भी ($x +
\lambda y$ का विघटन विघटनों का योग ही है, फिर से अद्वितीयता से)। $x_i \in
F_i$ के लिए विघटन स्वयं $x_i$ है, अतः $p_i(x_i) = x_i$: $p_i^2 = p_i$, और $j
\neq i$ के लिए $p_j(x_i) = 0$: $p_i p_j = 0$ ($p_j(x) \in F_j$)। घटकों को
जोड़ने पर $\sum_i p_i = \mathrm{id}$। अंत में $\operatorname{im} p_i = F_i$
और $\ker p_i = \bigoplus_{j \neq i} F_j$।

\textbf{12.} $p_i = p_i \circ \mathrm{id} = p_i\sum_j p_j = p_i^2 + \sum_{j
\neq i} p_i p_j = p_i^2$: अतः प्रत्येक $p_i$ एक प्रक्षेपक है। हर $x =
\mathrm{id}(x) = \sum_i p_i(x)$ $\sum_i \operatorname{im} p_i$ में है: अतः
प्रतिबिंबों का योग $E$ है। प्रत्यक्षता: मान लीजिए $y_i \in \operatorname{im}
p_i$ के साथ $y_1 + \dots + y_k = 0$, अतः $p_i(y_i) = y_i$। $p_j$ लगाइए: $i
\neq j$ के लिए $p_j (y_i) = p_j p_i (y_i) = 0$, अतः $0 = p_j\bigl(\sum
y_i\bigr) = y_j$, और यह हर $j$ के लिए। इस प्रकार $E = \bigoplus_i
\operatorname{im} p_i$।

\textbf{13.} $q = \mathrm{id} - p$, और प्रश्न 1 सीधे $pq = p - p^2 = 0 = qp$
दे देता है: दो प्रक्षेपकों के लिए तत्समक तक जुड़ जाना ही जोड़ी की लांबिकता
बाध्य कर देता है।

\textbf{14.} $r$ के प्रक्षेपक होने के साथ $p + q = \mathrm{id} - r$, और
$(\mathrm{id} - r)$ भी प्रक्षेपक है (प्रश्न 1): अतः $p + q$ एक प्रक्षेपक है,
और प्रश्न 6 $pq = qp = 0$ देता है। \omterm{def:b1:linmaps:projection}{सममिति} से ($q + r = \mathrm{id} - p$ और
$p + r = \mathrm{id} - q$) सभी युग्मशः गुणनफल शून्य हो जाते हैं, और प्रश्न
12 निष्कर्ष दे देता है: $E = \operatorname{im} p \oplus \operatorname{im} q
\oplus \operatorname{im} r$, स्वतः।

\textbf{15.} \emph{प्रमेयिका।} ग्रासमान के साथ आगमन से
(\cref{thm:b1:findim:grassmann}):
\[
\dim(F_1 + \dots + F_k) \leq \dim(F_1 + \dots + F_{k-1}) + \dim
F_k \leq \dots \leq \sum_i \dim F_i .
\]
यदि कुल योग समानता है, तो हर चरण समानता है: हर $j$ के लिए $(F_1 + \dots +
F_{j-1}) \cap F_j = \{0\}$, और कोई संबंध $y_1 + \dots + y_k = 0$ ($y_i \in
F_i$) दाईं ओर से ढह जाता है: $y_k \in (F_1 + \dots + F_{k-1}) \cap F_k =
\{0\}$, फिर $y_{k-1} = 0$, इत्यादि: अतः योग प्रत्यक्ष है। विलोमतः \omterm{def:b1:vspaces:sum}{प्रत्यक्ष
योग} की विमाएँ जुड़ती हैं (आधारों को एक साथ लिख दीजिए)। \emph{अनुप्रयोग:} $x
= \sum_i p_i(x)$ दिखाता है कि $E = \sum_i \operatorname{im} p_i$, अतः $n
\leq \sum_i \operatorname{rk} p_i$; परिकल्पना समानता देती है, इसलिए
प्रत्यक्षता। गुणनफल: $j$ और $y \in \operatorname{im} p_j$ नियत कीजिए। तब
$p_i(y) \in \operatorname{im} p_i$ के साथ $y = \sum_i p_i(y)$, जबकि $y = y$
भी एक विघटन है (केवल घटक $j$); अद्वितीयता $i \neq j$ के लिए $p_i(y) = 0$
बाध्य कर देती है। $y = p_j(x)$ पर लगाने से: $p_i p_j = 0$।

\textbf{16.} यदि $u^k(x) = 0$, तो $u^{k+1}(x) = u(0) = 0$। और
$\operatorname{im} u^{k+1} = u^k\bigl(u(E)\bigr) \subseteq u^k(E) =
\operatorname{im} u^k$।

\textbf{17.} मान लीजिए $\ker u^{r} = \ker u^{r+1}$ और $x \in \ker u^{r+2}$
लीजिए: तब $u(x) \in \ker u^{r+1} = \ker u^{r}$, अतः $u^{r+1}(x) = 0$: $x \in
\ker u^{r+1}$। प्रश्न 16 के साथ $\ker u^{r+1} = \ker u^{r+2}$, और आगमन से
आगे की सभी अष्टियाँ $\ker u^{r}$ के साथ एक ही हो जाती हैं। प्रतिबिंबों के
लिए: कोटि--शून्यता $\dim\operatorname{im} u^k = n - \dim\ker u^k$ देती है,
अतः \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} की विमाएँ ठीक तभी जम जाती हैं जब \omterm{def:b1:linmaps:kerim}{अष्टि} की विमाएँ जमती हैं, और
प्रश्न 16 की अंतर्विष्टताओं के साथ बराबर विमाओं का अर्थ बराबर उपसमष्टियाँ है
(\cref{thm:b1:findim:subspaces})।

\textbf{18.} अनुक्रम $\bigl(\dim\ker u^k\bigr)_k$ अह्रासमान है और उसके मान
$\intint{0}{n}$ में हैं; वह $n + 1$ बार सचमुच नहीं बढ़ सकता, अतः $r \leq n$
वाला कोई $\dim\ker u^{r} = \dim\ker u^{r+1}$ है, इसलिए $\ker u^{r} = \ker
u^{r+1}$ (अंतर्विष्टता और बराबर विमा)। सबसे छोटा $r$ लीजिए; प्रश्न 17 $r$ से
आगे सब कुछ जमा देता है, \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} भी।

\textbf{19.} सर्वनिष्ठ: मान लीजिए $x \in \ker u^{r} \cap \operatorname{im}
u^{r}$, यानी $u^{r}(x) = 0$ के साथ $x = u^{r}(y)$। तब $u^{2r}(y) = 0$, और
$\ker u^{2r} = \ker u^{r}$ (प्रश्न 17), अतः $x = u^{r}(y) = 0$। विमाएँ:
$u^{r}$ के लिए कोटि--शून्यता $\dim\ker u^{r} + \dim\operatorname{im} u^{r} =
n$ देती है; और तुच्छ सर्वनिष्ठ के साथ ग्रासमान योग को $n$ विमा वाली \omterm{def:b1:vspaces:subspace}{उपसमष्टि}
बना देता है: $E = \ker u^{r} \oplus \operatorname{im} u^{r}$।

\textbf{20.} स्थायित्व: $x \in \ker u^{r}$ के लिए $u^{r}(u(x)) = u(u^{r}(x))
= 0$; और $u(u^{r}(y)) = u^{r}(u(y)) \in \operatorname{im} u^{r}$। $N = \ker
u^{r}$ पर: $N$ की परिभाषा से $(u|_N)^{r} = 0$: अतः शून्यंभावी। $I =
\operatorname{im} u^{r}$ पर: $\ker(u|_I) = \ker u \cap I \subseteq \ker
u^{r} \cap I = \{0\}$, अतः $u|_I$ \omterm{def:b1:findim:def}{परिमित-विमीय} $I$ की \omterm{def:b1:logic:inj}{एकैकी} अंतःसमाकारिता
है, इसलिए \omterm{def:b1:logic:inj}{एकैकी आच्छादक} (\cref{cor:b1:linmaps:samedim})।

\textbf{21.} मान लीजिए $a \in N$, $b \in I$ के साथ $x = a + b$। तब $u(a) \in
N$ और $u(b) \in I$ के साथ $u(x) = u(a) + u(b)$ (प्रश्न 20): यही $u(x)$ का
विघटन \emph{है}, अतः $\pi(u(x)) = u(a) = u(\pi(x))$: $\pi u = u\pi$।

\textbf{22.} $u^2(x,y,z) = u(y, 0, z) = (0, 0, z)$ और $u^3(x,y,z) = u(0,0,z)
= (0,0,z) = u^2(x,y,z)$। अष्टियाँ: $\ker u = \{y = z = 0\} =
\operatorname{Vect}(e_1)$, $\ker u^2 = \{z = 0\} = \operatorname{Vect}(e_1,
e_2)$, $\ker u^3 = \ker u^2$: अतः $r = 2$ पर स्थिरीकरण। \omterm{def:b1:linmaps:kerim}{प्रतिबिंब}:
$\operatorname{im} u = \operatorname{Vect}(e_1, e_3)$, $\operatorname{im}
u^2 = \operatorname{Vect}(e_3)$। फिटिंग: $\R^3 = \operatorname{Vect}(e_1,
e_2) \oplus \operatorname{Vect}(e_3)$, और $\pi(x, y, z) = (x, y, 0)$। जाँच:
$\pi u(x,y,z) = \pi(y, 0, z) = (y, 0, 0)$ और $u\pi(x,y,z) = u(x, y, 0) = (y,
0, 0)$: बराबर। पहले गुणनखंड पर $u(x, y, 0) = (y, 0, 0)$, जिसका वर्ग $0$ है:
अतः शून्यंभावी; और दूसरे पर $u(0,0,z) = (0,0,z)$: अर्थात् तत्समक, जो \omterm{def:b1:logic:inj}{एकैकी
आच्छादक} है।

\textbf{23.} यदि $u^m = 0$, तो $\ker u^m = E$; और चूँकि अष्टियाँ $r$ से आगे
जमी हुई हैं, अतः $\ker u^{r} = \ker u^{\max(m, r)} = E$। विलोमतः $\ker u^{r}
= E$ का अर्थ $u^{r} = 0$ है। और $\ker u^{r} = E \iff$ फिटिंग प्रक्षेपक
$\{0\}$ के अनुदिश $E$ पर है, अर्थात् $\pi = \mathrm{id}$। अंत में $r \leq n$
(प्रश्न 18) देता है: हर शून्यंभावी अंतःसमाकारिता $u^{n} = 0$ संतुष्ट करती है
--- शून्यंभाविता सूचकांक विमा से कभी अधिक नहीं होता।

\textbf{24.} मान लीजिए $m$ $u|_A$ का कोई शून्यंभाविता सूचकांक है: $A
\subseteq \ker u^{m} \subseteq \ker u^{\max(m,r)} = \ker u^{r}$। चूँकि
$u|_B$ \omterm{def:b1:logic:inj}{एकैकी आच्छादक} है, अतः हर $k$ के लिए $B = u(B) = u^{k}(B) \subseteq
\operatorname{im} u^{k}$, विशेष रूप से $B \subseteq \operatorname{im}
u^{r}$। तब
\[
n = \dim A + \dim B \leq \dim\ker u^{r} +
\dim\operatorname{im} u^{r} = n :
\]
दोनों अंतर्विष्टताएँ विमाओं की समानताएँ हैं, अतः उपसमष्टियों की भी: $A =
\ker u^{r}$, $B = \operatorname{im} u^{r}$।

\textbf{25.} (क) भाग III एक शब्दकोश है: \omterm{thm:b1:logic:partition}{विभाजन} $E = F_1 \oplus \dots \oplus
F_k$ ठीक उन प्रक्षेपक-कुलों के अनुरूप हैं जिनके लिए $\sum p_i = \mathrm{id}$
और $p_i p_j = 0$, और $F_i$ उनके \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} हैं। (ख) $k = 3$ के लिए \omterm{def:b1:vspaces:sum}{पूरक}
$\mathrm{id} - p_i$ स्वयं प्रक्षेपक हैं, जिससे तर्क बिना किसी अतिरिक्त
परिकल्पना के बंद हो गया; व्यापक $k$ के लिए $\sum_i \operatorname{rk} p_i
\leq n$ चाहिए, और यह असमिका स्नातक वर्ष~2 का अनुरेख मुफ़्त में दे देता है
(प्रक्षेपक के लिए $\operatorname{tr} p = \operatorname{rk} p$, और अनुरेख
जुड़कर $\operatorname{tr} \mathrm{id} = n$ देते हैं)। (ग)
\cref{exo:b1:linmaps:7} फिटिंग की प्रमेयिका ही है, बस पहले से स्थिर हो चुकी
स्थिति $r \leq 1$ में; व्यापक रूप में \omterm{def:b1:linmaps:kerim}{अष्टि} और \omterm{def:b1:linmaps:kerim}{प्रतिबिंब} की शृंखलाओं को जमने
दिया जाता है, जिसमें अधिक से अधिक $n$ चरण लगते हैं। (घ) स्नातक वर्ष~2 के खंड
में $u - \lambda\, \mathrm{id}$ पर लगाने से शून्यंभावी गुणनखंड $\lambda$ पर
की व्यापकीकृत अभिलक्षणिक \omterm{def:b1:vspaces:subspace}{उपसमष्टि} बन जाता है, और भाग III के प्रक्षेपक
लघुकरण-सिद्धांत के वर्णक्रमीय प्रक्षेपक। भाग IV की प्रमेय \emph{फिटिंग की
प्रमेयिका} है।
\end{solution}
