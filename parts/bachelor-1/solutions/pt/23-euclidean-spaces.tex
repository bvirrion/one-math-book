\begin{solution}{exo:b1:euclid:4}
$F^\perp = \{(x,y,z) : x + y + z = 0\}$, \omterm{def:b1:vspaces:free}{base} $\bigl((1,-1,0),
(1,0,-1)\bigr)$. Com $e = \frac{1}{\sqrt3}(1,1,1)$: $p_F(x) =
\langle x, e\rangle e = \frac{x_1 + x_2 + x_3}{3}\,(1,1,1)$, logo
\[
\operatorname{Mat}(p_F) = \frac13

\]
Para $x = (1,2,3)$: $p_F(x) = (2,2,2)$ e $d(x, F) = \norm{(1,2,3)

\end{solution}

\begin{solution}{exo:b1:euclid:5}
Seja $p = \alpha(1,0,1) + \beta(0,1,1) = (\alpha, \beta, \alpha +
\beta)$. Ortogonalidade de $x - p$ aos geradores:
\[
\langle x - p, (1,0,1)\rangle = (1 - \alpha) + (4 - \alpha - \beta)
= 0,
\qquad
\langle x - p, (0,1,1)\rangle = (1 - \beta) + (4 - \alpha - \beta)
= 0,
\]
isto é, $2\alpha + \beta = 5$ e $\alpha + 2\beta = 5$: $\alpha =
\beta = \frac53$. Logo $p_F(x) = \bigl(\frac53, \frac53,
\frac{10}{3}\bigr)$ e
\[
d(x, F) = \norm{x - p} =
\norm{\bigl(-\tfrac23, -\tfrac23, \tfrac23\bigr)} =
\frac{2}{\sqrt 3} .
\]
Gram--Schmidt é desnecessário porque a propriedade que define a
\omterm{def:b1:linmaps:projection}{projeção} --- $x - p \perp F$ --- é ela própria um \omterm{def:b1:det:system}{sistema linear} (``equações
normais'') sobre os coeficientes em \emph{qualquer} família
geradora.
\end{solution}

\begin{solution}{exo:b1:euclid:6}
Cauchy--Schwarz com $g = 1$:
\[
\Bigl(\int_0^1 f \cdot 1\Bigr)^2 \leq \int_0^1 f^2 \cdot \int_0^1
1^2 = \int_0^1 f^2 ,
\]
igualdade se e somente se $f$ é proporcional a $1$, isto é,
constante. Versão discreta: em $\R^n$ com $x = (a_1, \dots, a_n)$, $y = (1, \dots,
1)$: $\bigl(\sum a_i\bigr)^2 \leq \norm x^2\, \norm y^2 = n \sum a_i^2$, com
igualdade se e somente se todos os $a_i$ são iguais.
\end{solution}

\begin{solution}{exo:b1:euclid:7}
Pelo \cref{thm:b1:euclid:projection}, $E = F \oplus F^\perp$, logo as dimensões se somam: $\dim F^\perp = \dim E - \dim F$. A inclusão
$F
\subseteq (F^\perp)^\perp$ é imediata (os vetores de $F$ são \omterm{def:b1:euclid:orthogonal}{ortogonais} a tudo em
$F^\perp$). Dimensões: $\dim(F^\perp)^\perp = \dim E - \dim F^\perp = \dim F$; uma inclusão com dimensões (finitas)
iguais é uma igualdade (o \cref{thm:b1:findim:subspaces}).
\end{solution}

\begin{solution}{exo:b1:euclid:9}
A quantidade é $\norm{f - (a + bX)}^2$ em $C(\intcc{0}{1})$ com o \omterm{def:b1:euclid:def}{produto interno} \omterm{thm:b1:integration:def}{integral}: mínima
exatamente na \omterm{thm:b1:euclid:projection}{projeção ortogonal} $a + bX = p_F(f) = X - \frac16$ (o \cref{ex:b1:euclid:bestapprox}). O mínimo é
\[
\norm{f - p_F(f)}^2 = \int_0^1 \Bigl(x^2 - x + \frac16\Bigr)^{\!2}
\dd x .
\]
Desenvolva: $\int_0^1 (x^2 - x + \frac16)^2 = \int_0^1 \bigl(x^4 - 2x^3
+ \frac{4}{3}x^2 - \frac13 x + \frac{1}{36}\bigr)\dd x = \frac15 -
\frac12 + \frac49 - \frac16 + \frac{1}{36} = \frac{1}{180}$. Logo o mínimo vale $\dfrac{1}{180}$.
\end{solution}

\begin{solution}{exo:b1:euclid:10}
Seja $y \in \ker(u - \mathrm{id})$ (isto é, $u(y) = y$) e $x \in E$. Então, usando a preservação
do \omterm{def:b1:euclid:def}{produto interno} ($\langle u(a),
u(b)\rangle = \langle a, b\rangle$):
\[
\langle x - u(x),\, y\rangle
= \langle x, y\rangle - \langle u(x), y\rangle
= \langle x, y\rangle - \langle u(x), u(y)\rangle
= \langle x, y\rangle - \langle x, y\rangle = 0 :
\]
todo vetor de $\operatorname{im}(u - \mathrm{id})$ é \omterm{def:b1:euclid:orthogonal}{ortogonal} a todo vetor fixo.

Portanto $\operatorname{im}(u - \mathrm{id}) \subseteq \ker(u -
\mathrm{id})^{\perp}$, e pelo teorema do núcleo e da \omterm{def:b1:linmaps:kerim}{imagem}, mais o \cref{exo:b1:euclid:7},
ambos têm dimensão $\dim E - \dim\ker(u -
\mathrm{id})$: eles são iguais. Então $E = \ker(u - \mathrm{id})
\oplus \ker(u - \mathrm{id})^\perp = \ker(u - \mathrm{id}) \oplus
\operatorname{im}(u - \mathrm{id})$.
\end{solution}

\begin{solution}{exo:b1:euclid:11}
\begin{enumerate}
  \item Suponha $Gc = 0$ para uma coluna $c = (c_1, \dots,
        c_k)^{\mathsf T}$. Então
        \[
        \Bigl\lVert\sum_i c_i v_i\Bigr\rVert^2
        = \sum_{i,j} c_i c_j \langle v_i, v_j\rangle
        = c^{\mathsf T} G\, c = 0,
        \]
        logo $\sum_i c_i v_i = 0$. Se a família é \omterm{def:b1:vspaces:free}{livre}, isto força $c = 0$: $G$ é
        invertível. Reciprocamente, uma relação não trivial $\sum_j c_j v_j = 0$
        dá, tomando o produto com cada $v_i$, a relação não trivial
        $Gc = 0$: $G$ singular. Liberdade $\iff \det G \neq 0$.
  \item $\langle X^{i-1}, X^{j-1}\rangle = \int_0^1 x^{i+j-2}\dd
        x = \frac{1}{i+j-1}$: a matriz de Gram de $(1, X, X^2)$ é exatamente a \omterm{pb:b1:det:1}{matriz de
        Hilbert} $H_3$, cujo determinante $\frac1{2160}$ foi calculado no
        problema de fim de semana do \cref{ch:b1:det}: não nulo, logo os
        monômios são \omterm{def:b1:vspaces:free}{livres} --- como se esperava, mas agora
        certificado por um número.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:euclid:12}
\begin{enumerate}
  \item $A^2 = A A = A^{\mathsf T} A = I$, logo $u^2 =
        \mathrm{id}$: $u$ é uma simetria, e o \cref{thm:b1:linmaps:projchar}~(2) dá
        $E = \ker(u -
        \mathrm{id}) \oplus \ker(u + \mathrm{id})$.
  \item Sejam $u(x) = x$ e $u(y) = -y$. Como $u$ preserva o \omterm{def:b1:euclid:def}{produto interno},
        \[
        \langle x, y\rangle = \langle u(x), u(y)\rangle
        = \langle x, -y\rangle = -\langle x, y\rangle ,
        \]
        logo $\langle x, y\rangle = 0$: os dois autoespaços são \omterm{def:b1:euclid:orthogonal}{ortogonais}, $\ker(u + \mathrm{id}) = F^\perp$ para
        $F =
        \ker(u - \mathrm{id})$, e $u$ é a simetria \omterm{def:b1:euclid:orthogonal}{ortogonal} em relação a $F$.
  \item $R_\theta$ é simétrica se e somente se $-\sin\theta = \sin\theta$, isto é, $\theta \in \{0, \pi\}$: as
        aplicações $\pm\mathrm{id}$ (identidade e simetria central). Toda $S_\theta$
        é simétrica: as reflexões em retas. Estas são exatamente as
        simetrias \omterm{def:b1:euclid:orthogonal}{ortogonais} do plano, com $F$ igual ao plano
        inteiro, a $\{0\}$, ou ao eixo da reflexão.
\end{enumerate}
\end{solution}

\begin{solution}{pb:b1:euclid:1}
\textbf{1.} $\norm{t_a(x) - t_a(y)} = \norm{x - y}$; uma \omterm{def:b1:euclid:isometry}{isometria} \omterm{def:b1:linmaps:def}{linear} preserva normas, logo
distâncias ($\norm{g(x) - g(y)} =
\norm{g(x - y)} = \norm{x-y}$); e uma composição de aplicações que preservam
distâncias preserva distâncias. Toda \omterm{def:b1:euclid:isometry}{isometria} é \omterm{def:b1:logic:inj}{injetiva} (pontos
distintos ficam a distância positiva) e, pela classificação abaixo,
\omterm{def:b1:logic:inj}{bijetiva}; a identidade e as inversas são \omterm{def:b1:euclid:isometry}{isometrias}: um \omterm{def:b1:structures:group}{grupo}.

\textbf{2.} $\norm{f(x)} = \norm{f(x) - f(0)} = \norm{x - 0} =
\norm x$. Polarização:
\[
\langle f(x), f(y)\rangle
= \frac{\norm{f(x)}^2 + \norm{f(y)}^2 - \norm{f(x) -
f(y)}^2}{2}
= \frac{\norm x^2 + \norm y^2 - \norm{x - y}^2}{2}
= \langle x, y\rangle .
\]

\textbf{3.} Desenvolvendo com a questão 2 (cada produto de $f$ é
igual ao produto dos argumentos):
\begin{align*}
\norm{f(x+y) - f(x) - f(y)}^2
&= \norm{x+y}^2 + \norm x^2 + \norm y^2\\
&\quad - 2\langle x+y, x\rangle - 2\langle x+y, y\rangle
+ 2\langle x, y\rangle = 0 ,
\end{align*}
como mostra uma verificação direta; do mesmo modo $\norm{f(\lambda x) -
\lambda f(x)}^2 = \norm{\lambda x}^2 - 2\lambda\langle \lambda
x, x\rangle + \lambda^2\norm x^2 = 0$. Logo $f(x + y) = f(x) +
f(y)$
e $f(\lambda x) = \lambda f(x)$: $f$ é \omterm{def:b1:linmaps:def}{linear} e preserva normas: $f \in O(\R^2)$.

\textbf{4.} Ponha $a = f(0)$ e $g = t_{-a}\circ f$: uma \omterm{def:b1:euclid:isometry}{isometria} que fixa $0$, logo
uma \omterm{def:b1:euclid:isometry}{isometria} \omterm{def:b1:linmaps:def}{linear} (questão 3), e $f = t_a
\circ g$. Unicidade: $t_a \circ g = t_{a'}\circ g'$ avaliado
em $0$ dá $a = a'$, e então $g = g'$.

\textbf{5.} $(t_a\circ g)\circ(t_{a'}\circ g') = t_{a +
g(a')}\circ(g\circ g')$, pois $g\,t_{a'} = t_{g(a')}\,g$ para $g$ \omterm{def:b1:linmaps:def}{linear}. Logo a parte \omterm{def:b1:linmaps:def}{linear}
de uma composição é $g \circ g'$, e $\det(gg') = \det g \det g'$: direta$\circ$direta $=$
indireta$\circ$indireta $=$ direta, e direta$\circ$indireta $=$
indireta --- a regra de sinais de $\pm1$.

\textbf{6.} $z \mapsto \eu^{\iu\theta}z$ leva $1$ em $(\cos\theta, \sin\theta)$ e $\iu$ em $\iu\eu^{\iu\theta} = (-\sin\theta, \cos\theta)$: as colunas de
$R_\theta$. E $z \mapsto \eu^{\iu\theta}\conj z$ leva $1$ em $(\cos\theta, \sin\theta)$ e $\iu$ em $-\iu\eu^{\iu\theta} =
(\sin\theta, -\cos\theta)$: as colunas de
$S_\theta$.

\textbf{7.} $a = 1$: $f = t_b$. $a \neq 1$: a equação de ponto fixo $az_0 + b = z_0$
tem a solução única $z_0 = b/(1 -
a)$, e então
\[
f(z) - z_0 = az + b - (az_0 + b) = a(z - z_0):
\]
no referencial centrado em $z_0$, $f$ é a multiplicação por $a
= \eu^{\iu\arg a}$:
a rotação de centro $z_0$ e ângulo $\arg a$.

\textbf{8.} $f(f(z)) = a\conj{(a\conj z + b)} + b = a\conj a\,z
+ a\conj b + b = z + v$: $f^2 = t_v$. Comutação: $f(z + v) =
a\conj z + a\conj v + b$ e $a\conj v = a\conj{(a\conj b + b)}
= b + a\conj b = v$, logo $f\circ t_v = t_v\circ f$.

\emph{Caso $v = 0$}: $f^2 = \mathrm{id}$. Para qualquer $z$, o ponto médio $m = \frac{z + f(z)}2$
satisfaz (sendo as \omterm{def:b1:euclid:isometry}{isometrias} afins, Parte I) $f(m) = \frac{f(z) + f^2(z)}{2} = m$: existem pontos
fixos. Conjugando pela translação até um ponto fixo, $f$ torna-se
uma \omterm{def:b1:euclid:isometry}{isometria} \omterm{def:b1:linmaps:def}{linear} indireta, isto é, algum $S_\theta$ (o \cref{thm:b1:euclid:plane}): uma
reflexão numa reta.

\emph{Caso $v \neq 0$}: $r = t_{-v/2}\circ f$ é indireta e
\[
r^2 = t_{-v/2}\,f\,t_{-v/2}\,f = t_{-v/2}\,t_{-v/2}\,f^2
= t_{-v}\,t_v = \mathrm{id}
\]
(usando a comutação), logo $r$ é uma reflexão, e $f =
t_{v/2}\circ r$. O seu eixo
é paralelo a $v$: $r$ comuta com $t_v$ (tanto $f$ quanto $t_{v/2}$
comutam), logo $t_v$ leva o eixo (a reta fixa de $r$) nele mesmo, o
que força $v$ a dirigi-lo. Portanto $f$ é uma \omterm{pb:b1:euclid:1}{reflexão deslizante}
com vetor de deslizamento $v/2$. Toda \omterm{def:b1:euclid:isometry}{isometria} está agora
classificada: translação ou rotação (direta), reflexão ou \omterm{pb:b1:euclid:1}{reflexão
deslizante} (indireta).

\textbf{9.} Rotações $f(z) = az + b$, $g(z) = a'z + b'$ com $a = \eu^{\iu\alpha}$, $a' = \eu^{\iu\beta}$: $g\circ f(z) =
a'az + (a'b + b')$ tem
coeficiente \omterm{def:b1:linmaps:def}{linear} $\eu^{\iu(\alpha+\beta)}$: uma rotação de ângulo $\alpha + \beta$ pela questão
7, ou uma translação quando $\eu^{\iu(\alpha+\beta)} =
1$. Reflexões $r_i(z) = a_i\conj z + b_i$ com $a_i =
\eu^{\iu\theta_i}$ (eixo
no ângulo $\theta_i/2$):
\[
r_2\circ r_1(z) = a_2\conj{a_1}\,z + (a_2\conj{b_1} + b_2),
\]
direta com ângulo $\theta_2 - \theta_1 = 2\bigl(\tfrac{\theta_2}2
- \tfrac{\theta_1}2\bigr)$: o dobro do ângulo entre os eixos; eixos
paralelos ($\theta_1 = \theta_2$) dão uma translação.

\textbf{10.} Uma rotação de centro $c$ e ângulo $\theta$ é o produto
de duas reflexões em retas por $c$ que fazem ângulo $\theta/2$ (questão
9, lida ao contrário); uma translação $t_v$ é o produto de duas
reflexões em retas paralelas \omterm{def:b1:euclid:orthogonal}{ortogonais} a $v$ à distância $\norm v/2$; uma
reflexão é uma reflexão; uma \omterm{pb:b1:euclid:1}{reflexão deslizante} é uma reflexão
composta com uma translação, logo três. Máximo: três.

\textbf{11.} $a = \iu$, $b = 1 + \iu$: $v = a\conj b + b =
\iu(1 - \iu) + 1 + \iu = (1 + \iu) + (1 + \iu) = 2 + 2\iu \neq
0$: uma \omterm{pb:b1:euclid:1}{reflexão deslizante} com
vetor de deslizamento $v/2 = 1 + \iu$. A reflexão $r = t_{-(1+\iu)}\circ f$ é $r(z) = \iu\conj z$, cujo eixo é a
reta no ângulo $\frac12\arg\iu = \frac\pi4$: a reta $y = x$. Logo $f$ é a \omterm{pb:b1:euclid:1}{reflexão
deslizante} de eixo $y = x$ e vetor $(1, 1)$.

\textbf{12.} $f(z) = \iu z$ e $g(z) = 1 + \iu(z - 1) = \iu z
+ 1 - \iu$, logo
\[
g\circ f(z) = \iu(\iu z) + 1 - \iu = -z + 1 - \iu :
\]
coeficiente \omterm{def:b1:linmaps:def}{linear} $-1 = \eu^{\iu\pi}$, uma rotação de ângulo $\pi$ (uma
meia-volta), de centro $z_0 = \frac{1 - \iu}{1 - (-1)}
= \frac{1 - \iu}{2}$.

\textbf{13.} $r_2\circ r_1(z) = \iu\,\conj{\conj z} = \iu z$: a rotação de ângulo $\frac\pi2$ em torno de $0$. Os
eixos (o eixo real, ângulo $0$; a reta $y = x$, ângulo $\frac\pi4$)
encontram-se num ângulo $\frac\pi4$, e o dobro disso é $\frac\pi2$: questão 9
confirmada.

\textbf{14.} Com $f = t_a\circ g$, $g$ \omterm{def:b1:linmaps:def}{linear}, e $\sum_i
\lambda_i = 1$:
\[
f\Bigl(\sum_i \lambda_i x_i\Bigr)
= a + \sum_i \lambda_i\,g(x_i)
= \sum_i \lambda_i\bigl(a + g(x_i)\bigr)
= \sum_i \lambda_i f(x_i) .
\]

\textbf{15.} Para $h \in G$, usando a questão 14 (os coeficientes
$\frac1{\abs G}$ somam $1$):
\[
h(c) = \frac{1}{\abs G}\sum_{g \in G} h\bigl(g(x_0)\bigr)
= \frac{1}{\abs G}\sum_{g' \in G} g'(x_0) = c ,
\]
pois $g \mapsto hg$ é uma bijeção de $G$ sobre si mesmo.

\textbf{16.} Os conjugados $t_{-c}\,g\,t_{c}$ ($g \in G$) formam um \omterm{def:b1:structures:group}{grupo} de
\omterm{def:b1:euclid:isometry}{isometrias} que fixam $0$, logo de \omterm{def:b1:euclid:isometry}{isometrias} \emph{\omterm{def:b1:linmaps:def}{lineares}}
(questão 3): um \omterm{def:b1:structures:subgroup}{subgrupo} finito de $O(\R^2)$. Se algum $s \in G$ é
indireto, a \omterm{def:b1:logic:map}{aplicação} $g \mapsto sg$ leva $G^+$ injetivamente em $G\setminus G^+$ e
$h \mapsto s^{-1}h$ é a sua inversa (regra de sinais da questão 5): $\abs G = 2\,\abs{G^+}$; caso
contrário $G = G^+$.

\textbf{17.} Se $G^{+} = \{\mathrm{id}\}$, ele é $C_1$. Caso contrário, escreva os seus
elementos como $R_\theta$, $\theta \in
\intco{0}{2\pi}$, e seja $\theta_0$ o menor ângulo positivo
que ocorre. Para $R_\theta \in G^{+}$, a divisão euclidiana $\theta = k\theta_0 + \rho$ com $0 \leq \rho < \theta_0$ dá
$R_\rho = R_\theta R_{\theta_0}^{-k} \in G^{+}$, logo $\rho = 0$ por minimalidade: $G^{+} = \langle R_{\theta_0}\rangle$. Dividir $2\pi$ por $\theta_0$
do mesmo modo mostra que $2\pi = n\theta_0$: $G^{+} \cong C_n$, gerado pela rotação de
ângulo $\frac{2\pi}n$.

\textbf{18.} $G = G^{+} \cup sG^{+}$ pela contagem da questão 16. Com $s(z) = a\conj z$ e $r(z) = \omega z$
(\omterm{def:b1:linmaps:forms}{formas lineares}, depois da questão 16; $\abs a = \abs\omega = 1$):
\[
s r s(z) = a\,\conj{\omega\,a\conj z}
= a\conj\omega\conj a\,z = \conj\omega\,z = r^{-1}(z) .
\]
Cada elemento $sr^k$ de $sG^{+}$ é indireto e satisfaz $(sr^k)^2 = (sr^ks)\,r^k = r^{-k}r^k = \mathrm{id}$: uma
\omterm{def:b1:euclid:isometry}{isometria} indireta involutiva que fixa $c$, isto é, uma reflexão.
Logo $G$ consiste em $n$ rotações e $n$ reflexões, com as relações
$r^n = s^2 = \mathrm{id}$, $srs = r^{-1}$: o \omterm{pb:b1:euclid:1}{grupo diedral} $D_n$.

\textbf{19.} Reunindo: um \omterm{def:b1:structures:group}{grupo} finito de \omterm{def:b1:euclid:isometry}{isometrias} do plano fixa
um ponto $c$ (questão 15), reduz-se a um \omterm{def:b1:structures:subgroup}{subgrupo} finito de $O(2)$
(questão 16), e é $C_n$ se contém apenas rotações (questão 17), e
$D_n$ caso contrário (questão 18): o \emph{teorema de Leonardo}.

\textbf{20.} Uma translação $t_v \neq \mathrm{id}$ tem potências $t_{kv}$, todas
distintas ($kv$ dois a dois distintos para $v \neq 0$): ordem infinita.
Uma \omterm{pb:b1:euclid:1}{reflexão deslizante} $f$ tem $f^2 = t_v$ com $v
\neq 0$: de novo ordem
infinita. Nenhuma das duas cabe num \omterm{def:b1:structures:group}{grupo} finito --- coerentemente
com as questões 15--19, que produziram apenas rotações e reflexões.

\textbf{21.} As $2n$ aplicações preservam o \omterm{def:b1:logic:sets}{conjunto} dos vértices:
$\omega^k
\omega^j = \omega^{k+j}$ e $\omega^k\conj{\omega^j} =
\omega^{k-j}$; sendo \omterm{def:b1:euclid:isometry}{isometrias}, elas preservam o polígono
(segmentos vão em segmentos). Reciprocamente, uma simetria preserva
o baricentro $0$ dos vértices (questão 14), logo é \omterm{def:b1:linmaps:def}{linear}, e permuta
os vértices (que são os pontos do polígono a distância máxima de
$0$). Uma \omterm{def:b1:euclid:isometry}{isometria} \omterm{def:b1:linmaps:def}{linear} que leva o vértice $1$ em $\omega^k$ é $z \mapsto \omega^k z$
se direta, e $z \mapsto \omega^k\conj z$ se indireta (a sua matriz é determinada por uma
coluna e pelo sinal): no máximo $2n$ simetrias, logo exatamente o
$D_n$ acima.

\textbf{22.} $C_1$: um triângulo escaleno (nenhuma simetria não
trivial). $D_1$: um triângulo isósceles não equilátero (uma
reflexão). $D_2$: um retângulo não quadrado (identidade, meia-volta
em torno do centro, as duas reflexões nos eixos). $C_3$: um trisquel
--- três braços dobrados congruentes fixados a \omterm{prop:b1:reals:intervals}{intervalos} de $120$
graus; a dobra mata toda reflexão mas deixa as rotações de ordem
$3$.

\textbf{23.} Com $\omega = \iu$: as rotações $z \mapsto z$, $\iu z$, $-z$, $-\iu z$
(ângulos $0, \frac\pi2, \pi,
\frac{3\pi}2$), e as reflexões
\[
z \mapsto \conj z \ (\text{eixo } y = 0), \quad
\iu\conj z \ (y = x), \quad
-\conj z \ (x = 0), \quad
-\iu\conj z \ (y = -x):
\]
as duas diagonais e as duas medianas do quadrado --- o \omterm{pb:b1:euclid:1}{grupo
diedral} $D_4$, de ordem $8$.

\textbf{24.} Escreva $a = \eu^{\iu\theta} \neq 1$: $f(z) = az +
c_1(1 - a)$ e $g(z) = az + c_2(1 - a)$. Então
\[
f\circ g(z) - g\circ f(z)
= (1-a)\bigl(ac_2 + c_1 - ac_1 - c_2\bigr)
= -(1 - a)^2(c_2 - c_1) \neq 0 ,
\]
logo $f g \neq g f$. As duas composições têm coeficiente \omterm{def:b1:linmaps:def}{linear} $a^2$, logo
$(fg)\circ(gf)^{-1}$ tem coeficiente \omterm{def:b1:linmaps:def}{linear} $1$: é a translação por $fg(z) - gf(z)$, a
constante não nula acima. Um \omterm{def:b1:structures:group}{grupo} que contenha $f$ e $g$ contém
essa translação e todas as suas potências: ele é infinito. Dois
centros de rotação são um a mais do que um \omterm{def:b1:structures:group}{grupo} finito suporta ---
o coração geométrico do teorema de Leonardo.

\textbf{25.} (i) As questões 3--4 mostram que toda \omterm{def:b1:logic:map}{aplicação} que
preserva distâncias é afim com parte \omterm{def:b1:linmaps:def}{linear} \omterm{def:b1:euclid:orthogonal}{ortogonal}, e a questão
15 prende todo \omterm{def:b1:structures:group}{grupo} finito a um ponto fixo: os dois resultados
convertem geometria métrica em álgebra linear na origem. (ii) A
lista completa é translação, rotação, reflexão e \omterm{pb:b1:euclid:1}{reflexão
deslizante}; o determinante da parte \omterm{def:b1:linmaps:def}{linear} separa as diretas das
indiretas, e a existência de pontos fixos separa os dois tipos
dentro de cada paridade. (iii) Pela questão 9, duas reflexões
compõem-se em qualquer rotação ou translação, logo as reflexões
geram o \omterm{def:b1:structures:group}{grupo} inteiro --- no máximo três bastam para qualquer
\omterm{def:b1:euclid:isometry}{isometria}. (iv) Na escala finita só duas famílias sobrevivem, os
\omterm{def:b1:structures:group}{grupos} cíclicos e os diedrais, e é por isso que os ornamentos em
roseta vêm em exatamente dois tipos (com ou sem eixos de espelho).
A Parte II demonstrou a \emph{classificação das \omterm{def:b1:euclid:isometry}{isometrias} do
plano}; a Parte IV demonstrou o \emph{teorema de Leonardo}.
\end{solution}
