\section*{Capítulo \ref{ch:b1:complex} --- Números Complexos}

\begin{solution}{exo:b1:complex:1}
$1 + \iu = \sqrt 2\, \eu^{\iu\pi/4}$;
$\;\sqrt 3 - \iu = 2\, \eu^{-\iu\pi/6}$;
$\;-5 = 5\, \eu^{\iu\pi}$;
\[
\frac{1 + \iu}{\sqrt 3 - \iu}
= \frac{\sqrt 2}{2}\, \eu^{\iu(\pi/4 + \pi/6)}
= \frac{\sqrt 2}{2}\, \eu^{5\iu\pi/12}.
\]
Calculando o mesmo quociente algebricamente (multiplique pelo \omterm{def:b1:complex:field}{conjugado}):
\[
\frac{(1 + \iu)(\sqrt 3 + \iu)}{4}
= \frac{(\sqrt 3 - 1) + \iu(\sqrt 3 + 1)}{4}.
\]
Identificando com $\frac{\sqrt 2}{2}(\cos\frac{5\pi}{12} +
\iu\sin\frac{5\pi}{12})$:
\[
\cos\frac{5\pi}{12} = \frac{\sqrt 6 - \sqrt 2}{4},
\qquad
\sin\frac{5\pi}{12} = \frac{\sqrt 6 + \sqrt 2}{4}.
\]
\end{solution}

\begin{solution}{exo:b1:complex:2}
$(1+\iu)^2 = 2\iu$, logo $(1+\iu)^{20} = (2\iu)^{10} = 2^{10}\,\iu^{10}
= 1024 \times (-1) = -1024$.

Na forma exponencial, $(1+\iu)^n = 2^{n/2}\, \eu^{\iu n\pi/4}$, que é real se,
e somente se, $\sin\frac{n\pi}{4} = 0$, isto é, $4 \mid n$. Logo $(1+\iu)^n
\in \R$
exatamente para os múltiplos de $4$ (com valor $(-4)^{n/4}$).
\end{solution}

\begin{solution}{exo:b1:complex:3}
$\abs{z - 2} = \abs{z + \iu}$: equidistância dos pontos $2$ e
$-\iu$ --- a mediatriz do segmento que liga $(2, 0)$
a $(0, -1)$.

$\abs{z - 1} = 2$: o círculo de centro $1$ e raio $2$.

$\frac{z-1}{z+1} \in \iu\R$: pela
\cref{prop:b1:complex:geometry}~(3), os pontos $M(z)$, $A(1)$,
$B(-1)$ satisfazem: as retas $MA$ e $MB$ são perpendiculares (ou $z = 1$,
caso em que o quociente é $0 \in \iu\R$). O \omterm{def:b1:logic:sets}{conjunto} é o círculo de
diâmetro $[-1, 1]$ (o círculo unitário), menos o ponto $-1$, em que o
quociente não está definido. \emph{Verificação por cálculo:} $z = \eu^{\iu\theta}$ dá
$\frac{z-1}{z+1} = \frac{\eu^{\iu\theta/2}(\eu^{\iu\theta/2} -
\eu^{-\iu\theta/2})}{\eu^{\iu\theta/2}(\eu^{\iu\theta/2} +
\eu^{-\iu\theta/2})} = \iu\tan\frac\theta2 \in \iu\R$.
\end{solution}

\begin{solution}{exo:b1:complex:4}
Linearização:
\[
\sin^4\theta
= \Bigl(\frac{\eu^{\iu\theta} - \eu^{-\iu\theta}}{2\iu}\Bigr)^{\!4}
= \frac{\eu^{4\iu\theta} - 4\eu^{2\iu\theta} + 6 - 4\eu^{-2\iu\theta}
+ \eu^{-4\iu\theta}}{16}
= \frac{\cos 4\theta - 4\cos 2\theta + 3}{8}.
\]
Expansão: por De Moivre, $\cos 4\theta = \Re\bigl((c +
\iu s)^4\bigr) = c^4 - 6c^2 s^2 + s^4$ com $c = \cos\theta$, $s =
\sin\theta$; substituindo $s^2 = 1 - c^2$:
\[
\cos 4\theta = c^4 - 6c^2(1 - c^2) + (1 - c^2)^2
= 8c^4 - 8c^2 + 1 .
\]
\end{solution}

\begin{solution}{exo:b1:complex:5}
$8\iu = 8\,\eu^{\iu\pi/2}$, de modo que as raízes cúbicas são $2\,\eu^{\iu(\pi/6 +
2k\pi/3)}$, $k = 0, 1, 2$:
\[
z_0 = 2\eu^{\iu\pi/6} = \sqrt 3 + \iu,\quad
z_1 = 2\eu^{5\iu\pi/6} = -\sqrt 3 + \iu,\quad
z_2 = 2\eu^{3\iu\pi/2} = -2\iu :
\]
um triângulo equilátero no círculo de raio $2$.

$-4 = 4\eu^{\iu\pi}$, de modo que $z^4 = -4$ tem soluções $\sqrt 2\,
\eu^{\iu(\pi/4 + k\pi/2)}$: $\;1 + \iu$, $-1 + \iu$, $-1 - \iu$,
$1 - \iu$. Emparelhando as raízes conjugadas:
\[
X^4 + 4 = \bigl(X^2 - 2X + 2\bigr)\bigl(X^2 + 2X + 2\bigr),
\]
pois $(X - (1+\iu))(X - (1-\iu)) = X^2 - 2X + 2$, e analogamente para
o outro par. (Expanda para conferir.)
\end{solution}

\begin{solution}{exo:b1:complex:6}
$C_n + \iu S_n = \sum_{k=0}^{n} \eu^{\iu k\theta} =
\frac{\eu^{\iu(n+1)\theta} - 1}{\eu^{\iu\theta} - 1}$ (soma geométrica,
de razão $\eu^{\iu\theta} \neq 1$). Fatoração do arco metade
(\cref{met:b1:complex:trig}~(4)) no numerador e no denominador:
\[
\frac{2\iu\sin\frac{(n+1)\theta}{2}\, \eu^{\iu(n+1)\theta/2}}
{2\iu\sin\frac{\theta}{2}\, \eu^{\iu\theta/2}}
= \frac{\sin\frac{(n+1)\theta}{2}}{\sin\frac{\theta}{2}}\,
\eu^{\iu n\theta/2}.
\]
Tomando as partes real e imaginária:
\[
C_n = \frac{\sin\frac{(n+1)\theta}{2}}{\sin\frac{\theta}{2}}
\cos\frac{n\theta}{2},
\qquad
S_n = \frac{\sin\frac{(n+1)\theta}{2}}{\sin\frac{\theta}{2}}
\sin\frac{n\theta}{2}.
\]
\end{solution}

\begin{solution}{exo:b1:complex:7}
Discriminante: $\Delta = (3 + 4\iu)^2 - 4(-1 + 5\iu) = 9 + 24\iu - 16 +
4 - 20\iu = -3 + 4\iu$. Raízes quadradas de $-3 + 4\iu$: resolva $x^2 - y^2
= -3$, $2xy = 4$, $x^2 + y^2 = 5$; então $x^2 = 1$, $y = 2/x$, com os mesmos
sinais: $\delta = \pm(1 + 2\iu)$. Portanto
\[
z = \frac{(3 + 4\iu) \pm (1 + 2\iu)}{2}
\in \{\, 2 + 3\iu,\; 1 + \iu \,\}.
\]
\emph{Verificação:} a soma vale $= 3 + 4\iu$ e o produto $(2+3\iu)(1+\iu) = -1 +
5\iu$, como os coeficientes exigem.
\end{solution}

\begin{solution}{exo:b1:complex:8}
\begin{enumerate}
  \item \cref{prop:b1:complex:sumroots} com $n = 5$.
  \item $u + v = \omega + \omega^2 + \omega^3 + \omega^4 = -1$, por (1).
        Para o produto, expanda e reduza os expoentes módulo $5$:
        \[
        uv = (\omega + \omega^4)(\omega^2 + \omega^3)
        = \omega^3 + \omega^4 + \omega^6 + \omega^7
        = \omega^3 + \omega^4 + \omega + \omega^2 = -1 .
        \]
        Assim, $u$ e $v$ têm soma $-1$ e produto $-1$: são as
        duas raízes de $X^2 + X - 1$.
  \item $u = \omega + \conj\omega = 2\cos\frac{2\pi}{5} > 0$ (o
        ângulo é agudo), e a raiz positiva de $X^2 + X - 1$ é
        $\frac{-1 + \sqrt 5}{2}$. Portanto, $\cos\frac{2\pi}{5} =
        \frac{\sqrt 5 - 1}{4}$.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:complex:9}
Expanda os dois quadrados como na demonstração da
\cref{prop:b1:complex:rules}~(4):
\[
\abs{z + w}^2 = \abs z^2 + \abs w^2 + 2\Re(z\conj w),
\qquad
\abs{z - w}^2 = \abs z^2 + \abs w^2 - 2\Re(z\conj w),
\]
e some. Geometricamente, $\abs{z+w}$ e $\abs{z-w}$ são os comprimentos das
duas diagonais do paralelogramo de vértices $0, z, z+w, w$,
ao passo que $\abs z$ e $\abs w$ são os comprimentos dos lados: a soma
dos quadrados das diagonais é igual à soma dos quadrados dos quatro
lados.
\end{solution}

\begin{solution}{exo:b1:complex:10}
Seja $j = \eu^{2\iu\pi/3}$, de modo que $j^2 + j + 1 = 0$ e $\eu^{\iu\pi/3} =
-j^2$, $\eu^{-\iu\pi/3} = -j$. O triângulo é equilátero direto se, e
somente se, $c - a = -j^2 (b - a)$, e indireto se, e somente se, $c - a = -j(b - a)$
(\cref{ex:b1:complex:equilateral}).

Considere $P = a + jb + j^2 c$ e $Q = a + j^2 b + jc$. Usando $1 +
j^2 = -j$ e $j^3 = 1$:
\[
c - a + j^2(b - a) = c + j^2 b - (1 + j^2)\,a = c + j^2 b + ja
= j\,(a + jb + j^2 c) = jP,
\]
de modo que o caso direto diz $jP = 0 \iff P = 0$; o mesmo cálculo
com $j$ no lugar de $j^2$ dá $c - a + j(b - a) = j^2 Q$, de sorte que o
caso indireto diz $Q = 0$. Portanto: equilátero (em qualquer orientação)
$\iff PQ = 0$. Expandindo, com $j + j^2 = -1$:
\[
PQ = a^2 + b^2 + c^2 + (j + j^2)(ab + bc + ca)
= a^2 + b^2 + c^2 - (ab + bc + ca).
\]
Assim, o triângulo é equilátero se, e somente se, $a^2 + b^2 + c^2 = ab +
bc + ca$.
\end{solution}

\begin{solution}{exo:b1:complex:11}
Como os $\omega^k$, $k = 0, \dots, n-1$, são exatamente as $n$ raízes
de $X^n - 1$ (\cref{thm:b1:complex:roots}), e o polinômio é
mônico:
\[
X^n - 1 = \prod_{k=0}^{n-1} (X - \omega^k)
= (X - 1) \prod_{k=1}^{n-1} (X - \omega^k).
\]
Dividindo por $X - 1$: $\;1 + X + \dots + X^{n-1} = \prod_{k=1}^{n-1}
(X - \omega^k)$. Avaliando em $X = 1$, obtém-se $P = n$.

Agora, $1 - \omega^k = -\eu^{\iu k\pi/n}\bigl(\eu^{\iu k\pi/n} -
\eu^{-\iu k\pi/n}\bigr) = -2\iu\,\eu^{\iu k\pi/n} \sin\frac{k\pi}{n}$,
de modo que, tomando \omterm{def:b1:complex:field}{módulos} em $P = n$ (cada $\sin\frac{k\pi}n > 0$ para $1 \leq k
\leq n-1$):
\[
n = \abs P = \prod_{k=1}^{n-1} 2\sin\frac{k\pi}{n}
= 2^{n-1} \prod_{k=1}^{n-1} \sin\frac{k\pi}{n},
\qquad\text{logo}\qquad
\prod_{k=1}^{n-1} \sin\frac{k\pi}{n} = \frac{n}{2^{n-1}} .
\]
\end{solution}

\begin{solution}{exo:b1:complex:12}
$z = 1$ não é solução ($2^n \neq 0$), de modo que a equação é
equivalente a $\bigl(\frac{z+1}{z-1}\bigr)^n = 1$, isto é,
$\frac{z+1}{z-1} = \omega^k$ com $\omega = \eu^{2\iu\pi/n}$ e
$k \in \{0, \dots, n-1\}$. O valor $k = 0$ está excluído ($z + 1 =
z - 1$ é impossível). Para $1 \leq k \leq n - 1$, resolver
$z + 1 = \omega^k(z - 1)$ dá $z(1 - \omega^k) = -1 - \omega^k$,
de modo que, com $\varphi = \frac{2k\pi}{n}$ e as fatorações do arco
metade $1 + \eu^{\iu\varphi} = 2\cos\frac\varphi2\,
\eu^{\iu\varphi/2}$, $\eu^{\iu\varphi} - 1 = 2\iu\sin\frac\varphi2
\,\eu^{\iu\varphi/2}$:
\[
z_k = \frac{1 + \omega^k}{\omega^k - 1}
= \frac{2\cos\frac{k\pi}{n}}{2\iu\,\sin\frac{k\pi}{n}}
= -\iu\,\frac{\cos\frac{k\pi}{n}}{\sin\frac{k\pi}{n}} ,
\]
puramente imaginário, como afirmado. A \omterm{def:b1:logic:map}{aplicação} $t \mapsto \cos t/\sin t$ é
\omterm{def:b1:logic:inj}{injetiva} em $\intoo0\pi$ (é estritamente decrescente), de modo que os
$n - 1$ valores $z_k$ são dois a dois distintos: a equação, de grau
$n - 1$ uma vez expandida (os termos $z^n$ se cancelam), tem exatamente
essas $n - 1$ soluções.
\end{solution}

\begin{solution}{pb:b1:complex:1}
\textbf{1.} $j = \cos\frac{2\pi}3 + \iu\sin\frac{2\pi}3 = -\frac12
+ \iu\frac{\sqrt3}2$; $j^2 = \eu^{4\iu\pi/3} = -\frac12 -
\iu\frac{\sqrt3}2 = \conj j$; $j^3 = 1$; $1 + j + j^2 = 0$ (soma das
raízes cúbicas da unidade, \cref{prop:b1:complex:sumroots}); $j^{-1}
= j^2$
(pois $j \cdot j^2 = 1$). No círculo unitário, $1$, $j$,
$j^2$ são os vértices de um triângulo equilátero direto.

\textbf{2.} A rotação de centro $a$ e ângulo $\theta$ fixa $a$
e gira de $\theta$ todo vetor com origem em $a$: $r(z) - a =
\eu^{\iu\theta}(z - a)$, isto é, $r(z) = a + \eu^{\iu\theta}(z - a)$.
Compondo $r(z) = a + \eu^{\iu\theta}(z-a)$ e $r'(z) = a' +
\eu^{\iu\theta'}(z-a')$:
\[
r' \circ r\,(z) = \eu^{\iu(\theta + \theta')} z +
\text{constante},
\]
uma \omterm{def:b1:logic:map}{aplicação} da forma $Az + B$ com $A = \eu^{\iu(\theta+\theta')}$ de
módulo $1$. Pela \cref{prop:b1:complex:similitude}, trata-se de uma rotação
de ângulo $\arg A = \theta + \theta'$ se $A \neq 1$, e de uma
translação se $A = 1$, isto é, se $\theta + \theta' \in 2\pi\Z$.

\textbf{3.} $(b, c, p)$ é equilátero se, e somente se, $\abs{p - b} = \abs{c -
b} = \abs{p - c}$. Escrevendo $q = \frac{p - b}{c - b}$, a primeira
igualdade diz que $\abs q = 1$ e a segunda que $\abs{q - 1} = 1$;
juntas, $q = \eu^{\pm\iu\pi/3}$ (os dois pontos de interseção dos
círculos $\abs q = 1$ e $\abs{q - 1} = 1$ são $\frac12 \pm
\iu\frac{\sqrt3}2$). Portanto, $p = b + \eu^{\pm\iu\pi/3}(c - b)$. Para
$a = 0$, $b = 1$, $c = \iu$ (um triângulo direto): a escolha
$\eu^{-\iu\pi/3}$ dá
\[
p = 1 + \Bigl(\tfrac12 - \iu\tfrac{\sqrt3}2\Bigr)(\iu - 1)
= \tfrac{1 + \sqrt3}2\,(1 + \iu) \approx 1.37 + 1.37\,\iu ,
\]
que fica do lado da reta $BC$ ($x + y = 1$) oposto a $a
= 0$: para fora. A escolha $\eu^{+\iu\pi/3}$ dá $p \approx -0.37
- 0.37\,\iu$, do mesmo lado que $a$: para dentro.

\textbf{4.} Multiplique $P = a + jb + j^2c$ por $j^2$: $j^2 P = j^2 a +
j^3 b + j^4 c = b + jc + j^2 a$ (usando $j^3 = 1$, $j^4 = j$); e
por $j$: $jP = ja + j^2 b + c$. Estas são as duas identidades. Se $P
= 0$, então $j^2 P = jP = 0$: o critério vale também para $(b, c, a)$ e
$(c, a, b)$ --- invariância cíclica (como deve ser: um triângulo
equilátero não se importa com qual vértice é listado primeiro).

\textbf{5.} Se $a = b$: $0 = a(1 + j) + j^2 c = -j^2 a + j^2 c$
(usando $1 + j = -j^2$), logo $c = a$. Se $b = c$: $0 = a + b(j + j^2)
= a - b$, logo $a = b$. Se $a = c$: $0 = a(1 + j^2) + jb = -ja + jb$,
logo $a = b$. Em cada caso, os três pontos coincidem.

\textbf{6.} $(a'z + b') \circ (az + b) = a'a\,z + (a'b + b')$ com
$a'a \neq 0$: mesma forma. A inversa de $z \mapsto az + b$ é $z
\mapsto \frac1a z - \frac ba$, de novo da mesma forma. Com a
\omterm{def:b1:logic:map}{aplicação} identidade como elemento neutro, as semelhanças diretas formam
um grupo com a composição.

\textbf{7.} $f(z) = az + b$ satisfaz $f(z_1) = w_1$, $f(z_2) =
w_2$ se, e somente se, $a z_1 + b = w_1$ e $a z_2 + b = w_2$; subtraindo,
$a(z_2 - z_1) = w_2 - w_1$, de modo que
\[
a = \frac{w_2 - w_1}{z_2 - z_1} \;(\neq 0), \qquad
b = w_1 - a z_1
\]
ficam forçados e, reciprocamente, essa escolha funciona: existência e
unicidade.

\textbf{8.} $\abs{f(z) - f(w)} = \abs{a(z - w)} = \abs a\,\abs{z -
w}$: todas as distâncias são multiplicadas por $\abs a$. Quanto à forma:
\[
\frac{f(c') - f(a')}{f(b') - f(a')} = \frac{a(c' -
a')}{a(b' - a')} = \frac{c' - a'}{b' - a'} .
\]
Se dois triângulos $(a', b', c')$ e $(a'', b'', c'')$ têm formas
iguais, seja $f$ a única semelhança direta com $f(a') = a''$
e $f(b') = b''$ (questão 7); então a forma de $(a'', b'',
f(c'))$
é igual à de $(a', b', c')$ e, portanto, à de $(a'', b'',
c'')$, e a forma determina o terceiro vértice a partir dos dois
primeiros: $f(c') = c''$. Reciprocamente, a igualdade das formas
decorre da invariância exibida.

\textbf{9.} $b = f(0) = 1$; $a + b = f(1) = \iu$ dá $a = \iu -
1$. Razão $\abs a = \sqrt2$, ângulo $\arg(\iu - 1) =
\frac{3\pi}4$. Ponto fixo:
\[
\zeta = \frac{b}{1 - a} = \frac1{2 - \iu} = \frac{2 + \iu}5 .
\]
Assim, $f$ é a semelhança direta de centro $\frac{2+\iu}5$, razão
$\sqrt2$ e ângulo $\frac{3\pi}4$.

\textbf{10.} Com $\alpha + \beta = 1$ e $f(z) = az + b$:
\[
f(\alpha a' + \beta b') = a\alpha a' + a\beta b' + b
= \alpha(a a' + b) + \beta(a b' + b)
= \alpha f(a') + \beta f(b') ,
\]
sendo a chave $b = (\alpha + \beta)b$. Os pontos médios ($\alpha = \beta =
\frac12$), os baricentros (itere) e os centros de Napoleão $\alpha b
+ \beta c$
abaixo são, portanto, \emph{equivariantes}: transformar o
triângulo os transforma na mesma medida. Assim, demonstrar um enunciado
invariante por semelhança para um triângulo bem posicionado demonstra-o
para todos --- a licença usada na questão 21.

\textbf{11.} O ápice externo sobre $[b, c]$ é $p_a = b +
\eu^{-\iu\pi/3}(c - b)$ (questão 3), de modo que o centro é
\[
n_a = \frac{b + c + p_a}3
= \frac{2b + c + \frac{1 - \iu\sqrt3}2\,(c - b)}3
= \frac{(3 + \iu\sqrt3)\,b + (3 - \iu\sqrt3)\,c}6
= \alpha b + \beta c ,
\]
com $\alpha = \frac{3 + \iu\sqrt3}6$ e $\beta = \frac{3 -
\iu\sqrt3}6 = \conj\alpha$. O mesmo cálculo sobre os lados
$[c, a]$ e $[a, b]$ dá $n_b = \alpha c + \beta a$ e $n_c =
\alpha a + \beta b$ (deslocamento cíclico dos papéis).

\textbf{12.} $\alpha + \beta = \frac{3 + \iu\sqrt3 + 3 -
\iu\sqrt3}6 = 1$. Portanto,
\[
\frac{n_a + n_b + n_c}3
= \frac{(\alpha + \beta)(a + b + c)}3 = \frac{a + b + c}3 :
\]
os dois triângulos têm o mesmo baricentro.

\textbf{13.} Agrupe por $\alpha$ e $\beta$ e aplique as identidades da
questão 4 a $P = a + jb + j^2c$:
\[
n_a + j n_b + j^2 n_c
= \alpha\,(b + jc + j^2 a) + \beta\,(c + ja + j^2 b)
= \alpha\,j^2 P + \beta\,j P
= (\alpha j^2 + \beta j)\,P .
\]

\textbf{14.} Com $j = \frac{-1 + \iu\sqrt3}2$:
\[
\alpha j = \frac{(3 + \iu\sqrt3)(-1 + \iu\sqrt3)}{12}
= \frac{-3 + 3\iu\sqrt3 - \iu\sqrt3 - 3}{12}
= \frac{-3 + \iu\sqrt3}6 = -\beta ,
\]
logo $\alpha j + \beta = 0$, e portanto $\alpha j^2 + \beta j = j(\alpha j
+ \beta) = 0$, e a questão 13 dá $n_a + jn_b + j^2n_c = 0$ para
todo triângulo $(a, b, c)$. Pelo critério (questões 4--5), os
centros formam um triângulo equilátero direto ou um único ponto:
o teorema de Napoleão.

\textbf{15.} O ápice interno é $b + \eu^{+\iu\pi/3}(c - b)$, e
o cálculo da questão 11 com $\eu^{+\iu\pi/3} = \frac{1 +
\iu\sqrt3}2$ troca $\alpha$ por $\beta$: $m_a = \beta b + \alpha
c$, $m_b = \beta c + \alpha a$, $m_c = \beta a + \alpha b$. Usando
as identidades análogas $b + j^2c + ja = j\,Q$ e $c + j^2a + jb
= j^2 Q$ para $Q = a + j^2b + jc$:
\[
m_a + j^2 m_b + j m_c
= \beta(b + j^2 c + ja) + \alpha(c + j^2 a + jb)
= (\beta j + \alpha j^2)\, Q = j(\beta + \alpha j)\,Q = 0 ,
\]
pois $\alpha j = -\beta$ (questão 14). Assim, os centros internos
satisfazem o critério equilátero \emph{indireto}: equiláteros com
a orientação oposta (ou um ponto).

\textbf{16.} Pela questão 5 aplicada à tripla $(n_a, n_b, n_c)$
(que satisfaz o critério direto), dois centros coincidem se, e somente
se, os três coincidem; e os três coincidem se, e somente se, \emph{ambos}
os critérios valem, isto é, se, adicionalmente, $n_a + j^2 n_b + j n_c = 0$. Calcule como na
questão 13, com as identidades $b + j^2c + ja = jQ$, $c + j^2a +
jb = j^2Q$:
\[
n_a + j^2 n_b + j n_c
= \alpha\,jQ + \beta\,j^2 Q = j(\alpha + \beta j)\,Q .
\]
Diretamente: $\beta j = \frac{(3 -
\iu\sqrt3)(-1 + \iu\sqrt3)}{12} = \frac{-3 + 3\iu\sqrt3 +
\iu\sqrt3 + 3}{12} = \frac{\iu\sqrt3}3$, de modo que $\alpha + \beta j =
\frac{3 + \iu\sqrt3 + 2\iu\sqrt3}6 = \frac{1 + \iu\sqrt3}2 \neq
0$. Portanto, o triângulo externo de Napoleão degenera se, e
somente se, $Q = a +
j^2b + jc = 0$, isto é, se, e somente se, $(a, b, c)$ é um triângulo equilátero indireto --- caso em que as
construções ``para fora'' apontam todas para a região circunscrita ao
triângulo e partilham um mesmo centro.

\textbf{17.} Com $a = 0$, $b = 1$, $c = \iu$:
\[
n_a = \alpha + \beta\iu = \frac{(3 + \sqrt3)(1 + \iu)}6,
\qquad
n_b = \alpha\iu = \frac{-\sqrt3 + 3\iu}6,
\qquad
n_c = \beta = \frac{3 - \iu\sqrt3}6 .
\]
Lados ao quadrado: $n_a - n_b = \frac{(3 + 2\sqrt3) +
\iu\sqrt3}6$ dá $\abs{n_a - n_b}^2 = \frac{(3 + 2\sqrt3)^2 +
3}{36} = \frac{24 + 12\sqrt3}{36} = \frac{2 + \sqrt3}3$; $n_b -
n_c = \frac{(3 + \sqrt3)(-1 + \iu)}6$ dá $\abs{n_b - n_c}^2 =
\frac{2(3 + \sqrt3)^2}{36} = \frac{24 + 12\sqrt3}{36}$; $n_c - n_a
= \frac{-\sqrt3 - \iu(3 + 2\sqrt3)}6$ dá de novo o mesmo valor.
Os três lados ao quadrado valem $\frac{2 + \sqrt3}3$: equilátero,
como prometido.

\textbf{18.} Expandindo:
\[
(a - b)(c - d) + (a - d)(b - c)
= (ac - ad - bc + bd) + (ab - ac - bd + cd)
= ab - ad - bc + cd ,
\]
e $(a - c)(b - d) = ab - ad - bc + cd$: iguais.

\textbf{19.} Tome \omterm{def:b1:complex:field}{módulos} na questão 18 e aplique a desigualdade
triangular (\cref{prop:b1:complex:rules}~(4)):
\[
AC \cdot BD = \abs{(a-b)(c-d) + (a-d)(b-c)}
\leq \abs{a-b}\,\abs{c-d} + \abs{a-d}\,\abs{b-c}
= AB \cdot CD + AD \cdot BC ,
\]
com igualdade se, e somente se, as duas parcelas estão numa mesma
semirreta com origem em $0$ (o caso de igualdade da desigualdade
triangular).

\textbf{20.} Fatoração do arco metade
(\cref{met:b1:complex:trig}~(4)):
$\eu^{\iu\theta} - \eu^{\iu\varphi} = 2\iu\,\sin\frac{\theta -
\varphi}2\;\eu^{\iu(\theta + \varphi)/2}$, e tomar \omterm{def:b1:complex:field}{módulos} mata
os fatores de módulo $1$: $\abs{\eu^{\iu\theta} - \eu^{\iu\varphi}}
= 2\,\abs{\sin\frac{\theta - \varphi}2}$.

\textbf{21.} Com $p = \eu^{\iu\theta}$, $\theta \in
\intoo{2\pi/3}{4\pi/3}$, a questão 20 dá
\[
PA = 2\,\abs{\sin\tfrac\theta2},
\quad
PB = 2\,\abs{\sin\bigl(\tfrac\theta2 - \tfrac\pi3\bigr)},
\quad
PC = 2\,\abs{\sin\bigl(\tfrac\theta2 - \tfrac{2\pi}3\bigr)} .
\]
No arco, $\frac\theta2 \in \intoo{\pi/3}{2\pi/3}$: então
$\sin\frac\theta2 > 0$; $\frac\theta2 - \frac\pi3 \in
\intoo0{\pi/3}$, de modo que o segundo seno é positivo; $\frac\theta2 -
\frac{2\pi}3 \in \intoo{-\pi/3}0$, de modo que o terceiro é negativo e
$PC = 2\sin\bigl(\frac{2\pi}3 - \frac\theta2\bigr)$. Transformando soma em
produto:
\[
\sin\Bigl(\frac\theta2 - \frac\pi3\Bigr) +
\sin\Bigl(\frac{2\pi}3 - \frac\theta2\Bigr)
= 2\,\sin\frac\pi6\,\cos\Bigl(\frac\theta2 - \frac\pi2\Bigr)
= \sin\frac\theta2 ,
\]
logo $PB + PC = 2\sin\frac\theta2 = PA$: o teorema de van Schooten.

\textbf{22.} Em $p = -1$: $PA = 2$, $PB = PC = 1$ e todos os lados do
triângulo equilátero têm comprimento $\sqrt3$, de modo que os dois lados
de Ptolomeu valem $2\sqrt3$. Algebricamente, usando $-1 - j^2 = j$ e
$1 + j = -j^2$:
\[
(a - b)(p - c) = (1 - j)\,(-1 - j^2) = (1 - j)j = j - j^2
= \iu\sqrt3 ,
\]
\[
(a - c)(b - p) = (1 - j^2)(j + 1) = 1 + j - j^2 - j^3
= j - j^2 = \iu\sqrt3 .
\]
Os dois termos são iguais, logo a sua razão é
$1 \in \R_{>0}$: o caso de igualdade da questão 19, correspondendo
exatamente a $PA
\cdot BC = PB \cdot AC + PC \cdot AB$.

\textbf{23.} Para $(a, b, c) = (1, j, j^2)$: $n_a = \alpha j +
\beta j^2 = -\beta + \beta j^2$ (questão 14) $= \beta(j^2 - 1)$;
numericamente, $\beta(j^2 - 1) = \frac{(3 - \iu\sqrt3)}6 \cdot
\bigl(-\frac32 - \iu\frac{\sqrt3}2\bigr) = -1$. Do mesmo modo, $n_b =
\alpha j^2 + \beta = -j$ e $n_c = \alpha + \beta j = -j^2$. O
triângulo externo de Napoleão de $(1, j, j^2)$ é $(-1, -j, -j^2)$: o
triângulo original refletido através do seu baricentro $0$ --- mesmo
tamanho, girado de meia volta. Um triângulo equilátero é uma forma fixa
da construção de Napoleão, e não um limite que encolhe.

\textbf{24.} (i) A multiplicação por um número de módulo $1$ como
rotação construiu os ápices e os centros (questões 2--3 e 11) e converteu
distâncias no círculo em senos (questão 20). (ii) A estrutura de
grupo e o invariante de forma justificaram normalizar o
círculo circunscrito ao círculo unitário e o triângulo a $(1, j, j^2)$
na questão 21, via a equivariância da questão 10. (iii) A
fatoração do arco metade alimentou tanto a questão 20 quanto o
passo de soma em produto que encerra van Schooten.

\textbf{25.} O dicionário converte enunciados geométricos em
identidades polinomiais nas quais toda hipótese é uma equação:
o que ele compra é mecanização --- o teorema de Napoleão reduzido a
$\alpha j + \beta = 0$, uma linha de aritmética em
$\Q(\iu\sqrt3)$; o que ele custa é visibilidade geométrica --- o
cálculo certifica, mas não mostra \emph{por que} os centros
se fecham num triângulo equilátero. Uma resposta justa é que a
identidade explica a \emph{inevitabilidade} do teorema (ela vale
identicamente em $a, b, c$, de modo que nenhuma esperteza de
configuração está em jogo), ao passo que a figura explica o seu
\emph{conteúdo}. O mesmo dicionário volta em forma matricial quando as
rotações se tornam as matrizes ortogonais $2 \times 2$ do \cref{ch:b1:matrices} e
do \cref{ch:b1:euclid}, em que ``multiplicar por
$\eu^{\iu\theta}$'' é o protótipo de uma isometria linear.
\end{solution}
