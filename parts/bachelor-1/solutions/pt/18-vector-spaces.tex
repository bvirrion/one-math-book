\begin{solution}{exo:b1:vspaces:1}
\begin{enumerate}
  \item Sim: contém $0$, e a equação que o define é linear
        (estável por $x + \lambda y$).
  \item Não: não contém $(0,0,0)$.
  \item Não: $(1, 0)$ e $(0, -1)$ pertencem ($xy = 0$), mas a sua

\begin{solution}{exo:b1:vspaces:2}
$(1,2,1) = a(1,0,1) + b(1,1,0)$ exige $a + b = 1$, $b = 2$, $a =
1$: incompatível ($a + b = 3 \neq 1$): não está no \omterm{def:b1:vspaces:span}{espaço gerado}. $(2,1,1) =
a(1,0,1) + b(1,1,0)$:
$b = 1$, $a = 1$, $a + b = 2$: compatível, logo
$(2,1,1) = (1,0,1) + (1,1,0)$, está no \omterm{def:b1:vspaces:span}{espaço gerado}.

Equação: $(x, y, z) = (a + b, b, a)$ significa $x = y + z$: o \omterm{def:b1:vspaces:span}{espaço gerado} é
o plano $\{x - y - z = 0\}$.
\end{solution}

\begin{solution}{exo:b1:vspaces:3}
Primeira família: $\lambda(1,1,0) + \mu(1,0,1) + \nu(0,1,1) = 0$ dá
$\lambda + \mu = 0$, $\lambda + \nu = 0$, $\mu + \nu = 0$: somando,
$2(\lambda + \mu + \nu) = 0$, e subtraindo cada equação,
$\lambda = \mu = \nu = 0$: \omterm{def:b1:vspaces:free}{livre}.

Segunda: $(2,4,6) = 2(1,2,3)$: linearmente dependente.

Terceira: quatro vetores em $\R^3$ --- necessariamente dependentes
assim que a dimensão estiver disponível (\cref{ch:b1:findim}); diretamente:
$(0,1,1) = -(1,0,0) +
0\cdot(1,1,0) + (1,1,1)$, de fato $(-1,0,0) + (1,1,1) = (0,1,1)$: uma relação não trivial.
\end{solution}

\begin{solution}{exo:b1:vspaces:4}
Os \omterm{def:b1:poly:def}{polinômios} $1, (X-1), (X-1)^2, (X-1)^3$ têm graus distintos $0,
1, 2, 3$: \omterm{def:b1:vspaces:free}{livres}
(\cref{prop:b1:vspaces:freecriteria}~(1)); quatro vetores \omterm{def:b1:vspaces:free}{livres} e geradores (todo $P \in \R_3[X]$ se
desenvolve em potências de $X - 1$, por exemplo por Taylor para
\omterm{def:b1:poly:def}{polinômios}, cf.\ a demonstração da \cref{prop:b1:poly:multiplicity}): uma \omterm{def:b1:vspaces:free}{base}. Para $X^3$,
Taylor em $1$: $P = X^3$, $P(1) = 1$, $P'(1) = 3$, $P''(1) = 6$, $P'''(1) =
6$:
\[
X^3 = 1 + 3(X - 1) + 3(X-1)^2 + (X-1)^3 ,
\]
\omterm{prop:b1:vspaces:coordinates}{coordenadas} $(1, 3, 3, 1)$ (a linha de Pascal, como se espera de $X^3 =
((X-1)+1)^3$).
\end{solution}

\begin{solution}{exo:b1:vspaces:5}
$F \cap G$: as condições $x = y = z$ e $x = t = 0$ dão juntas $x = 0$, donde $y = z = 0$,
e $t = 0$: a interseção é $\{0\}$. Soma: dado $(x,y,z,t)$, procure $(a,a,a,b) \in F$ e
$(0,c,d,0) \in G$ cuja soma seja ele: $a = x$, $b = t$, $c = y - x$, $d =
z - x$: sempre
possível. Logo $\R^4 = F \oplus G$, e
\[
(1,2,3,4) = (1,1,1,4) + (0,1,2,0) .
\]
\end{solution}

\begin{solution}{exo:b1:vspaces:6}
Um elemento de $G$ é $\lambda u$; se ele converge, então (como
$\lambda u_n = \lambda(-1)^n$ tem os dois limites de \omterm{def:b1:seq:subsequence}{subsequências} $\pm
\lambda$) necessariamente
$\lambda = 0$: $F \cap G = \{0\}$.

$F + G$ não é tudo: consiste nas sequências da forma
$c_n + \lambda(-1)^n$ com $(c_n)$ convergente. A sequência $v_n =
n$ não é dessa forma ($v_n - \lambda(-1)^n$ é ilimitada, nunca convergente).
Logo $F \oplus G \subsetneq$ (espaço de todas as sequências).
\end{solution}

\begin{solution}{exo:b1:vspaces:7}
($\supseteq$) Tanto $F \cap G$ quanto $F \cap H$ estão em $F$, e a sua soma está em
$G + (F \cap H)$: a inclusão segue porque o lado esquerdo é um \omterm{def:b1:vspaces:subspace}{subespaço} que
contém as duas partes --- concretamente, um elemento $g + h$ com
$g \in F\cap G$, $h \in F \cap H$ está em $F$ (soma de dois elementos de $F$) e em $G + (F \cap H)$.

($\subseteq$) Seja $x \in F$ com $x = g + h$, $g \in G$, $h \in F
\cap H$. Então $g = x - h \in F$
(diferença de elementos de $F$), logo $g \in F \cap G$, e $x = g + h \in (F \cap G) + (F \cap H)$.

Contraexemplo à distributividade plena em $\R^2$: $F =
\operatorname{Vect}(1,1)$, $G = \operatorname{Vect}(1,0)$,
$H =
\operatorname{Vect}(0,1)$. Então $G + H = \R^2$, logo $F \cap (G+H) =
F$, ao passo que $F \cap G = F \cap H = \{0\}$: o lado direito
é $\{0\}
\neq F$.
\end{solution}

\begin{solution}{exo:b1:vspaces:8}
\begin{enumerate}
  \item Suponha $\sum_{i} \lambda_i \eu^{a_i x} = 0$ para todo $x$.
        Multiplique por $\eu^{-a_p x}$: $\lambda_p + \sum_{i < p}
        \lambda_i \eu^{(a_i - a_p)x} \to \lambda_p$ quando $x \to
        +\infty$ (cada expoente
        $a_i - a_p < 0$). O lado esquerdo é identicamente $0$, logo $\lambda_p = 0$;
        itere para baixo.
  \item Seja $a\cos x + b \sin x + c \cos 2x + d\sin 2x = 0$ para todo $x$. Avalie em $x = 0$: $a + c = 0$; em
        $x = \pi$: $-a + c
        = 0$; logo $a = c = 0$, e a relação reduz-se a
        $b\sin
        x + d \sin 2x = 0$. Avalie em $x = \frac\pi2$: $b = 0$; depois em $x = \frac\pi4$: $d = 0$.
  \item Suponha $\sum \lambda_i \abs{x - a_i} = 0$ para todo $x$. A função $\sum_{i \neq j} \lambda_i\abs{x - a_i}$ é \omterm{def:b1:derivative:def}{derivável}
        em $a_j$ (cada termo o é, longe do seu próprio bico), logo
        $-\lambda_j \abs{x - a_j}$, a diferença delas, também deve ser \omterm{def:b1:derivative:def}{derivável} em
        $a_j$ --- o que força $\lambda_j = 0$ ($\abs{\,\cdot\,}$ tem um bico). Isto vale
        para todo $j$.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:vspaces:9}
Seja $h \in H$. Como $h \in H \subseteq F + H = F + G$, escreva $h = f
+ g$ com $f \in F$, $g \in G$.
Então $f = h - g \in H$ (os dois termos em $H$, usando $G \subseteq H$), logo $f \in F \cap H = F \cap G
\subseteq G$, e
$h = f + g \in G$. Portanto $H \subseteq G$, e com a hipótese $G \subseteq H$: igualdade.

Contraexemplo sem $G \subseteq H$: em $\R^2$, tome $F =
\operatorname{Vect}(1,0)$, $G = \operatorname{Vect}(0,1)$, $H =
\operatorname{Vect}(1,1)$:
então $F + G = F + H = \R^2$ e $F \cap G
= F \cap H = \{0\}$, mas $G \neq H$.
\end{solution}

\begin{solution}{exo:b1:vspaces:10}
Os dois \omterm{def:b1:logic:sets}{conjuntos} são \omterm{def:b1:vspaces:subspace}{subespaços} (as condições que os definem são
lineares e valem para $0$). Decomposição: para $P \in \R[X]$,
\[
P(X) = \underbrace{\frac{P(X) + P(-X)}{2}}_{\in\,\mathcal P}
+ \underbrace{\frac{P(X) - P(-X)}{2}}_{\in\,\mathcal I},
\]
e um \omterm{def:b1:poly:def}{polinômio} ao mesmo tempo par e ímpar satisfaz $P = -P$, logo
$P = 0$: a soma é direta e vale $\R[X]$.

Agora seja $P = \sum_k a_k X^k$ par. Então $P(X) - P(-X) = 2
\sum_{k \text{ ímpar}} a_k X^k$ é o \omterm{def:b1:poly:def}{polinômio} nulo, logo todo
coeficiente de grau ímpar se anula (\cref{def:b1:poly:def}): $P \in
\operatorname{Vect}(1, X^2, X^4, \dots)$, isto é, $P = Q(X^2)$
para um \omterm{def:b1:poly:def}{polinômio} $Q$. Reciprocamente, todo \omterm{def:b1:poly:def}{polinômio} em $X^2$ é
par.
\end{solution}

\begin{solution}{exo:b1:vspaces:11}
Suponha $a(x_1 + x_2) + b(x_2 + x_3) + c(x_3 + x_1) = 0$.
Reagrupando na \omterm{def:b1:vspaces:free}{família livre} $(x_1, x_2, x_3)$:
\[
(a + c)\,x_1 + (a + b)\,x_2 + (b + c)\,x_3 = 0
\implies a + c = a + b = b + c = 0 .
\]
Subtrair as duas primeiras equações dá $c = b$; a terceira então dá
$2b = 0$, logo $b = c = 0$, e depois $a = 0$: a família é \omterm{def:b1:vspaces:free}{livre}.

Para quatro vetores a família análoga é \emph{sempre} linearmente
dependente:
\[
(x_1 + x_2) - (x_2 + x_3) + (x_3 + x_4) - (x_4 + x_1) = 0
\]
é uma combinação nula não trivial (coeficientes $1, -1, 1, -1$), qualquer que
seja $(x_1, x_2, x_3, x_4)$. É a paridade do comprimento do ciclo que decide.
\end{solution}

\begin{solution}{exo:b1:vspaces:12}
\begin{enumerate}
  \item Se $F_1 \subseteq F_2$ ou $F_2 \subseteq F_1$, a união é um dos dois, logo própria.
        Caso contrário, escolha $x \in F_1
        \setminus F_2$ e $y \in F_2 \setminus F_1$, e considere
        $x + y$. Se $x + y \in F_1$, então $y = (x + y) - x \in
        F_1$: contradição. Se $x + y \in F_2$,
        então $x \in F_2$: contradição. Logo $x + y \notin F_1 \cup F_2$, e $E \neq
        F_1 \cup F_2$.
  \item Suponha, por absurdo, que $E = F_1 \cup \dots \cup
        F_k$, com $k$ escolhido
        \emph{mínimo} entre todas essas coberturas. A minimalidade
        proíbe $F_1 \subseteq F_2 \cup
        \dots \cup F_k$ (senão descarte $F_1$), logo existe $x \in F_1$
        com $x \notin F_i$ para todo $i \geq 2$. Como $F_1$ é próprio, escolha
        $y \notin F_1$. Para cada escalar $t$, o vetor $y + t x$ está em
        algum $F_i$. Ele nunca está em $F_1$: caso contrário
        $y = (y + tx) - tx \in F_1$ (pois $x \in
        F_1$). O \omterm{def:b1:structures:field}{corpo} é infinito, logo escolha $k$
        escalares distintos $t_1, \dots, t_k$: os $k$ vetores $y + t_j x$ caem nos
        $k - 1$ \omterm{def:b1:vspaces:subspace}{subespaços} $F_2, \dots, F_k$, e dois deles, digamos $y + t x$
        e $y + t' x$ com $t \neq t'$, estão no mesmo $F_i$ ($i \geq 2$). Então a
        sua diferença $(t - t')x \in F_i$, logo $x \in F_i$: contradição. Portanto
        não existe cobertura finita por \omterm{def:b1:vspaces:subspace}{subespaços} próprios.
\end{enumerate}
\end{solution}

\begin{solution}{pb:b1:vspaces:1}
\textbf{1.} $L_i$ é um produto de $n$ fatores lineares dividido por
uma constante não nula (os $x_i$ são distintos), logo $\deg L_i = n$.
Avaliando em $x_j$ com $j \neq i$: o fator $X - x_j$ do numerador se
anula, logo $L_i(x_j) = 0$. Em $x_i$, numerador e denominador coincidem:
$L_i(x_i) = 1$.

\textbf{2.} Suponha $\sum_i \lambda_i L_i = 0$. Avalie em $x_j$: todos os termos morrem
exceto $\lambda_j L_j(x_j) = \lambda_j$, logo $\lambda_j = 0$ para todo $j$: a família é \omterm{def:b1:vspaces:free}{livre}.

\textbf{3.} Seja $D = P - \sum_i P(x_i) L_i$. Então $\deg D \leq n$ e, pela questão 1, $D(x_j) = P(x_j) - P(x_j) = 0$ nos
$n + 1$ pontos distintos $x_0, \dots, x_n$. Um \omterm{def:b1:poly:def}{polinômio} não nulo de grau
$\leq n$ tem no máximo $n$ raízes (\cref{cor:b1:poly:nroots}), logo $D = 0$. Assim todo
$P \in \R_n[X]$ é uma combinação dos $L_i$: a família é geradora e, com a
questão 2, uma \omterm{def:b1:vspaces:free}{base}.

\textbf{4.} Dados $y_0, \dots, y_n$, o \omterm{def:b1:poly:def}{polinômio} $P = \sum_i
y_i L_i$ tem grau $\leq n$ e
interpola. Unicidade: um $P$ interpolador tem, pela questão 3,
\omterm{prop:b1:vspaces:coordinates}{coordenadas} $(P(x_0), \dots,
P(x_n)) = (y_0, \dots, y_n)$ na \omterm{def:b1:vspaces:free}{base} $(L_i)$, e as \omterm{prop:b1:vspaces:coordinates}{coordenadas} numa \omterm{def:b1:vspaces:free}{base} são
únicas (\cref{prop:b1:vspaces:coordinates}). As \omterm{prop:b1:vspaces:coordinates}{coordenadas} de $P$ na \omterm{def:b1:vspaces:free}{base} de Lagrange são os
seus \emph{valores nos nós} --- é este todo o interesse da \omterm{def:b1:vspaces:free}{base}.

\textbf{5.} Aplique a questão 3 a $P = X^k$ ($0 \leq k \leq n$):
\[
X^k = \sum_{i=0}^{n} x_i^{k} L_i ,
\]
e $k = 0$ dá $\sum_i L_i = 1$.

\textbf{6.} $\deg N_k = k$ exatamente: a família $(N_0, \dots, N_n)$
é uma escada de graus em $\R_n[X]$, logo uma \omterm{def:b1:vspaces:free}{base} pelo
\cref{ex:b1:vspaces:staircase} (liberdade pela \cref{prop:b1:vspaces:freecriteria}~(1), geração por descida finita sobre o
grau).

\textbf{7.} Da recorrência,
\[
f[x_0, x_1] = \frac{f(x_1) - f(x_0)}{x_1 - x_0},
\qquad
f[x_0, x_1, x_2] = \frac{f[x_1, x_2] - f[x_0, x_1]}{x_2 - x_0}.
\]
Para $f(x) = x^2$:
\[
f[x_0, x_1] = \frac{x_1^2 - x_0^2}{x_1 - x_0} = x_0 + x_1,
\]
e então
\[
f[x_0, x_1, x_2]
= \frac{(x_1 + x_2) - (x_0 + x_1)}{x_2 - x_0}
= \frac{x_2 - x_0}{x_2 - x_0} = 1 .
\]

\textbf{8.} $\deg S \leq n$, pois $Q, R$ têm grau $\leq n - 1$.
Em $x_0$: $S(x_0) = \frac{-(x_0 - x_n) R(x_0)}{x_n - x_0} =
R(x_0) = f(x_0)$. Em $x_n$: $S(x_n) = \frac{(x_n - x_0)
Q(x_n)}{x_n - x_0} = Q(x_n) = f(x_n)$. Num nó \omterm{def:b1:topology:closure}{interior} $x_i$
($1 \leq i \leq n-1$), tanto $Q$ quanto $R$ tomam o valor $f(x_i)$, logo
\[
S(x_i) = \frac{(x_i - x_0) - (x_i - x_n)}{x_n - x_0}\, f(x_i)
= f(x_i) .
\]

\textbf{9.} Indução sobre o número de nós. Um nó: o interpolante é
a constante $f(x_0) = f[x_0]$. Suponha a afirmação para $k$ nós e seja $S$
interpolando em $x_0, \dots, x_k$; pela unicidade (questão 4), $S$ é dado pelo
lema de Aitken a partir de $R$ (nós $x_0, \dots, x_{k-1}$) e $Q$ (nós $x_1, \dots, x_k$).
O coeficiente de $X^{k}$ em $S$ é
\[
\frac{[X^{k-1}]\,Q - [X^{k-1}]\,R}{x_k - x_0}
= \frac{f[x_1, \dots, x_k] - f[x_0, \dots, x_{k-1}]}{x_k - x_0}
= f[x_0, \dots, x_k]
\]
pela hipótese de indução e pela recorrência que as define.

\textbf{10.} Seja $P_k$ o interpolante de $f$ em $x_0, \dots, x_k$. A
diferença $P_k - P_{k-1}$ tem grau $\leq k$ e anula-se em $x_0, \dots, x_{k-1}$, logo, pelo
teorema do fator aplicado $k$ vezes (\cref{thm:b1:poly:factor}), ela vale $c\,N_k$ para
uma constante $c$; comparando os coeficientes de $X^{k}$ e usando a
questão 9, $c =
f[x_0, \dots, x_k]$. Telescopando a partir de $P_0 = f(x_0) N_0$ obtém-se a fórmula
de Newton. Para a forma \omterm{def:b1:topology:closed}{fechada}, escreva $P_k = \sum_{i \leq
k} f(x_i) L_i$ (Lagrange, nos nós
$x_0, \dots, x_k$) e leia o coeficiente de $X^{k}$: cada $L_i$ contribui com
$\frac{1}{\prod_{j \neq i}(x_i - x_j)}$, donde
\[
f[x_0, \dots, x_k] = \sum_{i=0}^{k}
\frac{f(x_i)}{\prod_{j \neq i,\, j \leq k}(x_i - x_j)} .
\]
O lado direito é invariante por qualquer \omterm{def:b1:counting:objects}{permutação} dos nós, logo a
diferença dividida não depende da ordenação deles.

\textbf{11.} Se $P = a X^m + (\text{graus inferiores})$, o teorema binomial dá
\[
\Delta P = a\bigl((X+1)^m - X^m\bigr) + \dots
= a\,m\,X^{m-1} + (\text{graus inferiores}),
\]
pois $(X+1)^m - X^m = m X^{m-1} + \dots$ e a parte de grau inferior de $P$ contribui com grau
$\leq m - 2$ depois de $\Delta$ (ou seja, com termos de grau $\leq m-2$). Logo
$\deg \Delta P = m - 1$ com coeficiente dominante $m a$. Uma constante $c$ dá $\Delta c = c - c = 0$.

\textbf{12.} $\deg B_k = k$: escada, logo uma \omterm{def:b1:vspaces:free}{base} de $\R_n[X]$ (\cref{ex:b1:vspaces:staircase}).
Para $\Delta B_k$ ($k \geq 1$), fatore o produto comum:
\begin{align*}
k!\,\Delta B_k
&= (X+1)X\cdots(X-k+2) - X(X-1)\cdots(X-k+1) \\
&= X(X-1)\cdots(X-k+2)\,\bigl[(X+1) - (X-k+1)\bigr] \\
&= k\,X(X-1)\cdots(X-k+2),
\end{align*}
logo $\Delta B_k = \frac{X(X-1)\cdots(X-k+2)}{(k-1)!} = B_{k-1}$.

\textbf{13.} Escreva $P = \sum_{k=0}^{n} c_k B_k$ (\omterm{def:b1:vspaces:free}{base}, questão 12). Aplique $\Delta^{j}$: pela
questão 12, $\Delta^{j} P = \sum_{k
\geq j} c_k B_{k-j}$. Avalie em $0$: $B_0(0) = 1$ e $B_m(0) =
0$ para $m \geq 1$ (o fator
$X$ se anula), logo $\bigl(\Delta^{j}P\bigr)(0) = c_j$. Esta é a fórmula das diferenças
progressivas.

\textbf{14.} Indução sobre $k$. Para $k = 0$ a identidade lê-se
$P(0) = P(0)$. Suponha-a para $k$ e aplique-a a $\Delta P$:
\[
\bigl(\Delta^{k+1} P\bigr)(0)
= \sum_{j=0}^{k} (-1)^{k-j}\binom kj \bigl(P(j+1) - P(j)\bigr).
\]
Junte o coeficiente de $P(i)$: ele é $(-1)^{k-i+1}\binom
k{i-1}\cdot(-1)^{0}$ na primeira soma
(deslocada) e $-(-1)^{k-i}\binom ki$ na segunda --- juntos
\[
(-1)^{k+1-i}\Bigl(\binom k{i-1} + \binom ki\Bigr)
= (-1)^{k+1-i}\binom{k+1}i
\]
pela regra de Pascal, que é a identidade na ordem $k + 1$.

\textbf{15.} Iterando a questão 11 a partir do grau $n$, com
coeficiente dominante $a_n$: depois de um $\Delta$, grau $n-1$ e
coeficiente dominante $n a_n$; depois de dois, $n(n-1)a_n$; depois de $n$
passos, grau $0$ e valor $n(n-1)\cdots 1\, a_n = n!\,a_n$, uma constante.
Mais um $\Delta$ a mata: $\Delta^{n+1}P = 0$.

\textbf{16.} Se $m \geq k$: $B_k(m) = \binom mk \in \N$. Se $0
\leq m < k$: um fator de $m(m-1)\cdots(m-k+1)$ é nulo,
logo $B_k(m) = 0$. Se $m = -q$ com $q \geq 1$:
\[
B_k(-q) = \frac{(-q)(-q-1)\cdots(-q-k+1)}{k!}
= (-1)^k\,\frac{q(q+1)\cdots(q+k-1)}{k!}
= (-1)^k \binom{q+k-1}{k},
\]
um inteiro. Logo todo $B_k$ leva $\Z$ em $\Z$.

\textbf{17.} ($\Leftarrow$) Se $P = \sum_k c_k B_k$ com $c_k \in
\Z$, então para $m \in \Z$, $P(m) = \sum_k c_k B_k(m) \in \Z$
pela questão 16. ($\Rightarrow$) Se $P$ é a valores inteiros, as suas
\omterm{prop:b1:vspaces:coordinates}{coordenadas} são $c_k = \bigl(\Delta^k P\bigr)(0) = \sum_{j=0}^k
(-1)^{k-j}\binom kj P(j)$ (questões 13 e 14), uma combinação inteira
dos inteiros $P(0), \dots, P(k)$. Esta é a caracterização de Pólya dos
\omterm{pb:b1:vspaces:1}{polinômios a valores inteiros}.

\textbf{18.} Ponha $Q(X) = P(X + a)$, um \omterm{def:b1:poly:def}{polinômio} de grau $\leq n$ com $Q(0), Q(1), \dots, Q(n) \in \Z$.
As suas \omterm{prop:b1:vspaces:coordinates}{coordenadas} em $(B_k)_{k \leq n}$ são $c_k = \sum_{j \leq k}(-1)^{k-j}\binom kj
Q(j) \in \Z$ (a questão 14 usa apenas os
valores em $0, \dots, k
\leq n$). Pela questão 17 ($\Leftarrow$), $Q$ é a valores
inteiros em todo o $\Z$, e portanto $P(X) = Q(X - a)$ também.

\textbf{19.} Um produto de $k$ inteiros consecutivos é $m(m-1)
\cdots(m-k+1) = k!\,B_k(m)$ para
algum $m \in \Z$, e $B_k(m) \in
\Z$ pela questão 16: o produto é divisível por
$k!$.

\textbf{20.} Valores de $P = \frac{X(X+1)(2X+1)}{6}$ em $0,1,2,3$: $0, 1, 5, 14$. Tabela de
diferenças: linha $\Delta$: $1, 4, 9$; linha $\Delta^2$: $3, 5$; linha
$\Delta^3$: $2$. Logo, pela questão 13,
\[
P = 0\cdot B_0 + 1\cdot B_1 + 3\,B_2 + 2\,B_3 ,
\]
com \omterm{prop:b1:vspaces:coordinates}{coordenadas} inteiras: $P$ é a valores inteiros (questão 17),
ao passo que os seus coeficientes monomiais $\frac13, \frac12, \frac16$ não são
inteiros. Cálculo direto:
\begin{align*}
\Delta P &= \frac{(X+1)(X+2)(2X+3) - X(X+1)(2X+1)}{6} \\
&= \frac{(X+1)\bigl[(X+2)(2X+3) - X(2X+1)\bigr]}{6}
= \frac{(X+1)(6X+6)}{6} = (X+1)^2 .
\end{align*}
Telescopando $P(m) = \sum_{j=0}^{m-1}\Delta P(j) = \sum_{j=1}^{m}
j^2$ (com $P(0) = 0$): a fórmula da soma dos quadrados.

\textbf{21.} Os valores $2^i$ em $i = 0, \dots, n$ têm tabela de diferenças
constantemente igual a $1$ na borda esquerda: $\Delta^k$ da sequência
$(2^i)$ é de novo $(2^i)$ (pois $2^{i+1} - 2^i = 2^i$), logo $\bigl(\Delta^k P\bigr)(0) = 2^0 = 1$ para todo $k \leq n$,
e $P =
B_0 + B_1 + \dots + B_n$ pela questão 13. Então
\[
P(n+1) = \sum_{k=0}^{n}\binom{n+1}{k}
= 2^{n+1} - \binom{n+1}{n+1} = 2^{n+1} - 1 \neq 2^{n+1}:
\]
o padrão quebra no primeiro ponto não controlado.

\textbf{22.} Pela questão 12, $B_k = \Delta B_{k+1}$, logo
\[
\sum_{j=0}^{m-1} B_k(j)
= \sum_{j=0}^{m-1}\bigl(B_{k+1}(j+1) - B_{k+1}(j)\bigr)
= B_{k+1}(m) - B_{k+1}(0) = B_{k+1}(m).
\]
Para $j < k$ os termos $B_k(j)$ se anulam, logo a soma começa de
fato em $j = k$: $\sum_{j=k}^{m-1}\binom jk = \binom m{k+1}$, a identidade do taco de hóquei.

\textbf{23.} As tabelas de diferenças (ou o desenvolvimento direto)
dão
\[
X^2 = B_1 + 2 B_2, \qquad X^3 = B_1 + 6 B_2 + 6 B_3
\]
(verificação: $B_1 + 2B_2 = X + X(X-1) = X^2$; em $X = 1, 2, 3$ o segundo dá $1, 8, 27$). A questão 22
fornece então
\[
\sum_{j=0}^{m-1} j^2 = B_2(m) + 2B_3(m)
= \binom m2 + 2\binom m3 = \frac{m(m-1)(2m-1)}{6},
\]
\[
\sum_{j=0}^{m-1} j^3 = B_2(m) + 6B_3(m) + 6B_4(m)
= \binom m2 + 6\binom m3 + 6\binom m4 .
\]
Desenvolvendo a última expressão: $\binom m2 + 6\binom m3 + 6\binom
m4 = \frac{m(m-1)}{2}\bigl[1 + 2(m-2) +
\frac{(m-2)(m-3)}{2}\bigr] = \frac{m^2(m-1)^2}{4} = \binom m2^2$. Substituindo $m$ por
$m + 1$: $1^3 + \dots + m^3 =
\bigl(\frac{m(m+1)}2\bigr)^2 = (1 + \dots + m)^2$, a identidade de Nicômaco.

\textbf{24.} Nós $0, 1, 2$, \omterm{def:b1:poly:def}{polinômio} $X^2$. \omterm{def:b1:vspaces:free}{Base} monomial
$(1, X, X^2)$: \omterm{prop:b1:vspaces:coordinates}{coordenadas} $(0, 0, 1)$. \omterm{def:b1:vspaces:free}{Base} de Lagrange: as \omterm{prop:b1:vspaces:coordinates}{coordenadas} são os
valores $(0, 1, 4)$ (questão 4). \omterm{def:b1:vspaces:free}{Base} de Newton $(1, X, X(X-1))$: \omterm{pb:b1:vspaces:1}{diferenças
divididas} $f[0] = 0$, $f[0,1] = 1$, $f[0,1,2] = \frac{3 - 1}{2} = 1$ (questão 9), logo \omterm{prop:b1:vspaces:coordinates}{coordenadas}
$(0, 1, 1)$ --- de fato $X + X(X-1) = X^2$. Três \omterm{def:b1:vspaces:free}{bases}, três vetores de \omterm{prop:b1:vspaces:coordinates}{coordenadas},
um só \omterm{def:b1:poly:def}{polinômio}.

\textbf{25.} (i) A questão 4 é automática porque $(L_i)$ é uma
\emph{\omterm{def:b1:vspaces:free}{base}}: existência e unicidade da interpolação são exatamente
existência e unicidade das \omterm{prop:b1:vspaces:coordinates}{coordenadas}. (ii) A \omterm{def:b1:vspaces:free}{base} de Newton é uma
escada, de modo que as \omterm{prop:b1:vspaces:coordinates}{coordenadas} se calculam por divisões
sucessivas --- cada novo nó acrescenta um termo sem perturbar os
anteriores --- ao passo que as \omterm{prop:b1:vspaces:coordinates}{coordenadas} de $P$ na \omterm{def:b1:vspaces:free}{base} de
Lagrange são os valores $P(x_i)$, disponíveis sem cálculo algum.
(iii) As duas \omterm{def:b1:vspaces:free}{bases} são \omterm{def:b1:vspaces:free}{livres} pelos mesmos dois critérios da
\cref{prop:b1:vspaces:freecriteria}: graus distintos para Newton, avaliação nos nós para
Lagrange. (iv) O teorema de Pólya diz que ``$P(\Z) \subseteq \Z$'', uma
propriedade dos valores, equivale à integralidade das \omterm{prop:b1:vspaces:coordinates}{coordenadas}
na \omterm{def:b1:vspaces:free}{base} $(B_k)$ --- a aritmética de um \omterm{def:b1:poly:def}{polinômio} só se torna
visível na \omterm{def:b1:vspaces:free}{base} adaptada à pergunta.
\end{solution}
