\section*{Capítulo \ref{ch:b1:findim} --- Dimensão Finita}

\begin{solution}{exo:b1:findim:1}
\begin{enumerate}
  \item $z = -x - y$: $F = \{(x, y, -x-y)\} =
        \operatorname{Vect}\bigl((1,0,-1), (0,1,-1)\bigr)$; os dois
        vetores são \omterm{def:b1:vspaces:free}{livres} (\omterm{prop:b1:vspaces:coordinates}{coordenadas}): $\dim F = 2$.
  \item $G = \{(x, x, 2t, t)\} =
        \operatorname{Vect}\bigl((1,1,0,0), (0,0,2,1)\bigr)$: \omterm{def:b1:vspaces:free}{livres}, $\dim G = 2$.
  \item $P(1) = P'(1) = 0$ significa $(X-1)^2 \mid P$
        (\cref{prop:b1:poly:multiplicity}): $P = (X-1)^2(aX + b)$.
        \omterm{def:b1:vspaces:free}{Base} $\bigl((X-1)^2, X(X-1)^2\bigr)$, dimensão $2$.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:findim:2}
$(3,5,7) = (1,1,1) + 2(1,2,3)$ e $(0,1,2) = (1,2,3) - (1,1,1)$: ambos são combinações dos dois primeiros, que são
\omterm{def:b1:vspaces:free}{livres} (não proporcionais). \omterm{def:b1:findim:rank}{Posto} $2$; \omterm{def:b1:vspaces:free}{base} do \omterm{def:b1:vspaces:span}{espaço gerado}:
$\bigl((1,1,1), (1,2,3)\bigr)$.
\end{solution}

\begin{solution}{exo:b1:findim:3}
Tente acrescentar $e_1 = (1,0,0,0)$ e $e_3 = (0,0,1,0)$. A família $\bigl((1,1,0,0), (0,0,1,1), e_1, e_3\bigr)$ é \omterm{def:b1:vspaces:free}{livre}: uma
combinação nula $\alpha(1,1,0,0) + \beta(0,0,1,1) + \gamma e_1 + \delta
e_3 = 0$ lê-se $(\alpha + \gamma, \alpha, \beta + \delta, \beta) =
0$, logo $\alpha = \beta = 0$, e depois $\gamma = \delta = 0$.
Quatro vetores \omterm{def:b1:vspaces:free}{livres} em dimensão $4$: uma \omterm{def:b1:vspaces:free}{base} (a \cref{prop:b1:findim:twoofthree}).
\end{solution}

\begin{solution}{exo:b1:findim:4}
Graus $0, 1, 2, 3$ dois a dois distintos: \omterm{def:b1:vspaces:free}{livres} (a \cref{prop:b1:vspaces:freecriteria}), quatro
vetores em dimensão $4$: \omterm{def:b1:vspaces:free}{base}. Para $X^3$: desenvolva para baixo,
\[
X(X-1)(X-2) = X^3 - 3X^2 + 2X,
\qquad
X(X-1) = X^2 - X,
\]
logo $X^3 - X(X-1)(X-2) = 3X^2 - 2X$; e $3X^2 - 2X = 3(X^2 - X) + X
= 3\,X(X-1) + X$. Logo
\[
X^3 = X(X-1)(X-2) + 3\,X(X-1) + 1\cdot X + 0\cdot 1 :
\]
\omterm{prop:b1:vspaces:coordinates}{coordenadas} $(0, 1, 3, 1)$ em $\bigl(1, X, X(X-1),
X(X-1)(X-2)\bigr)$. (São números de Stirling disfarçados.)
\end{solution}

\begin{solution}{exo:b1:findim:5}
Grassmann: $\dim(F \cap G) = 4 + 5 - \dim(F + G)$, e $F + G$ é um \omterm{def:b1:vspaces:subspace}{subespaço} de $E$ que contém $G$:
$5 \leq \dim(F+G) \leq 7$. Logo $\dim(F \cap G) \in \{2, 3, 4\}$. Realizações em $\R^7$ com a \omterm{def:b1:vspaces:free}{base} canônica
$(e_1, \dots, e_7)$, tomando $G =
\operatorname{Vect}(e_1, \dots, e_5)$:
\begin{itemize}
  \item $F = \operatorname{Vect}(e_1, e_2, e_6, e_7)$: $F + G = \R^7$, interseção $\operatorname{Vect}(e_1, e_2)$, dimensão $2$;
  \item $F = \operatorname{Vect}(e_1, e_2, e_3, e_6)$: interseção de dimensão $3$;
  \item $F = \operatorname{Vect}(e_1, e_2, e_3, e_4) \subseteq G$: dimensão $4$.
\end{itemize}
\end{solution}

\begin{solution}{exo:b1:findim:6}
$H$ é um \omterm{def:b1:vspaces:subspace}{subespaço} (o \cref{exo:b1:vspaces:1}~(4), palavra por palavra). Pelo
teorema do fator (\cref{thm:b1:poly:factor}), $P \in H \iff P =
(X-1)Q$ com $\deg Q \leq n - 1$: a \omterm{def:b1:logic:map}{aplicação} $Q \mapsto (X-1)Q$ é uma
bijeção linear de $\R_{n-1}[X]$ sobre $H$, logo uma \omterm{def:b1:vspaces:free}{base} de $H$ é
\[
\bigl((X-1),\ (X-1)X,\ (X-1)X^2,\ \dots,\ (X-1)X^{n-1}\bigr),
\qquad \dim H = n .
\]
Uma reta \omterm{def:b1:vspaces:sum}{suplementar}: $\operatorname{Vect}(1)$ (as constantes). De fato $H \cap \operatorname{Vect}(1) = \{0\}$ (uma
constante não nula não se anula em $1$) e as dimensões somam
$n + 1$: por Grassmann, $H
\oplus \operatorname{Vect}(1) = \R_n[X]$.
\end{solution}

\begin{solution}{exo:b1:findim:7}
Remoção: retirando $u_p$, o \omterm{def:b1:vspaces:span}{espaço gerado} só pode encolher; e
encolhe no máximo uma dimensão, pois recolocar $u_p$ numa \omterm{def:b1:vspaces:free}{base} do
espaço menor dá uma família geradora do maior com no máximo um
vetor a mais. Simetricamente, ao acrescentar um vetor $v$: o novo
\omterm{def:b1:vspaces:span}{espaço gerado} contém o antigo com no máximo um gerador a mais,
logo a sua dimensão é $r$ (se $v$ já estava no \omterm{def:b1:vspaces:span}{espaço gerado}) ou
$r + 1$ (caso contrário, pela \cref{prop:b1:vspaces:freecriteria}~(2) uma \omterm{def:b1:vspaces:free}{base} se estende). Os
dois enunciados juntos dão a conclusão ``o \omterm{def:b1:findim:rank}{posto} é
$1$-lipschitziano''.
\end{solution}

\begin{solution}{exo:b1:findim:8}
Ao longo de uma cadeia estritamente crescente, as dimensões
crescem estritamente ($F_i \subseteq F_{i+1}$, $F_i \neq F_{i+1}$ e o \cref{thm:b1:findim:subspaces}: a igualdade das
dimensões forçaria a igualdade dos espaços). Logo $\dim F_1 < \dim F_2 < \dots < \dim F_k$ é uma
sequência estritamente crescente de inteiros em $\intint{0}{n}$: no máximo
$n + 1$ valores, $k \leq n + 1$. Cadeia maximal em $\R^n$:
\[
\{0\} \subsetneq \operatorname{Vect}(e_1) \subsetneq
\operatorname{Vect}(e_1, e_2) \subsetneq \dots \subsetneq \R^n ,
\]
de comprimento exatamente $n + 1$.
\end{solution}

\begin{solution}{exo:b1:findim:9}
$F + G$ contém estritamente $F$ (pois $G \not\subseteq F$), logo $\dim(F + G) = n$ e
Grassmann dá $\dim(F \cap G) = (n-1) +
(n-1) - n = n - 2$.

Afirmação geral, por indução sobre $k$: a interseção $I_k$ de $k$
hiperplanos tem $\dim I_k \geq n - k$. Verdadeira para $k = 1$. Passo: $I_{k+1}
= I_k \cap H_{k+1}$, e
Grassmann dentro de $E$:
\[
\dim(I_k \cap H_{k+1}) = \dim I_k + (n - 1) - \dim(I_k + H_{k+1})
\geq \dim I_k + (n-1) - n \geq n - k - 1 . \qedhere
\]
\end{solution}

\begin{solution}{exo:b1:findim:10}
\begin{enumerate}
  \item A condição $u_{n+2} - 3u_{n+1} + 2u_n = 0$ é linear e satisfeita pela sequência
        nula: $E$ é um \omterm{def:b1:vspaces:subspace}{subespaço}. Por indução, $u_0$ e $u_1$
        determinam todo $u_n$, e a dependência é linear (cada
        passo é uma combinação linear dos dois valores
        anteriores); reciprocamente, todo par $(a, b)$ provém de
        exatamente uma sequência de $E$ (defina $u_n$ pela
        recorrência). Como no \cref{ex:b1:findim:dims}, $E$ é parametrizado \omterm{def:b1:logic:inj}{bijetiva}
        e linearmente por $(u_0, u_1) \in
        \R^2$: $\dim E = 2$.
  \item Constantes: $3\cdot1 - 2\cdot1 = 1$. Geométrica: $3\cdot
        2^{n+1} - 2\cdot 2^n = (6 - 2)2^n = 2^{n+2}$. Ambas estão em
        $E$. Liberdade: $a\cdot 1 + b\cdot 2^n = 0$ para todo $n$ dá, em $n = 0$ e
        $n = 1$: $a + b = 0$, $a + 2b = 0$, logo $a = b = 0$. Dois vetores \omterm{def:b1:vspaces:free}{livres}
        em dimensão $2$: uma \omterm{def:b1:vspaces:free}{base} (a \cref{prop:b1:findim:twoofthree}).
  \item Resolva $a + b = 0$, $a + 2b = 1$: $b = 1$, $a = -1$, logo $u_n = 2^n - 1$
        (a sequência de Mersenne).
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:findim:11}
\begin{enumerate}
  \item Grassmann: $\dim(F \cap G) = \dim F + \dim G - \dim(F +
        G) \geq \dim F + \dim G - n > 0$, logo $F \cap G \neq
        \{0\}$. Com $n = 100$: $51 + 50 - 100 = 1 > 0$, e a
        interseção contém uma reta.
  \item Se $F \cap H = \{0\}$, a soma é direta e $\dim F +
        (n - 1) = \dim(F \oplus H) \leq n$, logo $\dim F \leq 1$.
        (Reciprocamente, uma reta não contida em $H$ satisfaz
        isto: os hiperplanos deixam escapar quase nada.)
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:findim:12}
Indução decrescente sobre $d = \dim F = \dim G$, de $d = n$ até $d = 0$. Se
$d = n$: $F = G = E$ e $S = \{0\}$ serve. Suponha o enunciado válido para
pares de \omterm{def:b1:vspaces:subspace}{subespaços} de dimensão $d + 1 \leq n$, e seja $\dim F = \dim G = d < n$.

Se $F = G$: tome para $S$ um \omterm{def:b1:vspaces:sum}{suplementar} qualquer de $F$
(o \cref{thm:b1:findim:subspaces}). Se $F \neq G$: ambos são próprios, logo pelo \cref{exo:b1:vspaces:12} existe
$x \notin F \cup G$. As somas $F' = F \oplus \operatorname{Vect}(x)$ e $G' = G \oplus
\operatorname{Vect}(x)$ são diretas ($x \notin F$, $x \notin G$) e têm
dimensão $d + 1$; pela hipótese de indução elas admitem um
\omterm{def:b1:vspaces:sum}{suplementar} comum $S'$: $E = F' \oplus S' = G' \oplus
S'$. Ponha $S = \operatorname{Vect}(x) \oplus S'$ --- direta, porque
$S' \cap \operatorname{Vect}(x) \subseteq S' \cap F' =
\{0\}$.

Então $F + S = F + \operatorname{Vect}(x) + S' = F' + S' = E$, e $F \cap S = \{0\}$: se $f = \lambda x + s'$ com $f \in F$, $s'
\in S'$, então $s' = f - \lambda x \in F' \cap S' = \{0\}$,
logo $f
= \lambda x$, o que força $\lambda = 0$ ($x \notin F$) e $f = 0$. Logo $E = F \oplus S$, e
simetricamente $E = G \oplus S$.
\end{solution}

\begin{solution}{pb:b1:findim:1}
\textbf{1.} $\Q$ contém $0 \neq 1$, é estável por adição, multiplicação
e opostos, e todo racional não nulo tem inverso racional: um \omterm{def:b1:structures:field}{corpo}
(a \cref{def:b1:structures:field}). As definições de \omterm{def:b1:vspaces:span}{espaço gerado}, liberdade e \omterm{def:b1:vspaces:free}{base} e as
demonstrações dos Capítulos 18--19 usam apenas a adição de
vetores, a distributividade e a aritmética dos escalares num \omterm{def:b1:structures:field}{corpo}
--- nunca um valor absoluto, uma ordem ou um limite. Logo $\R$,
com a sua própria adição e a multiplicação $\Q \times \R \to \R$, é um \omterm{def:b1:vspaces:def}{espaço
vetorial} sobre $\Q$, e todos os teoremas se aplicam. A divisão por
um escalar ocorre uma única vez na demonstração do \cref{thm:b1:findim:exchange}: para
``isolar $g_k$'' divide-se pelo seu coeficiente não nulo.

\textbf{2.} Se $a + b\sqrt2 = 0$ com $a, b \in \Q$ não ambos nulos: $b \neq 0$ daria $\sqrt2 = -a/b \in \Q$,
contradizendo a irracionalidade de $\sqrt2$ (o \cref{exo:b1:arith:7} com $p = 2$);
logo $b = 0$, e então $a = 0$: \omterm{def:b1:vspaces:free}{livre} sobre $\Q$. Sobre $\R$ a
relação $\sqrt2\cdot 1 + (-1)\cdot\sqrt2 = 0$ é não trivial: linearmente dependente. Logo $\dim_\Q \Q(\sqrt2) = 2$,
com \omterm{prop:b1:vspaces:coordinates}{coordenadas} $(a, b)$ únicas.

\textbf{3.} $(a + b\sqrt2)(c + d\sqrt2) = (ac + 2bd) + (ad +
bc)\sqrt2 \in \Q(\sqrt2)$. Então
\[
\sigma(uv) = (ac + 2bd) - (ad + bc)\sqrt2
= (a - b\sqrt2)(c - d\sqrt2) = \sigma(u)\sigma(v),
\]
logo $N(uv) = uv\,\sigma(uv) = u\sigma(u)\,v\sigma(v) = N(u)N(v)$.
Se $N(u) = a^2 - 2b^2 = 0$ com $u \neq 0$: $b \neq 0$ daria $(a/b)^2 = 2$, uma raiz quadrada racional
de $2$; logo $b = 0$, e então $a = 0$ e $u = 0$: contradição.
Portanto $N(u) \neq 0$ para $u \neq
0$.

\textbf{4.} $u \cdot \dfrac{\sigma(u)}{N(u)} = \dfrac{N(u)}{N(u)}
= 1$, e $\sigma(u)/N(u) \in \Q(\sqrt2)$: todo elemento não nulo é invertível
dentro de $\Q(\sqrt2)$, que é portanto um subcorpo de $\R$. Exemplos:
$N(3 + 2\sqrt2) = 9 - 8 = 1$, logo
\[
\frac1{3 + 2\sqrt2} = 3 - 2\sqrt2 ;
\qquad
N(1 + \sqrt2) = -1, \quad
\frac1{1 + \sqrt2} = \sqrt2 - 1 .
\]

\textbf{5.} Se $\sqrt6 = p/q$ então $6q^2 = p^2$; o expoente de $2$ é $1 + 2v_2(q)$,
ímpar, à esquerda, e $2v_2(p)$, par, à direita: impossível. Agora
suponha $\sqrt3 = a + b\sqrt2$ com $a, b \in \Q$. Elevando ao quadrado: $3 = a^2 + 2b^2 + 2ab\sqrt2$. Se $ab
\neq 0$,
então $\sqrt2 = (3 - a^2 - 2b^2)/(2ab) \in \Q$: impossível. Se $b = 0$: $\sqrt3 = a \in \Q$, contradizendo o \cref{exo:b1:arith:7}
($p = 3$). Se $a = 0$: $\sqrt3 = b\sqrt2$, e multiplicando por $\sqrt2$:
$\sqrt6 = 2b \in \Q$: impossível. Logo $\sqrt3 \notin \Q(\sqrt2)$.

\textbf{6.} Os produtos dos vetores da \omterm{def:b1:vspaces:free}{base} são
\[
\sqrt2\,\sqrt3 = \sqrt6,\quad
\sqrt2\,\sqrt6 = 2\sqrt3,\quad
\sqrt3\,\sqrt6 = 3\sqrt2,\quad
(\sqrt2)^2 = 2,\ (\sqrt3)^2 = 3,\ (\sqrt6)^2 = 6,
\]
todos em $M$. Um produto de dois elementos de $M$ desenvolve-se
por bilinearidade em combinações racionais destes: $M$ é estável
por multiplicação.

\textbf{7.} Seja $x + y\sqrt3 = 0$ com $x, y \in K = \Q(\sqrt2)$. Se $y \neq 0$, então $\sqrt3 = -x/y \in K$
(questão 4: $K$ é um \omterm{def:b1:structures:field}{corpo}), contradizendo a questão 5. Logo
$y = 0$, e então $x = 0$: $(1,
\sqrt3)$ é \omterm{def:b1:vspaces:free}{livre} sobre $K$. E gera: $K + K\sqrt3 =
\{(a + b\sqrt2) + (c + d\sqrt2)\sqrt3\} = \operatorname{Vect}_\Q
(1, \sqrt2, \sqrt3, \sqrt6) = M$.
Logo $\dim_K M = 2$.

\textbf{8.} Uma relação $a + b\sqrt2 + c\sqrt3 + d\sqrt6 = 0$ reagrupa-se como $(a + b\sqrt2) + (c + d\sqrt2)\sqrt3 = 0$ com
coeficientes em $K$; pela questão 7 ambos se anulam, e pela
questão 2 $a = b = 0$ e $c = d = 0$. Logo a família é \omterm{def:b1:vspaces:free}{livre} e $\dim_\Q M
= 4$.

\textbf{9.} Todo $x \in M$ se escreve $x = \sum_j y_j f_j$ com $y_j
\in K$ (\omterm{def:b1:vspaces:free}{base} de $M$
sobre $K$), e cada $y_j = \sum_i
\lambda_{ij} e_i$ com $\lambda_{ij} \in \Q$ (\omterm{def:b1:vspaces:free}{base} de $K$ sobre $\Q$);
substituindo, $x = \sum_{i,j} \lambda_{ij}\, e_i f_j$: os produtos geram $M$ sobre $\Q$.

\textbf{10.} Suponha $\sum_{i,j} \lambda_{ij}\, e_i f_j = 0$ com $\lambda_{ij} \in \Q$. Reagrupe: $\sum_j \bigl(\sum_i \lambda_{ij}
e_i\bigr) f_j = 0$, e as somas
internas pertencem a $K$. A liberdade de $(f_j)$ sobre $K$ dá
$\sum_i \lambda_{ij} e_i = 0$ para cada $j$; a liberdade de $(e_i)$ sobre $\Q$ dá então
$\lambda_{ij} = 0$ para todos $i, j$.

\textbf{11.} Pelas questões 9 e 10, $(e_i f_j)_{i \leq m,\, j
\leq n}$ é uma \omterm{def:b1:vspaces:free}{base} de $M$ sobre
$\Q$ com $mn$ elementos:
\[
\dim_\Q M = m\,n = \dim_\Q K \cdot \dim_K M .
\]
Verificação: $\dim_\Q K = 2$ e $\dim_K M = 2$ (questões 2 e 7) dão $\dim_\Q M = 4$, que é a
questão 8.

\textbf{12.} Os vetores $u a_1, \dots, u a_d$ pertencem a $A$ (estabilidade).
Eles são \omterm{def:b1:vspaces:free}{livres}: se $\sum_i \lambda_i\, u a_i = 0$, então $u \sum_i \lambda_i a_i = 0$ em $\R$, e $u \neq 0$ força
$\sum_i \lambda_i a_i = 0$, donde $\lambda_i = 0$ (os $a_i$ formam uma \omterm{def:b1:vspaces:free}{base}). Logo $(u a_1, \dots, u a_d)$ é uma
\omterm{def:b1:vspaces:free}{família livre} de $d$ vetores em $A$, $\dim_\Q A = d$: uma \omterm{def:b1:vspaces:free}{base} (a \cref{prop:b1:findim:twoofthree}).
Em particular, $1 \in A$ decompõe-se como $1 = \sum_i \mu_i\, u a_i = u\,v$ com $v = \sum_i
\mu_i a_i \in A$: o
inverso de $u$ está em $A$. Isto recupera a questão 4 ($A = \Q(\sqrt2)$) e
demonstra de uma só vez que $M$ é um subcorpo de $\R$ (com a
questão 6).

\textbf{13.} Os $d + 1$ vetores $1, x, x^2, \dots, x^{d}$ estão todos em $A$
(estabilidade por produtos); uma \omterm{def:b1:vspaces:free}{família livre} de $A$ tem no
máximo $d$ vetores (a \cref{prop:b1:findim:twoofthree}), logo eles são linearmente
dependentes: existem racionais $\lambda_0, \dots, \lambda_d$, não todos nulos, com $\sum_k \lambda_k x^k = 0$.
O \omterm{def:b1:poly:def}{polinômio} $P
= \sum_k \lambda_k X^k$ é não nulo, tem coeficientes racionais, grau
$\leq d$, e $P(x) = 0$.

\textbf{14.} $s^2 = 2 + 2\sqrt6 + 3 = 5 + 2\sqrt6$: \omterm{prop:b1:vspaces:coordinates}{coordenadas} $(5, 0, 0, 2)$. Então
\[
s^3 = s\,s^2 = (\sqrt2 + \sqrt3)(5 + 2\sqrt6)
= 5\sqrt2 + 2\sqrt{12} + 5\sqrt3 + 2\sqrt{18}
= 11\sqrt2 + 9\sqrt3 ,
\]
\omterm{prop:b1:vspaces:coordinates}{coordenadas} $(0, 11, 9, 0)$ (usando $\sqrt{12} = 2\sqrt3$, $\sqrt{18} = 3\sqrt2$). Por fim $s^4 = (s^2)^2 = 25 + 20\sqrt6 +
24 = 49 + 20\sqrt6$:
\omterm{prop:b1:vspaces:coordinates}{coordenadas} $(49, 0, 0, 20)$.

\textbf{15.} Suponha $a + bs + cs^2 + ds^3 = 0$. Lendo as quatro \omterm{prop:b1:vspaces:coordinates}{coordenadas} em $(1, \sqrt2, \sqrt3, \sqrt6)$:
\[
a + 5c = 0,\qquad b + 11d = 0,\qquad b + 9d = 0,\qquad 2c = 0 .
\]
Logo $c = 0$, e então $a = 0$; subtraindo as equações do meio,
$2d = 0$, e então $b = 0$: $(1, s, s^2, s^3)$ é \omterm{def:b1:vspaces:free}{livre}. Quatro vetores \omterm{def:b1:vspaces:free}{livres} em
$M$, de dimensão $4$: uma \omterm{def:b1:vspaces:free}{base}, logo $\operatorname{Vect}_\Q(1, s, s^2, s^3) = M$. Da questão 14, $s^3 - 9s = 2\sqrt2$ e
$11s - s^3 = 2\sqrt3$:
\[
\sqrt2 = \frac{s^3 - 9s}{2},
\qquad
\sqrt3 = \frac{11s - s^3}{2} .
\]

\textbf{16.} $s^4 - 10s^2 + 1 = (49 + 20\sqrt6) - 10(5 +
2\sqrt6) + 1 = 0$. Um \omterm{def:b1:poly:def}{polinômio mônico} de grau $\leq 3$ que se
anulasse em $s$ produziria uma combinação nula não trivial de
$(1, s, s^2, s^3)$, contradizendo a questão 15: $X^4 - 10X^2 + 1$ tem grau mínimo. As suas
raízes: $X^2 = 5 \pm 2\sqrt6 = (\sqrt3 \pm
\sqrt2)^2$, logo as quatro raízes reais são $\pm(\sqrt3 + \sqrt2)$ e $\pm(\sqrt3 - \sqrt2)$,
isto é, $\pm\sqrt2 \pm \sqrt3$.

\textbf{17.} De $s^4 - 10s^2 + 1 = 0$: $s\,(10s - s^3) = 1$, logo
\[
\frac1s = 10s - s^3 = 10(\sqrt2 + \sqrt3) - (11\sqrt2 + 9\sqrt3)
= \sqrt3 - \sqrt2 ,
\]
coerente com $(\sqrt3 + \sqrt2)(\sqrt3 - \sqrt2) = 1$. Se $a +
b\sqrt2 + c\sqrt3 + d\sqrt6 = r \in \Q$, então $(a - r) + b\sqrt2
+ c\sqrt3 + d\sqrt6 = 0$, e a liberdade
(questão 8) força $b = c
= d = 0$ (e $a = r$). Para $\sqrt2 + \sqrt3 + \sqrt6$, as \omterm{prop:b1:vspaces:coordinates}{coordenadas}
$(0, 1, 1, 1)$ não são dessa forma: irracional --- um enunciado mais forte
do que o \cref{exo:b1:reals:7}, obtido sem nenhum truque de elevar ao quadrado.

\textbf{18.} Uma raiz racional $p/q$ (em forma irredutível) de
$X^3 - 2$ satisfaz $p^3 = 2q^3$, e o critério do \cref{exo:b1:poly:5} dá $p \mid 2$, $q \mid 1$:
candidatos $\pm1, \pm2$, cujos cubos são $\pm1, \pm8 \neq 2$. Nenhuma raiz racional;
em particular $t = 2^{1/3} \notin \Q$, logo $(1, t)$ é \omterm{def:b1:vspaces:free}{livre} sobre $\Q$ (como na
questão 2).

\textbf{19.} Suponha $(1, t, t^2)$ linearmente dependente: algum $P \in
\Q[X]$
não nulo com $\deg P \leq 2$ tem $P(t) = 0$; escolha um tal $P$ de grau
\emph{mínimo} $d \geq 1$. Pela questão 18, $d \neq 1$, logo $d = 2$. Divisão
euclidiana (\cref{thm:b1:poly:division}): $X^3 - 2
= PQ + R$ com $Q, R \in \Q[X]$, $\deg R < 2$. Avaliando em $t$:
$0 = P(t)Q(t) + R(t) = R(t)$, logo $R$ se anula em $t$; a minimalidade de $d$ força
$R = 0$. Então $X^3 - 2 = PQ$ com $\deg Q = 1$: a raiz racional de $Q$ é uma
raiz racional de $X^3 - 2$, contradizendo a questão 18. Logo
$(1, t, t^2)$ é \omterm{def:b1:vspaces:free}{livre} e $\dim_\Q A = 3$. Estabilidade: $t^3 = 2$ reduz todo
produto de $1, t, t^2$ a uma combinação deles ($t\cdot t^2 = 2$, $t^2\cdot
t^2 = 2t$); $A$
contém $1$: pela questão 12, $A$ é um subcorpo de $\R$.

\textbf{20.} Suponha $t \in M$. Então $t^2 \in M$ (estabilidade), logo
$A \subseteq M$, e $M$ é um \omterm{def:b1:vspaces:def}{espaço vetorial} sobre o \omterm{def:b1:structures:field}{corpo} $A$: os axiomas
são os da aritmética de $\R$, restrita. Ele tem \omterm{def:b1:findim:def}{dimensão finita}
sobre $A$ (uma família $\Q$-geradora finita gera a fortiori sobre
$A \supseteq \Q$). A \omterm{pb:b1:findim:1}{lei das torres} para $\Q \subseteq A \subseteq M$ dá
\[
4 = \dim_\Q M = \dim_\Q A \cdot \dim_A M = 3\,\dim_A M ,
\]
impossível: $3$ não \omterm{def:b1:arith:divides}{divide} $4$. Logo $2^{1/3} \notin M$: nenhuma combinação
racional de $1, \sqrt2, \sqrt3, \sqrt6$ é igual a $2^{1/3}$.

\textbf{21.} $\Q(\sqrt2) \subseteq M$, logo $t \notin \Q(\sqrt2)$. Se $t + \sqrt2 = r \in \Q$, então $t = r - \sqrt2 \in \Q(\sqrt2)$:
contradição --- $2^{1/3} + \sqrt2$ é irracional. Se $t = a +
b\sqrt3$, então $t \in \operatorname{Vect}_\Q(1, \sqrt3) \subseteq
M$:
contradição de novo.

\textbf{22.} $B$ é um \omterm{def:b1:vspaces:def}{espaço vetorial} sobre o \omterm{def:b1:structures:field}{corpo} $A$
(restrição dos escalares), de \omterm{def:b1:findim:def}{dimensão finita} já que $\dim_\Q B$ é
finita. A \omterm{pb:b1:findim:1}{lei das torres} para $\Q \subseteq A \subseteq B$ dá $\dim_\Q B = \dim_\Q A \cdot \dim_A B$: o fator da esquerda
\omterm{def:b1:arith:divides}{divide}. Os subcorpos de $M$ têm portanto $\Q$-dimensão $1$, $2$
ou $4$: a dimensão $1$ é o próprio $\Q$, e a dimensão $4$ é $M$.

\textbf{23.} Seja $F \subseteq M$ um subcorpo com $\dim_\Q F =
2$, e $w \in F \setminus \Q$: $(1, w)$
é \omterm{def:b1:vspaces:free}{livre}, logo uma \omterm{def:b1:vspaces:free}{base} de $F$. Pela questão 13 ($d = 2$), $w^2 = \alpha + \beta w$
para racionais $\alpha, \beta$. Pondo $v = w - \beta/2 \in F
\setminus \Q$:
\[
v^2 = w^2 - \beta w + \frac{\beta^2}4 = \alpha + \frac{\beta^2}4
\;\in\; \Q,
\]
e $F = \operatorname{Vect}(1, v)$. Escreva agora $v = a + b\sqrt2 +
c\sqrt3 + d\sqrt6$ e desenvolva:
\[
v^2 = (a^2 + 2b^2 + 3c^2 + 6d^2)
+ (2ab + 6cd)\sqrt2 + (2ac + 4bd)\sqrt3 + (2ad + 2bc)\sqrt6 .
\]
As três \omterm{prop:b1:vspaces:coordinates}{coordenadas} irracionais se anulam: $ab + 3cd = ac + 2bd =
ad + bc = 0$. Se $a \neq 0$:
$b = -3cd/a$, e a segunda condição torna-se $c(a^2 - 6d^2) = 0$; como $a^2 = 6d^2$ com $d
\neq 0$
tornaria $\sqrt6 = \abs{a/d}$ racional, ou $c = 0$ ou $d = 0$, e nos dois casos
as condições restantes forçam $b =
c = d = 0$, isto é, $v \in \Q$: excluído.
Logo $a = 0$, e as condições leem-se $3cd = 2bd = bc = 0$: no máximo um dentre
$b, c, d$ é não nulo. Portanto $v$ é um múltiplo racional de
$\sqrt2$, $\sqrt3$ ou $\sqrt6$, e
\[
F \in \bigl\{\Q(\sqrt2),\ \Q(\sqrt3),\ \Q(\sqrt6)\bigr\} ,
\]
cada um dos quais é de fato um subcorpo de dimensão $2$
(estabilidade como na questão 6, inversos pela questão 12):
exatamente três subcorpos quadráticos.

\textbf{24.} $(1 + \sqrt2 + \sqrt3)(1 + \sqrt2 - \sqrt3) = (1 +
\sqrt2)^2 - 3 = 2\sqrt2$, logo
\[
\frac1{1 + \sqrt2 + \sqrt3}
= \frac{1 + \sqrt2 - \sqrt3}{2\sqrt2}
= \frac{\sqrt2 + 2 - \sqrt6}{4}
= \frac12 + \frac14\sqrt2 + 0\cdot\sqrt3 - \frac14\sqrt6 .
\]
(Verificação: $(1 + \sqrt2 + \sqrt3)(2 + \sqrt2 - \sqrt6) = 4$ após o desenvolvimento.)

\textbf{25.} (i) A $\Q$-dimensão associa um \emph{inteiro} a cada
subcorpo de $\R$, e a \omterm{pb:b1:findim:1}{lei das torres} faz esses inteiros
multiplicarem-se ao longo das inclusões: a dimensão comporta-se
como um invariante aritmético, e as restrições de \omterm{def:b1:arith:divides}{divisibilidade}
tornam-se demonstrações de impossibilidade. (ii) A \omterm{def:b1:findim:def}{dimensão
finita} força todo elemento a satisfazer uma equação polinomial
racional não nula de grau no máximo igual à dimensão: finitude
significa algebricidade. (iii) A regra dois em três transformou
``a multiplicação por $u$ leva uma \omterm{def:b1:vspaces:free}{base} numa \omterm{def:b1:vspaces:free}{família livre} de
tamanho máximo'' em sobrejetividade, produzindo $u^{-1}$ sem
nenhuma fórmula: os inversos vieram da contagem. (iv) A \omterm{pb:b1:findim:1}{lei das
torres} proíbe um \omterm{def:b1:structures:field}{corpo} de dimensão $3$ dentro de um de dimensão
$4$, e é por isso que $2^{1/3}$ não pode ser alcançado a partir de
$\sqrt2$ e $\sqrt3$. O teorema da Parte III é a \emph{\omterm{pb:b1:findim:1}{lei das
torres} de Dedekind}, o lance de abertura da teoria de Galois,
desenvolvida no volume do terceiro ano de graduação.
\end{solution}
