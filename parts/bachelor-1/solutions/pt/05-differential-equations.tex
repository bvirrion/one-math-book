\section*{Capítulo \ref{ch:b1:diffeq} --- Equações Diferenciais Lineares}

\begin{solution}{exo:b1:diffeq:1}
Homogênea: $y_h = \lambda\,\eu^{-2x}$. Particular: tente $y_p =
c\,\eu^{3x}$ ($3$ não é raiz de $r + 2$): $3c + 2c = 1$, $c =
\frac15$. Solução geral: $y = \frac{\eu^{3x}}{5} +
\lambda\,\eu^{-2x}$. Com $y(0) = 1$: $\frac15 + \lambda = 1$,
$\lambda = \frac45$, de modo que $y = \frac{\eu^{3x} + 4\,\eu^{-2x}}{5}$.
\end{solution}

\begin{solution}{exo:b1:diffeq:2}
Em $\intoo{0}{+\infty}$, divida por $x$: $y' - \frac{2}{x}\,y = x^2$.
Aqui $A(x) = -2\ln x$, $\eu^{-A(x)} = x^2$: soluções homogêneas
$\lambda x^2$. \omterm{thm:b1:diffeq:voc}{Variação das constantes}: $\mu'(x) = x^2 \cdot x^{-2} =
1$, de modo que $\mu = x + \lambda$ e
\[
y(x) = x^3 + \lambda x^2, \qquad \lambda \in \R .
\]
\emph{Verificação:} $x(3x^2 + 2\lambda x) - 2(x^3 + \lambda x^2) = x^3$.
\end{solution}

\begin{solution}{exo:b1:diffeq:3}
$y'' - 3y' + 2y = 0$: raízes $1$ e $2$; $\;y = \lambda\,\eu^{x} +
\mu\,\eu^{2x}$.

$y'' + 4y' + 4y = 0$: raiz dupla $-2$; $\;y = (\lambda + \mu
x)\,\eu^{-2x}$.

$y'' - 2y' + 5y = 0$: raízes $1 \pm 2\iu$; $\;y = \eu^{x}(\lambda\cos
2x + \mu\sin 2x)$.
\end{solution}

\begin{solution}{exo:b1:diffeq:4}
Homogênea: raízes $\pm 1$, $y_h = \lambda\,\eu^x + \mu\,\eu^{-x}$.
Particular com lado direito polinomial ($\gamma = 0$ não é raiz):
$y_p = ax^2 + bx + c$; substituindo, $2a - (ax^2 + bx + c) = x^2$
dá $a = -1$, $b = 0$, $c = 2a = -2$: $y_p = -x^2 - 2$. Solução
geral: $y = -x^2 - 2 + \lambda\eu^x + \mu\eu^{-x}$.

Cauchy: $y(0) = -2 + \lambda + \mu = 0$ e $y'(0) = \lambda - \mu =
1$: $\lambda = \frac32$, $\mu = \frac12$. Logo,
$y = -x^2 - 2 + \frac{3\eu^x + \eu^{-x}}{2}$.
\end{solution}

\begin{solution}{exo:b1:diffeq:5}
$a(x) = \tan x$, $A(x) = -\ln(\cos x)$ (válido: $\cos > 0$ no
intervalo), $\eu^{-A} = \cos x$: soluções homogêneas $\lambda\cos x$.
\omterm{thm:b1:diffeq:voc}{Variação das constantes}: $\mu'(x) = \sin 2x \cdot \frac{1}{\cos x} =
2\sin x$, de modo que $\mu = -2\cos x + \lambda$ e
\[
y(x) = -2\cos^2 x + \lambda \cos x .
\]
\emph{Verificação:} $y' = 4\cos x \sin x - \lambda\sin x$ e $y\tan x =
-2\cos x\sin x + \lambda \sin x$; a soma vale $2\cos x\sin x =
\sin 2x$, como exigido.
\end{solution}

\begin{solution}{exo:b1:diffeq:6}
$\chi(r) = r^2 - 4r + 3 = (r-1)(r-3)$: $\gamma = 1$ é raiz simples
($m = 1$). Tente $y_p = x(ax + b)\,\eu^x$. Com $u = ax^2 + bx$,
\[
y_p'' - 4y_p' + 3y_p = \bigl(u'' + (2 - 4)u' + \chi(1) u\bigr)\eu^x
= \bigl(2a - 2(2ax + b)\bigr)\eu^x .
\]
Identificando com $(2x + 1)\eu^x$: $-4a = 2$ e $2a - 2b = 1$, de modo que $a =
-\frac12$, $b = -1$. Solução geral:
\[
y = -\Bigl(\frac{x^2}{2} + x\Bigr)\eu^{x} + \lambda\,\eu^{x} +
\mu\,\eu^{3x}, \qquad (\lambda, \mu) \in \R^2 .
\]
\end{solution}

\begin{solution}{exo:b1:diffeq:7}
Homogênea: $y_h = \lambda\cos 2x + \mu\sin 2x$.

Lado direito $x$ ($\gamma = 0$ não é raiz): $y_1 = ax + b$ com
$4(ax + b) = x$: $y_1 = \frac x4$.

Lado direito $\sin 2x = \Im(\eu^{2\iu x})$, sendo $2\iu$ raiz simples de
$r^2 + 4$: tente $z = c\,x\,\eu^{2\iu x}$; então $z'' + 4z =
4\iu c\,\eu^{2\iu x}$, igual a $\eu^{2\iu x}$ para $c =
\frac{1}{4\iu} = -\frac{\iu}{4}$. Logo, $z = -\frac{\iu x}{4}(\cos 2x +
\iu \sin 2x)$ e $y_2 = \Im(z) = -\frac{x\cos 2x}{4}$.

Por superposição:
\[
y = \frac{x}{4} - \frac{x\cos 2x}{4} + \lambda\cos 2x + \mu\sin 2x .
\]
\end{solution}

\begin{solution}{exo:b1:diffeq:8}
A equação $T' + kT = 20k$ tem a solução particular constante $20$ e
as soluções homogêneas $\lambda\eu^{-kt}$: $T(t) = 20 +
\lambda\,\eu^{-kt}$, e $T(0) = 80$ dá $\lambda = 60$:
\[
T(t) = 20 + 60\,\eu^{-kt} .
\]
$T(10) = 50$: $\eu^{-10k} = \frac12$, de modo que $k = \frac{\ln 2}{10}$. Então
$T(t) = 25$ exige $\eu^{-kt} = \frac{5}{60} = \frac{1}{12}$, isto é,
\[
t = \frac{\ln 12}{k} = 10\,\frac{\ln 12}{\ln 2} \approx 35.8
\text{ minutos.}
\]
\end{solution}

\begin{solution}{exo:b1:diffeq:9}
\begin{enumerate}
  \item $\chi(r) = r^2 + 2\varepsilon r + 1$, $\Delta = 4(\varepsilon^2
        - 1)$.
        Para $\varepsilon \in \intco{0}{1}$: raízes $-\varepsilon \pm
        \iu\sqrt{1 - \varepsilon^2}$, de modo que
        $y = \eu^{-\varepsilon t}\bigl(\lambda\cos\omega t +
        \mu\sin\omega t\bigr)$ com $\omega = \sqrt{1 -
        \varepsilon^2}$.
        Para $\varepsilon = 1$: raiz dupla $-1$, $y = (\lambda + \mu
        t)\,\eu^{-t}$.
        Para $\varepsilon > 1$: raízes reais $r_\pm = -\varepsilon \pm
        \sqrt{\varepsilon^2 - 1}$, ambas $< 0$, e $y =
        \lambda\eu^{r_+t} + \mu\eu^{r_-t}$.
  \item Para $\varepsilon \in \intoo{0}{1}$: $\abs y \leq
        \eu^{-\varepsilon t}(\abs\lambda + \abs\mu) \to 0$. Para
        $\varepsilon = 1$: $(\lambda + \mu t)\eu^{-t} \to 0$
        (a exponencial vence o polinômio,
        \cref{prop:b1:functions:powerrules}). Para $\varepsilon > 1$:
        as duas exponenciais decaem, pois $r_\pm < 0$ (com efeito,
        $\sqrt{\varepsilon^2 - 1} < \varepsilon$). Para $\varepsilon =
        0$: $y = \lambda\cos t + \mu\sin t$ tem amplitude constante
        $\sqrt{\lambda^2 + \mu^2} \neq 0$, a menos que $y = 0$.
  \item Escreva $\lambda\cos\omega t + \mu\sin\omega t =
        R\cos(\omega t - \varphi)$ com $R = \sqrt{\lambda^2 + \mu^2}
        > 0$. Os zeros de $y$ são os de $\cos(\omega t -
        \varphi)$ (o fator $\eu^{-\varepsilon t}$ nunca
        se anula): $\omega t - \varphi \equiv \frac\pi2 \pmod \pi$,
        uma progressão aritmética de passo $\frac{\pi}{\omega} =
        \frac{\pi}{\sqrt{1 - \varepsilon^2}}$.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b1:diffeq:10}
Ponha $x = y = 0$: $2f(0) = 2f(0)^2$, e $f(0) \neq 0$ dá $f(0) =
1$. Fixe $x$ e derive duas vezes a equação em relação a $y$:
\[
f''(x+y) + f''(x-y) = 2 f(x) f''(y) .
\]
Fazendo $y = 0$: $\;2f''(x) = 2 f(x) f''(0)$, isto é,
\[
f''(x) = c\,f(x), \qquad c = f''(0).
\]
\emph{Caso $c = \omega^2 > 0$:} $f(x) = \lambda\cosh\omega x +
\mu\sinh\omega x$; $f(0) = 1$ dá $\lambda = 1$. Substituindo na
equação funcional e usando as fórmulas de adição
(\cref{prop:b1:functions:hyprules}), a equação força $\mu = 0$
(compare os coeficientes de $\sinh\omega x \sinh\omega y$ ou
avalie em $x = y$): $f = \cosh\omega x$, que de fato satisfaz
$\cosh(x+y) + \cosh(x-y) = 2\cosh x\cosh y$.

\emph{Caso $c = -\omega^2 < 0$:} do mesmo modo, $f(x) = \cos\omega x$
($\omega \neq 0$), que satisfaz a equação.

\emph{Caso $c = 0$:} $f$ é afim com $f(0) = 1$: $f(x) = 1 + \mu x$;
a equação força $\mu = 0$, o que está excluído ($f$ não constante).

Conclusão: as soluções são $f(x) = \cos\omega x$ e $f(x) =
\cosh\omega x$, $\omega > 0$.
\end{solution}

\begin{solution}{exo:b1:diffeq:11}
Ponha $z(t) = y(\eu^t)$, de modo que $y(x) = z(\ln x)$ para $x > 0$. Então
\[
y'(x) = \frac{z'(\ln x)}x,
\qquad
y''(x) = \frac{z''(\ln x) - z'(\ln x)}{x^2} ,
\]
e, substituindo na equação:
\[
x^2y'' - xy' + y = \bigl(z'' - z'\bigr) - z' + z
= z'' - 2z' + z = 0 .
\]
\omterm{def:b1:diffeq:linear2}{Polinômio característico} $(r - 1)^2$: raiz dupla $1$, de modo que $z(t) =
(\lambda + \mu t)\,\eu^t$ e, de volta à variável $x = \eu^t$:
\[
y(x) = (\lambda + \mu\ln x)\,x,
\qquad \lambda, \mu \in \R .
\]
\end{solution}

\begin{solution}{exo:b1:diffeq:12}
\begin{enumerate}
  \item Na forma normalizada $y' - \frac2x\,y = 0$ em cada intervalo:
        $A(x) = -2\ln\abs x$, de modo que as soluções são $y = a x^2$ em
        $\intoo0{+\infty}$ e $y = b x^2$ em $\intoo{-\infty}0$,
        com constantes independentes
        (\cref{thm:b1:diffeq:homogeneous1}).
  \item Seja $y = ax^2$ para $x \geq 0$ e $bx^2$ para $x < 0$. Em
        cada semirreta aberta, $y$ é derivável com $xy' = 2y$.
        Em $0$: os quocientes de diferenças $\frac{y(h) - y(0)}h =
        ah$
        ou $bh$ tendem a $0$, de modo que $y'(0) = 0$ existe, e a
        equação em $x = 0$ diz que $0 \cdot y'(0) = 2y(0) = 0$:
        satisfeita. Logo, $y$ resolve a equação em todo o $\R$.
  \item As soluções em $\R$ são exatamente essas funções coladas: uma
        família a \emph{dois} parâmetros para uma equação de primeira
        ordem. Não há contradição com o \cref{thm:b1:diffeq:voc},
        cujas hipóteses falham aqui: escrito como $y' + a(x)y = 0$,
        o coeficiente $a(x) = -\frac2x$ não é contínuo em $0$
        --- nem sequer está definido --- de modo que $\R$ não é um
        intervalo no qual o teorema se aplique. A singularidade em $0$
        desconecta as duas semirretas, e o valor $y(0) = 0$
        fica forçado, sem transportar informação alguma de um lado ao
        outro. Todo dado de Cauchy em $x_0 \neq 0$ determina a solução
        apenas na semirreta que contém $x_0$.
\end{enumerate}
\end{solution}

\begin{solution}{pb:b1:diffeq:1}
\textbf{1.} $\chi(r) = r^2 + 2\lambda r + \omega_0^2$, $\Delta =
4(\lambda^2 - \omega_0^2) < 0$ para $0 < \lambda < \omega_0$: raízes
$-\lambda \pm \iu\omega_d$, de modo que, pelo
\cref{thm:b1:diffeq:homogeneous2},
\[
x(t) = \eu^{-\lambda t}\bigl(\lambda_1\cos\omega_d t +
\mu_1\sin\omega_d t\bigr), \qquad (\lambda_1, \mu_1) \in \R^2 .
\]
Para $\lambda = 0$: $x = \lambda_1\cos\omega_0 t +
\mu_1\sin\omega_0 t$. Os regimes crítico ($\lambda = \omega_0$) e
superamortecido ($\lambda > \omega_0$) são os do
\cref{exo:b1:diffeq:9} (após reescalonar o tempo): $(\lambda_1 +
\mu_1 t)\eu^{-\lambda t}$, resp.\ combinações de $\eu^{r_\pm t}$
com $r_\pm = -\lambda \pm \sqrt{\lambda^2 - \omega_0^2}$.

\textbf{2.} Subamortecido: $\abs x \leq \eu^{-\lambda
t}(\abs{\lambda_1} + \abs{\mu_1}) \to 0$. Crítico: $(\lambda_1 +
\mu_1 t)\eu^{-\lambda t} \to 0$, pois as exponenciais vencem os
polinômios (\cref{prop:b1:functions:powerrules}). Superamortecido:
$r_- < r_+ = -\lambda + \sqrt{\lambda^2 - \omega_0^2} < 0$
porque $\sqrt{\lambda^2 - \omega_0^2} < \lambda$; as duas
exponenciais decaem.

\textbf{3.} Ao longo de uma solução de $(H)$, usando $x'' = -2\lambda x' -
\omega_0^2 x$:
\[
\mathcal E'(t) = x'x'' + \omega_0^2 x x'
= x'\bigl(-2\lambda x' - \omega_0^2 x\bigr) + \omega_0^2 xx'
= -2\lambda\,x'^2 \leq 0 .
\]
Se $x(t_0) = x'(t_0) = 0$, então $\mathcal E(t_0) = 0$; $\mathcal E$
é não negativa e não crescente, de modo que $\mathcal E \equiv 0$ em
$\intco{t_0}{+\infty}$, o que força $x \equiv 0$ ali; para $t \leq
t_0$, aplique o mesmo argumento a $\tilde x(t) = x(2t_0 - t)$, que
resolve a equação com amortecimento $-\lambda$, mas ainda tem
$\tilde{\mathcal E}(t_0) = 0$ e $\tilde{\mathcal E}' =
+2\lambda\tilde x'^2 \geq 0$ com $\tilde{\mathcal E} \geq 0$;
não negativa, não decrescente e nula na extremidade direita de
$\intoc{-\infty}{t_0}$ significa nula em toda parte. Logo, $x \equiv 0$ em
$\R$ --- uma demonstração de unicidade por energia, válida para todo
$\lambda
\geq 0$.

\textbf{4.} $x(t + T_d) = R\,\eu^{-\lambda t}\eu^{-\lambda
T_d}\cos(\omega_d t + 2\pi - \varphi) = \eu^{-\lambda T_d}x(t)$.
O fator de encolhimento por pseudoperíodo é $\eu^{-\delta}$ com
$\delta = \lambda T_d = \frac{2\pi\lambda}{\omega_d}$. Para
$\omega_0 = 1$, $\lambda = 0.1$: $\omega_d = \sqrt{0.99} =
0.99499$, de modo que $\delta = \frac{0.62832}{0.99499} = 0.6315$: cada
oscilação guarda $\eu^{-0.63} \approx 53\%$ da sua amplitude.

\textbf{5.} O fator de amplitude é $\eu^{-\lambda t}$, que
decai por um fator $\eu$ ao longo de $t = \frac1\lambda$. Esse intervalo contém
$\frac{1/\lambda}{T_d} = \frac{\omega_d}{2\pi\lambda}$
pseudoperíodos. Para $\lambda \ll \omega_0$, $\omega_d \approx
\omega_0$ e isso vale $\approx \frac{\omega_0}{2\pi\lambda} =
\frac Q\pi$. Uma corda de violão com $Q = 300$ soa por cerca de cem
períodos; um amortecedor de porta com $Q = 1$ não completa nem um.

\textbf{6.} Substituir $z\,\eu^{\iu\Omega t}$ no lado esquerdo dá
$z\,(-\Omega^2 + 2\iu\lambda\Omega +
\omega_0^2)\,\eu^{\iu\Omega t}$, que é igual a $A\,\eu^{\iu\Omega
t}$
exatamente para $z = \frac{A}{\omega_0^2 - \Omega^2 +
2\iu\lambda\Omega}$ (o denominador é não nulo: a sua parte
imaginária é $2\lambda\Omega > 0$). Como os coeficientes são reais,
a parte real $x_p = \Re\bigl(z\eu^{\iu\Omega t}\bigr)$ resolve a
equação com lado direito $\Re\bigl(A\eu^{\iu\Omega t}\bigr) =
A\cos\Omega t$.

\textbf{7.} Escreva $\omega_0^2 - \Omega^2 + 2\iu\lambda\Omega =
\sqrt{D}\,\eu^{\iu\varphi}$ com $D = (\omega_0^2 - \Omega^2)^2 +
4\lambda^2\Omega^2$ e $\varphi \in \intoo0\pi$ (a parte imaginária
$2\lambda\Omega$ é positiva), de modo que $\tan\varphi =
\frac{2\lambda\Omega}{\omega_0^2 - \Omega^2}$. Então $z =
\frac{A}{\sqrt D}\eu^{-\iu\varphi}$ e
\[
x_p(t) = \Re\Bigl(\frac A{\sqrt D}\,\eu^{\iu(\Omega t -
\varphi)}\Bigr) = R\cos(\Omega t - \varphi),
\qquad R = \frac A{\sqrt D} .
\]

\textbf{8.} Quando $\Omega \to 0^+$: $D \to \omega_0^4$, de modo que $R \to
A/\omega_0^2$ e $\tan\varphi \to 0^+$ com $\varphi \in
\intoo0{\frac\pi2}$: $\varphi \to 0$. A massa segue a força
quase estaticamente, deslocada por força/rigidez. Quando $\Omega \to
+\infty$: $D \sim \Omega^4$, de modo que $R \sim A/\Omega^2 \to 0$, e
$\varphi \to \pi$ (o número complexo $\omega_0^2 - \Omega^2 +
2\iu\lambda\Omega$ vai para o segundo quadrante, com argumento $\to
\pi$): a massa mal se move, e em oposição de fase ---
a inércia domina.

\textbf{9.} Pelo \cref{thm:b1:diffeq:structure2}~(1), toda solução
é $x = x_p + x_h$ com $x_h$ resolvendo $(H)$; pela questão 2, $x_h(t)
\to 0$, de modo que $x(t) - x_p(t) \to 0$: todas as soluções convergem ao
mesmo regime permanente. As condições iniciais só moldam o transitório.

\textbf{10.} Se $x$ é uma solução periódica, $x - x_p = x_h$ é uma
solução periódica de $(H)$ que tende a $0$ em $+\infty$; uma
função periódica com limite $0$ é identicamente $0$ (os seus valores
num período se repetem para sempre, de modo que todo valor é limite de
uma subsequência que tende a $0$). Portanto, $x = x_p$.

\textbf{11.} Aqui $\lambda = 1$, $\omega_0^2 = 2$, $\Omega = 1$,
$A = 1$: $z = \frac1{2 - 1 + 2\iu} = \frac{1 - 2\iu}5$, de modo que
\[
x_p = \Re\Bigl(\frac{(1 - 2\iu)(\cos t + \iu\sin t)}5\Bigr)
= \frac{\cos t + 2\sin t}5 .
\]
Homogênea: as raízes de $r^2 + 2r + 2$ são $-1 \pm \iu$: $x_h =
\eu^{-t}(C\cos t + S\sin t)$. Condições: $x(0) = \frac15 + C = 0$
dá $C = -\frac15$; derivando, $x'(0) = \frac25 - C + S =
0$
dá $S = C - \frac25 = -\frac35$. Portanto,
\[
x(t) = \underbrace{\frac{\cos t + 2\sin t}5}_{\text{permanente}}
- \underbrace{\eu^{-t}\,\frac{\cos t + 3\sin
t}5}_{\text{transitório}} ,
\]
com o transitório morrendo como $\eu^{-t}$.

\textbf{12.} Expandindo, $g(u) = u^2 - 2(\omega_0^2 - 2\lambda^2)u
+ \omega_0^4 = (u - u_r)^2 + g(u_r)$ com $u_r = \omega_0^2 -
2\lambda^2$ e
\[
g(u_r) = \omega_0^4 - u_r^2 = (\omega_0^2 - u_r)(\omega_0^2 +
u_r) = 2\lambda^2\bigl(2\omega_0^2 - 2\lambda^2\bigr) =
4\lambda^2(\omega_0^2 - \lambda^2) .
\]
Se $2\lambda^2 < \omega_0^2$, então $u_r > 0$ é uma frequência ao quadrado
admissível: $g$ tem ali um mínimo estrito, de modo que $R = A/\sqrt
g$ tem um máximo estrito em $\Omega_r = \sqrt{u_r} =
\sqrt{\omega_0^2 - 2\lambda^2}$, com $R_{\max} = A/\sqrt{g(u_r)}
= \frac{A}{2\lambda\sqrt{\omega_0^2 - \lambda^2}}$.

\textbf{13.} $R(0) = A/\omega_0^2$, de modo que
\[
\frac{R_{\max}}{R(0)}
= \frac{\omega_0^2}{2\lambda\sqrt{\omega_0^2 - \lambda^2}}
= \frac{\omega_0}{2\lambda}\cdot
\frac{\omega_0}{\sqrt{\omega_0^2 - \lambda^2}}
= Q\,\Bigl(1 - \frac{\lambda^2}{\omega_0^2}\Bigr)^{-1/2}
\geq Q .
\]
Para amortecimento fraco, o fator de correção é próximo de $1$: a
\omterm{ex:b1:diffeq:oscillation}{ressonância} multiplica o deslocamento estático essencialmente por $Q$.

\textbf{14.} $V(\Omega)^2 = \frac{A^2 u}{g(u)}$ com $u =
\Omega^2$. A sua derivada tem o sinal de $g(u) - u\,g'(u) =
(\omega_0^2 - u)^2 + 4\lambda^2 u - u\bigl(2(u - \omega_0^2) +
4\lambda^2\bigr) = (\omega_0^2 - u)^2 + 2u(\omega_0^2 - u) =
(\omega_0^2 - u)(\omega_0^2 + u)$, positivo para $u < \omega_0^2$
e negativo além disso: máximo estrito exatamente em $\Omega =
\omega_0$, para todo amortecimento. Ali $\tan\varphi$ explode, com
$\varphi \in \intoo0\pi$: $\varphi = \frac\pi2$, e $x_p'(t) =
-R\Omega\sin(\Omega t - \frac\pi2) = R\Omega\cos(\Omega t)$ está
exatamente em fase com a força: transferência ótima de potência.

\textbf{15.} $R = R_{\max}/\sqrt2 \iff g(u) = 2g(u_r) \iff (u -
u_r)^2 = g(u_r) = 4\lambda^2(\omega_0^2 - \lambda^2)$, o que dá
\[
u_\pm = u_r \pm 2\lambda\sqrt{\omega_0^2 - \lambda^2},
\qquad
\Omega_+^2 - \Omega_-^2 = 4\lambda\sqrt{\omega_0^2 - \lambda^2} .
\]
Então $\Omega_+ - \Omega_- = \frac{\Omega_+^2 -
\Omega_-^2}{\Omega_+ + \Omega_-}$ e, para $\lambda \ll \omega_0$,
ambos $\Omega_\pm \approx \omega_0$: $\Omega_+ - \Omega_- \approx
\frac{4\lambda\omega_0}{2\omega_0} = 2\lambda$, de modo que
$\frac{\omega_0}{\Omega_+ - \Omega_-} \approx
\frac{\omega_0}{2\lambda} = Q$. Medir a largura de um pico de \omterm{ex:b1:diffeq:oscillation}{ressonância} é
medir o seu \omterm{pb:b1:diffeq:1}{fator de qualidade}.

\textbf{16.} $Q = 10$; $\Omega_r = \sqrt{1 - 2(0.05)^2} =
\sqrt{0.995} = 0.9975$; $R_{\max} = \frac1{2 \times 0.05
\sqrt{1 - 0.0025}} = \frac1{0.1 \times 0.99875} = 10.01$; resposta
estática $R(0) = 1$; largura de banda $\approx 2\lambda = 0.1$. Um pico
alto e fino, de altura $\approx Q$, sobre um platô de altura $1$.

\textbf{17.} Se $2\lambda^2 \geq \omega_0^2$, então $u_r \leq 0$ e
$g'(u) = 2(u - u_r) > 0$ para todo $u > 0$: $g$ cresce estritamente
em $\intoo0{+\infty}$, de modo que $R = A/\sqrt g$ decresce estritamente a
partir de $R(0) = A/\omega_0^2$: a resposta é máxima na frequência zero
e não há pico algum.

\textbf{18.} $\gamma = \iu\Omega$ não é raiz de $r^2 +
\omega_0^2$ (pois $\Omega \neq \omega_0$), de modo que o
\cref{met:b1:diffeq:particular} com $m = 0$ dá $x_p =
\frac{A\cos\Omega t}{\omega_0^2 - \Omega^2}$ (substitua e
verifique: $-\Omega^2 + \omega_0^2$ vezes o cosseno). Solução
geral:
\[
x(t) = \frac{A\cos\Omega t}{\omega_0^2 - \Omega^2}
+ \lambda_1\cos\omega_0 t + \mu_1\sin\omega_0 t .
\]

\textbf{19.} $x(0) = 0$ força $\lambda_1 = -\frac A{\omega_0^2 -
\Omega^2}$, e $x'(0) = 0$ força $\mu_1 = 0$:
\[
x(t) = \frac{A}{\omega_0^2 - \Omega^2}\,
\bigl(\cos\Omega t - \cos\omega_0 t\bigr)
= \frac{2A}{\omega_0^2 - \Omega^2}\,
\sin\Bigl(\frac{(\omega_0 - \Omega)t}2\Bigr)
\sin\Bigl(\frac{(\omega_0 + \Omega)t}2\Bigr) ,
\]
pela fórmula de transformação em produto $\cos a - \cos b =
2\sin\frac{b - a}2\sin\frac{b + a}2$ aplicada com $a = \Omega t$,
$b = \omega_0 t$.

\textbf{20.} Para $\Omega$ próximo de $\omega_0$, o segundo seno
oscila na frequência rápida $\frac{\omega_0 + \Omega}2
\approx \omega_0$, ao passo que o primeiro é um envelope lento de frequência
$\frac{\abs{\omega_0 - \Omega}}2$: a amplitude da oscilação rápida
cresce e mingua, com período do envelope
$\frac{2\pi}{\abs{\omega_0 - \Omega}}$ (dois batimentos por período do
envelope), atingindo máximos $\frac{2A}{\abs{\omega_0^2 - \Omega^2}}$.
Quando $\Omega \to \omega_0$, os batimentos ficam ao mesmo tempo mais lentos
(período $\to \infty$) e mais altos (amplitude $\to \infty$).

\textbf{21.} Fixe $t$. Quando $\Omega \to \omega_0$:
\[
\frac{2A}{\omega_0^2 - \Omega^2}\sin\Bigl(\frac{(\omega_0 -
\Omega)t}2\Bigr)
= \frac{2A}{\omega_0 + \Omega}\cdot
\frac{\sin\bigl(\frac{(\omega_0 - \Omega)t}2\bigr)}
{\omega_0 - \Omega}
\longrightarrow \frac{2A}{2\omega_0}\cdot\frac t2
= \frac{At}{2\omega_0} ,
\]
ao passo que $\sin\bigl(\frac{(\omega_0 + \Omega)t}2\bigr) \to
\sin(\omega_0 t)$: o limite é $x_\infty(t) =
\frac{At\sin\omega_0 t}{2\omega_0}$. Verificação direta: com $C =
\frac A{2\omega_0}$, $x_\infty = Ct\sin\omega_0 t$ tem
$x_\infty'' = 2C\omega_0\cos\omega_0 t - C\omega_0^2 t
\sin\omega_0 t$, de modo que $x_\infty'' + \omega_0^2 x_\infty =
2C\omega_0\cos\omega_0 t = A\cos\omega_0 t$, com $x_\infty(0) =
0$ e $x_\infty'(0) = C\sin 0 + C\omega_0 \cdot 0 \cdot \cos 0 =
0$ --- para $\omega_0 = A = 1$ isso é
exatamente o \cref{ex:b1:diffeq:oscillation}. A \omterm{ex:b1:diffeq:oscillation}{ressonância} é a
degeneração dos batimentos: o primeiro inchaço do envelope, esticado
até um comprimento infinito.

\textbf{22.} Sem amortecimento, a amplitude ressonante cresce
linearmente e sem limite; com amortecimento $\lambda > 0$, o crescimento
satura em $R_{\max} \approx Q\,\frac A{\omega_0^2}$. O mecanismo: a
dissipação retira energia a uma taxa que cresce com a amplitude
(questão 23), de modo que o acúmulo para exatamente quando o
amortecimento queima energia tão depressa quanto o forçamento a fornece.

\textbf{23.} Com $x_p' = -R\Omega\sin(\Omega t - \varphi)$, ao longo
de um período as médias $\langle\cos^2\rangle =
\langle\sin^2\rangle = \frac12$ e $\langle\sin\cos\rangle = 0$
dão:
\[
\langle P_{\mathrm{diss}}\rangle
= 2\lambda\,R^2\Omega^2\,\langle\sin^2(\Omega t -
\varphi)\rangle = \lambda R^2\Omega^2 ;
\]
e, expandindo $\sin(\Omega t - \varphi) = \sin\Omega t\cos\varphi
- \cos\Omega t\sin\varphi$:
\[
\langle P_{\mathrm{in}}\rangle
= -AR\Omega\,\bigl\langle\cos\Omega t\,\sin(\Omega t -
\varphi)\bigr\rangle
= AR\Omega\,\frac{\sin\varphi}2 .
\]
Como $\sin\varphi = \frac{2\lambda\Omega}{\sqrt D} =
\frac{2\lambda\Omega R}A$, isso vale $\frac{AR\Omega}2 \cdot
\frac{2\lambda\Omega R}A = \lambda R^2\Omega^2$: as potências injetada e
dissipada se equilibram exatamente --- a propriedade que define um
regime permanente.

\textbf{24.} (i) O teorema de estrutura separou cada solução em
regime permanente mais transitório (questões 9--11) e reduziu a
unicidade ao problema homogêneo. (ii) O método complexo
transformou a busca de uma solução particular numa única divisão de
números complexos (questão 6), com amplitude e fase lidas num
\omterm{def:b1:complex:field}{módulo} e num argumento. (iii) A curva de \omterm{ex:b1:diffeq:oscillation}{ressonância} é um puro
estudo de função --- uma quadrática em $u = \Omega^2$, o seu mínimo,
os seus \omterm{def:b1:logic:sets}{conjuntos} de nível --- ao estilo do \cref{ch:b1:functions}
(questões 12--17).

\textbf{25.} Livre e amortecido: pseudo-oscilações que decaem,
tempo de vida $\frac1\lambda$, cerca de $\frac Q\pi$ oscilações. Forçado e
amortecido: os transitórios morrem e sobrevive um único regime
permanente senoidal na frequência de forçamento, com amplitude que
atinge o pico perto de $\omega_0$ (altura $\approx Q \times$ a resposta estática,
largura $\approx
\frac{\omega_0}Q$) e fase varrendo de $0$ a $\pi$, passando por
$\frac\pi2$ em $\omega_0$. Livre e não amortecido: oscilação
perpétua. Forçado e não amortecido: batimentos, degenerando numa
\omterm{ex:b1:diffeq:oscillation}{ressonância} de crescimento linear na sintonia exata. Um único
número adimensional, $Q = \frac{\omega_0}{2\lambda}$, ajusta tudo ---
altura do pico, largura de banda e tempo de vida do transitório são três
leituras do mesmo mostrador. A continuação: reescrever $x'' + 2\lambda x' +
\omega_0^2x$ como um sistema de primeira ordem abre os métodos matriciais do
\cref{ch:b1:matrices} e do volume do segundo ano de graduação, e decompor um
forçamento periódico arbitrário em senoides (séries de Fourier, volume
do terceiro ano de graduação) torna a análise a uma só frequência deste
problema o tijolo universal: resolva para cada frequência e superponha.
\end{solution}
