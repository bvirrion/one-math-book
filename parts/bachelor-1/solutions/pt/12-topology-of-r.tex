\section*{Capítulo \ref{ch:b1:topology} --- Topologia da Reta Real}

\begin{solution}{exo:b1:topology:1}
$\intoo{0}{1} \cup \intoo{2}{3}$: \omterm{def:b1:topology:open}{aberto} (união de \omterm{def:b1:topology:open}{abertos}), não
\omterm{def:b1:topology:closed}{fechado} ($\frac 1n \to 0$ fora dele).

$\intco{0}{1}$: nenhum dos dois. Não é \omterm{def:b1:topology:open}{aberto} (nenhum \omterm{prop:b1:reals:intervals}{intervalo} em torno de $0$
está dentro); não é \omterm{def:b1:topology:closed}{fechado} ($1 - \frac1n \to 1 \notin$ do \omterm{def:b1:logic:sets}{conjunto}).

$\{0\} \cup \intcc{1}{2}$: \omterm{def:b1:topology:closed}{fechado} (união finita de \omterm{def:b1:topology:closed}{fechados}), não
\omterm{def:b1:topology:open}{aberto} (falha em $0$).

$\R \setminus \Z$: \omterm{def:b1:topology:open}{aberto} ($\Z$ é \omterm{def:b1:topology:closed}{fechado}), não \omterm{def:b1:topology:closed}{fechado}: a sequência
$\bigl(\frac 1n\bigr)$ está nele, mas o seu limite $0$ pertence a $\Z$,
isto é, escapa do \omterm{def:b1:logic:sets}{conjunto}.

$\Q \cap \intoo{0}{1}$: nenhum dos dois. Não é \omterm{def:b1:topology:open}{aberto}: todo \omterm{prop:b1:reals:intervals}{intervalo} em torno de um
racional contém irracionais. Não é \omterm{def:b1:topology:closed}{fechado}: ele contém sequências
que tendem ao irracional $\frac{\sqrt 2}{2}$ (\omterm{def:b1:topology:dense}{densidade}).
\end{solution}

\begin{solution}{exo:b1:topology:2}
$A = \intoc{0}{1} \cup \{2\}$: $\mathring A = \intoo{0}{1}$,
$\overline A = \intcc{0}{1} \cup \{2\}$, $\partial A = \{0, 1, 2\}$.

$A = \R \setminus \Q$: $\mathring A = \emptyset$ (todo \omterm{prop:b1:reals:intervals}{intervalo}
contém racionais), $\overline A = \R$ (\omterm{def:b1:topology:dense}{densidade} dos
irracionais), $\partial A = \R$.

$A = \bigl\{\frac{(-1)^n n}{n+1}\bigr\}$: os termos de índice par tendem a $1$,
e os de índice ímpar a $-1$, e nenhum deles pertence a $A$. $\mathring A =
\emptyset$ (pontos isolados), $\overline A = A \cup \{-1, 1\}$,
$\partial A = \overline A$.
\end{solution}

\begin{solution}{exo:b1:topology:3}
\emph{Complementos.} $F = \{a_1 < a_2 < \dots < a_k\}$: o complementar
é a união dos \omterm{prop:b1:reals:intervals}{intervalos} \omterm{def:b1:topology:open}{abertos} $\intoo{-\infty}{a_1}$,
$\intoo{a_i}{a_{i+1}}$, $\intoo{a_k}{+\infty}$ --- \omterm{def:b1:topology:open}{abertos} pela
\cref{prop:b1:topology:openstable}.

\emph{Sequências.} Sejam $u_n \in F$, $u_n \to \ell$. Com $\varepsilon
= \min\{\abs{a_i - a_j} : i \neq j\}/2 > 0$ (ou qualquer $\varepsilon$, se
$F$ é unitário): a partir de certo posto, todos os termos estão a menos
de $\varepsilon$ de $\ell$, logo a menos de $2\varepsilon$ uns dos outros,
o que os força a ser um único $a_i$ a partir desse posto; então
$\ell = a_i \in F$.
\end{solution}

\begin{solution}{exo:b1:topology:4}
$\subseteq$: $\overline A \cup \overline B$ é \omterm{def:b1:topology:closed}{fechado} (união finita)
e contém $A \cup B$, de modo que contém o \emph{menor} \omterm{def:b1:topology:closed}{fechado} que
contém $\overline{A \cup B}$. $\supseteq$: $A \subseteq A \cup B$
dá $\overline A \subseteq \overline{A \cup B}$ (o \omterm{def:b1:topology:closure}{fecho} é monótono: pontos aderentes a $A$ são
aderentes ao \omterm{def:b1:logic:sets}{conjunto} maior), e o mesmo para $B$.

Contraexemplo para as interseções: $A = \intoo{0}{1}$, $B =
\intoo{1}{2}$: $\overline{A \cap B} = \overline\emptyset = \emptyset$,
ao passo que $\overline A \cap \overline B = \{1\}$.
\end{solution}

\begin{solution}{exo:b1:topology:5}
Use a caracterização sequencial
(\cref{thm:b1:topology:seqclosed}). Sejam $S = \{u_n\} \cup \{\ell\}$
e $(v_k)$ uma sequência em $S$ com $v_k \to m$; mostremos que $m \in S$. Dois
casos. Se algum valor $v \in S$ é assumido por $(v_k)$ infinitas vezes,
uma \omterm{def:b1:seq:subsequence}{subsequência} constante dá $m = v \in S$. Caso contrário, todo
valor é assumido finitas vezes; em particular, para cada $n$, o termo
$u_n$ aparece finitas vezes, e $\ell$ também. Então, para todo $N$, os
índices $k$ com $v_k \in \{u_0, \dots, u_N, \ell\}$ são em número finito:
os demais $v_k$ são termos $u_n$ com $n > N$. Dado
$\varepsilon > 0$, escolha $N$ com $\abs{u_n - \ell} \leq
\varepsilon$ para $n > N$: todos os $v_k$, salvo finitos, satisfazem
$\abs{v_k - \ell} \leq \varepsilon$. Portanto, $v_k \to \ell$, de modo que $m =
\ell \in S$.
\end{solution}

\begin{solution}{exo:b1:topology:6}
$U + A = \bigcup_{a \in A} (U + a)$, e cada transladado $U + a$ é
\omterm{def:b1:topology:open}{aberto} (transladando os certificados de \omterm{prop:b1:reals:intervals}{intervalo}). Uma união de \omterm{def:b1:topology:open}{abertos}
é \omterm{def:b1:topology:open}{aberta} (\cref{prop:b1:topology:openstable}).

\omterm{def:b1:topology:closed}{Conjuntos fechados}: $\Z$ e $\sqrt 2\,\Z$ são \omterm{def:b1:topology:closed}{fechados} (como $\alpha\Z$:
sequências convergentes são constantes a partir de certo ponto, cf.\
\cref{ex:b1:topology:closed}). A sua soma $\Z + \sqrt 2\,\Z$ é \omterm{def:b1:topology:dense}{densa}
em $\R$ (\cref{exo:b1:reals:9}), mas não é $\R$ (ela é enumerável ou,
simplesmente: $\frac{\sqrt 2}{2} \notin \Z + \sqrt2\Z$, caso contrário $\sqrt 2$
seria racional --- escrever $\frac{\sqrt2}{2} = m + n\sqrt 2$
força $(2n - 1)\sqrt 2 = -2m$, de modo que $\sqrt 2 \in \Q$, a menos que $n =
\frac12$, o que é impossível). Um subconjunto próprio e \omterm{def:b1:topology:dense}{denso} não é
\omterm{def:b1:topology:closed}{fechado}: o seu \omterm{def:b1:topology:closure}{fecho} é $\R \neq$ ele mesmo.
\end{solution}

\begin{solution}{exo:b1:topology:7}
$\Z$: a \omterm{def:b1:topology:open}{vizinhança} $\intoo{n - \frac12}{n + \frac12}$ de $n$
encontra $\Z$ apenas em $n$.

$A = \{\frac1n : n \in \N^*\}$: em torno de $\frac 1n$, o \omterm{prop:b1:reals:intervals}{intervalo} de
raio $\frac{1}{n} - \frac{1}{n+1} = \frac{1}{n(n+1)}$ (dividido ao meio, digamos)
o isola dos seus vizinhos --- todos os pontos são isolados. E, no
entanto, $0 \in
\overline A \setminus A$: pontos isolados não impedem forasteiros
aderentes.

Se todos os pontos de $A$ são isolados: nenhum ponto de $A$ é \omterm{def:b1:topology:closure}{interior},
pois um ponto \omterm{def:b1:topology:closure}{interior} tem um \omterm{prop:b1:reals:intervals}{intervalo} inteiro de vizinhos em $A$ em
torno de si (um \omterm{prop:b1:reals:intervals}{intervalo} é infinito), o que contradiz o isolamento.
Logo, $\mathring A
= \emptyset$.
\end{solution}

\begin{solution}{exo:b1:topology:8}
Se $A \subseteq \intcc{-M}{M}$, o \omterm{def:b1:topology:closed}{fechado} $\intcc{-M}{M}$
contém $A$ e, portanto, contém $\overline A$ (o menor \omterm{def:b1:topology:closed}{fechado} que
o contém): $\overline A$ é limitado.

Seja $s = \sup A$ (finito). Como $A \subseteq \overline A$, $\sup
\overline A \geq s$. Reciprocamente, $\overline A \subseteq
\intoc{-\infty}{s}$: a semirreta é \omterm{def:b1:topology:closed}{fechada} e contém $A$; de modo que
todo elemento de $\overline A$ é $\leq s$, o que dá $\sup\overline A
\leq s$. Igualdade.

$s \in \overline A$: pela caracterização por $\varepsilon$
(\cref{prop:b1:reals:epsilon}), todo \omterm{prop:b1:reals:intervals}{intervalo} $\intoo{s -
\varepsilon}{s + \varepsilon}$ contém um elemento de $A$: $s$ é
aderente.
\end{solution}

\begin{solution}{exo:b1:topology:9}
Suponha $A$ \omterm{def:b1:topology:open}{aberto} e \omterm{def:b1:topology:closed}{fechado}, com $a \in A$ e $b \in \R
\setminus A$; sem perda de generalidade, $a < b$. O \omterm{def:b1:logic:sets}{conjunto} $B = A \cap
\intcc{a}{b}$ é não vazio ($a$) e limitado: seja $s = \sup B$. Pelo
\cref{exo:b1:topology:8}, $s \in \overline B \subseteq \overline A =
A$ ($A$ \omterm{def:b1:topology:closed}{fechado}). Note que $s \leq b$ e, como $b \notin A$: $s < b$.
Ora, $A$ é \omterm{def:b1:topology:open}{aberto}: algum \omterm{prop:b1:reals:intervals}{intervalo} $\intoo{s - r}{s + r}$ está em $A$,
e podemos tomar $r < b - s$. Então $s + \frac{r}{2}$ pertence a $A
\cap \intcc{a}{b} = B$ e excede $s$ --- contradizendo $s = \sup
B$. Portanto, não existe tal par $(a, b)$: um dentre $A$ e $\R \setminus A$ é vazio.
\end{solution}

\begin{solution}{exo:b1:topology:10}
$I_x$ é uma união de \omterm{prop:b1:reals:intervals}{intervalos} \omterm{def:b1:topology:open}{abertos} que contêm todos $x$: é \omterm{def:b1:topology:open}{aberto},
e é um \omterm{prop:b1:reals:intervals}{intervalo}, por ser convexo --- se $u < z < v$ com $u, v \in
I_x$, então $u$ e $v$ estão em subintervalos \omterm{def:b1:topology:open}{abertos} $J_u \ni x$, $J_v \ni
x$ de $U$, e $J_u \cup J_v$ é um \omterm{prop:b1:reals:intervals}{intervalo} (ambos contêm $x$)
dentro de $U$ que contém $z$; de modo que $z \in I_x$
(\cref{prop:b1:reals:intervals}).

Se $I_x \cap I_y \neq \emptyset$: $I_x \cup I_y$ é então um \omterm{prop:b1:reals:intervals}{intervalo}
\omterm{def:b1:topology:open}{aberto} (convexo: dois \omterm{prop:b1:reals:intervals}{intervalos} que se sobrepõem) contido em $U$ e
que contém $x$ e $y$, de sorte que $I_x \cup I_y \subseteq I_x$ e
$\subseteq I_y$, pela maximalidade de cada um: $I_x = I_y$.

Assim, $U$ é a união disjunta dos \omterm{def:b1:logic:sets}{conjuntos} distintos $I_x$ (cada $x \in
U$ está no seu próprio $I_x$). Enumerabilidade: cada \omterm{prop:b1:reals:intervals}{intervalo} \omterm{def:b1:topology:open}{aberto}
não vazio $I$ da família contém um racional $q_I$
(\cref{thm:b1:reals:density}), e \omterm{prop:b1:reals:intervals}{intervalos} disjuntos distintos recebem
racionais distintos: a família se injeta em $\Q$, que é enumerável
(está indexado por pares de inteiros). Logo, há no máximo
enumeravelmente muitos \omterm{prop:b1:reals:intervals}{intervalos}.
\end{solution}

\begin{solution}{exo:b1:topology:11}
$\overline A = A \cup A'$. ($\supseteq$) $A \subseteq \overline
A$ sempre; e, se $x \in A'$, toda \omterm{def:b1:topology:open}{vizinhança} de $x$ encontra $A
\setminus \{x\} \subseteq A$, de modo que $x$ é aderente. ($\subseteq$)
Seja $x \in \overline A$. Se $x \in A$, pronto. Se $x \notin A$,
toda \omterm{def:b1:topology:open}{vizinhança} de $x$ encontra $A = A \setminus \{x\}$: $x \in
A'$.

Consequentemente, $A$ é \omterm{def:b1:topology:closed}{fechado} $\iff$ $A = \overline A = A \cup A'$
$\iff$ $A' \subseteq A$.

$A = \{\frac 1n\}$: $0$ é ponto de acumulação ($\frac 1n \to
0$, com termos $\neq 0$); cada $\frac 1n$ é isolado
(\cref{exo:b1:topology:7}), logo não está em $A'$; e um ponto $x
\notin A \cup \{0\}$ tem um \omterm{prop:b1:reals:intervals}{intervalo} inteiro que evita $A$ (entre
os dois vizinhos de $x$ em $A \cup \{0\}$, ou além de $1$).
Portanto, $A' = \{0\}$.

$\Z' = \emptyset$: todo inteiro é isolado, e todo não inteiro
tem uma \omterm{def:b1:topology:open}{vizinhança} dentro de $\R \setminus \Z$.

$\Q' = \R$: todo \omterm{prop:b1:reals:intervals}{intervalo} em torno de qualquer real contém infinitos
racionais (\cref{thm:b1:reals:density}), em particular um
diferente do centro.
\end{solution}

\begin{solution}{exo:b1:topology:12}
\begin{enumerate}
  \item Para todo $a \in A$: $\abs{x - a} \leq \abs{x - y} +
        \abs{y - a}$, de modo que $d_A(x) \leq \abs{x - y} + \abs{y - a}$;
        tomando o \omterm{def:b1:reals:bounds}{ínfimo} em $a$: $d_A(x) \leq \abs{x - y} +
        d_A(y)$. Trocando $x$ e $y$, obtém-se a outra
        desigualdade: $\abs{d_A(x) - d_A(y)} \leq \abs{x - y}$.
  \item $d_A(x) = 0$ $\iff$ para todo $\varepsilon > 0$ existe
        $a \in A$ com $\abs{x - a} < \varepsilon$ $\iff$ todo
        \omterm{prop:b1:reals:intervals}{intervalo} em torno de $x$ encontra $A$ $\iff$ $x \in \overline A$.
        Se $F$ é \omterm{def:b1:topology:closed}{fechado} e $x \notin F = \overline F$, então
        $d_F(x) \neq 0$, isto é, $d_F(x) > 0$.
  \item Se $d_F(x) < \varepsilon$, ponha $r = \varepsilon -
        d_F(x) > 0$: para $\abs{y - x} < r$, a parte (1) dá
        $d_F(y) \leq d_F(x) + \abs{x - y} < \varepsilon$: o
        \omterm{prop:b1:reals:intervals}{intervalo} $\intoo{x - r}{x + r}$ está em $V_\varepsilon$,
        que é, portanto, \omterm{def:b1:topology:open}{aberto}; ele contém $F$, pois $d_F = 0$
        ali. Por fim, $x \in \bigcap_{\varepsilon>0}
        V_\varepsilon$ $\iff$ $d_F(x) < \varepsilon$ para todo
        $\varepsilon$ $\iff$ $d_F(x) = 0$ $\iff$ $x \in \overline
        F = F$. Como $\bigcap_{\varepsilon > 0} V_\varepsilon =
        \bigcap_{n \geq 1} V_{1/n}$, todo \omterm{def:b1:topology:closed}{fechado} é uma
        interseção enumerável de \omterm{def:b1:topology:open}{abertos}.
\end{enumerate}
\end{solution}

\begin{solution}{pb:b1:topology:1}
\textbf{1.} $C_1 = \intcc{0}{\frac13} \cup \intcc{\frac23}{1}$
e
\[
C_2 = \intcc{0}{\tfrac19} \cup \intcc{\tfrac29}{\tfrac13}
\cup \intcc{\tfrac23}{\tfrac79} \cup \intcc{\tfrac89}{1} .
\]
Indução: se $C_n$ é uma união disjunta de $2^n$ segmentos \omterm{def:b1:topology:closed}{fechados}
de comprimento $3^{-n}$, apagar o terço médio \omterm{def:b1:topology:open}{aberto} de cada um
deixa dois segmentos \omterm{def:b1:topology:closed}{fechados} de comprimento $3^{-n-1}$ por segmento
pai: $2^{n+1}$ segmentos, dois a dois disjuntos (filhos de pais
distintos ficam separados porque os pais ficavam; filhos de um mesmo
pai ficam separados pelo vão retirado).

\textbf{2.} Cada $C_n$ é uma união finita de segmentos, logo
\omterm{def:b1:topology:closed}{fechado}; $C = \bigcap C_n$ é uma interseção de \omterm{def:b1:topology:closed}{fechados}:
\omterm{def:b1:topology:closed}{fechado} (\cref{def:b1:topology:closed}); limitado ($\subseteq
\intcc{0}{1}$): compacto pelo \cref{thm:b1:topology:compact}.
Não vazio: $0$ está no segmento mais à esquerda de todo $C_n$. Seja
$a$ uma extremidade de um segmento $S$ de $C_n$. Para $m \leq n$,
$a \in C_n \subseteq C_m$. Para os estágios posteriores: a supressão do terço
médio nunca retira uma extremidade, e $a$ é de novo extremidade
de um dos dois filhos de $S$ (o filho que toca $a$); por
indução, $a \in C_m$ para todo $m \geq n$: $a \in C$.

\textbf{3.} Comprimento total de $C_n$: $2^n \cdot 3^{-n} =
(\frac23)^n \to 0$. Dado $\varepsilon > 0$, escolha $n$ com
$(\frac23)^n \leq \varepsilon$: então $C \subseteq C_n$, uma união
de finitos segmentos de comprimento total $\leq \varepsilon$.

\textbf{4.} Seja $I \subseteq C$ um \omterm{prop:b1:reals:intervals}{intervalo} com dois pontos
distintos. Para todo $n$: $I \subseteq C_n$ e, sendo convexo, $I$
deve estar dentro de um \emph{único} segmento de $C_n$ (encontrar dois
segmentos forçaria $I$ a conter um ponto do vão entre
eles, que está fora de $C_n$). Logo, o comprimento de $I$ é $\leq
3^{-n}$ para todo $n$: contradição. Assim, os únicos \omterm{prop:b1:reals:intervals}{intervalos}
dentro de $C$ são vazios ou unitários; em particular, nenhum
$\intoo{x-r}{x+r}$ cabe dentro de $C$: $\mathring C = \emptyset$.
Como $C$ é \omterm{def:b1:topology:closed}{fechado}, $\overline C = C$ tem \omterm{def:b1:topology:closure}{interior} vazio: $C$ é
nunca \omterm{def:b1:topology:dense}{denso}.

\textbf{5.} Escreva $\varphi_0(x) = \frac x3$ e $\varphi_2(x) =
\frac{2 + x}{3}$, bijeções afins crescentes de $\intcc{0}{1}$
sobre $\intcc{0}{\frac13}$ e $\intcc{\frac23}{1}$. Afirmação:
$C_{n+1} = \varphi_0(C_n) \cup \varphi_2(C_n)$. Para $n = 0$ isso é
a questão 1. Indução: uma \omterm{def:b1:logic:map}{aplicação} afim crescente leva o
terço médio de um segmento ao terço médio do segmento imagem,
de modo que apagar terços médios comuta com $\varphi_0$ e
$\varphi_2$; aplicar o passo de supressão a $C_{n+1} =
\varphi_0(C_n) \cup \varphi_2(C_n)$ dá $C_{n+2} =
\varphi_0(C_{n+1}) \cup \varphi_2(C_{n+1})$. Interceptando sobre
$n$: para $x \leq \frac13$, $x \in C \iff x \in \varphi_0(C_n)$
para todo $n$ $\iff 3x \in \bigcap C_n = C$; do mesmo modo em
$\intcc{\frac23}{1}$; e nenhum ponto de $\intoo{\frac13}{\frac23}$
está em $C_1$. Portanto, $C = \varphi_0(C) \cup \varphi_2(C)$,
disjuntamente.

\textbf{6.} Indução em $n$; o caso $n = 0$ diz que todo $x \in
\intcc{0}{1}$ tem um código, o que é o \cref{pb:b1:reals:1}
(questão 9 para $x < 1$; $1 = (0.\overline 2)_3$). Suponha a
equivalência no posto $n$. Se $x \in C_{n+1}$: pela questão 5, $x =
\varphi_i(z)$ com $z \in C_n$ e $i \in \{0, 2\}$; se $(e_k)$
é um código de $z$ cujos $n$ primeiros algarismos são sem $1$, então $(i, e_1,
e_2, \dots)$ tem somas parciais $\frac i3 + \frac13\sum_{k\leq m}
e_k 3^{-k} \to \varphi_i(z) = x$: um código de $x$ cujos $n + 1$ primeiros
algarismos são sem $1$. Reciprocamente, se $x$ tem um código $(d_k)$ com
$d_1, \dots, d_{n+1} \in \{0, 2\}$: a cadeia deslocada $(d_2,
d_3, \dots)$ tem algum valor $z \in \intcc{0}{1}$, os seus $n$ primeiros
algarismos são sem $1$, e o cálculo das somas parciais lido
ao contrário dá $x = \varphi_{d_1}(z)$; por indução, $z \in
C_n$, de modo que $x \in C_{n+1}$ pela questão 5. Por fim: um código
inteiramente sem $1$ põe $x$ em todo $C_n$, logo em $C$; reciprocamente,
se $x \in C$, então, para todo $n$, um dos no máximo dois códigos de
$x$ (\cref{pb:b1:reals:1}, questão 11) tem os seus $n$ primeiros algarismos
sem $1$; um mesmo código fixado deve servir para $n$ arbitrariamente
grandes (casa dos pombos entre dois códigos), e um código cujos $n$
primeiros algarismos são sem $1$ para $n$ arbitrariamente grandes é
sem $1$ por completo.

\textbf{7.} $0 = (0.\overline 0)_3$, $1 = (0.\overline 2)_3$,
$\frac13 = (0.0\overline{2})_3$ (o gêmeo impróprio de $(0.1)_3$),
$\frac23 = (0.2\overline{0})_3$. Somas geométricas:
\[
(0.\overline{02})_3 = \sum_{j\geq1} \frac{2}{9^{\,j}}
= \frac{2/9}{1 - 1/9} = \frac14 ,
\qquad
(0.\overline{20})_3 = \sum_{j\geq1} \frac{2}{3\cdot 9^{\,j-1}}
= \frac{2/3}{1 - 1/9} = \frac34 ,
\]
ambos sem $1$: $\frac14, \frac34 \in C$. (As somas infinitas
abreviam \omterm{def:b1:reals:bounds}{supremos} de somas parciais, como no \cref{pb:b1:reals:1}.)
As extremidades dos segmentos de $C_n$ são da forma $m/3^n$
(indução: as extremidades dos filhos são extremidades do pai ou
diferem de uma delas por um múltiplo de $3^{-n-1}$). Se $\frac14 =
\frac{m}{3^n}$, então $3^n = 4m$, e $4 \nmid 3^n$: impossível.
Logo, $\frac14 \in C$ sem jamais ser extremidade.

\textbf{8.} Suponha que $x$ tivesse dois códigos sem $1$ distintos. Ter
dois códigos, para começar, significa (\cref{pb:b1:reals:1}, questão 11, base
$3$) que $x = m/3^N \in \intoo{0}{1}$ e que os dois códigos são:
o que termina, com último algarismo não nulo $d_N \in \{1, 2\}$
seguido de $0$s, e o seu gêmeo, com $d_N - 1$ na posição $N$
seguido de $2$s. Se $d_N = 1$, o primeiro contém um $1$; se $d_N
= 2$, o gêmeo carrega $d_N - 1 = 1$. De um jeito ou de outro, no máximo um
do par é sem $1$: contradição. Logo, cada $x \in C$ tem
exatamente um código sem $1$ (existência pela questão 6), e
cadeias de $\{0,2\}$ distintas têm valores distintos. Toda
cadeia de $\{0,2\}$ tem valor em $\intcc{0}{1}$ (somas parciais $\leq
1$) com todos os prefixos sem $1$, logo valor em todo $C_n$,
isto é, em $C$: a \omterm{def:b1:logic:map}{aplicação} valor é uma bijeção das
cadeias de $\{0,2\}$ sobre $C$.

\textbf{9.} Seja $(d^{(k)})$ o código sem $1$ de $x_k$ e
ponha $e_k = 2 - d^{(k)}_k \in \{0, 2\}$: uma cadeia de $\{0,2\}$ cujo
valor $y$ está em $C$ e tem $(e_k)$ como o seu único código
sem $1$ (questão 8). Para cada $k$, os códigos de $y$ e de $x_k$
diferem na posição $k$, de modo que $y \neq x_k$: nenhuma \omterm{def:b1:logic:map}{aplicação} $\N^* \to C$ é
\omterm{def:b1:logic:inj}{sobrejetiva}. Um \omterm{def:b1:logic:sets}{conjunto} não enumerável de comprimento zero: grandeza
em \omterm{def:b1:counting:card}{cardinalidade} e pequenez em medida --- simultaneamente.

\textbf{10.} Compacidade: o fechamento da interseção infinita
mais a limitação (questão 2) --- a única propriedade
não carregada por uma única inclusão. Comprimento zero: $C \subseteq
C_n$, de comprimento total $(\frac23)^n$ (questão 3). Nunca
\omterm{def:b1:topology:dense}{denso}: $C \subseteq C_n$ força os \omterm{prop:b1:reals:intervals}{intervalos} dentro de $C$ a terem
comprimento $\leq 3^{-n}$ (questão 4). A não enumerabilidade não
cavalga em inclusão alguma: ela precisa da estrutura completa da
interseção, codificada na bijeção da questão 8.

\textbf{11.} Inverta $d_n$ para $2 - d_n$: a nova cadeia continua sendo
uma cadeia de $\{0,2\}$, de modo que o seu valor $x_n$ está em $C$; as
somas parciais além do posto $n$ diferem exatamente de $2\cdot3^{-n}$, de
sorte que $\abs{x_n - x} = 2\cdot3^{-n}$. Assim, $x_n \neq x$ e $x_n \to
x$: todo ponto de $C$ é limite de outros pontos de $C$ --- $C$
é \emph{perfeito}, sem ponto isolado.

\textbf{12.} Pela indução da questão 5, os segmentos de
$C_n$ são exatamente os $\intcc{t}{t + 3^{-n}}$, em que $t$ percorre
os valores das cadeias de $\{0,2\}$ de comprimento $n$. Dado $x \in C$
com código $(d_k)$, a truncagem $t_n$ (algarismos $d_1 \dots d_n$
e depois $0$s) é, portanto, uma extremidade esquerda, e $0 \leq x - t_n
\leq 3^{-n}$: as extremidades são \omterm{def:b1:topology:dense}{densas} em $C$. Elas formam um subconjunto
de $\{m/3^n : m, n\}$, um \omterm{def:b1:logic:sets}{conjunto} indexado por pares de inteiros, logo
enumerável (como para $\Q$ no \cref{exo:b1:topology:10}). Como $C$
é não enumerável (questão 9), todos os pontos de $C$, salvo
enumeravelmente muitos, não são extremidades --- $\frac14$ (questão 7)
é a ponta visível desse iceberg.

\textbf{13.} Tome $n$ com $3^{-n} < y - x$. Ambos $x, y \in
C_n$, e eles não podem estar no mesmo segmento (comprimento $3^{-n} <
y - x$): o vão retirado entre os seus segmentos fornece $z$
com $x < z < y$ e $z \notin C_n \supseteq C$. Portanto, dois pontos
quaisquer de $C$ são separados pelo complementar: os únicos subconjuntos
convexos de $C$ são unitários --- $C$ é totalmente desconexo.

\textbf{14.} Se $(d_k)$ é o código sem $1$ de $x$, a cadeia
$(2 - d_k)$ é de novo uma cadeia de $\{0,2\}$, com somas parciais
\[
\sum_{k=1}^{n} (2 - d_k)3^{-k} = (1 - 3^{-n}) - \sum_{k=1}^n
d_k 3^{-k} \longrightarrow 1 - x :
\]
de modo que $1 - x \in C$. Assim, $1 - C \subseteq C$ e, aplicando a \omterm{def:b1:logic:map}{aplicação}
duas vezes, $1 - C = C$: o \omterm{pb:b1:topology:1}{conjunto de Cantor} é simétrico em relação a
$\frac12$.

\textbf{15.} As somas parciais $t_n \to x$ e $t'_n \to x'$
(sequências crescentes convergem para o seu \omterm{def:b1:reals:bounds}{supremo}, isto é, para o
valor), de modo que $t_n + t'_n \to x + x'$, pelo
\cref{thm:b1:seq:operations}. Dado $y \in \intcc{0}{1}$ com
código $(e_k)$, escolha $(a_k, b_k) = (0,0), (0,2), (2,2)$
conforme $e_k = 0, 1, 2$: então $a_k + b_k = 2e_k$, as
cadeias $(a_k)$ e $(b_k)$ são cadeias de $\{0,2\}$ com valores $x,
x' \in C$, e
\[
x + x' = \lim_n\,(t_n + t'_n) = \lim_n 2\sum_{k=1}^n e_k 3^{-k}
= 2y :
\]
todo $y \in \intcc{0}{1}$ é o ponto médio de dois pontos de $C$.

\textbf{16.} A questão 15 dá $\intcc{0}{2} = 2\,\intcc{0}{1}
\subseteq C + C$, e $C + C \subseteq \intcc{0}{1} +
\intcc{0}{1} = \intcc{0}{2}$: igualdade. Então, usando $1 - C = C$:
\[
C - C = C + (C - 1) = (C + C) - 1 = \intcc{-1}{1} .
\]
Instância concreta: $1 = \frac14 + \frac34$, uma soma de dois
membros de $C$ que não são extremidades.

\textbf{17.} O comprimento mede quanto da reta o próprio \omterm{def:b1:logic:sets}{conjunto}
ocupa; ele nada diz sobre o \omterm{def:b1:logic:sets}{conjunto} das \emph{somas}, que é
a imagem da família a dois parâmetros $C \times C$ pela \omterm{def:b1:logic:map}{aplicação} $(x,
x') \mapsto x + x'$ --- as duas cadeias de algarismos são escolhidas
independentemente, e essa liberdade é exatamente o que preenche
$\intcc{0}{2}$. Teorema algum limita o comprimento de um \omterm{def:b1:logic:sets}{conjunto} soma
pelos comprimentos das parcelas, e $C$ é a prova de que nenhum pode.

\textbf{18.} Pelo \cref{pb:b1:reals:1} (questão 18), $x$ é
racional se, e somente se, a sua expansão própria é periódica a partir
de certo ponto. O código sem $1$ de $x \in C$ é ou essa expansão
própria, ou o gêmeo impróprio de uma que termina; uma cadeia que termina
e o seu gêmeo (constante $2$ a partir de certo ponto) são ambos
periódicos a partir de certo ponto, de modo que a periodicidade do
código sem $1$ é equivalente à racionalidade de $x$. Divisão longa de
$\frac1{13}$ na base $3$ ($r_0 = 1$): $3 = 13\cdot0 + 3$, $9 = 13\cdot0 + 9$, $27 =
13\cdot2 + 1$, e o resto volta a $1$: algarismos
$\overline{002}$, de modo que $\frac1{13} = (0.\overline{002})_3$,
sem $1$ e periódico: um membro racional de $C$. (Verificação:
$\frac{2/27}{1 - 1/27} = \frac{2}{26} = \frac1{13}$.)

\textbf{19.} A cadeia com $d_k = 2$ nas posições triangulares
$k = \frac{j(j+1)}{2}$ e $0$ nas demais é uma cadeia de
$\{0,2\}$, de modo que o seu valor $x^*$ pertence a $C$ (questão
8). Ela tem infinitos $2$s com espaçamentos $j + 1 \to \infty$
entre consecutivos, de sorte que não é periódica a partir de ponto algum
(um período $T$ acabaria forçando $2$s com espaçamentos $\leq T$: o
argumento dos espaçamentos crescentes do \cref{pb:b1:reals:1}, questão 20); pela
questão 18, $x^* \notin \Q$. E, pela questão 9 mais a
enumerabilidade de $\Q$, todos os membros de $C$, salvo enumeravelmente
muitos, são irracionais: $x^*$ é a norma, não a exceção.

\textbf{20.} Seja $y \in \intcc{0}{1}$: ele tem um código binário
$(c_k)$ com $c_k \in \{0, 1\}$ (\cref{pb:b1:reals:1}, questão
9, base $2$; $y = 1$ toma a cadeia só de $1$s). Então $(2c_k)$
é uma cadeia de $\{0,2\}$, o seu valor $x$ está em $C$, e $h(x)$ é
o valor de $(c_k)$, ou seja, $y$: $h$ leva $C$ sobre
$\intcc{0}{1}$. Se $C$ fosse imagem de uma \omterm{def:b1:logic:map}{aplicação} de $\N^*$,
compor com $h$ listaria todo $\intco{0}{1}$,
contradizendo o teorema diagonal do \cref{pb:b1:reals:1}
(questão 22): $C$ é não enumerável, de novo. Um \omterm{def:b1:logic:sets}{conjunto} de comprimento
zero que se projeta sobrejetivamente num segmento inteiro.

\textbf{21.} Pela questão 5, $C$ é a união disjunta de
$\varphi_0(C)$ e $\varphi_2(C)$, cada um um transladado da
cópia reescalonada $\frac13 C$. A aditividade, a escala e a invariância
por translação dão
\[
L(C) = L(\varphi_0(C)) + L(\varphi_2(C))
= \tfrac13 L(C) + \tfrac13 L(C) = \tfrac23\,L(C),
\]
de modo que $\frac13 L(C) = 0$: $L(C) = 0$. Só a autossemelhança já
condena $C$ a comprimento zero --- a questão 3 apenas executou a
sentença.

\textbf{22.} Um segmento de comprimento $l_n$ perde um \omterm{prop:b1:reals:intervals}{intervalo} central
de comprimento $4^{-(n+1)}$, restando dois segmentos de comprimento $l_{n+1}
= \frac{l_n - 4^{-(n+1)}}{2}$; a partir de $l_0 = 1$, a indução
confirma que $l_n = \frac{2^n + 1}{2\cdot4^n}$: com efeito,
$\frac12\Bigl(\frac{2^n+1}{2\cdot4^n} - \frac{1}{4^{n+1}}\Bigr)
= \frac{2(2^n + 1) - 1}{2\cdot4^{n+1}} = \frac{2^{n+1} +
1}{2\cdot4^{n+1}}$, e $l_n > 0$ sempre: a construção nunca
passa fome. $K = \bigcap K_n$ é \omterm{def:b1:topology:closed}{fechado} e limitado, logo compacto;
um \omterm{prop:b1:reals:intervals}{intervalo} dentro de $K$ está num segmento de $K_n$, de comprimento
$l_n \to 0$: \omterm{def:b1:topology:closure}{interior} vazio. Comprimento retirado: $\sum_{n\geq0} 2^n
\cdot 4^{-(n+1)} = \frac14\sum_{n\geq0}\bigl(\frac12\bigr)^n =
\frac12$, e cada $K_n$ tem comprimento total $2^n l_n = \frac{2^n +
1}{2^{n+1}} > \frac12$. Seja agora uma família finita de \omterm{prop:b1:reals:intervals}{intervalos} \omterm{def:b1:topology:open}{abertos}
de união $U \supseteq K$. Pelo enunciado companheiro do
\cref{ex:b1:topology:nested}, $U \supseteq K_n$ para algum $n$;
admitindo a aditividade do comprimento em uniões finitas de \omterm{prop:b1:reals:intervals}{intervalos},
o comprimento total dos \omterm{prop:b1:reals:intervals}{intervalos} da cobertura é ao menos o de
$K_n$, que excede $\frac12$. Assim, $K$ é nunca \omterm{def:b1:topology:dense}{denso} e, ainda assim,
não existe cobertura barata: pequenez topológica (nunca \omterm{def:b1:topology:dense}{denso}) e
pequenez métrica (comprimento zero) são noções genuinamente diferentes,
e $K$ as separa.

\textbf{23.} Atingimento: sejam $d = d(x, F)$ e $a_k \in F$
com $\abs{x - a_k} \leq d + \frac1k$: os $a_k$ são limitados, de modo que
Bolzano--Weierstrass (\cref{thm:b1:seq:bw}) extrai
$a_{\varphi(k)} \to a$, com $a \in F$ ($F$ \omterm{def:b1:topology:closed}{fechado},
\cref{thm:b1:topology:seqclosed}) e $\abs{x - a} = \lim
\abs{x - a_{\varphi(k)}} = d$. Agora, o máximo: se $y \in C$,
$d(y, C) = 0$; caso contrário, $y$ está num vão retirado em algum estágio
$n \geq 1$, um \omterm{prop:b1:reals:intervals}{intervalo} \omterm{def:b1:topology:open}{aberto} de comprimento $3^{-n}$ cujas duas
extremidades pertencem a $C$ (questão 2), de modo que $d(y, C) \leq
\frac{3^{-n}}{2} \leq \frac16$, com igualdade exigindo $n = 1$
e $y$ no centro do vão $\intoo{\frac13}{\frac23}$,
isto é, $y = \frac12$; e, de fato, $d\bigl(\frac12, C\bigr) =
\frac16$, pois $C \cap \intoo{\frac13}{\frac23} = \emptyset$ e
$\frac13, \frac23 \in C$. Portanto, $\max_{y\in\intcc{0}{1}} d(y, C)
= \frac16$, atingido exatamente em $\frac12$.

\textbf{24.} As extremidades formam um subconjunto enumerável e \omterm{def:b1:topology:dense}{denso}
de $C$ (questão 12): liste-as como uma única sequência $(e_j)_{j\geq1}$,
uma sequência em $C$. Os seus limites subsequenciais estão todos em $C$
($C$ \omterm{def:b1:topology:closed}{fechado}). Reciprocamente, fixe $x \in C$: para cada $n$, os
segmentos dos $C_m$ que contêm $x$ ($m \geq n$) têm as suas extremidades
a menos de $3^{-m} \leq 3^{-n}$ de $x$, de modo que infinitas extremidades
distintas ficam a menos de $3^{-n}$ de $x$; escolha índices $j_1 < j_2
< \dots$ com $\abs{e_{j_n} - x} \leq 3^{-n}$: uma \omterm{def:b1:seq:subsequence}{subsequência}
que converge para $x$. Assim, o \omterm{def:b1:logic:sets}{conjunto} dos limites subsequenciais de
$(e_j)$ é exatamente $C$ --- uma única sequência se acumulando em
uma quantidade não enumerável de pontos, o extremo oposto de uma
sequência convergente, cujo \omterm{def:b1:logic:sets}{conjunto} de acumulação é unitário.

\textbf{25.} (i) A construção consumiu: a estabilidade dos \omterm{def:b1:topology:closed}{fechados}
por interseção arbitrária (existência de $C$ como \omterm{def:b1:topology:closed}{fechado}),
o teorema de compacidade \cref{thm:b1:topology:compact}
(questões 2, 22 e 23) e as caracterizações sequenciais do
fechamento e da aderência (o argumento dos compactos encaixados do
\cref{ex:b1:topology:nested} e a questão 23). (ii) Os quatro
pares: comprimento zero e, ainda assim, não enumerável (questões 3 e 9);
\omterm{def:b1:topology:closed}{fechado} e, ainda assim, de \omterm{def:b1:topology:closure}{interior} vazio (questão 4); perfeito --- sem
ponto isolado --- e, ainda assim, totalmente desconexo (questões 11 e 13);
desprezível e, ainda assim, com $C + C = \intcc{0}{2}$ (questão 16). (iii)
A sobrejeção $h$ da questão 20, tornada contínua e
não decrescente, torna-se a escada do diabo na teoria das
funções contínuas; e, na teoria da medida do volume do terceiro ano,
$C$ é a testemunha padrão de que ``desprezível'' não significa
``enumerável'', com o seu \omterm{def:b1:arith:prime}{primo} gordo (questão 22) separando
``nunca \omterm{def:b1:topology:dense}{denso}'' de ``desprezível''.
\end{solution}
