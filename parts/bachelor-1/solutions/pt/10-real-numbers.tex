\section*{Capítulo \ref{ch:b1:reals} --- Números Reais}

\begin{solution}{exo:b1:reals:1}
$A$: todo elemento é $\leq 1$ e $1 \in A$: $\sup A = \max A = 1$.
Cotas inferiores: $0$ limita por baixo; para $\varepsilon > 0$, Arquimedes
fornece $n$ com $\frac 1n < \varepsilon$, de modo que nenhum número positivo
limita $A$ por baixo: $\inf A = 0$, não atingido (sem mínimo).

$B$: termos $0, -\frac12, \frac23, -\frac34, \frac45, \dots$ Os termos de
índice par $\frac{n}{n+1}$ ($n$ par) crescem rumo a $1$ sem alcançá-lo;
os de índice ímpar $-\frac{n}{n+1}$ decrescem rumo a $-1$. Logo, $\sup B = 1$ e
$\inf B = -1$, nenhum atingido: sem máximo, sem mínimo. (Cotas: $\abs{b}
< 1$ para todo $b \in B$; e $\frac{n}{n+1} = 1 - \frac{1}{n+1} > 1 -
\varepsilon$ para $n$ grande, analogamente por baixo.)

$C = \intoo{-\sqrt 3}{\sqrt 3}$: $\sup C = \sqrt 3$, $\inf C =
-\sqrt 3$, nenhum atingido.
\end{solution}

\begin{solution}{exo:b1:reals:2}
Escreva $x = \lfloor x \rfloor + u$, $y = \lfloor y \rfloor + v$ com
$u, v \in \intco{0}{1}$. Então $x + y = \lfloor x \rfloor + \lfloor y
\rfloor + (u + v)$ com $u + v \in \intco{0}{2}$. Se $u + v < 1$,
$\lfloor x + y\rfloor = \lfloor x\rfloor + \lfloor y \rfloor$; se
$1 \leq u + v < 2$, $\lfloor x+y \rfloor = \lfloor x \rfloor +
\lfloor y \rfloor + 1$. Os dois casos ocorrem: $(x, y) = (0.2,\, 0.3)$
dá a igualdade da esquerda, e $(0.7,\, 0.8)$, a da direita.
\end{solution}

\begin{solution}{exo:b1:reals:3}
Seja $k = \lfloor x \rfloor$, de modo que $k \leq x < k + 1$. Multiplicando por
$n$: $nk \leq nx < nk + n$ e, tomando \omterm{thm:b1:reals:floor}{partes inteiras} (operação crescente):
$nk \leq \lfloor nx \rfloor \leq nk + n
- 1$. Dividindo por $n$: $k \leq \frac{\lfloor nx \rfloor}{n} < k + 1$,
de modo que a \omterm{thm:b1:reals:floor}{parte inteira} externa é $k$.
\end{solution}

\begin{solution}{exo:b1:reals:4}
Todo elemento de $A$ está em $B$, de sorte que $\sup B$ limita $A$
superiormente: portanto, $\sup A \leq \sup B$ ($\sup A$ é a \emph{menor} \omterm{def:b1:reals:bounds}{cota
superior}). Simetricamente, $\inf B \leq \inf A$. Por fim, $\inf A \leq
\sup A$, pois $A$ é não vazio: qualquer $a \in A$ fica entre eles.
\end{solution}

\begin{solution}{exo:b1:reals:5}
Sejam $s = \sup A$, $t = \sup B$. Todo $a + b \leq s + t$: \omterm{def:b1:reals:bounds}{cota
superior}. Para $\varepsilon > 0$, escolha $a > s - \frac\varepsilon2$ e
$b > t - \frac\varepsilon2$ (\cref{prop:b1:reals:epsilon}): então $a +
b > s + t - \varepsilon$. Pela caracterização por $\varepsilon$,
$\sup(A+B) = s + t$.

Para $-A$: $m$ limita $-A$ superiormente $\iff$ $-m$ limita $A$
inferiormente; a menor \omterm{def:b1:reals:bounds}{cota superior} de $-A$ corresponde, portanto, à
maior cota inferior de $A$: $\sup(-A) = -\inf A$.
\end{solution}

\begin{solution}{exo:b1:reals:6}
Para todo $x$: $f(x) + g(x) \leq \sup f + \sup g$; tomando o \omterm{def:b1:reals:bounds}{supremo} do
lado esquerdo, obtém-se a desigualdade. Exemplo estrito: $E = \{0, 1\}$,
$f = \mathbf{1}_{\{0\}}$ (valor $1$ em $0$, e $0$ no resto), $g =
\mathbf{1}_{\{1\}}$: $\sup(f + g) = 1 < 2 = \sup f + \sup g$.

Nenhuma contradição com o \cref{exo:b1:reals:5}: lá, $a \in A$ e $b
\in B$ variam \emph{independentemente}; aqui o mesmo $x$ alimenta $f$
e $g$ --- o \omterm{def:b1:logic:sets}{conjunto} $\{f(x) + g(x) : x \in E\}$ é menor que o
\omterm{def:b1:logic:sets}{conjunto} $\{f(x) + g(y) : x, y \in E\}$.
\end{solution}

\begin{solution}{exo:b1:reals:7}
$\sqrt 6$ é irracional: $6 = 2 \times 3$ não é quadrado perfeito,
e $v_2(6q^2) = 1 + 2v_2(q)$ ímpar impede $6q^2 = r^2$ (como no
\cref{exo:b1:arith:7}). Suponha agora que $x = \sqrt 2 + \sqrt 3 \in \Q$.
Então $x^2 = 5 + 2\sqrt 6 \in \Q$, de modo que $\sqrt 6 = \frac{x^2 - 5}{2} \in
\Q$: contradição. Portanto, $\sqrt 2 + \sqrt 3 \notin \Q$.
\end{solution}

\begin{solution}{exo:b1:reals:8}
Sejam $x < y$. Tome $n \in \N$ com $2^n > \frac{1}{y - x}$ (Arquimedes:
$2^n \geq n + 1$ por uma indução fácil, de modo que alguma potência de $2$
excede qualquer real). Então, como na demonstração do \cref{thm:b1:reals:density}, com
$2^n$ no lugar de $n$: $m = \lfloor 2^n x \rfloor + 1$ satisfaz $x <
\frac{m}{2^n} < y$. Logo, $D$ é denso.
\end{solution}

\begin{solution}{exo:b1:reals:9}
\begin{enumerate}
  \item Suponha $\alpha > 0$. Primeiro, $\alpha \in G$. Suponha que não: pela
        caracterização por $\varepsilon$ do \omterm{def:b1:reals:bounds}{ínfimo} com
        $\varepsilon = \alpha$, existe $g \in G$ com $\alpha < g <
        2\alpha$ (estrita à esquerda, pois $\alpha \notin G$); então,
        com $\varepsilon = g - \alpha$, existe $h \in G$ com
        $\alpha < h < g$. Ora, $g - h \in G$ e $0 < g - h < g -
        \alpha < \alpha$: um elemento de $G \cap \intoo{0}{+\infty}$
        abaixo do seu \omterm{def:b1:reals:bounds}{ínfimo}, absurdo. Logo, $\alpha \in G$, e $\alpha\Z
        \subseteq G$ ($G$ é \omterm{def:b1:structures:group}{grupo}). Reciprocamente, para $x \in G$, seja
        $k = \lfloor x/\alpha \rfloor$: então $x - k\alpha \in G$ e
        $0 \leq x - k\alpha < \alpha$, e a definição de $\alpha$
        força $x - k\alpha = 0$. Portanto, $G = \alpha\Z$.
  \item Suponha $\alpha = 0$ e sejam $x < y$. Existe $g \in G$
        com $0 < g < y - x$. O múltiplo $kg$ com $k = \lfloor x/g
        \rfloor + 1$ satisfaz $x < kg \leq x + g < y$, e $kg \in
        G$: densidade.
\end{enumerate}
$G = \Z + \sqrt 2\,\Z$ é um \omterm{def:b1:structures:subgroup}{subgrupo} de $(\R, +)$. Ele não é da
forma $\alpha\Z$: caso contrário, $1 = p\alpha$ e $\sqrt 2 = q\alpha$
($p, q \in \Z$) dariam $\sqrt 2 = \frac qp \in \Q$,
contradição. Pela dicotomia, $G$ é denso em $\R$.
\end{solution}

\begin{solution}{exo:b1:reals:10}
Sejam $s = \sup A > 0$, $t = \sup B > 0$. Para $a \in A$, $b \in B$: $ab
\leq st$ (multiplicando desigualdades entre números \emph{positivos}).
Para $0 < \varepsilon < \min(s, t)$: escolha $a > s - \varepsilon$ e
$b > t - \varepsilon$; então
\[
ab > (s - \varepsilon)(t - \varepsilon)
= st - \varepsilon(s + t) + \varepsilon^2
> st - \varepsilon (s + t),
\]
e $\varepsilon(s+t)$ pode ser tornado arbitrariamente pequeno: pela
caracterização por $\varepsilon$ (na forma: nenhum número $< st$ limita
$AB$ superiormente), $\sup AB = st$.

A positividade é essencial: $A = B = \{-1, 0\}$ dá $AB = \{0, 1\}$,
$\sup AB = 1$, ao passo que $\sup A \cdot \sup B = 0 \times 0 = 0$.
\end{solution}

\begin{solution}{exo:b1:reals:11}
Escreva $s = \sup A$, $i = \inf A$. Para $a, a' \in A$: $a \leq s$
e $a' \geq i$ dão $a - a' \leq s - i$; por simetria, $\abs{a -
a'} \leq s - i$, de modo que $s - i$ limita superiormente o \omterm{def:b1:logic:sets}{conjunto} das
distâncias. Para $\varepsilon > 0$, escolha $a > s - \frac\varepsilon2$ e $a' < i
+ \frac\varepsilon2$ (\cref{prop:b1:reals:epsilon} e o seu espelho
para o \omterm{def:b1:reals:bounds}{ínfimo}): então $\abs{a - a'} \geq a - a' > s - i -
\varepsilon$. Pela caracterização por $\varepsilon$,
$\operatorname{diam} A = s - i$.

Todo $a \in A$ satisfaz $i \leq a \leq s$, de modo que $A \subseteq
\intcc{i}{s}$, um \omterm{prop:b1:reals:intervals}{intervalo} fechado de comprimento $\operatorname{diam}
A$. Se um \omterm{prop:b1:reals:intervals}{intervalo} fechado $\intcc{u}{v}$ contém $A$, então $v$ é
\omterm{def:b1:reals:bounds}{cota superior} e $u$ é cota inferior de $A$, de sorte que $u \leq i$ e $v
\geq s$: $\intcc{i}{s} \subseteq \intcc{u}{v}$. Portanto,
$\intcc{i}{s}$ é o menor deles.
\end{solution}

\begin{solution}{exo:b1:reals:12}
$0 \in E$, logo $E \neq \emptyset$. Se $x > \max(1, y)$, então $x^2 >
x > y$, de modo que $E$ é limitado superiormente por $\max(1, y)$: $s = \sup E$
existe (\cref{thm:b1:reals:sup}), e $s \geq \min(1, y) > 0$,
pois $\min(1, y) \in E$: com efeito, se $y \geq 1$, então $1^2 = 1
\leq y$, e, se $y < 1$, então $y^2 < y$.

\emph{$s^2 < y$ é impossível.} Escolha $0 < h < 1$ com $h <
\frac{y - s^2}{2s + 1}$. Então
\[
(s + h)^2 = s^2 + 2sh + h^2 \leq s^2 + (2s + 1)h < y ,
\]
de modo que $s + h \in E$, contradizendo que $s$ limita $E$ superiormente.

\emph{$s^2 > y$ é impossível.} Escolha $0 < h < s$ com $h <
\frac{s^2 - y}{2s}$. Então $(s - h)^2 = s^2 - 2sh + h^2 > s^2 -
2sh > y$; todo $x \in E$ satisfaz $x^2 \leq y < (s - h)^2$,
logo $x < s - h$ (ambos são $\geq 0$): $s - h$ é uma \omterm{def:b1:reals:bounds}{cota superior}
de $E$ menor que $s$, contradizendo a minimalidade.

Portanto, $s^2 = y$. Unicidade: se $0 < s < s'$, então $s^2 <
s'^2$, de modo que duas raízes positivas distintas não podem ambas
ter quadrado $y$. Monotonicidade: se $0 < y < y'$, então $\sqrt y \neq \sqrt{y'}$ e
$\sqrt y > \sqrt{y'}$ daria $y > y'$ ao elevar ao quadrado: logo, $\sqrt
y < \sqrt{y'}$.
\end{solution}

\begin{solution}{pb:b1:reals:1}
\textbf{1.} Para $\frac 18$: $10 = 8 \cdot 1 + 2$, $20 = 8 \cdot 2
+ 4$, $40 = 8 \cdot 5 + 0$; algarismos $1, 2, 5$, resto $0$ e, depois,
só zeros: $\frac 18 = 0.125$. Para $\frac 17$: $10 = 7 \cdot 1 +
3$, $30 = 7 \cdot 4 + 2$, $20 = 7 \cdot 2 + 6$, $60 = 7 \cdot 8 +
4$, $40 = 7 \cdot 5 + 5$, $50 = 7 \cdot 7 + 1$: algarismos $1, 4, 2,
8, 5, 7$, restos $3, 2, 6, 4, 5, 1$. O resto
voltou a $r = 1$, de modo que os seis passos se repetem literalmente
para sempre: $\frac 17 = 0.(142857)$, com os restos ciclando por $1, 3, 2, 6,
4, 5$.

\textbf{2.} $\frac 13$ na base $2$ ($r_0 = 1$): $2 = 3 \cdot 0 +
2$, $4 = 3 \cdot 1 + 1$, e $r = 1$ retorna: $\frac 13 =
(0.\overline{01})_2$. $\frac{5}{16}$ na base $2$: $10 = 16 \cdot
0 + 10$, $20 = 16 \cdot 1 + 4$, $8 = 16 \cdot 0 + 8$, $16 = 16
\cdot 1 + 0$: $\frac{5}{16} = (0.0101)_2$, terminando. $\frac 12$
na base $3$: $3 = 2 \cdot 1 + 1$, e $r = 1$ retorna de imediato:
$\frac 12 = (0.\overline{1})_3$.

\textbf{3.} O resto do algoritmo após $N$ passos é $r_N =
b^N p \bmod q$ (demonstrado formalmente na questão 14; aqui é a
observação de que cada passo multiplica o resto por $b$ e o
reduz módulo $q$). Todos os algarismos posteriores são $0$ se, e somente
se, algum $r_N = 0$, isto é, $q \mid b^N p$; como $\gcd(p, q) = 1$, o lema de
Gauss dá $q \mid b^N$. Se $q \mid b^N$, todo fator \omterm{def:b1:arith:prime}{primo} de
$q$ \omterm{def:b1:arith:divides}{divide} $b^N$ e, portanto, \omterm{def:b1:arith:divides}{divide} $b$ (primalidade). Reciprocamente,
se todo \omterm{def:b1:arith:prime}{primo} de $q = p_1^{a_1} \cdots p_r^{a_r}$ \omterm{def:b1:arith:divides}{divide} $b$, então,
com $A = \max_i a_i$, cada $p_i^{a_i}$ \omterm{def:b1:arith:divides}{divide} $b^A$, e os
$p_i^{a_i}$ são dois a dois \omterm{cor:b1:arith:bezout}{primos entre si}, de modo que $q \mid b^A$. Para $q = 20 =
2^2 \cdot 5$: os dois \omterm{def:b1:arith:prime}{primos} \omterm{def:b1:arith:divides}{dividem} $10$ ($\frac{1}{20} = 0.05$),
mas $2 \nmid 3$, de sorte que $\frac{1}{20}$ se repete para sempre na base $3$.

\textbf{4.} $1^2 = 1 < 2 < 4 = 2^2$ dá $s_0 = 1$. Depois, $1.4^2
= 1.96 < 2 < 2.25 = 1.5^2$: $\lfloor 10\sqrt 2 \rfloor = 14$,
$s_1 = 1.4$. Em seguida, $1.41^2 = 1.9881 < 2 < 2.0164 = 1.42^2$: $s_2 =
1.41$; $1.414^2 = 1.999396 < 2 < 2.002225 = 1.415^2$: $s_3 =
1.414$; $1.4142^2 = 1.99996164 < 2 < 2.00024449 = 1.4143^2$: $s_4
= 1.4142$. Em cada caso, as desigualdades exibidas dizem exatamente
$s_n \leq \sqrt 2 < s_n + 10^{-n}$, que é a definição da
\omterm{thm:b1:reals:floor}{parte inteira} de $10^n \sqrt 2$.

\textbf{5.} $A_0 = \lfloor x \rfloor = 0$, pois $0 \leq x < 1$.
De $A_{n-1} \leq b^{n-1} x < A_{n-1} + 1$, multiplique por $b$:
\[
b\,A_{n-1} \leq b^n x < b\,A_{n-1} + b .
\]
O inteiro $b\,A_{n-1}$ é $\leq b^n x$, de modo que $b\,A_{n-1} \leq
A_n$; e $b^n x < b\,A_{n-1} + b$, com $b\,A_{n-1} + b$ inteiro,
força $A_n \leq b\,A_{n-1} + b - 1$. Portanto, $0 \leq d_n =
A_n - b\,A_{n-1} \leq b - 1$: um algarismo.

\textbf{6.} Telescopando: $d_k b^{-k} = A_k b^{-k} - A_{k-1}
b^{-(k-1)}$, de modo que
\[
\sum_{k=1}^n d_k b^{-k} = A_n b^{-n} - A_0 = s_n .
\]
Dividir $A_n \leq b^n x < A_n + 1$ por $b^n$ dá $s_n \leq x <
s_n + b^{-n}$.

\textbf{7.} Indução: $b^0 = 1 \geq 1$ e $b^{n+1} = b \cdot
b^n \geq 2(n + 1) \geq n + 2$. Monotonicidade: $s_n - s_{n-1} = d_n
b^{-n} \geq 0$. Cada $s_n \leq x$ (questão 6): $x$ é \omterm{def:b1:reals:bounds}{cota
superior} de $\{s_n\}$. Para $\varepsilon > 0$, a \omterm{thm:b1:reals:archimedes}{propriedade
arquimediana} fornece $n$ com $n + 1 > \frac1\varepsilon$, donde
$b^{-n} < \varepsilon$, e então $s_n > x - b^{-n} > x -
\varepsilon$, pela questão 6. Pela \cref{prop:b1:reals:epsilon}, $x =
\sup_n s_n$.

\textbf{8.} Suponha $d_k = b - 1$ para todo $k > N$. Para $n > N$,
a soma geométrica finita dá
\[
s_n = s_N + (b - 1)\sum_{k=N+1}^{n} b^{-k}
= s_N + b^{-N} - b^{-n} .
\]
Assim, $x \geq s_n = s_N + b^{-N} - b^{-n}$ para todo $n$; fazendo
o último termo encolher abaixo de qualquer $\varepsilon$ (questão 7), $x
\geq s_N + b^{-N}$. Mas a questão 6 no posto $N$ diz que $x < s_N +
b^{-N}$: contradição. A cadeia $(d_n)$ é própria.

\textbf{9.} Limitada: $t_n \leq (b-1)\sum_{k=1}^n b^{-k} = 1 -
b^{-n} < 1$, e $(t_n)$ é não decrescente, de modo que $y = \sup t_n$
existe, com $0 \leq y \leq 1$. Fixe $n$. Para a estimativa
bilateral: $t_n \leq y$ é claro. Pela propriedade de ser própria, tome
$m > n$ com $e_m \leq b - 2$. Para $p \geq m$:
\[
t_p - t_n = \sum_{k=n+1}^{p} e_k b^{-k}
\leq (b^{-n} - b^{-p}) - b^{-m} < b^{-n} - b^{-m},
\]
com a soma do meio perdendo ao menos $b^{-m}$ contra o máximo com todos
os algarismos $(b-1)$; para $p \leq m$, também $t_p \leq t_m \leq t_n + b^{-n} -
b^{-m}$ (monotonicidade mais o caso $p = m$). Portanto, todo
$t_p \leq t_n + b^{-n} - b^{-m}$, de modo que $y \leq t_n + b^{-n} -
b^{-m} < t_n + b^{-n}$. (Com $n = 0$: $y < 1$, logo $y \in
\intco{0}{1}$.) Ora, $b^n t_n = \sum_{k \leq n} e_k b^{n-k}$ é um
inteiro, e $b^n t_n \leq b^n y < b^n t_n + 1$: assim, $\lfloor b^n
y \rfloor = b^n t_n$. Por fim, os algarismos de $y$: $d_n(y) = b^n
t_n - b \cdot b^{n-1} t_{n-1} = b^n(t_n - t_{n-1}) = e_n$.

\textbf{10.} A questão 9 diz: (valor de uma cadeia) tem por algarismos
(a cadeia); as questões 5--8 dizem: (algarismos de $x$) formam uma
cadeia própria cujas truncagens têm \omterm{def:b1:reals:bounds}{supremo} $x$ (questão 7). Logo, as
duas aplicações se compõem dando a identidade nas duas ordens: são
bijeções inversas uma da outra entre $\intco{0}{1}$ e as cadeias próprias.
Se duas cadeias próprias tivessem o mesmo valor, aplicar a \omterm{def:b1:logic:map}{aplicação}
dos algarismos as tornaria iguais: unicidade. Este é o \emph{teorema da
expansão $b$-ádica}.

\textbf{11.} Seja $e_n = b - 1$ para $n > M$, com $M \geq 0$ mínimo.
Como na questão 8, $t_n = t_M + b^{-M} - b^{-n}$ para $n \geq M$,
de modo que o valor é $\sup t_n = t_M + b^{-M}$. Se $M = 0$, o valor
é $0 + 1 = 1$: na base $10$, $0.999\dots = 1$ exatamente --- e não
aproximadamente. Se $M \geq 1$, a minimalidade dá $e_M \leq b - 2$,
e o valor é
\[
t_M + b^{-M} = \frac{b^M t_M + 1}{b^M} \in \intoo{0}{1},
\]
uma fração $b$-ádica, cuja expansão \emph{própria} é $e_1 \dots
e_{M-1}\,(e_M + 1)\,000\dots$ (uma cadeia que termina é própria,
e o seu valor é o mesmo número). Reciprocamente, um real com duas
representações deve ter uma imprópria (ser própria fixa a
representação, questão 10) e, portanto, ser dessa forma. E cada
$m/b^N \in \intoo{0}{1}$, escrito com último algarismo não nulo $d_N$,
de fato tem o gêmeo impróprio $d_1 \dots d_{N-1}(d_N -
1)(b-1)(b-1)\dots$: exatamente as frações $b$-ádicas carregam dois
nomes, e todos os demais reais, um só.

\textbf{12.} Digamos que as cadeias coincidam até $m - 1$, com truncagem
comum $P = s_{m-1}$, e que $d_m < e_m$. Pela questão 9 (estimativa
superior estrita no posto $m$), $x < P + d_m b^{-m} + b^{-m} = P +
(d_m + 1)b^{-m} \leq P + e_m b^{-m} \leq y$, o último passo
porque $P + e_m b^{-m}$ é a truncagem $t_m \leq y$ de $y$. Logo,
$d_m < e_m \implies x < y$; trocando os papéis, $e_m < d_m \implies
y < x$; e, como as cadeias diferem em $m$, uma das duas
coisas ocorre. As duas direções seguem.

\textbf{13.} Seja $z = bx - A_1 \in \intco{0}{1}$ (com efeito, $A_1
\leq bx < A_1 + 1$). Para $n \geq 0$: $b^n z = b^{n+1} x - b^n
A_1$ com $b^n A_1 \in \Z$, de modo que, por $\lfloor u - K \rfloor =
\lfloor u \rfloor - K$ ($K$ inteiro),
\[
A_n(z) = A_{n+1}(x) - b^n A_1(x) .
\]
Portanto, $d_n(z) = A_n(z) - b\,A_{n-1}(z) = A_{n+1} - b^n A_1 -
b\,A_n + b^n A_1 = d_{n+1}(x)$. Assim, a parte fracionária de $bx$
carrega os algarismos deslocados; iterando $m$ vezes, a parte
fracionária de $b^m x$ tem algarismos $(d_{n+m})_{n \geq 1}$.

\textbf{14.} Divisão euclidiana: $b^n p = q\,Q_n + r_n$ com $0
\leq r_n < q$. Divida por $q$: $b^n x = Q_n + \frac{r_n}{q}$ com
$0 \leq \frac{r_n}{q} < 1$, de modo que $Q_n = \lfloor b^n x \rfloor =
A_n$, o que dá $A_n = \frac{b^n p - r_n}{q}$. Quanto à recorrência:
$b^n p = b(q\,A_{n-1} + r_{n-1}) = q\,(b\,A_{n-1}) +
b\,r_{n-1}$, de sorte que $b^n p$ e $b\,r_{n-1}$ diferem por um múltiplo de
$q$: $r_n = (b\,r_{n-1}) \bmod q$.

\textbf{15.} Divida $b\,r_{n-1}$ por $q$: $b\,r_{n-1} = q\,c +
r_n$ com $c = \lfloor b\,r_{n-1}/q \rfloor$. Substituindo na
expressão da questão 14: $b^n p = q(b\,A_{n-1} + c) + r_n$,
e a unicidade da divisão euclidiana identifica $A_n = b\,A_{n-1}
+ c$, isto é, $d_n = c = \lfloor b\,r_{n-1}/q \rfloor$. O algarismo
$n$ depende apenas de $r_{n-1}$ --- o laço da divisão longa da Parte
I, agora certificado.

\textbf{16.} Os $q + 1$ restos $r_0, \dots, r_q$ assumem valores
no \omterm{def:b1:logic:sets}{conjunto} de $q$ elementos $\intint{0}{q-1}$: pelo princípio da casa dos
pombos (\cref{cor:b1:counting:pigeonhole}), dois coincidem, digamos
$r_N = r_{N+T}$ com $0 \leq N < N + T \leq q$. Como $r_n$
determina $r_{n+1}$ (questão 14), a indução dá $r_{n+T} =
r_n$
para todo $n \geq N$; e, como $r_{n-1}$ determina $d_n$
(questão 15), $d_{n+T} = d_n$ para todo $n \geq N + 1$. A expansão
de todo racional é periódica a partir de certo ponto, com pré-período
$\leq q$ e período $\leq q$.

\textbf{17.} Os algarismos da parte fracionária de $b^T y$ são
$(d_{n+T}) = (d_n)$ (lema do deslocamento, e depois periodicidade pura): a
mesma cadeia própria que a de $y$. Pela questão 10, os valores são
iguais: $b^T y - \lfloor b^T y \rfloor = y$, de modo que $(b^T - 1)\,y = \lfloor
b^T y \rfloor = A_T \in \N$ e
\[
y = \frac{A_T}{b^T - 1} ,
\]
racional, com denominador que \omterm{def:b1:arith:divides}{divide} $b^T - 1$; o numerador $A_T$
é o inteiro cujos algarismos na base $b$ são $d_1 \dots d_T$. Verificação:
$0.(142857) = \frac{142857}{999999}$, e $142857 \times 7 =
999999$, de modo que isso é $\frac 17$.

\textbf{18.} Se $d_{n+T} = d_n$ para $n > N$, a parte fracionária
$z$ de $b^N x$ tem algarismos $(d_{N+n})_{n\geq1}$ (lema do deslocamento),
que são puramente periódicos; pela questão 17, $z \in \Q$. Então $b^N
x = A_N + z$ dá $x = (A_N + z)/b^N \in \Q$. Com a questão 16:
$x$ racional $\iff$ expansão periódica a partir de certo ponto. O lado
direito menciona a base, e o esquerdo não: a periodicidade numa base é
equivalente à racionalidade e, portanto, à periodicidade em toda base.

\textbf{19.} Para $x = \frac 1q$, $r_n = b^n \bmod q$. Se
$\gcd(b, q) = 1$, então $r_T = r_0 = 1$ se, e somente se, $b^T \equiv
1 \pmod q$; tal $T$ existe (a casa dos pombos dá $b^i \equiv b^j$,
$i < j$, e $b$ é invertível módulo $q$, de modo que $b^{j-i} \equiv
1$), e o menor deles --- a ordem multiplicativa --- torna os
restos, e portanto os algarismos, puramente periódicos de período $T$.
Nenhum período menor é possível: um período $T'$ daria $(b^{T'}
- 1)\frac1q \in \N$ (questão 17), isto é, $q \mid b^{T'} - 1$.
Para $q = 7$, $b = 10$: $10 \equiv 3$, $10^2 \equiv 2$, $10^3
\equiv 6$, $10^4 \equiv 4$, $10^5 \equiv 5$, $10^6 \equiv 1
\pmod 7$: ordem $6$ e, de fato, $\frac 17$ tem período seis.

\textbf{20.} A cadeia tem infinitos $0$s (os algarismos de $x^*$ são
quase todos nulos), de modo que é própria, e $x^*$ está bem definido
(questão 9). Suponha os algarismos periódicos a partir de $N$, com
período $T$. Infinitos algarismos valem $1$ (um por número
triangular), de sorte que algum $1$ ocupa uma posição $j > N$; então a
periodicidade põe um $1$ em toda posição $j + kT$: de $j$ em
diante, os espaçamentos entre $1$s consecutivos são no máximo $T$. Mas
os $1$s ficam exatamente nos números triangulares, cujos espaçamentos
consecutivos $\frac{(j+1)(j+2)}{2} - \frac{j(j+1)}{2} = j + 1$ excedem $T$ a partir
de certo ponto: contradição. Não é periódica a partir de ponto algum,
de modo que, pela questão 18, $x^* \notin \Q$ --- irracionalidade lida
apenas no ritmo dos algarismos.

\textbf{21.} Dados $x < y$, a questão 7 fornece $n$ com $b^{-n}
< y - x$; ponha $m = \lfloor b^n x \rfloor + 1$. Então $b^n x < m
\leq b^n x + 1 < b^n y$, de modo que $x < \frac{m}{b^n} < y$: densidade,
para toda base de uma só vez ($b = 2$ recupera o
\cref{exo:b1:reals:8}). Para $\frac pq \in \intoo{0}{1}$ na base
$b = q$: o primeiro algarismo é $\lfloor q \cdot \frac pq \rfloor =
p$, e a parte fracionária de $q \cdot \frac pq = p$ é $0$: todos
os algarismos posteriores se anulam, uma expansão finita $\frac pq =
(0.p)_q$. Terminar depende da base; a periodicidade ---
a racionalidade --- não (questão 18).

\textbf{22.} Cada $e_k \in \{1, 2\}$ é um algarismo da base $10$, e
a cadeia nunca termina só com $9$s: é própria. O seu valor $y$ está em
$\intco{0}{1}$ e tem por algarismos exatamente os $(e_k)$ (questão 9). Fixe
$k$: o $k$-ésimo algarismo de $y$ é $e_k$, escolhido $\neq$ do
$k$-ésimo algarismo de $x_k$, de modo que as cadeias próprias de $y$ e
de $x_k$ diferem e, portanto, $y \neq x_k$ (questão 10: a codificação é
\omterm{def:b1:logic:inj}{injetiva}). Assim, $y$ não está em lista alguma: nenhuma \omterm{def:b1:logic:map}{aplicação}
$\N^* \to \intco{0}{1}$ é \omterm{def:b1:logic:inj}{sobrejetiva}. Os reais, ao contrário dos
racionais, não podem ser enumerados --- a não enumerabilidade, cuja
teoria o \cref{ch:b1:topology} desenvolve.

\textbf{23.} Se as expansões coincidem até $n$, então $x$ e $y$
têm a mesma truncagem $s_n$, e a questão 6 põe os dois em
$\intco{s_n}{s_n + b^{-n}}$, um \omterm{prop:b1:reals:intervals}{intervalo} de comprimento $b^{-n}$:
$\abs{x - y} < b^{-n}$. Recíproca: $x = 0.1$ e $y = 0.0999$
(finita, logo própria) satisfazem $\abs{x - y} = 10^{-4} <
10^{-3}$, e ainda assim as suas expansões diferem já no primeiro
algarismo. Os culpados são as frações $b$-ádicas da questão 11: perto
delas, um movimento minúsculo inverte todo algarismo exibido ($0.0999 \to
0.1000$), porque elas são precisamente os reais em que
espreita o gêmeo impróprio.

\textbf{24.} $p = 1$, $q = 10$, $b = 2$, $r_0 = 1$: $2 = 10 \cdot
0 + 2$, $4 = 10 \cdot 0 + 4$, $8 = 10 \cdot 0 + 8$, $16 = 10
\cdot 1 + 6$, $12 = 10 \cdot 1 + 2$ --- e $r_5 = 2 = r_1$: os
restos ciclam por $(2, 4, 8, 6)$ a partir do índice $1$. Algarismos: $d_1 =
0$, e depois o bloco que se repete $d_2 d_3 d_4 d_5 = 0, 0, 1, 1$:
\[
\tfrac{1}{10} = (0.0\overline{0011})_2 ,
\]
pré-período $1$, período $4$. Pela questão 3, uma expansão finita na
base $2$ exigiria que todo \omterm{def:b1:arith:prime}{primo} de $10$ dividisse $2$; o \omterm{def:b1:arith:prime}{primo}
$5$ se recusa. Assim, $0.1$ \emph{não} é representável por cadeia
binária finita alguma --- um computador que armazena finitos bits guarda
apenas uma truncagem, e os erros de truncagem acumulados são a razão
pela qual, em ponto flutuante, $0.1 + 0.2$ difere de $0.3$ nos últimos bits.

\textbf{25.} (i) A completude produziu os valores: $x = \sup s_n$
e $y = \sup t_n$ (questões 7 e 9) --- só com $\Q$, a
cadeia própria de $\sqrt 2$ não nomearia nada. (ii) A
\omterm{thm:b1:reals:archimedes}{propriedade arquimediana} fez $b^{-n}$ ficar, a partir de certo ponto,
menor que qualquer $\varepsilon$, forçando as truncagens a se fecharem sobre
o seu \omterm{def:b1:reals:bounds}{supremo} (questões 7 e 21). (iii) A cláusula de unicidade da
\omterm{thm:b1:reals:floor}{parte inteira} identificou $Q_n = A_n$ na questão 14 e legitimou toda
extração de algarismo $\lfloor u - K \rfloor = \lfloor u \rfloor - K$
(questão 13). (iv) O princípio da casa dos pombos, aplicado a finitos
restos, é o único motor da periodicidade (questão 16).
Moral: as cadeias próprias codificam $\intco{0}{1}$ fielmente e convertem
a racionalidade num ritmo visível; mas somar cadeias de algarismos
exige transportes vindos de infinitamente longe, à direita, de modo que
regra alguma em estágios finitos calcula sequer o primeiro algarismo de
uma soma --- ao passo que a interface do \omterm{def:b1:reals:bounds}{supremo} do \cref{thm:b1:reals:sup} dá
conta de toda a análise com um único axioma. Os algarismos são um
magnífico \emph{retrato} de $\R$; o \omterm{def:b1:reals:bounds}{supremo} é o seu \emph{motor}.
\end{solution}
