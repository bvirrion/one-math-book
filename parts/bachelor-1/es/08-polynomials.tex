\chapter{Polinomios}\label{ch:b1:poly}

Los \omterm{def:b1:poly:def}{polinomios} son las funciones favoritas del algebrista --- salvo
que aquí no se tratan como funciones, sino como expresiones formales en
una indeterminada $X$, que se suman y multiplican con las reglas de un
\omterm{def:b1:structures:ring}{anillo} conmutativo. La teoría corre asombrosamente en paralelo con la
del~\cref{ch:b1:arith}: una división euclídea, un mcd y relaciones de
Bézout, elementos irreducibles y una factorización única. En todo el
capítulo, $K$ denota $\Q$, $\R$ o $\C$.

\section{El anillo $K[X]$}

\begin{definition}[Polinomio, grado]\label{def:b1:poly:def}
Un \emph{polinomio}\index{polinomio} con coeficientes en $K$ es una
suma formal
\[
P = a_0 + a_1 X + a_2 X^2 + \dots + a_n X^n
= \sum_{k} a_k X^k,
\]
con $a_k \in K$ todos nulos a partir de cierto índice. Con la suma
natural y el producto
\[
\Bigl(\sum_i a_i X^i\Bigr)\Bigl(\sum_j b_j X^j\Bigr)
= \sum_k \Bigl(\sum_{i+j=k} a_i b_j\Bigr) X^k,
\]
el \omterm{def:b1:logic:sets}{conjunto} $K[X]$ es un \omterm{def:b1:structures:ring}{anillo} conmutativo. El \emph{grado} $\deg P$
de $P \neq 0$ es el mayor $n$ con $a_n \neq 0$; $a_n$ es el
\emph{coeficiente principal} ($P$ es \emph{mónico}\index{polinomio
mónico} cuando $a_n = 1$), y por convenio $\deg 0 = -\infty$. Todo
polinomio define una función $x \mapsto P(x)$ en $K$ por sustitución.
\end{definition}

\begin{proposition}[Reglas de los grados; dominio de integridad]\label{prop:b1:poly:degree}
Para $P, Q \in K[X]$:
\[
\deg(P + Q) \leq \max(\deg P, \deg Q),
\qquad
\deg(PQ) = \deg P + \deg Q .
\]
En consecuencia, $K[X]$ es un \omterm{def:b1:structures:field}{dominio de integridad}, y sus unidades son
las constantes no nulas.
\end{proposition}

\begin{proof}
La regla de la suma es clara (los coeficientes más allá del máximo se
anulan). Para el producto, sean $a_m$ y $b_n$ los coeficientes
principales: el coeficiente de $X^{m+n}$ en $PQ$ es $a_m b_n \neq 0$
($K$ es un \omterm{def:b1:structures:field}{cuerpo} y, por tanto, un \omterm{def:b1:structures:field}{dominio de integridad}), y todos los
coeficientes superiores se anulan. Si $P, Q \neq 0$, entonces
$\deg PQ = \deg P + \deg Q \geq 0$, luego $PQ \neq 0$: \omterm{def:b1:structures:field}{dominio de
integridad}. Si $PQ = 1$, entonces $\deg P + \deg Q = 0$ obliga a
$\deg P = \deg Q = 0$: los elementos invertibles son las constantes
invertibles, es decir, todo $K^*$.
\end{proof}

\begin{theorem}[División euclídea]\label{thm:b1:poly:division}
Sean $A, B \in K[X]$ con $B \neq 0$. Existe exactamente un par
$(Q, R)$ de \omterm{def:b1:poly:def}{polinomios} con
\[
A = BQ + R, \qquad \deg R < \deg B .
\]
\end{theorem}

\begin{proof}
\emph{Existencia}, por inducción fuerte sobre $\deg A$. Si
$\deg A < \deg B$, tómese $(Q, R) = (0, A)$. En caso contrario,
escríbanse $A = a X^m + \dots$, $B = b X^n + \dots$ con $m \geq n$; el
\omterm{def:b1:poly:def}{polinomio} $A_1 = A - \frac ab X^{m-n} B$ tiene grado $< m$ (los
términos principales se cancelan), luego, por inducción,
$A_1 = BQ_1 + R$ con $\deg R < \deg B$, y
$A = B(Q_1 + \frac ab X^{m-n}) + R$.

\emph{Unicidad}: si $BQ + R = BQ' + R'$, entonces $B(Q - Q') = R' - R$
con $\deg(R' - R) < \deg B$; por la regla de los grados, esto obliga a
$Q - Q' = 0$ y después a $R = R'$.
\end{proof}

\begin{example}\label{ex:b1:poly:division}
Divídase $A = X^4 + X^3 - 2X + 1$ entre $B = X^2 + 1$:
\[
X^4 + X^3 - 2X + 1 = (X^2 + 1)(X^2 + X - 1) + (-3X + 2).
\]
(Cálculo: réstese $X^2 B$, después $X B$ y después $-B$; el resto
$-3X + 2$ tiene grado $1 < 2$.)
\end{example}

\begin{method}[Esquema de Horner]\label{met:b1:poly:horner}
Para evaluar $P = a_nX^n + \dots + a_0$ en $x$, o para dividir $P$
entre $X - x$, evítese calcular potencias: léanse los coeficientes de
izquierda a derecha e itérese \emph{multiplicar por $x$ y sumar el
coeficiente siguiente}:
\[
b_n = a_n, \qquad b_{k} = a_{k} + x\,b_{k+1}
\quad (k = n-1, \dots, 0) .
\]
Entonces $b_0 = P(x)$, y los $b_k$ anteriores son los coeficientes del
cociente: $P = (X - x)(b_nX^{n-1} + \dots + b_1) + b_0$ (desarróllese y
compárese). Ejemplo: $P = X^4 - 5X^3 + 6X^2 + 4X - 8$ en $x = 2$: los
$b$ son $1, -3, 0, 4, 0$, luego $P(2) = 0$ y
$P = (X-2)(X^3 - 3X^2 + 4)$ --- una línea en lugar de una división
larga, y $n$ multiplicaciones en lugar de las $\approx n^2/2$ de la
evaluación ingenua. Iterar el esquema en el mismo punto extrae las
\omterm{def:b1:poly:derivative}{multiplicidades} (compárese con el~\cref{ex:b1:poly:multexample}).
\end{method}

\begin{remark}[{Aritmética de $K[X]$}]
Con la división euclídea en la mano, toda la aritmética
del~\cref{ch:b1:arith} se traslada a $K[X]$, con las mismas
demostraciones y haciendo el grado el papel del valor absoluto: mcd
(normalizado para que sea \omterm{def:b1:poly:def}{mónico}), \omterm{met:b1:arith:euclid}{algoritmo de Euclides} extendido,
identidad de Bézout, lema de Gauss, \omterm{def:b1:poly:def}{polinomios} irreducibles y
factorización única. Usaremos libremente estos resultados
trasladados, y el~\cref{exo:b1:poly:6} ensaya uno de ellos.
\end{remark}

\section{Raíces}

\begin{theorem}[Teorema del factor]\label{thm:b1:poly:factor}
Sean $P \in K[X]$ y $a \in K$. El resto de la división de $P$ entre
$X - a$ es la constante $P(a)$. En particular,
\[
P(a) = 0 \iff (X - a) \mid P .
\]
Más en general, raíces distintas $a_1, \dots, a_r$ de $P$ dan la
factorización $P = (X - a_1)\cdots(X - a_r)\, Q$.
\end{theorem}

\begin{proof}
Divídase: $P = (X - a) Q + R$ con $\deg R < 1$, luego $R$ es una
constante $c$; sustituyendo $X = a$ (la sustitución respeta sumas y
productos) se obtiene $P(a) = c$. La equivalencia se sigue. Para varias
raíces, hágase inducción sobre $r$: el caso $r = 1$ es la equivalencia
recién demostrada. Supóngase el \omterm{def:b1:logic:statement}{enunciado} para $r - 1$ raíces y sean
$a_1, \dots, a_r$ raíces distintas de $P$. Escríbase $P = (X - a_1)Q_1$;
para cada $i \geq 2$, sustituyendo $a_i$:
\[
0 = P(a_i) = (a_i - a_1)\,Q_1(a_i),
\qquad a_i - a_1 \neq 0 ,
\]
y, como $K$ no tiene divisores de cero, $Q_1(a_i) = 0$: los $r - 1$
puntos distintos $a_2, \dots, a_r$ son raíces de $Q_1$. La hipótesis de
inducción factoriza $Q_1 = (X - a_2)\cdots(X - a_r)\,Q$, y sustituyendo
hacia atrás se obtiene lo afirmado.
\end{proof}

\begin{corollary}[Un polinomio de grado $n$ tiene a lo sumo $n$ raíces]\label{cor:b1:poly:nroots}
Un $P \in K[X]$ no nulo de grado $n$ tiene a lo sumo $n$ raíces
distintas en $K$. En consecuencia, un \omterm{def:b1:poly:def}{polinomio} (de grado $\leq n$)
que se anula en $n + 1$ puntos distintos es el \omterm{def:b1:poly:def}{polinomio} nulo, y dos
\omterm{def:b1:poly:def}{polinomios} de grado $\leq n$ que coinciden en $n+1$ puntos son iguales.
\end{corollary}

\begin{proof}
Si $a_1, \dots, a_r$ son raíces distintas, el~\cref{thm:b1:poly:factor}
da $P = (X-a_1)\cdots(X-a_r) Q$, luego $n = \deg P \geq r$. Las dos
consecuencias se siguen por reducción al absurdo y por diferencia.
\end{proof}

\begin{example}[El truco del polinomio auxiliar]\label{ex:b1:poly:auxiliary}
Sea $P$ el \omterm{def:b1:poly:def}{polinomio} de grado $\leq n$ con
\[
P(k) = \frac{k}{k+1} \qquad (k = 0, 1, \dots, n) ;
\]
existe y es único por la \omterm{thm:b1:poly:lagrange}{interpolación de Lagrange} de más abajo.
¿Cuánto vale $P(n+1)$? Quítense denominadores: el \omterm{def:b1:poly:def}{polinomio}
$Q = (X+1)P - X$ tiene grado $\leq n + 1$ y se anula en los $n + 1$
puntos $0, 1, \dots, n$, luego, por el~\cref{thm:b1:poly:factor},
\[
Q = c\,X(X-1)(X-2)\cdots(X-n)
\]
para cierta constante $c$. Evalúese donde $Q$ se conoce de forma
independiente: en $X = -1$, $Q(-1) = 0 \cdot P(-1) + 1 = 1$, mientras
que el producto vale $(-1)(-2)\cdots(-1-n) = (-1)^{n+1}(n+1)!$; de ahí
$c = \frac{(-1)^{n+1}}{(n+1)!}$. Evalúese ahora en $X = n + 1$:
\[
(n+2)\,P(n+1) - (n+1) = Q(n+1) = c\,(n+1)! = (-1)^{n+1} ,
\]
de modo que $P(n+1) = \dfrac{(n+1) + (-1)^{n+1}}{n+2}$: igual a $1$
para $n$ impar y a $\frac{n}{n+2}$ para $n$ par --- el \omterm{def:b1:poly:def}{polinomio}
interpolador \emph{no} continúa el patrón $\frac{n+1}{n+2}$. El truco
que hay que recordar: codifíquense los datos como raíces de un
\omterm{def:b1:poly:def}{polinomio} auxiliar, identifíquese la constante desconocida en un punto
ajeno a los datos, y recójase la cosecha.
\end{example}

\begin{definition}[Derivada, multiplicidad]\label{def:b1:poly:derivative}
La \emph{derivada formal} de $P = \sum a_k X^k$ es
$P' = \sum_{k \geq 1} k\,a_k X^{k-1}$; cumple las reglas habituales
$(P+Q)' = P' + Q'$, $(PQ)' = P'Q + PQ'$ (se comprueban en los monomios
y se extienden por linealidad). Una raíz $a$ de $P$ tiene
\emph{multiplicidad}\index{multiplicidad de una raíz} $m \geq 1$ cuando
$(X-a)^m \mid P$ pero $(X-a)^{m+1} \nmid P$; la raíz es \emph{simple}
si $m = 1$ y \emph{múltiple} si $m \geq 2$.
\end{definition}

\begin{proposition}[Multiplicidad mediante derivadas]\label{prop:b1:poly:multiplicity}
$a$ es raíz de $P$ de \omterm{def:b1:poly:derivative}{multiplicidad} $\geq m$ si y solo si
\[
P(a) = P'(a) = \dots = P^{(m-1)}(a) = 0 .
\]
En particular, $a$ es raíz múltiple de $P$ si y solo si
$P(a) = P'(a) = 0$.
\end{proposition}

\begin{proof}
Escríbase $P = (X - a)^m Q + R$, donde $R$ es el resto de la división
entre $(X-a)^m$, con $\deg R < m$. Derivando $k \leq m - 1$ veces y
evaluando en $a$: el primer término aporta $0$ (cada derivada conserva
un factor $(X-a)$), luego $P^{(k)}(a) = R^{(k)}(a)$.

Ahora bien, un \omterm{def:b1:poly:def}{polinomio} $R$ de grado $< m$ queda determinado por
$R(a), R'(a), \dots, R^{(m-1)}(a)$: escribiendo
$R = \sum_{k < m} c_k (X - a)^k$ (es posible: desarróllense las
potencias de $X = (X - a) + a$), se obtiene $R^{(k)}(a) = k!\, c_k$.
Por tanto: todos los $P^{(k)}(a) = 0$ para $k < m$ $\iff$ todos los
$c_k = 0$ $\iff$ $R = 0$ $\iff$ $(X-a)^m \mid P$.
\end{proof}

\begin{example}[Cálculo de una multiplicidad]\label{ex:b1:poly:multexample}
¿Cuál es la \omterm{def:b1:poly:derivative}{multiplicidad} de la raíz $2$ en
$P = X^4 - 5X^3 + 6X^2 + 4X - 8$? Evalúense las derivadas sucesivas en $2$:
\[
P(2) = 16 - 40 + 24 + 8 - 8 = 0, \qquad
P'(2) = 32 - 60 + 24 + 4 = 0,
\]
\[
P''(2) = 48 - 60 + 12 = 0, \qquad
P'''(2) = 48 - 30 = 18 \neq 0
\]
(con $P' = 4X^3 - 15X^2 + 12X + 4$, $P'' = 12X^2 - 30X + 12$,
$P''' = 24X - 30$). Tres valores nulos y después uno no nulo:
\omterm{def:b1:poly:derivative}{multiplicidad} exactamente $3$. Dividiendo, $P = (X - 2)^3(X + 1)$ ---
lo que se comprueba desarrollando $(X-2)^3 = X^3 - 6X^2 + 12X - 8$ y
multiplicando por $X + 1$. La idea clave: las \omterm{def:b1:poly:derivative}{multiplicidades} se leen
en \emph{evaluaciones}, sin necesidad de factorizar --- que es
exactamente como se detectan cuando la factorización está fuera de
alcance.
\end{example}

\begin{example}[Detectar raíces múltiples con un mcd]\label{ex:b1:poly:gcdmultiple}
Cuando no se conoce ninguna raíz, la~\cref{prop:b1:poly:multiplicity}
sigue dando un detector \emph{global} de raíces múltiples: $a$ es raíz
múltiple de $P$ si y solo si es raíz común de $P$ y $P'$, luego $P$
tiene una raíz múltiple (en $\C$) si y solo si $\gcd(P, P') \neq 1$ ---
calculable con el \omterm{met:b1:arith:euclid}{algoritmo de Euclides} sin resolver nada. Muestra:
$P = X^3 - 3X + 2$, $P' = 3X^2 - 3 = 3(X - 1)(X + 1)$. Probando las
raíces $\pm1$ de $P'$ dentro de $P$: $P(1) = 0$ pero $P(-1) = 4$, luego
\[
\gcd(P, P') = X - 1 :
\]
la raíz $1$ es múltiple; dividiendo dos veces, $P = (X - 1)^2(X + 2)$.
El mcd informa incluso del \omterm{def:b1:logic:sets}{conjunto} completo de raíces múltiples, cada
una con la \omterm{def:b1:poly:derivative}{multiplicidad} rebajada en uno --- hecho que aprovecha todo
sistema de álgebra por computador para «factorizar libre de cuadrados»
antes de buscar raíz alguna, y el gemelo polinómico de los argumentos
sin raíces múltiples del~\cref{exo:b1:poly:9}.
\end{example}

\begin{theorem}[Teorema fundamental del álgebra]\label{thm:b1:poly:fta}
Todo \omterm{def:b1:poly:def}{polinomio} no constante de $\C[X]$ tiene una raíz en
$\C$.\index{teorema fundamental del álgebra}
\end{theorem}
\admitted

\begin{remark}
A pesar de su nombre, el teorema es un \omterm{def:b1:logic:statement}{enunciado} de \emph{análisis}:
toda demostración conocida usa de algún modo la completitud de $\R$, y
ninguna es puramente algebraica --- la demostración honesta se da en el
volumen del tercer año, una vez disponibles la integración compleja o
los argumentos de compacidad. Lo que este capítulo demuestra de verdad
es la \emph{reducción}: concedida una raíz para todo \omterm{def:b1:poly:def}{polinomio} no
constante, las factorizaciones completas sobre $\C$ y sobre $\R$ de más
abajo se siguen por puro álgebra.
\end{remark}

\begin{corollary}[Factorización sobre $\C$ y sobre $\R$]\label{cor:b1:poly:factorization}
\begin{enumerate}
  \item Todo $P \in \C[X]$ no nulo se factoriza como
        \[
        P = c\, (X - a_1)^{m_1} \cdots (X - a_r)^{m_r},
        \]
        con $c$ el coeficiente principal, $a_i$ las raíces complejas
        distintas y $\sum m_i = \deg P$: \emph{contadas con
        \omterm{def:b1:poly:derivative}{multiplicidad}, un \omterm{def:b1:poly:def}{polinomio} de grado $n$ tiene exactamente $n$
        raíces complejas}.
  \item Todo $P \in \R[X]$ no nulo se factoriza sobre $\R$ como
        \[
        P = c \prod_i (X - a_i)^{m_i} \prod_j (X^2 + p_j X +
        q_j)^{n_j},
        \]
        siendo los factores cuadráticos distintos y con
        $p_j^2 - 4q_j < 0$ (sin raíces reales).
\end{enumerate}
\end{corollary}

\begin{proof}
(1) Inducción sobre el grado, separando una raíz cada vez mediante
el~\cref{thm:b1:poly:factor}; el recuento de grados cuadra en cada paso.

(2) Sea $P$ de coeficientes reales. Si $z$ es una raíz compleja de
\omterm{def:b1:poly:derivative}{multiplicidad} $m$, también lo es $\conj z$: conjugando $P(z) = 0$ se
obtiene $P(\conj z) = \conj{P(z)} = 0$ (los coeficientes son sus
propios \omterm{def:b1:complex:field}{conjugados}), y lo mismo vale para las derivadas
(\cref{prop:b1:poly:multiplicity}). Agrúpense las raíces no reales en
pares \omterm{def:b1:complex:field}{conjugados}: cada par aporta
\[
(X - z)(X - \conj z) = X^2 - 2\Re(z)\, X + \abs z^2 ,
\]
un \omterm{def:b1:poly:def}{polinomio} real de segundo grado con discriminante negativo. Las
raíces reales aportan los factores lineales.
\end{proof}

\begin{example}\label{ex:b1:poly:factorexample}
$X^4 + 4$ se factorizó sobre $\R$ en el~\cref{exo:b1:complex:5}
emparejando las cuatro raíces complejas $\pm 1 \pm \iu$:
$X^4 + 4 = (X^2 - 2X + 2)(X^2 + 2X + 2)$. Ninguno de los dos factores
se descompone sobre $\R$ (discriminantes $-4$). Obsérvese: un
\omterm{def:b1:poly:def}{polinomio} real \emph{irreducible} tiene grado $1$ o $2$ --- que es
exactamente lo que dice el teorema de factorización. El mismo
emparejamiento de \omterm{def:b1:complex:field}{conjugados} aplicado a $X^4 + 1$, cuyas raíces son
$\eu^{\pm\iu\pi/4}$ y $\eu^{\pm3\iu\pi/4}$: cada par aporta $X^2 -
2\cos\theta\,X + 1$, luego
\[
X^4 + 1 = \bigl(X^2 - \sqrt2\,X + 1\bigr)
\bigl(X^2 + \sqrt2\,X + 1\bigr) ,
\]
identidad invisible para los intentos ingenuos de factorizar sobre $\Q$
--- el precio de insistir en coeficientes reales (aquí incluso
irracionales), y un ingrediente estándar para integrar
$\frac1{x^4 + 1}$ en el~\cref{ch:b1:integration}.
\end{example}

\begin{remark}[Errores frecuentes con polinomios]\label{rem:b1:poly:pitfalls}
\begin{enumerate}
  \item \emph{\omterm{def:b1:poly:def}{Polinomio} frente a función.} Sobre $K = \Q, \R, \C$
        las dos nociones coinciden (funciones iguales tienen
        coeficientes iguales, por el~\cref{cor:b1:poly:nroots} y ser
        $K$ infinito), pero conceptualmente un \omterm{def:b1:poly:def}{polinomio} \emph{es} su
        lista de coeficientes: sobre el \omterm{def:b1:structures:field}{cuerpo} de dos elementos
        $\Z/2\Z$ del~\cref{ch:b1:structures}, $X^2 + X$ se anula en los
        dos puntos y, sin embargo, no es el \omterm{def:b1:poly:def}{polinomio} nulo.
  \item \emph{Grados al sumar.} $\deg(P + Q)$ puede caer por debajo
        de $\max(\deg P, \deg Q)$ cuando los términos principales se
        cancelan; escribir «$\deg(P + Q) = \max(\dots)$» solo es
        seguro con grados distintos.
  \item \emph{Contar bien las raíces.} «$n$ raíces» en
        el~\cref{cor:b1:poly:factorization} significa \emph{con
        \omterm{def:b1:poly:derivative}{multiplicidad} y en $\C$}: $X^2 + 1$ no tiene raíces reales, y
        $(X-1)^2$ tiene una raíz distinta pero dos con
        \omterm{def:b1:poly:derivative}{multiplicidad}. Los \omterm{def:b1:logic:statement}{enunciados} que mezclan los tres recuentos
        son la fuente más frecuente de demostraciones falsas.
  \item \emph{La irreducibilidad depende del \omterm{def:b1:structures:field}{cuerpo}.} $X^2 - 2$ es
        irreducible sobre $\Q$ y se descompone sobre $\R$; $X^2 + 1$
        es irreducible sobre $\R$ y se descompone sobre $\C$. La
        palabra «irreducible» a secas no significa nada mientras no se
        nombre el \omterm{def:b1:structures:field}{cuerpo} de coeficientes.
\end{enumerate}
\end{remark}

\section{Coeficientes y raíces}

\begin{theorem}[Fórmulas de Vieta]\label{thm:b1:poly:vieta}
Sea $P = X^n + c_{n-1} X^{n-1} + \dots + c_0$ \omterm{def:b1:poly:def}{mónico} con raíces
$a_1, \dots, a_n \in \C$ (con \omterm{def:b1:poly:derivative}{multiplicidad}). Entonces\index{fórmulas
de Vieta}
\[
\sum_i a_i = -c_{n-1},
\qquad
\sum_{i < j} a_i a_j = c_{n-2},
\qquad \dots, \qquad
a_1 a_2 \cdots a_n = (-1)^n c_0 ,
\]
siendo la $k$-ésima función simétrica de las raíces $(-1)^k c_{n-k}$.
\end{theorem}

\begin{proof}
Por el~\cref{cor:b1:poly:factorization},
$P = (X - a_1)\cdots(X - a_n)$ (\omterm{def:b1:poly:def}{mónico}, con todas las raíces
enumeradas). Desarrollar el producto por distributividad produce un
término por cada manera de elegir, en cada factor, o bien $X$ o bien el
término de raíz $-a_i$: elegir las raíces en los factores de índices
$i_1 < \dots < i_k$ y $X$ en los $n - k$ restantes aporta
$(-a_{i_1})\cdots(-a_{i_k})\,X^{n-k}$. Agrupando por potencias de $X$:
\[
P = \sum_{k=0}^{n} (-1)^k
\Bigl(\sum_{i_1 < \dots < i_k} a_{i_1}\cdots a_{i_k}\Bigr)
X^{n-k} ,
\]
e identificando con $P = \sum_k c_{n-k}X^{n-k}$ (los coeficientes son
únicos, \cref{def:b1:poly:def}) se obtiene $c_{n-k} = (-1)^k \sigma_k$,
es decir, $\sigma_k = (-1)^kc_{n-k}$, donde $\sigma_k$ denota la
$k$-ésima función simétrica mostrada arriba. Los tres casos mostrados
son $k = 1$, $k = 2$ y $k = n$.
\end{proof}

\begin{example}\label{ex:b1:poly:vieta}
Para la cuadrática $X^2 - sX + p$: suma de raíces $s$, producto $p$ ---
ya usado repetidamente (\cref{exo:b1:complex:8}). Para una cúbica
mónica $X^3 + aX^2 + bX + c$ con raíces $\alpha, \beta, \gamma$:
\[
\alpha + \beta + \gamma = -a,
\quad
\alpha\beta + \beta\gamma + \gamma\alpha = b,
\quad
\alpha\beta\gamma = -c ,
\]
lo que permite calcular cantidades simétricas como
$\alpha^2 + \beta^2 + \gamma^2 = a^2 - 2b$ sin resolver.
\end{example}

\begin{example}[Transformar las raíces sin hallarlas]\label{ex:b1:poly:transformroots}
Sean $\alpha, \beta$ las raíces de $X^2 - 3X + 1$. ¿Qué cuadrática
mónica tiene por raíces $\alpha^2, \beta^2$? Por Vieta,
$\alpha + \beta = 3$ y $\alpha\beta = 1$, luego
\[
\alpha^2 + \beta^2 = (\alpha+\beta)^2 - 2\alpha\beta = 7,
\qquad
\alpha^2\beta^2 = (\alpha\beta)^2 = 1 :
\]
la respuesta es $X^2 - 7X + 1$ --- obtenida sin calcular
$\alpha = \frac{3 + \sqrt5}2$. (Comprobación:
$\alpha^2 = \frac{7 + 3\sqrt5}2$ y, en efecto,
$\alpha^2 + \beta^2 = 7$.) La misma estrategia sirve para los inversos
(transformaciones del tipo $X^2 - \frac ba X + \frac ca$), para los
desplazamientos y para cualquier dato simétrico: Vieta convierte
preguntas sobre raíces \emph{desconocidas} en álgebra sobre
coeficientes \emph{conocidos}. Servirá constantemente cuando las raíces
sean valores propios (\cref{ch:b1:det}).
\end{example}

\begin{example}[Ecuaciones palindrómicas]\label{ex:b1:poly:palindrome}
Resuélvase $X^4 + X^3 - 4X^2 + X + 1 = 0$. Los coeficientes se leen
igual en los dos sentidos, así que $0$ no es raíz y dividir entre $X^2$
no pierde soluciones:
\[
X^2 + X - 4 + \frac1X + \frac1{X^2} = 0 .
\]
Póngase $y = X + \frac1X$: entonces $X^2 + \frac1{X^2} = y^2 - 2$ y la
ecuación se reduce a
\[
y^2 + y - 6 = 0 \iff (y + 3)(y - 2) = 0 .
\]
Desenróllese cada valor mediante $X^2 - yX + 1 = 0$: para $y = 2$,
$X^2 - 2X + 1 = (X - 1)^2$ da la raíz doble $1$; para $y = -3$,
$X^2 + 3X + 1 = 0$ da $X = \frac{-3 \pm \sqrt5}2$. Cuatro raíces con
\omterm{def:b1:poly:derivative}{multiplicidad} para una cuártica, como exige
el~\cref{cor:b1:poly:factorization} --- obtenidas resolviendo dos
cuadráticas. El truco cubre todo \omterm{def:b1:poly:def}{polinomio} \emph{palindrómico}: sus
raíces vienen en pares de inversos $\{x, 1/x\}$ (sustitúyase $X$ por
$1/X$ y quítense denominadores), e $y = X + \frac1X$ es precisamente la
cantidad constante sobre esos pares, lo que reduce el grado a la mitad.
\end{example}

\begin{theorem}[Interpolación de Lagrange]\label{thm:b1:poly:lagrange}
Sean $x_0, \dots, x_n$ puntos distintos de $K$ e
$y_0, \dots, y_n \in K$. Existe exactamente un $P \in K[X]$ de grado
$\leq n$ con $P(x_i) = y_i$ para todo $i$, a saber\index{interpolación de Lagrange}
\[
P = \sum_{i=0}^{n} y_i\, L_i,
\qquad
L_i = \prod_{j \neq i} \frac{X - x_j}{x_i - x_j} .
\]
\end{theorem}

\begin{proof}
Cada $L_i$ tiene grado $n$ y cumple $L_i(x_i) = 1$, $L_i(x_j) = 0$ para
$j \neq i$ (cada factor se anula en el $x_j$ correspondiente). Así
pues, el $P$ mostrado tiene grado $\leq n$ e interpola. Unicidad: dos
\omterm{def:b1:poly:def}{polinomios} interpoladores de grado $\leq n$ coinciden en los $n+1$
puntos $x_i$ y, por tanto, son iguales (\cref{cor:b1:poly:nroots}).
\end{proof}

\begin{remark}[Interludio: los polinomios también son vectores]\label{rem:b1:poly:interlude}
Un cambio de punto de vista que el~\cref{ch:b1:vspaces} hará oficial:
los \omterm{def:b1:poly:def}{polinomios} de grado $\leq n$ forman un espacio en el que la suma y
el escalado se comportan exactamente como coordenadas --- un
\omterm{def:b1:poly:def}{polinomio} \emph{es} su lista de $n + 1$ coeficientes. Tres
\omterm{def:b1:logic:statement}{enunciados} de este capítulo son álgebra lineal encubierta. La
\omterm{thm:b1:poly:lagrange}{interpolación de Lagrange} (\cref{thm:b1:poly:lagrange}) dice que los
datos de evaluación $(P(x_0), \dots, P(x_n))$ determinan $P$ de manera
única: evaluar en $n + 1$ puntos es una \omterm{def:b1:logic:inj}{biyección} lineal, y los $L_i$
son la base adaptada a ella. El desarrollo $R = \sum c_k(X - a)^k$ de
la demostración de la~\cref{prop:b1:poly:multiplicity} dice que las
potencias de $(X - a)$ forman otro sistema de coordenadas, con
$c_k = R^{(k)}(a)/k!$ como coordenadas. Y
el~\cref{cor:b1:poly:nroots} --- más raíces que el grado obliga al
\omterm{def:b1:poly:def}{polinomio} nulo --- es el motor de todas las unicidades: se convertirá
en «una \omterm{def:b1:logic:map}{aplicación} lineal \omterm{def:b1:logic:inj}{inyectiva} sobre un espacio de dimensión
$n + 1$» en el~\cref{ch:b1:findim}. Cuando lleguen esos capítulos, el
espacio $K_n[X]$ será su ejemplo favorito; vale la pena llegar allí ya
familiarizado con él.
\end{remark}

\begin{remark}[Dónde se usa este capítulo]\label{rem:b1:poly:whereused}
La factorización sobre $\R$ y sobre $\C$
(\cref{cor:b1:poly:factorization}) es el motor de las fracciones
simples del~\cref{ch:b1:fractions} y, por tanto, de una amplia clase de
integrales del~\cref{ch:b1:integration}. El desarrollo de un
\omterm{def:b1:poly:def}{polinomio} en potencias de $(X - a)$, encontrado en la demostración de
la~\cref{prop:b1:poly:multiplicity}, es la sombra algebraica de las
fórmulas de Taylor del~\cref{ch:b1:taylor}. Los \omterm{def:b1:diffeq:linear2}{polinomios
característicos} ya aparecieron para las ecuaciones diferenciales
(\cref{ch:b1:diffeq}) y vuelven para las matrices en
el~\cref{ch:b1:det}; la \omterm{thm:b1:poly:lagrange}{interpolación de Lagrange} es el primer teorema
de existencia y unicidad del análisis numérico, y los
\omterm{pb:b1:poly:1}{polinomios de Chebyshev} del~\cref{exo:b1:poly:10} --- cuya optimalidad
establece el problema del fin de semana --- le dicen a esa disciplina
\emph{dónde} interpolar. Por último, toda la aritmética de $K[X]$,
copiada del~\cref{ch:b1:arith}, alimenta el estudio de los ideales de
$K[X]$ y de los \omterm{def:b1:structures:ring}{anillos} cociente en el volumen del segundo año.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:poly:1}
Háganse las divisiones euclídeas: $X^5 - 1$ entre $X^2 + X + 1$; y
después $2X^4 + X^3 - X + 3$ entre $X^2 - 2$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:poly:2}
¿Para qué $n \in \N$ \omterm{def:b1:arith:divides}{divide} $X^2 + X + 1$ a $X^{2n} + X^n + 1$?
\emph{Indicación: las raíces de $X^2 + X + 1$ son $j$ y $j^2$ con
$j = \eu^{2\iu\pi/3}$; discútase $n$ \omterm{def:b1:complex:field}{módulo} $3$.}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:poly:3}
Determínense los reales $a, b$ para que $(X-1)^2$ divida a
$P = X^4 + aX^3 + bX^2 + 1$, y factorícese después $P$ sobre $\R$ para esos valores.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:poly:4}
Factorícense sobre $\C$ y sobre $\R$: $X^3 - 1$; $\;X^4 + X^2 + 1$;
$\;X^6 - 1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:5}
Sea $P = X^3 - 6X^2 + 11X - 6$.
\begin{enumerate}
  \item Hállense las raíces racionales \emph{(una raíz racional $p/q$
        irreducible de un \omterm{def:b1:poly:def}{polinomio} entero \omterm{def:b1:poly:def}{mónico} es un entero que
        \omterm{def:b1:arith:divides}{divide} al término constante --- demuéstrese)}, y factorícese $P$.
  \item Sin resolver, calcúlense la suma de los cuadrados y la suma de
        los inversos de las raíces mediante Vieta, y compruébese con
        la factorización.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:6}
Calcúlese $\gcd(X^4 - 1,\; X^3 - X^2 + X - 1)$ con el \omterm{met:b1:arith:euclid}{algoritmo de
Euclides}, y escríbase como combinación $AU + BV$ de los dos
\omterm{def:b1:poly:def}{polinomios}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:7}
Sea $P \in \R[X]$ con $P(x) \geq 0$ para todo $x \in \R$. Demuéstrese
que $P$ es suma de dos cuadrados de \omterm{def:b1:poly:def}{polinomios} reales:
$P = A^2 + B^2$. \emph{Indicación: en la factorización real, las raíces
reales tienen \omterm{def:b1:poly:derivative}{multiplicidad} par; escríbanse los factores cuadráticos
como $(X - z)(X - \conj z)$ y úsese $\abs{\,\cdot\,}^2 = (\Re)^2 +
(\Im)^2$ sobre el producto de los $(X - z)$.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:8}
Hállese el \omterm{def:b1:poly:def}{polinomio} $P$ de grado $\leq 2$ con $P(0) = 1$, $P(1) = 3$,
$P(2) = 2$, primero con la fórmula de Lagrange y después resolviendo el
sistema lineal sobre los coeficientes. Compruébese que ambas coinciden.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:9}
Demuéstrese que $P = X^{2n+1} - 1$ tiene exactamente una raíz real, y
que, para todo $n \geq 1$, el \omterm{def:b1:poly:def}{polinomio} $1 + X + \frac{X^2}{2!} +
\dots + \frac{X^n}{n!}$ no tiene raíces múltiples \emph{(compárense $P$
y $P'$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:poly:10}
(\omterm{pb:b1:poly:1}{Polinomios de Chebyshev}) Defínanse $T_0 = 1$, $T_1 = X$ y
$T_{n+1} = 2X\,T_n - T_{n-1}$.
\begin{enumerate}
  \item Demuéstrese por inducción que $T_n(\cos\theta) = \cos n\theta$
        para todo $\theta$.
  \item Dedúzcanse las $n$ raíces de $T_n$ y su coeficiente principal.
  \item Demuéstrese que $\sup_{x \in \intcc{-1}{1}} \abs{T_n(x)} = 1$,
        alcanzado en $n + 1$ puntos de $\intcc{-1}{1}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:poly:11}
Sea $P \in \C[X]$ no constante, con raíces distintas $a_1, \dots, a_r$
(de \omterm{def:b1:poly:derivative}{multiplicidades} $m_1, \dots, m_r$). Demuéstrese la identidad de
funciones racionales
\[
\frac{P'(X)}{P(X)} = \sum_{i=1}^{r} \frac{m_i}{X - a_i},
\]
y dedúzcase el teorema de Gauss--Lucas: toda raíz de $P'$ está en la
envolvente convexa de las raíces de $P$ \emph{(evalúese la identidad en
una raíz $w$ de $P'$ que no sea raíz de $P$, tómense \omterm{def:b1:complex:field}{conjugados} y léase
el resultado como que $w$ es una media ponderada de los $a_i$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:poly:12}
(Filtro de \omterm{def:b1:complex:unity}{raíces de la unidad}) Sean $n \in \N^*$ y
$j = \eu^{2\iu\pi/3}$. Evaluando $(1 + X)^n$ en $1$, $j$ y $j^2$,
demuéstrese que
\[
\sum_{k \geq 0} \binom{n}{3k}
= \frac{2^n + 2\cos\frac{n\pi}{3}}{3} ,
\]
y compruébese la fórmula para $n = 3$ y $n = 6$. \emph{Indicación:
$1 + j^m + j^{2m}$ vale $3$ si $3 \mid m$ y $0$ en caso contrario; y
$1 + j = \eu^{\iu\pi/3}$.}
\end{exercise}

\section{Problema: los polinomios de Chebyshev y el polinomio más plano}

\begin{problem}\label{pb:b1:poly:1}
Entre todos los \omterm{def:b1:poly:def}{polinomios} \emph{\omterm{def:b1:poly:def}{mónicos}} de grado $n$, ¿cuál se
mantiene más cerca de cero en $\intcc{-1}1$? La respuesta --- el
teorema de Chebyshev, partida de nacimiento de la teoría de la
aproximación --- es $2^{1-n}T_n$, donde $T_n$ es el \omterm{def:b1:poly:def}{polinomio} de
Chebyshev del~\cref{exo:b1:poly:10}, y ningún competidor \omterm{def:b1:poly:def}{mónico} puede
mejorar su desviación $2^{1-n}$.\index{polinomios de Chebyshev} Este
problema desarrolla el álgebra de la familia $(T_n)$ (\omterm{def:b1:structures:law}{ley de
composición}, coeficientes explícitos, la familia de segunda especie
$U_n$, una ecuación diferencial), demuestra el teorema de extremalidad
con su caso de igualdad y recoge aplicaciones: nodos óptimos de
interpolación, el valor exacto de $\cos 36^\circ$ y una \omterm{def:b1:arith:congruence}{congruencia}
$T_p \equiv X^p \pmod p$. En todo el problema, $T_0 = 1$, $T_1 = X$,
$T_{n+1} = 2X\,T_n - T_{n-1}$, y usamos libremente
$T_n(\cos\theta) = \cos n\theta$ del~\cref{exo:b1:poly:10}.

\medskip
\noindent\textbf{Parte I --- La familia $(T_n)$.}
\begin{enumerate}
  \item Calcúlense $T_2, T_3, T_4, T_5$ con la recurrencia. (Compárese
        $T_3$ con la identidad $\cos3\theta = 4\cos^3\theta -
        3\cos\theta$ del~\cref{ex:b1:complex:cos3}.)
  \item Demuéstrese por inducción que $\deg T_n = n$ con coeficiente
        principal $2^{n-1}$ para $n \geq 1$, y que $T_n$ tiene la
        paridad de $n$ (solo aparecen potencias pares, o solo impares).
  \item Demuéstrese el principio de unicidad: $T_n$ es el \emph{único}
        \omterm{def:b1:poly:def}{polinomio} que cumple $P(\cos\theta) = \cos n\theta$ para todo
        $\theta$. (Dos \omterm{def:b1:poly:def}{polinomios} que coinciden en $\intcc{-1}1$
        coinciden en todas partes: \cref{cor:b1:poly:nroots}.)
  \item Dedúzcanse las \omterm{def:b1:structures:law}{leyes de composición} y de producto:
        \[
        T_m \circ T_n = T_{mn},
        \qquad
        2\,T_m T_n = T_{m+n} + T_{\abs{m-n}} .
        \]
  \item Recuérdense del~\cref{exo:b1:poly:10} las raíces
        $x_k = \cos\frac{(2k+1)\pi}{2n}$ y los puntos de equioscilación
        $y_k = \cos\frac{k\pi}n$ con $T_n(y_k) = (-1)^k$. Escríbase la
        factorización completa de $T_n$ sobre $\R$ y justifíquese que
        los $y_k$ se entrelazan: $y_n < x_{n-1} < y_{n-1} < \dots <
        x_0 < y_0$.
  \item Demuéstrese que $T_n(\cosh t) = \cosh(nt)$ para todo
        $t \in \R$ (misma inducción, usando
        la~\cref{prop:b1:functions:hyprules}) y dedúzcase, para
        $x \geq 1$, la forma cerrada
        \[
        T_n(x) = \frac{\bigl(x + \sqrt{x^2 - 1}\bigr)^n +
        \bigl(x - \sqrt{x^2 - 1}\bigr)^n}{2} ,
        \]
        de modo que $T_n(x) > 1$ para $x > 1$: fuera de $\intcc{-1}1$
        el \omterm{def:b1:poly:def}{polinomio} se dispara de inmediato.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Coeficientes, la familia $U_n$, una
ecuación diferencial.}
\begin{enumerate}[resume]
  \item A partir de la fórmula de De Moivre
        (\cref{cor:b1:complex:demoivre}), demuéstrese la expresión
        explícita
        \[
        T_n(x) = \sum_{0 \leq 2j \leq n} \binom{n}{2j}\,
        x^{\,n-2j}\,(x^2 - 1)^j ,
        \]
        y compruébese para $n = 3$.
  \item Calcúlense $T_n(1)$, $T_n(-1)$ y $T_n(0)$ para todo $n$.
  \item Defínase $U_n$ (\emph{de segunda especie}) por $U_0 = 1$,
        $U_1 = 2X$, $U_{n+1} = 2X\,U_n - U_{n-1}$. Demuéstrese que
        $U_n(\cos\theta) = \frac{\sin(n+1)\theta}{\sin\theta}$ para
        $\theta \notin \pi\Z$, y que $T_n' = n\,U_{n-1}$ para
        $n \geq 1$.
  \item Demuéstrese que $\abs{\sin n\theta} \leq n\,\abs{\sin\theta}$
        para todo $\theta$ (inducción) y dedúzcase la cota de tipo
        Markov
        \[
        \abs{T_n'(x)} \leq n^2
        \quad\text{on } \intcc{-1}1,
        \qquad\text{with } T_n'(\pm1) = (\pm1)^{n-1}\,n^2 .
        \]
  \item Véase que $y = T_n$ cumple la ecuación diferencial
        \[
        (1 - x^2)\,y'' - x\,y' + n^2\,y = 0 ,
        \]
        derivando la identidad
        $\sin\theta\, T_n'(\cos\theta) = n\sin n\theta$ respecto de
        $\theta$; compruébese
        directamente para $T_2$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El teorema de extremalidad de
Chebyshev.} Sea $\widetilde T_n = 2^{1-n}\,T_n$ (\omterm{def:b1:poly:def}{mónico} por la
pregunta 2) y escríbase $\norm{P}_\infty = \sup_{x \in \intcc{-1}1}\abs{P(x)}$.
\begin{enumerate}[resume]
  \item Justifíquese que $\norm{\widetilde T_n}_\infty = 2^{1-n}$,
        alcanzado con signos alternados en los $n + 1$ puntos
        $y_n < \dots < y_0$.
  \item Supóngase que algún $P$ \omterm{def:b1:poly:def}{mónico} de grado $n$ tuviese
        $\norm P_\infty < 2^{1-n}$, y póngase $D = \widetilde T_n - P$.
        Véase que $\deg D \leq n - 1$ y que $D(y_k)$ tiene el signo
        estricto de $(-1)^k$ para cada $k = 0, \dots, n$.
  \item Dedúzcase que $D$ tiene al menos $n$ raíces reales distintas
        (una en cada hueco, por la propiedad del valor intermedio,
        usada aquí al nivel del volumen anterior y demostrada en
        el~\cref{ch:b1:continuity}) y conclúyase el \emph{teorema de
        Chebyshev}: todo $P$ \omterm{def:b1:poly:def}{mónico} de grado $n$ cumple
        \[
        \norm{P}_\infty \geq 2^{1-n} .
        \]
  \item (Caso de igualdad, primer paso) Supóngase ahora
        $\norm P_\infty = 2^{1-n}$ exactamente, con $P$ \omterm{def:b1:poly:def}{mónico} de
        grado $n$, y sea $D = \widetilde T_n - P$. Véase que
        $(-1)^kD(y_k) \geq 0$ para todo $k$, y que si $D(y_k) = 0$ en
        un punto \emph{interior} $y_k$ ($0 < k < n$), entonces también
        $D'(y_k) = 0$. \emph{(En un $y_k$ interior, tanto
        $\widetilde T_n$ como $P$ alcanzan un extremo de valor
        absoluto $\norm{\cdot}_\infty$; una función derivable tiene
        derivada nula en un extremo interior --- usado aquí al nivel
        del volumen anterior y demostrado en
        el~\cref{ch:b1:derivative}.)}
  \item (Caso de igualdad, conclusión) Cuéntense las raíces de $D$ con
        \omterm{def:b1:poly:derivative}{multiplicidad} para ver que $D = 0$: el minimizador es
        \emph{único}, $P = \widetilde T_n$.
  \item Transpórtese a un segmento arbitrario $\intcc ab$: véase que
        la norma del supremo mínima de un \omterm{def:b1:poly:def}{polinomio mónico} de grado
        $n$ en $\intcc ab$ es $2\bigl(\frac{b-a}4\bigr)^n$, alcanzada
        por un \omterm{def:b1:poly:def}{polinomio} de Chebyshev reescalado. \emph{(Sustitúyase
        $x = \frac{a+b}2 + \frac{b-a}2\,t$ y síganse los coeficientes
        principales.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Aplicaciones.}
\begin{enumerate}[resume]
  \item Hágase a mano el caso $n = 3$: localícense los extremos de
        $\widetilde T_3 = X^3 - \frac34X$ en $\intcc{-1}1$,
        compruébese la equioscilación cuádruple con valor $\frac14$ y
        conclúyase que ninguna cúbica mónica lo hace mejor.
  \item (Nodos óptimos de interpolación) Para $n + 1$ nodos
        $x_0, \dots, x_n \in \intcc{-1}1$, el error de interpolación
        está gobernado por $\omega(X) = \prod_i (X - x_i)$ (como
        cuantificará el~\cref{ch:b1:taylor}). Demuéstrese que la
        elección que minimiza $\norm\omega_\infty$ es el \omterm{def:b1:logic:sets}{conjunto} de
        las $n + 1$ raíces de $T_{n+1}$, con $\norm\omega_\infty =
        2^{-n}$: los nodos de Chebyshev son los lugares adecuados para
        interpolar.
  \item Usando $T_5$, demuéstrese que $c = \cos 36^\circ$ cumple
        $16c^5 - 20c^3 + 5c + 1 = 0$, factorícese este \omterm{def:b1:poly:def}{polinomio} como
        $(x + 1)(4x^2 - 2x - 1)^2$ y conclúyase
        \[
        \cos 36^\circ = \frac{1 + \sqrt5}4 .
        \]
        Compruébese la coherencia con $\cos 72^\circ =
        \frac{\sqrt5 - 1}4$ del~\cref{exo:b1:complex:8}.
  \item Estímese $T_{10}(1.1)$ con la forma cerrada de la pregunta 6
        (bastan dos cifras significativas) e interprétese: un
        \omterm{def:b1:poly:def}{polinomio} acotado por $1$ en $\intcc{-1}1$ puede superar ya
        $40$ en $x = 1.1$. (Que $T_n$ crezca lo \emph{más deprisa}
        entre esos \omterm{def:b1:poly:def}{polinomios} es otra propiedad extremal de la
        familia, más allá de este problema.)
  \item Demuéstrese la \omterm{def:b1:arith:congruence}{congruencia}: para todo \omterm{def:b1:arith:prime}{primo} impar $p$,
        todos los coeficientes de $T_p - X^p$ son divisibles por $p$.
        \emph{(Úsese la pregunta 7 y $p \mid \binom p{2j}$ para
        $0 < 2j < p$, de la demostración
        del~\cref{thm:b1:arith:fermat}.)} Compruébese en $T_3$ y $T_5$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Síntesis.}
\begin{enumerate}[resume]
  \item Calcúlese explícitamente la cuadrática mónica de norma del
        supremo mínima en $\intcc01$ y su desviación. (Pregunta 17 con
        $n = 2$.)
  \item ¿Dónde ha usado exactamente el problema: (i) la rigidez de los
        \omterm{def:b1:poly:def}{polinomios} (\cref{cor:b1:poly:nroots}); (ii) la trigonometría
        de los \cref{ch:b1:complex} y \cref{ch:b1:functions}; (iii) la
        aritmética de los \omterm{def:b1:counting:objects}{coeficientes binomiales}
        del~\cref{ch:b1:arith}? Una frase para cada uno.
  \item Síntesis, en un párrafo breve: el teorema dice que el
        \omterm{def:b1:poly:def}{polinomio mónico} más plano es el que \emph{equioscila}, y la
        demostración convierte la optimalidad en un recuento de
        raíces. Coméntese este mecanismo, el papel de la sustitución
        $x = \cos\theta$ como puente entre álgebra y trigonometría, y
        nómbrense los dos lugares donde el problema ha necesitado
        hechos de análisis (valor intermedio, extremo interior) que
        demuestran capítulos posteriores.
\end{enumerate}
\end{problem}
