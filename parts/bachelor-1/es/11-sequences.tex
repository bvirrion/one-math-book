\chapter{Sucesiones}\label{ch:b1:seq}

Las sucesiones se manipularon en el volumen anterior con el concepto de
límite tomado a medias como acto de fe. Aquí la teoría se reconstruye
sobre la completitud de $\R$ (\cref{ch:b1:reals}): todo teorema clásico
--- convergencia monótona, \omterm{thm:b1:seq:adjacent}{sucesiones adyacentes},
Bolzano--Weierstrass, criterio de Cauchy --- es una cara de ese único
axioma. El capítulo termina con el estudio práctico de las sucesiones
definidas por $u_{n+1} = f(u_n)$.

\section{Convergencia}

\begin{definition}[Límite de una sucesión]\label{def:b1:seq:limit}
Una sucesión $(u_n)$ de reales \emph{converge} a $\ell \in \R$
cuando\index{límite!de una sucesión}
\[
\forall \varepsilon > 0,\ \exists N \in \N,\ \forall n \geq N,
\qquad \abs{u_n - \ell} \leq \varepsilon .
\]
Se escribe $u_n \to \ell$ o $\lim u_n = \ell$. Una sucesión que no
converge (a ningún real) \emph{diverge}. Divergencia \emph{a
$+\infty$}: $\forall M,\ \exists N,\ \forall n \geq N,\ u_n \geq M$
(análogamente, $-\infty$).
\end{definition}

\begin{example}[Una demostración $\varepsilon$--$N$, escrita una vez]\label{ex:b1:seq:epsilonN}
Afirmación: $u_n = \dfrac{n^2 + 1}{2n^2 - 3} \to \dfrac12$. Aíslese
primero el error:
\[
\Bigl| u_n - \frac12 \Bigr|
= \Bigl| \frac{2(n^2 + 1) - (2n^2 - 3)}{2(2n^2 - 3)} \Bigr|
= \frac{5}{2\,\abs{2n^2 - 3}}
= \frac{5}{2\,(2n^2 - 3)} \quad (n \geq 2).
\]
Domínese después con algo simple: para $n \geq 2$, $2n^2 - 3 \geq n^2$,
luego el error es $\leq \frac{5}{2n^2} \leq \frac 5{2n}$. Dado
$\varepsilon > 0$, la \omterm{thm:b1:reals:archimedes}{propiedad arquimediana} proporciona
$N \geq \max\bigl(2, \frac{5}{2\varepsilon}\bigr)$; para $n \geq N$ el
error es $\leq \varepsilon$. Hecho. La idea de cierre: una
demostración $\varepsilon$--$N$ tiene exactamente tres movimientos ---
calcular el error, acotarlo por una expresión elemental decreciente,
despejar el umbral --- y, tras los teoremas de este capítulo
(operaciones, emparedado), casi nunca se vuelve a escribir una
demostración así: los teoremas empaquetan los tres movimientos de una
vez por todas.
\end{example}

\begin{example}[Divergencia a infinito, certificada]\label{ex:b1:seq:toinfinity}
Afirmación: $u_n = n^2 - 100n \to +\infty$. Factorícese el término
dominante: $u_n = n^2\bigl(1 - \frac{100}{n}\bigr) \geq \frac{n^2}{2}$
para $n \geq 200$. Dado $M$, tómese $N = \max\bigl(200,
\lceil\sqrt{2M}\rceil\bigr)$: para $n \geq N$,
$u_n \geq \frac{n^2}{2} \geq M$. Se ven dos hábitos: factorizar el
término dominante convierte una competencia ($n^2$ contra $-100n$) en
una sola escala por un factor que tiende a $1$; y el umbral puede ser
enorme ($u_{100} = 0$, y la sucesión es incluso negativa antes de
$n = 100$) --- la divergencia a $+\infty$ es un \omterm{def:b1:logic:statement}{enunciado} sobre la
cola, indiferente a cualquier mal comportamiento finito.
\end{example}

\begin{proposition}[Primeras propiedades]\label{prop:b1:seq:first}
\begin{enumerate}
  \item El límite, si existe, es único.
  \item Una sucesión convergente está acotada.
  \item Si $u_n \to \ell$, modificar un número finito de términos no
        cambia ni la convergencia ni el límite.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Si $u_n \to \ell$ y $u_n \to \ell'$ con $\ell \neq \ell'$, tómese
$\varepsilon = \frac{\abs{\ell - \ell'}}{3}$: más allá de los dos
umbrales, $\abs{\ell - \ell'} \leq \abs{\ell - u_n} + \abs{u_n -
\ell'} \leq 2\varepsilon = \frac23 \abs{\ell - \ell'}$, absurdo.

(2) Con $\varepsilon = 1$: más allá de $N$,
$\abs{u_n} \leq \abs\ell + 1$; y los finitos términos anteriores
también están acotados, luego $\abs{u_n} \leq \max(\abs{u_0}, \dots,
\abs{u_{N-1}}, \abs\ell + 1)$.

(3) Con detalle: supóngase $v_n = u_n$ para $n \geq n_0$ y
$u_n \to \ell$. Dado $\varepsilon > 0$, tómese el umbral $N$ de
$(u_n)$: para $n \geq \max(N, n_0)$, $\abs{v_n - \ell} = \abs{u_n -
\ell} \leq \varepsilon$. Luego $v_n \to \ell$: la definición
cuantifica solo sobre $n \geq N$, y cualquier prefijo finito se puede
reescribir al precio de agrandar el umbral. (Por eso bastan en todo
este capítulo las hipótesis del tipo «para todo $n$ grande».)
\end{proof}

\begin{theorem}[Operaciones con límites]\label{thm:b1:seq:operations}
Si $u_n \to \ell$ y $v_n \to m$, entonces
\[
u_n + v_n \to \ell + m, \qquad u_n v_n \to \ell m, \qquad
\frac{u_n}{v_n} \to \frac{\ell}{m} \ (\text{if } m \neq 0),
\qquad \abs{u_n} \to \abs\ell .
\]
\end{theorem}

\begin{proof}
\emph{Suma:} $\abs{(u_n + v_n) - (\ell + m)} \leq \abs{u_n - \ell} +
\abs{v_n - m} \leq 2\varepsilon$ más allá del mayor de los umbrales.
\emph{Producto:} escríbase
\[
u_n v_n - \ell m = (u_n - \ell)\,v_n + \ell\,(v_n - m);
\]
$(v_n)$ está acotada por cierto $B$ (\cref{prop:b1:seq:first}), luego
el miembro derecho es $\leq B\abs{u_n - \ell} + \abs{\ell}\,\abs{v_n -
m}$, arbitrariamente pequeño. \emph{Cociente:} basta tratar
$\frac 1{v_n}$. Con $\varepsilon = \frac{\abs m}{2}$: más allá de
cierto $N_0$, $\abs{v_n} \geq \frac{\abs m}{2}$, luego
\[
\Bigl| \frac{1}{v_n} - \frac 1m \Bigr|
= \frac{\abs{m - v_n}}{\abs{v_n m}}
\leq \frac{2}{m^2}\,\abs{v_n - m} \longrightarrow 0 .
\]
\emph{Valor absoluto:} $\bigl|\abs{u_n} - \abs\ell\bigr| \leq
\abs{u_n - \ell}$ (desigualdad triangular inversa,
\cref{prop:b1:complex:rules}).
\end{proof}

\begin{example}[Operaciones más un truco algebraico]\label{ex:b1:seq:conjugatetrick}
Calcúlese $\lim\,\bigl(\sqrt{n^2 + n} - n\bigr)$. Las dos piezas por
separado tienden a $+\infty$: el teorema de las operaciones no dice
nada de su diferencia (una \emph{forma indeterminada}). Multiplíquese
por el \omterm{def:b1:complex:field}{conjugado}:
\[
\sqrt{n^2 + n} - n
= \frac{(n^2 + n) - n^2}{\sqrt{n^2 + n} + n}
= \frac{n}{\sqrt{n^2+n} + n}
= \frac{1}{\sqrt{1 + \frac1n} + 1} .
\]
Ahora todo converge: $\sqrt{1 + \frac1n} \to 1$, porque
$0 \leq \sqrt{1 + h} - 1 = \frac{h}{\sqrt{1+h} + 1} \leq h$
(\omterm{def:b1:complex:field}{conjugado} otra vez y emparedado con $h = \frac1n$); y entonces el
teorema de las operaciones da el límite $\frac{1}{1 + 1} = \frac12$.
La idea de cierre: el teorema de las operaciones no es una calculadora
para todos los límites --- las formas indeterminadas
($\infty - \infty$, $\frac00$, $0 \times \infty$, $1^\infty$) hay que
\emph{transformarlas} antes con álgebra (\omterm{def:b1:complex:field}{conjugados}, factorizar el
término dominante) hasta que cada pieza converja; la máquina
sistemática para los casos resistentes es el desarrollo asintótico
del~\cref{ch:b1:taylor}.
\end{example}

\begin{theorem}[Límites y orden]\label{thm:b1:seq:order}
\begin{enumerate}
  \item Si $u_n \leq v_n$ para todo $n$ grande y las dos convergen,
        entonces $\lim u_n \leq \lim v_n$. (Las desigualdades
        estrictas \emph{no} pasan al límite: $\frac 1n > 0$ pero
        $\lim = 0$.)
  \item (Teorema del emparedado\index{teorema del emparedado}) Si
        $u_n \leq w_n \leq v_n$ para todo $n$ grande y $u_n, v_n \to
        \ell$, entonces $w_n \to \ell$.
  \item Si $u_n \to \ell > 0$, entonces $u_n > \frac\ell2 > 0$ para
        todo $n$ grande.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Supóngase $\ell = \lim u_n > m = \lim v_n$; con
$\varepsilon = \frac{\ell - m}{3}$, los términos grandes cumplen
$v_n \leq m + \varepsilon < \ell - \varepsilon \leq u_n$, en contra de
$u_n \leq v_n$.

(2) Más allá de los umbrales:
$\ell - \varepsilon \leq u_n \leq w_n \leq v_n \leq \ell + \varepsilon$.

(3) es la~\cref{def:b1:seq:limit} con $\varepsilon = \frac\ell2$.
\end{proof}

\begin{example}[Dos emparedados]\label{ex:b1:seq:squeezes}
(i) $\dfrac{\sin n}{n} \to 0$: de $-\frac1n \leq \frac{\sin n}{n} \leq
\frac1n$, con las dos paredes cerrándose sobre $0$ --- sin necesidad
de entender en absoluto el errático numerador. (ii)
$(2^n + 3^n)^{1/n} \to 3$: encuádrese el interior,
\[
3^n \leq 2^n + 3^n \leq 2\cdot3^n
\quad\Longrightarrow\quad
3 \leq (2^n + 3^n)^{1/n} \leq 3\cdot 2^{1/n} ,
\]
y $2^{1/n} = \eu^{\frac{\ln 2}{n}} \to 1$ (como para $5^{1/n}$ en
el~\cref{exo:b1:seq:2}): el emparedado entrega $3$. La idea de cierre:
una suma de exponenciales en competencia se comporta como su término
\emph{mayor} --- las menores quedan absorbidas por un factor constante
inofensivo, que la raíz $n$-ésima borra después.
\end{example}

\section{Sucesiones monótonas}

\begin{theorem}[Teorema del límite monótono]\label{thm:b1:seq:monotone}
Una sucesión creciente y acotada superiormente converge, a
$\sup\{u_n : n \in \N\}$; una sucesión creciente y no acotada
superiormente diverge a $+\infty$. (\omterm{def:b1:logic:statement}{Enunciados} espejo para las
sucesiones decrecientes.)\index{teorema del límite monótono}
\end{theorem}

\begin{proof}
Sea $s = \sup\{u_n\}$ (\cref{thm:b1:reals:sup}). Dado
$\varepsilon > 0$, la caracterización con $\varepsilon$
(\cref{prop:b1:reals:epsilon}) da un $N$ con $u_N > s - \varepsilon$;
por monotonía, $s - \varepsilon < u_N \leq u_n \leq s$ para todo
$n \geq N$: convergencia a $s$. Si no está acotada: para todo $M$ hay
algún $u_N > M$, y la monotonía mantiene por encima de $M$ todos los
términos posteriores.
\end{proof}

\begin{example}[El teorema monótono como máquina de existencia]\label{ex:b1:seq:existencemachine}
Sea $u_n = \prod_{k=1}^{n} \bigl(1 + \frac{1}{2^k}\bigr)$. Cada factor
supera a $1$, luego $(u_n)$ es creciente. ¿Acotada superiormente?
Tómense logaritmos y úsese $\ln(1 + x) \leq x$
(el~\cref{ex:b1:derivative:convexineq} lo anticipa; o la burda
$1 + x \leq \eu^x$ del volumen anterior):
\[
\ln u_n = \sum_{k=1}^{n} \ln\Bigl(1 + \frac{1}{2^k}\Bigr)
\leq \sum_{k=1}^{n} \frac{1}{2^k} < 1 ,
\]
luego $u_n < \eu$. Creciente y acotada: $(u_n)$ converge a cierto
$\ell \in \intoc{u_1}{\eu}$ --- un número real perfectamente bien
definido y sin ninguna forma cerrada a la vista ($\ell = 2.384\dots$).
La idea de cierre: el teorema del límite monótono es la máquina de
existencia más barata del análisis; bautizó al propio $\eu$
(el~\cref{ex:b1:seq:e} de más abajo) y, en el~\cref{ch:b1:series},
decidirá la convergencia de toda serie de términos positivos por mera
acotación.
\end{example}

\begin{theorem}[Sucesiones adyacentes]\label{thm:b1:seq:adjacent}
Sean $(a_n)$ creciente y $(b_n)$ decreciente, con $b_n - a_n \to 0$.
Entonces las dos convergen a un límite \emph{común} $\ell$, y
$a_n \leq \ell \leq b_n$ para todo $n$.\index{sucesiones adyacentes}
\end{theorem}

\begin{proof}
Primero, $a_n \leq b_n$ para todo $n$: la sucesión $(b_n - a_n)$ es
decreciente y tiende a $0$, luego es $\geq 0$ (un término negativo la
dejaría congelada por debajo de $0$). Después, $(a_n)$ es creciente y
está acotada superiormente por $b_0$: converge a cierto $\ell$
(\cref{thm:b1:seq:monotone}); igualmente $(b_n) \to \ell'$; y
$\ell' - \ell = \lim (b_n - a_n) = 0$. Las desigualdades
$a_n \leq \ell \leq b_n$ se siguen de la monotonía
($\ell = \sup a_k \geq a_n$, etc.).
\end{proof}

\begin{example}[El número $\eu$]\label{ex:b1:seq:e}
Póngase $a_n = \sum_{k=0}^{n} \frac{1}{k!}$ y
$b_n = a_n + \frac{1}{n \cdot n!}$ ($n \geq 1$). Entonces $(a_n)$ crece; y
\[
b_{n+1} - b_n = \frac{1}{(n+1)!} + \frac{1}{(n+1)(n+1)!} -
\frac{1}{n\,n!}
= \frac{n(n+1) + n - (n+1)^2}{n(n+1)(n+1)!}
= \frac{-1}{n(n+1)(n+1)!} < 0 ,
\]
luego $(b_n)$ decrece, y $b_n - a_n \to 0$: son adyacentes. Su límite
común es (aquí por definición) el número $\eu \approx 2.71828$; las
desigualdades $a_n < \eu < b_n$ son lo bastante finas para demostrar
que $\eu \notin \Q$ (\cref{exo:b1:seq:9}).
\end{example}

\section{Subsucesiones y Bolzano--Weierstrass}

\begin{definition}[Subsucesión]\label{def:b1:seq:subsequence}
Una \emph{subsucesión}\index{subsucesión} de $(u_n)$ es una sucesión
$(u_{\varphi(n)})$ donde $\varphi \colon \N \to \N$ es estrictamente
creciente (obsérvese que $\varphi(n) \geq n$, por inducción).
\end{definition}

\begin{proposition}\label{prop:b1:seq:subsequences}
Si $u_n \to \ell$ ($\ell \in \R$ o $\pm\infty$), toda \omterm{def:b1:seq:subsequence}{subsucesión}
tiende a $\ell$. En consecuencia, una sucesión con dos subsucesiones
de límites distintos diverge. Recíprocamente, si $(u_{2n})$ y
$(u_{2n+1})$ convergen las dos al \emph{mismo} $\ell$, entonces
$u_n \to \ell$.
\end{proposition}

\begin{proof}
Más allá del umbral $N$ de $(u_n)$, todos los índices
$\varphi(n) \geq n \geq N$ sirven (la desigualdad $\varphi(n) \geq n$
es la inducción señalada en la~\cref{def:b1:seq:subsequence}:
$\varphi(0) \geq 0$, y $\varphi(n+1) > \varphi(n) \geq n$ obliga a
$\varphi(n+1) \geq n + 1$). Para el recíproco: dado $\varepsilon$,
tómense los dos umbrales $N_0$ (pares) y $N_1$ (impares); un índice
cualquiera $n \geq \max(2N_0, 2N_1 + 1)$ es o bien par, $n = 2k$ con
$k \geq N_0$, o bien impar, $n = 2k+1$ con $k \geq N_1$ --- y en los
dos casos $\abs{u_n - \ell} \leq \varepsilon$: todo índice queda
cubierto por una de las dos subsucesiones, y eso es todo lo que
hace falta.
\end{proof}

\begin{example}[Límites de subsucesiones]\label{ex:b1:seq:sublimits}
Para $u_n = (-1)^n \frac{n}{n+1}$: la \omterm{def:b1:seq:subsequence}{subsucesión} par tiende a $1$ y
la impar a $-1$, luego la sucesión diverge --- pero lo hace de manera
organizada, agrupándose en torno a los dos valores $\pm 1$. Para
$u_n = \cos\frac{2\pi n}{3}$: las tres subsucesiones de índices $3k$,
$3k + 1$, $3k + 2$ son constantes, iguales a $1$, $-\frac12$,
$-\frac12$; el \omterm{def:b1:logic:sets}{conjunto} de límites de subsucesiones es
$\{1, -\frac12\}$. La idea de cierre: una sucesión \emph{acotada}
converge exactamente cuando tiene un único límite de \omterm{def:b1:seq:subsequence}{subsucesión}
(\cref{exo:b1:seq:8}); la divergencia de una sucesión acotada
significa siempre al menos dos \omterm{def:b1:structures:group}{grupos}, y Bolzano--Weierstrass, más
abajo, garantiza que hay al menos uno.
\end{example}

\begin{theorem}[Bolzano--Weierstrass]\label{thm:b1:seq:bw}
Toda sucesión acotada de reales tiene una \omterm{def:b1:seq:subsequence}{subsucesión}
convergente.\index{teorema de Bolzano--Weierstrass}
\end{theorem}

\begin{proof}
Sea $u_n \in \intcc{a}{b}$ para todo $n$. Constrúyanse segmentos
encajados por dicotomía: póngase $\intcc{a_0}{b_0} = \intcc{a}{b}$;
dado $\intcc{a_k}{b_k}$ que contenga $u_n$ para infinitos $n$, una de
sus dos mitades sigue conteniendo $u_n$ para infinitos $n$ ---
llámese $\intcc{a_{k+1}}{b_{k+1}}$. Las sucesiones $(a_k)$, $(b_k)$
son adyacentes ($b_k - a_k = \frac{b-a}{2^k} \to 0$), con límite común
$\ell$ (\cref{thm:b1:seq:adjacent}).

Extracción: elíjase $\varphi(0)$ con
$u_{\varphi(0)} \in \intcc{a_0}{b_0}$ y después, por recurrencia,
$\varphi(k+1) > \varphi(k)$ con $u_{\varphi(k+1)} \in
\intcc{a_{k+1}}{b_{k+1}}$ --- posible, pues ese segmento contiene
infinitos términos. Entonces $a_k \leq u_{\varphi(k)} \leq b_k$, y el
teorema del emparedado da $u_{\varphi(k)} \to \ell$.
\end{proof}

\begin{remark}[Qué dice, y qué no dice, Bolzano--Weierstrass]
\emph{Sí} dice: solo con la acotación, alguna \omterm{def:b1:seq:subsequence}{subsucesión} converge ---
existencia sin fórmula, como deja claro la demostración por dicotomía
(nada nos dice \emph{qué} índices sobreviven). \emph{No} dice que el
límite sea único: $((-1)^n)$ tiene subsucesiones que convergen a $1$
y a $-1$, y el \omterm{def:b1:logic:sets}{conjunto} de límites de subsucesiones puede ser incluso
infinito (\cref{ex:b1:seq:sublimits}, y todo el \omterm{def:b1:logic:sets}{conjunto} de Cantor
del~\cref{pb:b1:topology:1}). No sobrevive sin la acotación: $(n)$ no
tiene ninguna \omterm{def:b1:seq:subsequence}{subsucesión} convergente --- aunque de toda sucesión no
acotada siempre se puede extraer una \omterm{def:b1:seq:subsequence}{subsucesión} que tienda a
$+\infty$ o a $-\infty$ (elíjase $\varphi(k)$ con
$u_{\varphi(k)} \geq k$, por ejemplo). Bien usado, el teorema es una
\emph{bomba de existencia}: aparece en el nudo del criterio de Cauchy
de más abajo, del teorema de Heine y del teorema de los valores
extremos --- siempre para producir un punto que ninguna construcción
explícita ofrece.
\end{remark}

\section{Sucesiones de Cauchy y completitud}

\begin{definition}\label{def:b1:seq:cauchy}
Una sucesión $(u_n)$ es una \emph{sucesión de
Cauchy}\index{sucesión de Cauchy} cuando sus términos se acercan
arbitrariamente \emph{entre sí}:
\[
\forall \varepsilon > 0,\ \exists N,\ \forall p, q \geq N,
\qquad \abs{u_p - u_q} \leq \varepsilon .
\]
\end{definition}

\begin{example}[Verificar a mano la propiedad de Cauchy]\label{ex:b1:seq:cauchyhand}
Sea $u_n = \sum_{k=0}^{n} \frac{\cos k}{2^k}$ --- sin monotonía y sin
límite adivinable. Para $p > q$:
\[
\abs{u_p - u_q}
= \Bigl| \sum_{k=q+1}^{p} \frac{\cos k}{2^k} \Bigr|
\leq \sum_{k=q+1}^{p} \frac{1}{2^k}
< \frac{1}{2^{q}} ,
\]
por la desigualdad triangular, $\abs{\cos k} \leq 1$ y una suma
geométrica finita. Dado $\varepsilon > 0$, elíjase $N$ con
$2^{-N} \leq \varepsilon$: todas las separaciones más allá de $N$ son
$\leq \varepsilon$, la sucesión es de Cauchy y, por tanto, converge
--- a un límite que nadie sabe nombrar en forma cerrada, que es
justamente lo importante. La idea de cierre: dominar geométricamente
los incrementos es la manera estándar de ganarse la propiedad de
Cauchy, y el~\cref{ch:b1:series} embotellará el argumento como «la
convergencia absoluta implica la convergencia».
\end{example}

\begin{theorem}[Completitud de $\R$]\label{thm:b1:seq:complete}
Una sucesión de reales converge si y solo si es una \omterm{def:b1:seq:cauchy}{sucesión de
Cauchy}.\index{completitud de R@completitud de $\R$}
\end{theorem}

\begin{proof}
($\Rightarrow$) Si $u_n \to \ell$: más allá del umbral de
$\frac\varepsilon2$, $\abs{u_p - u_q} \leq \abs{u_p - \ell} +
\abs{\ell - u_q} \leq \varepsilon$.

($\Leftarrow$) Sea $(u_n)$ de Cauchy. \emph{Está acotada}: con
$\varepsilon = 1$, más allá de $N$ todos los términos distan menos de
$1$ de $u_N$, y la cabeza es finita. \emph{Extracción}: por
el~\cref{thm:b1:seq:bw}, alguna \omterm{def:b1:seq:subsequence}{subsucesión} $u_{\varphi(n)} \to \ell$.
\emph{Conclusión}: dado $\varepsilon > 0$, tómense $N$ (Cauchy, para
$\frac\varepsilon2$) y $n \geq N$ con $\abs{u_{\varphi(n)} - \ell}
\leq \frac\varepsilon2$ y $\varphi(n) \geq N$; entonces, para todo
$p \geq N$:
\[
\abs{u_p - \ell} \leq \abs{u_p - u_{\varphi(n)}} +
\abs{u_{\varphi(n)} - \ell} \leq \varepsilon . \qedhere
\]
\end{proof}

\begin{remark}
El valor del criterio: certifica la convergencia \emph{sin nombrar el
límite}. Falla en $\Q$ (los truncamientos decimales de $\sqrt 2$ forman
una \omterm{def:b1:seq:cauchy}{sucesión de Cauchy} de racionales sin límite racional): la
completitud es una propiedad de $\R$, equivalente al axioma de la
\omterm{def:b1:reals:bounds}{cota superior}. Es además el caballo de batalla que hay detrás de la
convergencia de las series (\cref{ch:b1:series}).
\end{remark}

\begin{example}[Una sucesión de Cauchy de límite invisible]\label{ex:b1:seq:basel}
Sea $S_n = \sum_{k=1}^{n} \frac{1}{k^2}$. Para $p > q \geq 1$:
\[
S_p - S_q = \sum_{k=q+1}^{p} \frac{1}{k^2}
\leq \sum_{k=q+1}^{p} \frac{1}{k(k-1)}
= \sum_{k=q+1}^{p} \Bigl(\frac{1}{k-1} - \frac 1k\Bigr)
= \frac 1q - \frac 1p < \frac 1q ,
\]
de modo que, más allá de $N > \frac1\varepsilon$, todas las
separaciones son $\leq \varepsilon$: $(S_n)$ es de Cauchy y, por
tanto, converge. Obsérvese lo que acaba de pasar: hemos demostrado que
existe un número real concreto sin tener ningún nombre para él. (Es
$\frac{\pi^2}{6}$ --- una célebre identidad de Euler, demostrada en el
volumen del segundo año; nada de este capítulo podría decírnoslo.)
Este reparto de tareas --- existencia ahora, identificación después,
si acaso --- es todo el sentido del criterio de Cauchy y el motor de
la teoría de series del~\cref{ch:b1:series}.
\end{example}

\section{Sucesiones recurrentes}

\begin{method}[Estudiar $u_{n+1} = f(u_n)$]\label{met:b1:seq:recurrent}
Dados $f$ y un punto de partida $u_0$:\index{sucesión recurrente}
\begin{enumerate}
  \item \emph{\omterm{prop:b1:reals:intervals}{Intervalo} estable:} búsquese un \omterm{prop:b1:reals:intervals}{intervalo} $I$ con
        $f(I) \subseteq I$ que contenga $u_0$: entonces todos los
        $u_n \in I$ (por inducción).
  \item \emph{Candidatos a límite:} si $u_n \to \ell \in I$ y $f$ es
        continua en $\ell$ (\cref{ch:b1:continuity}), entonces $\ell$
        es un \emph{punto fijo}: $f(\ell) = \ell$. Resuélvase
        $f(x) = x$.
  \item \emph{Monotonía:} si $f$ es creciente en $I$, entonces
        $(u_n)$ es monótona (creciente si $u_1 \geq u_0$, decreciente
        en caso contrario); combinado con la acotación,
        el~\cref{thm:b1:seq:monotone} concluye. Si $f$ es decreciente,
        estúdiense las dos subsucesiones $(u_{2n})$ y $(u_{2n+1})$,
        que son monótonas para $f \circ f$.
  \item \emph{Control del error:} una desigualdad
        $\abs{f(x) - \ell} \leq k\abs{x - \ell}$ con $k < 1$ da
        $\abs{u_n - \ell} \leq k^n \abs{u_0 - \ell} \to 0$ directamente.
\end{enumerate}
\end{method}

\begin{example}[Método de Herón]\label{ex:b1:seq:heron}
Sean $u_0 = 2$ y $u_{n+1} = \dfrac12\Bigl(u_n + \dfrac{2}{u_n}\Bigr)$:
el antiguo algoritmo para
$\sqrt 2$.\index{método de Herón}
\begin{itemize}
  \item \emph{Estabilidad:} para $x > 0$, la desigualdad entre las
        medias aritmética y geométrica da $\frac12(x + \frac2x) \geq
        \sqrt{x \cdot \frac 2x} = \sqrt 2$; luego
        $I = \intco{\sqrt 2}{+\infty}$ es estable y contiene $u_1$ (en
        efecto, $u_1 = \frac32 \geq \sqrt 2$).
  \item \emph{Monotonía:} para $x \geq \sqrt 2$, $\;x - f(x) =
        \frac{x^2 - 2}{2x} \geq 0$: la sucesión decrece a partir de
        $u_1$ y está acotada inferiormente por $\sqrt 2$: converge.
  \item \emph{Límite:} los puntos fijos resuelven $x = \frac12(x +
        \frac2x)$, es decir, $x^2 = 2$: en $I$, $\ell = \sqrt 2$.
  \item \emph{Velocidad:} $u_{n+1} - \sqrt 2 = \frac{(u_n -
        \sqrt2)^2}{2u_n} \leq \frac{(u_n - \sqrt 2)^2}{2\sqrt 2}$: el
        número de cifras correctas aproximadamente se \emph{duplica}
        en cada paso (convergencia cuadrática).
\end{itemize}
\end{example}

\begin{remark}[Errores frecuentes con los límites]
Cuatro clásicos. (i) \emph{Los pasos pequeños no implican
convergencia}: $u_{n+1} - u_n \to 0$ es mucho más débil que la
propiedad de Cauchy --- las sumas armónicas $H_n$ tienen pasos
$\frac{1}{n+1} \to 0$ y, sin embargo, divergen a $+\infty$
(\cref{exo:b1:seq:5}); la condición de Cauchy controla
$\abs{u_p - u_q}$ para \emph{todos} los pares grandes, no solo los
consecutivos. (ii) \emph{Las desigualdades estrictas mueren en el
límite}: de $u_n < v_n$ para todo $n$ solo se obtiene
$\lim u_n \leq \lim v_n$ (\cref{thm:b1:seq:order}); $\frac1n > 0$ y,
pese a ello, $\lim = 0$. (iii) \emph{Acotada no es convergente}:
$((-1)^n)$ está acotada y diverge; acotación más \emph{monotonía} sí
converge, y la acotación sola solo garantiza una \omterm{def:b1:seq:subsequence}{subsucesión}
convergente (\cref{thm:b1:seq:bw}). (iv) \emph{La ecuación de punto
fijo va después, no antes}: para $u_{n+1} = f(u_n)$, resolver
$f(\ell) = \ell$ identifica el límite \emph{solo después} de haber
demostrado la convergencia. La recurrencia $u_{n+1} = 2u_n$ tiene el
único punto fijo $\ell = 0$ y, sin embargo, desde $u_0 = 1$ la
sucesión se va a $+\infty$: la ecuación $\ell = 2\ell$ nunca tuvo
derecho a un límite. Orden de las operaciones, siempre: existencia
primero (\cref{met:b1:seq:recurrent}, pasos 1--3), identificación
después.
\end{remark}

\begin{example}[Una $f$ decreciente: la recurrencia áurea]\label{ex:b1:seq:golden}
Sean $u_0 = 1$ y $u_{n+1} = \dfrac{1}{1 + u_n}$. Aquí
$f(x) = \frac{1}{1+x}$ es \emph{decreciente}, de modo que la sucesión
no es monótona (alterna alrededor de su límite); el paso de
contracción del~\cref{met:b1:seq:recurrent} es la herramienta
adecuada. Estabilidad: si $x \in \intcc{\frac12}{1}$, entonces
$1 + x \in \intcc{\frac32}{2}$, luego $f(x) \in
\intcc{\frac12}{\frac23} \subseteq \intcc{\frac12}{1}$, y
$u_1 = \frac12$ mete ahí a toda la sucesión. Punto fijo:
$\ell = \frac{1}{1+\ell}$ con $\ell > 0$ da $\ell^2 + \ell - 1 = 0$,
es decir,
\[
\ell = \frac{\sqrt5 - 1}{2} = 0.6180\dots
\]
(el inverso de la razón áurea). Contracción: para
$x, y \in \intcc{\frac12}{1}$,
\[
\abs{f(x) - f(y)} = \frac{\abs{x - y}}{(1+x)(1+y)}
\leq \frac{\abs{x-y}}{(3/2)^2} = \frac49\,\abs{x - y} ,
\]
luego $\abs{u_n - \ell} \leq \bigl(\frac49\bigr)^{n-1}\abs{u_1 -
\ell} \to 0$: convergencia, con velocidad geométrica y sin necesidad
de monotonía. La idea de cierre: los métodos monótonos y los métodos
de contracción se reparten el mundo recurrente --- una $f$ creciente
da órbitas monótonas, y una $f$ decreciente, órbitas alternadas
domadas por una constante de Lipschitz $< 1$ (la teoría sistemática es
el~\cref{exo:b1:derivative:11}).
\end{example}

\begin{remark}[Perspectivas dentro de este volumen]
Las sucesiones son el instrumento de medida que el resto del volumen
acerca a cada objeto. En el~\cref{ch:b1:topology} \emph{caracterizan}
la clausura y la compacidad; en el~\cref{ch:b1:continuity} transportan
límites de funciones; en el~\cref{ch:b1:integration}, las sumas de
Riemann son sucesiones que convergen a la integral; y
el~\cref{ch:b1:series} \emph{es} la teoría de una clase especial de
sucesiones, las sumas parciales. Hasta los capítulos de álgebra las
consumen: los iterados de una matriz en el~\cref{ch:b1:matrices} forman
sucesiones cuyo comportamiento (la convergencia de $A^n$) es una
cuestión de álgebra lineal con el vocabulario de este capítulo. Los dos
teoremas que hay que llevar a todas partes: el del límite monótono
(existencia a partir del orden) y el de Bolzano--Weierstrass
(existencia a partir de la acotación) --- entre los dos nace casi todo
límite de este libro.
\end{remark}

\begin{remark}[Sucesiones complejas]
Una sucesión $(z_n)$ de números complejos converge a $\ell$ cuando
$\abs{z_n - \ell} \to 0$; equivalentemente, cuando
$\Re(z_n) \to \Re(\ell)$ e $\Im(z_n) \to \Im(\ell)$ (compárese
$\abs{z}$ con $\abs{\Re z} + \abs{\Im z}$). Los teoremas que no
involucran el orden --- operaciones, Bolzano--Weierstrass (extráigase
dos veces), criterio de Cauchy --- se trasladan literalmente.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:seq:1}
Directamente a partir de la~\cref{def:b1:seq:limit}, demuéstrese que
$\dfrac{2n+1}{n+3} \to 2$ y que $(u_n) = ((-1)^n)$ diverge.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:seq:2}
Calcúlense los límites:
\[
\frac{n^2 - 3n + 1}{2n^2 + 5},
\qquad
\sqrt{n+1} - \sqrt n,
\qquad
\frac{2^n + n^3}{3^n - n^2},
\qquad
\sqrt[n]{5}\ \Bigl(= 5^{1/n}\Bigr).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:seq:3}
Demuéstrese la comparación estándar: si $\abs{q} < 1$, entonces
$q^n \to 0$ \emph{(escríbase $\frac{1}{\abs q} = 1 + h$, $h > 0$, y
úsese la desigualdad de Bernoulli $(1+h)^n \geq 1 + nh$, que se
demuestra por inducción)}. ¿Qué comportamientos hay para $q = 1$,
$q = -1$ y $\abs q > 1$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:seq:4}
Sean $u_{n+1} = \frac{u_n + 3}{2}$, $u_0 = 0$. Hállese el punto fijo
$\ell$, demuéstrese que $v_n = u_n - \ell$ es geométrica y dense una
fórmula explícita y el límite de $(u_n)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:seq:5}
(Serie armónica) Sea $H_n = \sum_{k=1}^{n} \frac 1k$. Demuéstrese que
$H_{2n} - H_n \geq \frac12$ para todo $n \geq 1$, y conclúyase que
$(H_n)$ \emph{no} es una \omterm{def:b1:seq:cauchy}{sucesión de Cauchy} y, por tanto, diverge (a
$+\infty$, por ser creciente).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:seq:6}
Supóngase que $(u_{2n})$, $(u_{2n+1})$ y $(u_{3n})$ convergen todas.
Demuéstrese que $(u_n)$ converge. \emph{(Búsquense subsucesiones
comunes para igualar los límites.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:seq:7}
Estúdiese la sucesión $u_0 = 0$, $u_{n+1} = \sqrt{2 + u_n}$:
estabilidad, monotonía, límite. Demuéstrese después la cota de error
$\abs{u_n - 2} \leq \dfrac{2}{3^{\,n}}$ \emph{(véase que
$2 - u_{n+1} = \dfrac{2 - u_n}{2 + \sqrt{2 + u_n}}$ y acótese el
denominador inferiormente por $3$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:seq:8}
Sea $(u_n)$ acotada y tal que toda \omterm{def:b1:seq:subsequence}{subsucesión} convergente de $(u_n)$
tenga el \emph{mismo} límite $\ell$. Demuéstrese que $u_n \to \ell$.
\emph{(Reducción al absurdo más Bolzano--Weierstrass.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:seq:9}
Con la notación del~\cref{ex:b1:seq:e}, supóngase $\eu = \frac pq$ con
$p, q \in \N^*$. Usando $a_q < \eu < b_q = a_q + \frac{1}{q\, q!}$,
multiplíquese por $q!$ y dedúzcase una contradicción entre dos
enteros. Conclúyase: $\eu$ es irracional.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:seq:10}
(Medias de Cesàro) Para una sucesión $(u_n)_{n \geq 1}$, póngase
$c_n = \frac{u_1 + \dots + u_n}{n}$.
\begin{enumerate}
  \item Demuéstrese que $u_n \to \ell$ implica $c_n \to \ell$
        \emph{(córtese la suma en un umbral $N$; acótese la cabeza por
        una cantidad fija dividida por $n$, y la cola por
        $\varepsilon$)}.
  \item Véase con un ejemplo que el recíproco falla.
  \item Dedúzcase que si $u_{n+1} - u_n \to \ell$, entonces
        $\frac{u_n}{n} \to \ell$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:seq:11}
Sea $(u_n)$ tal que $0 \leq u_{m+n} \leq u_m + u_n$ para todos $m, n$
(subaditividad). Demuéstrese que $\bigl(\frac{u_n}{n}\bigr)$ converge
a $\inf_{n \geq 1} \frac{u_n}{n}$. \emph{(Para $m$ fijo, escríbase
$n = qm + r$ y acótese $\frac{u_n}{n}$ usando $u_n \leq q\,u_m + u_r$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:seq:12}
Usando la densidad del \omterm{def:b1:structures:subgroup}{subgrupo} $\Z + 2\pi\Z$ de $(\R, +)$
(\cref{exo:b1:reals:9}), demuéstrese que la sucesión
$(\sin n)_{n \in \N}$ es densa en $\intcc{-1}{1}$ --- en particular,
diverge.
\end{exercise}

\section{Problema: Cesàro, Stolz y la lenta caída del seno}

\begin{problem}[{Problema del fin de semana --- el teorema de
Cesàro--Stolz y la asintótica $u_n \sim \sqrt{3/n}$ para
$u_{n+1} = \sin u_n$}]
\label{pb:b1:seq:1}
El teorema de Cesàro--Stolz es la regla de l'Hôpital discreta: para
hallar el límite de un cociente $a_n/b_n$ basta hallar el límite del
cociente de \emph{incrementos} $(a_{n+1} - a_n)/(b_{n+1} - b_n)$. Este
problema demuestra el teorema, recolecta con él límites clásicos y lo
apunta después a un blanco famoso: la sucesión $u_{n+1} = \sin u_n$,
que se arrastra hacia $0$ a la velocidad exactamente calculable
$u_n \sim \sqrt{3/n}$.\index{teorema de Cesaro--Stolz@teorema de
Cesàro--Stolz} Aquí se conceden dos hechos del volumen anterior, que
más adelante se redemuestran honestamente en este volumen: la
desigualdad de la recta tangente
\[
\tag{G1} \eu^{u} \geq 1 + u \quad (u \in \R),
\]
redemostrada por convexidad en el~\cref{ch:b1:derivative}, y el
encuadre del seno
\[
\tag{G2} x - \frac{x^3}{6} \;\leq\; \sin x \;\leq\; x -
\frac{x^3}{6} + \frac{x^5}{120} \quad (0 \leq x \leq 1), \qquad
\abs{\sin x} \leq \abs{x} \quad (x \in \R),
\]
redemostrado con la fórmula de Taylor en el~\cref{ch:b1:taylor}.

\medskip
\noindent\textbf{Parte I --- Sumas sin fórmula cerrada.}
\begin{enumerate}
  \item Usando $1 + 2 + \dots + n = \frac{n(n+1)}{2}$ y
        $1^2 + \dots + n^2 = \frac{n(n+1)(2n+1)}{6}$, calcúlense
        $\lim \frac{1 + 2 + \dots + n}{n^2}$ y
        $\lim \frac{1^2 + \dots + n^2}{n^3}$.
  \item Sea $T_n = \sum_{k=1}^n \sqrt k$, para el que no existe
        fórmula cerrada. Demuéstrese el encuadre
        \[
        \frac{1}{2\sqrt 2}\,n^{3/2} \;\leq\; T_n \;\leq\;
        n^{3/2}
        \]
        \emph{(consérvense solo los términos $k > \frac n2$ para la
        cota inferior)}. Así pues, $T_n$ es del orden de $n^{3/2}$
        --- pero ¿con qué constante? Guárdese la pregunta hasta la
        pregunta 8.
  \item (Lema del telescopio) Sea $(b_n)$ estrictamente creciente y
        supóngase que, para todo $k \geq N$,
        \[
        m \;\leq\; \frac{a_{k+1} - a_k}{b_{k+1} - b_k} \;\leq\;
        M .
        \]
        Demuéstrese que $m \leq \dfrac{a_n - a_N}{b_n - b_N} \leq M$
        para todo $n > N$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El teorema de Cesàro--Stolz.} Sea
$(b_n)$ estrictamente creciente con $b_n \to +\infty$, y supóngase
$\dfrac{a_{n+1} - a_n}{b_{n+1} - b_n} \to \ell \in \R$.
\begin{enumerate}[resume]
  \item Fíjese $\varepsilon > 0$. Véase que hay un $N$ tal que
        $\ell - \varepsilon \leq \dfrac{a_n - a_N}{b_n - b_N}
        \leq \ell + \varepsilon$ para todo $n > N$.
  \item Establézcase, para $n > N$, la identidad
        \[
        \frac{a_n}{b_n} - \ell
        = \frac{a_N - \ell\,b_N}{b_n}
        + \Bigl(1 - \frac{b_N}{b_n}\Bigr)
        \Bigl(\frac{a_n - a_N}{b_n - b_N} - \ell\Bigr),
        \]
        y conclúyase el teorema: $\dfrac{a_n}{b_n} \to \ell$.
  \item Demuéstrese la variante $+\infty$: si $\dfrac{a_{n+1} -
        a_n}{b_{n+1} - b_n} \to +\infty$ (con las mismas hipótesis
        sobre $(b_n)$), entonces $\dfrac{a_n}{b_n} \to +\infty$.
  \item Tómese $b_n = n$: recupérese el teorema de las medias de
        Cesàro del~\cref{exo:b1:seq:10}. Véase después que el
        recíproco de Cesàro--Stolz falla: para $a_n = (-1)^n$,
        $b_n = n$, el cociente $a_n/b_n$ converge mientras que el
        cociente de incrementos no. Stolz es una calle de un solo
        sentido.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Primeros dividendos.}
\begin{enumerate}[resume]
  \item Demuéstrese $(1+h)^{3/2} - 1 = \dfrac{3h + 3h^2 +
        h^3}{(1+h)^{3/2} + 1}$ por conjugación, dedúzcase
        $n\bigl((1 + \tfrac1n)^{3/2} - 1\bigr) \to \tfrac32$ y
        conclúyase con Cesàro--Stolz:
        \[
        T_n = \sum_{k=1}^{n} \sqrt k \;\sim\; \tfrac23\,
        n^{3/2} ,
        \]
        resolviendo el suspense de la pregunta 2.
  \item Solo a partir de (G1), dedúzcase el encuadre del logaritmo
        \[
        \frac{t}{1 + t} \;\leq\; \ln(1 + t) \;\leq\; t
        \qquad (t > -1)
        \]
        \emph{(aplíquese (G1) en $u = \ln(1+t)$ y en
        $u = -t/(1+t)$)}.
  \item Véase que $b_n = \ln n$ es estrictamente creciente con
        $\ln n \to +\infty$, y demuéstrese con Cesàro--Stolz y la
        pregunta 9 que
        \[
        H_n = \sum_{k=1}^{n} \frac 1k \;\sim\; \ln n .
        \]
        (La estructura más fina $H_n = \ln n + \gamma + o(1)$ es el
        problema del fin de semana del~\cref{ch:b1:series}.)
  \item (De los cocientes a las raíces) Sea $u_n > 0$ con
        $\frac{u_{n+1}}{u_n} \to L > 0$. Usando la pregunta 9, véase
        que $\ln\frac{u_{n+1}}{u_n} \to \ln L$; aplíquese Cesàro para
        concluir $\frac{\ln u_n}{n} \to \ln L$ y, después, con (G1),
        que $u_n^{1/n} \to L$. Aplicación: calcúlese
        $\lim\,\binom{2n}{n}^{1/n}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La lenta caída del seno.} Sean
$u_0 \in \R$ y $u_{n+1} = \sin u_n$.
\begin{enumerate}[resume]
  \item A partir de (G2), véase que $0 < \sin x < x$ para
        $0 < x \leq 1$. Dedúzcase: $u_1 \in \intcc{-1}{1}$; si
        $u_1 = 0$, la sucesión es nula a partir del rango $1$; y si
        $u_1 > 0$ (siendo simétrico el caso $u_1 < 0$, por ser $\sin$
        impar), entonces $(u_n)_{n \geq 1}$ es estrictamente
        decreciente, positiva y converge a $0$ \emph{(identifíquese el
        límite con $\ell = \sin \ell$, usando $\abs{\sin a - \sin b}
        \leq \abs{a - b}$, consecuencia a su vez de (G2) y de la
        fórmula de transformación de producto en suma)}.
  \item Supóngase de aquí en adelante $u_1 \in \intoc{0}{1}$. Véase,
        por emparedado y usando (G2):
        \[
        \frac{\sin u_n}{u_n} \to 1
        \qquad\text{and}\qquad
        \frac{u_n - \sin u_n}{u_n^{3}} \to \frac16 .
        \]
  \item Demuéstrese la factorización
        \[
        w_n := \frac{1}{u_{n+1}^{2}} - \frac{1}{u_n^{2}}
        = \frac{u_n - \sin u_n}{u_n^{3}} \cdot
        \frac{u_n + \sin u_n}{u_n} \cdot
        \Bigl(\frac{u_n}{\sin u_n}\Bigr)^{2},
        \]
        y dedúzcase $w_n \to \frac13$.
  \item Conclúyase con el~\cref{exo:b1:seq:10} (versión de
        incrementos) que $\frac{1}{n\,u_n^{2}} \to \frac13$ y,
        después, por un argumento de conjugación para la raíz
        cuadrada, el titular:
        \[
        \sqrt n\;u_n \longrightarrow \sqrt 3 ,
        \qquad\text{i.e.}\qquad
        u_n \sim \sqrt{\frac 3n} .
        \]
  \item Cuantifíquese la lentitud: véase que, a la larga,
        $\sqrt{2/n} \leq u_n \leq 2/\sqrt n$, de modo que llegar a
        $u_n \leq 10^{-2}$ exige más de $20\,000$ iteraciones (unas
        $30\,000$, por la asintótica). Contrástese con el
        \omterm{ex:b1:seq:heron}{método de Herón} (\cref{ex:b1:seq:heron}) y explíquese la
        razón estructural: en el punto fijo $0$, la pendiente de
        $\sin$ vale $1$ (un punto fijo \emph{neutro}), mientras que
        las iteraciones que \omterm{def:b1:arith:divides}{dividen} el error por dos necesitan una
        pendiente de \omterm{def:b1:complex:field}{módulo} $< 1$.
  \item Véase que, para \emph{todo} punto de partida $u_0 \in \R$, o
        bien $u_n = 0$ a partir del rango $1$, o bien
        $\abs{u_n} \sim \sqrt{3/n}$ --- la caída es universal, y solo
        el signo recuerda a $u_0$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- El principio general.} El seno es una
instancia de una máquina.
\begin{enumerate}[resume]
  \item Sean $u_n > 0$, $u_n \to 0$ y $\dfrac{u_n -
        u_{n+1}}{u_n^{2}} \to a > 0$. Demuéstrense sucesivamente:
        $\frac{u_{n+1}}{u_n} \to 1$; después $\frac{1}{u_{n+1}} -
        \frac{1}{u_n} \to a$; y después $n\,u_n \to \frac1a$.
  \item (Modelo exacto) Para $u_{n+1} = \dfrac{u_n}{1 + u_n}$ con
        $u_0 > 0$: véase que $\frac{1}{u_n}$ es aritmética, resuélvase
        exactamente y compruébese la conclusión de la pregunta 18
        contra la fórmula exacta.
  \item Para $u_{n+1} = u_n \eu^{-u_n}$ con $u_0 > 0$: véase que
        $u_n \to 0$, úsese (G1) para encajar
        $\frac{1 - \eu^{-t}}{t}$ entre $\frac{1}{1+t}$ y $1$ para
        $t > 0$, y conclúyase $u_n \sim \frac 1n$.
  \item (Contacto cúbico, telescopio al cuadrado) Sean $u_n > 0$,
        $u_n \to 0$ y $\dfrac{u_n - u_{n+1}}{u_n^{3}} \to a > 0$.
        Adáptese la factorización de la pregunta 14 para ver que
        $\frac{1}{u_{n+1}^2} - \frac{1}{u_n^2} \to 2a$ y conclúyase
        $n\,u_n^{2} \to \frac{1}{2a}$. Compruébese que $a = \frac16$
        recupera la parte IV.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Límites del método y moralejas.}
\begin{enumerate}[resume]
  \item Véase que la hipótesis $b_n \to +\infty$ no se puede
        suprimir: para $a_n = 2 - 2^{-n}$ y $b_n = 1 - 2^{-n}$, el
        cociente de incrementos tiende a $1$ mientras que
        $\frac{a_n}{b_n} \to 2$. Señálese la línea exacta de la
        demostración de la pregunta 5 que se rompe.
  \item (Stolz dos veces) Demuéstrese $\sum_{k=1}^{n} H_k \sim n \ln n$
        \emph{(una \omterm{def:b1:logic:map}{aplicación} de Cesàro--Stolz y después la
        pregunta 10; acótese $(n+1)\ln(n+1) - n\ln n$ con la
        pregunta 9)}.
  \item (Medias geométricas) Si $u_n > 0$ y $u_n \to \ell > 0$, véase
        que $(u_1 u_2 \cdots u_n)^{1/n} \to \ell$; y si
        $u_n \to +\infty$, que $(u_1 \cdots u_n)^{1/n} \to +\infty$.
        Dedúzcase $(n!)^{1/n} \to +\infty$.
  \item Síntesis, una frase para cada punto: (i) dónde ha entrado
        exactamente la completitud en este problema; (ii) en qué
        sentido es Cesàro--Stolz una regla de l'Hôpital discreta (su
        gemelo diferencial se apoya en el teorema del valor medio
        del~\cref{ch:b1:derivative}); (iii) enúnciese la heurística
        que liga el orden de contacto de $f$ en un punto fijo neutro
        con el exponente de decaimiento de $u_{n+1} = f(u_n)$;
        (iv) sígase el rastro de la constante $3$ de $\sqrt{3/n}$ a
        través de la cadena $\frac16 \to \frac13 \to 3$.
\end{enumerate}
\end{problem}
