\chapter{Curvas planas}\label{ch:b1:curves}

El cálculo de los \cref{ch:b1:derivative,ch:b1:taylor} se construyó
para funciones $y = f(x)$; pero la mayoría de las curvas de la
geometría y de la mecánica ---trayectorias, círculos rodando sobre
círculos, órbitas--- se niegan a esa forma y vienen, en su lugar,
como curvas \emph{parametrizadas} $t \mapsto (x(t), y(t))$ o como
curvas \emph{polares} $r = r(\theta)$. Este capítulo clásico estudia
las dos, y cierra con las \omterm{def:b1:curves:conics}{cónicas}.

\section{Curvas parametrizadas}

\begin{definition}\label{def:b1:curves:def}
Una \emph{curva parametrizada}\index{curva parametrizada} es una
\omterm{def:b1:logic:map}{aplicación} $f \colon I \to \R^2$, $t \mapsto M(t) = (x(t), y(t))$,
con $x, y$ de clase $C^1$ (al menos) en el \omterm{prop:b1:reals:intervals}{intervalo} $I$. El
\emph{vector velocidad} es $f'(t) = (x'(t), y'(t))$; el punto
$M(t_0)$ es \emph{regular} cuando $f'(t_0) \neq (0,0)$, y la
\emph{tangente} en él es la recta que pasa por $M(t_0)$ dirigida por
$f'(t_0)$.
\end{definition}

\begin{proof}[Por qué $f'$ dirige la tangente]
Por Taylor--Young componente a componente,
$M(t) = M(t_0) + (t - t_0) f'(t_0) + o(t - t_0)$: la dirección de la
cuerda $\frac{M(t) - M(t_0)}{t - t_0}$ tiende a $f'(t_0)$. La
hipótesis de regularidad es lo que hace del límite una
\emph{dirección}: si $f'(t_0) = 0$, la fórmula se colapsa en
$M(t) = M(t_0) + o(t - t_0)$ y no dice nada sobre cómo abandona la
curva el punto --- de ahí el tratamiento aparte de los puntos
singulares en el método de más abajo, donde la primera \omterm{def:b1:derivative:def}{derivada}
\emph{no nula} asume el papel de $f'$. Nótese además que la recta
tangente es un objeto geométrico: toda reparametrización cambia $f'$
en un factor escalar no nulo y deja la recta intacta.
\end{proof}

\begin{method}[Estudiar una curva parametrizada]\label{met:b1:curves:study}
\begin{enumerate}
  \item \emph{Redúzcase el dominio} usando las simetrías: relaciones
        entre $M(-t)$, $M(t + T)$, \dots\ y $M(t)$ (reflexiones,
        traslaciones, periodicidad), y dibújese después solo la
        parte reducida.
  \item \emph{Variaciones}: tabúlense conjuntamente los signos de
        $x'$ y de $y'$; la curva se mueve a la derecha o a la
        izquierda según $x'$, y hacia arriba o abajo según $y'$.
  \item \emph{Puntos notables}: tangente horizontal
        ($y' = 0 \neq x'$), tangente vertical ($x' = 0 \neq y'$); y
        en un punto \emph{singular} ($f' = 0$), examínense las
        \omterm{def:b1:derivative:def}{derivadas} de orden superior para la dirección tangente.
  \item \emph{Comportamiento asintótico} en los extremos de $I$, y
        después el boceto.
\end{enumerate}
\end{method}

\begin{example}[Ramas asintóticas, trabajadas]
\label{ex:b1:curves:asymptotes}
$x(t) = t$, $y(t) = t + \dfrac1t$ en $\intoo{0}{+\infty}$. Cuando
$t \to 0^{+}$: $x \to 0$ mientras $y \to +\infty$ --- la curva trepa
a lo largo de la \emph{asíntota vertical} $x = 0$ (límite finito para
una coordenada e infinito para la otra). Cuando $t \to +\infty$: las
dos \omterm{prop:b1:vspaces:coordinates}{coordenadas} explotan, así que pruébese con una recta: la
diferencia
\[
y(t) - x(t) = \frac1t \longrightarrow 0^{+}
\]
exhibe la \emph{asíntota oblicua} $y = x$, alcanzada por encima.
Entremedias, $y' = 1 - \frac1{t^2}$ se anula en $t = 1$: el punto
$(1, 2)$ es el punto bajo de la rama, y $x' = 1 > 0$ en todo el
recorrido, de modo que la curva siempre avanza hacia la derecha. Un
arco, dos asíntotas, un mínimo: una imagen completa a partir de tres
cálculos --- con la receta general visible por debajo: cuando
$x, y \to \infty$ juntas, examínese $y/x$ para una pendiente
candidata (aquí $\to 1$) y después $y - (\text{pendiente})\,x$ para
la ordenada en el origen y el lado de aproximación.
\end{example}

\begin{example}[La astroide]\label{ex:b1:curves:astroid}
$x(t) = \cos^3 t$, $y(t) = \sin^3 t$.\index{astroide} Simetrías:
$M(t + 2\pi) = M(t)$ (estúdiese un período); $M(-t)$ es el reflejo de
$M(t)$ respecto del eje $x$; $M(\pi - t)$, respecto del eje $y$; y
$M(\frac\pi2 - t)$, respecto de la diagonal $y = x$: basta estudiar
$t \in \intcc{0}{\frac\pi4}$ y desplegar.

Velocidad: $f'(t) = 3\sin t\cos t\,(-\cos t, \sin t)$. En
$\intoo{0}{\frac\pi2}$ todos los puntos son regulares con tangente
dirigida por $(-\cos t, \sin t)$; y en $t = 0$ (el punto $(1,0)$) la
velocidad se anula: un \emph{punto de retroceso}\index{punto de
retroceso}, donde la curva se invierte a lo largo de la dirección
tangente $(-1, 0)\cdot(\pm)$ --- por simetría, los cuatro
\omterm{ex:b1:curves:astroid}{puntos de retroceso} están en $(\pm1, 0), (0, \pm1)$.
\end{example}

\begin{example}[Un ocho, estudiado al completo]
\label{ex:b1:curves:lissajous}
$x(t) = \sin t$, $y(t) = \sin 2t$ (una \emph{curva de Lissajous}).
\emph{Simetrías}: $M(t + \pi) = (-x(t), y(t))$ (reflexión respecto
del eje $y$), $M(-t) = (-x(t), -y(t))$ (simetría central),
$M(\pi - t) = (x(t), -y(t))$ (reflexión respecto del eje $x$): basta
estudiar $t \in \intcc{0}{\frac\pi2}$ y desplegar.
\emph{Variaciones}: $x' = \cos t \geq 0$ en todo el recorrido,
mientras que $y' = 2\cos 2t$ es positiva antes de $t = \frac\pi4$ y
negativa después: el arco sube hacia la derecha hasta la cima
$\bigl(\frac{\sqrt2}2,\ 1\bigr)$ en $t = \frac\pi4$ (tangente
horizontal) y desciende después hacia la derecha hasta $(1, 0)$ en
$t = \frac\pi2$, donde $x' = 0 \neq y'$: tangente vertical.
\emph{Punto doble}: $M(0) = M(\pi) = (0,0)$, con dos velocidades
distintas
\[
f'(0) = (1,\ 2), \qquad f'(\pi) = (-1,\ 2) :
\]
dos ramas \emph{regulares} que se cruzan en el origen con ángulos
distintos --- un punto doble, no un punto singular: cada paso es
perfectamente suave, y los dos pasos simplemente comparten el lugar.
La curva entera es el ocho de más abajo.
\end{example}

\section{Curvas polares}

\begin{definition}\label{def:b1:curves:polar}
Una \emph{curva polar}\index{curva polar} viene dada por
$r = r(\theta)$: el punto de parámetro $\theta$ es
\[
M(\theta) = r(\theta)\,\vec u(\theta),
\qquad \vec u(\theta) = (\cos\theta, \sin\theta).
\]
Con $\vec v(\theta) = (-\sin\theta, \cos\theta) = \vec
u\,'(\theta)$, la velocidad es
\[
M'(\theta) = r'(\theta)\, \vec u(\theta) + r(\theta)\, \vec
v(\theta) .
\]
Consecuencias: donde $r \neq 0$, el punto es regular y la tangente
forma con el radio el ángulo $V$ dado por $\tan V = \frac{r}{r'}$ (el
ángulo entre $M'$ y $\vec u$); y donde $r(\theta_0) = 0$, la curva
pasa por el origen \emph{con tangente el radio $\theta = \theta_0$}
(dirección $\vec u(\theta_0)$, leída en
$M'(\theta_0) = r'(\theta_0)\vec u(\theta_0)$, o del límite de las
cuerdas: la cuerda de $O$ a $M(\theta)$ va dirigida por el propio
$\vec u(\theta)$, que tiende a $\vec u(\theta_0)$ --- así que la
regla vale incluso cuando $r'(\theta_0) = 0$ y el punto es singular,
que es por lo que los \omterm{ex:b1:curves:astroid}{puntos de retroceso} polares en el origen, como
el de la cardioide, obtienen su tangente gratis).
\end{definition}

\begin{example}[La cardioide]\label{ex:b1:curves:cardioid}
$r(\theta) = 1 + \cos\theta$.\index{cardioide} Simetría:
$r(-\theta) = r(\theta)$: reflexión respecto del eje $x$; estúdiese
$\theta \in \intcc{0}{\pi}$. $r$ decrece de $2$ a $0$; y en
$\theta = \pi$, $r = 0$: la curva alcanza el origen tangencialmente
al radio $\theta = \pi$ (el eje $x$), formando ahí un
\omterm{ex:b1:curves:astroid}{punto de retroceso} --- la punta del corazón. Tangente en
$\theta = 0$: $r' = 0$, luego $\tan V = \infty$: perpendicular al eje.

El ángulo $V$ merece una lectura más. En $\theta = \frac\pi2$:
$r = 1$ y $r' = -1$, luego $\tan V = \frac{r}{r'} = -1$: la tangente
forma tres cuartos de ángulo recto con el radio saliente --- la curva
ya se está doblando de vuelta hacia su \omterm{ex:b1:curves:astroid}{punto de retroceso}. La
fórmula $\tan V = r/r'$ entrega direcciones tangentes a lo largo de
toda la curva sin calcular en absoluto $M'(\theta)$: es el análogo
polar de leer una pendiente.
\end{example}

\begin{example}[De polar a cartesiana: un círculo escondido]
\label{ex:b1:curves:polarcircle}
¿Qué es la \omterm{def:b1:curves:polar}{curva polar} $r = 2\cos\theta$? Multiplíquese por $r$:
$r^2 = 2r\cos\theta$, es decir, $x^2 + y^2 = 2x$, es decir,
\[
(x - 1)^2 + y^2 = 1 :
\]
el círculo de centro $(1, 0)$ y radio $1$, que pasa por el origen. La
contabilidad importa: cuando $\theta$ recorre
$\intoc{-\frac\pi2}{\frac\pi2}$, todo el círculo se barre exactamente
una vez ($r$ se anula en los dos extremos), y en
$\theta = \pm\frac\pi2$ la regla «tangente en el origen a lo largo
del radio $\theta = \theta_0$» da ahí una tangente \emph{vertical}
--- de acuerdo con la geometría, ya que el eje vertical es en efecto
tangente a este círculo en $O$. Para $\theta$ más allá de ese rango,
$r < 0$ vuelve a recorrer el mismo círculo: un recordatorio de que
una \omterm{def:b1:curves:polar}{curva polar} es un objeto \emph{parametrizado}, con permiso para
pasar sobre sí misma.
\end{example}

\section{Longitud de arco}

\begin{definition}[Longitud de un arco]\label{def:b1:curves:arclength}
La \emph{longitud}\index{longitud de arco} de un arco de clase $C^1$
$f \colon \intcc{a}{b} \to \R^2$ es la \omterm{thm:b1:integration:def}{integral} de la rapidez:
\[
L = \int_a^b \norm{f'(t)}\,\dd t
= \int_a^b \sqrt{x'(t)^2 + y'(t)^2}\;\dd t .
\]
(Motivación: en un \omterm{prop:b1:reals:intervals}{intervalo} pequeño,
$M(t + h) \approx M(t) + h f'(t)$, de modo que el arco está próximo a
una poligonal cuyas longitudes de segmento suman una suma de Riemann
de $\norm{f'}$, \cref{thm:b1:integration:riemann}.) Para una
\omterm{def:b1:curves:polar}{curva polar} $r = r(\theta)$, la velocidad $r'\vec u + r\vec v$
tiene componentes \omterm{def:b1:euclid:orthogonal}{ortogonales}, luego
\[
L = \int_{\theta_1}^{\theta_2}
\sqrt{r'(\theta)^2 + r(\theta)^2}\;\dd\theta .
\]
La longitud no depende de la parametrización elegida (monótona y de
clase $C^1$): sustituir $t = \varphi(s)$ en la \omterm{thm:b1:integration:def}{integral}
(\cref{thm:b1:integration:parts}) multiplica $f'$ por $\varphi'$ y
$\dd t$ por $\varphi'^{-1}$.
\end{definition}

\begin{example}[Comprobación de coherencia: el círculo]\label{ex:b1:curves:circlelength}
Para $f(t) = (R\cos t, R\sin t)$ en $\intcc{0}{2\pi}$:
$\norm{f'} = R$, luego $L = 2\pi R$ --- la definición devuelve la
circunferencia. Prueba de reparametrización: la \omterm{def:b1:logic:map}{aplicación}
$g(t) = (R\cos 2t, R\sin 2t)$ en $\intcc{0}{\pi}$ dibuja el
\emph{mismo} círculo al doble de rapidez $\norm{g'} = 2R$, y
\[
\int_0^{\pi} 2R\,\dd t = 2\pi R
\]
otra vez: la mitad de tiempo, el doble de rapidez, la misma
\omterm{def:b1:curves:arclength}{longitud} --- la invariancia prometida en la definición, vista una
vez sobre números. (Ejecutar $g$ en todo $\intcc{0}{2\pi}$ daría
$4\pi R$: una curva recorrida dos veces es dos veces más larga
\emph{como trayecto}; la \omterm{def:b1:curves:arclength}{longitud} mide el camino parametrizado, y
llevar bien la cuenta del \omterm{prop:b1:reals:intervals}{intervalo} es parte del cálculo.) Para la
astroide $(\cos^3t, \sin^3t)$:
$\norm{f'} = 3\abs{\sin t\cos t} = \tfrac32\abs{\sin 2t}$ y, por
simetría, $L = 4\int_0^{\pi/2} \tfrac32\sin 2t\,\dd t = 6$: una curva
dibujada dentro del círculo unidad, de \omterm{def:b1:curves:arclength}{longitud} $6 < 2\pi$. El
problema del fin de semana mide el arco más famoso de todos.
\end{example}

\section{Cónicas}

\begin{definition}[Definición por foco y directriz]\label{def:b1:curves:conics}
Fíjense un punto $F$ (el \emph{foco}), una recta $D$ que no pase por
$F$ (la \emph{directriz}) y $e > 0$ (la \emph{excentricidad}). La
\emph{cónica}\index{cónica} de esos datos es
\[
\mathcal{C} = \{M : d(M, F) = e\; d(M, D)\}:
\]
una \emph{elipse} para $e < 1$, una \emph{parábola} para $e = 1$ y
una \emph{hipérbola} para $e > 1$. (El círculo aparece como el límite
degenerado $e \to 0$.)
\end{definition}

\begin{example}[La definición, comprobada en una parábola]
\label{ex:b1:curves:parabolacheck}
Tómese la parábola $y^2 = 4x$, es decir, $2p = 4$: foco $F = (1, 0)$
y directriz $D : x = -1$ (la forma reducida
del~\cref{thm:b1:curves:reduced} los coloca en $\pm\frac p2$). En el
punto $M = (1, 2)$ de la curva:
\[
MF = \sqrt{(1-1)^2 + 2^2} = 2,
\qquad
d(M, D) = 1 - (-1) = 2 :
\]
iguales, como exige $e = 1$. En $M' = (4, 4)$:
$MF' = \sqrt{9 + 16} = 5$ y $d(M', D) = 5$ otra vez. La definición
por foco y directriz no es una abstracción: es una pareja de
distancias que se puede medir en cualquier punto, y el álgebra de las
ecuaciones reducidas no es más que esa medición hecha de una vez por
todas.
\end{example}

\begin{theorem}[Ecuaciones reducidas]\label{thm:b1:curves:reduced}
En un sistema de referencia \omterm{def:b1:euclid:orthogonal}{ortonormal} bien elegido:
\[
\text{elipse: } \frac{x^2}{a^2} + \frac{y^2}{b^2} = 1
\quad (a \geq b > 0),
\qquad
\text{hipérbola: } \frac{x^2}{a^2} - \frac{y^2}{b^2} = 1,
\qquad
\text{parábola: } y^2 = 2px,
\]
con, para la elipse: focos en $(\pm c, 0)$, $c = \sqrt{a^2 - b^2}$,
$e = \frac ca$ y la caracterización bifocal $MF + MF' = 2a$; y para
la hipérbola: $c = \sqrt{a^2 + b^2}$, $e = \frac ca$,
$\abs{MF - MF'} = 2a$, con asíntotas $y = \pm\frac ba x$.
\end{theorem}

\begin{proof}
Tómese el foco en el origen y la directriz vertical, $x = -h$
($h > 0$): la condición $MF^2 = e^2 d(M, D)^2$ dice
$x^2 + y^2 = e^2 (x + h)^2$. Para $e = 1$: $y^2 = 2hx + h^2$, una
parábola tras el desplazamiento $x \mapsto x - \frac h2$ (con
$p = h$). Y para $e \neq 1$, completando el cuadrado en $x$,
\[
(1 - e^2)\Bigl(x - \frac{e^2 h}{1 - e^2}\Bigr)^{\!2} + y^2
= e^2h^2 + \frac{e^4 h^2}{1 - e^2}
= \frac{e^2 h^2}{1 - e^2} .
\]
Póngase $X = x - \frac{e^2h}{1-e^2}$ (un desplazamiento del sistema
de referencia). Cuando $e < 1$, divídase por el miembro derecho, que
es positivo: $\frac{X^2}{a^2} + \frac{y^2}{b^2} = 1$ con
$a = \frac{eh}{1 - e^2}$, $b = \frac{eh}{\sqrt{1-e^2}}$; y cuando
$e > 1$, los dos miembros de la fórmula cambian de signo como
corresponde y la misma división da
$\frac{X^2}{a^2} - \frac{y^2}{b^2} = 1$ con
$a = \frac{eh}{e^2 - 1}$, $b = \frac{eh}{\sqrt{e^2 - 1}}$. Los
valores indicados de $c$ se siguen ($c^2 = a^2 - b^2$ o $a^2 + b^2$
da $c = ea$ en los dos casos, colocando el foco correctamente), y las
propiedades bifocales son comprobaciones directas sobre las
ecuaciones reducidas. Con detalle para la elipse, con $F' = (c, 0)$ y
$M = (X, y)$ sobre la curva:
\[
MF'^2 = (X - c)^2 + y^2
= X^2 - 2cX + c^2 + b^2\Bigl(1 - \frac{X^2}{a^2}\Bigr)
= \frac{c^2}{a^2}X^2 - 2cX + a^2
= (a - eX)^2 ,
\]
usando $c^2 + b^2 = a^2$ y $e = \frac ca$; y como $\abs X \leq a$ y
$e < 1$, se tiene $a - eX > 0$, luego $MF' = a - eX$ sin extraer
jamás ninguna raíz cuadrada. El cálculo espejo da $MF = a + eX$, de
donde $MF + MF' = 2a$, constante. Para la hipérbola, la misma álgebra
da $MF = \abs{a + eX}$ y $MF' = \abs{eX - a}$, con diferencia
$\pm2a$ según la rama.
\end{proof}

\begin{example}\label{ex:b1:curves:conicexample}
$\frac{x^2}{25} + \frac{y^2}{9} = 1$: elipse, $a = 5$, $b = 3$,
$c = 4$: focos $(\pm4, 0)$, excentricidad $\frac45$. Su
parametrización: $(5\cos t, 3\sin t)$ --- un círculo estirado de
forma anisótropa; y la suma de las distancias a los focos de
cualquiera de sus puntos es $10$.
\end{example}

\begin{example}[Leer una órbita en su ecuación polar]
\label{ex:b1:curves:orbit}
La \omterm{def:b1:curves:conics}{cónica} polar $r = \dfrac{1}{1 + \frac12\cos\theta}$
(\cref{exo:b1:curves:7} con $p = 1$, $e = \frac12$) es una elipse con
un \emph{foco en el origen} --- la geometría de una órbita planetaria
con el sol en $O$. Extráigase todo de $p$ y de $e$: por la reducción
del~\cref{thm:b1:curves:reduced}, $a = \dfrac{p}{1 - e^2} =
\dfrac{1}{3/4} = \dfrac43$ y $c = ea = \dfrac23$. Y los dos ápsides
lo comprueban sin ninguna teoría:
\[
r(0) = \frac{1}{3/2} = \frac23 = a - c
\quad (\text{perihelio}),
\qquad
r(\pi) = \frac{1}{1/2} = 2 = a + c
\quad (\text{afelio}),
\]
y su suma $\frac23 + 2 = \frac83 = 2a$ recupera el eje mayor. La
forma polar es la natural siempre que un foco esté físicamente
distinguido; y la forma cartesiana reducida, siempre que lo estén los
ejes de simetría. Convertir una en otra es exactamente lo que hace el
cálculo de completar cuadrados del teorema.

La excentricidad como un dial, por último: manténgase $p = 1$ y
gírese $e$ en $r = \frac{1}{1 + e\cos\theta}$. En $e = 0$: el círculo
$r = 1$. En $e = \frac12$: la elipse recién estudiada, con $r$
oscilando entre $\frac23$ y $2$. En $e = 1$: $r(\theta) \to \infty$
cuando $\theta \to \pi$ --- la curva ya no se cierra: una parábola,
con su punto más lejano empujado al infinito. Y en $e = 2$: el
denominador se anula en $\cos\theta = -\frac12$, y solo sobrevive
$\theta \in \intoo{-\frac{2\pi}3}{\frac{2\pi}3}$: una rama de
hipérbola, que se escapa por dos direcciones asintóticas. Una
fórmula, toda la familia de las \omterm{def:b1:curves:conics}{cónicas} y el punto de transición
$e = 1$ visible como el momento en que el denominador alcanza el cero
por primera vez.
\end{example}

\begin{remark}[Errores frecuentes]
\label{rem:b1:curves:pitfalls}
\emph{Los puntos dobles no son puntos singulares}: en un
autocruce, cada rama es regular; «singular» se refiere a
$f'(t_0) = 0$ para un valor del parámetro
(\cref{ex:b1:curves:lissajous} frente a los \omterm{ex:b1:curves:astroid}{puntos de retroceso} de la
astroide).
\emph{Tangente vertical frente a \omterm{ex:b1:curves:astroid}{punto de retroceso}}:
$x'(t_0) = 0 \neq y'(t_0)$ es un punto regular con tangente
vertical; solo $x' = y' = 0$ exige el desarrollo de orden superior.
\emph{Los radios negativos se dibujan en el radio opuesto}: para
$r(\theta) < 0$, el punto es $-\abs{r}\,\vec u(\theta)$, de ángulo
$\theta + \pi$ (\cref{exo:b1:curves:5}); olvidarlo pierde los lazos
interiores o duplica curvas.
\emph{La \omterm{def:b1:curves:arclength}{longitud de arco} integra $\norm{f'}$, no $f'$}: pártase la
\omterm{thm:b1:integration:def}{integral} en los ceros de la rapidez --- para la astroide, integrar
$\frac32\sin 2t$ en un período completo sin valores absolutos da $0$,
no $6$.
\emph{Las \omterm{def:b1:curves:conics}{cónicas} hay que reducirlas antes de leerlas}: en
$9x^2 + 25y^2 - 36x - 50y - 164 = 0$ no se ven ni los ejes, ni el
centro, ni la excentricidad hasta completar los cuadrados
(\cref{exo:b1:curves:6}); y que la parte cuadrática se anule en una
variable significa parábola, no «elipse degenerada».
\end{remark}

\begin{remark}[Adónde van estas curvas]
\label{rem:b1:curves:whereused}
Las \omterm{def:b1:curves:def}{curvas parametrizadas} son el idioma de la mecánica: las
trayectorias son curvas, los vectores velocidad son vectores
tangentes y la \omterm{def:b1:curves:arclength}{longitud de arco} de la~\cref{def:b1:curves:arclength}
es la distancia recorrida. Las \omterm{def:b1:curves:conics}{cónicas} reaparecen allí donde actúa
una ley del inverso del cuadrado --- las órbitas planetarias son
elipses con el sol en un foco. En el~\cref{ch:b1:multivar}, las curvas
se convierten en los \omterm{def:b1:logic:sets}{conjuntos} de nivel de las funciones de dos
variables, y la tangente de este capítulo se encuentra con el
gradiente del siguiente. El volumen del segundo año añade la
curvatura y la forma canónica local de una curva; y el problema del
fin de semana de más abajo ya extrae, solo con las herramientas de
este año, todo lo que el siglo XVII sabía sobre su curva más célebre.
\end{remark}

\begin{remark}[Perspectivas dentro del libro 3]
\label{rem:b1:curves:perspectives}
Este capítulo consume toda la primera mitad del volumen y alimenta el
último capítulo. \emph{Consumido}: los vectores tangentes son
\omterm{def:b1:derivative:def}{derivadas} (\cref{ch:b1:derivative}), las \omterm{def:b1:curves:arclength}{longitudes de arco} y las
áreas son \omterm{thm:b1:integration:def}{integrales} (\cref{ch:b1:integration}), los
\omterm{ex:b1:curves:astroid}{puntos de retroceso} los resuelven los desarrollos de Taylor
(\cref{ch:b1:taylor}), la tautócrona es una ecuación diferencial
\omterm{def:b1:linmaps:def}{lineal} (\cref{ch:b1:diffeq}) y toda distancia y todo ángulo son
euclídeos (\cref{ch:b1:euclid}). \emph{Alimentado}: en
el~\cref{ch:b1:multivar}, una curva $t \mapsto (x(t), y(t))$ dibujada
dentro de un \omterm{def:b1:logic:sets}{conjunto} de nivel $\{f = c\}$ se deriva con la regla de
la cadena, y la identidad resultante
$\langle \nabla f,\ f'(t)\rangle = 0$ casa los vectores tangentes de
este capítulo con los gradientes del siguiente --- la recta tangente
de la curva y la dirección normal de la superficie son el mismo
cálculo visto desde las dos orillas.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:curves:1}
Estúdiese y esbócese la curva $x(t) = t^2$, $y(t) = t^3$ (simetrías,
variaciones, comportamiento en el punto singular $t = 0$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:curves:2}
Para la cicloide $x(t) = t - \sin t$, $y(t) = 1 - \cos t$ (la
trayectoria de un punto de una rueda de radio $1$ que rueda):
identifíquense la simetría de traslación por el período, los puntos
singulares y la dirección tangente en $t = 0$ \emph{(desarróllense
$x$ e $y$ hasta los primeros órdenes no nulos)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:curves:3}
Dese la recta tangente a la curva $(\cos^3 t, \sin^3 t)$ en
$t = \frac\pi4$, y compruébese que el segmento de esa tangente que
recortan los ejes tiene \omterm{def:b1:curves:arclength}{longitud} $1$ --- una famosa propiedad de la
astroide (cierta en todo punto regular).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:curves:4}
Esbócense las \omterm{def:b1:curves:polar}{curvas polares} $r = \cos 2\theta$ (la rosa de cuatro
pétalos) y $r = \frac{1}{\cos\theta}$ en
$\intoo{-\frac\pi2}{\frac\pi2}$ (reconózcase una recta).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:curves:5}
Estúdiese la \omterm{def:b1:curves:polar}{curva polar} $r = 1 + 2\cos\theta$ (un caracol de
Pascal): dominio donde $r \geq 0$ frente a $r < 0$ (los puntos
dibujados con radio negativo caen en el radio opuesto), paso por el
origen, lazo \omterm{def:b1:topology:closure}{interior} y boceto.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:curves:6}
Identifíquese la \omterm{def:b1:curves:conics}{cónica} $9x^2 + 25y^2 - 36x - 50y - 164 = 0$:
redúzcase completando cuadrados y dense el centro, los semiejes, los
focos y la excentricidad.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:curves:7}
Demuéstrese que la ecuación polar de una \omterm{def:b1:curves:conics}{cónica} con foco en el origen
es
\[
r = \frac{p}{1 + e\cos\theta}
\]
(con directriz vertical a distancia $\frac pe$ a la derecha del
foco). ¿Qué valores de $\theta$ se permiten cuando $e > 1$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:curves:8}
De la propiedad bifocal $MF + MF' = 2a$, dedúzcase la «construcción
del jardinero» de la elipse, y demuéstrese que la tangente en $M$
forma ángulos iguales con $MF$ y $MF'$ \emph{(propiedad de reflexión;
derívese $\norm{M(t) - F} + \norm{M(t) - F'} = 2a$ e interprétese la
suma nula de los productos con los vectores unitarios)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:curves:9}
La \emph{lemniscata de Bernoulli} es la \omterm{def:b1:curves:polar}{curva polar}
$r^2 = \cos 2\theta$ (tómese $r = \sqrt{\cos2\theta}$ donde esté
definida).
\begin{enumerate}
  \item Dense su dominio, sus simetrías y sus tangentes en el
        origen, y esbócese.
  \item Demuéstrese que es el lugar de los puntos $M$ con
        $MF \cdot MF' = \frac12$, donde
        $F, F' = \bigl(\pm\frac{1}{\sqrt2}, 0\bigr)$.
        \emph{(Calcúlese $MF^2\,MF'^2$ en \omterm{prop:b1:vspaces:coordinates}{coordenadas} polares.)}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:curves:10}
Calcúlese la \omterm{def:b1:curves:arclength}{longitud} total de la cardioide $r = 1 + \cos\theta$
con la fórmula polar de la~\cref{def:b1:curves:arclength} \emph{(la
identidad $1 + \cos\theta = 2\cos^2\frac\theta2$ convierte la raíz
cuadrada en $2\abs{\cos\frac\theta2}$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:curves:11}
Véase que la tangente a la elipse $(a\cos t, b\sin t)$ en el punto de
parámetro $t$ tiene ecuación
\[
\frac{x\cos t}{a} + \frac{y\sin t}{b} = 1 ,
\]
y dedúzcase la tangente a $\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1$ en
un punto $(x_0, y_0)$ de la elipse:
$\frac{x\,x_0}{a^2} + \frac{y\,y_0}{b^2} = 1$ (la regla del
«desdoblamiento de los cuadrados»).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:curves:12}
La \emph{espiral logarítmica} es la \omterm{def:b1:curves:polar}{curva polar}
$r = \eu^{k\theta}$ ($k > 0$ fijo, $\theta \in \R$).
\begin{enumerate}
  \item Véase que el ángulo $V$ entre el radio y la tangente es
        \emph{constante} ($\tan V = \frac1k$) --- la espiral cruza
        todos los radios con el mismo ángulo.
  \item Véase que girar la espiral un ángulo $c$ la lleva sobre su
        homotética de razón $\eu^{-kc}$: toda rotación de la espiral
        es una ampliación de ella misma (autosemejanza).
  \item Calcúlese la \omterm{def:b1:curves:arclength}{longitud} del arco
        $\theta \in \intoc{-\infty}{\theta_0}$ (como límite de las
        \omterm{def:b1:curves:arclength}{longitudes} sobre $\intcc{A}{\theta_0}$, con
        $A \to -\infty$) y obsérvese que es finita: una curva que da
        infinitas vueltas en torno al origen, de \omterm{def:b1:curves:arclength}{longitud} finita.
\end{enumerate}
\end{exercise}

\section{Problema: la cicloide, reina de las curvas}

\begin{problem}\label{pb:b1:curves:1}
Una rueda de radio $R$ rueda sin deslizar a lo largo del eje $x$; el
punto de la llanta que estaba inicialmente en el origen
\omterm{def:b1:matrices:transpose}{traza} la \emph{cicloide}. El siglo XVII se peleó por esta curva
---Galileo pesó recortes de papel con su forma, Wren la midió,
Roberval calculó su área, Huygens construyó relojes sobre ella--- y
todos sus resultados están al alcance de este capítulo. Demostramos
los cuatro clásicos: la construcción de la tangente, la \omterm{def:b1:curves:arclength}{longitud}
$8R$ de Wren, el área $3\pi R^2$ y la propiedad tautócrona de
Huygens.\index{cicloide}\index{tautócrona}

\medskip
\noindent\textbf{Parte I --- Rodadura y tangente.}
\begin{enumerate}
  \item Tras girar la rueda un ángulo $t$, su centro está en
        $\Omega(t) = (Rt, R)$ (rodadura sin deslizamiento: la
        distancia de contacto $=$ el arco rodado). Véase que el
        punto marcado está en
        \[
        M(t) = \bigl(R(t - \sin t),\; R(1 - \cos t)\bigr).
        \]
  \item Calcúlese $f'(t)$ y véase que
        $\norm{f'(t)} = 2R\,\abs{\sin\frac t2}$; localícense los
        puntos singulares (\omterm{ex:b1:curves:astroid}{puntos de retroceso} --- cf.\
        \cref{exo:b1:curves:2}) y la cima de cada arco.
  \item Sean $C(t) = (Rt, 0)$ el punto de contacto y
        $T(t) = (Rt, 2R)$ lo alto de la rueda. Demuéstrese que para
        $0 < t < 2\pi$ el vector $M - C$ es \emph{normal} a la curva
        en $M(t)$ y el vector $T - M$ es \emph{tangente}: para
        trazar la tangente a una cicloide, únase el punto con lo
        alto de su círculo rodante. \emph{(Factorícese todo a través
        de $\sin\frac t2$ y $\cos\frac t2$.)}
  \item Interprétese la pregunta 3 de forma cinemática: el punto de
        contacto es el \emph{centro instantáneo de rotación}, y la
        rapidez de $M$ es igual a su distancia a $C$ (para velocidad
        angular unidad). Compruébese
        $\norm{M - C} = 2R\abs{\sin\frac t2}$.
  \item Demuéstrese la relación altura--rapidez
        \[
        \norm{f'(t)}^2 = 2R\,y(t) :
        \]
        en una cicloide recorrida a velocidad angular unidad, la
        rapidez en cada punto es exactamente la de la caída \omterm{def:b1:vspaces:free}{libre}
        para un descenso igual a la altura actual. (Guárdese esto
        para la parte IV.)
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El teorema de Wren: el arco tiene
\omterm{def:b1:curves:arclength}{longitud} $8R$.}
\begin{enumerate}[resume]
  \item Con la~\cref{def:b1:curves:arclength}, calcúlese la
        \omterm{def:b1:curves:arclength}{longitud} de un arco:
        \[
        L = \int_0^{2\pi} 2R\sin\frac t2\,\dd t = 8R
        \]
        (teorema de Wren, 1658). Cuatro diámetros de rueda, y
        ningún $\pi$ por ninguna parte.
  \item Calcúlese la \omterm{def:b1:curves:arclength}{longitud de arco} desde el
        \omterm{ex:b1:curves:astroid}{punto de retroceso}: $s(t) = 4R\bigl(1 - \cos\frac t2\bigr)$,
        y compruébese $s(2\pi) = 8R$.
  \item Mídase ahora el arco desde el \emph{ápice} $t = \pi$:
        $\sigma(t) = \abs{4R\cos\frac t2}$. Demuéstrese la relación
        intrínseca
        \[
        \sigma^2 = 8R\,\bigl(2R - y\bigr) :
        \]
        la distancia de arco al cuadrado desde lo alto es
        proporcional al descenso de altura por debajo de lo alto.
  \item Comprobaciones: recupérese de la pregunta 8 que el medio
        arco del ápice al \omterm{ex:b1:curves:astroid}{punto de retroceso} tiene \omterm{def:b1:curves:arclength}{longitud} $4R$, y
        compárese con el cálculo de la astroide
        del~\cref{ex:b1:curves:circlelength} --- las dos curvas
        tienen \omterm{def:b1:curves:arclength}{longitudes} algebraicas, sin $\pi$; explíquese qué lo
        hace posible aunque las dos estén construidas a partir de
        círculos. \emph{(Mírese la forma de $\norm{f'}$.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El área de Roberval: $3\pi R^2$.}
\begin{enumerate}[resume]
  \item Justifíquese que el área entre un arco y el suelo es
        $A = \int_0^{2\pi} y(t)\,x'(t)\,\dd t$ \emph{(el cambio
        $x = x(t)$ en $\int y\,\dd x$,
        \cref{thm:b1:integration:parts}; $x$ es creciente)}.
  \item Calcúlese
        \[
        A = R^2\int_0^{2\pi}(1 - \cos t)^2\,\dd t = 3\pi R^2 :
        \]
        exactamente \emph{tres} veces el área de la rueda --- la
        razón que Galileo había conjeturado pesando.
  \item El mismo método para la astroide $(\cos^3t, \sin^3t)$:
        véase que el área encerrada es $\frac{3\pi}8$
        \emph{(linealícese $\sin^4 t\cos^2 t$; en un período
        completo solo sobrevive el término constante)}.
  \item Compruébese la fórmula de la pregunta 10 sobre la
        semicircunferencia unidad superior $(\cos t, \sin t)$, con
        $t$ de $\pi$ a $0$: ¿devuelve $\frac\pi2$?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La tautócrona de Huygens.} Dese la
vuelta al arco: una cuenta sin rozamiento se desliza, bajo la
gravedad $g$, dentro del cuenco cicloidal
\[
x(t) = R(t + \sin t), \qquad y(t) = R(1 - \cos t)
\qquad (t \in \intcc{-\pi}{\pi}),
\]
cuyo punto más bajo es el origen ($y$ medida hacia arriba).
\begin{enumerate}[resume]
  \item Calcúlense la rapidez $\norm{f'(t)} = 2R\cos\frac t2$ y la
        \omterm{def:b1:curves:arclength}{longitud de arco} desde el fondo $s(t) = 4R\sin\frac t2$, y
        demuéstrese la identidad clave
        \[
        y = \frac{s^2}{8R} .
        \]
  \item La cuenta soltada en reposo desde el punto de parámetro
        $t_0 > 0$ obedece la conservación de la energía: si
        $s(\tau)$ denota su posición de arco en el instante $\tau$,
        entonces
        $\frac12\bigl( \frac{\dd s}{\dd\tau}\bigr)^2 + g\,y = g\,y_0$.
        Reescríbase esto, usando la pregunta 14, como
        \[
        \Bigl(\frac{\dd s}{\dd\tau}\Bigr)^{\!2}
        = \frac{g}{4R}\,\bigl(s_0^2 - s^2\bigr),
        \qquad s_0 = s(t_0).
        \]
  \item Derívese respecto de $\tau$ y obténgase el oscilador
        armónico
        \[
        \frac{\dd^2 s}{\dd\tau^2} = -\frac{g}{4R}\,s ;
        \]
        resuélvase con el~\cref{thm:b1:diffeq:homogeneous2} y las
        condiciones iniciales: $s(\tau) = s_0\cos(\omega\tau)$,
        $\omega = \sqrt{g/4R}$.
  \item Dedúzcase la \emph{propiedad tautócrona} (Huygens, 1659):
        el tiempo hasta llegar al fondo,
        \[
        T_{\downarrow} = \frac{\pi}{2\omega}
        = \pi\sqrt{\frac Rg}\,,
        \]
        no depende del punto de partida --- dos cuentas soltadas a
        la vez desde dos alturas cualesquiera del cuenco llegan a la
        vez.
  \item Compruébese por sustitución que
        $s(\tau) = s_0\cos\omega\tau$ cumple exactamente la ecuación
        de energía de primer orden de la pregunta 15 (no solo la
        \omterm{def:b1:derivative:def}{derivada}), y explíquese en una frase por qué un péndulo
        circular es solo \emph{aproximadamente} isócrono mientras
        que la cicloide lo es exactamente.
  \item Numéricamente: ¿qué radio $R$ hace que el tiempo de
        descenso sea exactamente un segundo ($g = 9.81$)?
        Obsérvese lo cerca que está la respuesta de un metro, y
        cómo se compara el período completo de oscilación
        $2\pi\sqrt{4R/g}$ con la fórmula del péndulo de pequeñas
        oscilaciones $2\pi\sqrt{\ell/g}$ para $\ell = 4R$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dividendos, y síntesis.}
\begin{enumerate}[resume]
  \item Aplíquese la regla de la tangente de la pregunta 3 en
        $t = \frac\pi2$ (tómese $R = 1$): calcúlense $M$, $T$ y la
        dirección de $MT$, y compruébese contra $f'(\frac\pi2)$.
  \item (Trocoides) Márquese en su lugar un punto a distancia $d$
        del centro ($d \neq R$): la curva es
        $x = Rt - d\sin t$, $y = R - d\cos t$. Véase que para
        $d < R$ la curva es regular en todas partes y no presenta
        ningún comportamiento singular ($x' > 0$: avanza),
        mientras que para $d > R$ la abscisa $x'$ cambia de signo y
        la curva hace lazos --- la rueda de tren con pestaña, cuyos
        puntos del reborde viajan \emph{hacia atrás}.
  \item (Anticipo de la braquistócrona) Desde el
        \omterm{ex:b1:curves:astroid}{punto de retroceso} $(\pi R, 2R)$ del cuenco hasta el fondo,
        compárese el tiempo de descenso por la cicloide
        $\pi\sqrt{R/g}$ con el tiempo por la rampa recta que une los
        mismos puntos \emph{(aceleración constante $g\sin\alpha$ a
        lo largo de la cuerda)}: véase que la cuerda tarda
        $\sqrt{\pi^2 + 4}\,\sqrt{R/g} \approx 3.72\sqrt{R/g}$. La
        curva le gana a la recta --- de hecho, es la más rápida de
        todas las curvas, un resultado del cálculo de variaciones.
  \item Véase que en el arco ($0 < t < 2\pi$),
        \[
        \frac{\dd y}{\dd x} = \cot\frac t2,
        \qquad
        \frac{\dd^2 y}{\dd x^2} =
        -\frac{1}{4R\sin^4\frac t2} < 0 :
        \]
        el arco es cóncavo, con tangentes verticales exactamente en
        los \omterm{ex:b1:curves:astroid}{puntos de retroceso}.
  \item Recupérese geométricamente la tangente vertical en el
        \omterm{ex:b1:curves:astroid}{punto de retroceso} del~\cref{exo:b1:curves:2}: calcúlese la
        dirección límite de la cuerda $MT$ de la pregunta 3 cuando
        $t \to 0^{+}$, sin ningún desarrollo.
  \item Síntesis, en cuatro frases: qué tres teoremas con nombre
        demostró este problema (con sus números $8R$,
        $3\pi R^2$, $\pi\sqrt{R/g}$ y sus autores); qué único
        recurso de cálculo (las factorizaciones con $\sin\frac t2$
        y $\cos\frac t2$) movió las partes I, II y IV; cómo la
        relación intrínseca $y = s^2/8R$ convirtió la geometría en
        una ecuación diferencial \omterm{def:b1:linmaps:def}{lineal}; y en qué capítulo de este
        libro se apoyó cada parte.
\end{enumerate}
\end{problem}
