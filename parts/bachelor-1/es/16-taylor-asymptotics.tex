\chapter{Fórmulas de Taylor y desarrollos asintóticos}\label{ch:b1:taylor}

Cerca de un punto, una función regular vale tanto como un
\omterm{def:b1:poly:def}{polinomio} --- con un error controlable. Las fórmulas de Taylor lo
hacen exacto en tres versiones (resto \omterm{thm:b1:integration:def}{integral}, resto de Lagrange,
resto de Young), y los \emph{desarrollos asintóticos} resultantes,
manipulados algebraicamente, se convierten en la herramienta más
afilada del análisis elemental: límites, equivalentes, comportamiento
local, asíntotas.

\section{Notación de comparación}

\begin{definition}[Notación de Landau]\label{def:b1:taylor:landau}
Sean $f, g$ definidas cerca de $x_0$ ($x_0 \in \R$ o $\pm\infty$). Se
escribe, cuando $x \to x_0$:\index{notación de Landau}
\begin{itemize}
  \item $f = o(g)$ («o pequeña») cuando $f = \varepsilon g$ con
        $\varepsilon(x) \to 0$;
  \item $f = O(g)$ («O grande») cuando $f = u g$ con $u$ acotada
        cerca de $x_0$;
  \item $f \sim g$ («equivalente»\index{funciones equivalentes})
        cuando $f = (1 + \varepsilon) g$ con $\varepsilon \to 0$ ---
        equivalentemente, $f - g = o(g)$.
\end{itemize}
La misma notación se aplica a las sucesiones ($n \to \infty$).
\end{definition}

\begin{proposition}[Reglas]\label{prop:b1:taylor:landaurules}
Cuando $x \to x_0$:
\begin{enumerate}
  \item $\sim$ es una \omterm{def:b1:logic:equiv}{relación de equivalencia}; $f \sim g$ implica
        que $f$ y $g$ comparten los límites, los signos (cerca de
        $x_0$) y la ausencia de ceros;
  \item los equivalentes se \emph{multiplican y \omterm{def:b1:arith:divides}{dividen}}:
        $f_1 \sim g_1$, $f_2 \sim g_2$ implican $f_1 f_2 \sim g_1 g_2$
        y $\frac{f_1}{f_2} \sim \frac{g_1}{g_2}$;
  \item los equivalentes \emph{no} se suman: $x + 1 \sim x$ y
        $-x \sim -x + 2$ en $+\infty$ y, sin embargo, las sumas $1$ y
        $2$ no son equivalentes. Para sumar hay que volver a los
        desarrollos con términos $o(\cdot)$ explícitos;
  \item $o(g) + o(g) = o(g)$, $\;u \cdot o(g) = o(ug)$,
        $\;o(o(g)) = o(g)$, y $f \sim g \iff f = g + o(g)$.
\end{enumerate}
\end{proposition}

\begin{proof}
Cada una es una manipulación breve de las definiciones; por ejemplo,
$f_1 f_2 = (1+\varepsilon_1)(1+\varepsilon_2) g_1 g_2$ y
$(1 + \varepsilon_1)(1+\varepsilon_2) \to 1$. Dos puntos de (4)
merecen su línea. $u\cdot o(g) = o(ug)$: si $f = \varepsilon g$ con
$\varepsilon \to 0$, entonces $uf = \varepsilon\,(ug)$ con el mismo
$\varepsilon$. $o(o(g)) = o(g)$: si $f = \varepsilon_1 h$ y
$h = \varepsilon_2 g$ con los dos $\varepsilon_i \to 0$, entonces
$f = (\varepsilon_1\varepsilon_2) g$, y el producto de los dos
infinitésimos lo es. La equivalencia $f \sim g \iff f = g + o(g)$ es
la definición leída dos veces: $f - g = \varepsilon g$. El
contraejemplo de (3) es la demostración de (3).
\end{proof}

\begin{example}[La escala de comparación]\label{ex:b1:taylor:scale}
Cuando $x \to +\infty$, la escala estándar dice, en orden creciente
de fuerza:
\[
1 \;=\; o(\ln x), \quad
\ln x = o(x^{0.01}), \quad
x^{0.01} = o(\sqrt x), \quad
\sqrt x = o(x^{10}), \quad
x^{10} = o(\eu^{x}), \quad
\eu^x = o(\eu^{2x}) ,
\]
siendo cada paso una instancia de las comparaciones de crecimiento
de la~\cref{prop:b1:functions:powerrules} (las potencias ganan a los
logaritmos, las exponenciales ganan a las potencias y, dentro de una
misma familia, decide el exponente). Dos hábitos que conviene
adquirir: primero, un $O(\cdot)$ que cae en una clase menor asciende
en silencio ($O(\ln x)$ también es $o(x^{0.01})$); segundo, en
$x \to 0^+$ toda la escalera se invierte con la sustitución
$x \mapsto \frac1x$ ---ahí $\ln x = o(x^{-0.01})$---, de modo que
«$x^\alpha \ln x \to 0$» vale para todo $\alpha > 0$. Tener clara la
escala es la mitad de todo argumento asintótico
del~\cref{ch:b1:series}.
\end{example}

\begin{example}[Unicidad de los desarrollos, y un dividendo de paridad]\label{ex:b1:taylor:uniqueness}
Si una función admite dos desarrollos en $0$ al mismo orden,
\[
a_0 + a_1 x + \dots + a_n x^n + o(x^n)
= b_0 + b_1 x + \dots + b_n x^n + o(x^n),
\]
entonces $a_k = b_k$ para todo $k$: restando y poniendo
$c_k = a_k - b_k$, evalúese la identidad
$c_0 + c_1 x + \dots + c_n x^n = o(x^n)$ cuando $x \to 0$ para
obtener $c_0 = 0$; divídase por $x$ y repítase --- cada división es
legítima porque la expresión restante vuelve a ser $o(x^{n-k})$. Los
coeficientes son, por tanto, intrínsecos, y se pueden calcular por
\emph{cualquier} camino (\omterm{def:b1:derivative:def}{derivadas} de Taylor, álgebra sobre
desarrollos conocidos, integración): todos los caminos han de
coincidir. Dividendo: una función \emph{par} solo tiene potencias
pares en su desarrollo ---sustitúyase $x$ por $-x$ e invóquese la
unicidad---, e igualmente las impares tienen potencias impares. Por
eso $\cos$ lleva $o(x^{2p+1})$ y no $o(x^{2p})$ en la tabla de más
abajo: el término impar ausente es información gratis, un orden de
precisión por nada.
\end{example}

\section{Las tres fórmulas de Taylor}

\begin{theorem}[Taylor con resto integral]\label{thm:b1:taylor:integral}
Sea $f$ de clase $C^{n+1}$ en un \omterm{prop:b1:reals:intervals}{intervalo} que contiene a $a$ y a $x$.
Entonces\index{fórmula de Taylor}
\[
f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}\,(x - a)^k
+ \int_a^x \frac{(x - t)^n}{n!}\, f^{(n+1)}(t)\, \dd t .
\]
\end{theorem}

\begin{proof}
Inducción sobre $n$. Para $n = 0$:
$f(x) = f(a) + \int_a^x f'(t)\dd t$ es el teorema fundamental
(\cref{thm:b1:integration:ftc}). Paso: intégrese el resto por partes,
\[
\int_a^x \frac{(x-t)^n}{n!} f^{(n+1)}(t)\,\dd t
= \Bigl[-\frac{(x-t)^{n+1}}{(n+1)!} f^{(n+1)}(t)\Bigr]_a^x
+ \int_a^x \frac{(x-t)^{n+1}}{(n+1)!} f^{(n+2)}(t)\,\dd t ,
\]
contribuyendo el corchete con el término
$\frac{f^{(n+1)}(a)}{(n+1)!}(x - a)^{n+1}$.
\end{proof}

\begin{example}[Un desarrollo exacto con su resto]\label{ex:b1:taylor:lnexact}
Para $\ln(1 + x)$, el resto \omterm{thm:b1:integration:def}{integral} se puede hacer completamente
explícito sin derivar nada $n$ veces: intégrese la identidad
geométrica finita
$\frac{1}{1+t} = \sum_{k=0}^{n-1}(-t)^k + \frac{(-t)^n}{1+t}$ de $0$
a $x$:
\[
\ln(1 + x) = \sum_{k=1}^{n} \frac{(-1)^{k-1}x^k}{k}
+ (-1)^n \int_0^x \frac{t^n}{1 + t}\,\dd t ,
\]
y para $0 \leq x \leq 1$ el resto está acotado por
$\int_0^x t^n\,\dd t = \frac{x^{n+1}}{n+1}$. Esto es más fuerte que
Taylor--Young en dos sentidos: es una \emph{identidad} válida para un
$x$ fijo (no solo cuando $x \to 0$), y la cota del error es numérica.
El problema del fin de semana (\cref{pb:b1:taylor:1}) vive de formas
exactas así; Taylor--Young, más abajo, es la herramienta ligera para
los límites, donde solo importa la forma del error.
\end{example}

\begin{theorem}[Desigualdad de Taylor--Lagrange]\label{thm:b1:taylor:lagrange}
Sea $f$ de clase $C^{n+1}$ con $\abs{f^{(n+1)}} \leq M$ entre $a$ y
$x$. Entonces
\[
\Bigl| f(x) - \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k \Bigr|
\leq M\, \frac{\abs{x - a}^{n+1}}{(n+1)!} .
\]
\end{theorem}

\begin{proof}
Acótese el resto \omterm{thm:b1:integration:def}{integral}: $\bigl|\int_a^x \frac{(x-t)^n}{n!}
f^{(n+1)}(t)\,\dd t\bigr| \leq M \bigl|\int_a^x
\frac{\abs{x-t}^n}{n!}\dd t\bigr| = M\frac{\abs{x-a}^{n+1}}{(n+1)!}$.
\end{proof}

\begin{example}[Cálculo numérico certificado]\label{ex:b1:taylor:certified}
¿Cuánto vale $\sqrt{1.02}$? Aplíquese Taylor--Lagrange a
$f(t) = \sqrt{1 + t}$ en $a = 0$, al orden $2$, con $x = 0.02$:
\[
\sqrt{1.02} \approx 1 + \frac{0.02}{2} - \frac{0.02^2}{8}
= 1.00995 ,
\qquad
\abs{f'''(t)} = \frac{3}{8}(1+t)^{-5/2} \leq \frac 38 ,
\]
luego el error es a lo sumo
$\frac38 \cdot \frac{0.02^3}{6} = 5\cdot10^{-7}$:
$\sqrt{1.02} = 1.00995$ con seis decimales certificados (valor
verdadero $1.0099504938\dots$ --- la cota es casi óptima). La idea de
cierre: Taylor--Young dice solamente \emph{con qué rapidez} se anula
el error; Taylor--Lagrange convierte el mismo \omterm{def:b1:poly:def}{polinomio} en un
\emph{certificado}, un número más una barra de error demostrada. Cada
vez que en este libro se afirma un decimal, hay detrás una cota de
tipo Lagrange; el problema del fin de semana
(\cref{pb:b1:taylor:1}) industrializa la idea.
\end{example}

\begin{theorem}[Taylor--Young]\label{thm:b1:taylor:young}
Sea $f$ $n$ veces \omterm{def:b1:derivative:def}{derivable} en $a$. Entonces, cuando $x \to a$:
\[
f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}\,(x-a)^k
+ o\bigl((x-a)^n\bigr) .
\]
\end{theorem}

\begin{proof}
Inducción sobre $n$. Para $n = 1$, esto es la definición de la
\omterm{def:b1:derivative:def}{derivada} (\cref{def:b1:derivative:def}). Supóngase el \omterm{def:b1:logic:statement}{enunciado} al
orden $n - 1$ y sea $f$ $n$ veces \omterm{def:b1:derivative:def}{derivable} en $a$. Aplíquese la
hipótesis de inducción a $f'$ (que es $n-1$ veces \omterm{def:b1:derivative:def}{derivable} en $a$):
\[
f'(t) = \sum_{k=0}^{n-1} \frac{f^{(k+1)}(a)}{k!}(t-a)^k + r(t),
\qquad r(t) = o\bigl((t-a)^{n-1}\bigr).
\]
Sea $g(x) = f(x) - \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k$;
entonces $g' = r$ y $g(a) = 0$. Dado $\varepsilon > 0$, elíjase
$\delta$ con $\abs{r(t)} \leq \varepsilon\abs{t - a}^{n-1}$ para
$\abs{t-a} \leq \delta$; la desigualdad del valor medio
(\cref{thm:b1:derivative:mvt}) aplicada en el segmento de $a$ a $x$
(donde $\abs{g'} \leq \varepsilon\abs{x-a}^{n-1}$) da
$\abs{g(x)} \leq \varepsilon\abs{x - a}^n$: exactamente
$g(x) = o((x-a)^n)$.
\end{proof}

\begin{remark}[Tres fórmulas, tres precios, tres productos]
Las hipótesis se gradúan exactamente con las conclusiones.
Taylor--Young pide lo mínimo ($n$ \omterm{def:b1:derivative:def}{derivadas} \emph{en el punto}, nada
más) y da lo mínimo: un $o((x-a)^n)$ cualitativo, perfecto para los
límites e inútil para cifras certificadas. La desigualdad de Lagrange
pide $C^{n+1}$ \emph{en el \omterm{prop:b1:reals:intervals}{intervalo}} y una cota $M$ ahí, y devuelve
una barra numérica de error. La forma integral pide la misma
regularidad y devuelve lo máximo: el error como objeto explícito que
se puede transformar (integrar por partes, acotar a trozos, cambiar
de variable) --- es la forma que movía la máquina de irracionalidad
del~\cref{pb:b1:integration:1}. Elegir la fórmula más débil que
sostiene el objetivo no es pedantería: la función plana
del~\cref{pb:b1:taylor:1} cumple Taylor--Young a todos los órdenes,
mientras que toda conclusión más fuerte sobre ella es falsa lejos
de $0$.
\end{remark}

\begin{proposition}[Desarrollos usuales en $0$]
Cuando $x \to 0$, para todo orden $n$ fijo:
\index{desarrollo de Taylor!usuales}
\begin{align*}
\eu^x &= 1 + x + \frac{x^2}{2!} + \dots + \frac{x^n}{n!} + o(x^n),\\
\cos x &= 1 - \frac{x^2}{2!} + \frac{x^4}{4!} - \dots
+ \frac{(-1)^p x^{2p}}{(2p)!} + o(x^{2p+1}),\\
\sin x &= x - \frac{x^3}{3!} + \dots + \frac{(-1)^p
x^{2p+1}}{(2p+1)!} + o(x^{2p+2}),\\
\frac{1}{1 - x} &= 1 + x + x^2 + \dots + x^n + o(x^n),\\
\ln(1 + x) &= x - \frac{x^2}{2} + \frac{x^3}{3} - \dots +
\frac{(-1)^{n-1} x^n}{n} + o(x^n),\\
(1 + x)^\alpha &= 1 + \alpha x + \frac{\alpha(\alpha-1)}{2!}x^2 +
\dots + \binom{\alpha}{n} x^n + o(x^n),
\end{align*}
donde $\binom{\alpha}{n} = \frac{\alpha(\alpha - 1)\cdots(\alpha - n
+ 1)}{n!}$ para $\alpha$ real. ($\cosh$ y $\sinh$: igual que $\cos$ y
$\sin$, pero sin los signos alternados.)
\label{prop:b1:taylor:standard}
\end{proposition}

\begin{proof}
Cada función es regular cerca de $0$ con \omterm{def:b1:derivative:def}{derivadas} fáciles de
evaluar: $(\eu^x)^{(k)} = \eu^x$; las \omterm{def:b1:derivative:def}{derivadas} de $\sin$ y de $\cos$
son cíclicas de período $4$;
$\bigl((1+x)^\alpha\bigr)^{(k)} = \alpha(\alpha - 1)\cdots(\alpha - k
+ 1)(1+x)^{\alpha - k}$;
$\bigl(\ln(1+x)\bigr)^{(k)} = \frac{(-1)^{k-1}(k-1)!}{(1+x)^k}$.
Aplíquese Taylor--Young en $a = 0$. (El geométrico es exacto:
$\frac{1}{1-x} - \sum_0^n x^k = \frac{x^{n+1}}{1 - x} = o(x^n)$.)
\end{proof}

\begin{method}[Calcular con desarrollos]\label{met:b1:taylor:compute}
\begin{enumerate}
  \item \emph{Fíjese primero el orden objetivo} $n$ y trúnquese ahí
        todo resultado intermedio --- arrastrar términos de orden
        superior es trabajo perdido, y dejar caer los inferiores es
        un error.
  \item \emph{Sumas y productos}: desarróllese cada factor al orden
        $n$ y multiplíquese, descartando más allá de $x^n$.
  \item \emph{Composición} $f(u(x))$ con $u(x) \to 0$: sustitúyase el
        desarrollo de $u$ en el de $f$, orden a orden.
  \item \emph{Cocientes}: escríbase $\frac{1}{1 + v}$ con $v \to 0$ y
        úsese el desarrollo geométrico.
  \item \emph{Intégrese} un desarrollo término a término (derivar
        exige más cuidado --- justificación: la \omterm{thm:b1:integration:def}{integral} de $o(t^n)$
        de $0$ a $x$ es $o(x^{n+1})$, por acotación directa).
\end{enumerate}
\end{method}

\begin{example}[Composición, con la contabilidad a la vista]\label{ex:b1:taylor:esin}
Desarróllese $\eu^{\sin x}$ al orden $3$. Desarrollo \omterm{def:b1:topology:closure}{interior}:
$u = \sin x = x - \frac{x^3}{6} + o(x^3)$, que en efecto tiende a
$0$. Exterior: $\eu^u = 1 + u + \frac{u^2}{2} + \frac{u^3}{6} +
o(u^3)$, y $o(u^3) = o(x^3)$ puesto que $u \sim x$. Potencias de $u$,
truncadas en $x^3$:
\[
u^2 = x^2 + o(x^3), \qquad u^3 = x^3 + o(x^3)
\]
(el término cruzado $2x\cdot(-\frac{x^3}{6})$ ya es $x^4$).
Ensámblese:
\[
\eu^{\sin x} = 1 + \Bigl(x - \frac{x^3}{6}\Bigr) + \frac{x^2}{2}
+ \frac{x^3}{6} + o(x^3)
= 1 + x + \frac{x^2}{2} + o(x^3) :
\]
las dos contribuciones en $x^3$ se cancelan exactamente. La idea de
cierre: $\eu^{\sin x}$ y $\eu^x$ coinciden hasta el orden $3$ --- no
porque $\sin x \approx x$ groseramente, sino porque el primer
desacuerdo de los exponentes ($-\frac{x^3}{6}$) entra multiplicado
por $\eu^0 = 1$ y se topa después con el término cúbico de la
exponencial exterior; la contabilidad orden a orden detecta esas
conspiraciones, y el ojo nunca. (El término siguiente es
$-\frac{x^4}{8}$: la tregua acaba en el orden $4$.)
\end{example}

\begin{example}\label{ex:b1:taylor:tan}
Desarrollo de $\tan$ al orden $5$. Escríbase
$\tan x = \sin x \cdot \frac{1}{\cos x}$:
\[
\frac{1}{\cos x} = \frac{1}{1 - \bigl(\frac{x^2}{2} - \frac{x^4}{24}
+ o(x^5)\bigr)}
= 1 + \Bigl(\frac{x^2}{2} - \frac{x^4}{24}\Bigr)
+ \Bigl(\frac{x^2}{2}\Bigr)^{\!2} + o(x^5)
= 1 + \frac{x^2}{2} + \frac{5x^4}{24} + o(x^5),
\]
y después
\[
\tan x = \Bigl(x - \frac{x^3}{6} + \frac{x^5}{120}\Bigr)
\Bigl(1 + \frac{x^2}{2} + \frac{5x^4}{24}\Bigr) + o(x^5)
= x + \frac{x^3}{3} + \frac{2x^5}{15} + o(x^5) .
\]
\end{example}

\section{Aplicaciones}

\begin{example}[Límites]\label{ex:b1:taylor:limits}
\[
\lim_{x \to 0} \frac{x - \sin x}{x^3}:
\qquad
x - \sin x = \frac{x^3}{6} + o(x^3) \sim \frac{x^3}{6},
\qquad\text{luego el límite es } \frac16
\]
--- lo que zanja la cuestión planteada en
el~\cref{exo:b1:functions:9}. Del mismo modo,
$\displaystyle\lim_{x\to0}\Bigl(\frac{\sin x}{x}\Bigr)^{1/x^2}$: el
logaritmo es
\[
\frac{1}{x^2}\ln\Bigl(1 - \frac{x^2}{6} + o(x^2)\Bigr)
= \frac{1}{x^2}\Bigl(-\frac{x^2}{6} + o(x^2)\Bigr)
\longrightarrow -\frac16,
\qquad\text{límite } \eu^{-1/6}.
\]
\end{example}

\begin{remark}[Errores frecuentes con los desarrollos]
(i) \emph{No se suman ni se restan nunca equivalentes}: de
$\tan x \sim x$ y $\sin x \sim x$ \emph{no} se puede concluir
$\tan x - \sin x \sim 0$ (que no significa nada) --- el camino
honesto son los desarrollos:
\[
\tan x - \sin x
= \Bigl(x + \frac{x^3}{3}\Bigr) - \Bigl(x - \frac{x^3}{6}\Bigr)
+ o(x^3) = \frac{x^3}{2} + o(x^3) \sim \frac{x^3}{2} .
\]
(ii) \emph{Desarróllese más allá de la masacre}: en ese mismo
cálculo, el orden $1$ solo ve $x - x = 0$; siempre que los términos
dominantes se cancelen, súbase el orden hasta que sobreviva un
coeficiente no nulo y solo entonces vuélvase a un equivalente.
(iii) \emph{Los equivalentes no pasan por las exponenciales}:
$n^2 + n \sim n^2$ y, sin embargo, $\eu^{n^2+n} = \eu^{n}\,\eu^{n^2}$
\emph{no} es equivalente a $\eu^{n^2}$ --- exponéncense solo
\emph{desarrollos del exponente} cuyo error tienda a $0$, nunca
equivalentes del exponente. (Los logaritmos son más benignos: si
$u_n \sim v_n \to \ell \neq 1$, con $\ell > 0$, entonces
$\ln u_n \sim \ln v_n$.) (iv) \emph{El cálculo con $o(\cdot)$ es de
un solo sentido}: $o(x^2) + o(x^2) = o(x^2)$, $5\,o(x^2) = o(x^2)$,
$x\cdot o(x^2) = o(x^3)$ --- pero un $o(x^2)$ no es una función
concreta, así que no se cancelen nunca dos de ellos entre sí:
$o(x^2) - o(x^2)$ es $o(x^2)$, no $0$.
\end{remark}

\begin{proposition}[Comportamiento local]\label{prop:b1:taylor:local}
Supóngase $f(x) = f(a) + c\,(x - a)^p + o\bigl((x-a)^p\bigr)$ con
$c \neq 0$ (primer término no nulo tras la constante; $p \geq 2$ en
un punto crítico).
\begin{itemize}
  \item Si $p$ es par: $f$ tiene un mínimo local en $a$ si $c > 0$, y
        un máximo local si $c < 0$.
  \item Si $p$ es impar: no hay extremo ($f - f(a)$ cambia de signo);
        y si además el desarrollo empieza tras un término lineal
        $f'(a)(x - a)$, la gráfica cruza su tangente: es un
        \emph{punto de inflexión}.
\end{itemize}
\end{proposition}

\begin{proof}
Cerca de $a$, $f(x) - f(a) = (x-a)^p\bigl(c + o(1)\bigr)$ tiene el
signo de $c\,(x-a)^p$: constante si $p$ es par, cambiante si $p$ es impar.
\end{proof}

\begin{example}[Los exponentes hay que desarrollarlos hasta el $o(1)$]\label{ex:b1:taylor:expdiscipline}
Hállese un equivalente de $u_n = \bigl(1 + \frac1n\bigr)^{n^2}$.
Desarróllese el \emph{exponente} hasta que su error tienda a $0$:
\[
n^2 \ln\Bigl(1 + \frac1n\Bigr)
= n^2\Bigl(\frac1n - \frac{1}{2n^2} +
O\Bigl(\frac{1}{n^3}\Bigr)\Bigr)
= n - \frac12 + O\Bigl(\frac1n\Bigr),
\]
luego $u_n = \eu^{\,n - 1/2}\,\eu^{O(1/n)}$ con $\eu^{O(1/n)} \to
1$:
\[
u_n \;\sim\; \eu^{\,n - \frac12} .
\]
Obsérvese qué habría salido mal con menos cuidado: detener el
exponente en $n^2\cdot\frac1n = n + O(1)$ deja un factor
$\eu^{O(1)}$ ---acotado, pero que no tiende a $1$--- y no se puede
afirmar ningún equivalente. La regla de los errores frecuentes de más
arriba, en versión positiva: un equivalente de $\eu^{a_n}$ exige el
desarrollo de $a_n$ \emph{hasta un término que tienda a cero},
conservando exactamente todos los coeficientes anteriores.
\end{example}

\begin{example}[Clasificar un punto crítico plano]\label{ex:b1:taylor:flatmin}
Estúdiese $f(x) = \cos x + \frac{x^2}{2}$ cerca de $0$. Se tiene
$f'(0) = 0$ y $f''(0) = -\cos 0 + 1 = 0$: el criterio de la
segunda \omterm{def:b1:derivative:def}{derivada} enmudece. Desarróllese en su lugar:
\[
f(x) = \Bigl(1 - \frac{x^2}{2} + \frac{x^4}{24} +
o(x^4)\Bigr) + \frac{x^2}{2}
= 1 + \frac{x^4}{24} + o(x^4) :
\]
primer término no nulo $c\,x^p$ con $p = 4$ par y
$c = \frac{1}{24} > 0$: un mínimo local, de una planitud inusual (la
gráfica abandona su valor mínimo como $x^4$, no como $x^2$). La idea
de cierre: el desarrollo ve en una línea lo que la derivación
iterada oscurece --- y la~\cref{prop:b1:taylor:local} es el
diccionario sistemático que va del «primer término superviviente» a
la «forma local».
\end{example}

\begin{example}[Desarrollos en el infinito]\label{ex:b1:taylor:atinfinity}
Dos cálculos en los que la variable se va a $+\infty$ y la
sustitución $h = \frac1x \to 0^+$ importa todo el utillaje. Primero,
\[
x - x^2\ln\Bigl(1 + \frac1x\Bigr)
= x - x^2\Bigl(\frac1x - \frac{1}{2x^2} +
O\Bigl(\frac{1}{x^3}\Bigr)\Bigr)
= \frac12 + O\Bigl(\frac1x\Bigr) \longrightarrow \frac12 .
\]
Segundo, el \omterm{def:b1:functions:arc}{arcotangente} en el infinito: de
$\arctan x + \arctan\frac1x = \frac\pi2$ para $x > 0$
(\cref{prop:b1:functions:arcidentities}) y del desarrollo de
$\arctan$ en $0$ (\cref{exo:b1:taylor:3}),
\[
\arctan x = \frac\pi2 - \arctan\frac1x
= \frac\pi2 - \frac1x + \frac{1}{3x^3} +
o\Bigl(\frac{1}{x^3}\Bigr) :
\]
la gráfica se acerca a su asíntota $y = \frac\pi2$ por debajo, a
velocidad $\frac1x$. La idea de cierre: no hay una teoría aparte de
los desarrollos en el infinito --- una sola sustitución recíproca los
reduce a desarrollos en $0$, siempre que todo $O$ y todo $o$
intermedio se arrastre con honestidad.
\end{example}

\begin{example}[Asíntota por desarrollo]\label{ex:b1:taylor:asymptote}
Cuando $x \to +\infty$,
\[
\sqrt{x^2 + x} = x\sqrt{1 + \tfrac1x}
= x\Bigl(1 + \frac{1}{2x} - \frac{1}{8x^2} + o\bigl(\tfrac{1}{x^2}\bigr)\Bigr)
= x + \frac12 - \frac{1}{8x} + o\bigl(\tfrac 1x\bigr):
\]
la recta $y = x + \frac12$ es una asíntota, alcanzada \emph{por
debajo} (el término siguiente $-\frac{1}{8x}$ es negativo).
\end{example}

\begin{remark}[Dónde funcionan los desarrollos a continuación]
Los desarrollos asintóticos son el idioma corriente del resto del
libro: en el~\cref{ch:b1:series} deciden la convergencia (los
equivalentes alimentan los criterios de comparación, y el estudio de
$\sum \frac{1}{n^\alpha}$ es un desarrollo disfrazado); en el volumen
del segundo año se convierten en \emph{series de potencias}, donde el
\omterm{def:b1:poly:def}{polinomio} de Taylor adquiere infinitos términos y un radio de
convergencia; y toda linealización en física ---el péndulo, la
perturbación de primer orden--- es un \omterm{def:b1:logic:statement}{enunciado} de Taylor--Young con
el $o(\cdot)$ suprimido en silencio. El único aviso que merece
grabarse: un desarrollo describe una función solo \emph{cerca de un
punto} --- véase la función plana del problema del fin de semana,
cuyo desarrollo en $0$ es idénticamente nulo sin que la función lo sea.
\end{remark}

\begin{remark}[Perspectivas dentro de este volumen]
Los desarrollos son el idioma de trabajo del análisis restante y de
la geometría que viene. \Cref{ch:b1:series} los convierte en
veredictos de convergencia: un equivalente del término general es un
desarrollo truncado en su primer término, y los criterios más finos
(los alternados con control del error) consumen también el segundo.
\Cref{ch:b1:curves} lee la geometría local en los desarrollos de las
dos funciones coordenadas: que una curva parametrizada se cruce, sea
tangente o presente un punto de retroceso se decide por qué potencias
de $t$ sobreviven en $x(t)$ e $y(t)$ --- la versión plana de
la~\cref{prop:b1:taylor:local}. Y el~\cref{ch:b1:multivar} se detiene
en el orden uno a propósito: el plano tangente es un \omterm{def:b1:logic:statement}{enunciado} de
Taylor--Young en dos variables, y toda la teoría de segundo orden
(hessianas, puntos de silla) queda aplazada al volumen del segundo
año. El hilo común: toda pregunta «local» de este libro se responde
escribiendo el primer término superviviente de un desarrollo.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:taylor:1}
Dense los desarrollos en $0$: de $\eu^{2x}$ al orden $3$; de
$\;\ln(1 - x)$ al orden $4$; de $\;\sqrt{1 + x}$ al orden $3$; de
$\;\dfrac{1}{1 + x^2}$ al orden $6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:taylor:2}
Calcúlense los límites:
\[
\lim_{x\to 0} \frac{\eu^x - 1 - x}{x^2},
\qquad
\lim_{x\to 0} \frac{\cos x - \sqrt{1 - x^2}}{x^4},
\qquad
\lim_{x\to 0} \frac{\ln(1+x) - \sin x}{x^2}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:taylor:3}
Desarróllese $\arctan x$ en $0$ al orden $5$ integrando el desarrollo
de $\frac{1}{1 + x^2}$, y $\arcsin x$ al orden $5$ integrando el de
$(1 - x^2)^{-1/2}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:taylor:4}
Usando Taylor--Lagrange para $\exp$ en $\intcc{0}{1}$, demuéstrese que
\[
\Bigl| \eu - \sum_{k=0}^{n} \frac{1}{k!} \Bigr| \leq
\frac{3}{(n+1)!},
\]
y determínese un $n$ que garantice $6$ decimales exactos de $\eu$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:5}
Desarróllese al orden $2$ en $\frac1n$ y dedúzcanse el límite y la
velocidad de convergencia:
\[
\Bigl(1 + \frac 1n\Bigr)^{\!n}
= \eu\Bigl(1 - \frac{1}{2n} + \frac{11}{24n^2} +
o\Bigl(\frac{1}{n^2}\Bigr)\Bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:6}
Estúdiese el comportamiento local en $0$ de $f(x) = x^2 - x^4$ y de
$g(x) = x^3 + x^5$; y hállese la posición de la gráfica de
$h(x) = \eu^x$ respecto de su tangente en $a = 1$, primero localmente
y después globalmente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:7}
Determínense las asíntotas en $\pm\infty$ de
$f(x) = \sqrt[3]{x^3 + x^2}$ y la posición de la curva respecto de
ellas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:8}
Hállese el equivalente, cuando $n \to \infty$, de
\[
u_n = \sqrt{n+1} - \sqrt n, \qquad
v_n = \ln(n+1) - \ln n, \qquad
w_n = \sin\frac{1}{n} - \tan\frac{1}{n},
\]
cada uno como una potencia de $n$ por una constante.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:taylor:9}
Sea $f$ de clase $C^2$ en $\R$. Demuéstrese que para todos $x$ y $h > 0$:
\[
\abs{f'(x)} \leq \frac{\abs{f(x+h) - f(x-h)}}{2h} +
\frac{h}{2}\sup_{\intcc{x-h}{x+h}}\abs{f''} ,
\]
y dedúzcase la desigualdad de tipo Landau--Kolmogórov: si
$\abs f \leq M_0$ y $\abs{f''} \leq M_2$ en $\R$, entonces
$\abs{f'} \leq \sqrt{2 M_0 M_2}$ en todas partes.
\emph{(Optimícese en $h$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:taylor:10}
La sucesión $u_0 \in \intoo{0}{\pi}$, $u_{n+1} = \sin u_n$ decrece
hacia $0$ (justifíquese brevemente). Para hallar su velocidad,
considérese $v_n = \frac{1}{u_n^2}$:
\begin{enumerate}
  \item usando el desarrollo de $\sin$, demuéstrese
        $v_{n+1} - v_n \to \frac13$;
  \item con Cesàro (\cref{exo:b1:seq:10}), dedúzcase
        $\frac{v_n}{n} \to \frac13$ y después el equivalente
        $u_n \sim \sqrt{\dfrac{3}{n}}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:taylor:11}
(Diferencias de infinitos) Calcúlense
\[
\lim_{x \to 0} \Bigl(\frac{1}{x^2} - \frac{1}{\sin^2 x}\Bigr)
\qquad\text{y}\qquad
\lim_{x \to 0^+} \Bigl(\frac 1x - \frac{1}{\ln(1 + x)}\Bigr)
\]
reduciendo a común denominador y desarrollando numerador y
denominador por separado.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:taylor:12}
(Asintótica de raíces implícitas) Véase que para todo $k \in \N^*$ la
ecuación $\tan x = x$ tiene exactamente una solución $x_k$ en
$\intoo{k\pi - \frac\pi2}{k\pi + \frac\pi2}$, que
$x_k = k\pi + \frac\pi2 - \varepsilon_k$ con
$\varepsilon_k = \arctan\frac{1}{x_k}$, y dedúzcase el desarrollo
\[
x_k = k\pi + \frac\pi2 - \frac{1}{k\pi} + o\Bigl(\frac
1k\Bigr) \qquad (k \to \infty).
\]
\end{exercise}

\section{Problema: sumas alternadas, cifras certificadas y la
irracionalidad de $\cos 1$}

\begin{problem}[{Problema del fin de semana --- la estimación alternada
$\abs{S - S_n} \leq a_{n+1}$: $\ln 2$ y $\pi$ con decimales
demostrados, la fórmula de Machin y $\cos 1 \notin \Q$}]
\label{pb:b1:taylor:1}
Una suma alternada de términos decrecientes es el objeto más amable
del análisis numérico: su error está acotado por el primer término
omitido, \emph{con signo conocido}. Este problema demuestra ese
principio con el teorema de las \omterm{thm:b1:seq:adjacent}{sucesiones adyacentes} y después lo
gasta de tres maneras: decimales certificados para $\ln 2$ (tres
caminos rivales) y para $\pi$ (Leibniz y después la fórmula de Machin
de 1706, todavía la idea que hay detrás de los cálculos récord
durante siglos), la irracionalidad de $\cos 1$, $\sin 1$ y $\cosh 1$
y, como contrapeso, la \emph{igualdad} de Taylor--Lagrange y la
función plana cuyo desarrollo de Taylor miente.\index{estimación de
las series alternadas}\index{fórmula de Machin}\index{irracionalidad!de
cos 1@de $\cos 1$} En todo el problema, el lenguaje de «series» es
informal: toda suma de aquí es una \emph{sucesión de sumas
parciales}, como en el~\cref{ex:b1:seq:e}; la teoría propiamente
dicha se abre en el~\cref{ch:b1:series}.

\medskip
\noindent\textbf{Parte I --- La estimación alternada.} Sea
$(a_k)_{k \geq 0}$ decreciente hacia $0$ y
$S_n = \sum_{k=0}^{n} (-1)^k a_k$.
\begin{enumerate}
  \item Véase que $(S_{2n+1})$ es no decreciente, que $(S_{2n})$ es
        no creciente y que son adyacentes
        (\cref{thm:b1:seq:adjacent}): las dos convergen a un mismo
        $S$ con, para todo $n$,
        \[
        S_{2n+1} \leq S \leq S_{2n},
        \qquad
        \abs{S - S_n} \leq a_{n+1} ,
        \]
        teniendo el error el signo del primer término omitido.
        Véase además que si el decrecimiento es \emph{estricto},
        todas estas desigualdades son estrictas.
  \item Primer dividendo: para $x = 1$ en la serie exponencial,
        compárese con el~\cref{thm:b1:taylor:lagrange} en $a = 0$:
        véase que $T_n = \sum_{k=0}^{n} \frac{(-1)^k}{k!}$ converge
        a $\eu^{-1}$ con
        $\abs{\eu^{-1} - T_n} < \frac{1}{(n+1)!}$.
  \item (Leibniz, 1674) A partir de la identidad finita
        \emph{exacta}
        \[
        \frac{1}{1 + t^2} = \sum_{k=0}^{n} (-1)^k t^{2k}
        + \frac{(-1)^{n+1} t^{2n+2}}{1 + t^2} ,
        \]
        integrada en $\intcc{0}{1}$, demuéstrese
        \[
        \frac\pi4 = \sum_{k=0}^{n} \frac{(-1)^k}{2k+1} + \rho_n,
        \qquad \abs{\rho_n} \leq \frac{1}{2n+3} .
        \]
  \item Lentitud: ¿cuántos términos de Leibniz garantizan seis
        decimales exactos de $\pi$? (Unos dos millones.) Evalúese
        $4 S_4 = 4\bigl(1 - \frac13 + \frac15 - \frac17 +
        \frac19\bigr)$ y su distancia a $\pi$, para sentir el
        dolor.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- $\ln 2$ de tres maneras.}
\begin{enumerate}[resume]
  \item (Camino 1: la armónica alternada) De
        $\frac{1}{1+t} = \sum_{k=0}^{n-1}(-1)^k t^k +
        \frac{(-1)^n t^n}{1+t}$ integrada en $\intcc{0}{1}$:
        \[
        \ln 2 = \sum_{k=1}^{n} \frac{(-1)^{k-1}}{k} + (-1)^n
        R_n, \qquad \frac{1}{2(n+1)} \leq R_n \leq
        \frac{1}{n+1} :
        \]
        el error es de orden exactamente $\frac 1n$ --- un millón de
        términos para seis decimales.
  \item (Camino 2: la serie rápida) Intégrese
        $\frac{1}{1 - t^2} = \sum_{k=0}^{n} t^{2k} +
        \frac{t^{2n+2}}{1-t^2}$ de $0$ a $x \in \intoo{0}{1}$ y
        evalúese en $x = \frac13$ (nótese
        $\frac{1 + 1/3}{1 - 1/3} = 2$):
        \[
        \ln 2 = 2\sum_{k=0}^{n} \frac{(1/3)^{2k+1}}{2k+1} +
        \tilde\rho_n,
        \qquad
        0 < \tilde\rho_n \leq \frac{9}{4}\cdot
        \frac{(1/3)^{2n+3}}{2n+3} :
        \]
        convergencia geométrica, aproximadamente una cifra por
        término.
  \item (Camino 3: \omterm{thm:b1:integration:riemann}{sumas de Riemann} y una identidad oculta)
        Demuéstrese por inducción la identidad
        \[
        \sum_{k=1}^{2n} \frac{(-1)^{k-1}}{k} = H_{2n} - H_n =
        \sum_{k=1}^{n} \frac{1}{n + k} ,
        \]
        y recupérese $\ln 2$ como el límite de \omterm{thm:b1:integration:riemann}{sumas de Riemann}
        del~\cref{ex:b1:integration:riemannexample}: los caminos 1 y
        3 son en secreto el mismo número visto dos veces.
  \item Duelo con seis términos: compárese
        $\sum_{k=1}^{6} \frac{(-1)^{k-1}}{k} = 0.6167$ con el
        camino 2 en $n = 5$, que ya da $\ln 2 = 0.693147$ con error
        $\leq 1.1\cdot10^{-7}$. Explíquese la razón estructural
        (punto de evaluación bien dentro del \omterm{prop:b1:reals:intervals}{intervalo} de
        convergencia frente a en su \omterm{def:b1:topology:closure}{frontera}).
  \item ¿Cuántos términos del camino 2 certifican \emph{diez}
        decimales de $\ln 2$? Véase que basta $n = 10$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La fórmula de Machin.}
\begin{enumerate}[resume]
  \item Calcúlese $(5 + \iu)^4$ y compruébese la identidad compleja
        \[
        (5 + \iu)^4 = 2\,(1 + \iu)\,(239 + \iu) .
        \]
        Tomando argumentos (con los convenios
        del~\cref{ch:b1:complex}), dedúzcase la fórmula de Machin
        \[
        \frac\pi4 = 4\arctan\frac15 - \arctan\frac{1}{239} .
        \]
        (Compruébese que ningún argumento sale de
        $\intoo{-\frac\pi2}{\frac\pi2}$.)
  \item Como en la pregunta 3, establézcase, para $0 < x < 1$:
        \[
        \arctan x = \sum_{k=0}^{n} \frac{(-1)^k
        x^{2k+1}}{2k+1} + r_n(x), \qquad \abs{r_n(x)} \leq
        \frac{x^{2n+3}}{2n+3} .
        \]
  \item Certifíquense siete decimales de $\pi$ con seis términos:
        acótese el error total de
        \[
        \pi \approx 16\sum_{k=0}^{4}
        \frac{(-1)^k (1/5)^{2k+1}}{2k+1}
        - 4\Bigl(\frac{1}{239} - \frac{1}{3\cdot239^3}\Bigr)
        \]
        por $16\,\frac{(1/5)^{11}}{11} + 4\,\frac{(1/239)^5}{5} <
        5\cdot10^{-8}$, y dese el valor resultante
        $3.1415926\dots$
  \item Compárense los tres caminos hacia $\pi$ ya disponibles
        ---Leibniz (pregunta 4), las \omterm{thm:b1:integration:def}{integrales} de Dalzell
        del~\cref{pb:b1:integration:1} (error $4^{1-5m}$) y Machin
        (error $\approx 16\cdot 5^{-(2n+3)}$)--- en cifras por
        término, y explíquese por qué encoger el punto de evaluación
        gana a todo lo demás.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La trampa del entero, edición
alternada.}
\begin{enumerate}[resume]
  \item Supóngase $\cos 1 = \frac pq$. Multiplíquese el encuadre
        alternado estricto de $\sum_{k} \frac{(-1)^k}{(2k)!}$
        (pregunta 1) por $(2n)!$ con $2n \geq q$ y dedúzcase una
        contradicción: $\cos 1$ es irracional.
  \item Adáptese a $\sin 1 = \sum_k \frac{(-1)^k}{(2k+1)!}$
        (multiplíquese por $(2n+1)!$): $\sin 1 \notin \Q$. Los dos
        irracionales y, sin embargo, $\cos^2 1 + \sin^2 1 = 1$: la
        irracionalidad no es estable por operaciones algebraicas.
  \item El \omterm{def:b1:arith:prime}{primo} no alternado: $\cosh 1 = \sum_k \frac{1}{(2k)!}$
        (en el sentido de sumas parciales, con la cota bilateral de
        la cola $0 < \cosh 1 - \sum_{k \leq n} \frac{1}{(2k)!} <
        \frac{2}{(2n+2)!}$, que hay que demostrar). Conclúyase
        $\cosh 1 \notin \Q$ con la misma trampa.
  \item Empújese hasta $\cos\frac 1m$ para todo entero $m \geq 1$:
        multiplíquese por $m^{2n}(2n)!$ y conclúyase
        $\cos\frac1m \notin \Q$. ¿Dónde se rompe el mismo intento
        para $\cos\frac ab$ con $b > 1$ general? (Identifíquese el
        denominador que ya no se elimina.)
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Más fino y más oscuro: la forma con
igualdad, y una función que engaña a Taylor.}
\begin{enumerate}[resume]
  \item (Taylor--Lagrange, forma con igualdad) Sea $f$ $n + 1$
        veces \omterm{def:b1:derivative:def}{derivable} entre $a$ y $x$. Defínase
        $g(t) = f(x) - \sum_{k=0}^{n} \frac{f^{(k)}(t)}{k!}(x - t)^k
        - A\,\frac{(x-t)^{n+1}}{(n+1)!}$ con la constante $A$
        elegida de modo que $g(a) = 0$. Calcúlese $g'$ (la suma
        telescopa), aplíquese Rolle en $\intcc{a}{x}$ y conclúyase
        que existe $c$ estrictamente entre $a$ y $x$ con
        \[
        f(x) = \sum_{k=0}^{n} \frac{f^{(k)}(a)}{k!}(x-a)^k
        + \frac{f^{(n+1)}(c)}{(n+1)!}\,(x-a)^{n+1} .
        \]
  \item Dividendo de la igualdad: para $x > 0$ véase que
        \[
        \eu^x > 1 + x + \frac{x^2}{2!} + \dots +
        \frac{x^n}{n!}
        \]
        (estrictamente, para todo $n$), y localícese dónde se
        invierte la desigualdad para $x < 0$ según la paridad
        de $n$.
  \item Compárense los restos sobre $\sin(0.5)$ al orden $3$: Young
        da solo $o(x^3)$ (ningún número); Lagrange da
        $\abs{\sin 0.5 - (0.5 - \frac{0.5^3}{6})} \leq
        \frac{0.5^5}{120} = 2.61\cdot10^{-4}$; y la estimación
        alternada da la misma cota \emph{más} la información del
        signo $\sin 0.5 > 0.5 - \frac{0.5^3}{6}$. Compárese con el
        error verdadero $2.59\cdot10^{-4}$: la cota casi se alcanza.
        ¿A qué herramienta se echaría mano, y cuándo?
  \item (La función plana) Sea $f(x) = \eu^{-1/x^2}$ para
        $x \neq 0$ y $f(0) = 0$. Véase que $f$ es \omterm{def:b1:continuity:continuous}{continua} en $0$,
        que $f'(0) = 0$ y, más en general ---demostrando que toda
        \omterm{def:b1:derivative:def}{derivada} tiene la forma
        $f^{(k)}(x) = P_k\bigl(\frac1x\bigr)\eu^{-1/x^2}$ para un
        \omterm{def:b1:poly:def}{polinomio} $P_k$ (inducción)---, que $f^{(k)}(0) = 0$ para
        todo $k$ \emph{(\omterm{prop:b1:functions:powerrules}{comparación de crecimientos},
        \cref{prop:b1:functions:powerrules})}. Conclúyase: todos los
        \omterm{def:b1:poly:def}{polinomios} de Taylor de $f$ en $0$ se anulan y, sin embargo,
        $f(x) > 0$ para $x \neq 0$: Taylor--Young vale a todos los
        órdenes y no dice nada de $f$ lejos de $0$. Los desarrollos
        describen \emph{gérmenes}, no funciones.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- Síntesis.}
\begin{enumerate}[resume]
  \item Hágase correr la trampa una vez más, sobre
        $\eu^{-1} = \sum_k \frac{(-1)^k}{k!}$: multiplíquese el
        encuadre alternado estricto por $n!$ y conclúyase
        $\eu^{-1} \notin \Q$, y por tanto $\eu \notin \Q$ --- la
        tercera demostración de este hecho en el volumen.
        Enumérense las tres (\omterm{thm:b1:seq:adjacent}{sucesiones adyacentes},
        \cref{exo:b1:seq:9}; \omterm{thm:b1:integration:def}{integrales},
        \cref{pb:b1:integration:1}; sumas alternadas, aquí) y lo que
        necesitó cada una.
  \item La letra pequeña: la monotonía no es decorativa. Sea
        $b_k = \frac1k$ para $k$ impar y $b_k = \frac{1}{k^2}$ para
        $k$ par: los $b_k$ son positivos y tienden a $0$ y, sin
        embargo, las sumas parciales de $\sum (-1)^k b_k$ divergen a
        $-\infty$. Demuéstrese \emph{(pártase la suma parcial en la
        parte par, acotada mediante el~\cref{ex:b1:seq:basel}, y la
        parte impar, que domina la mitad de la serie armónica,
        \cref{exo:b1:seq:5})}, y dígase exactamente qué paso de la
        pregunta 1 usó la monotonía.
  \item Compruébese la identidad más sencilla de Euler
        $\arctan\frac12 + \arctan\frac13 = \frac\pi4$ mediante
        $(2 + \iu)(3 + \iu) = 5(1 + \iu)$, estímense los términos
        necesarios para seis decimales de $\pi$ por este camino
        (basta $n = 10$) y colóquese entre Leibniz y Machin en la
        clasificación de la pregunta 13.
  \item Síntesis, una frase para cada punto: (i) enúnciese la
        estimación alternada y sus dos productos (cota y signo);
        (ii) por qué las identidades finitas exactas con restos
        explícitos ganan a los \omterm{def:b1:logic:statement}{enunciados} con límites para el
        cálculo numérico certificado; (iii) inventario del problema
        ($\pi$ con error $10^{-7}$ a mano, $\ln 2$ con diez
        decimales, cuatro demostraciones de irracionalidad, un
        teorema con igualdad, un ejemplo de aviso); (iv) cuáles de
        estos hilos recogerá el~\cref{ch:b1:series} (el criterio de
        las series alternadas, la convergencia absoluta frente a la
        condicional y el drama de las reordenaciones de su problema
        del fin de semana).
\end{enumerate}
\end{problem}
