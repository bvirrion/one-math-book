\chapter{Matrices}\label{ch:b1:matrices}

Una matriz es una \omterm{def:b1:linmaps:def}{aplicación lineal} escrita en \omterm{prop:b1:vspaces:coordinates}{coordenadas}. Este
capítulo establece el diccionario ---la composición pasa a ser el
producto de matrices, la \omterm{def:b1:logic:inj}{biyectividad} pasa a ser la invertibilidad y
el \omterm{def:b1:matrices:changeofbasis}{cambio de base} pasa a ser la conjugación--- y también el lado
algorítmico: las \omterm{met:b1:matrices:gauss}{operaciones por filas} y el cálculo de rangos e
inversas. Vistas por primera vez en el volumen anterior, las matrices
quedan ahora fundadas en la teoría de los
\cref{ch:b1:vspaces,ch:b1:findim,ch:b1:linmaps}.

\section{Matrices y aplicaciones lineales}

\begin{definition}\label{def:b1:matrices:def}
$\mathcal{M}_{n,p}(K)$ es el \omterm{def:b1:vspaces:def}{espacio vectorial} de las tablas
$n \times p$ de escalares $A = (a_{ij})$ ($i$: fila, $j$: columna),
de dimensión $np$ (\omterm{def:b1:vspaces:free}{base}: las matrices $E_{ij}$ con un único $1$).
Dadas \omterm{def:b1:vspaces:free}{bases} $\mathcal{B} = (e_1, \dots, e_p)$ de $E$ y $\mathcal{C}$
de $F$ ($\dim F = n$), la \emph{matriz de $u \in \mathcal{L}(E, F)$}
es la tabla cuya columna $j$-ésima enumera las \omterm{prop:b1:vspaces:coordinates}{coordenadas} de
$u(e_j)$ en $\mathcal{C}$:
\[
\operatorname{Mat}_{\mathcal{B},\mathcal{C}}(u) = (a_{ij}),
\qquad u(e_j) = \sum_{i=1}^{n} a_{ij}\, f_i .
\]
La \omterm{def:b1:logic:map}{aplicación}
$u \mapsto \operatorname{Mat}_{\mathcal{B},\mathcal{C}}(u)$ es un
isomorfismo de $\mathcal{L}(E, F)$ sobre $\mathcal{M}_{n,p}(K)$
(\cref{prop:b1:linmaps:basis}: una \omterm{def:b1:linmaps:def}{aplicación lineal} es exactamente
una elección de imágenes de los $e_j$).
\end{definition}

\begin{example}[La derivada, como matriz]
\label{ex:b1:matrices:derivative}
Sea $D(P) = P'$ en $\R_3[X]$. En la \omterm{def:b1:vspaces:free}{base} monomial
$(1, X, X^2, X^3)$: $D(1) = 0$, $D(X) = 1$, $D(X^2) = 2X$,
$D(X^3) = 3X^2$, luego
\[
\operatorname{Mat}(D) =
\begin{pmatrix}
0 & 1 & 0 & 0\\
0 & 0 & 2 & 0\\
0 & 0 & 0 & 3\\
0 & 0 & 0 & 0
\end{pmatrix}.
\]
En la \omterm{def:b1:vspaces:free}{base} \emph{dividida}
$\bigl(1,\ X,\ \frac{X^2}2,\ \frac{X^3}6\bigr)$, cada vector de la
\omterm{def:b1:vspaces:free}{base} va al anterior
($D\bigl(\frac{X^k}{k!}\bigr) = \frac{X^{k-1}}{(k-1)!}$) y la matriz
se convierte en el desplazamiento puro: unos en la superdiagonal y
ceros en el resto. Dos moralejas: la matriz pertenece a la
\emph{pareja} (\omterm{def:b1:logic:map}{aplicación}, \omterm{def:b1:vspaces:free}{base}), no a la \omterm{def:b1:logic:map}{aplicación} sola; y una
buena \omterm{def:b1:vspaces:free}{base} vuelve visible la estructura de un vistazo --- la forma
de desplazamiento muestra al instante que $D^4 = 0$ en $\R_3[X]$,
pues cada potencia de la matriz empuja su diagonal de unos un paso
más lejos.
\end{example}

\begin{definition}[Producto]\label{def:b1:matrices:product}
Para $A \in \mathcal{M}_{n,p}$ y $B \in \mathcal{M}_{p,q}$:
\[
(AB)_{ik} = \sum_{j=1}^{p} a_{ij}\, b_{jk}
\qquad (1 \leq i \leq n,\ 1 \leq k \leq q).
\]
Esta es precisamente la matriz de la composición:
$\operatorname{Mat}(v \circ u) = \operatorname{Mat}(v)\,
\operatorname{Mat}(u)$ (con las \omterm{def:b1:vspaces:free}{bases} coincidiendo en el medio). De
igual modo, si $X$ es la columna de \omterm{prop:b1:vspaces:coordinates}{coordenadas} de $x$, la columna
de $u(x)$ es $AX$.
\end{definition}

\begin{proof}[Demostración de la fórmula de la composición]
\[
v(u(e_k)) = v\Bigl(\sum_j b_{jk} f_j\Bigr) = \sum_j b_{jk}\, v(f_j)
= \sum_j b_{jk} \sum_i a_{ij}\, g_i
= \sum_i \Bigl(\sum_j a_{ij} b_{jk}\Bigr) g_i . \qedhere
\]
\end{proof}

\begin{proposition}[El álgebra $\mathcal{M}_n(K)$]\label{prop:b1:matrices:ring}
Las matrices cuadradas $\mathcal{M}_n(K)$ forman un \omterm{def:b1:structures:ring}{anillo} (no
conmutativo para $n \geq 2$), con unidad $I_n$; su \omterm{def:b1:structures:group}{grupo} de unidades
es el \emph{grupo lineal general}\index{grupo lineal general}
$GL_n(K)$, que corresponde a los endomorfismos \omterm{def:b1:logic:inj}{biyectivos}. Para
$A, B \in \mathcal{M}_n(K)$:
\[
AB = I_n \implies A \in GL_n(K) \text{ y } B = A^{-1}
\]
(las inversas por un lado lo son por los dos, por
el~\cref{cor:b1:linmaps:samedim}).
\end{proposition}

\begin{proof}
Los axiomas de \omterm{def:b1:structures:ring}{anillo} se transportan desde $\mathcal{L}(E)$ por el
isomorfismo de la~\cref{def:b1:matrices:def}: convierte la composición
en producto y la suma en suma, de modo que la asociatividad, la
distributividad y el papel de $I_n$ se heredan de los hechos
correspondientes sobre aplicaciones, sin ninguna comprobación
entrada a entrada. No conmutatividad:
$E_{12}E_{21} = E_{11} \neq E_{22} = E_{21}E_{12}$. Si $AB = I_n$:
el endomorfismo $a$ de $A$ cumple $a \circ b = \mathrm{id}$, luego
$a$ es \omterm{def:b1:logic:inj}{sobreyectivo} ($x = a(b(x))$ exhibe una \omterm{def:b1:logic:map}{imagen recíproca} de
cada $x$) y, por tanto, \omterm{def:b1:logic:inj}{biyectivo} en \omterm{def:b1:findim:def}{dimensión finita}
(\cref{cor:b1:linmaps:samedim}); componiendo
$a \circ b = \mathrm{id}$ con $a^{-1}$ por la izquierda se obtiene
$b = a^{-1}$ y, entonces, también $b\circ a = \mathrm{id}$: la
inversa por un lado lo era por los dos desde el principio --- un
favor estrictamente \omterm{def:b1:findim:def}{de dimensión finita}.
\end{proof}

\begin{definition}[Traspuesta; traza]\label{def:b1:matrices:transpose}
La \emph{traspuesta} de $A = (a_{ij}) \in \mathcal{M}_{n,p}$ es
$A^{\mathsf T} = (a_{ji}) \in \mathcal{M}_{p,n}$; cumple
$(AB)^{\mathsf T} = B^{\mathsf T} A^{\mathsf T}$ y
$(A^{\mathsf T})^{\mathsf T} = A$. La \emph{traza}\index{traza} de
una matriz cuadrada es $\operatorname{tr} A = \sum_i a_{ii}$; es
\omterm{def:b1:linmaps:def}{lineal}, y
\[
\operatorname{tr}(AB) = \operatorname{tr}(BA)
\qquad (A \in \mathcal{M}_{n,p},\ B \in \mathcal{M}_{p,n}).
\]
\end{definition}

\begin{proof}[Demostración de la identidad de la traza]
$\operatorname{tr}(AB) = \sum_i \sum_j a_{ij} b_{ji}$ y
$\operatorname{tr}(BA) = \sum_j \sum_i b_{ji} a_{ij}$: la misma suma
doble.
\end{proof}

\begin{example}[La traza en acción]
\label{ex:b1:matrices:tracework}
La \omterm{def:b1:linmaps:projection}{proyección} del~\cref{ch:b1:linmaps} sobre
$\operatorname{Vect}(1,1)$ paralelamente a
$\operatorname{Vect}(0,1)$, $p(x, y) = (x, x)$, tiene matriz
$A = \begin{pmatrix} 1 & 0\\ 1 & 0\end{pmatrix}$ en la \omterm{def:b1:vspaces:free}{base}
canónica: en efecto, $A^2 = A$, y
\[
\operatorname{tr} A = 1 = \operatorname{rk} A ,
\]
lo que ilustra el~\cref{exo:b1:matrices:8}: para los idempotentes, la
\omterm{def:b1:matrices:transpose}{traza} \emph{cuenta} la dimensión de la \omterm{def:b1:linmaps:kerim}{imagen}, sea cual sea la
\omterm{def:b1:vspaces:free}{base} torcida en la que esté escrita la matriz. El mecanismo de la
invariancia es la identidad
$\operatorname{tr}(AB) = \operatorname{tr}(BA)$:
\[
\operatorname{tr}\bigl(P^{-1}(AP)\bigr) =
\operatorname{tr}\bigl((AP)P^{-1}\bigr) = \operatorname{tr} A ,
\]
de modo que todas las \omterm{thm:b1:matrices:conjugation}{matrices semejantes} a $A$ comparten su
\omterm{def:b1:matrices:transpose}{traza} --- el primer \emph{invariante numérico} de un endomorfismo,
al que se sumará el determinante en el~\cref{ch:b1:det} (la pareja
$(s, p)$ del problema del fin de semana de más abajo).
\end{example}

\begin{example}[Simétrica más antisimétrica]
\label{ex:b1:matrices:symsplit}
Dígase que $A$ es \emph{simétrica} cuando $A^{\mathsf T} = A$, y
\emph{antisimétrica} cuando $A^{\mathsf T} = -A$. Toda matriz
cuadrada se parte de forma única como una más la otra:
\[
A = \underbrace{\frac{A + A^{\mathsf T}}{2}}_{\text{simétrica}}
+ \underbrace{\frac{A - A^{\mathsf T}}{2}}_{\text{antisimétrica}},
\]
y una matriz que sea las dos cosas es nula ($A = -A$): los dos
\omterm{def:b1:logic:sets}{conjuntos} son \omterm{def:b1:vspaces:sum}{subespacios suplementarios} de $\mathcal{M}_n(K)$ ---
el análogo exacto de la separación par/impar de las funciones
(\cref{ex:b1:vspaces:evenodd}), con la \omterm{def:b1:matrices:transpose}{trasposición} en el papel de
$x \mapsto -x$. Dimensiones: una matriz simétrica es \omterm{def:b1:vspaces:free}{libre} en la
diagonal y por encima, y una antisimétrica, estrictamente por encima
(diagonal nula):
\[
\frac{n(n+1)}{2} + \frac{n(n-1)}{2} = n^2 ,
\]
y el que la cuenta cuadre es la confirmación de Grassmann de que la
suma es directa. Para $n = 2$:
$\begin{pmatrix} 1 & 5\\ 1 & 2\end{pmatrix} = \begin{pmatrix} 1 & 3\\
3 & 2\end{pmatrix} + \begin{pmatrix} 0 & 2\\ -2 & 0\end{pmatrix}$.
Las matrices simétricas vuelven como los datos de segunda \omterm{def:b1:derivative:def}{derivada}
del~\cref{ch:b1:multivar} (la terna de Monge $r, s, t$), y las
simétricas ortogonales se clasifican en el~\cref{exo:b1:euclid:12}.
\end{example}

\section{Cambio de base}

\begin{definition}\label{def:b1:matrices:changeofbasis}
Sean $\mathcal{B}, \mathcal{B}'$ \omterm{def:b1:vspaces:free}{bases} de $E$. La \emph{matriz de
cambio de base}\index{cambio de base}
$P = P_{\mathcal{B}\to\mathcal{B}'}$ tiene por columnas las
\omterm{prop:b1:vspaces:coordinates}{coordenadas} de los vectores de la \omterm{def:b1:vspaces:free}{base} \emph{nueva} en la \omterm{def:b1:vspaces:free}{base}
\emph{antigua}. Es invertible, $P^{-1} =
P_{\mathcal{B}'\to\mathcal{B}}$, y las \omterm{prop:b1:vspaces:coordinates}{coordenadas} se transforman por
$X = PX'$ (antiguas $=$ $P\,\cdot$ nuevas).
\end{definition}

\begin{example}[Leer la matriz de cambio de base]
\label{ex:b1:matrices:changerun}
En $\R^2$, de la canónica $\mathcal B$ a
$\mathcal B' = \bigl((1,1), (1,-1)\bigr)$:
\[
P = P_{\mathcal B\to\mathcal B'} =
\begin{pmatrix} 1 & 1\\ 1 & -1 \end{pmatrix}
\]
(los vectores nuevos escritos en \omterm{prop:b1:vspaces:coordinates}{coordenadas} antiguas, columna a
columna). El vector de \omterm{prop:b1:vspaces:coordinates}{coordenadas} antiguas
$X = (3, 1)^{\mathsf T}$ tiene \omterm{prop:b1:vspaces:coordinates}{coordenadas} nuevas
$X' = P^{-1}X = \frac12(3 + 1,\ 3 - 1)^{\mathsf T} =
(2, 1)^{\mathsf T}$: en efecto, $2(1,1) + 1(1,-1) = (3,1)$. Ojo con
el sentido --- la matriz $P$ se construye con la \omterm{def:b1:vspaces:free}{base} \emph{nueva}
pero convierte \emph{de nuevas a antiguas} las \omterm{prop:b1:vspaces:coordinates}{coordenadas}
($X = PX'$); pasar de antiguas a nuevas cuesta la inversa. Escribir
la comprobación $2(1,1) + (1,-1) = (3,1)$ tras cada conversión atrapa
el error de invertir $P$, que es el fallo más frecuente del capítulo.
\end{example}

\begin{theorem}[Cambio de base para una aplicación]\label{thm:b1:matrices:conjugation}
Sea $u \in \mathcal{L}(E)$ con matriz $A$ en $\mathcal{B}$ y $A'$ en
$\mathcal{B}'$, y sea $P = P_{\mathcal{B}\to\mathcal{B}'}$. Entonces
\[
A' = P^{-1} A\, P .
\]
Dos matrices relacionadas así se llaman
\emph{semejantes}\index{matrices semejantes}. (Para
$u \colon E \to F$ con dos parejas de \omterm{def:b1:vspaces:free}{bases}, la fórmula es
$A' = Q^{-1} A P$ --- matrices \emph{equivalentes}\index{matrices
equivalentes}.)
\end{theorem}

\begin{proof}
Para cualquier $x$: $X = PX'$ y la \omterm{def:b1:linmaps:kerim}{imagen} cumple $Y = AX$,
$Y = PY'$. Luego $PY' = APX'$, es decir, $Y' = (P^{-1}AP)X'$ para
todo $X'$: la matriz de $u$ en la \omterm{def:b1:vspaces:free}{base} nueva es $P^{-1}AP$ (tómense
como $X'$ las columnas canónicas).
\end{proof}

\begin{example}[Una buena base vuelve transparente una aplicación]
\label{ex:b1:matrices:conjugationrun}
Sea $u(x, y) = (y, x)$ (el intercambio), con matriz
$A = \begin{pmatrix} 0 & 1\\ 1 & 0\end{pmatrix}$ en la \omterm{def:b1:vspaces:free}{base}
canónica. En la base $\mathcal B' = \bigl((1,1), (1,-1)\bigr)$:
\[
P = \begin{pmatrix} 1 & 1\\ 1 & -1 \end{pmatrix},
\qquad
P^{-1} = \frac12\begin{pmatrix} 1 & 1\\ 1 & -1 \end{pmatrix},
\qquad
P^{-1} A P = \begin{pmatrix} 1 & 0\\ 0 & -1 \end{pmatrix}.
\]
En realidad no hacía falta ningún producto de matrices: $u$ deja fijo
$(1,1)$ e invierte $(1,-1)$, así que en $\mathcal B'$ su matriz
\emph{tiene} que ser $\operatorname{diag}(1, -1)$ --- el intercambio
es la reflexión respecto de la recta $y = x$. Hallar, para un
endomorfismo dado, una \omterm{def:b1:vspaces:free}{base} en la que su matriz sea diagonal es el
problema central del volumen del segundo año (la teoría de
reducción); y el problema del fin de semana de más abajo muestra
hasta dónde llegan ya las solas identidades polinómicas.
\end{example}

\begin{example}[Cambio de base, al revés]
\label{ex:b1:matrices:reverserun}
La \omterm{def:b1:linmaps:projection}{proyección} sobre $F = \operatorname{Vect}(1,1)$ paralelamente a
$G = \operatorname{Vect}(1,-1)$ tiene, en la \omterm{def:b1:vspaces:free}{base} adaptada
$\mathcal B' = \bigl((1,1),(1,-1)\bigr)$, la matriz transparente
$A' = \operatorname{diag}(1, 0)$. Para obtener su matriz en la
\omterm{def:b1:vspaces:free}{base} canónica, ejecútese al revés
el~\cref{thm:b1:matrices:conjugation}, $A = P A' P^{-1}$:
\[
P = \begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix},
\quad
P^{-1} = \frac12\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix},
\quad
A = P\begin{pmatrix} 1 & 0\\ 0 & 0\end{pmatrix}P^{-1}
= \frac12\begin{pmatrix} 1 & 1\\ 1 & 1\end{pmatrix}.
\]
Comprobación: $A^2 = A$ (idempotente),
$\operatorname{tr} A = 1 = \operatorname{rk} A$ y
$A\binom{1}{1} = \binom11$, $A\binom{1}{-1} = 0$, como se prescribió.
Este sentido inverso ---diséñese la matriz en la \omterm{def:b1:vspaces:free}{base} buena y
conjúguese después de vuelta--- es como se producen en la práctica
las matrices de rotación, de reflexión y de \omterm{def:b1:linmaps:projection}{proyección}.
\end{example}

\begin{theorem}[Forma normal del rango]\label{thm:b1:matrices:rank}
El \emph{rango} de una matriz (el rango de sus columnas,
equivalentemente el de la \omterm{def:b1:linmaps:def}{aplicación lineal} asociada) es el único
invariante de la equivalencia: toda $A \in \mathcal{M}_{n,p}$ de
rango $r$ es equivalente a
\[
J_r = \begin{pmatrix} I_r & 0 \\ 0 & 0 \end{pmatrix},
\]
y $\operatorname{rk}(A^{\mathsf T}) = \operatorname{rk}(A)$: el rango
por filas es igual al rango por columnas.
\end{theorem}

\begin{proof}
Sea $u \colon E \to F$ de rango $r$. Elíjase un \omterm{def:b1:vspaces:sum}{suplementario} $S$ de
$\ker u$ ($\dim S = r$, \cref{thm:b1:linmaps:ranknullity}) con base
$(e_1, \dots, e_r)$, completada por una \omterm{def:b1:vspaces:free}{base} de $\ker u$ hasta una
\omterm{def:b1:vspaces:free}{base} de $E$; las imágenes $f_i = u(e_i)$, $i \leq r$, forman una
\omterm{def:b1:vspaces:free}{base} de $\operatorname{im} u$ (la restricción es un isomorfismo),
completada hasta una \omterm{def:b1:vspaces:free}{base} de $F$. En esas \omterm{def:b1:vspaces:free}{bases}, la matriz de $u$ es
exactamente $J_r$. Luego $A = Q J_r P^{-1}$ con $P, Q$ invertibles.

Trasponiendo:
$A^{\mathsf T} = (P^{-1})^{\mathsf T} J_r^{\mathsf T} Q^{\mathsf T}$
con $J_r^{\mathsf T}$ de la misma forma (rango $r$) y los factores
exteriores invertibles (la \omterm{def:b1:matrices:transpose}{traspuesta} de una invertible es
invertible, aplicando $(AB)^{\mathsf T} = B^{\mathsf T}A^{\mathsf T}$
a $AA^{-1} = I$): $\operatorname{rk} A^{\mathsf T} = r$.
\end{proof}

\section{Operaciones por filas}

\begin{method}[Eliminación de Gauss sobre matrices]\label{met:b1:matrices:gauss}
Las tres \emph{operaciones elementales por
filas}\index{operaciones por filas} ---intercambiar dos filas,
multiplicar una fila por $\lambda \neq 0$, sumar a una fila un
múltiplo de otra--- no cambian el rango (cada una es una
multiplicación por la izquierda por una matriz invertible).
Algoritmo: créese un pivote (la entrada no nula más a la izquierda),
límpiese su columna por debajo y pásese a la fila y la columna
siguientes; el número de pivotes de la forma escalonada resultante es
el rango.

\emph{Cálculo de la inversa:} ejecútese el algoritmo sobre el bloque
$(A \mid I_n)$ hasta que el bloque izquierdo sea $I_n$ (posible si y
solo si $A$ es invertible); el bloque derecho es entonces $A^{-1}$
--- en efecto, el producto de las matrices elementales usadas es
igual a $A^{-1}$.
\end{method}

\begin{example}\label{ex:b1:matrices:inverse}
$A = \begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}$: redúzcase
$(A \mid I_2)$:
\[
\begin{pmatrix} 1 & 2 & 1 & 0\\ 3 & 4 & 0 & 1 \end{pmatrix}
\to
\begin{pmatrix} 1 & 2 & 1 & 0\\ 0 & -2 & -3 & 1 \end{pmatrix}
\to
\begin{pmatrix} 1 & 0 & -2 & 1\\ 0 & 1 & \tfrac32 & -\tfrac12
\end{pmatrix},
\]
(operaciones: $L_2 \leftarrow L_2 - 3L_1$; después
$L_1 \leftarrow L_1 + L_2$, $L_2 \leftarrow -\frac12 L_2$). Luego
$A^{-1} = \begin{pmatrix} -2 & 1 \\ \tfrac32 &
-\tfrac12\end{pmatrix}$. \emph{Comprobación:} $AA^{-1} = I_2$.
\end{example}

\begin{example}[Rango con un parámetro, solo por filas]
\label{ex:b1:matrices:rankparameter}
Para $m \in \R$, el rango de
$M_m = \begin{pmatrix} 1 & 1 & m\\ 1 & m & 1\\ m & 1 &
1\end{pmatrix}$. Redúzcase: $L_2 \leftarrow L_2 - L_1$ y
$L_3 \leftarrow L_3 - mL_1$ dan las filas
\[
(1,\ 1,\ m), \qquad (0,\ m - 1,\ 1 - m), \qquad
(0,\ 1 - m,\ 1 - m^2).
\]
\emph{Caso $m = 1$}: las dos últimas filas se anulan --- un pivote,
$\operatorname{rk} M_1 = 1$ (las tres filas originales eran iguales).
\emph{Caso $m \neq 1$}: escálese $L_2$ por $\frac1{m-1}$ y $L_3$ por
$\frac1{1-m}$ para obtener $(0, 1, -1)$ y $(0, 1, 1 + m)$, y después
$L_3 \leftarrow L_3 - L_2 = (0, 0, m + 2)$. Si $m = -2$: dos pivotes,
rango $2$; en los demás casos, tres pivotes, rango $3$. Resumen:
\[
\operatorname{rk} M_m =
\begin{cases}
1 & m = 1,\\
2 & m = -2,\\
3 & \text{en los demás casos}.
\end{cases}
\]
Los mismos umbrales saldrán de un solo cálculo de determinante en
el~\cref{ch:b1:det} (el \omterm{def:b1:poly:def}{polinomio} $-(m+2)(m-1)^2$
del~\cref{exo:b1:det:7}) --- pero obsérvese lo que da la eliminación
y no da el determinante: el \emph{valor} del rango en los casos
degenerados, y no solo el hecho de que haya bajado.
\end{example}

\begin{example}[Calcular potencias]\label{ex:b1:matrices:powers}
$A = \begin{pmatrix} 1 & 1 \\ 0 & 1\end{pmatrix} = I + N$ con
$N = E_{12}$, $N^2 = 0$. Como $I$ y $N$ conmutan, el teorema del
binomio (\cref{prop:b1:structures:binomial}) se trunca:
\[
A^k = I + kN = \begin{pmatrix} 1 & k \\ 0 & 1 \end{pmatrix}
\qquad (k \in \N, \text{ y } k \in \Z \text{ usando } A^{-1} = I -
N).
\]
\end{example}

\begin{method}[Calcular $A^n$: los tres caminos]
\label{met:b1:matrices:powers}
\begin{enumerate}
  \item \emph{Camino binomial}: si $A = \lambda I + N$ con $N$
        nilpotente, el teorema del binomio se trunca
        (\cref{ex:b1:matrices:powers}, \cref{exo:b1:matrices:5});
        y se aplica porque $\lambda I$ conmuta con todo.
  \item \emph{Camino polinómico}: hállese una identidad
        polinómica que cumpla $A$ (en dimensión $2$, siempre
        $A^2 = sA - pI$) y redúzcase $X^n$ \omterm{def:b1:complex:field}{módulo} ella; el problema
        del fin de semana de más abajo construye este camino por
        completo.
  \item \emph{Camino de la semejanza}: hállese una $P$ invertible
        con $P^{-1}AP = D$ sencilla (diagonal, o de
        desplazamiento), calcúlese $D^n$ y deshágase:
        $A^n = P D^n P^{-1}$ (\cref{thm:b1:matrices:conjugation},
        \cref{ex:b1:matrices:conjugationrun}); la búsqueda
        sistemática de esa $P$ es la teoría de reducción del
        segundo año.
\end{enumerate}
Sea cual sea el camino, compruébese el resultado en $n = 0, 1, 2$:
tres pruebas baratas que atrapan casi todos los resbalones.
\end{method}

\begin{remark}[Errores frecuentes: el precio de la no conmutatividad]
\label{rem:b1:matrices:pitfalls}
Toda identidad del álgebra escalar cuya demostración reordene
factores muere en $\mathcal{M}_n(K)$, $n \geq 2$.
\emph{Cuadrados}: $(A + B)^2 = A^2 + AB + BA + B^2$, y el centro se
colapsa a $2AB$ solo si $AB = BA$ (\cref{exo:b1:matrices:1}).
\emph{Potencias de productos}: $(AB)^k$ es $ABAB\cdots$, no
$A^kB^k$.
\emph{Divisores de cero}: $E_{12}E_{12} = 0$ con $E_{12} \neq 0$; y,
en consecuencia, \emph{no hay cancelación}: $AB = AC$ implica $B = C$
solo cuando $A$ es invertible (multiplíquese por $A^{-1}$ --- por el
lado correcto).
\emph{\omterm{def:b1:matrices:transpose}{Trazas}}: $\operatorname{tr}(AB) = \operatorname{tr}(BA)$
siempre, pero $\operatorname{tr}(AB) \neq
\operatorname{tr}A\operatorname{tr}B$ en general (tómese
$A = B = I_2$: $2 \neq 4$), y $\operatorname{tr}(ABC) =
\operatorname{tr}(BCA)$ (cíclica) mientras que
$\operatorname{tr}(ACB)$ puede ser distinta.
\emph{Las \omterm{def:b1:matrices:transpose}{traspuestas} invierten el orden}: $(AB)^{\mathsf T} =
B^{\mathsf T}A^{\mathsf T}$ --- olvidar la inversión es el error más
frecuente en los cálculos de ortogonalidad (\cref{ch:b1:euclid}).
En caso de duda, pruébese toda identidad supuesta con $E_{12}$ y
$E_{21}$: la pareja no conmutativa más pequeña refuta en una línea
casi todas las fórmulas falsas.
\end{remark}

\begin{remark}[Adónde va el diccionario]
\label{rem:b1:matrices:whereused}
El diccionario matricial se usa en todas las páginas restantes de
este volumen: el~\cref{ch:b1:det} asocia a cada matriz cuadrada un
único número que decide la invertibilidad, y resuelve $AX = B$ de
forma sistemática; el~\cref{ch:b1:euclid} señala las matrices que
conservan las longitudes (las matrices ortogonales); y en
el~\cref{ch:b1:multivar}, el comportamiento de segundo orden de una
función de dos variables es una matriz simétrica $2 \times 2$. La
\omterm{def:b1:matrices:transpose}{traza}, introducida arriba casi de pasada, se convierte en un
invariante potente: los \cref{exo:b1:matrices:6,exo:b1:matrices:8}
dan un primer sabor, y el volumen del segundo año construye sobre
ella la teoría de valores propios. El problema del fin de semana
desarrolla el otro caballo de batalla: las identidades polinómicas
que cumple una matriz, que convierten el cálculo de $A^n$ en una
\omterm{pb:b1:matrices:1}{recurrencia lineal} de dos términos.
\end{remark}

\begin{remark}[Perspectivas dentro del libro 3]
\label{rem:b1:matrices:perspectives}
Tres familias de matrices presentadas aquí tienen cita más adelante
en este volumen. Las \emph{matrices simétricas}
(\cref{ex:b1:matrices:symsplit}) llevan los datos de segundo orden de
las funciones de dos variables: el test de Monge
del~\cref{ch:b1:multivar} es un \omterm{def:b1:logic:statement}{enunciado} sobre el comportamiento del
signo de una matriz simétrica $2\times2$, y su determinante
$rt - s^2$ lo calcula la maquinaria del~\cref{ch:b1:det}. Las
\emph{matrices ortogonales} ($A^{\mathsf T}A = I$) son las isometrías
del~\cref{ch:b1:euclid}, donde la \omterm{def:b1:matrices:transpose}{traspuesta} adquiere por fin su
significado geométrico: es la sombra algebraica del producto escalar.
Y las \emph{matrices invertibles} encuentran su prueba práctica en
el~\cref{ch:b1:det} ---un solo número, $\det A \neq 0$---, cerrando
la búsqueda que este capítulo empezó con la reducción por filas. La
\omterm{def:b1:matrices:transpose}{traza} y el determinante viajan después como la pareja invariante
$(s, p)$ del problema del fin de semana, hasta la teoría de valores
propios del segundo año.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:matrices:1}
Sean $A = \begin{pmatrix} 1 & 2 \\ 0 & 1 \end{pmatrix}$ y
$B = \begin{pmatrix} 0 & 1 \\ 1 & 0\end{pmatrix}$. Calcúlense $AB$,
$BA$, $A^2 - B^2$ y $(A+B)(A-B)$; explíquese por qué los dos últimos
difieren.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:matrices:2}
Calcúlese el rango de
\[
M = \begin{pmatrix}
1 & 2 & 3\\
2 & 4 & 6\\
1 & 1 & 1
\end{pmatrix},
\qquad
N = \begin{pmatrix}
1 & 1 & 0 & 2\\
0 & 1 & 1 & 1\\
1 & 2 & 1 & 3
\end{pmatrix}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:matrices:3}
Inviértase, por reducción por filas, $A = \begin{pmatrix} 1 & 0 & 1\\
2 & 1 & 1\\ 1 & 1 & 1 \end{pmatrix}$, y compruébese con un producto.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:matrices:4}
Escríbase la matriz, en la \omterm{def:b1:vspaces:free}{base} canónica de $\R_2[X]$, del
endomorfismo $u(P) = P(X + 1)$. Explíquese, sin calcular, por qué es
invertible, y dese la matriz de $u^{-1}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:5}
Sea $A = \begin{pmatrix} 2 & 1 \\ 0 & 2\end{pmatrix}$. Escríbase
$A = 2I + N$, calcúlese $N^2$ y dedúzcase $A^k$ para todo $k \in \N$
con el teorema del binomio.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:6}
Demuéstrese que no hay matrices $A, B \in \mathcal{M}_n(K)$ (con
$K = \R$ o $\C$) tales que $AB - BA = I_n$. \emph{(Tómense \omterm{def:b1:matrices:transpose}{trazas}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:7}
Una matriz $A$ es \emph{nilpotente} cuando $A^m = 0$ para algún $m$.
Demuéstrese que entonces $I - A$ es invertible, con
\[
(I - A)^{-1} = I + A + A^2 + \dots + A^{m-1} .
\]
Aplicación: inviértase $\begin{pmatrix} 1 & 2 & 3\\ 0 & 1 & 2\\ 0 & 0
& 1\end{pmatrix}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:8}
Sea $A \in \mathcal{M}_n(\R)$ con $A^2 = A$ (idempotente).
Demuéstrese que $\operatorname{tr} A = \operatorname{rk} A$.
\emph{(Interprétese $A$ como una \omterm{def:b1:linmaps:projection}{proyección} y elíjase una \omterm{def:b1:vspaces:free}{base}
adaptada; el~\cref{thm:b1:matrices:conjugation} dice que la \omterm{def:b1:matrices:transpose}{traza} no
depende de la \omterm{def:b1:vspaces:free}{base}, ya que $\operatorname{tr}(P^{-1}MP) =
\operatorname{tr} M$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:matrices:9}
Sea $J \in \mathcal{M}_n(\R)$ la matriz de todo unos. Calcúlese
$J^2$ y dedúzcase, para $a, b \in \R$, la condición de
invertibilidad de $M = aI + bJ$ junto con $M^{-1}$ \emph{(búsquese
una inversa de la misma forma $\alpha I + \beta J$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:matrices:10}
(Desigualdades de rango) Para $A, B \in \mathcal{M}_n(K)$, demuéstrese
\[
\operatorname{rk}(A + B) \leq \operatorname{rk} A +
\operatorname{rk} B,
\qquad
\operatorname{rk}(AB) \geq \operatorname{rk} A + \operatorname{rk}
B - n .
\]
\emph{(Para la segunda ---la desigualdad de Sylvester--- aplíquese el
teorema del rango a la restricción de la \omterm{def:b1:logic:map}{aplicación} de $A$ a
$\operatorname{im} B$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:matrices:11}
Sea $D = \operatorname{diag}(d_1, \dots, d_n)$ con los $d_i$
\emph{distintos dos a dos}.
\begin{enumerate}
  \item Demuéstrese que una matriz $A$ conmuta con $D$ si y solo si
        $A$ es diagonal. \emph{(Compárense las entradas $(i,j)$ de
        $AD$ y de $DA$.)}
  \item Dedúzcase el \emph{centro} de $\mathcal{M}_n(K)$: las
        matrices que conmutan con \emph{todas} las matrices son
        exactamente las matrices escalares $\lambda I_n$.
        \emph{(Pruébese contra $D$ y después contra las $E_{ij}$.)}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:matrices:12}
(Matrices de rango uno) Sea $A \in \mathcal{M}_n(K)$, $A \neq 0$.
\begin{enumerate}
  \item Demuéstrese que $\operatorname{rk} A = 1$ si y solo si
        $A = CL$ para cierta columna no nula
        $C \in \mathcal{M}_{n,1}$ y cierta fila no nula
        $L \in \mathcal{M}_{1,n}$.
  \item Para una $A$ así, demuéstrese $A^2 = (\operatorname{tr} A)\,A$;
        dedúzcase que una matriz de rango uno es nilpotente si y solo
        si su \omterm{def:b1:matrices:transpose}{traza} es nula.
  \item Si $\operatorname{tr} A \neq -1$, demuéstrese que $I_n + A$
        es invertible con
        \[
        (I_n + A)^{-1} = I_n - \frac{1}{1 + \operatorname{tr}
        A}\,A ,
        \]
        y que $I_n + A$ \emph{no} es invertible cuando
        $\operatorname{tr} A = -1$. \emph{(Hállese un vector que
        mate $I_n + A$.)}
\end{enumerate}
\end{exercise}

\section{Problema: potencias de una matriz por división polinómica}

\begin{problem}\label{pb:b1:matrices:1}
Calcular $A^{100}$ entrada a entrada es desesperado; calcularlo
mediante una identidad polinómica que cumpla $A$ son tres líneas.
Este problema construye el método desde cero: la división euclídea de
$X^n$, la identidad $A^2 - sA + pI = 0$ que cumple toda matriz
$2 \times 2$ (el teorema de Cayley--Hamilton en dimensión $2$) y el
diccionario entre las potencias de una matriz y las
\omterm{pb:b1:matrices:1}{recurrencias lineales} --- con los \omterm{pb:b1:matrices:1}{números de Fibonacci} como ejemplo
recurrente.
\index{teorema de Cayley--Hamilton (dimensión 2)}
\index{números de Fibonacci}\index{recurrencia lineal}

\medskip
\noindent\textbf{Parte I --- El cálculo con restos.} Fíjense
$s, p \in K$ y $D = X^2 - sX + p$.
\begin{enumerate}
  \item Justifíquese que para cada $n \in \N$ hay unos únicos
        $Q_n \in K[X]$ y $(a_n, b_n) \in K^2$ con
        \[
        X^n = Q_n\,D + a_n X + b_n ,
        \]
        y calcúlense $(a_0, b_0)$ y $(a_1, b_1)$.
  \item Multiplicando por $X$ y dividiendo de nuevo, establézcanse
        las recurrencias
        \[
        a_{n+1} = s\,a_n + b_n,
        \qquad
        b_{n+1} = -p\,a_n ,
        \]
        y dedúzcase $a_{n+2} = s\,a_{n+1} - p\,a_n$: la sucesión de
        coeficientes obedece la \omterm{pb:b1:matrices:1}{recurrencia lineal} asociada a $D$.
  \item Supóngase que $D$ tiene dos raíces distintas
        $\lambda \neq \mu$. Evaluando la identidad de la división,
        demuéstrese
        \[
        a_n = \frac{\lambda^n - \mu^n}{\lambda - \mu},
        \qquad
        b_n = \frac{\lambda\mu^n - \mu\lambda^n}{\lambda - \mu} .
        \]
  \item Supóngase $D = (X - \lambda)^2$. Usando la \omterm{def:b1:derivative:def}{derivada} de la
        identidad de la división, demuéstrese $a_n = n\lambda^{n-1}$
        y $b_n = (1 - n)\lambda^{n}$.
  \item Véase que sustituir una matriz fija
        $M \in \mathcal{M}_k(K)$ en los \omterm{def:b1:poly:def}{polinomios} respeta las sumas
        y los productos: $(PQ)(M) = P(M)\,Q(M)$. Dedúzcase que si
        $D(M) = 0$, entonces
        \[
        M^n = a_n\,M + b_n\,I \qquad (n \in \N).
        \]
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Dimensión 2: \omterm{def:b1:matrices:transpose}{traza}, número
determinante, Cayley--Hamilton.} Para
$A = \begin{pmatrix} a & b\\ c & d\end{pmatrix}$, póngase
$s = a + d = \operatorname{tr} A$ y $p = ad - bc$ (el número que
el~\cref{ch:b1:det} llamará determinante).
\begin{enumerate}[resume]
  \item Compruébese por cálculo directo la \emph{identidad de
        Cayley--Hamilton en dimensión $2$}:
        \[
        A^2 - s\,A + p\,I_2 = 0 .
        \]
  \item Demuéstrese, desarrollando directamente, que $p$ es
        multiplicativo: con notación evidente,
        $p(AB) = p(A)\,p(B)$. Véase después que $A$ es invertible si
        y solo si $p \neq 0$, y en ese caso
        \[
        A^{-1} = \frac1p\,\bigl(s\,I_2 - A\bigr).
        \]
  \item Sea $A = \begin{pmatrix} 1 & 1\\ 0 & 2\end{pmatrix}$.
        Calcúlense $s$, $p$ y las raíces de $D$, y dedúzcase una
        fórmula \omterm{def:b1:topology:closed}{cerrada} para $A^n$; compruébese con un cálculo
        directo de $A^2$.
  \item Sea $A = \begin{pmatrix} 3 & 1\\ -1 & 1\end{pmatrix}$.
        Véase que $D$ tiene una raíz doble y calcúlese $A^n$;
        compruébese en $n = 2$.
  \item Sea $F = \begin{pmatrix} 1 & 1\\ 1 & 0\end{pmatrix}$ y
        defínanse los \omterm{pb:b1:matrices:1}{números de Fibonacci} por $F_0 = 0$, $F_1 = 1$,
        $F_{n+2} = F_{n+1} + F_n$. Demuéstrese
        \[
        F^n = \begin{pmatrix} F_{n+1} & F_n\\ F_n &
        F_{n-1}\end{pmatrix} \quad (n \geq 1),
        \]
        dedúzcase la fórmula de Binet
        $F_n = \dfrac{\varphi^n - \psi^n}{\sqrt5}$, donde
        $\varphi = \frac{1 + \sqrt5}2$, $\psi = \frac{1 - \sqrt5}2$
        y, usando la pregunta 7, la identidad de Cassini
        $F_{n+1}F_{n-1} - F_n^2 = (-1)^n$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- \omterm{pb:b1:matrices:1}{Recurrencias lineales},
estructuralmente.} Fíjense $s, p \in K$ con $p \neq 0$, y sea $E_D$
el \omterm{def:b1:logic:sets}{conjunto} de las sucesiones con
$u_{n+2} = s\,u_{n+1} - p\,u_n$ para todo $n$.
\begin{enumerate}[resume]
  \item Véase que $E_D$ es un \omterm{def:b1:vspaces:def}{espacio vectorial} de dimensión $2$
        (adáptese el~\cref{exo:b1:findim:10}).
  \item Véase que la sucesión $(a_n)$ de la parte I es el elemento
        de $E_D$ con valores iniciales $0, 1$, y que todo
        $u \in E_D$ cumple
        \[
        u_n = u_1\,a_n + u_0\,b_n \qquad (n \in \N),
        \]
        con $(b_n)$ como en la parte I: los restos de la división
        resuelven \emph{todas} las recurrencias a la vez.
  \item Si $\lambda \neq \mu$ son las raíces de $D$, véase que
        $\bigl((\lambda^n), (\mu^n)\bigr)$ es una \omterm{def:b1:vspaces:free}{base} de $E_D$; y
        si $D = (X-\lambda)^2$ con $\lambda \neq 0$, véase que lo es
        $\bigl((\lambda^n), (n\lambda^n)\bigr)$.
  \item Resuélvase por completo: $u_{n+2} = u_{n+1} + 6u_n$,
        $u_0 = 1$, $u_1 = 8$; compruébese la respuesta en $u_2$ y
        $u_3$.
  \item Sea $C = \begin{pmatrix} 0 & 1\\ -p & s\end{pmatrix}$ (la
        \emph{matriz compañera} de $D$). Véase que
        \[
        \begin{pmatrix} u_{n}\\ u_{n+1}\end{pmatrix}
        = C^n \begin{pmatrix} u_0\\ u_1\end{pmatrix}
        \quad (u \in E_D),
        \]
        y que $\operatorname{tr} C = s$ y $p(C) = p$: la recurrencia y
        la matriz llevan el mismo \omterm{def:b1:poly:def}{polinomio} $D$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Grado tres.} Sea
$D_3 = X^3 - \alpha X^2 - \beta X - \gamma$ y
\[
C_3 = \begin{pmatrix} 0 & 1 & 0\\ 0 & 0 & 1\\ \gamma & \beta &
\alpha \end{pmatrix}.
\]
\begin{enumerate}[resume]
  \item Véase que $D_3(C_3) = 0$. \emph{(Calcúlense las imágenes de
        los vectores de la \omterm{def:b1:vspaces:free}{base} canónica por las potencias de
        $C_3$: la \omterm{def:b1:logic:map}{aplicación} de $C_3$ manda
        $e_1 \mapsto \dots \mapsto$ una combinación que fuerza la
        última fila.)}
  \item Véase que si $D_3$ tiene tres raíces distintas
        $\lambda_1, \lambda_2, \lambda_3$, el resto $R_n$ de $X^n$
        dividido entre $D_3$ es el \emph{interpolador de Lagrange}
        de los valores $\lambda_i^n$ en los nodos $\lambda_i$
        (\cref{thm:b1:poly:lagrange}); dedúzcase que toda entrada de
        $C_3^{\,n}$ es una combinación lineal fija de
        $\lambda_1^n, \lambda_2^n, \lambda_3^n$.
  \item Resuélvase: $u_{n+3} = 2u_{n+2} + u_{n+1} - 2u_n$ con
        $u_0 = 0$, $u_1 = 1$, $u_2 = 1$. \emph{(Factorícese
        $D_3 = (X - 1)(X + 1)(X - 2)$.)} Compruébese en $u_3$.
  \item Calcúlese el resto de $X^n$ \omterm{def:b1:complex:field}{módulo} $(X - \lambda)^3$
        \emph{(desarrollo de Taylor de $X^n$ en $\lambda$)}, y
        dedúzcase una fórmula para $(\lambda I + N)^n$ cuando
        $N^3 = 0$ y $N$ conmuta con todo lo que hay a la vista;
        compruébese con el teorema del binomio.
  \item Véase que, para $D_3$ con raíces distintas, la solución
        general de la recurrencia de orden $3$ es
        $u_n = c_1 \lambda_1^n + c_2\lambda_2^n + c_3\lambda_3^n$:
        demuéstrese que las tres sucesiones geométricas forman una
        \omterm{def:b1:vspaces:free}{base} del espacio de soluciones. \emph{(Para la libertad,
        evalúese una combinación nula en $n = 0, 1, 2$ y
        reconózcase un sistema de interpolación en los nodos
        distintos $\lambda_i$.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dividendos de Fibonacci, y síntesis.}
\begin{enumerate}[resume]
  \item Demuéstrese $F_1 + F_2 + \dots + F_n = F_{n+2} - 1$.
  \item De $F^{m+n} = F^m F^n$, dedúzcase la fórmula de adición
        \[
        F_{m+n} = F_{m+1}F_n + F_m F_{n-1},
        \]
        y dedúzcase $F_{2n} = F_n(F_{n+1} + F_{n-1})$.
  \item Demuéstrese que $F_n$ es el entero más próximo a
        $\varphi^n/\sqrt5$ para todo $n \geq 0$.
  \item Sea $t_n = \operatorname{tr}(F^n) = F_{n+1} + F_{n-1}$ (los
        \emph{números de Lucas} $L_n$). Véase que
        $t_{n+2} = t_{n+1} + t_n$, $t_1 = 1$, $t_2 = 3$, que
        $L_n = \varphi^n + \psi^n$, y recupérese $F_{2n} = F_n L_n$.
  \item Síntesis, en cuatro frases: por qué las potencias de una
        matriz $2 \times 2$ viven en el plano
        $\operatorname{Vect}(I, A)$ de $\mathcal{M}_2(K)$ (qué
        argumento de dimensión garantiza una identidad cuadrática y
        qué identidad explícita produjo la parte II); cómo la
        división euclídea convierte la exponenciación en una
        recurrencia de dos términos; qué \omterm{def:b1:logic:statement}{enunciado} de este problema
        es el caso $n = 2$ de un teorema válido en todas las
        dimensiones (nómbrese, y dígase dónde se demuestra en esta
        \omterm{def:b1:series:def}{serie}); y qué añade al cuadro la construcción de la matriz
        compañera.
\end{enumerate}
\end{problem}
