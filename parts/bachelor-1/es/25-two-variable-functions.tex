\chapter{Funciones de dos variables}\label{ch:b1:multivar}

El año termina con un primer paseo por la dimensión superior: las funciones
$f(x, y)$ de dos variables reales. Todo se generaliza --- límites,
\omterm{def:b1:continuity:continuous}{continuidad}, \omterm{def:b1:derivative:def}{derivadas}, extremos --- pero cada noción gana un matiz:
a un límite se puede tender por todas las direcciones a la vez, las \omterm{def:b1:derivative:def}{derivadas}
se desdoblan en parciales y el \omterm{def:b1:multivar:partial}{gradiente} señala el camino cuesta arriba. La
teoría completa (diferenciales, $\R^n$ general, subvariedades) pertenece al
segundo año; aquí fijamos el vocabulario y los primeros teoremas
honestos.

\section{El plano $\R^2$; continuidad}

\begin{definition}\label{def:b1:multivar:topology}
En $\R^2$ se usa la norma euclídea $\norm{(x,y)} = \sqrt{x^2 + y^2}$
(\cref{ch:b1:euclid}). Las bolas \omterm{def:b1:topology:open}{abiertas}, los \omterm{def:b1:topology:open}{entornos} y los \emph{subconjuntos
\omterm{def:b1:topology:open}{abiertos}} de $\R^2$ se definen exactamente como en
\cref{ch:b1:topology}, con bolas en lugar de \omterm{prop:b1:reals:intervals}{intervalos}. Una función
$f \colon U \to \R$ ($U \subseteq \R^2$ \omterm{def:b1:topology:open}{abierto}) es \emph{\omterm{def:b1:continuity:continuous}{continua}
en} $a \in U$ cuando
\[
\forall\varepsilon > 0,\ \exists\delta > 0, \quad
\norm{X - a} \leq \delta \implies \abs{f(X) - f(a)} \leq
\varepsilon,
\]
con la misma caracterización secuencial que en una variable.
Sumas, productos, cocientes y composiciones con funciones \omterm{def:b1:continuity:continuous}{continuas}
de una variable preservan la \omterm{def:b1:continuity:continuous}{continuidad}; las aplicaciones \omterm{prop:b1:vspaces:coordinates}{coordenadas} son
\omterm{def:b1:continuity:continuous}{continuas}, y por tanto también lo son los \omterm{def:b1:poly:def}{polinomios} en $(x,y)$.
\end{definition}

\begin{example}[La cota polar, la manera limpia de probar un límite]
\label{ex:b1:multivar:polarbound}
Demuéstrese que $f(x, y) = \dfrac{x^2y^2}{x^2 + y^2}$ (con $f(0,0) =
0$) es \omterm{def:b1:continuity:continuous}{continua} en el origen. En \omterm{prop:b1:vspaces:coordinates}{coordenadas} polares $x =
\rho\cos\theta$, $y = \rho\sin\theta$:
\[
\abs{f} = \frac{\rho^4\cos^2\theta\sin^2\theta}{\rho^2}
= \rho^2\,(\cos\theta\sin\theta)^2 \leq \frac{\rho^2}{4}
\xrightarrow[\rho \to 0]{} 0 ,
\]
una cota \emph{independiente de $\theta$}: sea cual sea la dirección
de aproximación, los valores quedan atrapados hacia $0$. Esa uniformidad en
$\theta$ es toda la cuestión --- una cota como $\abs g =
\abs{\cos\theta\sin\theta}$ (sin $\rho$ alguno) no prueba nada,
y de hecho ese $g$ es la trampa radial discontinua del
ejemplo siguiente.
\end{example}

\begin{example}[La trampa radial]\label{ex:b1:multivar:trap}
Sea $f(x, y) = \dfrac{xy}{x^2 + y^2}$ para $(x,y) \neq (0,0)$, $f(0,
0) = 0$. A lo largo de cada eje, $f = 0 \to 0$; pero a lo largo de la diagonal $y
= x$, $f(x, x) = \frac12 \not\to 0$. \emph{No hay límite en el origen}:
aproximarse por todas las rectas, e incluso hallar el mismo límite por
cada una, no basta (aquí los límites por rectas discrepan; hay ejemplos peores
que coinciden por todas las rectas y fallan por una parábola,
\cref{exo:b1:multivar:3}). La \omterm{def:b1:continuity:continuous}{continuidad} en cada variable por separado
no implica la \omterm{def:b1:continuity:continuous}{continuidad}.
\end{example}

\section{Derivadas parciales}

\begin{definition}\label{def:b1:multivar:partial}
Las \emph{derivadas parciales}\index{derivada parcial} de $f$ en
$(a, b)$ son las \omterm{def:b1:derivative:def}{derivadas} de una variable a lo largo de los ejes:
\[
\frac{\partial f}{\partial x}(a,b)
= \lim_{h \to 0} \frac{f(a + h, b) - f(a,b)}{h},
\qquad
\frac{\partial f}{\partial y}(a,b)
= \lim_{k \to 0} \frac{f(a, b + k) - f(a,b)}{k}.
\]
$f$ es de clase $C^1$ en $U$ cuando ambas existen y son \omterm{def:b1:continuity:continuous}{continuas} en
$U$. El \emph{gradiente}\index{gradiente} es $\nabla f(a,b) =
\bigl(\frac{\partial f}{\partial x},\, \frac{\partial f}{\partial
y}\bigr)(a,b)$.
\end{definition}

\begin{theorem}[$C^1$ implica un plano tangente]\label{thm:b1:multivar:c1}
Sea $f$ de clase $C^1$ en $U$ y sea $(a,b) \in U$. Entonces, cuando $(h, k) \to
(0,0)$:
\[
f(a + h, b + k) = f(a, b)
+ h\,\frac{\partial f}{\partial x}(a,b)
+ k\,\frac{\partial f}{\partial y}(a,b)
+ o\bigl(\norm{(h,k)}\bigr) .
\]
En particular $f$ es \omterm{def:b1:continuity:continuous}{continua}, y la gráfica $z = f(x,y)$ tiene en
cada punto el plano tangente que se lee en la fórmula.
\end{theorem}

\begin{example}[La aproximación lineal en acción]
\label{ex:b1:multivar:linapprox}
Estímese $f(1.02,\ 0.99)$ para $f(x, y) = x^3y^2$. En $(1, 1)$:
$f = 1$, $\frac{\partial f}{\partial x} = 3x^2y^2 = 3$,
$\frac{\partial f}{\partial y} = 2x^3y = 2$, luego
\cref{thm:b1:multivar:c1} da
\[
f(1.02,\ 0.99) \approx 1 + 3\,(0.02) + 2\,(-0.01) = 1.04 ,
\]
frente al valor verdadero $1.02^3 \times 0.99^2 = 1.04006\dots$ ---
el error es de segundo orden en los incrementos, como promete el
$o(\norm{(h,k)})$. El plano tangente a la gráfica en
$(1, 1, 1)$ es $z = 1 + 3(x - 1) + 2(y - 1)$, la ecuación
implícita en la estimación.
\end{example}

\begin{proof}
Se mueve una coordenada cada vez:
\[
f(a+h, b+k) - f(a,b)
= \bigl[f(a+h, b+k) - f(a, b+k)\bigr] + \bigl[f(a, b+k) -
f(a,b)\bigr].
\]
Por el teorema del valor medio en una variable
(\cref{thm:b1:derivative:mvt}), el primer corchete vale $h\,
\frac{\partial f}{\partial x}(a + \theta h,\, b + k)$ para cierto
$\theta \in \intoo{0}{1}$, y el segundo $k\,\frac{\partial
f}{\partial y}(a,\, b + \theta' k)$. La \omterm{def:b1:continuity:continuous}{continuidad} de las parciales en
$(a,b)$ permite escribir cada uno como (valor en $(a,b)$) $+$
(error $\to 0$); el error total es $h\,\varepsilon_1 +
k\,\varepsilon_2 = o(\norm{(h,k)})$ ya que $\abs h, \abs k \leq
\norm{(h,k)}$.
\end{proof}

\begin{theorem}[Regla de la cadena]\label{thm:b1:multivar:chain}
Sea $f$ de clase $C^1$ en $U$ y sea $t \mapsto (x(t), y(t))$ de clase $C^1$ de
un \omterm{prop:b1:reals:intervals}{intervalo} en $U$. Entonces $g(t) = f\bigl(x(t), y(t)\bigr)$ es
$C^1$, con\index{regla de la cadena!dos variables}
\[
g'(t) = x'(t)\,\frac{\partial f}{\partial x}\bigl(x(t),y(t)\bigr)
+ y'(t)\,\frac{\partial f}{\partial y}\bigl(x(t),y(t)\bigr)
= \bigl\langle \nabla f,\ (x', y')\bigr\rangle .
\]
\end{theorem}

\begin{proof}
Aplíquese \cref{thm:b1:multivar:c1} en $(x(t), y(t))$ con $(h, k) =
(x(t+s) - x(t),\, y(t+s) - y(t))$: cuando $s \to 0$, la \omterm{def:b1:derivative:def}{derivabilidad}
en una variable da $h = s\,x'(t) + o(s)$ y $k = s\,y'(t)
+ o(s)$, luego $\norm{(h, k)} = O(s)$ y
\[
g(t+s) - g(t)
= h\,\frac{\partial f}{\partial x} + k\,\frac{\partial
f}{\partial y} + o\bigl(\norm{(h,k)}\bigr)
= s\,\Bigl(x'\,\frac{\partial f}{\partial x} +
y'\,\frac{\partial f}{\partial y}\Bigr) + o(s) ,
\]
donde el error final absorbe tanto los $o(s)$ de $h$ y $k$
(multiplicados por los valores fijos de las parciales) como el
$o(O(s))$ de la estimación del plano tangente. Divídase por $s$ y hágase
$s \to 0$. La \omterm{def:b1:continuity:continuous}{continuidad} de $g'$ se sigue de la de todos los
ingredientes.
\end{proof}

\begin{example}[La regla de la cadena, comprobada por los dos caminos]
\label{ex:b1:multivar:chainrun}
Sean $f(x, y) = x^2 y$ y $g(t) = f(t, t^2)$. Directamente: $g(t) =
t^2\cdot t^2 = t^4$, luego $g'(t) = 4t^3$. Por la regla de la cadena:
$\frac{\partial f}{\partial x} = 2xy$ y $\frac{\partial
f}{\partial y} = x^2$, evaluadas a lo largo de la curva $(t, t^2)$:
\[
g'(t) = 1\cdot 2t\cdot t^2 + 2t\cdot t^2 = 2t^3 + 2t^3 = 4t^3 .
\]
Los dos cálculos coinciden, y el desglose es significativo: $2t^3$
del crecimiento proviene de moverse \emph{hacia la derecha} por la
pendiente en $x$, y $2t^3$ de moverse \emph{hacia arriba} por la
pendiente en $y$. En las curvas donde no existe forma \omterm{def:b1:topology:closed}{cerrada} para $g$, solo
sobrevive el segundo cálculo --- ese es el sentido del teorema.
\end{example}

\begin{remark}[Lectura del gradiente]
A lo largo de una dirección unitaria $u$, la regla de la cadena aplicada a
$t \mapsto f(a + tu)$ da la \emph{\omterm{def:b1:derivative:def}{derivada} direccional} $\langle \nabla f(a),
u\rangle$: máxima cuando $u$ apunta en el sentido de $\nabla f(a)$ (por
Cauchy--Schwarz, \cref{thm:b1:euclid:cs}). El \omterm{def:b1:multivar:partial}{gradiente} es la
dirección de máxima pendiente de subida, y es \omterm{def:b1:euclid:orthogonal}{ortogonal} a las curvas
de nivel $\{f = c\}$ (derívese $f$ a lo largo de una curva trazada en un
\omterm{def:b1:logic:sets}{conjunto} de nivel: la regla de la cadena da $\langle\nabla f,\,
\text{tangente}\rangle = 0$).
\end{remark}

\begin{example}[Curvas de nivel y gradientes, sobre una sola función]
\label{ex:b1:multivar:levelcurves}
Tómese $f(x, y) = x^2 - y^2$. Sus \omterm{def:b1:logic:sets}{conjuntos} de nivel: $\{f = c\}$ es una
hipérbola que se abre a izquierda y derecha si $c > 0$, hacia arriba y abajo si $c < 0$,
y el par de rectas cruzadas $y = \pm x$ si $c = 0$ --- el
mapa de curvas de nivel de un puerto de montaña, con el \omterm{met:b1:multivar:monge}{punto de silla} en el
origen donde se cruzan las dos rectas de nivel cero. \omterm{def:b1:multivar:partial}{Gradiente}: $\nabla
f = (2x, -2y)$. En el punto $(2, 1)$ (sobre el nivel $c = 3$):
$\nabla f = (4, -2)$, mientras que el vector tangente de la curva
de nivel, parametrizada cerca de ese punto por $\bigl(t,
\sqrt{t^2 - 3}\bigr)$, es $\bigl(1, \frac{t}{\sqrt{t^2 -
3}}\bigr) = (1, 2)$ en $t = 2$ --- y en efecto
\[
\langle (4, -2),\ (1, 2)\rangle = 4 - 4 = 0 :
\]
\omterm{def:b1:multivar:partial}{gradiente} perpendicular a la curva de nivel, apuntando hacia valores
mayores de $f$ (aquí: alejándose del eje $y$). Dos lecturas
más: el \omterm{def:b1:multivar:partial}{gradiente} se anula exactamente en el \omterm{met:b1:multivar:monge}{punto de silla}, donde
el mapa de niveles se estrangula; y la \emph{recta} tangente a la
curva de nivel en $(2,1)$ es $4(x - 2) - 2(y - 1) = 0$, es decir, $2x -
y = 3$ --- la ecuación «$\langle \nabla f,\ M - M_0\rangle =
0$» que generaliza la tangente a la elipse de
\cref{exo:b1:curves:11}.
\end{example}

\begin{theorem}[Schwarz]\label{thm:b1:multivar:schwarz}
Si $f$ es de clase $C^2$ (las parciales de las parciales existen y son
\omterm{def:b1:continuity:continuous}{continuas}), entonces\index{teorema de Schwarz}
\[
\frac{\partial^2 f}{\partial x\,\partial y}
= \frac{\partial^2 f}{\partial y\,\partial x} .
\]
\end{theorem}
\admitted

\section{Extremos locales}

\begin{method}[Estudio de extremos, organizado]
\label{met:b1:multivar:extremamethod}
\begin{enumerate}
  \item \emph{Resuélvase $\nabla f = 0$ por completo.} Factorícese cada
        parcial siempre que sea posible (los productos de factores \omterm{def:b1:linmaps:def}{lineales}
        descomponen el sistema en casos transparentes, como en
        \cref{ex:b1:multivar:fourpoints} más abajo); un caso
        olvidado es un \omterm{thm:b1:multivar:critical}{punto crítico} olvidado.
  \item \emph{Clasifíquese cada punto} con los datos de Monge $r, s,
        t$ --- recalculados \emph{en cada punto}, nunca una sola vez
        para todos.
  \item \emph{Si $rt - s^2 = 0$}, examínese $f$ directamente a lo
        largo de curvas bien elegidas que pasen por el punto (primero
        rectas, luego parábolas), buscando o bien dos signos (no hay
        extremo) o bien un signo fijo con un argumento que cubra todas
        las direcciones.
  \item \emph{Retrocédase para ver el cuadro global}: revísese el
        comportamiento en el infinito (un mínimo local puede no ser
        global) y, si el dominio no es \omterm{def:b1:topology:open}{abierto}, trátese su \omterm{def:b1:topology:closure}{frontera}
        aparte (\cref{exo:b1:multivar:12}) --- el teorema del \omterm{thm:b1:multivar:critical}{punto
        crítico} solo ve puntos \omterm{def:b1:topology:closure}{interiores}.
\end{enumerate}
\end{method}

\begin{theorem}[Puntos críticos]\label{thm:b1:multivar:critical}
Si $f$ (de clase $C^1$ en el \omterm{def:b1:topology:open}{abierto} $U$) tiene un extremo local en $(a,b)
\in U$, entonces $\nabla f(a,b) = (0,0)$: el punto es
\emph{crítico}\index{punto crítico}.
\end{theorem}

\begin{proof}
Las funciones de una variable $x \mapsto f(x, b)$ e $y \mapsto f(a,
y)$ tienen extremos locales \omterm{def:b1:topology:closure}{interiores} en $a$ y en $b$ respectivamente:
\cref{prop:b1:derivative:fermat} anula ambas parciales.
\end{proof}

\begin{method}[Criterio de segundo orden (notación de Monge)]\label{met:b1:multivar:monge}
En un \omterm{thm:b1:multivar:critical}{punto crítico} de una función $C^2$, póngase
\[
r = \frac{\partial^2 f}{\partial x^2},
\qquad
s = \frac{\partial^2 f}{\partial x \partial y},
\qquad
t = \frac{\partial^2 f}{\partial y^2}
\qquad (\text{valores en el punto}).
\]
\begin{itemize}
  \item Si $rt - s^2 > 0$: extremo local --- mínimo si $r > 0$,
        máximo si $r < 0$;
  \item si $rt - s^2 < 0$: no hay extremo (un \emph{punto de
        silla}\index{punto de silla});
  \item si $rt - s^2 = 0$: el criterio calla; examínese directamente.
\end{itemize}
(La justificación --- un desarrollo de Taylor--Young de orden $2$ y el
estudio del signo de la forma cuadrática $r h^2 + 2shk + tk^2$ --- se
lleva a cabo en el segundo año; aquí el criterio se usa como
herramienta de trabajo.)
\end{method}

\begin{example}\label{ex:b1:multivar:extrema}
$f(x,y) = x^3 + y^3 - 3xy$. \omterm{thm:b1:multivar:critical}{Puntos críticos}: $\nabla f = (3x^2 -
3y,\; 3y^2 - 3x) = 0$ da $y = x^2$ y $x = y^2$, luego $x = x^4$:
$x \in \{0, 1\}$: los puntos $(0,0)$ y $(1,1)$.

\omterm{def:b1:derivative:def}{Derivadas} segundas: $r = 6x$, $s = -3$, $t = 6y$. En $(0,0)$: $rt -
s^2 = -9 < 0$: \omterm{met:b1:multivar:monge}{punto de silla}. En $(1,1)$: $rt - s^2 = 36 - 9 > 0$, $r = 6
> 0$: mínimo local, $f(1,1) = -1$. (No global: $f(x, 0) = x^3 \to
-\infty$.)
\end{example}

\begin{example}[Un estudio de cuatro puntos, completo]
\label{ex:b1:multivar:fourpoints}
$f(x, y) = xy\,(3 - x - y) = 3xy - x^2y - xy^2$. \omterm{def:b1:multivar:partial}{Gradiente}:
\[
\frac{\partial f}{\partial x} = y\,(3 - 2x - y),
\qquad
\frac{\partial f}{\partial y} = x\,(3 - x - 2y).
\]
\omterm{thm:b1:multivar:critical}{Puntos críticos}: si $y = 0$, la segunda ecuación da $x \in
\{0, 3\}$; si $x = 0$, la primera da $y \in \{0, 3\}$; si $xy
\neq 0$, se resuelve $2x + y = 3$, $x + 2y = 3$: $x = y = 1$. Cuatro
puntos: $(0,0)$, $(3,0)$, $(0,3)$, $(1,1)$. \omterm{def:b1:derivative:def}{Derivadas} segundas:
$r = -2y$, $s = 3 - 2x - 2y$, $t = -2x$.
\begin{itemize}
  \item $(1,1)$: $r = -2$, $s = -1$, $t = -2$: $rt - s^2 = 3 >
        0$, $r < 0$: máximo local, $f(1,1) = 1$.
  \item $(0,0)$: $r = t = 0$, $s = 3$: $rt - s^2 = -9 < 0$:
        \omterm{met:b1:multivar:monge}{punto de silla}; lo mismo en $(3, 0)$ ($s = -3$) y en $(0, 3)$:
        tres \omterm{met:b1:multivar:monge}{puntos de silla}.
\end{itemize}
El máximo es solo local: $f(-T, -T) = T^2(3 + 2T) \to
+\infty$. Comprobación de simetría: $f(x, y) = f(y, x)$, y en efecto el
\omterm{def:b1:logic:sets}{conjunto} crítico y la clasificación son simétricos en $x
\leftrightarrow y$. Interpretación: entre los rectángulos con holgura
$x, y \geq 0$, $x + y \leq 3$, el producto $xy(3 - x - y)$ de
las tres «partes» de $3$ es máximo cuando las partes son iguales
--- una sombra en dos variables de la desigualdad entre medias.
\end{example}

\begin{remark}[Errores frecuentes]
\label{rem:b1:multivar:pitfalls}
\emph{Las \omterm{def:b1:multivar:partial}{derivadas parciales} pueden existir en un punto de
discontinuidad}: la trampa radial $g(x,y) = \frac{xy}{x^2+y^2}$ de
\cref{ex:b1:multivar:trap} tiene $\frac{\partial g}{\partial
x}(0,0) = \frac{\partial g}{\partial y}(0,0) = 0$ (ambas restricciones
a los ejes son idénticamente nulas), y sin embargo $g$ no tiene límite en el
origen --- las parciales sondean solo dos direcciones y la \omterm{def:b1:continuity:continuous}{continuidad}
las necesita todas; solo la hipótesis $C^1$ restablece el orden
(\cref{thm:b1:multivar:c1}).
\emph{Los límites por rectas nunca bastan}: la función del
\cref{exo:b1:multivar:3} tiene límite $0$ por toda recta y aun así no
tiene límite --- pruébense siempre las parábolas (o cotas polares
válidas uniformemente en $\theta$).
\emph{Ser crítico es necesario, no suficiente}: los \omterm{met:b1:multivar:monge}{puntos de silla}
abundan (tres de los cuatro puntos de \cref{ex:b1:multivar:fourpoints}); y
el teorema vale solo en \emph{\omterm{def:b1:topology:open}{abiertos}}: en \omterm{def:b1:logic:sets}{conjuntos} como un disco \omterm{def:b1:topology:closed}{cerrado},
los extremos pueden estar en la \omterm{def:b1:topology:closure}{frontera} con \omterm{def:b1:multivar:partial}{gradiente} no nulo
(\cref{exo:b1:multivar:12}).
\emph{El caso mudo $rt - s^2 = 0$ es genuinamente mudo}: $x^4
+ y^4$ (mínimo) y $x^3 + y^3$ (ninguno de los dos) tienen ambos $r = s = t
= 0$ en el origen; solo un estudio directo del signo decide
(\cref{exo:b1:multivar:6}, función $h$).
\emph{El \omterm{def:b1:multivar:partial}{gradiente} es \omterm{def:b1:euclid:orthogonal}{ortogonal} a las curvas de nivel, no tangente
a ellas}: para seguir una curva de nivel hay que moverse perpendicularmente a $\nabla
f$; para subir lo más deprisa posible, en su dirección --- mezclar ambas cosas invierte
la geometría de todo mapa de niveles.
\end{remark}

\begin{remark}[Adónde llevan las dos variables]
\label{rem:b1:multivar:whereused}
Este capítulo es una puerta. El \omterm{def:b1:multivar:partial}{gradiente} y la regla de la cadena se extienden
literalmente a $n$ variables en el volumen del segundo año, donde el
$o(\norm{(h,k)})$ de \cref{thm:b1:multivar:c1} se convierte en la
\emph{diferencial} y el criterio de Monge se demuestra por completo mediante la
fórmula de Taylor de orden dos y las formas cuadráticas. El caso especial
que puede resolverse \emph{este} año --- las funciones cuadráticas,
para las que el desarrollo de segundo orden es exacto --- es el objeto
del problema de fin de semana, y resulta ser el caso que gobierna
el ajuste de datos en todo el mundo: la regresión por \omterm{pb:b1:multivar:1}{mínimos cuadrados}. Los
extremos con restricciones (\cref{exo:b1:multivar:5} fue un anticipo) se convierten
en los multiplicadores de Lagrange del segundo año; las funciones armónicas
(\cref{exo:b1:multivar:7}) reaparecen en el análisis complejo del
volumen del tercer año.
\end{remark}

\begin{remark}[Perspectivas dentro del Libro 3: el año, cerrado]
\label{rem:b1:multivar:perspectives}
En este capítulo se dan la mano las dos mitades del volumen. La
mitad de análisis aportó sus herramientas una \omterm{def:b1:derivative:def}{derivada} cada vez: el
teorema del valor medio impulsa \cref{thm:b1:multivar:c1}, los desarrollos
de Taylor impulsan los criterios de extremos, y los $\varepsilon$ de
\cref{ch:b1:topology} han vuelto con bolas en lugar de
\omterm{prop:b1:reals:intervals}{intervalos}. La mitad de álgebra aportó la geometría: el \omterm{def:b1:multivar:partial}{gradiente}
se lee a través del \omterm{def:b1:euclid:def}{producto escalar} de \cref{ch:b1:euclid}
(Cauchy--Schwarz lo convierte en la dirección de máxima pendiente), los datos
de Monge $(r, s, t)$ forman una matriz simétrica de
\cref{ch:b1:matrices} con el criterio del determinante de
\cref{ch:b1:det}, y el problema de fin de semana ejecuta una \omterm{thm:b1:euclid:projection}{proyección
ortogonal} sobre vectores de datos. Incluso las curvas de
\cref{ch:b1:curves} vuelven como \omterm{def:b1:logic:sets}{conjuntos} de nivel. Quien sea capaz de
reconstruir por qué funciona cada uno de estos cinco relevos habrá,
en efecto, repasado el año entero --- que es el verdadero propósito
de este capítulo final.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:multivar:1}
Calcúlense las \omterm{def:b1:multivar:partial}{derivadas parciales}:
$f(x,y) = x^2 y + \eu^{xy}$; $\;g(x,y) = \ln(x^2 + y^2)$ (en
$\R^2\setminus\{0\}$); $\;h(x,y) = \arctan\frac yx$ (en $x > 0$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:multivar:2}
Estúdiese la \omterm{def:b1:continuity:continuous}{continuidad} en $(0,0)$ (con valor $0$ allí) de:
\[
f(x,y) = \frac{x^2 y}{x^2 + y^2},
\qquad
g(x,y) = \frac{xy}{x^2 + y^2},
\qquad
h(x,y) = \frac{x^3 + y^3}{x^2 + y^2}.
\]
\emph{(Las \omterm{prop:b1:vspaces:coordinates}{coordenadas} polares $x = \rho\cos\theta$, $y = \rho\sin\theta$
ayudan: acótese por una función solo de $\rho$ cuando sea posible.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:3}
Sea $f(x,y) = \dfrac{x^2 y}{x^4 + y^2}$ ($f(0,0) = 0$). Demuéstrese que
$f$ tiene límite $0$ en el origen \emph{a lo largo de toda recta},
pero que $f\bigl(x, x^2\bigr) = \frac12$: $f$ no es \omterm{def:b1:continuity:continuous}{continua} en
$(0,0)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:multivar:4}
Verifíquese a mano el teorema de Schwarz en $f(x, y) = x^3 y^2 + \sin(xy)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:5}
Sea $f$ de clase $C^1$ en $\R^2$ y sea $g(t) = f(\cos t, \sin t)$. Exprésese
$g'(t)$ mediante la regla de la cadena. Dedúzcase que $f$ restringida a la
circunferencia unidad alcanza sus extremos en los puntos donde $\nabla f$ es
paralelo al vector radial.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:6}
Hállense y clasifíquense los \omterm{thm:b1:multivar:critical}{puntos críticos} de:
\[
f(x, y) = x^2 + xy + y^2 - 3x,
\qquad
g(x, y) = x^2 - y^2 + 4y,
\qquad
h(x, y) = x^4 + y^4 - 2(x - y)^2 .
\]
\emph{(Para $h$, el criterio del determinante calla en el origen:
examínense $h(x, x)$ y $h(x, -x)$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:7}
Una función $f$ es \emph{armónica} cuando $\frac{\partial^2 f}{\partial
x^2} + \frac{\partial^2 f}{\partial y^2} = 0$. Compruébese que $x^2 -
y^2$, $xy$, $\eu^x\cos y$ y $\ln(x^2 + y^2)$ (fuera del origen) son
armónicas.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:8}
Hállese el punto del plano $\{x + 2y - z = 4\} \subseteq \R^3$
más cercano al origen de dos maneras: por \omterm{thm:b1:euclid:projection}{proyección ortogonal}
(\cref{ch:b1:euclid}) y minimizando la función de dos variables
$f(x, y) = x^2 + y^2 + (x + 2y - 4)^2$ obtenida al eliminar $z$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:9}
(El laplaciano en \omterm{prop:b1:vspaces:coordinates}{coordenadas} polares, primer contacto) Sea $f$ de clase $C^2$ en
$\R^2 \setminus \{0\}$ y \emph{radial}: $f(x, y) =
\varphi\bigl(\sqrt{x^2+y^2}\bigr)$ con $\varphi$ de clase $C^2$ en
$\intoo{0}{+\infty}$. Demuéstrese que
\[
\frac{\partial^2 f}{\partial x^2} + \frac{\partial^2 f}{\partial
y^2} = \varphi''(\rho) + \frac{\varphi'(\rho)}{\rho},
\qquad \rho = \sqrt{x^2 + y^2},
\]
y hállense todas las funciones armónicas radiales en el plano perforado.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:10}
Dese la ecuación del plano tangente a la gráfica $z = xy$ en
el punto $(1, 1, 1)$. Hállense después todos los puntos de la gráfica de
$f(x,y) = x^3 + y^3 - 3xy$ donde el plano tangente es
\emph{horizontal}, y relaciónese la respuesta con
\cref{ex:b1:multivar:extrema}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:multivar:11}
Sea $f$ de clase $C^1$ en $\R^2$.
\begin{enumerate}
  \item Si $\frac{\partial f}{\partial x} = 0$ en todo punto, demuéstrese
        que $f(x, y)$ solo depende de $y$.
  \item Hállense todas las soluciones $C^1$ de la ecuación
        $\frac{\partial f}{\partial x} = \frac{\partial
        f}{\partial y}$ en $\R^2$. \emph{(Póngase $g(u, v) =
        f\bigl(\frac{u+v}2, \frac{u-v}2\bigr)$ y calcúlese
        $\frac{\partial g}{\partial v}$ por la regla de la cadena.)}
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:multivar:12}
Hállense el máximo y el mínimo globales de $f(x, y) = xy$ en el disco
\omterm{def:b1:topology:closed}{cerrado} $x^2 + y^2 \leq 1$. \emph{(Admítase el teorema de los valores
extremos en dos variables: una función \omterm{def:b1:continuity:continuous}{continua} en el disco \omterm{def:b1:topology:closed}{cerrado}
alcanza sus cotas --- se demuestra en el volumen del segundo año. Trátese el
disco \omterm{def:b1:topology:open}{abierto} por \omterm{thm:b1:multivar:critical}{puntos críticos} y la circunferencia \omterm{def:b1:topology:closure}{frontera} por la
parametrización de \cref{exo:b1:multivar:5}.)}
\end{exercise}

\section{Problema: mínimos cuadrados y recta de regresión}

\begin{problem}\label{pb:b1:multivar:1}
Dados $n$ puntos de datos, ¿qué recta pasa «lo más cerca» de
todos ellos? Legendre y Gauss respondieron: la recta que minimiza la
suma de los errores verticales \emph{al cuadrado} --- porque esa
minimización se resuelve exactamente con álgebra lineal. Este problema
demuestra primero de forma honesta el criterio de Monge de
\cref{met:b1:multivar:monge} para las funciones cuadráticas (el único
caso en que el desarrollo de segundo orden es exacto), y construye
después las \omterm{pb:b1:multivar:1}{ecuaciones normales}, la \omterm{pb:b1:multivar:1}{recta de regresión} y el coeficiente
de correlación sobre la geometría euclídea de
\cref{ch:b1:euclid}.\index{mínimos cuadrados}\index{recta de regresión}
\index{ecuaciones normales}\index{correlación}

\medskip
\noindent\textbf{Parte I --- Funciones cuadráticas: el criterio de Monge,
demostrado.} Fíjense números reales $r, s, t$ y sea $q(h, k) = r h^2 + 2s hk +
t k^2$.
\begin{enumerate}
  \item Supóngase $r \neq 0$. Establézcase la forma con el cuadrado completado
        \[
        q(h, k) = r\Bigl(h + \frac{s}{r}k\Bigr)^{\!2}
        + \frac{rt - s^2}{r}\,k^2 ,
        \]
        y dedúzcase: si $rt - s^2 > 0$, $q$ tiene el signo estricto
        de $r$ fuera del origen; si $rt - s^2 < 0$, $q$ toma ambos
        signos.
  \item Resuélvanse los casos restantes: $r = 0$, $t \neq 0$
        (simétrico); y $r = t = 0$ ($q = 2shk$). Conclúyase:
        $q$ toma ambos signos si y solo si $rt - s^2 < 0$, y $q$
        se anula solo en el origen si y solo si $rt - s^2 > 0$.
  \item Sea ahora $f(x, y) = \frac12\bigl(r x^2 + 2s xy + t
        y^2\bigr) + \beta x + \gamma y + \delta$ una función cuadrática
        con un \omterm{thm:b1:multivar:critical}{punto crítico} $X_0 = (a, b)$. Demuéstrese el
        desarrollo \emph{exacto}
        \[
        f(X_0 + (h,k)) = f(X_0) + \tfrac12\,q(h, k)
        \qquad (\text{sin resto}),
        \]
        y dedúzcase la clasificación de Monge para las funciones
        cuadráticas: mínimo global estricto si $rt - s^2 > 0$, $r >
        0$; máximo global estricto si $rt - s^2 > 0$, $r < 0$;
        \omterm{met:b1:multivar:monge}{punto de silla} si $rt - s^2 < 0$.
  \item Supóngase $rt - s^2 > 0$ y $r > 0$ (luego también $t > 0$).
        Demuéstrese la cota explícita de coercividad
        \[
        q(h, k) \geq c\,(h^2 + k^2),
        \qquad
        c = \frac{rt - s^2}{r + t} > 0 .
        \]
        \emph{(Muéstrese que la forma cuadrática $q - c(h^2 + k^2)$,
        de coeficientes $r - c$, $s$, $t - c$, sigue teniendo
        datos no negativos en el criterio del discriminante.)}
  \item Dedúzcase que una función cuadrática con parte cuadrática
        definida positiva tiende a $+\infty$ cuando $\norm{(x,y)} \to
        \infty$, y que por tanto tiene un \emph{único minimizador
        global}: su \omterm{thm:b1:multivar:critical}{punto crítico}. Contrástese con
        \cref{ex:b1:multivar:extrema}, donde un mínimo local de
        una función no cuadrática no era global.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Las \omterm{pb:b1:multivar:1}{ecuaciones normales}.} Sean $C_1,
C_2, b \in \R^n$ (\omterm{def:b1:euclid:def}{producto escalar} canónico), y
\[
f(u, v) = \norm{u\,C_1 + v\,C_2 - b}^2 .
\]
\begin{enumerate}[resume]
  \item Desarróllese $f$ y muéstrese que es una función cuadrática de $(u,
        v)$ cuya parte cuadrática tiene coeficientes $r = 2\norm{C_1}^2$,
        $s = 2\langle C_1, C_2\rangle$, $t = 2\norm{C_2}^2$.
  \item Muéstrese que $rt - s^2 > 0$ si y solo si $(C_1, C_2)$ es
        \omterm{def:b1:vspaces:free}{libre} \emph{(el caso de igualdad de Cauchy--Schwarz,
        \cref{thm:b1:euclid:cs})}; supóngase esto en adelante.
  \item Muéstrese que las ecuaciones críticas $\nabla f = 0$ son las
        \emph{\omterm{pb:b1:multivar:1}{ecuaciones normales}}
        \[
        \begin{pmatrix}
        \norm{C_1}^2 & \langle C_1, C_2\rangle\\
        \langle C_1, C_2\rangle & \norm{C_2}^2
        \end{pmatrix}
        \begin{pmatrix} u\\ v\end{pmatrix}
        =
        \begin{pmatrix} \langle C_1, b\rangle\\ \langle C_2,
        b\rangle\end{pmatrix}
        \]
        --- con la matriz de Gram de \cref{exo:b1:euclid:11} a la
        izquierda --- y que dicen exactamente: $b - (uC_1 + vC_2)
        \perp C_1, C_2$. Conclúyase con la Parte I y
        \cref{thm:b1:euclid:projection}: el único minimizador
        da $p = uC_1 + vC_2 = $ la \omterm{thm:b1:euclid:projection}{proyección ortogonal} de
        $b$ sobre $\operatorname{Vect}(C_1, C_2)$.
  \item Muéstrese que el valor mínimo es $\norm{b}^2 - \langle p,
        b\rangle = \norm{b - p}^2$, y dibújese la figura
        pitagórica.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La \omterm{pb:b1:multivar:1}{recta de regresión}.} Puntos de datos
$(x_1, y_1), \dots, (x_n, y_n)$, con los $x_i$ no todos iguales. Minimícese
\[
E(\alpha, \beta) = \sum_{i=1}^{n}\bigl(y_i - \alpha - \beta
x_i\bigr)^2 .
\]
Escríbanse $\bar x = \frac1n\sum x_i$, $\bar y = \frac1n\sum y_i$,
$v_x = \frac1n\sum x_i^2 - \bar x^2$, $v_y = \frac1n\sum y_i^2 -
\bar y^2$, $c_{xy} = \frac1n\sum x_i y_i - \bar x\bar y$.
\begin{enumerate}[resume]
  \item Reconózcase la Parte II con $C_1 = (1, \dots, 1)$, $C_2 =
        (x_1, \dots, x_n)$, $b = (y_1, \dots, y_n)$, y escríbanse
        las \omterm{pb:b1:multivar:1}{ecuaciones normales}
        \[
        n\alpha + \Bigl(\sum x_i\Bigr)\beta = \sum y_i,
        \qquad
        \Bigl(\sum x_i\Bigr)\alpha + \Bigl(\sum
        x_i^2\Bigr)\beta = \sum x_i y_i .
        \]
  \item Resuélvanse:
        \[
        \beta = \frac{c_{xy}}{v_x},
        \qquad
        \alpha = \bar y - \beta\,\bar x ,
        \]
        y obsérvese que la \emph{\omterm{pb:b1:multivar:1}{recta de regresión}} $y = \alpha
        + \beta x$ pasa por el punto medio $(\bar x, \bar
        y)$.
  \item Compruébese que $v_x > 0$ exactamente cuando los $x_i$ no son
        todos iguales, y relaciónese esto con la pregunta 7.
  \item Demuéstrese que el error mínimo es
        \[
        E_{\min} = n\Bigl(v_y - \frac{c_{xy}^2}{v_x}\Bigr)
        = n\,v_y\,(1 - \rho^2),
        \qquad
        \rho = \frac{c_{xy}}{\sqrt{v_x v_y}}
        \quad (v_y \neq 0).
        \]
        Dedúzcase $\abs\rho \leq 1$, con $\abs\rho = 1$ si y solo
        si los datos están perfectamente alineados.
  \item Ejemplo resuelto: para los datos $(0,0)$, $(1,1)$, $(2,3)$,
        $(3,4)$, calcúlense $\bar x, \bar y, v_x, c_{xy}$, la
        \omterm{pb:b1:multivar:1}{recta de regresión}, los cuatro residuos y $E_{\min}$.
  \item Demuéstrese que los residuos $\varepsilon_i = y_i - \alpha -
        \beta x_i$ de la recta óptima cumplen $\sum_i
        \varepsilon_i = 0$ y $\sum_i x_i\varepsilon_i = 0$,
        e interprétense ambas igualdades mediante la \omterm{def:b1:euclid:orthogonal}{ortogonalidad}.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Variantes y aplicaciones.}
\begin{enumerate}[resume]
  \item (La mejor constante) Muéstrese que la constante $\alpha$
        que minimiza $\sum_i (y_i - \alpha)^2$ es la media $\bar
        y$, y que el valor mínimo es $n\,v_y$: la varianza
        mide hasta qué punto los datos no son constantes.
  \item (Recta por el origen) Muéstrese que la pendiente que minimiza
        $\sum_i (y_i - \beta x_i)^2$ es $\beta_0 = \frac{\sum
        x_iy_i}{\sum x_i^2}$, y dese una condición sobre los datos
        para que $\beta_0$ coincida con la pendiente
        $c_{xy}/v_x$ de la pregunta 11.
  \item (La distancia a una recta, de nuevo) Para un punto $M$ y la recta
        $D$ que pasa por $P_0$ con vector director unitario $w$,
        minimícese $g(\tau) = \norm{P_0 + \tau w - M}^2$ y
        dedúzcase $d(M, D)^2 = \norm{M - P_0}^2 - \langle M - P_0,
        w\rangle^2$; recupérese la fórmula $d = \frac{\abs{a x_0 +
        b y_0 + c}}{\sqrt{a^2 + b^2}}$ para la recta $ax + by + c
        = 0$ del plano.
  \item (Análisis de la varianza) Demuéstrese la descomposición $v_y =
        \beta^2 v_x + \frac{E_{\min}}n$: la varianza de los
        $y_i$ se reparte entre la parte explicada por la recta y
        la varianza residual.
  \item (Ajuste parabólico) Para ajustar $y = a + bx + cx^2$, muéstrese que
        las \omterm{pb:b1:multivar:1}{ecuaciones normales} son el sistema $3 \times 3$ con
        la matriz de momentos
        \[
        \begin{pmatrix}
        n & \sum x_i & \sum x_i^2\\
        \sum x_i & \sum x_i^2 & \sum x_i^3\\
        \sum x_i^2 & \sum x_i^3 & \sum x_i^4
        \end{pmatrix},
        \]
        una matriz de Gram que es invertible en cuanto tres de
        los $x_i$ son distintos \emph{(libertad de $(1, X, X^2)$
        muestreada en los datos, \cref{exo:b1:euclid:11}; compárense
        las matrices de momentos del problema de fin de semana de
        \cref{ch:b1:det})}.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Robustez y síntesis.}
\begin{enumerate}[resume]
  \item Clasifíquense los \omterm{thm:b1:multivar:critical}{puntos críticos} de las tres funciones
        cuadráticas
        \[
        x^2 + xy + y^2, \qquad x^2 + 3xy + y^2, \qquad
        x^2 + 2xy + y^2 ,
        \]
        mediante la Parte I, tratando a mano el tercer caso degenerado
        (¿dónde se alcanza el mínimo?).
  \item (Experimento con un dato atípico) Añádase el punto $(10, 0)$ a los
        datos de la pregunta 14 y recalcúlese la pendiente $\beta$.
        ¿Qué ha ocurrido y por qué es el error \emph{al cuadrado} tan
        sensible a un solo punto lejano?
  \item (\omterm{pb:b1:multivar:1}{Mínimos cuadrados} ponderados) Dados pesos $w_i > 0$,
        minimícese $\sum_i w_i(y_i - \alpha - \beta x_i)^2$: muéstrese
        que la solución viene dada por las mismas fórmulas que en la
        pregunta 11 con medias, varianza y covarianza ponderadas
        (defínanse).
  \item Muéstrese que, para la recta óptima, $E_{\min} = 0$ obliga
        a que todos los puntos estén exactamente sobre ella, y conéctese con el
        caso de igualdad $\abs\rho = 1$ de la pregunta 13; pruébese con los
        datos alineados $(1,1), (2,3), (3,5)$.
  \item Síntesis, en cuatro frases: por qué las funciones cuadráticas
        son la única clase en la que el criterio de segundo orden de
        este capítulo no necesita ningún teorema admitido; cómo las \omterm{pb:b1:multivar:1}{ecuaciones normales}
        identifican la minimización analítica con la \omterm{thm:b1:euclid:projection}{proyección
        ortogonal} de \cref{ch:b1:euclid}; qué mide el coeficiente de
        correlación y qué desigualdad lo acota;
        y qué piezas de los capítulos 18--23 (\omterm{def:b1:vspaces:free}{bases}, matrices de Gram y
        de momentos, proyecciones) han reaparecido. Nómbrense el
        método y las ecuaciones estudiados en las Partes II--III.
\end{enumerate}
\end{problem}
