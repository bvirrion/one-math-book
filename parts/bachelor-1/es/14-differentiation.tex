\chapter{Derivación}\label{ch:b1:derivative}

Las \omterm{def:b1:derivative:def}{derivadas} se calcularon a lo largo de todo el volumen anterior;
lo que faltaba es la cadena de teoremas que convierte el cálculo en
información sobre las funciones: el teorema de Rolle, el teorema del
valor medio y sus consecuencias --- criterios de monotonía, cotas
lipschitzianas, \omterm{def:b1:derivative:convex}{convexidad}. Todo este capítulo trata de funciones
definidas en un \omterm{prop:b1:reals:intervals}{intervalo} $I$.

\section{La derivada}

\begin{definition}\label{def:b1:derivative:def}
$f \colon I \to \R$ es \emph{derivable en $x_0 \in
I$}\index{derivada} cuando el cociente incremental
$\frac{f(x) - f(x_0)}{x - x_0}$ tiene límite (finito) al tender
$x \to x_0$; ese límite se escribe $f'(x_0)$. Equivalentemente:
\[
f(x_0 + h) = f(x_0) + f'(x_0)\,h + h\,\varepsilon(h),
\qquad \varepsilon(h) \xrightarrow[h \to 0]{} 0 ,
\]
admitiendo entonces la gráfica la recta tangente
$y = f(x_0) + f'(x_0)(x - x_0)$. La derivabilidad en $x_0$ implica la
\omterm{def:b1:continuity:continuous}{continuidad} en $x_0$ (léase la fórmula anterior). $f$ es derivable
en $I$ cuando lo es en cada punto; $f$ es de clase $C^1$ si además
$f'$ es \omterm{def:b1:continuity:continuous}{continua}, y de clase
$C^k$\index{funcion Ck@función $C^k$} cuando $f$ se puede derivar $k$
veces con $f^{(k)}$ \omterm{def:b1:continuity:continuous}{continua}.
\end{definition}

\begin{example}\label{ex:b1:derivative:notc1}
El recíproco de «\omterm{def:b1:derivative:def}{derivable} $\Rightarrow$ \omterm{def:b1:continuity:continuous}{continua}» falla:
$\abs{\,\cdot\,}$ en $0$. Más sorprendente aún: ser \omterm{def:b1:derivative:def}{derivable} no
implica ser de clase $C^1$; la función $f(x) = x^2 \sin\frac 1x$
($f(0) = 0$) es \omterm{def:b1:derivative:def}{derivable} en todas partes, con $f'(0) = 0$, pero
$f'(x) = 2x \sin\frac1x - \cos\frac 1x$ no tiene límite en $0$
(\cref{exo:b1:derivative:2}).
\end{example}

\begin{example}[Derivable en exactamente un punto]\label{ex:b1:derivative:onepoint}
Sea $f(x) = x^2$ para $x \in \Q$ y $f(x) = 0$ para $x \notin \Q$. En
$0$: $\bigl|\frac{f(h) - 0}{h}\bigr| \leq \abs h \to 0$, luego $f$ es
\omterm{def:b1:derivative:def}{derivable} en $0$ con $f'(0) = 0$. En cualquier $x_0 \neq 0$, $f$ ni
siquiera es \omterm{def:b1:continuity:continuous}{continua}: sucesiones racionales e irracionales que
convergen a $x_0$ mandan $f$ a $x_0^2 \neq 0$ y a $0$,
respectivamente (\omterm{def:b1:topology:dense}{densidad}, \cref{thm:b1:reals:density}). Así pues, la
\omterm{def:b1:derivative:def}{derivabilidad} es una noción genuinamente \emph{puntual}: puede darse
en un punto de $\R$ y en ningún otro. La moraleja práctica: los
\omterm{def:b1:logic:statement}{enunciados} como el criterio de monotonía o el de Rolle exigen la
\omterm{def:b1:derivative:def}{derivada} \emph{en un \omterm{prop:b1:reals:intervals}{intervalo}} --- poseer $f'(x_0)$ en puntos
aislados, por muchos que sean, no sostiene ninguna conclusión global.
\end{example}

\begin{theorem}[Operaciones]\label{thm:b1:derivative:operations}
Si $f, g$ son \omterm{def:b1:derivative:def}{derivables} en $x_0$ (y donde las fórmulas tengan
sentido):
\[
(f + g)' = f' + g', \qquad
(fg)' = f'g + fg', \qquad
\Bigl(\frac fg\Bigr)' = \frac{f'g - fg'}{g^2},
\]
y, si $g$ es \omterm{def:b1:derivative:def}{derivable} en $f(x_0)$:
$\;(g \circ f)'(x_0) = g'\bigl(f(x_0)\bigr)\, f'(x_0)$ (regla de la
cadena\index{regla de la cadena}).
\end{theorem}

\begin{proof}
Suma: inmediato. Producto: escríbase
\[
f(x)g(x) - f(x_0)g(x_0)
= \bigl(f(x) - f(x_0)\bigr) g(x) + f(x_0)\bigl(g(x) - g(x_0)\bigr),
\]
divídase por $x - x_0$ y hágase $x \to x_0$ ($g$ es \omterm{def:b1:continuity:continuous}{continua} en
$x_0$). Cociente: trátese $\frac 1g$ mediante
$\frac{1/g(x) - 1/g(x_0)}{x - x_0} =
\frac{-1}{g(x)g(x_0)}\cdot\frac{g(x) - g(x_0)}{x - x_0}$ y aplíquese
después la regla del producto. Regla de la cadena: con
$y_0 = f(x_0)$, defínase $\theta(y) = \frac{g(y) - g(y_0)}{y - y_0}$
para $y \neq y_0$ y $\theta(y_0) = g'(y_0)$: $\theta$ es \omterm{def:b1:continuity:continuous}{continua} en
$y_0$ y, para $x \neq x_0$,
\[
\frac{g(f(x)) - g(f(x_0))}{x - x_0}
= \theta\bigl(f(x)\bigr)\cdot \frac{f(x) - f(x_0)}{x - x_0}
\longrightarrow g'(y_0)\, f'(x_0),
\]
el primer factor por composición de límites (este recurso trata con
limpieza el caso $f(x) = f(x_0)$, donde el ingenuo «multiplicar y
dividir por $f(x) - f(x_0)$» se rompe).
\end{proof}

\begin{theorem}[Derivada de una función inversa]\label{thm:b1:derivative:inverse}
Sea $f$ \omterm{def:b1:continuity:continuous}{continua} y estrictamente monótona en $I$, y \omterm{def:b1:derivative:def}{derivable} en
$x_0$ con $f'(x_0) \neq 0$. Entonces $f^{-1}$
(\cref{thm:b1:continuity:bijection}) es \omterm{def:b1:derivative:def}{derivable} en
$y_0 = f(x_0)$, con
\[
(f^{-1})'(y_0) = \frac{1}{f'(x_0)} = \frac{1}{f'\bigl(f^{-1}(y_0)\bigr)} .
\]
Si $f'(x_0) = 0$, la inversa tiene tangente vertical en $y_0$.
\end{theorem}

\begin{proof}
Para $y \to y_0$, póngase $x = f^{-1}(y)$: la \omterm{def:b1:continuity:continuous}{continuidad} de $f^{-1}$
da $x \to x_0$, y
\[
\frac{f^{-1}(y) - f^{-1}(y_0)}{y - y_0}
= \frac{x - x_0}{f(x) - f(x_0)}
= \frac{1}{\dfrac{f(x) - f(x_0)}{x - x_0}}
\longrightarrow \frac{1}{f'(x_0)} .
\]
Afirmación sobre la tangente vertical: si $f'(x_0) = 0$, el cociente
anterior es el inverso de una cantidad que tiende a $0$ manteniendo
un signo constante (para $f$ estrictamente creciente,
$\frac{f(x) - f(x_0)}{x - x_0} > 0$ para todo $x \neq x_0$): el
cociente incremental de $f^{-1}$ tiende, por tanto, a $+\infty$ (a
$-\infty$ para $f$ decreciente). La inversa sigue siendo \omterm{def:b1:continuity:continuous}{continua},
pero no es \omterm{def:b1:derivative:def}{derivable} en $y_0$ --- su gráfica, reflejo de la de $f$
respecto de la diagonal, se pone vertical exactamente donde la de $f$
iba horizontal, como ilustra $x^{1/3}$ en $0$ frente a $x^3$.
\end{proof}

\begin{example}[Derivadas de inversas, dos veces]\label{ex:b1:derivative:inverseaction}
El teorema recalcula las \omterm{def:b1:derivative:def}{derivadas} clásicas sin ningún trabajo con
límites. Para $\ln = \exp^{-1}$: en $y = \eu^x$,
\[
(\ln)'(y) = \frac{1}{\exp'(x)} = \frac{1}{\eu^{x}} = \frac1y ,
\]
válido para todo $y > 0$, ya que $\exp' = \exp$ no se anula nunca.
Para $\arctan = \tan^{-1}$: en $y = \tan x$,
\[
(\arctan)'(y) = \frac{1}{1 + \tan^2 x} = \frac{1}{1 + y^2} ,
\]
usando $\tan' = 1 + \tan^2 > 0$. La idea de cierre: la fórmula
convierte el conocimiento de una función en conocimiento de su
inversa al precio de una sustitución --- y la sustitución
($x = \ln y$, $x = \arctan y$) es exactamente el \omterm{def:b1:logic:statement}{enunciado} de que las
dos variables viven a lados opuestos de la \omterm{def:b1:logic:inj}{biyección}.
\end{example}

\section{Rolle y el teorema del valor medio}

\begin{proposition}[Extremo interior]\label{prop:b1:derivative:fermat}
Si $f$ es \omterm{def:b1:derivative:def}{derivable} en un punto \emph{\omterm{def:b1:topology:closure}{interior}} $x_0$ de $I$ y tiene
ahí un extremo local, entonces $f'(x_0) = 0$.
\end{proposition}

\begin{proof}
Digamos que se trata de un máximo local: hay $r > 0$ con
$f(x) \leq f(x_0)$ para $\abs{x - x_0} \leq r$, y la interioridad
garantiza que los dos lados de $x_0$ están disponibles dentro de $I$.
Para $0 < h \leq r$, el cociente $\frac{f(x_0 + h) - f(x_0)}{h}$ tiene
numerador $\leq 0$ y denominador $> 0$: es $\leq 0$, y su límite
$f'(x_0)$ hereda el $\leq 0$ (las desigualdades amplias pasan al
límite, \cref{thm:b1:seq:order}); para $-r \leq h < 0$ el cociente es
$\geq 0$, lo que da $f'(x_0) \geq 0$. Luego $f'(x_0) = 0$. (En un
extremo del \omterm{prop:b1:reals:intervals}{intervalo} solo se dispone de un signo: ahí la conclusión
falla --- piénsese en $x$ sobre $\intcc{0}{1}$, máximo en $1$ con
\omterm{def:b1:derivative:def}{derivada} $1$.)
\end{proof}

\begin{theorem}[Rolle]\label{thm:b1:derivative:rolle}
Sea $f$ \omterm{def:b1:continuity:continuous}{continua} en $\intcc{a}{b}$ y \omterm{def:b1:derivative:def}{derivable} en $\intoo{a}{b}$,
con $f(a) = f(b)$. Entonces $f'(c) = 0$ para algún
$c \in \intoo{a}{b}$.\index{teorema de Rolle}
\end{theorem}

\begin{proof}
Por el teorema de los valores extremos
(\cref{thm:b1:continuity:evt}), $f$ alcanza su máximo y su mínimo en
$\intcc{a}{b}$. Si los dos se alcanzan en los extremos, entonces
(puesto que $f(a) = f(b)$) máximo $=$ mínimo y $f$ es constante:
sirve cualquier $c$ \omterm{def:b1:topology:closure}{interior}. En caso contrario, un extremo se
alcanza en un punto \omterm{def:b1:topology:closure}{interior} $c$, y
la~\cref{prop:b1:derivative:fermat} da $f'(c) = 0$.
\end{proof}

\begin{theorem}[Teorema del valor medio]\label{thm:b1:derivative:mvt}
Sea $f$ \omterm{def:b1:continuity:continuous}{continua} en $\intcc{a}{b}$ y \omterm{def:b1:derivative:def}{derivable} en $\intoo{a}{b}$.
Existe $c \in \intoo{a}{b}$ con
\[
f(b) - f(a) = f'(c)\,(b - a) .
\]
\emph{Desigualdad del valor medio:} si además $m \leq f' \leq M$ en
$\intoo{a}{b}$, entonces $m(b-a) \leq f(b) - f(a) \leq M(b-a)$; en
particular, $\abs{f'} \leq K$ implica que $f$ es $K$-lipschitziana.
\index{teorema del valor medio}
\end{theorem}

\begin{proof}
Aplíquese Rolle a $g(x) = f(x) - \frac{f(b) - f(a)}{b - a}(x - a)$:
$g$ es \omterm{def:b1:continuity:continuous}{continua} en $\intcc{a}{b}$, \omterm{def:b1:derivative:def}{derivable} dentro, y
$g(a) = f(a) = g(b)$. En el punto $c$ donde $g'(c) = 0$:
$f'(c) = \frac{f(b)-f(a)}{b-a}$. La desigualdad se sigue acotando
$f'(c)$; y el \omterm{def:b1:logic:statement}{enunciado} lipschitziano la aplica a cada par de puntos.
\end{proof}

\begin{example}[El método de Newton es el de Herón]\label{ex:b1:derivative:newton}
El método de Newton para resolver $f(x) = 0$ sustituye la curva por su
tangente en la estimación actual $x_n$ y toma la raíz de la tangente
como estimación siguiente:
\[
0 = f(x_n) + f'(x_n)(x_{n+1} - x_n)
\quad\Longrightarrow\quad
x_{n+1} = x_n - \frac{f(x_n)}{f'(x_n)} .
\]
Ejecútese sobre $f(x) = x^2 - 2$:
\[
x_{n+1} = x_n - \frac{x_n^2 - 2}{2x_n}
= \frac{x_n}{2} + \frac{1}{x_n}
= \frac12\Bigl(x_n + \frac{2}{x_n}\Bigr) :
\]
exactamente la iteración de Herón (\cref{ex:b1:seq:heron}), dos
milenios antes. La velocidad cuadrática observada allí queda ahora
explicada por la imagen de la tangente: cerca de una raíz simple, la
curva y la tangente difieren en un error de segundo orden, de modo
que cada paso viene a elevar al cuadrado el error --- el \omterm{def:b1:logic:statement}{enunciado}
general se sigue de las cotas de Taylor del~\cref{ch:b1:taylor}. La
idea de cierre: mientras que la dicotomía
(\cref{ex:b1:continuity:dichotomy}) solo usa la \omterm{def:b1:continuity:continuous}{continuidad} y gana un
bit por paso, Newton gasta una \omterm{def:b1:derivative:def}{derivada} para duplicar el número de
cifras correctas en cada paso.
\end{example}

\begin{example}[El teorema del valor medio como estimador]\label{ex:b1:derivative:mvtestimate}
¿Cuánto vale $\sqrt{101}$? Aplíquese el teorema a $f(t) = \sqrt t$ en
$\intcc{100}{101}$: para algún $c \in \intoo{100}{101}$,
\[
\sqrt{101} - 10 = \frac{1}{2\sqrt c},
\qquad\text{luego}\qquad
\frac{1}{2\sqrt{101}} < \sqrt{101} - 10 < \frac{1}{20} = 0.05 ,
\]
y, como $\sqrt{101} < 10.05$, la cota izquierda supera
$\frac{1}{20.1} > 0.0497$: así pues,
$10.0497 < \sqrt{101} < 10.05$ (valor verdadero $10.049875\dots$) ---
tres decimales correctos a partir de una sola evaluación de la
\omterm{def:b1:derivative:def}{derivada}. Del mismo modo,
$\abs{\sin a - \sin b} \leq \abs{a - b}$ (acótese $\abs{\cos}\leq 1$):
todas las estimaciones lipschitzianas usadas desde
el~\cref{ch:b1:seq} son este teorema. La idea de cierre: el teorema
del valor medio es una fórmula de Taylor de orden cero --- cambia un
punto desconocido $c$ por una desigualdad firme, y el~\cref{ch:b1:taylor}
repetirá exactamente ese trueque.
\end{example}

\begin{corollary}[Criterio de monotonía]\label{cor:b1:derivative:monotone}
Sea $f$ \omterm{def:b1:continuity:continuous}{continua} en $I$ y \omterm{def:b1:derivative:def}{derivable} en el \omterm{def:b1:topology:closure}{interior}.
\begin{enumerate}
  \item $f' \geq 0$ en el \omterm{def:b1:topology:closure}{interior} $\iff$ $f$ es creciente; $f' = 0$
        $\iff$ $f$ constante.
  \item Si $f' > 0$ salvo en finitos puntos donde se anula, $f$ es
        \emph{estrictamente} creciente.
\end{enumerate}
\end{corollary}

\begin{proof}
Si $f' \geq 0$: para $x < y$ en $I$, el teorema del valor medio en
$\intcc{x}{y}$ da $f(y) - f(x) = f'(c)(y - x) \geq 0$.
Recíprocamente, los cocientes incrementales de una función creciente
son $\geq 0$, luego sus límites también. Caso constante: aplíquese lo
anterior a $f'$ y a $-f' \geq 0$. Versión estricta: $f$ es creciente;
una igualdad $f(x) = f(y)$ con $x < y$ congelaría $f$ en
$\intcc{x}{y}$, forzando $f' = 0$ ahí --- infinitos puntos.
\end{proof}

\begin{example}[Derivadas iguales, funciones distintas]\label{ex:b1:derivative:intervalmatters}
En $\R^* = \intoo{-\infty}{0} \cup \intoo{0}{+\infty}$, tanto
$f(x) = \ln\abs x$ como $g(x) = \ln\abs x + \mathbf{1}_{x>0}$ (que
suma $1$ solo en la semirrecta derecha) cumplen
$f' = g' = \frac1x$. Y no difieren en una constante: el criterio
«$f' = 0 \implies f$ constante» es un \omterm{def:b1:logic:statement}{enunciado} de
\emph{\omterm{prop:b1:reals:intervals}{intervalo}} --- su demostración corre el teorema del valor medio
entre dos puntos, lo que exige que todo el segmento que los une esté
en el dominio. En cada semirrecta por separado, las primitivas de
$\frac1x$ son $\ln\abs x + c$, con una constante por semirrecta: dos
constantes independientes en total. \Cref{ch:b1:integration} hereda
esta letra pequeña: «la» primitiva de una función está bien definida
salvo constante \emph{en cada \omterm{prop:b1:reals:intervals}{intervalo} de su dominio}, y las tablas
de primitivas suponen la conexión en silencio.
\end{example}

\begin{example}[Estrictez gratis]\label{ex:b1:derivative:strictness}
$x \mapsto x^3$ es \emph{estrictamente} creciente en $\R$ aunque su
\omterm{def:b1:derivative:def}{derivada} se anule en $0$: la cláusula «$f' > 0$ salvo en finitos
puntos» del criterio está pensada justo para esos puntos planos. En
cambio, $f' \geq 0$ por sí solo únicamente da crecimiento en sentido
amplio (una función constante lo cumple), y una \omterm{def:b1:derivative:def}{derivada} que se anula
en todo un subintervalo sí congela la función ahí. La regla práctica:
para afirmar monotonía estricta, enumérense los ceros de $f'$; que
sean finitos (o, más en general, que no llenen ningún subintervalo) es
inofensivo, y que formen un \omterm{prop:b1:reals:intervals}{intervalo} es fatal.
\end{example}

\begin{example}[Un estudio de variación completo]\label{ex:b1:derivative:variations}
Estúdiese $f(x) = x^3 - 3x + 1$ en $\R$. \omterm{def:b1:derivative:def}{Derivada}:
$f'(x) = 3(x^2 - 1)$, positiva en $\intoo{-\infty}{-1}$, negativa en
$\intoo{-1}{1}$ y positiva en $\intoo{1}{+\infty}$: por el criterio de
monotonía, $f$ crece, después decrece y después crece, con un máximo
local $f(-1) = 3$ y un mínimo local $f(1) = -1$. Límites: $\mp\infty$
en $\mp\infty$. Consecuencias, que se leen en la tabla de variación
con el teorema del valor intermedio en cada rama monótona: $f$ se
anula exactamente una vez en cada uno de
\[
\intoo{-\infty}{-1}, \qquad \intoo{-1}{1}, \qquad
\intoo{1}{+\infty}
\]
(los valores en los empalmes tienen signos opuestos: $3 > 0 > -1$), de
modo que la ecuación $x^3 - 3x + 1 = 0$ tiene exactamente tres raíces
reales; numéricamente están cerca de $-1.88$, $0.35$ y $1.53$. La
idea de cierre: una tabla de variación es un \emph{instrumento de
demostración}, no un boceto --- rama monótona más cambio de signo es
igual a exactamente una raíz, y la tabla enumera las ramas de forma
exhaustiva.
\end{example}

\begin{theorem}[Fórmula de Leibniz]\label{thm:b1:derivative:leibniz}
Si $f, g$ son $n$ veces \omterm{def:b1:derivative:def}{derivables}, también lo es $fg$,
y\index{fórmula de Leibniz}
\[
(fg)^{(n)} = \sum_{k=0}^{n} \binom nk f^{(k)}\, g^{(n-k)} .
\]
\end{theorem}

\begin{proof}
Inducción sobre $n$, exactamente paralela a la del teorema del
binomio. El caso $n = 1$ es la regla del producto. Suponiendo la
fórmula en el rango $n$, derívese una vez más:
\[
(fg)^{(n+1)} = \sum_{k=0}^{n} \binom nk
\Bigl( f^{(k+1)} g^{(n-k)} + f^{(k)} g^{(n-k+1)} \Bigr),
\]
y reindéxese después la primera suma con $j = k + 1$ y recójase el
coeficiente de $f^{(j)} g^{(n+1-j)}$: es
$\binom{n}{j-1} + \binom nj = \binom{n+1}{j}$ por la regla de Pascal
(\cref{prop:b1:counting:identities}), llevando los términos de
\omterm{def:b1:topology:closure}{frontera} $j = 0$ y $j = n + 1$ el valor
$\binom{n+1}{0} = \binom{n+1}{n+1} = 1$, como debe ser.
\end{proof}

\begin{example}[Leibniz en acción]\label{ex:b1:derivative:leibnizaction}
Calcúlese $\bigl(x^2 \eu^x\bigr)^{(n)}$ para $n \geq 2$. Tómense
$f = x^2$, cuyas \omterm{def:b1:derivative:def}{derivadas} mueren enseguida ($f' = 2x$, $f'' = 2$,
$f^{(k)} = 0$ para $k \geq 3$), y $g = \eu^x$: de la suma de Leibniz
solo sobreviven tres términos,
\[
\bigl(x^2\eu^x\bigr)^{(n)}
= \binom n0 x^2 \eu^x + \binom n1 (2x)\,\eu^x + \binom n2\,
2\,\eu^x
= \eu^x\bigl(x^2 + 2nx + n(n-1)\bigr).
\]
Comprobación en $n = 1$: $\eu^x(x^2 + 2x)$, que es en efecto
$(x^2\eu^x)'$. La idea de cierre: úsese Leibniz cuando uno de los
factores sea un \omterm{def:b1:poly:def}{polinomio} --- la suma tiene entonces solo $\deg + 1$
términos, y la fórmula es una expresión \omterm{def:b1:topology:closed}{cerrada}, no una identidad
abstracta. (Para dos factores infinitamente vivaces, como
$\eu^x\sin x$, las exponenciales complejas del~\cref{ch:b1:complex}
son mejor herramienta.)
\end{example}

\section{Convexidad}

\begin{definition}\label{def:b1:derivative:convex}
$f \colon I \to \R$ es \emph{convexa}\index{función convexa} cuando
toda cuerda queda por encima de la gráfica:
\[
\forall x, y \in I,\ \forall t \in \intcc{0}{1}, \quad
f\bigl(tx + (1-t)y\bigr) \leq t f(x) + (1-t) f(y).
\]
($f$ es \emph{cóncava} cuando $-f$ es convexa.)
\end{definition}

\begin{theorem}[Caracterizaciones diferenciales]\label{thm:b1:derivative:convexchar}
Sea $f$ \omterm{def:b1:derivative:def}{derivable} en $I$. Son equivalentes:
\begin{enumerate}
  \item $f$ es \omterm{def:b1:derivative:convex}{convexa};
  \item $f'$ es creciente en $I$;
  \item la gráfica queda por encima de toda tangente:
        $f(y) \geq f(x) + f'(x)(y - x)$ para todos $x, y \in I$.
\end{enumerate}
Si $f$ es dos veces \omterm{def:b1:derivative:def}{derivable}: $f$ \omterm{def:b1:derivative:convex}{convexa} $\iff f'' \geq 0$.
\end{theorem}

\begin{proof}
(1 $\Rightarrow$ 3) La \omterm{def:b1:derivative:convex}{convexidad}, escrita como
$\frac{f(x + t(y-x)) - f(x)}{t} \leq f(y) - f(x)$ para
$t \in \intoc{0}{1}$; hágase $t \to 0^+$:
$f'(x)(y - x) \leq f(y) - f(x)$.

(3 $\Rightarrow$ 2) Para $x < y$, las dos desigualdades de tangencia
en $x$ y en $y$ dan $f'(x)(y-x) \leq f(y) - f(x) \leq f'(y)(y - x)$,
luego $f'(x) \leq f'(y)$.

(2 $\Rightarrow$ 1) Fíjense $x < y$ y $t \in \intoo{0}{1}$, y sea
$z = tx + (1-t)y \in \intoo{x}{y}$. Por el teorema del valor medio en
$\intcc{x}{z}$ y en $\intcc{z}{y}$, hay $c_1 < z < c_2$ con
\[
\frac{f(z) - f(x)}{z - x} = f'(c_1) \leq f'(c_2)
= \frac{f(y) - f(z)}{y - z} ,
\]
y, quitando denominadores ($z - x = (1-t)(y-x)$, $y - z = t(y-x)$), se
reordena exactamente en la desigualdad de \omterm{def:b1:derivative:convex}{convexidad}.

Caso dos veces \omterm{def:b1:derivative:def}{derivable}: $f'' \geq 0 \iff f'$ creciente
(\cref{cor:b1:derivative:monotone}).
\end{proof}

\begin{example}[Desigualdades clásicas de convexidad]\label{ex:b1:derivative:convexineq}
$\exp$ es \omterm{def:b1:derivative:convex}{convexa} ($\exp'' = \exp > 0$): su tangente en $0$ da
$\eu^x \geq 1 + x$ para todo $x$. $\ln$ es cóncava: su tangente en $1$
da $\ln x \leq x - 1$; y sus cuerdas dan, para $0 < a \leq b$, la
desigualdad entre las medias geométrica y aritmética: tomando
$t = \frac12$ en la concavidad,
\[
\ln\frac{a + b}{2} \geq \frac{\ln a + \ln b}{2} = \ln\sqrt{ab},
\qquad\text{luego}\qquad
\sqrt{ab} \leq \frac{a+b}{2} .
\]
La desigualdad aritmético-geométrica general es
el~\cref{exo:b1:derivative:9}.
\end{example}

\begin{example}[Una desigualdad de convexidad desde cero]\label{ex:b1:derivative:xlnx}
La función $f(t) = t\ln t$ es \omterm{def:b1:derivative:convex}{convexa} en $\intoo{0}{+\infty}$:
$f''(t) = \frac1t > 0$. Su desigualdad del punto medio, multiplicada
por $2$, dice: para todos $a, b > 0$,
\[
a\ln a + b\ln b \;\geq\; (a + b)\,\ln\frac{a + b}{2} ,
\]
con igualdad si y solo si $a = b$ (\omterm{def:b1:derivative:convex}{convexidad} estricta). Prueba de
manejo: $a = 1$, $b = 3$ da $3\ln 3 = 3.296$ frente a
$4\ln 2 = 2.773$. Esta inocua desigualdad es el caso de dos puntos de
la comparación de \emph{entropías} que reaparece con la desigualdad de
Jensen (\cref{exo:b1:derivative:9}) y en las asintóticas de la teoría
de la información del volumen del tercer año. La idea de cierre: para
fabricar una desigualdad, búsquese una función cuya segunda
\omterm{def:b1:derivative:def}{derivada} tenga un signo y escríbase lo que dice la \omterm{def:b1:derivative:convex}{convexidad} --- la
caracterización diferencial convierte una sola comprobación de signo
en infinitas desigualdades.
\end{example}

\begin{remark}[Errores frecuentes con las derivadas]
(i) \emph{Una \omterm{def:b1:derivative:def}{derivada} positiva en un punto no da monotonía cerca de
él}: $f(x) = \frac x2 + x^2\sin\frac1x$ (con $f(0) = 0$) tiene
$f'(0) = \frac12 > 0$ y, sin embargo,
\[
f'(x) = \frac12 + 2x\sin\frac1x - \cos\frac1x
\]
vale $-\frac12$ en cada $x_n = \frac{1}{2\pi n}$: todo \omterm{def:b1:topology:open}{entorno} de
$0$ contiene descensos. La monotonía necesita $f' \geq 0$
\emph{en un \omterm{prop:b1:reals:intervals}{intervalo}} (\cref{cor:b1:derivative:monotone}); el signo
puntual solo controla el cruce con la recta tangente. (ii) \emph{Las
tres hipótesis de Rolle están todas activas}: $\abs x$ en
$\intcc{-1}{1}$ (sin \omterm{def:b1:derivative:def}{derivabilidad} \omterm{def:b1:topology:closure}{interior}), $x$ en $\intcc{0}{1}$
(extremos desiguales) y $x - \lfloor x\rfloor$ en $\intcc{0}{1}$ (la
\omterm{def:b1:continuity:continuous}{continuidad} falla en $1$) rompen cada uno exactamente una hipótesis
y también la conclusión. (iii) \emph{Las \omterm{def:b1:derivative:def}{derivadas} pueden ser
discontinuas, pero no de cualquier manera}: $f'$ puede oscilar
(\cref{ex:b1:derivative:notc1}), pero siempre cumple la propiedad del
valor intermedio (Darboux, \cref{exo:b1:derivative:10}): una
\omterm{def:b1:derivative:def}{derivada} no salta nunca --- si se calcula un «límite de la
\omterm{def:b1:derivative:def}{derivada}» lateral con un salto, es que se ha derivado una función
no \omterm{def:b1:derivative:def}{derivable}. (iv) \emph{La fórmula de la inversa necesita
$f' \neq 0$}: $x \mapsto x^3$ es una \omterm{def:b1:logic:inj}{biyección} suave y estrictamente
creciente cuya inversa $x^{1/3}$ tiene tangente vertical en $0$ --- la
\omterm{def:b1:derivative:def}{derivabilidad} de la inversa se pierde exactamente donde se anula $f'$
(\cref{thm:b1:derivative:inverse}).
\end{remark}

\begin{remark}[Dónde funciona a continuación el teorema del valor medio]
Casi todo \omterm{def:b1:logic:statement}{enunciado} cuantitativo de los capítulos siguientes es el
teorema del valor medio de este capítulo disfrazado: el teorema
fundamental del cálculo (\cref{ch:b1:integration}) deriva la función
área y concluye con el criterio de monotonía; la fórmula de
Taylor--Lagrange (\cref{ch:b1:taylor}) es el teorema del valor medio
iterado $n$ veces; el análisis del error del método de Newton y de las
iteraciones de punto fijo (\cref{exo:b1:derivative:11}) es la forma
lipschitziana; y el problema del fin de semana de este capítulo
(\cref{pb:b1:derivative:1}) convierte esa misma cota lipschitziana en
teoría de números --- una desigualdad de repulsión entre \omterm{pb:b1:logic:1}{números
algebraicos} y racionales, que produce el primer \omterm{pb:b1:logic:1}{número trascendente}
de la historia. En el volumen del segundo año, la \emph{desigualdad}
del valor medio sobrevive en varias variables allí donde la igualdad
no lo hace.
\end{remark}

\begin{example}[La desigualdad de Young a partir de la concavidad]\label{ex:b1:derivative:young}
Sean $p, q > 1$ con $\frac1p + \frac1q = 1$. Para todos $a, b > 0$:
\[
ab \;\leq\; \frac{a^p}{p} + \frac{b^q}{q} .
\]
Demostración con una sola \omterm{def:b1:logic:map}{aplicación} de la concavidad de $\ln$ con
pesos $\frac1p, \frac1q$ (la desigualdad de Jensen de dos puntos,
como en el~\cref{exo:b1:derivative:9}):
\[
\ln\Bigl(\frac{a^p}{p} + \frac{b^q}{q}\Bigr)
\;\geq\; \frac1p \ln(a^p) + \frac1q \ln(b^q)
= \ln a + \ln b = \ln(ab),
\]
y ser $\ln$ creciente convierte la desigualdad entre logaritmos en la
afirmación buscada; hay igualdad si y solo si $a^p = b^q$ (concavidad
estricta). El caso $p = q = 2$ es la desigualdad
aritmético-geométrica $ab \leq \frac{a^2 + b^2}{2}$ disfrazada. La
idea de cierre: la desigualdad de Young es la semilla algebraica de
las desigualdades de Hölder y de Minkowski del volumen del segundo
año --- un solo \omterm{def:b1:logic:statement}{enunciado} de concavidad sobre $\ln$, cosechado para
las normas.
\end{example}

\begin{remark}[Perspectivas dentro de este volumen]
La \omterm{def:b1:derivative:def}{derivada} adquiere tres vidas nuevas antes de que acabe el
volumen. En el~\cref{ch:b1:taylor} se itera: $n$ \omterm{def:b1:derivative:def}{derivadas} en un
punto se comprimen en un \omterm{def:b1:poly:def}{polinomio} más un error controlado, y el
teorema del valor medio pasa a ser el resto de Lagrange. En
el~\cref{ch:b1:curves}, la derivación se vuelve geométrica: para una
curva parametrizada $t \mapsto (x(t), y(t))$, el par
$(x'(t), y'(t))$ es un \emph{vector} velocidad, la tangencia se
convierte en colinealidad y los puntos críticos, en puntos de
retroceso que hay que clasificar. En el~\cref{ch:b1:multivar} se
congela una variable cada vez: las \omterm{def:b1:derivative:def}{derivadas} parciales repiten dos
veces este capítulo, y la recta tangente crece hasta ser un plano
tangente. Los tres capítulos heredan la misma gramática ---
aproximación lineal local más un término de error --- hablada aquí
por primera vez.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:derivative:1}
Derívense (precisando los dominios): $x^x$;
$\;\ln\bigl(x + \sqrt{x^2+1}\bigr)$; $\;\arctan\frac{1}{x}$;
$\;\sqrt{1 + \eu^{2x}}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:derivative:2}
Complétese el~\cref{ex:b1:derivative:notc1}: demuéstrese que
$f(x) = x^2 \sin\frac1x$, $f(0) = 0$, es \omterm{def:b1:derivative:def}{derivable} en $0$ con
$f'(0) = 0$, y que $f'$ no tiene límite en $0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:derivative:3}
Usando el teorema del valor medio o las desigualdades de tangencia,
demuéstrese que para todo $x > 0$:
\[
\frac{x}{1 + x} < \ln(1 + x) < x .
\]
Dedúzcase $\lim_{n\to\infty} \bigl(1 + \frac xn\bigr)^n = \eu^x$ para
todo $x > 0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:derivative:4}
Sea $P$ un \omterm{def:b1:poly:def}{polinomio} real con $k$ raíces reales distintas.
Demuéstrese que $P'$ tiene al menos $k - 1$ raíces reales distintas,
entrelazadas con las de $P$. Dedúzcase que si $P$ tiene todas sus
raíces reales, $P'$ también.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:5}
Sea $f$ \omterm{def:b1:derivative:def}{derivable} en $\R$ con $f' (x)\to \ell$ cuando
$x \to +\infty$. Demuéstrese que $\frac{f(x)}{x} \to \ell$
\emph{(teorema del valor medio en $\intcc{A}{x}$)}. ¿Se cumple también
$f(x+1) - f(x) \to \ell$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:6}
(Un Rolle discreto) Sea $f$ $n$ veces \omterm{def:b1:derivative:def}{derivable} en $I$ y que se
anula en $n + 1$ puntos distintos. Demuéstrese que $f^{(n)}$ se anula
al menos una vez. Aplicación: un \omterm{def:b1:poly:def}{polinomio} de grado $\leq n$ que se
anula en $n+1$ puntos es nulo (una vez más).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:7}
Sea $f$ dos veces \omterm{def:b1:derivative:def}{derivable} en $\intcc{a}{b}$ con
$f(a) = f(b) = 0$ y $f(x_0) > 0$ para algún $x_0$ \omterm{def:b1:topology:closure}{interior}.
Demuéstrese que $f''(c) < 0$ para algún $c \in \intoo{a}{b}$.
\emph{(Dos teoremas del valor medio y una comparación de pendientes.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:8}
Estúdiese la función $f(x) = \dfrac{\ln x}{x}$ en
$\intoo{0}{+\infty}$: variaciones, límites, máximo. Dedúzcase que
$a^b > b^a$ para todos los reales $\eu \leq a < b$, y resuélvase el
famoso caso particular: ¿cuál es mayor, $\eu^\pi$ o $\pi^\eu$?
Compruébese con las parejas de enteros pequeños $(2,3)$ y $(2,4)$:
¿por qué se comportan de forma distinta?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:derivative:9}
(Desigualdad aritmético-geométrica) Usando la concavidad de $\ln$ con
pesos generales (la desigualdad de Jensen para $n$ puntos, que hay que
demostrar por inducción sobre $n$), véase que para reales positivos
$a_1, \dots, a_n$:
\[
\sqrt[n]{a_1 a_2 \cdots a_n} \leq \frac{a_1 + \dots + a_n}{n},
\]
con igualdad si y solo si todos los $a_i$ son iguales.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:10}
(Darboux: las \omterm{def:b1:derivative:def}{derivadas} toman los valores intermedios) Sea $f$
\omterm{def:b1:derivative:def}{derivable} en $I$ y sean $a < b$ en $I$ con $f'(a) < v < f'(b)$.
Considerando $g(x) = f(x) - vx$ y el punto donde $g$ alcanza su mínimo
en $\intcc{a}{b}$, demuéstrese que $f'(c) = v$ para algún
$c \in \intoo{a}{b}$ --- aunque $f'$ no tenga por qué ser \omterm{def:b1:continuity:continuous}{continua}
(\cref{exo:b1:derivative:2}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:11}
Sea $f \colon \R \to \R$ \omterm{def:b1:derivative:def}{derivable} con $\abs{f'(x)} \leq k < 1$ para
todo $x$ (una \emph{contracción}). Demuéstrese que $f$ tiene
exactamente un punto fijo $\ell$, y que toda sucesión
$u_{n+1} = f(u_n)$ converge a $\ell$ con
$\abs{u_n - \ell} \leq k^n\abs{u_0 - \ell}$. \emph{(Existencia:
aplíquese el teorema del valor intermedio a $f(x) - x$ en un segmento
grande, usando la cota lipschitziana; o úsese la completitud con el
criterio de Cauchy.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:derivative:12}
(Teorema del valor medio de Cauchy y regla de l'Hôpital)
\begin{enumerate}
  \item Sean $f, g$ \omterm{def:b1:continuity:continuous}{continuas} en $\intcc{a}{b}$ y \omterm{def:b1:derivative:def}{derivables} en
        $\intoo{a}{b}$, con $g'$ nunca nula ahí. Demuéstrese que
        $g(b) \neq g(a)$ y que algún $c \in \intoo{a}{b}$ cumple
        \[
        \frac{f(b) - f(a)}{g(b) - g(a)} = \frac{f'(c)}{g'(c)}
        \]
        \emph{(aplíquese Rolle a $h = f - \lambda g$ con la constante
        $\lambda$ adecuada)}.
  \item Dedúzcase la regla de l'Hôpital en la forma $\frac00$ en un
        punto: si $f(a) = g(a) = 0$ y
        $\frac{f'(x)}{g'(x)} \to \ell$ cuando $x \to a^+$, entonces
        $\frac{f(x)}{g(x)} \to \ell$.
  \item Véase que el recíproco falla: para $f(x) = x^2\sin\frac1x$
        ($f(0) = 0$) y $g(x) = x$, el cociente $\frac{f}{g}$ tiene
        límite en $0$, pero $\frac{f'}{g'}$ no lo tiene.
\end{enumerate}
\end{exercise}

\section{Problema: la desigualdad de Liouville y el primer número
trascendente}

\begin{problem}[{Problema del fin de semana --- los \omterm{pb:b1:logic:1}{números algebraicos}
repelen a los racionales: $\abs{x - p/q} \geq C/q^d$, y la
trascendencia de $\sum 10^{-n!}$}]
\label{pb:b1:derivative:1}
Un número real es \emph{algebraico} cuando es raíz de un \omterm{def:b1:poly:def}{polinomio}
no nulo con coeficientes enteros, y \emph{trascendente} en caso
contrario. En 1844, Liouville produjo el primer número del que se
\emph{demostró} que era trascendente, y el motor de su demostración es
el teorema del valor medio de este capítulo: un \omterm{pb:b1:logic:1}{número algebraico} de
grado $d$ no se puede aproximar por racionales mejor que $C/q^d$ ---
de modo que un número aproximable \emph{más deprisa que toda potencia}
no puede ser algebraico. Este problema construye la desigualdad,
construye el número de Liouville $L = 0.110001000\dots$ (unos en las
posiciones factoriales, con la maquinaria de cifras
del~\cref{pb:b1:reals:1}), demuestra su trascendencia y termina con la
demostración rival de Cantor y con cotas efectivas para $\sqrt2$ y
$2^{1/3}$.\index{número trascendente}\index{desigualdad de
Liouville}\index{número algebraico}

\medskip
\noindent\textbf{Parte I --- ¿Con qué precisión se pueden aproximar
los racionales?}
\begin{enumerate}
  \item Véase que dos racionales distintos $\frac ab \neq \frac pq$
        (escritos con $b, q \geq 1$) cumplen
        $\bigl|\frac ab - \frac pq\bigr| \geq \frac{1}{bq}$.
        Dedúzcase: si $x = \frac ab$ y
        $0 < \bigl|x - \frac pq\bigr| < \frac{1}{bq}$, no existe tal
        $\frac pq$ --- un racional repele a todos los demás
        racionales a escala $\frac 1q$.
  \item Demuéstrese que para \emph{todo} racional $\frac pq$
        ($q \geq 1$): $\bigl|\sqrt2 - \frac pq\bigr| \geq
        \frac{1}{4q^2}$ \emph{(si la distancia supera $1$ está claro;
        en caso contrario, acótese $\abs{\sqrt2 + p/q} < 4$ y úsese
        el entero no nulo $\abs{p^2 - 2q^2} \geq 1$)}.
  \item En la otra dirección: compruébese que
        $(p, q) \mapsto (p + 2q, p + q)$ conserva
        $\abs{p^2 - 2q^2} = 1$, genérense a partir de $(1,1)$ las
        parejas $(3,2)$, $(7,5)$, $(17,12)$, $(41,29)$, $(99,70)$, y
        véase que cada una cumple
        \[
        \Bigl|\sqrt2 - \frac pq\Bigr| =
        \frac{1}{q^2\,(\sqrt2 + p/q)} < \frac{1}{2q^2} :
        \]
        infinitas aproximaciones de orden $2$. Junto con la
        pregunta 2: el exponente de aproximación de $\sqrt 2$ es
        \emph{exactamente} $2$.
  \item (Dirichlet) Sea $x$ irracional y $N \in \N^*$. Considérense
        las $N + 1$ partes fraccionarias de $0, x, 2x, \dots, Nx$ en
        las $N$ cajas $\intco{\frac kN}{\frac{k + 1}{N}}$: por el
        principio del palomar (\cref{cor:b1:counting:pigeonhole}), dos
        caen en la misma caja. Dedúzcanse $q \leq N$ y $p$ con
        $\abs{qx - p} < \frac 1N$, y por tanto infinitos racionales
        con $\bigl|x - \frac pq\bigr| < \frac{1}{q^2}$: \emph{todo}
        irracional es aproximable al orden $2$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La desigualdad de Liouville.} Sea $x$
irracional y algebraico.
\begin{enumerate}[resume]
  \item Véase que entre los \omterm{def:b1:poly:def}{polinomios} enteros no nulos que se
        anulan en $x$ hay uno, digamos $P$ de grado $d$, \emph{sin
        ninguna raíz racional}; y compruébese que $d \geq 2$
        \emph{(divídase por un factor $X - \frac ab$ sobre $\Q$ y
        quítense denominadores; el grado $1$ haría racional a $x$)}.
  \item Véase que para todo racional $\frac pq$ ($q \geq 1$):
        $\bigl|P\bigl(\frac pq\bigr)\bigr| \geq \frac{1}{q^d}$
        \emph{($q^d P(p/q)$ es un entero no nulo)}.
  \item Sea $M = \max_{\intcc{x-1}{x+1}} \abs{P'}$
        (\cref{thm:b1:continuity:evt}). Usando el teorema del valor
        medio entre $x$ y $\frac pq$, demuéstrese la
        \emph{desigualdad de Liouville}: con
        $C = \min\bigl(1, \frac 1M\bigr) > 0$,
        \[
        \Bigl| x - \frac pq \Bigr| \geq \frac{C}{q^{\,d}}
        \qquad\text{para todo racional } \frac pq,\ q \geq 1 .
        \]
  \item Llámese a $x$ \emph{número de Liouville} cuando para todo
        $n \in \N$ existe un racional $\frac pq$ con $q \geq 2$ y
        $0 < \bigl|x - \frac pq\bigr| < q^{-n}$. Demuéstrese que un
        número de Liouville es irracional \emph{(pregunta 1: elíjase
        $n$ con $2^{\,n-1} > b$)}.
  \item Demuéstrese el teorema de Liouville: \emph{un número de
        Liouville es trascendente} \emph{(combínense las preguntas 7
        y 8: la desigualdad $C < q^{\,d-n}$ falla para $n$ grande)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El número $L$.}
\begin{enumerate}[resume]
  \item Sea $L$ el valor (en el sentido del~\cref{pb:b1:reals:1}) de
        la cadena de cifras decimales con la cifra $1$ en las
        posiciones $n!$ ($n = 1, 2, 3, \dots$) y $0$ en las demás, es
        decir, $L = \sup_k t_k$ con $t_k = \sum_{n=1}^{k} 10^{-n!}$.
        Escríbanse las primeras $25$ cifras de $L$.
  \item Demuéstrese el encuadre de la cola, para todo $k \geq 1$:
        \[
        10^{-(k+1)!} \;\leq\; L - t_k \;\leq\;
        \frac{10}{9}\,10^{-(k+1)!} \;<\; 2\cdot 10^{-(k+1)!}
        \]
        \emph{(acótese toda suma parcial más allá de $t_k$ por una
        suma geométrica finita)}.
  \item Escríbase $t_k = \frac{p_k}{q_k}$ con $q_k = 10^{k!}$.
        Véase que $0 < L - \frac{p_k}{q_k} < \frac{2}{q_k^{\,k+1}}$ y
        conclúyase que $L$ es un número de Liouville en el sentido de
        la pregunta 8.
  \item Conclúyase: $L$ es trascendente --- el primer ejemplo
        explícito de la historia (Liouville, 1844). Compruébese
        además su irracionalidad directamente: sus cifras no son
        finalmente periódicas (huecos crecientes, como en
        el~\cref{pb:b1:reals:1}, pregunta 20).
  \item Generalícese: sustitúyase cada cifra $1$ por una cifra no
        nula arbitraria $d_n \in \intint{1}{9}$. Véase que el valor
        sigue siendo un número de Liouville, y dedúzcase ---por el
        argumento diagonal del~\cref{pb:b1:reals:1} (pregunta 22)
        aplicado a estas elecciones de cifras--- que hay no
        numerablemente muchos \omterm{pb:b1:logic:1}{números trascendentes} de esta forma.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La jerarquía de los órdenes de
aproximación.} Dígase que $x$ es \emph{aproximable al orden $\mu$}
cuando para alguna constante $c > 0$ hay infinitos racionales que
cumplen $\bigl|x - \frac pq\bigr| < \frac{c}{q^{\mu}}$.
\begin{enumerate}[resume]
  \item Móntese la jerarquía a partir de las partes I--III: los
        racionales son aproximables al orden $1$ y no mejor; $\sqrt 2$
        al orden $2$ y no mejor; todo irracional, al menos al orden
        $2$; un \omterm{pb:b1:logic:1}{número algebraico} de grado $d$, a ningún orden más
        allá de $d$; y los números de Liouville, a todo orden.
        Justifíquese cada afirmación citando la pregunta pertinente.
  \item Véase que $L + r$ es un número de Liouville para todo
        racional $r = \frac ab$ \emph{(tradúzcanse los aproximantes:
        los nuevos denominadores son $b\,q_k$)}. Conclúyase que los
        números de Liouville ---y por tanto los trascendentes--- son
        \omterm{def:b1:topology:dense}{densos} en $\R$.
  \item (Cantor, 1874) Demuéstrese que el \omterm{def:b1:logic:sets}{conjunto} de los
        \omterm{pb:b1:logic:1}{números algebraicos} es numerable: hay finitos \omterm{def:b1:poly:def}{polinomios}
        enteros cuyo grado más la suma de los
        $\abs{\text{coeficientes}}$ está acotada por $h$, cada uno con
        a lo sumo $\deg$ raíces; y una unión numerable de
        \omterm{def:b1:counting:card}{conjuntos finitos} es numerable. Como ninguna sucesión agota
        $\R$ (\cref{pb:b1:reals:1}, pregunta 22), existen \omterm{pb:b1:logic:1}{números
        trascendentes} --- y de hecho forman un \omterm{def:b1:logic:sets}{conjunto} no numerable.
        Compárense las dos demostraciones: ¿qué da la de Liouville que
        la de Cantor no puede dar?
  \item Demuéstrese directamente, a partir de la pregunta 2, que
        $\sqrt 2$ \emph{no} es un número de Liouville \emph{(para
        $n \geq 3$, la desigualdad $q^{-n} > \frac{1}{4q^2}$ acota
        $q$; y entonces solo quedan finitos racionales candidatos,
        todos a distancia positiva de $\sqrt2$)}. Generalícese:
        ningún \omterm{pb:b1:logic:1}{número algebraico} es de Liouville.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Constantes efectivas.}
\begin{enumerate}[resume]
  \item Para la pareja de Pell $(99, 70)$: compruébese
        $99^2 - 2\cdot70^2 = 1$ y evalúese el error exacto
        \[
        \sqrt2 - \frac{99}{70}
        = \frac{-1}{70^2\,\bigl(\sqrt2 + \frac{99}{70}\bigr)},
        \qquad
        \Bigl|\sqrt 2 - \frac{99}{70}\Bigr| \approx 7.2\cdot
        10^{-5} :
        \]
        cinco cifras correctas a partir de una fracción de tres
        cifras.
  \item Aplíquese la parte II a $x = 2^{1/3}$, $P = X^3 - 2$:
        compruébese que $P$ no tiene raíz racional, acótese
        $M = \max_{\intcc{x-1}{x+1}} 3t^2 \leq 3\,(1 + 2^{1/3})^2 <
        16$, y conclúyase la desigualdad efectiva
        \[
        \Bigl| 2^{1/3} - \frac pq \Bigr| \geq
        \frac{1}{16\,q^3} \qquad \text{para todo } \frac pq .
        \]
  \item Recompensa: véase que todo racional que aproxime $2^{1/3}$
        con error menor que $10^{-6}$ tiene denominador $q \geq 40$.
  \item Véase que la base $10$ es irrelevante: el análogo binario
        $\sum_{n\geq1} 2^{-n!}$ (valor de la cadena binaria con unos
        en las posiciones factoriales) también es un número de
        Liouville y, por tanto, trascendente.
\end{enumerate}

\medskip
\noindent\textbf{Parte VI --- \omterm{def:b1:topology:closure}{Fronteras} y síntesis.}
\begin{enumerate}[resume]
  \item Sea $x^\dagger$ el valor de la cadena decimal con unos
        exactamente en las posiciones $3^k$ ($k \geq 0$). Véase que
        $x^\dagger$ es aproximable al orden $3$, y dedúzcase de la
        desigualdad de Liouville que $x^\dagger$ no es racional ni
        irracional cuadrático. Explíquese por qué el método se atasca
        ahí: el orden $3$ es compatible con ser algebraico de grado
        $\geq 3$, y cerrar esa brecha (basta cualquier exponente
        $> 2$, para todo \omterm{pb:b1:logic:1}{número algebraico}) es el teorema de Roth,
        muy por encima de este volumen.
  \item Cuantifíquese a Cantor: véase que los \omterm{pb:b1:logic:1}{números algebraicos} de
        grado $\leq d$ dados por \omterm{def:b1:poly:def}{polinomios} con coeficientes en
        $\intint{-H}{H}$ son a lo sumo $d\,(2H + 1)^{d+1}$. (Esta
        finitud es lo que hizo funcionar la pregunta 17.)
  \item Síntesis, una frase para cada punto: (i) localícese el único
        ingrediente analítico de la demostración de Liouville (qué
        teorema de este capítulo y dónde se usa); (ii) enúnciese la
        tensión que la impulsa (la integralidad fuerza
        $\abs{P(p/q)} \geq q^{-d}$, la suavidad prohíbe
        $\abs{P(p/q)} > M\abs{x - p/q}$); (iii) contrástense las
        demostraciones de Liouville y de Cantor de la existencia de
        \omterm{pb:b1:logic:1}{números trascendentes}; (iv) nómbrese dónde vuelve este
        volumen sobre el tema --- el problema del fin de semana
        del~\cref{ch:b1:integration} demuestra que $\pi$ es irracional
        con el mismo emparedado de integralidad contra pequeñez, con
        integrales en lugar de \omterm{def:b1:derivative:def}{derivadas}.
\end{enumerate}
\end{problem}
