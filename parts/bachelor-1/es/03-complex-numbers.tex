\chapter{Números complejos}\label{ch:b1:complex}

Los números complejos aparecieron en el volumen anterior como un
recurso de cálculo para las ecuaciones de segundo grado. Este capítulo
los trata como un objeto central: la forma exponencial y sus
consecuencias (De Moivre, raíces $n$-ésimas, \omterm{def:b1:complex:unity}{raíces de la unidad}), la
traducción sistemática entre $\C$ y la geometría del plano, y el uso de
$\eu^{\iu\theta}$ como máquina de demostrar identidades
trigonométricas.

\section{El cuerpo $\C$, módulo y conjugado}

\begin{definition}[El cuerpo de los complejos]\label{def:b1:complex:field}
$\C = \{a + \iu b : a, b \in \R\}$ con la suma usual y la
multiplicación determinada por $\iu^2 = -1$. Todo $z = a + \iu b$ no
nulo tiene inverso: $z^{-1} = \frac{a - \iu b}{a^2 + b^2}$ --- en el
lenguaje del~\cref{ch:b1:structures}, $\C$ es un cuerpo. Se escribe
$a = \Re(z)$, $b = \Im(z)$, $\conj{z} = a - \iu b$ (el
\emph{conjugado}\index{conjugado}) y $\abs{z} = \sqrt{a^2 + b^2}$ (el
\emph{módulo}\index{módulo}).
\end{definition}

\begin{proposition}[Reglas del conjugado y del módulo]\label{prop:b1:complex:rules}
Para $z, w \in \C$:
\begin{enumerate}
  \item $\conj{z + w} = \conj z + \conj w$, $\conj{zw} = \conj z\,
        \conj w$, $\conj{\conj z} = z$;
  \item $z \conj z = \abs z^2$; $\;\Re(z) = \frac{z + \conj z}{2}$,
        $\Im(z) = \frac{z - \conj z}{2\iu}$;
  \item $\abs{zw} = \abs z\, \abs w$, y $\abs{z^{-1}} =
        \abs{z}^{-1}$ para $z \neq 0$;
  \item (desigualdad triangular\index{desigualdad triangular})
        $\abs{z + w} \leq \abs z + \abs w$, con igualdad si y solo si
        $z$ y $w$ están en una misma semirrecta desde $0$ ($w =
        \lambda z$ o $z = \lambda w$ con $\lambda \geq 0$);
  \item (desigualdad triangular inversa) $\bigl|\abs z - \abs w\bigr|
        \leq \abs{z - w}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) y (2) son cálculos directos sobre las partes real e imaginaria.
(3): $\abs{zw}^2 = zw\,\conj{zw} = z\conj z\, w \conj w = \abs z^2
\abs w^2$, y se toman raíces cuadradas; aplíquese a $z \cdot z^{-1} =
1$ para el inverso.

(4) Los dos miembros son no negativos, así que se comparan cuadrados:
\[
\abs{z+w}^2 = (z+w)(\conj z + \conj w)
= \abs z^2 + \abs w^2 + 2\,\Re(z \conj w),
\]
y $\Re(z\conj w) \leq \abs{z \conj w} = \abs z \abs w$ da
$\abs{z+w}^2 \leq (\abs z + \abs w)^2$. La igualdad obliga a
$\Re(z \conj w) = \abs{z \conj w}$, es decir, $z \conj w \in \R_+$; si
$w \neq 0$, esto da $z = \frac{z\conj w}{\abs w^2}\, w = \lambda w$ con
$\lambda \geq 0$ (y el caso $w = 0$ es trivial).

(5) $\abs z = \abs{(z - w) + w} \leq \abs{z-w} + \abs w$ da
$\abs z - \abs w \leq \abs{z - w}$; intercámbiense $z$ y $w$ para el otro signo.
\end{proof}

\begin{example}[Un ejercicio completo de módulo y argumento]\label{ex:b1:complex:workout}
Póngase $w = \dfrac{3 + 4\iu}{1 - 2\iu}$ en forma algebraica y
calcúlese su \omterm{def:b1:complex:field}{módulo} de dos maneras. Multiplicando por el
\omterm{def:b1:complex:field}{conjugado} del denominador:
\[
w = \frac{(3 + 4\iu)(1 + 2\iu)}{(1 - 2\iu)(1 + 2\iu)}
= \frac{3 + 6\iu + 4\iu - 8}{1 + 4}
= \frac{-5 + 10\iu}{5} = -1 + 2\iu .
\]
Directamente: $\abs w = \sqrt{1 + 4} = \sqrt5$. Por la regla del
cociente (\cref{prop:b1:complex:rules}~(3)): $\abs w = \frac{\abs{3 +
4\iu}}{\abs{1 - 2\iu}} = \frac5{\sqrt5} = \sqrt5$ --- la misma
respuesta, sin necesidad de forma algebraica. La lección se
generaliza: \omterm{def:b1:complex:field}{módulos} y argumentos viajan bien a través de productos y
cocientes; las partes real e imaginaria, a través de sumas. Elíjase la
representación que se ajuste a las operaciones en juego y conviértase
solo cuando no quede más remedio.
\end{example}

\begin{example}[Ecuaciones con el conjugado]\label{ex:b1:complex:conjeq}
Resuélvase en $\C$: $\;z + 2\conj z = 6 + 2\iu$. Una ecuación que mezcla
$z$ y $\conj z$ \emph{no} es polinómica en $z$; lo fiable es separar en
coordenadas reales. Con $z = x + \iu y$:
\[
z + 2\conj z = 3x - \iu y ,
\]
la ecuación se lee $3x = 6$ y $-y = 2$: la única solución es
$z = 2 - 2\iu$. (Comprobación: $(2 - 2\iu) + 2(2 + 2\iu) = 6 + 2\iu$.)
Alternativamente, se conjuga toda la ecuación para obtener
$\conj z + 2z = 6 - 2\iu$ y se resuelve el sistema lineal en las
incógnitas $z, \conj z$ --- la misma respuesta, y un truco útil cuando
los coeficientes son complejos. Las ecuaciones en $z$ y $\conj z$ son
en realidad sistemas de dos ecuaciones reales; esperar «grado $1$, una
solución» es seguro aquí, pero $z\conj z = -1$ (sin solución) muestra
que la intuición polinómica falla en cuanto aparecen productos.
\end{example}

\section{Forma exponencial}

\begin{definition}[Exponencial compleja de argumento imaginario]\label{def:b1:complex:expi}
Para $\theta \in \R$ se define
\[
\eu^{\iu\theta} = \cos\theta + \iu \sin\theta .
\]
Todo $z \neq 0$ se puede escribir $z = r\,\eu^{\iu\theta}$ con
$r = \abs z > 0$; $\theta$ es un \emph{argumento}\index{argumento} de
$z$, determinado salvo múltiplos de $2\pi$. El valor de
$\intoc{-\pi}{\pi}$ es el \emph{argumento principal}, escrito
$\arg z$.
\end{definition}

\begin{example}[Primeros valores, y una identidad célebre]\label{ex:b1:complex:eulerid}
Leyendo la definición en los ángulos notables:
\[
\eu^{\iu\pi/2} = \iu, \qquad
\eu^{\iu\pi} = -1, \qquad
\eu^{2\iu\pi} = 1, \qquad
\eu^{\iu\pi/4} = \frac{\sqrt2}2\,(1 + \iu) .
\]
La segunda, reescrita como $\eu^{\iu\pi} + 1 = 0$, es la célebre
identidad de Euler, que enlaza $\eu$, $\iu$, $\pi$, $1$ y $0$; a esta
altura del libro es un desarrollo de la definición más que un teorema,
y su contenido real --- por qué la función exponencial analítica
del~\cref{ch:b1:functions}, extendida a argumentos imaginarios, merece
el mismo nombre --- lo zanjan las series de potencias
del~\cref{ch:b1:series}. Entretanto, conviene memorizar la fórmula
anterior como tabla de conversión: se usa en silencio cada vez que se
lee un argumento en un dibujo.
\end{example}

\begin{theorem}[La ecuación funcional]\label{thm:b1:complex:funceq}
Para todos $\theta, \varphi \in \R$:
\[
\eu^{\iu\theta}\, \eu^{\iu\varphi} = \eu^{\iu(\theta + \varphi)},
\qquad
\abs{\eu^{\iu\theta}} = 1,
\qquad
\conj{\eu^{\iu\theta}} = \eu^{-\iu\theta} = (\eu^{\iu\theta})^{-1}.
\]
En consecuencia, $\abs{zw} = \abs z \abs w$ arrastra también los
argumentos: $\arg(zw) \equiv \arg z + \arg w \pmod{2\pi}$.
\end{theorem}

\begin{proof}
Se desarrolla el producto y se usan las fórmulas de adición:
\[
(\cos\theta + \iu\sin\theta)(\cos\varphi + \iu\sin\varphi)
= (\cos\theta\cos\varphi - \sin\theta\sin\varphi)
+ \iu\,(\sin\theta\cos\varphi + \cos\theta\sin\varphi),
\]
que es $\cos(\theta+\varphi) + \iu\sin(\theta+\varphi)$. El \omterm{def:b1:complex:field}{módulo} vale
$\sqrt{\cos^2\theta + \sin^2\theta} = 1$, y la fórmula del \omterm{def:b1:complex:field}{conjugado} es
la paridad del coseno y del seno; esta invierte $\eu^{\iu\theta}$, ya
que $\eu^{\iu\theta}\eu^{-\iu\theta} = \eu^0 = 1$.
\end{proof}

\begin{corollary}[Fórmula de De Moivre]\label{cor:b1:complex:demoivre}
Para $\theta \in \R$ y $n \in \Z$:
$\;(\cos\theta + \iu\sin\theta)^n = \cos(n\theta) +
\iu\sin(n\theta)$.\index{fórmula de De Moivre}
\end{corollary}

\begin{proof}
Para $n \geq 0$, se procede por inducción. El caso $n = 0$ dice
$1 = 1$. Suponiendo la fórmula para $n$, la ecuación funcional
(\cref{thm:b1:complex:funceq}) da
\[
\bigl(\eu^{\iu\theta}\bigr)^{n+1}
= \bigl(\eu^{\iu\theta}\bigr)^{n}\,\eu^{\iu\theta}
= \eu^{\iu n\theta}\,\eu^{\iu\theta}
= \eu^{\iu(n+1)\theta} ,
\]
que es la fórmula en el rango $n + 1$. Para $n < 0$, escríbase $n = -m$
con $m > 0$: como $\eu^{\iu m\theta}$ tiene inverso
$\eu^{-\iu m\theta}$ (de nuevo \cref{thm:b1:complex:funceq}),
\[
\bigl(\eu^{\iu\theta}\bigr)^{-m}
= \bigl(\eu^{\iu m\theta}\bigr)^{-1}
= \eu^{\iu(-m)\theta} . \qedhere
\]
\end{proof}

\begin{example}[Desarrollo de $\cos 3\theta$ por De Moivre]\label{ex:b1:complex:cos3}
Escríbase $c = \cos\theta$, $s = \sin\theta$. De Moivre y el teorema
del binomio dan
\[
\cos 3\theta + \iu \sin 3\theta = (c + \iu s)^3
= c^3 - 3cs^2 + \iu\,(3c^2 s - s^3),
\]
e igualando partes reales y sustituyendo después $s^2 = 1 - c^2$:
\[
\cos 3\theta = c^3 - 3c(1 - c^2) = 4\cos^3\theta - 3\cos\theta .
\]
La parte imaginaria da gratis $\sin 3\theta = 3\sin\theta -
4\sin^3\theta$: una identidad compleja arrastra siempre \emph{dos}
reales. Leída al revés, la identidad anterior es la clave de la
clásica ecuación de la trisección: construir $\cos(\theta/3)$ a partir
de $\cos\theta$ equivale a resolver la cúbica $4x^3 - 3x = \cos\theta$,
que es donde toma el relevo el álgebra del~\cref{ch:b1:poly}.
\end{example}

\begin{proposition}[Fórmulas de Euler]\label{prop:b1:complex:euler}
\[
\cos\theta = \frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2},
\qquad
\sin\theta = \frac{\eu^{\iu\theta} - \eu^{-\iu\theta}}{2\iu}.
\]
\end{proposition}

\begin{proof}
Súmense, respectivamente réstense, $\eu^{\iu\theta} = \cos\theta +
\iu\sin\theta$ y $\eu^{-\iu\theta} = \cos\theta - \iu\sin\theta$.
\end{proof}

\begin{method}[Trigonometría mediante exponenciales]\label{met:b1:complex:trig}
\begin{enumerate}
  \item \emph{Linealizar} $\cos^p\theta \sin^q\theta$ (convertir
        potencias en una suma de $\cos k\theta$, $\sin k\theta$):
        sustitúyanse las fórmulas de Euler, desarróllese con el teorema
        del binomio (\cref{thm:b1:counting:binomial}) y agrúpense los
        términos \omterm{def:b1:complex:field}{conjugados}.
  \item \emph{Desarrollar} $\cos n\theta$ como polinomio en
        $\cos\theta$: escríbase $\cos n\theta = \Re\bigl((\cos\theta +
        \iu\sin\theta)^n\bigr)$, desarróllese y conviértanse las
        potencias pares de $\sin$ mediante $\sin^2 = 1 - \cos^2$.
  \item \emph{Sumar series trigonométricas} como $\sum_k \cos k\theta$:
        recuérdese que son la parte real de una suma geométrica
        $\sum_k (\eu^{\iu\theta})^k$.
  \item La \emph{factorización por el ángulo mitad}: para todos $p, q$,
        \[
        \eu^{\iu p} + \eu^{\iu q}
        = 2 \cos\tfrac{p - q}{2}\; \eu^{\iu \frac{p+q}{2}},
        \qquad
        \eu^{\iu p} - \eu^{\iu q}
        = 2\iu \sin\tfrac{p - q}{2}\; \eu^{\iu \frac{p+q}{2}} .
        \]
\end{enumerate}
\end{method}

\begin{example}[Linealización]\label{ex:b1:complex:linearize}
\[
\cos^3\theta
= \Bigl(\frac{\eu^{\iu\theta} + \eu^{-\iu\theta}}{2}\Bigr)^{\!3}
= \frac{\eu^{3\iu\theta} + 3\eu^{\iu\theta} + 3\eu^{-\iu\theta} +
\eu^{-3\iu\theta}}{8}
= \frac{\cos 3\theta + 3\cos\theta}{4}.
\]
Esta forma se integra de inmediato --- justamente por eso importa la
linealización en el~\cref{ch:b1:integration}. Un producto mixto
funciona igual, solo que con las dos fórmulas de Euler a la vez:
\[
\sin^2\theta\cos^2\theta
= \Bigl(\frac{\sin2\theta}2\Bigr)^{\!2}
= \frac{1}{4}\cdot
\Bigl(\frac{\eu^{2\iu\theta} - \eu^{-2\iu\theta}}{2\iu}
\Bigr)^{\!2}
= \frac{2 - \eu^{4\iu\theta} - \eu^{-4\iu\theta}}{16}
= \frac{1 - \cos4\theta}{8} ,
\]
donde el atajo del ángulo doble en el primer paso ahorró un desarrollo
binomial --- siempre conviene buscarlo antes de mecanizar.
\end{example}

\begin{example}[Una suma trigonométrica binomial]\label{ex:b1:complex:binomsum}
Para $n \in \N$ y $\theta \in \R$, calcúlese
$S = \sum_{k=0}^{n}\binom nk \cos k\theta$. Reconózcase la parte real
de un desarrollo binomial:
\[
S = \Re\sum_{k=0}^n \binom nk \bigl(\eu^{\iu\theta}\bigr)^k
= \Re\bigl(1 + \eu^{\iu\theta}\bigr)^n ,
\]
y factorícese después el ángulo mitad (\cref{met:b1:complex:trig}~(4)):
$1 + \eu^{\iu\theta} = 2\cos\frac\theta2\,\eu^{\iu\theta/2}$, luego
\[
S = \Re\Bigl(2^n\cos^n\frac\theta2\;\eu^{\iu n\theta/2}\Bigr)
= 2^n \cos^n\frac\theta2\,\cos\frac{n\theta}2 .
\]
La parte imaginaria da gratis $\sum_k\binom nk\sin k\theta =
2^n\cos^n\frac\theta2\sin\frac{n\theta}2$. Comprobaciones: $\theta = 0$
recupera $\sum\binom nk = 2^n$, y $\theta = \pi$ da $S = 0$ para
$n \geq 1$ (cada factor $\cos\frac\pi2$ se anula), es decir, la suma
alternada de una fila del~\cref{ex:b1:counting:binomial}. El método
---«ver la suma de cosenos como la sombra de una potencia compleja y
factorizar después ángulos mitad»--- es exactamente el
del~\cref{exo:b1:complex:6}, con el teorema del binomio en lugar de la
serie geométrica.
\end{example}

\section{Raíces de números complejos}

\begin{theorem}[Raíces $n$-ésimas]\label{thm:b1:complex:roots}
Sean $a = r\,\eu^{\iu\alpha} \neq 0$ y $n \in \N^*$. La ecuación
$z^n = a$ tiene exactamente $n$ soluciones:
\[
z_k = r^{1/n}\, \eu^{\iu\left(\frac{\alpha}{n} +
\frac{2k\pi}{n}\right)},
\qquad k = 0, 1, \dots, n - 1 .
\]
\end{theorem}

\begin{proof}
Escríbase $z = \rho\,\eu^{\iu\theta}$ ($\rho > 0$). Entonces
$z^n = \rho^n \eu^{\iu n\theta} = r \eu^{\iu\alpha}$ si y solo si
$\rho^n = r$ (\omterm{def:b1:complex:field}{módulos}) y $n\theta \equiv \alpha \pmod{2\pi}$
(argumentos), es decir, $\rho = r^{1/n}$ y
$\theta = \frac{\alpha}{n} + \frac{2k\pi}{n}$ para algún $k \in \Z$.
Queda por ver cuándo dos enteros $k, k'$ dan el mismo número: eso
ocurre exactamente cuando los ángulos difieren en un múltiplo de $2\pi$,
\[
\frac{2k\pi}n - \frac{2k'\pi}n \in 2\pi\Z
\iff \frac{k - k'}n \in \Z
\iff n \mid k - k' .
\]
Por la división euclídea, todo $k \in \Z$ es congruente \omterm{def:b1:complex:field}{módulo} $n$ con
exactamente un elemento de $\{0, 1, \dots, n-1\}$, de modo que ese
rango enumera cada solución una sola vez y el recuento es exactamente
$n$. (El mismo argumento, hecho dentro de $\mathbb U_n$, muestra que
las raíces forman un $n$-ágono regular: los valores consecutivos de $k$
giran el ángulo fijo $\frac{2\pi}n$.)
\end{proof}

\begin{example}[Raíces cúbicas de $-27$]\label{ex:b1:complex:cuberoots}
Resuélvase $z^3 = -27$. Forma exponencial del miembro derecho:
$-27 = 27\,\eu^{\iu\pi}$, así que las tres raíces son
\[
z_k = 3\,\eu^{\iu(\frac\pi3 + \frac{2k\pi}3)}, \quad k = 0, 1, 2 :
\qquad
z_0 = 3\eu^{\iu\pi/3} = \frac32 + \frac{3\sqrt3}2\,\iu,
\quad
z_1 = -3,
\quad
z_2 = \conj{z_0} .
\]
Dos comprobaciones. Primera: la raíz real $-3$ es la evidente, y las
otras dos son sus rotaciones de $\pm\frac{2\pi}3$ --- equivalentemente,
$-3j$ y $-3j^2$. Segunda: el álgebra lo confirma,
$z^3 + 27 = (z + 3)(z^2 - 3z + 9)$, y la cuadrática tiene discriminante
$9 - 36 = -27 < 0$ con raíces $\frac{3 \pm 3\iu\sqrt3}2 = z_0,
\conj{z_0}$. La idea clave: para miembros derechos reales, las raíces
no reales vienen siempre en pares \omterm{def:b1:complex:field}{conjugados}, de modo que el dibujo del
\omterm{def:b1:logic:sets}{conjunto} de soluciones es simétrico respecto del eje real --- un
anticipo del teorema de factorización real del~\cref{ch:b1:poly}.
\end{example}

\begin{example}[Un miembro derecho no real]\label{ex:b1:complex:fourthroots}
Resuélvase $z^4 = -8 + 8\iu\sqrt3$. Forma exponencial del miembro
derecho: \omterm{def:b1:complex:field}{módulo} $\sqrt{64 + 192} = 16$ y argumento $\theta$ con
$\cos\theta = -\frac12$, $\sin\theta = \frac{\sqrt3}2$, es decir,
$\theta = \frac{2\pi}3$. Las cuatro raíces son
\[
z_k = 2\,\eu^{\iu(\frac\pi6 + \frac{k\pi}2)}, \quad k = 0, 1, 2,
3 :
\qquad
z_0 = \sqrt3 + \iu,\quad z_1 = \iu z_0 = -1 + \iu\sqrt3,
\]
\[
z_2 = -z_0 = -\sqrt3 - \iu,\qquad z_3 = -\iu z_0 = 1 - \iu\sqrt3 .
\]
(Comprobación: $z_0^2 = 2 + 2\iu\sqrt3$, luego $z_0^4 = (2 +
2\iu\sqrt3)^2 = 4 - 12 + 8\iu\sqrt3 = -8 + 8\iu\sqrt3$.) Una vez
hallada \emph{una} raíz, las otras tres salen gratis: son sus
rotaciones sucesivas de $\frac\pi2$, es decir, sus productos por las
raíces cuartas de la unidad --- la estructura general que hay detrás
del~\cref{thm:b1:complex:roots}, y que conviene aprovechar antes de
recalcular cada raíz desde cero. Esta vez no hay simetría de
conjugación: el miembro derecho no es real.
\end{example}

\begin{definition}[Raíces de la unidad]\label{def:b1:complex:unity}
Las \emph{raíces $n$-ésimas de la unidad}\index{raíces de la unidad}
son las soluciones de $z^n = 1$:
\[
\mathbb{U}_n = \bigl\{\, \omega^k : k = 0, \dots, n-1 \,\bigr\},
\qquad \omega = \eu^{2\iu\pi/n}.
\]
Forman un grupo con la multiplicación (\cref{ch:b1:structures}) y se
sitúan en los vértices de un $n$-ágono regular inscrito en la
circunferencia unidad.
\end{definition}

\begin{proposition}[Suma de las raíces de la unidad]\label{prop:b1:complex:sumroots}
Para $n \geq 2$, las raíces $n$-ésimas de la unidad suman cero:
$\sum_{k=0}^{n-1} \omega^k = 0$.
\end{proposition}

\begin{proof}
Suma geométrica de razón $\omega \neq 1$:
$\sum_{k=0}^{n-1} \omega^k = \frac{\omega^n - 1}{\omega - 1} = 0$,
puesto que $\omega^n = 1$.
\end{proof}

\begin{example}[Leer las partes real e imaginaria]\label{ex:b1:complex:sumrootsparts}
Separar $\sum_{k=0}^{n-1}\omega^k = 0$ en partes real e imaginaria da
gratis dos identidades trigonométricas:
\[
\sum_{k=0}^{n-1}\cos\frac{2k\pi}n = 0,
\qquad
\sum_{k=0}^{n-1}\sin\frac{2k\pi}n = 0
\qquad (n \geq 2).
\]
Geométricamente: el baricentro de un $n$-ágono regular inscrito en la
circunferencia unidad es su centro: los vértices se compensan
exactamente. Para $n = 5$, la primera da
$1 + 2\cos\frac{2\pi}5 + 2\cos\frac{4\pi}5 = 0$ (emparejando $k$ con
$n - k$), punto de partida del cálculo de $\cos\frac{2\pi}5$
del~\cref{exo:b1:complex:8}.
\end{example}

\begin{example}[Raíces cuadradas en forma algebraica]\label{ex:b1:complex:sqrt}
Para resolver $z^2 = 3 + 4\iu$ sin trigonometría, póngase $z = x + \iu y$:
\[
x^2 - y^2 = 3, \qquad 2xy = 4, \qquad x^2 + y^2 = \abs{3 + 4\iu} = 5 .
\]
Sumando la primera y la última: $x^2 = 4$, luego $x = \pm 2$, y
entonces $y = 2/x = \pm 1$ con el \emph{mismo} emparejamiento de signos
($xy = 2 > 0$): $z = \pm(2 + \iu)$. Combinado con la fórmula habitual,
esto resuelve toda ecuación de segundo grado con coeficientes complejos
(\cref{exo:b1:complex:7}).
\end{example}

\section{Números complejos y geometría del plano}

\begin{proposition}[Diccionario geométrico]\label{prop:b1:complex:geometry}
Identifíquese el punto $M(x, y)$ del plano con su \emph{afijo}
$z = x + \iu y$\index{afijo}. Para puntos distintos $A, B, C$ de afijos
$a, b, c$:
\begin{enumerate}
  \item $\abs{b - a}$ es la distancia $AB$;
  \item $\arg\dfrac{c - a}{b - a}$ es el ángulo entre los vectores
        $\vect{AB}$ y $\vect{AC}$ (\omterm{def:b1:complex:field}{módulo} $2\pi$);
  \item $A, B, C$ están alineados si y solo si $\dfrac{c - a}{b - a} \in
        \R$; las rectas $AB$ y $AC$ son perpendiculares si y solo si
        $\dfrac{c - a}{b - a} \in \iu\R$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) es la definición de \omterm{def:b1:complex:field}{módulo} aplicada a $b - a$, el afijo de
$\vect{AB}$. (2): escríbanse $b - a = r\eu^{\iu\theta}$,
$c - a = s\eu^{\iu\varphi}$; entonces $\frac{c-a}{b-a} = \frac sr
\eu^{\iu(\varphi - \theta)}$ tiene argumento $\varphi - \theta$, el
ángulo de $\vect{AB}$ a $\vect{AC}$. (3): estar alineados significa
ángulo $0$ o $\pi$, es decir, argumento en $\pi\Z$, es decir, cociente
real; ser perpendiculares significa ángulo $\pm\frac\pi2$, es decir,
cociente imaginario puro. (El cociente no es nulo, pues $C \neq A$.)
\end{proof}

\begin{remark}[Interludio: $\C$ es el plano con una multiplicación]\label{rem:b1:complex:interlude}
Vale la pena detenerse en lo que hace posible este capítulo: el plano
$\R^2$ tiene sumas en todas las direcciones, pero ninguna
multiplicación dada de antemano --- y $\C$ \emph{es} el plano dotado de
una, en la que multiplicar por un número fijo gira y escala. Esta única
estructura se exprimirá tres veces más en este volumen. En
el~\cref{ch:b1:matrices}, la multiplicación por $a + \iu b$ reaparece
como la matriz $2 \times 2$ de filas $(a, -b)$ y $(b, a)$: la
aritmética compleja es una primera familia de productos de matrices,
enteramente concreta. En el~\cref{ch:b1:euclid}, la fórmula
$\Re(\conj z\,w)$ resulta ser el producto escalar, y $\abs z$ la norma
euclídea: la desigualdad triangular demostrada aquí es el modelo de la
historia de Cauchy--Schwarz de allí. Y en el~\cref{ch:b1:curves}, un
punto móvil se escribe mejor $t \mapsto z(t)$, de modo que velocidad y
aceleración se vuelven derivadas de valores complejos --- el movimiento
circular, por ejemplo, es sin más $z(t) = R\,\eu^{\iu\omega t}$. Una
buena multiplicación, cuatro capítulos de dividendos.
\end{remark}

\begin{proposition}[Las aplicaciones $z \mapsto az + b$]\label{prop:b1:complex:similitude}
Sean $a \in \C^*$, $b \in \C$. La transformación $f(z) = az + b$ del
plano:
\begin{itemize}
  \item es una traslación si $a = 1$;
  \item en caso contrario tiene un único punto fijo
        $\zeta = \frac{b}{1-a}$, y $f(z) - \zeta = a\,(z - \zeta)$: $f$
        es la rotación de centro $\zeta$ y ángulo $\arg a$, compuesta
        con la homotecia de centro $\zeta$ y razón $\abs a$.
\end{itemize}
En particular, $z \mapsto \eu^{\iu\theta} z$ es la rotación de ángulo
$\theta$ alrededor del origen, y $\conj z$ es la simetría respecto del
eje real.
\end{proposition}

\begin{proof}
Si $a = 1$, $f(z) = z + b$ traslada según el vector de afijo $b$. Si
$a \neq 1$, la ecuación de punto fijo $z = az + b$ tiene la única
solución $\zeta = \frac{b}{1-a}$, y entonces $f(z) - \zeta = az + b -
(a\zeta + b) = a(z - \zeta)$. Escribiendo $a = \abs a\, \eu^{\iu\arg a}$,
multiplicar por $a$ escala las distancias a $\zeta$ por $\abs a$ y suma
$\arg a$ a los ángulos en $\zeta$, que es la composición anunciada.
\end{proof}

\begin{example}[Clasificar una aplicación $z \mapsto az+b$]\label{ex:b1:complex:classify}
Tómese $f(z) = \iu z + 1$. Aquí $a = \iu \neq 1$: el punto fijo es
\[
\zeta = \frac{b}{1 - a} = \frac1{1 - \iu}
= \frac{1 + \iu}{2},
\]
y, como $\abs a = 1$, no hay escala alguna: $f$ es la rotación pura de
centro $\frac{1+\iu}2$ y ángulo $\arg \iu = \frac\pi2$. Comprobación:
$f(\zeta) = \iu\,\frac{1+\iu}2 + 1 = \frac{\iu - 1}2 + 1 =
\frac{1 + \iu}2 = \zeta$, y $f(0) = 1$, $f(1) = 1 + \iu$,
$f(1 + \iu) = \iu\,(1 + \iu) + 1 = \iu$: los cuatro puntos
$0, 1, 1+\iu, \iu$ del cuadrado unidad rotan alrededor de su centro
$\zeta$, un cuarto de vuelta cada vez --- exactamente lo que debe hacer
una rotación de $\frac\pi2$ alrededor del centro del cuadrado.
\end{example}

\begin{remark}[Errores frecuentes con módulos y argumentos]\label{rem:b1:complex:pitfalls}
\begin{enumerate}
  \item \emph{En $\C$ no hay desigualdades.} Escribir $z \leq w$ para
        números no reales carece de sentido; solo se pueden comparar
        \omterm{def:b1:complex:field}{módulos}, partes reales y partes imaginarias.
  \item \emph{$\sqrt{\phantom z}$ está reservado para $\R_+$.} Todo
        complejo no nulo tiene \emph{dos} raíces cuadradas y ninguna es
        privilegiada: escríbase «sea $\delta$ una raíz cuadrada de
        $\Delta$» (calculada como en \cref{ex:b1:complex:sqrt}), nunca
        $\sqrt\Delta$ --- la regla $\sqrt{ab} = \sqrt a\sqrt b$ ya
        falla en $a = b = -1$.
  \item \emph{Los argumentos viven \omterm{def:b1:complex:field}{módulo} $2\pi$.} De
        $\eu^{\iu\alpha} = \eu^{\iu\beta}$ dedúzcase $\alpha
        \equiv \beta \pmod{2\pi}$, no $\alpha = \beta$; olvidar el
        $2k\pi$ es como los \omterm{def:b1:logic:sets}{conjuntos} de soluciones de $z^n = a$
        pierden $n - 1$ de sus $n$ elementos.
  \item \emph{$\abs{z + w}$ no es $\abs z + \abs w$.} La igualdad en la
        desigualdad triangular es el caso excepcional de alineación
        (\cref{prop:b1:complex:rules}~(4)); en general, el \omterm{def:b1:complex:field}{módulo} de una
        suma hay que acotarlo, no calcularlo.
\end{enumerate}
\end{remark}

\begin{example}\label{ex:b1:complex:equilateral}
$A, B, C$ (de afijos $a, b, c$, distintos dos a dos) forman un
triángulo equilátero con los vértices en orden directo (antihorario) si
y solo si $\frac{c - a}{b - a} = \eu^{\iu\pi/3}$: la rotación de centro
$A$ y ángulo $\frac\pi3$ envía $B$ a $C$. Las dos orientaciones juntas
quedan recogidas en la ecuación simétrica $a^2 + b^2 + c^2 = ab + bc +
ca$ (\cref{exo:b1:complex:10}).
\end{example}

\begin{remark}[Dónde se usa este capítulo]\label{rem:b1:complex:whereused}
La forma exponencial es el recurso de cálculo más reutilizado del
volumen. Las \omterm{def:b1:complex:unity}{raíces de la unidad} se convierten en el ejemplo estándar
de grupo cíclico en el~\cref{ch:b1:structures} y mueven la
factorización de $X^n - 1$ en el~\cref{ch:b1:poly}; agrupar allí las
raíces \omterm{def:b1:complex:field}{conjugadas} produce las factorizaciones reales que usan las
fracciones simples del~\cref{ch:b1:fractions}. La linealización
(\cref{met:b1:complex:trig}) es la preparación estándar para integrar
potencias trigonométricas en el~\cref{ch:b1:integration}, y las
ecuaciones características del~\cref{ch:b1:diffeq} tienen raíces
complejas cuyas partes real e imaginaria producen las soluciones
oscilantes $\eu^{\lambda t}\cos\omega t$. El diccionario geométrico
vuelve con ropaje matricial: las rotaciones y las semejanzas se
convierten en las matrices ortogonales de
los~\cref{ch:b1:matrices,ch:b1:euclid}, y las curvas parametrizadas
del~\cref{ch:b1:curves} se escriben a menudo mejor como aplicaciones
$t \mapsto z(t) \in \C$.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:complex:1}
Póngase en forma exponencial: $1 + \iu$; $\;\sqrt 3 - \iu$; $\;-5$;
$\;\dfrac{1 + \iu}{\sqrt 3 - \iu}$. Dedúzcanse
$\cos\frac{5\pi}{12}$ y $\sin\frac{5\pi}{12}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:2}
Calcúlese $(1 + \iu)^{20}$. ¿Para qué $n \in \N$ es $(1 + \iu)^n$ un
número real?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:3}
Descríbase geométricamente el \omterm{def:b1:logic:sets}{conjunto} de los $z \in \C$ tales que:
$\abs{z - 2} = \abs{z + \iu}$; $\;\abs{z - 1} = 2$;
$\;\dfrac{z - 1}{z + 1} \in \iu\R$ (para $z \neq -1$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:4}
Linealícese $\sin^4\theta$ y desarróllese $\cos 4\theta$ como polinomio
en $\cos\theta$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:complex:5}
Resuélvase $z^3 = 8\iu$ y sitúense las soluciones en un dibujo.
Resuélvase $z^4 = -4$ y factorícese $X^4 + 4$ en dos polinomios reales de segundo grado.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:6}
Para $\theta \in \R$ con $\eu^{\iu\theta} \neq 1$ y $n \in \N$,
calcúlense
\[
C_n = \sum_{k=0}^{n} \cos k\theta
\qquad\text{y}\qquad
S_n = \sum_{k=0}^{n} \sin k\theta
\]
sumando la serie geométrica $\sum_k \eu^{\iu k\theta}$ y usando la
factorización por el ángulo mitad.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:7}
Resuélvase en $\C$: $z^2 - (3 + 4\iu) z + (-1 + 5\iu) = 0$.
\emph{(Calcúlese el discriminante y extráiganse sus raíces cuadradas
como en \cref{ex:b1:complex:sqrt}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:8}
Sea $\omega = \eu^{2\iu\pi/5}$.
\begin{enumerate}
  \item Justifíquese $1 + \omega + \omega^2 + \omega^3 + \omega^4 = 0$.
  \item Póngase $u = \omega + \omega^4$ y $v = \omega^2 + \omega^3$.
        Calcúlense $u + v$ y $uv$, y dedúzcase que $u$ y $v$ son las
        raíces de $X^2 + X - 1$.
  \item Conclúyase que $\cos\frac{2\pi}{5} = \frac{\sqrt 5 - 1}{4}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:9}
Demuéstrese que, para todos $z, w \in \C$ (identidad del paralelogramo):
\[
\abs{z + w}^2 + \abs{z - w}^2 = 2\abs z^2 + 2\abs w^2 ,
\]
e interprétese geométricamente en el paralelogramo de vértices $0, z,
w, z + w$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:complex:10}
Demuéstrese que tres puntos distintos dos a dos, de afijos $a, b, c$,
forman un triángulo equilátero (con cualquier orientación) si y solo si
\[
a^2 + b^2 + c^2 = ab + bc + ca .
\]
\emph{Indicación: el caso directo es $\frac{c-a}{b-a} = -j^2$ y el
indirecto $\frac{c-a}{b-a} = -j$, donde $j = \eu^{2\iu\pi/3}$ cumple
$j^2 + j + 1 = 0$; o bien factorícense $a + jb + j^2c$ y $a + j^2 b
+ jc$.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:complex:11}
Para $n \in \N^*$, calcúlese $P = \prod_{k=1}^{n-1}
\bigl(1 - \omega^k\bigr)$, donde $\omega = \eu^{2\iu\pi/n}$.
\emph{Indicación: $X^n - 1 = \prod_{k=0}^{n-1} (X - \omega^k)$;
divídase por $X - 1$ y evalúese en $X = 1$.} Dedúzcase
$\prod_{k=1}^{n-1} \sin\frac{k\pi}{n} = \dfrac{n}{2^{n-1}}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:complex:12}
Sea $n \geq 2$. Resuélvase la ecuación $(z + 1)^n = (z - 1)^n$ en $\C$:
véase que tiene exactamente $n - 1$ soluciones, todas imaginarias
puras, a saber,
\[
z_k = -\iu\,\frac{\cos\frac{k\pi}{n}}{\sin\frac{k\pi}{n}},
\qquad k = 1, \dots, n - 1 .
\]
\emph{Indicación: $z = 1$ no es solución, así que se divide y se usan
las \omterm{def:b1:complex:unity}{raíces de la unidad}; aplíquese después la factorización por el
ángulo mitad del~\cref{met:b1:complex:trig}.}
\end{exercise}

\section{Problema: el teorema de Napoleón}

\begin{problem}\label{pb:b1:complex:1}
Levántese un triángulo equilátero hacia fuera sobre cada lado de un
triángulo \emph{arbitrario}: los tres centros de esos triángulos forman
siempre un triángulo equilátero. Este es el teorema de
Napoleón\index{teorema de Napoleón} --- un \omterm{def:b1:logic:statement}{enunciado} sin razón visible
para ser cierto, que el álgebra de $j = \eu^{2\iu\pi/3}$ demuestra en
tres líneas de cálculo. Este problema construye toda la caja de
herramientas (rotaciones, semejanzas directas, el criterio de $j$
del~\cref{exo:b1:complex:10}), demuestra el teorema de Napoleón interior
y exterior, localiza el caso degenerado y cierra con otra joya de la
misma escuela: el teorema de Van Schooten sobre triángulos equiláteros
y la desigualdad de Ptolomeo. En todo el problema, los puntos del plano
se identifican con sus afijos.

\medskip
\noindent\textbf{Parte I --- El número $j$ y las rotaciones.}
\begin{enumerate}
  \item Calcúlese la forma algebraica de $j$, y después $j^2$, $j^3$,
        $1 + j + j^2$, $\conj j$ y $j^{-1}$. Sitúense $1$, $j$, $j^2$
        en un esbozo de la circunferencia unidad.
  \item Pruébese que la rotación de centro $a$ y ángulo $\theta$ es
        $r(z) = a + \eu^{\iu\theta}(z - a)$, y que la composición de
        dos rotaciones, de ángulos $\theta$ y $\theta'$, es una
        rotación de ángulo $\theta + \theta'$ si $\theta + \theta'
        \notin 2\pi\Z$, y una traslación en caso contrario. (Úsese
        la~\cref{prop:b1:complex:similitude}.)
  \item Sea $b \neq c$. Pruébese que hay exactamente dos puntos $p$
        que hacen equilátero a $(b, c, p)$, a saber, $p = b +
        \eu^{\pm\iu\pi/3}(c - b)$. Para un triángulo \emph{directo}
        $(a, b, c)$, compruébese en el ejemplo $a = 0$, $b = 1$, $c =
        \iu$ que la elección $\eu^{-\iu\pi/3}$ es la que queda al otro
        lado de la recta $BC$ respecto de $a$ --- el vértice
        \emph{exterior}.
  \item De la solución del~\cref{exo:b1:complex:10}, $(a, b, c)$ es
        equilátero directo si y solo si $a + jb + j^2c = 0$.
        Demuéstrense las dos identidades de rotación
        \[
        b + jc + j^2a = j^2\,(a + jb + j^2c),
        \qquad
        c + ja + j^2b = j\,(a + jb + j^2c),
        \]
        y dedúzcase que el criterio es invariante por \omterm{def:b1:counting:objects}{permutación}
        cíclica de $(a, b, c)$.
  \item Pruébese que si $a + jb + j^2c = 0$ y dos de los tres puntos
        coinciden, entonces los tres coinciden. (De modo que el
        criterio caracteriza exactamente: triángulo equilátero
        directo, o un solo punto.)
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Semejanzas directas.}
\begin{enumerate}[resume]
  \item Pruébese que las aplicaciones $f(z) = az + b$ con $a \in \C^*$
        (\emph{semejanzas directas}) son estables por composición e
        inversión: la composición de dos de ellas, y la inversa de
        cualquiera de ellas, vuelve a ser de esta forma. (En el
        lenguaje del~\cref{ch:b1:structures}: forman un grupo.)
  \item Dados $z_1 \neq z_2$ y $w_1 \neq w_2$, pruébese que hay una
        \emph{única} semejanza directa $f$ con $f(z_1) = w_1$ y
        $f(z_2) = w_2$, y dense $a$ y $b$ explícitamente.
  \item Pruébese que una semejanza directa $f(z) = az + b$ multiplica
        todas las distancias por $\abs a$ y conserva la \emph{forma}
        $\frac{c - a'}{b' - a'}$ de todo triángulo $(a', b', c')$
        \emph{(numerador y denominador quedan ambos multiplicados por
        $a$)}. Dedúzcase que dos triángulos son directamente semejantes
        exactamente cuando sus formas coinciden.
  \item Determínese por completo la semejanza directa con $f(0) = 1$ y
        $f(1) = \iu$: dense $a$, $b$, el punto fijo, la razón y el
        ángulo.
  \item Sea $\alpha + \beta = 1$. Pruébese que el «punto ponderado»
        $g(a', b') = \alpha a' + \beta b'$ conmuta con toda semejanza
        directa: $f(\alpha a' + \beta b') = \alpha f(a') + \beta f(b')$.
        Dedúzcase que baricentros, puntos medios y los centros de
        Napoleón de más abajo son transportados por las semejanzas ---
        la licencia algebraica que hay detrás de todo argumento del
        tipo «sin pérdida de generalidad, tómese la circunferencia
        circunscrita como circunferencia unidad».
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El teorema de Napoleón.} Sea $(a, b, c)$
un triángulo directo. Sobre cada lado levántese el triángulo equilátero
exterior (pregunta 3) y sean $n_a$, $n_b$, $n_c$ los centros
(baricentros) de los triángulos levantados sobre $[b, c]$, $[c, a]$,
$[a, b]$, respectivamente.
\begin{enumerate}[resume]
  \item Pruébese que
        \[
        n_a = \alpha b + \beta c, \qquad
        n_b = \alpha c + \beta a, \qquad
        n_c = \alpha a + \beta b,
        \qquad\text{con}\quad
        \alpha = \frac{3 + \iu\sqrt3}{6},\ \beta = \conj\alpha .
        \]
  \item Compruébese que $\alpha + \beta = 1$ y dedúzcase que el
        triángulo $(n_a, n_b, n_c)$ tiene el \emph{mismo baricentro}
        que $(a, b, c)$.
  \item Usando las identidades de la pregunta 4, pruébese que
        \[
        n_a + j\,n_b + j^2 n_c
        = (\alpha j^2 + \beta j)\,(a + jb + j^2c) .
        \]
  \item Calcúlese $\alpha j + \beta$ y conclúyase: $\alpha j^2 +
        \beta j = j(\alpha j + \beta) = 0$, luego $n_a + j n_b + j^2
        n_c = 0$ para \emph{todo} triángulo: el triángulo de Napoleón
        exterior es equilátero directo (o un punto). El teorema de
        Napoleón queda demostrado.
  \item Levántense ahora los triángulos equiláteros \emph{hacia
        dentro} (elección $\eu^{+\iu\pi/3}$ en la pregunta 3) y sean
        $m_a, m_b, m_c$ sus centros. Pruébese que $m_a = \beta b +
        \alpha c$ (y cíclicamente), y demuéstrese después que
        $m_a + j^2 m_b + j\,m_c = 0$: el triángulo de Napoleón
        interior también es equilátero, con la orientación opuesta.
  \item Localícese la degeneración: pruébese que $n_a = n_b = n_c$
        ocurre exactamente cuando $a + j^2 b + jc = 0$, es decir,
        cuando $(a, b, c)$ es un triángulo equilátero
        \emph{indirecto}. \emph{(Dos de los puntos $n_a, n_b, n_c$
        coinciden si y solo si los tres coinciden, por la pregunta 5;
        úsese después $n_a + j^2 n_b + j n_c = j(\alpha + \beta j)(a +
        j^2b + jc)$ y compruébese que $\alpha + \beta j \neq 0$.)}
  \item Hágase todo el cálculo con el triángulo $a = 0$, $b = 1$,
        $c = \iu$: dense $n_a, n_b, n_c$ exactamente y verifíquese, por
        cálculo directo de las tres longitudes al cuadrado de los
        lados, que el triángulo es equilátero, de lado al cuadrado
        $\frac{2 + \sqrt3}{3}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Ptolomeo y Van Schooten.}
\begin{enumerate}[resume]
  \item Demuéstrese la identidad, válida para todos los complejos $a, b, c, d$:
        \[
        (a - b)(c - d) + (a - d)(b - c) = (a - c)(b - d) .
        \]
  \item Dedúzcase la \emph{desigualdad de
        Ptolomeo}\index{desigualdad de Ptolomeo}: para cuatro puntos
        cualesquiera $A, B, C, D$,
        \[
        AC \cdot BD \;\leq\; AB \cdot CD + AD \cdot BC ,
        \]
        con igualdad si y solo si $(a-b)(c-d)$ y $(a-d)(b-c)$ están
        en una misma semirrecta desde $0$.
  \item Pruébese que, para $\theta, \varphi \in \R$,
        $\abs{\eu^{\iu\theta} - \eu^{\iu\varphi}} =
        2\,\abs{\sin\frac{\theta - \varphi}2}$.
  \item (Teorema de Van Schooten\index{teorema de Van Schooten}) Sea
        $(a, b, c) = (1, j, j^2)$ --- por la parte II esto no supone
        pérdida de generalidad entre los triángulos equiláteros
        directos --- y sea $p = \eu^{\iu\theta}$ con $\theta \in
        \intoo{2\pi/3}{4\pi/3}$, un punto de la circunferencia
        circunscrita en el arco $BC$ que no contiene a $A$. Demuéstrese
        \[
        PA = PB + PC .
        \]
        \emph{(Exprésense las tres distancias con la pregunta 20 y
        úsese una fórmula de transformación de suma en producto.)}
  \item Verifíquese en $p = -1$ que esto es exactamente el caso de
        igualdad de la desigualdad de Ptolomeo para el orden cíclico
        $A, B, P, C$: calcúlense $(a - b)(p - c)$ y $(a - c)(b - p)$ y
        compruébese que su cociente es un real positivo.
\end{enumerate}

\noindent\textbf{Parte V --- Síntesis.}
\begin{enumerate}[resume]
  \item Calcúlese el triángulo de Napoleón exterior del propio
        triángulo equilátero $(1, j, j^2)$ y descríbase el resultado
        geométricamente.
  \item ¿Dónde ha usado exactamente el problema: (i) la multiplicación
        como rotación; (ii) la estructura de grupo de las semejanzas y
        el invariante de forma; (iii) la factorización por el ángulo
        mitad del~\cref{met:b1:complex:trig}? Una frase para cada uno.
  \item En un párrafo breve, formúlese la moraleja del problema: qué
        aporta el diccionario entre la geometría del plano y el álgebra
        de $\C$, qué cuesta y cuál de los dos pasos de la demostración
        --- la identidad $\alpha j + \beta = 0$ o el dibujo clásico ---
        \emph{explica} mejor, en su opinión, el teorema de Napoleón.
        Menciónese un lugar donde el diccionario reaparecerá en forma
        matricial más adelante en este volumen.
\end{enumerate}
\end{problem}
