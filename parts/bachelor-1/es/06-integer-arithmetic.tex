\chapter{Aritmética de los enteros}\label{ch:b1:arith}

La aritmética --- el estudio de la \omterm{def:b1:arith:divides}{divisibilidad} en $\Z$ --- se empezó
en el volumen anterior. Este capítulo la reconstruye por completo a
partir de la división euclídea y con demostraciones completas:
\omterm{thm:b1:arith:gcd}{máximo común divisor} y \omterm{met:b1:arith:euclid}{algoritmo de Euclides}, identidad de Bézout y
lema de Gauss, \omterm{thm:b1:arith:fta}{factorización en primos} y el cálculo de \omterm{def:b1:arith:congruence}{congruencias}
hasta el pequeño teorema de Fermat. Más allá de su encanto propio, esta
materia es el modelo que el~\cref{ch:b1:poly} imita para los polinomios.

\section{Divisibilidad y división euclídea}

\begin{definition}[Divisibilidad]\label{def:b1:arith:divides}
Para $a, b \in \Z$, se dice que $b$ \emph{divide} a $a$ (y se escribe
$b \mid a$)\index{divisibilidad} cuando $a = bq$ para algún
$q \in \Z$. Consecuencias básicas: si $b \mid a$ y $b \mid a'$,
entonces $b \mid (ua + va')$ para todos $u, v \in \Z$; si $b \mid a$ y
$a \neq 0$, entonces $\abs b \leq \abs a$; y $a \mid b$ junto con
$b \mid a$ obligan a $b = \pm a$.
\end{definition}

\begin{theorem}[División euclídea]\label{thm:b1:arith:division}
Para todos $a \in \Z$ y $b \in \N^*$ existe exactamente un par
$(q, r) \in \Z \times \N$ con
\[
a = bq + r, \qquad 0 \leq r < b .
\]
\end{theorem}

\begin{proof}
\emph{Existencia.} El \omterm{def:b1:logic:sets}{conjunto} $A = \{a - bk : k \in \Z\} \cap \N$ es
un subconjunto no vacío de $\N$ (tómese $k = -\abs a$: $a + b\abs a
\geq a + \abs a \geq 0$). Sea $r = a - bq$ su elemento mínimo. Si
$r \geq b$, entonces $r - b = a - b(q+1)$ sería un elemento menor de
$A$: contradicción. Luego $0 \leq r < b$.

\emph{Unicidad.} Si $bq + r = bq' + r'$ con $0 \leq r, r' < b$,
entonces $b(q - q') = r' - r$ y $\abs{r' - r} < b$: el múltiplo de $b$
del miembro izquierdo tiene que ser $0$, luego $q = q'$ y $r = r'$.
\end{proof}

\begin{example}[Numeración posicional por divisiones sucesivas]\label{ex:b1:arith:base7}
Escríbase $2026$ en base $7$. Divídase repetidamente por $7$,
guardando los restos:
\[
2026 = 7 \times 289 + 3, \quad
289 = 7 \times 41 + 2, \quad
41 = 7 \times 5 + 6, \quad
5 = 7 \times 0 + 5 .
\]
Leyendo los restos del último al primero: $2026 = (5\,6\,2\,3)_7$.
Comprobación: $5 \times 343 + 6 \times 49 + 2 \times 7 + 3 = 1715 +
294 + 14 + 3 = 2026$. La unicidad de la división euclídea es
exactamente lo que hace que cada cifra quede \emph{forzada}: en cada
paso, el resto es el único entero de $\intint06$ congruente con el
valor actual \omterm{def:b1:complex:field}{módulo} $7$, así que la escritura en base $7$ es única ---
hecho que se usa en silencio siempre que el problema del fin de semana
manipula «las cifras de $n$ en base $p$».
\end{example}

\section{Máximo común divisor}

\begin{theorem}[Subgrupos de $\Z$; existencia del mcd]\label{thm:b1:arith:gcd}
\begin{enumerate}
  \item Todo subgrupo de $(\Z, +)$ es de la forma $n\Z = \{nk : k
        \in \Z\}$ para un único $n \in \N$.
  \item Para $a, b \in \Z$ no ambos nulos, el \omterm{def:b1:logic:sets}{conjunto} $a\Z + b\Z =
        \{au + bv : u, v \in \Z\}$ es un subgrupo de $\Z$ y, por
        tanto, es igual a $d\,\Z$ para un único $d \in \N^*$. Ese $d$
        es el \emph{máximo común divisor}\index{máximo común divisor}
        $\gcd(a, b)$: \omterm{def:b1:arith:divides}{divide} a $a$ y a $b$, y todo divisor común de
        $a$ y $b$ \omterm{def:b1:arith:divides}{divide} a $d$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Sea $H \subseteq \Z$ un subgrupo (no vacío y estable por resta; la
definición formal está en el~\cref{ch:b1:structures}, y aquí solo se
usan esas dos propiedades). Si $H = \{0\}$, tómese $n = 0$. En caso
contrario, $H$ contiene un elemento no nulo y su opuesto, luego un
menor elemento estrictamente positivo $n$. Entonces $n\Z \subseteq H$.
Para $x \in H$, escríbase $x = nq + r$ con $0 \leq r < n$
(\cref{thm:b1:arith:division}); $r = x - nq \in H$, y la minimalidad de
$n$ obliga a $r = 0$: $x \in n\Z$. Unicidad: $n$ es el menor elemento
positivo de $n\Z$.

(2) $a\Z + b\Z$ contiene $0$ y es estable por resta, luego es $d\Z$ con
$d \geq 1$ (contiene $a$ o $b$, no nulos). Como $a, b \in d\Z$, $d$
\omterm{def:b1:arith:divides}{divide} a los dos. Y si $c$ \omterm{def:b1:arith:divides}{divide} a $a$ y a $b$, entonces $c$
\omterm{def:b1:arith:divides}{divide} a todo $au + bv$ --- en particular $c \mid d$, pues $d \in a\Z
+ b\Z$. Esta es la propiedad anunciada (e implica $\abs c \leq d$, de
modo que $d$ merece el nombre de \emph{máximo} común divisor).
\end{proof}

\begin{corollary}[Identidad de Bézout]\label{cor:b1:arith:bezout}
Para $a, b$ no ambos nulos existen $u, v \in \Z$ con
\[
au + bv = \gcd(a, b) .
\]
En particular ($\gcd(a,b) = 1$, el caso
\emph{coprimo}\index{coprimos}): $a$ y $b$ son \omterm{cor:b1:arith:bezout}{coprimos} si y solo si
$au + bv = 1$ tiene solución.
\end{corollary}

\begin{proof}
$\gcd(a,b) = d \in d\Z = a\Z + b\Z$. Para la equivalencia: si
$\gcd(a,b) = 1$, Bézout da la solución; recíprocamente, $au + bv = 1$
obliga a todo divisor común de $a, b$ a dividir a $1$.
\end{proof}

\begin{method}[Algoritmo de Euclides, extendido]\label{met:b1:arith:euclid}
Para calcular $\gcd(a, b)$ ($a > b > 0$): divídase $a = bq + r$;
entonces $\gcd(a, b) = \gcd(b, r)$ (los divisores comunes de $(a,b)$ y
los de $(b,r)$ coinciden, pues $r = a - bq$); itérese hasta que el
resto sea $0$; el último resto no nulo es el
mcd\index{algoritmo de Euclides}. Recorrer las divisiones hacia atrás
(o arrastrar los coeficientes al bajar) produce un par de Bézout
$(u, v)$.
\end{method}

\begin{example}\label{ex:b1:arith:euclid}
$\gcd(120, 23)$: $120 = 5 \times 23 + 5$; $23 = 4 \times 5 + 3$;
$5 = 1\times 3 + 2$; $3 = 1 \times 2 + 1$; $2 = 2 \times 1 + 0$. Luego
$\gcd = 1$. Hacia atrás:
\begin{align*}
1 &= 3 - 2 = 3 - (5 - 3) = 2\times 3 - 5 = 2(23 - 4\times 5) - 5 \\
&= 2 \times 23 - 9 \times 5 = 2\times 23 - 9(120 - 5\times 23)
= 47 \times 23 - 9 \times 120 .
\end{align*}
Comprobación: $47 \times 23 = 1081$, $9 \times 120 = 1080$.
\end{example}

\begin{theorem}[Lema de Gauss y consecuencias]\label{thm:b1:arith:gauss}
Sean $a, b, c \in \Z$.
\begin{enumerate}
  \item (Lema de Gauss\index{lema de Gauss}) Si $a \mid bc$ y
        $\gcd(a, b) = 1$, entonces $a \mid c$.
  \item Si $a \mid c$, $b \mid c$ y $\gcd(a,b) = 1$, entonces
        $ab \mid c$.
  \item Si $\gcd(a, b) = \gcd(a, c) = 1$, entonces $\gcd(a, bc) = 1$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Bézout: $au + bv = 1$. Multiplíquese por $c$: $acu + bcv = c$. Los
dos términos son divisibles por $a$ (el segundo porque $a \mid bc$),
luego $a \mid c$.

(2) Escríbase $c = aq$; de $b \mid aq$ y $\gcd(a, b) = 1$, el punto (1)
da $b \mid q$, luego $ab \mid aq = c$.

(3) $au + bv = 1$ y $au' + cv' = 1$. Multiplíquense las dos
relaciones:
\[
1 = (au + bv)(au' + cv')
= a\,\bigl(auu' + ucv' + u'bv\bigr) + bc\,(vv') ,
\]
una relación de Bézout entre $a$ y $bc$: por
el~\cref{cor:b1:arith:bezout}, $\gcd(a, bc) = 1$.
\end{proof}

\begin{example}[Resolución de una ecuación diofántica lineal]\label{ex:b1:arith:diophantine}
Hállense todos los $(x, y) \in \Z^2$ con $6x + 10y = 4$. Primero, el
\emph{test de existencia}: $\gcd(6, 10) = 2$ \omterm{def:b1:arith:divides}{divide} a $4$, luego hay
soluciones (si el mcd no dividiese al miembro derecho, el izquierdo
sería siempre múltiplo suyo y no habría ninguna). Divídase todo:
$3x + 5y = 2$. Se ve una solución particular: $(x_0, y_0) = (-1, 1)$.
Para la general, réstese: $3(x + 1) = -5(y - 1)$, luego
$3 \mid 5(y-1)$, y el lema de Gauss ($\gcd(3,5) = 1$) da
$3 \mid y - 1$: $y = 1 - 3k$ y después $x = -1 + 5k$. Recíprocamente,
todo par así sirve:
\[
(x, y) = (-1 + 5k,\ 1 - 3k), \qquad k \in \Z .
\]
El patrón es general: una solución particular más los múltiplos
enteros de $\bigl(\frac b{\gcd}, -\frac a{\gcd}\bigr)$ --- la misma
estructura de «particular más homogénea» que en
el~\cref{ch:b1:diffeq}, con el lema de Gauss haciendo el papel de la
unicidad.
\end{example}

\begin{definition}[Mínimo común múltiplo]\label{def:b1:arith:lcm}
$\operatorname{lcm}(a, b)$\index{mínimo común múltiplo} es el generador
en $\N$ del subgrupo $a\Z \cap b\Z$: es un múltiplo común de $a$ y $b$
que \omterm{def:b1:arith:divides}{divide} a todo múltiplo común y, para $a, b \in \N^*$,
\[
\gcd(a,b) \times \operatorname{lcm}(a,b) = ab
\qquad (\text{demostración en el } \cref{exo:b1:arith:5}).
\]
\end{definition}

\begin{example}[Los problemas de coincidencia son problemas de mcm]\label{ex:b1:arith:gears}
Dos ruedas dentadas engranadas tienen $84$ y $36$ dientes. ¿Tras
cuántos dientes de movimiento común vuelven las dos a su posición
inicial? La configuración se repite cuando el número de dientes
transcurridos es un múltiplo común de $84$ y $36$; la primera vez es
\[
\operatorname{lcm}(84, 36) = \frac{84 \times 36}{\gcd(84, 36)}
= \frac{3024}{12} = 252
\]
dientes --- es decir, $3$ vueltas de la rueda grande y $7$ de la
pequeña ($252/84$ y $252/36$). Obsérvese la vía práctica:
\emph{calcúlese primero el mcd} (Euclides: $84 = 2\times36 + 12$,
$36 = 3\times12$) y divídase después --- nunca se construya el mcm
enumerando múltiplos. Toda cuestión de coincidencias periódicas
(engranajes, alineaciones planetarias, decimales periódicos que se
encuentran) se reduce a este único cálculo.
\end{example}

\section{Números primos}

\begin{definition}\label{def:b1:arith:prime}
Un entero $p \geq 2$ es \emph{primo}\index{número primo} cuando sus
únicos divisores positivos son $1$ y $p$. Para $p$ primo y $a \in \Z$:
o bien $p \mid a$, o bien $\gcd(p, a) = 1$. En consecuencia
(\cref{thm:b1:arith:gauss}), se cumple el \emph{lema de Euclides}: si
$p \mid ab$, entonces $p \mid a$ o $p \mid b$.
\end{definition}

\begin{remark}[Probar la primalidad por divisiones sucesivas]\label{rem:b1:arith:trialdivision}
Si $n = ab$ con $2 \leq a \leq b$, entonces $a^2 \leq ab = n$, luego
$a \leq \sqrt n$: un $n$ compuesto tiene siempre un divisor \omterm{def:b1:arith:prime}{primo}
$\leq \sqrt n$. Por tanto, para comprobar si $n$ es \omterm{def:b1:arith:prime}{primo} basta
probar los \omterm{def:b1:arith:prime}{primos} hasta $\sqrt n$. Para $n = 271$: $\sqrt{271} < 17$,
y $271$ no es divisible por ninguno de $2, 3, 5, 7, 11, 13$ (es impar,
su suma de cifras es $10$, no acaba en $0$ ni en $5$, y
$271 = 7\cdot38 + 5 = 11\cdot24 + 7 = 13\cdot20 + 11$): \omterm{def:b1:arith:prime}{primo}, tras
seis divisiones en lugar de doscientas. La barrera $\sqrt n$ es un
umbral real: cruzarla con eficacia para números de cien cifras exige
los tests de primalidad modernos nacidos
del~\cref{thm:b1:arith:fermat}.
\end{remark}

\begin{theorem}[Euclides]\label{thm:b1:arith:euclidprimes}
Hay infinitos \omterm{def:b1:arith:prime}{primos}.
\end{theorem}

\begin{proof}
Todo entero $n \geq 2$ tiene un divisor \omterm{def:b1:arith:prime}{primo}: su menor divisor
$\geq 2$ es \omterm{def:b1:arith:prime}{primo} (una factorización propia suya produciría un divisor
menor de $n$). Supóngase ahora que $p_1, \dots, p_k$ fuesen todos los
\omterm{def:b1:arith:prime}{primos} y sea $N = p_1 p_2 \cdots p_k + 1 \geq 2$. Algún \omterm{def:b1:arith:prime}{primo} $p_i$
\omterm{def:b1:arith:divides}{divide} a $N$; pero $p_i$ \omterm{def:b1:arith:divides}{divide} también a $N - 1 = p_1\cdots p_k$,
luego $p_i \mid 1$ --- absurdo.
\end{proof}

\begin{theorem}[Teorema fundamental de la aritmética]\label{thm:b1:arith:fta}
Todo entero $n \geq 2$ es un producto de \omterm{def:b1:arith:prime}{primos}, y la factorización
\[
n = p_1^{\alpha_1} p_2^{\alpha_2} \cdots p_k^{\alpha_k}
\qquad (p_1 < p_2 < \dots < p_k \text{ primos},\ \alpha_i \in \N^*)
\]
es única.\index{factorización en primos}
\end{theorem}

\begin{proof}
\emph{Existencia}, por inducción fuerte
(\cref{thm:b1:logic:induction}): $n = 2$ es \omterm{def:b1:arith:prime}{primo}; para $n > 2$, o
bien $n$ es \omterm{def:b1:arith:prime}{primo}, o bien $n = ab$ con $2 \leq a, b < n$, y la
hipótesis de inducción factoriza $a$ y $b$.

\emph{Unicidad.} Supóngase $p_1 \cdots p_r = q_1 \cdots q_s$
(\omterm{def:b1:arith:prime}{primos} enumerados con repetición, digamos $r \leq s$) y hágase
inducción sobre $r$. Si $r = 0$, el miembro izquierdo es $1$, lo que
obliga a $s = 0$ (un producto no vacío de \omterm{def:b1:arith:prime}{primos} supera a $1$). Para
$r \geq 1$: el \omterm{def:b1:arith:prime}{primo} $p_1$ \omterm{def:b1:arith:divides}{divide} a $q_1(q_2\cdots q_s)$, luego, por
el lema de Euclides, $p_1 \mid q_1$ o $p_1 \mid q_2\cdots q_s$;
iterando, $p_1$ \omterm{def:b1:arith:divides}{divide} a algún $q_j$. Pero $q_j$ es \omterm{def:b1:arith:prime}{primo} y
$p_1 \geq 2$: necesariamente $p_1 = q_j$. Cancélese este factor común
(es legítimo: $\Z$ es un dominio de integridad) para obtener
\[
p_2 \cdots p_r = q_1 \cdots \widehat{q_j} \cdots q_s
\]
(el sombrero marca la omisión), una igualdad de productos más cortos;
la hipótesis de inducción dice que las dos listas $p_2, \dots, p_r$ y
$q_1, \dots, \widehat{q_j}, \dots, q_s$ coinciden salvo el orden, y
por tanto también lo hacían las originales. La forma con exponentes
agrupa los \omterm{def:b1:arith:prime}{primos} iguales.
\end{proof}

\begin{proposition}[Valuaciones]\label{prop:b1:arith:valuation}
Para $p$ \omterm{def:b1:arith:prime}{primo} y $n \in \N^*$, escríbase $v_p(n)$\index{valuacion
p-adica@valuación $p$-ádica} para el exponente de $p$ en la
factorización de $n$ (con $v_p(n) = 0$ si $p \nmid n$). Entonces
\[
v_p(mn) = v_p(m) + v_p(n),
\qquad
m \mid n \iff \forall p,\ v_p(m) \leq v_p(n),
\]
\[
v_p\bigl(\gcd(m,n)\bigr) = \min\bigl(v_p(m), v_p(n)\bigr),
\qquad
v_p\bigl(\operatorname{lcm}(m,n)\bigr) = \max\bigl(v_p(m),
v_p(n)\bigr).
\]
\end{proposition}

\begin{proof}
La primera identidad se cumple porque las factorizaciones se
multiplican y la de $mn$ es única. Si $m \mid n$, escríbase $n = mq$ y
aplíquese. Recíprocamente, si todos los $v_p(m) \leq v_p(n)$, el
entero $q = \prod_p p^{\,v_p(n) - v_p(m)}$ cumple $mq = n$. Fórmula
del mcd: el entero $d = \prod p^{\min}$ \omterm{def:b1:arith:divides}{divide} a los dos por el
criterio, y todo divisor común $c$ cumple $v_p(c) \leq \min$ para todo
$p$, luego $c \mid d$; el mismo razonamiento vale para el mcm con el
$\max$.
\end{proof}

\begin{example}[Cuadrados y cubos a través de las valuaciones]\label{ex:b1:arith:squarecube}
Un entero $n \geq 1$ es un cuadrado perfecto si y solo si todos los
$v_p(n)$ son pares (si $n = m^2$, entonces $v_p(n) = 2v_p(m)$;
recíprocamente, divídase por la mitad cada exponente). Lo análogo vale
para los cubos con múltiplos de $3$. Así, $21168 = 2^4 \times 3^3
\times 7^2$ no es un cuadrado ($v_3 = 3$ es impar) ni un cubo
($v_2 = 4$); el menor entero positivo $m$ tal que $21168\,m$ \emph{sí}
sea un cubo se halla completando cada exponente hasta el siguiente
múltiplo de $3$:
\[
m = 2^{6-4} \times 3^{3-3} \times 7^{3-2} = 2^2 \times 7 = 28,
\qquad
21168 \times 28 = 2^6\,3^3\,7^3 = (2^2 \times 3 \times 7)^3
= 84^3 .
\]
La idea clave: las cuestiones multiplicativas (cuadrados, cubos,
divisores, mcd, mcm) se vuelven cuestiones \emph{coordenada a
coordenada} sobre los vectores de exponentes $(v_2, v_3, v_5, \dots)$
--- la factorización única es el \omterm{def:b1:logic:statement}{enunciado} de que esas coordenadas
existen y están bien definidas.
\end{example}

\section{Congruencias}

\begin{definition}\label{def:b1:arith:congruence}
Para $n \in \N^*$: $a \equiv b \pmod n$\index{congruencia} cuando
$n \mid a - b$. Es una \omterm{def:b1:logic:equiv}{relación de equivalencia} compatible con la
suma y el producto: si $a \equiv b$ y $a' \equiv b'$ (mód $n$),
entonces $a + a' \equiv b + b'$, $aa' \equiv bb'$ y $a^k \equiv b^k$
para $k \in \N$.
\end{definition}

\begin{example}[La prueba del nueve]\label{ex:b1:arith:castingnines}
La compatibilidad con $+$ y $\times$ es un método de comprobación tan
viejo como el comercio. Como $10 \equiv 1 \pmod 9$, todo entero es
congruente \omterm{def:b1:complex:field}{módulo} $9$ con la suma de sus cifras (se demuestra en
el~\cref{exo:b1:arith:2}). Para comprobar la afirmación
$1234 \times 567 = 699\,678$: las sumas de cifras dan $1234 \equiv 1$
y $567 \equiv 18 \equiv 0 \pmod 9$, luego el producto debe ser
$\equiv 1 \times 0 = 0$; y, en efecto, $6 + 9 + 9 + 6 + 7 + 8 = 45
\equiv 0$. La comprobación pasa (y el producto es de hecho correcto).
Si alguien hubiese dado $699\,478$, la suma de cifras $43 \equiv 7
\not\equiv 0$ lo delataría al instante. El test es de un solo sentido
--- caza el error salvo que el propio error sea múltiplo de $9$ ---,
que es exactamente la lección de los seudoprimos
del~\cref{ex:b1:arith:pseudoprime} en miniatura: las comprobaciones
por \omterm{def:b1:arith:congruence}{congruencias} refutan, no certifican.
\end{example}

\begin{proposition}[Invertibilidad módulo $n$]\label{prop:b1:arith:invmod}
$a$ es \emph{invertible \omterm{def:b1:complex:field}{módulo} $n$} (es decir, $ab \equiv 1 \pmod n$
para algún $b$) si y solo si $\gcd(a, n) = 1$. El inverso es entonces
único \omterm{def:b1:complex:field}{módulo} $n$ y se calcula con el \omterm{met:b1:arith:euclid}{algoritmo de Euclides} extendido.
\end{proposition}

\begin{proof}
$ab \equiv 1 \pmod n$ significa $ab + nk = 1$ para algún $k$: una
relación de Bézout, que existe si y solo si $\gcd(a,n) = 1$
(\cref{cor:b1:arith:bezout}). Unicidad: si $ab \equiv ab' \equiv 1$,
entonces $b \equiv b(ab') = (ab)b' \equiv b' \pmod n$.
\end{proof}

\begin{example}[Invertir $7$ módulo $26$]\label{ex:b1:arith:inv7mod26}
Como $\gcd(7, 26) = 1$, la clase de $7$ es invertible \omterm{def:b1:complex:field}{módulo} $26$.
Euclides extendido:
\[
26 = 3 \times 7 + 5, \qquad
7 = 1 \times 5 + 2, \qquad
5 = 2 \times 2 + 1 ,
\]
y después hacia atrás:
\[
1 = 5 - 2 \times 2 = 5 - 2(7 - 5) = 3 \times 5 - 2 \times 7
= 3(26 - 3 \times 7) - 2 \times 7 = 3 \times 26 - 11 \times 7 .
\]
Por tanto $7 \times (-11) \equiv 1 \pmod{26}$, es decir,
$7^{-1} \equiv -11 \equiv 15 \pmod{26}$; comprobación:
$7 \times 15 = 105 = 4 \times 26 + 1$. Con el inverso en la mano,
cualquier \omterm{def:b1:arith:congruence}{congruencia} $7x \equiv c \pmod{26}$ se resuelve con una
multiplicación: $x \equiv 15c$. Esta inversión mecánica es el caballo
de batalla de la aritmética modular --- y de los protocolos de clave
pública mencionados en el~\cref{rem:b1:arith:whereused}, donde los
\omterm{def:b1:complex:field}{módulos} tienen cientos de cifras pero el algoritmo es exactamente este.
\end{example}

\begin{example}[Cuando el coeficiente no es invertible]\label{ex:b1:arith:noninvertible}
Resuélvase $12x \equiv 8 \pmod{20}$. Aquí $\gcd(12, 20) = 4$, así que
$12$ no es invertible \omterm{def:b1:complex:field}{módulo} $20$ --- pero la ecuación sigue siendo
tratable. La \omterm{def:b1:arith:congruence}{congruencia} dice que $20 \mid 12x - 8$; dividiendo toda
la relación por $4$ (divisor de los tres ingredientes), equivale a
$5 \mid 3x - 2$, es decir,
\[
3x \equiv 2 \pmod 5 .
\]
Ahora $\gcd(3, 5) = 1$ y $3^{-1} \equiv 2 \pmod 5$ ($3 \times 2 = 6
\equiv 1$), luego $x \equiv 4 \pmod 5$: las soluciones son
$x \equiv 4, 9, 14, 19 \pmod{20}$ --- \emph{cuatro} clases \omterm{def:b1:complex:field}{módulo}
$20$, tantas como el mcd. (Si el miembro derecho no hubiese sido
divisible por $4$, por ejemplo $12x \equiv 6 \pmod{20}$, no habría
ninguna solución: el miembro izquierdo es siempre $\equiv 0 \pmod 4$.)
Forma general: $ax \equiv b \pmod n$ tiene solución si y solo si
$\gcd(a, n) \mid b$, y entonces tiene exactamente $\gcd(a, n)$ clases
de soluciones --- divídase todo por el mcd e inviértase.
\end{example}

\begin{theorem}[Pequeño teorema de Fermat]\label{thm:b1:arith:fermat}
Sea $p$ \omterm{def:b1:arith:prime}{primo}. Para todo $a \in \Z$:
\[
a^p \equiv a \pmod p,
\]
y, si $p \nmid a$, entonces $a^{p-1} \equiv 1 \pmod
p$.\index{pequeño teorema de Fermat}
\end{theorem}

\begin{proof}
Primero, para $1 \leq k \leq p - 1$, el \omterm{def:b1:counting:objects}{coeficiente binomial}
$\binom pk = \frac{p!}{k!(p-k)!}$ es divisible por $p$: en efecto,
$k!\,(p-k)!\,\binom pk = p!$ y $p$ \omterm{def:b1:arith:divides}{divide} a $p!$ pero es \omterm{cor:b1:arith:bezout}{coprimo} con
$k!(p-k)!$ (todos sus factores son $< p$), luego el lema de Gauss da
$p \mid \binom pk$.

Pruébese ahora $a^p \equiv a$ para $a \in \N$ por inducción. Cierto
para $a = 0$. Si $a^p \equiv a$, entonces, por el teorema del binomio,
\[
(a+1)^p = \sum_{k=0}^{p} \binom pk a^k
\equiv a^p + 1 \equiv a + 1 \pmod p,
\]
anulándose \omterm{def:b1:complex:field}{módulo} $p$ todos los términos intermedios. Para $a < 0$,
aplíquese el resultado a $-a$ y sepárese $p = 2$ (donde $x \equiv -x$)
de $p$ impar (donde $(-a)^p = -a^p$). Por último, si $p \nmid a$,
multiplíquese $a^p \equiv a$ por un inverso de $a$ \omterm{def:b1:complex:field}{módulo} $p$
(\cref{prop:b1:arith:invmod}).
\end{proof}

\begin{example}[El recíproco de Fermat falla: $341$]\label{ex:b1:arith:pseudoprime}
El pequeño teorema de Fermat da un test barato de
\emph{composición}: si $a^{n-1} \not\equiv 1 \pmod n$ para algún $a$
\omterm{cor:b1:arith:bezout}{coprimo} con $n$, entonces $n$ no es \omterm{def:b1:arith:prime}{primo}. ¿Podría el test
certificar también la primalidad? No: tómese $n = 341 = 11 \times 31$,
compuesto, y $a = 2$. Como $2^{10} = 1024 = 3 \times 341 + 1$,
\[
2^{10} \equiv 1 \pmod{341}
\qquad\Longrightarrow\qquad
2^{340} = \bigl(2^{10}\bigr)^{34} \equiv 1 \pmod{341} :
\]
el compuesto $341$ pasa el test de Fermat en base $2$ (es el menor
\emph{seudoprimo} de esta clase). La base $3$ lo desenmascara
($3^{340} \not\equiv 1$), y por eso los tests de primalidad prácticos
se ejecutan en varias bases, además de con refinamientos --- las
versiones industriales de esta idea son las que certifican los grandes
\omterm{def:b1:arith:prime}{primos} del~\cref{rem:b1:arith:whereused}. Moraleja: una implicación
y su recíproca llevan vidas separadas
(\cref{rem:b1:logic:pitfalls}), incluso tratándose de teoremas.
\end{example}

\begin{example}[Cálculos prácticos con congruencias]\label{ex:b1:arith:congruences}
¿Cuál es el resto de $7^{2026}$ \omterm{def:b1:complex:field}{módulo} $11$? Por Fermat,
$7^{10} \equiv 1 \pmod{11}$. Como $2026 = 10 \times 202 + 6$:
\[
7^{2026} \equiv 7^6 = (7^2)^3 = 49^3 \equiv 5^3 = 125 \equiv 4
\pmod{11}.
\]
El resto es $4$. La estrategia: redúzcase el exponente \omterm{def:b1:complex:field}{módulo} el orden
que proporciona Fermat y redúzcanse las potencias intermedias en cada
paso.
\end{example}

\begin{remark}[Errores frecuentes en aritmética]\label{rem:b1:arith:pitfalls}
\begin{enumerate}
  \item \emph{Dividir una \omterm{def:b1:arith:congruence}{congruencia}.} De $ac \equiv bc \pmod n$
        \emph{no} se puede concluir $a \equiv b$ salvo si
        $\gcd(c, n) = 1$: $6 \equiv 2 \pmod 4$, pero
        $3 \not\equiv 1 \pmod 4$. La regla general correcta
        \omterm{def:b1:arith:divides}{divide} también el \omterm{def:b1:complex:field}{módulo}: $ac \equiv bc \pmod n \iff a
        \equiv b \pmod{n/\gcd(c,n)}$.
  \item \emph{Usar mal el lema de Euclides.} $a \mid bc$ implica
        $a \mid b$ o $a \mid c$ solo para $a$ \emph{\omterm{def:b1:arith:prime}{primo}} (o
        \omterm{cor:b1:arith:bezout}{coprimo} con uno de los factores): $6 \mid 4 \times 9$ y, sin
        embargo, $6$ no \omterm{def:b1:arith:divides}{divide} a ninguno de los dos.
  \item \emph{\omterm{cor:b1:arith:bezout}{Coprimo} es una relación, no una propiedad.} «$8$ y
        $9$ son \omterm{cor:b1:arith:bezout}{coprimos}» es cierto aunque ninguno sea \omterm{def:b1:arith:prime}{primo}; «\omterm{cor:b1:arith:bezout}{coprimos}
        dos a dos» es más fuerte que «\omterm{cor:b1:arith:bezout}{coprimos} en \omterm{def:b1:logic:sets}{conjunto}»
        ($\gcd(6, 10, 15) = 1$, pero ningún par es \omterm{cor:b1:arith:bezout}{coprimo}).
  \item \emph{Los exponentes no viven \omterm{def:b1:complex:field}{módulo} $n$.} En $a^k \bmod n$,
        el exponente solo se puede reducir \omterm{def:b1:complex:field}{módulo} el \emph{orden} de
        $a$ (por ejemplo, $p - 1$ cuando se aplica Fermat), nunca
        \omterm{def:b1:complex:field}{módulo} $n$: $2^{10} \bmod 11$ vale $1$, no
        $2^{10 \bmod 11} = 2^{10}$ --- la reducción que sí funciona es
        la que hace el~\cref{ex:b1:arith:congruences}.
\end{enumerate}
\end{remark}

\begin{remark}[Dónde se usa este capítulo]\label{rem:b1:arith:whereused}
Este capítulo es tanto una plantilla como una caja de herramientas.
Toda la cadena --- división euclídea, mcd, Bézout, Gauss,
factorización única --- se repite literalmente para los polinomios en
el~\cref{ch:b1:poly}, donde el «grado» hace el papel del valor
absoluto; comparar los dos capítulos en paralelo es la mejor manera de
entender ambos. El cálculo de \omterm{def:b1:arith:congruence}{congruencias} se convierte en el anillo
$\Z/n\Z$ en el~\cref{ch:b1:structures}, cuyos elementos invertibles
(\cref{prop:b1:arith:invmod}) forman el primer ejemplo no trivial de
grupo de unidades. Las valuaciones vuelven en el problema del fin de
semana (fórmula de Legendre) y sostienen las demostraciones de
irracionalidad del~\cref{ch:b1:reals}. Más allá de este volumen, la
inversión de Bézout \omterm{def:b1:complex:field}{módulo} $n$ es el motor de la criptografía de clave
pública, y el pequeño teorema de Fermat es el abuelo de los tests de
primalidad que certifican los grandes \omterm{def:b1:arith:prime}{primos} que allí se usan.
\end{remark}

\begin{remark}[Interludio: $\Z$ como plantilla]\label{rem:b1:arith:interlude}
Tómese distancia de los teoremas concretos y obsérvese la arquitectura
del capítulo: una herramienta (la división euclídea) produjo una
clasificación (los subgrupos $n\Z$), que produjo un teorema de
existencia (mcd, Bézout), que produjo un cálculo de
\omterm{def:b1:arith:divides}{divisibilidad} (Gauss), que produjo la factorización única --- cada
piso apoyado únicamente en el inmediatamente inferior. El mismo
edificio se levantará dos veces más en este volumen con plantas bajas
distintas: en el~\cref{ch:b1:poly}, donde dividir por el grado
sustituye a dividir por el tamaño y todo lo de arriba se repite
\emph{literalmente}; y, en miniatura, dentro de cada $\Z/n\Z$
del~\cref{ch:b1:structures}, donde las cuestiones de invertibilidad
(la~\cref{prop:b1:arith:invmod} de este capítulo) se vuelven
\omterm{def:b1:logic:statement}{enunciados} estructurales sobre anillos y cuerpos. Reconocer un
argumento como «el argumento de $\Z$, trasplantado» es la forma más
rápida de aprender esos capítulos --- y el primer sabor del hábito
central del álgebra: demostrar teoremas sobre \emph{axiomas} y no sobre
objetos.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:arith:1}
Calcúlese $\gcd(1\,001, 777)$ con el \omterm{met:b1:arith:euclid}{algoritmo de Euclides}, y un par
de Bézout para él.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:arith:2}
Demuéstrense los criterios de \omterm{def:b1:arith:divides}{divisibilidad} en base $10$: un entero es
congruente \omterm{def:b1:complex:field}{módulo} $9$ con la suma de sus cifras, y \omterm{def:b1:complex:field}{módulo} $11$ con la
suma alternada de sus cifras. ¿Cuánto vale $123\,456\,789$ \omterm{def:b1:complex:field}{módulo} $9$ y \omterm{def:b1:complex:field}{módulo} $11$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:arith:3}
Resuélvase en $\Z$: $91x \equiv 1 \pmod{237}$ \emph{(Euclides extendido)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:arith:4}
Hállense todos los pares $(x, y) \in \Z^2$ con $17x + 39y = 1$; y
después todos los pares con $17 x + 39 y = 5$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:5}
Demuéstrese que, para $a, b \in \N^*$, $\gcd(a,b) \times
\operatorname{lcm}(a,b) = ab$. \emph{(Úsense las fórmulas de
valuación de la~\cref{prop:b1:arith:valuation} y
$\min(\alpha,\beta) + \max(\alpha,\beta) = \alpha + \beta$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:6}
Sean $a = 2^{10} \times 3^4 \times 5^2$ y $b = 2^6 \times 3^7 \times
7$. Calcúlense $\gcd(a, b)$, $\operatorname{lcm}(a,b)$ y el número de
divisores positivos de $a$. \emph{(Demuéstrese la fórmula del número
de divisores $\prod_i (\alpha_i + 1)$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:7}
Demuéstrese que $\sqrt p$ es irracional para todo \omterm{def:b1:arith:prime}{primo} $p$, usando
valuaciones: compárese $v_p$ en los dos miembros de $p q^2 = r^2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:8}
(Problema chino de los restos) Hállense todos los enteros $x$ con
\[
x \equiv 2 \pmod 7, \qquad x \equiv 5 \pmod{11}.
\]
Demuéstrese de paso que, para $m, n$ \omterm{cor:b1:arith:bezout}{coprimos}, el par de
\omterm{def:b1:arith:congruence}{congruencias} $x \equiv a \ (m)$, $x \equiv b\ (n)$ tiene siempre
solución, única \omterm{def:b1:complex:field}{módulo} $mn$.\index{teorema chino del resto}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:arith:9}
Calcúlese $3^{1000}$ \omterm{def:b1:complex:field}{módulo} $7$, y las dos últimas cifras decimales de
$7^{100}$ \emph{(\omterm{def:b1:complex:field}{módulo} $100 = 4 \times 25$: úsese
el~\cref{exo:b1:arith:8})}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:arith:10}
Para $m, n \in \N^*$, demuéstrese que $\gcd(2^m - 1,\, 2^n - 1) =
2^{\gcd(m,n)} - 1$. \emph{Indicación: véase primero que el resto de
$2^m - 1$ \omterm{def:b1:complex:field}{módulo} $2^n - 1$ es $2^r - 1$, donde $r$ es el resto de $m$
\omterm{def:b1:complex:field}{módulo} $n$; síganse después los pasos del \omterm{met:b1:arith:euclid}{algoritmo de Euclides}.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:arith:11}
(Teorema de Wilson) Sea $p$ un \omterm{def:b1:arith:prime}{primo}. Demuéstrese que
\[
(p-1)! \equiv -1 \pmod p ,
\]
emparejando cada factor de $(p-1)!$ con su inverso \omterm{def:b1:complex:field}{módulo} $p$ e
identificando los factores emparejados consigo mismos (resuélvase
antes $x^2 \equiv 1 \pmod p$). Compruébese el recíproco: si $n \geq 2$
no es \omterm{def:b1:arith:prime}{primo}, entonces $(n-1)! \not\equiv -1 \pmod n$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:arith:12}
(Números de Fermat) Para $n \in \N$, sea $F_n = 2^{2^n} + 1$.
\begin{enumerate}
  \item Demuéstrese que $F_0 F_1 \cdots F_{n-1} = F_n - 2$ para
        $n \geq 1$ (inducción).
  \item Dedúzcase que los números de Fermat son \omterm{cor:b1:arith:bezout}{coprimos} dos a dos.
  \item Dedúzcase una segunda demostración, independiente
        del~\cref{thm:b1:arith:euclidprimes}, de que hay infinitos
        \omterm{def:b1:arith:prime}{primos}.
\end{enumerate}
\end{exercise}

\section{Problema: la fórmula de Legendre y los acarreos de Kummer}

\begin{problem}\label{pb:b1:arith:1}
¿En cuántos ceros termina la escritura decimal de $1000!$ y, más a
fondo, cuál es la potencia exacta de un \omterm{def:b1:arith:prime}{primo} $p$ que \omterm{def:b1:arith:divides}{divide} a $n!$, o
que \omterm{def:b1:arith:divides}{divide} a un \omterm{def:b1:counting:objects}{coeficiente binomial}? Las respuestas completas son dos
joyas de la aritmética elemental: la \emph{fórmula de Legendre}
\index{fórmula de Legendre} $v_p(n!) = \sum_{k\geq1}
\lfloor n/p^k \rfloor$, con su avatar digital $v_p(n!) = \frac{n
- s_p(n)}{p-1}$, y el \emph{teorema de Kummer}\index{teorema de
Kummer}: $v_p\binom{m+n}m$ cuenta los \emph{acarreos} al sumar $m$ y
$n$ en base $p$. Este problema demuestra los dos, los contrasta
numéricamente entre sí y recolecta las consecuencias clásicas ---
ceros finales, paridad del triángulo de Pascal y una primera cota en
la dirección del teorema de los \omterm{def:b1:arith:prime}{números primos}.
En todo el problema, $p$ es un \omterm{def:b1:arith:prime}{primo}, $\floor{x}$ es la parte entera y
$s_p(n)$ denota la suma de las cifras de $n$ escrito en base $p$.

\medskip
\noindent\textbf{Parte I --- Partes enteras, valuaciones y fórmula de
Legendre.}
\begin{enumerate}
  \item Calentamiento: calcúlese $10!$ y léase su número de ceros
        finales; calcúlense $v_2(10!)$ y $v_5(10!)$ directamente a
        partir de la factorización de cada factor $1, 2, \dots, 10$.
  \item Demuéstrese que, para $x \in \R$ y $n \in \N^*$,
        $\bigl\lfloor \lfloor x \rfloor / n \bigr\rfloor =
        \lfloor x/n \rfloor$.
  \item Demuéstrese que $v_p(a + b) \geq \min\bigl(v_p(a),
        v_p(b)\bigr)$ para todos $a, b \in \N^*$, con igualdad
        siempre que $v_p(a) \neq v_p(b)$.
  \item Pruébese que el número de múltiplos de $m$ en $\intint1n$ es
        $\lfloor n/m \rfloor$.
  \item Demuéstrese la \emph{fórmula de Legendre}: para todo
        $n \in \N^*$,
        \[
        v_p(n!) = \sum_{k=1}^{\infty}
        \Bigl\lfloor \frac{n}{p^k} \Bigr\rfloor
        \]
        (una suma finita: los términos se anulan en cuanto
        $p^k > n$). \emph{Cuéntense, para cada $k$, los factores de
        $\intint1n$ divisibles por $p^k$: cada uno aporta exactamente
        una unidad por cada nivel al que llega.}
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La forma digital y los ceros
finales.}
\begin{enumerate}[resume]
  \item Calcúlense $v_5(1000!)$ y $v_2(1000!)$ y conclúyase: ¿en
        cuántos ceros termina $1000!$?
  \item Demuéstrese la forma digital de la fórmula de Legendre:
        escribiendo $n = \sum_i a_i p^i$ en base $p$,
        \[
        v_p(n!) = \frac{n - s_p(n)}{p - 1} .
        \]
  \item Dos consecuencias para $p = 2$: pruébese que $2^n$ nunca
        \omterm{def:b1:arith:divides}{divide} a $n!$, y que $2^{n-1}$ \omterm{def:b1:arith:divides}{divide} a $n!$ exactamente
        cuando $n$ es una potencia de $2$.
  \item Acótese el defecto: pruébese que $\frac n{p-1} - \log_p(n) - 1
        \leq v_p(n!) < \frac n{p-1}$, de modo que
        $\frac{v_p(n!)}{n} \to \frac1{p-1}$: a la larga, se acumula
        una proporción $\frac1{p-1}$ de un factor $p$ por unidad.
  \item Sea $Z(n) = v_5(n!)$ el número de ceros finales de $n!$.
        Pruébese que $Z(n) - Z(n-1) = v_5(n)$, dedúzcase que $Z$ se
        salta por completo el valor $5$ (calcúlense $Z(24)$ y
        $Z(25)$) y demuéstrese que ningún factorial termina en
        exactamente cinco ceros.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El teorema de Kummer.}
\begin{enumerate}[resume]
  \item Demuéstrese que $\lfloor x + y \rfloor - \lfloor x \rfloor -
        \lfloor y \rfloor \in \{0, 1\}$ para todos $x, y \in \R$, y
        dedúzcase de la fórmula de Legendre que
        \[
        v_p\binom{m+n}m
        = \sum_{k\geq1}\Bigl(
        \Bigl\lfloor\frac{m+n}{p^k}\Bigr\rfloor
        - \Bigl\lfloor\frac{m}{p^k}\Bigr\rfloor
        - \Bigl\lfloor\frac{n}{p^k}\Bigr\rfloor\Bigr),
        \]
        una suma de términos iguales cada uno a $0$ o a $1$.
  \item Demuéstrese el \emph{teorema de Kummer}: el $k$-ésimo término
        de esa suma vale $1$ exactamente cuando la suma de $m$ y $n$
        en base $p$ produce un acarreo en la posición $k$; por tanto,
        $v_p\binom{m+n}m$ es el número total de acarreos.
        \emph{(Escríbanse $m = p^km_1 + m_0$ y $n = p^kn_1 + n_0$ con
        $0 \leq m_0, n_0 < p^k$ y examínese $\lfloor (m_0 +
        n_0)/p^k \rfloor$.)}
  \item Dedúzcase que, para $0 < j < p^k$:
        \[
        v_p\binom{p^k}{j} = k - v_p(j) ,
        \]
        contando los acarreos de la suma $j + (p^k - j)$. (En
        particular, $p \mid \binom p j$ para $0 < j < p$: el paso
        clave del~\cref{thm:b1:arith:fermat}, recuperado.)
  \item Demuéstrese que $v_2\binom{2n}n = s_2(n)$. Dedúzcase que el
        \omterm{def:b1:counting:objects}{coeficiente binomial} central es siempre par, y que
        $\binom{2n}n \equiv 2 \pmod 4$ exactamente cuando $n$ es una
        potencia de $2$.
  \item Pruébese, usando la identidad de Vandermonde
        (\cref{exo:b1:counting:7}) y la pregunta 13, que
        $\binom{2p}p \equiv 2 \pmod p$ para todo \omterm{def:b1:arith:prime}{primo} $p$.
  \item Calcúlese $v_3\binom{1000}{500}$ de dos maneras: una por
        Kummer (escríbase $500$ en base $3$ y cuéntense los acarreos
        de $500 + 500$) y otra por la forma digital de Legendre
        (calcúlense $s_3(500)$ y $s_3(1000)$); compruébese que las dos
        dan el mismo valor.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La paridad del triángulo de Pascal y una
cota de densidad de \omterm{def:b1:arith:prime}{primos}.}
\begin{enumerate}[resume]
  \item Demuéstrese el criterio de las cifras: $\binom nk$ es
        \emph{impar} si y solo si cada cifra binaria de $k$ es menor o
        igual que la cifra correspondiente de $n$. Enúnciese y
        demuéstrese el criterio análogo para $p \nmid \binom nk$ en
        base $p$.
  \item Dedúzcase que la fila $n$ del triángulo de Pascal contiene
        exactamente $2^{s_2(n)}$ entradas impares; compruébese en las
        filas $4$ y $5$.
  \item Dedúzcase que todas las entradas interiores $\binom nk$
        ($0 < k < n$) son pares si y solo si $n$ es una potencia
        de $2$.
  \item Demuéstrese que toda potencia de un \omterm{def:b1:arith:prime}{primo} que divida a
        $\binom{m+n}m$ es a lo sumo $m + n$: si $p^a \mid \binom{m+n}m$,
        entonces $p^a \leq m + n$. \emph{(¿Cuántos términos no nulos
        puede tener la suma de la pregunta 11?)}
  \item Dedúzcase que $\binom{2n}n$ \omterm{def:b1:arith:divides}{divide} a
        $\operatorname{lcm}(1, 2, \dots, 2n)$ y combínese con la cota
        inferior $\binom{2n}n \geq \frac{4^n}{2n+1}$ (que se
        demostrará: la entrada central es la mayor de las $2n + 1$
        entradas de la fila $2n$) para obtener
        \[
        \operatorname{lcm}(1, \dots, 2n) \geq \frac{4^n}{2n+1} :
        \]
        los múltiplos comunes de los primeros enteros crecen
        \emph{exponencialmente} --- un primer atisbo cuantitativo de
        la abundancia de \omterm{def:b1:arith:prime}{primos}.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Síntesis.}
\begin{enumerate}[resume]
  \item Hállese el menor $n$ tal que $n!$ termine en al menos $2026$
        ceros. \emph{(Estímese $Z(n) \approx n/4$ y ajústese después
        con la fórmula exacta.)}
  \item Una última comprobación cruzada: pruébese que $7$ \emph{no}
        \omterm{def:b1:arith:divides}{divide} a $\binom{100}{50}$, primero escribiendo $50$ en base
        $7$ y comprobando que la suma $50 + 50$ no tiene acarreos, y
        después calculando $v_7(100!)$ y $v_7(50!)$ con la fórmula de
        Legendre.
  \item ¿Dónde ha usado exactamente el problema: (i) la factorización
        única; (ii) la descomposición por división euclídea
        $n = p^k n_1 + n_0$; (iii) un argumento combinatorio
        del~\cref{ch:b1:counting}? Una frase para cada uno.
  \item Síntesis, en un párrafo breve: la fórmula de Legendre
        convierte una cuestión de \omterm{def:b1:arith:divides}{divisibilidad} en aritmética de
        cifras, y el teorema de Kummer lee la respuesta en los
        acarreos de una sola suma --- coméntese esta traducción, las
        comprobaciones de la pregunta 16 y lo que la cota de la
        pregunta 21 sugiere sobre los \omterm{def:b1:arith:prime}{primos} (el \omterm{def:b1:logic:statement}{enunciado} completo,
        el teorema de los \omterm{def:b1:arith:prime}{números primos}, queda muy lejos de este
        volumen; el análogo polinómico de las herramientas de este
        capítulo es el~\cref{ch:b1:poly}).
\end{enumerate}
\end{problem}
