\chapter{Espacios euclídeos}\label{ch:b1:euclid}

Añadir un \omterm{def:b1:euclid:def}{producto escalar} a un \omterm{def:b1:vspaces:def}{espacio vectorial} real compra las
nociones geométricas ---longitudes, ángulos, \omterm{def:b1:euclid:orthogonal}{ortogonalidad},
distancias--- y un teorema que domina el capítulo: todo \omterm{def:b1:vspaces:subspace}{subespacio}
admite una \omterm{thm:b1:euclid:projection}{proyección ortogonal}, calculable por Gram--Schmidt, que
realiza la distancia más corta. Las \omterm{def:b1:euclid:isometry}{isometrías} del plano cierran el
capítulo y la geometría del año.

En todo el capítulo, $E$ es un \omterm{def:b1:vspaces:def}{espacio vectorial} \emph{real}.

\section{Productos escalares}

\begin{definition}\label{def:b1:euclid:def}
Un \emph{producto escalar}\index{producto escalar} en $E$ es una
\omterm{def:b1:logic:map}{aplicación} $\langle\cdot,\cdot\rangle \colon E \times E \to \R$
bilineal, simétrica y definida positiva ($\langle x, x\rangle > 0$
para $x \neq 0$). Un espacio \omterm{def:b1:findim:def}{de dimensión finita} así equipado es un
\emph{espacio euclídeo}\index{espacio euclídeo}. La \emph{norma} de
$x$ es $\norm{x} = \sqrt{\langle x, x\rangle}$, y
$d(x, y) = \norm{x - y}$.
\end{definition}

\begin{example}\label{ex:b1:euclid:examples}
En $\R^n$: el producto canónico $\langle x, y\rangle = \sum x_i
y_i$. En $C(\intcc{a}{b})$: $\langle f, g \rangle = \int_a^b fg$ (que
sea definido positivo es el~\cref{thm:b1:integration:props}~(4)). En
$\R_n[X]$: $\langle P, Q\rangle = \int_0^1 PQ$, o bien
$\sum_{i} P(x_i)Q(x_i)$ sobre $n+1$ puntos distintos.
\end{example}

\begin{example}[El ángulo entre dos polinomios]
\label{ex:b1:euclid:polyangle}
Una vez elegido un \omterm{def:b1:euclid:def}{producto escalar}, \emph{cualesquiera} dos vectores
no nulos tienen un ángulo, vía
$\cos\theta = \frac{\langle x, y\rangle}{\norm x\,\norm y}$ (un
coseno legítimo por Cauchy--Schwarz). Para $X$ y $X^2$ con
$\int_0^1$:
\[
\langle X, X^2\rangle = \frac14, \qquad
\norm X = \frac1{\sqrt3}, \qquad \norm{X^2} = \frac1{\sqrt5},
\qquad
\cos\theta = \frac{1/4}{1/\sqrt{15}} = \frac{\sqrt{15}}{4}
\approx 0.968 :
\]
un ángulo de unos $14.5$ grados --- en $\intcc{0}{1}$, las gráficas
de $x$ y de $x^2$ son «casi paralelas» en el sentido de la media
cuadrática, y por eso quitar esa dirección compartida (Gram--Schmidt,
más abajo) deja solo la pequeña corrección $X^2 - X + \frac16$.
\end{example}

\begin{theorem}[Cauchy--Schwarz; propiedades de la norma]\label{thm:b1:euclid:cs}
Para todos $x, y \in E$:\index{desigualdad de Cauchy--Schwarz}
\[
\abs{\langle x, y\rangle} \leq \norm x\, \norm y ,
\]
con igualdad si y solo si $x, y$ son proporcionales. En consecuencia,
$\norm\cdot$ cumple la desigualdad triangular
$\norm{x + y} \leq \norm x + \norm y$ (y
$\norm{\lambda x} = \abs\lambda \norm x$, $\norm x = 0 \iff x = 0$).
Además:
\[
\norm{x+y}^2 = \norm x^2 + 2\langle x, y\rangle + \norm y^2,
\qquad
\langle x, y \rangle = \tfrac14\bigl(\norm{x+y}^2 -
\norm{x-y}^2\bigr) .
\]
\end{theorem}

\begin{proof}
Si $y = 0$, todo es trivial. En caso contrario, el trinomio
$t \mapsto \norm{x + ty}^2 = \norm x^2 + 2t\langle x, y\rangle + t^2
\norm y^2$ es $\geq 0$ para todo $t$: su discriminante es $\leq 0$,
que es Cauchy--Schwarz; y la igualdad significa una raíz doble $t_0$,
es decir, $x + t_0 y = 0$ (por ser definido): proporcionalidad.
Desigualdad triangular: desarróllese,
\[
\norm{x + y}^2 = \norm x^2 + 2\langle x, y\rangle + \norm y^2
\leq \norm x^2 + 2\norm x\,\norm y + \norm y^2
= \bigl(\norm x + \norm y\bigr)^2 ,
\]
siendo el paso central Cauchy--Schwarz; la igualdad fuerza
$\langle x, y\rangle = \norm x\norm y$, el caso de igualdad
\emph{positivo}, es decir, la proporcionalidad con razón no negativa
--- geométricamente, el triángulo solo degenera cuando los dos
vectores apuntan en el mismo sentido. Las dos últimas identidades son
desarrollos directos (la segunda, la \emph{identidad de
polarización}, recupera el producto a partir de la norma).
\end{proof}

\section{Ortogonalidad}

\begin{definition}\label{def:b1:euclid:orthogonal}
$x \perp y$ cuando $\langle x, y \rangle = 0$. Una familia es
\emph{ortogonal} cuando sus vectores son ortogonales dos a dos, y
\emph{ortonormal}\index{familia ortonormal} cuando además cada uno
tiene norma $1$. El \emph{complemento ortogonal} de un \omterm{def:b1:vspaces:subspace}{subespacio}
$F$ es
\[
F^{\perp} = \{x \in E : \forall y \in F,\ \langle x, y\rangle =
0\},
\]
un \omterm{def:b1:vspaces:subspace}{subespacio} de $E$.\index{complemento ortogonal}
\end{definition}

\begin{proposition}\label{prop:b1:euclid:orthfree}
(Pitágoras\index{teorema de Pitágoras}) Si $x \perp y$, entonces
$\norm{x+y}^2 = \norm x^2 + \norm y^2$. Una familia \omterm{def:b1:euclid:orthogonal}{ortogonal} de
vectores no nulos es \omterm{def:b1:vspaces:free}{libre}. Y en una \omterm{def:b1:vspaces:free}{base} \omterm{def:b1:euclid:orthogonal}{ortonormal}
$(e_1, \dots, e_n)$, las \omterm{prop:b1:vspaces:coordinates}{coordenadas} y los productos son
\[
x = \sum_{i} \langle x, e_i\rangle\, e_i,
\qquad
\langle x, y \rangle = \sum_i \langle x, e_i\rangle \langle y,
e_i\rangle,
\qquad
\norm x^2 = \sum_i \langle x, e_i\rangle^2 .
\]
\end{proposition}

\begin{proof}
Pitágoras: desarróllese. Libertad: tómese
$\langle\,\cdot\,, x_j\rangle$ de una combinación nula:
$\lambda_j \norm{x_j}^2 = 0$. \omterm{prop:b1:vspaces:coordinates}{Coordenadas}: escríbase
$x = \sum \lambda_i e_i$ y tómese el producto con $e_j$:
$\lambda_j = \langle x, e_j\rangle$; y las dos fórmulas se siguen por
bilinealidad.
\end{proof}

\begin{example}[Coordenadas ortonormales, con comprobación de Parseval]
\label{ex:b1:euclid:parsevalcheck}
Desarróllese $x = (1, 2, 3)$ en la \omterm{def:b1:vspaces:free}{base} \omterm{def:b1:euclid:orthogonal}{ortonormal}
del~\cref{exo:b1:euclid:3},
\[
e_1 = \tfrac{1}{\sqrt2}(1,1,0), \quad
e_2 = \tfrac{1}{\sqrt6}(1,-1,2), \quad
e_3 = \tfrac{1}{\sqrt3}(-1,1,1).
\]
Ningún sistema que resolver --- tres \omterm{def:b1:euclid:def}{productos escalares}:
\[
\langle x, e_1\rangle = \frac{3}{\sqrt2},
\qquad
\langle x, e_2\rangle = \frac{1 - 2 + 6}{\sqrt6} =
\frac{5}{\sqrt6},
\qquad
\langle x, e_3\rangle = \frac{-1 + 2 + 3}{\sqrt3} =
\frac{4}{\sqrt3}.
\]
Certificación con la fórmula de la norma de la proposición:
\[
\frac{9}{2} + \frac{25}{6} + \frac{16}{3}
= \frac{27 + 25 + 32}{6} = 14 = \norm{x}^2 = 1 + 4 + 9 .
\]
Esta comprobación de la suma de los cuadrados de las \omterm{prop:b1:vspaces:coordinates}{coordenadas} (una
identidad de Parseval finita) cuesta segundos y atrapa con casi total
seguridad los errores de signo y de normalización --- conviértase en
costumbre siempre que se calcule un desarrollo \omterm{def:b1:euclid:orthogonal}{ortonormal}; su
versión en dimensión infinita, para los coeficientes de Fourier
del~\cref{ex:b1:euclid:trigortho}, es un teorema del volumen del
tercer año.
\end{example}

\begin{theorem}[Gram--Schmidt]\label{thm:b1:euclid:gramschmidt}
Todo \omterm{def:b1:euclid:def}{espacio euclídeo} tiene \omterm{def:b1:vspaces:free}{bases} \omterm{def:b1:euclid:orthogonal}{ortonormales}. Explícitamente, a
partir de cualquier base $(v_1, \dots, v_n)$, la
receta\index{método de Gram--Schmidt}
\[
w_k = v_k - \sum_{i=1}^{k-1} \langle v_k, e_i\rangle\, e_i ,
\qquad
e_k = \frac{w_k}{\norm{w_k}}
\]
produce una \omterm{def:b1:vspaces:free}{base} \omterm{def:b1:euclid:orthogonal}{ortonormal} $(e_1, \dots, e_n)$ con
$\operatorname{Vect}(e_1, \dots, e_k) = \operatorname{Vect}(v_1,
\dots, v_k)$ para todo $k$.
\end{theorem}

\begin{proof}
Inducción sobre $k$. Suponiendo $(e_1, \dots, e_{k-1})$ \omterm{def:b1:euclid:orthogonal}{ortonormal} y
generadora de $\operatorname{Vect}(v_1, \dots, v_{k-1})$: el vector
$w_k$ es \omterm{def:b1:euclid:orthogonal}{ortogonal} a cada $e_j$ ($j < k$) por construcción
($\langle w_k, e_j \rangle = \langle v_k, e_j\rangle - \langle v_k,
e_j\rangle$), y $w_k \neq 0$ ya que
$v_k \notin \operatorname{Vect}(v_1, \dots, v_{k-1})$. Normalizar
conserva la \omterm{def:b1:euclid:orthogonal}{ortogonalidad}; y el \omterm{def:b1:logic:statement}{enunciado} sobre el \omterm{def:b1:vspaces:span}{subespacio
generado} vale porque $e_k$ es combinación de $v_k$ y de los $e_i$
anteriores, de forma invertible.
\end{proof}

\begin{example}[Gram--Schmidt sobre polinomios, al completo]
\label{ex:b1:euclid:legendre}
Ortonormalícese $(1, X, X^2)$ en $\R_2[X]$ con
$\langle P, Q\rangle = \int_0^1 PQ$. \emph{Paso 1}: $\norm{1}^2 = 1$,
luego $e_1 = 1$. \emph{Paso 2}: $w_2 = X - \langle X, 1\rangle\,1 =
X - \frac12$, y
$\norm{w_2}^2 = \int_0^1\bigl(x - \frac12\bigr)^2 \dd x =
\frac1{12}$: $e_2 = \sqrt{12}\,\bigl(X - \frac12\bigr)$.
\emph{Paso 3}: $\langle X^2, e_1\rangle = \frac13$ y
\[
\langle X^2, e_2\rangle
= \sqrt{12}\int_0^1 x^2\Bigl(x - \frac12\Bigr)\dd x
= \frac{\sqrt{12}}{12},
\qquad\text{luego}\qquad
w_3 = X^2 - \frac13 - \Bigl(X - \frac12\Bigr)
= X^2 - X + \frac16 .
\]
Su norma se calculó en el~\cref{exo:b1:euclid:9}:
$\norm{w_3}^2 = \frac1{180}$, de donde
$e_3 = \sqrt{180}\,\bigl(X^2 - X + \frac16\bigr)$. Los \omterm{def:b1:poly:def}{polinomios}
$1$, $X - \frac12$, $X^2 - X + \frac16$ son, salvo escala, los
primeros \emph{\omterm{def:b1:poly:def}{polinomios} de Legendre} del \omterm{prop:b1:reals:intervals}{intervalo}
$\intcc{0}{1}$; la construcción sigue de grado en grado, con cada
nuevo \omterm{def:b1:poly:def}{polinomio} \omterm{def:b1:euclid:orthogonal}{ortogonal} a todos sus predecesores. Obsérvese cómo
el algoritmo recicla el trabajo anterior: la \omterm{def:b1:linmaps:projection}{proyección} que se resta
en el paso 3 es exactamente la mejor aproximación afín de $X^2$
hallada en el~\cref{ex:b1:euclid:bestapprox} --- Gram--Schmidt
\emph{es} la \omterm{thm:b1:euclid:projection}{proyección ortogonal} iterada.
\end{example}

\begin{theorem}[Proyección ortogonal]\label{thm:b1:euclid:projection}
Sea $F$ un \omterm{def:b1:vspaces:subspace}{subespacio} del \omterm{def:b1:euclid:def}{espacio euclídeo} $E$. Entonces
\[
E = F \oplus F^{\perp},
\]
y la \omterm{def:b1:linmaps:projection}{proyección} asociada $p_F$ sobre $F$ (la \emph{proyección
ortogonal}\index{proyección ortogonal}) viene dada, en cualquier
\omterm{def:b1:vspaces:free}{base} \omterm{def:b1:euclid:orthogonal}{ortonormal} $(e_1, \dots, e_k)$ de $F$, por
$p_F(x) = \sum_i \langle x, e_i\rangle e_i$. Realiza la distancia a
$F$: para todo $y \in F$,
\[
\norm{x - p_F(x)} \leq \norm{x - y},
\]
con igualdad solo para $y = p_F(x)$; se escribe
$d(x, F) = \norm{x - p_F(x)}$.
\end{theorem}

\begin{proof}
Tómese una \omterm{def:b1:vspaces:free}{base} \omterm{def:b1:euclid:orthogonal}{ortonormal} $(e_i)_{i \leq k}$ de $F$
(\cref{thm:b1:euclid:gramschmidt} dentro de $F$) y póngase
$\pi(x) = \sum \langle x, e_i\rangle e_i \in F$. Entonces
$x - \pi(x) \perp e_j$ para cada $j$ (la misma cancelación de antes),
luego $x - \pi(x) \in F^\perp$: $E = F + F^\perp$. Y
$F \cap F^\perp = \{0\}$: un vector así cumple
$\langle x, x\rangle = 0$. Luego la suma es directa y $\pi = p_F$.

Distancia: para $y \in F$, descompóngase
$x - y = (x - p_F(x)) + (p_F(x) - y)$, piezas \omterm{def:b1:euclid:orthogonal}{ortogonales}
($F^\perp$ y $F$); Pitágoras:
\[
\norm{x - y}^2 = \norm{x - p_F(x)}^2 + \norm{p_F(x) - y}^2
\geq \norm{x - p_F(x)}^2,
\]
con igualdad si y solo si $y = p_F(x)$.
\end{proof}

\begin{example}[Las proyecciones nunca alargan]
\label{ex:b1:euclid:bessel}
Aplicando Pitágoras a la separación $x = p_F(x) + (x - p_F(x))$:
\[
\norm{p_F(x)}^2 = \norm x^2 - \norm{x - p_F(x)}^2 \leq
\norm x^2 ,
\]
con igualdad si y solo si $x \in F$. En una \omterm{def:b1:vspaces:free}{base} \omterm{def:b1:euclid:orthogonal}{ortonormal}
$(e_1, \dots, e_k)$ de $F$, esto dice
$\sum_{i \leq k}\langle x, e_i\rangle^2 \leq \norm x^2$ (una
\emph{desigualdad de Bessel}): por muchas direcciones \omterm{def:b1:euclid:orthogonal}{ortonormales}
que se midan, las \omterm{prop:b1:vspaces:coordinates}{coordenadas} al cuadrado nunca superan la longitud
al cuadrado --- compárese con la igualdad exacta
del~\cref{ex:b1:euclid:parsevalcheck} cuando la familia es una \omterm{def:b1:vspaces:free}{base}
completa. Esta desigualdad de una línea es lo que hace sumables los
coeficientes de Fourier en el volumen del tercer año; y aquí ya
explica por qué añadir más funciones de \omterm{def:b1:vspaces:free}{base} a un ajuste por mínimos
cuadrados solo puede reducir el residuo.
\end{example}

\begin{example}[La mejor aproximación cuadrática]\label{ex:b1:euclid:bestapprox}
En $C(\intcc{0}{1})$ con $\langle f, g\rangle = \int_0^1 fg$, el
\omterm{def:b1:poly:def}{polinomio} de grado $\leq 1$ más próximo a $f(x) = x^2$ en la
distancia asociada (la de la media cuadrática) es $p_F(f)$, con
$F = \R_1[X]$. Gram--Schmidt sobre $(1, X)$: $e_1 = 1$,
$w_2 = X - \frac12$,
$\norm{w_2}^2 = \int_0^1 (x - \frac12)^2 = \frac{1}{12}$,
$e_2 = \sqrt{12}\,(X - \tfrac12)$. Entonces
\[
p_F(f) = \langle f, e_1\rangle e_1 + \langle f, e_2\rangle e_2
= \frac13 + 12\Bigl(\int_0^1 x^2\bigl(x - \tfrac12\bigr)\dd
x\Bigr)\bigl(X - \tfrac12\bigr)
= X - \frac{1}{6},
\]
usando $\int_0^1 x^2(x - \frac12)\dd x = \frac14 - \frac16 =
\frac{1}{12}$. La idea de los «mínimos cuadrados» en una línea de
álgebra lineal.
\end{example}

\begin{method}[Tres caminos hacia una distancia $d(x, F)$]
\label{met:b1:euclid:distance}
\begin{enumerate}
  \item \emph{\omterm{def:b1:vspaces:free}{Base} \omterm{def:b1:euclid:orthogonal}{ortonormal} de $F$}: entonces
        $p_F(x) = \sum_i \langle x, e_i\rangle e_i$ y, por Pitágoras,
        \[
        d(x, F)^2 = \norm{x}^2 - \norm{p_F(x)}^2
        = \norm x^2 - \sum_i \langle x, e_i\rangle^2 ,
        \]
        a menudo más barato que calcular el propio $x - p_F(x)$.
  \item \emph{Ecuaciones normales}: con cualquier familia
        generadora de $F$, resuélvase
        $\langle x - p, v_j\rangle = 0$ para los coeficientes de $p$
        (\cref{exo:b1:euclid:5}) --- sin necesidad de
        ortonormalizar.
  \item \emph{Por el complemento}: si $F^\perp$ es más pequeño que
        $F$ (por ejemplo, $F$ un \omterm{def:b1:linmaps:forms}{hiperplano} y $F^\perp$ una recta
        $\operatorname{Vect}(n)$), proyéctese sobre $F^\perp$ en su
        lugar:
        \[
        d(x, F) = \norm{p_{F^\perp}(x)} =
        \frac{\abs{\langle x, n\rangle}}{\norm n} ,
        \]
        que es la fórmula clásica de la distancia a un plano (la usa
        el~\cref{exo:b1:multivar:8}).
\end{enumerate}
El camino 3 es un caso particular de un reflejo general: proyéctese
siempre sobre aquel de $F$, $F^\perp$ que tenga menor dimensión.
\end{method}

\begin{example}[Ortogonalidad trigonométrica: un anticipo de Fourier]
\label{ex:b1:euclid:trigortho}
En $C(\intcc{0}{2\pi})$ con
$\langle f, g\rangle = \frac1\pi\int_0^{2\pi} fg$, la familia
\[
\Bigl(\frac{1}{\sqrt2},\ \cos x,\ \sin x,\ \cos 2x,\ \sin 2x,\
\dots\Bigr)
\]
es \omterm{def:b1:euclid:orthogonal}{ortonormal}: por ejemplo,
$\langle \cos px, \cos qx\rangle = \frac1\pi\int_0^{2\pi}\cos px\cos
qx\,\dd x = 0$ para $p \neq q$ (linealícese el producto en
$\frac12[\cos(p{-}q)x + \cos(p{+}q)x]$ e intégrese sobre períodos
completos), mientras que $\frac1\pi\int_0^{2\pi}\cos^2 px\,\dd x = 1$.
La \omterm{thm:b1:euclid:projection}{proyección ortogonal} sobre el \omterm{def:b1:vspaces:span}{subespacio generado} por las
primeras $2N + 1$ de estas funciones tiene, por tanto, \omterm{prop:b1:vspaces:coordinates}{coordenadas}
$\langle f, e_i\rangle$ --- \omterm{thm:b1:integration:def}{integrales} contra cosenos y senos. Son
los \emph{coeficientes de Fourier} de $f$, y la \omterm{def:b1:linmaps:projection}{proyección} es su
mejor aproximación trigonométrica en media cuadrática; el volumen del
tercer año estudia su convergencia. La \omterm{def:b1:euclid:orthogonal}{ortogonalidad} hace todo el
trabajo: las fórmulas de los coeficientes son
el~\cref{thm:b1:euclid:projection} al pie de la letra.
\end{example}

\section{Isometrías del plano}

\begin{definition}\label{def:b1:euclid:isometry}
Un endomorfismo $u$ de un \omterm{def:b1:euclid:def}{espacio euclídeo} es una
\emph{isometría}\index{isometría} (o \omterm{def:b1:logic:map}{aplicación} \omterm{def:b1:euclid:orthogonal}{ortogonal}) cuando
conserva la norma: $\norm{u(x)} = \norm x$ para todo $x$ ---
equivalentemente (por polarización), conserva el \omterm{def:b1:euclid:def}{producto escalar};
equivalentemente, su matriz $A$ en una \omterm{def:b1:vspaces:free}{base} \omterm{def:b1:euclid:orthogonal}{ortonormal} cumple
$A^{\mathsf T} A = I$. Las isometrías forman un \omterm{def:b1:structures:group}{grupo}, el \omterm{def:b1:structures:group}{grupo}
\omterm{def:b1:euclid:orthogonal}{ortogonal} $O(E)$.
\end{definition}

\begin{example}[Reconocer una isometría a simple vista]
\label{ex:b1:euclid:recognize}
¿Es \omterm{def:b1:euclid:orthogonal}{ortogonal}
$A = \dfrac15\begin{pmatrix} 3 & -4\\ 4 & 3\end{pmatrix}$? Columnas:
normas $\frac15\sqrt{9 + 16} = 1$ y $\frac15\sqrt{16 + 9} = 1$;
producto $\frac1{25}(3\cdot(-4) + 4\cdot3) = 0$. Sí --- y
$\det A = \frac{9 + 16}{25} = 1$, así que es la rotación $R_\theta$
con $\cos\theta = \frac35$, $\sin\theta = \frac45$ (la «rotación
$3$-$4$-$5$», cuyo ángulo no es ninguna fracción notable de $\pi$).
En cambio, $B = \frac{1}{\sqrt2}\begin{pmatrix} 1 & 1\\ 0 &
1\end{pmatrix}$ tiene aspecto de estar escalada para tener determinante
unidad, pero su primera columna no es unitaria ($\frac1{\sqrt2}$): no es \omterm{def:b1:euclid:orthogonal}{ortogonal}
--- el determinante $\pm1$ por sí solo no certifica nada, hay que
comprobar las columnas.
\end{example}

\begin{theorem}[Isometrías del plano]\label{thm:b1:euclid:plane}
En una \omterm{def:b1:vspaces:free}{base} \omterm{def:b1:euclid:orthogonal}{ortonormal} de un plano euclídeo, las matrices de las
\omterm{def:b1:euclid:isometry}{isometrías} son exactamente
\[
R_\theta = \begin{pmatrix}
\cos\theta & -\sin\theta\\ \sin\theta & \cos\theta
\end{pmatrix}
\quad (\text{rotación de ángulo } \theta,\ \det = 1),
\]
\[
S_\theta = \begin{pmatrix}
\cos\theta & \sin\theta\\ \sin\theta & -\cos\theta
\end{pmatrix}
\quad (\det = -1),
\]
siendo la segunda la reflexión respecto de la recta que forma un
ángulo $\frac\theta2$ con el primer vector de la \omterm{def:b1:vspaces:free}{base}.
\end{theorem}

\begin{proof}
Sea $A = \begin{pmatrix} a & c\\ b & d\end{pmatrix}$ con
$A^{\mathsf T}A = I$: las columnas son unitarias y \omterm{def:b1:euclid:orthogonal}{ortogonales}. La
primera columna es $(\cos\theta, \sin\theta)$ para cierto $\theta$; y
la segunda, unitaria y \omterm{def:b1:euclid:orthogonal}{ortogonal} a ella, es
$\pm(-\sin\theta, \cos\theta)$. El signo $+$ da $R_\theta$ y el signo
$-$ da $S_\theta$. Se comprueba que $S_\theta^2 = I$ y que el vector
$(\cos\frac\theta2, \sin\frac\theta2)$ queda fijo mientras que su
\omterm{def:b1:euclid:orthogonal}{ortogonal} se invierte: una reflexión. (Y
$R_\alpha R_\beta = R_{\alpha+\beta}$: el \omterm{def:b1:structures:group}{grupo} de las rotaciones es
el \omterm{def:b1:structures:group}{grupo} de los ángulos --- compárese con
el~\cref{thm:b1:complex:funceq}.)
\end{proof}

\begin{remark}[Errores frecuentes]
\label{rem:b1:euclid:pitfalls}
\emph{La fórmula de la \omterm{def:b1:linmaps:projection}{proyección} necesita una \omterm{def:b1:vspaces:free}{base} \omterm{def:b1:euclid:orthogonal}{ortonormal}}:
para una familia $(v_i)$ de $F$ meramente generadora, la suma
$\sum_i\langle x, v_i\rangle v_i$ \emph{no} es $p_F(x)$ (pruébese con
$F = \R^2$, $v_1 = e_1$, $v_2 = e_1 + e_2$); con una familia
no \omterm{def:b1:euclid:orthogonal}{ortonormal}, resuélvanse en su lugar las ecuaciones normales
(\cref{met:b1:euclid:distance}~(2)).
\emph{Las familias \omterm{def:b1:euclid:orthogonal}{ortogonales} deben evitar el $0$ para ser \omterm{def:b1:vspaces:free}{libres}}:
el vector nulo es \omterm{def:b1:euclid:orthogonal}{ortogonal} a todo, incluido a sí mismo --- la
libertad en la~\cref{prop:b1:euclid:orthfree} exige vectores no nulos.
\emph{$F^\perp$ depende del \omterm{def:b1:euclid:def}{producto escalar}}: en $\R_1[X]$, el
complemento de $\operatorname{Vect}(X)$ para $\int_0^1 PQ$ no son las
constantes, sino $\operatorname{Vect}(1 - \frac32 X)$ --- calcúlese
$\int_0^1 x(1 - \frac32 x) = \frac12 - \frac12 = 0$;
«perpendicular» no significa nada mientras no se nombre el producto.
\emph{No se desarrolle $\norm{x + y}$ linealmente}: la identidad
correcta es
$\norm{x+y}^2 = \norm x^2 + 2\langle x, y\rangle + \norm y^2$; el
término cruzado se anula solo bajo \omterm{def:b1:euclid:orthogonal}{ortogonalidad} (Pitágoras), y la
desigualdad triangular es una \emph{desigualdad}.
\emph{No basta con llevar vectores unitarios a vectores unitarios}:
$u(x, y) = (x + y,\ 0)$ lleva los dos vectores de la \omterm{def:b1:vspaces:free}{base} canónica
al vector unitario $(1, 0)$ y, sin embargo,
$\norm{u(1,1)} = 2 \neq \sqrt2$: no hay \omterm{def:b1:euclid:isometry}{isometría}. La definición
exige $\norm{u(x)} = \norm x$ para \emph{todo} $x$; en términos
matriciales, $A^{\mathsf T}A = I$, es decir, columnas unitarias
\emph{y \omterm{def:b1:euclid:orthogonal}{ortogonales} dos a dos} --- las dos condiciones, comprobadas
juntas.
\end{remark}

\begin{remark}[Adónde va el producto escalar]
\label{rem:b1:euclid:whereused}
La \omterm{thm:b1:euclid:projection}{proyección ortogonal} es el teorema más aplicado del capítulo:
sostiene los mínimos cuadrados (el problema del fin de semana
del~\cref{ch:b1:multivar} construye sobre ella las rectas de
regresión), los coeficientes de Fourier
(\cref{ex:b1:euclid:trigortho}) y las ecuaciones normales
del~\cref{exo:b1:euclid:5}, que el análisis numérico resuelve a gran
escala. La clasificación $R_\theta / S_\theta$ se completa más abajo:
el problema del fin de semana clasifica \emph{todas} las
transformaciones del plano que conservan las distancias, \omterm{def:b1:linmaps:def}{lineales} o
no, y sus \omterm{def:b1:structures:group}{grupos} finitos --- las matemáticas de los rosetones y de
los polígonos regulares. En el volumen del segundo año, el
\omterm{def:b1:euclid:def}{producto escalar} se encuentra con la teoría de valores propios
(matrices simétricas, formas cuadráticas); y en el tercero, la
geometría euclídea de dimensión infinita se convierte en la teoría de
los espacios de Hilbert.
\end{remark}

\begin{remark}[Perspectivas dentro del libro 3]
\label{rem:b1:euclid:perspectives}
De este capítulo salen dos puentes. \emph{Hacia atrás}, al álgebra
lineal: la matriz de Gram del~\cref{exo:b1:euclid:11} empaqueta los
\omterm{def:b1:euclid:def}{productos escalares} en la maquinaria de determinantes
del~\cref{ch:b1:det}, y la \omterm{thm:b1:euclid:projection}{proyección ortogonal} es el proyector
particular del~\cref{ch:b1:linmaps} cuyo núcleo es $F^\perp$ --- toda
su álgebra ($p^2 = p$, $s = 2p - \mathrm{id}$) se aplica al pie de la
letra, ahora con la propina de que $\norm{x - p(x)}$ es una
\emph{distancia}. \emph{Hacia delante}, al análisis:
el~\cref{ch:b1:curves} mide la longitud de arco con la norma de este
capítulo y no clasifica nada sin sus \omterm{def:b1:euclid:isometry}{isometrías}; y
el~\cref{ch:b1:multivar} lee el gradiente a través de
Cauchy--Schwarz (el ascenso más pronunciado) y cierra el volumen con
los mínimos cuadrados, que son el~\cref{thm:b1:euclid:projection}
aplicado a un vector de datos de $\R^n$. El \omterm{def:b1:euclid:def}{producto escalar} es el
punto donde el álgebra del libro y su análisis se encuentran por fin.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:euclid:1}
En $\R^3$ canónico: calcúlense $\langle u, v\rangle$, $\norm u$,
$\norm v$ y el ángulo entre $u = (1, 2, 2)$ y $v = (2, -2, 1)$.
Compruébese Cauchy--Schwarz numéricamente.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:euclid:2}
Demuéstrese la identidad del paralelogramo
$\norm{x+y}^2 + \norm{x-y}^2 = 2\norm x^2 + 2\norm y^2$ en cualquier
\omterm{def:b1:euclid:def}{espacio euclídeo}, y úsese para ver que la norma del \omterm{def:b1:reals:bounds}{supremo} en
$\R^2$, $\norm{(x,y)}_\infty = \max(\abs x, \abs y)$, no proviene de
ningún \omterm{def:b1:euclid:def}{producto escalar}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:euclid:3}
Aplíquese Gram--Schmidt a $\bigl((1,1,0), (1,0,1), (0,1,1)\bigr)$ en
el $\R^3$ canónico.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:euclid:4}
En $\R^3$, sea $F = \operatorname{Vect}\bigl((1,1,1)\bigr)$.
Determínense $F^\perp$ (ecuación y \omterm{def:b1:vspaces:free}{base}), la matriz de $p_F$ en la
\omterm{def:b1:vspaces:free}{base} canónica y $d\bigl((1, 2, 3), F\bigr)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:euclid:5}
(Ecuaciones normales) Sean
$F = \operatorname{Vect}\bigl((1,0,1),(0,1,1)\bigr) \subseteq \R^3$
y $x = (1, 1, 4)$. Calcúlese $p_F(x)$ resolviendo
$\langle x - p, v \rangle = 0$ para los dos generadores
($p = \alpha(1,0,1) + \beta(0,1,1)$), y después $d(x, F)$. ¿Por qué
no hace falta aquí Gram--Schmidt?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:euclid:6}
Para $f$ \omterm{def:b1:continuity:continuous}{continua} en $\intcc{0}{1}$, demuéstrese
\[
\Bigl(\int_0^1 f\Bigr)^{\!2} \leq \int_0^1 f^2 ,
\]
con el caso de igualdad, como instancia de Cauchy--Schwarz en
$C(\intcc{0}{1})$. Demuéstrese después
$\bigl(\sum_{i=1}^n a_i\bigr)^2 \leq n \sum a_i^2$ para reales $a_i$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:euclid:7}
Demuéstrese que para todo \omterm{def:b1:vspaces:subspace}{subespacio} $F$ de un \omterm{def:b1:euclid:def}{espacio euclídeo}:
$(F^{\perp})^{\perp} = F$ y $\dim F^{\perp} = \dim E - \dim F$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:euclid:8}
Identifíquense las \omterm{def:b1:euclid:isometry}{isometrías} del plano de matrices
\[
A = \frac{1}{\sqrt 2}\begin{pmatrix} 1 & -1\\ 1 & 1\end{pmatrix},
\qquad
B = \frac{1}{5}\begin{pmatrix} 3 & 4\\ 4 & -3\end{pmatrix}
\]
(tipo, ángulo o eje). Calcúlense $A^8$ y $B^2$ sin multiplicar
matrices.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:euclid:9}
(El mínimo como \omterm{def:b1:linmaps:projection}{proyección}) Calcúlese
\[
\min_{(a, b) \in \R^2} \int_0^1 \bigl(x^2 - a - bx\bigr)^2 \dd x ,
\]
usando el~\cref{ex:b1:euclid:bestapprox}: el mínimo es
$\norm{f - p_F(f)}^2$ para $f = X^2$, $F = \R_1[X]$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:euclid:10}
Sea $u$ una \omterm{def:b1:euclid:isometry}{isometría} de un \omterm{def:b1:euclid:def}{espacio euclídeo} $E$. Demuéstrese que
$\ker(u - \mathrm{id}) \perp \operatorname{im}(u - \mathrm{id})$, y
dedúzcase
$E = \ker(u - \mathrm{id}) \oplus \operatorname{im}(u -
\mathrm{id})$. \emph{(Calcúlese $\langle x - u(x), y\rangle$ para
$u(y) = y$, usando la conservación del producto.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:euclid:11}
(Matriz de Gram) Para vectores $v_1, \dots, v_k$ de un \omterm{def:b1:euclid:def}{espacio
euclídeo}, sea
$G = \bigl(\langle v_i, v_j\rangle\bigr)_{1 \leq i, j \leq k}$ su
\emph{matriz de Gram}.
\begin{enumerate}
  \item Demuéstrese que $(v_1, \dots, v_k)$ es \omterm{def:b1:vspaces:free}{libre} si y solo si
        $G$ es invertible. \emph{(Si $Gc = 0$, calcúlese
        $\norm{\sum_i c_i v_i}^2$.)}
  \item Calcúlese la matriz de Gram de $(1, X, X^2)$ en $\R_2[X]$
        con $\langle P, Q\rangle = \int_0^1 PQ$, reconózcase la
        \omterm{pb:b1:det:1}{matriz de Hilbert} $H_3$ del problema del fin de semana
        del~\cref{ch:b1:det} y conclúyase la libertad a partir de
        $\det H_3 = \frac1{2160} \neq 0$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:euclid:12}
Sea $u$ un endomorfismo de un \omterm{def:b1:euclid:def}{espacio euclídeo} $E$ cuya matriz $A$
en una \omterm{def:b1:vspaces:free}{base} \omterm{def:b1:euclid:orthogonal}{ortonormal} es a la vez \omterm{def:b1:euclid:orthogonal}{ortogonal}
($A^{\mathsf T}A = I$) y simétrica ($A^{\mathsf T} = A$).
\begin{enumerate}
  \item Véase que $A^2 = I$ y dedúzcase (mediante
        el~\cref{thm:b1:linmaps:projchar}) que
        $E = \ker(u - \mathrm{id}) \oplus \ker(u + \mathrm{id})$.
  \item Véase que los dos \omterm{def:b1:vspaces:subspace}{subespacios} son \omterm{def:b1:euclid:orthogonal}{ortogonales}, de modo que
        $u$ es la \emph{simetría \omterm{def:b1:euclid:orthogonal}{ortogonal}} respecto de
        $F = \ker(u - \mathrm{id})$: la reflexión a través de $F$.
        \emph{(Para $u(x) = x$ y $u(y) = -y$, calcúlese
        $\langle x, y\rangle$ de dos maneras.)}
  \item Clasifíquese el caso plano: qué matrices
        del~\cref{thm:b1:euclid:plane} son simétricas y cuáles son
        las aplicaciones correspondientes.
\end{enumerate}
\end{exercise}

\section{Problema: isometrías del plano y teorema de Leonardo}

\begin{problem}\label{pb:b1:euclid:1}
Una \emph{\omterm{def:b1:euclid:isometry}{isometría} del plano} es cualquier \omterm{def:b1:logic:map}{aplicación}
$f \colon \R^2 \to \R^2$ que conserve las distancias:
$\norm{f(x) - f(y)} = \norm{x - y}$ para todos $x, y$ --- sin
suponer linealidad. Este problema demuestra que esas aplicaciones
son exactamente las traslaciones, las rotaciones, las reflexiones y
las \omterm{pb:b1:euclid:1}{reflexiones deslizantes} (la clasificación de las \omterm{def:b1:euclid:isometry}{isometrías} del
plano), calcula sus composiciones y determina todos sus \omterm{def:b1:structures:group}{grupos}
\emph{finitos}: el teorema de Leonardo, las matemáticas que hay
detrás de los rosetones. A partir de la parte II identificamos $\R^2$
con $\C$ (\cref{ch:b1:complex}): el \omterm{def:b1:euclid:def}{producto escalar} canónico es
$\langle z, w\rangle = \operatorname{Re}(z\conj w)$ y la norma es el
\omterm{def:b1:complex:field}{módulo}.\index{isometría}\index{reflexión deslizante}
\index{teorema de Leonardo}\index{grupo diédrico}

\medskip
\noindent\textbf{Parte I --- Toda \omterm{def:b1:euclid:isometry}{isometría} es afín.}
\begin{enumerate}
  \item Compruébese que las traslaciones $t_a(x) = x + a$, las
        \omterm{def:b1:euclid:isometry}{isometrías} \omterm{def:b1:linmaps:def}{lineales} y todas sus composiciones son
        \omterm{def:b1:euclid:isometry}{isometrías}, y que las \omterm{def:b1:euclid:isometry}{isometrías} forman un \omterm{def:b1:structures:group}{grupo} con la
        composición.
  \item Sea $f$ una \omterm{def:b1:euclid:isometry}{isometría} con $f(0) = 0$. Véase que $f$
        conserva las normas y después ---por polarización,
        \cref{thm:b1:euclid:cs}--- que
        $\langle f(x), f(y)\rangle = \langle x, y\rangle$ para todos
        $x, y$.
  \item Todavía con $f(0) = 0$: desarróllense
        $\norm{f(x + y) - f(x) - f(y)}^2$ y
        $\norm{f(\lambda x) - \lambda f(x)}^2$ usando la pregunta 2,
        y conclúyase que $f$ es \emph{\omterm{def:b1:linmaps:def}{lineal}}: $f \in O(\R^2)$.
  \item Dedúzcase que toda \omterm{def:b1:euclid:isometry}{isometría} $f$ se escribe de forma
        \emph{única} como $f = t_a \circ g$ con $a = f(0)$ y $g$ una
        \omterm{def:b1:euclid:isometry}{isometría} \omterm{def:b1:linmaps:def}{lineal} (la \emph{parte \omterm{def:b1:linmaps:def}{lineal}} de $f$).
  \item Dígase que $f$ es \emph{directa} si $\det g = 1$ e
        \emph{indirecta} si $\det g = -1$. Véase que la parte
        \omterm{def:b1:linmaps:def}{lineal} de una composición es la composición de las partes
        \omterm{def:b1:linmaps:def}{lineales}, y enúnciese la regla de signos resultante (las
        directas y las indirectas se componen como $+1/-1$).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Los cuatro tipos.} Por
el~\cref{thm:b1:euclid:plane}, las \omterm{def:b1:euclid:isometry}{isometrías} \omterm{def:b1:linmaps:def}{lineales} de $\C$ son
$z \mapsto az$ y $z \mapsto a\conj z$ con $\abs a = 1$; así que toda
\omterm{def:b1:euclid:isometry}{isometría} del plano es
\[
f(z) = a z + b \quad (\text{directa}) \qquad\text{o}\qquad
f(z) = a\conj z + b \quad (\text{indirecta}), \qquad \abs a = 1 .
\]
\begin{enumerate}[resume]
  \item Compruébese el diccionario: $R_\theta$ es
        $z \mapsto \eu^{\iu\theta}z$ y $S_\theta$ es
        $z \mapsto \eu^{\iu\theta}\conj z$ (compruébense los dos en
        $1$ y en $\iu$).
  \item (Caso directo) Sea $f(z) = az + b$, $\abs a = 1$. Véase
        que: si $a = 1$, $f$ es una traslación; y si $a \neq 1$, $f$
        tiene el único punto fijo $z_0 = b/(1 - a)$ y
        $f(z) - z_0 = a(z - z_0)$: una \emph{rotación} de centro
        $z_0$ y ángulo $\arg a$.
  \item (Caso indirecto) Sea $f(z) = a\conj z + b$ y
        $v = a\conj b + b$. Véase que $f \circ f = t_v$ y
        $f \circ t_v = t_v \circ f$. Si $v = 0$: véase que el punto
        medio de $z$ y $f(z)$ es un punto fijo y que $f$ es una
        \emph{reflexión} respecto de una recta. Y si $v \neq 0$:
        véase que $r = t_{-v/2}\circ f$ es una reflexión cuyo eje es
        paralelo a $v$, de modo que $f = t_{v/2} \circ r$ es una
        \emph{\omterm{pb:b1:euclid:1}{reflexión deslizante}}. Conclúyase: toda \omterm{def:b1:euclid:isometry}{isometría} del
        plano es una traslación, una rotación, una reflexión o una
        \omterm{pb:b1:euclid:1}{reflexión deslizante} (la clasificación de las \omterm{def:b1:euclid:isometry}{isometrías}
        del plano).
  \item (Composiciones) Véase que: la composición de rotaciones de
        ángulos $\alpha$ y $\beta$ es una rotación de ángulo
        $\alpha + \beta$ (una traslación si
        $\alpha + \beta \in 2\pi\Z$); y la composición de dos
        reflexiones es una rotación del doble del ángulo entre los
        ejes (una traslación si los ejes son paralelos).
  \item Dedúzcase que toda \omterm{def:b1:euclid:isometry}{isometría} del plano es composición de a
        lo sumo tres reflexiones.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Tres identificaciones.}
\begin{enumerate}[resume]
  \item Clasifíquese por completo $f(z) = \iu\conj z + 1 + \iu$:
        tipo, eje y vector de deslizamiento.
  \item Sea $f$ la rotación de ángulo $\frac\pi2$ en torno a $0$ y
        $g$ la rotación de ángulo $\frac\pi2$ en torno a $1$.
        Calcúlese $g \circ f$ en la forma $z \mapsto az + b$ e
        identifíquese (tipo, centro, ángulo).
  \item Sean $r_1(z) = \conj z$ (la reflexión respecto del eje real)
        y $r_2(z) = \iu\conj z$ (la reflexión respecto de la recta
        $y = x$). Calcúlese $r_2 \circ r_1$ y compruébese la
        pregunta 9 con este ejemplo.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- \omterm{def:b1:structures:group}{Grupos} finitos: el teorema de
Leonardo.} Sea $G$ un \omterm{def:b1:structures:group}{grupo} \emph{finito} de \omterm{def:b1:euclid:isometry}{isometrías} del plano.
\begin{enumerate}[resume]
  \item Véase que las \omterm{def:b1:euclid:isometry}{isometrías} conservan los baricentros: si
        $\lambda_1 + \dots + \lambda_m = 1$ y $f = t_a \circ g$ (con
        $g$ \omterm{def:b1:linmaps:def}{lineal}), entonces
        $f\bigl(\sum_i \lambda_i x_i\bigr) = \sum_i \lambda_i f(x_i)$.
  \item Póngase $c = \frac{1}{\abs G}\sum_{g \in G} g(x_0)$ para un
        $x_0$ elegido cualquiera. Véase que todo $h \in G$ deja fijo
        $c$: un \omterm{def:b1:structures:group}{grupo} finito de \omterm{def:b1:euclid:isometry}{isometrías} tiene un punto fijo común.
  \item Dedúzcase que, tras conjugar por $t_{-c}$, se puede suponer
        $G \subseteq O(\R^2)$. Sea $G^{+} = \{g \in G : \det g =
        1\}$; véase que o bien $G = G^{+}$, o bien $G^{+}$ tiene
        índice exactamente $2$ en $G$ (exhíbase una \omterm{def:b1:logic:inj}{biyección}
        $G^+ \to G \setminus G^+$).
  \item Véase que un \omterm{def:b1:structures:group}{grupo} finito de \emph{rotaciones} en torno a
        $c$ es cíclico: entre sus elementos, tómese la rotación
        $R_{\theta_0}$ de ángulo mínimo
        $\theta_0 \in \intoo{0}{2\pi}$ y úsese la división euclídea
        de ángulos para demostrar que genera; conclúyase
        $\theta_0 = \frac{2\pi}n$ y
        $G^{+} = \{R_{\theta_0}^{\,k}\} \cong C_n$.
  \item Supóngase $G \neq G^{+}$ y tómese una reflexión $s \in G$.
        Véase que $G = G^{+} \cup sG^{+}$, que $s r s = r^{-1}$ para
        toda rotación $r \in G^{+}$ y que los $n$ elementos de
        $sG^{+}$ son reflexiones: $G$ es el \emph{\omterm{pb:b1:euclid:1}{grupo diédrico}}
        $D_n$, de orden $2n$.
  \item Conclúyase (\emph{teorema de Leonardo}): todo \omterm{def:b1:structures:group}{grupo} finito
        de \omterm{def:b1:euclid:isometry}{isometrías} del plano es cíclico $C_n$ o diédrico $D_n$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dividendos, y síntesis.}
\begin{enumerate}[resume]
  \item Véase directamente que un \omterm{def:b1:structures:group}{grupo} finito de \omterm{def:b1:euclid:isometry}{isometrías} no
        puede contener ninguna traslación ni ninguna
        \omterm{pb:b1:euclid:1}{reflexión deslizante} distinta de la identidad (considérense
        las potencias de un elemento así).
  \item Sea $P_n$ el polígono regular de $n$ lados con vértices las
        raíces $n$-ésimas de la unidad ($n \geq 3$). Véase que su
        \omterm{def:b1:structures:group}{grupo} de simetrías es exactamente $D_n$: las $n$ rotaciones
        $z \mapsto \omega^k z$ y las $n$ reflexiones
        $z \mapsto \omega^k \conj z$, con
        $\omega = \eu^{2\iu\pi/n}$, y ninguna más.
  \item Exhíbanse figuras planas cuyos \omterm{def:b1:structures:group}{grupos} de simetrías sean,
        respectivamente, $C_1$, $D_1$, $D_2$ y $C_3$.
  \item Enumérense los ocho elementos del \omterm{def:b1:structures:group}{grupo} de simetrías del
        cuadrado de vértices $\pm1, \pm\iu$ como aplicaciones
        $z \mapsto \omega^k z$ o $z \mapsto \omega^k\conj z$, y dese
        el eje de cada una de las cuatro reflexiones.
  \item Sean $f, g$ rotaciones del mismo ángulo
        $\theta \notin 2\pi\Z$ en torno a centros \emph{distintos}
        $c_1 \neq c_2$. Calcúlese $f\circ g - g\circ f$ punto a
        punto y véase que $f\circ g \neq g\circ f$; véase además que
        $(f \circ g)\circ(g\circ f)^{-1}$ es una traslación no
        trivial, de modo que todo \omterm{def:b1:structures:group}{grupo} que contenga $f$ y $g$ es
        infinito --- una segunda explicación del centro único del
        teorema de Leonardo.
  \item Síntesis, en cuatro frases: qué dos resultados
        estructurales reducen las \omterm{def:b1:euclid:isometry}{isometrías} arbitrarias al álgebra
        lineal (preguntas 3--4) y los \omterm{def:b1:structures:group}{grupos} finitos arbitrarios a
        \omterm{def:b1:structures:subgroup}{subgrupos} de $O(2)$ (pregunta 15); cuál es la lista
        completa de las \omterm{def:b1:euclid:isometry}{isometrías} del plano y qué invariantes
        (directa/indirecta, puntos fijos) separan los cuatro tipos;
        por qué las reglas de composición de la pregunta 9 hacen de
        las reflexiones los generadores de todo; y qué añade el
        teorema de Leonardo a escala finita. Nómbrense los dos
        teoremas demostrados en las partes II y IV.
\end{enumerate}
\end{problem}
