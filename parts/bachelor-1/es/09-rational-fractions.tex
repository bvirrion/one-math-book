\chapter{Fracciones racionales}\label{ch:b1:fractions}

Una \omterm{def:b1:fractions:field}{fracción racional} es un cociente de \omterm{def:b1:poly:def}{polinomios}. El teorema
central de este capítulo breve --- la descomposición en fracciones
simples\index{descomposición en fracciones simples} --- rompe cualquier
cociente así en una suma de ladrillos elementales
$\frac{c}{(X - a)^k}$. Más allá de su interés algebraico, es la máquina
estándar para integrar funciones racionales (\cref{ch:b1:integration})
y para sumar ciertas series (\cref{ch:b1:series}).

\section{El cuerpo $K(X)$}

\begin{definition}\label{def:b1:fractions:field}
Una \emph{fracción racional}\index{fracción racional} sobre $K$ ($= \R$
o $\C$) es un cociente $F = \frac{A}{B}$ con $A, B \in K[X]$,
$B \neq 0$; dos cocientes $\frac AB$ y $\frac{A'}{B'}$ se identifican
cuando $AB' = A'B$. Toda fracción tiene una \emph{forma irreducible}
con $\gcd(A, B) = 1$, única salvo constantes. Con las operaciones
naturales, el \omterm{def:b1:logic:sets}{conjunto} $K(X)$ de las fracciones racionales es un \omterm{def:b1:structures:field}{cuerpo}.

Los \emph{polos}\index{polo} de $F$ (en forma irreducible) son las
raíces de $B$; el \emph{orden} de un polo es su \omterm{def:b1:poly:derivative}{multiplicidad} como raíz
de $B$. El \emph{grado} de $F$ es
$\deg F = \deg A - \deg B \in \Z \cup \{-\infty\}$.
\end{definition}

\begin{example}[Leer polos, órdenes y grado]\label{ex:b1:fractions:vocab}
Sea $F = \dfrac{X^3 - X}{X^4 - 2X^3 + X^2}$. Factorícense las dos
capas: numerador $X(X-1)(X+1)$, denominador $X^2(X-1)^2$; cancélese el
factor común $X(X-1)$:
\[
F = \frac{X + 1}{X(X - 1)} \quad\text{(reduced form).}
\]
\omterm{def:b1:fractions:field}{Polos}: $0$ y $1$, los dos \emph{simples} --- los órdenes se leen en
el denominador reducido, así que las aparentes raíces dobles del
original son irrelevantes. Grado: $\deg F = 1 - 2 = -1$, visible
asintóticamente ($xF(x) \to 1$ cuando $x \to \infty$). El grado se
comporta como el grado polinómico ($\deg FG = \deg F + \deg G$,
$\deg(F + G) \leq \max$), regla contable que se usa constantemente en
las cazas de coeficientes de más abajo: cada argumento del tipo
«límite de $xF(x)$» es un recuento de grados disfrazado.
\end{example}

\begin{proposition}[Parte entera]\label{prop:b1:fractions:integerpart}
Todo $F = \frac AB \in K(X)$ se escribe de manera única
$F = E + \frac RB$ con $E \in K[X]$ (la \emph{parte entera}, o parte
polinómica, de $F$) y $\deg R < \deg B$. Se tiene $E \neq 0$ si y solo
si $\deg F \geq 0$.
\end{proposition}

\begin{proof}
División euclídea $A = BE + R$ (\cref{thm:b1:poly:division}), dividida
entre $B$. Unicidad: si $E + \frac RB = E' + \frac{R'}{B}$, entonces
$(E - E')B = R' - R$ con $\deg(R' - R) < \deg B$, lo que obliga a
$E = E'$ y después a $R = R'$.
\end{proof}

\begin{example}[Redúzcase primero, divídase después]\label{ex:b1:fractions:integerpart}
Hállese la parte entera de $F = \dfrac{X^3 + 1}{X^2 - 1}$. Dividiendo a
ciegas: $X^3 + 1 = (X^2 - 1)X + (X + 1)$, luego
$F = X + \frac{X + 1}{X^2 - 1}$. Pero la fracción no era irreducible:
$X^3 + 1 = (X + 1)(X^2 - X + 1)$ y $X^2 - 1 = (X+1)(X-1)$ comparten el
factor $X + 1$, y
\[
F = \frac{X^2 - X + 1}{X - 1} = X + \frac{1}{X - 1} :
\]
la misma parte entera $X$, pero la parte fraccionaria se reduce a un
único ladrillo, y el «\omterm{def:b1:fractions:field}{polo}» en $-1$ nunca fue un \omterm{def:b1:fractions:field}{polo}. Redúzcase
siempre a forma irreducible antes de cazar \omterm{def:b1:fractions:field}{polos}: los \omterm{def:b1:fractions:field}{polos} de $F$
son las raíces del denominador \emph{reducido}. (La parte entera es
insensible a la simplificación, como garantiza la unicidad de
la~\cref{prop:b1:fractions:integerpart}.)
\end{example}

\section{Descomposición en fracciones simples sobre $\C$}

\begin{theorem}[Descomposición sobre $\C$]\label{thm:b1:fractions:complex}
Sea $F = \frac AB \in \C(X)$ en forma irreducible, con
$B = c\,(X - a_1)^{m_1} \cdots (X - a_r)^{m_r}$. Entonces $F$ es, de manera única,
\[
F = E + \sum_{i=1}^{r} \sum_{k=1}^{m_i}
\frac{c_{i,k}}{(X - a_i)^k},
\qquad E \in \C[X],\ c_{i,k} \in \C ,
\]
siendo $E$ la parte entera de $F$.
\end{theorem}

\begin{proof}
Por la~\cref{prop:b1:fractions:integerpart} podemos suponer
$\deg A < \deg B$ y demostrar la descomposición en suma con $E = 0$.

\emph{Separación de los \omterm{def:b1:fractions:field}{polos}.} Escríbase $B = (X - a_1)^{m_1} B_1$ con
$B_1(a_1) \neq 0$. Los \omterm{def:b1:poly:def}{polinomios} $(X-a_1)^{m_1}$ y $B_1$ son
\omterm{cor:b1:arith:bezout}{coprimos} (no tienen raíces comunes), luego, por Bézout en $\C[X]$
(véase la observación del~\cref{ch:b1:poly}), existen $U_0, V_0$ con
$U_0 (X-a_1)^{m_1} + V_0 B_1 = 1$; multiplicando por $A$ y poniendo
$U = AU_0$, $V = AV_0$, y dividiendo después entre $B$:
\[
\frac AB = \frac{V}{(X - a_1)^{m_1}} + \frac{U}{B_1} .
\]
Las condiciones de grado se pueden imponer: divídase $V$ entre
$(X-a_1)^{m_1}$, digamos $V = (X-a_1)^{m_1}Q + V_1$ con
$\deg V_1 < m_1$, y absórbase el cociente en el segundo término
($U_1 = U + QB_1$):
\[
\frac AB = \frac{V_1}{(X - a_1)^{m_1}} + \frac{U_1}{B_1},
\qquad \deg V_1 < m_1 ;
\]
comparando grados ($\deg A < \deg B$ y $\deg V_1 < m_1$) se obliga
también a $\deg U_1 < \deg B_1$. Iterando sobre $\frac{U_1}{B_1}$,
\omterm{def:b1:fractions:field}{polo} tras \omterm{def:b1:fractions:field}{polo}, todo se reduce al caso de un solo \omterm{def:b1:fractions:field}{polo} de más abajo.

\emph{Un solo \omterm{def:b1:fractions:field}{polo}.} Para $\frac{A}{(X-a)^m}$ con $\deg A < m$:
desarróllese $A$ en potencias de $(X - a)$,
$A = \sum_{j=0}^{m-1} \alpha_j (X - a)^j$ (desarrollo de Taylor de un
\omterm{def:b1:poly:def}{polinomio}, como en la demostración de
la~\cref{prop:b1:poly:multiplicity}); al dividir aparecen exactamente
los ladrillos $\frac{\alpha_j}{(X-a)^{m-j}}$. En concreto, para
$\frac{X^2 + 1}{(X+2)^3}$: sustituyendo $X = Y - 2$,
\[
X^2 + 1 = (Y - 2)^2 + 1 = Y^2 - 4Y + 5 ,
\qquad\text{so}\qquad
\frac{X^2+1}{(X+2)^3} = \frac1{Y} - \frac4{Y^2} + \frac5{Y^3}
\]
con $Y = X + 2$: los tres ladrillos aparecen con la mera división del
desarrollo desplazado --- la vía más rápida siempre que haya un único
\omterm{def:b1:fractions:field}{polo} de orden alto, y la recomendada para
el~\cref{exo:b1:fractions:3}.

\emph{Unicidad.} Supóngase que dos descomposiciones coinciden; su
diferencia es una identidad $0 = \sum_{i,k} \frac{d_{i,k}}{(X -
a_i)^k}$. Multiplíquese todo por $(X - a_1)^{m_1}$: todo término
adquiere un factor que se anula en $a_1$ \emph{salvo} el de $i = 1$,
$k = m_1$, cuyo coeficiente queda como $d_{1,m_1}$ más términos con al
menos un factor $(X - a_1)$. Evaluando en $a_1$ (es lícito: tras la
multiplicación ya no queda \omterm{def:b1:fractions:field}{polo} en $a_1$) se obtiene $d_{1,m_1} = 0$.
Eliminado el coeficiente superior, repítase con $(X - a_1)^{m_1 - 1}$ y
así sucesivamente hasta $k = 1$; pásese después al \omterm{def:b1:fractions:field}{polo} siguiente.
Todos los $d_{i,k}$ se anulan: la descomposición es única.
\end{proof}

\begin{method}[Cálculo de los coeficientes]\label{met:b1:fractions:compute}
En la práctica, evítese Bézout y combínense:
\begin{enumerate}
  \item \emph{tapado para la potencia más alta}: el coeficiente de
        $\frac{1}{(X-a)^m}$ (con $m$ el orden del \omterm{def:b1:fractions:field}{polo} $a$) es
        \[
        c_{a,m} = \Bigl[\,(X - a)^m F\,\Bigr]_{X = a};
        \]
        y, para un \omterm{def:b1:fractions:field}{polo} \emph{simple} de $F = \frac AB$, vale
        $\frac{A(a)}{B'(a)}$;
  \item \emph{evaluaciones} en puntos cómodos y \emph{límites} de
        $xF(x)$ cuando $x \to \infty$, para reunir relaciones lineales
        entre los coeficientes restantes;
  \item \emph{simetrías} de paridad o de conjugación, cuando las haya,
        para reducir el trabajo a la mitad.
\end{enumerate}
\end{method}

\begin{remark}[Errores frecuentes con las fracciones simples]\label{rem:b1:fractions:pitfalls}
\begin{enumerate}
  \item \emph{Saltarse la parte entera.} La descomposición en
        ladrillos se aplica a fracciones de grado $< 0$; cuando
        $\deg A \geq \deg B$, divídase primero
        (\cref{prop:b1:fractions:integerpart}), o la caza de
        coeficientes producirá contradicciones.
  \item \emph{Usar el tapado fuera de su alcance.} Multiplicar por
        $(X - a)^k$ y evaluar en $a$ da el coeficiente solo para
        $k = m$, el orden \emph{completo} del \omterm{def:b1:fractions:field}{polo}; los coeficientes
        de orden menor exigen otras relaciones (límites,
        evaluaciones) --- véase el~\cref{ex:b1:fractions:multiple}.
  \item \emph{Olvidarse de reducir.} Los \omterm{def:b1:fractions:field}{polos} se leen en la forma
        \emph{irreducible}; un factor común entre numerador y
        denominador crea \omterm{def:b1:fractions:field}{polos} fantasma
        (\cref{ex:b1:fractions:integerpart}).
  \item \emph{Ladrillos de forma equivocada sobre $\R$.} Sobre un
        factor cuadrático irreducible, los numeradores son
        \emph{afines} ($\alpha X + \beta$), no constantes; escribir
        solo $\frac{c}{X^2 + 1}$ pierde soluciones --- las formas
        correctas las dicta el~\cref{thm:b1:fractions:real}, nunca la
        improvisación.
\end{enumerate}
\end{remark}

\begin{proof}[Demostración de la fórmula del polo simple]
Cerca de un \omterm{def:b1:fractions:field}{polo} simple $a$: $B = (X - a) Q$ con $Q(a) \neq 0$, y
$B' = Q + (X - a) Q'$, luego $B'(a) = Q(a)$. El valor por tapado es
$\frac{A(a)}{Q(a)} = \frac{A(a)}{B'(a)}$.
\end{proof}

\begin{example}\label{ex:b1:fractions:simple}
Descompóngase $F = \dfrac{1}{X(X-1)(X-2)}$. Tres \omterm{def:b1:fractions:field}{polos} simples;
tapado en cada uno:
\[
c_0 = \frac{1}{(0-1)(0-2)} = \frac12,
\quad
c_1 = \frac{1}{1 \times (1 - 2)} = -1,
\quad
c_2 = \frac{1}{2 \times 1} = \frac12,
\]
luego $F = \dfrac{1/2}{X} - \dfrac{1}{X-1} + \dfrac{1/2}{X-2}$.
\emph{Comprobación en $X = 3$:} directamente,
$F(3) = \frac{1}{3\cdot2\cdot1} = \frac16$; y con la descomposición,
$\frac{1/2}{3} - \frac{1}{2} + \frac{1/2}{1} = \frac16 - \frac12 + \frac12 = \frac16$.
\end{example}

\begin{example}[Polo múltiple]\label{ex:b1:fractions:multiple}
Descompóngase $F = \dfrac{X}{(X-1)^2 (X+1)}$. Forma:
$\frac{a}{(X-1)^2} + \frac{b}{X-1} + \frac{c}{X+1}$.
\begin{itemize}
  \item Tapado en el \omterm{def:b1:fractions:field}{polo} doble: $a = \bigl[\frac{X}{X+1}\bigr]_{X=1}
        = \frac12$.
  \item Tapado en $-1$: $c = \bigl[\frac{X}{(X-1)^2}\bigr]_{X=-1} =
        -\frac14$.
  \item Límite de $xF(x)$ en $\infty$: $0 = b + c$, luego $b = \frac14$.
\end{itemize}
\[
F = \frac{1/2}{(X-1)^2} + \frac{1/4}{X-1} - \frac{1/4}{X+1} .
\]
\emph{Comprobación en $X = 0$:} $F(0) = 0$ y
$\frac12 - \frac14 - \frac14 = 0$.
\end{example}

\section{Descomposición sobre $\R$}

\begin{theorem}[Descomposición sobre $\R$]\label{thm:b1:fractions:real}
Sea $F \in \R(X)$ en forma irreducible con denominador
\[
B = c \prod_i (X - a_i)^{m_i} \prod_j (X^2 + p_j X + q_j)^{n_j}
\qquad (p_j^2 - 4q_j < 0).
\]
Entonces $F$ se descompone de manera única como su parte entera más términos
\[
\frac{c_{i,k}}{(X - a_i)^k} \quad (1 \leq k \leq m_i),
\qquad
\frac{\alpha_{j,l}\, X + \beta_{j,l}}{(X^2 + p_j X + q_j)^{l}}
\quad (1 \leq l \leq n_j),
\]
con coeficientes reales.
\end{theorem}

\begin{proof}
Descompóngase sobre $\C$ (\cref{thm:b1:fractions:complex}). Como $F$ es
real, el coeficiente sobre el \omterm{def:b1:fractions:field}{polo} $\conj a$ (en cada orden) es el
\omterm{def:b1:complex:field}{conjugado} del coeficiente sobre $a$ (aplíquese la conjugación a la
descomposición e invóquese la unicidad). Agrúpese cada par \omterm{def:b1:complex:field}{conjugado}:
\[
\frac{c}{(X - z)^l} + \frac{\conj c}{(X - \conj z)^l}
= \frac{c\,(X - \conj z)^l + \conj c\,(X - z)^l}{\bigl(X^2 - 2\Re(z)X
+ \abs z^2\bigr)^{l}},
\]
cuyo numerador es su propio \omterm{def:b1:complex:field}{conjugado}, luego es real, de grado $\leq l$;
separar múltiplos del factor cuadrático real lo baja a grado $\leq 1$
en cada nivel $l$ (una pequeña inducción descendente). Los \omterm{def:b1:fractions:field}{polos} reales
conservan sus coeficientes reales (la conjugación los deja fijos). La
unicidad se sigue de la unicidad sobre $\C$.
\end{proof}

\begin{example}[Ver el emparejamiento de conjugados]\label{ex:b1:fractions:pairing}
El mecanismo de la demostración, en el caso más pequeño: sobre $\C$,
los \omterm{def:b1:fractions:field}{polos} de $\frac1{X^2+1}$ son $\pm\iu$, con coeficientes por tapado
$\frac1{2\iu}$ en $\iu$ y $\frac1{-2\iu}$ en $-\iu$ --- \omterm{def:b1:complex:field}{conjugados} el
uno del otro, como predice el teorema:
\[
\frac{1}{X^2 + 1}
= \frac{1/(2\iu)}{X - \iu} - \frac{1/(2\iu)}{X + \iu} .
\]
Recombinando sobre el denominador común:
\[
\frac{1}{2\iu}\cdot\frac{(X + \iu) - (X - \iu)}{X^2 + 1}
= \frac{1}{2\iu}\cdot\frac{2\iu}{X^2 + 1} = \frac{1}{X^2+1} :
\]
las partes imaginarias se cancelan y el ladrillo real reaparece
intacto. Para integrandos reales normalmente no se sale nunca de $\R$
--- pero, al evaluar \emph{sumas} en puntos complejos (como hace el
problema del fin de semana con las \omterm{def:b1:complex:unity}{raíces de la unidad}), los ladrillos
complejos son la moneda natural, y este emparejamiento es el tipo de
cambio entre las dos descomposiciones.
\end{example}

\begin{example}\label{ex:b1:fractions:real}
Descompóngase $F = \dfrac{4}{(X^2+1)(X-1)^2}$ sobre $\R$. Forma:
$\frac{aX + b}{X^2 + 1} + \frac{c}{(X-1)^2} + \frac{d}{X - 1}$. Tapado
en el \omterm{def:b1:fractions:field}{polo} doble: $c = \bigl[\frac{4}{X^2+1}\bigr]_{X=1} = 2$. Tapado
en el \omterm{def:b1:fractions:field}{polo} complejo $\iu$ (el numerador sobre $X^2 + 1$ evaluado con la
descomposición compleja, o directamente): multiplíquese por $X^2 + 1$ y
hágase $X = \iu$:
\[
a\iu + b = \frac{4}{(\iu - 1)^2} = \frac{4}{-2\iu} = 2\iu ,
\]
luego $a = 2$, $b = 0$. Límite de $xF(x)$ en el infinito: $0 = a + d$,
luego $d = -2$. Por tanto
\[
F = \frac{2X}{X^2+1} + \frac{2}{(X-1)^2} - \frac{2}{X-1} .
\]
\emph{Comprobación en $X = 0$:} $F(0) = 4$ y $0 + 2 + 2 = 4$.
\end{example}

\begin{example}[Un denominador cúbico, de principio a fin]\label{ex:b1:fractions:cubic}
Descompóngase $F = \dfrac{1}{X^3 + 1}$ sobre $\R$. Factorícese primero:
$X^3 + 1 = (X + 1)(X^2 - X + 1)$, teniendo la cuadrática discriminante
$-3 < 0$. Forma: $\frac{a}{X+1} + \frac{bX + c}{X^2 - X + 1}$. Tapado
en el \omterm{def:b1:fractions:field}{polo} simple $-1$: $a = \bigl[\frac1{X^2 - X + 1}\bigr]_{X=-1} =
\frac13$. Límite de $xF(x)$ en el infinito: $0 = a + b$, luego
$b = -\frac13$. Evaluación en $X = 0$: $1 = a + c$, luego
$c = \frac23$. Por tanto
\[
\frac{1}{X^3+1} = \frac13\Bigl(\frac{1}{X+1}
+ \frac{-X + 2}{X^2 - X + 1}\Bigr) ,
\]
confirmado en $X = 1$: el miembro izquierdo vale $\frac12$ y el derecho
$\frac13\bigl(\frac12 + 1\bigr) = \frac12$. Obsérvese la economía: tres
incógnitas, tres hechos lineales baratos (un tapado, un límite, una
evaluación) y ningún desarrollo --- el flujo de trabajo
del~\cref{met:b1:fractions:compute} en estado puro.
\end{example}

\begin{remark}[Para qué sirve]
Una vez descompuesta, una función racional se integra término a
término: los ladrillos $\frac{1}{(x-a)^k}$ tienen primitivas
elementales, y los ladrillos $\frac{\alpha x + \beta}{(x^2 + px + q)^l}$
se reducen a $\ln$ y $\arctan$ (\cref{ch:b1:integration}). Las sumas
telescópicas son la otra \omterm{def:b1:logic:map}{aplicación} estándar (\cref{exo:b1:fractions:8}).
\end{remark}

\begin{remark}[Dónde se usa este capítulo]\label{rem:b1:fractions:whereused}
Las fracciones simples son ante todo un paso de
\emph{preprocesamiento}: el capítulo de integración
(\cref{ch:b1:integration}) pasa todo integrando racional por
el~\cref{thm:b1:fractions:real} antes de integrar, y el capítulo de
series (\cref{ch:b1:series}) telescopa términos racionales exactamente
como en el~\cref{exo:b1:fractions:8} y en el problema del fin de
semana --- que lleva la técnica hasta $\sum 1/k^2 = \pi^2/6$. La
derivada logarítmica $P'/P = \sum m_i/(X - a_i)$
(\cref{exo:b1:poly:11}) reaparece siempre que se estudia la
localización de las raíces. Más allá de este volumen, la
descomposición de $1/\chi(X)$ para un \omterm{def:b1:diffeq:linear2}{polinomio característico} $\chi$
sostiene el cálculo de potencias de matrices y de transformadas de
Laplace en el volumen del segundo año: los ladrillos
$\frac{c}{(X - a)^k}$ son la sombra algebraica de las soluciones
$t^{k-1}\eu^{at}$ del~\cref{ch:b1:diffeq}.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:fractions:1}
Descompóngase sobre $\R$: $\dfrac{1}{X^2 - 1}$;
$\;\dfrac{X}{X^2 - 3X + 2}$; $\;\dfrac{X^2 + 1}{X(X-1)}$ \emph{(ojo con
la parte entera)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:fractions:2}
Descompóngase sobre $\R$: $\dfrac{1}{X(X^2 + 1)}$ y
$\dfrac{X^3}{X^2 + X + 1}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:fractions:3}
Descompóngase $\dfrac{1}{X^2(X - 1)}$ y $\dfrac{X + 1}{(X - 1)^3}$
\emph{(para la segunda, sustitúyase $Y = X - 1$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:4}
Descompóngase sobre $\C$ y después sobre $\R$: $\dfrac{1}{X^4 - 1}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:5}
Descompóngase sobre $\R$: $\dfrac{X^2}{(X^2 + 1)^2}$, y dedúzcase una
primitiva de $x \mapsto \dfrac{x^2}{(x^2+1)^2}$ sabiendo que
$\int \frac{\dd x}{(x^2+1)^2} = \frac12\bigl(\arctan x +
\frac{x}{x^2+1}\bigr) + C$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:6}
Para $n \in \N^*$, descompóngase $F_n = \dfrac{n!}{X(X+1)\cdots(X+n)}$
\emph{(\omterm{def:b1:fractions:field}{polos} simples en $0, -1, \dots, -n$; úsese la fórmula del
tapado y reconózcanse \omterm{def:b1:counting:objects}{coeficientes binomiales})}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:7}
Usando la identidad del~\cref{exo:b1:poly:11} para $P = X^n - 1$,
demuéstrese que
\[
\sum_{k=0}^{n-1} \frac{1}{X - \omega^k} = \frac{n X^{n-1}}{X^n - 1},
\qquad \omega = \eu^{2\iu\pi/n},
\]
y evalúense los dos miembros en $X = 2$ para $n = 4$ como comprobación.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:8}
Descompóngase $\dfrac{1}{k(k+1)(k+2)}$ y calcúlense
\[
S_n = \sum_{k=1}^{n} \frac{1}{k(k+1)(k+2)},
\qquad\text{y después}\qquad
\lim_{n \to \infty} S_n .
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:fractions:9}
Sea $P \in \R[X]$ \omterm{def:b1:poly:def}{mónico} de grado $n$ con $n$ raíces reales distintas
$x_1 < \dots < x_n$. Demuéstrese que
\[
\sum_{i=1}^{n} \frac{1}{P'(x_i)} = 0 \quad (n \geq 2),
\qquad
\sum_{i=1}^{n} \frac{x_i^{\,n-1}}{P'(x_i)} = 1 .
\]
\emph{Indicación: descompóngase $\frac{X^m}{P}$ para $m \leq n - 1$ y
obsérvese el decaimiento de los coeficientes en el infinito --- o úsese
la \omterm{thm:b1:poly:lagrange}{interpolación de Lagrange} (\cref{thm:b1:poly:lagrange}) del
monomio $X^m$ en los nodos $x_i$.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:10}
Descompóngase $\dfrac{1}{X(X+1)^2}$ sobre $\R$. Admitiendo el valor
$\sum_{k \geq 1} \frac1{k^2} = \frac{\pi^2}6$ (demostrado en el problema
del fin de semana de este capítulo), dedúzcase
\[
\sum_{n=1}^{\infty} \frac{1}{n(n+1)^2} = 2 - \frac{\pi^2}{6} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fractions:11}
Descompóngase sobre $\R$: $\dfrac{1}{(X^2+1)(X^2+4)}$, y después
$\dfrac{X^2}{(X^2+1)(X^2+4)}$. \emph{Indicación: los dos denominadores
son \omterm{def:b1:poly:def}{polinomios} en $X^2$: descompóngase antes $\frac1{(Y+1)(Y+4)}$.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:fractions:12}
(Ecuaciones seculares) Sea $F = \sum_{i=1}^{r} \frac{c_i}{X - p_i}$ con
$p_1 < p_2 < \dots < p_r$ reales y todos los $c_i > 0$.
\begin{enumerate}
  \item Véase que $F$ es estrictamente decreciente en cada intervalo
        de su dominio, y dense sus límites en $\pm\infty$ y a los dos
        lados de cada \omterm{def:b1:fractions:field}{polo}.
  \item Dedúzcase que, para todo $\lambda > 0$, la ecuación
        $F(x) = \lambda$ tiene exactamente $r$ soluciones reales, una
        en cada intervalo $\intoo{p_i}{p_{i+1}}$ y otra más allá de
        $p_r$. (Ecuaciones así gobiernan las perturbaciones de valores
        propios; el teorema del valor intermedio se usa aquí al nivel
        del volumen anterior y se demuestra en el~\cref{ch:b1:continuity}.)
\end{enumerate}
\end{exercise}

\section{Problema: las fracciones simples como motor}

\begin{problem}\label{pb:b1:fractions:1}
La descomposición en fracciones simples parece contabilidad; este
problema muestra que es un motor. Alimentado con la fracción
$\frac1{X(X+1)\cdots(X+k)}$, telescopa familias enteras de sumas en
forma cerrada; alimentado con $\frac1{X^n - 1}$, produce identidades
trigonométricas como
\[
\sum_{k=1}^{n-1}\frac{1}{\sin^2\frac{k\pi}{n}} = \frac{n^2-1}{3} ;
\]
y, empujada un paso más, esa identidad exprime una de las fórmulas más
célebres de las matemáticas, la de Euler:
\[
\sum_{k=1}^{\infty}\frac1{k^2} = \frac{\pi^2}{6}
\]
--- obtenida aquí sin nada más que el álgebra de este capítulo y la
trigonometría anterior.\index{telescopio}\index{problema de Basilea}
En todo el problema, $\omega = \eu^{2\iu\pi/n}$; los límites de
sucesiones se usan al nivel del volumen anterior (el~\cref{ch:b1:seq}
los formaliza).

\medskip
\noindent\textbf{Parte I --- El telescopio.}
\begin{enumerate}
  \item Descompóngase $\frac1{X(X+1)}$ y calcúlese exactamente
        $\sum_{n=1}^{N} \frac1{n(n+1)}$; conclúyase que la suma tiende
        a $1$.
  \item Lo mismo para $\frac1{X(X+2)}$: véase que
        $\sum_{n=1}^{N} \frac1{n(n+2)} = \frac12\bigl(\frac32 -
        \frac1{N+1} - \frac1{N+2}\bigr) \to \frac34$. (Con un hueco
        sobreviven \emph{dos} términos de borde en cada extremo.)
  \item Formalícese el mecanismo: si $F(X) = G(X) - G(X+1)$ para alguna
        racional $G$ sin \omterm{def:b1:fractions:field}{polos} en $\intco1{+\infty}$, entonces
        $\sum_{n=1}^N F(n) = G(1) - G(N+1)$. Recupérese el valor
        $\frac14$ del~\cref{exo:b1:fractions:8} exhibiendo el testigo
        $G$ para $F = \frac1{X(X+1)(X+2)}$.
  \item Demuéstrese el telescopio factorial general: para $k \geq 1$,
        \[
        \frac{1}{X(X+1)\cdots(X+k)}
        = \frac1k\biggl(\frac{1}{X(X+1)\cdots(X+k-1)}
        - \frac{1}{(X+1)\cdots(X+k)}\biggr),
        \]
        y dedúzcase
        \[
        \sum_{n=1}^{\infty}\frac{1}{n(n+1)\cdots(n+k)}
        = \frac{1}{k \cdot k!} .
        \]
        Compruébese el caso $k = 2$ con la pregunta 3.
  \item Evalúese la descomposición del~\cref{exo:b1:fractions:6} en
        puntos bien elegidos para demostrar
        \[
        \sum_{j=0}^{n}(-1)^j\binom nj\,\frac1{j+1}
        = \frac1{n+1},
        \qquad
        \sum_{j=0}^{n}(-1)^j\binom nj\,\frac1{j+2}
        = \frac1{(n+1)(n+2)} .
        \]
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La fracción $1/(X^n - 1)$.}
\begin{enumerate}[resume]
  \item Véase, con la fórmula del tapado, que
        \[
        \frac{1}{X^n - 1}
        = \frac1n\sum_{k=0}^{n-1}\frac{\omega^k}{X - \omega^k} .
        \]
  \item Dos comprobaciones: verifíquese la fórmula directamente para
        $n = 2$ y véase que la suma de los $n$ coeficientes se anula
        para $n \geq 2$ --- explíquese por qué tiene que ser así
        (considérese $xF(x)$ cuando $x \to \infty$).
  \item Redúzcase de nuevo por tapado la identidad
        del~\cref{exo:b1:fractions:7}:
        $\frac{nX^{n-1}}{X^n-1} = \sum_k \frac1{X - \omega^k}$.
  \item Agrúpense los \omterm{def:b1:fractions:field}{polos} \omterm{def:b1:complex:field}{conjugados} para demostrar la
        descomposición real: con $\theta_k = \frac{2k\pi}n$,
        \[
        \frac{\omega^k}{X - \omega^k} +
        \frac{\omega^{n-k}}{X - \omega^{n-k}}
        = \frac{2\cos\theta_k\,X - 2}
        {X^2 - 2\cos\theta_k\,X + 1} ,
        \]
        y escríbase la descomposición real completa de
        $\frac1{X^n-1}$ (distínganse $n$ impar y $n$ par).
  \item Particularícese a $n = 4$ y compárese con
        el~\cref{exo:b1:fractions:4}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Sumas trigonométricas y el $\pi^2/6$ de
Euler.} Sea $P = 1 + X + \dots + X^{n-1}$, cuyas raíces son
$\omega, \omega^2, \dots, \omega^{n-1}$ (todas simples).
\begin{enumerate}[resume]
  \item Usando $\frac{P'}P = \sum_{k=1}^{n-1}\frac1{X - \omega^k}$
        (\cref{exo:b1:poly:11}), demuéstrese
        \[
        \sum_{k=1}^{n-1}\frac{1}{1 - \omega^k} = \frac{n-1}2 .
        \]
  \item Demuéstrese que $\dfrac1{1 - \eu^{\iu\theta}} = \dfrac12 +
        \dfrac\iu2\,\frac{\cos(\theta/2)}{\sin(\theta/2)}$ para
        $\theta \notin 2\pi\Z$ (factorización por el ángulo mitad,
        \cref{met:b1:complex:trig}) y dedúzcase de la pregunta 11 que
        $\sum_{k=1}^{n-1} \frac{\cos(k\pi/n)}{\sin(k\pi/n)} = 0$ ---
        visible también por la simetría $k \leftrightarrow n - k$.
  \item Derivando la identidad de la pregunta 11 (es decir, usando
        $\bigl(\frac{P'}P\bigr)' = \frac{P''}P -
        \bigl(\frac{P'}P\bigr)^2$ evaluado en $X = 1$), demuéstrese
        \[
        \sum_{k=1}^{n-1}\frac{1}{(1 - \omega^k)^2}
        = \frac{(n-1)(5-n)}{12} .
        \]
  \item Escribiendo $\cot t = \frac{\cos t}{\sin t}$, dedúzcanse de
        las preguntas 12--13 las dos formas cerradas
        \[
        \sum_{k=1}^{n-1}\cot^2\frac{k\pi}n = \frac{(n-1)(n-2)}3,
        \qquad
        \sum_{k=1}^{n-1}\frac1{\sin^2\frac{k\pi}n}
        = \frac{n^2-1}3 .
        \]
  \item Compruébense las dos fórmulas a mano para $n = 3$ y $n = 4$.
  \item Demuéstrense las desigualdades $\cot t < \frac1t <
        \frac1{\sin t}$ para $t \in \intoo0{\frac\pi2}$ (a partir de
        $\sin t < t < \tan t$) y dedúzcase, para $n = 2m + 1$ y
        $1 \leq k \leq m$:
        \[
        \cot^2\frac{k\pi}n \;<\; \frac{n^2}{k^2\pi^2}
        \;<\; \frac1{\sin^2\frac{k\pi}n} .
        \]
  \item Súmense estas desigualdades para $k = 1, \dots, m$ (usando la
        simetría $k \leftrightarrow n - k$ para reducir a la mitad las
        fórmulas de la pregunta 14) y aplíquese el emparedado:
        \[
        \sum_{k=1}^{\infty}\frac1{k^2} = \frac{\pi^2}6 .
        \]
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Ladrillos de orden superior.}
\begin{enumerate}[resume]
  \item Elevando al cuadrado la descomposición de $\frac1{X^2-1}$ y
        redescomponiendo el término cruzado, demuéstrese
        \[
        \frac1{(X^2-1)^2}
        = \frac14\Bigl(\frac1{(X-1)^2} + \frac1{(X+1)^2}\Bigr)
        - \frac14\Bigl(\frac1{X-1} - \frac1{X+1}\Bigr),
        \]
        y compruébese en $X = 0$.
  \item Combínense la pregunta 18, el telescopio y el valor de Euler
        (pregunta 17) para demostrar
        \[
        \sum_{n=2}^{\infty}\frac1{(n^2-1)^2}
        = \frac{\pi^2}{12} - \frac{11}{16} ,
        \]
        y confírmese el valor numéricamente con tres decimales.
  \item Derívese la identidad de la pregunta 8 para obtener una forma
        cerrada de $\sum_{k=0}^{n-1}\frac1{(X - \omega^k)^2}$, y
        compruébese en $X = 2$, $n = 2$.
  \item Demuéstrese que, para $k \geq 2$,
        \[
        \sum_{n=k}^{\infty}\frac{1}{\binom nk} = \frac{k}{k-1} .
        \]
        \emph{(Redúzcase a la pregunta 4 escribiendo $1/\binom nk$ con
        factoriales.)}
  \item Para $k = 3$, dese la suma parcial exacta
        $\sum_{n=3}^{N}\frac1{\binom n3}$ y su límite.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Síntesis.}
\begin{enumerate}[resume]
  \item Como cálculo de cierre, escríbase la descomposición real
        completa de $\dfrac1{X^6 - 1}$ y compruébese en $X = 0$.
  \item ¿Dónde ha usado exactamente el problema: (i) la unicidad de la
        descomposición; (ii) las \omterm{def:b1:complex:unity}{raíces de la unidad}
        del~\cref{ch:b1:complex}; (iii) la derivada logarítmica
        del~\cref{exo:b1:poly:11}? Una frase para cada uno.
  \item Síntesis, en un párrafo breve: una sola identidad algebraica
        --- romper una fracción en ladrillos --- ha generado sumas
        exactas, identidades trigonométricas y $\pi^2/6$. Coméntese el
        reparto de tareas entre el álgebra (descomposiciones exactas,
        válidas en todas partes) y el análisis (límites, emparedados),
        y señálese dónde se industrializa cada hilo: telescopios y
        comparación en el~\cref{ch:b1:series}, integración de los
        ladrillos en el~\cref{ch:b1:integration}.
\end{enumerate}
\end{problem}
