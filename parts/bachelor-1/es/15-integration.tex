\chapter{Integración en un segmento}\label{ch:b1:integration}

La \omterm{thm:b1:integration:def}{integral} del volumen anterior se fundaba en áreas tomadas de forma
intuitiva. Este capítulo la construye: primero para las \omterm{def:b1:integration:step}{funciones
escalonadas}, donde la \omterm{thm:b1:integration:def}{integral} es una suma finita, y después para las
funciones \omterm{def:b1:continuity:continuous}{continuas} (y \omterm{def:b1:continuity:continuous}{continuas} a trozos) por aproximación
uniforme --- el lugar donde el teorema de Heine
(\cref{thm:b1:continuity:heine}) se gana el sueldo. El teorema
fundamental del cálculo conecta después la construcción con las
primitivas, y las \omterm{thm:b1:integration:riemann}{sumas de Riemann} la conectan con los promedios
discretos.

En todo el capítulo, $a < b$ son reales.

\section{Funciones escalonadas}

\begin{definition}\label{def:b1:integration:step}
$\varphi \colon \intcc{a}{b} \to \R$ es una \emph{función
escalonada}\index{función escalonada} cuando existe una subdivisión
$a = x_0 < x_1 < \dots < x_n = b$ tal que $\varphi$ es constante,
igual a $c_i$, en cada \omterm{prop:b1:reals:intervals}{intervalo} \omterm{def:b1:topology:open}{abierto} $\intoo{x_{i-1}}{x_i}$ (los
valores en los nodos no están sujetos a restricción). Su \omterm{thm:b1:integration:def}{integral} es
\[
\int_a^b \varphi = \sum_{i=1}^{n} c_i\,(x_i - x_{i-1}),
\]
independiente de la subdivisión elegida (refínense dos subdivisiones
por la común: cada miembro queda invariante al refinar).
\end{definition}

\begin{proposition}\label{prop:b1:integration:stepprops}
Sobre las \omterm{def:b1:integration:step}{funciones escalonadas}, la \omterm{thm:b1:integration:def}{integral} es lineal, creciente
($\varphi \leq \psi \implies \int\varphi \leq \int\psi$) y cumple la
relación de Chasles $\int_a^b = \int_a^c + \int_c^b$ para $a < c < b$.
\end{proposition}

\begin{proof}
El motor es la \emph{invariancia por refinamiento}, enunciada en la
definición: insertar un nodo más $t \in \intoo{x_{i-1}}{x_i}$ en una
subdivisión sustituye el término $c_i(x_i - x_{i-1})$ por
$c_i(t - x_{i-1}) + c_i(x_i - t)$ ---el mismo número---, de modo que
la \omterm{thm:b1:integration:def}{integral} no cambia por ningún refinamiento finito. Tómense ahora
$\varphi$ con subdivisión $\sigma$ y $\psi$ con subdivisión
$\sigma'$: sobre el refinamiento común $\sigma \cup \sigma'$ las dos
son \omterm{def:b1:integration:step}{funciones escalonadas} con los \emph{mismos} nodos y, en cada
trozo, $\varphi + \lambda\psi$ es constante igual a
$c_i + \lambda d_i$: la linealidad se reduce a la linealidad de las
sumas finitas. Crecimiento: $c_i \leq d_i$ en cada trozo da
$\sum c_i \Delta_i \leq \sum d_i \Delta_i$ (las longitudes son
$\Delta_i \geq 0$). Chasles: insértese el nodo $c$ y pártase la suma
en él.
\end{proof}

\section{Integral de una función continua}

\begin{theorem}[Aproximación uniforme]\label{thm:b1:integration:approx}
Sea $f$ \omterm{def:b1:continuity:continuous}{continua} en $\intcc{a}{b}$. Para todo $\varepsilon > 0$ hay
\omterm{def:b1:integration:step}{funciones escalonadas} $\varphi, \psi$ con
\[
\varphi \leq f \leq \psi
\qquad\text{y}\qquad
\psi - \varphi \leq \varepsilon \text{ en } \intcc{a}{b}.
\]
\end{theorem}

\begin{proof}
Por el teorema de Heine (\cref{thm:b1:continuity:heine}), $f$ es
\omterm{def:b1:continuity:uniform}{uniformemente continua}: tómese $\delta$ para $\varepsilon$ y una
subdivisión de paso $< \delta$ (equidistante, digamos, con
$n > \frac{b - a}{\delta}$ trozos). En cada trozo \omterm{def:b1:topology:closed}{cerrado}
$\intcc{x_{i-1}}{x_i}$, $f$ alcanza un mínimo $m_i$ y un máximo $M_i$
(\cref{thm:b1:continuity:evt}), y $M_i - m_i \leq \varepsilon$ (los
dos puntos extremales distan menos de $\delta$). Defínanse
$\varphi = m_i$ y $\psi = M_i$ en $\intoo{x_{i-1}}{x_i}$ (y
$\varphi = \psi = f$ en los nodos).
\end{proof}

\begin{theorem}[Definición de la integral]\label{thm:b1:integration:def}
Sea $f$ \omterm{def:b1:continuity:continuous}{continua} en $\intcc{a}{b}$. Los dos números
\[
I_-(f) = \sup\Bigl\{\int_a^b \varphi : \varphi \text{ escalonada},\
\varphi \leq f\Bigr\},
\qquad
I_+(f) = \inf\Bigl\{\int_a^b \psi : \psi \text{ escalonada},\ \psi \geq
f\Bigr\}
\]
son iguales; su valor común es la
\emph{integral}\index{integral} $\int_a^b f$ (que también se escribe
$\int_a^b f(t)\,\dd t$). Coincide con la noción anterior sobre las
\omterm{def:b1:integration:step}{funciones escalonadas}, y se extiende a las funciones \omterm{def:b1:continuity:continuous}{continuas} a
trozos partiendo $\intcc{a}{b}$ en las discontinuidades (Chasles como
definición en ese caso).
\end{theorem}

\begin{proof}
Los dos \omterm{def:b1:logic:sets}{conjuntos} son no vacíos ($f$ está acotada) y toda \omterm{thm:b1:integration:def}{integral}
escalonada inferior es $\leq$ que toda superior (monotonía sobre las
\omterm{def:b1:integration:step}{funciones escalonadas}): luego $I_-(f) \leq I_+(f)$. Por
el~\cref{thm:b1:integration:approx}, para cada $\varepsilon$ hay una
pareja con $\int\psi - \int\varphi \leq \varepsilon(b - a)$: el
\omterm{def:b1:reals:bounds}{supremo} y el \omterm{def:b1:reals:bounds}{ínfimo} quedan apretados uno contra otro, $I_- = I_+$.
\end{proof}

\begin{example}[Continua a trozos, sin dramas]\label{ex:b1:integration:piecewise}
La función \omterm{thm:b1:reals:floor}{parte entera} en $\intcc{0}{3}$ es una \omterm{def:b1:integration:step}{función escalonada}
disfrazada: partiendo en sus saltos,
\[
\int_0^3 \lfloor t \rfloor\,\dd t
= \int_0^1 0 + \int_1^2 1 + \int_2^3 2 = 0 + 1 + 2 = 3 ,
\]
y los valores \emph{en} los puntos de salto $1, 2$ son irrelevantes:
cambiar una función en finitos puntos no cambia ninguna \omterm{thm:b1:integration:def}{integral} (las
\omterm{def:b1:integration:step}{funciones escalonadas} del encuadre no se ven afectadas). Ese es todo
el contenido de la extensión «\omterm{def:b1:continuity:continuous}{continua} a trozos»: córtese en las
finitas discontinuidades, intégrese cada trozo \omterm{def:b1:continuity:continuous}{continuo} y súmese ---
Chasles como definición.
\end{example}

\begin{example}[La definición calcula, una vez]\label{ex:b1:integration:defcompute}
Sea $f(x) = x$ en $\intcc{0}{1}$ y córtese en $n$ trozos iguales. Las
mejores \omterm{def:b1:integration:step}{funciones escalonadas} constantes en los trozos son
$\varphi = \frac{k-1}{n}$ y $\psi = \frac kn$ en el $k$-ésimo trozo,
con
\[
\int_0^1 \varphi = \sum_{k=1}^{n} \frac{k-1}{n}\cdot\frac1n
= \frac{n-1}{2n},
\qquad
\int_0^1 \psi = \sum_{k=1}^{n} \frac{k}{n}\cdot\frac1n
= \frac{n+1}{2n} .
\]
Toda \omterm{thm:b1:integration:def}{integral} inferior es $\leq I_-(f) \leq I_+(f) \leq$ que toda
superior, luego $\frac{n-1}{2n} \leq I_-(f) \leq I_+(f) \leq
\frac{n+1}{2n}$ para todo $n$: las dos se aprietan sobre $\frac12$, y
$\int_0^1 x\,\dd x = \frac12$ sale directamente de la definición. La
idea de cierre: esta es la primera y la última vez que integramos a
partir de la definición --- el teorema fundamental de más abajo
sustituye todos esos cálculos por una consulta de primitiva, que es
todo el sentido económico de este capítulo.
\end{example}

\begin{theorem}[Propiedades]\label{thm:b1:integration:props}
Para $f, g$ \omterm{def:b1:continuity:continuous}{continuas} (o \omterm{def:b1:continuity:continuous}{continuas} a trozos) en $\intcc{a}{b}$ y
$\lambda \in \R$:
\begin{enumerate}
  \item linealidad: $\int (f + \lambda g) = \int f + \lambda \int g$;
  \item monotonía: $f \leq g \implies \int_a^b f \leq \int_a^b g$;
        y $\bigl|\int_a^b f\bigr| \leq \int_a^b \abs f \leq (b -
        a)\, \sup\abs f$;
  \item Chasles: $\int_a^b = \int_a^c + \int_c^b$ (con el convenio
        $\int_b^a = -\int_a^b$, válido para cualquier orden de los
        límites);
  \item positividad estricta: si $f$ es \omterm{def:b1:continuity:continuous}{continua}, $f \geq 0$ y
        $\int_a^b f = 0$, entonces $f = 0$ en todo $\intcc{a}{b}$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1)--(3) pasan de las \omterm{def:b1:integration:step}{funciones escalonadas} al límite por medio de la
definición con sup e ínf. La linealidad merece los detalles una vez:
dado $\varepsilon > 0$, encuádrense $\varphi_f \leq f \leq \psi_f$ y
$\varphi_g \leq g \leq \psi_g$ con huecos $\leq \varepsilon$
(\cref{thm:b1:integration:approx}). Para $\lambda \geq 0$,
$\varphi_f + \lambda\varphi_g \leq f + \lambda g \leq \psi_f +
\lambda\psi_g$ es un encuadre por \omterm{def:b1:integration:step}{funciones escalonadas} con hueco
$\leq (1 + \lambda)\varepsilon$, y sus \omterm{thm:b1:integration:def}{integrales} escalonadas valen
$\int \varphi_f + \lambda\int\varphi_g$, etc.\
(\cref{prop:b1:integration:stepprops}): haciendo $\varepsilon \to 0$
se aprieta $\int(f + \lambda g)$ sobre $\int f + \lambda\int g$. Para
$\lambda < 0$, multiplicar por $\lambda$ \emph{invierte} el encuadre
de $g$ ---la \omterm{def:b1:integration:step}{función escalonada} inferior de $\lambda g$ es
$\lambda \psi_g$--- y el mismo apretón funciona con los papeles
intercambiados. La cota $\abs{\int f} \leq \int \abs f$ sale de
$-\abs f \leq f \leq \abs f$ y de la monotonía.

(4) Por contrarrecíproco: si $f(x_0) = m > 0$, la \omterm{def:b1:continuity:continuous}{continuidad}
proporciona un subintervalo de longitud $\eta > 0$ en el que
$f \geq \frac m2$; la \omterm{def:b1:integration:step}{función escalonada} que vale $\frac m2$ ahí y $0$
fuera es $\leq f$, luego $\int f \geq \frac{m\eta}{2} > 0$.
\end{proof}

\begin{example}[Chasles en acción: integrales con valores absolutos]\label{ex:b1:integration:chasleswork}
Para integrar un valor absoluto, córtese donde cambia el signo.
\[
\int_0^2 \abs{x - 1}\,\dd x
= \int_0^1 (1 - x)\,\dd x + \int_1^2 (x - 1)\,\dd x
= \frac12 + \frac12 = 1 ,
\]
y, cortando $\intcc{0}{2\pi}$ en $\pi$:
\[
\int_0^{2\pi} \abs{\sin t}\,\dd t
= \int_0^{\pi} \sin t\,\dd t - \int_{\pi}^{2\pi} \sin t\,\dd t
= 2 + 2 = 4 ,
\]
mientras que $\int_0^{2\pi} \sin t\,\dd t = 0$: la cancelación es real,
y por eso el \omterm{def:b1:logic:statement}{enunciado} de positividad estricta
(\cref{thm:b1:integration:props}~(4)) lleva la hipótesis
$f \geq 0$ --- sin ella, una \omterm{thm:b1:integration:def}{integral} nula no demuestra nada sobre
$f$. La idea de cierre: $\int \abs f$ mide \emph{área} y $\int f$ mide
\emph{balance con signo}; la desigualdad
$\abs{\int f} \leq \int\abs f$ es el registro exacto de lo que la
cancelación puede destruir.
\end{example}

\section{El teorema fundamental del cálculo}

\begin{theorem}[Teorema fundamental del cálculo]\label{thm:b1:integration:ftc}
Sea $f$ \omterm{def:b1:continuity:continuous}{continua} en un \omterm{prop:b1:reals:intervals}{intervalo} $I$ y sea $a \in I$. La función
\[
F(x) = \int_a^x f(t)\, \dd t
\]
es de clase $C^1$ en $I$, con $F' = f$: toda función \omterm{def:b1:continuity:continuous}{continua} en un
\omterm{prop:b1:reals:intervals}{intervalo} tiene primitivas. En consecuencia, para \emph{cualquier}
primitiva $G$ de $f$:\index{teorema fundamental del cálculo}
\[
\int_a^b f(t)\,\dd t = G(b) - G(a) .
\]
\end{theorem}

\begin{proof}
Fíjense $x_0 \in I$ y $\varepsilon > 0$; la \omterm{def:b1:continuity:continuous}{continuidad} en $x_0$
proporciona $\delta$ con $\abs{f(t) - f(x_0)} \leq \varepsilon$ para
$\abs{t - x_0} \leq \delta$. Para $0 < \abs{h} \leq \delta$ (y
$x_0 + h \in I$), Chasles da
\[
\frac{F(x_0 + h) - F(x_0)}{h} - f(x_0)
= \frac 1h \int_{x_0}^{x_0+h} \bigl(f(t) - f(x_0)\bigr)\dd t ,
\]
cuyo valor absoluto es
$\leq \frac{1}{\abs h}\cdot \abs h\, \varepsilon = \varepsilon$ (cota
(2), válida para cualquier orden de los límites). Luego
$F'(x_0) = f(x_0)$; y $F' = f$ es \omterm{def:b1:continuity:continuous}{continua}: $F$ es de clase $C^1$. Si
también $G' = f$, entonces $(G - F)' = 0$ en el \omterm{prop:b1:reals:intervals}{intervalo}, luego
$G = F + c$ (\cref{cor:b1:derivative:monotone}), y
$G(b) - G(a) = F(b) - F(a) = \int_a^b f$.
\end{proof}

\begin{example}[La simetría antes que el cálculo]\label{ex:b1:integration:parity}
En un \omterm{prop:b1:reals:intervals}{intervalo} simétrico, la paridad hace el trabajo: si $f$ es
impar, el cambio $t \mapsto -t$ manda $\int_{-a}^{0} f$ a
$-\int_0^a f$, luego
\[
\int_{-a}^{a} f(t)\,\dd t = 0 ;
\qquad\text{si $f$ es par,}\quad
\int_{-a}^{a} f = 2\int_0^a f .
\]
Así, $\int_{-1}^{1} \frac{t^3\cos t}{1 + t^4}\,\dd t = 0$ sin ninguna
primitiva a la vista (el integrando es impar), y
$\int_{-\pi}^{\pi} t^2\cos t\,\dd t = 2\int_0^\pi t^2\cos t\, \dd t$.
Compruébese la simetría antes de echar mano de las técnicas: la
\omterm{thm:b1:integration:def}{integral} más rápida es la que nunca se calcula.
\end{example}

\begin{example}[Reconocer una derivada a simple vista]\label{ex:b1:integration:halfangle}
Calcúlese $\displaystyle\int_0^{\pi/2} \frac{\dd x}{1 + \cos x}$. La
identidad del ángulo mitad $1 + \cos x = 2\cos^2\frac x2$ convierte el
integrando en $\frac{1}{2}\bigl(1 + \tan^2\frac x2\bigr)$, que es
exactamente la \omterm{def:b1:derivative:def}{derivada} de $\tan\frac x2$:
\[
\int_0^{\pi/2} \frac{\dd x}{1 + \cos x}
= \Bigl[\tan\frac x2\Bigr]_0^{\pi/2} = \tan\frac\pi4 = 1 .
\]
No ha hecho falta maquinaria de \omterm{thm:b1:integration:parts}{cambios de variable} --- solo el
reflejo de leer un integrando como la \omterm{def:b1:derivative:def}{derivada} de alguien, y el
teorema fundamental hace el resto. (La herramienta sistemática que hay
detrás de esas \omterm{thm:b1:integration:def}{integrales} trigonométricas, el cambio
$t = \tan\frac x2$, pertenece al utillaje estándar construido a partir
del~\cref{thm:b1:integration:parts}~(2).)
\end{example}

\begin{example}[Funciones definidas por integrales]\label{ex:b1:integration:definedby}
El teorema fundamental fabrica funciones. Sea
\[
F(x) = \int_0^x \eu^{-t^2}\,\dd t .
\]
Ninguna combinación de funciones clásicas tiene \omterm{def:b1:derivative:def}{derivada}
$\eu^{-t^2}$ (un teorema de Liouville, admitido); y, sin embargo, $F$
existe, es de clase $C^1$ con $F'(x) = \eu^{-x^2} > 0$,
estrictamente creciente, impar (sustitúyase $t \mapsto -t$) y
acotada: para $x \geq 1$,
\[
F(x) - F(1) = \int_1^x \eu^{-t^2}\dd t
\leq \int_1^x \eu^{-t}\dd t \leq \eu^{-1} ,
\]
luego $F \leq F(1) + \eu^{-1} \leq 1 + \eu^{-1}$. (El límite exacto,
$\frac{\sqrt\pi}{2}$, se calcula con \omterm{thm:b1:integration:def}{integrales} dobles en el volumen
del tercer año.) Regla de la cadena para límites móviles:
$\frac{\dd}{\dd x}\int_x^{x^2} \eu^{-t^2}\dd t =
2x\,\eu^{-x^4} - \eu^{-x^2}$. La idea de cierre: la integración
\emph{crea} funciones nuevas a partir de las viejas, con todas sus
propiedades legibles en el integrando --- la primitiva que no se puede
escribir es aun así una función que se controla por completo.
\end{example}

\begin{example}[Estimar sin evaluar]\label{ex:b1:integration:estimate}
Las \omterm{thm:b1:integration:def}{integrales} $R_n = \int_0^1 \frac{t^n}{1 + t}\,\dd t$ no tienen
ninguna expresión \omterm{def:b1:topology:closed}{cerrada} agradable y, aun así, la monotonía las
sujeta con precisión: en $\intcc{0}{1}$,
$\frac12 \leq \frac{1}{1+t} \leq 1$, luego
\[
\frac{1}{2(n+1)} = \frac12\int_0^1 t^n\,\dd t
\;\leq\; R_n \;\leq\; \int_0^1 t^n \,\dd t = \frac{1}{n+1} :
\]
el orden exacto de decrecimiento ($R_n \sim$ un múltiplo de
$\frac1n$; de hecho, $R_n \sim \frac{1}{2n}$) con dos líneas y sin
ninguna primitiva. Los problemas del fin de semana de este capítulo y
del siguiente corren sobre encuadres exactamente así --- el primer
instinto del analista ante una \omterm{thm:b1:integration:def}{integral} debe ser \emph{acotarla} y,
solo después, si hace falta, calcularla.
\end{example}

\begin{example}[Valores medios]\label{ex:b1:integration:average}
La \emph{media} de una $f$ \omterm{def:b1:continuity:continuous}{continua} en $\intcc{a}{b}$ es
$\frac{1}{b-a}\int_a^b f$. Para el arco del seno:
\[
\frac{1}{\pi}\int_0^\pi \sin t\,\dd t
= \frac{1}{\pi}\bigl[-\cos t\bigr]_0^\pi = \frac{2}{\pi}
\approx 0.637 :
\]
un arco positivo completo no promedia $\frac12$, sino $\frac2\pi$ ---
la curva pasa más tiempo arriba de lo que pasaría un triángulo. Por
el~\cref{exo:b1:integration:11} (teorema del valor medio para
\omterm{thm:b1:integration:def}{integrales}, con $g = 1$), la media es un \emph{valor}:
$\sin c = \frac2\pi$ para algún $c \in \intoo{0}{\pi}$. Y por las
\omterm{thm:b1:integration:riemann}{sumas de Riemann} de este capítulo, la media es el límite de
promedios ordinarios de $n$ muestras --- el puente entre la media
discreta de unos datos y la media \omterm{def:b1:continuity:continuous}{continua} de una señal, que es como
entra la \omterm{thm:b1:integration:def}{integral} en la física.
\end{example}

\begin{theorem}[Integración por partes; cambio de variable]\label{thm:b1:integration:parts}
\begin{enumerate}
  \item Si $u, v$ son de clase $C^1$ en
        $\intcc{a}{b}$:\index{integración por partes}
        \[
        \int_a^b u'v = \bigl[uv\bigr]_a^b - \int_a^b uv' .
        \]
  \item Si $\varphi$ es de clase $C^1$ en $\intcc{\alpha}{\beta}$ y
        $f$ es \omterm{def:b1:continuity:continuous}{continua} en $\varphi(\intcc{\alpha}{\beta})$:
        \index{cambio de variable}
        \[
        \int_{\alpha}^{\beta} f\bigl(\varphi(t)\bigr)\,\varphi'(t)\,
        \dd t = \int_{\varphi(\alpha)}^{\varphi(\beta)} f(x)\, \dd x .
        \]
\end{enumerate}
\end{theorem}

\begin{proof}
(1) $(uv)' = u'v + uv'$; intégrese en $\intcc{a}{b}$ y aplíquese el
teorema fundamental a la función de clase $C^1$, $uv$.

(2) Sea $F$ una primitiva de $f$ en el \omterm{prop:b1:reals:intervals}{intervalo} imagen
(\cref{thm:b1:integration:ftc}). Entonces
$(F \circ \varphi)' = (f \circ \varphi)\,\varphi'$ (regla de la
cadena), luego los dos miembros valen
$F(\varphi(\beta)) - F(\varphi(\alpha))$.
\end{proof}

\begin{example}\label{ex:b1:integration:parts}
$\int_0^1 t\,\eu^t \dd t = \bigl[t\,\eu^t\bigr]_0^1 - \int_0^1
\eu^t\dd t = \eu - (\eu - 1) = 1$. Y con el \omterm{thm:b1:integration:parts}{cambio de variable}
$x = \sin t$ ($t \in \intcc{0}{\frac\pi2}$):
\[
\int_0^1 \sqrt{1 - x^2}\, \dd x
= \int_0^{\pi/2} \cos t \cdot \cos t \, \dd t
= \int_0^{\pi/2} \frac{1 + \cos 2t}{2}\, \dd t = \frac\pi4 ,
\]
--- un cuarto del disco unidad, como exige la geometría.
(La linealización del~\cref{met:b1:complex:trig} en acción.)
\end{example}

\begin{remark}[Errores frecuentes en el cálculo integral]
(i) \emph{Los \omterm{thm:b1:integration:parts}{cambios de variable} deben ser de clase $C^1$ en todo el
\omterm{prop:b1:reals:intervals}{intervalo}}: el cambio $x = \frac1t$ es ilegal a caballo de $0$;
aplicado a ciegas a $\int_{-1}^{1}\frac{\dd x}{1 + x^2}$
«demuestra» que la \omterm{thm:b1:integration:def}{integral} es igual a su propia opuesta. Cuando un
\omterm{thm:b1:integration:parts}{cambio de variable} tiene una singularidad, córtese antes el \omterm{prop:b1:reals:intervals}{intervalo}
(Chasles), háganse los cambios en cada trozo y solo entonces
recompóngase. (ii) \emph{Las primitivas logarítmicas necesitan valores
absolutos}: $\int \frac{\dd x}{x - 2} = \ln\abs{x - 2} + C$ a cada
lado de $2$ por separado --- escribir $\ln(x - 2)$ en $\intoo{0}{1}$
es escribir el logaritmo de un número negativo; y la constante $C$
puede ser distinta a los dos lados de la singularidad. (iii) \emph{Una
\omterm{thm:b1:integration:def}{integral} nula no mata a la función}: $\int_0^{2\pi}\sin = 0$; hace
falta la positividad del integrando antes de concluir $f = 0$
(\cref{ex:b1:integration:chasleswork}). (iv) \emph{Las \omterm{thm:b1:integration:riemann}{sumas de
Riemann} hay que calibrarlas}: en
$\frac{b-a}{n}\sum f\bigl(a + k\frac{b-a}{n}\bigr)$, el paso de fuera
y los puntos de dentro deben corresponder a la misma subdivisión ---
el error frecuente es una suma $\sum_{k=1}^{n} f\bigl(\frac kn\bigr)$
\emph{sin} el factor $\frac1n$, que diverge en vez de converger a
$\int_0^1 f$. Lista de comprobación antes de invocar
el~\cref{thm:b1:integration:riemann}: sáquese el factor $\frac1n$,
reescríbase el sumando como $f$ de $\frac kn$, nómbrese $f$ y
compruébese su \omterm{def:b1:continuity:continuous}{continuidad}.
\end{remark}

\begin{example}[Adivinar, derivar, ajustar]\label{ex:b1:integration:guessdiff}
¿Cuánto vale $\int_1^x (\ln t)^2\,\dd t$? Adivínese una primitiva de
la forma $t\,P(\ln t)$ con $P$ \omterm{def:b1:poly:def}{polinomio} y derívese:
\[
\bigl(t\,P(\ln t)\bigr)' = P(\ln t) + P'(\ln t) .
\]
Hace falta $P(u) + P'(u) = u^2$: tómese $P(u) = u^2 - 2u + 2$
(igualando coeficientes hacia abajo desde $u^2$). Por tanto,
\[
\int_1^x (\ln t)^2\,\dd t
= \bigl[t\bigl((\ln t)^2 - 2\ln t + 2\bigr)\bigr]_1^x
= x(\ln x)^2 - 2x\ln x + 2x - 2 ,
\]
un resultado al que se llega, si no, con dos \omterm{thm:b1:integration:parts}{integraciones por partes}.
La idea de cierre: para integrandos de la forma (\omterm{def:b1:poly:def}{polinomio} en $\ln t$)
o (\omterm{def:b1:poly:def}{polinomio} por $\eu^{\lambda t}$), la primitiva tiene la misma
forma --- derivar una conjetura con forma convierte la integración en
álgebra lineal sobre los coeficientes, más rápida y menos propensa a
errores que las partes iteradas.
\end{example}

\begin{example}[La integral bumerán]\label{ex:b1:integration:boomerang}
Calcúlese $I = \int_0^{\pi/2} \eu^x \cos x\,\dd x$. Intégrese dos
veces por partes, derivando cada vez el factor trigonométrico:
\[
I = \bigl[\eu^x \sin x\bigr]_0^{\pi/2}
- \int_0^{\pi/2} \eu^x \sin x\,\dd x
= \eu^{\pi/2} - J,
\qquad
J = \bigl[-\eu^x\cos x\bigr]_0^{\pi/2}
+ \int_0^{\pi/2} \eu^x\cos x\,\dd x = 1 + I .
\]
La \omterm{thm:b1:integration:def}{integral} ha vuelto sobre sí misma: $I = \eu^{\pi/2} - 1 - I$, de
donde
\[
I = \frac{\eu^{\pi/2} - 1}{2} .
\]
La idea de cierre: cuando el integrando es un producto de dos
funciones que se reproducen al derivar ($\eu^{ax}$, $\cos bx$,
$\sin bx$), dos \omterm{thm:b1:integration:parts}{integraciones por partes} producen una ecuación lineal
para la \omterm{thm:b1:integration:def}{integral} desconocida --- resuélvase en lugar de integrar;
equivalentemente, pásese por $\eu^{(a + \iu b)x}$
(\cref{ch:b1:complex}) y tómense partes reales. Los dos caminos dan la
misma respuesta, y comprobar que lo hacen es una prueba de coherencia
gratuita.
\end{example}

\section{Sumas de Riemann}

\begin{theorem}[Sumas de Riemann]\label{thm:b1:integration:riemann}
Sea $f$ \omterm{def:b1:continuity:continuous}{continua} en $\intcc{a}{b}$. Entonces\index{sumas de Riemann}
\[
S_n = \frac{b - a}{n} \sum_{k=0}^{n-1}
f\Bigl(a + k\,\frac{b-a}{n}\Bigr)
\xrightarrow[n \to \infty]{} \int_a^b f(t)\, \dd t ,
\]
y lo mismo con cualquier punto de evaluación dentro de los subintervalos.
\end{theorem}

\begin{proof}
$S_n$ es la \omterm{thm:b1:integration:def}{integral} de la \omterm{def:b1:integration:step}{función escalonada} $\varphi_n$ igual a
$f(a + k\frac{b-a}{n})$ en el $k$-ésimo subintervalo. Dado
$\varepsilon > 0$, la \omterm{def:b1:continuity:uniform}{continuidad uniforme} (Heine) proporciona
$\delta$; para $n > \frac{b-a}{\delta}$, todo punto de un
subintervalo dista menos de $\delta$ de su punto de evaluación, luego
$\abs{f - \varphi_n} \leq \varepsilon$ en $\intcc{a}{b}$, de donde
\[
\Bigl| \int_a^b f - S_n \Bigr|
= \Bigl| \int_a^b (f - \varphi_n) \Bigr|
\leq (b-a)\,\varepsilon . \qedhere
\]
\end{proof}

\begin{example}\label{ex:b1:integration:riemannexample}
$\displaystyle\sum_{k=1}^{n} \frac{1}{n + k} = \frac 1n
\sum_{k=1}^{n} \frac{1}{1 + k/n} \xrightarrow[n\to\infty]{}
\int_0^1 \frac{\dd x}{1 + x} = \ln 2$: un límite invisible para las
acotaciones elementales y transparente como suma de Riemann.
\end{example}

\begin{example}[Una segunda suma de Riemann, con calibración]\label{ex:b1:integration:riemann2}
Hállese $\lim_{n\to\infty} \sum_{k=1}^{n} \dfrac{n}{(n+k)^2}$.
Calíbrese:
\[
\sum_{k=1}^{n} \frac{n}{(n+k)^2}
= \frac{1}{n}\sum_{k=1}^{n} \frac{n^2}{(n+k)^2}
= \frac1n \sum_{k=1}^{n} \frac{1}{\bigl(1 + \frac kn\bigr)^2} ,
\]
una suma de Riemann de la función \omterm{def:b1:continuity:continuous}{continua} $f(x) = \frac{1}{(1+x)^2}$
en $\intcc{0}{1}$: el límite es
\[
\int_0^1 \frac{\dd x}{(1 + x)^2}
= \Bigl[-\frac{1}{1+x}\Bigr]_0^1 = \frac12 .
\]
La idea de cierre: todo el arte está en la línea de en medio ---
fórcese el sumando a la forma $f(\frac kn)$ al precio de extraer
exactamente un factor $\frac1n$; una vez que la forma es la correcta,
el teorema hace el análisis y el teorema fundamental hace la
aritmética.
\end{example}

\begin{remark}[Dónde funciona la integral a continuación]
Cada una de las construcciones del capítulo tiene continuación. Las
\omterm{thm:b1:integration:riemann}{sumas de Riemann} vuelven en el~\cref{ch:b1:series} como puente entre
las series y las \omterm{thm:b1:integration:def}{integrales} (comparación de
$\sum \frac{1}{n^\alpha}$ con $\int \frac{\dd t}{t^\alpha}$); el resto
\omterm{thm:b1:integration:def}{integral} es la forma más fina de la fórmula de Taylor en
el~\cref{ch:b1:taylor}; y la definición basada en el $\sup$ es el
prototipo de la \omterm{thm:b1:integration:def}{integral} de Lebesgue del volumen del tercer año, donde
las mismas tres propiedades (linealidad, monotonía y un teorema de
convergencia) se reconstruyen sobre una clase de funciones mucho
mayor. Y el problema del fin de semana de más abajo convierte la
\omterm{thm:b1:integration:parts}{integración por partes} en aritmética: la irracionalidad de $\pi^2$.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:integration:1}
Calcúlense: $\displaystyle\int_0^1 \frac{\dd x}{x^2 - 4}$
\emph{(fracciones simples, \cref{ch:b1:fractions})};
$\displaystyle\int_1^{\eu} \ln t \,\dd t$;
$\displaystyle\int_0^{\pi} t \sin t\, \dd t$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:integration:2}
Calcúlense $\displaystyle\int_0^{1} \frac{t}{(t^2+1)^2}\,\dd t$
(\omterm{thm:b1:integration:parts}{cambio de variable}) y
$\displaystyle\int_0^{\pi/2} \cos^3 t\, \dd t$ (escríbase
$\cos^3 = \cos(1 - \sin^2)$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:integration:3}
Hállense los límites, como \omterm{thm:b1:integration:riemann}{sumas de Riemann}:
\[
u_n = \sum_{k=1}^{n} \frac{n}{n^2 + k^2},
\qquad
v_n = \frac{1}{n}\sqrt[n]{\frac{(2n)!}{n!\,n^n}}
\ \emph{(tómense logaritmos)}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:integration:4}
Sea $f$ \omterm{def:b1:continuity:continuous}{continua} en $\intcc{0}{1}$. Calcúlese
$\lim_{n\to\infty} \int_0^1 x^n f(x)\,\dd x$. \emph{(Córtese
$\intcc{0}{1}$ en $1 - \delta$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:integration:5}
(Cauchy--Schwarz) Para $f, g$ \omterm{def:b1:continuity:continuous}{continuas} en $\intcc{a}{b}$, demuéstrese
\[
\Bigl(\int_a^b fg\Bigr)^{\!2} \leq \int_a^b f^2 \cdot \int_a^b g^2 ,
\]
desarrollando $\int_a^b (f + \lambda g)^2 \geq 0$ como trinomio en
$\lambda$. ¿Cuándo hay igualdad?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:integration:6}
Sea $f$ \omterm{def:b1:continuity:continuous}{continua} en $\R$ y $T$-periódica. Demuéstrese que
$\int_a^{a+T} f$ no depende de $a$, y que
$\frac1x \int_0^x f(t)\,\dd t \to \frac 1T \int_0^T f$ cuando
$x \to +\infty$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:integration:7}
Para $f$ \omterm{def:b1:continuity:continuous}{continua} en $\intcc{0}{1}$ con $\int_0^1 f = \frac12$,
demuéstrese que $f$ tiene un punto fijo en $\intcc{0}{1}$.
\emph{(Intégrese $f(x) - x$ y úsese la positividad estricta,
\cref{thm:b1:integration:props}~(4), por su contrarrecíproco combinado
con el teorema del valor intermedio.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:integration:8}
(\omterm{thm:b1:integration:def}{Integrales} de Wallis) Sea $W_n = \int_0^{\pi/2} \sin^n t\,\dd t$.
\begin{enumerate}
  \item Demuéstrese la recurrencia $n W_n = (n-1) W_{n-2}$
        ($n \geq 2$) por partes, y calcúlense $W_0, W_1$ y después
        $W_{2p}$ y $W_{2p+1}$ en forma \omterm{def:b1:topology:closed}{cerrada}.
  \item Demuéstrese que $(W_n)$ es decreciente con
        $\frac{W_{n+1}}{W_n} \to 1$; demuéstrese que la cantidad
        $(n+1)\,W_{n+1} W_n$ es constante e igual a $\frac\pi2$; y
        dedúzcase el equivalente $W_n \sim \sqrt{\dfrac{\pi}{2n}}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:integration:9}
(Niven: $\pi$ es irracional) Supóngase $\pi = \frac ab$ con
$a, b \in \N^*$ y póngase, para un $n$ que se elegirá,
\[
P(x) = \frac{x^n (a - bx)^n}{n!},
\qquad
I_n = \int_0^{\pi} P(x)\sin x\, \dd x .
\]
\begin{enumerate}
  \item Demuéstrese que $0 < I_n \leq \pi\,\frac{(\pi a)^n}{n!}$,
        que es $< 1$ para $n$ grande.
  \item Demuéstrese que $P$ y todas sus \omterm{def:b1:derivative:def}{derivadas} toman valores
        \emph{enteros} en $0$ y en $\pi = \frac ab$.
        \emph{(Desarrollo binomial: los coeficientes de $P$ por $k!$
        son enteros para $k \geq n$; y $P(\pi - x) = P(x)$.)}
  \item Póngase $Q = P - P'' + P^{(4)} - \dots$ (suma finita).
        Compruébese que
        $\bigl(Q'\sin x - Q\cos x\bigr)' = P \sin x$, y dedúzcase
        que $I_n = Q(\pi) + Q(0)$ es un entero.
  \item Conclúyase.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:integration:10}
Sea $f$ de clase $C^1$ en $\intcc{a}{b}$. Demuéstrese el límite de
tipo Riemann--Lebesgue
\[
\int_a^b f(t)\sin(\lambda t)\,\dd t \xrightarrow[\lambda \to
+\infty]{} 0
\]
integrando por partes. Demuéstrese después de nuevo para $f$ solo
\omterm{def:b1:continuity:continuous}{continua}, por aproximación uniforme con \omterm{def:b1:integration:step}{funciones escalonadas}
(\cref{thm:b1:integration:approx}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:integration:11}
(Teorema del valor medio para \omterm{thm:b1:integration:def}{integrales}) Sean $f, g$ \omterm{def:b1:continuity:continuous}{continuas} en
$\intcc{a}{b}$ con $g \geq 0$. Demuéstrese que existe
$c \in \intcc{a}{b}$ con
\[
\int_a^b f(t)\,g(t)\,\dd t = f(c)\int_a^b g(t)\,\dd t ,
\]
y véase con un ejemplo que no se puede prescindir de la hipótesis
$g \geq 0$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:integration:12}
(Los momentos fuerzan ceros) Sea $f$ \omterm{def:b1:continuity:continuous}{continua} en $\intcc{a}{b}$ con
\[
\int_a^b f(t)\,t^k\,\dd t = 0 \qquad \text{para } k = 0, 1,
\dots, n .
\]
Demuéstrese que $f$ se anula en $n + 1$ puntos distintos de
$\intoo{a}{b}$. \emph{(Si $f$ cambia de signo solo en
$z_1 < \dots < z_m$ con $m \leq n$, intégrese $f$ contra
$P(t) = (t - z_1) \cdots (t - z_m)$ y úsese la positividad estricta.)}
\end{exercise}

\begin{remark}[Perspectivas dentro de este volumen]
Tres capítulos posteriores se apoyan directamente en este.
\Cref{ch:b1:taylor} lleva el resto \omterm{thm:b1:integration:def}{integral} --- la más fina de las tres
fórmulas de Taylor es una \omterm{thm:b1:integration:parts}{integración por partes} iterada $n$ veces.
\Cref{ch:b1:series} convierte el encuadre de sumas por \omterm{thm:b1:integration:def}{integrales} en el
criterio decisivo para $\sum n^{-\alpha}$, y su problema del fin de
semana afina ese encuadre hasta la constante de Euler.
\Cref{ch:b1:curves} vuelve geométrica la \omterm{thm:b1:integration:def}{integral}: la longitud de un
arco parametrizado es $\int \sqrt{x'(t)^2 + y'(t)^2}\,\dd t$, una
\omterm{thm:b1:integration:def}{integral} de una función \omterm{def:b1:continuity:continuous}{continua} en un segmento --- justo el objeto
construido aquí, sin necesidad de teoría impropia alguna. Y el hecho
más reutilizado será el más humilde:
$\bigl|\int f\bigr| \leq (b - a)\sup\abs{f}$, la desigualdad que
convierte toda estimación puntual en una estimación \omterm{thm:b1:integration:def}{integral}.
\end{remark}

\section{Problema: la máquina integral de irracionalidad}

\begin{problem}[{Problema del fin de semana --- $\eu$ y $\pi^2$ son
irracionales, $\eu$ con seis decimales, y $\frac{22}{7} > \pi$ con
demostración}]
\label{pb:b1:integration:1}
Un solo mecanismo mueve todo este problema: \emph{una expresión que
tiene que ser un entero positivo y que, sin embargo, es
demostrablemente menor que $1$ no puede existir}.
\Cref{exo:b1:integration:9} (Niven) lo hizo funcionar una vez para
demostrar $\pi \notin \Q$; aquí lo industrializamos. La máquina
necesita tres piezas: una entrada de \emph{integralidad} (valores en
los extremos de \omterm{def:b1:poly:def}{polinomios} bien elegidos), una entrada de
\emph{pequeñez} (un factor $\frac{1}{n!}$ que aplasta la \omterm{thm:b1:integration:def}{integral}) y
un \emph{puente} (la \omterm{thm:b1:integration:parts}{integración por partes}) que las conecta.
Demostramos que $\eu$ es irracional y lo calculamos con error
certificado, demostramos el teorema más fino de Legendre de que
$\pi^2$ es irracional, y terminamos con la \omterm{thm:b1:integration:def}{integral} más encantadora
del análisis: $\int_0^1 \frac{x^4(1 - x)^4}{1 + x^2}\dd x =
\frac{22}{7} - \pi$, que encuadra $\pi$ a mano.\index{irracionalidad!de
pi cuadrado@de $\pi^2$}\index{irracionalidad!de e@de $\eu$}

\medskip
\noindent\textbf{Parte I --- Combustible.}
\begin{enumerate}
  \item Demuéstrese que $\dfrac{c^{\,n}}{n!} \to 0$ para todo
        $c > 0$ fijo \emph{(más allá de $n \geq 2c$, cada paso al
        menos \omterm{def:b1:arith:divides}{divide} por dos el término)}.
  \item (\omterm{thm:b1:integration:def}{Integrales} beta) Demuéstrese, por inducción sobre $m$ con
        \omterm{thm:b1:integration:parts}{integración por partes}:
        \[
        \int_0^1 x^{\,k}\,(1 - x)^{\,m}\,\dd x =
        \frac{k!\,m!}{(k + m + 1)!}
        \qquad (k, m \in \N).
        \]
  \item Dedúzcase $\displaystyle\int_0^1 \bigl(x(1-x)\bigr)^n \dd x
        = \frac{1}{(2n+1)\binom{2n}{n}}$ y ---comparando con la cota
        $x(1 - x) \leq \frac14$--- la estimación
        $\binom{2n}{n} \geq \dfrac{4^n}{2n+1}$, acorde con
        $\binom{2n}{n}^{1/n} \to 4$ del~\cref{pb:b1:seq:1}.
  \item Demuéstrese el lema de pequeñez que se usa dos veces más
        abajo: para toda función \omterm{def:b1:continuity:continuous}{continua} $g > 0$ en $\intoo{0}{1}$,
        \[
        0 < \int_0^1 \bigl(x(1-x)\bigr)^n g(x)\,\dd x
        \leq \frac{\sup_{\intcc{0}{1}}\abs g}{4^{\,n}} .
        \]
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- $\eu$: irracionalidad y después seis
decimales.} Póngase $A_n = \displaystyle\int_0^1 x^n \eu^x \dd x$.
\begin{enumerate}[resume]
  \item Calcúlense $A_0$ y $A_1$, demuéstrese la recurrencia
        $A_n = \eu - n A_{n-1}$ y las cotas
        $0 < A_n \leq \dfrac{\eu}{n+1}$.
  \item Véase por inducción que $A_n = \alpha_n + \beta_n \eu$ con
        $\alpha_n, \beta_n \in \Z$.
  \item Dedúzcase que $\eu$ es irracional \emph{(si $\eu = \frac
        pq$, entonces $q A_n$ es un entero atrapado en
        $\intoo{0}{1}$ para $n$ grande)}. Compárese con la
        demostración del~\cref{exo:b1:seq:9}: misma conclusión,
        distinto combustible.
  \item Demuéstrese, por inducción y \omterm{thm:b1:integration:parts}{integración por partes}, la
        fórmula exacta del resto
        \[
        \eu = \sum_{k=0}^{n} \frac{1}{k!} + R_n,
        \qquad
        R_n = \frac{1}{n!}\int_0^1 (1 - t)^n\,\eu^{\,t}\,\dd t,
        \qquad
        \frac{1}{(n+1)!} \leq R_n \leq \frac{\eu}{(n+1)!} .
        \]
  \item Tómese $n = 9$: acótese $R_9$ usando $\eu < 2.75$ (a partir
        de $b_2 = 2.75$ en el~\cref{ex:b1:seq:e}), evalúese la suma y
        conclúyase el encuadre certificado
        $2.7182818 \leq \eu \leq 2.7182823$ --- seis decimales,
        $\eu \approx 2.718282$, con demostración.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El teorema de Legendre: $\pi^2$ es
irracional.} Sea $f(x) = \dfrac{x^n (1 - x)^n}{n!}$, y supóngase
$\pi^2 = \frac ab$ con $a, b \in \N^*$.
\begin{enumerate}[resume]
  \item Véase que $f(1 - x) = f(x)$ y que
        $0 < f \leq \dfrac{1}{4^n\,n!}$ en $\intoo{0}{1}$.
  \item Véase que $f^{(k)}(0)$ y $f^{(k)}(1)$ son enteros para todo
        $k \geq 0$ \emph{(desarróllese $x^n(1-x)^n$ con coeficientes
        enteros; $\frac{k!}{n!} \in \Z$ para $k \geq n$; y úsese
        después la simetría)}.
  \item Defínase
        \[
        G = b^{\,n} \sum_{k=0}^{n} (-1)^k\,
        \pi^{2n - 2k} f^{(2k)} .
        \]
        Véase que $G(0)$ y $G(1)$ son enteros \emph{(cada
        $b^n \pi^{2n-2k} = a^{\,n-k}\,b^{\,k}$)}.
  \item Compruébese el telescopaje $G'' + \pi^2 G = b^n \pi^{2n+2} f
        = \pi^2 a^n f$, y después
        \[
        \frac{\dd}{\dd x}\Bigl(G'(x)\sin \pi x - \pi\,G(x)\cos
        \pi x\Bigr) = \pi^2 a^n f(x)\sin \pi x .
        \]
  \item Intégrese en $\intcc{0}{1}$ y conclúyase
        \[
        \pi\,a^n \int_0^1 f(x)\sin(\pi x)\,\dd x = G(0) + G(1)
        \in \Z ,
        \]
        un entero \emph{positivo} acotado por
        $\dfrac{\pi a^n}{4^n\,n!}$.
  \item Conclúyase con la pregunta 1 que $\pi^2$ es irracional
        (Legendre, 1794), y que esto refuerza
        el~\cref{exo:b1:integration:9}: ¿por qué la irracionalidad de
        $\pi^2$ implica la de $\pi$, y no al revés?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Entender la máquina.}
\begin{enumerate}[resume]
  \item Localícense las dos fuerzas opuestas (la integralidad de
        los datos en los extremos; la pequeñez analítica de la
        \omterm{thm:b1:integration:def}{integral}) y el puente, en las partes II y III. Explíquese
        después por qué el factor $\frac{1}{n!}$ de $f$ es el
        quid: si se quita, la integralidad sobrevive, pero ¿qué
        desigualdad muere, y para qué supuestas fracciones
        $\frac ab$ falla entonces la demostración?
  \item Efectividad: supóngase que alguien afirma que
        $\pi^2 = \frac ab$ con $a \leq 10$. Véase que la
        contradicción llega ya en $n = 7$: calcúlese
        $\pi\,(10/4)^7/7! \approx 0.38 < 1$. La máquina no solo
        refuta; refuta en una etapa fija y calculable.
  \item Compruébese el puente de forma incondicional: demuéstrese,
        con dos \omterm{thm:b1:integration:parts}{integraciones por partes}, que
        \[
        \int_0^1 x(1 - x)\sin(\pi x)\,\dd x = \frac{4}{\pi^3},
        \]
        y concíliese con la pregunta 14 en $n = 1$ (manténgase
        $\pi^2$ simbólico: la identidad telescopada dice
        $\pi^3 \int_0^1 f_1 \sin \pi x = -(f_1''(0) + f_1''(1)) = 4$).
  \item ¿Qué hace elegibles a $\eu^x$ y a $\sin \pi x$ como núcleos
        de la máquina? Identifíquese la propiedad (cada una cumple
        una ecuación diferencial lineal con coeficientes constantes,
        de modo que la \omterm{thm:b1:integration:parts}{integración por partes} repetida vuelve al
        punto de partida) y nómbrese la \omterm{def:b1:topology:closure}{frontera}: la misma máquina,
        afinada por Hermite y Lindemann, demuestra que $\eu$ y $\pi$
        son \emph{trascendentes} --- fuera de este volumen.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- $\frac{22}{7}$ frente a $\pi$, y la
moraleja.}
\begin{enumerate}[resume]
  \item Establézcase la división de \omterm{def:b1:poly:def}{polinomios}
        \[
        \frac{x^4(1-x)^4}{1 + x^2}
        = x^6 - 4x^5 + 5x^4 - 4x^2 + 4 - \frac{4}{1 + x^2},
        \]
        y dedúzcase la célebre identidad
        \[
        \int_0^1 \frac{x^4 (1-x)^4}{1 + x^2}\,\dd x
        = \frac{22}{7} - \pi .
        \]
  \item El integrando es positivo: conclúyase $\pi < \frac{22}{7}$.
        Después, acotando $\frac{1}{1+x^2}$ entre $\frac12$ y $1$ y
        usando $\int_0^1 (x(1-x))^4 = \frac{1}{630}$ (pregunta 3),
        demuéstrese
        \[
        \frac{22}{7} - \frac{1}{630} \;\leq\; \pi \;\leq\;
        \frac{22}{7} - \frac{1}{1260} ,
        \]
        es decir, $3.14126 \leq \pi \leq 3.14207$: dos decimales
        correctos, a mano.
  \item Generalícese: dividiendo $x^{4m}(1-x)^{4m}$ entre $1 + x^2$,
        véase que el resto es la constante $(-4)^m$ \emph{(trabájese
        \omterm{def:b1:complex:field}{módulo} $x^2 + 1$: $(1-x)^2 \equiv -2x$)}, dedúzcanse
        racionales $r_m$ con
        \[
        \abs{\pi - r_m} \leq 4^{\,1 - 5m} ,
        \]
        y compruébese que $m = 1$ reproduce las preguntas 20--21.
  \item Confróntense estos racionales con la teoría de aproximación
        del~\cref{pb:b1:derivative:1}: calcúlese
        $\abs{\pi - \frac{22}{7}} \approx 1.26\cdot10^{-3}$ frente a
        la garantía de Dirichlet $\frac{1}{49}$, y cítese
        $\abs{\pi - \frac{355}{113}} \approx 2.7\cdot10^{-7}$ frente
        a $\frac{1}{113^2} \approx 7.8\cdot10^{-5}$: para $\pi$
        existen aproximaciones racionales excepcionalmente buenas ---
        coherente, ya que no se sabe que $\pi$ sea mal aproximable.
  \item (La trampa del entero, abstraída) Demuéstrese el lema que lo
        unifica todo: \emph{si $x \in \R$ y existen enteros
        $a_n, b_n$ con $0 < \abs{a_n + b_n x} \to 0$, entonces $x$
        es irracional.} Enumérense sus instancias en este problema,
        en el~\cref{exo:b1:integration:9}, en el~\cref{exo:b1:seq:9}
        y en el~\cref{pb:b1:derivative:1}.
  \item Síntesis, una frase para cada punto: (i) las tres piezas de
        la máquina y dónde vive cada una en el utillaje de este
        capítulo; (ii) qué aporta la \omterm{thm:b1:integration:def}{integral} que el teorema del
        valor medio del~\cref{pb:b1:derivative:1} no podía aportar;
        (iii) el inventario de resultados extraídos (dos
        irracionalidades, una constante con seis decimales, un
        encuadre de $\pi$, una cota binomial); (iv) la \omterm{def:b1:topology:closure}{frontera}
        (Hermite, Lindemann; y la misma trampa, aplicada a
        $\zeta(2)$ y $\zeta(3)$, en la aritmética del siglo XX).
\end{enumerate}
\end{problem}
