\chapter{Estructuras algebraicas}\label{ch:b1:structures}

Las mismas reglas de cálculo reaparecen una y otra vez: enteros,
números reales, números complejos, clases de \omterm{def:b1:arith:congruence}{congruencia} y, pronto,
polinomios (\cref{ch:b1:poly}), vectores y matrices
(\cref{ch:b1:vspaces,ch:b1:matrices}). El álgebra extrae los patrones
comunes y les pone nombre: \emph{\omterm{def:b1:structures:group}{grupo}}, \emph{\omterm{def:b1:structures:ring}{anillo}}, \emph{\omterm{def:b1:structures:field}{cuerpo}}.
Demostrar un hecho una sola vez, al nivel de la estructura, lo demuestra
de golpe para todos los ejemplos.

\section{Leyes de composición}

\begin{definition}\label{def:b1:structures:law}
Una \emph{ley de composición}\index{ley de composición} en un \omterm{def:b1:logic:sets}{conjunto}
$E$ es una \omterm{def:b1:logic:map}{aplicación} $E \times E \to E$, escrita
$(x, y) \mapsto x * y$. Es \emph{asociativa} cuando $(x*y)*z = x*(y*z)$
siempre, y \emph{conmutativa} cuando $x * y = y * x$ siempre. Un
elemento $e$ es \emph{neutro} cuando $e * x = x * e = x$ para todo $x$;
entonces $x'$ es un \emph{inverso} de $x$ cuando $x * x' = x' * x = e$.
\end{definition}

\begin{proposition}[Unicidad]\label{prop:b1:structures:unique}
Una ley tiene a lo sumo un elemento neutro; para una ley asociativa con
neutro, cada elemento tiene a lo sumo un inverso.
\end{proposition}

\begin{proof}
Si $e$ y $e'$ son neutros: $e = e * e' = e'$. Si $x'$ y $x''$ invierten
$x$: $x' = x' * e = x' * (x * x'') = (x' * x) * x'' = e * x''
= x''$.
\end{proof}

\section{Grupos}

\begin{definition}[Grupo]\label{def:b1:structures:group}
Un \emph{grupo}\index{grupo} $(G, *)$ es un \omterm{def:b1:logic:sets}{conjunto} con una ley
asociativa que admite neutro y en la que todo elemento tiene inverso. El
grupo es \emph{abeliano}\index{grupo abeliano} cuando la ley es
conmutativa.
\end{definition}

\begin{example}\label{ex:b1:structures:groups}
$(\Z, +)$, $(\Q, +)$, $(\R, +)$, $(\C, +)$; $(\Q^*, \times)$,
$(\R^*, \times)$, $(\C^*, \times)$, $(\mathbb{U}_n, \times)$ (\omterm{def:b1:complex:unity}{raíces de
la unidad}, \cref{def:b1:complex:unity}); el \omterm{def:b1:logic:sets}{conjunto} $\mathfrak{S}(E)$ de
las biyecciones de un \omterm{def:b1:logic:sets}{conjunto} $E$ en sí mismo, con la composición ---
el \emph{grupo simétrico}\index{grupo simétrico} de $E$, no \omterm{def:b1:structures:group}{abeliano} en
cuanto $\abs E \geq 3$. No son \omterm{def:b1:structures:group}{grupos}: $(\N, +)$ (faltan inversos),
$(\Z, \times)$ (solo $\pm 1$ es invertible).
\end{example}

\begin{proposition}[Reglas de cálculo]\label{prop:b1:structures:rules}
En un \omterm{def:b1:structures:group}{grupo} $G$ (escrito multiplicativamente, con neutro $e$):
\begin{enumerate}
  \item simplificación: $ax = ay \implies x = y$ y $xa = ya \implies
        x = y$;
  \item $(ab)^{-1} = b^{-1} a^{-1}$ y $(a^{-1})^{-1} = a$;
  \item para $a, b \in G$, cada una de las ecuaciones $ax = b$ y $xa = b$
        tiene solución única ($x = a^{-1}b$, resp.\ $x = b a^{-1}$).
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Multiplíquese por $a^{-1}$ por el lado adecuado, usando la
asociatividad. (2) $(b^{-1}a^{-1})(ab) = b^{-1}(a^{-1}a)b = b^{-1}b =
e$ y simétricamente; la unicidad del inverso concluye; el segundo punto
es la~\cref{prop:b1:structures:unique} aplicada a $a^{-1}$. (3)
Sustitúyase y úsese (1) para la unicidad.
\end{proof}

\begin{example}[Las simetrías de un rectángulo]\label{ex:b1:structures:klein}
Un rectángulo (no cuadrado) admite exactamente cuatro isometrías en sí
mismo: la identidad $e$, la simetría respecto del eje horizontal $h$, la
simetría respecto del eje vertical $v$ y el giro de media vuelta $r$
alrededor del centro. La composición hace de este \omterm{def:b1:logic:sets}{conjunto} de cuatro
elementos un \omterm{def:b1:structures:group}{grupo}: cada elemento es su propio inverso
($h^2 = v^2 = r^2 = e$) y el producto de dos elementos distintos y
distintos del neutro es el tercero ($hv = vh = r$: reflejar en los dos
ejes es el giro de media vuelta). La tabla completa es simétrica, luego
el \omterm{def:b1:structures:group}{grupo} es \omterm{def:b1:structures:group}{abeliano} --- y, sin embargo, \emph{no} es el mismo
\omterm{def:b1:structures:group}{grupo} que las rotaciones $\mathbb U_4$
del~\cref{ex:b1:structures:order}: allí $\iu$ tiene \omterm{def:b1:structures:order}{orden} $4$, mientras
que aquí todo elemento tiene \omterm{def:b1:structures:order}{orden} $\leq 2$. Dos \omterm{def:b1:structures:group}{grupos} del mismo
tamaño pueden, pues, tener estructuras multiplicativas genuinamente
distintas --- la figura de más abajo muestra las dos tablas una al lado
de la otra. Este \omterm{def:b1:structures:group}{grupo} de cuatro elementos vuelve como
$\{\pm1\} \times \{\pm1\}$, y el~\cref{exo:b1:structures:7} explica por
qué todo \omterm{def:b1:structures:group}{grupo} con todos los cuadrados triviales debe ser, como este,
\omterm{def:b1:structures:group}{abeliano}.
\end{example}

\begin{definition}[Subgrupo]\label{def:b1:structures:subgroup}
Un subconjunto $H$ de un \omterm{def:b1:structures:group}{grupo} $G$ es un
\emph{subgrupo}\index{subgrupo} (se escribe $H \leq G$) cuando contiene
$e$ y es estable por la ley y por la inversión. Entonces $H$ es a su vez
un \omterm{def:b1:structures:group}{grupo}.

\emph{Criterio:} un $H \subseteq G$ no vacío es un subgrupo si y solo
si
\[
\forall x, y \in H, \quad x y^{-1} \in H .
\]
\end{definition}

\begin{proof}[Demostración del criterio]
Un \omterm{def:b1:structures:subgroup}{subgrupo} lo cumple obviamente. Recíprocamente, sea
$H \neq \emptyset$ que lo cumpla y tómese $x_0 \in H$. Entonces
$e = x_0 x_0^{-1} \in H$; para $y \in H$, $y^{-1} = e\,y^{-1} \in H$; y
para $x, y \in H$, $xy =
x (y^{-1})^{-1} \in H$.
\end{proof}

\begin{example}\label{ex:b1:structures:subgroups}
$\mathbb{U}_n \leq (\C^*, \times)$: es no vacío y, para $z, w \in
\mathbb{U}_n$, $(zw^{-1})^n = z^n (w^n)^{-1} = 1$. Los \omterm{def:b1:structures:subgroup}{subgrupos} de
$(\Z, +)$ son exactamente los $n\Z$ (demostrado en
el~\cref{thm:b1:arith:gcd}). Una intersección de \omterm{def:b1:structures:subgroup}{subgrupos} es siempre un
\omterm{def:b1:structures:subgroup}{subgrupo}, pero una unión casi nunca lo es
(\cref{exo:b1:structures:6}).
\end{example}

\begin{remark}[Errores frecuentes con las estructuras]\label{rem:b1:structures:pitfalls}
\begin{enumerate}
  \item \emph{No basta con la estabilidad por la ley.} $\N$ es estable
        por la suma dentro de $\Z$ y contiene $0$ y, sin embargo, no es
        \omterm{def:b1:structures:subgroup}{subgrupo}: faltan los inversos. El criterio $xy^{-1} \in H$ lo
        comprueba todo de una vez --- pero solo después de verificar
        que $H \neq \emptyset$.
  \item \emph{Reflejos no \omterm{def:b1:structures:group}{abelianos}.} En un \omterm{def:b1:structures:group}{grupo} general,
        $(ab)^2 = abab$, que es $a^2b^2$ solo cuando $a$ y $b$
        conmutan; del mismo modo, $(ab)^{-1} = b^{-1}a^{-1}$, con el
        \omterm{def:b1:structures:order}{orden} invertido. Toda identidad importada del álgebra escolar
        hay que rededucirla de los axiomas o marcarla como conmutativa.
  \item \emph{\omterm{def:b1:structures:morphism}{Núcleo} frente a imagen.} $\ker f$ vive en el
        \emph{espacio de partida}; $\operatorname{im} f$, en el de
        llegada; «$f$ es \omterm{def:b1:logic:inj}{inyectiva} si y solo si $\ker f$ es trivial»
        (\cref{prop:b1:structures:kernel}) no tiene análogo con la
        imagen ($\operatorname{im} f = G'$ es la \omterm{def:b1:logic:inj}{sobreyectividad}).
  \item \emph{Los \omterm{def:b1:structures:ring}{anillos} no son \omterm{def:b1:structures:group}{grupos} para $\times$.} En un
        \omterm{def:b1:structures:ring}{anillo}, la mayoría de los elementos no tienen por qué ser
        invertibles, y simplificar por $a$ exige que $a$ sea una unidad
        o que el \omterm{def:b1:structures:ring}{anillo} sea un \omterm{def:b1:structures:field}{dominio de integridad}: en $\Z/12\Z$,
        $\overline3\,\overline2 =
        \overline3\,\overline6$ yet $\overline2 \neq \overline6$
        (\cref{ex:b1:structures:zncomputation}).
\end{enumerate}
\end{remark}

\begin{definition}[Morfismo de grupos]\label{def:b1:structures:morphism}
Sean $(G, *)$ y $(G', \star)$ \omterm{def:b1:structures:group}{grupos}. Una \omterm{def:b1:logic:map}{aplicación}
$f \colon G \to G'$ es un \emph{morfismo}\index{morfismo!de grupos} cuando
\[
\forall x, y \in G, \qquad f(x * y) = f(x) \star f(y).
\]
Entonces $f(e_G) = e_{G'}$ y $f(x^{-1}) = f(x)^{-1}$. El
\emph{núcleo}\index{núcleo} y la \emph{imagen} de $f$ son
\[
\ker f = f^{-1}(\{e_{G'}\}) \leq G,
\qquad
\operatorname{im} f = f(G) \leq G' .
\]
Un morfismo \omterm{def:b1:logic:inj}{biyectivo} es un \emph{isomorfismo}; su \omterm{def:b1:logic:map}{aplicación} inversa
es entonces automáticamente un morfismo.
\end{definition}

\begin{proof}[Demostración de las afirmaciones]
$f(e) = f(e * e) = f(e)\star f(e)$ y, simplificando $f(e)$, se obtiene
$e_{G'} = f(e)$. Después, $f(x)\star f(x^{-1}) = f(x x^{-1}) = e_{G'}$
identifica $f(x^{-1})$ como el inverso. \omterm{def:b1:structures:morphism}{Núcleo}: $e \in \ker f$; si
$x, y \in \ker f$, $f(xy^{-1}) = f(x)f(y)^{-1} = e$; se aplica el
criterio. Imagen: el mismo criterio con $f(x)f(y)^{-1} = f(xy^{-1})$.
Inversa de un isomorfismo: para $u, v \in G'$, escríbanse $u = f(x)$,
$v = f(y)$; entonces $f^{-1}(u \star v) = f^{-1}(f(xy)) = xy =
f^{-1}(u) f^{-1}(v)$.
\end{proof}

\begin{proposition}[Inyectividad mediante el núcleo]\label{prop:b1:structures:kernel}
Un \omterm{def:b1:structures:morphism}{morfismo} de \omterm{def:b1:structures:group}{grupos} $f$ es \omterm{def:b1:logic:inj}{inyectivo} si y solo si $\ker f =
\{e\}$.
\end{proposition}

\begin{proof}
Si $f$ es \omterm{def:b1:logic:inj}{inyectivo}, $\ker f$ solo puede contener la única
\omterm{def:b1:logic:map}{imagen recíproca} de $e_{G'}$, que es $e$. Recíprocamente, si
$\ker f = \{e\}$ y $f(x) = f(y)$, entonces $f(xy^{-1}) = f(x) f(y)^{-1}
= e_{G'}$, luego $xy^{-1} = e$,
i.e.\ $x = y$.
\end{proof}

\begin{example}\label{ex:b1:structures:expmorphism}
$\exp \colon (\R, +) \to (\R_+^*, \times)$ es un \omterm{def:b1:structures:morphism}{morfismo}
($\eu^{x+y} = \eu^x \eu^y$) y es \omterm{def:b1:logic:inj}{biyectivo}
(\cref{prop:b1:functions:expln}): las estructuras aditiva y
multiplicativa son isomorfas --- la razón de ser histórica de los
logaritmos. Otro \omterm{def:b1:structures:morphism}{morfismo}: $\theta \mapsto \eu^{\iu\theta}$ de
$(\R, +)$ sobre la circunferencia unidad $(\mathbb{U}, \times)$, con \omterm{def:b1:structures:morphism}{núcleo}
$2\pi\Z$.
\end{example}

\begin{example}[El morfismo del signo]\label{ex:b1:structures:signmorphism}
La \omterm{def:b1:logic:map}{aplicación} $s \colon (\R^*, \times) \to (\{\pm1\}, \times)$ que
envía $x$ a su signo es un \omterm{def:b1:structures:morphism}{morfismo}: el signo de un producto es el
producto de los signos. Su \omterm{def:b1:structures:morphism}{núcleo} es $\intoo0{+\infty}$ (un
\omterm{def:b1:structures:subgroup}{subgrupo}, como promete la~\cref{def:b1:structures:morphism}) y su imagen
es todo $\{\pm1\}$: \omterm{def:b1:logic:inj}{sobreyectivo} y masivamente no \omterm{def:b1:logic:inj}{inyectivo}. Dos
lecciones generales en miniatura. Primera: un \omterm{def:b1:structures:morphism}{morfismo} puede aplastar
información; $s$ no recuerda nada de $x$ salvo un bit, y esa es su
virtud --- los argumentos de signo son exactamente los cálculos que
pasan por $s$. Segunda: los \omterm{def:b1:structures:morphism}{morfismos} en $\{\pm1\}$ son los
«invariantes» más simples; la signatura de las permutaciones,
construida en el problema del fin de semana de este capítulo, es el
mismo fenómeno sobre el \omterm{def:b1:structures:group}{grupo} $\mathfrak S_n$, y todos los argumentos
de paridad que mueve descienden por un \omterm{def:b1:structures:morphism}{morfismo} bivaluado así.
\end{example}

\begin{definition}[Potencias, orden de un elemento]\label{def:b1:structures:order}
En un \omterm{def:b1:structures:group}{grupo} $G$ (notación multiplicativa), póngase $x^0 = e$, $x^{k+1} =
x^k x$ and $x^{-k} = (x^k)^{-1}$ for $k \in \N$; then $x^{k+l} = x^k
x^l$ for all $k, l \in \Z$, so $k \mapsto x^k$ is a morphism $(\Z,
+) \to G$ cuya imagen $\langle x \rangle = \{x^k : k \in \Z\}$ es un
\omterm{def:b1:structures:subgroup}{subgrupo}, el \omterm{def:b1:structures:subgroup}{subgrupo} \emph{generado} por $x$. El
\emph{orden}\index{orden de un elemento} de $x$ es el menor $m \geq
1$ with $x^m = e$ if one exists (then $\langle x\rangle = \{e, x,
\dots, x^{m-1}\}$ has exactly $m$ elements, and $x^k = e \iff m
\mid k$), y $\infty$ en caso contrario.
\end{definition}

\begin{example}\label{ex:b1:structures:order}
En $(\C^*, \times)$: $\iu$ tiene \omterm{def:b1:structures:order}{orden} $4$, con $\langle \iu \rangle
= \{1, \iu, -1, -\iu\} = \mathbb{U}_4$; more generally $\omega =
\eu^{2\iu\pi/n}$ has order $n$ and $\langle\omega\rangle =
\mathbb{U}_n$. En $(\Z, +)$, todo $x \neq 0$ tiene \omterm{def:b1:structures:order}{orden} infinito.
Por qué valen las afirmaciones de la definición: si $x$ tiene \omterm{def:b1:structures:order}{orden}
$m$, divídase cualquier $k$ entre $m$ ($k = mq + r$, $0 \leq r < m$,
\cref{thm:b1:arith:division}): $x^k = (x^m)^q x^r = x^r$, de modo que
las potencias se repiten con período $m$, los elementos enumerados son
distintos dos a dos por minimalidad de $m$, y $x^k = e$ obliga a
$r = 0$. Los órdenes de las permutaciones se calculan en el problema
del fin de semana.
\end{example}

\begin{example}[Órdenes dentro de $\mathbb U_{12}$]\label{ex:b1:structures:orderU12}
¿Cuál es el \omterm{def:b1:structures:order}{orden} de $\omega^k$ en $\mathbb U_n$, para $\omega =
\eu^{2\iu\pi/n}$? Se tiene $(\omega^k)^m = 1$ si y solo si $n \mid km$
y, escribiendo $d = \gcd(n, k)$, $n = dn'$, $k = dk'$ con
$\gcd(n', k') = 1$: $n \mid km \iff n' \mid k'm \iff n' \mid m$ (lema
de Gauss, \cref{thm:b1:arith:gauss}). El menor $m \geq 1$ así es
$n' = \frac{n}{\gcd(n,k)}$. En $\mathbb U_{12}$, por ejemplo,
$\omega^8$ tiene \omterm{def:b1:structures:order}{orden} $\frac{12}{\gcd(12,8)} = 3$ (en efecto,
$\omega^8 = \eu^{4\iu\pi/3} \in \mathbb U_3$), mientras que $\omega^5$
tiene \omterm{def:b1:structures:order}{orden} $12$: genera todo el \omterm{def:b1:structures:group}{grupo}, aunque no sea el generador
«estándar». Contar los generadores --- los $k$ con $\gcd(k, n) = 1$ ---
recupera los recuentos de \omterm{cor:b1:arith:bezout}{coprimos} del~\cref{ex:b1:counting:coprime}:
la teoría de \omterm{def:b1:structures:group}{grupos} y la combinatoria se encuentran.
\end{example}

\section{Anillos y cuerpos}

\begin{definition}[Anillo]\label{def:b1:structures:ring}
Un \emph{anillo}\index{anillo} $(A, +, \times)$ es un \omterm{def:b1:logic:sets}{conjunto} con dos
leyes tales que: $(A, +)$ es un \omterm{def:b1:structures:group}{grupo abeliano} (con neutro $0$);
$\times$ es asociativa con elemento neutro $1$; y $\times$ es
distributiva respecto de $+$ por los dos lados. El anillo es
\emph{conmutativo} cuando lo es $\times$. Un elemento $a$ es
\emph{invertible} (una \emph{unidad}) cuando $ab = ba = 1$ para algún
$b$; las unidades forman un \omterm{def:b1:structures:group}{grupo} $(A^\times, \times)$.
\end{definition}

\begin{proof}[Demostración de que las unidades forman un grupo]
Estabilidad: si $a, a'$ son unidades con inversos $b, b'$, entonces
\[
(aa')(b'b) = a(a'b')b = a\,1\,b = ab = 1,
\qquad (b'b)(aa') = 1
\]
y simétricamente, luego $aa'$ es una unidad. El elemento $1$ es una
unidad (su propio inverso), la asociatividad se hereda de $A$, y el
inverso $b$ de una unidad $a$ es a su vez una unidad (con inverso $a$).
Así pues, $(A^\times, \times)$ cumple todos los axiomas de \omterm{def:b1:structures:group}{grupo}. Todo
\omterm{def:b1:structures:group}{grupo} de este libro que no esté construido con permutaciones surge
así: $\Q^* = \Q^\times$, $\R^*$, $\C^*$, las unidades de $\Z/n\Z$ de
más abajo y, más adelante, las matrices invertibles (\cref{ch:b1:matrices}).
\end{proof}

\begin{example}\label{ex:b1:structures:rings}
$\Z, \Q, \R, \C$ son \omterm{def:b1:structures:ring}{anillos} conmutativos; $\Z^\times = \{1, -1\}$,
$\Q^\times = \Q^*$. Más adelante: \omterm{def:b1:structures:ring}{anillos} de polinomios $K[X]$
(\cref{ch:b1:poly}), \omterm{def:b1:structures:ring}{anillos} de matrices (no conmutativos,
\cref{ch:b1:matrices}) y $\Z/n\Z$ más abajo. En todo \omterm{def:b1:structures:ring}{anillo},
$0 \times a = 0$ (por distributividad: $0a = (0+0)a = 0a + 0a$), y
$(-1)a = -a$.
\end{example}

\begin{example}[Idempotentes: fenómenos nuevos en anillos nuevos]\label{ex:b1:structures:idempotents}
En $\Z$, la ecuación $x^2 = x$, es decir, $x(x - 1) = 0$, solo tiene las
soluciones $0$ y $1$. En $\Z/6\Z$, probando todas las clases:
$\overline0^2 = \overline0$, $\overline1^2 = \overline1$,
$\overline3^2 = \overline9 = \overline3$ y $\overline4^2 =
\overline{16} = \overline4$ --- \emph{cuatro} idempotentes. Los dos
exóticos vienen de los divisores de cero: $\overline3\,(\overline3 -
\overline1) = \overline3 \times \overline2 = \overline6 =
\overline0$ sin que ninguno de los factores sea cero. Cálculos así
calibran la intuición: los hechos familiares sobre ecuaciones
sobreviven en los \omterm{def:b1:structures:field}{dominios de integridad} y en los \omterm{def:b1:structures:field}{cuerpos}, pero un
\omterm{def:b1:structures:ring}{anillo} general puede comportarse de otro modo, y lo hace --- véanse
también los \omterm{def:b1:structures:ring}{anillos} booleanos del~\cref{exo:b1:structures:10}, donde
\emph{todo} elemento es idempotente.
\end{example}

\begin{proposition}[Teorema del binomio en un anillo conmutativo]\label{prop:b1:structures:binomial}
Si $a, b$ son elementos de un \omterm{def:b1:structures:ring}{anillo} conmutativo (más en general, si
$ab = ba$), entonces, para $n \in \N$:
\[
(a+b)^n = \sum_{k=0}^n \binom nk a^k b^{n-k},
\qquad
a^n - b^n = (a - b) \sum_{k=0}^{n-1} a^k b^{\,n-1-k} .
\]
\end{proposition}

\begin{proof}
Las demostraciones del~\cref{thm:b1:counting:binomial} y de la identidad
geométrica solo usan la asociatividad, la conmutatividad de los dos
elementos y la distributividad --- se aplican literalmente.
\end{proof}

\begin{example}[El teorema del binomio en un anillo poco familiar]\label{ex:b1:structures:binomialring}
Dos réditos rápidos de la generalidad. En $\Z/p\Z$ ($p$ \omterm{def:b1:arith:prime}{primo}), los
\omterm{def:b1:counting:objects}{coeficientes binomiales} intermedios se anulan (primer paso
del~\cref{thm:b1:arith:fermat}), de modo que el teorema se reduce al
\emph{sueño del novato}
\[
(a + b)^p = a^p + b^p \qquad \text{en } \Z/p\Z ,
\]
identidad genuina allí, por criminal que parezca sobre $\R$. Y en
cualquier \omterm{def:b1:structures:ring}{anillo} conmutativo que contenga un elemento $\varepsilon$
con $\varepsilon^2 = 0$, el teorema se trunca:
$(a + \varepsilon)^n = a^n + n\,a^{n-1}\varepsilon$, pues todos los
términos superiores llevan un factor $\varepsilon^2 = 0$. El
coeficiente $n\,a^{n-1}$ de $\varepsilon$ es la derivada de $x^n$ ---
no es casualidad, y es un primer indicio de que las derivadas son
álgebra tanto como análisis (compárese con la derivada formal de
\cref{ch:b1:poly}).
\end{example}

\begin{definition}[Dominio de integridad, cuerpo]\label{def:b1:structures:field}
Un \omterm{def:b1:structures:ring}{anillo} conmutativo $A \neq \{0\}$ es un \emph{dominio de
integridad}\index{dominio de integridad} cuando no tiene divisores de
cero: $ab = 0 \implies a = 0$ o $b = 0$. Es un
\emph{cuerpo}\index{cuerpo} cuando todo elemento no nulo es invertible.
Todo cuerpo es un dominio de integridad ($ab = 0$ y $a \neq 0$ dan
$b = a^{-1}ab = 0$).
\end{definition}

\begin{example}
$\Q$, $\R$, $\C$ son \omterm{def:b1:structures:field}{cuerpos}; $\Z$ es un \omterm{def:b1:structures:field}{dominio de integridad} pero
no un \omterm{def:b1:structures:field}{cuerpo}. En un \omterm{def:b1:structures:field}{dominio de integridad} vale la simplificación
para $\times$: $ab = ac$ y $a \neq 0$ implican $b = c$.
\end{example}

\section{El anillo $\Z/n\Z$}

\begin{definition}\label{def:b1:structures:zn}
Fíjese $n \in \N^*$. Las clases de \omterm{def:b1:arith:congruence}{congruencia} \omterm{def:b1:complex:field}{módulo} $n$
(\cref{ex:b1:logic:congruence}) forman un \omterm{def:b1:logic:sets}{conjunto}
$\Z/n\Z$\index{Z/nZ@$\Z/n\Z$} de $n$ elementos, escritas $\overline 0,
\overline 1, \dots, \overline{n-1}$. Las operaciones
\[
\overline a + \overline b = \overline{a + b},
\qquad
\overline a \times \overline b = \overline{ab}
\]
están bien definidas --- las clases de los resultados no dependen de los
representantes, precisamente porque la \omterm{def:b1:arith:congruence}{congruencia} es compatible con
$+$ y $\times$ (\cref{def:b1:arith:congruence}) --- y hacen de $\Z/n\Z$
un \omterm{def:b1:structures:ring}{anillo} conmutativo.
\end{definition}

\begin{theorem}[Unidades de $\Z/n\Z$; los cuerpos $\Z/p\Z$]\label{thm:b1:structures:zp}
\begin{enumerate}
  \item $\overline a$ es invertible en $\Z/n\Z$ si y solo si
        $\gcd(a, n) = 1$.
  \item $\Z/n\Z$ es un \omterm{def:b1:structures:field}{cuerpo} si y solo si $n$ es \omterm{def:b1:arith:prime}{primo}.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) es la~\cref{prop:b1:arith:invmod} reescrita con clases.

(2) Si $n = p$ es \omterm{def:b1:arith:prime}{primo}, todo $\overline a \neq \overline 0$ tiene $p
\nmid a$, so $\gcd(a, p) = 1$: invertible by (1) --- a field. If $n =
ab$ con $1 < a, b < n$, entonces $\overline a\, \overline b = \overline
n = \overline 0$ with $\overline a, \overline b \neq \overline 0$:
divisores de cero, así que ni siquiera es un \omterm{def:b1:structures:field}{dominio de integridad}; y
$n = 1$ da el \omterm{def:b1:structures:ring}{anillo} nulo, excluido.
\end{proof}

\begin{example}[¿Cuántas raíces cuadradas de $1$?]\label{ex:b1:structures:sqrtone}
Resuélvase $x^2 = \overline 1$ en $\Z/8\Z$ y en $\Z/7\Z$. Probando las
ocho clases \omterm{def:b1:complex:field}{módulo} $8$: $1^2 = 1$, $3^2 = 9 \equiv 1$, $5^2 = 25
\equiv 1$, $7^2 = 49 \equiv 1$ --- \emph{cuatro} soluciones
$\{\overline1, \overline3, \overline5, \overline7\}$, aunque el
polinomio $X^2 - 1$ tenga grado $2$. En el \omterm{def:b1:structures:field}{cuerpo} $\Z/7\Z$, en
cambio, $x^2 = \overline1$ significa $(x - \overline1)(x +
\overline1) = \overline0$, and a field has no zero divisors: $x =
\pm\overline1$, solo dos soluciones. El fallo \omterm{def:b1:complex:field}{módulo} $8$ es
rastreable: $(3-1)(3+1) = 2 \times 4 = 8 \equiv 0$ sin que se anule
ninguno de los factores. Moraleja: la regla familiar «una ecuación de
grado $d$ tiene a lo sumo $d$ raíces» es un teorema sobre
\emph{\omterm{def:b1:structures:field}{dominios de integridad}} (el~\cref{cor:b1:poly:nroots} lo demuestra
sobre \omterm{def:b1:structures:field}{cuerpos}); en los \omterm{def:b1:structures:ring}{anillos} con divisores de cero falla en
silencio --- que es exactamente por lo que la demostración por
emparejamiento del teorema de Wilson (\cref{exo:b1:arith:11}) necesitaba
$p$ \omterm{def:b1:arith:prime}{primo}.
\end{example}

\begin{example}[Calcular en $\Z/n\Z$]\label{ex:b1:structures:zncomputation}
En $\Z/12\Z$: las unidades son $\overline 1, \overline 5, \overline 7,
\overline{11}$ (las clases coprimas con $12$), y cada una es su
propio inverso ($5^2 = 25 \equiv 1$, $7^2 = 49 \equiv 1$, $11^2 = 121
\equiv 1$). La ecuación $\overline 3\, x = \overline 6$ tiene
\emph{tres} soluciones ($x \in \{\overline 2, \overline 6,
\overline{10}\}$): sin invertibilidad no hay simplificación. En
$\Z/11\Z$, en cambio, toda ecuación $\overline a x = \overline b$ con
$\overline a \neq \overline 0$ tiene exactamente una solución.
\end{example}

\begin{example}[Los axiomas de grupo como licencia para resolver]\label{ex:b1:structures:solving}
En el \omterm{def:b1:structures:group}{grupo} $\bigl((\Z/7\Z)^*, \times\bigr)$, resuélvase $\overline
3\,x = \overline 5$. Por la~\cref{prop:b1:structures:rules}~(3), la
solución existe, es única y vale $\overline3^{-1}\,
\overline5$; since $\overline3 \times \overline5 = \overline{15}
= \overline1$, el inverso de $\overline 3$ es $\overline 5$, luego
\[
x = \overline5 \times \overline5 = \overline{25} = \overline4,
\qquad\text{comprobación: } \overline3 \times \overline4 =
\overline{12} = \overline5 .
\]
Lo importante no es tanto la respuesta como la garantía: en un
\omterm{def:b1:structures:group}{grupo}, toda ecuación así tiene solución única \emph{antes} de
cualquier cálculo, de modo que un procedimiento de resolución nunca
puede toparse con «no hay solución» o «hay varias». Compárese con
$\overline3\,x = \overline6$ en $\Z/12\Z$ de más arriba, donde la
garantía falla --- saber en qué estructura se está es saber qué se
puede dar por supuesto.
\end{example}

\begin{example}[Productos directos]\label{ex:b1:structures:product}
Si $G$ y $H$ son \omterm{def:b1:structures:group}{grupos}, el \omterm{def:b1:logic:sets}{conjunto} producto $G \times H$ con la ley
componente a componente $(g, h)(g', h') = (gg', hh')$ es un \omterm{def:b1:structures:group}{grupo}:
los axiomas se comprueban coordenada a coordenada, con neutro
$(e_G, e_H)$ e inversos $(g^{-1}, h^{-1})$. Los órdenes se combinan
por el mcm: $(g, h)^m = (g^m, h^m)$ es el neutro si y solo si el
\omterm{def:b1:structures:order}{orden} de $g$ y el de $h$ \omterm{def:b1:arith:divides}{dividen} los dos a $m$. Así, en $\Z/2\Z
\times \Z/2\Z$ (additive) every nonzero element has order $2$
--- que es exactamente el \omterm{def:b1:structures:group}{grupo} del rectángulo
del~\cref{ex:b1:structures:klein} en coordenadas ---, mientras que
$\Z/4\Z$ tiene un elemento de \omterm{def:b1:structures:order}{orden} $4$: una segunda demostración, sin
cálculo alguno, de que los dos \omterm{def:b1:structures:group}{grupos} de tamaño $4$ no son isomorfos
(un isomorfismo conserva los órdenes). Los productos son la manera más
fácil de fabricar \omterm{def:b1:structures:group}{grupos} nuevos a partir de otros viejos, y el plano
$\R^2 = \R \times \R$ del~\cref{ch:b1:vspaces} es la instancia más
importante de esta construcción.
\end{example}

\begin{remark}[Fermat, estructuralmente]
En el \omterm{def:b1:structures:field}{cuerpo} $\Z/p\Z$, las clases no nulas forman un \omterm{def:b1:structures:group}{grupo}
multiplicativo de $p - 1$ elementos, y el pequeño teorema de Fermat
(\cref{thm:b1:arith:fermat}) dice: todo elemento $x$ de ese \omterm{def:b1:structures:group}{grupo}
cumple $x^{p-1} = \overline 1$. Es un caso particular de un hecho
general sobre \omterm{def:b1:structures:group}{grupos} finitos (el teorema de Lagrange), demostrado en
el segundo año; la demostración por emparejamiento del teorema de
Wilson (\cref{exo:b1:arith:11}) ya tenía este sabor de teoría de \omterm{def:b1:structures:group}{grupos}.
\end{remark}

\begin{remark}[Interludio: qué compra la abstracción]\label{rem:b1:structures:interlude}
Es legítimo preguntarse qué se ha ganado demostrando, por ejemplo,
la~\cref{prop:b1:structures:unique} para una ley abstracta en lugar de
para números. La respuesta es apalancamiento. Ese argumento de dos
líneas cubre ya, de golpe: los inversos de funciones para la
composición (\cref{thm:b1:logic:inverse}, cuya demostración de unicidad
repite palabra por palabra), los inversos \omterm{def:b1:complex:field}{módulo} $n$
(\cref{prop:b1:arith:invmod}), los inversos de los reales no nulos, los
de las unidades de cualquier \omterm{def:b1:structures:ring}{anillo} y --- sin haberlas visto todavía
--- los de las matrices invertibles del~\cref{ch:b1:matrices}, donde la
unicidad de $A^{-1}$ no necesitará ni una línea de demostración. La
misma economía vale para la~\cref{prop:b1:structures:kernel} (un solo
criterio de \omterm{def:b1:logic:inj}{inyectividad}, reutilizado para las aplicaciones lineales en
el~\cref{ch:b1:linmaps}) y para el criterio de \omterm{def:b1:structures:subgroup}{subgrupo}. La
abstracción no es aquí generalidad porque sí: es la negativa a
demostrar cinco veces el mismo lema con cinco nombres distintos. El
precio --- llevar la cuenta de qué axiomas usa realmente cada
\omterm{def:b1:logic:statement}{enunciado} --- es exactamente lo que entrenan los ejercicios de este
capítulo.
\end{remark}

\begin{remark}[Dónde se usa este capítulo]\label{rem:b1:structures:whereused}
El vocabulario de este capítulo es la gramática del resto del volumen.
Los \omterm{def:b1:structures:ring}{anillos} y los \omterm{def:b1:structures:field}{cuerpos} organizan el~\cref{ch:b1:poly} ($K[X]$ es
un \omterm{def:b1:structures:ring}{anillo} que imita a $\Z$) y el~\cref{ch:b1:fractions} ($K(X)$ es su
\omterm{def:b1:structures:field}{cuerpo} de fracciones); los espacios vectoriales
(\cref{ch:b1:vspaces}) son \omterm{def:b1:structures:group}{grupos abelianos} sobre los que actúa un
\omterm{def:b1:structures:field}{cuerpo}; las matrices (\cref{ch:b1:matrices}) forman el primer
\omterm{def:b1:structures:ring}{anillo} seriamente no conmutativo del volumen, y sus elementos
invertibles un \omterm{def:b1:structures:group}{grupo} cuyo estudio es el álgebra lineal misma. Los
\omterm{def:b1:structures:morphism}{morfismos} y los \omterm{def:b1:structures:morphism}{núcleos} vuelven como aplicaciones lineales y
\omterm{def:b1:structures:morphism}{núcleos} en el~\cref{ch:b1:linmaps} --- la~\cref{prop:b1:structures:kernel}
\emph{es} el criterio de \omterm{def:b1:logic:inj}{inyectividad} de aquel capítulo, demostrado
aquí de una vez por todas. El \omterm{ex:b1:structures:groups}{grupo simétrico}, protagonista del
problema del fin de semana, suministra la signatura sobre la que se
construyen los determinantes en el~\cref{ch:b1:det}.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:structures:1}
En $E = \R \setminus \{1\}$ se define $x * y = x + y - xy$. Demuéstrese
que $(E, *)$ es un \omterm{def:b1:structures:group}{grupo abeliano}. \emph{(Identifíquense el neutro y el
inverso de $x$; compruébese la estabilidad: ¿por qué $x * y \neq 1$?)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:structures:2}
¿Cuáles de los siguientes son \omterm{def:b1:structures:group}{grupos}?
\begin{enumerate}
  \item $(\intoo{0}{+\infty}, \times)$;
  \item $(\{-1, 0, 1\}, +)$;
  \item $(\Q^*, \times)$;
  \item el \omterm{def:b1:logic:sets}{conjunto} de los enteros impares con la suma.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:structures:3}
Escríbase la tabla de composición del \omterm{ex:b1:structures:groups}{grupo simétrico}
$\mathfrak{S}_3$ de $\{1,2,3\}$ (seis biyecciones: la identidad, tres
transposiciones y dos $3$-ciclos), y exhíbanse dos elementos que no conmuten.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:structures:4}
Demuéstrese que $H = \{z \in \C^* : \abs z = 1\}$ es un \omterm{def:b1:structures:subgroup}{subgrupo} de $(\C^*,
\times)$, y que $\R_+^*$ es otro; ¿es $H \cup \R_+^*$ un
\omterm{def:b1:structures:subgroup}{subgrupo}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:5}
Sea $f \colon (\R, +) \to (\C^*, \times)$, $\theta \mapsto
\eu^{\iu\theta}$. Demuéstrese que $f$ es un \omterm{def:b1:structures:morphism}{morfismo}, calcúlense
$\ker f$ e $\operatorname{im} f$ y dedúzcase de
la~\cref{prop:b1:structures:kernel} que $f$ no es \omterm{def:b1:logic:inj}{inyectivo}. Restrínjase
el dominio para hacerlo \omterm{def:b1:logic:inj}{inyectivo} en un intervalo lo mayor posible.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:6}
Sean $H, K$ \omterm{def:b1:structures:subgroup}{subgrupos} de $G$. Demuéstrese que $H \cap K$ es un
\omterm{def:b1:structures:subgroup}{subgrupo} y que $H \cup K$ es un \omterm{def:b1:structures:subgroup}{subgrupo} \emph{solo} cuando
$H \subseteq K$ o $K \subseteq H$. \emph{(Si $h \in H \setminus K$ y
$k \in K \setminus H$, ¿dónde puede vivir $hk$?)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:7}
Un \omterm{def:b1:structures:group}{grupo} $G$ cumple $x^2 = e$ para todo $x \in G$. Demuéstrese que $G$
es \omterm{def:b1:structures:group}{abeliano}. \emph{(Desarróllese $(xy)^2$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:8}
En $\Z/18\Z$: enumérense las unidades y hállese el inverso de
$\overline 5$; resuélvase $\overline 5\, x = \overline 7$; resuélvase $\overline 6\, x =
\overline 3$ and $\overline 6\, x = \overline{12}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:9}
Demuéstrese que el \omterm{def:b1:logic:sets}{conjunto} $\Z[\sqrt 2] = \{a + b\sqrt 2 : a, b \in \Z\}$
es un \omterm{def:b1:structures:ring}{anillo} (un subanillo de $\R$), y que $1 + \sqrt 2$ es una unidad
suya con infinitas potencias distintas --- de modo que
$\Z[\sqrt 2]^\times$ es infinito, a diferencia de $\Z^\times$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:structures:10}
(\omterm{def:b1:structures:ring}{Anillos} booleanos) Sea $A$ un \omterm{def:b1:structures:ring}{anillo} en el que $x^2 = x$ para todo
$x$. Demuéstrese que $x + x = 0$ para todo $x$ y que $A$ es conmutativo.
\emph{(Desarróllense $(x+x)^2$ y $(x+y)^2$.)} Dese un ejemplo de un
\omterm{def:b1:structures:ring}{anillo} así con $\mathcal{P}(E)$, tomando la diferencia simétrica como
suma y la intersección como producto.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:structures:11}
Sea $G$ un \omterm{def:b1:structures:group}{grupo} en el que, para cierto $n \geq 1$ fijo,
$(xy)^n = x^n y^n$, $(xy)^{n+1} = x^{n+1}y^{n+1}$ y
$(xy)^{n+2} = x^{n+2}y^{n+2}$ para todos $x, y$. Demuéstrese que $G$ es
\omterm{def:b1:structures:group}{abeliano}. \emph{(De las tres identidades dedúzcase primero
$y^n x = x y^n$, después $y^{n+1} x = x y^{n+1}$, y conclúyase.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:structures:12}
\begin{enumerate}
  \item Determínense todos los \omterm{def:b1:structures:morphism}{morfismos} de \omterm{def:b1:structures:group}{grupos} de $(\Z, +)$ en $(\Z, +)$.
  \item Demuéstrese que el único \omterm{def:b1:structures:morphism}{morfismo} de \omterm{def:b1:structures:group}{grupos} de $(\Q, +)$ en
        $(\Z, +)$ es el \omterm{def:b1:structures:morphism}{morfismo} nulo. \emph{(Para $x \in \Q$ y
        $n \in \N^*$, compárense $f(x)$ y $n\,f(x/n)$.)}
\end{enumerate}
\end{exercise}

\section{Problema: el grupo simétrico y el rompecabezas del 8}

\begin{problem}\label{pb:b1:structures:1}
El \omterm{def:b1:structures:group}{grupo} $\mathfrak S_n$ de las permutaciones de $\intint1n$ es el
\omterm{def:b1:structures:group}{grupo} más antiguo de las matemáticas y sigue siendo el más
instructivo. Este problema construye su teoría de estructura desde cero
--- ciclos, generación por transposiciones, el \omterm{def:b1:structures:morphism}{morfismo}
\emph{signatura} $\varepsilon \colon \mathfrak S_n \to \{\pm1\}$ (cuya
existencia no es en absoluto trivial) y el \omterm{pb:b1:structures:1}{grupo alternado}
$\mathfrak A_n$ generado por los $3$-ciclos --- y lo rentabiliza después
en un rompecabezas clásico: en el juego de fichas deslizantes
$3 \times 3$, ninguna sucesión de movimientos puede intercambiar dos
fichas y dejar todo lo demás en su sitio.
\index{grupo simétrico}\index{signatura}\index{grupo alternado}
Las permutaciones actúan sobre $\intint1n$; los productos
$\sigma\tau$ significan «aplíquese primero $\tau$»; $[\,v_1, \dots,
v_n]$ denota la \omterm{def:b1:counting:objects}{permutación} que envía $i$ a $v_i$.

\medskip
\noindent\textbf{Parte I --- Ciclos y transposiciones.}
\begin{enumerate}
  \item Justifíquese $\abs{\mathfrak S_n} = n!$
        (\cref{thm:b1:counting:counts}). En $\mathfrak S_3$, calcúlense
        los dos productos de $\sigma = [2, 3, 1]$ y $\tau = [1, 3, 2]$,
        y conclúyase que $\mathfrak S_3$ no es \omterm{def:b1:structures:group}{abeliano}.
  \item Un \emph{$k$-ciclo} $(a_1\ a_2\ \dots\ a_k)$ ($k \geq 2$, con
        los $a_i$ distintos dos a dos) envía $a_1 \mapsto a_2
        \mapsto \dots \mapsto a_k \mapsto a_1$ y deja fijo todo lo
        demás; su \emph{soporte} es $\{a_1, \dots, a_k\}$.
        Demuéstrese que dos ciclos de soportes disjuntos conmutan.
  \item Demuéstrese que toda $\sigma \in \mathfrak S_n$ es producto de
        ciclos de soportes disjuntos dos a dos, y que esa
        descomposición es única salvo el \omterm{def:b1:structures:order}{orden} de los factores.
        \emph{(Considérese, para cada $i$, la sucesión $i, \sigma(i),
        \sigma^2(i), \dots$: tiene que volver a $i$; las \emph{órbitas}
        resultantes parten $\intint1n$, y $\sigma$ actúa sobre cada
        una como un ciclo.)}
  \item Descompóngase $\sigma = [4, 1, 5, 2, 3, 7, 8, 6] \in
        \mathfrak S_8$ en ciclos disjuntos. Definiendo el
        \emph{\omterm{def:b1:structures:order}{orden}} de $\sigma$ como en
        el~\cref{def:b1:structures:order}, demuéstrese que el \omterm{def:b1:structures:order}{orden} de un
        producto de ciclos disjuntos es el mcm de sus longitudes, y
        calcúlese el \omterm{def:b1:structures:order}{orden} de esta $\sigma$.
  \item Demuéstrese la identidad telescópica
        \[
        (a_1\ a_2\ \dots\ a_k)
        = (a_1\ a_k)(a_1\ a_{k-1})\cdots(a_1\ a_2) ,
        \]
        y conclúyase que toda \omterm{def:b1:counting:objects}{permutación} es producto de
        transposiciones. Escríbase la $\sigma$ de la pregunta 4 como un
        producto así.
  \item Véase además que bastan las transposiciones \emph{contiguas}
        $(i\ \ i{+}1)$: para $a < b$,
        \[
        (a\ b) = (a\ \ a{+}1)(a{+}1\ \ a{+}2)\cdots(b{-}1\ \ b)
        \cdots(a{+}1\ \ a{+}2)(a\ \ a{+}1),
        \]
        un producto de $2(b - a) - 1$ transposiciones contiguas ---
        un número \emph{impar} (esta paridad importará dos veces más
        abajo).
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La signatura existe.} Para $\sigma
\in \mathfrak S_n$, sea
\[
N(\sigma) = \#\bigl\{(i, j) : i < j,\ \sigma(i) >
\sigma(j)\bigr\}
\]
su número de \emph{inversiones}, y póngase $\varepsilon(\sigma) =
(-1)^{N(\sigma)}$.
\begin{enumerate}[resume]
  \item Calcúlense $N$ y $\varepsilon$ para la identidad, para una
        transposición $(i\ \ i{+}1)$ y para $[2, 3, 1]$.
  \item Demuéstrese que, para toda $\sigma$ y toda transposición
        contigua $\tau = (i\ \ i{+}1)$: $N(\sigma\tau) = N(\sigma) \pm
        1$. \emph{(Componer con $\tau$ por la derecha intercambia los
        valores de las posiciones $i$ e $i + 1$; exactamente un par
        cambia su condición de inversión.)}
  \item Dedúzcase, usando la pregunta 6, que para \emph{cualquier}
        transposición $\tau$ se tiene $\varepsilon(\sigma\tau) =
        -\varepsilon(\sigma)$; conclúyase que si $\sigma$ es producto
        de $p$ transposiciones, entonces $\varepsilon(\sigma) =
        (-1)^p$ --- en particular, la paridad de $p$ solo depende de
        $\sigma$ y no de la factorización elegida --- y que
        $\varepsilon \colon \mathfrak S_n \to \{\pm 1\}$ es un
        \omterm{def:b1:structures:morphism}{morfismo} de \omterm{def:b1:structures:group}{grupos}.
  \item Véase que un $k$-ciclo tiene signatura $(-1)^{k-1}$ y que, en
        general, $\varepsilon(\sigma) = (-1)^{n - c(\sigma)}$, donde
        $c(\sigma)$ es el número de órbitas de $\sigma$ (incluidos los
        puntos fijos).
  \item El \emph{\omterm{pb:b1:structures:1}{grupo alternado}} es $\mathfrak A_n =
        \ker\varepsilon$. Justifíquese que es un \omterm{def:b1:structures:subgroup}{subgrupo} y
        demuéstrese que $\abs{\mathfrak A_n} = \frac{n!}2$ para
        $n \geq 2$. \emph{(Fíjese una transposición $\tau_0$ y
        considérese $\sigma
        \mapsto \sigma\tau_0$.)}
  \item Comprobación de coherencia sobre $\sigma = [4, 1, 5, 2, 3, 7,
        8, 6]$: calcúlese $\varepsilon(\sigma)$ de tres maneras ---
        contando inversiones, a partir del tipo cíclico con la
        pregunta 10, y con el recuento de transposiciones de la pregunta 5.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- $\mathfrak A_n$ está generado por
$3$-ciclos.}
\begin{enumerate}[resume]
  \item Sean $a, b, c, d$ distintos dos a dos. Compruébense las dos
        identidades
        \[
        (a\ b)(a\ c) = (a\ c\ b),
        \qquad
        (a\ b)(c\ d) = (a\ c\ b)(a\ c\ d) .
        \]
  \item Demuéstrese que, para $n \geq 3$, todo elemento de
        $\mathfrak A_n$ es producto de $3$-ciclos. \emph{(Una
        \omterm{def:b1:counting:objects}{permutación} par es producto de un número par de
        transposiciones; absórbanse de dos en dos.)}
  \item Escríbanse $(1\ 2)(3\ 4)$ y el $5$-ciclo $(1\ 2\ 3\ 4\ 5)$
        explícitamente como productos de $3$-ciclos.
  \item Demuéstrese la fórmula de conjugación: para toda $\sigma \in
        \mathfrak S_n$,
        \[
        \sigma\,(a_1\ \dots\ a_k)\,\sigma^{-1}
        = \bigl(\sigma(a_1)\ \dots\ \sigma(a_k)\bigr) .
        \]
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El rompecabezas del 8.} Las fichas
$1, \dots, 8$ se deslizan en un marco $3 \times 3$ con una casilla
vacía; un \emph{movimiento} desliza a la casilla vacía una ficha
contigua a ella. Numérense las casillas $1, \dots, 9$ (fila a fila; la
posición resuelta tiene la ficha $i$ en la casilla $i$ y la casilla
vacía en la casilla $9$). Trátese la casilla vacía como una novena
ficha, de modo que una posición es una \omterm{def:b1:counting:objects}{permutación} $\sigma \in \mathfrak
S_9$ (tile $\sigma(i)$ sits in cell $i$).
\begin{enumerate}[resume]
  \item Véase que un movimiento sustituye $\sigma$ por
        $\sigma \circ \tau$, donde $\tau$ es la transposición de las
        dos casillas implicadas; dedúzcase que cada movimiento cambia
        el signo de
        $\varepsilon(\sigma)$.
  \item Sea $d(\sigma)$ la distancia de taxi (filas más columnas)
        entre la casilla que ocupa el hueco y su casilla de destino
        $9$. Véase que cada movimiento cambia $d$ en $\pm1$, de modo
        que cada movimiento también cambia el signo de
        $(-1)^{d(\sigma)}$. Conclúyase que
        \[
        I(\sigma) = \varepsilon(\sigma)\cdot(-1)^{d(\sigma)}
        \]
        es \emph{invariante} bajo todo movimiento.
  \item Demuéstrese la imposibilidad clásica del rompecabezas: la
        posición que intercambia las fichas $7$ y $8$ y deja todo lo
        demás (incluida la casilla vacía) en su sitio no se puede
        alcanzar desde la posición resuelta.
  \item Admitimos el recíproco (su demostración es una inducción
        instructiva pero larga): toda posición con $I = +1$ es
        alcanzable. Dedúzcase que exactamente la mitad de las $8!$
        posiciones con la casilla vacía en su sitio son resolubles, es
        decir,
        $\frac{8!}2 = 20\,160$.
  \item Dedúzcase de la pregunta 20 que las disposiciones de fichas
        alcanzables con la casilla vacía en su sitio forman exactamente
        el \omterm{def:b1:structures:subgroup}{subgrupo} $\mathfrak A_8 \leq \mathfrak S_8$.
  \item Aplicaciones del invariante: ¿se puede alcanzar (a) la posición
        en la que las fichas $1, 2, 3$ están permutadas cíclicamente y
        todo lo demás, incluida la casilla vacía, está en su sitio?
        ¿(b) la posición en la que la ficha $5$ y la casilla vacía han
        intercambiado sus lugares y todas las demás fichas están en su
        sitio? Justifíquense las dos respuestas con $I$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Síntesis.}
\begin{enumerate}[resume]
  \item Demuéstrese que, para $n \geq 3$, los únicos \omterm{def:b1:structures:morphism}{morfismos} de
        \omterm{def:b1:structures:group}{grupos} $f \colon \mathfrak S_n \to \{\pm 1\}$ son el
        \omterm{def:b1:structures:morphism}{morfismo} constante y $\varepsilon$. \emph{(Usando la
        pregunta 16 y la conmutatividad de $\{\pm1\}$, véase que $f$
        toma el mismo valor en todas las transposiciones.)}
  \item ¿Dónde ha usado exactamente el problema: (i) el concepto de
        \omterm{def:b1:structures:morphism}{morfismo} y la~\cref{prop:b1:structures:kernel}; (ii) los
        principios de recuento del~\cref{ch:b1:counting}; (iii) la
        cuestión de la buena definición que resuelven las preguntas
        8--9? Una frase para cada uno.
  \item Síntesis, en un párrafo breve: una sola función de paridad,
        demostrada bien definida una única vez, organiza a la vez la
        estructura interna de $\mathfrak S_n$ (el \omterm{def:b1:structures:subgroup}{subgrupo}
        $\mathfrak A_n$), decide un rompecabezas físico y ---
        mediante la fórmula $\det = \sum_\sigma
        \varepsilon(\sigma)\cdots$ --- definirá los determinantes en
        el~\cref{ch:b1:det}. Coméntese el patrón recurrente: los
        invariantes convierten «pruébense todas las sucesiones de
        movimientos» en un único cálculo.
\end{enumerate}
\end{problem}
