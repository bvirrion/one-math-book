\chapter{Combinatoria}\label{ch:b1:counting}

Contar \omterm{def:b1:counting:card}{conjuntos finitos} suena elemental --- y se vuelve sutil enseguida.
Este capítulo define correctamente el \omterm{def:b1:counting:card}{cardinal} (mediante biyecciones, en el
espíritu del~\cref{ch:b1:logic}), establece el puñado de principios de
recuento de los que se sigue todo lo demás y deduce los recuentos
clásicos: listas, permutaciones, subconjuntos, \omterm{def:b1:counting:objects}{coeficientes binomiales}.

\section{Cardinal de un conjunto finito}

\begin{definition}[Conjunto finito, cardinal]\label{def:b1:counting:card}
Para $n \in \N^*$, se escribe $\intint{1}{n} = \{1, 2, \dots,
n\}$. Un \omterm{def:b1:logic:sets}{conjunto} $E$ es \emph{finito}\index{conjunto finito} cuando
$E = \emptyset$ o existe una \omterm{def:b1:logic:inj}{biyección} de $\intint{1}{n}$ sobre $E$
para algún $n \in \N^*$; ese $n$ es único (\cref{thm:b1:counting:welldef})
y es el \emph{cardinal}\index{cardinal} de $E$, escrito $\abs{E}$ (con
$\abs{\emptyset} = 0$).
\end{definition}

\begin{theorem}[El cardinal está bien definido]\label{thm:b1:counting:welldef}
Si $m \neq n$, no existe ninguna \omterm{def:b1:logic:inj}{biyección} de $\intint{1}{m}$ sobre
$\intint{1}{n}$. Con más precisión: si $m > n$, no existe ninguna
\omterm{def:b1:logic:inj}{inyección} de $\intint{1}{m}$ en $\intint{1}{n}$.
\end{theorem}

\begin{proof}
Demostramos por inducción sobre $n$ el \omterm{def:b1:logic:statement}{enunciado}: \emph{para todo $m > n$
no hay ninguna \omterm{def:b1:logic:inj}{inyección} $\intint{1}{m} \to \intint{1}{n}$}. Para
$n = 0$ el \omterm{def:b1:logic:sets}{conjunto} de llegada es vacío y $m \geq 1$: no existe
\omterm{def:b1:logic:map}{aplicación} alguna. Supongamos el \omterm{def:b1:logic:statement}{enunciado} para $n$ y sea $f \colon
\intint{1}{m} \to \intint{1}{n+1}$ una \omterm{def:b1:logic:inj}{inyección} con $m > n + 1$. Si el
valor $n + 1$ no se alcanza, $f$ es una \omterm{def:b1:logic:inj}{inyección} en $\intint{1}{n}$,
en contra de la hipótesis de inducción. En caso contrario,
$f(a) = n + 1$ para exactamente un $a$; se intercambian $f(a)$ y $f(m)$
(formalmente: se compone con la transposición de los dos valores), de
modo que la nueva \omterm{def:b1:logic:inj}{inyección} $g$ cumple $g(m) = n + 1$. Entonces la
restricción de $g$ a $\intint{1}{m-1}$ es una \omterm{def:b1:logic:inj}{inyección} en
$\intint{1}{n}$ con $m - 1 > n$ --- otra contradicción.
\end{proof}

\begin{corollary}[Principio del palomar]\label{cor:b1:counting:pigeonhole}
Si $\abs{E} > \abs{F}$, ninguna \omterm{def:b1:logic:map}{aplicación} $f \colon E \to F$ es
\omterm{def:b1:logic:inj}{inyectiva}: dos elementos de $E$ comparten imagen\index{principio del palomar}.
\end{corollary}

\begin{proof}
Escríbase $\abs E = m$, $\abs F = n$ con $m > n$, y elíjanse biyecciones
$u \colon \intint1m \to E$ y $v \colon F \to \intint1n$. Si $f$ fuese
\omterm{def:b1:logic:inj}{inyectiva}, $v \circ f \circ u$ sería una \omterm{def:b1:logic:inj}{inyección} de $\intint1m$ en
$\intint1n$ (composición de inyecciones,
\cref{prop:b1:logic:comp}), en contra
del~\cref{thm:b1:counting:welldef}.
\end{proof}

\begin{remark}[Interludio: ¿por qué el intercambio en la demostración?]\label{rem:b1:counting:swapinterlude}
La demostración del~\cref{thm:b1:counting:welldef} contiene el primer
paso realmente ingenioso del capítulo, que vale la pena repasar
despacio. El obstáculo: para aplicar la hipótesis de inducción se
quiere borrar el último punto $m$ de salida \emph{y} el último punto
$n+1$ de llegada, pero $f$ puede enviar otro punto $a$ a $n + 1$, y
entonces borrar el punto de llegada estropea la \omterm{def:b1:logic:map}{aplicación} en otro
sitio. El remedio: componer $f$ con la transposición de los dos
\emph{valores} $f(a)$ y $f(m)$ --- una \omterm{def:b1:logic:inj}{biyección} del \omterm{def:b1:logic:sets}{conjunto} de
llegada, con lo que la \omterm{def:b1:logic:inj}{inyectividad} se conserva ---, tras lo cual el
valor problemático $n + 1$ queda en la posición inofensiva $m$ y los
dos borrados son limpios. Este patrón de «normalizar primero, cortar
después» reaparece: es como la recurrencia de los desarreglos redirige
$\sigma^{-1}(n+1)$ en el problema del fin de semana de este capítulo, y
como se remiendan las permutaciones en todo el problema
del~\cref{ch:b1:structures} sobre el grupo simétrico.
\end{remark}

\begin{proposition}[Inyecciones, sobreyecciones y cardinal]\label{prop:b1:counting:injsur}
Sean $E, F$ \omterm{def:b1:counting:card}{conjuntos finitos} con $\abs{E} = \abs{F}$, y sea $f \colon E
\to F$. Entonces
\[
f \text{ inyectiva} \iff f \text{ sobreyectiva} \iff f \text{ biyectiva}.
\]
\end{proposition}

\begin{proof}
Supongamos $f$ \omterm{def:b1:logic:inj}{inyectiva}. Entonces $f$ es una \omterm{def:b1:logic:inj}{biyección} de $E$ sobre
$f(E)$, luego $\abs{f(E)} = \abs{E} = \abs{F}$. Si $f(E)$ dejase de
alcanzar un punto $y_0$ de $F$, entonces $f$ sería una \omterm{def:b1:logic:inj}{inyección} de $E$
en $F \setminus \{y_0\}$, un \omterm{def:b1:logic:sets}{conjunto} de \omterm{def:b1:counting:card}{cardinal} $\abs{F} - 1 < \abs{E}$
--- imposible por el principio del palomar. Luego $f(E) = F$: $f$ es
\omterm{def:b1:logic:inj}{sobreyectiva} y, por tanto, \omterm{def:b1:logic:inj}{biyectiva}.

Supongamos $f$ \omterm{def:b1:logic:inj}{sobreyectiva}. Elíjase para cada $y \in F$ una
\omterm{def:b1:logic:map}{imagen recíproca} $s(y) \in E$; entonces $f \circ s = \mathrm{id}_F$,
luego $s$ es \omterm{def:b1:logic:inj}{inyectiva} (\cref{prop:b1:logic:comp}). Por el párrafo
anterior aplicado a $s$ (los \omterm{def:b1:counting:card}{cardinales} son iguales), $s$ es
\omterm{def:b1:logic:inj}{biyectiva}. De $f \circ s = \mathrm{id}_F$ se obtiene
$f = \mathrm{id}_F \circ s^{-1} = s^{-1}$, luego $f$ es \omterm{def:b1:logic:inj}{biyectiva}. Por
último, una \omterm{def:b1:logic:map}{aplicación} \omterm{def:b1:logic:inj}{biyectiva} es por definición \omterm{def:b1:logic:inj}{inyectiva} y
\omterm{def:b1:logic:inj}{sobreyectiva}, lo que cierra el ciclo de implicaciones.
\end{proof}

\begin{example}[La finitud es esencial]\label{ex:b1:counting:finiteonly}
Sobre un \omterm{def:b1:logic:sets}{conjunto} \emph{\omterm{def:b1:counting:card}{finito}}, la~\cref{prop:b1:counting:injsur} es un
atajo poderoso: toda \omterm{def:b1:logic:map}{aplicación} \omterm{def:b1:logic:inj}{inyectiva} de $E$ en sí mismo es
automáticamente una \omterm{def:b1:counting:objects}{permutación} de $E$ --- la mitad de la \omterm{def:b1:logic:inj}{biyectividad}
sale gratis. Las dos implicaciones se derrumban en los \omterm{def:b1:logic:sets}{conjuntos}
infinitos: $n \mapsto n + 1$ es \omterm{def:b1:logic:inj}{inyectiva} de $\N$ en $\N$ pero no
alcanza $0$, y la \omterm{def:b1:logic:map}{aplicación} $\N \to \N$ que envía $0 \mapsto 0$ y
$n \mapsto n - 1$ para $n \geq 1$ es \omterm{def:b1:logic:inj}{sobreyectiva} pero no \omterm{def:b1:logic:inj}{inyectiva}.
Siempre que se invoca esta proposición, la hipótesis de finitud está
haciendo un trabajo real --- un tema que el problema del fin de semana
del~\cref{ch:b1:logic} explora desde el otro lado, donde los \omterm{def:b1:logic:sets}{conjuntos}
infinitos son precisamente los que admiten tales aplicaciones de un
\omterm{def:b1:logic:sets}{conjunto} en sí mismo.
\end{example}

\begin{example}[La mitad del trabajo, gratis]\label{ex:b1:counting:mod7}
Considérese la \omterm{def:b1:logic:map}{aplicación} $f$ en $\{0, 1, \dots, 6\}$ que envía $k$ al
resto de la división de $3k$ por $7$; su tabla de valores es
\[
0,\ 3,\ 6,\ 2,\ 5,\ 1,\ 4 .
\]
¿Es $f$ una \omterm{def:b1:logic:inj}{biyección}? Basta con la \omterm{def:b1:logic:inj}{inyectividad}
(\cref{prop:b1:counting:injsur}): si $3k$ y $3k'$ dejan el mismo resto,
$7$ divide a $3(k - k')$ y, como $7$ es primo y no divide a $3$, divide
a $k - k'$ (lema de Euclides, usado aquí al nivel del volumen anterior y
demostrado en el~\cref{ch:b1:arith}); con $\abs{k - k'} \leq 6$ esto
obliga a $k = k'$. La \omterm{def:b1:logic:inj}{sobreyectividad} sale gratis --- no hace falta
resolver $3k \equiv c$ para cada $c$, aunque la tabla confirme que cada
valor aparece exactamente una vez. El atajo es un caballo de batalla:
demuestra la invertibilidad de la multiplicación modular
(\cref{ch:b1:arith}), sostiene el emparejamiento del teorema de Wilson
y vuelve en álgebra lineal como «un endomorfismo de un espacio de
dimensión finita es \omterm{def:b1:logic:inj}{inyectivo} si y solo si es \omterm{def:b1:logic:inj}{sobreyectivo}»
(\cref{ch:b1:findim}).
\end{example}

\section{Los principios de recuento}

\begin{proposition}[Reglas de la suma y del producto]\label{prop:b1:counting:rules}
Sean $E, F$ \omterm{def:b1:counting:card}{conjuntos finitos}.
\begin{enumerate}
  \item Si $E \cap F = \emptyset$, entonces $\abs{E \cup F} = \abs{E} +
        \abs{F}$; más en general, para una \omterm{thm:b1:logic:partition}{partición} de $E$ en trozos
        $E_1, \dots, E_k$, $\abs{E} = \sum_i \abs{E_i}$.
  \item En general, $\abs{E \cup F} = \abs{E} + \abs{F} - \abs{E \cap
        F}$.
  \item $\abs{E \times F} = \abs{E} \times \abs{F}$.
  \item El \omterm{def:b1:logic:sets}{conjunto} $F^E$ de todas las aplicaciones de $E$ en $F$
        cumple $\abs{F^E} = \abs{F}^{\abs{E}}$.
  \item $\abs{\mathcal{P}(E)} = 2^{\abs{E}}$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Se concatenan enumeraciones: si $E = \{x_1, \dots, x_m\}$ y
$F = \{y_1, \dots, y_n\}$ sin repeticiones, entonces $x_1, \dots, x_m,
y_1, \dots, y_n$ enumera $E \cup F$ sin repetición (por la
disyunción de los dos \omterm{def:b1:logic:sets}{conjuntos}). La inducción lo extiende a $k$
trozos.

(2) $E \cup F$ es la unión disjunta de $E$ y $F \setminus E$, y $F$ es
la unión disjunta de $F \cap E$ y $F \setminus E$; luego
$\abs{E \cup F} = \abs{E} + \abs{F \setminus E} = \abs{E} + \abs{F} -
\abs{E \cap F}$.

(3) $E \times F$ es la unión disjunta, para $x \in E$, de los \omterm{def:b1:logic:sets}{conjuntos}
$\{x\} \times F$, cada uno de \omterm{def:b1:counting:card}{cardinal} $\abs{F}$; aplíquese (1).

(4) Una \omterm{def:b1:logic:map}{aplicación} de $E = \{x_1, \dots, x_m\}$ en $F$ es exactamente
la elección de la $m$-tupla $(f(x_1), \dots, f(x_m)) \in F^m$; esta
correspondencia es una \omterm{def:b1:logic:inj}{biyección}, y $\abs{F^m} = \abs{F}^m$ por (3) e
inducción.

(5) Los subconjuntos de $E$ se corresponden biyectivamente con las
aplicaciones $E \to \{0, 1\}$ (envíese $A$ a su función indicadora);
aplíquese (4).
\end{proof}

\begin{example}[Contar por el complementario]\label{ex:b1:counting:complement}
¿Cuántos códigos PIN de $4$ cifras (cifras $0$--$9$, el orden importa,
se permiten repeticiones) contienen \emph{al menos una} cifra repetida?
Contarlos directamente obliga a manejar los casos «exactamente una
pareja, dos parejas, un trío, un cuarteto» --- cinco configuraciones
que se solapan. Cuéntese en su lugar el complementario: los códigos
son $10^4 = 10\,000$ en total (regla del producto) y los códigos con
cuatro cifras distintas son $10 \times 9 \times 8 \times 7 = 5\,040$
($4$-variaciones), de modo que la respuesta es
\[
10^4 - 10 \cdot 9 \cdot 8 \cdot 7 = 10\,000 - 5\,040 = 4\,960 .
\]
Casi la mitad de los PIN repiten alguna cifra. La idea clave: siempre
que un recuento se formule con «al menos» o «no todos», pruébese
primero con el complementario --- la regla de la suma garantiza que
$\abs{A} = \abs{E} - \abs{\overline A}$, y el complementario suele ser
una sola configuración limpia.
\end{example}

\begin{example}[Caminos en una cuadrícula]\label{ex:b1:counting:paths}
Cuéntense los caminos más cortos de la esquina $(0,0)$ a la esquina
$(4, 3)$ de una cuadrícula, moviéndose cada vez un paso a la derecha
(R) o un paso hacia arriba (A). Todo camino de estos consta
exactamente de $7$ pasos, de los cuales $4$ son R y $3$ son A;
recíprocamente, toda palabra de longitud $7$ en las letras R y A con
cuatro R describe exactamente un camino. Los caminos se corresponden,
pues, biyectivamente con las elecciones de las posiciones de las R:
\[
\binom{7}{4} = 35 .
\]
La idea clave es la \emph{codificación}: el recuento se volvió trivial
en cuanto cada camino se tradujo a una palabra, es decir, a un
subconjunto de posiciones --- una instancia más del lema de que un
recuento correcto es una \omterm{def:b1:logic:inj}{biyección} disfrazada (\cref{met:b1:counting:which}).
\end{example}

\section{Listas, permutaciones, subconjuntos}

\begin{definition}[Variaciones, permutaciones, combinaciones]\label{def:b1:counting:objects}
Sea $E$ un \omterm{def:b1:logic:sets}{conjunto} con $\abs{E} = n$ y sea $0 \leq k \leq n$.
\begin{itemize}
  \item Una \emph{$k$-variación} de $E$ es una $k$-tupla \omterm{def:b1:logic:inj}{inyectiva} de
        elementos de $E$ (una selección ordenada sin repetición);
  \item una \emph{permutación}\index{permutación} de $E$ es una
        \omterm{def:b1:logic:inj}{biyección} de $E$ en sí mismo --- equivalentemente, una
        $n$-variación;
  \item una \emph{$k$-combinación} es un subconjunto de $E$ con $k$
        elementos (una selección no ordenada y sin repetición). Su
        número se escribe $\binom{n}{k}$, que se lee «$n$ sobre $k$»
        \index{coeficiente binomial}.
\end{itemize}
\end{definition}

\begin{theorem}[Los tres recuentos]\label{thm:b1:counting:counts}
Con $n = \abs{E}$ y $0 \leq k \leq n$:
\begin{enumerate}
  \item el número de $k$-variaciones de $E$ es $n (n-1) \cdots
        (n-k+1) = \dfrac{n!}{(n-k)!}$;
  \item el número de permutaciones de $E$ es $n!$;
  \item $\dbinom{n}{k} = \dfrac{n!}{k!\,(n-k)!}$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Se elige la primera coordenada ($n$ maneras), después la segunda
($n - 1$ elecciones restantes), \dots, y por último la $k$-ésima
($n - k + 1$ elecciones). Formalmente, se procede por inducción sobre
$k$. Para $k = 1$ hay $n$ tuplas \omterm{def:b1:logic:inj}{inyectivas} de un término. Supóngase
el recuento para $k - 1$. Cada $k$-variación $(x_1, \dots, x_k)$ se
obtiene a partir de exactamente una $(k-1)$-variación --- su
truncamiento $(x_1, \dots, x_{k-1})$ --- añadiendo una última
coordenada fuera de $\{x_1, \dots, x_{k-1}\}$, para la cual hay
exactamente $n - (k - 1)$ valores disponibles. El truncamiento
reparte así las $k$-variaciones en clases de tamaño común
$n - k + 1$ indexadas por las $(k-1)$-variaciones, y la regla de la
suma da
\[
\frac{n!}{(n-k+1)!}\;(n - k + 1) = \frac{n!}{(n-k)!} .
\]

(2) es (1) con $k = n$.

(3) Cada $k$-subconjunto se ordena de $k!$ maneras distintas en
$k$-variaciones, y toda $k$-variación proviene de exactamente un
subconjunto: luego $\frac{n!}{(n-k)!} = \binom nk \cdot k!$.
\end{proof}

\begin{example}[Mesas redondas: cocientar por una simetría]\label{ex:b1:counting:roundtable}
¿De cuántas maneras pueden sentarse $n$ invitados alrededor de una mesa
redonda, considerando idénticas dos disposiciones cuando cada invitado
tiene los mismos vecinos a izquierda y derecha, es decir, salvo
rotación? Cada disposición circular corresponde a exactamente $n$
disposiciones lineales (córtese el círculo por cualquiera de los $n$
lugares), de modo que los $n!$ órdenes lineales se agrupan en bloques
de $n$:
\[
\frac{n!}{n} = (n-1)! \quad\text{circular seatings.}
\]
Equivalentemente: siéntese en cualquier sitio a un invitado
distinguido (con lo que se elimina la libertad de rotación) y
ordénense después los $n - 1$ invitados restantes en el sentido de las
agujas del reloj. Para $n = 6$: $120$ mesas. Las dos soluciones
ilustran los dos remedios habituales contra el recuento por exceso:
dividir por el número exacto de repeticiones, o \emph{romper la
simetría} fijando un objeto. Ambos exigen que el grupo de repeticiones
tenga el mismo tamaño para toda configuración --- algo que la
demostración anterior de la fórmula $\binom nk = \frac{n!}{k!\,(n-k)!}$
también usó, con $k!$ en lugar de $n$.
\end{example}

\begin{example}[Añadir una restricción]\label{ex:b1:counting:constraint}
Sigamos con la mesa redonda: entre las $(n-1)!$ mesas de $n \geq 3$
invitados, ¿cuántas sientan \emph{separados} (no contiguos) a dos
invitados dados $A$ y $B$? Cuéntese el complementario. Mesas en las que
$A$ y $B$ se sientan juntos: péguense en un solo bloque --- quedan
$n - 1$ objetos alrededor de la mesa, es decir, $(n-2)!$ disposiciones
circulares --- y ordénese después la pareja dentro de su bloque ($2$
maneras): $2\,(n-2)!$ mesas con ellos contiguos. Por tanto
\[
(n-1)! - 2\,(n-2)! = (n-2)!\,\bigl((n - 1) - 2\bigr)
= (n-3)\,(n-2)!
\]
mesas los mantienen separados. Comprobaciones: $n = 3$ da $0$
(alrededor de un triángulo todos se tocan) y $n = 4$ da $2$, fáciles de
enumerar a mano. El truco del pegado --- tratar un bloque forzado como
un solo objeto y contar después sus disposiciones internas --- es el
remedio estándar para las restricciones de contigüidad, lineales o
circulares.
\end{example}

\begin{proposition}[Identidades básicas]\label{prop:b1:counting:identities}
Para $0 \leq k \leq n$:
\[
\binom{n}{k} = \binom{n}{n-k},
\qquad
\binom{n}{k} = \binom{n-1}{k-1} + \binom{n-1}{k}
\quad (1 \leq k \leq n-1),
\qquad
\sum_{k=0}^{n} \binom{n}{k} = 2^n .
\]
\end{proposition}

\begin{proof}
Primera identidad: $A \mapsto E \setminus A$ es una \omterm{def:b1:logic:inj}{biyección} entre los
$k$-subconjuntos y los $(n-k)$-subconjuntos. Regla de
Pascal\index{regla de Pascal}: fíjese un elemento $a \in E$; los
$k$-subconjuntos se reparten entre los que contienen $a$ (elíjanse los
$k - 1$ restantes: $\binom{n-1}{k-1}$) y los que evitan $a$
($\binom{n-1}{k}$). Tercera identidad: los dos miembros cuentan todos
los subconjuntos de $E$, repartidos por tamaño en el de la izquierda
(\cref{prop:b1:counting:rules}~(1) y (5)).
\end{proof}

\begin{theorem}[Teorema del binomio]\label{thm:b1:counting:binomial}
Para todos $a, b$ de un anillo conmutativo (por ejemplo $\R$ o $\C$) y todo $n \in \N$:
\[
(a + b)^n = \sum_{k=0}^{n} \binom{n}{k} a^k b^{\,n-k} .
\]
\end{theorem}

\begin{proof}
Al desarrollar $(a+b)(a+b)\cdots(a+b)$ por distributividad se obtiene
un término por cada elección, en cada factor, de $a$ o de $b$: el
término $a^k b^{n-k}$ aparece una vez por cada manera de elegir cuáles
$k$ de los $n$ factores aportan $a$ --- es decir, $\binom nk$ veces.
(Alternativamente: inducción sobre $n$ usando la regla de Pascal.)
\end{proof}

\begin{example}\label{ex:b1:counting:binomial}
Dos especializaciones clásicas: $a = b = 1$ recupera
$\sum_k \binom nk = 2^n$; $a = -1$, $b = 1$ da
$\sum_{k} (-1)^k \binom nk = 0$ para $n \geq 1$: entre los
subconjuntos de un \omterm{def:b1:logic:sets}{conjunto} no vacío, exactamente la mitad tienen
\omterm{def:b1:counting:card}{cardinal} par.
\end{example}

\begin{example}[Una identidad, dos demostraciones]\label{ex:b1:counting:threen}
La especialización $a = 2$, $b = 1$ del teorema del binomio dice
\[
\sum_{k=0}^{n} \binom nk\,2^k = 3^n .
\]
He aquí la misma identidad sin nada de álgebra. El miembro derecho
cuenta las palabras de longitud $n$ sobre el alfabeto $\{0, 1, 2\}$
(regla del producto). Clasifíquese cada palabra por el \omterm{def:b1:logic:sets}{conjunto} $K$ de
las posiciones que llevan una letra no nula: elegir $K$ con
$\abs K = k$ cuesta $\binom nk$, y después cada posición de $K$ lleva
independientemente $1$ o $2$: $2^k$ maneras. La regla de la suma sobre
$k$ da el miembro izquierdo. Más allá del placer de que coincidan, las
dos demostraciones tienen virtudes distintas: la algebraica se
generaliza a cualquier valor de $a$; la combinatoria \emph{explica} la
fórmula y se adapta a restricciones (prohibir la letra $2$ en la última
posición, por ejemplo) que ninguna sustitución captura. Mantener vivas
las dos técnicas es la destreza práctica que entrena este capítulo.
\end{example}

\begin{method}[¿Qué recuento se aplica?]\label{met:b1:counting:which}
Antes de calcular, respóndanse dos preguntas sobre la selección: ¿importa
el \emph{orden}?, ¿se permiten \emph{repeticiones}?
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{l|c|c}
 & importa el orden & no importa el orden \\
\hline
sin repetición & $\dfrac{n!}{(n-k)!}$ & $\dbinom{n}{k}$ \\[6pt]
con repetición & $n^k$ & (\cref{exo:b1:counting:10}) \\
\end{tabular}
\end{center}
Búsquese después una \omterm{def:b1:logic:inj}{biyección} o una \omterm{thm:b1:logic:partition}{partición} que reduzca el problema a
estos recuentos modelo; un recuento correcto es una \omterm{def:b1:logic:inj}{biyección} disfrazada.
\end{method}

\begin{remark}[Errores frecuentes al contar]\label{rem:b1:counting:pitfalls}
\begin{enumerate}
  \item \emph{Sumar casos no disjuntos.} La regla de la suma exige una
        \omterm{thm:b1:logic:partition}{partición}; si una configuración puede cumplir dos casos a la
        vez, se cuenta dos veces --- el remedio es la
        inclusión--exclusión (\cref{thm:b1:counting:inclexcl}) o un
        reparto de casos más fino.
  \item \emph{Ordenado frente a no ordenado.} Elegir «un comité de dos»
        es $\binom n2$, no $n(n-1)$: decídase \emph{antes de calcular}
        si la selección lleva un orden y, si el recuento ordenado
        resulta más fácil, divídase al final por el número de
        ordenaciones --- pero solo cuando cada objeto no ordenado
        provenga del \emph{mismo} número de objetos ordenados.
  \item \emph{Elecciones por etapas que no son independientes.} La regla
        del producto exige que el número de opciones de cada etapa no
        dependa de las elecciones anteriores. «Elíjase un capitán y
        después un subcapitán distinto» está bien ($n(n-1)$); «elíjanse
        dos jugadores que se lleven bien» no es un producto por etapas
        en absoluto.
  \item \emph{Contar dos veces por construcción.} Construir cada objeto
        dos veces --- por ejemplo, contar las manos con \emph{al menos}
        un as como (elegir un as) $\times$ (elegir $4$ cartas más) ---
        cuenta de más las manos con dos ases. «Al menos» pide casi
        siempre el complementario
        (\cref{ex:b1:counting:complement}).
\end{enumerate}
\end{remark}

\begin{example}[Un recuento de póquer]\label{ex:b1:counting:poker}
De una baraja de $52$ cartas, el número de manos de $5$ cartas es
$\binom{52}{5} = 2\,598\,960$. Manos con exactamente un as: elíjase el
as ($4$ maneras) y después $4$ cartas entre las $48$ que no son ases:
$4 \binom{48}{4} = 778\,320$. La regla del producto se aplica porque la
elección se reparte en etapas independientes.
\end{example}

\begin{method}[Doble recuento]\label{met:b1:counting:doublecount}
Para demostrar una identidad entre dos expresiones combinatorias,
búsquese un único \omterm{def:b1:counting:card}{conjunto finito} que ambos miembros cuenten
---típicamente un \omterm{def:b1:logic:sets}{conjunto} de \emph{pares}--- y evalúese su
\omterm{def:b1:counting:card}{cardinal} de dos maneras distintas. El prototipo es el \emph{lema de
los apretones de manos}: en una fiesta, cuéntense los pares (persona,
mano estrechada). Sumando sobre las personas se obtiene $\sum_p d_p$
(el número de apretones de cada persona $p$); sumando sobre los
apretones se obtiene el doble del número de apretones (cada uno
involucra a dos personas). Luego $\sum_p d_p$ es par --- de modo que el
número de personas que dieron un número impar de apretones es siempre
par, conclusión nada trivial obtenida sin fórmula alguna. El mismo
motor mueve el~\cref{exo:b1:counting:12} y varias preguntas del
problema del fin de semana.
\end{method}

\begin{example}[El subconjunto medio]\label{ex:b1:counting:averagesubset}
¿Cuál es el \omterm{def:b1:counting:card}{cardinal} medio de un subconjunto de un \omterm{def:b1:logic:sets}{conjunto} $E$ de $n$
elementos, siendo los $2^n$ subconjuntos igualmente probables? Cuéntense
dos veces los pares $(A, a)$ con $a \in A$: sumando sobre los
subconjuntos se obtiene $\sum_A \abs A$, el total buscado; sumando sobre
los elementos se obtiene $n \cdot 2^{n-1}$ (cada uno de los $n$
elementos está exactamente en la mitad de los subconjuntos ---
emparéjese cada $A$ que contiene $a$ con $A \setminus \{a\}$). Por tanto
\[
\frac{1}{2^n}\sum_{A \subseteq E} \abs A
= \frac{n\,2^{n-1}}{2^n} = \frac n2 :
\]
los subconjuntos están, en promedio, medio llenos --- como también
predice la simetría $A \leftrightarrow \overline A$ (que empareja los
tamaños $k$ y $n - k$). Dos demostraciones, una sola respuesta, y ambas
evitan el cálculo directo $\sum_k k\binom nk$
del~\cref{exo:b1:counting:5}: un emparejamiento bien elegido sustituye a
menudo a una identidad.
\end{example}

\section{Inclusión--exclusión}

\begin{theorem}[Inclusión--exclusión]\label{thm:b1:counting:inclexcl}
Para \omterm{def:b1:counting:card}{conjuntos finitos} $A_1, \dots, A_p$:
\[
\Bigl|\, \bigcup_{i=1}^{p} A_i \,\Bigr|
= \sum_{\emptyset \neq I \subseteq \intint{1}{p}}
(-1)^{\abs{I}+1} \Bigl|\, \bigcap_{i \in I} A_i \,\Bigr| .
\]
Para $p = 3$:
$\abs{A \cup B \cup C} = \abs A + \abs B + \abs C - \abs{A \cap B} -
\abs{A \cap C} - \abs{B \cap C} + \abs{A \cap B \cap C}$.
\end{theorem}

\begin{proof}
Fíjese un elemento $x$ de la unión y cuéntese su contribución al
miembro derecho. Sea $J = \{i : x \in A_i\}$, de \omterm{def:b1:counting:card}{cardinal} $m \geq 1$. El
elemento $x$ se cuenta una vez en $\abs{\bigcap_{i \in I} A_i}$
exactamente cuando $\emptyset \neq I \subseteq J$, con signo
$(-1)^{\abs I + 1}$; su contribución total es
\[
\sum_{k=1}^{m} \binom{m}{k} (-1)^{k+1}
= 1 - \sum_{k=0}^{m} \binom mk (-1)^k = 1 - 0 = 1
\]
por el~\cref{ex:b1:counting:binomial}. Así pues, cada elemento de la
unión se cuenta exactamente una vez.
\end{proof}

\begin{example}[Contar enteros coprimos]\label{ex:b1:counting:coprime}
¿Cuántos enteros de $\intint1{120}$ son coprimos con
$120 = 2^3 \times 3 \times 5$? Un entero comparte un factor con $120$
exactamente cuando es divisible por $2$, $3$ o $5$, así que se cuenta el
complementario de $A_2 \cup A_3 \cup A_5$, donde $A_d$ reúne los
múltiplos de $d$. Dentro de $\intint1{120}$, los múltiplos de $d$ son
$120/d$ siempre que $d$ divida a $120$ --- sin necesidad de partes
enteras --- y $A_2 \cap A_3 = A_6$, etc. Inclusión--exclusión:
\[
\abs{A_2 \cup A_3 \cup A_5}
= 60 + 40 + 24 - 20 - 12 - 8 + 4 = 88 ,
\]
luego $120 - 88 = 32$ enteros son coprimos con $120$. Es instructivo
reagrupar el cálculo como un producto:
\[
120 - 88 = 120\Bigl(1 - \frac12\Bigr)\Bigl(1 -
\frac13\Bigr)\Bigl(1 - \frac15\Bigr) = 120 \cdot \frac12 \cdot
\frac23 \cdot \frac45 = 32 :
\]
desarrollar los tres paréntesis reproduce exactamente los ocho términos
con signo de la inclusión--exclusión, uno por cada subconjunto de
$\{2, 3, 5\}$. Esta forma de producto define la función indicatriz de
Euler, cuyo papel aritmético asoma con las congruencias
del~\cref{ch:b1:arith} y se desarrolla en el volumen del segundo año.
\end{example}

\begin{example}[Desarreglos]\label{ex:b1:counting:derangement}
Un \emph{desarreglo} es una \omterm{def:b1:counting:objects}{permutación} sin puntos fijos. Sea $A_i$ el
\omterm{def:b1:logic:sets}{conjunto} de las permutaciones de $\intint{1}{n}$ que fijan $i$;
entonces $\abs{\bigcap_{i \in I} A_i} = (n - \abs I)!$, y la
inclusión--exclusión cuenta las permutaciones con al menos un punto
fijo; los desarreglos son
\[
D_n = n! \sum_{k=0}^{n} \frac{(-1)^k}{k!} .
\]
Como $\sum (-1)^k / k! \to \eu^{-1}$ (véase el~\cref{ch:b1:series}),
alrededor del $37\,\%$ de todas las permutaciones son desarreglos, sea cual
sea $n$.
\end{example}

\begin{remark}[Dónde se usa este capítulo]\label{rem:b1:counting:whereused}
Los \omterm{def:b1:counting:objects}{coeficientes binomiales} son los objetos más reutilizados de este
capítulo: mueven el teorema del binomio en el~\cref{ch:b1:poly}
(desarrollo de $(X + a)^n$), la fórmula de Leibniz para la derivada
$n$-ésima de un producto en el~\cref{ch:b1:derivative} y los
coeficientes de los desarrollos de Taylor en el~\cref{ch:b1:taylor}. Las
permutaciones vuelven como grupo --- con la signatura construida a
partir del recuento de inversiones --- en el~\cref{ch:b1:structures}, y
la signatura define a su vez los determinantes en el~\cref{ch:b1:det}.
La inclusión--exclusión y los principios de recuento son el esqueleto
\omterm{def:b1:counting:card}{finito} de la probabilidad discreta, desarrollada en el volumen del
segundo año; los números de desarreglos del~\cref{ex:b1:counting:derangement}
se estudian a fondo en el problema del fin de semana.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b1:counting:1}
Una matrícula consta de dos letras (A--Z), después tres cifras y
después dos letras. ¿Cuántas matrículas son posibles? ¿Y cuántas sin
ninguna letra repetida entre las cuatro?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:2}
¿Cuántos anagramas (reordenaciones de las letras, con sentido o sin él)
tiene la palabra \textsc{cuerpo}? ¿Y \textsc{banana}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:3}
Se elige un comité de $4$ personas entre $7$ mujeres y $5$ hombres.
¿Cuántos comités hay en total? ¿Cuántos con exactamente $2$ mujeres?
¿Cuántos con al menos un hombre?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:4}
Demuéstrese que en cualquier grupo de $13$ personas dos comparten mes de
nacimiento, y que entre $n + 1$ enteros cualesquiera elegidos de
$\intint{1}{2n}$ hay dos consecutivos. \emph{(Palomar las dos veces:
nómbrense las cajas.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:counting:5}
Calcúlese $\sum_{k=0}^{n} k \binom{n}{k}$. \emph{Indicación: derívese
$(1 + x)^n$, o úsese $k \binom nk = n \binom{n-1}{k-1}$ (demuéstrese).}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:6}
¿Cuántas aplicaciones estrictamente crecientes hay de $\intint{1}{k}$
en $\intint{1}{n}$? Dedúzcase el número de aplicaciones crecientes (no
necesariamente en sentido estricto). \emph{Indicación para el segundo
recuento: $f$ creciente $\mapsto$ $g(i) = f(i) + i - 1$.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:7}
(Vandermonde) Demuéstrese, contando los $k$-subconjuntos de un
\omterm{def:b1:logic:sets}{conjunto} repartido en dos bloques de tamaños $m$ y $n$:
\[
\binom{m+n}{k} = \sum_{j=0}^{k} \binom{m}{j} \binom{n}{k-j} .
\]
Dedúzcase $\sum_{j=0}^{n} \binom nj^2 = \binom{2n}{n}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:8}
¿Cuántos enteros de $\intint{1}{1000}$ son divisibles por $2$, por $3$ o
por $5$? (Inclusión--exclusión; $\lfloor 1000/6 \rfloor$ cuenta los
múltiplos de $6$, etc.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:9}
Cuéntense las sobreyecciones de un \omterm{def:b1:logic:sets}{conjunto} de $4$ elementos sobre un
\omterm{def:b1:logic:sets}{conjunto} de $2$ elementos; después, sobre uno de $3$ elementos.
\emph{Indicación: cuéntense las aplicaciones no \omterm{def:b1:logic:inj}{sobreyectivas} con
inclusión--exclusión sobre los valores no alcanzados.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:counting:10}
(Estrellas y barras) Demuéstrese que el número de selecciones de $k$
objetos entre $n$ \emph{con} repetición, sin tener en cuenta el orden
---equivalentemente, el número de $(x_1, \dots, x_n) \in \N^n$ con
$x_1 + \dots + x_n = k$--- es $\binom{n + k - 1}{k}$. \emph{Indicación:
codifíquese una solución como una fila de $k$ estrellas y $n - 1$ barras.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:counting:11}
Demuéstrese con detalle la fórmula del~\cref{ex:b1:counting:derangement}
para $D_n$ y dedúzcase $n! = \sum_{k=0}^{n} \binom{n}{k} D_{n-k}$
(demuéstrese esta identidad también de forma directa, clasificando las
permutaciones por su \omterm{def:b1:logic:sets}{conjunto} de puntos fijos).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:counting:12}
Para $n \in \N^*$, demuéstrese mediante un doble recuento de pares
(subconjunto, elemento marcado):
\[
\sum_{k=1}^{n} k \binom{n}{k} = n\, 2^{n-1},
\qquad\text{y después}\qquad
\sum_{k=1}^{n} k^2 \binom{n}{k} = n(n+1)\, 2^{n-2} .
\]
\emph{Para la segunda: cuéntense los pares de elementos marcados, iguales o no.}
\end{exercise}

\section{Problema: desarreglos, o las cartas mal repartidas}

\begin{problem}\label{pb:b1:counting:1}
Una secretaria mete $n$ cartas al azar en $n$ sobres con destinatario:
¿qué probabilidad hay de que \emph{nadie} reciba su carta? Esta
pregunta clásica (Montmort, 1708) conduce a los números de desarreglos
$D_n$ del~\cref{ex:b1:counting:derangement}. La fórmula de
inclusión--exclusión es solo la jugada de apertura: este problema
desarrolla las recurrencias que calculan $D_n$, dos demostraciones
independientes más de la fórmula, el llamativo teorema de que $D_n$ es
el entero más próximo a $n!/\eu$, la distribución completa de los
puntos fijos de una \omterm{def:b1:counting:objects}{permutación} aleatoria y la curiosa aritmética de la
sucesión $(D_n)$.\index{desarreglo}
En todo el problema, $D_n$ denota el número de desarreglos
(permutaciones sin puntos fijos) de $\intint1n$, con el convenio
$D_0 = 1$ (la \omterm{def:b1:counting:objects}{permutación} vacía no tiene puntos fijos).

\medskip
\noindent\textbf{Parte I --- Casos pequeños y censo de puntos
fijos.}
\begin{enumerate}
  \item Calcúlense $D_1, D_2, D_3$ directamente, y $D_4$ enumerando los
        desarreglos de $\{1, 2, 3, 4\}$ agrupados según el valor de
        $\sigma(1)$. (Debe salir $D_4 = 9$.)
  \item Para $0 \leq k \leq n$, pruébese que el número $P_k(n)$ de
        permutaciones de $\intint1n$ con \emph{exactamente} $k$ puntos
        fijos es $\binom nk D_{n-k}$.
  \item Compruébese el censo para $n = 4$: calcúlense $P_0(4), \dots,
        P_4(4)$ y véase que suman $4! = 24$. ¿Qué es más probable con
        cuatro cartas: ningún acierto o exactamente un acierto?
  \item Mediante un doble recuento (\cref{met:b1:counting:doublecount})
        de los pares $(\sigma, i)$ con $\sigma(i) = i$, pruébese que
        \[
        \sum_{\sigma} \abs{\mathrm{Fix}(\sigma)} = n! :
        \]
        en promedio, una \omterm{def:b1:counting:objects}{permutación} aleatoria tiene \emph{exactamente
        un} punto fijo, sea cual sea $n \geq 1$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Dos recurrencias y dos demostraciones
nuevas de la fórmula.}
\begin{enumerate}[resume]
  \item Demuéstrese combinatoriamente, para $n \geq 1$:
        \[
        D_{n+1} = n\,(D_n + D_{n-1}) .
        \]
        (Clasifíquense los desarreglos $\sigma$ de $\intint1{n+1}$ según
        $j = \sigma(n+1)$ y después según si $\sigma(j) = n + 1$; en el
        caso $\sigma(j) \neq n+1$, constrúyase una \omterm{def:b1:logic:inj}{biyección} con los
        desarreglos de $\intint1n$ redirigiendo hacia $j$ la
        \omterm{def:b1:logic:map}{imagen recíproca} de $n + 1$.) Compruébese la recurrencia
        numéricamente hasta $D_6$.
  \item Poniendo $u_n = D_n - n D_{n-1}$, dedúzcase de la pregunta 5
        que $u_{n+1} = -u_n$, y conclúyase la segunda recurrencia:
        \[
        D_n = n D_{n-1} + (-1)^n \qquad (n \geq 1).
        \]
  \item A partir de la pregunta 6, demuéstrese por inducción la fórmula
        del~\cref{ex:b1:counting:derangement},
        \[
        D_n = n! \sum_{k=0}^{n} \frac{(-1)^k}{k!},
        \]
        --- una demostración totalmente independiente de la inclusión--exclusión.
  \item (Inversión binomial) Sean $(a_n)$ y $(b_n)$ dos sucesiones
        tales que $a_n = \sum_{k=0}^n \binom nk b_k$ para todo $n$.
        Demuéstrese que
        \[
        b_n = \sum_{k=0}^{n} (-1)^{n-k} \binom nk a_k
        \qquad (n \in \N).
        \]
        (Establézcase primero la \emph{identidad trinomial}
        $\binom nk \binom kj = \binom nj \binom{n-j}{k-j}$ y úsese
        después la suma alternada de una fila
        del~\cref{ex:b1:counting:binomial}.)
  \item Aplíquese la pregunta 8 a la identidad
        $n! = \sum_k \binom nk D_{n-k}$ del~\cref{exo:b1:counting:11}
        para obtener una \emph{tercera} demostración de la fórmula
        de $D_n$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El entero más próximo a
$n!/\eu$.} Admítase en esta parte --- la teoría se construye en
el~\cref{ch:b1:series} --- que
$\eu^{-1} = \lim_{n \to \infty} s_n$, donde $s_n = \sum_{k=0}^{n}
\frac{(-1)^k}{k!}$, con la cota estricta de series alternadas
$\abs{\eu^{-1} - s_n} < \frac1{(n+1)!}$ para todo $n$.
\begin{enumerate}[resume]
  \item Pruébese que $\bigl| D_n - n!/\eu \bigr| < \frac1{n+1}$ para
        todo $n \in \N$.
  \item Dedúzcase el teorema estrella: \emph{para todo $n \geq 1$,
        $D_n$ es el entero más próximo a $n!/\eu$}. ¿Por qué necesita el
        argumento que $n \geq 1$?
  \item Determínese el signo del error: pruébese que $D_n > n!/\eu$
        exactamente cuando $n$ es par. (Localícese el primer término
        despreciado de la serie alternada.)
  \item Calcúlense $D_7$ hasta $D_{10}$ con la recurrencia de la
        pregunta 5 y compruébese después $D_{10}$ frente a $10!/\eu$
        ($10! = 3\,628\,800$, $\eu \approx 2.718281828$).
  \item (La probabilidad del guardarropa) Sea $p_n = D_n/n!$ la
        probabilidad de que una \omterm{def:b1:counting:objects}{permutación} tomada al azar
        uniformemente sea un desarreglo. Pruébese que
        $\abs{p_n - \eu^{-1}} < \frac1{(n+1)!}$ y calcúlese $p_6$ con
        cinco decimales. Coméntese: ¿por qué la respuesta a la pregunta
        de Montmort es esencialmente independiente de $n$, ya con una
        docena de cartas?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La distribución de los puntos fijos.}
\begin{enumerate}[resume]
  \item Fíjese $k \in \N$. Pruébese que la proporción de
        permutaciones de $\intint1n$ con exactamente $k$ puntos fijos cumple
        \[
        \frac{P_k(n)}{n!} = \frac{s_{n-k}}{k!}
        \;\xrightarrow[n \to \infty]{}\; \frac{\eu^{-1}}{k!} .
        \]
        (Estos valores límite, que suman $1$, forman la
        \emph{distribución de Poisson} de parámetro $1$, objeto central
        del curso de probabilidad del volumen del segundo año.)
  \item Mediante un doble recuento de las ternas $(\sigma, i, j)$ con
        $i \neq j$ fijados ambos por $\sigma$, pruébese que
        $\sum_\sigma \abs{\mathrm{Fix}(\sigma)}\,
        (\abs{\mathrm{Fix}(\sigma)} - 1) = n!$ para $n \geq 2$.
        Combinado con la pregunta 4: el promedio de
        $\abs{\mathrm{Fix}}^2$ es $2$, de modo que la «dispersión»
        (varianza) del número de puntos fijos vale $1$ --- de nuevo
        independiente de $n$, de nuevo acorde con la ley de Poisson.
  \item Calcúlese la proporción de permutaciones con al menos un punto
        fijo para $n = 4, 5, 6$ (como fracciones y con cuatro
        decimales) y compárese con $1 - \eu^{-1} \approx 0.6321$.
  \item Pruébese directamente --- sin necesidad de límites --- que
        $s_{n+2} - s_n = (-1)^{n+1}\bigl(\frac1{(n+1)!} -
        \frac1{(n+2)!}\bigr)$, y dedúzcase que las probabilidades
        $p_n = s_n$ de la pregunta 14 oscilan: $p_0 > p_2 > p_4 > \dots$
        y $p_1 < p_3 < p_5 < \dots$, decreciendo los valores pares
        (resp.\ creciendo los impares) hacia el límite común $\eu^{-1}$.
  \item (Amigo invisible) $n$ personas sacan cada una un nombre de una
        bolsa; si alguien saca su propio nombre, \emph{todo} el sorteo
        se repite desde cero. Usando el hecho estándar de que un suceso
        de probabilidad $p$ requiere en promedio $1/p$ intentos,
        estímese el número medio de sorteos completos necesarios y
        conclúyase que el procedimiento cuesta en promedio unos
        $\eu \approx 2.72$ sorteos, esencialmente con independencia
        de $n$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- La aritmética de $D_n$, y una
síntesis.}
\begin{enumerate}[resume]
  \item Refínese la pregunta 5: pruébese que, para $j \in \intint2n$
        fijo, los desarreglos de $\intint1n$ con $\sigma(1) = j$ son
        exactamente $D_{n-1} + D_{n-2}$, independientemente de $j$.
        Dedúzcase que $n - 1$ divide a $D_n$ para todo $n \geq 2$.
  \item Demuéstrese que $D_n$ es impar si y solo si $n$ es par.
        (Trabájese módulo $2$ en la recurrencia de la pregunta 6.)
  \item Demuéstrese que $D_n \equiv (-1)^n \pmod n$ para $n \geq 1$ y
        compruébese la congruencia en la última cifra de $D_{10}$.
  \item Pruébese, a partir de la pregunta 6, que
        $\dfrac{D_n}{D_{n-1}} = n + \dfrac{(-1)^n}{D_{n-1}}$ para
        $n \geq 3$, de modo que el cociente de dos números de
        desarreglos consecutivos es \emph{casi exactamente} $n$;
        explíquese en una frase por qué esto es coherente con
        $D_n \approx n!/\eu$.
  \item ¿Dónde ha usado exactamente este problema: (i) las reglas de la
        suma y del producto; (ii) el doble recuento; (iii) el teorema
        del binomio; (iv) la cota admitida de series alternadas? Una
        frase para cada uno.
  \item Síntesis. La fórmula de $D_n$ tiene ya tres demostraciones
        (inclusión--exclusión, recurrencia más inducción, inversión
        binomial). En un párrafo breve, compárese qué \emph{explica}
        cada una: cuál calcula más rápido, cuál se generaliza a otros
        recuentos de puntos fijos y cuál revela por qué aparece $\eu$
        en un problema sobre sobres.
\end{enumerate}
\end{problem}
