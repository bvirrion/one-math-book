\chapter{Números negativos}\label{ch:g7:negatives}

Temperaturas abaixo de \omterm{def:g1:counting:zero}{zero}, andares abaixo do térreo, dinheiro devido:
muitas grandezas descem abaixo de $0$. Os números negativos estendem a
reta numérica para a esquerda do \omterm{def:g1:counting:zero}{zero}. Este capítulo ensina a
compará-los e a somá-los e subtraí-los --- a multiplicação deles espera
o \cref{ch:g8:negprod}.

\section{Números relativos}

\begin{definition}[Números relativos]\label{def:g7:negatives:def}
Um \emph{número relativo}\index{número relativo} é formado por um sinal
($+$ ou $-$) e uma \emph{distância até o \omterm{def:g1:counting:zero}{zero}}: $+3$ e $-3$ estão os
dois à distância $3$ de $0$, em lados opostos. Os números positivos
costumam ser escritos sem o sinal: $+3 = 3$. Os números $+3$ e $-3$ são
\emph{opostos}\index{oposto} um do outro.
\end{definition}

  \draw[-Stealth] (-5.6,0) -- (5.6,0);
  \foreach \x in {-5,...,5} \draw (\x,-0.08) -- (\x,0.08);
  \foreach \x in {-5,...,5} \node[below, font=\small] at (\x,-0.1) {$\x$};
  \fill[omThm] (-3,0) circle (2pt);
  \fill[omDef] (3,0) circle (2pt);
  \draw[Stealth-Stealth, omExo] (-3,0.35) -- (0,0.35)
    node[midway, above, font=\small] {$3$};
  \draw[Stealth-Stealth, omExo] (0,0.35) -- (3,0.35)
    node[midway, above, font=\small] {$3$};
\end{tikzpicture}

{\small A reta numérica estendida para a esquerda: $-3$ e $+3$ são
números opostos, à mesma distância de $0$.}
\end{omfigure}

\begin{method}[Comparar números relativos]\label{met:g7:negatives:compare}
Na reta numérica, menor significa mais à esquerda. Na prática:
\begin{enumerate}
  \item um número negativo é sempre menor que um positivo: $-7 < 2$;
  \item entre dois positivos, vale a ordem de sempre;
  \item entre dois negativos, o de \emph{maior} distância até o \omterm{def:g1:counting:zero}{zero} é
        o \emph{menor}: $-7 < -2$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:negatives:compare}
Ordene do menor para o maior: $3$; $-5$; $0$; $-1.5$; $2.8$. Os
negativos primeiro, com o mais negativo na frente:
\[
-5 < -1.5 < 0 < 2.8 < 3 .
\]
\end{example}

\section{Somar números relativos}

Pense num \omterm{def:g7:negatives:def}{número relativo} como um movimento ao longo da reta: $+5$ quer
dizer ``ande $5$ passos para a direita'' e $-3$ quer dizer ``ande $3$
passos para a esquerda''. Somar números é encadear as caminhadas.

\begin{method}[Somar dois números relativos]\label{met:g7:negatives:add}
\begin{enumerate}
  \item \emph{Mesmo sinal}: some as distâncias e mantenha o sinal
        comum: $(-3) + (-4) = -7$.
  \item \emph{Sinais opostos}: subtraia a distância menor da maior e
        tome o sinal do maior: $(-7) + (+2) = -5$ e
        $(+7) + (-2) = +5$.
  \item Os opostos se cancelam: $(+4) + (-4) = 0$.
\end{enumerate}
\end{method}

\begin{tikzpicture}[scale=1.05]
  \draw[-Stealth] (-5.4,0) -- (4.4,0);
  \foreach \x in {-5,...,4} \draw (\x,-0.07) -- (\x,0.07);
  \foreach \x in {-5,-3,-1,0,2,4}
    \node[below, font=\small] at (\x,-0.09) {$\x$};
  \draw[-Stealth, omThm, very thick] (0,0.35) -- (-3.5,0.35);
  \draw[-Stealth, omDef, very thick] (-3.5,0.75) -- (-1.5,0.75);
  \node[omThm, font=\small] at (-1.75,0.18) {$-3.5$};
  \node[omDef, font=\small] at (-2.5,1.0) {$+2$};
  \fill (-1.5,0) circle (2pt) node[below=6pt, font=\small] {$-1.5$};
\end{tikzpicture}

{\small Somar é caminhar: partindo de $0$, a caminhada $-3.5$ seguida
da caminhada $+2$ chega a $(-3.5) + 2 = -1.5$.}
\end{omfigure}

\begin{example}\label{ex:g7:negatives:add}
Passo a passo, agrupando como convém:
\begin{align*}
(-8) + (+3) &= -5 && \text{(sinais opostos: } 8 - 3 = 5,
\text{ com o sinal do } 8)\\
(-2.5) + (-1.5) &= -4 && \text{(mesmo sinal: some as distâncias)}\\
(+6) + (-9) + (+3) &= (-3) + (+3) = 0 .
\end{align*}
\end{example}

\section{Subtrair números relativos}

\begin{theorem}[Regra da subtração]\label{thm:g7:negatives:sub}
Subtrair um \omterm{def:g7:negatives:def}{número relativo} é o mesmo que \emph{somar o oposto dele}:
\[
a - b = a + (-b) .
\]
\end{theorem}

\begin{proof}[Por que funciona]
$a - b$ é o número que, somado a $b$, dá $a$. Confira que $a + (-b)$
faz o serviço:
$\bigl(a + (-b)\bigr) + b = a + \bigl((-b) + b\bigr) = a + 0 = a$.
\end{proof}

\begin{example}\label{ex:g7:negatives:sub}
Toda subtração vira uma adição e, depois, aplica-se o
\cref{met:g7:negatives:add}:
\begin{align*}
5 - 9 &= 5 + (-9) = -4, \\
(-3) - (+4) &= (-3) + (-4) = -7, \\
(-2) - (-6) &= (-2) + (+6) = +4 .
\end{align*}
A última linha é a famosa: \emph{subtrair um negativo é somar}. Tirar
uma dívida deixa você mais rico!
\end{example}

\begin{example}[Diferenças de temperatura]\label{ex:g7:negatives:temperature}
Num dia, a temperatura passou de $-6$ graus para $+9$ graus. A subida é
a \omterm{ex:g1:subtraction:difference}{diferença}
\[
9 - (-6) = 9 + 6 = 15 \text{ graus}.
\]
Na reta numérica: de $-6$ até $0$ são $6$ graus, e de $0$ até $9$ são
mais $9$.
\end{example}

\section{Coordenadas no plano inteiro}

\begin{definition}[Coordenadas com sinal]\label{def:g7:negatives:coords}
Com dois eixos graduados \omterm{def:g6:lines:perp}{perpendiculares} que se cruzam na origem $O$,
todo ponto do plano ganha duas coordenadas relativas $(x, y)$: $x$
negativo quer dizer à esquerda do eixo vertical, e $y$ negativo quer
dizer abaixo do eixo horizontal.
\end{definition}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g7:negatives:1}
Dê o oposto de: $7$; \ $-3.5$; \ $0$; \ $+12$. Depois complete: ``dois
números são opostos quando a \omterm{def:g1:addition:def}{soma} deles é~?''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:2}
Copie e complete com $<$ ou $>$:
\[
-4 \;?\; 1, \qquad
-4 \;?\; -7, \qquad
-2.5 \;?\; -2.4, \qquad
0 \;?\; -0.1 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:3}
Ordene do menor para o maior: $-3$; \ $2.5$; \ $-0.5$; \ $-3.1$;
\ $0$; \ $1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:4}
Calcule:
\[
(-5) + (-9), \qquad
(-12) + (+7), \qquad
(+3.5) + (-1.5), \qquad
(-6) + (+6).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:5}
Transforme em adições e depois calcule:
\[
4 - 11, \qquad
(-5) - (+3), \qquad
(-7) - (-10), \qquad
0 - (-8).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:6}
Calcule agrupando com esperteza (os opostos primeiro):
\[
(+7) + (-12) + (+3) + (-8), \qquad
(-5.5) + (+9) + (-4.5) + (+1).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:7}
A temperatura era de $-3$ graus de manhã; subiu $8$ graus durante o dia
e depois caiu $12$ graus à noite. Escreva um único cálculo que dê a
temperatura final e calcule-o.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:8}
Marque num sistema de coordenadas: $A(2, 3)$; $B(-3, 1)$; $C(-2, -2)$;
$D(3, -1)$; $E(0, -3)$. Que ponto está sobre um eixo?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:9}
Calcule a \omterm{ex:g1:subtraction:difference}{diferença} entre a temperatura máxima e a mínima: Moscou em
janeiro, máxima $-4$ graus, mínima $-13$ graus; Helsinque, máxima $2$,
mínima $-9$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:10}
Uma mergulhadora está na altitude $-12$ m (doze metros abaixo da
superfície). Ela sobe $7$ m e depois desce $4$ m. Em que altitude ela
está? Quanto ainda precisa subir para chegar à superfície?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:11}
Marque $A(-2, 1)$, $B(2, 1)$, $C(2, -2)$. Encontre as coordenadas do
ponto $D$ para que $ABCD$ seja um retângulo e calcule o \omterm{def:g6:measure:perimeter}{perímetro} dele
(conte os quadradinhos da malha para os comprimentos dos lados).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:negatives:12}
No quadrado mágico abaixo, toda linha, coluna e diagonal tem de ter a
mesma \omterm{def:g1:addition:def}{soma}. Copie e complete:
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|}
\hline
$?$ & $-7$ & $0$ \\
\hline
$-3$ & $-2$ & $?$ \\
\hline
$-4$ & $?$ & $?$ \\
\hline
\end{tabular}
\end{center}
(Comece descobrindo a \omterm{def:g1:addition:def}{soma} mágica pela diagonal que está quase
completa.)
\end{exercise}

\section{Problema: o ano zero que nunca existiu}

\begin{problem}[{Problema de fim de semana --- números relativos na
linha do tempo da história: durações que atravessam a fronteira das
eras, o truque dos astrônomos e o princípio da variação total}]
\label{pb:g7:negatives:1}
A reta numérica deste capítulo tem uma gêmea famosa no mundo real: a
linha do tempo da história, com os anos a.C. se estendendo para a
esquerda e os anos d.C. para a direita. Mas a linha dos historiadores
esconde uma armadilha que já estragou muita conta: \emph{não existe ano
\omterm{def:g1:counting:zero}{zero}} --- o ano 1 a.C. é seguido imediatamente pelo ano 1 d.C. Este
problema calcula com temperaturas, elevadores e contas bancárias e
depois conserta o \omterm{def:g1:counting:zero}{zero} quebrado da história do jeito que os astrônomos
fazem: com \omterm{def:g7:negatives:def}{números relativos}.

\medskip
\noindent\textbf{Parte I --- Aquecimento e uma primeira armadilha.}
\begin{enumerate}
  \item O cume do Monte Branco está na altitude $+4\,808$ m; a margem
        do mar Morto, em $-430$ m. Calcule a \omterm{ex:g1:subtraction:difference}{diferença} de altitude
        entre os dois.
  \item Uma conta bancária tem $-45$ reais (está no vermelho). O dono
        deposita $120$ reais, paga uma conta de $60$ reais e depois
        deposita $15$ reais. Escreva o saldo como uma única cadeia de
        adições e calcule-o. Que depósito único teria levado os $-45$
        iniciais exatamente a $0$?
  \item Júlio César nasceu em $100$ a.C. e morreu em $44$ a.C.
        Contando os anos a.C. como números negativos ($100$ a.C. como
        $-100$ e $44$ a.C. como $-44$), calcule $(-44) - (-100)$:
        quantos anos César viveu?
  \item O imperador Augusto nasceu em $63$ a.C. e morreu em $14$ d.C.
        A mesma receita dá $14 - (-63) = 77$ anos --- mas os
        historiadores insistem que ele morreu aos $76$. A objeção
        deles: a contagem $77$ passa por um ano que nunca existiu.
        Qual?
  \item Os astrônomos consertam a linha do tempo renomeando: os anos
        d.C. mantêm o número ($+14$ continua $+14$), mas $1$ a.C. vira
        $0$, $2$ a.C. vira $-1$ e, em geral, o ano $n$ a.C. vira
        $-(n - 1)$. Converta $63$ a.C. e refaça a conta da questão 4
        com os números dos astrônomos. Agora a resposta satisfaz os
        historiadores?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Calcular atravessando as eras.}
De agora em diante, use a renomeação dos astrônomos em todas as contas,
convertendo de volta para a.C./d.C. no fim (um ano rotulado $0$ ou
negativo $-m$ é o ano $m + 1$ a.C. dos historiadores).
\begin{enumerate}[resume]
  \item Converta para números de astrônomo: $1$ a.C.; $10$ a.C.;
        $476$ d.C.; e $753$ a.C., a fundação lendária de Roma.
  \item Quantos anos se passaram da fundação de Roma ($753$ a.C.) até
        a queda do Império do Ocidente ($476$ d.C.)?
  \item Um cometa volta a cada $76$ anos e apareceu em $12$ a.C. Dê os
        anos dos historiadores para as três aparições seguintes e para
        a aparição imediatamente anterior a $12$ a.C.
  \item Quantos anos se passaram de $5$ a.C. até $5$ d.C.? (A resposta
        tentadora é $10$; a linha do tempo diz outra coisa.)
  \item Uma criança nascida em $3$ a.C. completou dez anos em que ano
        (a.C. ou d.C.)?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O princípio da variação total.}
\begin{enumerate}[resume]
  \item A \emph{distância} entre dois números na reta é o maior menos o
        menor --- sempre positiva. Calcule a distância entre $-7$ e
        $-2$ e depois entre $-3.5$ e $4.5$.
  \item Um elevador parte do andar $0$ e faz os movimentos $+3$, $-5$,
        $+2$, $-1$, $+4$. Calcule o andar final agrupando com esperteza
        (\cref{met:g7:negatives:add}, os opostos primeiro sempre que
        possível). Fazer os movimentos noutra ordem mudaria o destino?
        Por quê?
  \item Dia a dia, a temperatura varia $+2$, $-4$, $+1$, $0$, $-3$,
        $+5$, $-2$ graus ao longo de uma semana. Calcule a
        \emph{variação total}. Se a semana começou a $-2$ graus, em
        quanto ela terminou? Enuncie o princípio: valor final $=$ valor
        inicial $+$ (\omterm{def:g1:addition:def}{soma} de todas as variações).
  \item A caminhada de uma montanhista a traz de volta exatamente à
        altitude de partida. Quanto tem de valer a \omterm{def:g1:addition:def}{soma} de todas as
        variações de altitude dela, e por quê? Das seis variações
        registradas, cinco são $+4$, $-7$, $+2$, $+3$, $-5$ (em
        centenas de metros): encontre a sexta.
  \item Final, de volta a Roma. Um historiador calcula: ``a República
        durou de $509$ a.C. a $27$ a.C., e depois o Império até
        $476$ d.C.; ao todo, de $509$ a.C. a $476$ d.C., ou seja,
        $509 + 476 = 985$ anos de Estado romano''. Refaça a conta com
        os números dos astrônomos. Qual é o total correto --- e que
        ano, mais uma vez, causou o erro?
\end{enumerate}
\end{problem}
