\chapter{Simetria central e paralelogramos}\label{ch:g7:central}

Gire uma figura de meia volta em torno de um ponto: isso é a
\emph{\omterm{def:g7:central:def}{simetria central}}, a segunda transformação do livro depois das
reflexões do \cref{ch:g6:symmetry}. A figura-estrela dela é o
\omterm{def:g7:central:parallelogram}{paralelogramo} --- o \omterm{def:g4:geometry:polygon}{quadrilátero} que é \omterm{def:g7:central:def}{simétrico} em relação ao ponto de
cruzamento das próprias diagonais.

\section{Simetria central}

\begin{definition}[Simétrico em relação a um ponto]\label{def:g7:central:def}
O \emph{simétrico}\index{simetria central} de um ponto $M$ em relação a
um ponto $O$ é o ponto $M'$ tal que $O$ é o \emph{ponto médio} de
$[MM']$. A figura simétrica é aquilo em que a original se transforma
depois de uma meia volta ($180^\circ$) em torno de $O$.
\end{definition}

\begin{method}[Construir o simétrico de um ponto]\label{met:g7:central:construct}
Para construir o \omterm{def:g7:central:def}{simétrico} $M'$ de $M$ em relação a $O$:
\begin{enumerate}
  \item trace a reta $(MO)$ e prolongue-a além de $O$;
  \item meça $MO$ (com régua ou compasso);
  \item marque $M'$ sobre o prolongamento, com $OM' = OM$.
\end{enumerate}
No papel quadriculado: conte os passos horizontais e verticais de $M$
até $O$ e repita os \emph{mesmos} passos a partir de $O$ para chegar a
$M'$.
\end{method}

\begin{example}[Na malha quadriculada]\label{ex:g7:central:grid}
Se $M$ está $3$ quadradinhos à esquerda e $1$ acima de $O$, o \omterm{def:g7:central:def}{simétrico}
$M'$ fica $3$ quadradinhos à \emph{direita} e $1$ \emph{abaixo} de $O$:
a \omterm{def:g7:central:def}{simetria central} inverte as duas direções de uma vez (uma reflexão
inverte só uma).
\end{example}

\begin{proposition}[O que uma meia volta preserva]\label{prop:g7:central:preserved}
A \omterm{def:g7:central:def}{simetria central} preserva comprimentos, ângulos, \omterm{def:g6:measure:perimeter}{perímetros} e \omterm{def:g6:measure:area}{áreas};
a imagem de uma reta é uma reta \emph{paralela}, a imagem de um
\omterm{def:g6:lines:objects}{segmento} é um \omterm{def:g6:lines:objects}{segmento} paralelo de mesmo comprimento, e a imagem de um
círculo é um círculo de mesmo \omterm{def:g3:shapes:shapes}{raio} (com centro no \omterm{def:g7:central:def}{simétrico} do
centro).
\end{proposition}
\admitted

\begin{remark}
Compare com as reflexões (\cref{prop:g6:symmetry:preserved}): as duas
preservam comprimentos e ângulos; mas a reflexão vira as figuras do
avesso (um ``b'' vira um ``d''), enquanto a meia volta as mantém
legíveis --- de cabeça para baixo (um ``b'' vira um ``q''). E, ao
contrário de uma reflexão, uma meia volta leva toda reta a uma reta
paralela a ela.
\end{remark}

\begin{definition}[Centro de simetria]\label{def:g7:central:center}
Um ponto $O$ é um \emph{centro de simetria}\index{centro de simetria}
de uma figura quando a meia volta em torno de $O$ leva a figura
exatamente sobre ela mesma. Por exemplo, as letras S, N, Z têm centro
de simetria; um círculo tem um (o centro dele); um triângulo nunca
tem.
\end{definition}

\section{Paralelogramos}

\begin{definition}[Paralelogramo]\label{def:g7:central:parallelogram}
Um \emph{paralelogramo}\index{paralelogramo} é um \omterm{def:g4:geometry:polygon}{quadrilátero} cujas
diagonais se cruzam no ponto médio comum delas. Esse ponto de
cruzamento é então um \omterm{def:g7:central:center}{centro de simetria} da figura: cada \omterm{def:g5:solids:def}{vértice} é a
imagem, pela meia volta, do \omterm{def:g5:solids:def}{vértice} oposto.
\end{definition}

\begin{theorem}[Propriedades de um paralelogramo]\label{thm:g7:central:properties}
Num \omterm{def:g7:central:parallelogram}{paralelogramo} $ABCD$ de centro $O$:
\begin{enumerate}
  \item os lados opostos são paralelos: $(AB) \parallel (CD)$ e
        $(BC) \parallel (AD)$;
  \item os lados opostos têm o mesmo comprimento: $AB = CD$ e
        $BC = AD$;
  \item os ângulos opostos são iguais: $\widehat A = \widehat C$ e
        $\widehat B = \widehat D$.
\end{enumerate}
\end{theorem}

\begin{proof}
A meia volta em torno de $O$ leva $A$ a $C$ e $B$ a $D$ (pela
definição: $O$ é o ponto médio das duas diagonais). Logo ela leva o
\omterm{def:g6:lines:objects}{segmento} $[AB]$ ao \omterm{def:g6:lines:objects}{segmento} $[CD]$; pela
\cref{prop:g7:central:preserved}, a imagem é paralela a $[AB]$ e de
mesmo comprimento: pontos 1 e 2. Ela também leva o ângulo em $A$ ao
ângulo em $C$, e os ângulos são preservados: ponto 3.
\end{proof}

\begin{theorem}[Reconhecer um paralelogramo]\label{thm:g7:central:recognize}
Um \omterm{def:g4:geometry:polygon}{quadrilátero} (sem lados que se cruzem) é um \omterm{def:g7:central:parallelogram}{paralelogramo} assim que
\emph{uma} das condições abaixo valer:
\begin{enumerate}
  \item as diagonais dele têm o mesmo ponto médio (a definição);
  \item os lados opostos são paralelos dois a dois;
  \item dois lados opostos são paralelos \emph{e} de mesmo
        comprimento.
\end{enumerate}
\end{theorem}
\admitted

\begin{example}\label{ex:g7:central:recognize}
$ABCD$ tem $(AB) \parallel (CD)$ e $AB = CD = 5$ cm. Vale o critério
3: $ABCD$ é um \omterm{def:g7:central:parallelogram}{paralelogramo} --- então, sem medir mais nada, também
sabemos que $AD = BC$, que $(AD) \parallel (BC)$ e que as diagonais
dele se cortam ao meio.
\end{example}

\begin{remark}[Paralelogramos especiais]
O retângulo, o \omterm{def:g6:shapes:quadrilaterals}{losango} e o quadrado do \cref{ch:g6:shapes} são
exatamente os \omterm{def:g7:central:parallelogram}{paralelogramos} com uma propriedade a mais: diagonais
iguais (retângulo), diagonais \omterm{def:g6:lines:perp}{perpendiculares} (\omterm{def:g6:shapes:quadrilaterals}{losango}), as duas coisas
(quadrado). O argumento de ângulos prometido no
\cref{ex:g6:shapes:recognize} agora funciona: num \omterm{def:g7:central:parallelogram}{paralelogramo}, os
ângulos consecutivos são suplementares (\omterm{def:g7:triangles:pairs}{ângulos alternos internos} e
correspondentes com as \omterm{def:g6:lines:perp}{paralelas},
\cref{thm:g7:triangles:equalities}), então um \omterm{def:g3:shapes:rightangle}{ângulo reto} obriga os
quatro.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g7:central:1}
No papel quadriculado, marque $O$ e depois $M$ quatro quadradinhos à
direita de $O$ e um acima. Construa o \omterm{def:g7:central:def}{simétrico} $M'$ de $M$ em relação
a $O$ e o \omterm{def:g7:central:def}{simétrico} de $M$ em relação à \emph{linha vertical da malha}
que passa por $O$. Compare as duas imagens.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:2}
Trace um triângulo $ABC$ e um ponto $O$ fora dele. Construa a imagem
$A'B'C'$ dele pela meia volta em torno de $O$. O que você pode dizer
sobre os comprimentos $A'B'$ e $AB$? E sobre as retas $(A'B')$ e
$(AB)$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:3}
Que algarismos, escritos como num visor de calculadora ($0$ a $9$), têm
\omterm{def:g7:central:center}{centro de simetria}? E que letras maiúsculas entre H, A, N, S, T, Z, O?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:4}
$KLMN$ é um \omterm{def:g7:central:parallelogram}{paralelogramo} de centro $O$ com $KL = 7$ cm, $LM = 4$ cm e
$\widehat K = 115^\circ$. Dê, com justificativa: $MN$, $NK$,
$\widehat M$ e o ponto médio de $[LN]$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:5}
Construa um \omterm{def:g7:central:parallelogram}{paralelogramo} $ABCD$ a partir das diagonais: $AC = 8$ cm e
$BD = 5$ cm, cruzando-se nos pontos médios com um ângulo de $60^\circ$
entre elas. (Trace primeiro as diagonais!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:6}
Num \omterm{def:g7:central:parallelogram}{paralelogramo}, um ângulo mede $115^\circ$. Usando a observação
sobre ângulos consecutivos suplementares, dê os outros três ângulos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:7}
Marque $A(1, 1)$, $B(4, 2)$ e o ponto $O(2.5, 2.5)$. Calcule as
coordenadas das imagens $A'$ e $B'$ pela meia volta em torno de $O$
(use a contagem na malha do \cref{ex:g7:central:grid}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:8}
$ABCD$ é um \omterm{def:g4:geometry:polygon}{quadrilátero} em que $AB = CD$. Isso basta para que ele seja
um \omterm{def:g7:central:parallelogram}{paralelogramo}? Se não, esboce um contraexemplo (um trapézio
\omterm{def:g6:shapes:triangles}{isósceles} ajuda) e diga o que precisa ser acrescentado à hipótese.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:9}
Seja $ABC$ um triângulo e $O$ o ponto médio de $[BC]$. Construa o
\omterm{def:g7:central:def}{simétrico} $A'$ de $A$ em relação a $O$. Mostre que $ABA'C$ é um
\omterm{def:g7:central:parallelogram}{paralelogramo}. (Que critério do \cref{thm:g7:central:recognize} sai de
graça aqui?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:10}
Verdadeiro ou falso, com justificativa: ``um \omterm{def:g4:geometry:polygon}{quadrilátero} cujos ângulos
opostos são iguais dois a dois é um \omterm{def:g7:central:parallelogram}{paralelogramo}''; ``um \omterm{def:g7:central:parallelogram}{paralelogramo}
com diagonais \omterm{def:g6:lines:perp}{perpendiculares} é um \omterm{def:g6:shapes:quadrilaterals}{losango}''; ``um \omterm{def:g7:central:parallelogram}{paralelogramo} com um
\omterm{def:g3:shapes:rightangle}{ângulo reto} é um retângulo''.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:central:11}
Seja $ABCD$ um \omterm{def:g4:geometry:polygon}{quadrilátero} qualquer e $I$, $J$, $K$, $L$ os pontos
médios dos lados $[AB]$, $[BC]$, $[CD]$, $[DA]$ dele. Faça vários
exemplos. O que $IJKL$ sempre parece ser? (A demonstração, com o
teorema dos pontos médios, vem no \cref{ch:g8:midpoints} --- e de novo
com vetores, no volume do ensino médio.)
\end{exercise}

\section{Problema: o bolo, a mediana e o segundo centro impossível}

\begin{problem}[{Problema de fim de semana --- meias voltas em ação:
cortes justos de bolo, um limite para as medianas e por que nenhuma
forma pode ter dois centros de simetria}]
\label{pb:g7:central:1}
A meia volta parece a mais simples das transformações --- e, no
entanto, ela corta bolos em \omterm{def:g3:division:half}{metades} comprovadamente justas, mede as
medianas de um triângulo e esconde uma pequena pérola de matemática
pura: uma figura limitada pode ter um \omterm{def:g7:central:center}{centro de simetria}, mas
\emph{nunca dois}. A demonstração usa uma descoberta que você fará na
Parte I: duas meias voltas, feitas uma depois da outra, dão um
deslizamento.

\medskip
\noindent\textbf{Parte I --- Ginástica de meias voltas.}
\begin{enumerate}
  \item No papel quadriculado, tome a origem $O(0, 0)$ como centro.
        Usando a contagem na malha do \cref{ex:g7:central:grid}, dê as
        imagens de $M(3, 1)$, $N(-2, 4)$ e $P(0, -5)$ pela meia volta
        em torno de $O$. Qual é a regra geral para $(x, y)$?
  \item Agora o centro é $C(2, 1)$. Calcule a imagem de $M(5, 3)$ e
        confira a receita geral: cada coordenada da imagem é \emph{o
        dobro da do centro menos a do ponto}.
  \item As cartas de baralho são desenhadas para se ler igual de cabeça
        para baixo --- uma \omterm{def:g7:central:def}{simetria central}, para que nenhum jogador
        veja a carta ``errada''. Explique por que, numa carta com um
        número \emph{\omterm{def:g3:numbers:evenodd}{ímpar}} de naipes, um deles tem de ficar exatamente
        no centro da carta.
  \item Duas meias voltas seguidas: tome os centros $O_1(2, 0)$ e
        $O_2(5, 0)$. Leve o ponto $A(1, 1)$ pela meia volta em torno de
        $O_1$ e depois leve a imagem pela meia volta em torno de $O_2$.
        Compare partida e chegada. Repita com $B(0, 3)$. Em que
        transformação única e simples as duas meias voltas
        resultaram?
  \item Meça o deslizamento da questão 4 em relação à distância
        $O_1 O_2$ e enuncie a descoberta. (Compare com os dois espelhos
        paralelos do \cref{pb:g6:symmetry:1}: o mesmo fenômeno, com
        outros atores.)
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Bolos e medianas.}
\begin{enumerate}[resume]
  \item Seja $O$ o centro de um \omterm{def:g7:central:parallelogram}{paralelogramo}. Explique por que toda
        reta que passa por $O$ encontra o contorno em dois pontos
        \omterm{def:g7:central:def}{simétricos} em relação a $O$, e por que a reta corta o
        \omterm{def:g7:central:parallelogram}{paralelogramo} em duas peças que são cópias exatas uma da outra
        pela meia volta --- e, portanto, de \omterm{def:g6:measure:area}{áreas} iguais.
  \item O teorema do confeiteiro: um bolo retangular é repartido com
        perfeita justiça por \emph{qualquer} corte reto que passe pelo
        centro dele. Melhor ainda: um bolo retangular tem um buraco
        retangular (de qualquer tamanho, em qualquer posição, até
        inclinado). Descreva o único corte reto que dá duas partes com
        exatamente a mesma quantidade de bolo --- e justifique-o com a
        questão 6.
  \item Três \omterm{def:g5:solids:def}{vértices} de um \omterm{def:g7:central:parallelogram}{paralelogramo} $ABCD$ são $A(0, 0)$,
        $B(6, 1)$ e $C(8, 5)$. Calcule as coordenadas de $D$ e do
        centro $O$.
  \item O \cref{exo:g7:central:9} construiu, a partir de um triângulo
        $ABC$ e do ponto médio $O$ de $[BC]$, o \omterm{def:g7:central:parallelogram}{paralelogramo} $ABA'C$
        em que $A'$ é o \omterm{def:g7:central:def}{simétrico} de $A$ em relação a $O$. Use-o, junto
        com a desigualdade triangular
        (\cref{thm:g7:triangles:inequality}) no triângulo $ABA'$, para
        demonstrar o \emph{limite da mediana}: a mediana $[AO]$
        satisfaz
        \[
        AO < \frac{AB + AC}{2} .
        \]
        (Repare que $AA' = 2\,AO$ e $BA' = AC$.)
  \item Num triângulo com $AB = 5$ cm e $AC = 7$ cm, entre que dois
        valores tem de ficar o comprimento da mediana que sai de $A$?
        (Use os dois sentidos da desigualdade triangular em $ABA'$.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Quantos centros uma figura pode ter?}
\begin{enumerate}[resume]
  \item Para cada figura, diga se ela tem \omterm{def:g7:central:center}{centro de simetria} e onde: um
        \omterm{def:g6:lines:objects}{segmento}; uma reta inteira; um \omterm{def:g6:shapes:triangles}{triângulo equilátero}; um
        quadrado; um círculo; as letras S e Z.
  \item Demonstre a sua afirmação para o \omterm{def:g6:shapes:triangles}{triângulo equilátero}: uma meia
        volta que levasse o triângulo a si mesmo emparelharia os três
        \omterm{def:g5:solids:def}{vértices} --- mas três é \omterm{def:g3:numbers:evenodd}{ímpar}, então algum \omterm{def:g5:solids:def}{vértice} seria levado
        a si mesmo. O que isso obrigaria esse \omterm{def:g5:solids:def}{vértice} a ser, e por que
        isso é impossível?
  \item Generalize a questão 12: explique por que \emph{nenhum}
        \omterm{def:g4:geometry:polygon}{polígono} com um \omterm{def:g3:numbers:evenodd}{número ímpar} de \omterm{def:g5:solids:def}{vértices} tem \omterm{def:g7:central:center}{centro de
        simetria}.
  \item A pérola: suponha que uma figura tivesse \emph{dois} \omterm{def:g7:central:center}{centros de
        simetria} distintos $O_1$ e $O_2$. Faça as duas meias voltas uma
        depois da outra e use a descoberta da questão 5: que movimento
        teria de levar a figura exatamente sobre ela mesma? Explique
        por que uma figura que cabe numa folha de papel não sobrevive a
        isso e conclua.
  \item A conclusão da questão 14 deixa uma brecha: as figuras
        \emph{ilimitadas}. Verifique, num friso infinito --- digamos,
        uma faixa sem fim de pegadas, esquerda, direita, esquerda,
        direita\,\dots\ ---, que ele tem (pelo menos) dois \omterm{def:g7:central:center}{centros de
        simetria} diferentes, e identifique a translação que a questão
        14 previu.
\end{enumerate}
\end{problem}
