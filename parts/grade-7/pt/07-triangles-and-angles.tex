\chapter{Triângulos e ângulos}\label{ch:g7:triangles}

Dá para traçar um triângulo de lados $10$, $3$ e $2$? E com ângulos
$80^\circ$, $70^\circ$ e $50^\circ$? Este capítulo responde às duas
perguntas com dois dos fatos mais úteis da geometria plana: a
desigualdade triangular e a \omterm{def:g1:addition:def}{soma} dos ângulos igual a $180^\circ$ ---
mais o vocabulário de ângulos que vem com retas que se cruzam e \omterm{def:g6:lines:perp}{retas
paralelas}.

\section{Ângulos em torno de retas que se cruzam}

\begin{definition}[Pares de ângulos]\label{def:g7:triangles:pairs}
\begin{itemize}
  \item Dois ângulos são \emph{complementares} quando as medidas deles
        somam $90^\circ$, e \emph{suplementares} quando somam
        $180^\circ$.
  \item Quando duas retas se cruzam, os ângulos que ficam um de frente
        para o outro através do ponto de cruzamento são \emph{opostos
        pelo vértice}\index{ângulos opostos pelo vértice}.
  \item Quando uma reta cruza duas outras retas, os ângulos situados na
        mesma posição em cada cruzamento são \emph{ângulos
        correspondentes}; os ângulos entre as duas retas e em lados
        opostos da reta que as cruza são \emph{ângulos alternos
        internos}\index{ângulos alternos internos}.
\end{itemize}
\end{definition}

\begin{theorem}[Igualdades de ângulos]\label{thm:g7:triangles:equalities}
\begin{enumerate}
  \item \omterm{def:g7:triangles:pairs}{Ângulos opostos pelo vértice} são iguais.
  \item Se duas retas são \omterm{def:g6:lines:perp}{paralelas}, os ângulos correspondentes são
        iguais e os \omterm{def:g7:triangles:pairs}{ângulos alternos internos} são iguais. Reciprocamente,
        ângulos correspondentes (ou alternos internos) iguais obrigam as
        retas a serem \omterm{def:g6:lines:perp}{paralelas}.
\end{enumerate}
\end{theorem}

\begin{proof}[Demonstração de 1]
Os dois ângulos $\widehat{1}$ e $\widehat{2}$ de um lado de uma reta
formam um ângulo raso: $\widehat 1 + \widehat 2 = 180^\circ$. O ângulo
$\widehat 3$, oposto pelo \omterm{def:g5:solids:def}{vértice} a $\widehat 1$, também satisfaz
$\widehat 2 + \widehat 3 = 180^\circ$ (pela mesma razão). Logo
$\widehat 1 = 180^\circ - \widehat 2 = \widehat 3$. O ponto 2 é
admitido neste nível.
\end{proof}

\section{A soma dos ângulos de um triângulo}

\begin{theorem}[Soma dos ângulos]\label{thm:g7:triangles:sum}
Em todo triângulo, os três ângulos somam $180^\circ$.
\end{theorem}

\begin{proof}
Seja $ABC$ um triângulo. Trace a reta que passa por $A$ e é paralela a
$(BC)$. No \omterm{def:g5:solids:def}{vértice} $A$, três ângulos se alinham ao longo dessa paralela
e formam um ângulo raso, $180^\circ$: o do meio é $\widehat{A}$, e os
dois de fora são alternos internos com $\widehat B$ e $\widehat C$
(\omterm{def:g6:lines:perp}{retas paralelas}!), então valem $\widehat B$ e $\widehat C$. Portanto,
$\widehat B + \widehat A + \widehat C = 180^\circ$.
\end{proof}

\begin{example}\label{ex:g7:triangles:sum}
Dois ângulos de um triângulo medem $67^\circ$ e $58^\circ$. O terceiro
mede
\[
180 - (67 + 58) = 180 - 125 = 55^\circ .
\]
Casos especiais que vale a pena decorar: cada ângulo de um \omterm{def:g6:shapes:triangles}{triângulo
equilátero} mede $180 \div 3 = 60^\circ$; num \omterm{def:g6:shapes:triangles}{triângulo retângulo}, os
dois ângulos agudos são complementares (somam $180 - 90 = 90^\circ$);
os ângulos da base de um \omterm{def:g6:shapes:triangles}{triângulo isósceles} são iguais, então cada um
vale $\frac{180 - \text{ângulo do ápice}}{2}$.
\end{example}

\begin{example}[Não existe triângulo com dois ângulos retos]\label{ex:g7:triangles:tworight}
Um triângulo com dois ângulos de $90^\circ$ já gastaria $180^\circ$,
deixando $0^\circ$ para o terceiro --- impossível. A \omterm{def:g1:addition:def}{soma} dos ângulos
proíbe muitos triângulos de uma vez.
\end{example}

\section{A desigualdade triangular}

\begin{theorem}[Desigualdade triangular]\label{thm:g7:triangles:inequality}
Em todo triângulo, cada lado é menor que a \omterm{def:g1:addition:def}{soma} dos outros dois: para
um triângulo de lados $a$, $b$, $c$,
\[
a < b + c .
\]
Reciprocamente, três comprimentos em que o \emph{maior} é menor que a
\omterm{def:g1:addition:def}{soma} dos outros dois sempre formam um triângulo. Se o maior for
\emph{igual} à \omterm{def:g1:addition:def}{soma} dos outros, o ``triângulo'' fica achatado: os três
\omterm{def:g5:solids:def}{vértices} ficam alinhados.
\end{theorem}
\admitted

\begin{example}\label{ex:g7:triangles:inequality}
Os lados podem ser $10$, $3$ e $2$? Teste o maior: será que
$10 < 3 + 2 = 5$? Não --- não existe triângulo assim. O compasso mostra
por quê: arcos de \omterm{def:g3:shapes:shapes}{raios} $3$ e $2$ traçados a partir das pontas de um
\omterm{def:g6:lines:objects}{segmento} de comprimento $10$ nunca se encontram.

Lados $7$, $5$ e $4$? Maior: $7 < 5 + 4 = 9$: sim, o triângulo existe
(basta conferir o maior lado).
\end{example}

\begin{method}[Existência e construção de um triângulo]\label{met:g7:triangles:construct}
Dados três comprimentos:
\begin{enumerate}
  \item compare o maior com a \omterm{def:g1:addition:def}{soma} dos outros dois; se ele não for
        estritamente menor, pare: não há triângulo;
  \item caso contrário, construa como no
        \cref{met:g6:shapes:construct} (\omterm{def:g6:lines:objects}{segmento} de base, dois arcos de
        compasso);
  \item depois de qualquer construção a partir de ângulos, confira de
        cabeça que os ângulos dados somam menos de $180^\circ$ --- com
        o terceiro ângulo completando exatamente $180^\circ$.
\end{enumerate}
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g7:triangles:1}
Dê o complemento (até $90^\circ$) e o suplemento (até $180^\circ$) de:
$30^\circ$; \ $45^\circ$; \ $72^\circ$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:2}
Duas retas se cruzam; um dos quatro ângulos mede $35^\circ$. Dê as
medidas dos outros três, com uma razão para cada uma.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:3}
Calcule o terceiro ângulo de um triângulo cujos dois primeiros ângulos
medem: (a) $40^\circ$ e $60^\circ$; \
(b) $90^\circ$ e $28^\circ$; \
(c) $75^\circ$ e $75^\circ$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:4}
Um \omterm{def:g6:shapes:triangles}{triângulo isósceles} tem o ângulo do ápice igual a $40^\circ$.
Calcule os dois ângulos da base dele. Outro \omterm{def:g6:shapes:triangles}{triângulo isósceles} tem um
ângulo da base de $70^\circ$: calcule o ângulo do ápice.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:5}
Que ternos podem ser os lados de um triângulo? Justifique pela
desigualdade triangular (olhe o maior lado!):
\[
(6,\ 8,\ 12); \qquad
(5,\ 5,\ 11); \qquad
(4,\ 9,\ 13); \qquad
(7,\ 7,\ 7).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:6}
Um triângulo tem ângulos $x$, $2x$ e $3x$. Encontre $x$ e os três
ângulos. Que tipo de triângulo é esse?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:7}
Construa um triângulo $ABC$ com $BC = 6$ cm,
$\widehat{ABC} = 50^\circ$ e $\widehat{ACB} = 60^\circ$ (transferidor
em $B$ e em $C$). Antes de construir, preveja a medida de
$\widehat{BAC}$ e depois confira na sua figura.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:8}
Duas \omterm{def:g6:lines:perp}{retas paralelas} são cortadas por uma terceira reta, formando um
ângulo de $54^\circ$ no primeiro cruzamento (entre a reta que corta e
uma das \omterm{def:g6:lines:perp}{paralelas}). Desenhe a situação e dê as medidas de todos os oito
ângulos formados.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:9}
Dois lados de um triângulo medem $8$ cm e $3$ cm. Entre que dois
valores tem de ficar o terceiro lado? Dê um comprimento inteiro que
sirva e um que não sirva.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:10}
Num triângulo $ABC$, $\widehat A = 2\widehat B$ e
$\widehat C = 90^\circ$. Calcule $\widehat A$ e $\widehat B$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:triangles:11}
Os ângulos de qualquer \omterm{def:g4:geometry:polygon}{quadrilátero} $ABCD$ podem ser estudados
cortando-o por uma diagonal em dois triângulos. Use isso para
demonstrar que os quatro ângulos de todo \omterm{def:g4:geometry:polygon}{quadrilátero} somam
$360^\circ$ e deduza a \omterm{def:g1:addition:def}{soma} dos ângulos de um \omterm{def:g4:geometry:polygon}{pentágono} (cinco lados).
\end{exercise}

\section{Problema: dando a volta no quarteirão}

\begin{problem}[{Problema de fim de semana --- os ângulos de um
polígono qualquer: $(n-2) \times 180^\circ$ por dentro, sempre
$360^\circ$ de giro por fora, e por que só três formas regulares
ladrilham um piso}]
\label{pb:g7:triangles:1}
O \cref{exo:g7:triangles:11} cortou um \omterm{def:g4:geometry:polygon}{quadrilátero} em dois triângulos
e encontrou $360^\circ$. Este problema leva a ideia a \omterm{def:g4:geometry:polygon}{polígonos} com
qualquer número de lados, depois descobre uma segunda demonstração,
completamente diferente --- \emph{caminhando} em volta da forma e
somando os giros --- e termina no piso do seu banheiro: entre todos os
\omterm{def:g4:geometry:polygon}{polígonos} regulares, exatamente três conseguem ladrilhar um plano sem
falhas nem sobreposições, e as abelhas escolheram o delas.

\medskip
\noindent\textbf{Parte I --- Os ângulos de dentro.}
\begin{enumerate}
  \item A partir de um \omterm{def:g5:solids:def}{vértice} de um \omterm{def:g4:geometry:polygon}{hexágono} (seis lados), trace todas
        as diagonais que saem desse \omterm{def:g5:solids:def}{vértice}. Em quantos triângulos o
        \omterm{def:g4:geometry:polygon}{hexágono} fica cortado, e qual é a \omterm{def:g1:addition:def}{soma} de \emph{todos} os
        ângulos internos dele (\cref{thm:g7:triangles:sum})?
  \item Generalize: a partir de um \omterm{def:g5:solids:def}{vértice} de um \omterm{def:g4:geometry:polygon}{polígono} de $n$ lados,
        quantos triângulos o leque de diagonais produz? Deduza a
        fórmula da \omterm{def:g1:addition:def}{soma} dos ângulos internos e confira-a para $n = 3$,
        $4$, $5$, $6$.
  \item Calcule a \omterm{def:g1:addition:def}{soma} dos ângulos internos de um decágono ($10$ lados)
        e de um dodecágono ($12$ lados).
  \item Num \omterm{def:g4:geometry:polygon}{polígono} \emph{regular}, todos os ângulos internos são
        iguais. Calcule o ângulo interno do \omterm{def:g6:shapes:triangles}{triângulo equilátero}, do
        quadrado, do \omterm{def:g4:geometry:polygon}{pentágono}, do \omterm{def:g4:geometry:polygon}{hexágono} e do \omterm{def:g4:geometry:polygon}{octógono} regulares.
  \item Explique, sem nenhuma fórmula, por que a \omterm{def:g1:addition:def}{soma} dos ângulos
        cresce exatamente $180^\circ$ a cada vez que o \omterm{def:g4:geometry:polygon}{polígono} ganha
        um \omterm{def:g5:solids:def}{vértice}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A volta no quarteirão.}
Caminhe ao longo do contorno de um quarteirão poligonal, sempre para a
frente. Em cada esquina você \emph{gira} um certo ângulo --- o
\emph{ângulo externo} dessa esquina; os giros continuam até você voltar
ao ponto de partida, olhando na direção de partida.
\begin{enumerate}[resume]
  \item Numa esquina cujo ângulo interno é $\widehat A$, de quanto você
        gira? (Ângulo interno e ângulo externo são suplementares.)
        Calcule os três giros para um triângulo de ângulos $40^\circ$,
        $60^\circ$, $80^\circ$.
  \item Some os três giros da questão 6. Que fato de volta completa
        você observa?
  \item Explique por que os giros de uma volta completa em torno de
        \emph{qualquer} \omterm{def:g4:geometry:polygon}{polígono} --- de três lados ou de trinta ---
        somam sempre exatamente $360^\circ$. (O que o seu nariz fez,
        somados todos os giros, quando você volta?)
  \item Deduza a fórmula dos ângulos internos uma segunda vez: em cada
        uma das $n$ esquinas, interno $+$ giro $= 180^\circ$; some
        sobre todas as esquinas e use a questão 8.
  \item Num \omterm{def:g4:geometry:polygon}{polígono} regular, todos os giros são iguais, então cada
        ângulo externo é $\frac{360^\circ}{n}$. Use isso para responder
        na hora: que \omterm{def:g4:geometry:polygon}{polígono} regular tem ângulos internos de
        $150^\circ$? E por que \emph{nenhum} \omterm{def:g4:geometry:polygon}{polígono} regular tem
        ângulos internos de $155^\circ$?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Ladrilhar o piso.}
Em torno de cada ponto de um piso ladrilhado, os cantos dos ladrilhos
que se encontram têm de somar exatamente $360^\circ$ --- sem falha e
sem sobreposição (\cref{pb:g6:lines:1}).
\begin{enumerate}[resume]
  \item Para cada um destes --- \omterm{def:g6:shapes:triangles}{triângulo equilátero}, quadrado e
        \omterm{def:g4:geometry:polygon}{hexágono} regular ---, quantas cópias se encontram num ponto de
        canto do ladrilhamento? Verifique os $360^\circ$ em cada caso.
  \item Mostre que \omterm{def:g4:geometry:polygon}{pentágonos} regulares não conseguem ladrilhar o piso:
        quanto somam três cantos, e quanto somariam quatro?
  \item Mostre que nenhum \omterm{def:g4:geometry:polygon}{polígono} regular com \emph{mais} de seis
        lados consegue ladrilhar: pelo menos três ladrilhos têm de se
        encontrar em cada canto (cada ângulo é menor que $180^\circ$),
        e o que acontece com três cantos de $135^\circ$ ou mais?
        Conclua a lista completa dos ladrilhadores regulares.
  \item Pisos de banheiro muitas vezes misturam \omterm{def:g4:geometry:polygon}{octógonos} regulares com
        quadradinhos. Verifique que este canto funciona:
        $135^\circ + 135^\circ + 90^\circ$. Outro clássico mistura um
        quadrado, um \omterm{def:g4:geometry:polygon}{hexágono} regular e um dodecágono regular em cada
        canto: verifique-o também (o ângulo interno do dodecágono sai
        do método da questão 10).
  \item A escolha das abelhas: as células do favo de mel são \omterm{def:g4:geometry:polygon}{hexágonos}
        regulares. Verifique os cantos delas ($3$ ângulos) e explique
        em uma frase --- com a ajuda da descoberta de Dido
        (\cref{pb:g6:measure:1}) --- por que, dos três ladrilhos
        regulares possíveis, o \omterm{def:g4:geometry:polygon}{hexágono} é o mais barato em cera para o
        mel que guarda.
\end{enumerate}
\end{problem}
