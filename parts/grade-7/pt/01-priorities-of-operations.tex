\chapter{Prioridades das operações}\label{ch:g7:priorities}

Quanto é $3 + 4 \times 5$? Se você calcular da esquerda para a direita,
dá $35$; se multiplicar primeiro, dá $23$. A matemática precisa de uma
resposta única, então existem regras de trânsito para os cálculos ---
as prioridades. Este capítulo as enuncia e apresenta uma primeira
identidade algébrica, a distributividade.

\section{As regras de prioridade}

\begin{theorem}[Regras de prioridade]\label{thm:g7:priorities:rules}
Num cálculo sem parênteses:
\begin{enumerate}
  \item as multiplicações e as divisões são feitas \emph{antes} das
        adições e das subtrações;
  \item as operações do mesmo nível são feitas da esquerda para a
        direita.
\end{enumerate}
Os parênteses passam na frente de tudo: o que está dentro deles é
calculado primeiro, começando pelos mais internos.
\end{theorem}

\begin{example}\label{ex:g7:priorities:basic}
Uma etapa por linha, sublinhando (mentalmente) o que está sendo
calculado:
\begin{align*}
3 + 4 \times 5 &= 3 + 20 = 23
&&\text{(a multiplicação primeiro)}\\
20 - 8 + 2 &= 12 + 2 = 14
&&\text{(mesmo nível: da esquerda para a direita)}\\
18 \div 3 \times 2 &= 6 \times 2 = 12
&&\text{(mesmo nível: da esquerda para a direita)}\\
(3 + 4) \times 5 &= 7 \times 5 = 35
&&\text{(os parênteses primeiro).}
\end{align*}
Repare nas armadilhas: $20 - 8 + 2$ \emph{não} é $20 - 10$, e
$18 \div 3 \times 2$ \emph{não} é $18 \div 6$.
\end{example}

\begin{example}[Parênteses encaixados]\label{ex:g7:priorities:nested}
\begin{align*}
50 - 2 \times \bigl(3 + (10 - 4)\bigr)
&= 50 - 2 \times (3 + 6) && \text{(parêntese mais interno)}\\
&= 50 - 2 \times 9 && \text{(parêntese restante)}\\
&= 50 - 18 && \text{(multiplicação)}\\
&= 32 .
\end{align*}
\end{example}

\begin{remark}[A barra de fração esconde parênteses]
A barra de \omterm{def:g6:fractions:def}{fração} agrupa o \omterm{def:g4:fractions:def}{numerador} e o \omterm{def:g4:fractions:def}{denominador} dela, como se cada
um tivesse parênteses invisíveis:
\[
\frac{12 + 8}{4} = \frac{20}{4} = 5,
\qquad\text{e não}\quad 12 + \frac{8}{4} = 14 .
\]
\end{remark}

\begin{method}[Calcular com segurança]\label{met:g7:priorities:safe}
\begin{enumerate}
  \item Identifique a operação a fazer primeiro (parênteses mais
        internos, depois $\times$ e $\div$, depois $+$ e $-$);
  \item calcule \emph{só ela} e reescreva a expressão inteira com o
        resultado no lugar;
  \item repita até sobrar um único número --- uma operação por linha,
        sem atalhos.
\end{enumerate}
\end{method}

\section{Distributividade}

\begin{theorem}[Distributividade]\label{thm:g7:priorities:distributivity}
Para quaisquer números $k$, $a$, $b$:
\[
k \times (a + b) = k \times a + k \times b,
\qquad
k \times (a - b) = k \times a - k \times b .
\]
\end{theorem}

\begin{proof}[Demonstração por um desenho]
Conte a \omterm{def:g6:measure:area}{área} de um retângulo de largura $k$ dividido em dois retângulos
menores de comprimentos $a$ e $b$: inteiro, ele é $k \times (a+b)$; em
duas partes, ele é $k \times a + k \times b$. O mesmo retângulo, a
mesma \omterm{def:g6:measure:area}{área}.
\end{proof}

\begin{tikzpicture}[scale=0.75]
  \fill[omDef!14] (0,0) rectangle (4.2,1.8);
  \fill[omThm!14] (4.2,0) rectangle (6.4,1.8);
  \draw[thick] (0,0) rectangle (6.4,1.8);
  \draw[thick] (4.2,0) -- (4.2,1.8);
  \node[below, font=\small] at (2.1,0) {$a$};
  \node[below, font=\small] at (5.3,0) {$b$};
  \node[left, font=\small] at (0,0.9) {$k$};
  \node[font=\small] at (2.1,0.9) {$k \times a$};
  \node[font=\small] at (5.3,0.9) {$k \times b$};
\end{tikzpicture}

{\small A distributividade em \omterm{def:g6:measure:area}{áreas}: o retângulo grande
$k \times (a + b)$ é a \omterm{def:g1:addition:def}{soma} dos dois menores.}
\end{omfigure}

\begin{example}[Nos dois sentidos]\label{ex:g7:priorities:distributivity}
\emph{Desenvolver} (da esquerda para a direita) facilita o cálculo
mental:
\[
7 \times 103 = 7 \times (100 + 3) = 700 + 21 = 721,
\qquad
6 \times 98 = 6 \times (100 - 2) = 600 - 12 = 588 .
\]
\emph{Fatorar} (da direita para a esquerda) recolhe um fator comum:
\[
23 \times 12 + 23 \times 8 = 23 \times (12 + 8) = 23 \times 20 = 460 .
\]
\end{example}

\begin{example}[Traduzir palavras em cálculo]\label{ex:g7:priorities:words}
``A \omterm{def:g1:addition:def}{soma} de $8$ com o \omterm{def:g2:mult:def}{produto} de $5$ por $3$'' é
$8 + 5 \times 3 = 23$. ``O \omterm{def:g2:mult:def}{produto} de $5$ pela \omterm{def:g1:addition:def}{soma} de $8$ com $3$'' é
$5 \times (8 + 3) = 55$: as palavras ``o \omterm{def:g2:mult:def}{produto} de \dots\ pela \omterm{def:g1:addition:def}{soma}
de\dots'' põem parênteses em torno da \omterm{def:g1:addition:def}{soma}.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g7:priorities:1}
Calcule, uma operação por linha:
\[
5 + 3 \times 6, \qquad
(5 + 3) \times 6, \qquad
30 - 12 \div 4, \qquad
(30 - 12) \div 4 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:2}
Calcule:
\[
24 - 6 + 2, \qquad
24 \div 6 \times 2, \qquad
24 - 6 \times 2, \qquad
24 \div (6 \times 2).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:3}
Calcule passo a passo:
\[
7 + 3 \times (10 - 6), \qquad
40 - (2 + 3) \times 4, \qquad
60 \div \bigl(2 \times (7 - 4)\bigr).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:4}
Calcule as frações (parênteses invisíveis!):
\[
\frac{18 + 6}{8}, \qquad
\frac{30}{7 - 2}, \qquad
\frac{5 \times 8}{4 + 6} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:5}
Use a distributividade para calcular de cabeça:
\[
8 \times 102, \qquad
25 \times 99, \qquad
14 \times 12 + 14 \times 8, \qquad
37 \times 45 - 37 \times 35 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:6}
Escreva o cálculo numa linha só, com parênteses onde for preciso, e
depois calcule: ``a \omterm{ex:g1:subtraction:difference}{diferença} entre $50$ e o \omterm{def:g2:mult:def}{produto} de $6$ por $7$'';
``o \omterm{def:g2:mult:def}{produto} da \omterm{def:g1:addition:def}{soma} de $4$ com $5$ pela \omterm{ex:g1:subtraction:difference}{diferença} entre $10$ e $8$''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:7}
Amir comprou $3$ cadernos a $2.50$ cada um e $2$ canetas a $1.20$ cada
uma. Escreva o total dele numa única expressão e depois calcule.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:priorities:8}
Coloque parênteses em $4 + 2 \times 5 - 1$ para que o resultado seja:
(a) $13$; (b) $29$; (c) $24$. (Um dos três não precisa de parêntese
nenhum --- qual?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:priorities:9}
$12 + 4 \times k$ e $16 \times k$ são a mesma coisa para todo número
$k$? Teste com $k = 1$ e $k = 2$, e conclua. Que erro de prioridade
levaria de um ao outro?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:priorities:10}
Um ingresso de cinema custa $9$ reais, mas um grupo de $n$ pessoas paga
$12$ reais fixos mais $6$ reais por pessoa.
\begin{enumerate}
  \item Escreva uma expressão para o preço do grupo e calcule-a para
        $n = 5$.
  \item A partir de quantas pessoas a tarifa de grupo fica mais barata
        que $n$ ingressos individuais? (Teste alguns valores de $n$.)
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:priorities:11}
Usando cada um dos algarismos $1$, $2$, $3$, $4$ exatamente uma vez, as
quatro operações e parênteses à vontade, escreva expressões iguais a
$24$, a $10$ e a $1$.
\end{exercise}

\section{Problema: multiplicar como um escriba egípcio}

\begin{problem}[{Problema de fim de semana --- truques de cálculo
mental vindos da distributividade, e a multiplicação por dobros e
metades usada há quatro mil anos}]
\label{pb:g7:priorities:1}
Muito antes das calculadoras --- e antes de as tabuadas serem
decoradas na escola ---, escribas egípcios e camponeses russos
multiplicavam dois números quaisquer usando apenas \emph{dobrar,
dividir ao meio e somar}. O método deles parece mágica; na verdade é a
distributividade (\cref{thm:g7:priorities:distributivity}) trabalhando
a todo vapor. Este problema afia o seu cálculo mental, depois ensina o
método antigo e, por fim, abre o capô para mostrar exatamente por que
ele funciona.

\medskip
\noindent\textbf{Parte I --- A arte de não calcular.}
\begin{enumerate}
  \item Calcule de cabeça, desenvolvendo como no
        \cref{ex:g7:priorities:distributivity}: $7 \times 102$ e
        $9 \times 98$.
  \item Calcule de cabeça, fatorando: $17 \times 13 + 17 \times 7$ e
        $45 \times 101 - 45$.
  \item O truque do ``vezes $5$'': para multiplicar por $5$,
        multiplique por $10$ e divida ao meio. Calcule $5 \times 87$
        assim e explique por que o truque é legítimo.
  \item Mais dois truques para justificar e usar: multiplicar por $25$
        multiplicando por $100$ e dividindo por $4$ (calcule
        $25 \times 32$); multiplicar por $99$ multiplicando por $100$ e
        subtraindo uma vez (calcule $99 \times 47$).
  \item Um tiro de alerta: a distributividade espalha um fator sobre
        uma \emph{\omterm{def:g1:addition:def}{soma}}, nunca sobre um \omterm{def:g2:mult:def}{produto}. Compare
        $8 \times (5 + 3)$ com $8 \times 5 + 8 \times 3$ e depois
        $8 \times (5 \times 3)$ com
        $(8 \times 5) \times (8 \times 3)$: qual par bate, e qual não
        bate?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O método de dobrar e dividir ao meio.}
Eis a receita para $13 \times 24$. Escreva duas colunas. Coluna da
esquerda: comece em $13$ e vá \emph{dividindo ao meio}, jogando fora
qualquer \omterm{def:g3:division:remainder}{resto}, até chegar a $1$: \ $13,\ 6,\ 3,\ 1$. Coluna da
direita: comece em $24$ e vá \emph{dobrando} a mesma quantidade de
vezes: \ $24,\ 48,\ 96,\ 192$. Risque toda linha cuja número da
\emph{esquerda} for par e some os números da direita que sobraram.
\begin{enumerate}[resume]
  \item Execute a receita para $13 \times 24$: que linha é riscada, que
        \omterm{def:g1:addition:def}{soma} resta, e ela é igual a $13 \times 24$?
  \item Multiplique $21 \times 17$ pelo método (divida $21$ ao meio,
        dobre $17$) e confira a resposta com a multiplicação comum.
  \item Multiplique $16 \times 25$ pelo método. Quantas linhas
        sobrevivem, e o que há de especial numa coluna da esquerda que
        começa em $1$, $2$, $4$, $8$, $16, \dots$?
  \item Agora, \emph{por que} funciona, um passo de cada vez.
        Justifique cada igualdade com a distributividade:
        \[
        13 \times 24 = (12 + 1) \times 24
        = 12 \times 24 + 24 = 6 \times 48 + 24 .
        \]
        Moral: dividir a esquerda ao meio e dobrar a direita mantém o
        \omterm{def:g2:mult:def}{produto} --- mas, quando o número da esquerda é \emph{\omterm{def:g3:numbers:evenodd}{ímpar}},
        uma cópia do número da direita se destaca e tem de ir para o
        cofre. Que número a primeira linha da tabela põe no cofre?
  \item Continue a escrituração tabela abaixo:
        $6 \times 48 = 3 \times 96$, depois
        $3 \times 96 = 1 \times 192 + 96$, depois
        $1 \times 192 = 192$. Junte tudo o que foi para o cofre pelo
        caminho e explique por que os números guardados são exatamente
        os números da coluna da direita das linhas de esquerda \omterm{def:g3:numbers:evenodd}{ímpar}
        --- as linhas que a receita mantém.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Sob o capô: \omterm{def:g1:addition:def}{somas} de dobros.}
\begin{enumerate}[resume]
  \item Na questão 6, as linhas sobreviventes valiam $1$, $4$ e $8$
        dobros de $24$: o método escreveu, sem dizer,
        $13 \times 24 = (1 + 4 + 8) \times 24$. Confira que
        $1 + 4 + 8 = 13$ e descubra que dobros ($1$, $2$, $4$, $8$,
        $16$) o método guardou para o $21$ da questão 7.
  \item Escreva $25$ e $42$ como \omterm{def:g1:addition:def}{somas} de números da lista de dobros
        $1, 2, 4, 8, 16, 32$, usando cada um no máximo uma vez.
  \item Explique por que \emph{todo} número inteiro pode ser escrito
        assim: tome o maior número de dobro que couber, subtraia,
        repita --- por que isso tem de parar? Escreva $100$ como uma
        \omterm{def:g1:addition:def}{soma} dessas.
  \item Os escribas do Egito faziam exatamente isso, quatro mil anos
        atrás: para multiplicar $19 \times 35$, montavam uma tabela de
        dobros de $35$ ($1 \to 35$, $2 \to 70$, $4 \to 140$,
        $8 \to 280$, $16 \to 560$), escolhiam as linhas que somavam
        $19$ e somavam. Faça a conta do jeito deles.
  \item Final: os computadores modernos, que contam com $0$ e $1$,
        multiplicam essencialmente como os escribas --- por dobros e
        adições. Que três adições um computador faria para
        $13 \times 24$? (Por que todo número se reparte em dobros, e o
        que os $0$ e os $1$ têm a ver com isso, é o problema de fim de
        semana do capítulo de potências: \cref{pb:g8:powers:1}.)
\end{enumerate}
\end{problem}
