\chapter{Organizando dados}\label{ch:g7:stats}

Antes de calcular qualquer coisa, os dados precisam ser coletados,
contados e apresentados. Este capítulo trata de transformar uma lista
bagunçada de observações numa tabela ou numa imagem clara: \omterm{def:g7:stats:frequency}{frequências
absolutas} e relativas, dados agrupados, gráficos de barras e gráficos
de setores. Resumir dados por um único número (média, mediana) é
assunto do \cref{ch:g9:statproba}.

\section{Frequências absolutas e relativas}

\begin{definition}[Frequência absoluta e relativa]\label{def:g7:stats:frequency}
Para cada valor possível de uma pesquisa, a \emph{frequência
absoluta}\index{frequência absoluta} dele é o número de vezes que ele
aparece; a \emph{frequência relativa}\index{frequência relativa} é a
frequência absoluta dividida pelo número total de observações:
\[
\text{frequência relativa}
= \frac{\text{frequência absoluta}}{\text{total}} .
\]
As frequências relativas podem ser escritas como frações, decimais ou
porcentagens, e somam sempre $1$ (ou seja, $100\,\%$).
\end{definition}

\begin{example}\label{ex:g7:stats:frequency}
Esporte preferido dos $25$ alunos de uma turma, da lista bruta à
tabela:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cccc|c}
esporte & futebol & natação & tênis & judô & total \\
\hline
freq. absoluta & $10$ & $6$ & $5$ & $4$ & $25$ \\
freq. relativa & $0.40$ & $0.24$ & $0.20$ & $0.16$ & $1$ \\
\end{tabular}
\end{center}
Para o futebol: $\frac{10}{25} = \frac{40}{100} = 0.40 = 40\,\%$. A
linha das \omterm{def:g7:stats:frequency}{frequências relativas} tem de somar $1$ --- uma conferência
permanente e gratuita.
\end{example}

\begin{remark}[Por que frequências relativas?]
\omterm{def:g7:stats:frequency}{Frequências absolutas} não podem ser comparadas entre grupos de tamanhos
diferentes: $10$ torcedores de futebol numa turma de $25$ (isto é,
$40\,\%$) mostram mais entusiasmo que $12$ num ano escolar de $120$
($10\,\%$). As \omterm{def:g7:stats:frequency}{frequências relativas} põem tudo na mesma escala.
\end{remark}

\section{Agrupar dados em classes}

\begin{example}[Classes]\label{ex:g7:stats:classes}
As alturas (em cm) de $20$ alunos vão de $148$ a $172$. Listar cada
altura separadamente daria uma tabela com quase tantas colunas quantos
alunos; em vez disso, agrupamos em \emph{classes}:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|ccc|c}
altura (cm) & $\intco{145}{155}$ & $\intco{155}{165}$ &
$\intco{165}{175}$ & total \\
\hline
freq. absoluta & $5$ & $9$ & $6$ & $20$ \\
freq. relativa & $0.25$ & $0.45$ & $0.30$ & $1$ \\
\end{tabular}
\end{center}
Cada altura é contada em exatamente uma classe ($155$ entra em
$\intco{155}{165}$, e não nas duas --- é por isso que os intervalos são
semiabertos). Agrupar perde detalhe, mas ganha legibilidade.
\end{example}

\section{Gráficos}

\begin{method}[Escolher e traçar um gráfico]\label{met:g7:stats:charts}
\begin{enumerate}
  \item \emph{Gráfico de barras}: uma barra por valor ou classe, com
        altura proporcional à \omterm{def:g7:stats:frequency}{frequência absoluta} (ou à relativa). Bom
        para comparar valores.
  \item \emph{Gráfico de setores}: um setor por valor, com ângulo
        proporcional à \omterm{def:g7:stats:frequency}{frequência relativa} --- o círculo inteiro
        ($360^\circ$) representa o total, então
        \[
        \text{ângulo} = \text{frequência relativa} \times 360^\circ .
        \]
        Bom para mostrar partes de um todo.
  \item Nos dois casos: título, rótulos e uma graduação ou as
        porcentagens escritas no gráfico.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:stats:pie}
Ângulos para os esportes do \cref{ex:g7:stats:frequency}: futebol
$0.40 \times 360 = 144^\circ$; natação $0.24 \times 360 = 86.4^\circ$;
tênis $72^\circ$; judô $57.6^\circ$. Confere:
$144 + 86.4 + 72 + 57.6 = 360$.
\end{example}

\begin{example}[Ler com espírito crítico]\label{ex:g7:stats:critical}
Um gráfico cujo eixo vertical começa em $9$ em vez de $0$ faz uma barra
de $10$ parecer dez vezes mais alta que uma barra de $9.1$ ---
visualmente enganoso, ainda que os números sejam honestos. Primeiro
reflexo diante de qualquer gráfico: veja onde o eixo começa e quanto
vale uma graduação (\cref{met:g6:propdata:read}).
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g7:stats:1}
Eis as notas de $20$ alunos: $12$, $15$, $9$, $12$, $14$, $9$, $12$,
$15$, $11$, $12$, $9$, $14$, $12$, $11$, $15$, $9$, $12$, $14$, $11$,
$12$. Monte a tabela de \omterm{def:g7:stats:frequency}{frequências absolutas} (uma coluna por nota).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:2}
Complete a tabela do \cref{exo:g7:stats:1} com uma linha de
\omterm{def:g7:stats:frequency}{frequências relativas} (como frações de $20$ e depois como
porcentagens). Confira a \omterm{def:g1:addition:def}{soma}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:3}
Numa pesquisa com $50$ famílias, $30$ têm um carro. Qual é a
\omterm{def:g7:stats:frequency}{frequência relativa} de ``um carro'', como decimal e como porcentagem?
Em outra pesquisa, $45$ famílias em $90$ têm um carro: que pesquisa
mostra a proporção maior?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:4}
As \omterm{def:g2:measure:units}{massas} (kg) de $15$ cães: $8$, $23$, $31$, $12$, $9$, $27$, $35$,
$14$, $22$, $18$, $29$, $11$, $25$, $33$, $16$. Agrupe-as nas classes
$\intco{5}{15}$, $\intco{15}{25}$, $\intco{25}{35}$,
$\intco{35}{45}$ e dê as \omterm{def:g7:stats:frequency}{frequências absolutas}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:5}
Trace o gráfico de barras da tabela:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|cccc}
bichos de estimação & $0$ & $1$ & $2$ & $3$ \\
\hline
famílias & $8$ & $12$ & $6$ & $4$ \\
\end{tabular}
\end{center}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:6}
Para a tabela do \cref{exo:g7:stats:5}, calcule as \omterm{def:g7:stats:frequency}{frequências
relativas} e os ângulos do gráfico de setores de cada valor (confira:
total $360^\circ$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:7}
Um gráfico de setores sobre as estações preferidas mostra: verão
$162^\circ$, primavera $90^\circ$, outono $54^\circ$, inverno
$54^\circ$. Que porcentagem escolheu cada estação? Se $60$ pessoas
responderam, quantas escolheram o verão?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:8}
Na turma A, $12$ alunos em $30$ vão a pé para a escola; na turma B,
$14$ em $40$. Que turma tem a maior proporção de alunos que vão a pé?
Justifique com \omterm{def:g7:stats:frequency}{frequências relativas}, e não com as absolutas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:9}
Uma revista publica um gráfico de barras das vendas mensais: $9\,800$,
depois $10\,000$, depois $10\,100$, com o eixo vertical começando em
$9\,700$. Descreva o que o leitor vê e o que um eixo honesto, começando
em $0$, mostraria.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:10}
Uma tabela de frequências tem três valores com \omterm{def:g7:stats:frequency}{frequências relativas}
$0.35$, $0.4$ e $f$. Encontre $f$. Se o total for $80$, dê as três
\omterm{def:g7:stats:frequency}{frequências absolutas}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:stats:11}
Numa escola, $55\,\%$ dos alunos são meninas. Entre as meninas,
$40\,\%$ almoçam no refeitório; entre os meninos, $60\,\%$ almoçam. Em
$400$ alunos, quantos almoçam no refeitório? (Calcule os tamanhos dos
quatro grupos passo a passo.) Que porcentagem geral é essa?
\end{exercise}

\section{Problema: como os dados mentem --- e como pegá-los}

\begin{problem}[{Problema de fim de semana --- frequências absolutas
contra relativas, a pesquisa que enganou um país e o paradoxo dos dois
hospitais}]
\label{pb:g7:stats:1}
Os números não mentem, mas podem ser arranjados para enganar: uma
contagem enorme pode esconder uma frequência pequena, uma pesquisa
gigantesca pode não valer nada e --- o mais estranho de tudo --- um
hospital pode vencer o rival em \emph{todas} as categorias de pacientes
e ainda assim perder nos números gerais. As três armadilhas são armadas
neste problema, munidas apenas das frequências deste capítulo
(\cref{def:g7:stats:frequency}).

\medskip
\noindent\textbf{Parte I --- \omterm{def:g7:stats:frequency}{Frequência absoluta} não é \omterm{def:g7:stats:frequency}{frequência
relativa}.}
\begin{enumerate}
  \item A escola A recicla $120$ das $400$ caixinhas de suco dela; a
        escola B recicla $90$ das $250$ dela. Que escola tem a maior
        frequência \emph{absoluta} de caixinhas recicladas? E a maior
        frequência \emph{relativa}? Que escola merece o prêmio de
        reciclagem?
  \item Um horário novo é submetido a pesquisa: $18$ dos $30$
        professores gostam dele, e $210$ dos $600$ alunos também.
        Calcule as duas \omterm{def:g7:stats:frequency}{frequências relativas} e depois a \omterm{def:g7:stats:frequency}{frequência
        relativa} entre as $630$ pessoas juntas. Por que o número
        combinado fica tão perto do dos alunos?
  \item Um gráfico de setores publicado mostra setores rotulados
        $45\,\%$, $30\,\%$, $20\,\%$ e $10\,\%$. Sem nenhuma outra
        informação, como você sabe que houve erro?
  \item Uma tabela de frequências sobre $40$ observações está meio
        apagada: \omterm{def:g7:stats:frequency}{frequências absolutas} $14$, $10$, $?$, $?$ e
        relativas $0.35$, $0.25$, $0.15$, $?$. Reconstrua as entradas
        que faltam.
  \item Uma pesquisa termina com \omterm{def:g7:stats:frequency}{frequências relativas} $0.40$, $0.35$ e
        $0.25$. Calcule os três ângulos do gráfico de setores
        (\cref{met:g7:stats:charts}) e confira que eles fecham o
        círculo.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A pesquisa que enganou um país.}
Em 1936, uma revista norte-americana enviou dez milhões de cédulas pelo
correio a endereços tirados de listas telefônicas e de registros de
automóveis, recebeu $2.4$ milhões de respostas e previu uma vitória
esmagadora do candidato Landon. Um jovem estatístico, George Gallup,
perguntou a apenas cerca de cinquenta mil pessoas --- escolhidas para
se parecerem com a população inteira --- e previu o contrário.
Roosevelt ganhou de goleada.
\begin{enumerate}[resume]
  \item Que \omterm{def:g7:stats:frequency}{frequência relativa} das cédulas enviadas voltou?
  \item Em 1936, telefones e automóveis eram luxos. Explique em uma ou
        duas frases por que a amostra da revista, embora enorme, estava
        condenada --- e que lição da questão 1 ela repete em escala
        nacional.
  \item Um modelo de brinquedo. Uma cidade tem $1\,000$ eleitores
        ricos, dos quais $70\,\%$ apoiam L, e $9\,000$ eleitores
        modestos, dos quais $30\,\%$ apoiam L. Calcule o apoio
        verdadeiro a L na cidade. Uma ``pesquisa da lista telefônica''
        alcança $500$ ricos e $500$ modestos: que apoio ela prevê?
  \item A armadilha silenciosa: uma empresa tem $200$ clientes
        insatisfeitos e $800$ satisfeitos. Ela consulta todo mundo;
        $40\,\%$ dos insatisfeitos respondem, mas só $10\,\%$ dos
        satisfeitos. Calcule o número de respostas de cada tipo e a
        \omterm{def:g7:stats:frequency}{frequência relativa} de insatisfação \emph{entre as respostas}.
        Compare com a frequência verdadeira.
  \item Conserte a pesquisa da questão 8: mantenha $1\,000$ entrevistas,
        mas distribua-as nas proporções verdadeiras da cidade. O que a
        pesquisa consertada prevê?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O paradoxo dos dois hospitais.}
Dois hospitais publicam as contagens de curas deles, separadas por
gravidade:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cc}
 & casos leves & casos graves \\
\hline
hospital A & $90$ curados de $100$ & $280$ curados de $400$ \\
hospital B & $340$ curados de $400$ & $65$ curados de $100$ \\
\end{tabular}
\end{center}
\begin{enumerate}[resume]
  \item Calcule a taxa de cura do hospital A nos casos leves, nos casos
        graves e no total (os $500$ pacientes).
  \item As mesmas três contas para o hospital B. Que hospital vence nos
        casos leves? E nos casos graves?
  \item Compare as duas taxas gerais. Enuncie com clareza a coisa
        estranha que aconteceu.
  \item Explique o truque: compare as \emph{misturas} de pacientes dos
        dois hospitais e diga por que a taxa geral pode trair as duas
        taxas por categoria. Se você tivesse um caso grave, que
        hospital deveria escolher?
  \item Uma manchete de jornal diz: ``Hospital B tem a melhor taxa de
        cura: $81\,\%$ contra $74\,\%$''. Escreva a carta de duas
        frases ao editor que este problema inteiro ensinou você a
        escrever --- uma frase sobre o que a manchete ignora e outra
        sobre a moral geral (compare o comparável e preste atenção aos
        pesos).
\end{enumerate}
\end{problem}
