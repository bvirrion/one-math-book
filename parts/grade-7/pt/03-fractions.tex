\chapter{Frações: comparar e somar}\label{ch:g7:fractions}

O Ano 6 apresentou as frações como partes e como \omterm{def:g3:division:remainder}{quocientes} exatos
(\cref{ch:g6:fractions}). Este capítulo aprende a \emph{calcular} com
elas: reconhecer frações iguais, comparar, somar e subtrair --- por
enquanto com \omterm{def:g4:fractions:def}{denominadores} amigáveis; o caso geral vem no
\cref{ch:g9:fractions}.

\section{Frações iguais}

\begin{theorem}[Frações iguais]\label{thm:g7:fractions:equal}
Uma \omterm{def:g6:fractions:def}{fração} não muda quando o \omterm{def:g4:fractions:def}{numerador} e o \omterm{def:g4:fractions:def}{denominador} dela são
multiplicados, ou divididos, pelo mesmo número não nulo:
\[
\frac{a}{b} = \frac{a \times k}{b \times k} .
\]
\end{theorem}
\admitted

\begin{example}\label{ex:g7:fractions:equal}
$\dfrac{3}{4} = \dfrac{3 \times 5}{4 \times 5} = \dfrac{15}{20}$ e, no
outro sentido, $\dfrac{42}{30} = \dfrac{42 \div 6}{30 \div 6}
= \dfrac{7}{5}$ (simplificada por $6$).

Para testar se $\dfrac{8}{12}$ e $\dfrac{10}{15}$ são iguais,
simplifique as duas: $\dfrac{8}{12} = \dfrac{2}{3}$ e
$\dfrac{10}{15} = \dfrac{2}{3}$ --- sim, são iguais.
\end{example}

\section{Comparar frações}

\begin{method}[Comparar duas frações]\label{met:g7:fractions:compare}
\begin{enumerate}
  \item \emph{Mesmo \omterm{def:g4:fractions:def}{denominador}}: compare os \omterm{def:g4:fractions:def}{numeradores}:
        $\frac{5}{9} < \frac{7}{9}$.
  \item \emph{Um \omterm{def:g4:fractions:def}{denominador} é \omterm{def:g5:division:multiple}{múltiplo} do outro}: reescreva a \omterm{def:g6:fractions:def}{fração}
        mais grossa e depois compare: $\frac{5}{6}$ contra
        $\frac{7}{12}$: $\frac56 = \frac{10}{12} > \frac{7}{12}$.
  \item \emph{Compare com $1$ ou com $\frac12$} quando der:
        $\frac{9}{8} > 1 > \frac{7}{9}$ resolve $\frac98$ contra
        $\frac79$ na hora.
\end{enumerate}
\end{method}

\section{Somar e subtrair}

\begin{theorem}[Mesmo denominador]\label{thm:g7:fractions:add}
Frações de mesmo \omterm{def:g4:fractions:def}{denominador} são somadas (ou subtraídas) somando (ou
subtraindo) os \omterm{def:g4:fractions:def}{numeradores}:
\[
\frac{a}{d} + \frac{b}{d} = \frac{a + b}{d},
\qquad
\frac{a}{d} - \frac{b}{d} = \frac{a - b}{d} .
\]
\end{theorem}

\begin{proof}[Por quê]
$a$ partes de tamanho $\frac1d$ mais $b$ partes do mesmo tamanho fazem
$a + b$ partes desse tamanho.
\end{proof}

\begin{method}[Denominadores diferentes]\label{met:g7:fractions:add}
Quando um \omterm{def:g4:fractions:def}{denominador} é \omterm{def:g5:division:multiple}{múltiplo} do outro:
\begin{enumerate}
  \item reescreva a \omterm{def:g6:fractions:def}{fração} de \omterm{def:g4:fractions:def}{denominador} menor para que as duas fiquem
        com o \omterm{def:g4:fractions:def}{denominador} maior (\cref{thm:g7:fractions:equal});
  \item some ou subtraia os \omterm{def:g4:fractions:def}{numeradores};
  \item simplifique o resultado, se der.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:fractions:add}
Vamos calcular $\dfrac{3}{4} + \dfrac{5}{8}$, passo a passo:
\begin{enumerate}
  \item $8$ é \omterm{def:g5:division:multiple}{múltiplo} de $4$: reescreva
        $\dfrac34 = \dfrac{3 \times 2}{4 \times 2} = \dfrac{6}{8}$;
  \item some: $\dfrac{6}{8} + \dfrac{5}{8} = \dfrac{11}{8}$;
  \item não há o que simplificar: o resultado é $\dfrac{11}{8}$, ou
        seja, $1 + \dfrac38$.
\end{enumerate}
Outro: $2 - \dfrac{4}{5} = \dfrac{10}{5} - \dfrac{4}{5} =
\dfrac{6}{5}$ (escreva primeiro o número inteiro sobre o \omterm{def:g4:fractions:def}{denominador}
$5$).
\end{example}

\begin{example}[Uma armadilha clássica]\label{ex:g7:fractions:trap}
$\dfrac12 + \dfrac13$ \emph{não} é $\dfrac{2}{5}$! Somar \omterm{def:g4:fractions:def}{numeradores} e
\omterm{def:g4:fractions:def}{denominadores} separadamente está errado --- a resposta $\frac25$ seria
menor que $\frac12$, o que é absurdo quando se \emph{\omterm{def:g1:addition:def}{soma}} algo
positivo a $\frac12$. (A \omterm{def:g1:addition:def}{soma} correta, $\frac56$, precisa do
\omterm{def:g4:fractions:def}{denominador} comum $6$; o método geral está no
\cref{ch:g9:fractions}.)
\end{example}

\section{Multiplicar uma fração por um número}

\begin{proposition}[Fração vezes um número]\label{prop:g7:fractions:mult}
Para um número $k$:
\[
k \times \frac{a}{b} = \frac{k \times a}{b} .
\]
Tomar $\frac ab$ \emph{de} uma quantidade é multiplicar por
$\frac ab$.
\end{proposition}
\admitted

\begin{example}\label{ex:g7:fractions:mult}
$6 \times \dfrac{2}{3} = \dfrac{12}{3} = 4$: seis vezes dois terços dá
quatro inteiros. E ``$\frac34$ de $60$ reais'' é
$\frac34 \times 60 = \frac{180}{4} = 45$ reais --- a mesma resposta de
dividir por $4$ e depois multiplicar por $3$
(\cref{met:g6:fractions:ofquantity}).
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g7:fractions:1}
Complete: $\dfrac{2}{5} = \dfrac{?}{20}$; \quad
$\dfrac{18}{24} = \dfrac{3}{?}$; \quad
$\dfrac{7}{3} = \dfrac{28}{?}$; \quad
$\dfrac{?}{9} = \dfrac{20}{36}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:2}
Simplifique o máximo possível:
$\dfrac{12}{16}$; \quad $\dfrac{30}{45}$; \quad $\dfrac{27}{36}$;
\quad $\dfrac{48}{12}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:3}
$\dfrac{15}{35}$ e $\dfrac{21}{49}$ são iguais? E $\dfrac{16}{28}$ e
$\dfrac{20}{36}$? Justifique simplificando.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:4}
Compare (escreva o raciocínio, não só a resposta):
\[
\frac{7}{11} \text{ e } \frac{9}{11}; \qquad
\frac{5}{6} \text{ e } \frac{11}{18}; \qquad
\frac{13}{12} \text{ e } \frac{19}{20}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:5}
Calcule e simplifique quando der:
\[
\frac{5}{9} + \frac{2}{9}, \qquad
\frac{11}{7} - \frac{4}{7}, \qquad
\frac{5}{8} + \frac{7}{8} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:6}
Calcule usando o \cref{met:g7:fractions:add}:
\[
\frac{2}{3} + \frac{5}{6}, \qquad
\frac{7}{10} - \frac{2}{5}, \qquad
\frac{3}{4} + \frac{5}{12} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:7}
Calcule:
\[
3 - \frac{5}{4}, \qquad
1 + \frac{3}{8}, \qquad
2 - \frac{7}{6} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:8}
Calcule:
\[
5 \times \frac{3}{10}, \qquad
\frac{7}{8} \times 4, \qquad
\frac{2}{3} \text{ de } 45 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:fractions:9}
Marcos comeu $\frac14$ de uma pizza e Júlia comeu $\frac38$ da mesma
pizza. Que \omterm{def:g6:fractions:def}{fração} da pizza eles comeram juntos? Que \omterm{def:g6:fractions:def}{fração} sobrou?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:fractions:10}
Uma garrafa contém $\frac34$ L de suco. Lia serve $\frac16$ L duas
vezes. Quanto suco resta? (\omterm{def:g4:fractions:def}{Denominador} comum: $12$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:fractions:11}
Explique, com o argumento do \cref{ex:g7:fractions:trap}, por que
$\frac35 + \frac12$ não pode ser $\frac{4}{7}$, sem calcular a \omterm{def:g1:addition:def}{soma}
correta.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:fractions:12}
Calcule a \omterm{def:g1:addition:def}{soma}
\[
\frac12 + \frac14 + \frac18 + \frac{1}{16}
\]
passo a passo (da esquerda para a direita). Observe o padrão das \omterm{def:g1:addition:def}{somas}
parciais: de que número elas se aproximam à medida que mais e mais
termos $\frac{1}{32}, \frac{1}{64}, \dots$ são somados?
\end{exercise}

\section{Problema: as frações escondidas num compasso de música}

\begin{problem}[{Problema de fim de semana --- os valores das notas são
frações que têm de somar exatamente, a contagem de ritmos esconde uma
sequência famosa, e a paridade manda no ritmo balcânico}]
\label{pb:g7:fractions:1}
A música se escreve em frações. Uma \emph{semibreve} dura $1$ compasso
(no comum ``quatro por quatro''); uma \emph{mínima} dura $\frac12$;
depois vêm a \emph{semínima} $\frac14$, a \emph{colcheia} $\frac18$ e a
\emph{semicolcheia} $\frac{1}{16}$. Uma lei inviolável: \emph{as
durações dentro de um compasso têm de somar exatamente o total do
compasso} --- $1$ no quatro por quatro. Cada questão abaixo é a
aritmética deste capítulo (\cref{thm:g7:fractions:add},
\cref{met:g7:fractions:add}) posta em música; pelo caminho você vai
encontrar uma sequência de contagem descoberta por sábios indianos há
mil anos.

\medskip
\noindent\textbf{Parte I --- Notas que têm de fechar a conta.}
\begin{enumerate}
  \item Confira que cada um destes compassos é legal no quatro por
        quatro: (a) mínima $+$ semínima $+$ semínima; (b) semínima $+$
        colcheia $+$ colcheia $+$ mínima.
  \item Um copista escreveu: mínima $+$ semínima $+$ colcheia. Que
        \omterm{def:g6:fractions:def}{fração} do compasso está escrita, e que nota única a completa?
  \item Quantas semicolcheias enchem um compasso inteiro? E quantas
        semicolcheias fazem uma colcheia?
  \item Um \emph{ponto de aumento} depois de uma nota acrescenta a
        \omterm{def:g3:division:half}{metade} do valor dela: uma mínima pontuada é $\frac12 +
        \frac14$, e uma semínima pontuada é $\frac14 + \frac18$.
        Calcule os dois valores.
  \item Uma valsa se escreve em três por quatro: cada compasso tem de
        totalizar $\frac34$. Confira que uma única mínima pontuada
        enche um compasso de valsa e escreva outros dois compassos de
        valsa legais usando apenas semínimas e colcheias.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A oficina do copista.}
\begin{enumerate}[resume]
  \item Um compasso de quatro por quatro contém: semínima pontuada $+$
        colcheia $+$ semínima. O que está faltando?
  \item O que dura mais, uma colcheia pontuada ou uma semínima? Compare
        em semicolcheias.
  \item Uma \emph{ligadura} cola durações num só som prolongado: uma
        mínima ligada a uma colcheia dura quanto? Quantas semicolcheias
        são?
  \item As notas de um compasso somam $\frac{11}{16}$; o \omterm{def:g3:division:remainder}{resto} é
        silêncio (uma \emph{pausa}). Quanto dura a pausa?
  \item Um baterista preenche uma semínima ($\frac{4}{16}$ do compasso)
        usando apenas colcheias ($\frac{2}{16}$) e semicolcheias
        ($\frac{1}{16}$), importando a ordem: por exemplo,
        colcheia--semicolcheia--semicolcheia ou
        semicolcheia--colcheia--semicolcheia. Confira que os dois
        exemplos são legais e depois liste \emph{todos} os ritmos
        possíveis. Quantos são?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Uma sequência de mil anos e um compasso
\omterm{def:g3:numbers:evenodd}{ímpar}.}
\begin{enumerate}[resume]
  \item Vamos contar os ritmos do baterista para notas mais longas.
        Todo ritmo termina ou com uma semicolcheia, ou com uma
        colcheia; explique por que isso dá a regra: (ritmos que enchem
        $n$ semicolcheias) $=$ (ritmos que enchem $n - 1$) $+$ (ritmos
        que enchem $n - 2$). Partindo de $1$ ritmo para uma
        semicolcheia e $2$ para duas, monte a tabela até $n = 8$:
        quantos ritmos enchem uma mínima?
  \item Os números que você encontrou --- $1, 2, 3, 5, 8, 13, 21, 34$
        --- foram publicados pelo sábio indiano Hemachandra por volta
        do ano 1150, contando exatamente ritmos desses, duas gerações
        antes de Fibonacci os anotar na Itália. Enuncie o padrão que os
        define em uma frase e confira-o nas três últimas entradas da
        sua tabela.
  \item O estilo \emph{swing} substitui duas colcheias iguais por um
        par longo--curto: $\frac16$ e depois $\frac{1}{12}$. Verifique
        que a troca é legal (mesma duração total), usando o \omterm{def:g4:fractions:def}{denominador}
        comum $12$.
  \item Uma \emph{quiáltera} espreme três notas iguais dentro de uma
        semínima. Quanto dura cada nota (confira a sua resposta
        multiplicando por $3$, \cref{prop:g7:fractions:mult})? A mesma
        pergunta para três notas iguais preenchendo uma mínima.
  \item A música de dança dos Bálcãs usa o sete por oito: cada compasso
        totaliza $\frac78$. Verifique que semínima pontuada $+$ semínima
        $+$ semínima o preenche. Depois explique por que
        \emph{nenhuma} combinação só de mínimas e semínimas, por mais
        numerosas que sejam, consegue preencher um compasso de sete por
        oito. (Conte em colcheias e pense em par e \omterm{def:g3:numbers:evenodd}{ímpar}.)
\end{enumerate}
\end{problem}
