\chapter{Expressões literais}\label{ch:g7:literal}

Uma letra num cálculo representa um número que ainda não conhecemos, ou
\emph{todos} os números de uma vez: escrever $P = 4c$ dá o \omterm{def:g6:measure:perimeter}{perímetro} de
todo quadrado numa fórmula só. Este capítulo ensina as convenções de
escrita, como substituir valores e como simplificar expressões simples
--- o começo da álgebra, que continua no \cref{ch:g8:equations}.

\section{Escrever com letras}

\begin{definition}[Expressão literal]\label{def:g7:literal:def}
Uma \emph{expressão literal}\index{expressão literal} é um cálculo que
contém uma ou mais letras, cada uma representando um número. Convenções
de escrita:
\begin{itemize}
  \item o sinal $\times$ é omitido antes de uma letra ou de um
        parêntese: $3 \times a = 3a$, \ $a \times b = ab$, \
        $2 \times (x + 5) = 2(x + 5)$;
  \item mas ele fica entre dois números: $3 \times 5$ não pode ser
        escrito $35$!
  \item $a \times a$ escreve-se $a^2$ (``$a$ ao quadrado'') e
        $a \times a \times a = a^3$;
  \item $1a$ escreve-se $a$, e $0a$ é $0$.
\end{itemize}
\end{definition}

\begin{example}\label{ex:g7:literal:write}
Simplifique a escrita:
\[
5 \times x + 3 \times y = 5x + 3y,
\qquad
a \times 7 = 7a \ \text{(o número na frente)},
\qquad
x \times x \times 4 = 4x^2 .
\]
\end{example}

\begin{example}[As fórmulas são expressões literais]\label{ex:g7:literal:formulas}
Para um retângulo de comprimento $L$ e largura $w$: \omterm{def:g6:measure:perimeter}{perímetro}
$P = 2(L + w)$, \omterm{def:g6:measure:area}{área} $A = Lw$. Uma fórmula só resume de uma vez os
infinitos retângulos --- é para isso que servem as letras.
\end{example}

\section{Substituir}

\begin{method}[Calcular o valor de uma expressão]\label{met:g7:literal:substitute}
Para calcular o valor de uma expressão com valores dados das letras:
\begin{enumerate}
  \item reescreva a expressão devolvendo os sinais $\times$
        escondidos: $3a + a^2$ é $3 \times a + a \times a$;
  \item substitua cada letra pelo valor dela, mantendo parênteses em
        volta se for preciso;
  \item calcule, respeitando as prioridades do
        \cref{ch:g7:priorities}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:literal:substitute}
Calcule $E = 3x + x^2$ para $x = 5$:
\[
E = 3 \times 5 + 5 \times 5 = 15 + 25 = 40 .
\]
Calcule $F = 2(a + 3b)$ para $a = 4$ e $b = 0.5$:
\[
F = 2 \times (4 + 3 \times 0.5) = 2 \times (4 + 1.5) = 2 \times 5.5
= 11 .
\]
\end{example}

\section{Reduzir expressões}

\begin{proposition}[Juntar termos semelhantes]\label{prop:g7:literal:reduce}
Os termos que têm a mesma parte literal podem ser juntados somando as
partes numéricas deles (isso é a distributividade,
\cref{thm:g7:priorities:distributivity}, lida ao contrário):
\[
5x + 3x = (5 + 3)x = 8x,
\qquad
7a - 2a = 5a .
\]
Termos com partes literais diferentes (como $3x$ e $5y$, ou $x$ e
$x^2$) não podem ser juntados.
\end{proposition}

\begin{example}\label{ex:g7:literal:reduce}
Reduza $E = 4x + 7 + 3x + 2$, separando os termos:
\[
E = (4x + 3x) + (7 + 2) = 7x + 9 .
\]
Cuidado: $7x + 9$ \emph{não} é $16x$! Teste com $x = 1$:
$4 + 7 + 3 + 2 = 16$ e $7 \times 1 + 9 = 16$ coincidem, mas $16x$
também daria $16$ --- teste com $x = 2$: a expressão original dá
$8 + 7 + 6 + 2 = 23$, e $7 \times 2 + 9 = 23$, enquanto
$16 \times 2 = 32$. A redução $7x + 9$ está certa; $16x$ está errada.
\end{example}

\begin{method}[Testar uma igualdade]\label{met:g7:literal:test}
Para verificar se duas expressões podem ser iguais para todo valor:
\begin{enumerate}
  \item substitua o mesmo valor nas duas, ao menos duas vezes (por
        exemplo, $x = 2$ e $x = 10$; evite $0$ e $1$, que escondem
        \omterm{ex:g1:subtraction:difference}{diferenças} demais);
  \item se os resultados diferirem uma única vez, as expressões
        \emph{não} são iguais --- um contraexemplo derruba uma
        afirmação geral;
  \item se os resultados coincidirem, a igualdade é apenas
        \emph{plausível}: a demonstração exige álgebra (desenvolver,
        reduzir).
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:literal:test}
$2(x + 3)$ é igual a $2x + 6$ para todo $x$? Desenvolver demonstra
isso: $2(x+3) = 2x + 2 \times 3 = 2x + 6$. E $(x + 1)^2$ é igual a
$x^2 + 1$? Teste $x = 2$: $(2+1)^2 = 9$, mas $2^2 + 1 = 5$. Não --- e
um contraexemplo é uma demonstração completa do ``não''.
\end{example}

\section{Produzir expressões}

\begin{example}[Da situação à expressão]\label{ex:g7:literal:produce}
Um táxi cobra $4$ reais mais $2$ reais por quilômetro. Para uma corrida
de $k$ quilômetros, o preço é
\[
p = 4 + 2k \quad\text{reais}.
\]
Para $k = 7$: $p = 4 + 14 = 18$ reais. A expressão \emph{é} a resposta
geral; substituir produz cada resposta particular.
\end{example}

\begin{example}[Truques com números]\label{ex:g7:literal:trick}
``Pense num número, dobre-o, some $10$, divida por $2$, subtraia o seu
número.'' Chame o número de $n$ e siga os passos:
\[
n \ \to\ 2n \ \to\ 2n + 10 \ \to\ \frac{2n + 10}{2} = n + 5 \ \to\
n + 5 - n = 5 .
\]
Todo mundo obtém $5$, qualquer que tenha sido $n$ --- as letras
explicam a mágica.
\end{example}

7 \times x, \qquad

x \times x, \qquad
2 \times (a + 4).

5ab, \qquad

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:literal:3}
Calcule o valor para $x = 3$: \ $5x + 1$; \quad $x^2$; \quad

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:literal:4}
Calcule $3a + 2b$ e $a b + 5$ para $a = 4$ e $b = 1.5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:literal:5}
Reduza:
\[
8x + 5x, \qquad
9a - 4a + a, \qquad
6y + 2 + 3y + 7, \qquad
5t + 3 - 2t - 3 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:literal:6}
Dê uma fórmula para: o \omterm{def:g6:measure:perimeter}{perímetro} de um quadrado de lado $c$; o
\omterm{def:g6:measure:perimeter}{perímetro} de um \omterm{def:g6:shapes:triangles}{triângulo isósceles} de base $b$ e lados iguais $s$; o
preço de $n$ pães a $1.20$ reais cada um.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:literal:7}
Ligue cada expressão a uma descrição: $2n$; $n + 2$; $n^2$;
$\frac n2$. Descrições: ``a \omterm{def:g3:division:half}{metade} de $n$''; ``o dobro de $n$''; ``o
quadrado de $n$''; ``duas unidades a mais que $n$''.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:literal:8}
Usando o \cref{met:g7:literal:test}, decida (com demonstração ou
contraexemplo) se estas igualdades valem para todo $x$:
\[
3(x + 4) = 3x + 12; \qquad
2x + 5 = 7x; \qquad
x + x = x^2 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:literal:9}
Um retângulo tem largura $w$ e comprimento $3$ cm maior que a largura.
\begin{enumerate}
  \item Expresse o comprimento e depois o \omterm{def:g6:measure:perimeter}{perímetro} em função de $w$, e
        reduza.
  \item Calcule o \omterm{def:g6:measure:perimeter}{perímetro} para $w = 5$ cm, diretamente e com a sua
        fórmula reduzida. A resposta é a mesma?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:literal:10}
Refaça o truque do \cref{ex:g7:literal:trick} com letras: ``pense num

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:literal:11}
A figura abaixo é um quadrado de lado $a$ com um canto quadrado de lado
$b$ retirado ($b$ menor que $a$). Escreva duas expressões diferentes
para a \omterm{def:g6:measure:area}{área} dela (uma como \omterm{ex:g1:subtraction:difference}{diferença} e outra cortando o L em dois

\end{exercise}

\section{Problema: o oráculo dos palitos de fósforo}

\begin{problem}[{Problema de fim de semana --- padrões que crescem e o
termo geral deles: três fórmulas para um mesmo padrão, e o padrão
famoso que quebra no sexto passo}]
\label{pb:g7:literal:1}
Monte palitos de fósforo numa fileira de quadrados: um quadrado, depois
dois, depois três\,\dots\ De quantos palitos a centésima figura vai
precisar? Contar um a um levaria o fim de semana inteiro; uma
\emph{fórmula em $n$} responde na hora para todas as figuras de uma vez
--- é exatamente para isso que servem as letras
(\cref{ex:g7:literal:formulas}). Mas atenção: este problema termina com
um padrão célebre que joga limpo por cinco passos e depois trai todos
os que confiaram nele.

\medskip
\noindent\textbf{Parte I --- Uma fileira de quadrados.}
A figura $1$ é um quadrado de palitos ($4$ palitos); cada figura
seguinte acrescenta mais um quadrado que compartilha um lado com o
anterior.
\begin{enumerate}
  \item Desenhe as figuras $1$ a $4$ e conte os palitos delas.
  \item Explique por que cada quadrado novo custa exatamente $3$
        palitos a mais e use isso para prever a contagem da figura $10$
        sem nenhuma fórmula.
  \item Justifique a fórmula: a figura $n$ precisa de $3n + 1$ palitos.
        (De onde vem o ``$+1$'' solitário?) Confira-a com a questão 1.
  \item Amir raciocina de outro jeito: ``$4$ palitos para o primeiro
        quadrado e depois $3$ para cada um dos outros $n - 1$'', o que
        dá $4 + 3(n - 1)$. Desenvolva e reduza a expressão dele
        (\cref{prop:g7:literal:reduce}): ele concorda com a questão 3?
  \item Lena conta de uma terceira maneira: ``$2n$ palitos horizontais
        e $n + 1$ verticais''. Escreva a fórmula dela, reduza-a e
        conclua. Depois responda à pergunta de abertura: quantos
        palitos para a figura $100$?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Triângulos, mesas e o caminho de
volta.}
\begin{enumerate}[resume]
  \item Uma fileira de triângulos de palitos (cada triângulo novo se
        apoia no anterior): conte os palitos para $1$, $2$, $3$
        triângulos e depois justifique a fórmula $2n + 1$ para $n$
        triângulos.
  \item De trás para a frente: com exatamente $49$ palitos, quantos
        triângulos a fileira pode ter? (Desfaça a fórmula passo a
        passo.)
  \item Mesas quadradas de banquete acomodam um convidado por lado
        livre. Quando $n$ mesas são encostadas numa fileira comprida,
        explique por que a fórmula de lugares é $2n + 2$ e confira-a
        para $1$, $2$, $3$ mesas.
  \item Uma festa espera $30$ convidados. De quantas mesas uma fileira
        precisa? O bufê, em vez disso, divide as mesmas mesas em
        \emph{duas} fileiras de sete: quantos convidados cabem agora?
        Explique o ganho com a fórmula.
  \item Padrão de pontos: a figura $n$ é um L de pontos, com $n$ pontos
        de altura e $n$ pontos de largura (o canto contado uma vez só).
        Conte os pontos das figuras $1$, $2$, $3$, encontre a fórmula e
        dê o nome dos números que ela produz. (Encaixados uns nos
        outros, esses Ls ladrilham um quadrado --- história continuada
        no \cref{pb:g8:equations:1}.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Confiar, testar, demonstrar.}
\begin{enumerate}[resume]
  \item Um colega afirma que a figura $n$ da Parte I precisa de
        $3(n + 2) - 5$ palitos. Resolva a questão desenvolvendo e
        reduzindo --- sem precisar testar.
  \item ``Elevar ao quadrado deixa os números maiores, então
        $n^2 \geq n$ para todo número $n$.'' Teste a afirmação para
        $n = 2$, $3$, $10$; depois encontre um número em que ela falha.
        O que os três testes bem-sucedidos demonstraram, e o que a
        única falha demonstra (\cref{met:g7:literal:test})?
  \item Marque pontos sobre um círculo e trace \emph{todos} os
        \omterm{def:g6:lines:objects}{segmentos} entre eles; depois conte as regiões dentro do
        círculo. Faça isso para $2$, $3$, $4$ e $5$ pontos (ponha os
        pontos de modo desigual, para que três \omterm{def:g6:lines:objects}{segmentos} nunca se
        cruzem num mesmo ponto interior). Anote as contagens. Que
        fórmula o padrão sussurra?
  \item Agora $6$ pontos, desenhados grandes e desigualmente, contando
        com muito cuidado (há muitas regiõezinhas). Quantas regiões
        você encontra --- e o que aconteceu com a conjectura?
  \item A moral, em duas frases: o que teria sido preciso para
        \emph{demonstrar} a conjectura da duplicação, e por que as
        fórmulas dos palitos das Partes I e II nunca correram esse
        perigo? Comemore com uma previsão demonstrada: quantos palitos
        para a fileira de $2\,026$ quadrados?
\end{enumerate}
\end{problem}
