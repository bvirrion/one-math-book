\chapter{Breuken: vergelijken en optellen}\label{ch:g7:fractions}

In jaar 6 kwamen \omterm{def:g6:fractions:def}{breuken} aan de orde als delen en als precieze \omterm{def:g3:division:remainder}{quotiënten}
(\cref{ch:g6:fractions}). In dit hoofdstuk leer je ermee \emph{rekenen}: gelijke
\omterm{def:g6:fractions:def}{breuken} herkennen, vergelijken, optellen en aftrekken --- voorlopig met
vriendelijke \omterm{def:g4:fractions:def}{noemers}; het algemene geval komt in \cref{ch:g9:fractions}.

\section{Gelijke breuken}

\begin{theorem}[Gelijke breuken]\label{thm:g7:fractions:equal}
Een \omterm{def:g6:fractions:def}{breuk} verandert niet als je haar \omterm{def:g4:fractions:def}{teller} en haar \omterm{def:g4:fractions:def}{noemer} met hetzelfde getal
(niet \omterm{def:g1:counting:zero}{nul}) vermenigvuldigt of erdoor deelt:
\[
\frac{a}{b} = \frac{a \times k}{b \times k} .
\]
\end{theorem}
\admitted

\begin{example}\label{ex:g7:fractions:equal}
$\dfrac{3}{4} = \dfrac{3 \times 5}{4 \times 5} = \dfrac{15}{20}$, en in de andere
richting $\dfrac{42}{30} = \dfrac{42 \div 6}{30 \div 6} = \dfrac{7}{5}$
(vereenvoudigd door $6$).

Om te testen of $\dfrac{8}{12}$ en $\dfrac{10}{15}$ gelijk zijn, vereenvoudig je
beide: $\dfrac{8}{12} = \dfrac{2}{3}$ en $\dfrac{10}{15} = \dfrac{2}{3}$ --- ja,
gelijk.
\end{example}

\section{Breuken vergelijken}

\begin{method}[Twee breuken vergelijken]\label{met:g7:fractions:compare}
\begin{enumerate}
  \item \emph{Dezelfde \omterm{def:g4:fractions:def}{noemer}}: vergelijk de \omterm{def:g4:fractions:def}{tellers}:
        $\frac{5}{9} < \frac{7}{9}$.
  \item \emph{De ene \omterm{def:g4:fractions:def}{noemer} is een \omterm{def:g5:division:multiple}{veelvoud} van de andere}: herschrijf de
        grovere \omterm{def:g6:fractions:def}{breuk} en vergelijk daarna: $\frac{5}{6}$ tegen $\frac{7}{12}$:
        $\frac56 = \frac{10}{12} > \frac{7}{12}$.
  \item \emph{Vergelijk met $1$ of met $\frac12$} als dat kan:
        $\frac{9}{8} > 1 > \frac{7}{9}$ beslist $\frac98$ tegen $\frac79$
        onmiddellijk.
\end{enumerate}
\end{method}

\section{Optellen en aftrekken}

\begin{theorem}[Dezelfde noemer]\label{thm:g7:fractions:add}
\omterm{def:g6:fractions:def}{Breuken} met dezelfde \omterm{def:g4:fractions:def}{noemer} tel je op (of trek je af) door de \omterm{def:g4:fractions:def}{tellers} op te
tellen (of af te trekken):
\[
\frac{a}{d} + \frac{b}{d} = \frac{a + b}{d},
\qquad
\frac{a}{d} - \frac{b}{d} = \frac{a - b}{d} .
\]
\end{theorem}

\begin{proof}[Waarom]
$a$ delen van grootte $\frac1d$ plus $b$ delen van dezelfde grootte maken
$a + b$ delen van die grootte.
\end{proof}

\begin{method}[Verschillende noemers]\label{met:g7:fractions:add}
Als de ene \omterm{def:g4:fractions:def}{noemer} een \omterm{def:g5:division:multiple}{veelvoud} van de andere is:
\begin{enumerate}
  \item herschrijf de \omterm{def:g6:fractions:def}{breuk} met de kleinste \omterm{def:g4:fractions:def}{noemer}, zodat beide de grootste
        \omterm{def:g4:fractions:def}{noemer} hebben (\cref{thm:g7:fractions:equal});
  \item tel de \omterm{def:g4:fractions:def}{tellers} op, of trek ze af;
  \item vereenvoudig de uitkomst als dat kan.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:fractions:add}
Reken $\dfrac{3}{4} + \dfrac{5}{8}$ uit, stap voor stap:
\begin{enumerate}
  \item $8$ is een \omterm{def:g5:division:multiple}{veelvoud} van $4$: herschrijf
        $\dfrac34 = \dfrac{3 \times 2}{4 \times 2} = \dfrac{6}{8}$;
  \item tel op: $\dfrac{6}{8} + \dfrac{5}{8} = \dfrac{11}{8}$;
  \item er valt niets te vereenvoudigen: de uitkomst is $\dfrac{11}{8}$, dus
        $1 + \dfrac38$.
\end{enumerate}
Nog een: $2 - \dfrac{4}{5} = \dfrac{10}{5} - \dfrac{4}{5} = \dfrac{6}{5}$
(schrijf het hele getal eerst op \omterm{def:g4:fractions:def}{noemer} $5$).
\end{example}

\begin{example}[Een klassieke valstrik]\label{ex:g7:fractions:trap}
$\dfrac12 + \dfrac13$ is \emph{niet} $\dfrac{2}{5}$! \omterm{def:g4:fractions:def}{Tellers} en \omterm{def:g4:fractions:def}{noemers} apart
optellen is fout --- de uitkomst $\frac25$ zou kleiner zijn dan $\frac12$, en dat
is onzin als je iets positiefs \emph{bij} $\frac12$ optelt. (De juiste \omterm{def:g1:addition:def}{som},
$\frac56$, heeft de gemeenschappelijke \omterm{def:g4:fractions:def}{noemer} $6$ nodig; de algemene methode staat
in \cref{ch:g9:fractions}.)
\end{example}

\section{Een breuk met een getal vermenigvuldigen}

\begin{proposition}[Breuk keer een getal]\label{prop:g7:fractions:mult}
Voor een getal $k$ geldt:
\[
k \times \frac{a}{b} = \frac{k \times a}{b} .
\]
$\frac ab$ \emph{van} een hoeveelheid nemen betekent met $\frac ab$
vermenigvuldigen.
\end{proposition}
\admitted

\begin{example}\label{ex:g7:fractions:mult}
$6 \times \dfrac{2}{3} = \dfrac{12}{3} = 4$: zes keer twee derde is vier gehelen.
En ``$\frac34$ van $60$ euro'' is $\frac34 \times 60 = \frac{180}{4} = 45$ euro
--- hetzelfde antwoord als door $4$ delen en dan met $3$ vermenigvuldigen
(\cref{met:g6:fractions:ofquantity}).
\end{example}

\section{Opgave: de breuken die in een muziekmaat verstopt zitten}

\begin{problem}[{Weekendopgave --- notenwaarden zijn breuken die precies moeten
uitkomen, het tellen van ritmes bergt een beroemde rij, en pariteit regeert de
Balkanmaat}]
\label{pb:g7:fractions:1}
Muziek wordt in \omterm{def:g6:fractions:def}{breuken} geschreven. Een \emph{hele noot} duurt $1$ maat (in de
gewone vierkwartsmaat); een \emph{halve noot} duurt $\frac12$; dan volgen de
\emph{kwartnoot} $\frac14$, de \emph{achtste noot} $\frac18$ en de
\emph{zestiende noot} $\frac{1}{16}$. Eén onwrikbare wet: \emph{de duren binnen
één maat moeten precies het totaal van die maat opleveren} --- $1$ in de
vierkwartsmaat. Elke vraag hieronder is de rekenkunde van dit hoofdstuk
(\cref{thm:g7:fractions:add}, \cref{met:g7:fractions:add}) op muziek gezet; en
onderweg ontmoet je een telrij die Indiase geleerden duizend jaar geleden hebben
ontdekt.

\medskip
\noindent\textbf{Deel I --- Noten die moeten uitkomen.}
\begin{enumerate}
  \item Ga na dat elk van deze maten in de vierkwartsmaat geldig is: (a) halve
        $+$ \omterm{def:g3:division:half}{kwart} $+$ \omterm{def:g3:division:half}{kwart}; (b) \omterm{def:g3:division:half}{kwart} $+$ achtste $+$ achtste $+$ halve.
  \item Een kopiist schreef: halve $+$ \omterm{def:g3:division:half}{kwart} $+$ achtste. Welke \omterm{def:g6:fractions:def}{breuk} van de
        maat staat er, en welke enkele noot maakt haar vol?
  \item Hoeveel zestiende noten vullen één hele maat? En hoeveel zestienden maken
        één achtste noot?
  \item Een \emph{punt} achter een noot voegt de \omterm{def:g3:division:half}{helft} van de notenwaarde toe: een
        gepunteerde halve is $\frac12 + \frac14$, een gepunteerde \omterm{def:g3:division:half}{kwart} is
        $\frac14 + \frac18$. Reken beide waarden uit.
  \item Een wals wordt in de driekwartsmaat geschreven: elke maat moet samen
        $\frac34$ zijn. Ga na dat één gepunteerde halve een walsmaat vult, en
        schrijf twee andere geldige walsmaten op met alleen \omterm{def:g3:division:half}{kwarten} en achtsten.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Het atelier van de kopiist.}
\begin{enumerate}[resume]
  \item Een vierkwartsmaat bevat: gepunteerde \omterm{def:g3:division:half}{kwart} $+$ achtste $+$ \omterm{def:g3:division:half}{kwart}. Wat
        ontbreekt er?
  \item Wat duurt langer, een gepunteerde achtste of een kwartnoot? Vergelijk via
        zestienden.
  \item Een \emph{bindboog} plakt duren tot één lange klank aan elkaar: hoe lang
        duurt een halve noot die aan een achtste gebonden is? Hoeveel zestienden
        is dat?
  \item De noten van een maat maken samen $\frac{11}{16}$; de \omterm{def:g3:division:remainder}{rest} is stilte (een
        \emph{rust}). Hoe lang duurt die rust?
  \item Een drummer vult één kwartnoot ($\frac{4}{16}$ van de maat) met alleen
        achtsten ($\frac{2}{16}$) en zestienden ($\frac{1}{16}$), waarbij de
        volgorde meetelt: bijvoorbeeld achtste--zestiende--zestiende, of
        zestiende--achtste--zestiende. Ga na dat beide voorbeelden geldig zijn en
        noem daarna \emph{alle} mogelijke ritmes. Hoeveel zijn er?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Een duizend jaar oude rij, en een \omterm{def:g3:numbers:evenodd}{oneven} maat.}
\begin{enumerate}[resume]
  \item Laten we de ritmes van de drummer voor langere noten tellen. Elk ritme
        eindigt op een zestiende of op een achtste; leg uit waarom dat de regel
        geeft: (ritmes die $n$ zestienden vullen) $=$ (ritmes die $n - 1$ vullen)
        $+$ (ritmes die $n - 2$ vullen). Bouw, vanaf $1$ ritme voor één zestiende
        en $2$ voor twee, de tabel op tot $n = 8$: hoeveel ritmes vullen een halve
        noot?
  \item De getallen die je vond --- $1, 2, 3, 5, 8, 13, 21, 34$ --- werden rond
        het jaar 1150 gepubliceerd door de Indiase geleerde Hemachandra, die
        precies zulke ritmes telde, twee generaties voordat Fibonacci ze in Italië
        opschreef. Formuleer hun patroon in één zin en controleer het op de laatste
        drie getallen van je tabel.
  \item De ``swing''-stijl vervangt twee gelijke achtsten door een lang--kort
        paar: $\frac16$ en dan $\frac{1}{12}$. Ga na dat de ruil geldig is
        (dezelfde totale duur), met de gemeenschappelijke \omterm{def:g4:fractions:def}{noemer} $12$.
  \item Een \emph{triool} perst drie gelijke noten in één kwartnoot. Hoe lang
        duurt elke noot (controleer je antwoord door het met $3$ te
        vermenigvuldigen, \cref{prop:g7:fractions:mult})? Dezelfde vraag voor drie
        gelijke noten die een halve noot vullen.
  \item Balkandansmuziek gebruikt de zevenachtstemaat: elke maat is samen
        $\frac78$. Ga na dat gepunteerde \omterm{def:g3:division:half}{kwart} $+$ \omterm{def:g3:division:half}{kwart} $+$ \omterm{def:g3:division:half}{kwart} haar vult. Leg
        daarna uit waarom \emph{geen} enkele combinatie van alleen halve en
        kwartnoten, hoeveel je er ook neemt, ooit een zevenachtstemaat kan vullen.
        (Tel in achtsten en denk aan \omterm{def:g3:numbers:evenodd}{even} en \omterm{def:g3:numbers:evenodd}{oneven}.)
\end{enumerate}
\end{problem}
