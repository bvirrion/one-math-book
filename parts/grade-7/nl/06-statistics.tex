\chapter{Gegevens ordenen}\label{ch:g7:stats}

Voordat je iets kunt uitrekenen, moeten gegevens verzameld, geteld en weergegeven
worden. Dit hoofdstuk gaat over het omzetten van een rommelige lijst waarnemingen
in een heldere tabel of tekening: aantallen, \omterm{def:g7:stats:frequency}{frequenties}, gegroepeerde gegevens,
staafdiagrammen en cirkeldiagrammen. Gegevens samenvatten in één getal (gemiddelde,
mediaan) is het onderwerp van \cref{ch:g9:statproba}.

\section{Aantallen en frequenties}

\begin{definition}[Aantal en frequentie]\label{def:g7:stats:frequency}
Voor elke mogelijke waarde in een onderzoek is het \emph{aantal}\index{aantal} het
aantal keren dat ze voorkomt; haar \emph{frequentie}\index{frequentie} is dat
aantal gedeeld door het totale aantal waarnemingen:
\[
\text{frequentie} = \frac{\text{aantal}}{\text{totaal}} .
\]
Frequenties kun je als \omterm{def:g6:fractions:def}{breuk}, als \omterm{def:g6:decimals:places}{decimaal getal} of als percentage schrijven, en
ze zijn samen altijd $1$ (dus $100\,\%$).
\end{definition}

\begin{example}\label{ex:g7:stats:frequency}
De favoriete sport van de $25$ leerlingen van een klas, van de ruwe lijst naar de
tabel:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cccc|c}
sport & voetbal & zwemmen & tennis & judo & totaal \\
\hline
\omterm{def:g7:stats:frequency}{aantal} & $10$ & $6$ & $5$ & $4$ & $25$ \\
\omterm{def:g7:stats:frequency}{frequentie} & $0.40$ & $0.24$ & $0.20$ & $0.16$ & $1$ \\
\end{tabular}
\end{center}
Voor voetbal: $\frac{10}{25} = \frac{40}{100} = 0.40 = 40\,\%$. De rij met
\omterm{def:g7:stats:frequency}{frequenties} moet samen $1$ zijn --- een permanente, gratis controle.
\end{example}

\begin{remark}[Waarom frequenties?]
Aantallen kun je niet vergelijken tussen groepen van verschillende grootte: $10$
voetbalfans in een klas van $25$ (dat is $40\,\%$) laten meer enthousiasme zien dan
$12$ in een jaarlaag van $120$ ($10\,\%$). \omterm{def:g7:stats:frequency}{Frequenties} zetten alles op dezelfde
schaal.
\end{remark}

\section{Gegevens in klassen groeperen}

\begin{example}[Klassen]\label{ex:g7:stats:classes}
De lengtes (in cm) van $20$ leerlingen lopen van $148$ tot $172$. Elke lengte
apart opsommen zou een tabel met bijna even veel kolommen als leerlingen geven;
groepeer ze in plaats daarvan in \emph{klassen}:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|ccc|c}
lengte (cm) & $\intco{145}{155}$ & $\intco{155}{165}$ &
$\intco{165}{175}$ & totaal \\
\hline
\omterm{def:g7:stats:frequency}{aantal} & $5$ & $9$ & $6$ & $20$ \\
\omterm{def:g7:stats:frequency}{frequentie} & $0.25$ & $0.45$ & $0.30$ & $1$ \\
\end{tabular}
\end{center}
Elke lengte wordt in precies één klasse geteld ($155$ hoort bij
$\intco{155}{165}$, niet bij beide --- daarom zijn de intervallen half open).
Groeperen kost detail, maar levert leesbaarheid op.
\end{example}

\section{Diagrammen}

\begin{method}[Een diagram kiezen en tekenen]\label{met:g7:stats:charts}
\begin{enumerate}
  \item \emph{Staafdiagram}: één staaf per waarde of klasse, met een hoogte die
        \omterm{def:g6:propdata:def}{evenredig} is met het \omterm{def:g7:stats:frequency}{aantal} (of met de \omterm{def:g7:stats:frequency}{frequentie}). Goed om waarden te
        vergelijken.
  \item \emph{Cirkeldiagram}: één sector per waarde, met een hoek die \omterm{def:g6:propdata:def}{evenredig} is
        met de \omterm{def:g7:stats:frequency}{frequentie} --- de volle cirkel ($360^\circ$) stelt het totaal voor,
        dus
        \[
        \text{hoek} = \text{frequentie} \times 360^\circ .
        \]
        Goed om delen van een geheel te laten zien.
  \item In beide gevallen: een titel, labels, en streepjes of de percentages op het
        diagram zelf.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:stats:pie}
De hoeken voor de sporten van \cref{ex:g7:stats:frequency}: voetbal
$0.40 \times 360 = 144^\circ$; zwemmen $0.24 \times 360 = 86.4^\circ$; tennis
$72^\circ$; judo $57.6^\circ$. Controle: $144 + 86.4 + 72 + 57.6 = 360$.
\end{example}

\begin{example}[Kritisch lezen]\label{ex:g7:stats:critical}
Een diagram waarvan de verticale as bij $9$ begint in plaats van bij $0$, laat een
staaf van $10$ tien keer zo hoog lijken als een staaf van $9.1$ --- visueel
misleidend, ook al zijn de getallen eerlijk. Eerste reflex bij elk diagram:
controleer waar de as begint en hoeveel één streepje waard is
(\cref{met:g6:propdata:read}).
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g7:stats:1}
Hier zijn de cijfers van $20$ leerlingen: $12$, $15$, $9$, $12$, $14$, $9$, $12$,
$15$, $11$, $12$, $9$, $14$, $12$, $11$, $15$, $9$, $12$, $14$, $11$, $12$. Bouw de
tabel met de aantallen (één kolom per cijfer).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:2}
Vul de tabel van \cref{exo:g7:stats:1} aan met een rij \omterm{def:g7:stats:frequency}{frequenties} (als \omterm{def:g6:fractions:def}{breuken} van
$20$, en daarna als percentages). Controleer de \omterm{def:g1:addition:def}{som}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:3}
In een onderzoek onder $50$ gezinnen hebben $30$ gezinnen één auto. Wat is de
\omterm{def:g7:stats:frequency}{frequentie} van ``één auto'', als \omterm{def:g6:decimals:places}{decimaal getal} en als percentage? In een ander
onderzoek hebben $45$ van de $90$ gezinnen één auto: welk onderzoek toont het
hoogste aandeel?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:4}
De \omterm{def:g2:measure:units}{massa}'s (kg) van $15$ honden: $8$, $23$, $31$, $12$, $9$, $27$, $35$, $14$,
$22$, $18$, $29$, $11$, $25$, $33$, $16$. Groepeer ze in de klassen
$\intco{5}{15}$, $\intco{15}{25}$, $\intco{25}{35}$, $\intco{35}{45}$ en geef de
aantallen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:5}
Teken het staafdiagram van de tabel:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|cccc}
huisdieren & $0$ & $1$ & $2$ & $3$ \\
\hline
gezinnen & $8$ & $12$ & $6$ & $4$ \\
\end{tabular}
\end{center}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:6}
Reken voor de tabel van \cref{exo:g7:stats:5} de \omterm{def:g7:stats:frequency}{frequenties} en de hoeken voor een
cirkeldiagram van elke waarde uit (controle: samen $360^\circ$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:7}
Een cirkeldiagram over favoriete seizoenen toont: zomer $162^\circ$, lente
$90^\circ$, herfst $54^\circ$, winter $54^\circ$. Welk percentage koos elk seizoen?
Als er $60$ mensen geantwoord hebben, hoeveel kozen dan de zomer?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:8}
In klas A komen $12$ van de $30$ leerlingen lopend naar school; in klas B $14$ van
de $40$. Welke klas heeft het hoogste aandeel lopers? Verantwoord het met
\omterm{def:g7:stats:frequency}{frequenties}, niet met aantallen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:9}
Een tijdschrift drukt een staafdiagram van de maandverkoop af: $9\,800$, dan
$10\,000$, dan $10\,100$, met een verticale as die bij $9\,700$ begint. Beschrijf
wat de lezer ziet, en wat een eerlijke as vanaf $0$ in plaats daarvan zou laten
zien.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:10}
Een frequentietabel heeft drie waarden met \omterm{def:g7:stats:frequency}{frequenties} $0.35$, $0.4$ en $f$. Zoek
$f$. Als het totale \omterm{def:g7:stats:frequency}{aantal} $80$ is, geef dan de drie aantallen.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:stats:11}
Op een school is $55\,\%$ van de leerlingen meisje. Van de meisjes eet $40\,\%$ in
de kantine; van de jongens $60\,\%$. Hoeveel van de $400$ leerlingen eten in de
kantine? (Reken de vier groepsgroottes stap voor stap uit.) Welk percentage van de
school is dat?
\end{exercise}

\section{Opgave: hoe gegevens liegen --- en hoe je ze betrapt}

\begin{problem}[{Weekendopgave --- aantallen tegen frequenties, de opiniepeiling
die een land om de tuin leidde, en de paradox van de twee ziekenhuizen}]
\label{pb:g7:stats:1}
Getallen liegen niet, maar je kunt ze wel laten misleiden: een enorm \omterm{def:g7:stats:frequency}{aantal} kan een
kleine \omterm{def:g7:stats:frequency}{frequentie} verbergen, een gigantisch onderzoek kan waardeloos zijn, en ---
het vreemdst van al --- een ziekenhuis kan zijn concurrent in \emph{elke} categorie
patiënten verslaan en toch op de totaalcijfers verliezen. Alle drie de valstrikken
gaan in deze opgave af, met niets anders in handen dan de aantallen en \omterm{def:g7:stats:frequency}{frequenties}
van dit hoofdstuk (\cref{def:g7:stats:frequency}).

\medskip
\noindent\textbf{Deel I --- Aantallen zijn geen \omterm{def:g7:stats:frequency}{frequenties}.}
\begin{enumerate}
  \item School A recyclet $120$ van haar $400$ drinkpakjes; school B $90$ van haar
        $250$. Welke school heeft het grootste \emph{\omterm{def:g7:stats:frequency}{aantal}} gerecyclede pakjes? En
        de grootste \emph{\omterm{def:g7:stats:frequency}{frequentie}}? Welke school verdient de recycleprijs?
  \item Een nieuw lesrooster wordt onderzocht: $18$ van de $30$ leraren vinden het
        goed, en $210$ van de $600$ leerlingen. Reken de twee \omterm{def:g7:stats:frequency}{frequenties} uit, en
        daarna de \omterm{def:g7:stats:frequency}{frequentie} onder alle $630$ mensen samen. Waarom ligt het
        gecombineerde cijfer zo dicht bij dat van de leerlingen?
  \item Een gepubliceerd cirkeldiagram toont sectoren met de labels $45\,\%$,
        $30\,\%$, $20\,\%$ en $10\,\%$. Hoe weet je zonder verdere informatie dat
        er een fout is gemaakt?
  \item Een frequentietabel over $40$ waarnemingen is half uitgewist: aantallen
        $14$, $10$, $?$, $?$ en \omterm{def:g7:stats:frequency}{frequenties} $0.35$, $0.25$, $0.15$, $?$. Herstel de
        ontbrekende gegevens.
  \item Een onderzoek eindigt met de \omterm{def:g7:stats:frequency}{frequenties} $0.40$, $0.35$ en $0.25$. Reken de
        drie hoeken voor het cirkeldiagram uit
        (\cref{met:g7:stats:charts}) en ga na dat ze de cirkel sluiten.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De peiling die een land om de tuin leidde.}
In 1936 stuurde een Amerikaans tijdschrift tien miljoen stembiljetten naar adressen
uit telefoongidsen en autoregistraties, ontving $2.4$ miljoen antwoorden, en
voorspelde een verpletterende overwinning voor kandidaat Landon. Een jonge
statisticus, George Gallup, ondervroeg maar ongeveer vijftigduizend mensen ---
gekozen om op de hele bevolking te lijken --- en voorspelde het tegendeel.
Roosevelt won met een enorme meerderheid.
\begin{enumerate}[resume]
  \item Welke \omterm{def:g7:stats:frequency}{frequentie} van de verstuurde biljetten kwam terug?
  \item In 1936 waren telefoons en auto's luxe. Leg in één of twee zinnen uit waarom
        de steekproef van het tijdschrift, hoe enorm ook, gedoemd was --- en welke
        les van vraag 1 zij op nationale schaal herhaalt.
  \item Een speelgoedmodel. Een dorp heeft $1\,000$ rijke kiezers, waarvan $70\,\%$
        L steunt, en $9\,000$ minder rijke kiezers, waarvan $30\,\%$ L steunt. Reken
        de echte steun voor L in het dorp uit. Een ``telefoongidspeiling'' bereikt
        $500$ rijke en $500$ minder rijke kiezers: welke steun voorspelt zij?
  \item De stille valstrik: een bedrijf heeft $200$ ontevreden en $800$ tevreden
        klanten. Het ondervraagt iedereen; $40\,\%$ van de ontevreden klanten
        antwoordt, maar slechts $10\,\%$ van de tevreden. Reken het \omterm{def:g7:stats:frequency}{aantal}
        antwoorden van elke soort uit, en de \omterm{def:g7:stats:frequency}{frequentie} van ontevredenheid
        \emph{onder de antwoorden}. Vergelijk met de echte \omterm{def:g7:stats:frequency}{frequentie}.
  \item Repareer de peiling van vraag 8: houd $1\,000$ gesprekken, maar verdeel ze
        volgens de echte verhoudingen van het dorp. Wat voorspelt de gerepareerde
        peiling?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De paradox van de twee ziekenhuizen.}
Twee ziekenhuizen publiceren hun genezingsaantallen, opgesplitst naar ernst:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cc}
 & lichte gevallen & ernstige gevallen \\
\hline
ziekenhuis A & $90$ genezen van $100$ & $280$ genezen van $400$ \\
ziekenhuis B & $340$ genezen van $400$ & $65$ genezen van $100$ \\
\end{tabular}
\end{center}
\begin{enumerate}[resume]
  \item Reken de genezingsfrequentie van ziekenhuis A uit voor de lichte gevallen,
        voor de ernstige gevallen, en in het totaal (alle $500$ patiënten).
  \item Doe dezelfde drie berekeningen voor ziekenhuis B. Welk ziekenhuis wint bij
        de lichte gevallen? En bij de ernstige?
  \item Vergelijk de twee totaalfrequenties. Formuleer duidelijk wat er vreemds
        gebeurd is.
  \item Verklaar de truc: vergelijk de \emph{samenstelling} van de patiënten van de
        twee ziekenhuizen, en zeg waarom de totaalfrequentie beide
        categoriefrequenties kan verraden. Welk ziekenhuis zou je kiezen met een
        ernstig geval?
  \item Een krantenkop luidt: ``Ziekenhuis B geneest beter: $81\,\%$ tegen
        $74\,\%$.'' Schrijf de brief van twee zinnen aan de redactie die deze hele
        opgave je heeft leren schrijven --- één zin over wat de kop negeert, en één
        over de algemene moraal (vergelijk gelijk met gelijk, en let op de
        gewichten).
\end{enumerate}
\end{problem}
