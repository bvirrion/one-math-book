\chapter{Negatieve getallen}\label{ch:g7:negatives}

Temperaturen onder \omterm{def:g1:counting:zero}{nul}, verdiepingen onder de grond, geld dat je schuldig bent:
heel wat grootheden gaan onder $0$. Negatieve getallen breiden de getallenlijn
uit naar links van \omterm{def:g1:counting:zero}{nul}. Dit hoofdstuk leert je ze vergelijken en ze optellen en
aftrekken --- hun vermenigvuldiging wacht tot \cref{ch:g8:negprod}.

\section{Positieve en negatieve getallen}

\begin{definition}[Getallen met een teken]\label{def:g7:negatives:def}
Een \emph{getal met een teken}\index{getal met een teken} bestaat uit een teken
($+$ of $-$) en een \emph{afstand tot \omterm{def:g1:counting:zero}{nul}}: $+3$ en $-3$ liggen beide op afstand
$3$ van $0$, aan weerszijden. Positieve getallen schrijf je meestal zonder hun
teken: $+3 = 3$. De getallen $+3$ en $-3$ zijn elkaars
\emph{tegengestelde}\index{tegengestelde}.
\end{definition}

\begin{omfigure}
\begin{tikzpicture}[scale=1.15]
  \draw[-Stealth] (-5.6,0) -- (5.6,0);
  \foreach \x in {-5,...,5} \draw (\x,-0.08) -- (\x,0.08);
  \foreach \x in {-5,...,5} \node[below, font=\small] at (\x,-0.1) {$\x$};
  \fill[omThm] (-3,0) circle (2pt);
  \fill[omDef] (3,0) circle (2pt);
  \draw[Stealth-Stealth, omExo] (-3,0.35) -- (0,0.35)
    node[midway, above, font=\small] {$3$};
  \draw[Stealth-Stealth, omExo] (0,0.35) -- (3,0.35)
    node[midway, above, font=\small] {$3$};
\end{tikzpicture}

{\small De getallenlijn, naar links uitgebreid: $-3$ en $+3$ zijn \omterm{def:g7:negatives:def}{tegengestelde}
getallen, op dezelfde afstand van $0$.}
\end{omfigure}

\begin{method}[Getallen met een teken vergelijken]\label{met:g7:negatives:compare}
Op de getallenlijn betekent kleiner: verder naar links. In de praktijk:
\begin{enumerate}
  \item een negatief getal is altijd kleiner dan een positief: $-7 < 2$;
  \item tussen twee positieve getallen geldt de gewone orde;
  \item tussen twee negatieve getallen is dat met de \emph{grotere} afstand tot
        \omterm{def:g1:counting:zero}{nul} het \emph{kleinste}: $-7 < -2$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:negatives:compare}
Zet op volgorde, van klein naar groot: $3$; $-5$; $0$; $-1.5$; $2.8$. Eerst de
negatieve, met de meest negatieve vooraan:
\[
-5 < -1.5 < 0 < 2.8 < 3 .
\]
\end{example}

\section{Getallen met een teken optellen}

Zie een \omterm{def:g7:negatives:def}{getal met een teken} als een beweging langs de \omterm{def:g6:lines:objects}{lijn}: $+5$ betekent ``loop
$5$ stappen naar rechts'', $-3$ betekent ``loop $3$ stappen naar links''.
Optellen is de wandelingen aan elkaar rijgen.

\begin{method}[Twee getallen met een teken optellen]\label{met:g7:negatives:add}
\begin{enumerate}
  \item \emph{Hetzelfde teken}: tel de afstanden op en houd het gemeenschappelijke
        teken: $(-3) + (-4) = -7$.
  \item \emph{\omterm{def:g7:negatives:def}{Tegengestelde} tekens}: trek de kleinste afstand van de grootste af
        en neem het teken van de grootste: $(-7) + (+2) = -5$ en
        $(+7) + (-2) = +5$.
  \item \omterm{def:g7:negatives:def}{Tegengestelden} heffen elkaar op: $(+4) + (-4) = 0$.
\end{enumerate}
\end{method}

  \foreach \x in {-5,...,4} \draw (\x,-0.07) -- (\x,0.07);

  \node[omDef, font=\small] at (-2.5,1.0) {$+2$};

\end{tikzpicture}

{\small Optellen als lopen: vanaf $0$ brengen de wandeling $-3.5$ en daarna de
wandeling $+2$ je op $(-3.5) + 2 = -1.5$.}
\end{omfigure}

\begin{example}\label{ex:g7:negatives:add}
Stap voor stap, handig gegroepeerd:
\begin{align*}
(-8) + (+3) &= -5 && \text{(tegengestelde tekens: } 8 - 3 = 5,
\text{ teken van } 8)\\
(-2.5) + (-1.5) &= -4 && \text{(zelfde teken: afstanden optellen)}\\
(+6) + (-9) + (+3) &= (-3) + (+3) = 0 .
\end{align*}
\end{example}

\section{Getallen met een teken aftrekken}

\begin{theorem}[Regel voor het aftrekken]\label{thm:g7:negatives:sub}
Een \omterm{def:g7:negatives:def}{getal met een teken} aftrekken is hetzelfde als \emph{zijn \omterm{def:g7:negatives:def}{tegengestelde}
optellen}:
\[
a - b = a + (-b) .
\]
\end{theorem}

\begin{proof}[Waarom het werkt]
$a - b$ is het getal dat, opgeteld bij $b$, $a$ geeft. Ga na dat $a + (-b)$ dat
doet: $\bigl(a + (-b)\bigr) + b = a + \bigl((-b) + b\bigr) = a + 0 = a$.
\end{proof}

\begin{example}\label{ex:g7:negatives:sub}
Elke aftrekking wordt een optelling, en dan geldt
\cref{met:g7:negatives:add}:
\begin{align*}
5 - 9 &= 5 + (-9) = -4, \\
(-3) - (+4) &= (-3) + (-4) = -7, \\
(-2) - (-6) &= (-2) + (+6) = +4 .
\end{align*}
De laatste regel is de beroemde: \emph{een negatief getal aftrekken is
optellen}. Een schuld die verdwijnt, maakt je rijker!
\end{example}

\begin{example}[Temperatuurverschillen]\label{ex:g7:negatives:temperature}
Op één dag ging de temperatuur van $-6$ graden naar $+9$ graden. De stijging is
het \omterm{ex:g1:subtraction:difference}{verschil}
\[
9 - (-6) = 9 + 6 = 15 \text{ graden}.
\]
Op de getallenlijn: van $-6$ tot $0$ is $6$ graden, en van $0$ tot $9$ nog $9$
erbij.
\end{example}

\section{Coördinaten in het hele vlak}

\begin{definition}[Coördinaten met tekens]\label{def:g7:negatives:coords}
Met twee loodrechte assen met streepjes, die elkaar in de oorsprong $O$ snijden,
krijgt elk punt van het vlak twee coördinaten $(x, y)$ met een teken: een
negatieve $x$ betekent links van de verticale as, een negatieve $y$ betekent
onder de horizontale as.
\end{definition}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g7:negatives:1}
Geef het \omterm{def:g7:negatives:def}{tegengestelde} van: $7$; \ $-3.5$; \ $0$; \ $+12$. Vul daarna aan:
``twee getallen zijn tegengesteld als hun \omterm{def:g1:addition:def}{som} gelijk is aan~?''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:2}
Neem over en vul in met $<$ of $>$:
\[
-4 \;?\; 1, \qquad
-4 \;?\; -7, \qquad
-2.5 \;?\; -2.4, \qquad
0 \;?\; -0.1 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:3}
Zet op volgorde, van klein naar groot: $-3$; \ $2.5$; \ $-0.5$; \ $-3.1$;
\ $0$; \ $1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:4}
Reken uit:
\[
(-5) + (-9), \qquad
(-12) + (+7), \qquad
(+3.5) + (-1.5), \qquad
(-6) + (+6).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:5}
Maak er optellingen van en reken daarna uit:
\[
4 - 11, \qquad
(-5) - (+3), \qquad
(-7) - (-10), \qquad
0 - (-8).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:6}
Reken uit door slim te groeperen (eerst de \omterm{def:g7:negatives:def}{tegengestelden}):
\[
(+7) + (-12) + (+3) + (-8), \qquad
(-5.5) + (+9) + (-4.5) + (+1).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:7}
Vanmorgen was het $-3$ graden; in de loop van de dag steeg de temperatuur $8$
graden, en 's nachts daalde ze $12$ graden. Schrijf één berekening op die de
eindtemperatuur geeft, en reken die uit.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:8}
Zet in een assenstelsel: $A(2, 3)$; $B(-3, 1)$; $C(-2, -2)$; $D(3, -1)$;
$E(0, -3)$. Welk punt ligt op een as?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:9}
Reken het \omterm{ex:g1:subtraction:difference}{verschil} tussen de hoogste en de laagste temperatuur uit: Moskou in
januari, maximaal $-4$ graden, minimaal $-13$ graden; Helsinki, maximaal $2$,
minimaal $-9$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:10}
Een duikster bevindt zich op hoogte $-12$ m (twaalf meter onder het oppervlak).
Ze stijgt $7$ m en daalt daarna $4$ m. Op welke hoogte is ze dan? Hoeveel moet ze
nog stijgen om het oppervlak te bereiken?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:11}
Zet $A(-2, 1)$, $B(2, 1)$ en $C(2, -2)$ in een assenstelsel. Zoek de coördinaten
van het punt $D$ waarvoor $ABCD$ een rechthoek is, en reken zijn \omterm{def:g6:measure:perimeter}{omtrek} uit (tel
de rastervakjes voor de lengtes van de zijden).
\end{exercise}

\section{Opgave: het jaar nul dat nooit bestond}

\begin{problem}[{Weekendopgave --- getallen met een teken op de tijdlijn van de
geschiedenis: tijdsduren over de grens van de jaartelling, de truc van de
astronomen, en het beginsel van de nettoverandering}]
\label{pb:g7:negatives:1}
De getallenlijn van dit hoofdstuk heeft een beroemde tweeling in de echte
wereld: de tijdlijn van de geschiedenis, met de jaren v.Chr. naar links en de
jaren n.Chr. naar rechts. Maar de \omterm{def:g6:lines:objects}{lijn} van de historici bergt een valstrik die al
veel berekeningen heeft verpest: \emph{er is geen jaar \omterm{def:g1:counting:zero}{nul}} --- op het jaar $1$
v.Chr. volgt onmiddellijk het jaar $1$ n.Chr. In deze opgave reken je met
temperaturen, liften en bankrekeningen, en repareer je daarna de kapotte \omterm{def:g1:counting:zero}{nul} van
de geschiedenis zoals astronomen dat doen: met getallen met een teken.

\medskip
\noindent\textbf{Deel I --- Opwarmers, en een eerste valstrik.}
\begin{enumerate}
  \item De top van de Mont Blanc ligt op hoogte $+4\,808$ m; de oever van de Dode
        Zee op $-430$ m. Reken het hoogteverschil tussen die twee uit.
  \item Op een bankrekening staat $-45$ euro (roodstand). De eigenaar stort $120$
        euro, betaalt een rekening van $60$ euro en stort daarna $15$ euro.
        Schrijf het saldo als één rij optellingen en reken het uit. Welke
        eenmalige storting zou de oorspronkelijke $-45$ precies op $0$ hebben
        gebracht?
  \item Julius Caesar werd geboren in $100$ v.Chr. en stierf in $44$ v.Chr. Reken
        met de jaren v.Chr. als negatieve getallen ($100$ v.Chr. als $-100$, $44$
        v.Chr. als $-44$) het \omterm{ex:g1:subtraction:difference}{verschil} $(-44) - (-100)$ uit: hoeveel jaar heeft
        Caesar geleefd?
  \item Keizer Augustus werd geboren in $63$ v.Chr. en stierf in $14$ n.Chr.
        Hetzelfde recept geeft $14 - (-63) = 77$ jaar --- maar historici houden
        vol dat hij op zijn $76$ste stierf. Hun bezwaar: de telling $77$ loopt
        door een jaar dat nooit heeft bestaan. Welk jaar?
  \item Astronomen repareren de tijdlijn door te herbenoemen: de jaren n.Chr.
        houden hun nummer ($+14$ blijft $+14$), maar $1$ v.Chr. wordt $0$, $2$
        v.Chr. wordt $-1$, en in het algemeen wordt het jaar $n$ v.Chr.
        $-(n - 1)$. Zet $63$ v.Chr. om en maak de berekening van vraag 4 opnieuw
        met de getallen van de astronomen. Voldoet het antwoord nu aan de
        historici?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Rekenen over de grens van de jaartelling.}
Gebruik van nu af aan bij elke berekening de herbenoeming van de astronomen, en
zet aan het einde terug om naar v.Chr./n.Chr. (een jaar met label $0$ of met een
negatief label $-m$ is het jaar $m + 1$ v.Chr. van de historici).
\begin{enumerate}[resume]
  \item Zet om naar de getallen van de astronomen: $1$ v.Chr.; $10$ v.Chr.; $476$
        n.Chr.; en $753$ v.Chr., de legendarische stichting van Rome.
  \item Hoeveel jaar liggen er tussen de stichting van Rome ($753$ v.Chr.) en de
        val van het West-Romeinse Rijk ($476$ n.Chr.)?
  \item Een komeet komt elke $76$ jaar terug en verscheen in $12$ v.Chr. Geef de
        jaren van de historici voor haar volgende drie verschijningen, en voor de
        verschijning net vóór $12$ v.Chr.
  \item Hoeveel jaar liggen er tussen $5$ v.Chr. en $5$ n.Chr.? (Het verleidelijke
        antwoord is $10$; de tijdlijn zegt iets anders.)
  \item Een kind dat in $3$ v.Chr. werd geboren, werd tien jaar in welk jaar
        (v.Chr. of n.Chr.)?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Het beginsel van de nettoverandering.}
\begin{enumerate}[resume]
  \item De \emph{afstand} tussen twee getallen op de \omterm{def:g6:lines:objects}{lijn} is het grootste min het
        kleinste --- altijd positief. Reken de afstand tussen $-7$ en $-2$ uit,
        en daarna die tussen $-3.5$ en $4.5$.
  \item Een lift begint op verdieping $0$ en maakt de bewegingen $+3$, $-5$,
        $+2$, $-1$, $+4$. Reken de eindverdieping uit door slim te groeperen
        (\cref{met:g7:negatives:add}, eerst de \omterm{def:g7:negatives:def}{tegengestelden} waar dat kan). Zou
        de lift op een andere verdieping uitkomen als je de bewegingen in een
        andere volgorde uitvoert? Waarom?
  \item Dag na dag verandert de temperatuur in een week met $+2$, $-4$, $+1$,
        $0$, $-3$, $+5$, $-2$ graden. Reken de \emph{nettoverandering} uit.
        Begon de week op $-2$ graden, waar eindigde ze dan? Formuleer het
        beginsel: eindwaarde $=$ beginwaarde $+$ (\omterm{def:g1:addition:def}{som} van alle veranderingen).
  \item Een wandelaarster komt precies op haar starthoogte terug. Wat moet de \omterm{def:g1:addition:def}{som}
        van al haar hoogteveranderingen zijn, en waarom? Van haar zes opgetekende
        veranderingen zijn er vijf $+4$, $-7$, $+2$, $+3$, $-5$ (in honderden
        meters): zoek de zesde.
  \item De finale, terug in Rome. Een historicus rekent: ``de Republiek liep van
        $509$ v.Chr. tot $27$ v.Chr., en daarna het Keizerrijk tot $476$ n.Chr.;
        alles samen, van $509$ v.Chr. tot $476$ n.Chr., dus $509 + 476 = 985$
        jaar Romeinse staat.'' Maak de berekening opnieuw met de getallen van de
        astronomen. Wat is het juiste totaal --- en welk jaar heeft, opnieuw, de
        fout veroorzaakt?
\end{enumerate}
\end{problem}
