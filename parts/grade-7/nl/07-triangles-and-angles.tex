\chapter{Driehoeken en hoeken}\label{ch:g7:triangles}

Kun je een driehoek tekenen met zijden $10$, $3$ en $2$? En met hoeken
$80^\circ$, $70^\circ$ en $50^\circ$? Dit hoofdstuk antwoordt op beide vragen met
twee van de nuttigste feiten van de vlakke meetkunde: de driehoeksongelijkheid en de
hoekensom van $180^\circ$ --- plus de woorden voor hoeken die bij snijdende en
\omterm{def:g6:lines:perp}{parallelle lijnen} horen.

\section{Hoeken bij snijdende lijnen}

\begin{definition}[Paren hoeken]\label{def:g7:triangles:pairs}
\begin{itemize}
  \item Twee hoeken zijn \emph{complementair} als hun maten samen $90^\circ$ zijn,
        en \emph{supplementair} als ze samen $180^\circ$ zijn.
  \item Als twee \omterm{def:g6:lines:objects}{lijnen} elkaar snijden, heten de hoeken die dwars over het snijpunt
        tegenover elkaar liggen \emph{overstaande
        hoeken}\index{overstaande hoeken}.
  \item Als een \omterm{def:g6:lines:objects}{lijn} twee andere \omterm{def:g6:lines:objects}{lijnen} snijdt, heten hoeken die bij elk snijpunt
        op dezelfde plaats liggen \emph{overeenkomstige hoeken}; hoeken tussen de
        twee \omterm{def:g6:lines:objects}{lijnen} aan weerszijden van de snijdende \omterm{def:g6:lines:objects}{lijn} heten
        \emph{verwisselende hoeken}\index{verwisselende hoeken}.
\end{itemize}
\end{definition}

\begin{theorem}[Gelijke hoeken]\label{thm:g7:triangles:equalities}
\begin{enumerate}
  \item \omterm{def:g7:triangles:pairs}{Overstaande hoeken} zijn gelijk.
  \item Zijn twee \omterm{def:g6:lines:objects}{lijnen} \omterm{def:g6:lines:perp}{parallel}, dan zijn overeenkomstige hoeken gelijk, en
        \omterm{def:g7:triangles:pairs}{verwisselende hoeken} ook. Omgekeerd dwingen gelijke overeenkomstige (of
        verwisselende) hoeken de \omterm{def:g6:lines:objects}{lijnen} \omterm{def:g6:lines:perp}{parallel} te zijn.
\end{enumerate}
\end{theorem}

\begin{proof}[Bewijs van 1]
De twee hoeken $\widehat{1}$ en $\widehat{2}$ aan één kant van een \omterm{def:g6:lines:objects}{lijn} vormen een
gestrekte hoek: $\widehat 1 + \widehat 2 = 180^\circ$. De hoek $\widehat 3$ die
tegenover $\widehat 1$ ligt, voldoet om dezelfde reden aan
$\widehat 2 + \widehat 3 = 180^\circ$. Dus is
$\widehat 1 = 180^\circ - \widehat 2 = \widehat 3$. Punt 2 nemen we op dit niveau
zonder bewijs aan.
\end{proof}

\section{De som van de hoeken van een driehoek}

\begin{theorem}[Hoekensom]\label{thm:g7:triangles:sum}
In elke driehoek zijn de drie hoeken samen $180^\circ$.
\end{theorem}

\begin{proof}
Zij $ABC$ een driehoek. Teken door $A$ de \omterm{def:g6:lines:objects}{lijn} die \omterm{def:g6:lines:perp}{parallel} is aan $(BC)$. In het
\omterm{def:g5:solids:def}{hoekpunt} $A$ liggen drie hoeken op een rij langs die \omterm{def:g6:lines:perp}{parallel} en vormen ze een
gestrekte hoek, $180^\circ$: de middelste is $\widehat{A}$, en de twee buitenste
zijn \omterm{def:g7:triangles:pairs}{verwisselende hoeken} met $\widehat B$ en $\widehat C$ (\omterm{def:g6:lines:perp}{parallelle lijnen}!) en
dus daaraan gelijk. Bijgevolg is
$\widehat B + \widehat A + \widehat C = 180^\circ$.
\end{proof}

\begin{example}\label{ex:g7:triangles:sum}
Twee hoeken van een driehoek meten $67^\circ$ en $58^\circ$. De derde meet
\[
180 - (67 + 58) = 180 - 125 = 55^\circ .
\]
Bijzondere gevallen die het onthouden waard zijn: elke hoek van een \omterm{def:g6:shapes:triangles}{gelijkzijdige
driehoek} is $180 \div 3 = 60^\circ$; in een \omterm{def:g6:shapes:triangles}{rechthoekige driehoek} zijn de twee
scherpe hoeken complementair (ze zijn samen $180 - 90 = 90^\circ$); de basishoeken
van een \omterm{def:g6:shapes:triangles}{gelijkbenige driehoek} zijn gelijk, dus is elk
$\frac{180 - \text{tophoek}}{2}$.
\end{example}

\begin{example}[Geen driehoek met twee rechte hoeken]\label{ex:g7:triangles:tworight}
Een driehoek met twee hoeken van $90^\circ$ zou al $180^\circ$ opgebruiken en
$0^\circ$ voor de derde overlaten --- onmogelijk. De hoekensom verbiedt in één keer
heel veel driehoeken.
\end{example}

\section{De driehoeksongelijkheid}

\begin{theorem}[Driehoeksongelijkheid]\label{thm:g7:triangles:inequality}
In elke driehoek is elke zijde korter dan de \omterm{def:g1:addition:def}{som} van de twee andere: voor een
driehoek met zijden $a$, $b$ en $c$ geldt
\[
a < b + c .
\]
Omgekeerd kun je van drie lengtes waarvan de \emph{grootste} kleiner is dan de \omterm{def:g1:addition:def}{som}
van de twee andere, altijd een driehoek maken. Is de grootste \emph{gelijk} aan de
\omterm{def:g1:addition:def}{som} van de andere, dan is de ``driehoek'' plat: de drie \omterm{def:g5:solids:def}{hoekpunten} liggen op één
\omterm{def:g6:lines:objects}{lijn}.
\end{theorem}
\admitted

\begin{example}\label{ex:g7:triangles:inequality}
Kunnen de zijden $10$, $3$ en $2$ zijn? Test de grootste: is
$10 < 3 + 2 = 5$? Nee --- zo'n driehoek bestaat niet. De passer laat zien waarom:
bogen met \omterm{def:g3:shapes:shapes}{radius} $3$ en $2$, getekend vanuit de uiteinden van een \omterm{def:g6:lines:objects}{lijnstuk} van $10$,
ontmoeten elkaar nooit.

Zijden $7$, $5$ en $4$? De grootste: $7 < 5 + 4 = 9$: ja, de driehoek bestaat (de
grootste zijde controleren is genoeg).
\end{example}

\begin{method}[Bestaan en constructie van een driehoek]\label{met:g7:triangles:construct}
Bij drie gegeven lengtes:
\begin{enumerate}
  \item vergelijk de grootste met de \omterm{def:g1:addition:def}{som} van de twee andere; is ze niet strikt
        kleiner, stop dan: geen driehoek;
  \item construeer anders zoals in \cref{met:g6:shapes:construct} (basislijnstuk,
        twee passerbogen);
  \item ga na een constructie uit hoeken in gedachten na dat de gegeven hoeken
        samen minder dan $180^\circ$ zijn --- en dat de derde hoek precies
        $180^\circ$ vol maakt.
\end{enumerate}
\end{method}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g7:triangles:1}
Geef het complement (tot $90^\circ$) en het supplement (tot $180^\circ$) van:
$30^\circ$; \ $45^\circ$; \ $72^\circ$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:2}
Twee \omterm{def:g6:lines:objects}{lijnen} snijden elkaar; een van de vier hoeken meet $35^\circ$. Geef de maten
van de drie andere, met bij elk een reden.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:3}
Reken de derde hoek uit van een driehoek waarvan de eerste twee hoeken meten:
(a) $40^\circ$ en $60^\circ$; \
(b) $90^\circ$ en $28^\circ$; \
(c) $75^\circ$ en $75^\circ$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:4}
Een \omterm{def:g6:shapes:triangles}{gelijkbenige driehoek} heeft een tophoek van $40^\circ$. Reken zijn twee
basishoeken uit. Een andere \omterm{def:g6:shapes:triangles}{gelijkbenige driehoek} heeft een basishoek van
$70^\circ$: reken zijn tophoek uit.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:5}
Welke van deze drietallen kunnen de zijden van een driehoek zijn? Verantwoord het
met de driehoeksongelijkheid (de grootste zijde!):
\[
(6,\ 8,\ 12); \qquad
(5,\ 5,\ 11); \qquad
(4,\ 9,\ 13); \qquad
(7,\ 7,\ 7).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:6}
Een driehoek heeft hoeken $x$, $2x$ en $3x$. Zoek $x$ en de drie hoeken. Wat voor
driehoek is het?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:7}
Construeer een driehoek $ABC$ met $BC = 6$ cm, $\widehat{ABC} = 50^\circ$ en
$\widehat{ACB} = 60^\circ$ (gradenboog in $B$ en in $C$). Voorspel voor het
construeren de maat van $\widehat{BAC}$, en controleer die daarna op je figuur.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:8}
Twee \omterm{def:g6:lines:perp}{parallelle lijnen} worden door een derde \omterm{def:g6:lines:objects}{lijn} gesneden, met bij het eerste
snijpunt een hoek van $54^\circ$ (tussen de snijdende \omterm{def:g6:lines:objects}{lijn} en een van de
parallellen). Teken de situatie en geef de maten van alle acht hoeken die ontstaan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:9}
Twee zijden van een driehoek meten $8$ cm en $3$ cm. Tussen welke twee waarden moet
de derde zijde liggen? Geef een hele lengte die werkt en een die niet werkt.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:10}
In een driehoek $ABC$ is $\widehat A = 2\widehat B$ en $\widehat C = 90^\circ$.
Reken $\widehat A$ en $\widehat B$ uit.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:triangles:11}
De hoeken van elke \omterm{def:g4:geometry:polygon}{vierhoek} $ABCD$ kun je onderzoeken door hem langs een diagonaal
in twee driehoeken te knippen. Bewijs daarmee dat de vier hoeken van elke \omterm{def:g4:geometry:polygon}{vierhoek}
samen $360^\circ$ zijn, en leid de hoekensom van een \omterm{def:g4:geometry:polygon}{vijfhoek} (vijf zijden) af.
\end{exercise}

\section{Opgave: een rondje om het blok}

\begin{problem}[{Weekendopgave --- de hoeken van elke veelhoek:
$(n-2) \times 180^\circ$ binnenin, altijd $360^\circ$ draaien buitenom, en waarom
maar drie regelmatige figuren een vloer betegelen}]
\label{pb:g7:triangles:1}
\Cref{exo:g7:triangles:11} knipte een \omterm{def:g4:geometry:polygon}{vierhoek} in twee driehoeken en vond
$360^\circ$. Deze opgave duwt dat idee door naar \omterm{def:g4:geometry:polygon}{veelhoeken} met een willekeurig
aantal zijden, ontdekt daarna een tweede, volstrekt ander bewijs --- door om de
figuur \emph{heen te lopen} en de draaiingen op te tellen --- en eindigt op de vloer
van je badkamer: van alle regelmatige \omterm{def:g4:geometry:polygon}{veelhoeken} kunnen er precies drie een vlak
betegelen zonder gaten of overlap, en de bijen hebben de hunne gekozen.

\medskip
\noindent\textbf{Deel I --- De hoeken binnenin.}
\begin{enumerate}
  \item Teken vanuit één \omterm{def:g5:solids:def}{hoekpunt} van een \omterm{def:g4:geometry:polygon}{zeshoek} (zes zijden) alle diagonalen die
        uit dat \omterm{def:g5:solids:def}{hoekpunt} vertrekken. In hoeveel driehoeken is de \omterm{def:g4:geometry:polygon}{zeshoek} verdeeld,
        en wat is de \omterm{def:g1:addition:def}{som} van \emph{al} haar binnenhoeken
        (\cref{thm:g7:triangles:sum})?
  \item Algemener: hoeveel driehoeken levert de bundel diagonalen uit één \omterm{def:g5:solids:def}{hoekpunt}
        van een \omterm{def:g4:geometry:polygon}{veelhoek} met $n$ zijden op? Leid de formule voor de \omterm{def:g1:addition:def}{som} van de
        binnenhoeken af, en controleer haar voor $n = 3$, $4$, $5$ en $6$.
  \item Reken de \omterm{def:g1:addition:def}{som} van de binnenhoeken uit van een tienhoek ($10$ zijden) en van
        een twaalfhoek ($12$ zijden).
  \item In een \emph{regelmatige} \omterm{def:g4:geometry:polygon}{veelhoek} zijn alle binnenhoeken gelijk. Reken de
        binnenhoek uit van de \omterm{def:g6:shapes:triangles}{gelijkzijdige driehoek}, het vierkant, en de
        regelmatige \omterm{def:g4:geometry:polygon}{vijfhoek}, \omterm{def:g4:geometry:polygon}{zeshoek} en \omterm{def:g4:geometry:polygon}{achthoek}.
  \item Leg zonder formule uit waarom de hoekensom precies $180^\circ$ groeit
        telkens als de \omterm{def:g4:geometry:polygon}{veelhoek} één \omterm{def:g5:solids:def}{hoekpunt} krijgt.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Het rondje om het blok.}
Loop langs de rand van een veelhoekig huizenblok, altijd vooruit. In elke hoek
\emph{draai} je over een zekere hoek --- de \emph{buitenhoek} van die hoek; het
draaien gaat door tot je terug bent bij je startpunt, in je startrichting.
\begin{enumerate}[resume]
  \item Over hoeveel draai je in een hoek waarvan de binnenhoek $\widehat A$ is?
        (Binnenhoek en buitenhoek zijn supplementair.) Reken de drie draaiingen uit
        voor een driehoek met hoeken $40^\circ$, $60^\circ$ en $80^\circ$.
  \item Tel de drie draaiingen van vraag 6 op. Welk feit over de volle cirkel merk
        je op?
  \item Leg uit waarom de draaiingen van een volledig rondje om \emph{elke}
        \omterm{def:g4:geometry:polygon}{veelhoek} --- drie zijden of dertig --- altijd precies $360^\circ$ zijn.
        (Wat heeft je neus gedaan, alle draaiingen samen, als je terugkomt?)
  \item Leid de formule voor de binnenhoeken een tweede keer af: in elk van de $n$
        hoeken is binnenhoek $+$ draaiing $= 180^\circ$; tel over alle hoeken op en
        gebruik vraag 8.
  \item In een regelmatige \omterm{def:g4:geometry:polygon}{veelhoek} zijn alle draaiingen gelijk, dus is elke
        buitenhoek $\frac{360^\circ}{n}$. Antwoord daarmee onmiddellijk: welke
        regelmatige \omterm{def:g4:geometry:polygon}{veelhoek} heeft binnenhoeken van $150^\circ$? En waarom heeft
        \emph{geen} regelmatige \omterm{def:g4:geometry:polygon}{veelhoek} binnenhoeken van $155^\circ$?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De vloer betegelen.}
Om elk punt van een betegelde vloer moeten de hoeken van de tegels die daar
samenkomen precies $360^\circ$ zijn --- geen gat, geen overlap
(\cref{pb:g6:lines:1}).
\begin{enumerate}[resume]
  \item Voor de \omterm{def:g6:shapes:triangles}{gelijkzijdige driehoek}, het vierkant en de regelmatige \omterm{def:g4:geometry:polygon}{zeshoek}:
        hoeveel exemplaren komen er in een \omterm{def:g5:solids:def}{hoekpunt} van de betegeling samen?
        Controleer elke keer de $360^\circ$.
  \item Laat zien dat regelmatige \omterm{def:g4:geometry:polygon}{vijfhoeken} de vloer niet kunnen betegelen: hoeveel
        zijn drie hoeken samen, en hoeveel zouden vier hoeken zijn?
  \item Laat zien dat geen regelmatige \omterm{def:g4:geometry:polygon}{veelhoek} met \emph{meer} dan zes zijden kan
        betegelen: in elke hoek moeten minstens drie tegels samenkomen (elke hoek is
        kleiner dan $180^\circ$), en wat gebeurt er met drie hoeken van $135^\circ$
        of meer? Besluit met de volledige lijst regelmatige betegelaars.
  \item Badkamervloeren mengen vaak regelmatige \omterm{def:g4:geometry:polygon}{achthoeken} met kleine vierkanten. Ga
        na dat deze hoek werkt: $135^\circ + 135^\circ + 90^\circ$. Een andere
        klassieker mengt in elke hoek een vierkant, een regelmatige \omterm{def:g4:geometry:polygon}{zeshoek} en een
        regelmatige twaalfhoek: ga die ook na (de binnenhoek van de twaalfhoek volgt
        uit de methode van vraag 10).
  \item De keuze van de bijen: de cellen van een honingraat zijn regelmatige
        \omterm{def:g4:geometry:polygon}{zeshoeken}. Ga hun hoeken na ($3$ hoeken), en leg in één zin uit --- met hulp
        van de ontdekking van Dido (\cref{pb:g6:measure:1}) --- waarom van de drie
        mogelijke regelmatige tegels de \omterm{def:g4:geometry:polygon}{zeshoek} het goedkoopst is in was voor de
        honing die ze bevat.
\end{enumerate}
\end{problem}
