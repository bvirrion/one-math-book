\chapter{Fracciones: comparar y sumar}\label{ch:g7:fractions}

El año pasado presentamos las fracciones como repartos y como \omterm{def:g3:division:remainder}{cocientes} exactos
(\cref{ch:g6:fractions}). Este capítulo aprende a \emph{calcular} con
ellas: reconocer fracciones iguales, comparar, sumar y restar
--- por ahora con \omterm{def:g4:fractions:def}{denominadores} amables; el caso general llega en el
\cref{ch:g9:fractions}.

\section{Fracciones iguales}

\begin{theorem}[Fracciones iguales]\label{thm:g7:fractions:equal}
Una \omterm{def:g6:fractions:def}{fracción} no cambia cuando su \omterm{def:g4:fractions:def}{numerador} y su \omterm{def:g4:fractions:def}{denominador} se
multiplican, o se dividen, por un mismo número distinto de \omterm{def:g1:counting:zero}{cero}:
\[
\frac{a}{b} = \frac{a \times k}{b \times k} .
\]
\end{theorem}
\admitted

\begin{example}\label{ex:g7:fractions:equal}
$\dfrac{3}{4} = \dfrac{3 \times 5}{4 \times 5} = \dfrac{15}{20}$ y,
en el otro sentido, $\dfrac{42}{30} = \dfrac{42 \div 6}{30 \div 6}
= \dfrac{7}{5}$ (simplified by $6$).

Para comprobar si $\dfrac{8}{12}$ y $\dfrac{10}{15}$ son iguales,
simplifica las dos: $\dfrac{8}{12} = \dfrac{2}{3}$ y
$\dfrac{10}{15} = \dfrac{2}{3}$ --- sí, son iguales.
\end{example}

\section{Comparar fracciones}

\begin{method}[Comparar dos fracciones]\label{met:g7:fractions:compare}
\begin{enumerate}
  \item \emph{Mismo \omterm{def:g4:fractions:def}{denominador}}: compara los \omterm{def:g4:fractions:def}{numeradores}:
        $\frac{5}{9} < \frac{7}{9}$.
  \item \emph{un \omterm{def:g4:fractions:def}{denominador} es \omterm{def:g5:division:multiple}{múltiplo} del otro}: reescribe la
        \omterm{def:g6:fractions:def}{fracción} más basta y compara después:
        $\frac{5}{6}$ frente a $\frac{7}{12}$: $\frac56 = \frac{10}{12} >
        \frac{7}{12}$.
  \item \emph{compara con $1$ o con $\frac12$} cuando se pueda:
        $\frac{9}{8} > 1 > \frac{7}{9}$ resuelve
        $\frac98$ frente a $\frac79$ al instante.
\end{enumerate}
\end{method}

\section{Sumar y restar}

\begin{theorem}[Mismo denominador]\label{thm:g7:fractions:add}
Las fracciones con el mismo \omterm{def:g4:fractions:def}{denominador} se suman (o se restan)
sumando (o restando) los \omterm{def:g4:fractions:def}{numeradores}:
\[
\frac{a}{d} + \frac{b}{d} = \frac{a + b}{d},
\qquad
\frac{a}{d} - \frac{b}{d} = \frac{a - b}{d} .
\]
\end{theorem}

\begin{proof}[Por qué]
$a$ partes de tamaño $\frac1d$ más $b$ partes del mismo tamaño hacen
$a + b$ partes de ese tamaño.
\end{proof}

\begin{method}[Denominadores distintos]\label{met:g7:fractions:add}
Cuando un \omterm{def:g4:fractions:def}{denominador} es \omterm{def:g5:division:multiple}{múltiplo} del otro:
\begin{enumerate}
  \item reescribe la \omterm{def:g6:fractions:def}{fracción} de \omterm{def:g4:fractions:def}{denominador} menor para que las dos tengan
        el mayor (\cref{thm:g7:fractions:equal});
  \item \omterm{def:g1:addition:def}{suma} o resta los \omterm{def:g4:fractions:def}{numeradores};
  \item simplifica el resultado si se puede.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:fractions:add}
Calcula $\dfrac{3}{4} + \dfrac{5}{8}$, paso a paso:
\begin{enumerate}
  \item $8$ es \omterm{def:g5:division:multiple}{múltiplo} de $4$: reescribe
        $\dfrac34 = \dfrac{3 \times 2}{4 \times 2} = \dfrac{6}{8}$;
  \item \omterm{def:g1:addition:def}{suma}: $\dfrac{6}{8} + \dfrac{5}{8} = \dfrac{11}{8}$;
  \item no hay nada que simplificar: el resultado es $\dfrac{11}{8}$, es decir,
        $1 + \dfrac38$.
\end{enumerate}
Otro: $2 - \dfrac{4}{5} = \dfrac{10}{5} - \dfrac{4}{5} =
\dfrac{6}{5}$ (escribe el número entero con \omterm{def:g4:fractions:def}{denominador} $5$
primero).
\end{example}

\begin{example}[Una trampa clásica]\label{ex:g7:fractions:trap}
¡$\dfrac12 + \dfrac13$ \emph{no} es $\dfrac{2}{5}$! Sumar \omterm{def:g4:fractions:def}{numeradores}
y \omterm{def:g4:fractions:def}{denominadores} por separado está mal --- la respuesta $\frac25$ sería
menor que $\frac12$, algo absurdo al \emph{sumar} algo positivo
a $\frac12$. (La \omterm{def:g1:addition:def}{suma} correcta, $\frac56$, necesita el \omterm{def:g4:fractions:def}{denominador} común
$6$; el método general está en el \cref{ch:g9:fractions}.)
\end{example}

\section{Multiplicar una fracción por un número}

\begin{proposition}[Fracción por un número]\label{prop:g7:fractions:mult}
Para un número $k$:
\[
k \times \frac{a}{b} = \frac{k \times a}{b} .
\]
Tomar $\frac ab$ \emph{de} una cantidad significa multiplicar por
$\frac ab$.
\end{proposition}
\admitted

\begin{example}\label{ex:g7:fractions:mult}
$6 \times \dfrac{2}{3} = \dfrac{12}{3} = 4$: seis veces dos tercios son
cuatro unidades. Y «$\frac34$ de $60$ euros» es
$\frac34 \times 60 = \frac{180}{4} = 45$ euros --- la misma respuesta que
dividir entre $4$ y multiplicar después por $3$
(\cref{met:g6:fractions:ofquantity}).
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g7:fractions:1}
Completa: $\dfrac{2}{5} = \dfrac{?}{20}$; \quad
$\dfrac{18}{24} = \dfrac{3}{?}$; \quad
$\dfrac{7}{3} = \dfrac{28}{?}$; \quad
$\dfrac{?}{9} = \dfrac{20}{36}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:2}
Simplifica todo lo posible:
$\dfrac{12}{16}$; \quad $\dfrac{30}{45}$; \quad $\dfrac{27}{36}$;
\quad $\dfrac{48}{12}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:3}
¿Son iguales $\dfrac{15}{35}$ y $\dfrac{21}{49}$? ¿Y $\dfrac{16}{28}$
y $\dfrac{20}{36}$? Justifícalo simplificando.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:4}
Compara (escribe el razonamiento, no solo la respuesta):
\[
\frac{7}{11} \text{ y } \frac{9}{11}; \qquad
\frac{5}{6} \text{ y } \frac{11}{18}; \qquad
\frac{13}{12} \text{ y } \frac{19}{20}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:5}
Calcula y simplifica si se puede:
\[
\frac{5}{9} + \frac{2}{9}, \qquad
\frac{11}{7} - \frac{4}{7}, \qquad
\frac{5}{8} + \frac{7}{8} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:6}
Calcula usando el \cref{met:g7:fractions:add}:
\[
\frac{2}{3} + \frac{5}{6}, \qquad
\frac{7}{10} - \frac{2}{5}, \qquad
\frac{3}{4} + \frac{5}{12} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:7}
Calcula:
\[
3 - \frac{5}{4}, \qquad
1 + \frac{3}{8}, \qquad
2 - \frac{7}{6} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:8}
Calcula:
\[
5 \times \frac{3}{10}, \qquad
\frac{7}{8} \times 4, \qquad
\frac{2}{3} \text{ de } 45 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:fractions:9}
Marcos se comió $\frac14$ de una pizza y Julia, $\frac38$ de la misma pizza.
¿Qué \omterm{def:g6:fractions:def}{fracción} de la pizza se comieron entre los dos? ¿Qué \omterm{def:g6:fractions:def}{fracción}
queda?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:fractions:10}
Una botella contiene $\frac34$ L de zumo. Lea sirve $\frac16$ L
dos veces. ¿Cuánto zumo queda? (\omterm{def:g4:fractions:def}{Denominador} común: $12$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:fractions:11}
Explica, con el argumento del \cref{ex:g7:fractions:trap}, por qué
$\frac35 + \frac12$ no puede ser $\frac{4}{7}$, sin calcular la
\omterm{def:g1:addition:def}{suma} correcta.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:fractions:12}
Calcula la \omterm{def:g1:addition:def}{suma}
\[
\frac12 + \frac14 + \frac18 + \frac{1}{16}
\]
paso a paso (de izquierda a derecha). Observa el patrón de las \omterm{def:g1:addition:def}{sumas} parciales:
¿a qué número se acercan a medida que se van añadiendo términos
$\frac{1}{32}, \frac{1}{64}, \dots$?
\end{exercise}

\section{Problema: Las fracciones escondidas en un compás de música}

\begin{problem}[{Problema de fin de semana --- las figuras musicales son fracciones
que tienen que sumar exactamente, contar el ritmo esconde una sucesión famosa
y la paridad manda en el compás balcánico}]
\label{pb:g7:fractions:1}
La música se escribe con fracciones. Una \emph{redonda} dura $1$
compás (en el corriente compás de «cuatro por cuatro»); una \emph{blanca}
dura $\frac12$; después vienen la \emph{negra}, $\frac14$, la
\emph{corchea}, $\frac18$, y la \emph{semicorchea},
$\frac{1}{16}$. Una ley inviolable: \emph{las duraciones de un
compás tienen que sumar exactamente el total del compás} --- $1$ en el
compás de cuatro por cuatro. Todas las preguntas de abajo son la aritmética de este capítulo
(\cref{thm:g7:fractions:add}, \cref{met:g7:fractions:add}) puesta en
música; por el camino conocerás una sucesión de recuento descubierta por
eruditos indios hace mil años.

\medskip
\noindent\textbf{Parte I --- Figuras que tienen que sumar.}
\begin{enumerate}
  \item comprueba que cada uno de estos compases es legal en el compás de cuatro
        por cuatro: (a) blanca $+$ negra $+$ negra; (b) negra $+$
        corchea $+$ corchea $+$ blanca;
  \item un copista escribió: blanca $+$ negra $+$ corchea. ¿Qué
        \omterm{def:g6:fractions:def}{fracción} del compás está escrita y qué figura única
        lo completa?
  \item ¿cuántas semicorcheas llenan un compás entero? ¿Cuántas
        semicorcheas hacen una corchea?
  \item un \emph{puntillo} detrás de una figura le añade la \omterm{def:g3:division:half}{mitad} de su valor: una
        blanca con puntillo es $\frac12 + \frac14$ y una negra con puntillo es
        $\frac14 + \frac18$. Calcula los dos valores;
  \item un vals se escribe en compás de tres por cuatro: cada compás debe
        sumar $\frac34$. Comprueba que una sola blanca con puntillo llena un
        compás de vals y escribe otros dos compases de vals legales
        usando solo negras y corcheas.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El taller del copista.}
\begin{enumerate}[resume]
  \item un compás de cuatro por cuatro contiene: negra con puntillo $+$ corchea
        $+$ negra. ¿Qué falta?
  \item ¿qué dura más, una corchea con puntillo o una negra?
        Compáralas en semicorcheas;
  \item una \emph{ligadura} pega duraciones en un solo sonido
        largo: una blanca ligada a una corchea, ¿cuánto dura? ¿Cuántas
        semicorcheas son?
  \item las figuras de un compás suman $\frac{11}{16}$; el
        \omterm{def:g3:division:remainder}{resto} es un \emph{silencio}. ¿Cuánto dura ese
        silencio?
  \item un batería rellena una negra ($\frac{4}{16}$ del
        compás) usando solo corcheas ($\frac{2}{16}$) y
        semicorcheas ($\frac{1}{16}$), y el orden importa: por
        ejemplo, corchea--semicorchea--semicorchea o
        semicorchea--corchea--semicorchea. Comprueba que los dos ejemplos
        son legales y enumera después \emph{todos} los ritmos posibles.
        ¿Cuántos hay?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Una sucesión de mil años y un compás
\omterm{def:g3:numbers:evenodd}{impar}.}
\begin{enumerate}[resume]
  \item contemos los ritmos del batería para figuras más largas.
        Todo ritmo termina o bien en semicorchea o bien en
        corchea; explica por qué eso da la regla: (ritmos que
        llenan $n$ semicorcheas) $=$ (ritmos que llenan $n - 1$)
        $+$ (ritmos que llenan $n - 2$). Partiendo de $1$ ritmo
        para una semicorchea y $2$ para dos, construye la tabla hasta
        $n = 8$: ¿cuántos ritmos llenan una blanca?
  \item los números que has hallado --- $1, 2, 3, 5, 8, 13, 21, 34$ ---
        los publicó el erudito indio Hemachandra
        hacia el año 1150, contando exactamente estos ritmos, dos
        generaciones antes de que Fibonacci los escribiera en Italia.
        Enuncia en una frase el patrón que los define y compruébalo
        en las tres últimas entradas de tu tabla;
  \item el estilo «swing» sustituye dos corcheas iguales por una pareja
        larga--corta: $\frac16$ y después $\frac{1}{12}$. Verifica
        que el cambio es legal (la misma duración total), usando el
        \omterm{def:g4:fractions:def}{denominador} común $12$;
  \item un \emph{tresillo} mete tres figuras iguales dentro de una
        negra. ¿Cuánto dura cada una (comprueba tu respuesta
        multiplicándola por $3$,
        \cref{prop:g7:fractions:mult})? La misma pregunta para tres
        figuras iguales que llenan una blanca;
  \item la música de danza balcánica usa el compás de siete por ocho: cada compás
        \omterm{def:g1:addition:def}{suma} $\frac78$. Verifica que negra con puntillo $+$ negra
        $+$ negra lo llena. Después explica por qué \emph{ninguna}
        combinación de blancas y negras solamente,
        por muchas que sean, puede llenar un compás de siete por ocho.
        (Cuenta en corcheas y piensa en \omterm{def:g3:numbers:evenodd}{pares} e \omterm{def:g3:numbers:evenodd}{impares}.)
\end{enumerate}
\end{problem}
