\chapter{Organizar datos}\label{ch:g7:stats}

Antes de calcular nada hay que recoger, contar y presentar los
datos. Este capítulo trata de convertir una lista desordenada de observaciones
en una tabla o en una imagen clara: \omterm{def:g7:stats:frequency}{frecuencias absolutas}, \omterm{def:g7:stats:frequency}{frecuencias relativas}, datos agrupados,
diagramas de barras y diagramas de sectores. Resumir los datos con un solo número (media,
mediana) es el tema del \cref{ch:g9:statproba}.

\section{Frecuencias absolutas y relativas}

\begin{definition}[Frecuencia absoluta y frecuencia relativa]\label{def:g7:stats:frequency}
Para cada valor posible de una encuesta, su \emph{frecuencia absoluta}\index{frecuencia absoluta} es
el número de veces que aparece; su \emph{frecuencia relativa}\index{frecuencia relativa}
es la frecuencia absoluta dividida entre el número total de observaciones:
\[
\text{frecuencia relativa} = \frac{\text{frecuencia absoluta}}{\text{total}} .
\]
Las frecuencias relativas se pueden escribir como fracciones, decimales o porcentajes, y
siempre suman $1$ (es decir, $100\,\%$).
\end{definition}

\begin{example}\label{ex:g7:stats:frequency}
Deporte favorito de los $25$ alumnos de una clase, de la lista en bruto a
la tabla:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cccc|c}
deporte & fútbol & natación & tenis & judo & total \\
\hline
frec. absoluta & $10$ & $6$ & $5$ & $4$ & $25$ \\
frec. relativa & $0.40$ & $0.24$ & $0.20$ & $0.16$ & $1$ \\
\end{tabular}
\end{center}
Para el fútbol: $\frac{10}{25} = \frac{40}{100} = 0.40 = 40\,\%$. La fila de
\omterm{def:g7:stats:frequency}{frecuencias relativas} tiene que sumar $1$ --- una comprobación permanente y gratuita.
\end{example}

\begin{remark}[¿Por qué frecuencias relativas?]
Las \omterm{def:g7:stats:frequency}{frecuencias absolutas} no se pueden comparar entre grupos de tamaños distintos: $10$
aficionados al fútbol en una clase de $25$ (o sea, el $40\,\%$) muestran más
entusiasmo que $12$ en un curso de $120$ (el $10\,\%$).
Las \omterm{def:g7:stats:frequency}{frecuencias relativas} lo ponen todo en la misma escala.
\end{remark}

\section{Agrupar datos en clases}

\begin{example}[Clases]\label{ex:g7:stats:classes}
Las estaturas (en cm) de $20$ alumnos van de $148$ a $172$.
Listar cada estatura por separado daría una tabla con casi tantas
columnas como alumnos; en su lugar, agrupamos en \emph{clases}:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|ccc|c}
estatura (cm) & $\intco{145}{155}$ & $\intco{155}{165}$ &
$\intco{165}{175}$ & total \\
\hline
frec. absoluta & $5$ & $9$ & $6$ & $20$ \\
frec. relativa & $0.25$ & $0.45$ & $0.30$ & $1$ \\
\end{tabular}
\end{center}
Cada estatura se cuenta en exactamente una clase (el $155$ va en
$\intco{155}{165}$, no en las dos --- por eso los intervalos son
semiabiertos). Agrupar pierde detalle, pero gana legibilidad.
\end{example}

\section{Gráficos}

\begin{method}[Elegir y dibujar un gráfico]\label{met:g7:stats:charts}
\begin{enumerate}
  \item \emph{Diagrama de barras}: una barra por valor o por clase, con la altura
        proporcional a la \omterm{def:g7:stats:frequency}{frecuencia absoluta} (o a la relativa). Bueno para
        comparar valores.
  \item \emph{Diagrama de sectores}: un sector por valor, con el ángulo proporcional a
        la \omterm{def:g7:stats:frequency}{frecuencia relativa} --- el círculo completo ($360^\circ$) representa el
        total, así que
        \[
        \text{ángulo} = \text{frecuencia relativa} \times 360^\circ .
        \]
        Bueno para mostrar partes de un todo.
  \item en los dos casos: título, etiquetas y una graduación o los
        porcentajes escritos en el gráfico.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:stats:pie}
Ángulos para los deportes del \cref{ex:g7:stats:frequency}: fútbol,
$0.40 \times 360 = 144^\circ$; natación, $0.24 \times 360 =
86.4^\circ$; tenis, $72^\circ$; judo, $57.6^\circ$. Comprobación:
$144 + 86.4 + 72 + 57.6 = 360$.
\end{example}

\begin{example}[Leer con espíritu crítico]\label{ex:g7:stats:critical}
Un gráfico cuyo eje vertical empieza en $9$ en vez de en $0$ hace que una barra
de $10$ parezca diez veces más alta que una de $9.1$ --- visualmente
engañoso, aunque los números sean honestos. Primer reflejo ante
cualquier gráfico: comprueba dónde empieza el eje y cuánto vale una
graduación (\cref{met:g6:propdata:read}).
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g7:stats:1}
Estas son las notas de $20$ alumnos: $12$, $15$, $9$, $12$, $14$,
$9$, $12$, $15$, $11$, $12$, $9$, $14$, $12$, $11$, $15$, $9$, $12$,
$14$, $11$, $12$. Construye la tabla de \omterm{def:g7:stats:frequency}{frecuencias absolutas} (una columna por nota).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:2}
Completa la tabla del \cref{exo:g7:stats:1} con una fila de \omterm{def:g7:stats:frequency}{frecuencias relativas}
(como fracciones de $20$ y después como porcentajes). Comprueba la \omterm{def:g1:addition:def}{suma}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:3}
En una encuesta a $50$ familias, $30$ tienen un coche. ¿Cuál es la
\omterm{def:g7:stats:frequency}{frecuencia relativa} de «un coche», como decimal y como porcentaje? En otra
encuesta, $45$ familias de $90$ tienen un coche: ¿qué encuesta muestra
la proporción mayor?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:4}
Las \omterm{def:g2:measure:units}{masas} (kg) de $15$ perros: $8$, $23$, $31$, $12$, $9$, $27$, $35$,
$14$, $22$, $18$, $29$, $11$, $25$, $33$, $16$. Agrúpalas en las
clases $\intco{5}{15}$, $\intco{15}{25}$, $\intco{25}{35}$ y
$\intco{35}{45}$, y da las \omterm{def:g7:stats:frequency}{frecuencias absolutas}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:5}
Dibuja el diagrama de barras de la tabla:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|cccc}
mascotas & $0$ & $1$ & $2$ & $3$ \\
\hline
familias & $8$ & $12$ & $6$ & $4$ \\
\end{tabular}
\end{center}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:6}
Para la tabla del \cref{exo:g7:stats:5}, calcula las \omterm{def:g7:stats:frequency}{frecuencias relativas} y
los ángulos del diagrama de sectores de cada valor (comprobación: total $360^\circ$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:7}
Un diagrama de sectores sobre la estación preferida muestra: verano $162^\circ$, primavera
$90^\circ$, otoño $54^\circ$, invierno $54^\circ$. ¿Qué porcentaje
eligió cada estación? Si respondieron $60$ personas, ¿cuántas eligieron el verano?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:8}
En la clase A, $12$ alumnos de $30$ van al colegio andando; en la clase B,
$14$ de $40$. ¿Qué clase tiene mayor proporción de alumnos que van andando?
Justifícalo con \omterm{def:g7:stats:frequency}{frecuencias relativas}, no con absolutas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:9}
Una revista publica un diagrama de barras de las ventas mensuales: $9\,800$, después
$10\,000$ y después $10\,100$, con el eje vertical empezando en
$9\,700$. Describe lo que ve el lector y lo que mostraría en cambio un eje honesto
que empezara en $0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:10}
Una tabla de \omterm{def:g7:stats:frequency}{frecuencias relativas} tiene tres valores con frecuencias $0.35$, $0.4$ y
$f$. Halla $f$. Si el total de observaciones es $80$, da las tres \omterm{def:g7:stats:frequency}{frecuencias absolutas}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:stats:11}
En un colegio, el $55\,\%$ de los alumnos son chicas. Entre las chicas, el
$40\,\%$ come en el comedor; entre los chicos, lo hace el $60\,\%$. De
$400$ alumnos, ¿cuántos comen en el comedor? (Calcula los cuatro
tamaños de grupo paso a paso.) ¿Qué porcentaje global es eso?
\end{exercise}

\section{Problema: Cómo mienten los datos --- y cómo pillarlos}

\begin{problem}[{Problema de fin de semana --- frecuencias absolutas frente a relativas,
la encuesta que engañó a todo un país y la paradoja de los dos
hospitales}]
\label{pb:g7:stats:1}
Los números no mienten, pero se les puede hacer engañar: una \omterm{def:g7:stats:frequency}{frecuencia absoluta} enorme
puede esconder una \omterm{def:g7:stats:frequency}{frecuencia relativa} pequeña, una encuesta gigantesca puede no valer nada
y --- lo más extraño de todo --- un hospital puede ganar a su rival en
\emph{todas} las categorías de pacientes y aun así perder en las cifras
globales. Las tres trampas saltan en este problema, armados solo
con las \omterm{def:g7:stats:frequency}{frecuencias absolutas} y relativas de este capítulo
(\cref{def:g7:stats:frequency}).

\medskip
\noindent\textbf{Parte I --- Las \omterm{def:g7:stats:frequency}{frecuencias absolutas} no son relativas.}
\begin{enumerate}
  \item el colegio A recicla $120$ de sus $400$ briks de zumo;
        el colegio B, $90$ de $250$. ¿Qué colegio tiene la mayor
        \emph{\omterm{def:g7:stats:frequency}{frecuencia absoluta}} de briks reciclados? ¿Y la mayor
        \emph{\omterm{def:g7:stats:frequency}{frecuencia relativa}}? ¿Qué colegio merece el premio de
        reciclaje?
  \item se encuesta sobre un horario nuevo: a $18$ de los $30$ profesores
        les gusta y a $210$ de los $600$ alumnos también. Calcula las
        dos \omterm{def:g7:stats:frequency}{frecuencias relativas} y después la \omterm{def:g7:stats:frequency}{frecuencia relativa} entre las $630$
        personas juntas. ¿Por qué queda la cifra conjunta tan
        cerca de la de los alumnos?
  \item un diagrama de sectores publicado muestra sectores etiquetados $45\,\%$,
        $30\,\%$, $20\,\%$ y $10\,\%$. Sin ninguna información
        más, ¿cómo sabes que hay un error?
  \item una tabla de frecuencias sobre $40$ observaciones está medio borrada:
        \omterm{def:g7:stats:frequency}{frecuencias absolutas} $14$, $10$, $?$, $?$ y relativas $0.35$,
        $0.25$, $0.15$, $?$. Reconstruye las entradas que faltan;
  \item una encuesta termina con \omterm{def:g7:stats:frequency}{frecuencias relativas} $0.40$, $0.35$ y
        $0.25$. Calcula los tres ángulos del diagrama de sectores
        (\cref{met:g7:stats:charts}) y comprueba que cierran el
        círculo.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La encuesta que engañó a todo un país.}
En 1936, una revista estadounidense envió por correo diez millones de papeletas a
direcciones sacadas de guías telefónicas y de registros de
vehículos, recibió $2.4$ millones de respuestas y predijo una victoria
aplastante del candidato Landon. Un joven estadístico, George
Gallup, preguntó solo a unas cincuenta mil personas --- elegidas para
parecerse al conjunto de la población --- y predijo lo contrario.
Roosevelt ganó por una goleada.
\begin{enumerate}[resume]
  \item ¿qué \omterm{def:g7:stats:frequency}{frecuencia relativa} de las papeletas enviadas volvió?
  \item en 1936 el teléfono y el coche eran lujos. Explica en
        una o dos frases por qué la muestra de la revista, aunque
        enorme, estaba condenada --- y qué lección de la pregunta 1
        repite a escala nacional;
  \item un modelo de juguete. Un pueblo tiene $1\,000$ votantes ricos, de los cuales
        el $70\,\%$ apoya a L, y $9\,000$ votantes modestos, de los cuales
        el $30\,\%$ apoya a L. Calcula el apoyo real a L en el
        pueblo. Una «encuesta de guía telefónica» llega a $500$ ricos
        y $500$ modestos: ¿qué apoyo predice?
  \item la trampa silenciosa: una empresa tiene $200$ clientes descontentos
        y $800$ contentos. Los encuesta a todos; responde el $40\,\%$ de
        los descontentos, pero solo el $10\,\%$ de los contentos.
        Calcula el número de respuestas de cada clase y la
        \omterm{def:g7:stats:frequency}{frecuencia relativa} de descontento \emph{entre las respuestas}.
        Compárala con la \omterm{def:g7:stats:frequency}{frecuencia relativa} verdadera;
  \item repara la encuesta de la pregunta 8: mantén $1\,000$
        entrevistas, pero repártelas en las proporciones verdaderas
        del pueblo. ¿Qué predice la encuesta reparada?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La paradoja de los dos
hospitales.}
Dos hospitales publican sus curaciones, separadas por gravedad:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cc}
                 & casos leves & casos graves \\
\hline
hospital A & $90$ curados de $100$ & $280$ curados de $400$ \\
hospital B & $340$ curados de $400$ & $65$ curados de $100$ \\
\end{tabular}
\end{center}
\begin{enumerate}[resume]
  \item calcula la \omterm{def:g7:stats:frequency}{frecuencia relativa} de curación del hospital A en los casos leves, en los
        graves y en total (los $500$ pacientes);
  \item los mismos tres cálculos para el hospital B. ¿Qué hospital
        gana en los casos leves? ¿Y en los graves?
  \item compara las dos frecuencias globales. Enuncia claramente lo
        extraño que ha ocurrido;
  \item explica el truco: compara las \emph{mezclas} de pacientes de los
        dos hospitales y di por qué la \omterm{def:g7:stats:frequency}{frecuencia relativa} global puede
        traicionar a las dos frecuencias por categorías. Si tuvieras
        un caso grave, ¿qué hospital elegirías?
  \item un titular de periódico dice: «El hospital B tiene mejor
        tasa de curación: $81\,\%$ frente a $74\,\%$». Escribe la
        carta al director de dos frases que todo este
        problema te ha enseñado a escribir --- una frase sobre
        lo que ignora el titular y otra sobre la moraleja general
        (compara lo comparable y cuidado con los pesos).
\end{enumerate}
\end{problem}
