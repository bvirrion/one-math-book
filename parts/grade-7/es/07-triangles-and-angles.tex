\chapter{Triángulos y ángulos}\label{ch:g7:triangles}

¿Se puede dibujar un triángulo de lados $10$, $3$ y $2$? ¿Y con ángulos
$80^\circ$, $70^\circ$ y $50^\circ$? Este capítulo responde a las dos
preguntas con dos de los hechos más útiles de la geometría plana: la
desigualdad triangular y la \omterm{def:g1:addition:def}{suma} de $180^\circ$ --- más el vocabulario
de ángulos que llega con las rectas secantes y \omterm{def:g6:lines:perp}{paralelas}.

\section{Ángulos alrededor de rectas secantes}

\begin{definition}[Parejas de ángulos]\label{def:g7:triangles:pairs}
\begin{itemize}
  \item dos ángulos son \emph{complementarios} cuando sus medidas suman
        $90^\circ$, y \emph{suplementarios} cuando suman
        $180^\circ$.
  \item cuando dos rectas se cortan, los ángulos que quedan enfrentados a través del
        punto de corte son \emph{opuestos por el
        vértice}\index{ángulos opuestos por el vértice};
  \item cuando una recta corta a otras dos, los ángulos situados en la
        misma posición en cada corte son
        \emph{ángulos correspondientes}; los ángulos situados entre las dos rectas y a
        lados opuestos de la secante son \emph{ángulos
        alternos}\index{ángulos alternos}.
\end{itemize}
\end{definition}

\begin{theorem}[Igualdades de ángulos]\label{thm:g7:triangles:equalities}
\begin{enumerate}
  \item los \omterm{def:g7:triangles:pairs}{ángulos opuestos por el vértice} son iguales;
  \item si dos rectas son \omterm{def:g6:lines:perp}{paralelas}, los ángulos correspondientes son iguales y
        los \omterm{def:g7:triangles:pairs}{ángulos alternos} son iguales. Recíprocamente, unos ángulos correspondientes
        (o alternos) iguales obligan a que las rectas sean \omterm{def:g6:lines:perp}{paralelas}.
\end{enumerate}
\end{theorem}

\begin{proof}[Demostración de 1]
Los dos ángulos $\widehat{1}$ y $\widehat{2}$ situados a un lado de una recta
forman un ángulo llano: $\widehat 1 + \widehat 2 = 180^\circ$. El ángulo
$\widehat 3$, opuesto por el \omterm{def:g5:solids:def}{vértice} a $\widehat 1$, cumple también
$\widehat 2 + \widehat 3 = 180^\circ$ (por la misma razón). Así que
$\widehat 1 = 180^\circ - \widehat 2 = \widehat 3$. El punto 2 se
admite a este nivel.
\end{proof}

\section{La suma de los ángulos de un triángulo}

\begin{theorem}[Suma de los ángulos]\label{thm:g7:triangles:sum}
En todo triángulo, los tres ángulos suman $180^\circ$.
\end{theorem}

\begin{proof}
Sea $ABC$ un triángulo. Traza la recta que pasa por $A$ paralela a
$(BC)$. En el \omterm{def:g5:solids:def}{vértice} $A$ se alinean tres ángulos sobre esa paralela
y forman un ángulo llano, $180^\circ$: el del medio es
$\widehat{A}$ y los dos exteriores son \omterm{def:g7:triangles:pairs}{ángulos alternos} con
$\widehat B$ y $\widehat C$ (¡\omterm{def:g6:lines:perp}{rectas paralelas}!), así que valen
$\widehat B$ y $\widehat C$. Por tanto,
$\widehat B + \widehat A + \widehat C = 180^\circ$.
\end{proof}

\begin{example}\label{ex:g7:triangles:sum}
Dos ángulos de un triángulo miden $67^\circ$ y $58^\circ$. El tercero
mide
\[
180 - (67 + 58) = 180 - 125 = 55^\circ .
\]
Casos especiales que conviene memorizar: cada ángulo de un \omterm{def:g6:shapes:triangles}{triángulo equilátero}
mide $180 \div 3 = 60^\circ$; en un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} los dos ángulos agudos
son complementarios (suman $180 - 90 = 90^\circ$); los ángulos
de la base de un \omterm{def:g6:shapes:triangles}{triángulo isósceles} son iguales, así que cada uno vale
$\frac{180 - \text{ángulo desigual}}{2}$.
\end{example}

\begin{example}[No hay triángulos con dos ángulos rectos]\label{ex:g7:triangles:tworight}
Un triángulo con dos ángulos de $90^\circ$ ya gastaría
$180^\circ$ y dejaría $0^\circ$ para el tercero --- imposible. La \omterm{def:g1:addition:def}{suma}
de los ángulos prohíbe muchos triángulos de golpe.
\end{example}

\section{La desigualdad triangular}

\begin{theorem}[Desigualdad triangular]\label{thm:g7:triangles:inequality}
En todo triángulo, cada lado es más corto que la \omterm{def:g1:addition:def}{suma} de los otros
dos: para un triángulo de lados $a$, $b$ y $c$,
\[
a < b + c .
\]
Recíprocamente, tres longitudes cuya \emph{mayor} sea menor que la \omterm{def:g1:addition:def}{suma}
de las otras dos siempre forman un triángulo. Si la mayor
\emph{es igual} a la \omterm{def:g1:addition:def}{suma} de las otras, el «triángulo» queda plano: los
tres \omterm{def:g5:solids:def}{vértices} están alineados.
\end{theorem}
\admitted

\begin{example}\label{ex:g7:triangles:inequality}
¿Pueden los lados ser $10$, $3$ y $2$? Comprueba el mayor: ¿es $10 < 3 + 2 = 5$?
No --- ese triángulo no existe. El compás muestra por qué: los arcos de \omterm{def:g3:shapes:shapes}{radios}
$3$ y $2$ trazados desde los dos extremos de un \omterm{def:g6:lines:objects}{segmento} de longitud $10$ no
se encuentran nunca.

¿Lados $7$, $5$ y $4$? El mayor: $7 < 5 + 4 = 9$: sí, el triángulo
existe (basta con comprobar el lado mayor).
\end{example}

\begin{method}[Existencia y construcción de un triángulo]\label{met:g7:triangles:construct}
Dadas tres longitudes:
\begin{enumerate}
  \item compara la mayor con la \omterm{def:g1:addition:def}{suma} de las otras dos; si no es
        estrictamente menor, para: no hay triángulo;
  \item si lo es, construye como en el \cref{met:g6:shapes:construct}
        (\omterm{def:g6:lines:objects}{segmento} base y dos arcos de compás);
  \item después de cualquier construcción a partir de ángulos, comprueba mentalmente que los
        ángulos dados suman menos de $180^\circ$ --- y que el tercer
        ángulo completa exactamente $180^\circ$.
\end{enumerate}
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g7:triangles:1}
Da el complementario (a $90^\circ$) y el suplementario (a
$180^\circ$) de: $30^\circ$; \ $45^\circ$; \ $72^\circ$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:2}
Dos rectas se cortan; uno de los cuatro ángulos mide $35^\circ$. Da las
medidas de los otros tres, con una razón para cada uno.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:3}
Calcula el tercer ángulo de un triángulo cuyos dos primeros ángulos miden:
(a) $40^\circ$ y $60^\circ$; \
(b) $90^\circ$ y $28^\circ$; \
(c) $75^\circ$ y $75^\circ$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:4}
Un \omterm{def:g6:shapes:triangles}{triángulo isósceles} tiene el ángulo desigual igual a $40^\circ$. Calcula
sus dos ángulos de la base. Otro \omterm{def:g6:shapes:triangles}{triángulo isósceles} tiene un ángulo de la base de
$70^\circ$: calcula su ángulo desigual.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:5}
¿Cuáles de estos tríos pueden ser los lados de un triángulo? Justifícalo con la
desigualdad triangular (¡el lado mayor!):
\[
(6,\ 8,\ 12); \qquad
(5,\ 5,\ 11); \qquad
(4,\ 9,\ 13); \qquad
(7,\ 7,\ 7).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:6}
Un triángulo tiene ángulos $x$, $2x$ y $3x$. Halla $x$ y los tres
ángulos. ¿Qué clase de triángulo es?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:7}
Construye un triángulo $ABC$ con $BC = 6$ cm,
$\widehat{ABC} = 50^\circ$ y $\widehat{ACB} = 60^\circ$ (transportador
en $B$ y en $C$). Antes de construir, predice la medida de
$\widehat{BAC}$ y compruébala después en tu figura.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:8}
Dos \omterm{def:g6:lines:perp}{rectas paralelas} son cortadas por una tercera recta, que forma un ángulo de
$54^\circ$ en el primer corte (entre la secante y una de las
\omterm{def:g6:lines:perp}{paralelas}). Dibuja la situación y da las medidas de los ocho
ángulos que se forman.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:9}
Dos lados de un triángulo miden $8$ cm y $3$ cm. ¿Entre qué dos
valores tiene que estar el tercer lado? Da una longitud entera que valga
y otra que no.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:10}
En un triángulo $ABC$, $\widehat A = 2\widehat B$ y
$\widehat C = 90^\circ$. Calcula $\widehat A$ y $\widehat B$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:triangles:11}
Los ángulos de un \omterm{def:g4:geometry:polygon}{cuadrilátero} $ABCD$ cualquiera se pueden estudiar
cortándolo por una diagonal en dos triángulos. Úsalo para demostrar que
los cuatro ángulos de todo \omterm{def:g4:geometry:polygon}{cuadrilátero} suman $360^\circ$ y
deduce la \omterm{def:g1:addition:def}{suma} de los ángulos de un \omterm{def:g4:geometry:polygon}{pentágono} (cinco lados).
\end{exercise}

\section{Problema: Dar la vuelta a la manzana}

\begin{problem}[{Problema de fin de semana --- los ángulos de un polígono cualquiera:
$(n-2) \times 180^\circ$ por dentro, siempre $360^\circ$ de giro
por fuera, y por qué solo tres figuras regulares embaldosan un suelo}]
\label{pb:g7:triangles:1}
El \Cref{exo:g7:triangles:11} cortó un \omterm{def:g4:geometry:polygon}{cuadrilátero} en dos triángulos
y obtuvo $360^\circ$. Este problema lleva la idea a \omterm{def:g4:geometry:polygon}{polígonos}
de cualquier número de lados, descubre después una segunda demostración completamente
distinta --- \emph{caminando} alrededor de la figura y sumando
los giros --- y termina en el suelo de tu \omterm{def:g3:division:half}{cuarto} de baño: entre todos los
\omterm{def:g4:geometry:polygon}{polígonos} regulares, exactamente tres pueden embaldosar el plano sin huecos ni
solapamientos, y las abejas ya han elegido el suyo.

\medskip
\noindent\textbf{Parte I --- Los ángulos de dentro.}
\begin{enumerate}
  \item desde un \omterm{def:g5:solids:def}{vértice} de un \omterm{def:g4:geometry:polygon}{hexágono} (seis lados), traza todas las
        diagonales que salen de ese \omterm{def:g5:solids:def}{vértice}. ¿En cuántos triángulos queda
        cortado el \omterm{def:g4:geometry:polygon}{hexágono} y cuál es la \omterm{def:g1:addition:def}{suma} de \emph{todos} sus
        ángulos interiores (\cref{thm:g7:triangles:sum})?
  \item generaliza: desde un \omterm{def:g5:solids:def}{vértice} de un \omterm{def:g4:geometry:polygon}{polígono} de $n$ lados,
        ¿cuántos triángulos produce el abanico de diagonales?
        Deduce la fórmula de la \omterm{def:g1:addition:def}{suma} de los ángulos interiores
        y compruébala para $n = 3$, $4$, $5$ y $6$;
  \item calcula la \omterm{def:g1:addition:def}{suma} de los ángulos interiores de un decágono
        ($10$ lados) y de un dodecágono ($12$ lados);
  \item en un \omterm{def:g4:geometry:polygon}{polígono} \emph{regular} todos los ángulos interiores son
        iguales. Calcula el ángulo interior del \omterm{def:g6:shapes:triangles}{triángulo
        equilátero}, del cuadrado, del \omterm{def:g4:geometry:polygon}{pentágono}, el \omterm{def:g4:geometry:polygon}{hexágono} y el
        \omterm{def:g4:geometry:polygon}{octógono} regulares;
  \item explica, sin ninguna fórmula, por qué la \omterm{def:g1:addition:def}{suma} de los ángulos crece
        exactamente $180^\circ$ cada vez que el \omterm{def:g4:geometry:polygon}{polígono} gana un
        \omterm{def:g5:solids:def}{vértice}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El paseo alrededor de la manzana.}
Camina por el borde de una manzana poligonal, siempre hacia delante. En
cada esquina \emph{giras} un cierto ángulo --- el \emph{ángulo
exterior} de esa esquina; los giros siguen hasta que vuelves al
punto de partida mirando en la dirección de partida.
\begin{enumerate}[resume]
  \item en una esquina cuyo ángulo interior es $\widehat A$, ¿cuánto
        giras? (El ángulo interior y el exterior son
        suplementarios.) Calcula
        los tres giros de un triángulo de ángulos $40^\circ$,
        $60^\circ$, $80^\circ$.
  \item \omterm{def:g1:addition:def}{suma} los tres giros de la pregunta 6. ¿Qué hecho sobre la vuelta
        completa observas?
  \item explica por qué los giros de un paseo completo alrededor de
        \emph{cualquier} \omterm{def:g4:geometry:polygon}{polígono} --- de tres lados o de treinta --- suman
        siempre exactamente $360^\circ$. (¿Qué ha hecho tu nariz, sumando todos
        los giros, cuando vuelves al principio?)
  \item deduce por segunda vez la fórmula de los ángulos interiores: en cada
        una de las $n$ esquinas, interior $+$ giro $= 180^\circ$; \omterm{def:g1:addition:def}{suma}
        sobre todas las esquinas y usa la pregunta 8;
  \item en un \omterm{def:g4:geometry:polygon}{polígono} regular todos los giros son iguales, así que cada
        ángulo exterior es $\frac{360^\circ}{n}$. Úsalo para
        responder al instante: ¿qué \omterm{def:g4:geometry:polygon}{polígono} regular tiene ángulos
        interiores de $150^\circ$? ¿Y por qué \emph{ningún}
        \omterm{def:g4:geometry:polygon}{polígono} regular tiene ángulos interiores de $155^\circ$?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Embaldosar el suelo.}
Alrededor de cada punto de un suelo embaldosado, las esquinas de las baldosas
que se juntan deben sumar exactamente $360^\circ$ --- sin huecos ni solapamientos
(\cref{pb:g6:lines:1}).
\begin{enumerate}[resume]
  \item para el \omterm{def:g6:shapes:triangles}{triángulo equilátero}, el cuadrado y el
        \omterm{def:g4:geometry:polygon}{hexágono} regular: ¿cuántas copias se juntan en un punto de esquina
        del embaldosado? Verifica los $360^\circ$ en cada caso;
  \item demuestra que los \omterm{def:g4:geometry:polygon}{pentágonos} regulares no pueden embaldosar el suelo: ¿cuánto
        suman tres esquinas y cuánto sumarían cuatro?
  \item demuestra que ningún \omterm{def:g4:geometry:polygon}{polígono} regular de \emph{más} de seis
        lados puede embaldosar: en cada esquina tienen que juntarse al menos tres baldosas
        (cada ángulo es menor que $180^\circ$), y ¿qué
        pasa con tres esquinas de $135^\circ$ o más? Concluye
        con la lista completa de las figuras regulares que embaldosan;
  \item los suelos de los \omterm{def:g3:division:half}{cuartos} de baño mezclan a menudo \omterm{def:g4:geometry:polygon}{octógonos} regulares con
        cuadrados pequeños. Verifica que esta esquina funciona:
        $135^\circ + 135^\circ + 90^\circ$. Otro clásico mezcla
        un cuadrado, un \omterm{def:g4:geometry:polygon}{hexágono} regular y un dodecágono regular en
        cada esquina: verifícalo también (el ángulo interior del
        dodecágono sale del método de la pregunta 10);
  \item la elección de las abejas: las celdas del panal son \omterm{def:g4:geometry:polygon}{hexágonos} regulares.
        Verifica sus esquinas ($3$ ángulos) y explica en una
        frase --- con la ayuda del descubrimiento de Dido
        (\cref{pb:g6:measure:1}) --- por qué, de las tres baldosas
        regulares posibles, el \omterm{def:g4:geometry:polygon}{hexágono} es la más barata en cera para
        la miel que guarda.
\end{enumerate}
\end{problem}
