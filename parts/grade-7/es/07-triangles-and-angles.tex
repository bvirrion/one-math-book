\chapter{Triángulos y ángulos}\label{ch:g7:triangles}

¿Puedes dibujar un triángulo con lados? $10$, $3$ y $2$? Con ángulos
$80^\circ $, $70^\circ $ y $50^\circ $? Este capítulo responde a ambos
preguntas con dos de los hechos más útiles de la geometría plana: el
desigualdad del triángulo y la $180^\circ $ ángulo \omterm{def:g1:addition:def}{suma} --- más el ángulo
vocabulario que viene con el cruce y \omterm{def:g6:lines:perp}{lineas paralelas}.

\section{Ángulos alrededor de líneas que se cruzan}

\begin{definition}[pares de ángulos]\label{def:g7:triangles:pairs}
\begin{itemize}
  \item Dos ángulos son \emph{complementario} cuando sus medidas suman
        a $90^\circ $, \emph{suplementario} cuando suman
        $180^\circ $.
  \item Cuando dos líneas se cruzan, los ángulos que se enfrentan a lo largo de la
        punto de cruce son \emph{verticalmente opuesto}\index{verticalmente opuesto}.
  \item Cuando una línea cruza otras dos líneas, los ángulos ubicados en la
        misma posición en cada cruce
        \emph{ángulos correspondientes}; ángulos entre las dos líneas en
        lados opuestos de la línea de cruce son \emph{ángulos alternos}\index{ángulos alternos}.
\end{itemize}
\end{definition}

\begin{theorem}[Igualdades de ángulos]\label{thm:g7:triangles:equalities}
\begin{enumerate}
  \item \omterm{def:g7:triangles:pairs}{Ángulos verticalmente opuestos} son iguales.
  \item Si dos líneas son \omterm{def:g6:lines:perp}{paralelo}, los ángulos correspondientes son iguales, y
        \omterm{def:g7:triangles:pairs}{ángulos alternos} son iguales. Por el contrario, igual correspondiente
        (o alternos) los ángulos obligan a las líneas a ser \omterm{def:g6:lines:perp}{paralelo}.
\end{enumerate}
\end{theorem}

\begin{proof}[Prueba de 1]
Los dos ángulos $\widehat{1}$ y $\widehat{2}$ a un lado de una línea
formar un ángulo recto: $\widehat 1 + \widehat 2 = 180^\circ $. El
\omterm{def:g7:triangles:pairs}{verticalmente opuesto} ángulo $\widehat 3$ de $\widehat 1$ también satisface
$\widehat 2 + \widehat 3 = 180^\circ $ (misma razón). Entonces
$\widehat 1 = 180^\circ - \widehat 2 = \widehat 3$. El punto 2 es
admitido en este nivel.
\end{proof}

\section{La suma de los ángulos de un triángulo.}

\begin{theorem}[Suma de ángulos]\label{thm:g7:triangles:sum}
En todo triángulo los tres ángulos suman $180^\circ $.
\end{theorem}

\begin{proof}
Sea $ ABC $ un triángulo. Dibuja la línea a través $ A $ \omterm{def:g6:lines:perp}{paralelo} a
$(a.\,C.)$. Al \omterm{def:g5:solids:def}{vértice} $ A $, tres ángulos se alinean a lo largo de este \omterm{def:g6:lines:perp}{paralelo}
y hacer un ángulo recto, $180^\circ $: el del medio es
$\widehat{A}$, y los dos exteriores son \omterm{def:g7:triangles:pairs}{ángulos alternos} con
$\widehat B $ y $\widehat C $ (\omterm{def:g6:lines:perp}{lineas paralelas}!), por lo que son iguales
$\widehat B $ y $\widehat C $. Por eso
$\widehat B + \widehat A + \widehat C = 180^\circ $.
\end{proof}

\begin{example}\label{ex:g7:triangles:sum}
Dos ángulos de un triángulo miden $67^\circ $ y $58^\circ $. el tercero
medidas
\[
180 - (67 + 58) = 180 - 125 = 55^\circ .
\]
Casos especiales que vale la pena memorizar: cada ángulo de un \omterm{def:g6:shapes:triangles}{triángulo equilatero}
es $180 \div 3 = 60^\circ $; en un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} los dos angulos agudos
son complementarios (suman $180 - 90 = 90^\circ $); la base
ángulos de un \omterm{def:g6:shapes:triangles}{triángulo isósceles} son iguales, por lo que cada uno es
$\frac{180 - \text{apex}}{2}$.
\end{example}

\begin{example}[Ningún triángulo con dos ángulos rectos.]\label{ex:g7:triangles:tworight}
Un triángulo con dos ángulos de $90^\circ $ ya se agotaría
$180^\circ $, partida $0^\circ $ para el tercero --- imposible. el angulo
\omterm{def:g1:addition:def}{suma} prohíbe muchos triángulos a la vez.
\end{example}

\section{La desigualdad del triángulo}

\begin{theorem}[Desigualdad triangular]\label{thm:g7:triangles:inequality}
En todo triángulo, cada lado es más corto que el \omterm{def:g1:addition:def}{suma} de los dos
otros: para un triángulo con lados $ a $, $ b $, $ c $,
\[
a < b + c .
\]
Por el contrario, tres longitudes con el \emph{más grande} más pequeño que
el \omterm{def:g1:addition:def}{suma}
de los otros dos siempre se puede formar un triángulo. si el mas grande
\emph{es igual} el \omterm{def:g1:addition:def}{suma} de los demás, el "triángulo" es plano: el
tres \omterm{def:g5:solids:def}{vértices} están alineados.
\end{theorem}
\admitted

\begin{example}\label{ex:g7:triangles:inequality}
¿Pueden los lados ser $10$, $3$ y $2$? Prueba el más grande: es $10 < 3 + 2 = 5$?
No: no existe tal triángulo. La brújula muestra por qué: arcos de radios
$3$ y $2$ extraído de los dos extremos de un \omterm{def:g6:lines:objects}{segmento} de longitud $10$ nunca
reunirse.

Lados $7$, $5$ y $4$? Más grande: $7 < 5 + 4 = 9$: si, el triángulo
existe (con comprobar el lado más grande es suficiente).
\end{example}

\begin{method}[Existencia y construcción de un triángulo.]\label{met:g7:triangles:construct}
Dadas tres longitudes:
\begin{enumerate}
  \item comparar el más grande con el \omterm{def:g1:addition:def}{suma} de los otros dos; si es
        no estrictamente más pequeño, parada: sin triángulo;
  \item de lo contrario construir como en \cref{met:g6:shapes:construct}
        (base \omterm{def:g6:lines:objects}{segmento}, dos arcos de compás);
  \item después de cualquier construcción desde ángulos, comprobar mentalmente que
        los ángulos dados suman menos que $180^\circ $ --- con el tercero
        ángulo haciendo exactamente $180^\circ $.
\end{enumerate}
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g7:triangles:1}
Dar el complemento (a $90^\circ $) y el suplemento (a
$180^\circ $) de: $30^\circ $; \ $45^\circ $; \ $72^\circ $.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:triangles:2}
Dos líneas se cruzan; uno de los cuatro angulos mide $35^\circ $. Dale el
medidas de los otros tres, con una razón para cada uno.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:triangles:3}
Calcula el tercer ángulo de un triángulo cuyos dos primeros ángulos miden:
(un) $40^\circ $ y $60^\circ $; \
(b) $90^\circ $ y $28^\circ $; \
(c) $75^\circ $ y $75^\circ $.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:triangles:4}
Un \omterm{def:g6:shapes:triangles}{triángulo isósceles} tiene su ángulo en el vértice igual a $40^\circ $. calcular
sus dos ángulos base. Otro \omterm{def:g6:shapes:triangles}{triángulo isósceles} tiene un ángulo base de
$70^\circ $: calcula su ángulo de vértice.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:triangles:5}
¿Cuáles de estos triples pueden ser los lados de un triángulo? Justificar por el
desigualdad del triángulo (¡lado más grande!):
\[
(6,\ 8,\ 12); \qquad
(5,\ 5,\ 11); \qquad
(4,\ 9,\ 13); \qquad
(7,\ 7,\ 7).
\]
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:triangles:6}
Un triángulo tiene angulos $ x $, $2x $ y $3x $. Encontrar $ x $ y los tres
ángulos. ¿Qué tipo de triángulo es?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:triangles:7}
construir un triángulo $ ABC $ con $ a.\,C. = 6$ centímetro,
$\widehat{ABC} = 50^\circ $ y $\widehat{ACB} = 60^\circ $ (transportador
en $ B $ y en $ C $). Antes de construir, prediga la medida de
$\widehat{BAC}$, luego revisa tu figura.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g7:triangles:8}
Dos \omterm{def:g6:lines:perp}{lineas paralelas} se cortan por una tercera línea, formando un ángulo de
$54^\circ $ en el primer cruce (entre la línea de cruce y un
\omterm{def:g6:lines:perp}{paralelo}). Dibuja la situación y da las medidas de los ocho.
ángulos formados.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g7:triangles:9}
Dos lados de un triángulo miden $8$ centímetros y $3$ centímetro. entre cuales dos
¿De qué valores debe estar el tercer lado? Dar una longitud de número entero que funcione
y otro que no.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g7:triangles:10}
en un triángulo $ ABC $, $\widehat A = 2\widehat B $ y
$\widehat C = 90^\circ $. Calcular $\widehat A $ y $\widehat B $.
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g7:triangles:11}
Los tres ángulos de cualquier \omterm{def:g4:geometry:polygon}{cuadrilátero} $ ABCD $ puede ser estudiado por
cortándolo a lo largo de una diagonal en dos triángulos. Usa esto para demostrar que
los cuatro ángulos de cada \omterm{def:g4:geometry:polygon}{cuadrilátero} sumar $360^\circ $, y
deducir el ángulo \omterm{def:g1:addition:def}{suma} de un \omterm{def:g4:geometry:polygon}{pentágono} (cinco lados).
\end{exercise}

\section{Problema: caminar alrededor de la cuadra}

\begin{problem}[{Problema de fin de semana --- los ángulos de cualquier polígono: $(n-2) \times 180^\circ$ adentro, siempre $360^\circ$ de giro afuera, y por qué solo tres formas regulares cubren un piso}]
\label{pb:g7:triangles:1}
\Cref{exo:g7:triangles:11} cortar un \omterm{def:g4:geometry:polygon}{cuadrilátero} en dos triangulos
y encontrado $360^\circ $. Este problema impulsa la idea de \omterm{def:g4:geometry:polygon}{polígonos}
con cualquier número de lados, luego descubre un segundo, completamente
prueba diferente --- por \emph{caminando} alrededor de la forma y agregando
sube las vueltas --- y termina en el suelo de tu baño: entre todos
regular \omterm{def:g4:geometry:polygon}{polígonos}, exactamente tres pueden revestir un plano sin espacios ni
se superponen y las abejas han elegido el suyo.

\medskip
\noindent\textbf{Parte I --- The angles inside.}
\begin{enumerate}
  \item De uno \omterm{def:g5:solids:def}{vértice} de un \omterm{def:g4:geometry:polygon}{hexágono} (seis lados), dibuja todos los
        diagonales dejando eso \omterm{def:g5:solids:def}{vértice}. en cuantos triangulos esta
        el \omterm{def:g4:geometry:polygon}{hexágono} corte, y cual es el \omterm{def:g1:addition:def}{suma} de \emph{todo} es
        ángulos interiores (\cref{thm:g7:triangles:sum})?
  \item Generalizar: de uno \omterm{def:g5:solids:def}{vértice} de un \omterm{def:g4:geometry:polygon}{polígono} con $ n $ lados,
        ¿cuantos triangulos produce el abanico de diagonales?
        Deduzca la fórmula para \omterm{def:g1:addition:def}{suma} de los ángulos interiores,
        y compruébalo para $ n = 3$, $4$, $5$, $6$.
  \item Calcular el \omterm{def:g1:addition:def}{suma} de los ángulos interiores de un decágono
        ($10$ lados) y de un dodecágono ($12$ lados).
  \item en un \emph{regular} \omterm{def:g4:geometry:polygon}{polígono} todos los ángulos interiores son
        igual. Calcula el ángulo interior del \omterm{def:g6:shapes:triangles}{triángulo equilatero}, el cuadrado, el regular \omterm{def:g4:geometry:polygon}{pentágono}, \omterm{def:g4:geometry:polygon}{hexágono} y
        \omterm{def:g4:geometry:polygon}{octágono}.
  \item Explica, sin ninguna fórmula, por qué el ángulo \omterm{def:g1:addition:def}{suma} crece por
        exactamente $180^\circ $ cada vez que el \omterm{def:g4:geometry:polygon}{polígono} gana uno
        \omterm{def:g5:solids:def}{vértice}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- The walk around the block.}
Camine por el límite de un bloque poligonal, siempre hacia adelante. en
cada rincón tu \emph{doblar} por algún ángulo --- el \emph{exteriores
ángulo} de ese rincón; el giro continúa hasta regresar a
su punto de partida, mirando hacia su dirección inicial.
\begin{enumerate}[resume]
  \item En una esquina cuyo ángulo interior es $\widehat A $, por como
        ¿Cuánto giras? (Los ángulos interior y exterior son
        suplementario.) Calcular
        las tres vueltas de un triángulo con angulos $40^\circ $,
        $60^\circ $, $80^\circ $.
  \item Sume los tres giros de la pregunta 6. ¿Qué círculo completo
        hecho observas?
  \item Explica por qué las vueltas de un paseo completo
        \emph{cualquier} \omterm{def:g4:geometry:polygon}{polígono} --- tres lados o treinta --- siempre
        total exactamente $360^\circ $. (¿Qué te ha hecho la nariz, todos?
        giros combinados, cuando regreses?)
  \item Deduzca la fórmula del ángulo interior por segunda vez: en cada
        de la $ n $ esquinas, interior $+$ doblar $= 180^\circ $; \omterm{def:g1:addition:def}{suma}
        en todas las esquinas y utilice la pregunta 8.
  \item en un regular \omterm{def:g4:geometry:polygon}{polígono}, todas las vueltas son iguales, por lo que cada
        El ángulo exterior es $\frac{360^\circ}{n}$. Usa esto para
        Responde al instante: ¿Qué regular? \omterm{def:g4:geometry:polygon}{polígono} tiene interior
        ángulos de $150^\circ $? ¿Y por qué \emph{No} regular
        \omterm{def:g4:geometry:polygon}{polígono} tener ángulos interiores de $155^\circ $?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Tiling the floor.}
Alrededor de cada punto de un suelo de baldosas, las esquinas del encuentro
los mosaicos deben sumar exactamente $360^\circ $ --- sin espacios, sin superposiciones
(\cref{pb:g6:lines:1}).
\begin{enumerate}[resume]
  \item Para cada uno de los \omterm{def:g6:shapes:triangles}{triángulo equilatero}, el cuadrado y el
        regular \omterm{def:g4:geometry:polygon}{hexágono}: cuantas copias se encuentran en un punto de esquina
        del mosaico? Verificar el $360^\circ $ siempre.
  \item Muestra eso regularmente \omterm{def:g4:geometry:polygon}{pentágonos} no puedo embaldosar el piso: ¿qué?
        ¿Hacen tres esquinas en total y cuáles serían cuatro en total?
  \item Mostrar que no hay regular \omterm{def:g4:geometry:polygon}{polígono} con \emph{más} de seis
        Los lados pueden colocar mosaicos: al menos tres mosaicos deben encontrarse en cada uno.
        esquina (cada ángulo es menor que $180^\circ $), y que
        sucede en tres esquinas de $135^\circ $ o más? Concluir
        la lista completa de alicatadores habituales.
  \item Los pisos del baño a menudo mezclan regular \omterm{def:g4:geometry:polygon}{octágonos} con pequeño
        cuadrados. Verifica que este rincón funcione:
        $135^\circ + 135^\circ + 90^\circ $. Otra mezcla clásica
        un cuadrado, un regular \omterm{def:g4:geometry:polygon}{hexágono} y un dodecágono regular en
        cada esquina: verifíquelo también (el interior del dodecágono
        ángulo se desprende del método de la pregunta 10).
  \item La elección de las abejas: las células del panal son regulares \omterm{def:g4:geometry:polygon}{hexágonos}.
        Verifique sus esquinas ($3$ ángulos), y explicar en uno
        frase --- con la ayuda del descubrimiento de Dido
        (\cref{pb:g6:measure:1}) --- por qué, de los tres posibles
        azulejos regulares, el \omterm{def:g4:geometry:polygon}{hexágono} es la cera más barata para
        la miel que contiene.
\end{enumerate}
\end{problem}
