\chapter{Jerarquía de las operaciones}\label{ch:g7:priorities}

¿Cuánto vale $3 + 4 \times 5$? Si calculas de izquierda a derecha obtienes $35$;
si multiplicas primero obtienes $23$. Las matemáticas necesitan una única
respuesta, así que hay reglas de tráfico para los cálculos --- las prioridades.
Este capítulo las enuncia y presenta una primera identidad algebraica,
la distributividad.

\section{Las reglas de prioridad}

\begin{theorem}[Reglas de prioridad]\label{thm:g7:priorities:rules}
En un cálculo sin paréntesis:
\begin{enumerate}
  \item las multiplicaciones y las divisiones se hacen \emph{antes} que
        las \omterm{def:g1:addition:def}{sumas} y las restas;
  \item las operaciones del mismo nivel se hacen de izquierda a derecha.
\end{enumerate}
Los paréntesis mandan sobre todo lo demás: lo que hay dentro de un paréntesis se calcula
primero, empezando por los más interiores.
\end{theorem}

\begin{example}\label{ex:g7:priorities:basic}
Un paso por línea, subrayando (mentalmente) lo que se calcula:
\begin{align*}
3 + 4 \times 5 &= 3 + 20 = 23
&&\text{(primero la multiplicación)}\\
20 - 8 + 2 &= 12 + 2 = 14
&&\text{(mismo nivel: de izquierda a derecha)}\\
18 \div 3 \times 2 &= 6 \times 2 = 12
&&\text{(mismo nivel: de izquierda a derecha)}\\
(3 + 4) \times 5 &= 7 \times 5 = 35
&&\text{(primero los paréntesis).}
\end{align*}
Fíjate en las trampas: $20 - 8 + 2$ \emph{no} es $20 - 10$, y
$18 \div 3 \times 2$ \emph{no} es $18 \div 6$.
\end{example}

\begin{example}[Paréntesis anidados]\label{ex:g7:priorities:nested}
\begin{align*}
50 - 2 \times \bigl(3 + (10 - 4)\bigr)
&= 50 - 2 \times (3 + 6) && \text{(el paréntesis interior)}\\
&= 50 - 2 \times 9 && \text{(el paréntesis restante)}\\
&= 50 - 18 && \text{(la multiplicación)}\\
&= 32 .
\end{align*}
\end{example}

\begin{remark}[La barra de fracción esconde paréntesis]
La barra de una \omterm{def:g6:fractions:def}{fracción} agrupa su \omterm{def:g4:fractions:def}{numerador} y su \omterm{def:g4:fractions:def}{denominador}, como si cada uno
llevara paréntesis invisibles:
\[
\frac{12 + 8}{4} = \frac{20}{4} = 5,
\qquad\text{y no}\quad 12 + \frac{8}{4} = 14 .
\]
\end{remark}

\begin{method}[Calcular sin errores]\label{met:g7:priorities:safe}
\begin{enumerate}
  \item localiza la operación que hay que hacer primero (los paréntesis más interiores, luego
        $\times$ y $\div$, y luego $+$ y $-$);
  \item calcula \emph{solo esa} y reescribe la expresión entera
        con el resultado en su sitio;
  \item repite hasta que quede un solo número --- una operación por línea, sin
        atajos.
\end{enumerate}
\end{method}

\section{La distributividad}

\begin{theorem}[Distributividad]\label{thm:g7:priorities:distributivity}
Para todos los números $k$, $a$ y $b$:
\[
k \times (a + b) = k \times a + k \times b,
\qquad
k \times (a - b) = k \times a - k \times b .
\]
\end{theorem}

\begin{proof}[Demostración con un dibujo]
Cuenta el \omterm{def:g6:measure:area}{área} de un rectángulo de ancho $k$ partido en dos rectángulos
menores de largos $a$ y $b$: entero vale $k \times (a+b)$;
en dos partes vale $k \times a + k \times b$. El mismo rectángulo, la misma
\omterm{def:g6:measure:area}{área}.
\end{proof}

\begin{omfigure}
\begin{tikzpicture}[scale=0.75]
  \fill[omDef!14] (0,0) rectangle (4.2,1.8);
  \fill[omThm!14] (4.2,0) rectangle (6.4,1.8);
  \draw[thick] (0,0) rectangle (6.4,1.8);
  \draw[thick] (4.2,0) -- (4.2,1.8);
  \node[below, font=\small] at (2.1,0) {$a$};
  \node[below, font=\small] at (5.3,0) {$b$};
  \node[left, font=\small] at (0,0.9) {$k$};
  \node[font=\small] at (2.1,0.9) {$k \times a$};
  \node[font=\small] at (5.3,0.9) {$k \times b$};
\end{tikzpicture}

{\small La distributividad como \omterm{def:g6:measure:area}{áreas}: el rectángulo grande
$k \times (a + b)$ es la \omterm{def:g1:addition:def}{suma} de los dos pequeños.}
\end{omfigure}

\begin{example}[Los dos sentidos]\label{ex:g7:priorities:distributivity}
\emph{Desarrollar} (de izquierda a derecha) facilita el cálculo mental:
\[
7 \times 103 = 7 \times (100 + 3) = 700 + 21 = 721,
\qquad
6 \times 98 = 6 \times (100 - 2) = 600 - 12 = 588 .
\]
\emph{Factorizar} (de derecha a izquierda) saca un factor común:
\[
23 \times 12 + 23 \times 8 = 23 \times (12 + 8) = 23 \times 20 = 460 .
\]
\end{example}

\begin{example}[Traducir palabras a un cálculo]\label{ex:g7:priorities:words}
«La \omterm{def:g1:addition:def}{suma} de $8$ y el \omterm{def:g2:mult:def}{producto} de $5$ por $3$» es $8 + 5 \times 3 =
23$. «El \omterm{def:g2:mult:def}{producto} de $5$ por la \omterm{def:g1:addition:def}{suma} de $8$ y $3$» es
$5 \times (8 + 3) = 55$: las palabras «el \omterm{def:g2:mult:def}{producto} de \dots\ por la \omterm{def:g1:addition:def}{suma}
de\dots» ponen paréntesis alrededor de la \omterm{def:g1:addition:def}{suma}.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g7:priorities:1}
Calcula, una operación por línea:
\[
5 + 3 \times 6, \qquad
(5 + 3) \times 6, \qquad
30 - 12 \div 4, \qquad
(30 - 12) \div 4 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:2}
Calcula:
\[
24 - 6 + 2, \qquad
24 \div 6 \times 2, \qquad
24 - 6 \times 2, \qquad
24 \div (6 \times 2).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:3}
Calcula paso a paso:
\[
7 + 3 \times (10 - 6), \qquad
40 - (2 + 3) \times 4, \qquad
60 \div \bigl(2 \times (7 - 4)\bigr).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:4}
Calcula las fracciones (¡paréntesis invisibles!):
\[
\frac{18 + 6}{8}, \qquad
\frac{30}{7 - 2}, \qquad
\frac{5 \times 8}{4 + 6} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:5}
Usa la distributividad para calcular de cabeza:
\[
8 \times 102, \qquad
25 \times 99, \qquad
14 \times 12 + 14 \times 8, \qquad
37 \times 45 - 37 \times 35 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:6}
Escribe el cálculo en una sola línea, con paréntesis donde haga falta, y
calcúlalo: «la \omterm{ex:g1:subtraction:difference}{diferencia} de $50$ y el \omterm{def:g2:mult:def}{producto} de $6$ por $7$»;
«el \omterm{def:g2:mult:def}{producto} de la \omterm{def:g1:addition:def}{suma} de $4$ y $5$ por la \omterm{ex:g1:subtraction:difference}{diferencia} de $10$ y
$8$''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:7}
Amir compró $3$ cuadernos a $2.50$ cada uno y $2$ bolígrafos a $1.20$ cada uno.
Escribe su total en una sola expresión y calcúlalo después.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:priorities:8}
Coloca paréntesis en $4 + 2 \times 5 - 1$ para que el resultado sea: (a) $13$;
(b) $29$; (c) $24$. (Uno de los tres no necesita ningún paréntesis ---
¿cuál?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:priorities:9}
¿Son $12 + 4 \times k$ y $16 \times k$ lo mismo para todo número $k$?
Pruébalo con $k = 1$ y $k = 2$ y saca una conclusión. ¿Qué error de prioridad
llevaría de uno al otro?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:priorities:10}
Una entrada de cine cuesta $9$ euros, pero un grupo de $n$ personas paga una tarifa fija
de $12$ euros más $6$ euros por persona.
\begin{enumerate}
  \item escribe una expresión para el precio del grupo y calcúlalo para
        $n = 5$.
  \item ¿a partir de cuántas personas sale más barata la tarifa de grupo que $n$
        entradas individuales? (Prueba con algunos valores de $n$.)
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:priorities:11}
Usando cada una de las cifras $1$, $2$, $3$ y $4$ exactamente una vez, las cuatro
operaciones y todos los paréntesis que quieras, escribe expresiones iguales a $24$, a
$10$ y a $1$.
\end{exercise}

\section{Problema: Multiplicar como un escriba egipcio}

\begin{problem}[{Problema de fin de semana --- trucos de cálculo mental que salen de
la distributividad, y la multiplicación por duplicación y mitades usada durante
cuatro mil años}]
\label{pb:g7:priorities:1}
Mucho antes de las calculadoras --- y antes de que las tablas de multiplicar se
machacaran en la escuela ---, los escribas egipcios y los campesinos rusos
multiplicaban dos números cualesquiera usando solo \emph{duplicar, partir por la \omterm{def:g3:division:half}{mitad} y
sumar}. Su método parece magia; en realidad es la
distributividad (\cref{thm:g7:priorities:distributivity}) trabajando
a pleno rendimiento. Este problema afila tu cálculo mental, después
te enseña el método antiguo y luego levanta el capó para ver exactamente
por qué funciona.

\medskip
\noindent\textbf{Parte I --- El arte de no calcular.}
\begin{enumerate}
  \item calcula de cabeza, desarrollando como en el
        \cref{ex:g7:priorities:distributivity}:
        $7 \times 102$ y $9 \times 98$;
  \item calcula de cabeza, factorizando:
        $17 \times 13 + 17 \times 7$ y
        $45 \times 101 - 45$.
  \item el truco del «por $5$»: para multiplicar por $5$, multiplica por
        $10$ y parte por la \omterm{def:g3:division:half}{mitad}. Calcula $5 \times 87$ así y
        explica por qué el truco es legítimo;
  \item dos trucos más para justificar y usar: multiplicar por $25$
        multiplicando por $100$ y dividiendo entre $4$ (calcula
        $25 \times 32$); multiplicar por $99$ multiplicando por $100$
        y restando una vez (calcula $99 \times 47$);
  \item un aviso: la distributividad reparte un factor sobre una
        \emph{\omterm{def:g1:addition:def}{suma}}, nunca sobre un \omterm{def:g2:mult:def}{producto}. Compara
        $8 \times (5 + 3)$ con $8 \times 5 + 8 \times 3$, y
        después $8 \times (5 \times 3)$ con
        $(8 \times 5) \times (8 \times 3)$: ¿qué pareja coincide
        y cuál no?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El método de duplicar y partir por la \omterm{def:g3:division:half}{mitad}.}
Esta es la receta para $13 \times 24$. Escribe dos columnas. Columna
izquierda: empieza en $13$ y \emph{parte por la \omterm{def:g3:division:half}{mitad}}, tirando cualquier \omterm{def:g3:division:remainder}{resto},
hasta llegar a $1$: \ $13,\ 6,\ 3,\ 1$. Columna derecha: empieza en $24$ y
\emph{duplica} otras tantas veces: \ $24,\ 48,\ 96,\ 192$. Tacha
todas las filas cuyo número de la \emph{izquierda} sea \omterm{def:g3:numbers:evenodd}{par} y \omterm{def:g1:addition:def}{suma} los
números de la derecha que sobrevivan.
\begin{enumerate}[resume]
  \item aplica la receta a $13 \times 24$: ¿qué fila se
        tacha, qué \omterm{def:g1:addition:def}{suma} queda y coincide con
        $13 \times 24$?
  \item multiplica $21 \times 17$ con el método (parte $21$ por la \omterm{def:g3:division:half}{mitad} y duplica
        $17$) y comprueba la respuesta con la multiplicación ordinaria;
  \item multiplica $16 \times 25$ con el método. ¿Cuántas filas
        sobreviven y qué tiene de especial una columna izquierda que
        empieza en $1$, $2$, $4$, $8$, $16, \dots$?
  \item ahora, \emph{por qué} funciona, paso a paso. Justifica cada
        igualdad con la distributividad:
        \[
        13 \times 24 = (12 + 1) \times 24
        = 12 \times 24 + 24 = 6 \times 48 + 24 .
        \]
        Moraleja: partir por la \omterm{def:g3:division:half}{mitad} la izquierda y duplicar la derecha conserva el
        \omterm{def:g2:mult:def}{producto} --- pero, cuando el número de la izquierda es \emph{\omterm{def:g3:numbers:evenodd}{impar}}, se
        desgaja una copia del número de la derecha que hay que guardar.
        ¿Qué número guarda la primera fila de la tabla?
  \item sigue la contabilidad tabla abajo:
        $6 \times 48 = 3 \times 96$, después
        $3 \times 96 = 1 \times 192 + 96$, después
        $1 \times 192 = 192$. Reúne todo lo que se ha ido guardando
        por el camino y explica por qué los números guardados son
        exactamente los de la columna derecha de las filas con izquierda \omterm{def:g3:numbers:evenodd}{impar}
        --- las filas que la receta conserva.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Bajo el capó: \omterm{def:g1:addition:def}{sumas} de
duplicaciones.}
\begin{enumerate}[resume]
  \item en la pregunta 6, las filas supervivientes valían $1$, $4$
        y $8$ duplicaciones de $24$: el método escribió en silencio
        $13 \times 24 = (1 + 4 + 8) \times 24$. Comprueba que
        $1 + 4 + 8 = 13$ y halla qué duplicaciones ($1$, $2$,
        $4$, $8$, $16$) conservó el método para $21$ en la pregunta 7;
  \item escribe $25$ y $42$ como \omterm{def:g1:addition:def}{sumas} de números de la lista de
        duplicaciones $1, 2, 4, 8, 16, 32$, usando cada uno como mucho una vez;
  \item explica por qué \emph{todo} número entero se puede escribir
        así: toma el mayor número de duplicación que quepa,
        resta y repite --- ¿por qué tiene que pararse esto? Escribe $100$ como
        una \omterm{def:g1:addition:def}{suma} así;
  \item los escribas de Egipto hacían exactamente esto hace cuatro mil
        años: para multiplicar $19 \times 35$, hacían una
        tabla de duplicaciones de $35$ ($1 \to 35$, $2 \to 70$,
        $4 \to 140$, $8 \to 280$, $16 \to 560$), elegían las filas
        que sumaban $19$ y las sumaban. Haz el cálculo a su
        manera;
  \item final: los ordenadores modernos, que cuentan con $0$ y
        $1$, multiplican esencialmente como los escribas --- a base de
        duplicaciones y \omterm{def:g1:addition:def}{sumas}. ¿Qué tres \omterm{def:g1:addition:def}{sumas} haría un
        ordenador para $13 \times 24$? (Por qué todo número
        se descompone en duplicaciones, y qué tienen que ver los $0$ y los $1$
        con ello, es el problema de fin de semana del capítulo de las potencias:
        \cref{pb:g8:powers:1}.)
\end{enumerate}
\end{problem}
