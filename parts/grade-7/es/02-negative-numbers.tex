\chapter{Los números relativos}\label{ch:g7:negatives}

Temperaturas bajo \omterm{def:g1:counting:zero}{cero}, plantas por debajo del suelo, dinero que se debe: muchas
magnitudes bajan de $0$. Los números negativos prolongan la recta numérica hacia la
izquierda del \omterm{def:g1:counting:zero}{cero}. Este capítulo enseña a compararlos y a sumarlos
y restarlos --- su multiplicación espera al
\cref{ch:g8:negprod}.

\section{Los números relativos}

\begin{definition}[Números relativos]\label{def:g7:negatives:def}
Un \emph{número relativo}\index{número relativo} está formado por un signo
($+$ o $-$) y una \emph{distancia al \omterm{def:g1:counting:zero}{cero}}: $+3$ y $-3$ están los dos a
distancia $3$ de $0$, en lados opuestos. Los números positivos se escriben normalmente
sin su signo: $+3 = 3$. Los números $+3$ y $-3$ son
\emph{opuestos}\index{opuesto} el uno del otro.
\end{definition}

\begin{method}[Comparar números relativos]\label{met:g7:negatives:compare}
En la recta numérica, menor significa más a la izquierda. En la práctica:
\begin{enumerate}
  \item un número negativo es siempre menor que uno positivo:
        $-7 < 2$;
  \item entre dos positivos, el orden de siempre;
  \item entre dos negativos, el de \emph{mayor} distancia
        al \omterm{def:g1:counting:zero}{cero} es el \emph{menor}: $-7 < -2$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:negatives:compare}
Ordena de menor a mayor: $3$; $-5$; $0$; $-1.5$; $2.8$.
Primero los negativos, encabezados por el más negativo:
\[
-5 < -1.5 < 0 < 2.8 < 3 .
\]
\end{example}

\section{Sumar números relativos}

Piensa en un \omterm{def:g7:negatives:def}{número relativo} como un movimiento a lo largo de la recta: $+5$ significa
«camina $5$ pasos a la derecha» y $-3$, «camina $3$ pasos a la izquierda». Sumar
números es encadenar los paseos.

\begin{method}[Sumar dos números relativos]\label{met:g7:negatives:add}
\begin{enumerate}
  \item \emph{Mismo signo}: \omterm{def:g1:addition:def}{suma} las distancias y conserva el signo común:
        $(-3) + (-4) = -7$.
  \item \emph{signos opuestos}: resta la distancia menor de la
        mayor y toma el signo de la mayor:
        $(-7) + (+2) = -5$ y $(+7) + (-2) = +5$;
  \item los opuestos se anulan: $(+4) + (-4) = 0$.
\end{enumerate}
\end{method}

  \foreach \x in {-5,...,4} \draw (\x,-0.07) -- (\x,0.07);

  \node[omDef, font=\small] at (-2.5,1.0) {$+2$};
  \fill (-1.5,0) circle (2pt) node[below=6pt, font=\small] {$-1.5$};
\end{tikzpicture}

{\small Sumar como caminar: saliendo de $0$, el paseo $-3.5$ y después el
paseo $+2$ llevan a $(-3.5) + 2 = -1.5$.}
\end{omfigure}

\begin{example}\label{ex:g7:negatives:add}
Paso a paso, agrupando como convenga:
\begin{align*}
(-8) + (+3) &= -5 && \text{(signos opuestos: } 8 - 3 = 5,
\text{ signo del } 8)\\
(-2.5) + (-1.5) &= -4 && \text{(mismo signo: se suman las distancias)}\\
(+6) + (-9) + (+3) &= (-3) + (+3) = 0 .
\end{align*}
\end{example}

\section{Restar números relativos}

\begin{theorem}[Regla de la resta]\label{thm:g7:negatives:sub}
Restar un \omterm{def:g7:negatives:def}{número relativo} es lo mismo que \emph{sumar su
opuesto}:
\[
a - b = a + (-b) .
\]
\end{theorem}

\begin{proof}[Por qué funciona]
$a - b$ es el número que, sumado a $b$, da $a$. Comprobemos que
$a + (-b)$ cumple ese papel:
$\bigl(a + (-b)\bigr) + b = a + \bigl((-b) + b\bigr) = a + 0 = a$.
\end{proof}

\begin{example}\label{ex:g7:negatives:sub}
Toda resta se convierte en una \omterm{def:g1:addition:def}{suma} y entonces se aplica el
\cref{met:g7:negatives:add}:
\begin{align*}
5 - 9 &= 5 + (-9) = -4, \\
(-3) - (+4) &= (-3) + (-4) = -7, \\
(-2) - (-6) &= (-2) + (+6) = +4 .
\end{align*}
La última línea es la famosa: \emph{restar un negativo es
sumar}. ¡Quitar una deuda te hace más rico!
\end{example}

\begin{example}[Diferencias de temperatura]\label{ex:g7:negatives:temperature}
En un día la temperatura pasó de $-6$ grados a $+9$ grados. La
subida es la \omterm{ex:g1:subtraction:difference}{diferencia}
\[
9 - (-6) = 9 + 6 = 15 \text{ grados}.
\]
En la recta numérica: de $-6$ hasta $0$ hay $6$ grados y de $0$ a
$9$ hay $9$ más.
\end{example}

\section{Coordenadas en todo el plano}

\begin{definition}[Coordenadas con signo]\label{def:g7:negatives:coords}
Con dos ejes graduados \omterm{def:g6:lines:perp}{perpendiculares} que se cortan en el origen $O$,
cada punto del plano recibe dos coordenadas relativas $(x, y)$:
una $x$ negativa significa a la izquierda del eje vertical y una $y$ negativa, por debajo
del eje horizontal.
\end{definition}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g7:negatives:1}
Di el opuesto de: $7$; \ $-3.5$; \ $0$; \ $+12$. Después completa:
«dos números son opuestos cuando su \omterm{def:g1:addition:def}{suma} es~?».
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:2}
Copia y completa con $<$ o $>$:
\[
-4 \;?\; 1, \qquad
-4 \;?\; -7, \qquad
-2.5 \;?\; -2.4, \qquad
0 \;?\; -0.1 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:3}
Ordena de menor a mayor: $-3$; \ $2.5$; \ $-0.5$; \ $-3.1$;
\ $0$; \ $1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:4}
Calcula:
\[
(-5) + (-9), \qquad
(-12) + (+7), \qquad
(+3.5) + (-1.5), \qquad
(-6) + (+6).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:5}
Transforma en \omterm{def:g1:addition:def}{sumas} y calcula después:
\[
4 - 11, \qquad
(-5) - (+3), \qquad
(-7) - (-10), \qquad
0 - (-8).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:6}
Calcula agrupando con astucia (¡primero los opuestos!):
\[
(+7) + (-12) + (+3) + (-8), \qquad
(-5.5) + (+9) + (-4.5) + (+1).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:7}
Esta mañana la temperatura era de $-3$ grados; subió $8$ grados
durante el día y bajó $12$ grados por la noche. Escribe un solo
cálculo que dé la temperatura final y calcúlalo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:8}
Sitúa en unos ejes de coordenadas: $A(2, 3)$; $B(-3, 1)$; $C(-2, -2)$;
$D(3, -1)$; $E(0, -3)$. ¿Qué punto está sobre un eje?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:9}
Calcula la \omterm{ex:g1:subtraction:difference}{diferencia} entre la temperatura máxima y la mínima:
Moscú en enero, máxima $-4$ grados, mínima $-13$ grados; Helsinki, máxima
$2$, mínima $-9$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:10}
Una buceadora está a una altitud de $-12$ m (doce metros por debajo de la superficie).
Sube $7$ m y después baja $4$ m. ¿A qué altitud está? ¿Cuánto
le falta por subir para llegar a la superficie?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:11}
Sitúa $A(-2, 1)$, $B(2, 1)$ y $C(2, -2)$. Halla las coordenadas del
punto $D$ para que $ABCD$ sea un rectángulo y calcula su \omterm{def:g6:measure:perimeter}{perímetro}
(cuenta las casillas de la cuadrícula para las longitudes de los lados).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:negatives:12}
En el cuadrado mágico de abajo, cada fila, cada columna y cada diagonal deben tener la
misma \omterm{def:g1:addition:def}{suma}. Cópialo y complétalo:
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|}
\hline
$?$ & $-7$ & $0$ \\
\hline
$-3$ & $-2$ & $?$ \\
\hline
$-4$ & $?$ & $?$ \\
\hline
\end{tabular}
\end{center}
(Empieza hallando la \omterm{def:g1:addition:def}{suma} mágica con la diagonal que está casi
completa.)
\end{exercise}

\section{Problema: El año cero que nunca existió}

\begin{problem}[{Problema de fin de semana --- los números relativos en la
línea del tiempo de la historia: duraciones que cruzan el cambio de era, el
truco de los astrónomos y el principio de la variación neta}]
\label{pb:g7:negatives:1}
La recta numérica de este capítulo tiene un gemelo famoso en el mundo real: la
línea del tiempo de la historia, con los años a.\,C. extendiéndose hacia la izquierda y los
años d.\,C. hacia la derecha. Pero la recta de los historiadores esconde una trampa
que ha estropeado muchos cálculos: \emph{no hay año \omterm{def:g1:counting:zero}{cero}}
--- al año 1 a.\,C. le sigue inmediatamente el 1 d.\,C. Este problema
calcula con temperaturas, ascensores y cuentas bancarias, y después repara
el \omterm{def:g1:counting:zero}{cero} roto de la historia como hacen los astrónomos: con \omterm{def:g7:negatives:def}{números
relativos}.

\medskip
\noindent\textbf{Parte I --- Calentamiento y una primera trampa.}
\begin{enumerate}
  \item la cima del Mont Blanc está a una altitud de $+4\,808$ m; la
        orilla del mar Muerto, a $-430$ m. Calcula la \omterm{ex:g1:subtraction:difference}{diferencia}
        de altitud entre las dos;
  \item una cuenta bancaria tiene $-45$ euros (un descubierto). Su
        titular ingresa $120$ euros, paga una factura de $60$ euros
        y después ingresa $15$ euros. Escribe el saldo como una sola
        cadena de \omterm{def:g1:addition:def}{sumas} y calcúlalo. ¿Qué ingreso único
        habría llevado los $-45$ iniciales exactamente a
        $0$?
  \item Julio César nació en el $100$ a.\,C. y murió en el $44$ a.\,C.
        Contando los años a.\,C. como números negativos (el $100$ a.\,C. como
        $-100$ y el $44$ a.\,C. como $-44$), calcula $(-44) - (-100)$: ¿cuántos
        años vivió César?
  \item el emperador Augusto nació en el $63$ a.\,C. y murió en el $14$
        d.\,C. La misma receta da $14 - (-63) = 77$ años ---
        pero los historiadores insisten en que murió con $76$. Su objeción:
        el recuento $77$ atraviesa un año que nunca existió.
        ¿Cuál?
  \item los astrónomos reparan la línea del tiempo cambiando las etiquetas: los años d.\,C.
        conservan su número (el $+14$ sigue siendo $+14$), pero el $1$ a.\,C.
        pasa a ser $0$, el $2$ a.\,C. pasa a ser $-1$ y, en general, el año
        $n$ a.\,C. pasa a ser $-(n - 1)$. Convierte el $63$ a.\,C. y rehaz
        el cálculo de la pregunta 4 con los números de los astrónomos.
        ¿Satisface ahora la respuesta a los historiadores?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Calcular a través de las eras.}
A partir de ahora, usa el cambio de etiquetas de los astrónomos en todos los
cálculos y vuelve a a.\,C./d.\,C. al final (un año etiquetado
$0$ o negativo $-m$ es el año $m + 1$ a.\,C. de los historiadores).
\begin{enumerate}[resume]
  \item convierte a números de astrónomo: $1$ a.\,C.; $10$ a.\,C.; $476$
        d.\,C.; y $753$ a.\,C., la fundación legendaria de Roma;
  \item ¿cuántos años pasaron desde la fundación de Roma
        ($753$ a.\,C.) hasta la caída del Imperio de Occidente ($476$ d.\,C.)?
  \item un \omterm{pb:g6:shapes:1}{cometa} vuelve cada $76$ años y apareció en el
        $12$ a.\,C. Di los años de los historiadores de sus tres siguientes
        apariciones y el de la aparición justo anterior al $12$ a.\,C.;
  \item ¿cuántos años pasaron del $5$ a.\,C. al $5$ d.\,C.? (La
        respuesta tentadora es $10$; la línea del tiempo dice otra cosa.)
  \item un niño nacido en el $3$ a.\,C. cumplió diez años ¿en qué
        año (a.\,C. o d.\,C.)?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El principio de la variación neta.}
\begin{enumerate}[resume]
  \item la \emph{distancia} entre dos números de la recta es
        el mayor menos el menor --- siempre positiva.
        Calcula la distancia entre $-7$ y $-2$, y después entre
        $-3.5$ y $4.5$;
  \item un ascensor sale de la planta $0$ y hace los movimientos $+3$,
        $-5$, $+2$, $-1$, $+4$. Calcula la planta final
        agrupando con astucia (\cref{met:g7:negatives:add},
        primero los opuestos cuando se pueda). ¿Cambiaría el destino hacer
        los movimientos en otro orden? ¿Por qué?
  \item día a día, la temperatura varía en $+2$, $-4$, $+1$,
        $0$, $-3$, $+5$ y $-2$ grados a lo largo de una semana. Calcula la
        \emph{variación neta}. Si la semana empezó a $-2$ grados,
        ¿en cuánto terminó? Enuncia el principio: valor final $=$
        valor inicial $+$ (\omterm{def:g1:addition:def}{suma} de todas las variaciones);
  \item el paseo de una senderista la devuelve exactamente a su altitud
        de partida. ¿Cuánto tiene que valer la \omterm{def:g1:addition:def}{suma} de todas sus variaciones de altitud
        y por qué? De sus seis variaciones registradas, cinco son
        $+4$, $-7$, $+2$, $+3$ y $-5$ (en centenares de metros):
        halla la sexta;
  \item el final, de vuelta en Roma. Un historiador calcula: «la
        República fue del $509$ a.\,C. al $27$ a.\,C., y después el Imperio
        hasta el $476$ d.\,C.; en total, del $509$ a.\,C. al $476$ d.\,C., es decir,
        $509 + 476 = 985$ años de Estado romano». Rehaz el
        cálculo con los números de los astrónomos. ¿Cuál es el
        total correcto y qué año, una vez más, ha causado el
        error?
\end{enumerate}
\end{problem}
