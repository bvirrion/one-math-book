\chapter{Expresiones literales}\label{ch:g7:literal}

Una letra en un cálculo representa un número que todavía no conocemos, o
\emph{todos} los números a la vez: escribir $P = 4c$ da el \omterm{def:g6:measure:perimeter}{perímetro}
de cualquier cuadrado en una sola fórmula. Este capítulo enseña las convenciones
de escritura, cómo sustituir valores y cómo simplificar expresiones
sencillas --- el comienzo del álgebra, que continúa en el
\cref{ch:g8:equations}.

\section{Escribir con letras}

\begin{definition}[Expresión literal]\label{def:g7:literal:def}
Una \emph{expresión literal}\index{expresión literal} es un cálculo
que contiene una o más letras, cada una en lugar de un número. Convenciones
de escritura:
\begin{itemize}
  \item el signo $\times$ se omite delante de una letra o de un paréntesis:
        $3 \times a = 3a$, \ $a \times b = ab$, \
        $2 \times (x + 5) = 2(x + 5)$;
  \item pero se mantiene entre dos números: ¡$3 \times 5$ no se puede
        escribir $35$!
  \item $a \times a$ se escribe $a^2$ («$a$ al cuadrado»),
        $a \times a \times a = a^3$;
  \item $1a$ se escribe $a$, y $0a$ es $0$.
\end{itemize}
\end{definition}

\begin{example}\label{ex:g7:literal:write}
Simplifica la escritura:
\[
5 \times x + 3 \times y = 5x + 3y,
\qquad
a \times 7 = 7a \ \text{(primero el número)},
\qquad
x \times x \times 4 = 4x^2 .
\]
\end{example}

\begin{example}[Las fórmulas son expresiones literales]\label{ex:g7:literal:formulas}
Para un rectángulo de largo $L$ y ancho $a$:
\omterm{def:g6:measure:perimeter}{perímetro} $P = 2(L + a)$, \omterm{def:g6:measure:area}{área} $A = La$. Una sola fórmula resume
infinitos rectángulos a la vez --- ese es todo el sentido de las
letras.
\end{example}

\section{Sustituir}

\begin{method}[Evaluar una expresión]\label{met:g7:literal:substitute}
Para evaluar una expresión con valores dados de sus letras:
\begin{enumerate}
  \item reescribe la expresión devolviendo los signos $\times$ escondidos:
        $3a + a^2$ es $3 \times a + a \times a$;
  \item sustituye cada letra por su valor, poniéndole paréntesis
        si hace falta;
  \item calcula respetando la jerarquía del
        \cref{ch:g7:priorities}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:literal:substitute}
Evalúa $E = 3x + x^2$ para $x = 5$:
\[
E = 3 \times 5 + 5 \times 5 = 15 + 25 = 40 .
\]
Evalúa $F = 2(a + 3b)$ para $a = 4$ y $b = 0.5$:
\[
F = 2 \times (4 + 3 \times 0.5) = 2 \times (4 + 1.5) = 2 \times 5.5
= 11 .
\]
\end{example}

\section{Reducir expresiones}

\begin{proposition}[Agrupar términos semejantes]\label{prop:g7:literal:reduce}
Los términos que tienen la misma parte literal se pueden agrupar sumando sus
partes numéricas (esto es la distributividad,
\cref{thm:g7:priorities:distributivity}, leída al revés):
\[
5x + 3x = (5 + 3)x = 8x,
\qquad
7a - 2a = 5a .
\]
Los términos con parte literal distinta (como $3x$ y $5y$, o $x$ y
$x^2$) no se pueden agrupar.
\end{proposition}

\begin{example}\label{ex:g7:literal:reduce}
Reduce $E = 4x + 7 + 3x + 2$ ordenando los términos:
\[
E = (4x + 3x) + (7 + 2) = 7x + 9 .
\]
Cuidado: ¡$7x + 9$ \emph{no} es $16x$! Pruébalo con $x = 1$:
$4 + 7 + 3 + 2 = 16$ y $7 \times 1 + 9 = 16$ coinciden, pero $16x$ también
daría $16$ --- prueba con $x = 2$: la expresión original da
$8 + 7 + 6 + 2 = 23$ y $7 \times 2 + 9 = 23$, mientras que
$16 \times 2 = 32$. La reducción $7x + 9$ es correcta y $16x$ es falsa.
\end{example}

\begin{method}[Poner a prueba una igualdad]\label{met:g7:literal:test}
Para comprobar si dos expresiones podrían ser iguales para todo valor:
\begin{enumerate}
  \item sustituye el mismo valor en las dos, al menos dos veces (por ejemplo,
        $x = 2$ y $x = 10$; evita $0$ y $1$, que esconden demasiadas
        \omterm{ex:g1:subtraction:difference}{diferencias});
  \item si los resultados difieren aunque sea una vez, las expresiones
        \emph{no} son iguales --- un solo contraejemplo mata una afirmación general;
  \item si los resultados coinciden, la igualdad es solo \emph{verosímil}: la
        demostración necesita álgebra (desarrollar, reducir).
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:literal:test}
¿Es $2(x + 3)$ igual a $2x + 6$ para todo $x$? Desarrollar lo demuestra:
$2(x+3) = 2x + 2 \times 3 = 2x + 6$. ¿Es $(x + 1)^2$ igual a
$x^2 + 1$? Prueba con $x = 2$: $(2+1)^2 = 9$ pero $2^2 + 1 = 5$. No ---
y un solo contraejemplo es una demostración completa del «no».
\end{example}

\section{Producir expresiones}

\begin{example}[De la situación a la expresión]\label{ex:g7:literal:produce}
Un taxi cobra $4$ euros más $2$ euros por kilómetro. Para un trayecto de
$k$ kilómetros, el precio es
\[
p = 4 + 2k \quad\text{euros}.
\]
Para $k = 7$: $p = 4 + 14 = 18$ euros. La expresión \emph{es} la
respuesta general; sustituir produce todas las respuestas particulares.
\end{example}

\begin{example}[Trucos con números]\label{ex:g7:literal:trick}
«Piensa un número, dóblalo, súmale $10$, divide entre $2$ y resta tu
número.» Llama $n$ al número y sigue los pasos:
\[
n \ \to\ 2n \ \to\ 2n + 10 \ \to\ \frac{2n + 10}{2} = n + 5 \ \to\
n + 5 - n = 5 .
\]
Todo el mundo obtiene $5$, fuera cual fuera $n$ --- las letras explican la
magia.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g7:literal:1}
Simplifica la escritura:
\[
7 \times x, \qquad
y \times 3, \qquad
a \times b \times 5, \qquad
x \times x, \qquad
2 \times (a + 4).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:literal:2}
Devuelve los signos $\times$:
\[
5ab, \qquad
3(x + 2), \qquad
x^2 + 4x, \qquad
2a^3 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:literal:3}
Evalúa para $x = 3$: \ $5x + 1$; \quad $x^2$; \quad $2x^2 - x$;
\quad $10 - 2x$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:literal:4}
Evalúa $3a + 2b$ y $a b + 5$ para $a = 4$ y $b = 1.5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:literal:5}
Reduce:
\[
8x + 5x, \qquad
9a - 4a + a, \qquad
6y + 2 + 3y + 7, \qquad
5t + 3 - 2t - 3 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:literal:6}
Da una fórmula para: el \omterm{def:g6:measure:perimeter}{perímetro} de un cuadrado de lado $c$; el
\omterm{def:g6:measure:perimeter}{perímetro} de un \omterm{def:g6:shapes:triangles}{triángulo isósceles} de base $b$ y lados iguales $s$;
el precio de $n$ panes a $1.20$ euros cada uno.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:literal:7}
Empareja cada expresión con una descripción: $2n$; $n + 2$; $n^2$;
$\frac n2$. Descripciones: «la \omterm{def:g3:division:half}{mitad} de $n$»; «el doble de $n$»;
«el cuadrado de $n$»; «dos más que $n$».
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:literal:8}
Usando el \cref{met:g7:literal:test}, decide (con demostración o
contraejemplo) si estas igualdades valen para todo $x$:
\[
3(x + 4) = 3x + 12; \qquad
2x + 5 = 7x; \qquad
x + x = x^2 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:literal:9}
Un rectángulo tiene ancho $a$ y su largo es $3$ cm mayor que su
ancho.
\begin{enumerate}
  \item expresa el largo y después el \omterm{def:g6:measure:perimeter}{perímetro} en función de $a$, y
        redúcelo;
  \item calcula el \omterm{def:g6:measure:perimeter}{perímetro} para $a = 5$ cm, directamente y con tu
        fórmula reducida. ¿Sale lo mismo?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:literal:10}
Repite el truco del \cref{ex:g7:literal:trick} con letras: «piensa
un número, súmale $6$, multiplica por $3$, resta $18$ y divide entre tu
número de partida» (supón que no es \omterm{def:g1:counting:zero}{cero}). ¿Qué obtiene todo el mundo?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:literal:11}
La figura de abajo es un cuadrado de lado $a$ al que se le ha quitado una esquina cuadrada
de lado $b$ ($b$ menor que $a$). Escribe dos expresiones distintas
para su \omterm{def:g6:measure:area}{área} (una como \omterm{ex:g1:subtraction:difference}{diferencia} y otra cortando la figura en L en dos
rectángulos) y comprueba que coinciden para $a = 5$ y $b = 2$.
\end{exercise}

\section{Problema: El oráculo de las cerillas}

\begin{problem}[{Problema de fin de semana --- patrones que crecen y su
término $n$-ésimo: tres fórmulas para un mismo patrón y el famoso
patrón que se rompe en el sexto paso}]
\label{pb:g7:literal:1}
Coloca cerillas formando una fila de cuadrados: un cuadrado, luego dos, luego
tres\,\dots\ ¿Cuántas cerillas necesitará la centésima figura?
Contarlas una a una llevaría todo el fin de semana; una \emph{fórmula
en $n$} responde al instante para todas las figuras a la vez --- que es
exactamente para lo que sirven las letras
(\cref{ex:g7:literal:formulas}). Pero cuidado: este problema termina
con un patrón célebre que juega limpio durante cinco pasos y luego
traiciona a todo el que se fio de él.

\medskip
\noindent\textbf{Parte I --- Una fila de cuadrados.}
La figura $1$ es un cuadrado de cerillas ($4$ cerillas); cada figura
siguiente añade un cuadrado más que comparte un lado con el anterior.
\begin{enumerate}
  \item dibuja las figuras $1$ a $4$ y cuenta sus cerillas;
  \item explica por qué cada cuadrado nuevo cuesta exactamente $3$ cerillas
        más y úsalo para predecir el recuento de la figura
        $10$ sin ninguna fórmula;
  \item justifica la fórmula: la figura $n$ necesita $3n + 1$ cerillas.
        (¿De dónde sale ese «$+1$» solitario?) Compruébala
        con la pregunta 1;
  \item Amir razona de otra manera: «$4$ cerillas para el primer
        cuadrado y luego $3$ por cada uno de los otros $n - 1$», lo que da
        $4 + 3(n - 1)$. Desarrolla y reduce su expresión
        (\cref{prop:g7:literal:reduce}): ¿coincide con la
        pregunta 3?
  \item Lena cuenta de una tercera manera: «$2n$ cerillas horizontales y
        $n + 1$ verticales». Escribe su fórmula, redúcela
        y concluye. Después responde a la pregunta inicial: ¿cuántas
        cerillas para la figura $100$?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Triángulos, mesas y marcha
atrás.}
\begin{enumerate}[resume]
  \item una fila de triángulos de cerillas (cada triángulo nuevo se apoya en
        el anterior): cuenta las cerillas de $1$, $2$ y $3$
        triángulos y justifica después la fórmula $2n + 1$ para $n$
        triángulos;
  \item marcha atrás: con exactamente $49$ cerillas, ¿cuántos
        triángulos puede tener la fila? (Deshaz la fórmula paso a
        paso.)
  \item las mesas cuadradas de banquete sientan a un invitado por lado libre. Cuando
        se juntan $n$ mesas en una fila larga, explica por qué
        la fórmula de plazas es $2n + 2$ y compruébala para $1$, $2$ y
        $3$ mesas;
  \item a una fiesta se esperan $30$ invitados. ¿Cuántas mesas hacen falta
        en una sola fila? El cáterin reparte en cambio las mismas mesas en
        \emph{dos} filas de siete: ¿cuántos invitados caben ahora?
        Explica la ganancia con la fórmula;
  \item patrón de puntos: la figura $n$ es una L de puntos, de $n$ puntos de alto
        y $n$ puntos de ancho (la esquina se cuenta una vez). Cuenta los
        puntos de las figuras $1$, $2$ y $3$, halla la fórmula y di el nombre
        de los números que produce. (Apiladas, estas L recubren
        un cuadrado --- una historia que continúa en el
        \cref{pb:g8:equations:1}.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Fiarse, probar, demostrar.}
\begin{enumerate}[resume]
  \item un compañero afirma que la figura $n$ de la Parte I necesita
        $3(n + 2) - 5$ cerillas. Resuelve la afirmación desarrollando y
        reduciendo --- sin necesidad de probar valores;
  \item «elevar al cuadrado hace los números más grandes, así que $n^2 \geq n$ para
        todo número $n$.» Prueba la afirmación con $n = 2$, $3$ y
        $10$; después halla un número para el que falle. ¿Qué demostraron las
        tres pruebas con éxito y qué demuestra el único
        fallo (\cref{met:g7:literal:test})?
  \item marca puntos sobre una circunferencia y traza \emph{todos} los \omterm{def:g6:lines:objects}{segmentos}
        entre ellos; después cuenta las regiones del interior del círculo.
        Hazlo con $2$, $3$, $4$ y $5$ puntos (coloca los puntos
        de forma irregular, para que no se corten tres \omterm{def:g6:lines:objects}{segmentos} en un mismo
        punto interior). Anota los recuentos. ¿Qué fórmula te
        susurra el patrón?
  \item ahora $6$ puntos, dibujados grandes e irregulares, contando con
        muchísimo cuidado (hay muchas regiones pequeñas). ¿Cuántas
        regiones encuentras y qué le ha pasado a la
        conjetura?
  \item la moraleja, en dos frases: qué habría hecho falta
        para \emph{demostrar} la conjetura del doble y por qué las
        fórmulas de las cerillas de las Partes I y II nunca corrieron ese
        peligro. Celébralo con una predicción demostrada: ¿cuántas
        cerillas para la fila de $2\,026$ cuadrados?
\end{enumerate}
\end{problem}
