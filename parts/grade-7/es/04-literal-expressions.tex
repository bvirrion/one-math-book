\chapter{Expresiones literales}\label{ch:g7:literal}

Una letra en un cálculo representa un número que aún no conocemos, o
para \emph{todo} números a la vez: escritura $P = 4c $ da el \omterm{def:g6:measure:perimeter}{perímetro}
de cada cuadrado en una fórmula. Este capítulo enseña la escritura.
convenciones, cómo sustituir valores y cómo simplificar simples
expresiones --- el comienzo del álgebra, continúa en
\cref{ch:g8:equations}.

\section{Escribiendo con letras}

\begin{definition}[expresión literal]\label{def:g7:literal:def}
A \emph{expresión literal}\index{expresión literal} es un cálculo
que contiene una o más letras, cada una de las cuales representa un número. escribiendo
convenciones:
\begin{itemize}
  \item el signo $\times $ se elimina antes de una letra o un corchete:
        $3 \times a = 3a $, \ $ a \times b = ab $, \
        $2 \times (x + 5) = 2(x + 5)$;
  \item pero se queda entre dos números: $3 \times 5$ no puede ser
        escrito $35$!
  \item $ a \times a $ esta escrito $ a^2$ (``$ a $ al cuadrado''),
        $ a \times a \times a = a^3$;
  \item $1a $ esta escrito $ a $, y $0a $ es $0$.
\end{itemize}
\end{definition}

\begin{example}\label{ex:g7:literal:write}
Simplifica la escritura:
\[
5 \times x + 3 \times y = 5x + 3y,
\qquad
a \times 7 = 7a \ \text{(number first)},
\qquad
x \times x \times 4 = 4x^2 .
\]
\end{example}

\begin{example}[Las fórmulas son expresiones literales.]\label{ex:g7:literal:formulas}
Para un rectángulo de longitud $ L $ y ancho $ w $:
\omterm{def:g6:measure:perimeter}{perímetro} $ P = 2(L + w)$, \omterm{def:g6:measure:area}{área} $ A = Lw $. Una fórmula resume la
infinitos rectángulos a la vez --- ese es el objetivo de
letras.
\end{example}

\section{Sustituyendo}

\begin{method}[Evaluar una expresión]\label{met:g7:literal:substitute}
Para evaluar una expresión para valores dados de sus letras:
\begin{enumerate}
  \item reescribe la expresión poniendo lo oculto $\times $ firma de nuevo:
        $3a + a^2$ es $3 \times a + a \times a $;
  \item reemplace cada letra por su valor, manteniendo corchetes alrededor
        si es necesario;
  \item calcular, respetando las prioridades de
        \cref{ch:g7:priorities}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:literal:substitute}
Evaluar $ E = 3x + x^2$ para $ x = 5$:
\[
E = 3 \times 5 + 5 \times 5 = 15 + 25 = 40 .
\]
Evaluar $ F = 2(a + 3b)$ para $ a = 4$ y $ b = 0.5$:
\[
F = 2 \times (4 + 3 \times 0.5) = 2 \times (4 + 1.5) = 2 \times 5.5
= 11 .
\]
\end{example}

\section{Reducir expresiones}

\begin{proposition}[Recopilando términos semejantes]\label{prop:g7:literal:reduce}
Los términos que contienen la misma parte de letra se pueden recopilar agregando sus
partes numéricas (esto es distributividad,
\cref{thm:g7:priorities:distributivity}, lee al revés):
\[
5x + 3x = (5 + 3)x = 8x,
\qquad
7a - 2a = 5a .
\]
Términos con diferentes partes de letras (como $3x $ y $5y $, o $ x $ y
$ x^2$) no se puede recoger.
\end{proposition}

\begin{example}\label{ex:g7:literal:reduce}
Reducir $ E = 4x + 7 + 3x + 2$, ordenando los términos:
\[
E = (4x + 3x) + (7 + 2) = 7x + 9 .
\]
Cuidadoso: $7x + 9$ es \emph{no} $16x $! Prueba con $ x = 1$:
$4 + 7 + 3 + 2 = 16$ y $7 \times 1 + 9 = 16$ de acuerdo, pero $16x $ haría
también ser $16$ --- prueba con $ x = 2$: el original da
$8 + 7 + 6 + 2 = 23$, y $7 \times 2 + 9 = 23$, mientras
$16 \times 2 = 32$. la reduccion $7x + 9$ tiene razón, $16x $ está mal.
\end{example}

\begin{method}[Probando una igualdad]\label{met:g7:literal:test}
Para comprobar si dos expresiones pueden ser iguales para cada valor:
\begin{enumerate}
  \item sustituya el mismo valor en ambos, al menos dos veces (p. ej.\
        $ x = 2$ y $ x = 10$; evitar $0$ y $1$, esconden demasiados
        \omterm{ex:g1:subtraction:difference}{diferencias});
  \item Si los resultados difieren aunque sea una vez, las expresiones son
        \emph{no} igual --- un contraejemplo mata una afirmación general;
  \item si los resultados concuerdan, la igualdad es sólo \emph{plausible}: un
        la prueba necesita álgebra (expandir, reducir).
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:literal:test}
Es $2(x + 3)$ igual a $2x + 6$ por cada $ x $? La ampliación lo demuestra:
$2(x+3) = 2x + 2 \times 3 = 2x + 6$. Es $(x + 1)^2$ igual a
$ x^2 + 1$? Prueba $ x = 2$: $(2+1)^2 = 9$ pero $2^2 + 1 = 5$. No ---
y un contraejemplo es una prueba completa del "no".
\end{example}

\section{Producir expresiones}

\begin{example}[De la situación a la expresión]\label{ex:g7:literal:produce}
un taxi cobra $4$ euros más $2$ euros por kilómetro. para un viaje de
$ k $ kilómetros el precio es
\[
p = 4 + 2k \quad\text{euros}.
\]
Para $ k = 7$: $ p = 4 + 14 = 18$ euros. la expresión \emph{es} el
respuesta general; la sustitución produce cada uno en particular.
\end{example}

\begin{example}[trucos de números]\label{ex:g7:literal:trick}
``Piensa en un número, duplícalo, súmalo $10$, dividir por $2$, resta tu
número." Llame al número $ n $ y sigue los pasos:
\[
n \ \to\ 2n \ \to\ 2n + 10 \ \to\ \frac{2n + 10}{2} = n + 5 \ \to\
n + 5 - n = 5 .
\]
todos obtienen $5$, lo que $ n $ era --- las letras explican el
magia.
\end{example}

\section{Problema: el oráculo de las cerillas}

\begin{problem}[{Problema de fin de semana --- Patrones de crecimiento y su término $n$: tres fórmulas para un patrón y el famoso patrón que se rompe en el sexto paso.}]
\label{pb:g7:literal:1}
Coloque las cerillas en una fila de cuadrados: un cuadrado, luego dos, luego
tres\,\dots\ ¿Cuántas cerillas necesitará la centésima cifra?
Contarlos uno por uno llevaría todo el fin de semana; a \emph{fórmula
en $ n $} respuestas instantáneas para cada cifra a la vez --- es decir
exactamente para qué sirven las letras
(\cref{ex:g7:literal:formulas}). Pero ojo: este problema se acaba
con un patrón famoso que juega limpio durante cinco pasos y luego
traiciona a todos los que confiaron en él.

\medskip
\noindent\textbf{Parte I --- A row of squares.}
Cifra $1$ es un cuadrado de cerilla ($4$ partidos); cada uno de los siguientes
La figura suma un cuadrado más compartiendo lado con el anterior.
\begin{enumerate}
  \item dibujar figuras $1$ a $4$ y contar sus coincidencias.
  \item Explica por qué cada cuadrado nuevo cuesta exactamente $3$ adicional
        coincidencias y utilícelo para predecir el recuento de la figura
        $10$ sin ninguna fórmula.
  \item Justifique la fórmula: figura $ n $ necesidades $3n + 1$ partidos.
        (¿De dónde viene el solitario ``$+1$'' viene de?) Compruébalo
        contra la pregunta 1.
  \item Amir razona de otra manera: ``$4$ partidos por primera
        cuadrado, entonces $3$ para cada uno de los $ n - 1$ otros'', dando
        $4 + 3(n - 1)$. Ampliar y reducir su expresión.
        (\cref{prop:g7:literal:reduce}): ¿está de acuerdo con
        pregunta 3?
  \item Lena cuenta una tercera vía: ``$2n $ coincidencias horizontales, y
        $ n + 1$ los verticales. '' Escribe su fórmula, redúcela,
        y concluir. Luego responda la pregunta inicial: ¿cuántos
        coincidencias para figura $100$?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Triangles, tables and going
backwards.}
\begin{enumerate}[resume]
  \item Una fila de triángulos de cerillas (cada nuevo triángulo se apoya en
        el anterior): cuenta los partidos para $1$, $2$, $3$
        triángulos, luego justifica la fórmula $2n + 1$ para $ n $
        triángulos.
  \item Yendo hacia atrás: con exactamente $49$ partidos, cuantos
        ¿Qué triángulos puede tener la fila? (Deshacer la fórmula paso a paso
        paso.)
  \item Las mesas de banquete cuadradas tienen capacidad para un invitado por lado libre. Cuando
        $ n $ Las tablas están agrupadas en una fila larga, explique por qué
        la fórmula de los asientos es $2n + 2$ y compruébalo para $1$, $2$,
        $3$ mesas.
  \item Una fiesta espera $30$ huéspedes. ¿Cuántas tablas tiene una fila?
        necesita? En cambio, el proveedor divide las mismas mesas en
        \emph{dos} filas de siete: ¿cuántos invitados caben ahora?
        Explica la ganancia con la fórmula.
  \item Patrón de puntos: figura $ n $ es una L de puntos, $ n $ puntos altos
        y $ n $ puntos de ancho (la esquina se cuenta una vez). contar el
        puntos de figuras $1$, $2$, $3$, encuentra la fórmula y el nombre
        los números que produce. (Apilados juntos, estos mosaicos L
        un cuadrado --- una historia continuada en
        \cref{pb:g8:equations:1}.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Trust, test, prove.}
\begin{enumerate}[resume]
  \item Un compañero reclama cifra $ n $ de la Parte I necesita
        $3(n + 2) - 5$ partidos. Resuelva el reclamo ampliando y
        reduciendo --- no se necesitan pruebas.
  \item "Al elevar al cuadrado los números son más grandes, entonces $ n^2 \geq n $ para
        cada número $ n $.'' Pruebe la afirmación de $ n = 2$, $3$,
        $10$; luego encuentre un número para el cual falla. ¿Qué hizo el
        tres pruebas exitosas, ¿y qué significa el único
        prueba de fracaso (\cref{met:g7:literal:test})?
  \item Marca puntos en un círculo y dibuja. \emph{cada} \omterm{def:g6:lines:objects}{segmento}
        entre ellos, luego cuenta las regiones dentro del círculo.
        hazlo por $2$, $3$, $4$ y $5$ puntos (coloque los puntos
        desigualmente, de modo que no haya tres \omterm{def:g6:lines:objects}{segmentos} cruzar en un interior
        punto). Registre los conteos. ¿Qué fórmula tiene el patrón?
        susurrar?
  \item Ahora $6$ puntos, dibujados de forma grande y desigual, contando con
        Mucho cuidado (hay muchas regiones pequeñas). cuantos
        regiones que encuentras --- y qué pasó con las
        conjetura?
  \item La moraleja, en dos frases: lo que hubiera hecho falta
        a \emph{probar} la conjetura de la duplicación y por qué
        fórmulas de cerillas de las Partes I y II nunca en tal
        peligro? Celebre con una predicción comprobada: ¿cuántos
        coincidencias para la fila de $2\,026$ ¿cuadrícula?
\end{enumerate}
\end{problem}
