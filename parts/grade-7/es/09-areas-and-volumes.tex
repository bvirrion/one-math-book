\chapter{Áreas y volúmenes}\label{ch:g7:areas}

Rectángulos medidos de grado 6, \omterm{def:g6:shapes:triangles}{triangulos rectángulos} y cajas
(\cref{ch:g6:measure}). Con unas tijeras --- corta un trozo aquí,
pégalo allí --- las mismas ideas ahora dan el \omterm{def:g6:measure:area}{áreas} de \omterm{def:g7:central:parallelogram}{paralelogramos}, de \emph{todo} triángulos y del disco, y el \omterm{def:g6:measure:volume}{volúmenes} de \omterm{def:g7:areas:prism}{prismas} y
\omterm{def:g7:areas:prism}{cilindros}.

\section{Área de un paralelogramo}

\begin{theorem}[Paralelogramo]\label{thm:g7:areas:parallelogram}
A \omterm{def:g7:central:parallelogram}{paralelogramo} con un lado de longitud $b $ (es \emph{base}) y
distancia $ h $ entre ese lado y el opuesto (su
\emph{altura}) tiene \omterm{def:g6:measure:area}{área}
\[
A = b \times h .
\]
\end{theorem}

\begin{proof}[Prueba con tijeras]
Cortar el \omterm{def:g6:shapes:triangles}{en ángulo recto} triángulo que sobresale de un lado y pégalo
por el otro: el \omterm{def:g7:central:parallelogram}{paralelogramo} se convierte en un $ b \times h $ rectángulo con
lo mismo \omterm{def:g6:measure:area}{área}.
\end{proof}

\begin{remark}
La altura es la \emph{\omterm{def:g6:lines:perp}{perpendicular}} distancia entre las dos bases,
¡No la longitud del lado inclinado! Un muy inclinado \omterm{def:g7:central:parallelogram}{paralelogramo} puede
tener lados largos y un pequeño \omterm{def:g6:measure:area}{área}.
\end{remark}

\section{Área de un triángulo}

\begin{theorem}[Triángulo]\label{thm:g7:areas:triangle}
Un triángulo con base $ b $ y altura correspondiente $ h $ (el
\omterm{def:g6:lines:perp}{perpendicular} distancia del opuesto \omterm{def:g5:solids:def}{vértice} hasta la línea base) tiene
\omterm{def:g6:measure:area}{área}
\[
A = \frac{b \times h}{2} .
\]
\end{theorem}

\begin{proof}
Dos copias del triángulo, una girada media vuelta alrededor del
punto medio de un lado, encajan en un \omterm{def:g7:central:parallelogram}{paralelogramo} con base $ b $
y altura $ h $ (eso es \omterm{def:g7:central:def}{céntrico
simetría} en el trabajo,
\cref{prop:g7:central:preserved}). El triángulo es \omterm{def:g3:division:half}{medio} de ello:
$\frac{bh}{2}$.
\end{proof}

\begin{tikzpicture}[scale=0.9]
  \fill[omDef!14] (0,0) -- (4,0) -- (1.4,2.1) -- cycle;
  \draw[thick] (0,0) -- (4,0) -- (1.4,2.1) -- cycle;
  \fill[omThm!12] (4,0) -- (1.4,2.1) -- (5.4,2.1) -- cycle;
  \draw[thick, omThm, dashed] (4,0) -- (5.4,2.1) -- (1.4,2.1);
  \draw[dashed] (1.4,0) -- (1.4,2.1);
  \draw[thin] (1.4,0) ++(-0.2,0) -- ++(0,0.2) -- ++(0.2,0);
  \node[below, font=\small] at (2,0) {$ b $};
  \node[right, font=\small] at (1.42,1.0) {$ h $};
\end{tikzpicture}

{\small Un triángulo y su copia a medio girar (discontinua) forman una
\omterm{def:g7:central:parallelogram}{paralelogramo}: el triángulo \omterm{def:g6:measure:area}{área} es $\frac{b\times h}{2}$. Cualquiera de los
tres lados pueden servir como base --- con \emph{su propio} altura.}
\end{omfigure}

\begin{example}\label{ex:g7:areas:triangle}
Un triángulo tiene una base de $9$ cm y las medidas de altura correspondientes
$4$ centímetro:
\[
A = \frac{9 \times 4}{2} = \frac{36}{2} = 18 \text{ cm}^2 .
\]
por un \emph{bien} triángulo, los dos catetos son una base y su altura:
recuperamos la fórmula de \cref{prop:g6:measure:formulas}.
\end{example}

\section{Área de un disco}

\begin{theorem}[Disco]\label{thm:g7:areas:disk}
un disco de \omterm{def:g3:shapes:shapes}{radio} $ r $ tiene \omterm{def:g6:measure:area}{área}
\[
A = \pi r^2 .
\]
\end{theorem}

\begin{proof}[Idea de prueba]
Cortar el disco en muchas rodajas finas iguales y colocarlas de la cabeza a la cola:
casi forman un \omterm{def:g7:central:parallelogram}{paralelogramo} de altura $ r $ cuya base es \omterm{def:g3:division:half}{medio} el
circunferencia, $\pi r $ (\cref{prop:g6:measure:circle}). Es \omterm{def:g6:measure:area}{área} esta cerca de $\pi r \times r = \pi r^2$, y la aproximación se convierte en
Perfecto a medida que las rodajas se vuelven más finas. Una prueba completa necesita integral
Cálculo, del volumen High School.
\end{proof}

  \node[below, font=\small] at (2.5,-0.1) {base $\approx \pi r $};
  \end{scope}
\end{tikzpicture}

{\small Cortar el disco y desenrollar las rodajas: casi un
\omterm{def:g7:central:parallelogram}{paralelogramo} de base $\pi r $ (\omterm{def:g3:division:half}{medio} el borde) y la altura $ r $.}
\end{omfigure}

\begin{example}\label{ex:g7:areas:disk}
Una mesa redonda tiene \omterm{def:g3:shapes:shapes}{radio} $60$ centímetro. Es \omterm{def:g6:measure:area}{área} es
$\pi \times 60^2 = 3600\pi \approx 11\,310$ centímetro $^2$, un poco más que
$1.1$ metro $^2$. Mira el \omterm{ex:g1:subtraction:difference}{diferencia}: el \emph{\omterm{def:g6:measure:perimeter}{perímetro}} $2\pi r $ usos
$ r $, el \emph{\omterm{def:g6:measure:area}{área}} $\pi r^2$ usos $ r^2$.
\end{example}

\section{Prismas y cilindros}

\begin{definition}[Prisma, cilindro]\label{def:g7:areas:prism}
A \emph{prisma}\index{prisma} es un \omterm{def:g5:solids:def}{sólido} con dos identicos \omterm{def:g6:lines:perp}{paralelo}
poligonal \omterm{def:g5:solids:def}{caras} (el \emph{bases}) unidos por rectángulos; a
\emph{cilindro}\index{cilindro} Lo mismo ocurre con los discos como bases. El
\emph{altura} es la distancia entre las dos bases.
\end{definition}

\begin{theorem}[Volúmenes]\label{thm:g7:areas:volume}
por un \omterm{def:g7:areas:prism}{prisma} o un \omterm{def:g7:areas:prism}{cilindro} con base \omterm{def:g6:measure:area}{área} $ B $ y altura $ h $:
\[
V = B \times h .
\]
\end{theorem}
\admitted

\begin{example}\label{ex:g7:areas:volume}
Una tienda de campaña es un \omterm{def:g7:areas:prism}{prisma} acostado de lado: sus bases son triángulos de base
$2$ m y altura $1.5$ m, y la tienda es $3$ m de largo.
\begin{enumerate}
  \item Base \omterm{def:g6:measure:area}{área}: $ B = \frac{2 \times 1.5}{2} = 1.5$ metro $^2$.
  \item \omterm{def:g6:measure:volume}{Volumen}: $ V = B \times h = 1.5 \times 3 = 4.5$ metro $^3$.
\end{enumerate}
una lata de \omterm{def:g3:shapes:shapes}{radio} $4$ cm y altura $11$ centímetro:
$ V = \pi \times 4^2 \times 11 = 176\pi \approx 553$ centímetro $^3$, apenas
\omterm{def:g3:division:half}{medio} un litro.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g7:areas:1}
Calcular el \omterm{def:g6:measure:area}{área} de un \omterm{def:g7:central:parallelogram}{paralelogramo} con base $8$ cm y altura
$5$ centímetro; de otro con base $6.5$ m y altura $4$ metro.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:areas:2}
A \omterm{def:g7:central:parallelogram}{paralelogramo} tiene lados $10$ centímetros y $6$ cm, y la altura relativa
al $10$ medidas base cm $4$ centímetro. Calcular su \omterm{def:g6:measure:area}{área}. ¿Por qué es el
responder no $60$ centímetro $^2$?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:areas:3}
Calcular el \omterm{def:g6:measure:area}{área} de un triángulo con base $12$ cm y altura $7$ centímetro;
de un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} con piernas $9$ y $6$; de un triángulo con base
$5.5$ m y altura $2$ metro.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:areas:4}
Calcular el \omterm{def:g6:measure:area}{área} de un disco de \omterm{def:g3:shapes:shapes}{radio} $5$ cm, luego de un medio disco de
\omterm{def:g3:shapes:shapes}{radio} $10$ cm (valores exactos con $\pi $, luego redondeado a la unidad).
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:areas:5}
Calcular el \omterm{def:g6:measure:perimeter}{perímetro} \emph{y} el \omterm{def:g6:measure:area}{área} de un disco de \omterm{def:g3:shapes:shapes}{radio} $3$ centímetro.
cual de las dos respuestas esta en cm y cual en cm $^2$?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:areas:6}
Calcular el \omterm{def:g6:measure:volume}{volumen} de un \omterm{def:g7:areas:prism}{prisma} con base \omterm{def:g6:measure:area}{área} $24$ centímetro $^2$ y altura
$10$ centímetro;
de un \omterm{def:g7:areas:prism}{cilindro} de \omterm{def:g3:shapes:shapes}{radio} $3$ cm y altura $8$ cm (exacto,
luego redondeado).
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:areas:7}
Un triángulo tiene \omterm{def:g6:measure:area}{área} $28$ centímetro $^2$ y base $8$ centímetro. Encuentra el
altura correspondiente, escribiendo primero la ecuación.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g7:areas:8}
Dibuja cualquier triángulo y mide cuidadosamente los tres pares base-altura.
Calcular $\frac{b \times h}{2}$ para cada par: los tres resultados deben
de acuerdo (hasta error de medición). ¿Por qué?
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g7:areas:9}
Un jardín en forma de trapezoide se puede dividir, por una diagonal, en dos
triangulos que comparten la misma altura $ h = 20$ m, con bases $30$ m y
$18$ metro. Calcula su total \omterm{def:g6:measure:area}{área}. ¿Puedes adivinar una fórmula general para
el trapezoide?
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g7:areas:10}
Un jarrón cilíndrico de \omterm{def:g3:shapes:shapes}{radio} $6$ cm contiene agua hasta una altura de
$15$ centímetro. Toda el agua se vierte en un recipiente vacío. \omterm{def:g5:solids:def}{caja}
$18$ centímetro $\times $ $12$ cm en la base. ¿Qué altura del agua se alcanza en
el \omterm{def:g5:solids:def}{caja}? (Mismo \omterm{def:g6:measure:volume}{volumen}, base diferente.)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g7:areas:11}
una pizza de \omterm{def:g4:geometry:circle}{diámetro} $30$ cm costos $9$ euros; uno de \omterm{def:g4:geometry:circle}{diámetro} $40$ cm costos $14$ euros. Calcular los dos \omterm{def:g6:measure:area}{áreas} y el precio por $100$
centímetro $^2$. ¿Qué pizza es la mejor oferta?
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g7:areas:12}
Las dos piernas de un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} medida $6$ centímetros y $8$ cm, y su
medidas de hipotenusa $10$ centímetro. Calcular su \omterm{def:g6:measure:area}{área} con las piernas, luego use
eso \omterm{def:g6:measure:area}{área} para encontrar la altura relativa a la \emph{hipotenusa}.
\end{exercise}

\section{Problema: el cuadrado que desaparece}

\begin{problem}[{Problema de fin de semana --- una famosa ``prueba'' de que $64 = 65$, desenmascarada por zonas y pistas, con pizzas y aros de postre}]
\label{pb:g7:areas:1}
Un mago corta un $8 \times 8$ cuadrado en cuatro pedazos, diapositivas
alrededor y los vuelve a ensamblar en un $5 \times 13$ rectángulo.
La audiencia cuenta: $8 \times 8 = 64$, pero
$5 \times 13 = 65$ --- una unidad cuadrada ha aparecido de la nada
aire! este capítulo \omterm{def:g6:measure:area}{área} Las fórmulas son exactamente las herramientas que
capta el truco. El problema entonces se pone honesto. \omterm{def:g6:measure:area}{área} razonamiento para
trabajo: sobre trapecios, sobre pilas inclinadas, sobre economía de la pizza --- y
finalmente regresa al mago por un segundo, \omterm{def:g3:numbers:evenodd}{incluso} extraño
rendimiento.

\medskip
\noindent\textbf{Parte I --- The trick, performed and unmasked.}
Las cuatro piezas del $8 \times 8$ cuadrados son: dos \omterm{def:g6:shapes:triangles}{triangulos rectángulos} con piernas $8$ y $3$, y dos trapecios derechos con
\omterm{def:g6:lines:perp}{paralelo} lados $5$ y $3$ y altura $5$.
\begin{enumerate}
  \item Calcular el \omterm{def:g6:measure:area}{área} del cuadrado original y el \omterm{def:g6:measure:area}{área} de
        el rectángulo reclamado. Exprese el escándalo en una línea.
  \item Calcular el \omterm{def:g6:measure:area}{área} de cada una de las cuatro piezas
        (\cref{thm:g7:areas:triangle}; para los trapecios, cortar
        o usar \cref{exo:g7:areas:9}).
  \item Suma los cuatro \omterm{def:g6:measure:area}{áreas}. ¿Cuál de las dos figuras --- cuadrado
        o rectángulo --- ¿se pueden llenar realmente las piezas?
  \item En la disposición del rectángulo, la hipotenusa de un triángulo
        (levantándose $3$ al otro lado de $8$) se supone que continúa un
        borde inclinado del trapezoide (ascendente $2$ al otro lado de $5$) en uno
        recta ``diagonal''. Pruebe la alineación: son $3$ al otro lado de
        $8$ y $2$ al otro lado de $5$ la misma pendiente
        (\cref{thm:g7:prop:cross})?
  \item Explica dónde se esconde el cuadrado sexagésimo quinto: qué forma tiene
        la brecha a lo largo de la falsa diagonal, ¿cuál es su exactitud? \omterm{def:g6:measure:area}{área},
        ¿Y por qué el ojo no lo ve? (¿Qué ancho, en promedio, tiene un
        astilla de eso \omterm{def:g6:measure:area}{área} estirado sobre el rectángulo
        diagonal, aproximadamente $13$ unidades de largo?)
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Honest \omterm{def:g6:measure:area}{área} reasoning.}
\begin{enumerate}[resume]
  \item dibuja dos \omterm{def:g6:lines:perp}{lineas paralelas} $4$ cm de distancia, una base de $6$ cm en
        uno de ellos, y tres triángulos diferentes en esa base
        con vértices en varios puntos de la otra línea. calcular
        los tres \omterm{def:g6:measure:area}{áreas}. ¿Por qué son todos iguales, sin importar cuál sea el vértice?
        se desliza a lo largo de su línea?
  \item Demuestre la fórmula del trapezoide en general: un trapezoide con
        \omterm{def:g6:lines:perp}{paralelo} lados $ B $ y $ b $ y altura $ h $, cortar a lo largo de un
        diagonal, da dos triángulos de la misma altura $ h $. Concluir:
        \[
        A = \frac{(B + b) \times h}{2} .
        \]
  \item Aplicarlo a un trapezoide con $ B = 9$ centímetro, $ b = 5$ centímetro,
        $ h = 4$ centímetro.
  \item Los carpinteros utilizan la misma fórmula disfrazada: \omterm{def:g6:measure:area}{área} $=$
        (promedio de los dos \omterm{def:g6:lines:perp}{paralelo} lados) $\times $ altura.
        Explique por qué esta es la misma regla y vuelva a calcular la pregunta.
        8 de esa manera.
  \item Incline una pila ordenada de naipes en una pila inclinada.
        Explique por qué el \omterm{def:g6:measure:volume}{volumen} no ha cambiado, y lo que esto
        dice sobre la altura en el \omterm{def:g7:areas:prism}{prisma} fórmula
        $ V = B \times h $ (\cref{thm:g7:areas:volume}): cual
        altura que se debe tomar para una pila inclinada --- la inclinación
        ¿Largo o el vertical? (Comparar
        \cref{exo:g7:areas:2}, una dimensión hacia arriba.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Rings, pizzas, and the magician's
return.}
\begin{enumerate}[resume]
  \item Un estanque circular de \omterm{def:g3:shapes:shapes}{radio} $3$ m se sienta en el centro de un
        césped circular de \omterm{def:g3:shapes:shapes}{radio} $5$ metro. Calcular el \omterm{def:g6:measure:area}{área} de la
        anillo de hierba (exactamente con $\pi $, luego redondeado a la
        m$^2$).
  \item Encuentra el \omterm{def:g3:shapes:shapes}{radio} del disco único cuyo \omterm{def:g6:measure:area}{área} es igual
        ese anillo \omterm{def:g6:measure:area}{área} exactamente. (Los tres radios que tienes ahora
        forman un famoso triple, cuya historia se cuenta en
        \cref{ch:g8:pythagoras}.)
  \item ¿Cuál es más pizza: una pizza de \omterm{def:g4:geometry:circle}{diámetro} $40$ centímetros, o
        dos pizzas de \omterm{def:g4:geometry:circle}{diámetro} $28$ ¿centímetro? Calcula y compara.
  \item A \emph{doble} una pizza \omterm{def:g6:measure:area}{área}, aproximadamente por qué factor
        debe ser \omterm{def:g3:shapes:shapes}{radio} ¿crecer? comprobar el candidato $1.4$ (recordar
        eso \omterm{def:g6:measure:area}{áreas} seguir
el \emph{cuadrado} de la escala
        factores, \cref{pb:g6:propdata:1}), y diga dónde
        factor exacto --- el número cuyo cuadrado es $2$ --- hace
        su entrada oficial
        (\cref{ex:g8:pythagoras:diagonal}).
  \item El mago regresa: corta un $13 \times 13$ cuadrado
        con los números $5$, $8$, $13$ desempeñando los papeles que
        $3$, $5$, $8$ jugado antes, y vuelve a montar un
        $8 \times 21$ rectángulo. Calcular ambos \omterm{def:g6:measure:area}{áreas}: que
        desaparece esta vez? los números $3, 5, 8, 13, 21$ son
        términos consecutivos de la secuencia de
        \cref{pb:g7:fractions:1} --- indicar el patrón de la
        ganancias y pérdidas del mago, y la moraleja: ¿por qué \omterm{def:g6:measure:area}{áreas}
        ¿Nunca mientes?
\end{enumerate}
\end{problem}
