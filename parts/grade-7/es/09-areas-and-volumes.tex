\chapter{Áreas y volúmenes}\label{ch:g7:areas}

El año pasado medimos rectángulos, \omterm{def:g6:shapes:triangles}{triángulos rectángulos} y \omterm{def:g5:solids:def}{ortoedros}
(\cref{ch:g6:measure}). Con unas tijeras --- cortar un trozo aquí,
pegarlo allá --- las mismas ideas dan ahora las \omterm{def:g6:measure:area}{áreas} de los \omterm{def:g7:central:parallelogram}{paralelogramos},
de \emph{todos} los triángulos y del disco, y los volúmenes de los \omterm{def:g7:areas:prism}{prismas} y
los \omterm{def:g7:areas:prism}{cilindros}.

\section{Área de un paralelogramo}

\begin{theorem}[Paralelogramo]\label{thm:g7:areas:parallelogram}
Un \omterm{def:g7:central:parallelogram}{paralelogramo} con un lado de longitud $b$ (su \emph{base}) y
distancia $h$ entre ese lado y el opuesto (su
\emph{altura}) tiene \omterm{def:g6:measure:area}{área}
\[
A = b \times h .
\]
\end{theorem}

\begin{proof}[Demostración con tijeras]
Corta el \omterm{def:g6:shapes:triangles}{triángulo rectángulo} que sobresale por un lado y pégalo
en el otro: el \omterm{def:g7:central:parallelogram}{paralelogramo} se convierte en un rectángulo $b \times h$ con
la misma \omterm{def:g6:measure:area}{área}.
\end{proof}

\begin{tikzpicture}[scale=0.82]
  \fill[omDef!14] (0,0) -- (4,0) -- (5.2,1.8) -- (1.2,1.8) -- cycle;
  \draw[thick] (0,0) -- (4,0) -- (5.2,1.8) -- (1.2,1.8) -- cycle;
  \fill[omThm!25] (0,0) -- (1.2,1.8) -- (1.2,0) -- cycle;
  \draw[omThm, thick] (0,0) -- (1.2,1.8) -- (1.2,0) -- cycle;
  \draw[dashed] (1.2,0) -- (1.2,1.8);
  \node[below, font=\small] at (2.0,0) {$b$};
  \draw[Stealth-Stealth] (6.0,0) -- (6.0,1.8)
    node[midway, right, font=\small] {$h$};
  \begin{scope}[xshift=8.2cm]
  \fill[omDef!14] (0,0) rectangle (4,1.8);
  \draw[thick] (0,0) rectangle (4,1.8);
  \fill[omThm!25] (2.8,0) -- (4,1.8) -- (4,0) -- cycle;
  \draw[omThm, thick] (2.8,0) -- (4,1.8) -- (4,0) -- cycle;
  \node[below, font=\small] at (2.0,0) {$b$};
  \end{scope}
\end{tikzpicture}

{\small La demostración con tijeras: deslizar el triángulo rojo del extremo izquierdo
al derecho convierte el \omterm{def:g7:central:parallelogram}{paralelogramo} en un rectángulo --- la misma
\omterm{def:g6:measure:area}{área}, $b \times h$.}
\end{omfigure}

\begin{remark}
¡La altura es la distancia \emph{perpendicular} entre las dos bases,
no la longitud del lado inclinado! Un \omterm{def:g7:central:parallelogram}{paralelogramo} muy inclinado puede
tener lados largos y un \omterm{def:g6:measure:area}{área} pequeña.
\end{remark}

\section{Área de un triángulo}

\begin{theorem}[Triángulo]\label{thm:g7:areas:triangle}
Un triángulo de base $b$ y altura correspondiente $h$ (la
distancia perpendicular del \omterm{def:g5:solids:def}{vértice} opuesto a la recta de la base) tiene
\omterm{def:g6:measure:area}{área}
\[
A = \frac{b \times h}{2} .
\]
\end{theorem}

\begin{proof}
Dos copias del triángulo, una girada media vuelta alrededor del
punto medio de un lado, encajan en un \omterm{def:g7:central:parallelogram}{paralelogramo} de base $b$
y altura $h$ (esa es la \omterm{def:g7:central:def}{simetría central} en acción,
\cref{prop:g7:central:preserved}). El triángulo es la \omterm{def:g3:division:half}{mitad}:
$\frac{bh}{2}$.
\end{proof}

\begin{omfigure}
\begin{tikzpicture}[scale=0.9]
  \fill[omDef!14] (0,0) -- (4,0) -- (1.4,2.1) -- cycle;
  \draw[thick] (0,0) -- (4,0) -- (1.4,2.1) -- cycle;
  \fill[omThm!12] (4,0) -- (1.4,2.1) -- (5.4,2.1) -- cycle;
  \draw[thick, omThm, dashed] (4,0) -- (5.4,2.1) -- (1.4,2.1);
  \draw[dashed] (1.4,0) -- (1.4,2.1);
  \draw[thin] (1.4,0) ++(-0.2,0) -- ++(0,0.2) -- ++(0.2,0);
  \node[below, font=\small] at (2,0) {$b$};
  \node[right, font=\small] at (1.42,1.0) {$h$};
\end{tikzpicture}

{\small Un triángulo y su copia girada media vuelta (discontinua) forman un
\omterm{def:g7:central:parallelogram}{paralelogramo}: el \omterm{def:g6:measure:area}{área} del triángulo es $\frac{b\times h}{2}$. Cualquiera de los
tres lados puede hacer de base --- con \emph{su propia} altura.}
\end{omfigure}

\begin{example}\label{ex:g7:areas:triangle}
Un triángulo tiene una base de $9$ cm y la altura correspondiente mide
$4$ cm:
\[
A = \frac{9 \times 4}{2} = \frac{36}{2} = 18 \text{ cm}^2 .
\]
Para un triángulo \emph{rectángulo}, los dos catetos son una base y su altura:
recuperamos la fórmula de la \cref{prop:g6:measure:formulas}.
\end{example}

\section{Área de un disco}

\begin{theorem}[Disco]\label{thm:g7:areas:disk}
Un disco de \omterm{def:g3:shapes:shapes}{radio} $r$ tiene \omterm{def:g6:measure:area}{área}
\[
A = \pi r^2 .
\]
\end{theorem}

\begin{proof}[Idea de la demostración]
Corta el disco en muchos sectores finos e iguales y colócalos alternando arriba y abajo:
casi forman un \omterm{def:g7:central:parallelogram}{paralelogramo} de altura $r$ cuya base es la \omterm{def:g3:division:half}{mitad} de la
longitud de la circunferencia, $\pi r$ (\cref{prop:g6:measure:circle}). Su \omterm{def:g6:measure:area}{área} es
próxima a $\pi r \times r = \pi r^2$, y la aproximación se vuelve
perfecta a medida que los sectores adelgazan. Una demostración completa necesita el cálculo
integral, del \omterm{def:g6:measure:volume}{volumen} de secundaria superior.
\end{proof}

\begin{tikzpicture}[scale=0.85]
  \foreach \a in {0,45,...,315}
    \fill[omDef!20, draw=omDef] (0,0) -- (\a:1.4)
      arc (\a:\a+45:1.4) -- cycle;
  \begin{scope}[xshift=4.3cm, yshift=-0.15cm]
  \foreach \i in {0,1,2,3} {
    \fill[omDef!20, draw=omDef]
      ({\i*1.1},0) -- ({\i*1.1+0.55},1.4) -- ({\i*1.1+1.1},0) -- cycle;
    \fill[omDef!20, draw=omDef]
      ({\i*1.1+0.55},1.4) -- ({\i*1.1+1.1},0) -- ({\i*1.1+1.65},1.4)
      -- cycle;
  }
  \draw[Stealth-Stealth] (5.2,0) -- (5.2,1.4)
    node[midway, right, font=\small] {$\approx r$};
  \node[below, font=\small] at (2.5,-0.1) {base $\approx \pi r$};
  \end{scope}
\end{tikzpicture}

{\small Cortar el disco en sectores y desenrollarlos: casi un
\omterm{def:g7:central:parallelogram}{paralelogramo} de base $\pi r$ (la \omterm{def:g3:division:half}{mitad} del borde) y altura $r$.}
\end{omfigure}

\begin{example}\label{ex:g7:areas:disk}
El tablero de una mesa redonda tiene \omterm{def:g3:shapes:shapes}{radio} $60$ cm. Su \omterm{def:g6:measure:area}{área} es
$\pi \times 60^2 = 3600\pi \approx 11\,310$ cm$^2$, algo más de
$1.1$ m$^2$. Fíjate en la \omterm{ex:g1:subtraction:difference}{diferencia}: el \emph{\omterm{def:g6:measure:perimeter}{perímetro}} $2\pi r$ usa
$r$ y el \emph{\omterm{def:g6:measure:area}{área}} $\pi r^2$ usa $r^2$.
\end{example}

\section{Prismas y cilindros}

\begin{definition}[Prisma, cilindro]\label{def:g7:areas:prism}
Un \emph{prisma}\index{prisma} es un cuerpo con dos \omterm{def:g5:solids:def}{caras} poligonales
iguales y \omterm{def:g6:lines:perp}{paralelas} (las \emph{bases}) unidas por rectángulos; un
\emph{cilindro}\index{cilindro} es lo mismo con discos como bases. La
\emph{altura} es la distancia entre las dos bases.
\end{definition}

\begin{theorem}[Volúmenes]\label{thm:g7:areas:volume}
Para un \omterm{def:g7:areas:prism}{prisma} o un \omterm{def:g7:areas:prism}{cilindro} de \omterm{def:g6:measure:area}{área} de la base $B$ y altura $h$:
\[
V = B \times h .
\]
\end{theorem}
\admitted

\begin{example}\label{ex:g7:areas:volume}
Una tienda de campaña es un \omterm{def:g7:areas:prism}{prisma} tumbado: sus bases son triángulos de base
$2$ m y altura $1.5$ m, y la tienda mide $3$ m de largo.
\begin{enumerate}
  \item \omterm{def:g6:measure:area}{área} de la base: $B = \frac{2 \times 1.5}{2} = 1.5$ m$^2$;
  \item \omterm{def:g6:measure:volume}{volumen}: $V = B \times h = 1.5 \times 3 = 4.5$ m$^3$.
\end{enumerate}
Una lata de \omterm{def:g3:shapes:shapes}{radio} $4$ cm y altura $11$ cm:
$V = \pi \times 4^2 \times 11 = 176\pi \approx 553$ cm$^3$, aproximadamente
medio litro.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g7:areas:1}
Calcula el \omterm{def:g6:measure:area}{área} de un \omterm{def:g7:central:parallelogram}{paralelogramo} de base $8$ cm y altura
$5$ cm; y la de otro de base $6.5$ m y altura $4$ m.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:areas:2}
Un \omterm{def:g7:central:parallelogram}{paralelogramo} tiene lados de $10$ cm y $6$ cm, y la altura relativa
a la base de $10$ cm mide $4$ cm. Calcula su \omterm{def:g6:measure:area}{área}. ¿Por qué la
respuesta no es $60$ cm$^2$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:areas:3}
Calcula el \omterm{def:g6:measure:area}{área} de un triángulo de base $12$ cm y altura $7$ cm;
la de un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} de catetos $9$ y $6$; y la de un triángulo de base
$5.5$ m y altura $2$ m.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:areas:4}
Calcula el \omterm{def:g6:measure:area}{área} de un disco de \omterm{def:g3:shapes:shapes}{radio} $5$ cm y después la de un semidisco de
\omterm{def:g3:shapes:shapes}{radio} $10$ cm (valores exactos con $\pi$ y después redondeados a la unidad).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:areas:5}
Calcula el \omterm{def:g6:measure:perimeter}{perímetro} \emph{y} el \omterm{def:g6:measure:area}{área} de un disco de \omterm{def:g3:shapes:shapes}{radio} $3$ cm.
¿Cuál de las dos respuestas va en cm y cuál en cm$^2$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:areas:6}
Calcula el \omterm{def:g6:measure:volume}{volumen} de un \omterm{def:g7:areas:prism}{prisma} de \omterm{def:g6:measure:area}{área} de la base $24$ cm$^2$ y altura
$10$ cm; y el de un \omterm{def:g7:areas:prism}{cilindro} de \omterm{def:g3:shapes:shapes}{radio} $3$ cm y altura $8$ cm (exacto
y después redondeado).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:areas:7}
Un triángulo tiene \omterm{def:g6:measure:area}{área} $28$ cm$^2$ y base $8$ cm. Halla la
altura correspondiente, escribiendo antes la ecuación.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:areas:8}
Dibuja un triángulo cualquiera y mide con cuidado las tres parejas base--altura.
Calcula $\frac{b \times h}{2}$ para cada pareja: los tres resultados deberían
coincidir (salvo error de medida). ¿Por qué?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:areas:9}
Un jardín con forma de trapecio se puede partir, con una diagonal, en dos
triángulos que comparten la misma altura $h = 20$ m, con bases de $30$ m y
$18$ m. Calcula su \omterm{def:g6:measure:area}{área} total. ¿Sabrías adivinar una fórmula general para
el trapecio?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:areas:10}
Un jarrón cilíndrico de \omterm{def:g3:shapes:shapes}{radio} $6$ cm contiene agua hasta una altura de
$15$ cm. Toda el agua se vierte en un \omterm{def:g5:solids:def}{ortoedro} vacío de base
$18$ cm $\times$ $12$ cm. ¿Qué altura de agua se alcanza en el
\omterm{def:g5:solids:def}{ortoedro}? (El mismo \omterm{def:g6:measure:volume}{volumen}, distinta base.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:areas:11}
Una pizza de \omterm{def:g4:geometry:circle}{diámetro} $30$ cm cuesta $9$ euros; otra de \omterm{def:g4:geometry:circle}{diámetro} $40$ cm
cuesta $14$ euros. Calcula las dos \omterm{def:g6:measure:area}{áreas} y el precio por cada $100$
cm$^2$. ¿Qué pizza sale mejor de precio?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:areas:12}
Los dos catetos de un \omterm{def:g6:shapes:triangles}{triángulo rectángulo} miden $6$ cm y $8$ cm, y su
hipotenusa mide $10$ cm. Calcula su \omterm{def:g6:measure:area}{área} con los catetos y usa después
ese \omterm{def:g6:measure:area}{área} para hallar la altura relativa a la \emph{hipotenusa}.
\end{exercise}

\section{Problema: El cuadrado que se esfuma}

\begin{problem}[{Problema de fin de semana --- una «demostración» famosa de que
$64 = 65$, desenmascarada con áreas y pendientes, con pizzas y coronas de
postre}]
\label{pb:g7:areas:1}
Un mago corta un cuadrado de $8 \times 8$ en cuatro piezas, las desliza
y las vuelve a montar formando un rectángulo de $5 \times 13$.
El público cuenta: $8 \times 8 = 64$, pero
$5 \times 13 = 65$ --- ¡ha aparecido un cuadradito de la nada!
Las fórmulas de \omterm{def:g6:measure:area}{área} de este capítulo son exactamente las herramientas que
pillan el truco. Después el problema pone a trabajar el razonamiento honesto con \omterm{def:g6:measure:area}{áreas}:
en trapecios, en pilas inclinadas, en la economía de las pizzas --- y,
por último, vuelve al mago para una segunda función aún más
extraña.

\medskip
\noindent\textbf{Parte I --- El truco, ejecutado y desenmascarado.}
Las cuatro piezas del cuadrado de $8 \times 8$ son: dos \omterm{def:g6:shapes:triangles}{triángulos
rectángulos} de catetos $8$ y $3$, y dos trapecios rectángulos de
lados paralelos $5$ y $3$ y altura $5$.
\begin{enumerate}
  \item calcula el \omterm{def:g6:measure:area}{área} del cuadrado original y el \omterm{def:g6:measure:area}{área} del
        rectángulo que se afirma. Enuncia el escándalo en una línea;
  \item calcula el \omterm{def:g6:measure:area}{área} de cada una de las cuatro piezas
        (\cref{thm:g7:areas:triangle}; para los trapecios, córtalos
        o usa el \cref{exo:g7:areas:9});
  \item \omterm{def:g1:addition:def}{suma} las cuatro \omterm{def:g6:measure:area}{áreas}. ¿Cuál de las dos figuras --- el cuadrado
        o el rectángulo --- pueden llenar realmente las piezas?
  \item en la disposición rectangular, la hipotenusa de un triángulo
        (que sube $3$ en $8$) debería continuar el borde inclinado de
        un trapecio (que sube $2$ en $5$) formando una sola
        «diagonal» recta. Comprueba la alineación: ¿son $3$ en
        $8$ y $2$ en $5$ la misma inclinación
        (\cref{thm:g7:prop:cross})?
  \item explica dónde se esconde el sexagésimo quinto cuadradito: qué forma tiene
        el hueco a lo largo de la falsa diagonal, cuál es su \omterm{def:g6:measure:area}{área} exacta
        y por qué se le escapa al ojo. (¿Qué anchura media tiene una
        astilla de esa \omterm{def:g6:measure:area}{área} estirada a lo largo de la diagonal del rectángulo,
        que mide unas $13$ unidades?)
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Razonar honradamente con \omterm{def:g6:measure:area}{áreas}.}
\begin{enumerate}[resume]
  \item traza dos \omterm{def:g6:lines:perp}{rectas paralelas} separadas $4$ cm, una base de $6$ cm sobre
        una de ellas y tres triángulos distintos sobre esa base
        con los \omterm{def:g5:solids:def}{vértices} en puntos diversos de la otra recta. Calcula
        las tres \omterm{def:g6:measure:area}{áreas}. ¿Por qué son todas iguales, deslice el \omterm{def:g5:solids:def}{vértice}
        donde deslice a lo largo de su recta?
  \item demuestra la fórmula del trapecio en general: un trapecio de
        lados paralelos $B$ y $b$ y altura $h$, cortado por una
        diagonal, da dos triángulos de la misma altura $h$.
        Concluye:
        \[
        A = \frac{(B + b) \times h}{2} .
        \]
  \item aplícala a un trapecio con $B = 9$ cm, $b = 5$ cm y
        $h = 4$ cm.
  \item los carpinteros usan la misma fórmula disfrazada: \omterm{def:g6:measure:area}{área} $=$
        (media de los dos lados paralelos) $\times$ altura.
        Explica por qué es la misma regla y recalcula la pregunta
        8 de esa manera;
  \item inclina un montón bien apilado de cartas hasta dejarlo torcido.
        Explica por qué el \omterm{def:g6:measure:volume}{volumen} no ha cambiado y qué dice esto
        sobre la altura en la fórmula del \omterm{def:g7:areas:prism}{prisma}
        $V = B \times h$ (\cref{thm:g7:areas:volume}): ¿qué
        altura hay que tomar en una pila torcida, la longitud
        inclinada o la vertical? (Compara con el
        \cref{exo:g7:areas:2}, una dimensión más arriba.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Coronas, pizzas y el regreso del
mago.}
\begin{enumerate}[resume]
  \item un estanque circular de \omterm{def:g3:shapes:shapes}{radio} $3$ m está en el centro de un
        césped circular de \omterm{def:g3:shapes:shapes}{radio} $5$ m. Calcula el \omterm{def:g6:measure:area}{área} de la
        corona de césped (exacta con $\pi$ y después redondeada al
        m$^2$).
  \item halla el \omterm{def:g3:shapes:shapes}{radio} del único disco cuya \omterm{def:g6:measure:area}{área} sea exactamente
        igual al \omterm{def:g6:measure:area}{área} de esa corona. (Los tres \omterm{def:g3:shapes:shapes}{radios} que ya tienes
        forman un trío famoso, cuya historia se cuenta en el
        \cref{ch:g8:pythagoras}.)
  \item ¿qué es más pizza: una pizza de \omterm{def:g4:geometry:circle}{diámetro} $40$ cm o
        dos pizzas de \omterm{def:g4:geometry:circle}{diámetro} $28$ cm? Calcula y compara;
  \item para \emph{duplicar} el \omterm{def:g6:measure:area}{área} de una pizza, ¿por qué factor
        aproximado tiene que crecer su \omterm{def:g3:shapes:shapes}{radio}? Comprueba el candidato $1.4$ (recuerda
        que las \omterm{def:g6:measure:area}{áreas} siguen el \emph{cuadrado} del factor de
        ampliación, \cref{pb:g6:propdata:1}) y di dónde hace su
        entrada oficial el factor exacto --- el número cuyo cuadrado es $2$
        (\cref{ex:g8:pythagoras:diagonal}).
  \item el mago vuelve: corta un cuadrado de $13 \times 13$
        con los números $5$, $8$ y $13$ haciendo los papeles que
        antes hacían $3$, $5$ y $8$, y monta un
        rectángulo de $8 \times 21$. Calcula las dos \omterm{def:g6:measure:area}{áreas}: ¿qué
        se esfuma esta vez? Los números $3, 5, 8, 13, 21$ son
        términos consecutivos de la sucesión del
        \cref{pb:g7:fractions:1} --- enuncia el patrón de las
        ganancias y pérdidas del mago, y la moraleja: ¿por qué las \omterm{def:g6:measure:area}{áreas}
        no mienten nunca?
\end{enumerate}
\end{problem}
