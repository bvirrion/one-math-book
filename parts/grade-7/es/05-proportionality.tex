\section{Reconocer y completar tablas.}

\begin{definition}[tabla de proporcionalidad]\label{def:g7:prop:table}
Una tabla de dos filas es una \emph{tabla de proporcionalidad}\index{tabla de proporcionalidad} cuando los números de la segunda fila son
los del primero multiplicados por un número fijo, el
\emph{coeficiente}. Equivalentemente: todas las columnas \omterm{def:g3:division:remainder}{cocientes}
$\frac{\text{bottom}}{\text{top}}$ son iguales.
\end{definition}

\begin{example}\label{ex:g7:prop:recognize}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|ccc}
top & $4$ & $10$ & $14$ \\
\hline
bottom & $6$ & $15$ & $21$ \\
\end{tabular}
\end{center}
\omterm{def:g3:division:remainder}{Cocientes}: $\frac64 = 1.5$, $\frac{15}{10} = 1.5$,
$\frac{21}{14} = 1.5$: \omterm{def:g6:propdata:def}{proporcional}, coeficiente $1.5$.
\end{example}

\begin{theorem}[regla cruzada]\label{thm:g7:prop:cross}
en un \omterm{def:g7:prop:table}{tabla de proporcionalidad} con columnas $\begin{pmatrix} a \\ b
\end{pmatrix}$ y $\begin{pmatrix} c \\ d \end{pmatrix}$, la cruz
\omterm{def:g2:mult:def}{productos} son iguales:
\[
a \times d = b \times c .
\]
En consecuencia, el cuarto valor se puede calcular a partir de los otros tres:
$ d = \dfrac{b \times c}{a}$.
\end{theorem}

\begin{proof}
Sea $ k $ sea el coeficiente: $ b = ka $ y $ d = kc $. Entonces
\[
a \times d = a \times kc = k \times ac,
\qquad
b \times c = ka \times c = k \times ac :
\]
ambos se cruzan \omterm{def:g2:mult:def}{productos} igual $ k \times ac $. Divisor $ ad = bc $ por $ a $
da la fórmula para $ d $.
\end{proof}

\begin{example}[Cuarto proporcional]\label{ex:g7:prop:fourth}
Si $7$ pesan libros idénticos $2.8$ kg ¿cuánto pesa? $12$ tales libros
pesar? Regla de tabla y cruz:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cc}
books & $7$ & $12$ \\
\hline
kg & $2.8$ & $ m $ \\
\end{tabular}
\end{center}
\[
7 \times m = 2.8 \times 12
\quad\Longrightarrow\quad
m = \frac{2.8 \times 12}{7} = \frac{33.6}{7} = 4.8 \text{ kg}.
\]
Consultar por unidad: un libro pesa $0.4$ kg, pesan doce $4.8$ kg.
\end{example}

\section{Porcentajes}

\begin{method}[Aplicar y encontrar porcentajes.]\label{met:g7:prop:percent}
\begin{enumerate}
  \item \emph{Aplicar} $ t\,\%$: multiplicar por $\frac{t}{100}$
        ($12\,\%$ de $350$ es $0.12 \times 350 = 42$);
  \item \emph{encontrar} el porcentaje que representa una parte: calcular
        $\frac{\text{part}}{\text{whole}}$ y reescribir $100$
        ($21$ fuera de $60$: $\frac{21}{60} = \frac{35}{100} = 35\,\%$);
  \item pregunta siempre: \emph{¿un porcentaje de qué?} El conjunto de referencia
        importa.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:prop:percent}
En una escuela, $180$ del $400$ los estudiantes son niñas. Porcentaje:
\[
\frac{180}{400} = \frac{180 \div 4}{400 \div 4} = \frac{45}{100}
= 45\,\% .
\]
Al revés: ¿cuántos son los $55\,\%$ chicos?
$0.55 \times 400 = 220$. Controlar: $180 + 220 = 400$.
\end{example}

\section{Balanza}

\begin{definition}[Escala]\label{def:g7:prop:scale}
En un mapa o modelo en \emph{escalar}\index{escalar}
$\frac{1}{n}$ (escrito $1 : n $), las distancias en el mapa son
\omterm{def:g6:propdata:def}{proporcional} a las distancias reales, con coeficiente $\frac1n $:
\[
\text{map distance} = \frac{1}{n} \times \text{real distance},
\]
ambos medidos en el \emph{misma unidad}.
\end{definition}

\begin{example}\label{ex:g7:prop:scale}
en un $1 : 50\,000$ mapa, dos ciudades son $7$ cm de distancia. Distancia real:
\[
7 \times 50\,000 = 350\,000 \text{ cm} = 3\,500 \text{ m} = 3.5
\text{ km}.
\]
Por el contrario, un $12$ km medidas de la ruta de senderismo en el mapa
$12$ kilómetros $= 1\,200\,000$ cm, entonces $1\,200\,000 \div 50\,000 = 24$ centímetro.
\end{example}

\begin{remark}
Los gráficos son la prueba más rápida de \omterm{def:g7:prop:table}{proporcionalidad}
(\cref{ch:g6:propdata}): puntos en una línea recta \emph{a través del
origen} significar \omterm{def:g6:propdata:def}{proporcional} cantidades; una línea a la que le falta el origen (como
una tarifa de taxi con un cargo fijo) significa \emph{no} \omterm{def:g6:propdata:def}{proporcional}, incluso
aunque es una línea recta. Funciones afines, en
\cref{ch:g9:linfunc}, hará esto preciso.
\end{remark}
