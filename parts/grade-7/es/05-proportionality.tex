\chapter{La proporcionalidad}\label{ch:g7:prop}

El año pasado vimos las tablas \omterm{def:g6:propdata:def}{proporcionales} (\cref{ch:g6:propdata}); este capítulo
convierte la \omterm{def:g7:prop:table}{proporcionalidad} en una herramienta completa: la regla de los \omterm{def:g2:mult:def}{productos} cruzados para
hallar un \omterm{def:g3:division:half}{cuarto} valor, los porcentajes y las \omterm{def:g7:prop:scale}{escalas} de mapas y maquetas.
La velocidad, la \omterm{def:g7:prop:table}{proporcionalidad} más famosa de todas, se estudia en el
\cref{ch:g8:speed}.

\section{Reconocer y completar tablas}

\begin{definition}[Tabla de proporcionalidad]\label{def:g7:prop:table}
Una tabla de dos filas es una \emph{tabla de
proporcionalidad}\index{proporcionalidad} cuando los números de la segunda fila son
los de la primera multiplicados por un número fijo, el
\emph{coeficiente}. Equivalentemente: todos los \omterm{def:g3:division:remainder}{cocientes} por columnas
$\frac{\text{fila de abajo}}{\text{fila de arriba}}$ son iguales.
\end{definition}

\begin{example}\label{ex:g7:prop:recognize}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|ccc}
arriba & $4$ & $10$ & $14$ \\
\hline
abajo & $6$ & $15$ & $21$ \\
\end{tabular}
\end{center}
\omterm{def:g3:division:remainder}{Cocientes}: $\frac64 = 1.5$, $\frac{15}{10} = 1.5$,
$\frac{21}{14} = 1.5$: proporcional, de coeficiente $1.5$.
\end{example}

\begin{theorem}[Regla de los productos cruzados]\label{thm:g7:prop:cross}
En una \omterm{def:g7:prop:table}{tabla de proporcionalidad} con columnas $\begin{pmatrix} a \\ b
\end{pmatrix}$ y $\begin{pmatrix} c \\ d \end{pmatrix}$, los \omterm{def:g2:mult:def}{productos}
cruzados son iguales:
\[
a \times d = b \times c .
\]
Por consiguiente, el \omterm{def:g3:division:half}{cuarto} valor se puede calcular a partir de los otros tres:
$d = \dfrac{b \times c}{a}$.
\end{theorem}

\begin{proof}
Sea $k$ el coeficiente: $b = ka$ y $d = kc$. Entonces
\[
a \times d = a \times kc = k \times ac,
\qquad
b \times c = ka \times c = k \times ac :
\]
los dos \omterm{def:g2:mult:def}{productos} cruzados valen $k \times ac$. Dividir $ad = bc$ entre $a$
da la fórmula de $d$.
\end{proof}

\begin{example}[La cuarta proporcional]\label{ex:g7:prop:fourth}
Si $7$ libros iguales pesan $2.8$ kg, ¿cuánto pesan $12$ libros
como esos? Tabla y regla de los \omterm{def:g2:mult:def}{productos} cruzados:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cc}
libros & $7$ & $12$ \\
\hline
kg & $2.8$ & $m$ \\
\end{tabular}
\end{center}
\[
7 \times m = 2.8 \times 12
\quad\Longrightarrow\quad
m = \frac{2.8 \times 12}{7} = \frac{33.6}{7} = 4.8 \text{ kg}.
\]
Comprobación por la unidad: un libro pesa $0.4$ kg y doce pesan $4.8$ kg.
\end{example}

\section{Porcentajes}

\begin{method}[Aplicar y hallar porcentajes]\label{met:g7:prop:percent}
\begin{enumerate}
  \item \emph{aplicar} el $t\,\%$: multiplica por $\frac{t}{100}$
        (el $12\,\%$ de $350$ es $0.12 \times 350 = 42$);
  \item \emph{hallar} el porcentaje que representa una parte: calcula
        $\frac{\text{parte}}{\text{total}}$ y reescríbelo sobre $100$
        ($21$ de $60$: $\frac{21}{60} = \frac{35}{100} = 35\,\%$);
  \item pregúntate siempre: \emph{¿porcentaje de qué?} El total de referencia
        importa.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:prop:percent}
En un colegio, $180$ de los $400$ alumnos son chicas. Porcentaje:
\[
\frac{180}{400} = \frac{180 \div 4}{400 \div 4} = \frac{45}{100}
= 45\,\% .
\]
Al revés: ¿cuántos son el $55\,\%$ de chicos?
$0.55 \times 400 = 220$. Comprobación: $180 + 220 = 400$.
\end{example}

\section{Escalas}

\begin{definition}[Escala]\label{def:g7:prop:scale}
En un mapa o una maqueta a \emph{escala}\index{escala}
$\frac{1}{n}$ (que se escribe $1 : n$), las distancias del mapa son
\omterm{def:g6:propdata:def}{proporcionales} a las distancias reales, con coeficiente $\frac1n$:
\[
\text{distancia en el mapa} = \frac{1}{n} \times \text{distancia real},
\]
medidas las dos en la \emph{misma unidad}.
\end{definition}

\begin{example}\label{ex:g7:prop:scale}
En un mapa $1 : 50\,000$, dos pueblos distan $7$ cm. Distancia real:
\[
7 \times 50\,000 = 350\,000 \text{ cm} = 3\,500 \text{ m} = 3.5
\text{ km}.
\]
A la inversa, un sendero de $12$ km mide en el mapa:
$12$ km $= 1\,200\,000$ cm, así que $1\,200\,000 \div 50\,000 = 24$ cm.
\end{example}

\end{axis}
\end{tikzpicture}

{\small Una \omterm{def:g7:prop:scale}{escala} es una \omterm{def:g7:prop:table}{proporcionalidad}: las distancias del mapa frente a las
distancias reales se alinean en una recta que pasa por el origen (aquí,
$1$ cm $\leftrightarrow$ $4$ km; las líneas discontinuas leen $7$ cm
$\leftrightarrow$ $28$ km).}
\end{omfigure}

\begin{remark}
Los gráficos son la prueba más rápida de \omterm{def:g7:prop:table}{proporcionalidad}
(\cref{ch:g6:propdata}): puntos alineados en una recta \emph{que pasa por el
origen} significan magnitudes \omterm{def:g6:propdata:def}{proporcionales}; una recta que no pasa por el origen (como
la tarifa de un taxi con un cargo fijo) significa que \emph{no} son \omterm{def:g6:propdata:def}{proporcionales}, aunque
sea una recta. Las funciones afines, en el
\cref{ch:g9:linfunc}, precisarán esto.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g7:prop:1}
¿Cuáles son \omterm{def:g7:prop:table}{tablas de proporcionalidad}? Da el coeficiente cuando
exista.
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|ccc}
 & $3$ & $7$ & $11$ \\
\hline
 & $7.5$ & $17.5$ & $27.5$ \\
\end{tabular}
\qquad
\begin{tabular}{l|ccc}
 & $2$ & $5$ & $9$ \\
\hline
 & $6$ & $15$ & $28$ \\
\end{tabular}
\end{center}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:prop:2}
Completa con la regla de los \omterm{def:g2:mult:def}{productos} cruzados, escribiendo el cálculo:
$5$ kg de patatas cuestan $6$ euros; ¿cuánto cuestan $8$ kg?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:prop:3}
Una impresora imprime $36$ páginas en $3$ minutos. Al mismo ritmo, ¿cuántas
páginas en $5$ minutos? ¿Cuánto tarda en $96$ páginas?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:prop:4}
Calcula: el $30\,\%$ de $250$; \quad el $15\,\%$ de $60$; \quad el $t\,\%$ tal
que el $t\,\%$ de $80$ sea $20$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:prop:5}
De $25$ lanzamientos, una jugadora de baloncesto encestó $18$. ¿Cuál es su
porcentaje de acierto? (Reescribe la \omterm{def:g6:fractions:def}{fracción} sobre $100$.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:prop:6}
En un mapa $1 : 25\,000$, un camino mide $9$ cm. ¿Cuál es su longitud
real en km? Un lago mide $2$ km de largo: ¿cuánto mide en el mapa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:prop:7}
Una maqueta de coche está construida a \omterm{def:g7:prop:scale}{escala} $1 : 43$. El coche real mide $4.3$ m de largo.
¿Cuánto mide la maqueta, en cm?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:prop:8}
Una receta usa $240$ g de chocolate para $8$ raciones. Alina tiene
$300$ g de chocolate. ¿Para cuántas raciones le llega (número
entero de raciones)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:prop:9}
El precio de una chaqueta baja de $80$ a $60$ euros.
\begin{enumerate}
  \item ¿cuál es la rebaja en euros? ¿Y en porcentaje del precio
        inicial?
  \item más tarde el precio vuelve a subir de $60$ a $80$ euros. Explica
        por qué esa subida \emph{no} es del $25\,\%$ --- calcula el
        porcentaje de aumento correcto.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:prop:10}
Dos filas de una tabla son \omterm{def:g6:propdata:def}{proporcionales}:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|ccc}
 & $4$ & $x$ & $10$ \\
\hline
 & $y$ & $10.5$ & $17.5$ \\
\end{tabular}
\end{center}
Halla el coeficiente y después $x$ e $y$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:prop:11}
Una fotocopiadora reduce los documentos al $80\,\%$ de su tamaño. Se copia un \omterm{def:g6:lines:objects}{segmento}
de $10$ cm y después se vuelve a copiar la \emph{copia} con el
mismo ajuste.
\begin{enumerate}
  \item ¿cuánto mide el \omterm{def:g6:lines:objects}{segmento} después de una copia? ¿Y después de dos?
  \item ¿por qué la respuesta tras dos copias no es el $60\,\%$ del
        original? ¿Qué porcentaje único corresponde a dos copias?
\end{enumerate}
\end{exercise}

\section{Problema: Medir la Tierra con un palo}

\begin{problem}[{Problema de fin de semana --- las sombras son una
proporcionalidad, y cómo Eratóstenes calculó la circunferencia
de la Tierra en el año 240 a.\,C.}]
\label{pb:g7:prop:1}
Hace veintidós siglos, sin telescopio, sin satélite y sin
calculadora, el bibliotecario de Alejandría midió la Tierra ---
con un palo, un pozo, una caravana de camellos y las matemáticas de este
capítulo. Este problema recorre sus pasos: primero la
\omterm{def:g7:prop:table}{proporcionalidad} de las sombras, después el famoso cálculo en sí
y, por último, cómo queda su respuesta a \omterm{def:g7:prop:scale}{escala} humana.

\medskip
\noindent\textbf{Parte I --- La sombra de un palo.}
El Sol está tan lejos que sus rayos nos llegan paralelos unos a
otros. En un instante dado, todos los objetos verticales y sus
sombras son, por tanto, \omterm{def:g6:propdata:def}{proporcionales}.
\begin{enumerate}
  \item a mediodía, un palo vertical de $1$ m proyecta una sombra de
        $0.4$ m y la sombra de un árbol mide $3.2$ m. ¿Cuánto mide
        el árbol (\cref{thm:g7:prop:cross})?
  \item en el mismo instante, una torre proyecta una sombra de $26$ m: ¿cuánto
        mide la torre? ¿Y cuánto mide la sombra de un niño
        de $1.5$ m?
  \item explica en una frase por qué todos estos cálculos solo son
        válidos en el \emph{mismo instante} del día;
  \item cuenta la leyenda que Tales midió la Gran \omterm{def:g5:solids:def}{Pirámide} esperando
        al momento en que \emph{su propia sombra medía exactamente lo
        mismo que él}. ¿Cuál es el coeficiente de \omterm{def:g7:prop:table}{proporcionalidad}
        en ese momento y cuánto mide la \omterm{def:g5:solids:def}{pirámide}
        si la punta de su sombra queda a $147$ m del punto
        del suelo situado justo debajo de su cúspide?
  \item resume la Parte I: en un instante dado, ¿qué magnitud es
        la misma para el palo, el árbol, la torre y la
        \omterm{def:g5:solids:def}{pirámide} (\cref{def:g7:prop:table})?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El cálculo de Eratóstenes.}
Eratóstenes sabía dos cosas. En la ciudad de Siena, a mediodía del
solsticio de verano, el Sol quedaba \emph{exactamente en la vertical}: la luz
llegaba al fondo de los pozos más profundos y los palos verticales no proyectaban
ninguna sombra. En Alejandría, $800$ km al norte, en el mismo instante,
un palo vertical \emph{sí} proyectaba sombra y los rayos del Sol
formaban un ángulo de $7.2^\circ$ con la vertical.
\begin{enumerate}[resume]
  \item explica por qué estas dos observaciones juntas demuestran que
        el suelo de Siena y el de Alejandría no son
        paralelos --- es decir, que la superficie de la Tierra es curva.
        (¿Qué harían unos rayos paralelos con dos palos en una Tierra
        plana?)
  \item en un dibujo de la Tierra redonda con rayos solares paralelos,
        los $7.2^\circ$ de Alejandría reaparecen en el centro de
        la Tierra, como el ángulo entre las direcciones de las
        dos ciudades. Haz el dibujo y convéncete (la
        justificación limpia usa las parejas de ángulos iguales del
        \cref{ch:g7:triangles}).
  \item ¿qué \omterm{def:g6:fractions:def}{fracción} de vuelta completa son $7.2^\circ$?
  \item el arco de Siena a Alejandría ($800$ km) corresponde
        a $7.2^\circ$; la circunferencia entera corresponde a
        $360^\circ$. Plantea la \omterm{def:g7:prop:table}{proporcionalidad} y calcula la
        circunferencia de la Tierra;
  \item deduce el \omterm{def:g4:geometry:circle}{diámetro} de la Tierra usando $\pi \approx
        3.14$ (\cref{prop:g6:measure:circle}) y redondeando a la
        centena de kilómetros más próxima. (El valor moderno es
        $12\,742$ km: ¿cuánto se acercó un hombre con un palo en el
        año 240 a.\,C.?)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La Tierra a \omterm{def:g7:prop:scale}{escala} humana.}
\begin{enumerate}[resume]
  \item se construye un globo terráqueo a \omterm{def:g7:prop:scale}{escala} $1 : 40\,000\,000$. Con la
        circunferencia hallada en la pregunta 9, muestra que la
        circunferencia del globo es exactamente $1$ m. ¿Cuál es su
        \omterm{def:g4:geometry:circle}{diámetro}, al centímetro más próximo?
  \item el Mont Blanc se eleva unos $4.8$ km. Convierte $4.8$ km a
        centímetros, divide entre $40\,000\,000$ y expresa la
        altura de la montaña en el globo en milímetros. ¿Qué dice
        la respuesta sobre lo «rugosa» que es en realidad la Tierra?
  \item un caminante recorre $40$ km al día. A ese ritmo, ¿cuántos
        días necesita para la circunferencia entera de Eratóstenes y
        aproximadamente cuántos años son?
  \item la atmósfera respirable se concentra aproximadamente en los
        primeros $10$ km sobre el suelo. ¿Qué porcentaje del
        \omterm{def:g3:shapes:shapes}{radio} de la Tierra (unos $6\,370$ km) es eso
        (\cref{met:g7:prop:percent})? Redondea a la décima
        de punto porcentual;
  \item los historiadores estiman que el valor anunciado por Eratóstenes,
        convertido a unidades modernas, pudo ser de unos
        $42\,000$ km. Tomando $40\,000$ km como valor verdadero,
        calcula su porcentaje de error. Concluye en una
        frase sobre palos, pozos y \omterm{def:g7:prop:table}{proporcionalidad}.
\end{enumerate}
\end{problem}
