\chapter{Simetría central y paralelogramos}\label{ch:g7:central}

Gira una figura media vuelta alrededor de un punto: esto es la \emph{\omterm{def:g7:central:def}{simetría
central}}, la segunda transformación del libro después de las reflexiones
del \cref{ch:g6:symmetry}. Su figura estrella es el \omterm{def:g7:central:parallelogram}{paralelogramo} --- el
\omterm{def:g4:geometry:polygon}{cuadrilátero} \omterm{def:g7:central:def}{simétrico} respecto del punto de corte de sus propias
diagonales.

\section{La simetría central}

\begin{definition}[Simétrico respecto de un punto]\label{def:g7:central:def}
El \emph{simétrico}\index{simetría central} de un punto $M$ respecto de un
punto $O$ es el punto $M'$ tal que $O$ es el \emph{punto medio} de
$[MM']$. La figura simétrica es en lo que se convierte la original tras un
medio giro ($180^\circ$) alrededor de $O$.
\end{definition}

\begin{method}[Construir el simétrico de un punto]\label{met:g7:central:construct}
Para construir el \omterm{def:g7:central:def}{simétrico} $M'$ de $M$ respecto de $O$:
\begin{enumerate}
  \item traza la recta $(MO)$ y prolóngala más allá de $O$;
  \item mide $MO$ (con regla o compás);
  \item coloca $M'$ sobre la prolongación con $OM' = OM$.
\end{enumerate}
En papel cuadriculado: cuenta los pasos horizontales y verticales de $M$ a
$O$ y repite los \emph{mismos} pasos desde $O$ para llegar a $M'$.
\end{method}

\begin{example}[En la cuadrícula]\label{ex:g7:central:grid}
Si $M$ está $3$ casillas a la izquierda y $1$ casilla por encima de $O$, su \omterm{def:g7:central:def}{simétrico}
$M'$ está $3$ casillas a la \emph{derecha} y $1$ casilla \emph{por debajo} de $O$:
la \omterm{def:g7:central:def}{simetría central} invierte las dos direcciones a la vez (una reflexión
invierte solo una).
\end{example}

\begin{proposition}[Qué conserva un medio giro]\label{prop:g7:central:preserved}
La \omterm{def:g7:central:def}{simetría central} conserva longitudes, ángulos, \omterm{def:g6:measure:perimeter}{perímetros} y \omterm{def:g6:measure:area}{áreas}; la
imagen de una recta es una recta \emph{paralela}, la imagen de un \omterm{def:g6:lines:objects}{segmento} es un
\omterm{def:g6:lines:objects}{segmento} paralelo de la misma longitud y la imagen de una circunferencia es una circunferencia
del mismo \omterm{def:g3:shapes:shapes}{radio} (con centro en el \omterm{def:g7:central:def}{simétrico} del centro).
\end{proposition}
\admitted

\begin{remark}
Compárala con las reflexiones (\cref{prop:g6:symmetry:preserved}): las dos
conservan longitudes y ángulos; pero la reflexión da la vuelta a las figuras
(una «b» se convierte en una «d»), mientras que el medio giro las deja legibles ---
del revés (una «b» se convierte en una «q»). Y, a \omterm{ex:g1:subtraction:difference}{diferencia} de una reflexión, un
medio giro envía toda recta a una recta paralela a ella.
\end{remark}

\begin{definition}[Centro de simetría]\label{def:g7:central:center}
Un punto $O$ es un \emph{centro de simetría}\index{centro de simetría}
de una figura cuando el medio giro alrededor de $O$ envía la figura exactamente
sobre sí misma. Por ejemplo, las letras S, N y Z tienen centro de
simetría; una circunferencia tiene uno (su centro); un triángulo no tiene ninguno.
\end{definition}

\section{Los paralelogramos}

\begin{definition}[Paralelogramo]\label{def:g7:central:parallelogram}
Un \emph{paralelogramo}\index{paralelogramo} es un \omterm{def:g4:geometry:polygon}{cuadrilátero} cuyas
diagonales se cortan en su punto medio común. Ese punto de corte es entonces
un \omterm{def:g7:central:center}{centro de simetría} de la figura: cada \omterm{def:g5:solids:def}{vértice} es la imagen por el medio giro
del \omterm{def:g5:solids:def}{vértice} opuesto.
\end{definition}

\begin{theorem}[Propiedades de un paralelogramo]\label{thm:g7:central:properties}
En un \omterm{def:g7:central:parallelogram}{paralelogramo} $ABCD$ de centro $O$:
\begin{enumerate}
  \item los lados opuestos son paralelos: $(AB) \parallel (CD)$ y
        $(BC) \parallel (AD)$;
  \item los lados opuestos tienen la misma longitud: $AB = CD$ y
        $BC = AD$;
  \item los ángulos opuestos son iguales: $\widehat A = \widehat C$ y
        $\widehat B = \widehat D$.
\end{enumerate}
\end{theorem}

\begin{proof}
El medio giro alrededor de $O$ envía $A$ a $C$ y $B$ a $D$ (por definición:
$O$ es el punto medio de las dos diagonales). Así que envía el \omterm{def:g6:lines:objects}{segmento}
$[AB]$ al \omterm{def:g6:lines:objects}{segmento} $[CD]$; por la
\cref{prop:g7:central:preserved}, la imagen es paralela a $[AB]$ y
de la misma longitud: puntos 1 y 2. También envía el ángulo en $A$
al ángulo en $C$, y los ángulos se conservan: punto 3.
\end{proof}

\begin{theorem}[Reconocer un paralelogramo]\label{thm:g7:central:recognize}
Un \omterm{def:g4:geometry:polygon}{cuadrilátero} (sin lados que se crucen) es un \omterm{def:g7:central:parallelogram}{paralelogramo} en cuanto se cumple
\emph{una} de estas condiciones:
\begin{enumerate}
  \item sus diagonales tienen el mismo punto medio (la definición);
  \item sus lados opuestos son paralelos dos a dos;
  \item dos lados opuestos son paralelos \emph{y} de la misma longitud.
\end{enumerate}
\end{theorem}
\admitted

\begin{example}\label{ex:g7:central:recognize}
$ABCD$ cumple $(AB) \parallel (CD)$ y $AB = CD = 5$ cm. Se aplica el criterio 3:
$ABCD$ es un \omterm{def:g7:central:parallelogram}{paralelogramo} --- así que, sin ninguna medición
más, sabemos también que $AD = BC$, que $(AD) \parallel (BC)$ y que sus
diagonales se cortan en su punto medio.
\end{example}

\begin{remark}[Paralelogramos especiales]
El rectángulo, el \omterm{def:g6:shapes:quadrilaterals}{rombo} y el cuadrado del \cref{ch:g6:shapes} son exactamente
los \omterm{def:g7:central:parallelogram}{paralelogramos} con una propiedad de más: diagonales iguales
(rectángulo), diagonales \omterm{def:g6:lines:perp}{perpendiculares} (\omterm{def:g6:shapes:quadrilaterals}{rombo}) o las dos cosas (cuadrado). El
argumento de los ángulos prometido en el \cref{ex:g6:shapes:recognize} ya funciona:
en un \omterm{def:g7:central:parallelogram}{paralelogramo}, los ángulos consecutivos son suplementarios
(\omterm{def:g7:triangles:pairs}{ángulos alternos} o correspondientes con las \omterm{def:g6:lines:perp}{paralelas},
\cref{thm:g7:triangles:equalities}), así que un \omterm{def:g3:shapes:rightangle}{ángulo recto} obliga a los
cuatro.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g7:central:1}
En papel cuadriculado, sitúa $O$ y después $M$ cuatro casillas a la derecha de $O$ y una
casilla más arriba. Construye el \omterm{def:g7:central:def}{simétrico} $M'$ de $M$ respecto de $O$ y el
\omterm{def:g7:central:def}{simétrico} de $M$ respecto de la \emph{línea vertical de la cuadrícula} que pasa por $O$.
Compara las dos imágenes.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:2}
Dibuja un triángulo $ABC$ y un punto $O$ fuera de él. Construye su imagen
$A'B'C'$ por el medio giro alrededor de $O$. ¿Qué puedes decir de las
longitudes $A'B'$ y $AB$? ¿Y de las rectas $(A'B')$ y $(AB)$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:3}
¿Qué cifras, escritas como en la pantalla de una calculadora (del $0$ al $9$), tienen
\omterm{def:g7:central:center}{centro de simetría}? ¿Y qué letras mayúsculas de entre H, A, N, S, T, Z y O?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:4}
$KLMN$ es un \omterm{def:g7:central:parallelogram}{paralelogramo} de centro $O$ con $KL = 7$ cm,
$LM = 4$ cm y $\widehat K = 115^\circ$. Da, con razones: $MN$,
$NK$, $\widehat M$ y el punto medio de $[LN]$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:5}
Construye un \omterm{def:g7:central:parallelogram}{paralelogramo} $ABCD$ a partir de sus diagonales: $AC = 8$ cm y
$BD = 5$ cm, que se cortan en sus puntos medios formando un ángulo de $60^\circ$
entre ellas. (¡Traza primero las diagonales!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:6}
En un \omterm{def:g7:central:parallelogram}{paralelogramo}, un ángulo mide $115^\circ$. Usando la observación
sobre los ángulos consecutivos suplementarios, da los otros tres ángulos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:7}
Sitúa $A(1, 1)$, $B(4, 2)$ y el punto $O(2.5, 2.5)$. Calcula las
coordenadas de las imágenes $A'$ y $B'$ por el medio giro alrededor de $O$
(usa el recuento en la cuadrícula del \cref{ex:g7:central:grid}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:8}
$ABCD$ es un \omterm{def:g4:geometry:polygon}{cuadrilátero} en el que $AB = CD$. ¿Basta eso para que sea
un \omterm{def:g7:central:parallelogram}{paralelogramo}? Si no, dibuja un contraejemplo (un trapecio \omterm{def:g6:shapes:triangles}{isósceles}
ayuda) y di qué hay que añadir a la hipótesis.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:9}
Sea $ABC$ un triángulo y $O$ el punto medio de $[BC]$. Construye el
\omterm{def:g7:central:def}{simétrico} $A'$ de $A$ respecto de $O$. Demuestra que $ABA'C$ es un
\omterm{def:g7:central:parallelogram}{paralelogramo}. (¿Qué criterio del \cref{thm:g7:central:recognize} sale
gratis aquí?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:10}
¿Verdadero o falso? Justifícalo: «un \omterm{def:g4:geometry:polygon}{cuadrilátero} cuyos ángulos opuestos
son iguales dos a dos es un \omterm{def:g7:central:parallelogram}{paralelogramo}»; «un \omterm{def:g7:central:parallelogram}{paralelogramo} con
diagonales \omterm{def:g6:lines:perp}{perpendiculares} es un \omterm{def:g6:shapes:quadrilaterals}{rombo}»; «un \omterm{def:g7:central:parallelogram}{paralelogramo} con un
\omterm{def:g3:shapes:rightangle}{ángulo recto} es un rectángulo».
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:central:11}
Sea $ABCD$ un \omterm{def:g4:geometry:polygon}{cuadrilátero} cualquiera y sean $I$, $J$, $K$ y $L$ los puntos medios
de sus lados $[AB]$, $[BC]$, $[CD]$ y $[DA]$. Dibuja varios ejemplos.
¿Qué parece ser siempre $IJKL$? (La demostración, con el teorema
del punto medio, llega en el \cref{ch:g8:midpoints} --- y otra vez con vectores,
en el \omterm{def:g6:measure:volume}{volumen} de secundaria superior.)
\end{exercise}

\section{Problema: La tarta, la mediana y el imposible segundo
centro}

\begin{problem}[{Problema de fin de semana --- el medio giro en acción: cortes justos
de tarta, una cota para las medianas y por qué ninguna figura puede tener dos centros
de simetría}]
\label{pb:g7:central:1}
El medio giro parece la más sencilla de las transformaciones --- y, sin embargo,
corta tartas en \omterm{def:g3:division:half}{mitades} demostrablemente justas, mide las medianas de
un triángulo y esconde una perlita de matemática pura: una
figura acotada puede tener un \omterm{def:g7:central:center}{centro de simetría}, pero \emph{nunca
dos}. La demostración usa un descubrimiento que harás en la Parte I: dos
medios giros, hechos uno detrás de otro, equivalen a un desplazamiento.

\medskip
\noindent\textbf{Parte I --- Gimnasia de medios giros.}
\begin{enumerate}
  \item en papel cuadriculado, toma el origen $O(0, 0)$ como centro.
        Con el recuento en la cuadrícula del \cref{ex:g7:central:grid},
        da las imágenes de $M(3, 1)$, $N(-2, 4)$ y
        $P(0, -5)$ por el medio giro alrededor de $O$. ¿Cuál es la
        regla general para $(x, y)$?
  \item ahora el centro es $C(2, 1)$. Calcula la imagen de
        $M(5, 3)$ y comprueba la receta general: cada coordenada
        de la imagen es \emph{el doble de la del centro menos la
        del punto};
  \item las cartas de la baraja se diseñan para leerse igual del revés
        --- una \omterm{def:g7:central:def}{simetría central}, para que ningún jugador vea la carta
        «al revés». Explica por qué, en una carta con un número
        \emph{\omterm{def:g3:numbers:evenodd}{impar}} de figuras, una de ellas tiene que estar exactamente en el
        centro de la carta;
  \item dos medios giros seguidos: toma los centros $O_1(2, 0)$
        y $O_2(5, 0)$. Manda el punto $A(1, 1)$ por el
        medio giro alrededor de $O_1$ y manda después la imagen por el
        medio giro alrededor de $O_2$. Compara la salida y la llegada. Repítelo
        con $B(0, 3)$. ¿A qué transformación única y sencilla equivalen
        los dos medios giros?
  \item mide el desplazamiento de la pregunta 4 frente a la distancia
        $O_1 O_2$ y enuncia el descubrimiento. (Compáralo con los
        dos espejos paralelos del \cref{pb:g6:symmetry:1}: el mismo
        fenómeno con actores nuevos.)
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Tartas y medianas.}
\begin{enumerate}[resume]
  \item sea $O$ el centro de un \omterm{def:g7:central:parallelogram}{paralelogramo}. Explica por qué
        toda recta que pase por $O$ corta al borde en dos puntos
        \omterm{def:g7:central:def}{simétricos} respecto de $O$, y por qué la recta corta el
        \omterm{def:g7:central:parallelogram}{paralelogramo} en dos trozos que son copias exactas por medio giro
        uno del otro --- y, por tanto, de \omterm{def:g6:measure:area}{áreas} iguales;
  \item el teorema del pastelero: una tarta rectangular se reparte
        con perfecta justicia con \emph{cualquier} corte recto que pase por su
        centro. Aún mejor: una tarta rectangular tiene un agujero rectangular
        (de cualquier tamaño, en cualquier posición e incluso inclinado). Describe el
        único corte recto que da dos partes con la misma cantidad
        de tarta --- y justifícalo con la pregunta 6;
  \item tres \omterm{def:g5:solids:def}{vértices} de un \omterm{def:g7:central:parallelogram}{paralelogramo} $ABCD$ son $A(0, 0)$,
        $B(6, 1)$ y $C(8, 5)$. Calcula las coordenadas de $D$
        y las del centro $O$;
  \item el \Cref{exo:g7:central:9} construyó, a partir de un triángulo $ABC$
        y del punto medio $O$ de $[BC]$, el \omterm{def:g7:central:parallelogram}{paralelogramo}
        $ABA'C$, donde $A'$ es el \omterm{def:g7:central:def}{simétrico} de $A$ respecto de $O$.
        Úsalo, junto con la desigualdad triangular
        (\cref{thm:g7:triangles:inequality}) en el triángulo
        $ABA'$, para demostrar la \emph{cota de la mediana}: la mediana
        $[AO]$ cumple
        \[
        AO < \frac{AB + AC}{2} .
        \]
        (Observa que $AA' = 2\,AO$ y $BA' = AC$.)
  \item en un triángulo con $AB = 5$ cm y $AC = 7$ cm, ¿entre
        qué dos valores tiene que estar la longitud de la mediana desde $A$?
        (Usa los dos sentidos de la desigualdad triangular en
        $ABA'$.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- ¿Cuántos centros puede tener una
figura?}
\begin{enumerate}[resume]
  \item para cada figura, di si tiene \omterm{def:g7:central:center}{centro de
        simetría} y dónde: un \omterm{def:g6:lines:objects}{segmento}; una recta entera; un
        \omterm{def:g6:shapes:triangles}{triángulo equilátero}; un cuadrado; una circunferencia; las letras S
        y Z;
  \item demuestra tu respuesta para el \omterm{def:g6:shapes:triangles}{triángulo equilátero}: un
        medio giro que enviara el triángulo sobre sí mismo emparejaría
        los tres \omterm{def:g5:solids:def}{vértices} --- pero tres es \omterm{def:g3:numbers:evenodd}{impar}, así que algún \omterm{def:g5:solids:def}{vértice}
        iría a parar a sí mismo. ¿A qué obligaría eso a ese
        \omterm{def:g5:solids:def}{vértice} y por qué es imposible?
  \item generaliza la pregunta 12: explica por qué \emph{ningún} \omterm{def:g4:geometry:polygon}{polígono}
        con un \omterm{def:g3:numbers:evenodd}{número impar} de \omterm{def:g5:solids:def}{vértices} tiene \omterm{def:g7:central:center}{centro de
        simetría};
  \item la perla: supón que una figura tuviera \emph{dos} \omterm{def:g7:central:center}{centros
        de simetría} distintos $O_1$ y $O_2$. Haz los dos
        medios giros uno detrás de otro y usa el descubrimiento de la
        pregunta 5: ¿qué movimiento tendría que enviar la figura exactamente
        sobre sí misma? Explica por qué una figura que cabe en una hoja
        de papel no puede sobrevivir a eso y concluye;
  \item la conclusión de la pregunta 14 deja un resquicio: las figuras
        \emph{no acotadas}. Verifica en un friso infinito
        --- por ejemplo, una fila interminable de huellas, izquierda, derecha,
        izquierda, derecha\,\dots\ --- que tiene (al menos) dos
        \omterm{def:g7:central:center}{centros de simetría} distintos e identifica la
        traslación que predecía la pregunta 14.
\end{enumerate}
\end{problem}
