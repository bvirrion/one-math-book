\chapter{Simetría central y paralelogramos}\label{ch:g7:central}

Girar una figura por \omterm{def:g3:division:half}{medio} un giro alrededor de un punto: esto es \emph{\omterm{def:g7:central:def}{céntrico
simetría}}, la segunda transformación del libro tras las reflexiones
de \cref{ch:g6:symmetry}. Su figura estrella es la \omterm{def:g7:central:parallelogram}{paralelogramo} --- el
\omterm{def:g4:geometry:polygon}{cuadrilátero} eso es \omterm{def:g7:central:def}{simétrico} sobre el punto de cruce de su propio
diagonales.

\section{simetría central}

\begin{definition}[Simétrico respecto de un punto]\label{def:g7:central:def}
El \emph{simetrico}\index{simetrico} de un
punto $M $ sobre un
punto $ O $ es el punto $ M'$ tal que $ O $ es el \emph{punto medio} de
$[MM']$. La figura simétrica es en lo que se convierte el original después de un
media vuelta ($180^\circ $) alrededor $ O $.
\end{definition}

\begin{method}[Construyendo la simetría de un punto.]\label{met:g7:central:construct}
Para construir el \omterm{def:g7:central:def}{simétrico} $ M'$ de $ M $ acerca de $ O $:
\begin{enumerate}
  \item poner límites $(MO)$ y extenderlo más allá $ O $;
  \item medida $ MO $ (regla o compás);
  \item lugar $ M'$ en la extensión con $ OM' = OM $.
\end{enumerate}
En papel cuadriculado: cuente los pasos horizontales y verticales desde $ M $ a
$ O $, y repita el \emph{mismo} pasos de $ O $ alcanzar $ M'$.
\end{method}

\begin{example}[en una grilla]\label{ex:g7:central:grid}
Si $ M $ es $3$ cuadrados a la izquierda y $1$ cuadrar desde $ O $, es \omterm{def:g7:central:def}{simétrico}
$ M'$ es $3$ cuadrícula \emph{bien} y $1$ cuadrado \emph{abajo} de $ O $:
\omterm{def:g7:central:def}{céntrico
simetría} invierte ambas direcciones a la vez (un reflejo
invierte sólo uno).
\end{example}

\begin{proposition}[¡Qué media vuelta conserva!]\label{prop:g7:central:preserved}
\omterm{def:g7:central:def}{simetría central} conserva longitudes, ángulos, \omterm{def:g6:measure:perimeter}{perímetros} y \omterm{def:g6:measure:area}{áreas}; el
La imagen de una línea es una \emph{\omterm{def:g6:lines:perp}{paralelo}} línea, la imagen de un \omterm{def:g6:lines:objects}{segmento} es un
\omterm{def:g6:lines:perp}{paralelo} \omterm{def:g6:lines:objects}{segmento} de la misma longitud, la imagen de un círculo es un círculo
del mismo \omterm{def:g3:shapes:shapes}{radio} (centrado en el \omterm{def:g7:central:def}{simétrico} del centro).
\end{proposition}
\admitted

\begin{remark}
Comparar con reflejos (\cref{prop:g6:symmetry:preserved}): ambos
preservar longitudes y ángulos; pero el reflejo voltea las figuras
(una ``b'' se convierte en una ``d''), mientras que la media vuelta los mantiene legibles ---
al revés (una ``b'' se convierte en una ``q''). Y a diferencia de un reflejo, un
media vuelta envía cada línea a una línea \omterm{def:g6:lines:perp}{paralelo} lo.
\end{remark}

\begin{definition}[centro de simetría]\label{def:g7:central:center}
un punto $ O $ es un \emph{centro de simetria}\index{centro de simetria}
de una figura cuando la media vuelta $ O $ envía la figura exactamente
sobre sí mismo. Por ejemplo, las letras S, N, Z tienen un centro de
simetría; un círculo tiene uno (su centro); un triángulo nunca tiene uno.
\end{definition}

\section{paralelogramos}

\begin{definition}[Paralelogramo]\label{def:g7:central:parallelogram}
A \emph{paralelogramo}\index{paralelogramo} es un \omterm{def:g4:geometry:polygon}{cuadrilátero} cuyo
las diagonales se cruzan en su punto medio común. Ese punto de cruce es entonces
un \omterm{def:g7:central:center}{centro de simetría} de la figura: cada \omterm{def:g5:solids:def}{vértice} es la imagen de media vuelta
de lo contrario \omterm{def:g5:solids:def}{vértice}.
\end{definition}

\begin{theorem}[Propiedades de un paralelogramo]\label{thm:g7:central:properties}
en un \omterm{def:g7:central:parallelogram}{paralelogramo} $ ABCD $ con centro $ O $:
\begin{enumerate}
  \item lados opuestos son \omterm{def:g6:lines:perp}{paralelo}: $(AB) \parallel (CD)$ y
        $(a.\,C.) \parallel (d.\,C.)$;
  \item los lados opuestos tienen la misma longitud: $ AB = CD $ y
        $ a.\,C. = d.\,C. $;
  \item los ángulos opuestos son iguales: $\widehat A = \widehat C $ y
        $\widehat B = \widehat D $.
\end{enumerate}
\end{theorem}

\begin{proof}
La media vuelta $ O $ envía $ A $ a $ C $ y $ B $ a $ D $ (definición:
$ O $ es el punto medio de ambas diagonales). Entonces envía el \omterm{def:g6:lines:objects}{segmento}
$[AB]$ hacia \omterm{def:g6:lines:objects}{segmento} $[CD]$; por
\cref{prop:g7:central:preserved} la imagen es \omterm{def:g6:lines:perp}{paralelo} a $[AB]$ y
de la misma longitud: puntos 1 y 2. También envía el ángulo en $ A $
al ángulo en $ C $, y se conservan los ángulos: punto 3.
\end{proof}

\begin{theorem}[Reconocer un paralelogramo]\label{thm:g7:central:recognize}
A \omterm{def:g4:geometry:polygon}{cuadrilátero} (sin lados cruzados) es un \omterm{def:g7:central:parallelogram}{paralelogramo} tan pronto como
\emph{uno} de las siguientes retenciones:
\begin{enumerate}
  \item sus diagonales tienen el mismo punto medio (la definición);
  \item sus lados opuestos son \omterm{def:g6:lines:perp}{paralelo} de dos en dos;
  \item dos lados opuestos son \omterm{def:g6:lines:perp}{paralelo} \emph{y} de la misma longitud.
\end{enumerate}
\end{theorem}
\admitted

\begin{example}\label{ex:g7:central:recognize}
$ ABCD $ tiene $(AB) \parallel (CD)$ y $ AB = CD = 5$ centímetro. Criterio 3
se aplica: $ ABCD $ es un \omterm{def:g7:central:parallelogram}{paralelogramo} --- así que sin más
midiendo también sabemos $ d.\,C. = a.\,C. $, $(d.\,C.) \parallel (a.\,C.)$, y que su
las diagonales se cortan entre sí por la mitad.
\end{example}

\begin{remark}[paralelogramos especiales]
El rectángulo, \omterm{def:g6:shapes:quadrilaterals}{rombo} y cuadrado de \cref{ch:g6:shapes} son exactamente
el \omterm{def:g7:central:parallelogram}{paralelogramos} con una propiedad extra: diagonales iguales
(rectángulo), \omterm{def:g6:lines:perp}{perpendicular} diagonales (\omterm{def:g6:shapes:quadrilaterals}{rombo}), ambos (cuadrado). el
argumento angular prometido en \cref{ex:g6:shapes:recognize} ahora funciona:
en un \omterm{def:g7:central:parallelogram}{paralelogramo}, los ángulos consecutivos son suplementarios
(ángulos alternos/correspondientes con el \omterm{def:g6:lines:perp}{paralelas},
\cref{thm:g7:triangles:equalities}), entonces uno \omterm{def:g3:shapes:rightangle}{ángulo recto} obliga a todos
cuatro.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g7:central:1}
En papel cuadriculado, coloque $ O $, entonces $ M $ cuatro cuadrados a la derecha de $ O $ y uno
cuadrar. Construir el \omterm{def:g7:central:def}{simétrico} $ M'$ de $ M $ acerca de $ O $, y el
\omterm{def:g7:central:def}{simétrico} de $ M $ a través del \emph{línea de cuadrícula vertical} a través de $ O $.
Compara las dos imágenes.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:central:2}
dibuja un triángulo $ ABC $ y un punto $ O $ fuera de él. Construir su imagen
$ A'B'C'$ por la media vuelta $ O $. ¿Qué puedes decir sobre el
longitudes $ A'B'$ y $ AB $? sobre las lineas $(A'B')$ y $(AB)$?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:central:3}
¿Qué dígitos, escritos como en la pantalla de una calculadora ($0$ a $9$), tener un
\omterm{def:g7:central:center}{centro de simetría}? ¿Qué letras mayúsculas entre H, A, N, S, T, Z, O?
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:central:4}
$ KLMN $ es un \omterm{def:g7:central:parallelogram}{paralelogramo} de centro $ O $ con $ KL = 7$ centímetro,
$ LM = 4$ centímetros y $\widehat K = 115^\circ $. Dé, con razones: $ MN $,
$ NK $, $\widehat M $, y el punto medio de $[LN]$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:central:5}
Construir un \omterm{def:g7:central:parallelogram}{paralelogramo} $ ABCD $ de sus diagonales: $ AC = 8$ centímetros y
$ BD = 5$ cm, cruzándose en sus puntos medios con un ángulo de $60^\circ $
entre ellos. (¡Primero dibuja las diagonales!)
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g7:central:6}
en un \omterm{def:g7:central:parallelogram}{paralelogramo}, un ángulo mide $115^\circ $. Usando el comentario
en ángulos consecutivos suplementarios, calcula los otros tres ángulos.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g7:central:7}
Trama $ A(1, 1)$, $ B(4, 2)$ y el punto $ O(2.5, 2.5)$. Calcular el
coordenadas de las imágenes $ A'$ y $ B'$ por la media vuelta $ O $
(use el conteo de cuadrículas de \cref{ex:g7:central:grid}).
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g7:central:8}
$ ABCD $ es un \omterm{def:g4:geometry:polygon}{cuadrilátero} en el cual $ AB = CD $. ¿Es eso suficiente para hacer
es un \omterm{def:g7:central:parallelogram}{paralelogramo}? Si no, esboce un contraejemplo (un \omterm{def:g6:shapes:triangles}{isósceles}
el trapezoide ayuda), y exponga lo que se debe agregar a la hipótesis.
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g7:central:9}
Sea $ ABC $ un triángulo y $ O $ el punto medio de $[a.\,C.]$. Construir el
\omterm{def:g7:central:def}{simétrico} $ A'$ de $ A $ acerca de $ O $. mostrar eso $ ABA'C $ es un
\omterm{def:g7:central:parallelogram}{paralelogramo}. (¿Qué criterio de \cref{thm:g7:central:recognize} es
gratis aquí?)
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g7:central:10}
Verdadero o falso, con una razón: ``a \omterm{def:g4:geometry:polygon}{cuadrilátero} cuyos ángulos opuestos
son iguales dos a dos es un \omterm{def:g7:central:parallelogram}{paralelogramo}''; ``un \omterm{def:g7:central:parallelogram}{paralelogramo} con
\omterm{def:g6:lines:perp}{perpendicular} diagonales es un \omterm{def:g6:shapes:quadrilaterals}{rombo}''; ``un \omterm{def:g7:central:parallelogram}{paralelogramo} con uno
\omterm{def:g3:shapes:rightangle}{ángulo recto} es un rectángulo''.
\end{exercise}

\begin{exercise}[$\star\star\star $]\label{exo:g7:central:11}
Sea $ ABCD $ cualquiera \omterm{def:g4:geometry:polygon}{cuadrilátero} y $ I $, $ J $, $ K $, $ L $ los puntos medios
de sus lados $[AB]$, $[a.\,C.]$, $[CD]$, $[DA]$. Dibuja varios ejemplos.
¿Qué hace? $ IJKL $ siempre parece serlo? (La prueba, con el punto medio
teorema, entra \cref{ch:g8:midpoints} --- y nuevamente con vectores,
en el volumen de High School.)
\end{exercise}

\section{Problema: el pastel, la mediana y el segundo imposible
centro}

\begin{problem}[{Problema de fin de semana --- medias vueltas en acción: cortes de pastel justos, un límite en las medianas y por qué ninguna figura puede tener dos centros de simetría}]
\label{pb:g7:central:1}
La media vuelta parece la más simple de las transformaciones, pero
corta pasteles en mitades demostrablemente justas, mide las medianas de
un triángulo, y esconde una pequeña perla de la matemática pura: un
figura acotada puede tener uno \omterm{def:g7:central:center}{centro de simetría}, pero \emph{nunca
dos}. La prueba utiliza un descubrimiento que harás en la Parte I: dos
medias vueltas, realizadas una tras otra, suman un deslizamiento.

\medskip
\noindent\textbf{Parte I --- Half-turn gymnastics.}
\begin{enumerate}
  \item En papel cuadriculado, toma el origen. $ O(0, 0)$ como centro.
        Usando el conteo de cuadrículas de \cref{ex:g7:central:grid},
        dar las imagenes de $ M(3, 1)$, $ N(-2, 4)$ y
        $ P(0, -5)$ por la media vuelta $ O $. cual es el
        regla general para $(x, y)$?
  \item Ahora el centro es $ C(2, 1)$. Calcular la imagen de
        $ M(5, 3)$, y consulta la receta general: cada coordenada
        de la imagen es \emph{el doble del centro menos el
        puntos}.
  \item Los naipes están diseñados para leerse igual al revés
        --- un \omterm{def:g7:central:def}{céntrico
simetría}, por lo que ningún jugador ve la carta
        ``incorrectamente''. Explique por qué, en una tarjeta con un \emph{\omterm{def:g3:numbers:evenodd}{extraño}}
        número de pips, un pip debe estar exactamente en el punto de la tarjeta
        centro.
  \item Dos medias vueltas seguidas: coger los centros $ O_1(2, 0)$
        y $ O_2(5, 0)$. enviar el punto $ A(1, 1)$ a través del
        media vuelta $ O_1$, luego envíe la imagen a través del
        media vuelta $ O_2$. Compara el inicio y el final. repetir
        con $ B(0, 3)$. ¿Qué única y simple transformación hizo
        ¿A cuánto ascienden las dos medias vueltas?
  \item Mida la diapositiva de la pregunta 4 contra la distancia
        $ O_1 O_2$y declarar el descubrimiento. (Compárese con el
        dos \omterm{def:g6:lines:perp}{paralelo} espejos de \cref{pb:g6:symmetry:1}: igual
        fenómeno, nuevos actores.)
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Cakes and medians.}
\begin{enumerate}[resume]
  \item Sea $ O $ el centro de una \omterm{def:g7:central:parallelogram}{paralelogramo}. Explica por qué
        cualquier línea a través $ O $ encuentra el límite en dos puntos
        que son \omterm{def:g7:central:def}{simétrico} acerca de $ O $, y por qué la línea corta el
        \omterm{def:g7:central:parallelogram}{paralelogramo} en dos piezas que sean exactamente media vuelta
        copias entre sí --- por lo tanto de igual \omterm{def:g6:measure:area}{áreas}.
  \item El teorema del panadero: se comparte un pastel rectangular
        perfectamente y justamente por \emph{cualquier} corte recto a través de su
        centro. Mejor: un pastel rectangular tiene una forma rectangular.
        agujero (cualquier tamaño, cualquier posición, \omterm{def:g3:numbers:evenodd}{incluso} inclinado). Describe el
        un corte recto que da dos partes con la misma cantidad
        pastel --- y justificarlo con la pregunta 6.
  \item Tres \omterm{def:g5:solids:def}{vértices} de un \omterm{def:g7:central:parallelogram}{paralelogramo} $ ABCD $ son $ A(0, 0)$,
        $ B(6, 1)$ y $ C(8, 5)$. Calcula las coordenadas de $ D $
        y del centro $ O $.
  \item \Cref{exo:g7:central:9} construido, a partir de un triángulo $ ABC $
        y el punto medio $ O $ de $[a.\,C.]$, el \omterm{def:g7:central:parallelogram}{paralelogramo}
        $ ABA'C $ dónde $ A'$ es el \omterm{def:g7:central:def}{simétrico} de $ A $ acerca de $ O $.
        Úselo, junto con la desigualdad del triángulo.
        (\cref{thm:g7:triangles:inequality}) en el triángulo
        $ ABA'$, para demostrar la \emph{límite medio}: la mediana
        $[AO]$ satisface
        \[
        AO < \frac{AB + AC}{2} .
        \]
        (Tenga en cuenta que $ AA' = 2\,AO $ y $ BA' = AC $.)
  \item en un triángulo con $ AB = 5$ centímetros y $ AC = 7$ centímetros, entre
        ¿Qué dos valores deben ser la longitud de la mediana desde $ A $
        ¿mentir? (Utilice ambas direcciones de la desigualdad del triángulo en
        $ ABA'$.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- How many centers can a figure
have?}
\begin{enumerate}[resume]
  \item Para cada figura, diga si tiene una \omterm{def:g7:central:center}{centro de simetría}, y donde: un \omterm{def:g6:lines:objects}{segmento}; una línea completa; un
        \omterm{def:g6:shapes:triangles}{triángulo equilatero}; un cuadrado; un círculo; las letras s
        y Z.
  \item Demuestre su reclamo por el \omterm{def:g6:shapes:triangles}{triángulo equilatero}: un
        media vuelta enviando el triángulo hacia sí mismo se emparejaría
        los tres \omterm{def:g5:solids:def}{vértices} --- pero tres son \omterm{def:g3:numbers:evenodd}{extraño}, entonces algunos \omterm{def:g5:solids:def}{vértice}
        sería enviado a sí mismo. ¿Qué obligaría eso a
        \omterm{def:g5:solids:def}{vértice} ser, ¿y por qué es imposible?
  \item Generalice la pregunta 12: explique por qué \emph{No} \omterm{def:g4:geometry:polygon}{polígono}
        con un \omterm{def:g3:numbers:evenodd}{impar
        número} de \omterm{def:g5:solids:def}{vértices} tiene un \omterm{def:g7:central:center}{centro de simetría}.
  \item La perla: supongamos que una figura hubiera \emph{dos} distinto
        centros de simetría $ O_1$ y $ O_2$. realizar los dos
        medias vueltas una tras otra y utilizar el descubrimiento de
        pregunta 5: qué movimiento debe enviar la figura exactamente
        ¿Sobre sí mismo? Explica por qué una figura que cabe en una hoja.
        El papel no puede sobrevivir a eso y concluir.
  \item La conclusión de la pregunta 14 deja un vacío legal:
        \emph{ilimitado} cifras. Verificar en un friso infinito
        --- digamos, una franja interminable de huellas, izquierda, derecha,
        izquierda, derecha\,\dots\ --- que tenga (al menos) dos
        diferentes centros de simetría e identificar los
        traducción que predijo la pregunta 14.
\end{enumerate}
\end{problem}
