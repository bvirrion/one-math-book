\chapter{الأعداد السالبة}\label{ch:g7:negatives}

درجات الحرارة تحت \omterm{def:g1:counting:zero}{الصفر}، والطوابق تحت الأرض، والمال المستدان: مقادير
كثيرة تنزل تحت $0$. والأعداد السالبة تمدّ \omterm{def:g6:lines:objects}{مستقيم} الأعداد إلى
يسار \omterm{def:g1:counting:zero}{الصفر}. ويعلّم هذا الفصل كيف تُقارَن وكيف تُجمع
وتُطرح --- أما ضربها فينتظر
\cref{ch:g8:negprod}.

\section{الأعداد النسبية}

\begin{definition}[الأعداد النسبية]\label{def:g7:negatives:def}
\emph{العدد النسبي}\index{عدد نسبي} مكوّن من إشارة
($+$ أو $-$) و\emph{مسافة إلى \omterm{def:g1:counting:zero}{الصفر}}: فالعددان $+3$ و $-3$ كلاهما على
مسافة $3$ من $0$، في جهتين متقابلتين. والأعداد الموجبة تُكتب عادةً
دون إشارتها: $+3 = 3$. والعددان $+3$ و $-3$
\emph{متعاكسان}\index{معاكس}.
\end{definition}

\end{omfigure}

\begin{method}[مقارنة الأعداد النسبية]\label{met:g7:negatives:compare}
على \omterm{def:g6:lines:objects}{مستقيم} الأعداد، الأصغر هو الأبعد نحو اليسار. وعمليًّا:
\begin{enumerate}
  \item العدد السالب أصغر دائمًا من العدد الموجب:
        $-7 < 2$؛
  \item وبين موجبين، الترتيب المعتاد؛
  \item وبين سالبين، الذي مسافته إلى \omterm{def:g1:counting:zero}{الصفر} \emph{أكبر} هو
        \emph{الأصغر}: $-7 < -2$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:negatives:compare}
رتّب من الأصغر إلى الأكبر: $3$؛ و $-5$؛ و $0$؛ و $-1.5$؛ و $2.8$.
السوالب أولًا، وأشدّها سلبًا في المقدمة:
\[
-5 < -1.5 < 0 < 2.8 < 3 .
\]
\end{example}

\section{جمع الأعداد النسبية}

تصوّر \omterm{def:g7:negatives:def}{العدد النسبي} حركةً على \omterm{def:g6:lines:objects}{المستقيم}: فالعدد $+5$ يعني
«اِمشِ $5$ خطوات يمينًا»، و $-3$ يعني «اِمشِ $3$ خطوات يسارًا». وجمع
الأعداد يعني ربط المشيات.

\begin{method}[جمع عددين نسبيين]\label{met:g7:negatives:add}
\begin{enumerate}
  \item \emph{الإشارة نفسها}: اِجمع المسافتين، واِحتفظ بالإشارة المشتركة:
        $(-3) + (-4) = -7$.
  \item \emph{إشارتان متعاكستان}: اِطرح المسافة الأصغر من
        الأكبر، وخذ إشارة الأكبر:
        $(-7) + (+2) = -5$ و $(+7) + (-2) = +5$.
  \item \omterm{def:g7:negatives:def}{والمتعاكسان} يتلاشيان: $(+4) + (-4) = 0$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:negatives:add}
خطوة خطوة، بالتجميع كما يناسب:
\begin{align*}
(-8) + (+3) &= -5 && \text{(إشارتان متعاكستان: } 8 - 3 = 5,
\text{ وإشارة } 8)\\
(-2.5) + (-1.5) &= -4 && \text{(الإشارة نفسها: اِجمع المسافتين)}\\
(+6) + (-9) + (+3) &= (-3) + (+3) = 0 .
\end{align*}
\end{example}

\section{طرح الأعداد النسبية}

\begin{theorem}[قاعدة الطرح]\label{thm:g7:negatives:sub}
طرح \omterm{def:g7:negatives:def}{عدد نسبي} هو نفسه \emph{جمع
معاكسه}:
\[
a - b = a + (-b) .
\]
\end{theorem}

\begin{proof}[لماذا تعمل]
$a - b$ هو العدد الذي إذا جُمع إلى $b$ أعطى $a$. وتحقّق من أن
$a + (-b)$ يقوم بالمهمة:
$\bigl(a + (-b)\bigr) + b = a + \bigl((-b) + b\bigr) = a + 0 = a$.
\end{proof}

\begin{example}\label{ex:g7:negatives:sub}
كل طرح يصير جمعًا، ثم تنطبق
\cref{met:g7:negatives:add}:
\begin{align*}
5 - 9 &= 5 + (-9) = -4, \\
(-3) - (+4) &= (-3) + (-4) = -7, \\
(-2) - (-6) &= (-2) + (+6) = +4 .
\end{align*}
والسطر الأخير هو المشهور: \emph{فطرح عدد سالب يعني
الجمع}. فإسقاط دَين يجعلك أغنى!
\end{example}

\begin{example}[فروق درجات الحرارة]\label{ex:g7:negatives:temperature}
في يوم واحد اِنتقلت درجة الحرارة من $-6$ درجة إلى $+9$ درجة.
والارتفاع هو \omterm{ex:g1:subtraction:difference}{الفرق}
\[
9 - (-6) = 9 + 6 = 15 \text{ درجة}.
\]
وعلى \omterm{def:g6:lines:objects}{مستقيم} الأعداد: من $-6$ صعودًا إلى $0$ هو $6$ درجات، ومن $0$ إلى
$9$ هو $9$ درجات أخرى.
\end{example}

\section{الإحداثيات في المستوي كله}

\begin{definition}[إحداثيات بإشارات]\label{def:g7:negatives:coords}
بمحورين مدرّجين \omterm{def:g6:lines:perp}{متعامدين} يتقاطعان في المبدأ $O$،
تأخذ كل نقطة من المستوي إحداثيين نسبيين $(x, y)$:
فالإحداثي $x$ السالب يعني يسار المحور العمودي، والإحداثي $y$ السالب يعني تحت
المحور الأفقي.
\end{definition}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g7:negatives:1}
أعطِ \omterm{def:g7:negatives:def}{معاكس} كل من: $7$؛ \ و $-3.5$؛ \ و $0$؛ \ و $+12$. ثم أكمل:

\end{exercise}

\end{exercise}

\end{exercise}

(-5) + (-9), \qquad
(-12) + (+7), \qquad

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:9}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:10}
غوّاصة على اِرتفاع $-12$ م (أي اِثني عشر مترًا تحت السطح). فترتفع

\section{مسألة: سنة الصفر التي لم توجد قطّ}

\begin{problem}[{مسألة نهاية الأسبوع --- الأعداد النسبية على
خطّ زمن التاريخ: المدد عبر حدّ الحقبتين، وحيلة
الفلكيين، ومبدأ التغيّر الصافي}]
\label{pb:g7:negatives:1}
\omterm{def:g6:lines:objects}{لمستقيم} الأعداد في هذا الفصل توأم مشهور في الواقع: وهو
خطّ زمن التاريخ، حيث تمتدّ سنوات ما قبل الميلاد يسارًا
وسنوات ما بعده يمينًا. لكن خطّ المؤرّخين يخفي فخًّا
أفسد حسابات كثيرة: \emph{فليست هناك سنة \omterm{def:g1:counting:zero}{صفر}}
--- إذ تلي سنةَ 1 قبل الميلاد مباشرةً سنةُ 1 بعد الميلاد. وتحسب هذه المسألة
بدرجات الحرارة والمصاعد والحسابات المصرفية، ثم تصلح
\omterm{def:g1:counting:zero}{صفر} التاريخ المكسور كما يفعل الفلكيون: بالأعداد
النسبية.

\medskip
\noindent\textbf{الجزء الأول --- تمارين إحماء، وفخّ أول.}
\begin{enumerate}
  \item قمّة مون بلان على اِرتفاع $+4\,808$ م؛ وشاطئ
        البحر الميت على $-430$ م. اِحسب \omterm{ex:g1:subtraction:difference}{الفرق}
        في الارتفاع بينهما.
  \item حساب مصرفي فيه $-45$ يورو (أي كشف). فيودع
        صاحبه $120$ يورو، ويدفع فاتورة $60$ يورو،
        ثم يودع $15$ يورو. اُكتب الرصيد سلسلة
        جموع واحدة واحسبه. وأيّ إيداع واحد
        كان يعيد المبلغ الأصلي $-45$ إلى
        $0$ بالضبط؟
  \item وُلد يوليوس قيصر سنة $100$ قبل الميلاد ومات سنة $44$ قبل الميلاد.
        وبعدّ سنوات ما قبل الميلاد أعدادًا سالبة (فسنة $100$ قبل الميلاد هي
        $-100$، وسنة $44$ قبل الميلاد هي $-44$)، اِحسب $(-44) - (-100)$: فكم
        سنة عاش قيصر؟
  \item وُلد الإمبراطور أغسطس سنة $63$ قبل الميلاد ومات سنة
        $14$ بعد الميلاد. والوصفة نفسها تعطي $14 - (-63) = 77$ سنة ---
        لكن المؤرّخين يصرّون على أنه مات عن $76$. واِعتراضهم:
        أن العدّ $77$ يمرّ بسنة لم توجد قطّ.
        فأيّ سنة هي؟
  \item يصلح الفلكيون خطّ الزمن بإعادة التسمية: فسنوات ما بعد الميلاد
        تحتفظ بعددها ($+14$ تبقى $+14$)، لكن سنة $1$ قبل الميلاد
        تصير $0$، وسنة $2$ قبل الميلاد تصير $-1$، وعمومًا السنة
        $n$ قبل الميلاد تصير $-(n - 1)$. حوّل سنة $63$ قبل الميلاد، وأعِد
        حساب السؤال 4 بأعداد الفلكيين.
        فهل يرضي الجوابُ المؤرّخين الآن؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الحساب عبر الحقبتين.}
من الآن فصاعدًا، اِستعمل إعادة تسمية الفلكيين في كل
حساب، مع التحويل إلى ما قبل الميلاد وما بعده في النهاية (فالسنة التي تحمل الاسم
$0$ أو السالب $-m$ هي سنة المؤرّخين $m + 1$ قبل الميلاد).
\begin{enumerate}[resume]
  \item حوّل إلى أعداد الفلكيين: $1$ قبل الميلاد؛ و $10$ قبل الميلاد؛ و
        $476$ بعد الميلاد؛ و $753$ قبل الميلاد، وهي السنة الأسطورية لتأسيس روما.
  \item كم سنة مرّت من تأسيس روما
        (سنة $753$ قبل الميلاد) إلى سقوط الإمبراطورية الغربية (سنة $476$ بعد الميلاد)؟
  \item مذنّب يعود كل $76$ سنة وظهر سنة
        $12$ قبل الميلاد. أعطِ سنوات المؤرّخين لظهوراته الثلاثة
        التالية، وللظهور الذي سبق سنة $12$ قبل الميلاد مباشرةً.
  \item كم سنة مرّت من سنة $5$ قبل الميلاد إلى سنة $5$ بعد الميلاد؟ (والجواب
        المغري هو $10$؛ لكن خطّ الزمن يقول غير ذلك.)
  \item طفل وُلد سنة $3$ قبل الميلاد بلغ عشر سنوات في أيّ
        سنة (قبل الميلاد أم بعده)؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- مبدأ التغيّر الصافي.}
\begin{enumerate}[resume]
  \item \emph{المسافة} بين عددين على \omterm{def:g6:lines:objects}{المستقيم} هي
        الأكبر ناقص الأصغر --- وهي موجبة دائمًا.
        اِحسب المسافة بين $-7$ و $-2$، ثم بين
        $-3.5$ و $4.5$.
  \item مصعد ينطلق من الطابق $0$ ويقوم بالحركات $+3$
        و $-5$ و $+2$ و $-1$ و $+4$. اِحسب الطابق النهائي
        بتجميع ذكيّ (\cref{met:g7:negatives:add}،
        والمتعاكسات أولًا حيث أمكن). وهل يغيّر إنجاز
        الحركات بترتيب مختلف الوجهةَ؟ ولماذا؟
  \item يومًا بعد يوم، تتغيّر درجة الحرارة بمقدار $+2$ و $-4$ و $+1$
        و $0$ و $-3$ و $+5$ و $-2$ درجة على مدى أسبوع. اِحسب
        \emph{التغيّر الصافي}. وإذا بدأ الأسبوع عند $-2$ درجة،
        فأين اِنتهى؟ واُذكر المبدأ: القيمة النهائية $=$
        القيمة الابتدائية $+$ (\omterm{def:g1:addition:def}{مجموع} كل التغيّرات).
  \item مشية متجوّلة تعيدها بالضبط إلى اِرتفاع
        اِنطلاقها. فما الذي يجب أن يكون عليه \omterm{def:g1:addition:def}{مجموع} كل تغيّرات اِرتفاعها،
        ولماذا؟ ومن تغيّراتها الستة المسجّلة، خمسة هي
        $+4$ و $-7$ و $+2$ و $+3$ و $-5$ (بمئات الأمتار):
        فجِد السادس.
  \item الخاتمة، عودًا إلى روما. يحسب مؤرّخ: «اِمتدّت
        الجمهورية من سنة $509$ قبل الميلاد إلى سنة $27$ قبل الميلاد، ثم الإمبراطورية
        إلى سنة $476$ بعد الميلاد؛ فمن سنة $509$ قبل الميلاد إلى سنة $476$ بعد الميلاد، أي
