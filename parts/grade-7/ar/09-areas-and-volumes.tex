\chapter{المساحات والأحجام}\label{ch:g7:areas}

قاست السنة السادسة المستطيلات والمثلثات القائمة الزاوية والصناديق
(\cref{ch:g6:measure}). وبمقصّ --- تقطع \omterm{def:g6:lines:objects}{قطعة} هنا،
وتلصقها هناك --- تعطي الأفكار نفسها الآن مساحات متوازيات الأضلاع،
و\emph{كل} المثلثات والقرص، وأحجام الموشورات
والأسطوانات.

\section{مساحة متوازي الأضلاع}

\begin{theorem}[متوازي الأضلاع]\label{thm:g7:areas:parallelogram}
\omterm{def:g7:central:parallelogram}{متوازي أضلاع} طول أحد أضلاعه $b$ (وهو \emph{قاعدته}) والمسافة
$h$ بين ذلك الضلع والمقابل له (وهي
\emph{اِرتفاعه}) مساحته
\[
A = b \times h .
\]
\end{theorem}

\begin{proof}[برهان بالمقصّ]
اِقطع \omterm{def:g6:shapes:triangles}{المثلث القائم الزاوية} البارز في جهة واِلصقه
في الأخرى: فيصير \omterm{def:g7:central:parallelogram}{متوازي الأضلاع} مستطيلًا $b \times h$ \omterm{def:g6:measure:area}{بالمساحة}
نفسها.
\end{proof}

تكون له أضلاع طويلة \omterm{def:g6:measure:area}{ومساحة} صغيرة.
\end{remark}

\section{مساحة المثلث}

\begin{theorem}[المثلث]\label{thm:g7:areas:triangle}
مثلث قاعدته $b$ واِرتفاعه الموافق $h$ (وهو
المسافة العمودية من الرأس المقابل إلى \omterm{def:g6:lines:objects}{مستقيم} القاعدة)
مساحته
\[
A = \frac{b \times h}{2} .
\]
\end{theorem}

\begin{proof}
نسختان من المثلث، إحداهما مُدارة نصف دورة حول
منتصف أحد الأضلاع، تتلاءمان في \omterm{def:g7:central:parallelogram}{متوازي أضلاع} قاعدته $b$
واِرتفاعه $h$ (وهذا هو \omterm{def:g7:central:def}{التماثل المركزي} في العمل،
\cref{prop:g7:central:preserved}). والمثلث نصفه:
أي $\frac{bh}{2}$.
\end{proof}

\omterm{def:g7:central:parallelogram}{متوازي أضلاع}: \omterm{def:g6:measure:area}{فمساحة} المثلث $\frac{b\times h}{2}$. وأيّ ضلع من
الثلاثة يصلح قاعدة --- مع اِرتفاعه \emph{هو}.}
\end{omfigure}

\begin{example}\label{ex:g7:areas:triangle}
مثلث قاعدته $9$ سم واِرتفاعه الموافق يقيس
$4$ سم:
\[
A = \frac{9 \times 4}{2} = \frac{36}{2} = 18 \text{ سم}^2 .
\]
وفي المثلث \emph{\omterm{def:g6:shapes:triangles}{القائم الزاوية}}، يكون ضلعا القائمة قاعدة واِرتفاعها:
فنستعيد صيغة \cref{prop:g6:measure:formulas}.
\end{example}

\section{مساحة القرص}

\begin{theorem}[القرص]\label{thm:g7:areas:disk}
قرص نصف قطره $r$ مساحته
\[
A = \pi r^2 .
\]
\end{theorem}

\begin{proof}[فكرة البرهان]
اِقطع القرص إلى شرائح رفيعة متساوية كثيرة وصُفَّها رأسًا لذيل:
فتكاد تكوّن \omterm{def:g7:central:parallelogram}{متوازي أضلاع} اِرتفاعه $r$ وقاعدته نصف
\omterm{def:g6:measure:perimeter}{المحيط}، أي $\pi r$ (\cref{prop:g6:measure:circle}). ومساحته
قريبة من $\pi r \times r = \pi r^2$، ويصير \omterm{def:g6:decimals:rounding}{التقريب}
تامًّا كلما رقّت الشرائح. والبرهان الكامل يحتاج حساب
التكامل، من مجلّد المرحلة الثانوية.
\end{proof}

{\small تشريح القرص وفرد الشرائح: فيكاد يكون
\omterm{def:g7:central:parallelogram}{متوازي أضلاع} قاعدته $\pi r$ (نصف الحدّ) واِرتفاعه $r$.}

\begin{example}\label{ex:g7:areas:disk}
سطح طاولة مدوّرة نصف قطره $60$ سم. ومساحته
$\pi \times 60^2 = 3600\pi \approx 11\,310$ سم$^2$، أي أكثر قليلًا من
$1.1$ م$^2$. واِنتبه \omterm{ex:g1:subtraction:difference}{للفرق}: \omterm{def:g6:measure:perimeter}{فالمحيط} $2\pi r$ يستعمل
$r$، \omterm{def:g6:measure:area}{والمساحة} $\pi r^2$ تستعمل $r^2$.
\end{example}

\section{الموشورات والأسطوانات}

\begin{definition}[الموشور والأسطوانة]\label{def:g7:areas:prism}
\emph{الموشور}\index{موشور} \omterm{def:g5:solids:def}{مجسّم} له وجهان مضلّعان متوازيان
متطابقان (وهما \emph{القاعدتان}) تصلهما مستطيلات؛
و\emph{الأسطوانة}\index{أسطوانة} مثله بقاعدتين قرصيتين.
و\emph{الارتفاع} هو المسافة بين القاعدتين.
\end{definition}

\begin{theorem}[الأحجام]\label{thm:g7:areas:volume}
من أجل \omterm{def:g7:areas:prism}{موشور} أو \omterm{def:g7:areas:prism}{أسطوانة} \omterm{def:g6:measure:area}{مساحة} قاعدته $B$ واِرتفاعه $h$:
\[
V = B \times h .
\]
\end{theorem}
\admitted

\begin{example}\label{ex:g7:areas:volume}
خيمة \omterm{def:g7:areas:prism}{موشور} مضطجع على جنبه: قاعدتاه مثلثان قاعدة كل منهما
$2$ م واِرتفاعه $1.5$ م، وطول الخيمة $3$ م.
\begin{enumerate}
  \item \omterm{def:g6:measure:area}{مساحة} القاعدة: $B = \frac{2 \times 1.5}{2} = 1.5$ م$^2$.
  \item \omterm{def:g6:measure:volume}{والحجم}: $V = B \times h = 1.5 \times 3 = 4.5$ م$^3$.
\end{enumerate}
وعلبة معدنية نصف قطرها $4$ سم واِرتفاعها $11$ سم:
$V = \pi \times 4^2 \times 11 = 176\pi \approx 553$ سم$^3$، أي نحو
نصف لتر.
\end{example}

\begin{exercise}[$\star$]\label{exo:g7:areas:3}

\begin{exercise}[$\star$]\label{exo:g7:areas:5}
اِحسب \omterm{def:g6:measure:perimeter}{محيط} قرص نصف قطره $3$ سم \emph{و} مساحته.

\begin{exercise}[$\star$]\label{exo:g7:areas:6}

\begin{exercise}[$\star$]\label{exo:g7:areas:7}
مثلث مساحته $28$ سم$^2$ وقاعدته $8$ سم. جِد

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:areas:8}
اُرسم مثلثًا كيفما كان وقِس بعناية أزواج القاعدة والارتفاع الثلاثة.
واِحسب $\frac{b \times h}{2}$ من أجل كل زوج: فالنتائج الثلاث يجب أن

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:areas:9}
حديقة على شكل شبه منحرف يمكن شقّها، \omterm{def:g4:geometry:circle}{بقطر} واحد، إلى
مثلثين لهما الارتفاع نفسه $h = 20$ م، وقاعدتاهما $30$ م و

لشبه المنحرف؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:areas:10}
مزهرية أسطوانية نصف قطرها $6$ سم فيها ماء إلى اِرتفاع
$15$ سم. ويُسكب الماء كله في صندوق فارغ
قاعدته $18$ سم $\times$ $12$ سم. فأيّ اِرتفاع للماء يُبلَغ في
الصندوق؟ (\omterm{def:g6:measure:volume}{الحجم} نفسه، وقاعدة مختلفة.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:areas:11}
بيتزا قطرها $30$ سم ثمنها $9$ يورو؛ وأخرى قطرها $40$ سم

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:areas:12}
ضلعا القائمة في \omterm{def:g6:shapes:triangles}{مثلث قائم الزاوية} يقيسان $6$ سم و $8$ سم، ووتره

\end{exercise}

\section{مسألة: المربع المتلاشي}

\begin{problem}[{مسألة نهاية الأسبوع --- «برهان» مشهور على أن
$64 = 65$، تفضحه المساحات والميول، مع بيتزا وحلقات
للحلوى}]
\label{pb:g7:areas:1}
يقطع ساحر مربعًا $8 \times 8$ إلى أربع قطع، ويزلقها
هنا وهناك، ويعيد تركيبها مستطيلًا $5 \times 13$.
فيعدّ الجمهور: $8 \times 8 = 64$، لكن
$5 \times 13 = 65$ --- فقد ظهرت وحدة مربعة من
العدم! وصيغ \omterm{def:g6:measure:area}{المساحة} في هذا الفصل هي بالضبط الأدوات التي
تمسك الحيلة. ثم تضع المسألة اِستدلال \omterm{def:g6:measure:area}{المساحة} الأمين في
العمل: على أشباه المنحرف، وعلى الأكداس المائلة، وعلى اِقتصاد البيتزا ---
وأخيرًا تعود إلى الساحر في عرض ثانٍ أغرب
بعدُ.

\medskip
\noindent\textbf{الجزء الأول --- الحيلة، منجزةً ومفضوحة.}
\omterm{def:g6:lines:objects}{والقطع} الأربع من المربع $8 \times 8$ هي: مثلثان قائما
الزاوية ضلعا قائمتهما $8$ و $3$، وشبها منحرف قائمان
ضلعاهما المتوازيان $5$ و $3$ واِرتفاعهما $5$.
\begin{enumerate}
  \item اِحسب \omterm{def:g6:measure:area}{مساحة} المربع الأصلي \omterm{def:g6:measure:area}{ومساحة}
        المستطيل المدّعى. واُذكر الفضيحة في سطر واحد.
  \item اِحسب \omterm{def:g6:measure:area}{مساحة} كل \omterm{def:g6:lines:objects}{قطعة} من \omterm{def:g6:lines:objects}{القطع} الأربع
        (\cref{thm:g7:areas:triangle}؛ ومن أجل شبهَي المنحرف، اِقطع
        أو اِستعمل \cref{exo:g7:areas:9}).
  \item اِجمع المساحات الأربع. فأيّ الشكلين --- المربع
        أم المستطيل --- تستطيع \omterm{def:g6:lines:objects}{القطع} ملأه حقًّا؟
  \item في ترتيب المستطيل، يُفترض أن وتر مثلث
        (يصعد $3$ على مدى $8$) يواصل حرفًا
        مائلًا لشبه منحرف (يصعد $2$ على مدى $5$) في
        «\omterm{def:g4:geometry:circle}{قطر}» \omterm{def:g6:lines:objects}{مستقيم} واحد. اِختبر المحاذاة: أيكون $3$ على مدى
        $8$ و $2$ على مدى $5$ بالميل نفسه
        (\cref{thm:g7:prop:cross})؟
  \item اِشرح أين يختبئ المربع الخامس والستون: فما شكل
        الفجوة على طول \omterm{def:g4:geometry:circle}{القطر} الزائف، وما مساحتها المضبوطة،
        ولماذا تفوت العين؟ (فكم يبلغ، في المتوسط، عرض
        شظية بتلك \omterm{def:g6:measure:area}{المساحة} ممدودة على \omterm{def:g4:geometry:circle}{قطر} المستطيل
        الذي يبلغ طوله نحو $13$ وحدة؟)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- اِستدلال \omterm{def:g6:measure:area}{المساحة} الأمين.}
\begin{enumerate}[resume]
  \item اُرسم \omterm{def:g6:lines:objects}{مستقيمين} \omterm{def:g6:lines:perp}{متوازيين} تفصل بينهما $4$ سم، وقاعدة $6$ سم على
        أحدهما، وثلاثة مثلثات مختلفة على تلك القاعدة
        رؤوسها في نقط شتّى من \omterm{def:g6:lines:objects}{المستقيم} الآخر. اِحسب
        المساحات الثلاث. ولماذا تتساوى كلها، مهما اِنزلق
        الرأس على مستقيمه؟
  \item بَرهِن على صيغة شبه المنحرف عمومًا: فشبه منحرف
        ضلعاه المتوازيان $B$ و $b$ واِرتفاعه $h$، مقطوعًا على طول
        \omterm{def:g4:geometry:circle}{قطر}، يعطي مثلثين لهما الارتفاع نفسه $h$.
        فاِستنتج:
        \[
        A = \frac{(B + b) \times h}{2} .
        \]
  \item طبّقها على شبه منحرف فيه $B = 9$ سم و $b = 5$ سم
        و $h = 4$ سم.
  \item ويستعمل النجّارون الصيغة نفسها متنكّرة: \omterm{def:g6:measure:area}{المساحة} $=$
        (متوسط الضلعين \omterm{def:g6:lines:perp}{المتوازيين}) $\times$ الارتفاع.
        اِشرح لماذا تكون القاعدة نفسها، وأعِد حساب
        السؤال 8 بتلك الطريقة.
  \item أمِل كدسًا مرتّبًا من ورق اللعب فيصير كدسًا مائلًا.
        اِشرح لماذا لم يتغيّر \omterm{def:g6:measure:volume}{الحجم}، وماذا يقول ذلك
        عن الارتفاع في صيغة \omterm{def:g7:areas:prism}{الموشور}
        $V = B \times h$ (\cref{thm:g7:areas:volume}): فأيّ
        اِرتفاع يجب أن يُؤخذ من أجل كدس مائل --- الطول المائل
        أم العمودي؟ (قارن
        \cref{exo:g7:areas:2}، ببُعد إضافي.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الحلقات والبيتزا وعودة
الساحر.}
\begin{enumerate}[resume]
  \item بركة دائرية نصف قطرها $3$ م تقع في مركز
        مرج دائري نصف قطره $5$ م. اِحسب \omterm{def:g6:measure:area}{مساحة}
        حلقة العشب (مضبوطة بالعدد $\pi$، ثم مقرّبة إلى
        م$^2$).
  \item جِد \omterm{def:g3:shapes:shapes}{نصف قطر} القرص الوحيد الذي تساوي مساحته
        \omterm{def:g6:measure:area}{مساحة} تلك الحلقة بالضبط. (\omterm{def:g3:shapes:shapes}{وأنصاف الأقطار} الثلاثة التي عندك الآن
        تكوّن ثلاثية مشهورة، تُروى قصتها في
        \cref{ch:g8:pythagoras}.)
  \item أيّهما بيتزا أكثر: بيتزا واحدة قطرها $40$ سم، أم
        بيتزاتان \omterm{def:g4:geometry:circle}{قطر} كل منهما $28$ سم؟ اِحسب وقارن.
  \item \emph{ولمضاعفة} \omterm{def:g6:measure:area}{مساحة} بيتزا، بأيّ معامل تقريبًا
        يجب أن ينمو نصف قطرها؟ اِختبر المرشّح $1.4$ (وتذكّر
        أن المساحات تتبع \emph{مربّع} معامل
        \omterm{def:g7:prop:scale}{السلّم}، \cref{pb:g6:propdata:1})، وقل أين يدخل
        المعامل المضبوط --- وهو العدد الذي مربّعه $2$ ---
        دخوله الرسمي
        (\cref{ex:g8:pythagoras:diagonal}).
  \item ويعود الساحر: فيقطع مربعًا $13 \times 13$
        والأعداد $5$ و $8$ و $13$ تلعب الأدوار التي لعبتها
        $3$ و $5$ و $8$ من قبل، ويعيد تركيب
        مستطيل $8 \times 21$. اِحسب المساحتين: فما الذي
        يتلاشى هذه المرة؟ والأعداد $3, 5, 8, 13, 21$
        حدود متتالية من متتالية
        \cref{pb:g7:fractions:1} --- اُذكر نمط
        مكاسب الساحر وخسائره، والدرس: لماذا لا تكذب المساحات
        أبدًا؟
\end{enumerate}
\end{problem}
