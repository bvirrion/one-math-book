\chapter{تنظيم المعطيات}\label{ch:g7:stats}

قبل حساب أيّ شيء، يجب جمع المعطيات وعدّها
وعرضها. ويدور هذا الفصل حول تحويل قائمة ملاحظات فوضوية
إلى جدول أو صورة واضحة: التعدادات والتواترات والمعطيات المجمّعة ومخططات
الأعمدة ومخططات القطاعات. وأما تلخيص المعطيات بعدد واحد (المتوسط
والوسيط) فموضوع \cref{ch:g9:statproba}.

\section{التعدادات والتواترات}

\begin{definition}[التعداد والتواتر]\label{def:g7:stats:frequency}
من أجل كل قيمة ممكنة في اِستطلاع، \emph{تعدادها}\index{تعداد} هو
عدد المرات التي تظهر فيها؛ و\emph{تواترها}\index{تواتر}
هو التعداد مقسومًا على العدد الكلي للملاحظات:
\[
\text{التواتر} = \frac{\text{التعداد}}{\text{المجموع}} .
\]
ويمكن كتابة التواترات كسورًا أو أعدادًا عشرية أو نسبًا مئوية، وهي
تتجمّع دائمًا إلى $1$ (أي\ $100\,\%$).
\end{definition}

\begin{example}\label{ex:g7:stats:frequency}
الرياضة المفضّلة عند تلاميذ قسم عددهم $25$، من القائمة الخام إلى
الجدول:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cccc|c}
الرياضة & كرة القدم & السباحة & التنس & الجودو & \omterm{def:g1:addition:def}{المجموع} \\
\hline
\omterm{def:g7:stats:frequency}{التعداد} & $10$ & $6$ & $5$ & $4$ & $25$ \\
\omterm{def:g7:stats:frequency}{التواتر} & $0.40$ & $0.24$ & $0.20$ & $0.16$ & $1$ \\
\end{tabular}
\end{center}
من أجل كرة القدم: $\frac{10}{25} = \frac{40}{100} = 0.40 = 40\,\%$. ويجب
أن يكون \omterm{def:g1:addition:def}{مجموع} سطر \omterm{def:g7:stats:frequency}{التواتر} $1$ --- وهو تحقّق دائم مجّاني.
\end{example}

\begin{remark}[لماذا التواترات؟]
لا يمكن مقارنة التعدادات بين مجموعات مختلفة \omterm{def:g6:measure:volume}{الحجم}: فالعدد $10$
من محبّي كرة القدم في قسم من $25$ (أي $40\,\%$) يدلّ على حماسة
أكبر من $12$ في مستوى دراسي من $120$ ($10\,\%$).
والتواترات تضع كل شيء على \omterm{def:g7:prop:scale}{السلّم} نفسه.
\end{remark}

\section{تجميع المعطيات في فئات}

\begin{example}[الفئات]\label{ex:g7:stats:classes}
اِرتفاعات (بالسنتيمتر) $20$ تلميذًا تتراوح بين $148$ و $172$.
وسرد كل اِرتفاع على حدة يعطي جدولًا فيه من الأعمدة ما يقارب عدد
التلاميذ؛ وبدلًا من ذلك، جمّعها في \emph{فئات}:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|ccc|c}
الارتفاع (سم) & $\intco{145}{155}$ & $\intco{155}{165}$ &
$\intco{165}{175}$ & \omterm{def:g1:addition:def}{المجموع} \\
\hline
\omterm{def:g7:stats:frequency}{التعداد} & $5$ & $9$ & $6$ & $20$ \\
\omterm{def:g7:stats:frequency}{التواتر} & $0.25$ & $0.45$ & $0.30$ & $1$ \\
\end{tabular}
\end{center}
ويُعدّ كل اِرتفاع في فئة واحدة بالضبط (فالعدد $155$ يدخل في
$\intco{155}{165}$، لا في الاثنتين --- ولهذا كانت الفترات
نصف مفتوحة). والتجميع يفقد التفصيل ويكسب الوضوح.
\end{example}

\section{المخططات}

\begin{method}[اِختيار مخطط ورسمه]\label{met:g7:stats:charts}
\begin{enumerate}
  \item \emph{مخطط الأعمدة}: عمود واحد لكل قيمة أو فئة، واِرتفاعه
        متناسب مع \omterm{def:g7:stats:frequency}{التعداد} (أو \omterm{def:g7:stats:frequency}{التواتر}). وهو جيد
        لمقارنة القيم.
  \item \emph{مخطط القطاعات}: قطاع واحد لكل قيمة، وزاويته متناسبة مع
        \omterm{def:g7:stats:frequency}{التواتر} --- فالدائرة الكاملة ($360^\circ$) تمثّل
        \omterm{def:g1:addition:def}{المجموع}، ومنه
        \[
        \text{الزاوية} = \text{التواتر} \times 360^\circ .
        \]
        وهو جيد لبيان الحصص من كلّ.
  \item وفي الحالتين: عنوان، ووسوم، وتدريج أو
        النسب المئوية مكتوبة على المخطط.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:stats:pie}
زوايا رياضات \cref{ex:g7:stats:frequency}: كرة القدم
$0.40 \times 360 = 144^\circ$؛ والسباحة $0.24 \times 360 =
86.4^\circ$؛ والتنس $72^\circ$؛ والجودو $57.6^\circ$. والتحقّق:
$144 + 86.4 + 72 + 57.6 = 360$.
\end{example}

\begin{example}[القراءة النقدية]\label{ex:g7:stats:critical}
المخطط الذي يبدأ محوره العمودي من $9$ بدل $0$ يجعل عمود
$10$ يبدو أطول عشر مرات من عمود $9.1$ --- وهو تضليل
بصري، وإن كانت الأعداد أمينة. وأول ردّ فعل أمام
أيّ مخطط: تحقّق من أين يبدأ المحور وكم تساوي التدريجة الواحدة
(\cref{met:g6:propdata:read}).
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g7:stats:1}
هذه علامات $20$ تلميذًا: $12$ و $15$ و $9$ و $12$ و $14$
و $9$ و $12$ و $15$ و $11$ و $12$ و $9$ و $14$ و $12$ و $11$ و $15$ و $9$ و $12$
و $14$ و $11$ و $12$. اِبنِ جدول التعدادات (عمود واحد لكل علامة).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:2}
أكمل جدول \cref{exo:g7:stats:1} بسطر \omterm{def:g7:stats:frequency}{تواتر}
(كسورًا من $20$، ثم نسبًا مئوية). وتحقّق من \omterm{def:g1:addition:def}{المجموع}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:3}
في اِستطلاع شمل $50$ أسرة، عند $30$ منها سيارة واحدة. فما
\omterm{def:g7:stats:frequency}{تواتر} «سيارة واحدة»، عددًا عشريًّا ونسبةً مئوية؟ وفي اِستطلاع
آخر، عند $45$ أسرة من $90$ سيارة واحدة: فأيّ اِستطلاع يبيّن
النسبة الأعلى؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:4}
كتل (بالكيلوغرام) $15$ كلبًا: $8$ و $23$ و $31$ و $12$ و $9$ و $27$ و $35$
و $14$ و $22$ و $18$ و $29$ و $11$ و $25$ و $33$ و $16$. جمّعها في
الفئات $\intco{5}{15}$ و $\intco{15}{25}$ و $\intco{25}{35}$
و $\intco{35}{45}$ وأعطِ التعدادات.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:5}
اُرسم مخطط الأعمدة للجدول:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|cccc}
الحيوانات الأليفة & $0$ & $1$ & $2$ & $3$ \\
\hline
الأسر & $8$ & $12$ & $6$ & $4$ \\
\end{tabular}
\end{center}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:6}
من أجل جدول \cref{exo:g7:stats:5}، اِحسب التواترات
وزوايا مخطط القطاعات لكل قيمة (والتحقّق: \omterm{def:g1:addition:def}{المجموع} $360^\circ$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:7}
مخطط قطاعات عن الفصول المفضّلة يبيّن: الصيف $162^\circ$، والربيع
$90^\circ$، والخريف $54^\circ$، والشتاء $54^\circ$. فأيّ نسبة مئوية
اِختارت كل فصل؟ وإذا أجاب $60$ شخصًا، فكم اِختاروا الصيف؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:8}
في القسم أ، يمشي $12$ تلميذًا من $30$ إلى المدرسة؛ وفي القسم ب،
$14$ من $40$. فأيّ قسم فيه نسبة الماشين أعلى؟
بَرِّر بالتواترات لا بالتعدادات.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:9}
تطبع مجلة مخطط أعمدة لمبيعات شهرية: $9\,800$، ثم
$10\,000$، ثم $10\,100$، والمحور العمودي يبدأ من
$9\,700$. صِف ما يراه القارئ، وما الذي يبيّنه محور أمين
يبدأ من $0$ بدل ذلك.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:10}
جدول \omterm{def:g7:stats:frequency}{تواتر} فيه ثلاث قيم تواتراتها $0.35$ و $0.4$ و
$f$. جِد $f$. وإذا كان \omterm{def:g7:stats:frequency}{التعداد} الكلي $80$، فأعطِ التعدادات الثلاثة.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:stats:11}
في مدرسة، $55\,\%$ من التلاميذ بنات. ومن بين البنات،
تأكل $40\,\%$ في المطعم؛ ومن بين الأولاد، يأكل $60\,\%$. ومن بين
$400$ تلميذ، كم يأكل في المطعم؟ (اِحسب أحجام الفئات
الأربع خطوة خطوة.) وأيّ نسبة مئوية إجمالية هي تلك؟
\end{exercise}

\section{مسألة: كيف تكذب المعطيات --- وكيف تمسكها}

\begin{problem}[{مسألة نهاية الأسبوع --- التعدادات مقابل التواترات،
والاستطلاع الذي خدع بلدًا، ومفارقة
المستشفيين}]
\label{pb:g7:stats:1}
الأعداد لا تكذب، لكن يمكن أن تُساق لتضلّل: \omterm{def:g7:stats:frequency}{فتعداد} ضخم
قد يخفي تواترًا صغيرًا، واِستطلاع هائل قد يكون بلا قيمة،
و--- وهذا أغربها --- قد يغلب مستشفى منافسه في
\emph{كل} فئة من المرضى ويخسر مع ذلك في الأرقام
الإجمالية. وتُنصب الفخاخ الثلاثة كلها في هذه المسألة، بسلاح
تعدادات هذا الفصل وتواتراته لا غير
(\cref{def:g7:stats:frequency}).

\medskip
\noindent\textbf{الجزء الأول --- التعدادات ليست تواترات.}
\begin{enumerate}
  \item تعيد المدرسة أ تدوير $120$ من علب العصير $400$ التي عندها؛
        وتعيد المدرسة ب تدوير $90$ من علبها $250$. فأيّ مدرسة عندها
        \emph{\omterm{def:g7:stats:frequency}{التعداد}} الأكبر من العلب المعاد تدويرها؟ وأيّهما عندها
        \emph{\omterm{def:g7:stats:frequency}{التواتر}} الأكبر؟ وأيّ مدرسة تستحقّ جائزة
        التدوير؟
  \item يُستطلَع رأي في جدول زمني جديد: فيعجب $18$ من المعلّمين $30$،
        ويعجب $210$ من التلاميذ $600$. اِحسب
        التواترين، ثم \omterm{def:g7:stats:frequency}{التواتر} بين الأشخاص $630$
        كلهم معًا. ولماذا يقع الرقم المشترك بهذا
        القرب من رقم التلاميذ؟
  \item مخطط قطاعات منشور يبيّن قطاعات موسومة $45\,\%$
        و $30\,\%$ و $20\,\%$ و $10\,\%$. فدون أيّ معلومة
        أخرى، كيف تعرف أن خطأ قد وقع؟
  \item جدول \omterm{def:g7:stats:frequency}{تواتر} على $40$ ملاحظة نصفه ممحوّ:
        التعدادات $14$ و $10$ و $?$ و $?$ والتواترات $0.35$
        و $0.25$ و $0.15$ و $?$. أعِد بناء المداخل الناقصة.
  \item يُختتم اِستطلاع بالتواترات $0.40$ و $0.35$ و
        $0.25$. اِحسب زوايا مخطط القطاعات الثلاث
        (\cref{met:g7:stats:charts}) وتحقّق من أنها تغلق
        الدائرة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الاستطلاع الذي خدع بلدًا.}
في سنة 1936، أرسلت مجلة أمريكية عشرة ملايين بطاقة اِقتراع إلى
عناوين مأخوذة من أدلّة الهاتف وقوائم تسجيل
السيارات، وتلقّت $2.4$ مليون جواب، وتوقّعت اِنتصارًا
ساحقًا للمرشّح لاندون. وسأل إحصائي شابّ، هو جورج
غالوب، نحو خمسين ألف شخص فقط --- مختارين ليشبهوا
السكان كلهم --- وتوقّع العكس.
وفاز روزفلت فوزًا كاسحًا.
\begin{enumerate}[resume]
  \item وأيّ \omterm{def:g7:stats:frequency}{تواتر} من بطاقات الاقتراع المرسلة عاد؟
  \item في سنة 1936 كانت الهواتف والسيارات كماليات. اِشرح في
        جملة أو جملتين لماذا كانت عيّنة المجلة، على
        ضخامتها، محكومة بالفشل --- وأيّ درس من السؤال 1
        تكرّره على \omterm{def:g7:prop:scale}{السلّم} الوطني.
  \item نموذج مبسّط. في مدينة $1\,000$ ناخب ثريّ، يؤيّد
        $70\,\%$ منهم المرشّح ل، و $9\,000$ ناخب متواضع، يؤيّد
        $30\,\%$ منهم ل. اِحسب تأييد ل الحقيقي في
        المدينة. واِستطلاع «دليل الهاتف» يبلغ $500$ ثريًّا
        و $500$ متواضعًا: فأيّ تأييد يتوقّع؟
  \item والفخّ الصامت: عند شركة $200$ زبون غير راضٍ
        و $800$ راضٍ. فتستطلع الجميع؛ فيجيب $40\,\%$ من
        غير الراضين، ولا يجيب من الراضين إلا $10\,\%$.
        اِحسب عدد الأجوبة من كل نوع، \omterm{def:g7:stats:frequency}{وتواتر}
        عدم الرضا \emph{بين الأجوبة}.
        وقارنه \omterm{def:g7:stats:frequency}{بالتواتر} الحقيقي.
  \item أصلح اِستطلاع السؤال 8: اِحتفظ بعدد $1\,000$
        مقابلة، لكن وزّعها على نسب المدينة
        الحقيقية. فماذا يتوقّع الاستطلاع المُصلَح؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- مفارقة
المستشفيين.}
ينشر مستشفيان تعدادات شفائهما، مقسومة بحسب الخطورة:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cc}
 & الحالات الخفيفة & الحالات الخطيرة \\
\hline
المستشفى أ & $90$ شُفوا من $100$ & $280$ شُفوا من $400$ \\
المستشفى ب & $340$ شُفوا من $400$ & $65$ شُفوا من $100$ \\
\end{tabular}
\end{center}
\begin{enumerate}[resume]
  \item اِحسب \omterm{def:g7:stats:frequency}{تواتر} الشفاء في المستشفى أ من أجل الحالات الخفيفة، ومن أجل
        الحالات الخطيرة، وإجمالًا (المرضى $500$ كلهم).
  \item والحسابات الثلاثة نفسها من أجل المستشفى ب. فأيّ مستشفى
        يفوز في الحالات الخفيفة؟ وفي الحالات الخطيرة؟
  \item قارن التواترين الإجماليين. واُذكر بوضوح
        الأمر الغريب الذي حدث.
  \item اِشرح الحيلة: قارن \emph{خلطتَي} المرضى
        في المستشفيين، وقل لماذا يمكن \omterm{def:g7:stats:frequency}{للتواتر}
        الإجمالي أن يخون تواترَي الفئتين معًا. ولو كانت
        عندك حالة خطيرة، فأيّ مستشفى تختار؟
  \item يقول عنوان في صحيفة: «المستشفى ب أفضل في
        نسبة الشفاء: $81\,\%$ مقابل $74\,\%$.» اُكتب
        رسالة من جملتين إلى المحرّر مما علّمتك
        هذه المسألة كلها --- جملة عمّا
        يتجاهله العنوان، وجملة عن الدرس العامّ
        (قارن المتماثل بالمتماثل، واِنتبه للأوزان).
\end{enumerate}
\end{problem}
