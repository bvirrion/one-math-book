\chapter{المثلثات والزوايا}\label{ch:g7:triangles}

أتستطيع رسم مثلث أضلاعه $10$ و $3$ و $2$؟ أو زواياه
$80^\circ$ و $70^\circ$ و $50^\circ$؟ يجيب هذا الفصل عن السؤالين
بواقعتين من أنفع وقائع هندسة المستوي: المتراجحة
المثلثية \omterm{def:g1:addition:def}{ومجموع} الزوايا $180^\circ$ --- مع مفردات
الزوايا التي تأتي مع \omterm{def:g6:lines:objects}{المستقيمات} المتقاطعة والمتوازية.

\section{الزوايا حول مستقيمين متقاطعين}

\begin{definition}[أزواج الزوايا]\label{def:g7:triangles:pairs}
\begin{itemize}
  \item تكون زاويتان \emph{متتامّتين} حين يبلغ \omterm{def:g1:addition:def}{مجموع} قياسيهما
        $90^\circ$، و\emph{متكاملتين} حين يبلغ
        $180^\circ$.
  \item وحين يتقاطع \omterm{def:g6:lines:objects}{مستقيمان}، تكون الزاويتان المتقابلتان عبر
        نقطة التقاطع \emph{متقابلتين
        بالرأس}\index{زاويتان متقابلتان بالرأس}.
  \item وحين يقطع \omterm{def:g6:lines:objects}{مستقيم} \omterm{def:g6:lines:objects}{مستقيمين} آخرين، تكون الزوايا الواقعة في
        الموضع نفسه عند كل تقاطع
        \emph{زوايا متناظرة}؛ والزوايا الواقعة بين \omterm{def:g6:lines:objects}{المستقيمين} في
        جهتين متقابلتين من \omterm{def:g6:lines:objects}{المستقيم} القاطع \emph{زوايا
        متبادلة}\index{زوايا متبادلة}.
\end{itemize}
\end{definition}

\begin{theorem}[تساويات الزوايا]\label{thm:g7:triangles:equalities}
\begin{enumerate}
  \item \omterm{def:g7:triangles:pairs}{الزاويتان المتقابلتان بالرأس} متساويتان.
  \item وإذا كان \omterm{def:g6:lines:objects}{مستقيمان} \omterm{def:g6:lines:perp}{متوازيين}، فالزوايا المتناظرة متساوية،
        \omterm{def:g7:triangles:pairs}{والزوايا المتبادلة} متساوية. وبالعكس، تساوي الزوايا المتناظرة
        (أو المتبادلة) يفرض توازي \omterm{def:g6:lines:objects}{المستقيمين}.
\end{enumerate}
\end{theorem}

\begin{proof}[برهان 1]
الزاويتان $\widehat{1}$ و $\widehat{2}$ في جهة واحدة من \omterm{def:g6:lines:objects}{مستقيم}
تكوّنان زاوية مستقيمة: أي $\widehat 1 + \widehat 2 = 180^\circ$. والزاوية
$\widehat 3$ المتقابلة بالرأس مع $\widehat 1$ تحقّق أيضًا
$\widehat 2 + \widehat 3 = 180^\circ$ (للسبب نفسه). إذن
$\widehat 1 = 180^\circ - \widehat 2 = \widehat 3$. وأما النقطة 2 فمقبولة
في هذا المستوى.
\end{proof}

\section{مجموع زوايا المثلث}

\begin{theorem}[مجموع الزوايا]\label{thm:g7:triangles:sum}
في كل مثلث، يبلغ \omterm{def:g1:addition:def}{مجموع} الزوايا الثلاث $180^\circ$.
\end{theorem}

\begin{proof}
ليكن $ABC$ مثلثًا. اُرسم \omterm{def:g6:lines:objects}{المستقيم} المارّ بالنقطة $A$ \omterm{def:g6:lines:perp}{الموازي} \omterm{def:g6:lines:objects}{للمستقيم}
$(BC)$. فعند الرأس $A$، تصطفّ ثلاث زوايا على هذا \omterm{def:g6:lines:perp}{الموازي}
وتكوّن زاوية مستقيمة، أي $180^\circ$: الوسطى منها
$\widehat{A}$، والطرفيتان زاويتان متبادلتان مع
$\widehat B$ و $\widehat C$ (\omterm{def:g6:lines:perp}{فالمستقيمان متوازيان}!)، إذن تساويان
$\widehat B$ و $\widehat C$. ومنه
$\widehat B + \widehat A + \widehat C = 180^\circ$.
\end{proof}

\begin{example}\label{ex:g7:triangles:sum}
زاويتان من مثلث قياسهما $67^\circ$ و $58^\circ$. والثالثة
قياسها
\[
180 - (67 + 58) = 180 - 125 = 55^\circ .
\]
وحالات خاصّة تستحقّ الحفظ: كل زاوية من \omterm{def:g6:shapes:triangles}{المثلث المتساوي الأضلاع}
قياسها $180 \div 3 = 60^\circ$؛ وفي \omterm{def:g6:shapes:triangles}{المثلث القائم الزاوية} تكون الزاويتان الحادّتان
متتامّتين (فمجموعهما $180 - 90 = 90^\circ$)؛ وزاويتا قاعدة
\omterm{def:g6:shapes:triangles}{المثلث المتساوي الساقين} متساويتان، فكل منهما
$\frac{180 - \text{الرأس}}{2}$.
\end{example}

\begin{example}[لا مثلث بزاويتين قائمتين]\label{ex:g7:triangles:tworight}
المثلث ذو الزاويتين $90^\circ$ يستهلك أصلًا
$180^\circ$، فلا يترك للثالثة إلا $0^\circ$ --- وهذا مستحيل. \omterm{def:g1:addition:def}{فمجموع}
الزوايا يمنع مثلثات كثيرة دفعة واحدة.
\end{example}

\section{المتراجحة المثلثية}

\begin{theorem}[المتراجحة المثلثية]\label{thm:g7:triangles:inequality}
في كل مثلث، يكون كل ضلع أقصر من \omterm{def:g1:addition:def}{مجموع} الضلعين
الآخرين: فمن أجل مثلث أضلاعه $a$ و $b$ و $c$،
\[
a < b + c .
\]
وبالعكس، ثلاثة أطوال \emph{أكبرها} أصغر من \omterm{def:g1:addition:def}{مجموع}
الآخرين يمكن دائمًا جعلها مثلثًا. وإذا كان الأكبر
\emph{يساوي} \omterm{def:g1:addition:def}{مجموع} الآخرين، كان ذلك «المثلث» مسطّحًا: \omterm{def:g4:geometry:polygon}{فرؤوسه}
الثلاثة مستقيمية.
\end{theorem}
\admitted

\begin{example}\label{ex:g7:triangles:inequality}
أيمكن أن تكون الأضلاع $10$ و $3$ و $2$؟ اِختبر الأكبر: أهو $10 < 3 + 2 = 5$؟
لا --- فلا وجود لمثلث كهذا. والفرجار يبيّن السبب: فقوسان نصفا قطريهما
$3$ و $2$ مرسومان من طرفَي \omterm{def:g6:lines:objects}{قطعة} طولها $10$ لا يتقاطعان
أبدًا.

والأضلاع $7$ و $5$ و $4$؟ الأكبر: $7 < 5 + 4 = 9$: نعم، فالمثلث
موجود (ويكفي اِختبار الضلع الأكبر).
\end{example}

\begin{method}[وجود مثلث وإنشاؤه]\label{met:g7:triangles:construct}
من أجل ثلاثة أطوال معطاة:
\begin{enumerate}
  \item قارن الأكبر \omterm{def:g1:addition:def}{بمجموع} الآخرين؛ فإن لم يكن
        أصغر تمامًا، فتوقّف: لا مثلث؛
  \item وإلا فأنشئه كما في \cref{met:g6:shapes:construct}
        (\omterm{def:g6:lines:objects}{قطعة} القاعدة، وقوسا الفرجار)؛
  \item وبعد أيّ إنشاء من الزوايا، تحقّق ذهنيًّا من أن
        الزوايا المعطاة مجموعها أصغر من $180^\circ$ --- مع الزاوية
        الثالثة التي تجعله $180^\circ$ بالضبط.
\end{enumerate}
\end{method}

\section{تمارين}

أعطِ متمّمة (إلى $90^\circ$) ومكمّلة (إلى
$180^\circ$) كل من: $30^\circ$؛ \ و $45^\circ$؛ \ و $72^\circ$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:2}
يتقاطع \omterm{def:g6:lines:objects}{مستقيمان}؛ وإحدى الزوايا الأربع قياسها $35^\circ$. أعطِ
قياسات الثلاث الأخرى، مع سبب لكل واحدة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:3}
اِحسب الزاوية الثالثة لمثلث قياسا زاويتيه الأوليين:
(أ) $40^\circ$ و $60^\circ$؛ \
(ب) $90^\circ$ و $28^\circ$؛ \
(ج) $75^\circ$ و $75^\circ$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:4}
\omterm{def:g6:shapes:triangles}{مثلث متساوي الساقين} زاوية رأسه $40^\circ$. اِحسب
زاويتَي قاعدته. ومثلث آخر \omterm{def:g6:shapes:triangles}{متساوي الساقين} إحدى زاويتَي قاعدته
$70^\circ$: اِحسب زاوية رأسه.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:5}
أيّ هذه الثلاثيات يمكن أن تكون أضلاع مثلث؟ بَرِّر بالمتراجحة
المثلثية (الضلع الأكبر!):
\[
(6,\ 8,\ 12); \qquad
(5,\ 5,\ 11); \qquad
(4,\ 9,\ 13); \qquad
(7,\ 7,\ 7).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:6}
مثلث زواياه $x$ و $2x$ و $3x$. جِد $x$ والزوايا

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:7}
اِبنِ مثلثًا $ABC$ حيث $BC = 6$ سم،
و $\widehat{ABC} = 50^\circ$ و $\widehat{ACB} = 60^\circ$ (بالمنقلة
عند $B$ وعند $C$). وقبل الإنشاء، توقّع قياس

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:8}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:9}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:10}
في مثلث $ABC$، $\widehat A = 2\widehat B$ و

\begin{exercise}[$\star\star\star$]\label{exo:g7:triangles:11}

\section{مسألة: المشي حول المربّع السكني}

\begin{problem}[{مسألة نهاية الأسبوع --- زوايا أيّ مضلع:
$(n-2) \times 180^\circ$ في الداخل، ودائمًا $360^\circ$ من الدوران
في الخارج، ولماذا لا تبلّط الأرض إلا ثلاثة أشكال منتظمة}]
\label{pb:g7:triangles:1}
قطع \cref{exo:g7:triangles:11} رباعيًا إلى مثلثين
فوجد $360^\circ$. وتدفع هذه المسألة الفكرة إلى المضلعات
ذات أيّ عدد من الأضلاع، ثم تكتشف برهانًا ثانيًا مختلفًا
تمامًا --- عن طريق \emph{المشي} حول الشكل وجمع
الدورات --- وتنتهي على أرض حمّامك: فمن بين كل
المضلعات المنتظمة، ثلاثة بالضبط تبلّط المستوي دون فراغ ولا
تراكب، وقد اِختار النحل مضلعه.

\medskip
\noindent\textbf{الجزء الأول --- الزوايا في الداخل.}
\begin{enumerate}
  \item من رأس واحد \omterm{def:g4:geometry:polygon}{لسداسي} (ستة أضلاع)، اُرسم كل
        الأقطار الخارجة من ذلك الرأس. فإلى كم مثلثًا يُقطع
        \omterm{def:g4:geometry:polygon}{السداسي}، وما \omterm{def:g1:addition:def}{مجموع} \emph{كل} زواياه
        الداخلية (\cref{thm:g7:triangles:sum})؟
  \item عمّم: من رأس واحد \omterm{def:g4:geometry:polygon}{لمضلع} ذي $n$ ضلعًا،
        كم مثلثًا تنتج مروحة الأقطار؟
        اِستنتج صيغة \omterm{def:g1:addition:def}{مجموع} الزوايا الداخلية،
        وتحقّق منها من أجل $n = 3$ و $4$ و $5$ و $6$.
  \item اِحسب \omterm{def:g1:addition:def}{مجموع} الزوايا الداخلية لعشاري
        ($10$ أضلاع) ولاثني عشري ($12$ ضلعًا).
  \item في \omterm{def:g4:geometry:polygon}{المضلع} \emph{المنتظم} تتساوى كل الزوايا
        الداخلية. اِحسب الزاوية الداخلية \omterm{def:g6:shapes:triangles}{للمثلث المتساوي
        الأضلاع}، وللمربع، وللخماسي المنتظم، وللسداسي
        وللمثمّن.
  \item اِشرح، دون أيّ صيغة، لماذا ينمو \omterm{def:g1:addition:def}{مجموع} الزوايا بمقدار
        $180^\circ$ بالضبط كلما كسب \omterm{def:g4:geometry:polygon}{المضلع} رأسًا
        واحدًا.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- المشي حول المربّع السكني.}
اِمشِ على حدّ مربّع سكني مضلّع، إلى الأمام دائمًا. وعند
كل ركن \emph{تدور} بزاوية ما --- وهي \emph{الزاوية
الخارجية} لذلك الركن؛ ويستمرّ الدوران حتى تعود إلى
نقطة اِنطلاقك، مواجهًا اِتجاه اِنطلاقك.
\begin{enumerate}[resume]
  \item عند ركن زاويته الداخلية $\widehat A$، بكم
        تدور؟ (فالزاويتان الداخلية والخارجية
        متكاملتان.) اِحسب
        الدورات الثلاث لمثلث زواياه $40^\circ$
        و $60^\circ$ و $80^\circ$.
  \item اِجمع الدورات الثلاث في السؤال 6. فأيّ واقعة عن
        الدورة الكاملة تلاحظ؟
  \item اِشرح لماذا تبلغ دورات مشية كاملة حول
        \emph{أيّ} \omterm{def:g4:geometry:polygon}{مضلع} --- ثلاثة أضلاع أو ثلاثين --- دائمًا
        $360^\circ$ بالضبط. (فماذا فعل أنفك، بكل
        الدورات مجتمعة، حين تعود؟)
  \item اِستنتج صيغة الزوايا الداخلية مرة ثانية: عند كل
        ركن من الأركان $n$، الداخلية $+$ الدورة $= 180^\circ$؛ فاِجمع
        على كل الأركان واِستعمل السؤال 8.
  \item في \omterm{def:g4:geometry:polygon}{المضلع} المنتظم، تتساوى كل الدورات، فتكون كل
        زاوية خارجية $\frac{360^\circ}{n}$. اِستعمل هذا
        لتجيب في الحال: أيّ \omterm{def:g4:geometry:polygon}{مضلع} منتظم زواياه
        الداخلية $150^\circ$؟ ولماذا \emph{لا} يوجد \omterm{def:g4:geometry:polygon}{مضلع}
        منتظم زواياه الداخلية $155^\circ$؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- تبليط الأرض.}
حول كل نقطة من أرض مبلّطة، يجب أن يبلغ \omterm{def:g1:addition:def}{مجموع} أركان
البلاطات الملتقية $360^\circ$ بالضبط --- بلا فراغ ولا تراكب
(\cref{pb:g6:lines:1}).
\begin{enumerate}[resume]
  \item من أجل كل من \omterm{def:g6:shapes:triangles}{المثلث المتساوي الأضلاع} والمربع \omterm{def:g4:geometry:polygon}{والسداسي}
        المنتظم: كم نسخة تلتقي عند نقطة ركن
        من التبليط؟ وتحقّق من $360^\circ$ في كل مرة.
  \item بيّن أن الخماسيات المنتظمة لا تستطيع تبليط الأرض: فما
        \omterm{def:g1:addition:def}{مجموع} ثلاثة أركان، وما \omterm{def:g1:addition:def}{مجموع} أربعة؟
  \item بيّن أنه لا يمكن لأيّ \omterm{def:g4:geometry:polygon}{مضلع} منتظم \emph{بأكثر} من ستة
        أضلاع أن يبلّط: فلا بدّ أن تلتقي ثلاث بلاطات على الأقلّ عند كل
        ركن (فكل زاوية أصغر من $180^\circ$)، وماذا
        يحدث لثلاثة أركان قياس كل منها $135^\circ$ أو أكثر؟ واِستنتج
        القائمة الكاملة للمبلِّطات المنتظمة.
  \item كثيرًا ما تخلط أرضيات الحمّامات مثمّنات منتظمة بمربعات
        صغيرة. تحقّق من أن هذا الركن يعمل:
        $135^\circ + 135^\circ + 90^\circ$. وخلطة كلاسيكية أخرى تجمع
        مربعًا وسداسيًّا منتظمًا واِثني عشريًّا منتظمًا عند
        كل ركن: تحقّق منها أيضًا (وزاوية الاثني عشري
        الداخلية تنتج من طريقة السؤال 10).
  \item اِختيار النحل: خلايا قرص العسل سداسيات منتظمة.
        تحقّق من أركانها ($3$ زوايا)، واِشرح في جملة
        واحدة --- بمساعدة اِكتشاف ديدو
        (\cref{pb:g6:measure:1}) --- لماذا يكون \omterm{def:g4:geometry:polygon}{السداسي}، من بين البلاطات
        المنتظمة الثلاث الممكنة، أرخصها في الشمع مقابل
        العسل الذي يحمله.
\end{enumerate}
\end{problem}
