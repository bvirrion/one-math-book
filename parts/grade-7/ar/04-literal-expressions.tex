\chapter{العبارات الحرفية}\label{ch:g7:literal}

الحرف في حساب يمثّل عددًا لا نعرفه بعدُ، أو
\emph{كل} الأعداد دفعة واحدة: فكتابة $P = 4c$ تعطي \omterm{def:g6:measure:perimeter}{محيط}
كل مربع في صيغة واحدة. ويعلّم هذا الفصل اِصطلاحات
الكتابة، وكيف تُعوَّض القيم، وكيف تُختزل العبارات
البسيطة --- وهي بداية الجبر، وتُتابَع في
\cref{ch:g8:equations}.

\section{الكتابة بالحروف}

\begin{definition}[العبارة الحرفية]\label{def:g7:literal:def}
\emph{العبارة الحرفية}\index{عبارة حرفية} حساب
يحوي حرفًا أو أكثر، كل منها يمثّل عددًا. واِصطلاحات
الكتابة:
\begin{itemize}
  \item تُحذف إشارة $\times$ قبل حرف أو قوس:
        $3 \times a = 3a$، \ و $a \times b = ab$، \
        و $2 \times (x + 5) = 2(x + 5)$؛
  \item لكنها تبقى بين عددين: \omterm{def:g2:mult:def}{فالجداء} $3 \times 5$ لا يمكن أن
        يُكتب $35$!
  \item \omterm{def:g2:mult:def}{والجداء} $a \times a$ يُكتب $a^2$ («مربع $a$»)،
        و $a \times a \times a = a^3$؛
  \item و $1a$ يُكتب $a$، و $0a$ هو $0$.
\end{itemize}
\end{definition}

\begin{example}\label{ex:g7:literal:write}
بسّط الكتابة:
\[
5 \times x + 3 \times y = 5x + 3y,
\qquad
a \times 7 = 7a \ \text{(العدد أولًا)},
\qquad
x \times x \times 4 = 4x^2 .
\]
\end{example}

\begin{example}[الصيغ عبارات حرفية]\label{ex:g7:literal:formulas}
من أجل مستطيل طوله $L$ وعرضه $w$:
\omterm{def:g6:measure:perimeter}{المحيط} $P = 2(L + w)$، \omterm{def:g6:measure:area}{والمساحة} $A = Lw$. فصيغة واحدة تلخّص
المستطيلات التي لا نهاية لعددها دفعة واحدة --- وذلك هو مغزى
الحروف كله.
\end{example}

\section{التعويض}

\begin{method}[حساب قيمة عبارة]\label{met:g7:literal:substitute}
لحساب قيمة عبارة من أجل قيم معطاة لحروفها:
\begin{enumerate}
  \item أعِد كتابة العبارة بإرجاع إشارات $\times$ المخفية:
        فالعبارة $3a + a^2$ هي $3 \times a + a \times a$؛
  \item وعوّض كل حرف بقيمته، مع الإبقاء على الأقواس حوله
        عند اللزوم؛
  \item ثم اِحسب، محترمًا أولويات
        \cref{ch:g7:priorities}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:literal:substitute}
اِحسب قيمة $E = 3x + x^2$ من أجل $x = 5$:
\[
E = 3 \times 5 + 5 \times 5 = 15 + 25 = 40 .
\]
واِحسب قيمة $F = 2(a + 3b)$ من أجل $a = 4$ و $b = 0.5$:
\[
F = 2 \times (4 + 3 \times 0.5) = 2 \times (4 + 1.5) = 2 \times 5.5
= 11 .
\]
\end{example}

\section{اِختزال العبارات}

\begin{proposition}[تجميع الحدود المتشابهة]\label{prop:g7:literal:reduce}
الحدود التي لها الجزء الحرفي نفسه يمكن تجميعها بجمع
أجزائها العددية (وهذه هي التوزيعية،
\cref{thm:g7:priorities:distributivity}، مقروءةً بالمقلوب):
\[
5x + 3x = (5 + 3)x = 8x,
\qquad
7a - 2a = 5a .
\]
أما الحدود ذات الأجزاء الحرفية المختلفة (مثل $3x$ و $5y$، أو $x$ و
$x^2$) فلا يمكن تجميعها.
\end{proposition}

\begin{example}\label{ex:g7:literal:reduce}
اِختزل $E = 4x + 7 + 3x + 2$، بترتيب الحدود:
\[
E = (4x + 3x) + (7 + 2) = 7x + 9 .
\]
واِنتبه: $7x + 9$ \emph{ليس} $16x$! اِختبر بالقيمة $x = 1$:
$4 + 7 + 3 + 2 = 16$ و $7 \times 1 + 9 = 16$ متوافقان، لكن $16x$ سيكون
أيضًا $16$ --- فاختبر بالقيمة $x = 2$: تعطي العبارة الأصلية
$8 + 7 + 6 + 2 = 23$، و $7 \times 2 + 9 = 23$، بينما
$16 \times 2 = 32$. فالاختزال $7x + 9$ صحيح، و $16x$ خطأ.
\end{example}

\begin{method}[اِختبار تساوٍ]\label{met:g7:literal:test}
للتحقّق مما إذا كانت عبارتان قد تتساويان من أجل كل قيمة:
\begin{enumerate}
  \item عوّض القيمة نفسها في كلتيهما، مرتين على الأقلّ (مثلًا
        $x = 2$ و $x = 10$؛ وتجنّب $0$ و $1$، فهما يخفيان فروقًا
        كثيرة جدًّا)؛
  \item وإن اِختلفت النتيجتان ولو مرة واحدة، فالعبارتان
        \emph{غير} متساويتين --- فمثال مضادّ واحد يقتل اِدّعاءً عامًّا؛
  \item وإن توافقت النتائج، فالتساوي \emph{محتمل} لا غير: إذ
        يحتاج البرهان إلى الجبر (نشر واِختزال).
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:literal:test}
هل $2(x + 3)$ يساوي $2x + 6$ من أجل كل $x$؟ النشر يبرهن على ذلك:
$2(x+3) = 2x + 2 \times 3 = 2x + 6$. وهل $(x + 1)^2$ يساوي
$x^2 + 1$؟ اِختبر بالقيمة $x = 2$: $(2+1)^2 = 9$ لكن $2^2 + 1 = 5$. لا ---
ومثال مضادّ واحد برهان كامل على «لا».
\end{example}

\section{إنتاج العبارات}

\begin{example}[من الوضعية إلى العبارة]\label{ex:g7:literal:produce}
سيارة أجرة تأخذ $4$ يورو زائد $2$ يورو لكل كيلومتر. ومن أجل رحلة من
$k$ كيلومترًا يكون الثمن
\[
p = 4 + 2k \quad\text{يورو}.
\]
ومن أجل $k = 7$: $p = 4 + 14 = 18$ يورو. فالعبارة \emph{هي} الجواب
العامّ؛ والتعويض ينتج كل جواب خاصّ.
\end{example}

\begin{example}[حيل الأعداد]\label{ex:g7:literal:trick}
«فكّر في عدد، وضاعِفه، وأضِف $10$، واقسم على $2$، واطرح
عددك.» سمِّ العدد $n$ واِتّبع الخطوات:
\[
n \ \to\ 2n \ \to\ 2n + 10 \ \to\ \frac{2n + 10}{2} = n + 5 \ \to\
n + 5 - n = 5 .
\]
فيحصل الجميع على $5$، مهما كان $n$ --- فالحروف تشرح
السحر.
\end{example}

\section{مسألة: عرّاف أعواد الثقاب}

\begin{problem}[{مسألة نهاية الأسبوع --- الأنماط المتنامية وحدّها
ذو الرتبة $n$: ثلاث صيغ لنمط واحد، والنمط
المشهور الذي ينكسر عند الخطوة السادسة}]
\label{pb:g7:literal:1}
ضع أعواد ثقاب في صفّ مربعات: مربع واحد، ثم اِثنان، ثم
ثلاثة\,\dots\ فكم عودًا يحتاج الشكل المئة؟
عدّها واحدًا واحدًا يستغرق نهاية الأسبوع كلها؛ وأما \emph{الصيغة
بدلالة $n$} فتجيب في الحال عن كل شكل دفعة واحدة --- وذلك
بالضبط ما وُجدت الحروف من أجله
(\cref{ex:g7:literal:formulas}). لكن اِحذر: فهذه المسألة تنتهي
بنمط شهير يلعب بنزاهة خمس خطوات ثم
يخون كل من وثق به.

\medskip
\noindent\textbf{الجزء الأول --- صفّ مربعات.}
الشكل $1$ مربع من أعواد الثقاب ($4$ أعواد)؛ وكل شكل تالٍ
يضيف مربعًا آخر يشترك في ضلع مع سابقه.
\begin{enumerate}
  \item اُرسم الأشكال من $1$ إلى $4$ وعُدّ أعوادها.
  \item اِشرح لماذا يكلّف كل مربع جديد $3$ أعواد إضافية
        بالضبط، واِستعمل ذلك لتوقّع عدد أعواد الشكل
        $10$ دون أيّ صيغة.
  \item بَرِّر الصيغة: الشكل $n$ يحتاج $3n + 1$ عودًا.
        (فمن أين تأتي «$+1$» الوحيدة؟) وتحقّق منها
        بمقارنتها بالسؤال 1.
  \item ويستدلّ أمير بطريقة أخرى: «$4$ أعواد للمربع
        الأول، ثم $3$ لكل واحد من المربعات $n - 1$ الباقية»، فيعطي
        $4 + 3(n - 1)$. اُنشر عبارته واِختزلها
        (\cref{prop:g7:literal:reduce}): فهل يوافق
        السؤال 3؟
  \item وتعدّ لينا بطريقة ثالثة: «$2n$ عودًا أفقيًّا،
        و $n + 1$ عودًا عموديًّا.» اُكتب صيغتها، واِختزلها،
        واِستنتج. ثم أجب عن السؤال الافتتاحي: كم عودًا
        يحتاج الشكل $100$؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- المثلثات والطاولات والسير
إلى الوراء.}
\begin{enumerate}[resume]
  \item صفّ مثلثات من أعواد الثقاب (كل مثلث جديد يتّكئ على
        السابق): عُدّ الأعواد من أجل $1$ و $2$ و $3$
        مثلثات، ثم بَرِّر الصيغة $2n + 1$ من أجل $n$
        مثلثًا.
  \item والسير إلى الوراء: بعدد $49$ عودًا بالضبط، كم
        مثلثًا يمكن أن يضمّ الصفّ؟ (اِعكس الصيغة خطوة
        خطوة.)
  \item طاولات ولائم مربعة تُجلس ضيفًا واحدًا على كل ضلع حرّ. وحين
        تُضمّ $n$ طاولة في صفّ واحد طويل، اِشرح لماذا تكون
        صيغة المقاعد $2n + 2$، وتحقّق منها من أجل $1$ و $2$
        و $3$ طاولات.
  \item حفلة تتوقّع $30$ ضيفًا. فكم طاولة يحتاج صفّ
        واحد؟ ثم يقسم المُطعِم الطاولات نفسها إلى
        صفّين من سبع: فكم ضيفًا يتّسع الآن؟
        اِشرح الكسب بالصيغة.
  \item نمط النقط: الشكل $n$ هو حرف L من النقط، اِرتفاعه $n$
        نقطة وعرضه $n$ نقطة (والركن يُعدّ مرة واحدة). عُدّ
        نقط الأشكال $1$ و $2$ و $3$، وجِد الصيغة، وسمِّ
        الأعداد التي تنتجها. (وبتكديسها معًا، تبلّط هذه الحروف
        مربعًا --- وهي قصة تُتابَع في
        \cref{pb:g8:equations:1}.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- ثِق، واِختبر، وبَرهِن.}
\begin{enumerate}[resume]
  \item يدّعي زميل أن الشكل $n$ من الجزء الأول يحتاج
        $3(n + 2) - 5$ عودًا. اِحسم الادّعاء بالنشر
        والاختزال --- دون حاجة إلى اِختبار.
  \item «التربيع يكبّر الأعداد، إذن $n^2 \geq n$ من أجل
        كل عدد $n$.» اِختبر الادّعاء من أجل $n = 2$ و $3$
        و $10$؛ ثم جِد عددًا يفشل فيه. فماذا برهنت
        الاختبارات الثلاثة الناجحة، وماذا يبرهن الفشل الواحد
        (\cref{met:g7:literal:test})؟
  \item علّم نقطًا على دائرة واُرسم \emph{كل} \omterm{def:g6:lines:objects}{قطعة}
        بينها، ثم عُدّ المناطق داخل الدائرة.
        اِفعل ذلك من أجل $2$ و $3$ و $4$ و $5$ نقط (وضع النقط
        بتباعد غير منتظم، بحيث لا تتقاطع ثلاث قطع في نقطة
        داخلية واحدة). سجّل الأعداد. فأيّ صيغة يهمس بها
        النمط؟
  \item والآن $6$ نقط، مرسومة كبيرة وبتباعد غير منتظم، مع العدّ
        بعناية شديدة (فالمناطق الصغيرة كثيرة). فكم منطقة
        تجد --- وماذا حلّ
        بالتخمين؟
  \item والدرس، في جملتين: ما الذي كان يلزم
        \emph{للبرهنة} على تخمين المضاعفة، ولماذا لم تكن صيغ
        أعواد الثقاب في الجزأين الأول والثاني في مثل هذا
        الخطر قطّ؟ واِحتفل بتوقّع مبرهن عليه: كم عودًا
        يحتاج صفّ $2\,026$ مربعًا؟
\end{enumerate}
\end{problem}
