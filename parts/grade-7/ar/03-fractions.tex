\chapter{الكسور: المقارنة والجمع}\label{ch:g7:fractions}

قدّمت السنة السادسة الكسور حصصًا وخوارجَ قسمة مضبوطة
(\cref{ch:g6:fractions}). ويتعلّم هذا الفصل \emph{الحساب} بها:
التعرّف على الكسور المتساوية، والمقارنة، والجمع والطرح
--- بمقامات ودّية في الوقت الحاضر؛ وأما الحالة العامّة فتأتي في
\cref{ch:g9:fractions}.

\section{الكسور المتساوية}

\begin{theorem}[الكسور المتساوية]\label{thm:g7:fractions:equal}
لا يتغيّر \omterm{def:g6:fractions:def}{الكسر} حين يُضرب بسطه ومقامه، أو يُقسمان،
على العدد نفسه غير المعدوم:
\[
\frac{a}{b} = \frac{a \times k}{b \times k} .
\]
\end{theorem}
\admitted

\begin{example}\label{ex:g7:fractions:equal}
$\dfrac{3}{4} = \dfrac{3 \times 5}{4 \times 5} = \dfrac{15}{20}$، وفي
الاتجاه الآخر $\dfrac{42}{30} = \dfrac{42 \div 6}{30 \div 6}
= \dfrac{7}{5}$ (مختزلًا على $6$).

ولاختبار تساوي $\dfrac{8}{12}$ و $\dfrac{10}{15}$،
اِختزلهما معًا: $\dfrac{8}{12} = \dfrac{2}{3}$ و
$\dfrac{10}{15} = \dfrac{2}{3}$ --- نعم، متساويان.
\end{example}

\section{مقارنة الكسور}

\begin{method}[مقارنة كسرين]\label{met:g7:fractions:compare}
\begin{enumerate}
  \item \emph{\omterm{def:g4:fractions:def}{المقام} نفسه}: قارن البسطين:
        $\frac{5}{9} < \frac{7}{9}$.
  \item \emph{أحد المقامين \omterm{def:g5:division:multiple}{مضاعف} للآخر}: أعِد كتابة
        \omterm{def:g6:fractions:def}{الكسر} الأخشن، ثم قارن:
        $\frac{5}{6}$ مقابل $\frac{7}{12}$: $\frac56 = \frac{10}{12} >
        \frac{7}{12}$.
  \item \emph{المقارنة بالعدد $1$ أو $\frac12$} حين يمكن:
        فالمتراجحة $\frac{9}{8} > 1 > \frac{7}{9}$ تحسم
        $\frac98$ مقابل $\frac79$ في الحال.
\end{enumerate}
\end{method}

\section{الجمع والطرح}

\begin{theorem}[المقام نفسه]\label{thm:g7:fractions:add}
الكسور ذات \omterm{def:g4:fractions:def}{المقام} نفسه تُجمع (أو تُطرح) بجمع
(أو طرح) البسوط:
\[
\frac{a}{d} + \frac{b}{d} = \frac{a + b}{d},
\qquad
\frac{a}{d} - \frac{b}{d} = \frac{a - b}{d} .
\]
\end{theorem}

\begin{proof}[لماذا]
$a$ جزءًا \omterm{def:g6:measure:volume}{حجم} كل منها $\frac1d$ زائد $b$ جزءًا من \omterm{def:g6:measure:volume}{الحجم} نفسه تعطي
$a + b$ جزءًا من ذلك \omterm{def:g6:measure:volume}{الحجم}.
\end{proof}

\begin{method}[مقامان مختلفان]\label{met:g7:fractions:add}
حين يكون أحد المقامين مضاعفًا للآخر:
\begin{enumerate}
  \item أعِد كتابة \omterm{def:g6:fractions:def}{الكسر} ذي \omterm{def:g4:fractions:def}{المقام} الأصغر بحيث يصير للاثنين
        \omterm{def:g4:fractions:def}{المقام} الأكبر (\cref{thm:g7:fractions:equal})؛
  \item واجمع البسطين أو اِطرحهما؛
  \item واِختزل النتيجة إن أمكن.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:fractions:add}
اِحسب $\dfrac{3}{4} + \dfrac{5}{8}$، خطوة خطوة:
\begin{enumerate}
  \item العدد $8$ \omterm{def:g5:division:multiple}{مضاعف} للعدد $4$: فأعِد الكتابة
        $\dfrac34 = \dfrac{3 \times 2}{4 \times 2} = \dfrac{6}{8}$؛
  \item ثم اِجمع: $\dfrac{6}{8} + \dfrac{5}{8} = \dfrac{11}{8}$؛
  \item ولا شيء يُختزل: فالنتيجة $\dfrac{11}{8}$، أي\
        $1 + \dfrac38$.
\end{enumerate}
ومثال آخر: $2 - \dfrac{4}{5} = \dfrac{10}{5} - \dfrac{4}{5} =
\dfrac{6}{5}$ (اُكتب العدد الصحيح على \omterm{def:g4:fractions:def}{المقام} $5$
أولًا).
\end{example}

\begin{example}[فخّ كلاسيكي]\label{ex:g7:fractions:trap}
$\dfrac12 + \dfrac13$ \emph{ليس} $\dfrac{2}{5}$! فجمع البسوط
والمقامات كلًّا على حدة خطأ --- إذ إن الجواب $\frac25$ سيكون
أصغر من $\frac12$، وهو عبث حين \emph{نجمع} شيئًا موجبًا
إلى $\frac12$. (\omterm{def:g1:addition:def}{والمجموع} الصحيح، $\frac56$، يحتاج مقامًا مشتركًا
هو $6$؛ والطريقة العامّة في \cref{ch:g9:fractions}.)
\end{example}

\section{ضرب كسر في عدد}

\begin{proposition}[كسر مضروب في عدد]\label{prop:g7:fractions:mult}
من أجل عدد $k$:
\[
k \times \frac{a}{b} = \frac{k \times a}{b} .
\]
وأخذ $\frac ab$ \emph{من} مقدار يعني الضرب في
$\frac ab$.
\end{proposition}
\admitted

\begin{example}\label{ex:g7:fractions:mult}
$6 \times \dfrac{2}{3} = \dfrac{12}{3} = 4$: فستّ مرات ثلثين تساوي
أربع وحدات. وعبارة «$\frac34$ من $60$ يورو» هي
$\frac34 \times 60 = \frac{180}{4} = 45$ يورو --- وهو الجواب نفسه الذي تعطيه
القسمة على $4$ ثم الضرب في $3$
(\cref{met:g6:fractions:ofquantity}).
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g7:fractions:1}
أكمل: $\dfrac{2}{5} = \dfrac{?}{20}$؛ \quad
$\dfrac{18}{24} = \dfrac{3}{?}$؛ \quad
$\dfrac{7}{3} = \dfrac{28}{?}$؛ \quad
$\dfrac{?}{9} = \dfrac{20}{36}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:2}
اِختزل إلى أبسط صورة ممكنة:
$\dfrac{12}{16}$؛ \quad $\dfrac{30}{45}$؛ \quad $\dfrac{27}{36}$؛
\quad $\dfrac{48}{12}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:3}
هل $\dfrac{15}{35}$ و $\dfrac{21}{49}$ متساويان؟ وماذا عن $\dfrac{16}{28}$
و $\dfrac{20}{36}$؟ بَرِّر بالاختزال.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:4}
قارن (اُكتب الاستدلال، لا الجواب وحده):
\[
\frac{7}{11} \text{ و } \frac{9}{11}; \qquad
\frac{5}{6} \text{ و } \frac{11}{18}; \qquad
\frac{13}{12} \text{ و } \frac{19}{20}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:5}
اِحسب واِختزل إن أمكن:
\[
\frac{5}{9} + \frac{2}{9}, \qquad
\frac{11}{7} - \frac{4}{7}, \qquad
\frac{5}{8} + \frac{7}{8} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:6}
اِحسب مستعملًا \cref{met:g7:fractions:add}:
\[
\frac{2}{3} + \frac{5}{6}, \qquad
\frac{7}{10} - \frac{2}{5}, \qquad
\frac{3}{4} + \frac{5}{12} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:7}
اِحسب:
\[
3 - \frac{5}{4}, \qquad
1 + \frac{3}{8}, \qquad
2 - \frac{7}{6} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:fractions:8}
اِحسب:
\[
5 \times \frac{3}{10}, \qquad
\frac{7}{8} \times 4, \qquad
\frac{2}{3} \text{ من } 45 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:fractions:9}
أكل مالك $\frac14$ من بيتزا وأكلت مريم $\frac38$ من البيتزا نفسها.
فأيّ \omterm{def:g6:fractions:def}{كسر} من البيتزا أكلا معًا؟ وأيّ \omterm{def:g6:fractions:def}{كسر}
بقي؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:fractions:10}
قارورة فيها $\frac34$ لتر من العصير. تسكب لينا منها $\frac16$ لتر
مرتين. فكم عصيرًا يبقى؟ (\omterm{def:g4:fractions:def}{المقام} المشترك: $12$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:fractions:11}
اِشرح، بحجّة \cref{ex:g7:fractions:trap}، لماذا لا يمكن أن يكون
$\frac35 + \frac12$ مساويًا $\frac{4}{7}$، دون حساب
\omterm{def:g1:addition:def}{المجموع} الصحيح.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:fractions:12}
اِحسب \omterm{def:g1:addition:def}{المجموع}
\[
\frac12 + \frac14 + \frac18 + \frac{1}{16}
\]
خطوة خطوة (من اليسار إلى اليمين). ولاحِظ نمط المجاميع الجزئية:
فأيّ عدد تقترب منه كلما أُضيفت حدود أكثر
$\frac{1}{32}, \frac{1}{64}, \dots$؟
\end{exercise}

\section{مسألة: الكسور المختبئة في ميزان موسيقي}

\begin{problem}[{مسألة نهاية الأسبوع --- قيم العلامات كسور
يجب أن تتجمّع بالضبط، وعدّ الإيقاع يخفي متتالية مشهورة،
والزوجية تحكم إيقاع البلقان}]
\label{pb:g7:fractions:1}
تُكتب الموسيقى بالكسور. فالعلامة الكاملة تدوم $1$
ميزانًا (في الميزان الشائع «أربعة أرباع»)؛ وعلامة النصف
تدوم $\frac12$؛ ثم تأتي علامة الربع $\frac14$،
وعلامة الثمن $\frac18$ وعلامة الجزء من ستة عشر
$\frac{1}{16}$. وقانون واحد لا يُخرق: \emph{\omterm{def:g1:addition:def}{مجموع} المدد داخل
الميزان يجب أن يساوي بالضبط \omterm{def:g1:addition:def}{مجموع} الميزان} --- أي $1$ في
ميزان أربعة أرباع. وكل سؤال أدناه هو حساب هذا الفصل
(\cref{thm:g7:fractions:add}، \cref{met:g7:fractions:add}) موضوعًا
على الموسيقى؛ وستلتقي في الطريق متتالية عدّ اِكتشفها
علماء الهند قبل ألف سنة.

\medskip
\noindent\textbf{الجزء الأول --- علامات يجب أن تتجمّع.}
\begin{enumerate}
  \item تحقّق من أن كلًّا من هذه الموازين مشروع في ميزان
        أربعة أرباع: (أ) نصف $+$ ربع $+$ ربع؛ (ب) ربع $+$
        ثمن $+$ ثمن $+$ نصف.
  \item كتب ناسخ: نصف $+$ ربع $+$ ثمن. فأيّ
        \omterm{def:g6:fractions:def}{كسر} من الميزان كُتب، وأيّ علامة واحدة
        تكمله؟
  \item كم علامة جزء من ستة عشر تملأ ميزانًا كاملًا؟ وكم
        جزءًا من ستة عشر يصنع علامة ثمن واحدة؟
  \item \emph{النقطة} بعد العلامة تضيف نصف قيمتها: فالنصف
        المنقوط هو $\frac12 + \frac14$، والربع المنقوط هو
        $\frac14 + \frac18$. اِحسب القيمتين.
  \item تُكتب الفالس في ميزان ثلاثة أرباع: فيجب أن يبلغ كل ميزان
        $\frac34$. تحقّق من أن نصفًا منقوطًا واحدًا يملأ
        ميزان فالس، واُكتب ميزانَي فالس مشروعين آخرين
        لا تستعمل فيهما إلا الأرباع والأثمان.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- ورشة الناسخ.}
\begin{enumerate}[resume]
  \item ميزان أربعة أرباع يحوي: ربعًا منقوطًا $+$ ثمنًا
        $+$ ربعًا. فما الناقص؟
  \item أيّهما أطول، الثمن المنقوط أم علامة الربع؟
        قارن بالأجزاء من ستة عشر.
  \item \emph{الرباط} يلصق المدد معًا في صوت واحد
        طويل: فكم تدوم علامة نصف مربوطة بثمن؟ وكم
        جزءًا من ستة عشر يساوي ذلك؟
  \item \omterm{def:g1:addition:def}{مجموع} علامات ميزان هو $\frac{11}{16}$؛
        \omterm{def:g3:division:remainder}{والباقي} صمت (أي \emph{سكتة}). فكم تدوم
        السكتة؟
  \item يملأ قارع الطبل علامة ربع واحدة ($\frac{4}{16}$ من
        الميزان) مستعملًا الأثمان ($\frac{2}{16}$)
        والأجزاء من ستة عشر ($\frac{1}{16}$) فقط، مع أهمية الترتيب: فمثلًا
        ثمن--جزء--جزء، أو
        جزء--ثمن--جزء. تحقّق من أن المثالين
        مشروعان، ثم اُذكر \emph{كل} الإيقاعات الممكنة.
        فكم عددها؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- متتالية عمرها ألف سنة،
وميزان فردي.}
\begin{enumerate}[resume]
  \item لنعُدّ إيقاعات قارع الطبل من أجل علامات أطول.
        فكل إيقاع ينتهي إما بجزء من ستة عشر وإما
        بثمن؛ اِشرح لماذا يعطي هذا القاعدة: (الإيقاعات
        التي تملأ $n$ جزءًا) $=$ (الإيقاعات التي تملأ $n - 1$)
        $+$ (الإيقاعات التي تملأ $n - 2$). واِنطلاقًا من إيقاع $1$
        لجزء واحد و $2$ لجزأين، اِبنِ الجدول إلى
        $n = 8$: فكم إيقاعًا يملأ علامة نصف؟
  \item والأعداد التي وجدتها --- $1, 2, 3, 5, 8, 13, 21, 34$
        --- نشرها العالِم الهندي هيماشندرا
        نحو سنة 1150، عادًّا هذه الإيقاعات بالضبط، قبل
        جيلين من تدوين فيبوناتشي لها في إيطاليا.
        اُذكر نمطها المميِّز في جملة واحدة، وتحقّق
        منه على المداخل الثلاثة الأخيرة من جدولك.
  \item يستبدل أسلوب التأرجح بثمنين متساويين زوجًا
        طويلًا--قصيرًا: $\frac16$ ثم $\frac{1}{12}$. تحقّق
        من أن التبديل مشروع (المدّة الكلية نفسها)، مستعملًا
        \omterm{def:g4:fractions:def}{المقام} المشترك $12$.
  \item \emph{الثلاثية} تحشر ثلاث علامات متساوية في علامة
        ربع واحدة. فكم تدوم كل علامة (تحقّق من جوابك
        بضربه في $3$،
        \cref{prop:g7:fractions:mult})؟ والسؤال نفسه من أجل ثلاث
        علامات متساوية تملأ علامة نصف.
  \item تستعمل موسيقى رقص البلقان ميزان سبعة أثمان: فيبلغ كل ميزان
        $\frac78$. تحقّق من أن ربعًا منقوطًا $+$ ربعًا
        $+$ ربعًا يملؤه. ثم اِشرح لماذا \emph{لا}
        يمكن لأيّ تركيب من علامات النصف وعلامات الربع وحدها،
        مهما كثُر، أن يملأ ميزان سبعة أثمان.
        (عُدّ بالأثمان وفكّر في الزوجي والفردي.)
\end{enumerate}
\end{problem}
