\chapter{التماثل المركزي ومتوازيات الأضلاع}\label{ch:g7:central}

أدِر شكلًا نصف دورة حول نقطة: هذا هو \emph{\omterm{def:g7:central:def}{التماثل
المركزي}}، وهو ثاني تحويل في الكتاب بعد اِنعكاسات
\cref{ch:g6:symmetry}. وشكله النجم هو \omterm{def:g7:central:parallelogram}{متوازي الأضلاع} ---
أي \omterm{def:g4:geometry:polygon}{الرباعي} المتماثل بالنسبة إلى نقطة تقاطع
قطريه.

\section{التماثل المركزي}

\begin{definition}[التماثل بالنسبة إلى نقطة]\label{def:g7:central:def}
\emph{مماثلة}\index{تماثل مركزي} النقطة $M$ بالنسبة إلى
النقطة $O$ هي النقطة $M'$ التي تكون فيها $O$ \emph{منتصف}
\omterm{def:g6:lines:objects}{القطعة} $[MM']$. والشكل المماثل هو ما يصير إليه الأصل بعد
نصف دورة ($180^\circ$) حول $O$.
\end{definition}

\begin{method}[إنشاء مماثلة نقطة]\label{met:g7:central:construct}
لإنشاء \omterm{def:g7:central:def}{مماثلة} $M'$ للنقطة $M$ بالنسبة إلى $O$:
\begin{enumerate}
  \item اُرسم \omterm{def:g6:lines:objects}{المستقيم} $(MO)$ ومدّه إلى ما بعد $O$؛
  \item وقِس $MO$ (بالمسطرة أو الفرجار)؛
  \item وضع $M'$ على الامتداد حيث $OM' = OM$.
\end{enumerate}
وعلى الورق المربّعات: عُدّ الخطوات الأفقية والعمودية من $M$ إلى
$O$، وكرّر الخطوات \emph{نفسها} من $O$ لتبلغ $M'$.
\end{method}

\begin{example}[على الشبكة]\label{ex:g7:central:grid}
إذا كانت $M$ على بُعد $3$ مربعات يسارًا ومربع $1$ أعلى من $O$، فمماثلتها
$M'$ على بُعد $3$ مربعات \emph{يمينًا} ومربع $1$ \emph{أسفل} من $O$:
\omterm{def:g7:central:def}{فالتماثل المركزي} يعكس الاتجاهين معًا (أما الانعكاس
فيعكس واحدًا فقط).
\end{example}

\begin{proposition}[ما يحفظه نصف الدورة]\label{prop:g7:central:preserved}
\omterm{def:g7:central:def}{التماثل المركزي} يحفظ الأطوال والزوايا والمحيطات والمساحات؛
وصورة \omterm{def:g6:lines:objects}{المستقيم} \omterm{def:g6:lines:objects}{مستقيم} \emph{\omterm{def:g6:lines:perp}{موازٍ}} له، وصورة \omterm{def:g6:lines:objects}{القطعة} \omterm{def:g6:lines:objects}{قطعة}
موازية بالطول نفسه، وصورة الدائرة دائرة
\omterm{def:g3:shapes:shapes}{بنصف القطر} نفسه (مركزها \omterm{def:g7:central:def}{مماثلة} المركز).
\end{proposition}
\admitted

\begin{remark}
قارن بالانعكاسات (\cref{prop:g6:symmetry:preserved}): فكلاهما
يحفظ الأطوال والزوايا؛ لكن الانعكاس يقلب الأشكال
(فحرف «b» يصير «d»)، بينما نصف الدورة يبقيها مقروءة ---
مقلوبةً رأسًا على عقب (فحرف «b» يصير «q»). وخلافًا للانعكاس، يرسل
نصف الدورة كل \omterm{def:g6:lines:objects}{مستقيم} إلى \omterm{def:g6:lines:objects}{مستقيم} \omterm{def:g6:lines:perp}{موازٍ} له.
\end{remark}

\begin{definition}[مركز التماثل]\label{def:g7:central:center}
تكون النقطة $O$ \emph{مركز تماثل}\index{مركز تماثل}
لشكل حين يرسل نصف الدورة حول $O$ الشكلَ بالضبط
على نفسه. فمثلًا، للحروف S و N و Z مركز
تماثل؛ وللدائرة واحد (مركزها)؛ وليس للمثلث مركز أبدًا.
\end{definition}

\section{متوازيات الأضلاع}

\begin{definition}[متوازي الأضلاع]\label{def:g7:central:parallelogram}
\emph{متوازي الأضلاع}\index{متوازي أضلاع} \omterm{def:g4:geometry:polygon}{رباعي}
يتقاطع قطراه في منتصفهما المشترك. وتكون نقطة التقاطع تلك حينئذٍ
\omterm{def:g7:central:center}{مركز تماثل} للشكل: فكل رأس هو صورة الرأس
المقابل بنصف الدورة.
\end{definition}

\begin{theorem}[خاصّيات متوازي الأضلاع]\label{thm:g7:central:properties}
في \omterm{def:g7:central:parallelogram}{متوازي أضلاع} $ABCD$ مركزه $O$:
\begin{enumerate}
  \item الأضلاع المتقابلة متوازية: $(AB) \parallel (CD)$ و
        $(BC) \parallel (AD)$؛
  \item والأضلاع المتقابلة لها الطول نفسه: $AB = CD$ و
        $BC = AD$؛
  \item والزوايا المتقابلة متساوية: $\widehat A = \widehat C$ و
        $\widehat B = \widehat D$.
\end{enumerate}
\end{theorem}

\begin{proof}
نصف الدورة حول $O$ يرسل $A$ إلى $C$ و $B$ إلى $D$ (فبالتعريف
$O$ منتصف القطرين معًا). إذن يرسل \omterm{def:g6:lines:objects}{القطعة}
$[AB]$ إلى \omterm{def:g6:lines:objects}{القطعة} $[CD]$؛ وحسب
\cref{prop:g7:central:preserved} تكون الصورة موازية \omterm{def:g6:lines:objects}{للقطعة} $[AB]$
وبالطول نفسه: وهما النقطتان 1 و2. ويرسل أيضًا الزاوية عند $A$
إلى الزاوية عند $C$، والزوايا محفوظة: وهي النقطة 3.
\end{proof}

\begin{theorem}[التعرّف على متوازي الأضلاع]\label{thm:g7:central:recognize}
يكون \omterm{def:g4:geometry:polygon}{الرباعي} (الذي لا تتقاطع أضلاعه) \omterm{def:g7:central:parallelogram}{متوازي أضلاع} بمجرد أن
يتحقّق \emph{أحد} ما يلي:
\begin{enumerate}
  \item لقطريه المنتصف نفسه (وهو التعريف)؛
  \item وأضلاعه المتقابلة متوازية مثنى مثنى؛
  \item وضلعان متقابلان متوازيان \emph{و} لهما الطول نفسه.
\end{enumerate}
\end{theorem}
\admitted

\begin{example}\label{ex:g7:central:recognize}
في $ABCD$ لدينا $(AB) \parallel (CD)$ و $AB = CD = 5$ سم. فينطبق المحكّ 3:
إذن $ABCD$ \omterm{def:g7:central:parallelogram}{متوازي أضلاع} --- ومنه نعرف أيضًا دون أيّ
قياس آخر أن $AD = BC$ و $(AD) \parallel (BC)$، وأن
قطريه يقطع كل منهما الآخر في منتصفه.
\end{example}

\begin{remark}[متوازيات أضلاع خاصّة]
المستطيل \omterm{def:g6:shapes:quadrilaterals}{والمعيّن} والمربع في \cref{ch:g6:shapes} هي بالضبط
متوازيات الأضلاع ذات خاصّية زائدة: قطران متساويان
(المستطيل)، وقطران متعامدان (\omterm{def:g6:shapes:quadrilaterals}{المعيّن})، والاثنان معًا (المربع). والآن
تعمل حجّة الزوايا الموعودة في \cref{ex:g6:shapes:recognize}:
ففي \omterm{def:g7:central:parallelogram}{متوازي الأضلاع} تكون الزاويتان المتتاليتان متكاملتين
(\omterm{def:g7:triangles:pairs}{زوايا متبادلة} أو متناظرة مع \omterm{def:g6:lines:perp}{المتوازيين}،
\cref{thm:g7:triangles:equalities})، إذن \omterm{def:g3:shapes:rightangle}{زاوية قائمة} واحدة تفرض
الأربع.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g7:central:1}
على الورق المربّعات، ضع $O$، ثم $M$ على بُعد أربعة مربعات يمين $O$ ومربع
واحد أعلى. واِبنِ \omterm{def:g7:central:def}{مماثلة} $M'$ للنقطة $M$ بالنسبة إلى $O$، \omterm{def:g7:central:def}{ومماثلة}
$M$ بالنسبة إلى \emph{الخط العمودي من الشبكة} المارّ بالنقطة $O$.
وقارن الصورتين.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:2}
اُرسم مثلثًا $ABC$ ونقطة $O$ خارجه. واِبنِ صورته
$A'B'C'$ بنصف الدورة حول $O$. فماذا تقول عن
الطولين $A'B'$ و $AB$؟ وعن \omterm{def:g6:lines:objects}{المستقيمين} $(A'B')$ و $(AB)$؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:3}
أيّ الأرقام، مكتوبة كما على شاشة آلة حاسبة (من $0$ إلى $9$)، له
\omterm{def:g7:central:center}{مركز تماثل}؟ وأيّ الحروف الكبيرة من بين H و A و N و S و T و Z و O؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:4}
$KLMN$ \omterm{def:g7:central:parallelogram}{متوازي أضلاع} مركزه $O$ حيث $KL = 7$ سم،
و $LM = 4$ سم و $\widehat K = 115^\circ$. أعطِ، مع الأسباب: $MN$
و $NK$ و $\widehat M$، ومنتصف $[LN]$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:5}
اِبنِ \omterm{def:g7:central:parallelogram}{متوازي أضلاع} $ABCD$ من قطريه: $AC = 8$ سم و
$BD = 5$ سم، متقاطعين في منتصفيهما بزاوية $60^\circ$
بينهما. (اُرسم القطرين أولًا!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:6}
في \omterm{def:g7:central:parallelogram}{متوازي أضلاع}، قياس إحدى الزوايا $115^\circ$. وباستعمال الملاحظة
عن الزاويتين المتتاليتين المتكاملتين، أعطِ الزوايا الثلاث الأخرى.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:7}
ضع النقط $A(1, 1)$ و $B(4, 2)$ والنقطة $O(2.5, 2.5)$. واِحسب
إحداثيَي الصورتين $A'$ و $B'$ بنصف الدورة حول $O$
(اِستعمل عدّ الشبكة في \cref{ex:g7:central:grid}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:8}
$ABCD$ \omterm{def:g4:geometry:polygon}{رباعي} فيه $AB = CD$. فهل يكفي ذلك ليكون
\omterm{def:g7:central:parallelogram}{متوازي أضلاع}؟ وإن لم يكفِ، فاِرسم مثالًا مضادًّا (وشبه المنحرف
\omterm{def:g6:shapes:triangles}{المتساوي الساقين} يساعد)، واُذكر ما يجب إضافته إلى الفرض.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:9}
ليكن $ABC$ مثلثًا و $O$ منتصف $[BC]$. اِبنِ
\omterm{def:g7:central:def}{مماثلة} $A'$ للنقطة $A$ بالنسبة إلى $O$. وبيّن أن $ABA'C$
\omterm{def:g7:central:parallelogram}{متوازي أضلاع}. (فأيّ محكّ من \cref{thm:g7:central:recognize}
مجّاني هنا؟)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:10}
صواب أم خطأ، مع سبب: «\omterm{def:g4:geometry:polygon}{الرباعي} الذي زواياه المتقابلة
متساوية مثنى مثنى \omterm{def:g7:central:parallelogram}{متوازي أضلاع}»؛ و«\omterm{def:g7:central:parallelogram}{متوازي الأضلاع} ذو
القطرين \omterm{def:g6:lines:perp}{المتعامدين} \omterm{def:g6:shapes:quadrilaterals}{معيّن}»؛ و«\omterm{def:g7:central:parallelogram}{متوازي الأضلاع} ذو \omterm{def:g3:shapes:rightangle}{الزاوية
القائمة} مستطيل».
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:central:11}
ليكن $ABCD$ رباعيًا كيفما كان و $I$ و $J$ و $K$ و $L$ منتصفات
أضلاعه $[AB]$ و $[BC]$ و $[CD]$ و $[DA]$. اُرسم أمثلة عدّة.
فماذا يبدو $IJKL$ دائمًا؟ (والبرهان، بمبرهنة المنتصفين،
يأتي في \cref{ch:g8:midpoints} --- ومرة أخرى بالمتجهات،
في مجلّد المرحلة الثانوية.)
\end{exercise}

\section{مسألة: الكعكة، والمتوسط، والمركز الثاني
المستحيل}

\begin{problem}[{مسألة نهاية الأسبوع --- نصف الدورة في العمل: قطع
الكعكة العادل، وحدّ للمتوسطات، ولماذا لا يمكن لأيّ شكل أن يكون له
مركزا تماثل}]
\label{pb:g7:central:1}
يبدو نصف الدورة أبسط التحويلات --- ومع ذلك
يقطع الكعك إلى نصفين عادلين بالبرهان، ويقيس متوسطات
مثلث، ويخفي لؤلؤة صغيرة من الرياضيات الخالصة: فالشكل
المحدود يمكن أن يكون له \omterm{def:g7:central:center}{مركز تماثل} واحد، لكن \emph{لا مركزان
أبدًا}. ويستعمل البرهان اِكتشافًا ستصنعه في الجزء الأول: أن
نصفَي دورة، ينجزان واحدًا بعد الآخر، يبلغان اِنزلاقًا.

\medskip
\noindent\textbf{الجزء الأول --- رياضة نصف الدورة.}
\begin{enumerate}
  \item على الورق المربّعات، خذ المبدأ $O(0, 0)$ مركزًا.
        وباستعمال عدّ الشبكة في \cref{ex:g7:central:grid}،
        أعطِ صور $M(3, 1)$ و $N(-2, 4)$ و
        $P(0, -5)$ بنصف الدورة حول $O$. فما
        القاعدة العامّة من أجل $(x, y)$؟
  \item والآن المركز هو $C(2, 1)$. اِحسب صورة
        $M(5, 3)$، وتحقّق من الوصفة العامّة: كل إحداثي
        للصورة هو \emph{ضعف إحداثي المركز ناقص إحداثي
        النقطة}.
  \item تُصمَّم ورق اللعب لتُقرأ كما هي مقلوبةً رأسًا على عقب
        --- أي \omterm{def:g7:central:def}{تماثل مركزي}، فلا يرى أيّ لاعب الورقة
        «بالمقلوب». اِشرح لماذا يجب، في ورقة عدد نقطها
        \emph{فردي}، أن تقع نقطة واحدة بالضبط في مركز
        الورقة.
  \item نصفا دورة متتاليان: خذ المركزين $O_1(2, 0)$
        و $O_2(5, 0)$. أرسِل النقطة $A(1, 1)$ بنصف
        الدورة حول $O_1$، ثم أرسِل الصورة بنصف
        الدورة حول $O_2$. وقارن البداية بالنهاية. وكرّر
        مع $B(0, 3)$. فأيّ تحويل واحد بسيط بلغه
        نصفا الدورة؟
  \item قِس اِنزلاق السؤال 4 بالنسبة إلى المسافة
        $O_1 O_2$، واُذكر الاكتشاف. (قارن مع
        المرآتين المتوازيتين في \cref{pb:g6:symmetry:1}: الظاهرة نفسها،
        وممثّلون جدد.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الكعك والمتوسطات.}
\begin{enumerate}[resume]
  \item ليكن $O$ مركز \omterm{def:g7:central:parallelogram}{متوازي أضلاع}. اِشرح لماذا
        يلتقي أيّ \omterm{def:g6:lines:objects}{مستقيم} يمرّ بالنقطة $O$ الحدَّ في نقطتين
        متماثلتين بالنسبة إلى $O$، ولماذا يقطع \omterm{def:g6:lines:objects}{المستقيم}
        \omterm{def:g7:central:parallelogram}{متوازي الأضلاع} إلى \omterm{def:g6:lines:objects}{قطعتين} كل منهما نسخة مطابقة
        للأخرى بنصف دورة --- ومنه متساويتا \omterm{def:g6:measure:area}{المساحة}.
  \item مبرهنة الخبّاز: كعكة مستطيلة تُتقاسم
        بعدل تامّ بقطع \omterm{def:g6:lines:objects}{مستقيم} \emph{أيّ} كان يمرّ
        بمركزها. وأفضل من ذلك: كعكة مستطيلة فيها ثقب
        مستطيل (بأيّ \omterm{def:g6:measure:volume}{حجم} وفي أيّ موضع، ومائل أيضًا). صِف
        \omterm{def:g6:lines:objects}{القطع} \omterm{def:g6:lines:objects}{المستقيم} الوحيد الذي يعطي جزأين فيهما من
        الكعكة قدر متساوٍ --- وبَرِّره بالسؤال 6.
  \item ثلاثة \omterm{def:g5:solids:def}{رؤوس} من \omterm{def:g7:central:parallelogram}{متوازي أضلاع} $ABCD$ هي $A(0, 0)$
        و $B(6, 1)$ و $C(8, 5)$. اِحسب إحداثيَي $D$
        وإحداثيَي المركز $O$.
  \item بنى \cref{exo:g7:central:9}، من مثلث $ABC$
        ومنتصف $O$ \omterm{def:g6:lines:objects}{للقطعة} $[BC]$، \omterm{def:g7:central:parallelogram}{متوازي الأضلاع}
        $ABA'C$ حيث $A'$ \omterm{def:g7:central:def}{مماثلة} $A$ بالنسبة إلى $O$.
        اِستعمله، مع المتراجحة المثلثية
        (\cref{thm:g7:triangles:inequality}) في المثلث
        $ABA'$، لتبرهن على \emph{حدّ المتوسط}: فالمتوسط
        $[AO]$ يحقّق
        \[
        AO < \frac{AB + AC}{2} .
        \]
        (ولاحِظ أن $AA' = 2\,AO$ و $BA' = AC$.)
  \item في مثلث فيه $AB = 5$ سم و $AC = 7$ سم، بين أيّ
        قيمتين يجب أن يقع طول المتوسط الصادر من $A$؟
        (اِستعمل اِتجاهَي المتراجحة المثلثية معًا في
        $ABA'$.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- كم مركزًا يمكن أن يكون
لشكل؟}
\begin{enumerate}[resume]
  \item من أجل كل شكل، قل هل له \omterm{def:g7:central:center}{مركز
        تماثل}، وأين: \omterm{def:g6:lines:objects}{قطعة}؛ \omterm{def:g6:lines:objects}{ومستقيم} كامل؛ \omterm{def:g6:shapes:triangles}{ومثلث
        متساوي الأضلاع}؛ ومربع؛ ودائرة؛ والحرفان S
        و Z.
  \item بَرهِن على اِدّعائك في \omterm{def:g6:shapes:triangles}{المثلث المتساوي الأضلاع}: فنصف
        دورة يرسل المثلث إلى نفسه يزاوج بين
        \omterm{def:g5:solids:def}{الرؤوس} الثلاثة --- لكن الثلاثة \omterm{def:g3:numbers:evenodd}{عدد فردي}، فيُرسل رأس
        ما إلى نفسه. فماذا يفرض ذلك على
        الرأس، ولماذا يستحيل؟
  \item عمّم السؤال 12: اِشرح لماذا \emph{لا} يوجد \omterm{def:g4:geometry:polygon}{مضلع}
        ذو \omterm{def:g3:numbers:evenodd}{عدد فردي} من \omterm{def:g5:solids:def}{الرؤوس} له \omterm{def:g7:central:center}{مركز
        تماثل}.
  \item اللؤلؤة: اِفترض أن لشكل مركزَي تماثل
        \emph{متمايزين} $O_1$ و $O_2$. أنجز نصفَي
        الدورة واحدًا بعد الآخر واِستعمل اِكتشاف
        السؤال 5: فأيّ حركة يجب أن ترسل الشكل بالضبط
        على نفسه؟ واِشرح لماذا لا يستطيع شكل يسع في ورقة
        أن يصمد لذلك، واِستنتج.
  \item وخاتمة السؤال 14 تترك ثغرة:
        الأشكال \emph{غير المحدودة}. تحقّق على إفريز لا نهائي
        --- ليكن شريطًا لا ينتهي من آثار الأقدام، يسرى، ويمنى،
        ويسرى، ويمنى\,\dots\ --- من أن له (على الأقلّ) مركزَي
        تماثل مختلفين، وعيّن الانسحاب الذي
        توقّعه السؤال 14.
\end{enumerate}
\end{problem}
