\chapter{التناسب}\label{ch:g7:prop}

اِلتقت السنة السادسة جداول \omterm{def:g7:prop:table}{التناسب} (\cref{ch:g6:propdata})؛ ويحوّل هذا الفصل
\omterm{def:g7:prop:table}{التناسب} أداةً كاملة النموّ: قاعدة الضرب المتقاطع لإيجاد
قيمة رابعة، والنسب المئوية، وسلالم الخرائط \omterm{def:g5:solids:def}{والمجسّمات} المصغّرة.
وأما السرعة، وهي أشهر التناسبات جميعًا، فتُدرس في
\cref{ch:g8:speed}.

\section{التعرّف على الجداول وإكمالها}

\begin{definition}[جدول التناسب]\label{def:g7:prop:table}
جدول من سطرين هو \emph{جدول
تناسب}\index{تناسب} حين تكون أعداد السطر الثاني
هي أعداد الأول مضروبةً في عدد ثابت واحد، وهو
\emph{المعامل}. وبعبارة مكافئة: كل خوارج قسمة الأعمدة
$\frac{\text{الأسفل}}{\text{الأعلى}}$ متساوية.
\end{definition}

\begin{example}\label{ex:g7:prop:recognize}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|ccc}
الأعلى & $4$ & $10$ & $14$ \\
\hline
الأسفل & $6$ & $15$ & $21$ \\
\end{tabular}
\end{center}
\omterm{def:g3:division:remainder}{خوارج القسمة}: $\frac64 = 1.5$، و $\frac{15}{10} = 1.5$،
و $\frac{21}{14} = 1.5$: إذن متناسب، ومعامله $1.5$.
\end{example}

\begin{theorem}[قاعدة الضرب المتقاطع]\label{thm:g7:prop:cross}
في \omterm{def:g7:prop:table}{جدول تناسب} عموداه $\begin{pmatrix} a \\ b
\end{pmatrix}$ و $\begin{pmatrix} c \\ d \end{pmatrix}$، يتساوى
الجداءان المتقاطعان:
\[
a \times d = b \times c .
\]
ومنه يمكن حساب القيمة الرابعة من الثلاث الأخرى:
$d = \dfrac{b \times c}{a}$.
\end{theorem}

\begin{proof}
ليكن $k$ المعامل: $b = ka$ و $d = kc$. إذن
\[
a \times d = a \times kc = k \times ac,
\qquad
b \times c = ka \times c = k \times ac :
\]
فكلا الجداءين المتقاطعين يساوي $k \times ac$. وبقسمة $ad = bc$ على $a$
نحصل على صيغة $d$.
\end{proof}

\begin{example}[الرابع المتناسب]\label{ex:g7:prop:fourth}
إذا كانت $7$ كتب متطابقة كتلتها $2.8$ كغ، فكم تبلغ \omterm{def:g2:measure:units}{كتلة} $12$ كتابًا
مثلها؟ الجدول وقاعدة الضرب المتقاطع:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cc}
الكتب & $7$ & $12$ \\
\hline
كغ & $2.8$ & $m$ \\
\end{tabular}
\end{center}
\[
7 \times m = 2.8 \times 12
\quad\Longrightarrow\quad
m = \frac{2.8 \times 12}{7} = \frac{33.6}{7} = 4.8 \text{ كغ}.
\]
والتحقّق بالوحدة: كتاب واحد كتلته $0.4$ كغ، واثنا عشر كتلتها $4.8$ كغ.
\end{example}

\section{النسب المئوية}

\begin{method}[تطبيق النسب المئوية وإيجادها]\label{met:g7:prop:percent}
\begin{enumerate}
  \item \emph{تطبيق} $t\,\%$: اِضرب في $\frac{t}{100}$
        (و $12\,\%$ من $350$ هي $0.12 \times 350 = 42$)؛
  \item \emph{وإيجاد} النسبة المئوية التي يمثّلها جزء: اِحسب
        $\frac{\text{الجزء}}{\text{الكلّ}}$ وأعِد كتابتها على \omterm{def:g4:fractions:def}{المقام} $100$
        (فالعدد $21$ من $60$: $\frac{21}{60} = \frac{35}{100} = 35\,\%$)؛
  \item واِسأل دائمًا: \emph{نسبة مئوية من ماذا؟} فالكلّ المرجعي
        مهمّ.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:prop:percent}
في مدرسة، $180$ من التلاميذ $400$ بنات. والنسبة المئوية:
\[
\frac{180}{400} = \frac{180 \div 4}{400 \div 4} = \frac{45}{100}
= 45\,\% .
\]
وفي الاتجاه الآخر: كم يبلغ الأولاد $55\,\%$؟
$0.55 \times 400 = 220$. والتحقّق: $180 + 220 = 400$.
\end{example}

\section{السلالم}

\begin{definition}[السلّم]\label{def:g7:prop:scale}
على خريطة أو \omterm{def:g5:solids:def}{مجسّم} مصغّر ذي \emph{سلّم}\index{سلم}
$\frac{1}{n}$ (يُكتب $1 : n$)، تكون المسافات على الخريطة
متناسبة مع المسافات الحقيقية، بمعامل $\frac1n$:
\[
\text{المسافة على الخريطة} = \frac{1}{n} \times \text{المسافة الحقيقية},
\]
وكلتاهما مقيستان \emph{بالوحدة نفسها}.
\end{definition}

\begin{example}\label{ex:g7:prop:scale}
على خريطة $1 : 50\,000$، تبعد مدينتان $7$ سم إحداهما عن الأخرى. والمسافة الحقيقية:
\[
7 \times 50\,000 = 350\,000 \text{ سم} = 3\,500 \text{ م} = 3.5
\text{ كم}.
\]
وبالعكس، مسار مشي طوله $12$ كم يقيس على الخريطة
$12$ كم $= 1\,200\,000$ سم، إذن $1\,200\,000 \div 50\,000 = 24$ سم.
\end{example}

\begin{remark}
المنحنيات أسرع اِختبار \omterm{def:g7:prop:table}{للتناسب}
(\cref{ch:g6:propdata}): فالنقط الواقعة على \omterm{def:g6:lines:objects}{مستقيم} \emph{يمرّ
بالمبدأ} تعني مقادير متناسبة؛ \omterm{def:g6:lines:objects}{والمستقيم} الذي لا يمرّ بالمبدأ (مثل
أجرة سيارة أجرة فيها مبلغ ثابت) يعني \emph{عدم} \omterm{def:g7:prop:table}{التناسب}، وإن
كان مستقيمًا. والدوال التآلفية، في
\cref{ch:g9:linfunc}، ستدقّق هذا.
\end{remark}

 & $3$ & $7$ & $11$ \\

 & $7.5$ & $17.5$ & $27.5$ \\

\qquad
\begin{tabular}{l|ccc}
 & $2$ & $5$ & $9$ \\

\begin{exercise}[$\star$]\label{exo:g7:prop:2}
أكمل مستعملًا قاعدة الضرب المتقاطع، مع كتابة الحساب:
$5$ كغ من البطاطا ثمنها $6$ يورو؛ فكم ثمن $8$ كغ؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:prop:3}
طابعة تطبع $36$ صفحة في $3$ دقائق. وبالوتيرة نفسها، كم
صفحة في $5$ دقائق؟ وكم من الوقت من أجل $96$ صفحة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:prop:4}
اِحسب: $30\,\%$ من $250$؛ \quad و $15\,\%$ من $60$؛ \quad و $t\,\%$ بحيث
تكون $t\,\%$ من $80$ هي $20$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:prop:5}
من بين $25$ تسديدة، سجّلت لاعبة كرة سلة $18$. فما نسبة
نجاحها المئوية؟ (أعِد كتابة \omterm{def:g6:fractions:def}{الكسر} على \omterm{def:g4:fractions:def}{المقام} $100$.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:prop:6}
على خريطة $1 : 25\,000$، يقيس مسار $9$ سم. فما طوله
الحقيقي بالكيلومتر؟ وبحيرة طولها $2$ كم: فكم يبلغ طولها على الخريطة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:prop:7}
سيارة مصغّرة مبنية \omterm{def:g7:prop:scale}{بسلّم} $1 : 43$. والسيارة الحقيقية طولها $4.3$ م.
فكم يبلغ طول المصغّرة، بالسنتيمتر؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:prop:8}
وصفة تستعمل $240$ غ من الشوكولاتة من أجل $8$ حصص. وعند أمينة
$300$ غ من الشوكولاتة. فلكم حصة تكفي (عدد صحيح
من الحصص)؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:prop:9}
ينخفض ثمن سترة من $80$ إلى $60$ يورو.
\begin{enumerate}
  \item فما التخفيض باليورو؟ وبالنسبة المئوية من الثمن
        الأصلي؟
  \item ثم يعود الثمن فيرتفع من $60$ إلى $80$ يورو. اِشرح
        لماذا لا يكون هذا الارتفاع $25\,\%$ --- واِحسب النسبة
        المئوية الصحيحة للزيادة.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:prop:10}
سطرا جدول متناسبان:
\begin{center}

 & $4$ & $x$ & $10$ \\

 & $y$ & $10.5$ & $17.5$ \\

\begin{exercise}[$\star\star\star$]\label{exo:g7:prop:11}
آلة نسخ تصغّر الوثائق إلى $80\,\%$ من حجمها. وتُنسخ \omterm{def:g6:lines:objects}{قطعة}
طولها $10$ سم، ثم تُنسخ \emph{النسخة} من جديد بالضبط
نفسه.
\begin{enumerate}
  \item فكم يبلغ طول \omterm{def:g6:lines:objects}{القطعة} بعد نسخة واحدة؟ وبعد نسختين؟
  \item ولماذا لا يكون الجواب بعد نسختين $60\,\%$ من
        الأصل؟ وأيّ نسبة مئوية واحدة توافق نسختين؟
\end{enumerate}
\end{exercise}

\section{مسألة: قياس الأرض بعصا}

\begin{problem}[{مسألة نهاية الأسبوع --- الظلال تناسب،
وكيف حسب إراتوستينس محيط
الأرض سنة 240 قبل الميلاد}]
\label{pb:g7:prop:1}
قبل اِثنين وعشرين قرنًا، دون منظار ولا قمر اِصطناعي ولا
آلة حاسبة، قاس أمين مكتبة الإسكندرية الأرض ---
بعصا وبئر وقافلة جِمال ورياضيات هذا
الفصل. وتعيد هذه المسألة تتبّع خطواته: أولًا
\omterm{def:g7:prop:table}{تناسب} الظلال، ثم الحساب المشهور نفسه،
وأخيرًا كيف يبدو جوابه على \omterm{def:g7:prop:scale}{السلّم} البشري.

\medskip
\noindent\textbf{الجزء الأول --- ظلّ عصا.}
الشمس بعيدة جدًّا حتى إن أشعتها تبلغنا متوازية بعضها
مع بعض. وفي لحظة معطاة، تكون كل الأشياء العمودية وظلالها
متناسبة إذن.
\begin{enumerate}
  \item عند الظهر، تلقي عصا عمودية طولها $1$ م ظلًّا طوله
        $0.4$ م، ويقيس ظلّ شجرة $3.2$ م. فكم يبلغ طول
        الشجرة (\cref{thm:g7:prop:cross})؟
  \item وفي اللحظة نفسها، يلقي برج ظلًّا طوله $26$ م: فكم
        يبلغ اِرتفاع البرج؟ وكم يبلغ طول ظلّ طفل
        طوله $1.5$ م؟
  \item اِشرح في جملة واحدة لماذا لا تصحّ هذه الحسابات كلها
        إلا في \emph{اللحظة نفسها} من النهار.
  \item تقول الأسطورة إن طاليس قاس \omterm{def:g5:solids:def}{الهرم} الأكبر باِنتظار
        اللحظة التي يكون فيها \emph{ظلّه هو نفسه بطول
        قامته بالضبط}. فما معامل \omterm{def:g7:prop:table}{التناسب}
        في تلك اللحظة، وكم يبلغ اِرتفاع \omterm{def:g5:solids:def}{الهرم}
        إذا كان طرف ظلّه يبعد $147$ م عن نقطة
        الأرض الواقعة تحت قمّته مباشرةً؟
  \item لخّص الجزء الأول: في لحظة معطاة، أيّ مقدار يكون
        هو نفسه للعصا والشجرة والبرج
        \omterm{def:g5:solids:def}{والهرم} (\cref{def:g7:prop:table})؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- حساب إراتوستينس.}
عرف إراتوستينس واقعتين. ففي مدينة أسوان، عند الظهر في
اِنقلاب الصيف، كانت الشمس \emph{فوق \omterm{def:g5:solids:def}{الرؤوس} تمامًا}: فيبلغ ضوءها
قاع أعمق الآبار، ولا تلقي العصيّ العمودية
ظلًّا. وفي الإسكندرية، على بُعد $800$ كم شمالًا تمامًا، في اللحظة نفسها،
كانت عصا عمودية \emph{تلقي} ظلًّا، وتصنع أشعة الشمس
زاوية $7.2^\circ$ مع العمودي.
\begin{enumerate}[resume]
  \item اِشرح لماذا يبرهن هذان الرصدان معًا على أن
        أرض أسوان وأرض الإسكندرية ليستا
        متوازيتين --- أي إن سطح الأرض منحنٍ.
        (فماذا كانت الأشعة المتوازية تفعل بعصوين على أرض
        مسطّحة؟)
  \item على رسم للأرض المدوّرة بأشعة شمس متوازية،
        تظهر الزاوية $7.2^\circ$ التي في الإسكندرية من جديد في مركز
        الأرض، زاويةً بين اِتجاهَي
        المدينتين. اُرسم الرسم واِقتنع بذلك (والتبرير
        النظيف يستعمل أزواج الزوايا المتساوية في
        \cref{ch:g7:triangles}).
  \item وأيّ \omterm{def:g6:fractions:def}{كسر} من الدورة الكاملة هي $7.2^\circ$؟
  \item القوس من أسوان إلى الإسكندرية ($800$ كم) يوافق
        $7.2^\circ$؛ \omterm{def:g6:measure:perimeter}{والمحيط} كله يوافق
        $360^\circ$. أقِم \omterm{def:g7:prop:table}{التناسب} واِحسب
        \omterm{def:g6:measure:perimeter}{محيط} الأرض.
  \item اِستنتج \omterm{def:g4:geometry:circle}{قطر} الأرض، مستعملًا $\pi \approx
        3.14$ (\cref{prop:g6:measure:circle}) \omterm{def:g6:decimals:rounding}{والتقريب} إلى
        أقرب مئة كيلومتر. (والقيمة الحديثة هي
        $12\,742$ كم --- فكم كان قريبًا رجل بعصا
        سنة 240 قبل الميلاد؟)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الأرض على \omterm{def:g7:prop:scale}{السلّم} البشري.}
\begin{enumerate}[resume]
  \item كرة أرضية مبنية \omterm{def:g7:prop:scale}{بسلّم} $1 : 40\,000\,000$. وباستعمال
        \omterm{def:g6:measure:perimeter}{المحيط} الموجود في السؤال 9، بيّن أن
        \omterm{def:g6:measure:perimeter}{محيط} الكرة هو $1$ م بالضبط. فما
        قطرها، إلى أقرب سنتيمتر؟
  \item يرتفع مون بلان نحو $4.8$ كم. حوّل $4.8$ كم إلى
        سنتيمترات، واقسم على $40\,000\,000$، وعبّر عن
        اِرتفاع الجبل على الكرة بالميليمتر. فماذا يقول
        الجواب عن مقدار «تضرّس» الأرض حقًّا؟
  \item يقطع ماشٍ $40$ كم في اليوم. فبهذه الوتيرة، كم
        يومًا يلزم \omterm{def:g6:measure:perimeter}{لمحيط} إراتوستينس كله --- وكم
        سنة يساوي ذلك تقريبًا؟
  \item الجوّ الصالح للتنفّس مركّز تقريبًا في
        أول $10$ كم فوق الأرض. فأيّ نسبة مئوية من
        \omterm{def:g3:shapes:shapes}{نصف قطر} الأرض (نحو $6\,370$ كم) هي تلك
        (\cref{met:g7:prop:percent})؟ قرّب إلى أقرب
        عُشر من النسبة.
  \item يقدّر المؤرّخون أن القيمة التي أعلنها إراتوستينس،
        محوّلةً إلى الوحدات الحديثة، ربما كانت نحو
        $42\,000$ كم. وبأخذ $40\,000$ كم قيمةً حقيقية،
        اِحسب نسبة خطئه المئوية. واختم بجملة
        واحدة عن العصيّ والآبار \omterm{def:g7:prop:table}{والتناسب}.
\end{enumerate}
\end{problem}
