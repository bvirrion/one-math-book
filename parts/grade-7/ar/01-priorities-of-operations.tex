\chapter{أولويات العمليات}\label{ch:g7:priorities}

ما قيمة $3 + 4 \times 5$؟ إن حسبت من اليسار إلى اليمين وجدت $35$؛
وإن ضربت أولًا وجدت $23$. والرياضيات تحتاج جوابًا واحدًا
لا غير، فكانت للحسابات قواعد سير --- هي الأولويات.
ويذكر هذا الفصل هذه القواعد ويقدّم أول متطابقة جبرية،
وهي التوزيعية.

\section{قواعد الأولوية}

\begin{theorem}[قواعد الأولوية]\label{thm:g7:priorities:rules}
في حساب بلا أقواس:
\begin{enumerate}
  \item تُنجَز الضروب والقسمات \emph{قبل}
        الجموع والطروح؛
  \item وتُنجَز عمليات المستوى نفسه من اليسار إلى اليمين.
\end{enumerate}
والأقواس تعلو على كل شيء: فما بداخلها يُحسَب
أولًا، اِبتداءً من الأعمق.
\end{theorem}

\begin{example}\label{ex:g7:priorities:basic}
عملية واحدة في كل سطر، مع تسطير (ذهني) لما يُحسَب:
\begin{align*}
3 + 4 \times 5 &= 3 + 20 = 23
&&\text{(الضرب أولًا)}\\
20 - 8 + 2 &= 12 + 2 = 14
&&\text{(المستوى نفسه: من اليسار إلى اليمين)}\\
18 \div 3 \times 2 &= 6 \times 2 = 12
&&\text{(المستوى نفسه: من اليسار إلى اليمين)}\\
(3 + 4) \times 5 &= 7 \times 5 = 35
&&\text{(الأقواس أولًا).}
\end{align*}
واِنتبه للفخّين: $20 - 8 + 2$ \emph{ليس} $20 - 10$،
كما أن $18 \div 3 \times 2$ \emph{ليس} $18 \div 6$.
\end{example}

\begin{example}[أقواس متداخلة]\label{ex:g7:priorities:nested}
\begin{align*}
50 - 2 \times \bigl(3 + (10 - 4)\bigr)
&= 50 - 2 \times (3 + 6) && \text{(القوس الأعمق)}\\
&= 50 - 2 \times 9 && \text{(القوس الباقي)}\\
&= 50 - 18 && \text{(الضرب)}\\
&= 32 .
\end{align*}
\end{example}

\begin{remark}[شريط الكسر يخفي أقواسًا]
شريط \omterm{def:g6:fractions:def}{الكسر} يجمع بسطه ومقامه، كأن لكل
منهما أقواسًا غير مرئية:
\[
\frac{12 + 8}{4} = \frac{20}{4} = 5,
\qquad\text{لا}\quad 12 + \frac{8}{4} = 14 .
\]
\end{remark}

\begin{method}[الحساب بأمان]\label{met:g7:priorities:safe}
\begin{enumerate}
  \item حدّد العملية التي تُنجَز أولًا (الأقواس الأعمق، ثم
        $\times$ و $\div$، ثم $+$ و $-$)؛
  \item واحسب \emph{تلك وحدها}، وأعد كتابة العبارة كلها
        مع وضع النتيجة مكانها؛
  \item وكرّر حتى يبقى عدد واحد --- عملية واحدة في كل سطر، دون
        اِختصارات.
\end{enumerate}
\end{method}

\section{التوزيعية}

\begin{theorem}[التوزيعية]\label{thm:g7:priorities:distributivity}
من أجل كل الأعداد $k$ و $a$ و $b$:
\[
k \times (a + b) = k \times a + k \times b,
\qquad
k \times (a - b) = k \times a - k \times b .
\]
\end{theorem}

\begin{proof}[برهان بالصورة]
اِحسب \omterm{def:g6:measure:area}{مساحة} مستطيل عرضه $k$ مقسوم إلى مستطيلين
أصغر طولاهما $a$ و $b$: فهو ككلّ $k \times (a+b)$؛
وهو في جزأين $k \times a + k \times b$. المستطيل نفسه، \omterm{def:g6:measure:area}{والمساحة}
نفسها.
\end{proof}

\begin{example}[الاتجاهان معًا]\label{ex:g7:priorities:distributivity}
\emph{النشر} (من اليسار إلى اليمين) يسهّل الحساب الذهني:
\[
7 \times 103 = 7 \times (100 + 3) = 700 + 21 = 721,
\qquad
6 \times 98 = 6 \times (100 - 2) = 600 - 12 = 588 .
\]
\emph{التعميل} (من اليمين إلى اليسار) يجمع عاملًا مشتركًا:
\[
23 \times 12 + 23 \times 8 = 23 \times (12 + 8) = 23 \times 20 = 460 .
\]
\end{example}

\begin{example}[ترجمة الكلمات إلى حساب]\label{ex:g7:priorities:words}
عبارة «\omterm{def:g1:addition:def}{مجموع} $8$ \omterm{def:g2:mult:def}{وجداء} $5$ في $3$» هي $8 + 5 \times 3 =
23$. وعبارة «\omterm{def:g2:mult:def}{جداء} $5$ في \omterm{def:g1:addition:def}{مجموع} $8$ و $3$» هي
$5 \times (8 + 3) = 55$: فكلمات «\omterm{def:g2:mult:def}{جداء} \dots\ في \omterm{def:g1:addition:def}{مجموع}
\dots» تضع أقواسًا حول \omterm{def:g1:addition:def}{المجموع}.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g7:priorities:1}
اِحسب، عملية واحدة في كل سطر:
\[
5 + 3 \times 6, \qquad
(5 + 3) \times 6, \qquad
30 - 12 \div 4, \qquad
(30 - 12) \div 4 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:2}
اِحسب:
\[
24 - 6 + 2, \qquad
24 \div 6 \times 2, \qquad
24 - 6 \times 2, \qquad
24 \div (6 \times 2).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:3}
اِحسب خطوة خطوة:
\[
7 + 3 \times (10 - 6), \qquad
40 - (2 + 3) \times 4, \qquad
60 \div \bigl(2 \times (7 - 4)\bigr).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:4}
اِحسب الكسور (أقواس غير مرئية!):
\[
\frac{18 + 6}{8}, \qquad
\frac{30}{7 - 2}, \qquad
\frac{5 \times 8}{4 + 6} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:5}
اِستعمل التوزيعية للحساب الذهني:
\[
8 \times 102, \qquad
25 \times 99, \qquad
14 \times 12 + 14 \times 8, \qquad
37 \times 45 - 37 \times 35 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:6}
اُكتب الحساب في سطر واحد، مع الأقواس عند اللزوم، ثم
اِحسب: «\omterm{ex:g1:subtraction:difference}{فرق} $50$ \omterm{def:g2:mult:def}{وجداء} $6$ في $7$»؛
و«\omterm{def:g2:mult:def}{جداء} \omterm{def:g1:addition:def}{مجموع} $4$ و $5$ في \omterm{ex:g1:subtraction:difference}{فرق} $10$ و
$8$».
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:7}
اِشترى أمير $3$ دفاتر ثمن الواحد $2.50$ و $2$ قلمين ثمن الواحد $1.20$.
اُكتب \omterm{def:g1:addition:def}{مجموع} ما دفع في عبارة واحدة، ثم اِحسبه.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:priorities:8}
ضع أقواسًا في $4 + 2 \times 5 - 1$ لتجعل النتيجة: (أ) $13$؛
(ب) $29$؛ (ج) $24$. (وواحدة من الثلاث لا تحتاج أقواسًا البتة ---
فأيّها؟)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:priorities:9}
هل $12 + 4 \times k$ و $16 \times k$ متساويان من أجل كل عدد $k$؟
اِختبر بالقيمتين $k = 1$ و $k = 2$، واِستنتج. وأيّ خطأ في الأولوية
يقود من إحداهما إلى الأخرى؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:priorities:10}
تذكرة السينما ثمنها $9$ يورو، لكن مجموعة من $n$ شخصًا تدفع مبلغًا
ثابتًا $12$ يورو زائد $6$ يورو لكل شخص.
\begin{enumerate}
  \item اُكتب عبارة لثمن المجموعة، واحسبها من أجل
        $n = 5$.
  \item واِبتداءً من كم شخصًا يصير سعر المجموعة أرخص من $n$
        تذكرة فردية؟ (جرّب بضع قيم للعدد $n$.)
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:priorities:11}
باستعمال كل رقم من الأرقام $1$ و $2$ و $3$ و $4$ مرة واحدة بالضبط، مع
العمليات الأربع والأقواس كما تشاء، اُكتب عبارات تساوي $24$ و
$10$ و $1$.
\end{exercise}

\section{مسألة: الضرب على طريقة الكاتب المصري}

\begin{problem}[{مسألة نهاية الأسبوع --- حيل الحساب الذهني من
التوزيعية، وضرب المضاعفة والتنصيف المستعمل
منذ أربعة آلاف سنة}]
\label{pb:g7:priorities:1}
قبل الحاسبات بزمن طويل --- وقبل أن تُحفَّظ جداول الضرب
في المدرسة --- كان كتبة مصر وفلاحو روسيا
يضربون أيّ عددين مستعملين \emph{المضاعفة والتنصيف
والجمع} وحدها. وتبدو طريقتهم سحرًا؛ وهي في الحقيقة
التوزيعية (\cref{thm:g7:priorities:distributivity}) تعمل
بكل طاقتها. وتشحذ هذه المسألة حسابك الذهني، ثم
تعلّمك الطريقة القديمة، ثم ترفع الغطاء لترى بالضبط
لماذا تعمل.

\medskip
\noindent\textbf{الجزء الأول --- فنّ عدم الحساب.}
\begin{enumerate}
  \item اِحسب ذهنيًّا، بالنشر كما في
        \cref{ex:g7:priorities:distributivity}:
        $7 \times 102$ و $9 \times 98$.
  \item اِحسب ذهنيًّا، بالتعميل:
        $17 \times 13 + 17 \times 7$ و
        $45 \times 101 - 45$.
  \item حيلة «الضرب في $5$»: للضرب في $5$، اِضرب في
        $10$ ثم نصّف. اِحسب $5 \times 87$ بهذه الطريقة،
        واِشرح لماذا تكون الحيلة مشروعة.
  \item وحيلتان أخريان للتبرير والاستعمال: اِضرب في $25$
        بالضرب في $100$ والقسمة على $4$ (اِحسب
        $25 \times 32$)؛ واِضرب في $99$ بالضرب في $100$
        والطرح مرة واحدة (اِحسب $99 \times 47$).
  \item وطلقة تحذير: التوزيعية تنشر عاملًا على
        \emph{\omterm{def:g1:addition:def}{مجموع}}، لا على \omterm{def:g2:mult:def}{جداء}. قارن
        $8 \times (5 + 3)$ بالمقدار $8 \times 5 + 8 \times 3$، ثم
        $8 \times (5 \times 3)$ بالمقدار
        $(8 \times 5) \times (8 \times 3)$: فأيّ زوج يتطابق،
        وأيّهما لا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- طريقة المضاعفة والتنصيف.}
وإليك وصفة $13 \times 24$. اُكتب عمودين. العمود
الأيسر: اِبدأ من $13$ و\emph{نصّف}، مهملًا أيّ باقٍ،
نزولًا إلى $1$: \ $13,\ 6,\ 3,\ 1$. والعمود الأيمن: اِبدأ من $24$
و\emph{ضاعِف} العدد نفسه من المرات: \ $24,\ 48,\ 96,\ 192$. ثم اِشطب
كل سطر عدده \emph{الأيسر} زوجي، واجمع الأعداد اليمنى
الباقية.
\begin{enumerate}[resume]
  \item نفّذ الوصفة من أجل $13 \times 24$: فأيّ سطر
        يُشطب، وأيّ \omterm{def:g1:addition:def}{مجموع} يبقى، وهل يساوي
        $13 \times 24$؟
  \item اِضرب $21 \times 17$ بالطريقة (نصّف $21$، وضاعِف
        $17$)، وتحقّق من الجواب بالضرب العادي.
  \item اِضرب $16 \times 25$ بالطريقة. فكم سطرًا
        ينجو، وما الخاصّ في عمود أيسر
        يبدأ من $1$ و $2$ و $4$ و $8$ و $16, \dots$؟
  \item والآن \emph{لماذا} تعمل، خطوة خطوة. بَرِّر كل
        تساوٍ بالتوزيعية:
        \[
        13 \times 24 = (12 + 1) \times 24
        = 12 \times 24 + 24 = 6 \times 48 + 24 .
        \]
        والدرس: التنصيف على اليسار والمضاعفة على اليمين يحفظان
        \omterm{def:g2:mult:def}{الجداء} --- لكن حين يكون العدد الأيسر \emph{فرديًّا}، تنفصل
        نسخة واحدة من العدد الأيمن ويجب إيداعها.
        فأيّ عدد يودعه السطر الأول من الجدول؟
  \item تابع المحاسبة نزولًا في الجدول:
        $6 \times 48 = 3 \times 96$، ثم
        $3 \times 96 = 1 \times 192 + 96$، ثم
        $1 \times 192 = 192$. واجمع كل ما أُودع
        في الطريق، واِشرح لماذا تكون الأعداد المودَعة
        بالضبط أعداد العمود الأيمن للسطور ذات اليسار
        الفردي --- وهي السطور التي تحتفظ بها الوصفة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- تحت الغطاء: مجاميع
\omterm{def:g5:division:multiple}{المضاعفات}.}
\begin{enumerate}[resume]
  \item في السؤال 6، كانت السطور الناجية تساوي $1$ و $4$
        و $8$ \omterm{def:g5:division:multiple}{مضاعفات} للعدد $24$: فقد كتبت الطريقة ضمنًا
        $13 \times 24 = (1 + 4 + 8) \times 24$. تحقّق من أن
        $1 + 4 + 8 = 13$، وجِد أيّ \omterm{def:g5:division:multiple}{المضاعفات} ($1$ و $2$
        و $4$ و $8$ و $16$) اِحتفظت بها الطريقة من أجل $21$ في السؤال 7.
  \item اُكتب $25$ و $42$ مجموعَي أعداد من قائمة المضاعفة
        $1, 2, 4, 8, 16, 32$، مستعملًا كل واحد مرة واحدة على الأكثر.
  \item اِشرح لماذا يمكن كتابة \emph{كل} عدد صحيح
        بهذه الطريقة: خذ أكبر عدد مضاعفة يدخل فيه،
        واطرح، وكرّر --- ولماذا يجب أن يتوقف هذا؟ واُكتب $100$
        مجموعًا كهذا.
  \item فعل كتبة مصر هذا بالضبط، قبل أربعة آلاف
        سنة: فلضرب $19 \times 35$، صنعوا
        جدول مضاعفة للعدد $35$ ($1 \to 35$، و $2 \to 70$،
        و $4 \to 140$، و $8 \to 280$، و $16 \to 560$)، واِختاروا السطور
        التي مجموعها $19$، وجمعوها. أنجز الحساب على
        طريقتهم.
  \item الخاتمة: الحاسوب الحديث، الذي يعدّ بالرمزين $0$ و
        $1$، يضرب أساسًا كما يضرب الكتبة --- \omterm{def:g5:division:multiple}{بالمضاعفات}
        والجمع. فأيّ ثلاث عمليات جمع يجريها
        الحاسوب من أجل $13 \times 24$؟ (وأما لماذا ينقسم كل عدد
        إلى \omterm{def:g5:division:multiple}{مضاعفات}، وما علاقة الرمزين $0$ و $1$
        بذلك، فمسألة نهاية الأسبوع في فصل القوى:
        \cref{pb:g8:powers:1}.)
\end{enumerate}
\end{problem}
