\chapter{Nombres relatifs}\label{ch:g7:negatives}

Températures sous \omterm{def:g1:counting:zero}{zéro}, étages sous le sol, dettes~: bien des
grandeurs passent sous $0$. Les nombres négatifs prolongent la droite
numérique à gauche de \omterm{def:g1:counting:zero}{zéro}. Ce chapitre enseigne à les comparer et à
les additionner et soustraire --- leur multiplication attend le
\cref{ch:g8:negprod}.

\section{Nombres relatifs}

\begin{definition}[Nombres relatifs]\label{def:g7:negatives:def}
Un \emph{nombre relatif}\index{nombre relatif} est formé d'un signe
($+$ ou $-$) et d'une \emph{distance à \omterm{def:g1:counting:zero}{zéro}}~: $+3$ et $-3$ sont tous
deux à distance $3$ de $0$, de part et d'autre. Les nombres positifs
s'écrivent d'ordinaire sans leur signe~: $+3 = 3$. Les nombres $+3$
et $-3$ sont \emph{opposés}\index{opposé} l'un de l'autre.
\end{definition}

\begin{omfigure}
\begin{tikzpicture}[scale=1.15]
  \draw[-Stealth] (-5.6,0) -- (5.6,0);
  \foreach \x in {-5,...,5} \draw (\x,-0.08) -- (\x,0.08);
  \foreach \x in {-5,...,5} \node[below, font=\small] at (\x,-0.1) {$\x$};
  \fill[omThm] (-3,0) circle (2pt);
  \fill[omDef] (3,0) circle (2pt);
  \draw[Stealth-Stealth, omExo] (-3,0.35) -- (0,0.35)
    node[midway, above, font=\small] {$3$};
  \draw[Stealth-Stealth, omExo] (0,0.35) -- (3,0.35)
    node[midway, above, font=\small] {$3$};
\end{tikzpicture}

{\small La droite numérique prolongée à gauche~: $-3$ et $+3$ sont
des nombres opposés, à la même distance de $0$.}
\end{omfigure}

\begin{method}[Comparer des nombres relatifs]\label{met:g7:negatives:compare}
Sur la droite numérique, plus petit signifie plus à gauche. En
pratique~:
\begin{enumerate}
  \item un nombre négatif est toujours plus petit qu'un positif~:
        $-7 < 2$~;
  \item entre deux positifs, l'ordre habituel~;
  \item entre deux négatifs, celui de plus \emph{grande} distance à
        \omterm{def:g1:counting:zero}{zéro} est le plus \emph{petit}~: $-7 < -2$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:negatives:compare}
Ranger du plus petit au plus grand~: $3$~; $-5$~; $0$~; $-1.5$~;
$2.8$. Les négatifs d'abord, le plus négatif en tête~:
\[
-5 < -1.5 < 0 < 2.8 < 3 .
\]
\end{example}

\section{Additionner des nombres relatifs}

Penser un \omterm{def:g7:negatives:def}{nombre relatif} comme un déplacement sur la droite~: $+5$
signifie «~avancer de $5$ pas à droite~», $-3$ signifie «~avancer de
$3$ pas à gauche~». Additionner des nombres, c'est enchaîner les
déplacements.

\begin{method}[Additionner deux nombres relatifs]\label{met:g7:negatives:add}
\begin{enumerate}
  \item \emph{Même signe}~: ajouter les distances, garder le signe
        commun~: $(-3) + (-4) = -7$.
  \item \emph{Signes contraires}~: soustraire la plus petite distance
        de la plus grande, prendre le signe de la plus grande~:
        $(-7) + (+2) = -5$ et $(+7) + (-2) = +5$.
  \item Les opposés s'annulent~: $(+4) + (-4) = 0$.
\end{enumerate}
\end{method}

\begin{tikzpicture}[scale=1.05]
  \draw[-Stealth] (-5.4,0) -- (4.4,0);
  \foreach \x in {-5,...,4} \draw (\x,-0.07) -- (\x,0.07);
  \foreach \x in {-5,-3,-1,0,2,4}
    \node[below, font=\small] at (\x,-0.09) {$\x$};
  \draw[-Stealth, omThm, very thick] (0,0.35) -- (-3.5,0.35);
  \draw[-Stealth, omDef, very thick] (-3.5,0.75) -- (-1.5,0.75);
  \node[omThm, font=\small] at (-1.75,0.18) {$-3.5$};
  \node[omDef, font=\small] at (-2.5,1.0) {$+2$};
  \fill (-1.5,0) circle (2pt) node[below=6pt, font=\small] {$-1.5$};
\end{tikzpicture}

{\small Additionner comme se déplacer~: partant de $0$, le
déplacement $-3.5$ puis le déplacement $+2$ aboutit à
$(-3.5) + 2 = -1.5$.}
\end{omfigure}

\begin{example}\label{ex:g7:negatives:add}
Étape par étape, en regroupant à convenance~:
\begin{align*}
(-8) + (+3) &= -5 && \text{(signes contraires~: } 8 - 3 = 5,
\text{ signe de } 8)\\
(-2.5) + (-1.5) &= -4 && \text{(même signe~: ajouter les distances)}\\
(+6) + (-9) + (+3) &= (-3) + (+3) = 0 .
\end{align*}
\end{example}

\section{Soustraire des nombres relatifs}

\begin{theorem}[Règle de soustraction]\label{thm:g7:negatives:sub}
Soustraire un \omterm{def:g7:negatives:def}{nombre relatif} revient à \emph{ajouter son opposé}~:
\[
a - b = a + (-b) .
\]
\end{theorem}

\begin{proof}[Pourquoi cela fonctionne]
$a - b$ est le nombre qui, ajouté à $b$, donne $a$. Vérifier que
$a + (-b)$ fait le travail~:
$\bigl(a + (-b)\bigr) + b = a + \bigl((-b) + b\bigr) = a + 0 = a$.
\end{proof}

\begin{example}\label{ex:g7:negatives:sub}
Toute soustraction devient une addition, puis
\cref{met:g7:negatives:add} s'applique~:
\begin{align*}
5 - 9 &= 5 + (-9) = -4, \\
(-3) - (+4) &= (-3) + (-4) = -7, \\
(-2) - (-6) &= (-2) + (+6) = +4 .
\end{align*}
La dernière ligne est la célèbre~: \emph{soustraire un négatif,
c'est ajouter}. Supprimer une dette vous enrichit~!
\end{example}

\begin{example}[Écarts de température]\label{ex:g7:negatives:temperature}
En une journée, la température est passée de $-6$ degrés à $+9$
degrés. La hausse est la \omterm{ex:g1:subtraction:difference}{différence}
\[
9 - (-6) = 9 + 6 = 15 \text{ degrés}.
\]
Sur la droite numérique~: de $-6$ jusqu'à $0$ c'est $6$ degrés, de
$0$ à $9$ c'est $9$ de plus.
\end{example}

\section{Coordonnées dans tout le plan}

\begin{definition}[Coordonnées avec signes]\label{def:g7:negatives:coords}
Avec deux axes gradués \omterm{def:g6:lines:perp}{perpendiculaires} se croisant à l'origine $O$,
tout point du plan reçoit deux coordonnées relatives $(x, y)$~: un
$x$ négatif signifie à gauche de l'axe vertical, un $y$ négatif
signifie au-dessous de l'axe horizontal.
\end{definition}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g7:negatives:1}
Donner l'opposé de~: $7$~; \ $-3.5$~; \ $0$~; \ $+12$. Puis
compléter~: «~deux nombres sont opposés lorsque leur \omterm{def:g1:addition:def}{somme}
vaut~?~».
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:2}
Recopier et compléter avec $<$ ou $>$~:
\[
-4 \;?\; 1, \qquad
-4 \;?\; -7, \qquad
-2.5 \;?\; -2.4, \qquad
0 \;?\; -0.1 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:3}
Ranger du plus petit au plus grand~: $-3$~; \ $2.5$~; \ $-0.5$~;
\ $-3.1$~; \ $0$~; \ $1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:4}
Calculer~:
\[
(-5) + (-9), \qquad
(-12) + (+7), \qquad
(+3.5) + (-1.5), \qquad
(-6) + (+6).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:5}
Transformer en additions, puis calculer~:
\[
4 - 11, \qquad
(-5) - (+3), \qquad
(-7) - (-10), \qquad
0 - (-8).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:6}
Calculer en regroupant astucieusement (les opposés d'abord)~:
\[
(+7) + (-12) + (+3) + (-8), \qquad
(-5.5) + (+9) + (-4.5) + (+1).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:7}
La température était de $-3$ degrés ce matin~; elle a monté de $8$
degrés dans la journée, puis baissé de $12$ degrés la nuit. Écrire un
seul calcul donnant la température finale, et le mener.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:negatives:8}
Placer dans un repère~: $A(2, 3)$~; $B(-3, 1)$~; $C(-2, -2)$~;
$D(3, -1)$~; $E(0, -3)$. Quel point est sur un axe~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:9}
Calculer l'écart entre la température la plus haute et la plus
basse~: Moscou en janvier, max $-4$ degrés, min $-13$ degrés~;
Helsinki, max $2$, min $-9$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:10}
Une plongeuse est à l'altitude $-12$ m (douze mètres sous la
surface). Elle remonte de $7$ m, puis descend de $4$ m. À quelle
altitude est-elle~? De combien doit-elle encore remonter pour
atteindre la surface~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:negatives:11}
Placer $A(-2, 1)$, $B(2, 1)$, $C(2, -2)$. Trouver les coordonnées du
point $D$ pour que $ABCD$ soit un rectangle, et calculer son
\omterm{def:g6:measure:perimeter}{périmètre} (compter les carreaux pour les longueurs des côtés).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:negatives:12}
Dans le carré magique ci-dessous, chaque ligne, colonne et diagonale
doit avoir la même \omterm{def:g1:addition:def}{somme}. Recopier et le compléter~:
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|}
\hline
$?$ & $-7$ & $0$ \\
\hline
$-3$ & $-2$ & $?$ \\
\hline
$-4$ & $?$ & $?$ \\
\hline
\end{tabular}
\end{center}
(Commencer par trouver la \omterm{def:g1:addition:def}{somme} magique à l'aide de la diagonale
presque complète.)
\end{exercise}

\section{Problème~: l'an zéro qui n'a jamais existé}

\begin{problem}[{Devoir maison --- les nombres relatifs sur la frise
de l'histoire~: des durées à cheval sur le changement d'ère, l'astuce
des astronomes, et le principe de la variation totale}]
\label{pb:g7:negatives:1}
La droite graduée de ce chapitre a un célèbre jumeau dans le monde
réel~: la frise de l'histoire, avec les années av. J.-C. qui
s'étirent vers la gauche et les années apr. J.-C. vers la droite.
Mais la frise des historiens cache un piège qui a gâché bien des
calculs~: \emph{il n'y a pas d'an \omterm{def:g1:counting:zero}{zéro}} --- l'an $1$ av. J.-C. est
immédiatement suivi de l'an $1$ apr. J.-C. Ce problème calcule
d'abord sur des températures, des ascenseurs et des comptes en
banque, puis répare le \omterm{def:g1:counting:zero}{zéro} cassé de l'histoire comme le font les
astronomes~: avec des \omterm{def:g7:negatives:def}{nombres relatifs}.

\medskip
\noindent\textbf{Partie I --- Échauffements, et un premier piège.}
\begin{enumerate}
  \item Le \omterm{def:g5:solids:def}{sommet} du mont Blanc est à l'altitude $+4\,808$ m~; la
        rive de la mer Morte à $-430$ m. Calculer leur \omterm{ex:g1:subtraction:difference}{différence}
        d'altitude.
  \item Un compte en banque affiche $-45$ euros (un découvert). Son
        titulaire dépose $120$ euros, règle une facture de $60$
        euros, puis dépose $15$ euros. Écrire le solde comme une
        seule chaîne d'additions et le calculer. Quel dépôt unique
        aurait ramené le $-45$ initial exactement à $0$~?
  \item Jules César est né en $100$ av. J.-C. et mort en $44$
        av. J.-C. En comptant les années av. J.-C. comme des nombres
        négatifs ($100$ av. J.-C. pour $-100$, $44$ av. J.-C. pour
        $-44$), calculer $(-44) - (-100)$~: combien d'années César
        a-t-il vécu~?
  \item L'empereur Auguste est né en $63$ av. J.-C. et mort en $14$
        apr. J.-C. La même recette donne $14 - (-63) = 77$ ans ---
        mais les historiens affirment qu'il est mort à $76$ ans.
        Leur objection~: le compte $77$ traverse une année qui n'a
        jamais existé. Laquelle~?
  \item Les astronomes réparent la frise en renumérotant~: les
        années apr. J.-C. gardent leur numéro ($+14$ \omterm{def:g3:division:remainder}{reste} $+14$),
        mais l'an $1$ av. J.-C. devient $0$, l'an $2$ av. J.-C.
        devient $-1$, et plus généralement l'an $n$ av. J.-C.
        devient $-(n - 1)$. Convertir $63$ av. J.-C., et refaire le
        calcul de la question 4 avec les nombres des astronomes. La
        réponse satisfait-elle désormais les historiens~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Calculer d'une ère à l'autre.}
Dans toute la suite, utiliser la renumérotation des astronomes pour
chaque calcul, et reconvertir en av./apr. J.-C. à la fin (une année
numérotée $0$ ou négative $-m$ est l'an $m + 1$ av. J.-C. des
historiens).
\begin{enumerate}[resume]
  \item Convertir en nombres d'astronomes~: $1$ av. J.-C.~; $10$
        av. J.-C.~; $476$ apr. J.-C.~; et $753$ av. J.-C., la
        fondation légendaire de Rome.
  \item Combien d'années se sont écoulées de la fondation de Rome
        ($753$ av. J.-C.) à la chute de l'Empire romain d'Occident
        ($476$ apr. J.-C.)~?
  \item Une comète revient tous les $76$ ans et est apparue en $12$
        av. J.-C. Donner, dans le comput des historiens, les années
        de ses trois apparitions suivantes, et celle de l'apparition
        juste avant $12$ av. J.-C.
  \item Combien d'années se sont écoulées de $5$ av. J.-C. à $5$
        apr. J.-C.~? (La réponse tentante est $10$~; la frise dit
        autre chose.)
  \item Un enfant né en $3$ av. J.-C. a eu dix ans en quelle année
        (av. ou apr. J.-C.)~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Le principe de la variation totale.}
\begin{enumerate}[resume]
  \item La \emph{distance} entre deux nombres de la droite est le
        plus grand moins le plus petit --- toujours positive.
        Calculer la distance entre $-7$ et $-2$, puis entre $-3.5$
        et $4.5$.
  \item Un ascenseur part de l'étage $0$ et effectue les
        déplacements $+3$, $-5$, $+2$, $-1$, $+4$. Calculer l'étage
        d'arrivée en regroupant habilement
        (\cref{met:g7:negatives:add}, les opposés d'abord quand
        c'est possible). Effectuer les déplacements dans un autre
        ordre changerait-il la destination~? Pourquoi~?
  \item Jour après jour, la température varie de $+2$, $-4$, $+1$,
        $0$, $-3$, $+5$, $-2$ degrés au cours d'une semaine.
        Calculer la \emph{variation totale}. Si la semaine a
        commencé à $-2$ degrés, à quelle température s'est-elle
        terminée~? Énoncer le principe~: valeur finale $=$ valeur
        initiale $+$ (\omterm{def:g1:addition:def}{somme} de toutes les variations).
  \item La marche d'une randonneuse la ramène exactement à son
        altitude de départ. Que doit valoir la \omterm{def:g1:addition:def}{somme} de toutes ses
        variations d'altitude, et pourquoi~? De ses six variations
        relevées, cinq valent $+4$, $-7$, $+2$, $+3$, $-5$ (en
        centaines de mètres)~: trouver la sixième.
  \item Bouquet final, retour à Rome. Un historien calcule~: «~la
        République a duré de $509$ av. J.-C. à $27$ av. J.-C., puis
        l'Empire jusqu'en $476$ apr. J.-C.~; en tout, de $509$
        av. J.-C. à $476$ apr. J.-C., cela fait $509 + 476 = 985$
        années d'État romain.~» Refaire le calcul avec les nombres
        des astronomes. Quel est le total correct --- et quelle
        année, une fois de plus, a causé l'erreur~?
\end{enumerate}
\end{problem}
