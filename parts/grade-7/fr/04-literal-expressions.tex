\chapter{Expressions littérales}\label{ch:g7:literal}

Une lettre dans un calcul désigne un nombre que l'on ne connaît pas
encore, ou \emph{tous} les nombres à la fois~: écrire $P = 4c$ donne
le \omterm{def:g6:measure:perimeter}{périmètre} de tout carré en une seule formule. Ce chapitre enseigne
les conventions d'écriture, comment substituer des valeurs, et comment
simplifier des expressions simples --- le début de l'algèbre, poursuivi
au \cref{ch:g8:equations}.

\section{Écrire avec des lettres}

\begin{definition}[Expression littérale]\label{def:g7:literal:def}
Une \emph{expression littérale}\index{expression littérale} est un
calcul contenant une ou plusieurs lettres, chacune désignant un
nombre. Conventions d'écriture~:
\begin{itemize}
  \item le signe $\times$ s'omet devant une lettre ou une
        parenthèse~: $3 \times a = 3a$, \ $a \times b = ab$, \
        $2 \times (x + 5) = 2(x + 5)$~;
  \item mais il reste entre deux nombres~: $3 \times 5$ ne peut pas
        s'écrire $35$~!
  \item $a \times a$ s'écrit $a^2$ («~$a$ au carré~»),
        $a \times a \times a = a^3$~;
  \item $1a$ s'écrit $a$, et $0a$ vaut $0$.
\end{itemize}
\end{definition}

\begin{example}\label{ex:g7:literal:write}
Simplifier l'écriture~:
\[
5 \times x + 3 \times y = 5x + 3y,
\qquad
a \times 7 = 7a \ \text{(nombre d'abord)},
\qquad
x \times x \times 4 = 4x^2 .
\]
\end{example}

\begin{example}[Les formules sont des expressions littérales]\label{ex:g7:literal:formulas}
Pour un rectangle de longueur $L$ et de largeur $w$~:
\omterm{def:g6:measure:perimeter}{périmètre} $P = 2(L + w)$, \omterm{def:g6:measure:area}{aire} $A = Lw$. Une formule résume
l'infinité des rectangles d'un coup --- c'est tout l'intérêt des
lettres.
\end{example}

\begin{tikzpicture}[scale=0.8]
  \draw[thick] (0,0) rectangle (4.4,2.0);
  \node[below, font=\small] at (2.2,0) {$L$};
  \node[left, font=\small] at (0,1.0) {$w$};
  \node[font=\small] at (2.2,1.0) {$A = Lw$};
  \node[above, font=\small, omDef] at (2.2,2.05) {$P = 2(L + w)$};
\end{tikzpicture}

{\small Deux formules, valables pour tout rectangle~: substituer les
valeurs de $L$ et $w$ pour obtenir des nombres.}
\end{omfigure}

\section{Substituer}

\begin{method}[Évaluer une expression]\label{met:g7:literal:substitute}
Pour évaluer une expression pour des valeurs données de ses
lettres~:
\begin{enumerate}
  \item réécrire l'expression en remettant les signes $\times$
        cachés~: $3a + a^2$ devient $3 \times a + a \times a$~;
  \item remplacer chaque lettre par sa valeur, en gardant des
        parenthèses autour si besoin~;
  \item calculer, en respectant les priorités du
        \cref{ch:g7:priorities}.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:literal:substitute}
Évaluer $E = 3x + x^2$ pour $x = 5$~:
\[
E = 3 \times 5 + 5 \times 5 = 15 + 25 = 40 .
\]
Évaluer $F = 2(a + 3b)$ pour $a = 4$ et $b = 0.5$~:
\[
F = 2 \times (4 + 3 \times 0.5) = 2 \times (4 + 1.5) = 2 \times 5.5
= 11 .
\]
\end{example}

\section{Réduire des expressions}

\begin{proposition}[Regrouper les termes semblables]\label{prop:g7:literal:reduce}
Les termes contenant la même partie littérale se regroupent en
additionnant leurs parties numériques (c'est la distributivité,
\cref{thm:g7:priorities:distributivity}, lue à l'envers)~:
\[
5x + 3x = (5 + 3)x = 8x,
\qquad
7a - 2a = 5a .
\]
Les termes de parties littérales différentes (comme $3x$ et $5y$,
ou $x$ et $x^2$) ne se regroupent pas.
\end{proposition}

\begin{example}\label{ex:g7:literal:reduce}
Réduire $E = 4x + 7 + 3x + 2$, en triant les termes~:
\[
E = (4x + 3x) + (7 + 2) = 7x + 9 .
\]
Attention~: $7x + 9$ n'est \emph{pas} $16x$~! Tester avec $x = 1$~:
$4 + 7 + 3 + 2 = 16$ et $7 \times 1 + 9 = 16$ coïncident, mais $16x$
vaudrait aussi $16$ --- tester avec $x = 2$~: l'original donne
$8 + 7 + 6 + 2 = 23$, et $7 \times 2 + 9 = 23$, tandis que
$16 \times 2 = 32$. La réduction $7x + 9$ est juste, $16x$ est
fausse.
\end{example}

\begin{method}[Tester une égalité]\label{met:g7:literal:test}
Pour vérifier si deux expressions pourraient être égales pour toute
valeur~:
\begin{enumerate}
  \item substituer la même valeur dans les deux, au moins deux fois
        (par ex.\ $x = 2$ et $x = 10$~; éviter $0$ et $1$, qui
        masquent trop de \omterm{ex:g1:subtraction:difference}{différences})~;
  \item si les résultats diffèrent ne serait-ce qu'une fois, les
        expressions ne sont \emph{pas} égales --- un contre-exemple
        suffit à tuer une affirmation générale~;
  \item si les résultats coïncident, l'égalité n'est que
        \emph{plausible}~: une preuve demande de l'algèbre
        (développer, réduire).
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:literal:test}
$2(x + 3)$ est-elle égale à $2x + 6$ pour tout $x$~? Développer le
prouve~: $2(x+3) = 2x + 2 \times 3 = 2x + 6$. $(x + 1)^2$ est-elle
égale à $x^2 + 1$~? Tester $x = 2$~: $(2+1)^2 = 9$ mais
$2^2 + 1 = 5$. Non --- et un seul contre-exemple est une preuve
complète du «~non~».
\end{example}

\section{Produire des expressions}

\begin{example}[De la situation à l'expression]\label{ex:g7:literal:produce}
Un taxi facture $4$ euros plus $2$ euros par kilomètre. Pour un
trajet de $k$ kilomètres le prix est
\[
p = 4 + 2k \quad\text{euros}.
\]
Pour $k = 7$~: $p = 4 + 14 = 18$ euros. L'expression \emph{est} la
réponse générale~; substituer \omterm{def:g2:mult:def}{produit} chaque cas particulier.
\end{example}

\begin{example}[Tours de magie numériques]\label{ex:g7:literal:trick}
«~Pense à un nombre, double-le, ajoute $10$, divise par $2$,
soustrais ton nombre.~» Appeler le nombre $n$ et suivre les
étapes~:
\[
n \ \to\ 2n \ \to\ 2n + 10 \ \to\ \frac{2n + 10}{2} = n + 5 \ \to\
n + 5 - n = 5 .
\]
Tout le monde obtient $5$, quel que soit $n$ --- les lettres
expliquent la magie.
\end{example}

\section{Problème~: l'oracle des allumettes}

\begin{problem}[{Devoir maison --- les motifs qui grandissent et leur
terme de rang $n$~: trois formules pour un même motif, et le motif
célèbre qui trahit à la sixième étape}]
\label{pb:g7:literal:1}
Aligner des allumettes en une rangée de carrés~: un carré, puis
deux, puis trois\,\dots\ Combien d'allumettes faudra-t-il à la
centième figure~? Les compter une à une prendrait tout le week-end~;
une \emph{formule en $n$} répond instantanément pour toutes les
figures à la fois --- c'est exactement à cela que servent les lettres
(\cref{ex:g7:literal:formulas}). Mais attention~: ce problème se
termine par un motif célèbre qui joue franc jeu pendant cinq étapes,
puis trahit tous ceux qui lui avaient fait confiance.

\medskip
\noindent\textbf{Partie I --- Une rangée de carrés.}
La figure $1$ est un carré d'allumettes ($4$ allumettes)~; chaque
figure suivante ajoute un carré de plus, partageant un côté avec le
précédent.
\begin{enumerate}
  \item Dessiner les figures $1$ à $4$ et compter leurs allumettes.
  \item Expliquer pourquoi chaque nouveau carré coûte exactement $3$
        allumettes supplémentaires, et s'en servir pour prévoir le
        compte de la figure $10$ sans aucune formule.
  \item Justifier la formule~: la figure $n$ demande $3n + 1$
        allumettes. (D'où vient le «~$+1$~» solitaire~?) La
        vérifier sur la question 1.
  \item Amir raisonne autrement~: «~$4$ allumettes pour le premier
        carré, puis $3$ pour chacun des $n - 1$ autres~», ce qui
        donne $4 + 3(n - 1)$. Développer et réduire son expression
        (\cref{prop:g7:literal:reduce})~: est-il d'accord avec la
        question 3~?
  \item Léna compte d'une troisième façon~: «~$2n$ allumettes
        horizontales, et $n + 1$ verticales~». Écrire sa formule, la
        réduire, et conclure. Répondre alors à la question du
        début~: combien d'allumettes pour la figure $100$~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Triangles, tables, et marche
arrière.}
\begin{enumerate}[resume]
  \item Une rangée de triangles d'allumettes (chaque nouveau
        triangle s'appuie sur le précédent)~: compter les allumettes
        pour $1$, $2$, $3$ triangles, puis justifier la formule
        $2n + 1$ pour $n$ triangles.
  \item Marche arrière~: avec exactement $49$ allumettes, combien de
        triangles la rangée peut-elle compter~? (Défaire la formule
        étape par étape.)
  \item Des tables de banquet carrées accueillent un convive par
        côté libre. Lorsque $n$ tables sont accolées en une seule
        longue rangée, expliquer pourquoi la formule du nombre de
        places est $2n + 2$, et la vérifier pour $1$, $2$, $3$
        tables.
  \item Une réception attend $30$ convives. Combien de tables faut-il
        pour une seule rangée~? Le traiteur répartit plutôt les mêmes
        tables en \emph{deux} rangées de sept~: combien de convives
        tiennent alors~? Expliquer le gain à l'aide de la formule.
  \item Motif de points~: la figure $n$ est un L de points, haut de
        $n$ points et large de $n$ points (le coin compté une seule
        fois). Compter les points des figures $1$, $2$, $3$, trouver
        la formule, et nommer les nombres qu'elle \omterm{def:g2:mult:def}{produit}.
        (Emboîtés les uns dans les autres, ces L pavent un carré ---
        une histoire poursuivie dans \cref{pb:g8:equations:1}.)
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Croire, tester, démontrer.}
\begin{enumerate}[resume]
  \item Un camarade affirme que la figure $n$ de la partie I demande
        $3(n + 2) - 5$ allumettes. Trancher en développant et en
        réduisant --- aucun test n'est nécessaire.
  \item «~Élever au carré rend les nombres plus grands, donc
        $n^2 \geq n$ pour tout nombre $n$.~» Tester l'affirmation
        pour $n = 2$, $3$, $10$~; puis trouver un nombre pour lequel
        elle échoue. Qu'ont prouvé les trois tests réussis, et que
        prouve l'unique échec (\cref{met:g7:literal:test})~?
  \item Marquer des points sur un cercle et tracer \emph{tous} les
        \omterm{def:g6:lines:objects}{segments} qui les joignent, puis compter les régions
        délimitées à l'intérieur du cercle. Le faire pour $2$, $3$,
        $4$ et $5$ points (placer les points irrégulièrement, pour
        que jamais trois \omterm{def:g6:lines:objects}{segments} ne se croisent en un même point
        intérieur). Noter les comptes. Quelle formule le motif
        souffle-t-il~?
  \item Passer maintenant à $6$ points, en dessinant grand et
        irrégulièrement, et en comptant avec le plus grand soin (les
        petites régions sont nombreuses). Combien de régions
        trouve-t-on --- et qu'est-il arrivé à la conjecture~?
  \item La morale, en deux phrases~: qu'aurait-il fallu pour
        \emph{démontrer} la conjecture du doublement, et pourquoi
        les formules d'allumettes des parties I et II n'ont-elles
        jamais couru pareil danger~? Fêter cela par une prévision
        démontrée~: combien d'allumettes pour la rangée de $2\,026$
        carrés~?
\end{enumerate}
\end{problem}
