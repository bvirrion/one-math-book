\section*{Capítulo \ref{ch:g7:negatives} --- Los números relativos}

\begin{solution}{exo:g7:negatives:1}
Opuestos: $-7$; \ $+3.5$; \ $0$ (opuesto de sí mismo); \ $-12$. Dos
números son opuestos cuando su \omterm{def:g1:addition:def}{suma} es $0$.
\end{solution}

\begin{solution}{exo:g7:negatives:2}
$-4 < 1$; \quad $-4 > -7$; \quad $-2.5 < -2.4$; \quad $0 > -0.1$.
\end{solution}

\begin{solution}{exo:g7:negatives:3}
$-3.1 < -3 < -0.5 < 0 < 1 < 2.5$.
\end{solution}

\begin{solution}{exo:g7:negatives:4}
$(-5) + (-9) = -14$; \quad
$(-12) + (+7) = -5$; \quad
$(+3.5) + (-1.5) = +2$; \quad
$(-6) + (+6) = 0$.
\end{solution}

\begin{solution}{exo:g7:negatives:5}
$4 - 11 = 4 + (-11) = -7$.

$(-5) - (+3) = (-5) + (-3) = -8$.

$(-7) - (-10) = (-7) + (+10) = +3$.

$0 - (-8) = 0 + (+8) = +8$.
\end{solution}

\begin{solution}{exo:g7:negatives:6}
$(+7) + (-12) + (+3) + (-8) = (+10) + (-20) = -10$ (los positivos
juntos y los negativos juntos).

$(-5.5) + (+9) + (-4.5) + (+1) = (-10) + (+10) = 0$.
\end{solution}

\begin{solution}{exo:g7:negatives:7}
$(-3) + 8 - 12 = 5 - 12 = -7$: por la noche hace $-7$ grados.
\end{solution}

\begin{solution}{exo:g7:negatives:9}

\begin{solution}{exo:g7:negatives:10}
$(-12) + 7 - 4 = -5 - 4 = -9$: está a $-9$ m. Para llegar a la superficie
(altitud $0$) tiene que subir $0 - (-9) = 9$ m.
\end{solution}

\begin{solution}{exo:g7:negatives:11}
$D(-2, -2)$: la misma primera coordenada que $A$ y la misma segunda coordenada que
$C$. Lados: $AB = 4$ (de $-2$ a $2$) y $BC = 3$ (de $1$ a
$-2$), así que el \omterm{def:g6:measure:perimeter}{perímetro} es $2 \times (4 + 3) = 14$.
\end{solution}

\begin{solution}{exo:g7:negatives:12}
La diagonal completa $0, -2, -4$ da la \omterm{def:g1:addition:def}{suma} mágica
$0 + (-2) + (-4) = -6$. Entonces, casilla a casilla:
arriba a la izquierda $= -6 - (-7) - 0 = 1$; centro derecha $= -6 - (-3) - (-2) =
-1$; abajo a la derecha $= -6 - 0 - (-1) = -5$; abajo en el centro
$= -6 - (-7) - (-2) = 3$.
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|}
\hline
$1$ & $-7$ & $0$ \\
\hline
$-3$ & $-2$ & $-1$ \\
\hline
$-4$ & $3$ & $-5$ \\
\hline
\end{tabular}
\end{center}
Cada fila, cada columna y cada diagonal suman $-6$.
\end{solution}

\begin{solution}{pb:g7:negatives:1}
\textbf{1.} $4\,808 - (-430) = 4\,808 + 430 = 5\,238$ m
(\cref{thm:g7:negatives:sub}: restar un negativo es
sumar).

\textbf{2.} Saldo:
\[
(-45) + 120 + (-60) + 15 = 75 + (-60) + 15 = 15 + 15 = 30
\text{ euros}.
\]
Para devolver los $-45$ iniciales a $0$, hay que ingresar su opuesto:
$45$ euros.

\textbf{3.} $(-44) - (-100) = (-44) + 100 = 56$: César vivió
$56$ años. (Aquí no hay trampa: los dos años están del mismo lado del
cambio de era.)

\textbf{4.} El año $0$: contar desde $-63$ hasta $+14$ pasa
por $0$, pero el calendario de los historiadores salta directamente del
$1$ a.\,C. al $1$ d.\,C. --- el año $0$ nunca existió, así que el recuento
ingenuo sobra un año.

\textbf{5.} El $63$ a.\,C. pasa a ser $-(63 - 1) = -62$. Entonces,
$14 - (-62) = 14 + 62 = 76$ años: exactamente la cifra de los
historiadores. En la recta reetiquetada de los astrónomos no hay huecos,
así que la resta ordinaria da duraciones verdaderas.

\textbf{6.} $1$ a.\,C. $\to 0$; \ $10$ a.\,C. $\to -9$; \ $476$
d.\,C. $\to +476$; \ $753$ a.\,C. $\to -752$.

\textbf{7.} $476 - (-752) = 476 + 752 = 1\,228$ años.

\textbf{8.} El $12$ a.\,C. es $-11$. Apariciones siguientes:
$-11 + 76 = 65$, después $141$ y después $217$: los años $65$, $141$ y
$217$ d.\,C. La anterior: $-11 - 76 = -87$, que es el
año $87 + 1 = 88$ a.\,C. de los historiadores.

\textbf{9.} El $5$ a.\,C. es $-4$, así que $5 - (-4) = 9$ años --- no
$10$. El año \omterm{def:g1:counting:zero}{cero} que falta ataca siempre que una duración cruza
el cambio de era.

\textbf{10.} El $3$ a.\,C. es $-2$; diez años después es $-2 + 10 = +8$:
el niño cumplió diez años en el año $8$ d.\,C.

\textbf{11.} Entre $-7$ y $-2$: $(-2) - (-7) = 5$. Entre
$-3.5$ y $4.5$: $4.5 - (-3.5) = 8$.

\textbf{12.} Agrupando opuestos y números amigos:
\[
3 + (-5) + 2 + (-1) + 4
= \bigl(3 + 2\bigr) + (-5) + (-1) + 4
= 0 + (-1) + 4 = +3 :
\]
el ascensor termina en la planta $3$. El orden no importa: una \omterm{def:g1:addition:def}{suma} de
\omterm{def:g7:negatives:def}{números relativos} se puede reorganizar libremente (cada movimiento se recorre
sobre la misma recta y el paseo total es el mismo).

\textbf{13.} Variación neta:
$2 + (-4) + 1 + 0 + (-3) + 5 + (-2) = -1$ grado. Empezando en
$-2$: temperatura final $-2 + (-1) = -3$ grados. En general,
el valor final es el valor inicial más la \omterm{def:g1:addition:def}{suma} de todas las
variaciones --- no hace falta seguir las subidas y bajadas una a una.

\textbf{14.} Volver al punto de partida significa variación neta $0$: las subidas y
las bajadas tienen que cancelarse exactamente. Las cinco variaciones conocidas suman
$4 + (-7) + 2 + 3 + (-5) = -3$, así que la sexta es la
opuesta: $+3$ (trescientos metros de subida).

\textbf{15.} Con los astrónomos: $509$ a.\,C. $\to -508$, y
$476 - (-508) = 984$ años. Los $509 + 476 = 985$ del historiador
cuentan un año de más --- una vez más el año \omterm{def:g1:counting:zero}{cero} fantasma,
incluido en silencio en la \omterm{def:g1:addition:def}{suma} pero ausente de la historia.
(Como comprobación, las dos eras: República de $-508$ a $-26$, $482$
años; Imperio de $-26$ a $+476$, $502$ años;
$482 + 502 = 984$.)
\end{solution}
