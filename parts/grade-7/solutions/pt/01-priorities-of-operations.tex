\begin{solution}{exo:g7:priorities:1}
$5 + 3 \times 6 = 5 + 18 = 23$.

$(5 + 3) \times 6 = 8 \times 6 = 48$.

$30 - 12 \div 4 = 30 - 3 = 27$.

$(30 - 12) \div 4 = 18 \div 4 = 4.5$.
\end{solution}

\begin{solution}{exo:g7:priorities:2}
$24 - 6 + 2 = 18 + 2 = 20$ (da esquerda para a direita).

$24 \div 6 \times 2 = 4 \times 2 = 8$ (da esquerda para a direita).

$24 - 6 \times 2 = 24 - 12 = 12$ (a multiplicação primeiro).

$24 \div (6 \times 2) = 24 \div 12 = 2$.
\end{solution}

\begin{solution}{exo:g7:priorities:3}
$7 + 3 \times (10 - 6) = 7 + 3 \times 4 = 7 + 12 = 19$.

$40 - (2 + 3) \times 4 = 40 - 5 \times 4 = 40 - 20 = 20$.

$60 \div \bigl(2 \times (7 - 4)\bigr) = 60 \div (2 \times 3)
= 60 \div 6 = 10$.
\end{solution}

\begin{solution}{exo:g7:priorities:4}
$\dfrac{18 + 6}{8} = \dfrac{24}{8} = 3$; \quad
$\dfrac{30}{7 - 2} = \dfrac{30}{5} = 6$; \quad
$\dfrac{5 \times 8}{4 + 6} = \dfrac{40}{10} = 4$.
\end{solution}

\begin{solution}{exo:g7:priorities:6}
``A \omterm{ex:g1:subtraction:difference}{diferença} entre $50$ e o \omterm{def:g2:mult:def}{produto} de $6$ por $7$'':
$50 - 6 \times 7 = 50 - 42 = 8$ (sem parênteses, a prioridade faz o
trabalho).

``O \omterm{def:g2:mult:def}{produto} da \omterm{def:g1:addition:def}{soma} de $4$ com $5$ pela \omterm{ex:g1:subtraction:difference}{diferença} entre $10$ e $8$'':
$(4 + 5) \times (10 - 8) = 9 \times 2 = 18$.
\end{solution}

\begin{solution}{exo:g7:priorities:9}
$k = 1$: $12 + 4 \times 1 = 16$ e $16 \times 1 = 16$ --- coincidem (é
por isso que $k = 1$ é um teste ruim!). $k = 2$:
$12 + 4 \times 2 = 20$, mas $16 \times 2 = 32$: diferentes. Logo as
duas expressões \emph{não} são iguais em geral. O erro
$12 + 4 \times k = 16 \times k$ vem de somar $12 + 4$ primeiro,
ignorando a prioridade do $\times$.
\end{solution}

\begin{solution}{exo:g7:priorities:10}
\emph{1.} Preço do grupo: $12 + 6 \times n$. Para $n = 5$:
$12 + 30 = 42$ reais.

\emph{2.} Preço individual: $9n$. Comparando: $n = 3$: $30$ contra
$27$ (individual mais barato); $n = 4$: $36$ contra $36$ (igual);
$n = 5$: $42$ contra $45$ (grupo mais barato). A partir de $5$ pessoas,
a tarifa de grupo ganha (cada pessoa a mais custa $6$ em vez de $9$,
então a \omterm{ex:g1:subtraction:difference}{diferença} só aumenta).
\end{solution}

\begin{solution}{pb:g7:priorities:1}
\textbf{1.} $7 \times 102 = 7 \times (100 + 2) = 700 + 14 = 714$;
\quad $9 \times 98 = 9 \times (100 - 2) = 900 - 18 = 882$.

\textbf{2.} $17 \times 13 + 17 \times 7 = 17 \times (13 + 7) =
17 \times 20 = 340$; \quad
$45 \times 101 - 45 = 45 \times (101 - 1) = 45 \times 100 = 4\,500$.

\textbf{3.} $5 \times 87 = 870 \div 2 = 435$. Legitimidade:
$5 = 10 \div 2$, então multiplicar por $5$ é multiplicar por $10$ e
depois dividir por $2$ (a ordem de $\times$ e $\div$, no mesmo nível e
da esquerda para a direita, faz o \omterm{def:g3:division:remainder}{resto}).

\textbf{4.} $25 \times 32 = 3\,200 \div 4 = 800$, porque
$25 = 100 \div 4$. E
$99 \times 47 = (100 - 1) \times 47 = 4\,700 - 47 = 4\,653$:
distributividade sobre a \omterm{ex:g1:subtraction:difference}{diferença}
(\cref{thm:g7:priorities:distributivity}).

\textbf{5.} $8 \times (5 + 3) = 64$ e
$8 \times 5 + 8 \times 3 = 40 + 24 = 64$: batem --- é a
distributividade. Mas $8 \times (5 \times 3) = 8 \times 15 = 120$,
enquanto $(8 \times 5) \times (8 \times 3) = 40 \times 24 = 960$: não
batem. O fator se espalha pelos \emph{termos de uma \omterm{def:g1:addition:def}{soma}}, e não pelos
fatores de um \omterm{def:g2:mult:def}{produto}.

\textbf{6.}
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{ccl}
$13$ & $24$ & mantida ($13$ \omterm{def:g3:numbers:evenodd}{ímpar}) \\
$6$ & $48$ & riscada ($6$ par) \\
$3$ & $96$ & mantida \\
$1$ & $192$ & mantida \\
\end{tabular}
\end{center}
\omterm{def:g1:addition:def}{Soma} dos números da direita que sobraram: $24 + 96 + 192 = 312$, e de
fato $13 \times 24 = 312$.

\textbf{7.} Dividindo $21$ ao meio: $21, 10, 5, 2, 1$; dobrando $17$:
$17, 34, 68, 136, 272$. As esquerdas pares $10$ e $2$ são riscadas; os
sobreviventes dão $17 + 68 + 272 = 357$. Confere:
$21 \times 17 = 357$.

\textbf{8.} Dividindo $16$ ao meio: $16, 8, 4, 2, 1$; dobrando $25$:
$25, 50, 100, 200, 400$. Todos os números da esquerda, exceto o $1$
final, são pares: sobrevive uma única linha, e a resposta é $400$. Uma
coluna da esquerda feita só de dobros mantém apenas a última linha ---
o método se reduz a dobrar.

\textbf{9.} $13 = 12 + 1$, então, pela distributividade,
$13 \times 24 = 12 \times 24 + 1 \times 24$; e
$12 \times 24 = 6 \times 48$, pois dividir um fator ao meio dobrando o
outro deixa o \omterm{def:g2:mult:def}{produto} inalterado
($12 \times 24 = 6 \times 2 \times 24 = 6 \times 48$). O \omterm{def:g3:numbers:evenodd}{número ímpar}
$13$ destacou uma cópia do número da direita: a primeira linha põe $24$
no cofre.

\textbf{10.} $6 \times 48 = 3 \times 96$ (\omterm{def:g3:division:half}{metade} e dobro, sem nada para
o cofre: $6$ é par). $3 \times 96 = 2 \times 96 + 96 =
1 \times 192 + 96$: o \omterm{def:g3:numbers:evenodd}{ímpar} $3$ guarda $96$. Por fim,
$1 \times 192 = 192$: o \omterm{def:g3:numbers:evenodd}{ímpar} $1$ guarda $192$ e a tabela acaba. Total
guardado: $24 + 96 + 192 = 312$. Uma linha guarda o número da direita
exatamente quando o número da esquerda é \omterm{def:g3:numbers:evenodd}{ímpar} --- então a receita
``mantenha as linhas de esquerda \omterm{def:g3:numbers:evenodd}{ímpar} e some'' é precisamente a
escrituração da distributividade.

\textbf{11.} $1 + 4 + 8 = 13$. Para o $21$, as linhas sobreviventes
foram as de $17$, $68 = 4 \times 17$ e $272 = 16 \times 17$: os dobros
$1 + 4 + 16 = 21$.

\textbf{12.} $25 = 16 + 8 + 1$ e $42 = 32 + 8 + 2$.

\textbf{13.} Tome o maior número de dobro que não ultrapasse o alvo e
subtraia-o; o que sobra é menor e é menor também que o número de dobro
recém-usado (senão um maior teria cabido). Repetindo, o \omterm{def:g3:division:remainder}{resto} encolhe
estritamente e tem de chegar a $0$: o processo para, e os dobros usados
são todos diferentes. Para $100$: $64$ cabe (sobram $36$), $32$ cabe
(sobram $4$), $4$ cabe (sobra $0$): $100 = 64 + 32 + 4$.

\textbf{14.} Tabela de dobros de $35$: $1 \to 35$, $2 \to 70$,
$4 \to 140$, $8 \to 280$, $16 \to 560$. Como $19 = 16 + 2 + 1$,
escolhemos essas linhas: $560 + 70 + 35 = 665$. Confere:
$19 \times 35 = 665$.

\textbf{15.} As mesmas três do escriba e do camponês:
$24 + 96 + 192$ --- os dobros de $24$ selecionados por
$13 = 1 + 4 + 8$. Escrever números como \omterm{def:g1:addition:def}{somas} de dobros é exatamente
escrevê-los com $0$ e $1$; a história completa está no
\cref{pb:g8:powers:1}.
\end{solution}
