\section*{Capítulo \ref{ch:g7:areas} --- Áreas e volumes}

\begin{solution}{exo:g7:areas:1}
$8 \times 5 = 40$ cm$^2$; \qquad $6.5 \times 4 = 26$ m$^2$.
\end{solution}

\begin{solution}{exo:g7:areas:2}
$A = 10 \times 4 = 40$ cm$^2$. A resposta não é $10 \times 6 = 60$
porque o lado de $6$ cm é inclinado: a fórmula usa a \emph{altura}
(distância perpendicular), que é $4$ cm.
\end{solution}

\begin{solution}{exo:g7:areas:3}
$\frac{12 \times 7}{2} = 42$ cm$^2$; \quad
$\frac{9 \times 6}{2} = 27$; \quad
$\frac{5.5 \times 2}{2} = 5.5$ m$^2$.
\end{solution}

\begin{solution}{exo:g7:areas:4}
Disco: $\pi \times 5^2 = 25\pi \approx 79$ cm$^2$. Semidisco de \omterm{def:g3:shapes:shapes}{raio}
$10$: $\frac{\pi \times 10^2}{2} = 50\pi \approx 157$ cm$^2$.
\end{solution}

\begin{solution}{exo:g7:areas:5}
\omterm{def:g6:measure:perimeter}{Perímetro}: $2\pi \times 3 = 6\pi \approx 18.8$ cm (um comprimento, em
cm). \omterm{def:g6:measure:area}{Área}: $\pi \times 3^2 = 9\pi \approx 28.3$ cm$^2$ (uma superfície,
em cm$^2$).
\end{solution}

\begin{solution}{exo:g7:areas:6}
\omterm{def:g7:areas:prism}{Prisma}: $24 \times 10 = 240$ cm$^3$. \omterm{def:g7:areas:prism}{Cilindro}:
$\pi \times 3^2 \times 8 = 72\pi \approx 226$ cm$^3$.
\end{solution}

\begin{solution}{exo:g7:areas:7}
$\frac{8 \times h}{2} = 28$, então $4h = 28$ e $h = 7$ cm.
\end{solution}

\begin{solution}{exo:g7:areas:8}
Os três \omterm{def:g2:mult:def}{produtos} coincidem porque cada um calcula a \emph{mesma}
grandeza --- a \omterm{def:g6:measure:area}{área} do triângulo. Isso dá uma conferência prática das
construções e uma maneira de calcular uma altura a partir de outro par
base--altura (usada no \cref{exo:g7:areas:12}).
\end{solution}

\begin{solution}{exo:g7:areas:9}
Os dois triângulos têm \omterm{def:g6:measure:area}{áreas} $\frac{30 \times 20}{2} = 300$ m$^2$ e
$\frac{18 \times 20}{2} = 180$ m$^2$: total $480$ m$^2$. Fórmula geral
sugerida: para lados paralelos $a$ e $b$ à distância $h$,
\[
A = \frac{(a + b) \times h}{2}
\]
--- a fórmula do trapézio (aqui, $\frac{(30+18)\times 20}{2} = 480$).
\end{solution}

\begin{solution}{exo:g7:areas:10}
\omterm{def:g6:measure:volume}{Volume} de água: $\pi \times 6^2 \times 15 = 540\pi \approx 1\,696$
cm$^3$. Base da caixa: $18 \times 12 = 216$ cm$^2$. Altura atingida:
$\frac{540\pi}{216} = 2.5\pi \approx 7.9$ cm.
\end{solution}

\begin{solution}{exo:g7:areas:11}
\omterm{def:g3:shapes:shapes}{Raios} $15$ e $20$ cm. \omterm{def:g6:measure:area}{Áreas}: $225\pi \approx 707$ cm$^2$ e
$400\pi \approx 1257$ cm$^2$. Preço por $100$ cm$^2$:
$\frac{9}{7.07} \approx 1.27$ reais e
$\frac{14}{12.57} \approx 1.11$ reais. A pizza grande vale mais a pena
--- as \omterm{def:g6:measure:area}{áreas} crescem com o \emph{quadrado} do \omterm{def:g4:geometry:circle}{diâmetro}.
\end{solution}

\begin{solution}{exo:g7:areas:12}
Com os catetos: $A = \frac{6 \times 8}{2} = 24$ cm$^2$. Com a
hipotenusa como base: $24 = \frac{10 \times h}{2} = 5h$, então
$h = 4.8$ cm.
\end{solution}

\begin{solution}{pb:g7:areas:1}
\textbf{1.} Quadrado: $8 \times 8 = 64$. Retângulo:
$5 \times 13 = 65$. As mesmas quatro peças parecem cobrir $64$
unidades de \omterm{def:g6:measure:area}{área} numa figura e $65$ na outra: uma unidade de \omterm{def:g6:measure:area}{área} foi
criada do nada --- supostamente.

\textbf{2.} Cada triângulo: $\frac{8 \times 3}{2} = 12$. Cada
trapézio: $\frac{(5 + 3) \times 5}{2} = 20$.

\textbf{3.} $12 + 12 + 20 + 20 = 64$: as peças somam $64$, então elas
podem preencher exatamente o quadrado --- e \emph{não} podem preencher
o retângulo de $65$ unidades. A imagem do retângulo tem de conter um
buraco.

\textbf{4.} Mesma inclinação significaria $3$ ao longo de $8$
proporcional a $2$ ao longo de $5$; \omterm{def:g2:mult:def}{produtos} cruzados:
$3 \times 5 = 15$ e $2 \times 8 = 16$. Não são iguais: a hipotenusa e a
borda inclinada \emph{não} se alinham. A ``diagonal'' do retângulo é
uma mentira --- duas inclinações ligeiramente diferentes se encontrando
num ângulo raso.

\textbf{5.} Ao longo da falsa diagonal, as quatro peças deixam uma
falha comprida e extremamente fina --- uma lasca em forma de
\omterm{def:g7:central:parallelogram}{paralelogramo} bem achatado --- cuja \omterm{def:g6:measure:area}{área} é exatamente a unidade que
falta, $65 - 64 = 1$. Esticada ao longo de uma diagonal de cerca de
$13$ unidades, a largura média dela é de uns
$1 \div 13 \approx 0.08$ unidades: num desenho de verdade, mais fina
que o traço de lápis que a esconde.

\textbf{6.} Cada triângulo tem base $6$ cm e altura $4$ cm --- a
distância entre as \omterm{def:g6:lines:perp}{paralelas}, esteja o ápice onde estiver ---, então
cada \omterm{def:g6:measure:area}{área} é $\frac{6 \times 4}{2} = 12$ cm$^2$
(\cref{thm:g7:areas:triangle}). Deslizar o ápice ao longo da paralela
muda a forma, nunca a base nem a altura: \omterm{def:g6:measure:area}{áreas} iguais para sempre.

\textbf{7.} A diagonal corta o trapézio num triângulo de base $B$ e
noutro de base $b$, ambos de altura $h$ (a distância entre os lados
paralelos):
\[
A = \frac{B \times h}{2} + \frac{b \times h}{2}
= \frac{(B + b) \times h}{2} .
\]

\textbf{8.} $A = \frac{(9 + 5) \times 4}{2} = \frac{56}{2} = 28$
cm$^2$.

\textbf{9.} A média dos lados paralelos é $\frac{B + b}{2}$, então
(média) $\times h = \frac{(B+b) \times h}{2}$: a mesma fórmula,
fatorada de outro jeito. Confere: média $= \frac{9+5}{2} = 7$, e
$7 \times 4 = 28$ cm$^2$.

\textbf{10.} A pilha inclinada contém exatamente as mesmas cartas ---
as mesmas camadas, cada uma de mesma \omterm{def:g6:measure:area}{área} ---, logo o mesmo \omterm{def:g6:measure:volume}{volume}. Em
$V = B \times h$, a altura tem de ser medida \emph{perpendicularmente}
à base (a altura vertical da pilha), e não ao longo da inclinação:
exatamente como a \omterm{def:g6:measure:area}{área} do \omterm{def:g7:central:parallelogram}{paralelogramo} quer a altura perpendicular, e
não o lado inclinado (\cref{exo:g7:areas:2}), uma dimensão acima.

\textbf{11.} Anel $= \pi \times 5^2 - \pi \times 3^2 = 25\pi - 9\pi =
16\pi \approx 50$ m$^2$ (\cref{thm:g7:areas:disk}).

\textbf{12.} Um disco de \omterm{def:g6:measure:area}{área} $16\pi$ tem \omterm{def:g3:shapes:shapes}{raio} $4$, já que
$\pi \times 4^2 = 16\pi$. Os \omterm{def:g3:shapes:shapes}{raios} $3$, $4$, $5$: o terno mais famoso
da matemática, protagonista do \cref{ch:g8:pythagoras}.

\textbf{13.} Uma pizza de $40$ cm: \omterm{def:g3:shapes:shapes}{raio} $20$, \omterm{def:g6:measure:area}{área}
$\pi \times 400 \approx 1\,257$ cm$^2$. Duas pizzas de $28$ cm: \omterm{def:g3:shapes:shapes}{raio}
$14$, juntas $2 \times \pi \times 196 = 392\pi \approx 1\,232$ cm$^2$.
A pizza grande sozinha é \emph{mais} pizza que as duas médias.

\textbf{14.} Multiplicar o \omterm{def:g3:shapes:shapes}{raio} por $1.4$ multiplica a \omterm{def:g6:measure:area}{área} por
$1.4^2 = 1.96$ --- quase o dobro. O fator exato é o número cujo
quadrado é $2$, cerca de $1.414$: o número da diagonal do
\cref{ex:g8:pythagoras:diagonal}. (Regra de bolso: \omterm{def:g3:division:half}{metade} mais larga é
quase o dobro de pizza.)

\textbf{15.} $13 \times 13 = 169$, mas $8 \times 21 = 168$: desta vez
uma unidade de \omterm{def:g6:measure:area}{área} \emph{desaparece} --- as peças se sobrepõem numa
lasca fina, em vez de deixarem uma falha. Com termos consecutivos $3,
5, 8, 13, 21$ da sequência de Hemachandra e Fibonacci, os \omterm{def:g2:mult:def}{produtos}
alternam entre uma unidade a mais e uma a menos que o quadrado
($5 \times 13 = 65 = 8^2 + 1$, e depois
$8 \times 21 = 168 = 13^2 - 1$): o mágico alternadamente ``ganha'' e
``perde'' uma unidade. As \omterm{def:g6:measure:area}{áreas} nunca mentem: peças que somam $64$
cobrem $64$, deslizem para onde deslizarem --- toda unidade que falta
ou sobra está escondida numa lasca, esperando uma conferência de
inclinação para ser exposta.
\end{solution}
