\section*{Capítulo \ref{ch:g7:negatives} --- Números negativos}

\begin{solution}{exo:g7:negatives:1}
Opostos: $-7$; \ $+3.5$; \ $0$ (oposto de si mesmo); \ $-12$. Dois
números são opostos quando a \omterm{def:g1:addition:def}{soma} deles é $0$.
\end{solution}

\begin{solution}{exo:g7:negatives:2}
$-4 < 1$; \quad $-4 > -7$; \quad $-2.5 < -2.4$; \quad $0 > -0.1$.
\end{solution}

\begin{solution}{exo:g7:negatives:3}
$-3.1 < -3 < -0.5 < 0 < 1 < 2.5$.
\end{solution}

\begin{solution}{exo:g7:negatives:4}
$(-5) + (-9) = -14$; \quad
$(-12) + (+7) = -5$; \quad
$(+3.5) + (-1.5) = +2$; \quad
$(-6) + (+6) = 0$.
\end{solution}

\begin{solution}{exo:g7:negatives:5}
$4 - 11 = 4 + (-11) = -7$.

$(-5) - (+3) = (-5) + (-3) = -8$.

$(-7) - (-10) = (-7) + (+10) = +3$.

$0 - (-8) = 0 + (+8) = +8$.
\end{solution}

\begin{solution}{exo:g7:negatives:6}
$(+7) + (-12) + (+3) + (-8) = (+10) + (-20) = -10$ (os positivos
juntos, os negativos juntos).

$(-5.5) + (+9) + (-4.5) + (+1) = (-10) + (+10) = 0$.
\end{solution}

\begin{solution}{exo:g7:negatives:7}
$(-3) + 8 - 12 = 5 - 12 = -7$: são $-7$ graus à noite.
\end{solution}

\begin{solution}{exo:g7:negatives:8}
Marcação livre; $E(0, -3)$ está sobre o eixo vertical (a primeira
coordenada dele é $0$).
\end{solution}

\begin{solution}{exo:g7:negatives:9}
Moscou: $(-4) - (-13) = -4 + 13 = 9$ graus de \omterm{ex:g1:subtraction:difference}{diferença}. Helsinque:
$2 - (-9) = 11$ graus.
\end{solution}

\begin{solution}{exo:g7:negatives:10}
$(-12) + 7 - 4 = -5 - 4 = -9$: ela está em $-9$ m. Para chegar à
superfície (altitude $0$), precisa subir $0 - (-9) = 9$ m.
\end{solution}

\begin{solution}{exo:g7:negatives:11}
$D(-2, -2)$: a mesma primeira coordenada de $A$ e a mesma segunda
coordenada de $C$. Lados: $AB = 4$ (de $-2$ a $2$) e $BC = 3$ (de $1$
até $-2$), então o \omterm{def:g6:measure:perimeter}{perímetro} é $2 \times (4 + 3) = 14$.
\end{solution}

\begin{solution}{exo:g7:negatives:12}
A diagonal completa $0, -2, -4$ dá a \omterm{def:g1:addition:def}{soma} mágica
$0 + (-2) + (-4) = -6$. Depois, casa por casa: canto superior esquerdo
$= -6 - (-7) - 0 = 1$; centro à direita $= -6 - (-3) - (-2) = -1$;
canto inferior direito $= -6 - 0 - (-1) = -5$; centro de baixo
$= -6 - (-7) - (-2) = 3$.
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|}
\hline
$1$ & $-7$ & $0$ \\
\hline
$-3$ & $-2$ & $-1$ \\
\hline
$-4$ & $3$ & $-5$ \\
\hline
\end{tabular}
\end{center}
Toda linha, coluna e diagonal \omterm{def:g1:addition:def}{soma} $-6$.
\end{solution}

\begin{solution}{pb:g7:negatives:1}
\textbf{1.} $4\,808 - (-430) = 4\,808 + 430 = 5\,238$ m
(\cref{thm:g7:negatives:sub}: subtrair um negativo é somar).

\textbf{2.} Saldo:
\[
(-45) + 120 + (-60) + 15 = 75 + (-60) + 15 = 15 + 15 = 30
\text{ reais}.
\]
Para levar os $-45$ iniciais a $0$, bastava depositar o oposto deles:
$45$ reais.

\textbf{3.} $(-44) - (-100) = (-44) + 100 = 56$: César viveu $56$
anos. (Sem armadilha aqui: os dois anos estão do mesmo lado da
fronteira das eras.)

\textbf{4.} O ano $0$: contar de $-63$ até $+14$ passa por $0$, mas o
calendário dos historiadores salta direto de $1$ a.C. para $1$ d.C. ---
o ano $0$ nunca existiu, então a contagem ingênua é um ano longa
demais.

\textbf{5.} $63$ a.C. vira $-(63 - 1) = -62$. Então
$14 - (-62) = 14 + 62 = 76$ anos: exatamente o número dos
historiadores. Na reta renomeada dos astrônomos não há buracos, então a
subtração comum dá durações verdadeiras.

\textbf{6.} $1$ a.C. $\to 0$; \ $10$ a.C. $\to -9$; \
$476$ d.C. $\to +476$; \ $753$ a.C. $\to -752$.

\textbf{7.} $476 - (-752) = 476 + 752 = 1\,228$ anos.

\textbf{8.} $12$ a.C. é $-11$. Aparições seguintes:
$-11 + 76 = 65$, depois $141$, depois $217$: os anos $65$ d.C.,
$141$ d.C. e $217$ d.C. Antes: $-11 - 76 = -87$, que é o ano
$87 + 1 = 88$ a.C. dos historiadores.

\textbf{9.} $5$ a.C. é $-4$, então $5 - (-4) = 9$ anos --- e não $10$.
O ano \omterm{def:g1:counting:zero}{zero} que falta ataca sempre que uma duração atravessa a fronteira
das eras.

\textbf{10.} $3$ a.C. é $-2$; dez anos depois é $-2 + 10 = +8$: a
criança completou dez anos em $8$ d.C.

\textbf{11.} Entre $-7$ e $-2$: $(-2) - (-7) = 5$. Entre $-3.5$ e
$4.5$: $4.5 - (-3.5) = 8$.

\textbf{12.} Agrupando opostos e amigos:
\[
3 + (-5) + 2 + (-1) + 4
= \bigl(3 + 2\bigr) + (-5) + (-1) + 4
= 0 + (-1) + 4 = +3 :
\]
o elevador termina no andar $3$. A ordem não importa: uma \omterm{def:g1:addition:def}{soma} de
\omterm{def:g7:negatives:def}{números relativos} pode ser reorganizada à vontade (cada movimento é
percorrido sobre a mesma reta, e a caminhada total é a mesma).

\textbf{13.} Variação total:
$2 + (-4) + 1 + 0 + (-3) + 5 + (-2) = -1$ grau. Partindo de $-2$:
temperatura final $-2 + (-1) = -3$ graus. Em geral, o valor final é o
valor inicial mais a \omterm{def:g1:addition:def}{soma} de todas as variações --- não é preciso
acompanhar as subidas e descidas uma a uma.

\textbf{14.} Voltar ao ponto de partida significa variação total $0$:
as subidas e descidas têm de se cancelar exatamente. As cinco variações
conhecidas somam $4 + (-7) + 2 + 3 + (-5) = -3$, então a sexta variação
é a oposta: $+3$ (trezentos metros de subida).

\textbf{15.} Pelos astrônomos: $509$ a.C. $\to -508$, e
$476 - (-508) = 984$ anos. Os $509 + 476 = 985$ do historiador contam
um ano a mais --- mais uma vez o ano \omterm{def:g1:counting:zero}{zero} fantasma, incluído em
silêncio na adição, mas ausente da história. (Como conferência, as duas
eras: República de $-508$ a $-26$, $482$ anos; Império de $-26$ a
$+476$, $502$ anos; $482 + 502 = 984$.)
\end{solution}
