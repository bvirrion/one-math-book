\section*{الفصل \ref{ch:g7:areas} --- المساحات والأحجام}

\begin{solution}{exo:g7:areas:1}
$8 \times 5 = 40$ سم$^2$؛ \qquad $6.5 \times 4 = 26$ م$^2$.
\end{solution}

\begin{solution}{exo:g7:areas:2}
$A = 10 \times 4 = 40$ سم$^2$. والجواب ليس
$10 \times 6 = 60$ لأن الضلع $6$ سم مائل: فالصيغة
تستعمل \emph{الارتفاع} (المسافة العمودية)، وهو $4$ سم.
\end{solution}

\begin{solution}{exo:g7:areas:3}
$\frac{12 \times 7}{2} = 42$ سم$^2$؛ \quad
$\frac{9 \times 6}{2} = 27$؛ \quad
$\frac{5.5 \times 2}{2} = 5.5$ م$^2$.
\end{solution}

\begin{solution}{exo:g7:areas:4}
القرص: $\pi \times 5^2 = 25\pi \approx 79$ سم$^2$. ونصف القرص ذو \omterm{def:g3:shapes:shapes}{نصف القطر}
$10$: $\frac{\pi \times 10^2}{2} = 50\pi \approx 157$ سم$^2$.
\end{solution}

\begin{solution}{exo:g7:areas:5}
\omterm{def:g6:measure:perimeter}{المحيط}: $2\pi \times 3 = 6\pi \approx 18.8$ سم (وهو طول، بالسنتيمتر).
\omterm{def:g6:measure:area}{والمساحة}: $\pi \times 3^2 = 9\pi \approx 28.3$ سم$^2$ (وهي سطح، بوحدة
سم$^2$).
\end{solution}

\begin{solution}{exo:g7:areas:6}
\omterm{def:g7:areas:prism}{الموشور}: $24 \times 10 = 240$ سم$^3$. \omterm{def:g7:areas:prism}{والأسطوانة}:
$\pi \times 3^2 \times 8 = 72\pi \approx 226$ سم$^3$.
\end{solution}

\begin{solution}{exo:g7:areas:7}
$\frac{8 \times h}{2} = 28$، إذن $4h = 28$ و $h = 7$ سم.
\end{solution}

\begin{solution}{exo:g7:areas:8}
تتوافق \omterm{def:g2:mult:def}{الجداءات} الثلاثة لأن كل واحد منها يحسب المقدار
\emph{نفسه} --- وهو \omterm{def:g6:measure:area}{مساحة} المثلث. وهذا يعطي تحقّقًا عمليًّا
من الإنشاءات، وطريقة لحساب اِرتفاع من زوج
قاعدة--اِرتفاع آخر (مستعملة في \cref{exo:g7:areas:12}).
\end{solution}

\begin{solution}{exo:g7:areas:9}
للمثلثين مساحتان $\frac{30 \times 20}{2} = 300$ م$^2$
و $\frac{18 \times 20}{2} = 180$ م$^2$: \omterm{def:g1:addition:def}{والمجموع} $480$ م$^2$. والصيغة
العامّة المقترحة: من أجل ضلعين \omterm{def:g6:lines:perp}{متوازيين} $a$ و $b$ على مسافة $h$،
\[
A = \frac{(a + b) \times h}{2}
\]
--- وهي صيغة شبه المنحرف (وهنا $\frac{(30+18)\times 20}{2} = 480$).
\end{solution}

\begin{solution}{exo:g7:areas:10}
\omterm{def:g6:measure:volume}{حجم} الماء: $\pi \times 6^2 \times 15 = 540\pi \approx 1\,696$
سم$^3$. وقاعدة الصندوق: $18 \times 12 = 216$ سم$^2$. والارتفاع المبلوغ:
$\frac{540\pi}{216} = 2.5\pi \approx 7.9$ سم.
\end{solution}

\begin{solution}{exo:g7:areas:11}
نصفا القطرين $15$ و $20$ سم. والمساحتان: $225\pi \approx 707$ سم$^2$ و
$400\pi \approx 1257$ سم$^2$. والثمن لكل $100$ سم$^2$:
$\frac{9}{7.07} \approx 1.27$ يورو و
$\frac{14}{12.57} \approx 1.11$ يورو. فالبيتزا الكبيرة صفقة
أفضل --- فالمساحات تنمو مع \emph{مربّع} \omterm{def:g4:geometry:circle}{القطر}.
\end{solution}

\begin{solution}{exo:g7:areas:12}
بضلعَي القائمة: $A = \frac{6 \times 8}{2} = 24$ سم$^2$. وبالوتر
قاعدةً: $24 = \frac{10 \times h}{2} = 5h$، إذن
$h = 4.8$ سم.
\end{solution}

\begin{solution}{pb:g7:areas:1}
\textbf{1.} المربع: $8 \times 8 = 64$. والمستطيل:
$5 \times 13 = 65$. \omterm{def:g6:lines:objects}{فالقطع} الأربع نفسها تبدو وكأنها تغطّي $64$
وحدة مربعة في شكل و $65$ في الآخر: أي إن وحدة
\omterm{def:g6:measure:area}{مساحة} خُلقت من العدم --- زعمًا.

\textbf{2.} كل مثلث: $\frac{8 \times 3}{2} = 12$. وكل
شبه منحرف: $\frac{(5 + 3) \times 5}{2} = 20$.

\textbf{3.} $12 + 12 + 20 + 20 = 64$: \omterm{def:g1:addition:def}{فمجموع} \omterm{def:g6:lines:objects}{القطع} $64$،
إذن تستطيع ملء المربع بالضبط --- و\emph{لا} تستطيع ملء
المستطيل ذي $65$ وحدة. فلا بدّ أن تحوي صورة المستطيل
ثقبًا.

\textbf{4.} الميل نفسه يعني أن $3$ على مدى $8$
متناسب مع $2$ على مدى $5$؛ والجداءان المتقاطعان:
$3 \times 5 = 15$ و $2 \times 8 = 16$. وهما غير متساويين: إذن
الوتر والحرف المائل \emph{لا} يصطفّان. و«\omterm{def:g4:geometry:circle}{قطر}»
المستطيل كذبة --- أي ميلان مختلفان قليلًا يلتقيان بزاوية ضحلة.

\textbf{5.} على طول \omterm{def:g4:geometry:circle}{القطر} الزائف تترك \omterm{def:g6:lines:objects}{القطع} الأربع
فجوة طويلة رفيعة جدًّا --- شظية على شكل \omterm{def:g7:central:parallelogram}{متوازي أضلاع}
مسطّح جدًّا --- مساحتها بالضبط الوحدة المربعة الناقصة
$65 - 64 = 1$. وممدودةً على \omterm{def:g4:geometry:circle}{قطر} طوله نحو
$13$ وحدة، يكون عرضها المتوسط نحو
$1 \div 13 \approx 0.08$ وحدة: أي، على رسم حقيقي، أرفع من
خطّ القلم الذي يخفيها.

\textbf{6.} كل مثلث قاعدته $6$ سم واِرتفاعه $4$ سم ---
وهو المسافة بين \omterm{def:g6:lines:perp}{المتوازيين}، أينما وقع الرأس ---
إذن \omterm{def:g6:measure:area}{مساحة} كل واحد $\frac{6 \times 4}{2} = 12$ سم$^2$
(\cref{thm:g7:areas:triangle}). وإزلاق الرأس على
\omterm{def:g6:lines:perp}{الموازي} يغيّر الشكل، لا القاعدة ولا الارتفاع:
فالمساحات متساوية إلى الأبد.

\textbf{7.} \omterm{def:g4:geometry:circle}{القطر} يقطع شبه المنحرف إلى مثلث
قاعدته $B$ ومثلث قاعدته $b$، كلاهما باِرتفاع $h$ (وهو
المسافة بين الضلعين \omterm{def:g6:lines:perp}{المتوازيين}):
\[
A = \frac{B \times h}{2} + \frac{b \times h}{2}
= \frac{(B + b) \times h}{2} .
\]

\textbf{8.} $A = \frac{(9 + 5) \times 4}{2} = \frac{56}{2} =
28$ سم$^2$.

\textbf{9.} متوسط الضلعين \omterm{def:g6:lines:perp}{المتوازيين} هو
$\frac{B + b}{2}$، إذن (المتوسط) $\times h =
\frac{(B+b) \times h}{2}$: أي الصيغة نفسها، معمّلة
تعميلًا مختلفًا. والتحقّق: المتوسط $= \frac{9+5}{2} = 7$، و
$7 \times 4 = 28$ سم$^2$.

\textbf{10.} الكدس المائل يحوي الورق نفسه بالضبط
--- الطبقات نفسها، وكل واحدة \omterm{def:g6:measure:area}{بالمساحة} نفسها --- إذن \omterm{def:g6:measure:volume}{الحجم}
نفسه. وفي $V = B \times h$، يجب أن يُقاس الارتفاع
\emph{عموديًّا} على القاعدة (أي اِرتفاع الكدس العمودي)،
لا على طول الميل: تمامًا كما تريد \omterm{def:g6:measure:area}{مساحة} \omterm{def:g7:central:parallelogram}{متوازي
الأضلاع} الارتفاعَ العمودي، لا الضلع المائل
(\cref{exo:g7:areas:2})، ببُعد إضافي.

\textbf{11.} الحلقة $= \pi \times 5^2 - \pi \times 3^2 = 25\pi -
9\pi = 16\pi \approx 50$ م$^2$ (\cref{thm:g7:areas:disk}).

\textbf{12.} القرص ذو \omterm{def:g6:measure:area}{المساحة} $16\pi$ نصف قطره $4$، لأن
$\pi \times 4^2 = 16\pi$. \omterm{def:g3:shapes:shapes}{وأنصاف الأقطار} $3$ و $4$ و $5$: أشهر
ثلاثية في الرياضيات، تلعب البطولة في
\cref{ch:g8:pythagoras}.

\textbf{13.} البيتزا الواحدة $40$ سم: نصف قطرها $20$،
ومساحتها $\pi \times 400 \approx 1\,257$ سم$^2$. والبيتزاتان $28$ سم:
\omterm{def:g3:shapes:shapes}{نصف القطر} $14$، ومعًا
$2 \times \pi \times 196 = 392\pi \approx 1\,232$ سم$^2$. فالبيتزا
الكبيرة الواحدة بيتزا \emph{أكثر} من المتوسطتين.

\textbf{14.} تكبير \omterm{def:g3:shapes:shapes}{نصف القطر} بالمعامل $1.4$ يكبّر \omterm{def:g6:measure:area}{المساحة} بالمعامل
$1.4^2 = 1.96$ --- أي الضعف تقريبًا. والمعامل المضبوط هو
العدد الذي مربّعه $2$، ونحوه $1.414$: وهو عدد \omterm{def:g4:geometry:circle}{القطر}
في \cref{ex:g8:pythagoras:diagonal}. (وقاعدة تقريبية: نصف زيادة
في العرض يعطي ضعف البيتزا تقريبًا.)

\textbf{15.} $13 \times 13 = 169$، لكن $8 \times 21 = 168$:
ففي هذه المرة \emph{تختفي} وحدة مربعة --- إذ تتراكب
\omterm{def:g6:lines:objects}{القطع} في شظية رفيعة بدل أن تترك فجوة. ومع
الحدود المتتالية $3, 5, 8, 13, 21$ من متتالية
هيماشندرا--فيبوناتشي، تتقلّب \omterm{def:g2:mult:def}{الجداءات} بين واحد
زائد وواحد ناقص عن المربع ($5 \times 13 = 65 =
8^2 + 1$، ثم $8 \times 21 = 168 = 13^2 - 1$): فالساحر
«يكسب» و«يخسر» وحدة بالتناوب. والمساحات لا تكذب أبدًا:
\omterm{def:g6:lines:objects}{فالقطع} التي مجموعها $64$ تغطّي $64$، أينما اِنزلقت --- وكل
وحدة ناقصة أو زائدة مختبئة في شظية، تنتظر فحص ميل
يفضحها.
\end{solution}
