\section*{الفصل \ref{ch:g7:priorities} --- أولويات العمليات}

\begin{solution}{exo:g7:priorities:1}
$5 + 3 \times 6 = 5 + 18 = 23$.

$(5 + 3) \times 6 = 8 \times 6 = 48$.

$30 - 12 \div 4 = 30 - 3 = 27$.

$(30 - 12) \div 4 = 18 \div 4 = 4.5$.
\end{solution}

\begin{solution}{exo:g7:priorities:2}
$24 - 6 + 2 = 18 + 2 = 20$ (من اليسار إلى اليمين).

$24 \div 6 \times 2 = 4 \times 2 = 8$ (من اليسار إلى اليمين).

$24 - 6 \times 2 = 24 - 12 = 12$ (الضرب أولًا).

$24 \div (6 \times 2) = 24 \div 12 = 2$.
\end{solution}

\begin{solution}{exo:g7:priorities:3}
$7 + 3 \times (10 - 6) = 7 + 3 \times 4 = 7 + 12 = 19$.

$40 - (2 + 3) \times 4 = 40 - 5 \times 4 = 40 - 20 = 20$.

$60 \div \bigl(2 \times (7 - 4)\bigr) = 60 \div (2 \times 3)
= 60 \div 6 = 10$.
\end{solution}

\begin{solution}{exo:g7:priorities:4}
$\dfrac{18 + 6}{8} = \dfrac{24}{8} = 3$; \quad
$\dfrac{30}{7 - 2} = \dfrac{30}{5} = 6$; \quad
$\dfrac{5 \times 8}{4 + 6} = \dfrac{40}{10} = 4$.
\end{solution}

\begin{solution}{exo:g7:priorities:5}
$8 \times 102 = 8 \times (100 + 2) = 800 + 16 = 816$.

$25 \times 99 = 25 \times (100 - 1) = 2500 - 25 = 2475$.

$14 \times 12 + 14 \times 8 = 14 \times (12 + 8) = 14 \times 20 = 280$.

$37 \times 45 - 37 \times 35 = 37 \times (45 - 35) = 37 \times 10 =
370$.
\end{solution}

\begin{solution}{exo:g7:priorities:6}
«\omterm{ex:g1:subtraction:difference}{فرق} $50$ \omterm{def:g2:mult:def}{وجداء} $6$ في $7$»:
$50 - 6 \times 7 = 50 - 42 = 8$ (لا حاجة إلى أقواس، فالأولوية
تقوم بالعمل).

«\omterm{def:g2:mult:def}{جداء} \omterm{def:g1:addition:def}{مجموع} $4$ و $5$ في \omterm{ex:g1:subtraction:difference}{فرق} $10$ و
$8$»: $(4 + 5) \times (10 - 8) = 9 \times 2 = 18$.
\end{solution}

\begin{solution}{exo:g7:priorities:8}
(أ) دون أقواس: $4 + 2 \times 5 - 1 = 4 + 10 - 1 = 13$.

(b) $(4 + 2) \times 5 - 1 = 30 - 1 = 29$.

(c) $(4 + 2) \times (5 - 1) = 6 \times 4 = 24$.

\begin{solution}{exo:g7:priorities:9}
$k = 1$: $12 + 4 \times 1 = 16$ و $16 \times 1 = 16$ --- فهما يتوافقان
(ولهذا كانت $k = 1$ اِختبارًا سيّئًا!). ومن أجل $k = 2$:
$12 + 4 \times 2 = 20$ لكن $16 \times 2 = 32$: مختلفان. إذن العبارتان
\emph{غير} متساويتين عمومًا. والخطأ
$12 + 4 \times k = 16 \times k$ يأتي من جمع $12 + 4$ أولًا،
وإهمال أولوية $\times$.
\end{solution}

\begin{solution}{exo:g7:priorities:10}
\emph{1.} ثمن المجموعة: $12 + 6 \times n$. ومن أجل $n = 5$:
$12 + 30 = 42$ يورو.

\emph{2.} الثمن الفردي: $9n$. والمقارنة: $n = 3$: $30$ مقابل $27$
(فالفردي أرخص)؛ و $n = 4$: $36$ مقابل $36$ (متساويان)؛ و $n = 5$: $42$ مقابل
$45$ (فالمجموعة أرخص). واِبتداءً من $5$ أشخاص يفوز سعر المجموعة
(فكل شخص إضافي يكلّف $6$ بدل $9$، فلا تزداد الفجوة إلا اِتساعًا).
\end{solution}

\end{solution}

\begin{solution}{pb:g7:priorities:1}
\textbf{1.} $7 \times 102 = 7 \times (100 + 2) = 700 + 14 =
714$؛ \quad و $9 \times 98 = 9 \times (100 - 2) = 900 - 18 = 882$.

\textbf{2.} $17 \times 13 + 17 \times 7 = 17 \times (13 + 7) =
17 \times 20 = 340$؛ \quad و $45 \times 101 - 45 = 45 \times (101
- 1) = 45 \times 100 = 4\,500$.

\textbf{3.} $5 \times 87 = 870 \div 2 = 435$. والمشروعية:
$5 = 10 \div 2$، إذن الضرب في $5$ هو الضرب في $10$
ثم القسمة على $2$ (وترتيب $\times$ و $\div$ في
المستوى نفسه، من اليسار إلى اليمين، يقوم \omterm{def:g3:division:remainder}{بالباقي}).

\textbf{4.} $25 \times 32 = 3\,200 \div 4 = 800$، لأن
$25 = 100 \div 4$. وكذلك
$99 \times 47 = (100 - 1) \times 47 = 4\,700 - 47 = 4\,653$:
وهي التوزيعية على \omterm{ex:g1:subtraction:difference}{الفرق}
(\cref{thm:g7:priorities:distributivity}).

\textbf{5.} $8 \times (5 + 3) = 64$ و $8 \times 5 + 8 \times 3
= 40 + 24 = 64$: فهما يتطابقان --- وتلك هي التوزيعية. لكن
$8 \times (5 \times 3) = 8 \times 15 = 120$، بينما
$(8 \times 5) \times (8 \times 3) = 40 \times 24 = 960$: فلا
تطابق. فالعامل ينتشر على \emph{حدود \omterm{def:g1:addition:def}{المجموع}}، لا
على عوامل \omterm{def:g2:mult:def}{الجداء}.

\textbf{6.}
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{ccl}
$13$ & $24$ & يُحتفظ به ($13$ فردي) \\
$6$ & $48$ & يُشطب ($6$ زوجي) \\
$3$ & $96$ & يُحتفظ به \\
$1$ & $192$ & يُحتفظ به \\
\end{tabular}
\end{center}
\omterm{def:g1:addition:def}{مجموع} الأعداد اليمنى الناجية: $24 + 96 + 192 = 312$، وفعلًا
$13 \times 24 = 312$.

\textbf{7.} تنصيف $21$: $21, 10, 5, 2, 1$؛ ومضاعفة $17$:
$17, 34, 68, 136, 272$. والعددان الأيسران الزوجيان $10$ و $2$ يُشطبان؛
والناجون $17 + 68 + 272 = 357$. والتحقّق:
$21 \times 17 = 357$.

\textbf{8.} تنصيف $16$: $16, 8, 4, 2, 1$؛ ومضاعفة $25$:
$25, 50, 100, 200, 400$. وكل عدد أيسر عدا الأخير $1$
زوجي: فينجو سطر واحد، والجواب $400$. والعمود
الأيسر المصنوع من \omterm{def:g5:division:multiple}{مضاعفات} صرفة لا يحتفظ إلا بسطره الأخير ---
فتنحصر الطريقة في المضاعفة وحدها.

\textbf{9.} $13 = 12 + 1$، إذن بالتوزيعية
$13 \times 24 = 12 \times 24 + 1 \times 24$؛ و
$12 \times 24 = 6 \times 48$ لأن تنصيف عامل مع
مضاعفة الآخر يترك \omterm{def:g2:mult:def}{الجداء} دون تغيير
($12 \times 24 = 6 \times 2 \times 24 = 6 \times 48$). والعدد
الأيسر الفردي $13$ فصل نسخة واحدة من العدد الأيمن: فالسطر
الأول يودع $24$.

\textbf{10.} $6 \times 48 = 3 \times 96$ (تنصيف ومضاعفة، دون
إيداع: فالعدد $6$ زوجي). و $3 \times 96 = 2 \times 96 + 96 =
1 \times 192 + 96$: فالفردي $3$ يودع $96$. وأخيرًا
$1 \times 192 = 192$: فالفردي $1$ يودع $192$ وينتهي الجدول.
\omterm{def:g1:addition:def}{ومجموع} المودَع: $24 + 96 + 192 = 312$. والسطر يودع عدده
الأيمن بالضبط حين يكون عدده الأيسر فرديًّا --- إذن
وصفة «اِحتفظ بالسطور ذات اليسار الفردي واجمع» هي بالضبط
محاسبة التوزيعية.

\textbf{11.} $1 + 4 + 8 = 13$. ومن أجل $21$، كانت السطور الناجية
سطور $17$ و $68 = 4 \times 17$ و $272 = 16 \times 17$: أي
\omterm{def:g5:division:multiple}{المضاعفات} $1 + 4 + 16 = 21$.

\textbf{12.} $25 = 16 + 8 + 1$ و $42 = 32 + 8 + 2$.

\textbf{13.} خذ أكبر عدد مضاعفة لا يتجاوز
الهدف واطرحه؛ فما يبقى أصغر، وأصغر
من عدد المضاعفة المستعمل للتوّ (وإلا لدخل أكبر
منه). وبالتكرار، يتقلّص \omterm{def:g3:division:remainder}{الباقي} تمامًا
ويجب أن يبلغ $0$: فتتوقف العملية، وأعداد المضاعفة المستعملة
كلها مختلفة. ومن أجل $100$: يدخل $64$ (فيبقى $36$)، ويدخل $32$
(فيبقى $4$)، ويدخل $4$ (فيبقى $0$): أي $100 = 64 + 32 + 4$.

\textbf{14.} جدول مضاعفة العدد $35$: $1 \to 35$، و $2 \to 70$،
و $4 \to 140$، و $8 \to 280$، و $16 \to 560$. وبما أن
$19 = 16 + 2 + 1$، تُختار تلك السطور:
$560 + 70 + 35 = 665$. والتحقّق: $19 \times 35 = 665$.

\textbf{15.} الثلاث نفسها التي عند الكاتب والفلاح:
$24 + 96 + 192$ --- وهي \omterm{def:g5:division:multiple}{مضاعفات} العدد $24$ التي اِختارها
$13 = 1 + 4 + 8$. وكتابة الأعداد مجاميعَ \omterm{def:g5:division:multiple}{مضاعفات} هي
بالضبط كتابتها بالرمزين $0$ و $1$؛ والقصة الكاملة تُروى
في \cref{pb:g8:powers:1}.
\end{solution}
