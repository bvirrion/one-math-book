\begin{solution}{exo:g7:stats:1}

\begin{center}

\begin{tabular}{l|ccccc|c}

\hline
\omterm{def:g7:stats:frequency}{التعداد} & $4$ & $3$ & $7$ & $3$ & $3$ & $20$ \\

\end{center}
\end{solution}

\begin{solution}{exo:g7:stats:2}
التواترات: $\frac{4}{20} = 20\,\%$، و $\frac{3}{20} = 15\,\%$،
و $\frac{7}{20} = 35\,\%$، و $15\,\%$، و $15\,\%$. \omterm{def:g1:addition:def}{والمجموع}:
$20 + 15 + 35 + 15 + 15 = 100\,\%$. \checkmark
\end{solution}

\begin{solution}{exo:g7:stats:3}
الاستطلاع الأول: $\frac{30}{50} = 0.6 = 60\,\%$. والثاني:
$\frac{45}{90} = 0.5 = 50\,\%$. فالاستطلاع الأول يبيّن النسبة
الأعلى، وإن كان فيه أسر أقلّ عندها سيارة واحدة \omterm{def:g7:stats:frequency}{بالتعداد}
المطلق.
\end{solution}

\begin{solution}{exo:g7:stats:4}
$\intco{5}{15}$: $8, 12, 9, 14, 11$ --- \omterm{def:g7:stats:frequency}{التعداد} $5$.
$\intco{15}{25}$: $23, 22, 18, 16$ --- \omterm{def:g7:stats:frequency}{التعداد} $4$.
$\intco{25}{35}$: $31, 27, 29, 25, 33$ --- \omterm{def:g7:stats:frequency}{التعداد} $5$.
$\intco{35}{45}$: $35$ --- \omterm{def:g7:stats:frequency}{التعداد} $1$. \omterm{def:g1:addition:def}{والمجموع} $15$. \checkmark
\end{solution}

\begin{solution}{exo:g7:stats:5}
أربعة أعمدة اِرتفاعاتها $8$ و $12$ و $6$ و $4$ فوق القيم $0$ و $1$
و $2$ و $3$ (والتدريج كل $2$ يعمل جيدًا).
\end{solution}

\begin{solution}{exo:g7:stats:6}
\omterm{def:g1:addition:def}{المجموع} $30$ أسرة. والتواترات: $\frac{8}{30}$، و $\frac{12}{30}$،
و $\frac{6}{30}$، و $\frac{4}{30}$. والزوايا ($\times 360^\circ$): $96^\circ$،
و $144^\circ$، و $72^\circ$، و $48^\circ$؛ \omterm{def:g1:addition:def}{والمجموع}
$96 + 144 + 72 + 48 = 360^\circ$. \checkmark
\end{solution}

\begin{solution}{exo:g7:stats:7}
النسب المئوية: الصيف $\frac{162}{360} = 45\,\%$؛ والربيع
$\frac{90}{360} = 25\,\%$؛ والخريف والشتاء
$\frac{54}{360} = 15\,\%$ لكل منهما. ومن بين $60$ شخصًا، اِختار الصيف
$0.45 \times 60 = 27$ منهم.
\end{solution}

\begin{solution}{exo:g7:stats:8}
القسم أ: $\frac{12}{30} = 40\,\%$. والقسم ب: $\frac{14}{40} = 35\,\%$.
فالقسم أ فيه نسبة الماشين أعلى، وإن كان في القسم ب
ماشون أكثر \omterm{def:g7:stats:frequency}{بالتعداد}.
\end{solution}

\begin{solution}{exo:g7:stats:9}
بمحور يبدأ من $9\,700$، تكون للأعمدة الثلاثة اِرتفاعات
مرئية $100$ و $300$ و $400$ وحدة: فيبدو الأخير أربعة أمثال
الأول، موحيًا بأن المبيعات تضاعفت أربع مرات. وبمحور من $0$، تكون الأعمدة
متساوية تقريبًا (فمن $9\,800$ إلى $10\,100$ اِرتفاع نحو $3\,\%$):
فالصورة الأمينة تبيّن مبيعات مسطّحة تقريبًا.
\end{solution}

\begin{solution}{exo:g7:stats:10}
\omterm{def:g1:addition:def}{مجموع} التواترات $1$: إذن $f = 1 - 0.35 - 0.4 = 0.25$. والتعدادات (\omterm{def:g1:addition:def}{والمجموع}
$80$): $0.35 \times 80 = 28$؛ و $0.4 \times 80 = 32$؛
و $0.25 \times 80 = 20$. والتحقّق: $28 + 32 + 20 = 80$.
\end{solution}

\begin{solution}{exo:g7:stats:11}
البنات: $55\,\%$ من $400 = 220$؛ والأولاد: $180$. والبنات الآكلات في المطعم:
$0.4 \times 220 = 88$؛ والأولاد: $0.6 \times 180 = 108$. \omterm{def:g1:addition:def}{والمجموع}:
$88 + 108 = 196$ تلميذًا، أي\ $\frac{196}{400} = 49\,\%$ من
المدرسة --- بين $40\,\%$ و $60\,\%$، وأقرب إلى نسبة البنات
لأن البنات أكثر عددًا.
\end{solution}

\begin{solution}{pb:g7:stats:1}
\textbf{1.} التعدادات: $120 > 90$، فتفوز المدرسة أ. والتواترات:
$\frac{120}{400} = 0.30 = 30\,\%$ مقابل
$\frac{90}{250} = 0.36 = 36\,\%$: فتفوز المدرسة ب. والجائزة
يجب أن تتبع \omterm{def:g7:stats:frequency}{التواتر} --- فالمدرسة ب تعيد تدوير حصة أكبر
مما تستعمل؛ والمدرسة أ لا تشرب إلا عصيرًا أكثر.

\textbf{2.} المعلّمون: $\frac{18}{30} = 0.60 = 60\,\%$. والتلاميذ:
$\frac{210}{600} = 0.35 = 35\,\%$. ومعًا:
$\frac{18 + 210}{630} = \frac{228}{630} \approx 0.36 =
36\,\%$. والتلاميذ أكثر عددًا عشرين مرة، فيُجذب
\omterm{def:g7:stats:frequency}{التواتر} المشترك كله تقريبًا إلى جهتهم ---
فالمجموعة الكبيرة تحمل الوزن الكبير.

\textbf{3.} يجب أن يبلغ \omterm{def:g1:addition:def}{مجموع} التواترات $100\,\%$
(\cref{def:g7:stats:frequency})، لكن
$45 + 30 + 20 + 10 = 105$: فقطاع واحد على الأقلّ خطأ.

\textbf{4.} \omterm{def:g7:stats:frequency}{التعداد} الموافق \omterm{def:g7:stats:frequency}{للتواتر} $0.15$:
$0.15 \times 40 = 6$. والتعدادات إلى الآن: $14 + 10 + 6 = 30$، إذن
\omterm{def:g7:stats:frequency}{التعداد} الأخير $40 - 30 = 10$، وتواتره
$\frac{10}{40} = 0.25$. (والتحقّق: $0.35 + 0.25 + 0.15 + 0.25 =
1$.)

\textbf{5.} الزوايا: $0.40 \times 360 = 144^\circ$،
و $0.35 \times 360 = 126^\circ$، و $0.25 \times 360 = 90^\circ$؛
و $144 + 126 + 90 = 360^\circ$: فتنغلق الدائرة.

\textbf{6.} $\frac{2.4}{10} = 0.24 = 24\,\%$ من بطاقات الاقتراع
عادت.

\textbf{7.} كانت أدلّة الهاتف وسجلّات السيارات تُدرج
أُسَرًا ثرية في معظمها، وكان تصويتها يختلف عن تصويت
البلد؛ وكانت الأجوبة $2.4$ مليون تعدادًا عملاقًا مسحوبًا
من السكان الخطأ. وكما في السؤال 1: فالمهمّ ليس
كم تعدّ، بل من تعدّ --- \omterm{def:g7:stats:frequency}{فالتواتر} المحسوب على عيّنة
مشوّهة يصف العيّنة لا البلد.

\textbf{8.} التأييد الحقيقي:
$0.7 \times 1\,000 + 0.3 \times 9\,000 = 700 + 2\,700 = 3\,400$
مؤيّدًا من $10\,000$: أي $34\,\%$. والاستطلاع:
$0.7 \times 500 + 0.3 \times 500 = 350 + 150 = 500$ من
$1\,000$: أي $50\,\%$ --- أي أعلى بستّ عشرة نقطة، لأن
الأثرياء نصف العيّنة وعُشر المدينة.

\textbf{9.} الأجوبة: $0.4 \times 200 = 80$ من غير الراضين و
$0.1 \times 800 = 80$ من الراضين، إذن $160$ جوابًا منها $80$
من غير الراضين: فعدم الرضا المقيس $\frac{80}{160} = 50\,\%$. وعدم
الرضا الحقيقي: $\frac{200}{1\,000} = 20\,\%$. ولم يكذب أحد ---
فغير الراضين يجيبون أكثر لا غير، والاستطلاع يسمعهم
أعلى.

\textbf{10.} النسب الحقيقية: عُشر أثرياء، إذن $100$
مقابلة ثرية و $900$ متواضعة:
$0.7 \times 100 + 0.3 \times 900 = 70 + 270 = 340$ من
$1\,000$: فيتوقّع الاستطلاع المُصلَح $34\,\%$ --- وهي حقيقة
السؤال 8. فخمسون ألف مقابلة حسنة الاختيار تغلب مليونين
سيّئَي الاختيار.

\textbf{11.} المستشفى أ: الخفيفة $\frac{90}{100} = 90\,\%$؛ والخطيرة
$\frac{280}{400} = 70\,\%$؛ وإجمالًا
$\frac{90 + 280}{500} = \frac{370}{500} = 74\,\%$.

\textbf{12.} المستشفى ب: الخفيفة $\frac{340}{400} = 85\,\%$؛
والخطيرة $\frac{65}{100} = 65\,\%$؛ وإجمالًا
$\frac{340 + 65}{500} = \frac{405}{500} = 81\,\%$. فالمستشفى أ
يفوز في الحالات الخفيفة ($90 > 85$) \emph{وفي} الحالات الخطيرة
($70 > 65$).

\textbf{13.} وإجمالًا، يبيّن ب $81\,\%$ مقابل $74\,\%$ للمستشفى أ:
فالمستشفى الذي يخسر في \emph{كل} فئة يفوز في
\omterm{def:g1:addition:def}{المجموع}. (ولهذا الانقلاب اِسم: مفارقة سيمبسون.)

\textbf{14.} الخلطتان متعاكستان: فمرضى أ خطيرون في
معظمهم ($400$ من $500$)، ومرضى ب خفيفون في معظمهم ($400$ من $500$).
والحالات الخطيرة تُشفى أقلّ أينما عولجت، فيُجرّ رقم أ
الإجمالي إلى الأسفل بسبب الحالات الصعبة التي يقبلها ---
فالإجمالي مزيج موزون، والأوزان تختلف بين
المستشفيين. والمريض الخطير يجب أن يختار المستشفى أ: فالنسبة $70\,\%$
تغلب $65\,\%$ في السطر الوحيد الذي يعنيه.

\textbf{15.} فمثلًا: «مقارنتك تخلط المرضى الخفيفين
والخطيرين، والمستشفيان يعالجان خلطتين متعاكستين ---
فبالتقسيم حسب الخطورة، يشفي المستشفى أ حصة أعلى من
\emph{النوعين} معًا. والأرقام الإجمالية لا تُقارن إلا حين تكون
المجموعات وراءها متشابهة؛ وإلا فالأوزان، لا
الجودة، هي التي تقرّر الفائز.»
\end{solution}
