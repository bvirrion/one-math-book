\begin{solution}{exo:g7:negatives:10}

\begin{solution}{exo:g7:negatives:11}
$D(-2, -2)$: पहला निर्देशांक $A$ वाला, दूसरा $C$ वाला। भुजाएँ:
$AB = 4$ ($-2$ से $2$ तक) और $BC = 3$ ($1$ से नीचे $-2$ तक),
इसलिए \omterm{def:g6:measure:perimeter}{परिमाप} $2 \times (4 + 3) = 14$ है।
\end{solution}

\begin{solution}{exo:g7:negatives:12}
पूरा विकर्ण $0, -2, -4$ जादुई \omterm{def:g1:addition:def}{योगफल}
$0 + (-2) + (-4) = -6$ देता है। फिर, ख़ाना-दर-ख़ाना:
ऊपर-बाएँ $= -6 - (-7) - 0 = 1$; बीच-दाएँ $= -6 - (-3) - (-2) =
-1$; नीचे-दाएँ $= -6 - 0 - (-1) = -5$; नीचे-बीच
$= -6 - (-7) - (-2) = 3$।
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|}
\hline
$1$ & $-7$ & $0$ \\
\hline
$-3$ & $-2$ & $-1$ \\
\hline
$-4$ & $3$ & $-5$ \\
\hline
\end{tabular}
\end{center}
हर पंक्ति, हर स्तंभ और हर विकर्ण का \omterm{def:g1:addition:def}{योगफल} $-6$ है।
\end{solution}

\begin{solution}{pb:g7:negatives:1}
\textbf{1.} $4\,808 - (-430) = 4\,808 + 430 = 5\,238$ m
(\cref{thm:g7:negatives:sub}: ऋण संख्या घटाना यानी
जोड़ना)।

\textbf{2.} शेष:
\[
(-45) + 120 + (-60) + 15 = 75 + (-60) + 15 = 15 + 15 = 30
\text{ यूरो}.
\]
शुरू के $-45$ को $0$ पर लाने के लिए उसकी \omterm{def:g7:negatives:def}{विपरीत संख्या} जमा करो:
$45$ यूरो।

\textbf{3.} $(-44) - (-100) = (-44) + 100 = 56$: सीज़र
$56$ साल जिया। (यहाँ कोई जाल नहीं: दोनों वर्ष युग-सीमा की एक ही
ओर हैं।)

\textbf{4.} वर्ष $0$: $-63$ से $+14$ तक गिनने पर बीच में
$0$ आता है, पर इतिहासकारों का पंचांग ईसा-पूर्व $1$ से सीधे ईसा
के बाद $1$ पर कूद जाता है --- वर्ष $0$ कभी था ही नहीं, इसलिए
सीधी गिनती एक साल ज़्यादा निकलती है।

\textbf{5.} ईसा-पूर्व $63$, $-(63 - 1) = -62$ बन जाता है। फिर
$14 - (-62) = 14 + 62 = 76$ साल: ठीक इतिहासकारों वाला
आँकड़ा। खगोलविदों की नई नामवाली रेखा में कोई ख़ाली जगह नहीं,
इसलिए साधारण घटाव सही अवधि दे देता है।

\textbf{6.} ईसा-पूर्व $1 \to 0$; \ ईसा-पूर्व $10 \to -9$; \ ईसा
के बाद $476 \to +476$; \ ईसा-पूर्व $753 \to -752$।

\textbf{7.} $476 - (-752) = 476 + 752 = 1\,228$ साल।

\textbf{8.} ईसा-पूर्व $12$ यानी $-11$। अगले दर्शन:
$-11 + 76 = 65$, फिर $141$, फिर $217$: यानी ईसा के बाद $65$,
$141$ और $217$। उससे पहले: $-11 - 76 = -87$, जो इतिहासकारों का
ईसा-पूर्व $87 + 1 = 88$ है।

\textbf{9.} ईसा-पूर्व $5$ यानी $-4$, इसलिए $5 - (-4) = 9$ साल
--- $10$ नहीं। जब भी कोई अवधि युग-सीमा पार करती है, ग़ायब
शून्यवाँ वर्ष वार कर देता है।

\textbf{10.} ईसा-पूर्व $3$ यानी $-2$; दस साल बाद $-2 + 10 = +8$:
बच्चा ईसा के बाद $8$ में दस साल का हुआ।

\textbf{11.} $-7$ और $-2$ के बीच: $(-2) - (-7) = 5$।
$-3.5$ और $4.5$ के बीच: $4.5 - (-3.5) = 8$।

\textbf{12.} विपरीत और मिलनसार संख्याओं के समूह बनाकर:
\[
3 + (-5) + 2 + (-1) + 4
= \bigl(3 + 2\bigr) + (-5) + (-1) + 4
= 0 + (-1) + 4 = +3 :
\]
लिफ़्ट मंज़िल $3$ पर रुकती है। क्रम से फ़र्क़ नहीं पड़ता: \omterm{def:g7:negatives:def}{सापेक्ष
संख्याओं} का \omterm{def:g1:addition:def}{योगफल} जैसे चाहे फिर से जमाया जा सकता है (हर चाल उसी
रेखा पर चली जाती है, और कुल चाल वही रहती है)।

\textbf{13.} शुद्ध परिवर्तन:
$2 + (-4) + 1 + 0 + (-3) + 5 + (-2) = -1$ डिग्री। $-2$ से शुरू
करने पर आख़िरी तापमान $-2 + (-1) = -3$ डिग्री। आम तौर पर,
आख़िरी मान शुरुआती मान और सारे परिवर्तनों के \omterm{def:g1:addition:def}{योगफल} के बराबर होता
है --- उतार-चढ़ाव एक-एक करके देखने की ज़रूरत नहीं।

\textbf{14.} शुरू पर लौटने का मतलब है शुद्ध परिवर्तन $0$: चढ़ाव
और उतार ठीक-ठीक कट जाने चाहिए। पाँचों ज्ञात परिवर्तनों का \omterm{def:g1:addition:def}{योगफल}
$4 + (-7) + 2 + 3 + (-5) = -3$ है, इसलिए छठा परिवर्तन उसकी
\omterm{def:g7:negatives:def}{विपरीत संख्या} है: $+3$ (तीन सौ मीटर ऊपर)।

\textbf{15.} खगोलविदों के अनुसार: ईसा-पूर्व $509 \to -508$, और
$476 - (-508) = 984$ साल। इतिहासकार का $509 + 476 = 985$
एक साल ज़्यादा गिनता है --- फिर वही भूतिया शून्यवाँ वर्ष, जो जोड़
में चुपचाप शामिल है पर इतिहास में है ही नहीं।
(जाँच के लिए दोनों युग: गणतंत्र $-508$ से $-26$, $482$
साल; साम्राज्य $-26$ से $+476$, $502$ साल;
$482 + 502 = 984$।)

\end{solution}
