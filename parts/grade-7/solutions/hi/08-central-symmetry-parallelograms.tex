\section*{अध्याय \ref{ch:g7:central} --- केंद्रीय सममिति और समांतर चतुर्भुज}

\begin{solution}{exo:g7:central:1}
आधे चक्कर वाला प्रतिबिंब: $M'$, $O$ से चार ख़ाने \emph{बाईं} ओर और
एक ख़ाना \emph{नीचे} है (दोनों दिशाएँ उलट गईं)। खड़ी रेखा के बारे
में परावर्तन: चार ख़ाने बाईं ओर, पर अब भी एक ख़ाना \emph{ऊपर}।
दोनों प्रतिबिंब अलग हैं: वे एक-दूसरे के खड़े दर्पण-प्रतिबिंब हैं।
\end{solution}

\begin{solution}{exo:g7:central:2}
$A'B' = AB$ (\omterm{def:g3:division:half}{आधा} चक्कर लंबाइयाँ बचाए रखता है) और
$(A'B') \parallel (AB)$ (किसी रेखा का प्रतिबिंब \omterm{def:g6:lines:perp}{समांतर} रेखा होती है,
\cref{prop:g7:central:preserved})।
\end{solution}

\begin{solution}{exo:g7:central:3}
\omterm{def:g7:central:center}{सममिति केंद्र} वाले कैलकुलेटर अंक: $0$, $2$, $5$, $8$ (और $1$ भी,
अगर वह सादी छड़ की तरह लिखा जाए)। अक्षर: H, N, S, Z, O का \omterm{def:g7:central:center}{सममिति
केंद्र} है; A और T का नहीं (उनकी जगह एक अक्ष है)।
\end{solution}

\begin{solution}{exo:g7:central:4}
आमने-सामने की भुजाएँ बराबर: $MN = KL = 7$ cm और $NK = LM = 4$ cm।
आमने-सामने के कोण बराबर: $\widehat M = \widehat K = 115^\circ$।
विकर्ण एक-दूसरे को अपने साझे मध्यबिंदु पर काटते हैं, और वही बिंदु
$O$ है: इसलिए $[LN]$ का मध्यबिंदु $O$ है।
\end{solution}

\begin{solution}{exo:g7:central:5}
$8$ cm का एक \omterm{def:g6:lines:objects}{रेखाखंड} $[AC]$ खींचो, उसका मध्यबिंदु $O$ अंकित करो,
$O$ से होकर $(AC)$ के साथ $60^\circ$ बनाती एक रेखा खींचो, और उस पर
$O$ से दोनों तरफ़ $2.5$ cm पर $B$ तथा $D$ रखो। $A$, $B$, $C$, $D$
को जोड़ो: विकर्ण अपने मध्यबिंदुओं पर कटते हैं, इसलिए परिभाषा से ही
$ABCD$ एक \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} है।
\end{solution}

\begin{solution}{exo:g7:central:6}
लगातार आने वाले कोण संपूरक होते हैं: $115^\circ$ के बग़ल में
$180 - 115 = 65^\circ$ बैठता है। आमने-सामने के कोण बराबर होते हैं,
इसलिए चारों कोण $115^\circ$, $65^\circ$, $115^\circ$, $65^\circ$
हैं (योग $360^\circ$)।
\end{solution}

\begin{solution}{exo:g7:central:7}
$A(1,1)$ से $O(2.5, 2.5)$ तक: $1.5$ दाएँ और $1.5$ ऊपर; उतना ही आगे
चलने पर: $A'(4, 4)$। $B(4,2)$ से $O$ तक: $1.5$ बाएँ और $0.5$ ऊपर;
आगे चलने पर: $B'(1, 3)$।
\end{solution}

\begin{solution}{exo:g7:central:8}
नहीं। \omterm{def:g6:shapes:triangles}{समद्विबाहु} समलंब की दोनों तिरछी भुजाएँ बराबर होती हैं, फिर भी
वह \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} नहीं होता (उसकी दोनों \omterm{def:g6:lines:perp}{समांतर} भुजाओं की लंबाई
अलग-अलग होती है)। जो काफ़ी है वह है
\cref{thm:g7:central:recognize} की कसौटी 3: आमने-सामने की दो
भुजाएँ बराबर \emph{और \omterm{def:g6:lines:perp}{समांतर}}।
\end{solution}

\begin{solution}{exo:g7:central:9}
$O$, चुनाव से ही $[BC]$ का मध्यबिंदु है, और \omterm{def:g7:central:def}{सममित} बिंदु की रचना से
वह $[AA']$ का भी मध्यबिंदु है। \omterm{def:g4:geometry:polygon}{चतुर्भुज} $ABA'C$ के विकर्ण $[AA']$
और $[BC]$ मध्यबिंदु $O$ साझा करते हैं: कसौटी 1 (यानी परिभाषा) उसे
बिना किसी ज़्यादा मेहनत के \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} बना देती है।
\end{solution}

\begin{solution}{exo:g7:central:10}
``आमने-सामने के कोण बराबर $\Rightarrow$ \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज}'':
\emph{सही} (यह पहचानने की एक और कसौटी है; कोणों का योग $360^\circ$
होने से, बराबर आमने-सामने के जोड़े लगातार कोणों को संपूरक होने पर
मजबूर करते हैं, और तब \cref{thm:g7:triangles:equalities} से भुजाएँ
\omterm{def:g6:lines:perp}{समांतर} हो जाती हैं)।

``\omterm{def:g6:lines:perp}{लंब} विकर्ण $\Rightarrow$ \omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज}'': \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} के लिए
\emph{सही} (उसके विकर्ण पहले से ही मध्यबिंदुओं पर कटते हैं)।

``एक \omterm{def:g6:shapes:triangles}{समकोण} $\Rightarrow$ आयत'': \emph{सही} --- लगातार आने वाले कोण
संपूरक होते हैं, इसलिए चारों \omterm{def:g6:shapes:triangles}{समकोण} बन जाते हैं।
\end{solution}

\begin{solution}{exo:g7:central:11}
\omterm{def:g4:geometry:polygon}{चतुर्भुज} चाहे जैसा हो --- बहुत बेढब भी --- मध्यबिंदु $I$, $J$, $K$,
$L$ हमेशा एक \emph{\omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज}} बनाते हैं। मध्यबिंदु प्रमेय
वाला प्रमाण \cref{ch:g8:midpoints} में है
(\cref{exo:g8:midpoints:9})।
\end{solution}

\begin{solution}{pb:g7:central:1}
\textbf{1.} $M(3, 1) \to (-3, -1)$; $N(-2, 4) \to (2, -4)$;
$P(0, -5) \to (0, 5)$। नियम: मूलबिंदु के चारों ओर \omterm{def:g3:division:half}{आधा} चक्कर
$(x, y)$ को $(-x, -y)$ पर भेजता है --- दोनों निर्देशांकों का चिह्न
बदल जाता है।

\textbf{2.} $M(5, 3)$ से: $C$ के तीन ख़ाने दाईं ओर होना तीन ख़ाने
बाईं ओर हो जाता है, और दो ऊपर होना दो नीचे: प्रतिबिंब
$(-1, -1)$। नुस्ख़े से जाँच: केंद्र का दुगुना घटा बिंदु,
$(2 \times 2 - 5,\ 2 \times 1 - 3) = (-1, -1)$।

\textbf{3.} पत्ते के केंद्र के चारों ओर \omterm{def:g3:division:half}{आधा} चक्कर निशानों को
जोड़ों में बदल देता है। \omterm{def:g3:numbers:evenodd}{विषम संख्या} में निशान होने पर एक निशान
जोड़ीदार के बिना रह जाता है: उसे अपना ही प्रतिबिंब होना पड़ेगा, और
अपना प्रतिबिंब सिर्फ़ केंद्र ही होता है। इसलिए 3, 5, 7\,\dots\ के
बीच वाला निशान ठीक बीचोंबीच बैठता है (असली पत्ता देखो: सचमुच ऐसा
ही है)।

\textbf{4.} $O_1(2,0)$ के बारे में: $A(1,1) \to (3,-1)$;
$O_2(5,0)$ के बारे में: $(3,-1) \to (7,1)$। शुरुआत $A(1,1)$, अंत
$(7,1)$: $6$ दाईं ओर सरका, पलटा नहीं। $B(0,3) \to (4,-3) \to
(6,3)$ में भी वही: $6$ दाईं ओर। दोनों आधे चक्कर मिलकर एक
\emph{स्थानांतरण} बन जाते हैं।

\textbf{5.} सरकाव $6$ है और $O_1 O_2 = 3$: स्थानांतरण दोनों
केंद्रों के बीच की दूरी का \emph{दुगुना} तय करता है ($O_1$ से
$O_2$ की दिशा में) --- ठीक वैसे ही जैसे
\cref{pb:g6:symmetry:1} में दो \omterm{def:g6:lines:perp}{समांतर} दर्पण अपने बीच की दूरी के
दुगुने से स्थानांतरण करते थे।

\textbf{6.} $O$ के चारों ओर \omterm{def:g3:division:half}{आधा} चक्कर \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} को उसी पर
भेजता है (उसके लिए ``\omterm{def:g7:central:center}{सममिति केंद्र}'' का यही मतलब है,
\cref{thm:g7:central:properties})। $O$ से होकर जाती रेखा भी उसी पर
भेजी जाती है, इसलिए वह सीमा को जिन दो बिंदुओं पर काटती है वे आपस
में बदल जाते हैं: वे $O$ के बारे में \omterm{def:g7:central:def}{सममित} हैं। रेखा के दोनों तरफ़
के \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} के दोनों टुकड़े भी आपस में बदल जाते हैं, इसलिए
वे एक-दूसरे की ठीक नक़लें हैं (\cref{prop:g7:central:preserved})
--- यानी बराबर \omterm{def:g6:measure:area}{क्षेत्रफल} के।

\textbf{7.} \emph{दोनों केंद्रों} से होकर जाती रेखा के साथ काटो:
केक वाले आयत का केंद्र और छेद वाले आयत का केंद्र। केक वाले आयत पर
सवाल 6 लगाने से कटाई केक को आधा-आधा कर देती है; छेद वाले आयत पर
लगाने से (रेखा उसके केंद्र से भी गुज़रती है) छेद भी आधा-आधा बँट
जाता है। दोनों हिस्सों को \omterm{def:g3:division:half}{आधा} केक घटा \omterm{def:g3:division:half}{आधा} छेद मिलता है: बिलकुल
बराबर।

\textbf{8.} $D = (0 + 8 - 6,\ 0 + 5 - 1) = (2, 4)$ (विकर्ण अपना
मध्यबिंदु साझा करते हैं, इसलिए $D$, केंद्र के बारे में $B$ का आधे
चक्कर वाला प्रतिबिंब है)। केंद्र: $[AC]$ का मध्यबिंदु,
$O(4,\ 2.5)$।

\textbf{9.} \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} $ABA'C$ में आमने-सामने की भुजाएँ बराबर
हैं: $BA' = AC$। त्रिभुज $ABA'$ में त्रिभुज असमिका से
$AA' < AB + BA' = AB + AC$। और $O$, $[AA']$ का मध्यबिंदु है,
इसलिए $AA' = 2\,AO$, यानी $2\,AO < AB + AC$, यानी
$AO < \frac{AB + AC}{2}$।

\textbf{10.} ऊपरी सीमा: $AO < \frac{5 + 7}{2} = 6$ cm। निचली
सीमा: $ABA'$ में $AA' > BA' - AB = 7 - 5 = 2$, इसलिए
$AO > 1$ cm। $A$ से खींची माध्यिका ठीक $1$ और $6$ cm के बीच
पड़ती है।

\textbf{11.} \omterm{def:g6:lines:objects}{रेखाखंड}: हाँ, उसका मध्यबिंदु। रेखा: हाँ --- उसका
प्रत्येक बिंदु एक केंद्र है। \omterm{def:g6:shapes:triangles}{समबाहु त्रिभुज}: कोई नहीं। वर्ग: हाँ,
विकर्णों का कटान। वृत्त: हाँ, उसका केंद्र। S और Z: हाँ, उनका बीच
वाला बिंदु (पन्ना उलटकर देखो: वे वैसे ही पढ़े जाते हैं)।

\textbf{12.} \omterm{def:g3:division:half}{आधा} चक्कर तीनों \omterm{def:g5:solids:def}{शीर्षों} के आपस में जोड़े बनाएगा; तीन
विषम है, इसलिए कोई एक \omterm{def:g5:solids:def}{शीर्ष} $A$ अपना ही प्रतिबिंब होगा, और तब $A$
को ही आधे चक्कर का केंद्र होना पड़ेगा। तब बाक़ी दोनों \omterm{def:g5:solids:def}{शीर्ष} $B$ और
$C$, $A$ के बारे में \omterm{def:g7:central:def}{सममित} होंगे --- यानी $A$, $[BC]$ का मध्यबिंदु
होगा, और तीनों \omterm{def:g5:solids:def}{शीर्ष} एक ही रेखा पर आ जाएँगे। किसी त्रिभुज के तीन
\omterm{def:g5:solids:def}{शीर्ष} एक रेखा पर नहीं होते: विरोधाभास। ऐसा कोई केंद्र है ही नहीं।

\textbf{13.} वही सम-विषम वाला तर्क: \omterm{def:g4:geometry:polygon}{बहुभुज} को बचाए रखने वाला \omterm{def:g3:division:half}{आधा}
चक्कर उसके \omterm{def:g5:solids:def}{शीर्षों} के जोड़े बनाता है; \omterm{def:g5:solids:def}{शीर्षों} की \omterm{def:g3:numbers:evenodd}{विषम संख्या} किसी
एक \omterm{def:g5:solids:def}{शीर्ष} को अपनी जगह जमा देती है, और उसे केंद्र होना पड़ता है तथा
दो और \omterm{def:g5:solids:def}{शीर्षों} को जोड़ने वाले \omterm{def:g6:lines:objects}{रेखाखंड} का मध्यबिंदु भी --- यानी तीन
\omterm{def:g5:solids:def}{शीर्ष} एक ही रेखा पर, जो किसी \omterm{def:g4:geometry:polygon}{बहुभुज} में असंभव है\,\dots\ इसलिए \omterm{def:g3:numbers:evenodd}{विषम
संख्या} में \omterm{def:g5:solids:def}{शीर्षों} वाले किसी \omterm{def:g4:geometry:polygon}{बहुभुज} का \omterm{def:g7:central:center}{सममिति केंद्र} नहीं होता।

\textbf{14.} $O_1$ के बारे में और फिर $O_2$ के बारे में \omterm{def:g3:division:half}{आधा} चक्कर,
दोनों बार आकृति को उसी पर भेजते हैं --- इसलिए उनका मेल भी वही करता
है। सवाल 5 से वह मेल दूरी $O_1 O_2$ के दुगुने का स्थानांतरण है, जो
\omterm{def:g1:counting:zero}{शून्य} नहीं है क्योंकि दोनों केंद्र अलग-अलग हैं। काग़ज़ के एक पन्ने
पर समा जाने वाली आकृति किसी तयशुदा, \omterm{def:g1:counting:zero}{शून्य} से बड़ी दूरी तक सरकने पर
अपने बराबर नहीं रह सकती (उतनी बार सरकाओ तो वह पन्ना ही छोड़ देगी)।
इसलिए किसी घिरी हुई आकृति का ज़्यादा से ज़्यादा एक \omterm{def:g7:central:center}{सममिति केंद्र}
होता है।

\textbf{15.} कभी न ख़त्म होने वाली पट्टी पर, किसी बाएँ पैर के निशान
और उसके बाद वाले दाएँ निशान के ठीक बीच वाले बिंदु के चारों ओर \omterm{def:g3:division:half}{आधा}
चक्कर पूरे नमूने को उसी पर भेज देता है; और उससे एक पूरा क़दम आगे
वाला बिंदु भी यही करता है: दो अलग-अलग केंद्र। दोनों आधे चक्करों का
मेल आधे क़दम के दुगुने का स्थानांतरण देता है --- यानी ठीक वही एक
क़दम का सरकाव जो अनंत पट्टी-नमूने को साफ़-साफ़ उसी पर भेजता है, जैसा
सवाल 14 कहता है। बिना सीमा वाले नमूने अलग नियमों से चलते हैं: यही
दीवार-काग़ज़ का गणित है।

\end{solution}
