\section*{अध्याय \ref{ch:g7:prop} --- समानुपात}

\begin{solution}{exo:g7:prop:1}
पहली सारणी: \omterm{def:g3:division:remainder}{भागफल} $\frac{7.5}{3} = 2.5$, $\frac{17.5}{7} = 2.5$,
$\frac{27.5}{11} = 2.5$: \omterm{def:g6:propdata:def}{समानुपाती}, गुणांक $2.5$।

दूसरी सारणी: $\frac62 = 3$ और $\frac{15}{5} = 3$, पर
$\frac{28}{9} \approx 3.1$: \omterm{def:g6:propdata:def}{समानुपाती} नहीं।
\end{solution}

\begin{solution}{exo:g7:prop:2}
तिरछे नियम से: $5 \times p = 6 \times 8$, इसलिए
$p = \frac{48}{5} = 9.60$ यूरो।
\end{solution}

\begin{solution}{exo:g7:prop:3}
रफ़्तार: $36 \div 3 = 12$ पन्ने प्रति मिनट। $5$ मिनट में:
$60$ पन्ने। $96$ पन्नों के लिए: $96 \div 12 = 8$ मिनट।
\end{solution}

\begin{solution}{exo:g7:prop:4}
$250$ का $30\,\%$: $0.3 \times 250 = 75$। \quad
$60$ का $15\,\%$: $0.15 \times 60 = 9$। \quad
$80$ का $t\,\%$, $20$ है: $\frac{20}{80} = \frac{25}{100}$, इसलिए
$t = 25$।
\end{solution}

\begin{solution}{exo:g7:prop:5}
$\frac{18}{25} = \frac{18 \times 4}{25 \times 4} = \frac{72}{100} =
72\,\%$.
\end{solution}

\begin{solution}{exo:g7:prop:6}
पगडंडी: $9 \times 25\,000 = 225\,000$ cm $= 2.25$ km।
झील: ज़मीन पर $2$ km $= 200\,000$ cm, इसलिए नक़्शे पर
$200\,000 \div 25\,000 = 8$ cm।
\end{solution}

\begin{solution}{exo:g7:prop:7}
$4.3$ m $= 430$ cm, और $430 \div 43 = 10$ cm।
\end{solution}

\begin{solution}{exo:g7:prop:8}
एक हिस्से की चॉकलेट: $240 \div 8 = 30$ g। $300$ g से:
$300 \div 30 = 10$ हिस्से, ठीक-ठीक।
\end{solution}

\begin{solution}{exo:g7:prop:9}
\emph{1.} छूट: $20$ यूरो, यानी पुराने दाम का
$\frac{20}{80} = 25\,\%$।

\emph{2.} $20$ यूरो की बढ़त की तुलना अब \emph{नए} आधार $60$ से
होती है: $\frac{20}{60} = \frac13 \approx 33\,\%$। प्रतिशत हमेशा
किसी पूरे के सापेक्ष होता है; पूरा बदल गया, इसलिए प्रतिशत भी बदल
गया।
\end{solution}

\begin{solution}{exo:g7:prop:10}
पूरे भरे स्तंभ से गुणांक: $\frac{17.5}{10} = 1.75$। फिर
$y = 4 \times 1.75 = 7$ और $x = \frac{10.5}{1.75} = 6$।
\end{solution}

\begin{solution}{exo:g7:prop:11}
\emph{1.} एक नक़ल के बाद: $10 \times 0.8 = 8$ cm। दो के बाद:
$8 \times 0.8 = 6.4$ cm।

\emph{2.} दूसरी बार का छोटा करना पहले से छोटी हो चुकी नक़ल पर लगता
है, असली पर नहीं: गुणक आपस में गुणा होते हैं,
$0.8 \times 0.8 = 0.64$, इसलिए दो नक़लों के बाद आकार असली का
$64\,\%$ रह जाता है --- $60\,\%$ नहीं।
\end{solution}

\begin{solution}{pb:g7:prop:1}
\textbf{1.} ऊँचाइयाँ और परछाइयाँ \omterm{def:g6:propdata:def}{समानुपाती} हैं, इसलिए तिरछे नियम
से (\cref{thm:g7:prop:cross}), डंडे के स्तंभ $(1,\ 0.4)$ और पेड़
के स्तंभ $(h,\ 3.2)$ के साथ: $0.4 \times h = 1 \times 3.2$, यानी
$h = \frac{3.2}{0.4} = 8$ m।

\textbf{2.} मीनार: $\frac{26}{0.4} = 65$ m। बच्चे की परछाईं:
$1.5 \times 0.4 = 0.6$ m (परछाईं $=$ ऊँचाई $\times$ उस पल का
गुणांक $0.4$)।

\textbf{3.} जैसे-जैसे सूरज आसमान में सरकता है, ऊँचाइयों को
परछाइयों से जोड़ने वाला गुणांक बदलता जाता है; एक ही गुणांक सिर्फ़
उन्हीं नापों का होता है जो एक ही पल पर ली गई हों।

\textbf{4.} उस पल गुणांक $1$ है: ऊँचाई और परछाईं बराबर। इसलिए
\omterm{def:g5:solids:def}{पिरामिड} का शिखर $147$ m ऊँचा है --- बड़े \omterm{def:g5:solids:def}{पिरामिड} की ऊँचाई (शिखर
के नीचे वाले बिंदु से नापा गया उसकी परछाईं का सिरा ही काफ़ी है)।

\textbf{5.} स्तंभ का \omterm{def:g3:division:remainder}{भागफल} $\frac{\text{परछाईं}}{\text{ऊँचाई}}$
--- उस पल का \omterm{def:g7:prop:table}{समानुपात} गुणांक --- सारी खड़ी चीज़ों में एक ही रहता
है: ठीक इसी से ऊँचाइयों और परछाइयों की सारणी एक \omterm{def:g7:prop:table}{समानुपात सारणी} बन
जाती है (\cref{def:g7:prop:table})।

\textbf{6.} सूरज की किरणें \omterm{def:g6:lines:perp}{समांतर} आती हैं। \emph{चपटी} पृथ्वी पर
दो \omterm{def:g6:lines:perp}{समांतर} किरणें दो खड़े डंडों से एक ही कोण पर टकरातीं: दोनों की
परछाइयाँ \omterm{def:g6:propdata:def}{समानुपाती} होतीं --- या तो दोनों की कोई परछाईं नहीं, या
दोनों की परछाईं। एक ही पल पर एक डंडे की परछाईं न बनना (स्येने) और
दूसरे की बनना (सिकंदरिया) चपटी पृथ्वी पर असंभव है: दोनों खड़ी
दिशाओं को अलग-अलग तरफ़ इशारा करना ही होगा --- सतह मुड़ी हुई है।

\textbf{7.} चित्र में स्येने की खड़ी दिशा सीधे सूरज की ओर जाती है;
सिकंदरिया की खड़ी दिशा उतनी झुकी है जितना केंद्र से देखने पर दोनों
नगरों की दिशाओं के बीच का कोण है। किरणें \omterm{def:g6:lines:perp}{समांतर} हैं, इसलिए वह झुकाव
ठीक वही $7.2^\circ$ है जो सिकंदरिया में किरण और डंडे के बीच नापा
गया।

\textbf{8.} $\frac{7.2}{360} = \frac{72}{3600} = \frac{1}{50}$:
पूरे चक्कर का पचासवाँ भाग।

\textbf{9.} चापों की लंबाई कोणों के \omterm{def:g6:propdata:def}{समानुपाती} होती है: अगर
$7.2^\circ$ --- चक्कर का पचासवाँ भाग --- $800$ km के बराबर है, तो
पूरा चक्कर होगा
\[
50 \times 800 = 40\,000 \text{ km}.
\]

\textbf{10.} \omterm{def:g4:geometry:circle}{व्यास} $= \frac{\text{परिधि}}{\pi}
\approx \frac{40\,000}{3.14} \approx 12\,700$ km --- आज के
$12\,742$ km के सामने: एक डंडे से नापकर, एक प्रतिशत से भी कम की
ग़लती।

\textbf{11.} $40\,000$ km $= 4\,000\,000\,000$ cm, और
$4\,000\,000\,000 \div 40\,000\,000 = 100$ cm $= 1$ m परिधि।
\omterm{def:g4:geometry:circle}{व्यास}: $100 \div 3.14 \approx 32$ cm --- मेज़ पर सजने लायक़ शानदार
भूगोलक।

\textbf{12.} $4.8$ km $= 480\,000$ cm, और
$480\,000 \div 40\,000\,000 = 0.012$ cm $= 0.12$ mm। आल्प्स का
सबसे ऊँचा पहाड़ एक मीटर परिधि वाले भूगोलक पर मिलीमीटर का दसवाँ
भाग है: धूल का एक कण। इस \omterm{def:g7:prop:scale}{मापनी} पर पृथ्वी ज़्यादातर पॉलिश की हुई
गेंदों से भी चिकनी है।

\textbf{13.} $40\,000 \div 40 = 1\,000$ दिन, और
$1\,000 \div 365 \approx 2.7$: क़रीब तीन साल की पैदल यात्रा।

\textbf{14.} $\frac{10}{6\,370} \approx 0.0016$, यानी क़रीब
$0.2\,\%$ (और ठीक-ठीक $0.16\,\%$): वायुमंडल इस ग्रह पर काग़ज़ से
भी पतली एक झिल्ली है।

\textbf{15.} ग़लती: $42\,000 - 40\,000 = 2\,000$ km, और
$\frac{2\,000}{40\,000} = \frac{5}{100} = 5\,\%$। एक वाक्य में:
एक डंडे, एक कुएँ, एक नापी हुई सड़क और एक \omterm{def:g7:prop:table}{समानुपात} से एरातोस्थनीज़
ने पूरा ग्रह कुछ प्रतिशत की ग़लती के भीतर नाप लिया --- गणित बहुत
थोड़े साज़-सामान पर बहुत दूर तक चला जाता है।

\end{solution}
