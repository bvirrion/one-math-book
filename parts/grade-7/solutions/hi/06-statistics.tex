\section*{अध्याय \ref{ch:g7:stats} --- आँकड़ों को सँवारना}

\begin{solution}{exo:g7:stats:1}
प्रत्येक अंक की \omterm{def:g7:stats:frequency}{गिनती}:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|ccccc|c}
अंक & $9$ & $11$ & $12$ & $14$ & $15$ & कुल \\
\hline
\omterm{def:g7:stats:frequency}{गिनती} & $4$ & $3$ & $7$ & $3$ & $3$ & $20$ \\
\end{tabular}
\end{center}
\end{solution}

\begin{solution}{exo:g7:stats:2}
आवृत्तियाँ: $\frac{4}{20} = 20\,\%$, $\frac{3}{20} = 15\,\%$,
$\frac{7}{20} = 35\,\%$, $15\,\%$, $15\,\%$। योग:
$20 + 15 + 35 + 15 + 15 = 100\,\%$. \checkmark
\end{solution}

\begin{solution}{exo:g7:stats:3}
पहला सर्वेक्षण: $\frac{30}{50} = 0.6 = 60\,\%$। दूसरा:
$\frac{45}{90} = 0.5 = 50\,\%$। अनुपात पहले सर्वेक्षण में
ज़्यादा है, भले ही उसमें एक कार वाले परिवारों की \omterm{def:g7:stats:frequency}{गिनती} कम हो।
\end{solution}

\begin{solution}{exo:g7:stats:4}
$\intco{5}{15}$: $8, 12, 9, 14, 11$ --- \omterm{def:g7:stats:frequency}{गिनती} $5$.
$\intco{15}{25}$: $23, 22, 18, 16$ --- \omterm{def:g7:stats:frequency}{गिनती} $4$.
$\intco{25}{35}$: $31, 27, 29, 25, 33$ --- \omterm{def:g7:stats:frequency}{गिनती} $5$.
$\intco{35}{45}$: $35$ --- \omterm{def:g7:stats:frequency}{गिनती} $1$. कुल $15$. \checkmark
\end{solution}

\begin{solution}{exo:g7:stats:5}
मानों $0$, $1$, $2$, $3$ के ऊपर $8$, $12$, $6$, $4$ ऊँचाई की चार
पट्टियाँ (हर दो पर एक अंकन ठीक रहता है)।
\end{solution}

\begin{solution}{exo:g7:stats:6}
कुल $30$ परिवार। आवृत्तियाँ: $\frac{8}{30}$, $\frac{12}{30}$,
$\frac{6}{30}$, $\frac{4}{30}$। कोण ($\times 360^\circ$):
$96^\circ$, $144^\circ$, $72^\circ$, $48^\circ$; योग
$96 + 144 + 72 + 48 = 360^\circ$. \checkmark
\end{solution}

\begin{solution}{exo:g7:stats:7}
प्रतिशत: गर्मी $\frac{162}{360} = 45\,\%$; वसंत
$\frac{90}{360} = 25\,\%$; पतझड़ और जाड़ा $\frac{54}{360} = 15\,\%$
अलग-अलग। $60$ लोगों में से गर्मी $0.45 \times 60 = 27$ ने चुनी।
\end{solution}

\begin{solution}{exo:g7:stats:8}
कक्षा A: $\frac{12}{30} = 40\,\%$। कक्षा B:
$\frac{14}{40} = 35\,\%$। पैदल आने वालों का अनुपात कक्षा A में
ज़्यादा है, भले ही \omterm{def:g7:stats:frequency}{गिनती} में कक्षा B के पैदल आने वाले ज़्यादा हों।
\end{solution}

\begin{solution}{exo:g7:stats:9}
धुरी $9\,700$ से शुरू होने पर तीनों पट्टियों की दिखने वाली
ऊँचाइयाँ $100$, $300$ और $400$ इकाई हैं: आख़िरी पहली से चार गुनी
लगती है, मानो बिक्री चौगुनी हो गई हो। $0$ से शुरू होने वाली धुरी
पर पट्टियाँ लगभग बराबर हैं ($9\,800$ से $10\,100$ यानी क़रीब
$3\,\%$ की बढ़त): ईमानदार तस्वीर में बिक्री लगभग सपाट है।
\end{solution}

\begin{solution}{exo:g7:stats:10}
आवृत्तियों का योग $1$ होता है: $f = 1 - 0.35 - 0.4 = 0.25$।
गिनतियाँ (कुल $80$): $0.35 \times 80 = 28$; $0.4 \times 80 = 32$;
$0.25 \times 80 = 20$। जाँच: $28 + 32 + 20 = 80$।
\end{solution}

\begin{solution}{exo:g7:stats:11}
लड़कियाँ: $400$ का $55\,\% = 220$; लड़के: $180$। भोजनालय जाने
वाली लड़कियाँ: $0.4 \times 220 = 88$; लड़के:
$0.6 \times 180 = 108$। कुल: $88 + 108 = 196$ विद्यार्थी, यानी
स्कूल का $\frac{196}{400} = 49\,\%$ --- $40\,\%$ और $60\,\%$ के
बीच, और लड़कियों वाली दर के ज़्यादा पास, क्योंकि लड़कियाँ
ज़्यादा हैं।
\end{solution}

\begin{solution}{pb:g7:stats:1}
\textbf{1.} \omterm{def:g7:stats:frequency}{गिनती}: $120 > 90$, स्कूल A जीता। \omterm{def:g7:stats:frequency}{आवृत्ति}:
$\frac{120}{400} = 0.30 = 30\,\%$ बनाम
$\frac{90}{250} = 0.36 = 36\,\%$: स्कूल B जीता। इनाम \omterm{def:g7:stats:frequency}{आवृत्ति} के
साथ जाना चाहिए --- स्कूल B जितना इस्तेमाल करता है उसका बड़ा हिस्सा
पुनर्चक्रित करता है; स्कूल A तो बस ज़्यादा रस पीता है।

\textbf{2.} शिक्षक: $\frac{18}{30} = 0.60 = 60\,\%$।
विद्यार्थी: $\frac{210}{600} = 0.35 = 35\,\%$। दोनों मिलाकर:
$\frac{18 + 210}{630} = \frac{228}{630} \approx 0.36 = 36\,\%$।
विद्यार्थी बीस गुना ज़्यादा हैं, इसलिए मिली-जुली \omterm{def:g7:stats:frequency}{आवृत्ति} लगभग
पूरी तरह उन्हीं की तरफ़ खिंच जाती है --- बड़े समूह का भार बड़ा
होता है।

\textbf{3.} आवृत्तियों का योग $100\,\%$ होना ही चाहिए
(\cref{def:g7:stats:frequency}), पर
$45 + 30 + 20 + 10 = 105$: कम से कम एक फाँक ग़लत है।

\textbf{4.} \omterm{def:g7:stats:frequency}{आवृत्ति} $0.15$ की \omterm{def:g7:stats:frequency}{गिनती}: $0.15 \times 40 = 6$। अब तक
की गिनतियाँ: $14 + 10 + 6 = 30$, इसलिए आख़िरी \omterm{def:g7:stats:frequency}{गिनती}
$40 - 30 = 10$ है, जिसकी \omterm{def:g7:stats:frequency}{आवृत्ति} $\frac{10}{40} = 0.25$। (जाँच:
$0.35 + 0.25 + 0.15 + 0.25 = 1$।)

\textbf{5.} कोण: $0.40 \times 360 = 144^\circ$,
$0.35 \times 360 = 126^\circ$, $0.25 \times 360 = 90^\circ$;
और $144 + 126 + 90 = 360^\circ$: वृत्त पूरा हो जाता है।

\textbf{6.} भेजे गए मतपत्रों का $\frac{2.4}{10} = 0.24 = 24\,\%$
वापस आया।

\textbf{7.} टेलीफ़ोन-निर्देशिकाओं और गाड़ियों के पंजीकरण में
ज़्यादातर अमीर घर दर्ज थे, जिनका वोट देश के वोट से अलग था; वे
$24$ लाख जवाब ग़लत आबादी से खींची गई एक विशाल \omterm{def:g7:stats:frequency}{गिनती} भर थे। सवाल 1
की तरह: मायने यह नहीं रखता कि तुमने कितने गिने, बल्कि यह कि
किन्हें --- टेढ़े नमूने पर निकाली गई \omterm{def:g7:stats:frequency}{आवृत्ति} उस नमूने का ब्योरा
देती है, देश का नहीं।

\textbf{8.} असली समर्थन:
$0.7 \times 1\,000 + 0.3 \times 9\,000 = 700 + 2\,700 = 3\,400$
समर्थक $10\,000$ में से: $34\,\%$। सर्वेक्षण:
$0.7 \times 500 + 0.3 \times 500 = 350 + 150 = 500$, $1\,000$ में
से: $50\,\%$ --- सोलह अंक ज़्यादा, क्योंकि अमीर नमूने के आधे हैं
पर शहर के दसवें भाग।

\textbf{9.} जवाब: $0.4 \times 200 = 80$ नाख़ुश और
$0.1 \times 800 = 80$ ख़ुश, यानी $160$ जवाब जिनमें $80$ नाख़ुश:
नापी गई नाख़ुशी $\frac{80}{160} = 50\,\%$। असली नाख़ुशी:
$\frac{200}{1\,000} = 20\,\%$। झूठ किसी ने नहीं बोला --- नाख़ुश
लोग ज़्यादा जवाब देते हैं, और सर्वेक्षण उन्हीं की आवाज़ ऊँची
सुनता है।

\textbf{10.} असली अनुपात: दसवाँ भाग अमीर, इसलिए $100$ अमीर और
$900$ साधारण बातचीत:
$0.7 \times 100 + 0.3 \times 900 = 70 + 270 = 340$, $1\,000$ में
से: सुधरा हुआ सर्वेक्षण $34\,\%$ बताता है --- सवाल 8 वाला सच।
सोच-समझकर चुनी गई पचास हज़ार बातचीत बीस लाख बेतरतीब बातचीत से
बेहतर है।

\textbf{11.} अस्पताल A: हल्के $\frac{90}{100} = 90\,\%$; गंभीर
$\frac{280}{400} = 70\,\%$; कुल
$\frac{90 + 280}{500} = \frac{370}{500} = 74\,\%$।

\textbf{12.} अस्पताल B: हल्के $\frac{340}{400} = 85\,\%$; गंभीर
$\frac{65}{100} = 65\,\%$; कुल
$\frac{340 + 65}{500} = \frac{405}{500} = 81\,\%$। हल्के मामलों
में ($90 > 85$) \emph{और} गंभीर मामलों में ($70 > 65$) --- दोनों
में अस्पताल A जीतता है।

\textbf{13.} कुल मिलाकर B $81\,\%$ दिखाता है और A $74\,\%$: जो
अस्पताल \emph{प्रत्येक} श्रेणी में हारता है वही कुल में जीत जाता
है। (इस उलटफेर का नाम है: सिंपसन का विरोधाभास।)

\textbf{14.} दोनों के मिश्रण उलटे हैं: A के मरीज़ ज़्यादातर गंभीर
हैं ($500$ में से $400$), B के ज़्यादातर हल्के ($500$ में से
$400$)। गंभीर मामले कहीं भी हों, कम ही ठीक होते हैं, इसलिए A का
कुल आँकड़ा उन्हीं कठिन मामलों से नीचे खिंच जाता है जिन्हें वह भर्ती
करता है --- कुल \omterm{def:g7:stats:frequency}{आवृत्ति} एक भार-युक्त मिलावट है, और भार दोनों
अस्पतालों में अलग-अलग हैं। गंभीर मरीज़ को अस्पताल A चुनना चाहिए:
जिस अकेली पंक्ति से उसका वास्ता है, उसमें $70\,\%$, $65\,\%$ से
बेहतर है।

\textbf{15.} जैसे: ``आपकी तुलना हल्के और गंभीर मरीज़ों को घोल
देती है, और दोनों अस्पतालों के मरीज़ों के मिश्रण उलटे हैं ---
गंभीरता से बाँटकर देखें तो अस्पताल A \emph{दोनों} तरह के मरीज़ों
का बड़ा हिस्सा ठीक करता है। कुल आँकड़े तभी तुले जा सकते हैं जब
उनके पीछे के समूह एक जैसे हों; वरना जीत गुणवत्ता नहीं, भार तय
करते हैं।''

\end{solution}
