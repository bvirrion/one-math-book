\section*{Chapitre \ref{ch:g7:stats} --- Organiser des données}

\begin{solution}{exo:g7:stats:1}
En comptant chaque note~:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|ccccc|c}
note & $9$ & $11$ & $12$ & $14$ & $15$ & total \\
\hline
\omterm{def:g7:stats:frequency}{effectif} & $4$ & $3$ & $7$ & $3$ & $3$ & $20$ \\
\end{tabular}
\end{center}
\end{solution}

\begin{solution}{exo:g7:stats:2}
\omterm{def:g7:stats:frequency}{Fréquences}~: $\frac{4}{20} = 20\,\%$, $\frac{3}{20} = 15\,\%$,
$\frac{7}{20} = 35\,\%$, $15\,\%$, $15\,\%$. \omterm{def:g1:addition:def}{Somme}~:
$20 + 15 + 35 + 15 + 15 = 100\,\%$. \checkmark
\end{solution}

\begin{solution}{exo:g7:stats:3}
Première enquête~: $\frac{30}{50} = 0.6 = 60\,\%$. Seconde~:
$\frac{45}{90} = 0.5 = 50\,\%$. La première enquête montre la
proportion la plus élevée, bien qu'elle ait moins de familles avec
une voiture en \omterm{def:g7:stats:frequency}{effectif} absolu.
\end{solution}

\begin{solution}{exo:g7:stats:4}
$\intco{5}{15}$~: $8, 12, 9, 14, 11$ --- \omterm{def:g7:stats:frequency}{effectif} $5$.
$\intco{15}{25}$~: $23, 22, 18, 16$ --- \omterm{def:g7:stats:frequency}{effectif} $4$.
$\intco{25}{35}$~: $31, 27, 29, 25, 33$ --- \omterm{def:g7:stats:frequency}{effectif} $5$.
$\intco{35}{45}$~: $35$ --- \omterm{def:g7:stats:frequency}{effectif} $1$. Total $15$. \checkmark
\end{solution}

\begin{solution}{exo:g7:stats:5}
Quatre barres de hauteurs $8$, $12$, $6$, $4$ au-dessus des valeurs
$0$, $1$, $2$, $3$ (graduation de $2$ en $2$ convient bien).
\end{solution}

\begin{solution}{exo:g7:stats:6}
Total $30$ familles. \omterm{def:g7:stats:frequency}{Fréquences}~: $\frac{8}{30}$, $\frac{12}{30}$,
$\frac{6}{30}$, $\frac{4}{30}$. Angles ($\times 360^\circ$)~:
$96^\circ$, $144^\circ$, $72^\circ$, $48^\circ$~; \omterm{def:g1:addition:def}{somme}
$96 + 144 + 72 + 48 = 360^\circ$. \checkmark
\end{solution}

\begin{solution}{exo:g7:stats:7}
Pourcentages~: été $\frac{162}{360} = 45\,\%$~; printemps
$\frac{90}{360} = 25\,\%$~; automne et hiver
$\frac{54}{360} = 15\,\%$ chacun. Sur $60$ personnes, l'été a été
choisi par $0.45 \times 60 = 27$ d'entre elles.
\end{solution}

\begin{solution}{exo:g7:stats:8}
Classe A~: $\frac{12}{30} = 40\,\%$. Classe B~: $\frac{14}{40} =
35\,\%$. La classe A a la proportion de piétons la plus élevée, même
si la classe B a plus de piétons en \omterm{def:g7:stats:frequency}{effectif}.
\end{solution}

\begin{solution}{exo:g7:stats:9}
Avec l'axe commençant à $9\,700$, les trois barres ont des hauteurs
visibles de $100$, $300$ et $400$ unités~: la dernière paraît quatre
fois la première, suggérant que les ventes ont quadruplé. Avec un axe
partant de $0$, les barres sont presque égales ($9\,800$ à $10\,100$
est une hausse d'environ $3\,\%$)~: l'image honnête montre des ventes
presque plates.
\end{solution}

\begin{solution}{exo:g7:stats:10}
Les \omterm{def:g7:stats:frequency}{fréquences} somment à $1$~: $f = 1 - 0.35 - 0.4 = 0.25$.
\omterm{def:g7:stats:frequency}{Effectifs} (total $80$)~: $0.35 \times 80 = 28$~; $0.4 \times 80 =
32$~; $0.25 \times 80 = 20$. Vérification~: $28 + 32 + 20 = 80$.
\end{solution}

\begin{solution}{exo:g7:stats:11}
Filles~: $55\,\%$ de $400 = 220$~; garçons~: $180$. Filles à la
cantine~: $0.4 \times 220 = 88$~; garçons~: $0.6 \times 180 = 108$.
Total~: $88 + 108 = 196$ élèves, soit $\frac{196}{400} = 49\,\%$ de
l'établissement --- entre $40\,\%$ et $60\,\%$, plus proche du taux
des filles car elles sont plus nombreuses.
\end{solution}

\begin{solution}{pb:g7:stats:1}
\textbf{1.} \omterm{def:g7:stats:frequency}{Effectifs}~: $120 > 90$, l'école A l'emporte.
\omterm{def:g7:stats:frequency}{Fréquences}~: $\frac{120}{400} = 0.30 = 30\,\%$ contre
$\frac{90}{250} = 0.36 = 36\,\%$~: l'école B l'emporte. Le prix doit
suivre la \omterm{def:g7:stats:frequency}{fréquence} --- l'école B recycle une plus grande part de ce
qu'elle consomme~; l'école A boit simplement plus de jus de fruits.

\textbf{2.} Professeurs~: $\frac{18}{30} = 0.60 = 60\,\%$. Élèves~:
$\frac{210}{600} = 0.35 = 35\,\%$. Ensemble~:
$\frac{18 + 210}{630} = \frac{228}{630} \approx 0.36 = 36\,\%$. Les
élèves sont vingt fois plus nombreux~: la \omterm{def:g7:stats:frequency}{fréquence} globale est donc
tirée presque entièrement de leur côté --- le grand groupe pèse
lourd.

\textbf{3.} Les \omterm{def:g7:stats:frequency}{fréquences} doivent totaliser $100\,\%$
(\cref{def:g7:stats:frequency}), or $45 + 30 + 20 + 10 = 105$~: au
moins un secteur est faux.

\textbf{4.} \omterm{def:g7:stats:frequency}{Effectif} de la \omterm{def:g7:stats:frequency}{fréquence} $0.15$~:
$0.15 \times 40 = 6$. \omterm{def:g7:stats:frequency}{Effectifs} cumulés jusque-là~:
$14 + 10 + 6 = 30$, donc le dernier \omterm{def:g7:stats:frequency}{effectif} vaut
$40 - 30 = 10$, de \omterm{def:g7:stats:frequency}{fréquence} $\frac{10}{40} = 0.25$. (Vérification~:
$0.35 + 0.25 + 0.15 + 0.25 = 1$.)

\textbf{5.} Angles~: $0.40 \times 360 = 144^\circ$,
$0.35 \times 360 = 126^\circ$, $0.25 \times 360 = 90^\circ$~; et
$144 + 126 + 90 = 360^\circ$~: le disque se referme.

\textbf{6.} $\frac{2.4}{10} = 0.24 = 24\,\%$ des bulletins sont
revenus.

\textbf{7.} Les annuaires téléphoniques et les registres
d'automobiles recensaient surtout des foyers aisés, dont le vote
différait de celui du pays~; les $2.4$ millions de réponses
formaient un \omterm{def:g7:stats:frequency}{effectif} géant tiré de la mauvaise population. Comme à
la question 1~: ce qui compte n'est pas combien on compte, mais qui
--- une \omterm{def:g7:stats:frequency}{fréquence} calculée sur un échantillon déformé décrit
l'échantillon, pas le pays.

\textbf{8.} Soutien réel~:
$0.7 \times 1\,000 + 0.3 \times 9\,000 = 700 + 2\,700 = 3\,400$
partisans sur $10\,000$, soit $34\,\%$. Le sondage~:
$0.7 \times 500 + 0.3 \times 500 = 350 + 150 = 500$ sur $1\,000$,
soit $50\,\%$ --- seize points de trop, parce que les électeurs
aisés font la \omterm{def:g3:division:half}{moitié} de l'échantillon mais un dixième de la ville.

\textbf{9.} Réponses~: $0.4 \times 200 = 80$ mécontents et
$0.1 \times 800 = 80$ satisfaits, soit $160$ réponses dont $80$
mécontentes~: mécontentement mesuré $\frac{80}{160} = 50\,\%$.
Mécontentement réel~: $\frac{200}{1\,000} = 20\,\%$. Personne n'a
menti --- les mécontents répondent simplement davantage, et
l'enquête les entend plus fort.

\textbf{10.} Proportions réelles~: un dixième d'électeurs aisés, soit
$100$ entretiens auprès des aisés et $900$ auprès des modestes~:
$0.7 \times 100 + 0.3 \times 900 = 70 + 270 = 340$ sur $1\,000$~: le
sondage réparé prédit $34\,\%$ --- la vérité de la question 8.
Cinquante mille entretiens bien choisis battent deux millions
d'entretiens mal choisis.

\textbf{11.} Hôpital A~: bénins $\frac{90}{100} = 90\,\%$~; graves
$\frac{280}{400} = 70\,\%$~; global
$\frac{90 + 280}{500} = \frac{370}{500} = 74\,\%$.

\textbf{12.} Hôpital B~: bénins $\frac{340}{400} = 85\,\%$~; graves
$\frac{65}{100} = 65\,\%$~; global
$\frac{340 + 65}{500} = \frac{405}{500} = 81\,\%$. L'hôpital A
l'emporte sur les cas bénins ($90 > 85$) \emph{et} sur les cas
graves ($70 > 65$).

\textbf{13.} Globalement, B affiche $81\,\%$ contre $74\,\%$ pour
A~: l'hôpital qui perd dans \emph{toutes} les catégories gagne au
total. (Ce renversement porte un nom~: le paradoxe de Simpson.)

\textbf{14.} Les compositions sont opposées~: les patients de A sont
surtout des cas graves ($400$ sur $500$), ceux de B surtout des cas
bénins ($400$ sur $500$). Les cas graves guérissent moins souvent où
qu'ils soient soignés~: le chiffre global de A est donc tiré vers le
bas par les cas difficiles qu'il accepte --- le global est un mélange
pondéré, et les poids diffèrent d'un hôpital à l'autre. Avec un cas
grave, il faut choisir l'hôpital A~: $70\,\%$ bat $65\,\%$ dans la
seule ligne qui nous concerne.

\textbf{15.} Par exemple~: «~Votre comparaison mélange les cas
bénins et les cas graves, or les deux hôpitaux accueillent des
patientèles opposées --- à gravité égale, l'hôpital A guérit une
plus grande part des \emph{deux} sortes de cas. Des chiffres
globaux ne se comparent que si les groupes qui les portent se
ressemblent~; sinon, ce sont les poids, et non la qualité, qui
désignent le vainqueur.~»
\end{solution}
