\section*{Chapitre \ref{ch:g7:areas} --- Aires et volumes}

\begin{solution}{exo:g7:areas:1}
$8 \times 5 = 40$ cm$^2$~; \qquad $6.5 \times 4 = 26$ m$^2$.
\end{solution}

\begin{solution}{exo:g7:areas:2}
$A = 10 \times 4 = 40$ cm$^2$. La réponse n'est pas
$10 \times 6 = 60$ car le côté de $6$ cm est oblique~: la formule
utilise la \emph{hauteur} (distance \omterm{def:g6:lines:perp}{perpendiculaire}), qui vaut $4$ cm.
\end{solution}

\begin{solution}{exo:g7:areas:3}
$\frac{12 \times 7}{2} = 42$ cm$^2$~; \quad
$\frac{9 \times 6}{2} = 27$~; \quad
$\frac{5.5 \times 2}{2} = 5.5$ m$^2$.
\end{solution}

\begin{solution}{exo:g7:areas:4}
Disque~: $\pi \times 5^2 = 25\pi \approx 79$ cm$^2$. Demi-disque de
\omterm{def:g3:shapes:shapes}{rayon} $10$~: $\frac{\pi \times 10^2}{2} = 50\pi \approx 157$ cm$^2$.
\end{solution}

\begin{solution}{exo:g7:areas:5}
\omterm{def:g6:measure:perimeter}{Périmètre}~: $2\pi \times 3 = 6\pi \approx 18.8$ cm (une longueur, en
cm). \omterm{def:g6:measure:area}{Aire}~: $\pi \times 3^2 = 9\pi \approx 28.3$ cm$^2$ (une surface,
en cm$^2$).
\end{solution}

\begin{solution}{exo:g7:areas:6}
\omterm{def:g7:areas:prism}{Prisme}~: $24 \times 10 = 240$ cm$^3$. \omterm{def:g7:areas:prism}{Cylindre}~:
$\pi \times 3^2 \times 8 = 72\pi \approx 226$ cm$^3$.
\end{solution}

\begin{solution}{exo:g7:areas:7}
$\frac{8 \times h}{2} = 28$, donc $4h = 28$ et $h = 7$ cm.
\end{solution}

\begin{solution}{exo:g7:areas:8}
Les trois \omterm{def:g2:mult:def}{produits} coïncident car chacun calcule la \emph{même}
quantité --- l'\omterm{def:g6:measure:area}{aire} du triangle. Cela donne un contrôle pratique des
constructions, et un moyen de calculer une hauteur à partir d'une
autre paire base--hauteur (utilisé en \cref{exo:g7:areas:12}).
\end{solution}

\begin{solution}{exo:g7:areas:9}
Les deux triangles ont pour \omterm{def:g6:measure:area}{aires} $\frac{30 \times 20}{2} = 300$
m$^2$ et $\frac{18 \times 20}{2} = 180$ m$^2$~: total $480$ m$^2$.
Formule générale suggérée~: pour des côtés \omterm{def:g6:lines:perp}{parallèles} $a$ et $b$ à
distance $h$,
\[
A = \frac{(a + b) \times h}{2}
\]
--- la formule du trapèze (ici $\frac{(30+18)\times 20}{2} = 480$).
\end{solution}

\begin{solution}{exo:g7:areas:10}
\omterm{def:g6:measure:volume}{Volume} d'eau~: $\pi \times 6^2 \times 15 = 540\pi \approx 1\,696$
cm$^3$. Base de la boîte~: $18 \times 12 = 216$ cm$^2$. Hauteur
atteinte~: $\frac{540\pi}{216} = 2.5\pi \approx 7.9$ cm.
\end{solution}

\begin{solution}{exo:g7:areas:11}
\omterm{def:g3:shapes:shapes}{Rayons} $15$ et $20$ cm. \omterm{def:g6:measure:area}{Aires}~: $225\pi \approx 707$ cm$^2$ et
$400\pi \approx 1257$ cm$^2$. Prix pour $100$ cm$^2$~:
$\frac{9}{7.07} \approx 1.27$ euros et
$\frac{14}{12.57} \approx 1.11$ euros. La grande pizza est la
meilleure affaire --- les \omterm{def:g6:measure:area}{aires} croissent avec le \emph{carré} du
\omterm{def:g4:geometry:circle}{diamètre}.
\end{solution}

\begin{solution}{exo:g7:areas:12}
Avec les côtés de l'\omterm{def:g3:shapes:rightangle}{angle droit}~: $A = \frac{6 \times 8}{2} = 24$
cm$^2$. Avec l'hypoténuse comme base~: $24 = \frac{10 \times h}{2} =
5h$, donc $h = 4.8$ cm.
\end{solution}

\begin{solution}{pb:g7:areas:1}
\textbf{1.} Carré~: $8 \times 8 = 64$. Rectangle~:
$5 \times 13 = 65$. Les mêmes quatre pièces semblent couvrir $64$
unités d'\omterm{def:g6:measure:area}{aire} dans une figure et $65$ dans l'autre~: une unité
d'\omterm{def:g6:measure:area}{aire} aurait été créée à partir de rien.

\textbf{2.} Chaque triangle~: $\frac{8 \times 3}{2} = 12$. Chaque
trapèze~: $\frac{(5 + 3) \times 5}{2} = 20$.

\textbf{3.} $12 + 12 + 20 + 20 = 64$~: les pièces totalisent $64$,
elles peuvent donc remplir exactement le carré --- et ne peuvent
\emph{pas} remplir le rectangle de $65$ unités. L'image du rectangle
contient forcément un trou.

\textbf{4.} Une même pente exigerait que monter de $3$ sur $8$ soit
proportionnel à monter de $2$ sur $5$~; \omterm{def:g2:mult:def}{produits} en croix~:
$3 \times 5 = 15$ et $2 \times 8 = 16$. Ils diffèrent~: l'hypoténuse
et le côté oblique ne s'alignent \emph{pas}. La «~diagonale~» du
rectangle est un mensonge --- deux pentes légèrement différentes qui
se rejoignent sous un angle très ouvert.

\textbf{5.} Le long de la fausse diagonale, les quatre pièces
laissent un vide long et extrêmement fin --- une écharde en forme de
\omterm{def:g7:central:parallelogram}{parallélogramme} très aplati --- dont l'\omterm{def:g6:measure:area}{aire} vaut exactement l'unité
manquante, $65 - 64 = 1$. Étirée sur une diagonale d'environ $13$
unités, sa largeur moyenne vaut environ
$1 \div 13 \approx 0.08$ unité~: sur un vrai dessin, moins que le
trait de crayon qui la dissimule.

\textbf{6.} Chaque triangle a une base de $6$ cm et une hauteur de
$4$ cm --- la distance entre les \omterm{def:g6:lines:perp}{parallèles}, où que soit le \omterm{def:g5:solids:def}{sommet}
--- donc chaque \omterm{def:g6:measure:area}{aire} vaut $\frac{6 \times 4}{2} = 12$ cm$^2$
(\cref{thm:g7:areas:triangle}). Faire glisser le \omterm{def:g5:solids:def}{sommet} le long de
la \omterm{def:g6:lines:perp}{parallèle} change la forme, jamais la base ni la hauteur~: des
\omterm{def:g6:measure:area}{aires} égales à jamais.

\textbf{7.} La diagonale découpe le trapèze en un triangle de base
$B$ et un triangle de base $b$, tous deux de hauteur $h$ (la
distance entre les côtés \omterm{def:g6:lines:perp}{parallèles})~:
\[
A = \frac{B \times h}{2} + \frac{b \times h}{2}
= \frac{(B + b) \times h}{2} .
\]

\textbf{8.} $A = \frac{(9 + 5) \times 4}{2} = \frac{56}{2} =
28$ cm$^2$.

\textbf{9.} La moyenne des côtés \omterm{def:g6:lines:perp}{parallèles} vaut $\frac{B + b}{2}$,
donc (moyenne) $\times h = \frac{(B+b) \times h}{2}$~: la même
formule, factorisée autrement. Vérification~: moyenne
$= \frac{9+5}{2} = 7$, et $7 \times 4 = 28$ cm$^2$.

\textbf{10.} La pile penchée contient exactement les mêmes cartes
--- les mêmes couches, chacune de même \omterm{def:g6:measure:area}{aire} --- donc le même \omterm{def:g6:measure:volume}{volume}.
Dans $V = B \times h$, la hauteur doit se mesurer
\emph{perpendiculairement} à la base (la hauteur verticale de la
pile), et non le long de l'inclinaison~: exactement comme l'\omterm{def:g6:measure:area}{aire} du
\omterm{def:g7:central:parallelogram}{parallélogramme} réclame la hauteur \omterm{def:g6:lines:perp}{perpendiculaire} et non le côté
oblique (\cref{exo:g7:areas:2}), une dimension plus haut.

\textbf{11.} Anneau $= \pi \times 5^2 - \pi \times 3^2 = 25\pi -
9\pi = 16\pi \approx 50$ m$^2$ (\cref{thm:g7:areas:disk}).

\textbf{12.} Un disque d'\omterm{def:g6:measure:area}{aire} $16\pi$ a pour \omterm{def:g3:shapes:shapes}{rayon} $4$, puisque
$\pi \times 4^2 = 16\pi$. Les \omterm{def:g3:shapes:shapes}{rayons} $3$, $4$, $5$~: le triplet le
plus célèbre des mathématiques, qui tient la vedette au
\cref{ch:g8:pythagoras}.

\textbf{13.} Une pizza de $40$ cm~: \omterm{def:g3:shapes:shapes}{rayon} $20$, \omterm{def:g6:measure:area}{aire}
$\pi \times 400 \approx 1\,257$ cm$^2$. Deux pizzas de $28$ cm~:
\omterm{def:g3:shapes:shapes}{rayon} $14$, à elles deux
$2 \times \pi \times 196 = 392\pi \approx 1\,232$ cm$^2$. La grande
pizza seule fait \emph{plus} de pizza que les deux moyennes.

\textbf{14.} Multiplier le \omterm{def:g3:shapes:shapes}{rayon} par $1.4$ multiplie l'\omterm{def:g6:measure:area}{aire} par
$1.4^2 = 1.96$ --- presque le double. Le facteur exact est le nombre
dont le carré vaut $2$, soit environ $1.414$~: le nombre de la
diagonale de l'\cref{ex:g8:pythagoras:diagonal}. (Règle de pouce~:
une \omterm{def:g3:division:half}{moitié} plus large, c'est presque deux fois plus de pizza.)

\textbf{15.} $13 \times 13 = 169$, mais $8 \times 21 = 168$~: cette
fois, une unité d'\omterm{def:g6:measure:area}{aire} \emph{disparaît} --- les pièces se
chevauchent le long d'une écharde au lieu de laisser un vide. Avec
les termes consécutifs $3, 5, 8, 13, 21$ de la suite de
Hemachandra--Fibonacci, les \omterm{def:g2:mult:def}{produits} basculent alternativement une
unité au-dessus et une unité au-dessous du carré
($5 \times 13 = 65 = 8^2 + 1$, puis
$8 \times 21 = 168 = 13^2 - 1$)~: le magicien «~gagne~» et
«~perd~» une unité tour à tour. Les \omterm{def:g6:measure:area}{aires} ne mentent jamais~: des
pièces totalisant $64$ couvrent $64$, où qu'on les fasse glisser ---
chaque unité manquante ou excédentaire se cache dans une écharde, en
attendant qu'un contrôle de pente la démasque.
\end{solution}
