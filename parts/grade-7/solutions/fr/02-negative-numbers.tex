\section*{Chapitre \ref{ch:g7:negatives} --- Nombres relatifs}

\begin{solution}{exo:g7:negatives:1}
Opposés~: $-7$~; \ $+3.5$~; \ $0$ (son propre opposé)~; \ $-12$. Deux
nombres sont opposés lorsque leur \omterm{def:g1:addition:def}{somme} vaut $0$.
\end{solution}

\begin{solution}{exo:g7:negatives:2}
$-4 < 1$~; \quad $-4 > -7$~; \quad $-2.5 < -2.4$~; \quad $0 > -0.1$.
\end{solution}

\begin{solution}{exo:g7:negatives:3}
$-3.1 < -3 < -0.5 < 0 < 1 < 2.5$.
\end{solution}

\begin{solution}{exo:g7:negatives:4}
$(-5) + (-9) = -14$~; \quad
$(-12) + (+7) = -5$~; \quad
$(+3.5) + (-1.5) = +2$~; \quad
$(-6) + (+6) = 0$.
\end{solution}

\begin{solution}{exo:g7:negatives:5}
$4 - 11 = 4 + (-11) = -7$.

$(-5) - (+3) = (-5) + (-3) = -8$.

$(-7) - (-10) = (-7) + (+10) = +3$.

$0 - (-8) = 0 + (+8) = +8$.
\end{solution}

\begin{solution}{exo:g7:negatives:6}
$(+7) + (-12) + (+3) + (-8) = (+10) + (-20) = -10$ (positifs
ensemble, négatifs ensemble).

$(-5.5) + (+9) + (-4.5) + (+1) = (-10) + (+10) = 0$.
\end{solution}

\begin{solution}{exo:g7:negatives:7}
$(-3) + 8 - 12 = 5 - 12 = -7$~: il fait $-7$ degrés la nuit.
\end{solution}

\begin{solution}{exo:g7:negatives:8}
Tracé libre~; $E(0, -3)$ est sur l'axe vertical (sa première
coordonnée est $0$).
\end{solution}

\begin{solution}{exo:g7:negatives:9}
Moscou~: $(-4) - (-13) = -4 + 13 = 9$ degrés d'écart.
Helsinki~: $2 - (-9) = 11$ degrés.
\end{solution}

\begin{solution}{exo:g7:negatives:10}
$(-12) + 7 - 4 = -5 - 4 = -9$~: elle est à $-9$ m. Pour atteindre la
surface (altitude $0$) elle doit remonter $0 - (-9) = 9$ m.
\end{solution}

\begin{solution}{exo:g7:negatives:11}
$D(-2, -2)$~: même première coordonnée que $A$, même seconde
coordonnée que $C$. Côtés~: $AB = 4$ (de $-2$ à $2$) et $BC = 3$ (de
$1$ jusqu'à $-2$), donc le \omterm{def:g6:measure:perimeter}{périmètre} est $2 \times (4 + 3) = 14$.
\end{solution}

\begin{solution}{exo:g7:negatives:12}
La diagonale complète $0$, $-2$, $-4$ donne la \omterm{def:g1:addition:def}{somme} magique
$0 + (-2) + (-4) = -6$. Puis, case par case~:
en haut à gauche $= -6 - (-7) - 0 = 1$~; au milieu à droite
$= -6 - (-3) - (-2) = -1$~; en bas à droite $= -6 - 0 - (-1) = -5$~;
en bas au milieu $= -6 - (-7) - (-2) = 3$.
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|}
\hline
$1$ & $-7$ & $0$ \\
\hline
$-3$ & $-2$ & $-1$ \\
\hline
$-4$ & $3$ & $-5$ \\
\hline
\end{tabular}
\end{center}
Chaque ligne, colonne et diagonale a pour \omterm{def:g1:addition:def}{somme} $-6$.
\end{solution}

\begin{solution}{pb:g7:negatives:1}
\textbf{1.} $4\,808 - (-430) = 4\,808 + 430 = 5\,238$ m
(\cref{thm:g7:negatives:sub}~: soustraire un négatif, c'est
ajouter).

\textbf{2.} Solde~:
\[
(-45) + 120 + (-60) + 15 = 75 + (-60) + 15 = 15 + 15 = 30
\text{ euros}.
\]
Pour ramener le $-45$ initial à $0$, il fallait déposer son
opposé~: $45$ euros.

\textbf{3.} $(-44) - (-100) = (-44) + 100 = 56$~: César a vécu $56$
ans. (Aucun piège ici~: les deux années sont du même côté du
changement d'ère.)

\textbf{4.} L'an $0$~: compter de $-63$ jusqu'à $+14$ passe par $0$,
alors que le calendrier des historiens saute directement de l'an $1$
av. J.-C. à l'an $1$ apr. J.-C. --- l'an $0$ n'a jamais existé, donc
le compte naïf est trop long d'une année.

\textbf{5.} $63$ av. J.-C. devient $-(63 - 1) = -62$. Alors
$14 - (-62) = 14 + 62 = 76$ ans~: exactement le chiffre des
historiens. Sur la droite renumérotée des astronomes, il n'y a plus
de trou~: la soustraction ordinaire donne donc les durées exactes.

\textbf{6.} $1$ av. J.-C. $\to 0$~; \ $10$ av. J.-C. $\to -9$~; \
$476$ apr. J.-C. $\to +476$~; \ $753$ av. J.-C. $\to -752$.

\textbf{7.} $476 - (-752) = 476 + 752 = 1\,228$ ans.

\textbf{8.} $12$ av. J.-C., c'est $-11$. Apparitions suivantes~:
$-11 + 76 = 65$, puis $141$, puis $217$~: les années $65$, $141$ et
$217$ apr. J.-C. Avant~: $-11 - 76 = -87$, soit l'an
$87 + 1 = 88$ av. J.-C. des historiens.

\textbf{9.} $5$ av. J.-C., c'est $-4$, donc $5 - (-4) = 9$ ans ---
et non $10$. L'an \omterm{def:g1:counting:zero}{zéro} manquant frappe chaque fois qu'une durée
franchit le changement d'ère.

\textbf{10.} $3$ av. J.-C., c'est $-2$~; dix ans plus tard,
$-2 + 10 = +8$~: l'enfant a eu dix ans en l'an $8$ apr. J.-C.

\textbf{11.} Entre $-7$ et $-2$~: $(-2) - (-7) = 5$. Entre $-3.5$ et
$4.5$~: $4.5 - (-3.5) = 8$.

\textbf{12.} En regroupant les opposés et les nombres amis~:
\[
3 + (-5) + 2 + (-1) + 4
= \bigl(3 + 2\bigr) + (-5) + (-1) + 4
= 0 + (-1) + 4 = +3 :
\]
l'ascenseur s'arrête à l'étage $3$. L'ordre n'a pas d'importance~:
une \omterm{def:g1:addition:def}{somme} de \omterm{def:g7:negatives:def}{nombres relatifs} se réorganise librement (chaque
déplacement se parcourt sur la même droite, et le trajet total est le
même).

\textbf{13.} Variation totale~:
$2 + (-4) + 1 + 0 + (-3) + 5 + (-2) = -1$ degré. En partant de
$-2$~: température finale $-2 + (-1) = -3$ degrés. En général, la
valeur finale est la valeur initiale augmentée de la \omterm{def:g1:addition:def}{somme} de toutes
les variations --- inutile de suivre les hausses et les baisses une
à une.

\textbf{14.} Revenir au point de départ signifie une variation
totale nulle~: les hausses et les baisses doivent se compenser
exactement. Les cinq variations connues totalisent
$4 + (-7) + 2 + 3 + (-5) = -3$, donc la sixième est l'opposée~:
$+3$ (trois cents mètres de montée).

\textbf{15.} Chez les astronomes~: $509$ av. J.-C. $\to -508$, et
$476 - (-508) = 984$ ans. Le $509 + 476 = 985$ de l'historien compte
une année de trop --- une fois encore l'an \omterm{def:g1:counting:zero}{zéro} fantôme,
silencieusement inclus dans l'addition mais absent de l'histoire.
(Vérification par les deux périodes~: République de $-508$ à $-26$,
soit $482$ ans~; Empire de $-26$ à $+476$, soit $502$ ans~;
$482 + 502 = 984$.)
\end{solution}
