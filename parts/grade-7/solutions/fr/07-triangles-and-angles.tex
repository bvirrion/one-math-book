\begin{solution}{exo:g7:triangles:2}
L'angle opposé par le \omterm{def:g5:solids:def}{sommet} mesure aussi $35^\circ$
(\cref{thm:g7:triangles:equalities})~; les deux angles restants lui
sont supplémentaires~: $180 - 35 = 145^\circ$ chacun.
\end{solution}

\begin{solution}{exo:g7:triangles:3}
(a)~$180 - 100 = 80^\circ$. \quad
(b)~$180 - 118 = 62^\circ$. \quad
(c)~$180 - 150 = 30^\circ$.
\end{solution}

\begin{solution}{exo:g7:triangles:4}
Angle au \omterm{def:g5:solids:def}{sommet} $40^\circ$~: les angles à la base se partagent
$180 - 40 = 140^\circ$ également~: $70^\circ$ chacun.

Angle à la base $70^\circ$~: les deux angles à la base sont égaux,
donc l'angle au \omterm{def:g5:solids:def}{sommet} vaut $180 - 2 \times 70 = 40^\circ$.
\end{solution}

\begin{solution}{exo:g7:triangles:5}
$(6, 8, 12)$~: $12 < 6 + 8 = 14$~: le triangle existe.

$(5, 5, 11)$~: $11 > 5 + 5 = 10$~: impossible.

$(4, 9, 13)$~: $13 = 4 + 9$~: «~triangle~» plat --- les points sont
alignés, pas de vrai triangle.

$(7, 7, 7)$~: $7 < 14$~: \omterm{def:g6:shapes:triangles}{triangle équilatéral}.
\end{solution}

\begin{solution}{exo:g7:triangles:6}
$x + 2x + 3x = 6x = 180^\circ$, donc $x = 30^\circ$~: les angles sont
$30^\circ$, $60^\circ$ et $90^\circ$ --- un \omterm{def:g6:shapes:triangles}{triangle rectangle}.
\end{solution}

\begin{solution}{exo:g7:triangles:7}
Prédiction~: $\widehat{BAC} = 180 - (50 + 60) = 70^\circ$~; le
rapporteur sur la figure terminée le confirme.
\end{solution}

\begin{solution}{exo:g7:triangles:8}
À la première intersection, les quatre angles sont $54^\circ$,
$126^\circ$, $54^\circ$, $126^\circ$ (paires d'\omterm{def:g7:triangles:pairs}{opposés par le sommet} et
suppléments). La \omterm{def:g6:lines:perp}{parallèle} reproduit les mêmes quatre mesures à la
seconde intersection (angles correspondants)~: huit angles au total,
quatre de $54^\circ$ et quatre de $126^\circ$.
\end{solution}

\begin{solution}{exo:g7:triangles:9}
Soit $c$ le troisième côté. L'inégalité triangulaire exige
$c < 8 + 3 = 11$ et aussi $8 < c + 3$, c'est-à-dire $c > 5$. Donc
$5 < c < 11$~: la longueur $7$ cm convient~; la longueur $12$ cm (ou
$4$ cm) ne convient pas.
\end{solution}

\begin{solution}{exo:g7:triangles:10}
$\widehat A + \widehat B = 180 - 90 = 90^\circ$, et
$\widehat A = 2\widehat B$, donc $3\widehat B = 90^\circ$~:
$\widehat B = 30^\circ$ et $\widehat A = 60^\circ$.
\end{solution}

\begin{solution}{exo:g7:triangles:11}
La diagonale $[AC]$ découpe $ABCD$ en triangles $ABC$ et $ACD$. Les
quatre angles du \omterm{def:g4:geometry:polygon}{quadrilatère} sont exactement les six angles des deux
triangles, regroupés (les angles en $A$ et en $C$ sont chacun
partagés en deux). Total~: $180 + 180 = 360^\circ$. Un \omterm{def:g4:geometry:polygon}{pentagone} se
découpe depuis un \omterm{def:g5:solids:def}{sommet} en trois triangles~: $3 \times 180 =
540^\circ$.
\end{solution}

\begin{solution}{pb:g7:triangles:1}
\textbf{1.} Les diagonales issues d'un \omterm{def:g5:solids:def}{sommet} d'un \omterm{def:g4:geometry:polygon}{hexagone} le
découpent en $4$ triangles~: les angles intérieurs totalisent donc
$4 \times 180 = 720^\circ$.

\textbf{2.} Depuis un \omterm{def:g5:solids:def}{sommet}, les diagonales vont vers tous les
\omterm{def:g5:solids:def}{sommets} sauf lui-même et ses deux voisins~: $n - 3$ diagonales, qui
découpent le \omterm{def:g4:geometry:polygon}{polygone} en $n - 2$ triangles. Chaque angle intérieur du
\omterm{def:g4:geometry:polygon}{polygone} se répartit entre les triangles~: la \omterm{def:g1:addition:def}{somme} vaut donc
\[
(n - 2) \times 180^\circ .
\]
Vérification~: $n = 3$~: $180^\circ$~; $n = 4$~: $360^\circ$
(\cref{exo:g7:triangles:11})~; $n = 5$~: $540^\circ$~; $n = 6$~:
$720^\circ$.

\textbf{3.} Décagone~: $8 \times 180 = 1\,440^\circ$. Dodécagone~:
$10 \times 180 = 1\,800^\circ$.

\textbf{4.} En divisant par $n$~: triangle $60^\circ$~; carré
$90^\circ$~; \omterm{def:g4:geometry:polygon}{pentagone} $\frac{540}{5} = 108^\circ$~; \omterm{def:g4:geometry:polygon}{hexagone}
$\frac{720}{6} = 120^\circ$~; \omterm{def:g4:geometry:polygon}{octogone}
$\frac{1080}{8} = 135^\circ$.

\textbf{5.} Ajouter un \omterm{def:g5:solids:def}{sommet} ajoute un triangle de plus à
l'éventail~: le nouveau \omterm{def:g4:geometry:polygon}{polygone} demande $n - 1$ triangles là où
l'ancien en demandait $n - 2$. Un triangle de plus, un $180^\circ$
de plus.

\textbf{6.} Le marcheur tourne du supplémentaire~:
$180^\circ - \widehat A$. Pour le triangle $40$--$60$--$80$, les
virages valent $140^\circ$, $120^\circ$ et $100^\circ$.

\textbf{7.} $140 + 120 + 100 = 360^\circ$~: un tour complet.

\textbf{8.} Sur l'ensemble du parcours, notre nez part et revient
dans la même direction, après avoir balayé exactement un tour~:
tous les virages pris aux coins totalisent une révolution complète,
$360^\circ$ --- quels que soient le nombre de coins et la forme du
pâté de maisons. (Le fait est magnifiquement indépendant de $n$.)

\textbf{9.} À chaque coin, intérieur $+$ virage $= 180^\circ$. En
sommant sur les $n$ coins~:
\[
\text{(somme des intérieurs)} + 360^\circ = 180^\circ \times n,
\qquad\text{donc}\qquad
\text{somme des intérieurs} = (n - 2) \times 180^\circ
\]
--- la formule de la question 2, redémontrée en marchant.

\textbf{10.} Un angle intérieur de $150^\circ$ signifie un virage de
$30^\circ$, et $n = \frac{360}{30} = 12$~: le dodécagone régulier.
Un angle intérieur de $155^\circ$ signifierait des virages de
$25^\circ$, et $\frac{360}{25} = 14.4$ n'est pas un nombre entier de
coins~: un tel \omterm{def:g4:geometry:polygon}{polygone} régulier n'existe pas.

\textbf{11.} Triangle ($60^\circ$)~: $6$ se rejoignent,
$6 \times 60 = 360$. Carré ($90^\circ$)~: $4$ se rejoignent.
\omterm{def:g4:geometry:polygon}{Hexagone} ($120^\circ$)~: $3$ se rejoignent,
$3 \times 120 = 360$. Les trois se referment parfaitement.

\textbf{12.} Trois coins de \omterm{def:g4:geometry:polygon}{pentagone}~: $3 \times 108 = 324^\circ$
--- il reste un trou de $36^\circ$. Quatre coins~:
$4 \times 108 = 432^\circ > 360^\circ$ --- chevauchement. Ni l'un ni
l'autre ne convient~: les \omterm{def:g4:geometry:polygon}{pentagones} réguliers ne peuvent pas paver
le plan.

\textbf{13.} À un coin de pavage, au moins $3$ carreaux se
rejoignent. Un \omterm{def:g4:geometry:polygon}{polygone} régulier à plus de six côtés a des angles
intérieurs \emph{supérieurs} à $120^\circ$ (question 4 et croissance
de l'angle avec $n$)~: trois coins dépassent donc
$3 \times 120 = 360^\circ$~: chevauchement, impossible. Avec six
côtés exactement, $3 \times 120 = 360$ convient~; en dessous de six,
les questions 11 et 12 règlent les cas. Liste complète des paveurs
réguliers~: triangle, carré, \omterm{def:g4:geometry:polygon}{hexagone}.

\textbf{14.} \omterm{def:g4:geometry:polygon}{Octogones} et carrés~: $135 + 135 + 90 = 360^\circ$~: le
coin se referme --- le motif classique des trottoirs. Carré,
\omterm{def:g4:geometry:polygon}{hexagone}, dodécagone~: le virage du dodécagone vaut
$\frac{360}{12} = 30^\circ$, son angle intérieur vaut donc
$150^\circ$, et $90 + 120 + 150 = 360^\circ$~: cela fonctionne
aussi.

\textbf{15.} Coins du \omterm{def:g3:shapes:shapes}{rayon} de miel~: $3 \times 120 = 360^\circ$,
parfait. Parmi les carreaux triangulaires, carrés et hexagonaux de
même \omterm{def:g6:measure:area}{aire}, l'\omterm{def:g4:geometry:polygon}{hexagone} a le bord le plus court --- c'est celui des
trois qui se rapproche le plus du cercle de Didon
(\cref{pb:g6:measure:1}) --- donc des alvéoles hexagonales
enferment le même miel avec le moins de cire~: les abeilles pavent
en géomètres.
\end{solution}
