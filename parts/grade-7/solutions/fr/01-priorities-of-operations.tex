\section*{Chapitre \ref{ch:g7:priorities} --- Priorités opératoires}

\begin{solution}{exo:g7:priorities:1}
$5 + 3 \times 6 = 5 + 18 = 23$.

$(5 + 3) \times 6 = 8 \times 6 = 48$.

$30 - 12 \div 4 = 30 - 3 = 27$.

$(30 - 12) \div 4 = 18 \div 4 = 4.5$.
\end{solution}

\begin{solution}{exo:g7:priorities:2}
$24 - 6 + 2 = 18 + 2 = 20$ (gauche à droite).

$24 \div 6 \times 2 = 4 \times 2 = 8$ (gauche à droite).

$24 - 6 \times 2 = 24 - 12 = 12$ (multiplication d'abord).

$24 \div (6 \times 2) = 24 \div 12 = 2$.
\end{solution}

\begin{solution}{exo:g7:priorities:3}
$7 + 3 \times (10 - 6) = 7 + 3 \times 4 = 7 + 12 = 19$.

$40 - (2 + 3) \times 4 = 40 - 5 \times 4 = 40 - 20 = 20$.

$60 \div \bigl(2 \times (7 - 4)\bigr) = 60 \div (2 \times 3)
= 60 \div 6 = 10$.
\end{solution}

\begin{solution}{exo:g7:priorities:4}
$\dfrac{18 + 6}{8} = \dfrac{24}{8} = 3$~; \quad
$\dfrac{30}{7 - 2} = \dfrac{30}{5} = 6$~; \quad
$\dfrac{5 \times 8}{4 + 6} = \dfrac{40}{10} = 4$.
\end{solution}

\begin{solution}{exo:g7:priorities:5}
$8 \times 102 = 8 \times (100 + 2) = 800 + 16 = 816$.

$25 \times 99 = 25 \times (100 - 1) = 2500 - 25 = 2475$.

$14 \times 12 + 14 \times 8 = 14 \times (12 + 8) = 14 \times 20 = 280$.

$37 \times 45 - 37 \times 35 = 37 \times (45 - 35) = 37 \times 10 =

\begin{solution}{exo:g7:priorities:6}
«~La \omterm{ex:g1:subtraction:difference}{différence} de $50$ et du \omterm{def:g2:mult:def}{produit} de $6$ par $7$~»~:
$50 - 6 \times 7 = 50 - 42 = 8$ (pas de parenthèses nécessaires, la
priorité fait le travail).

«~Le \omterm{def:g2:mult:def}{produit} de la \omterm{def:g1:addition:def}{somme} de $4$ et $5$ par la \omterm{ex:g1:subtraction:difference}{différence} de $10$ et
$8$~»~: $(4 + 5) \times (10 - 8) = 9 \times 2 = 18$.
\end{solution}

\begin{solution}{exo:g7:priorities:8}
(a)~sans parenthèses~: $4 + 2 \times 5 - 1 = 4 + 10 - 1 = 13$.

(b)~$(4 + 2) \times 5 - 1 = 30 - 1 = 29$.

(c)~$(4 + 2) \times (5 - 1) = 6 \times 4 = 24$.

\begin{solution}{exo:g7:priorities:9}
$k = 1$~: $12 + 4 \times 1 = 16$ et $16 \times 1 = 16$ --- elles
coïncident (c'est pourquoi $k = 1$ est un mauvais test~!). $k = 2$~:
$12 + 4 \times 2 = 20$ mais $16 \times 2 = 32$~: différentes. Donc
les deux expressions ne sont \emph{pas} égales en général. L'erreur
$12 + 4 \times k = 16 \times k$ vient d'additionner $12 + 4$
d'abord, en ignorant la priorité de $\times$.
\end{solution}

\begin{solution}{exo:g7:priorities:10}
\emph{1.} Prix du groupe~: $12 + 6 \times n$. Pour $n = 5$~:
$12 + 30 = 42$ euros.

\emph{2.} Prix individuel~: $9n$. Comparer~: $n = 3$~: $30$ contre
$27$ (individuel moins cher)~; $n = 4$~: $36$ contre $36$ (égalité)~;
$n = 5$~: $42$ contre $45$ (groupe moins cher). À partir de $5$
personnes, le tarif groupe l'emporte (chaque personne
supplémentaire coûte $6$ au lieu de $9$, donc l'écart ne fait
qu'augmenter).
\end{solution}

\begin{solution}{pb:g7:priorities:1}
\textbf{1.} $7 \times 102 = 7 \times (100 + 2) = 700 + 14 = 714$~;
\quad $9 \times 98 = 9 \times (100 - 2) = 900 - 18 = 882$.

\textbf{2.} $17 \times 13 + 17 \times 7 = 17 \times (13 + 7) =
17 \times 20 = 340$~; \quad $45 \times 101 - 45 = 45 \times (101
- 1) = 45 \times 100 = 4\,500$.

\textbf{3.} $5 \times 87 = 870 \div 2 = 435$. Légitimité~:
$5 = 10 \div 2$, donc multiplier par $5$ revient à multiplier par
$10$ puis à diviser par $2$ (l'ordre de $\times$ et de $\div$, de
même priorité, de la gauche vers la droite, fait le \omterm{def:g3:division:remainder}{reste}).

\textbf{4.} $25 \times 32 = 3\,200 \div 4 = 800$, car
$25 = 100 \div 4$. Et
$99 \times 47 = (100 - 1) \times 47 = 4\,700 - 47 = 4\,653$~:
distributivité sur la \omterm{ex:g1:subtraction:difference}{différence}
(\cref{thm:g7:priorities:distributivity}).

\textbf{5.} $8 \times (5 + 3) = 64$ et
$8 \times 5 + 8 \times 3 = 40 + 24 = 64$~: cela concorde --- c'est
la distributivité. Mais $8 \times (5 \times 3) = 8 \times 15 = 120$,
alors que $(8 \times 5) \times (8 \times 3) = 40 \times 24 = 960$~:
aucune concordance. Le facteur se répartit sur les \emph{termes
d'une \omterm{def:g1:addition:def}{somme}}, pas sur les facteurs d'un \omterm{def:g2:mult:def}{produit}.

\textbf{6.}
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{ccl}
$13$ & $24$ & conservée ($13$ \omterm{def:g3:numbers:evenodd}{impair}) \\
$6$ & $48$ & barrée ($6$ \omterm{def:g3:numbers:evenodd}{pair}) \\
$3$ & $96$ & conservée \\
$1$ & $192$ & conservée \\
\end{tabular}
\end{center}
\omterm{def:g1:addition:def}{Somme} des nombres de droite survivants~:
$24 + 96 + 192 = 312$, et l'on a bien $13 \times 24 = 312$.

\textbf{7.} \omterm{def:g3:division:half}{Moitiés} de $21$~: $21, 10, 5, 2, 1$~; doublements de
$17$~: $17, 34, 68, 136, 272$. Les gauches paires $10$ et $2$ sont
barrées~; survivants $17 + 68 + 272 = 357$. Vérification~:
$21 \times 17 = 357$.

\textbf{8.} \omterm{def:g3:division:half}{Moitiés} de $16$~: $16, 8, 4, 2, 1$~; doublements de
$25$~: $25, 50, 100, 200, 400$. Tous les nombres de gauche sauf le
dernier $1$ sont \omterm{def:g3:numbers:evenodd}{pairs}~: une seule ligne survit, et la réponse est
$400$. Une colonne de gauche formée de doublements purs ne conserve
que sa dernière ligne --- la méthode se réduit au doublement seul.

\textbf{9.} $13 = 12 + 1$, donc par distributivité
$13 \times 24 = 12 \times 24 + 1 \times 24$~; et
$12 \times 24 = 6 \times 48$ puisque prendre la \omterm{def:g3:division:half}{moitié} d'un facteur
en doublant l'autre laisse le \omterm{def:g2:mult:def}{produit} inchangé
($12 \times 24 = 6 \times 2 \times 24 = 6 \times 48$). Le nombre de
gauche \omterm{def:g3:numbers:evenodd}{impair} $13$ a détaché un exemplaire du nombre de droite~: la
première ligne met $24$ de côté.

\textbf{10.} $6 \times 48 = 3 \times 96$ (moitié--double, rien à
mettre de côté~: $6$ est \omterm{def:g3:numbers:evenodd}{pair}). $3 \times 96 = 2 \times 96 + 96 =
1 \times 192 + 96$~: le $3$ \omterm{def:g3:numbers:evenodd}{impair} met $96$ de côté. Enfin
$1 \times 192 = 192$~: le $1$ \omterm{def:g3:numbers:evenodd}{impair} met $192$ de côté et le tableau
s'achève. Total mis de côté~: $24 + 96 + 192 = 312$. Une ligne met
son nombre de droite de côté exactement lorsque son nombre de gauche
est \omterm{def:g3:numbers:evenodd}{impair} --- la recette «~garder les lignes de gauche impaire et
additionner~» est donc précisément la comptabilité de la
distributivité.

\textbf{11.} $1 + 4 + 8 = 13$. Pour $21$, les lignes survivantes
étaient celles de $17$, de $68 = 4 \times 17$ et de
$272 = 16 \times 17$~: les doublements $1 + 4 + 16 = 21$.

\textbf{12.} $25 = 16 + 8 + 1$ et $42 = 32 + 8 + 2$.

\textbf{13.} Prendre le plus grand doublement ne dépassant pas
l'objectif et le soustraire~; ce qui reste est plus petit, et plus
petit que le doublement qui vient d'être utilisé (sinon un plus
grand aurait tenu). En recommençant, le \omterm{def:g3:division:remainder}{reste} diminue strictement et
doit atteindre $0$~: le procédé s'arrête, et les doublements
utilisés sont tous différents. Pour $100$~: $64$ tient (il reste
$36$), $32$ tient (il reste $4$), $4$ tient (il reste $0$)~:
$100 = 64 + 32 + 4$.

\textbf{14.} Tableau des doublements de $35$~: $1 \to 35$,
$2 \to 70$, $4 \to 140$, $8 \to 280$, $16 \to 560$. Comme
$19 = 16 + 2 + 1$, on retient ces lignes~:
$560 + 70 + 35 = 665$. Vérification~: $19 \times 35 = 665$.

\textbf{15.} Les trois mêmes que le scribe et le paysan~:
$24 + 96 + 192$ --- les doublements de $24$ sélectionnés par
$13 = 1 + 4 + 8$. Écrire les nombres comme \omterm{def:g1:addition:def}{sommes} de doublements,
c'est exactement les écrire avec des $0$ et des $1$~; toute
l'histoire est racontée dans \cref{pb:g8:powers:1}.
\end{solution}
