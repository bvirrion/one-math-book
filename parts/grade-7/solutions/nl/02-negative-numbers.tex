\section*{Hoofdstuk \ref{ch:g7:negatives} --- Negatieve getallen}

\begin{solution}{exo:g7:negatives:1}
\omterm{def:g7:negatives:def}{Tegengestelden}: $-7$; \ $+3.5$; \ $0$ (zijn eigen \omterm{def:g7:negatives:def}{tegengestelde}); \ $-12$. Twee
getallen zijn tegengesteld als hun \omterm{def:g1:addition:def}{som} $0$ is.
\end{solution}

\begin{solution}{exo:g7:negatives:2}
$-4 < 1$; \quad $-4 > -7$; \quad $-2.5 < -2.4$; \quad $0 > -0.1$.
\end{solution}

\begin{solution}{exo:g7:negatives:3}
$-3.1 < -3 < -0.5 < 0 < 1 < 2.5$.
\end{solution}

\begin{solution}{exo:g7:negatives:4}
$(-5) + (-9) = -14$; \quad
$(-12) + (+7) = -5$; \quad
$(+3.5) + (-1.5) = +2$; \quad
$(-6) + (+6) = 0$.
\end{solution}

\begin{solution}{exo:g7:negatives:5}
$4 - 11 = 4 + (-11) = -7$.

$(-5) - (+3) = (-5) + (-3) = -8$.

$(-7) - (-10) = (-7) + (+10) = +3$.

$0 - (-8) = 0 + (+8) = +8$.
\end{solution}

\begin{solution}{exo:g7:negatives:6}
$(+7) + (-12) + (+3) + (-8) = (+10) + (-20) = -10$ (de positieve samen, de
negatieve samen).

$(-5.5) + (+9) + (-4.5) + (+1) = (-10) + (+10) = 0$.
\end{solution}

\begin{solution}{exo:g7:negatives:7}
$(-3) + 8 - 12 = 5 - 12 = -7$: 's nachts is het $-7$ graden.
\end{solution}

\begin{solution}{exo:g7:negatives:8}

\end{solution}

\begin{solution}{exo:g7:negatives:9}
Moskou: $(-4) - (-13) = -4 + 13 = 9$ graden \omterm{ex:g1:subtraction:difference}{verschil}. Helsinki:

\end{solution}

\begin{solution}{exo:g7:negatives:10}
$(-12) + 7 - 4 = -5 - 4 = -9$: ze is op $-9$ m. Om het oppervlak (hoogte $0$) te
bereiken, moet ze nog $0 - (-9) = 9$ m stijgen.
\end{solution}

\begin{solution}{exo:g7:negatives:11}
$D(-2, -2)$: dezelfde eerste coördinaat als $A$, dezelfde tweede coördinaat als
$C$. Zijden: $AB = 4$ (van $-2$ tot $2$) en $BC = 3$ (van $1$ naar $-2$), dus is
de \omterm{def:g6:measure:perimeter}{omtrek} $2 \times (4 + 3) = 14$.
\end{solution}

\begin{solution}{exo:g7:negatives:12}
De volledige diagonaal $0, -2, -4$ geeft de magische \omterm{def:g1:addition:def}{som}
$0 + (-2) + (-4) = -6$. Dan, vakje na vakje: linksboven
$= -6 - (-7) - 0 = 1$; midden rechts $= -6 - (-3) - (-2) = -1$; rechtsonder
$= -6 - 0 - (-1) = -5$; midden onder $= -6 - (-7) - (-2) = 3$.
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|c|c|c|}
\hline
$1$ & $-7$ & $0$ \\
\hline
$-3$ & $-2$ & $-1$ \\
\hline
$-4$ & $3$ & $-5$ \\
\hline
\end{tabular}
\end{center}
Elke rij, kolom en diagonaal heeft \omterm{def:g1:addition:def}{som} $-6$.
\end{solution}

\begin{solution}{pb:g7:negatives:1}
\textbf{1.} $4\,808 - (-430) = 4\,808 + 430 = 5\,238$ m
(\cref{thm:g7:negatives:sub}: een negatief getal aftrekken is optellen).

\textbf{2.} Saldo:
\[
(-45) + 120 + (-60) + 15 = 75 + (-60) + 15 = 15 + 15 = 30
\text{ euro}.
\]
Om de beginstand $-45$ op $0$ te brengen, stort je het \omterm{def:g7:negatives:def}{tegengestelde}: $45$ euro.

\textbf{3.} $(-44) - (-100) = (-44) + 100 = 56$: Caesar heeft $56$ jaar geleefd.
(Hier geen valstrik: beide jaren liggen aan dezelfde kant van de grens van de
jaartelling.)

\textbf{4.} Het jaar $0$: van $-63$ naar $+14$ tellen gaat door $0$, maar de
kalender van de historici springt rechtstreeks van $1$ v.Chr. naar $1$ n.Chr. ---
het jaar $0$ heeft nooit bestaan, dus is de naïeve telling een jaar te lang.

\textbf{5.} $63$ v.Chr. wordt $-(63 - 1) = -62$. Dan is
$14 - (-62) = 14 + 62 = 76$ jaar: precies het cijfer van de historici. Op de
herbenoemde \omterm{def:g6:lines:objects}{lijn} van de astronomen zitten geen gaten, dus geeft gewoon aftrekken
echte tijdsduren.

\textbf{6.} $1$ v.Chr. $\to 0$; \ $10$ v.Chr. $\to -9$; \ $476$ n.Chr.
$\to +476$; \ $753$ v.Chr. $\to -752$.

\textbf{7.} $476 - (-752) = 476 + 752 = 1\,228$ jaar.

\textbf{8.} $12$ v.Chr. is $-11$. Volgende verschijningen: $-11 + 76 = 65$, dan
$141$, dan $217$: de jaren $65$, $141$ en $217$ n.Chr. Daarvoor:
$-11 - 76 = -87$, en dat is het jaar $87 + 1 = 88$ v.Chr. van de historici.

\textbf{9.} $5$ v.Chr. is $-4$, dus $5 - (-4) = 9$ jaar --- geen $10$. Het
ontbrekende jaar \omterm{def:g1:counting:zero}{nul} slaat toe zodra een tijdsduur de grens van de jaartelling
oversteekt.

\textbf{10.} $3$ v.Chr. is $-2$; tien jaar later is $-2 + 10 = +8$: het kind werd
tien in het jaar $8$ n.Chr.

\textbf{11.} Tussen $-7$ en $-2$: $(-2) - (-7) = 5$. Tussen $-3.5$ en $4.5$:
$4.5 - (-3.5) = 8$.

\textbf{12.} Door \omterm{def:g7:negatives:def}{tegengestelden} en handige paren te groeperen:
\[
3 + (-5) + 2 + (-1) + 4
= \bigl(3 + 2\bigr) + (-5) + (-1) + 4
= 0 + (-1) + 4 = +3 :
\]
de lift eindigt op verdieping $3$. De volgorde maakt niets uit: een \omterm{def:g1:addition:def}{som} van
getallen met een teken mag je vrij herschikken (elke beweging wordt langs dezelfde
\omterm{def:g6:lines:objects}{lijn} gelopen, en de hele wandeling blijft dezelfde).

\textbf{13.} Nettoverandering:
$2 + (-4) + 1 + 0 + (-3) + 5 + (-2) = -1$ graad. Beginnend op $-2$: de
eindtemperatuur is $-2 + (-1) = -3$ graden. In het algemeen is de eindwaarde de
beginwaarde plus de \omterm{def:g1:addition:def}{som} van alle veranderingen --- de op- en neergangen hoef je
niet één voor één te volgen.

\textbf{14.} Terug bij het begin betekent nettoverandering $0$: de stijgingen en
dalingen moeten elkaar precies opheffen. De vijf bekende veranderingen geven samen
$4 + (-7) + 2 + 3 + (-5) = -3$, dus is de zesde het \omterm{def:g7:negatives:def}{tegengestelde}: $+3$
(driehonderd meter omhoog).

\textbf{15.} Met de astronomen: $509$ v.Chr. $\to -508$, en
$476 - (-508) = 984$ jaar. De $509 + 476 = 985$ van de historicus telt één jaar te
veel --- opnieuw het spookjaar \omterm{def:g1:counting:zero}{nul}, dat stilzwijgend in de optelling meegaat maar
in de geschiedenis ontbreekt. (Als controle de twee tijdvakken: Republiek $-508$
tot $-26$, $482$ jaar; Keizerrijk $-26$ tot $+476$, $502$ jaar;
$482 + 502 = 984$.)
\end{solution}
