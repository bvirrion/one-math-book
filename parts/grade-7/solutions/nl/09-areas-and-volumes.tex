\section*{Hoofdstuk \ref{ch:g7:areas} --- Oppervlaktes en volumes}

\begin{solution}{exo:g7:areas:1}
$8 \times 5 = 40$ cm$^2$; \qquad $6.5 \times 4 = 26$ m$^2$.
\end{solution}

\begin{solution}{exo:g7:areas:2}
$A = 10 \times 4 = 40$ cm$^2$. Het antwoord is niet $10 \times 6 = 60$, omdat de
zijde van $6$ cm schuin staat: de formule gebruikt de \emph{hoogte} (de loodrechte
afstand), en die is $4$ cm.
\end{solution}

\begin{solution}{exo:g7:areas:3}
$\frac{12 \times 7}{2} = 42$ cm$^2$; \quad
$\frac{9 \times 6}{2} = 27$; \quad
$\frac{5.5 \times 2}{2} = 5.5$ m$^2$.
\end{solution}

\begin{solution}{exo:g7:areas:4}
Schijf: $\pi \times 5^2 = 25\pi \approx 79$ cm$^2$. Halve schijf met \omterm{def:g3:shapes:shapes}{radius} $10$:
$\frac{\pi \times 10^2}{2} = 50\pi \approx 157$ cm$^2$.
\end{solution}

\begin{solution}{exo:g7:areas:5}
\omterm{def:g6:measure:perimeter}{Omtrek}: $2\pi \times 3 = 6\pi \approx 18.8$ cm (een lengte, in cm). \omterm{def:g6:measure:area}{Oppervlakte}:
$\pi \times 3^2 = 9\pi \approx 28.3$ cm$^2$ (een oppervlak, in cm$^2$).
\end{solution}

\begin{solution}{exo:g7:areas:6}
\omterm{def:g7:areas:prism}{Prisma}: $24 \times 10 = 240$ cm$^3$. \omterm{def:g7:areas:prism}{Cilinder}:
$\pi \times 3^2 \times 8 = 72\pi \approx 226$ cm$^3$.
\end{solution}

\begin{solution}{exo:g7:areas:7}
$\frac{8 \times h}{2} = 28$, dus $4h = 28$ en $h = 7$ cm.
\end{solution}

\begin{solution}{exo:g7:areas:8}
De drie \omterm{def:g2:mult:def}{producten} komen overeen omdat elk van hen \emph{dezelfde} grootheid uitrekent
--- de \omterm{def:g6:measure:area}{oppervlakte} van de driehoek. Dat geeft een praktische controle op constructies,
en een manier om een hoogte uit een ander paar basis en hoogte te berekenen (gebruikt
in \cref{exo:g7:areas:12}).
\end{solution}

\begin{solution}{exo:g7:areas:9}
De twee driehoeken hebben \omterm{def:g6:measure:area}{oppervlaktes} $\frac{30 \times 20}{2} = 300$ m$^2$ en
$\frac{18 \times 20}{2} = 180$ m$^2$: samen $480$ m$^2$. De algemene formule die
zich aandient: voor parallelle zijden $a$ en $b$ op afstand $h$,
\[
A = \frac{(a + b) \times h}{2}
\]
--- de trapeziumformule (hier $\frac{(30+18)\times 20}{2} = 480$).
\end{solution}

\begin{solution}{exo:g7:areas:10}
Watervolume: $\pi \times 6^2 \times 15 = 540\pi \approx 1\,696$ cm$^3$. Grondvlak van
de \omterm{def:g5:solids:def}{balk}: $18 \times 12 = 216$ cm$^2$. Bereikte hoogte:
$\frac{540\pi}{216} = 2.5\pi \approx 7.9$ cm.
\end{solution}

\begin{solution}{exo:g7:areas:11}
Radii $15$ en $20$ cm. \omterm{def:g6:measure:area}{Oppervlaktes}: $225\pi \approx 707$ cm$^2$ en
$400\pi \approx 1257$ cm$^2$. Prijs per $100$ cm$^2$:
$\frac{9}{7.07} \approx 1.27$ euro en $\frac{14}{12.57} \approx 1.11$ euro. De grote
pizza is voordeliger --- \omterm{def:g6:measure:area}{oppervlaktes} groeien met het \emph{kwadraat} van de
\omterm{def:g4:geometry:circle}{diameter}.
\end{solution}

\begin{solution}{exo:g7:areas:12}
Met de rechthoekszijden: $A = \frac{6 \times 8}{2} = 24$ cm$^2$. Met de schuine zijde
als basis: $24 = \frac{10 \times h}{2} = 5h$, dus $h = 4.8$ cm.
\end{solution}

\begin{solution}{pb:g7:areas:1}
\textbf{1.} Vierkant: $8 \times 8 = 64$. Rechthoek: $5 \times 13 = 65$. Dezelfde vier
stukken lijken in de ene figuur $64$ vierkante eenheden te bedekken en in de andere
$65$: er zou dus één eenheid \omterm{def:g6:measure:area}{oppervlakte} uit het niets geschapen zijn.

\textbf{2.} Elke driehoek: $\frac{8 \times 3}{2} = 12$. Elk trapezium:
$\frac{(5 + 3) \times 5}{2} = 20$.

\textbf{3.} $12 + 12 + 20 + 20 = 64$: de stukken zijn samen $64$, dus kunnen ze het
vierkant precies vullen --- en de rechthoek van $65$ eenheden \emph{niet}. In het
rechthoekplaatje moet dus een gat zitten.

\textbf{4.} Dezelfde steilheid zou betekenen dat $3$ over $8$ \omterm{def:g6:propdata:def}{evenredig} is met $2$
over $5$; de kruisproducten: $3 \times 5 = 15$ en $2 \times 8 = 16$. Niet gelijk: de
schuine zijde en de schuine rand liggen \emph{niet} in één \omterm{def:g6:lines:objects}{lijn}. De ``diagonaal'' van
de rechthoek is een leugen --- twee licht verschillende hellingen die onder een heel
flauwe hoek samenkomen.

\textbf{5.} Langs de valse diagonaal laten de vier stukken een lange, extreem dunne
spleet open --- een spie in de vorm van een heel plat \omterm{def:g7:central:parallelogram}{parallellogram} --- waarvan de
\omterm{def:g6:measure:area}{oppervlakte} precies de ontbrekende $65 - 64 = 1$ vierkante eenheid is. Uitgerekt over
een diagonaal van ongeveer $13$ eenheden is haar gemiddelde breedte ongeveer
$1 \div 13 \approx 0.08$ eenheid: op een echte tekening dunner dan de potloodstreep
die haar verbergt.

\textbf{6.} Elke driehoek heeft basis $6$ cm en hoogte $4$ cm --- de afstand tussen de
parallellen, waar de top ook ligt --- dus is elke \omterm{def:g6:measure:area}{oppervlakte}
$\frac{6 \times 4}{2} = 12$ cm$^2$ (\cref{thm:g7:areas:triangle}). De top langs de
\omterm{def:g6:lines:perp}{parallel} schuiven verandert de vorm, nooit de basis of de hoogte: eeuwig gelijke
\omterm{def:g6:measure:area}{oppervlaktes}.

\textbf{7.} De diagonaal knipt het trapezium in een driehoek met basis $B$ en een met
basis $b$, beide met hoogte $h$ (de afstand tussen de parallelle zijden):
\[
A = \frac{B \times h}{2} + \frac{b \times h}{2}
= \frac{(B + b) \times h}{2} .
\]

\textbf{8.} $A = \frac{(9 + 5) \times 4}{2} = \frac{56}{2} = 28$ cm$^2$.

\textbf{9.} Het gemiddelde van de parallelle zijden is $\frac{B + b}{2}$, dus is
(gemiddelde) $\times h = \frac{(B+b) \times h}{2}$: dezelfde formule, anders
gegroepeerd. Controle: gemiddelde $= \frac{9+5}{2} = 7$, en $7 \times 4 = 28$ cm$^2$.

\textbf{10.} De scheve stapel bevat precies dezelfde kaarten --- dezelfde lagen, elk
met dezelfde \omterm{def:g6:measure:area}{oppervlakte} --- en dus hetzelfde \omterm{def:g6:measure:volume}{volume}. In $V = B \times h$ moet de
hoogte \emph{\omterm{def:g6:lines:perp}{loodrecht}} op het grondvlak gemeten worden (de verticale hoogte van de
stapel), en niet langs de schuinte: precies zoals de \omterm{def:g6:measure:area}{oppervlakte} van het
\omterm{def:g7:central:parallelogram}{parallellogram} de loodrechte hoogte wil en niet de schuine zijde
(\cref{exo:g7:areas:2}), één dimensie hoger.

\textbf{11.} Ring $= \pi \times 5^2 - \pi \times 3^2 = 25\pi - 9\pi = 16\pi \approx
50$ m$^2$ (\cref{thm:g7:areas:disk}).

\textbf{12.} Een schijf met \omterm{def:g6:measure:area}{oppervlakte} $16\pi$ heeft \omterm{def:g3:shapes:shapes}{radius} $4$, want
$\pi \times 4^2 = 16\pi$. De radii $3$, $4$ en $5$: het beroemdste drietal van de
wiskunde, met een hoofdrol in \cref{ch:g8:pythagoras}.

\textbf{13.} Één pizza van $40$ cm: \omterm{def:g3:shapes:shapes}{radius} $20$, \omterm{def:g6:measure:area}{oppervlakte}
$\pi \times 400 \approx 1\,257$ cm$^2$. Twee pizza's van $28$ cm: \omterm{def:g3:shapes:shapes}{radius} $14$, samen
$2 \times \pi \times 196 = 392\pi \approx 1\,232$ cm$^2$. De ene grote pizza is
\emph{meer} pizza dan de twee middelgrote.

\textbf{14.} De \omterm{def:g3:shapes:shapes}{radius} met $1.4$ schalen schaalt de \omterm{def:g6:measure:area}{oppervlakte} met
$1.4^2 = 1.96$ --- bijna het dubbele. De precieze factor is het getal waarvan het
kwadraat $2$ is, ongeveer $1.414$: het diagonaalgetal van
\cref{ex:g8:pythagoras:diagonal}. (Vuistregel: de \omterm{def:g3:division:half}{helft} breder is bijna twee keer zo
veel pizza.)

\textbf{15.} $13 \times 13 = 169$, maar $8 \times 21 = 168$: deze keer
\emph{verdwijnt} er een vierkante eenheid --- de stukken overlappen in een dunne spie
in plaats van een gat open te laten. Met de opeenvolgende termen $3, 5, 8, 13, 21$ van
de rij van Hemachandra en Fibonacci wippen de \omterm{def:g2:mult:def}{producten} heen en weer tussen één meer en
één minder dan het kwadraat ($5 \times 13 = 65 = 8^2 + 1$, en dan
$8 \times 21 = 168 = 13^2 - 1$): de goochelaar ``wint'' en ``verliest'' afwisselend een
eenheid. \omterm{def:g6:measure:area}{Oppervlaktes} liegen nooit: stukken die samen $64$ zijn, bedekken $64$, waar je
ze ook heen schuift --- elke ontbrekende of extra eenheid zit in een spie verstopt, en
wacht op een hellingcontrole om ontmaskerd te worden.
\end{solution}
