\section*{Hoofdstuk \ref{ch:g7:stats} --- Gegevens ordenen}

\begin{solution}{exo:g7:stats:1}
Elk cijfer geteld:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|ccccc|c}
cijfer & $9$ & $11$ & $12$ & $14$ & $15$ & totaal \\
\hline
\omterm{def:g7:stats:frequency}{aantal} & $4$ & $3$ & $7$ & $3$ & $3$ & $20$ \\
\end{tabular}
\end{center}
\end{solution}

\begin{solution}{exo:g7:stats:2}
\omterm{def:g7:stats:frequency}{Frequenties}: $\frac{4}{20} = 20\,\%$, $\frac{3}{20} = 15\,\%$,
$\frac{7}{20} = 35\,\%$, $15\,\%$, $15\,\%$. \omterm{def:g1:addition:def}{Som}:
$20 + 15 + 35 + 15 + 15 = 100\,\%$. \checkmark
\end{solution}

\begin{solution}{exo:g7:stats:3}
Eerste onderzoek: $\frac{30}{50} = 0.6 = 60\,\%$. Tweede:
$\frac{45}{90} = 0.5 = 50\,\%$. Het eerste onderzoek toont het hoogste aandeel, ook
al zijn er in absolute aantallen minder gezinnen met één auto.
\end{solution}

\begin{solution}{exo:g7:stats:4}
$\intco{5}{15}$: $8, 12, 9, 14, 11$ --- \omterm{def:g7:stats:frequency}{aantal} $5$.
$\intco{15}{25}$: $23, 22, 18, 16$ --- \omterm{def:g7:stats:frequency}{aantal} $4$.
$\intco{25}{35}$: $31, 27, 29, 25, 33$ --- \omterm{def:g7:stats:frequency}{aantal} $5$.
$\intco{35}{45}$: $35$ --- \omterm{def:g7:stats:frequency}{aantal} $1$. Totaal $15$. \checkmark
\end{solution}

\begin{solution}{exo:g7:stats:5}
Vier staven van hoogte $8$, $12$, $6$ en $4$ boven de waarden $0$, $1$, $2$ en $3$
(een streepje per $2$ werkt goed).
\end{solution}

\begin{solution}{exo:g7:stats:6}
Totaal $30$ gezinnen. \omterm{def:g7:stats:frequency}{Frequenties}: $\frac{8}{30}$, $\frac{12}{30}$,
$\frac{6}{30}$, $\frac{4}{30}$. Hoeken ($\times 360^\circ$): $96^\circ$,
$144^\circ$, $72^\circ$, $48^\circ$; \omterm{def:g1:addition:def}{som} $96 + 144 + 72 + 48 = 360^\circ$.
\checkmark
\end{solution}

\begin{solution}{exo:g7:stats:7}
Percentages: zomer $\frac{162}{360} = 45\,\%$; lente $\frac{90}{360} = 25\,\%$;
herfst en winter elk $\frac{54}{360} = 15\,\%$. Van $60$ mensen koos
$0.45 \times 60 = 27$ de zomer.
\end{solution}

\begin{solution}{exo:g7:stats:8}
Klas A: $\frac{12}{30} = 40\,\%$. Klas B: $\frac{14}{40} = 35\,\%$. Klas A heeft
het hoogste aandeel lopers, ook al heeft klas B er in \omterm{def:g7:stats:frequency}{aantal} meer.
\end{solution}

\begin{solution}{exo:g7:stats:9}
Met de as vanaf $9\,700$ hebben de drie staven zichtbare hoogtes $100$, $300$ en
$400$ eenheden: de laatste lijkt vier keer de eerste, wat suggereert dat de verkoop
verviervoudigd is. Met een as vanaf $0$ zijn de staven bijna gelijk (van $9\,800$
naar $10\,100$ is een stijging van ongeveer $3\,\%$): het eerlijke plaatje toont een
bijna vlakke verkoop.
\end{solution}

\begin{solution}{exo:g7:stats:10}
\omterm{def:g7:stats:frequency}{Frequenties} zijn samen $1$: $f = 1 - 0.35 - 0.4 = 0.25$. De aantallen (totaal
$80$): $0.35 \times 80 = 28$; $0.4 \times 80 = 32$; $0.25 \times 80 = 20$.
Controle: $28 + 32 + 20 = 80$.
\end{solution}

\begin{solution}{exo:g7:stats:11}
Meisjes: $55\,\%$ van $400 = 220$; jongens: $180$. Meisjes in de kantine:
$0.4 \times 220 = 88$; jongens: $0.6 \times 180 = 108$. Samen:
$88 + 108 = 196$ leerlingen, dus $\frac{196}{400} = 49\,\%$ van de school ---
tussen $40\,\%$ en $60\,\%$, en dichter bij het cijfer van de meisjes omdat er meer
meisjes zijn.
\end{solution}

\begin{solution}{pb:g7:stats:1}
\textbf{1.} Aantallen: $120 > 90$, school A wint. \omterm{def:g7:stats:frequency}{Frequenties}:
$\frac{120}{400} = 0.30 = 30\,\%$ tegen $\frac{90}{250} = 0.36 = 36\,\%$: school B
wint. De prijs moet de \omterm{def:g7:stats:frequency}{frequentie} volgen --- school B recyclet een groter deel van
wat ze gebruikt; school A drinkt alleen meer sap.

\textbf{2.} Leraren: $\frac{18}{30} = 0.60 = 60\,\%$. Leerlingen:
$\frac{210}{600} = 0.35 = 35\,\%$. Samen:
$\frac{18 + 210}{630} = \frac{228}{630} \approx 0.36 = 36\,\%$. De leerlingen zijn
twintig keer zo talrijk, dus wordt de gecombineerde \omterm{def:g7:stats:frequency}{frequentie} bijna volledig naar
hun kant getrokken --- de grote groep weegt zwaar.

\textbf{3.} \omterm{def:g7:stats:frequency}{Frequenties} moeten samen $100\,\%$ zijn
(\cref{def:g7:stats:frequency}), maar $45 + 30 + 20 + 10 = 105$: minstens één
sector is fout.

\textbf{4.} Het \omterm{def:g7:stats:frequency}{aantal} bij \omterm{def:g7:stats:frequency}{frequentie} $0.15$: $0.15 \times 40 = 6$. De aantallen tot
nu toe: $14 + 10 + 6 = 30$, dus is het laatste \omterm{def:g7:stats:frequency}{aantal} $40 - 30 = 10$, met
\omterm{def:g7:stats:frequency}{frequentie} $\frac{10}{40} = 0.25$. (Controle:
$0.35 + 0.25 + 0.15 + 0.25 = 1$.)

\textbf{5.} Hoeken: $0.40 \times 360 = 144^\circ$, $0.35 \times 360 = 126^\circ$,
$0.25 \times 360 = 90^\circ$; en $144 + 126 + 90 = 360^\circ$: de cirkel sluit.

\textbf{6.} $\frac{2.4}{10} = 0.24 = 24\,\%$ van de biljetten kwam terug.

\textbf{7.} In telefoongidsen en autoregistraties stonden vooral rijke huishoudens,
waarvan de stem afweek van die van het land; de $2.4$ miljoen antwoorden waren een
gigantisch \omterm{def:g7:stats:frequency}{aantal}, getrokken uit de verkeerde bevolking. Zoals in vraag 1: het gaat
er niet om hoeveel mensen je telt, maar wie --- een \omterm{def:g7:stats:frequency}{frequentie} berekend op een
vertekende steekproef beschrijft de steekproef, niet het land.

\textbf{8.} Echte steun:
$0.7 \times 1\,000 + 0.3 \times 9\,000 = 700 + 2\,700 = 3\,400$ steunbetuigingen
van $10\,000$: $34\,\%$. De peiling:
$0.7 \times 500 + 0.3 \times 500 = 350 + 150 = 500$ van $1\,000$: $50\,\%$ ---
zestien punten te hoog, omdat de rijken de \omterm{def:g3:division:half}{helft} van de steekproef zijn maar een
tiende van het dorp.

\textbf{9.} Antwoorden: $0.4 \times 200 = 80$ ontevreden en $0.1 \times 800 = 80$
tevreden, dus $160$ antwoorden waarvan $80$ ontevreden: de gemeten ontevredenheid
is $\frac{80}{160} = 50\,\%$. De echte ontevredenheid:
$\frac{200}{1\,000} = 20\,\%$. Niemand heeft gelogen --- de ontevreden klanten
antwoorden gewoon vaker, en het onderzoek hoort hen luider.

\textbf{10.} De echte verhoudingen: een tiende rijk, dus $100$ rijke en $900$
minder rijke gesprekken: $0.7 \times 100 + 0.3 \times 900 = 70 + 270 = 340$ van
$1\,000$: de gerepareerde peiling voorspelt $34\,\%$ --- de waarheid van vraag 8.
Vijftigduizend goed gekozen gesprekken verslaan twee miljoen slecht gekozen
gesprekken.

\textbf{11.} Ziekenhuis A: licht $\frac{90}{100} = 90\,\%$; ernstig
$\frac{280}{400} = 70\,\%$; in het totaal
$\frac{90 + 280}{500} = \frac{370}{500} = 74\,\%$.

\textbf{12.} Ziekenhuis B: licht $\frac{340}{400} = 85\,\%$; ernstig
$\frac{65}{100} = 65\,\%$; in het totaal
$\frac{340 + 65}{500} = \frac{405}{500} = 81\,\%$. Ziekenhuis A wint bij de lichte
gevallen ($90 > 85$) \emph{en} bij de ernstige ($70 > 65$).

\textbf{13.} In het totaal laat B $81\,\%$ zien tegen de $74\,\%$ van A: het
ziekenhuis dat in \emph{elke} categorie verliest, wint het totaal. (Deze omkering
heeft een naam: de paradox van Simpson.)

\textbf{14.} De samenstellingen zijn tegengesteld: de patiënten van A zijn meestal
ernstig ($400$ van $500$), die van B meestal licht ($400$ van $500$). Ernstige
gevallen genezen minder vaak, waar ze ook behandeld worden, dus wordt het totaal van
A naar beneden getrokken door de moeilijke gevallen die het aanneemt --- het totaal
is een gewogen mengsel, en de gewichten verschillen per ziekenhuis. Een ernstige
patiënt moet ziekenhuis A kiezen: $70\,\%$ verslaat $65\,\%$ in de enige rij die
hem aangaat.

\textbf{15.} Bijvoorbeeld: ``Uw vergelijking mengt lichte en ernstige patiënten, en
de twee ziekenhuizen behandelen \omterm{def:g7:negatives:def}{tegengestelde} samenstellingen --- opgesplitst naar
ernst geneest ziekenhuis A van \emph{beide} soorten een groter deel. Totaalcijfers
mag je alleen vergelijken als de groepen erachter op elkaar lijken; anders beslissen
de gewichten en niet de kwaliteit wie er wint.''
\end{solution}
