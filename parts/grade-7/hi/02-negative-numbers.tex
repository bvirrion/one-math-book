\chapter{ऋण संख्याएँ}\label{ch:g7:negatives}

\omterm{def:g1:counting:zero}{शून्य} से नीचे का तापमान, ज़मीन के नीचे की मंज़िलें, चुकाया जाने
वाला क़र्ज़: कई राशियाँ $0$ से नीचे चली जाती हैं। ऋण संख्याएँ
संख्या रेखा को \omterm{def:g1:counting:zero}{शून्य} के बाईं ओर बढ़ा देती हैं। यह अध्याय उनकी
तुलना करना और उन्हें जोड़ना-घटाना सिखाता है --- उनका गुणा
\cref{ch:g8:negprod} में आएगा।

\section{सापेक्ष संख्याएँ}

\begin{definition}[सापेक्ष संख्याएँ]\label{def:g7:negatives:def}
\emph{सापेक्ष संख्या}\index{सापेक्ष संख्या} एक चिह्न
($+$ या $-$) और \emph{\omterm{def:g1:counting:zero}{शून्य} से दूरी} से बनती है: $+3$ और $-3$
दोनों $0$ से $3$ की दूरी पर हैं, पर उलटी ओर। धन संख्याएँ आम तौर
पर बिना चिह्न के लिखी जाती हैं: $+3 = 3$। संख्याएँ $+3$ और $-3$
एक-दूसरे की \emph{विपरीत}\index{विपरीत संख्या} हैं।
\end{definition}

\begin{tikzpicture}[scale=1.15]
  \draw[-Stealth] (-5.6,0) -- (5.6,0);
  \foreach \x in {-5,...,5} \draw (\x,-0.08) -- (\x,0.08);
  \foreach \x in {-5,...,5} \node[below, font=\small] at (\x,-0.1) {$\x$};

    node[midway, above, font=\small] {$3$};
\end{tikzpicture}

{\small संख्या रेखा बाईं ओर बढ़ी हुई: $-3$ और $+3$ \omterm{def:g7:negatives:def}{विपरीत
संख्याएँ} हैं, $0$ से एक ही दूरी पर।}
\end{omfigure}

\begin{method}[सापेक्ष संख्याओं की तुलना]\label{met:g7:negatives:compare}
संख्या रेखा पर जितना बाईं ओर, उतनी छोटी। व्यवहार में:
\begin{enumerate}
  \item ऋण संख्या हमेशा किसी भी धन संख्या से छोटी होती है:
        $-7 < 2$;
  \item दो धन संख्याओं में वही पुराना क्रम;
  \item दो ऋण संख्याओं में जिसकी \omterm{def:g1:counting:zero}{शून्य} से दूरी \emph{ज़्यादा} हो
        वही \emph{छोटी} है: $-7 < -2$।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:negatives:compare}
सबसे छोटी से सबसे बड़ी के क्रम में लगाओ: $3$; $-5$; $0$; $-1.5$; $2.8$।
पहले ऋण संख्याएँ, सबसे अधिक ऋण वाली आगे:
\[
-5 < -1.5 < 0 < 2.8 < 3 .
\]
\end{example}

\section{सापेक्ष संख्याएँ जोड़ना}

किसी \omterm{def:g7:negatives:def}{सापेक्ष संख्या} को रेखा पर चाल की तरह सोचो: $+5$ यानी
``दाईं ओर $5$ कदम चलो'', $-3$ यानी ``बाईं ओर $3$ कदम चलो''।
संख्याएँ जोड़ना यानी चालों को एक के बाद एक जोड़ना।

\begin{method}[दो सापेक्ष संख्याएँ जोड़ना]\label{met:g7:negatives:add}
\begin{enumerate}
  \item \emph{एक ही चिह्न}: दूरियाँ जोड़ो, साझा चिह्न रखो:
        $(-3) + (-4) = -7$।
  \item \emph{उलटे चिह्न}: बड़ी दूरी में से छोटी घटाओ, और बड़ी
        वाली का चिह्न लो:
        $(-7) + (+2) = -5$ और $(+7) + (-2) = +5$।
  \item \omterm{def:g7:negatives:def}{विपरीत संख्याएँ} कट जाती हैं: $(+4) + (-4) = 0$।
\end{enumerate}
\end{method}

\begin{tikzpicture}[scale=1.05]
  \draw[-Stealth] (-5.4,0) -- (4.4,0);
  \foreach \x in {-5,...,4} \draw (\x,-0.07) -- (\x,0.07);
  \foreach \x in {-5,-3,-1,0,2,4}
    \node[below, font=\small] at (\x,-0.09) {$\x$};
  \draw[-Stealth, omThm, very thick] (0,0.35) -- (-3.5,0.35);
  \draw[-Stealth, omDef, very thick] (-3.5,0.75) -- (-1.5,0.75);
  \node[omThm, font=\small] at (-1.75,0.18) {$-3.5$};
  \node[omDef, font=\small] at (-2.5,1.0) {$+2$};
  \fill (-1.5,0) circle (2pt) node[below=6pt, font=\small] {$-1.5$};
\end{tikzpicture}

{\small चलने के रूप में जोड़: $0$ से शुरू करके पहले चाल $-3.5$ और
फिर चाल $+2$, $(-3.5) + 2 = -1.5$ पर पहुँचा देती है।}
\end{omfigure}

\begin{example}\label{ex:g7:negatives:add}
कदम-दर-कदम, जहाँ सुविधा हो वहाँ समूह बनाकर:
\begin{align*}
(-8) + (+3) &= -5 && \text{(उलटे चिह्न: } 8 - 3 = 5,
\text{ चिह्न } 8 \text{ वाला)}\\
(-2.5) + (-1.5) &= -4 && \text{(एक ही चिह्न: दूरियाँ जोड़ो)}\\
(+6) + (-9) + (+3) &= (-3) + (+3) = 0 .
\end{align*}
\end{example}

\section{सापेक्ष संख्याएँ घटाना}

\begin{theorem}[घटाव का नियम]\label{thm:g7:negatives:sub}
किसी \omterm{def:g7:negatives:def}{सापेक्ष संख्या} को घटाना उसकी \emph{\omterm{def:g7:negatives:def}{विपरीत संख्या} जोड़ने} के
बराबर है:
\[
a - b = a + (-b) .
\]
\end{theorem}

\begin{proof}[यह चलता क्यों है]
$a - b$ वह संख्या है जिसे $b$ में जोड़ने पर $a$ मिले। जाँचो कि
$a + (-b)$ यह काम कर देती है:
$\bigl(a + (-b)\bigr) + b = a + \bigl((-b) + b\bigr) = a + 0 = a$।
\end{proof}

\begin{example}\label{ex:g7:negatives:sub}
हर घटाव जोड़ बन जाता है, और फिर
\cref{met:g7:negatives:add} लागू होती है:
\begin{align*}
5 - 9 &= 5 + (-9) = -4, \\
(-3) - (+4) &= (-3) + (-4) = -7, \\
(-2) - (-6) &= (-2) + (+6) = +4 .
\end{align*}
आख़िरी पंक्ति मशहूर है: \emph{ऋण संख्या घटाना यानी जोड़ना}।
क़र्ज़ हट जाए तो तुम अमीर हो जाते हो!
\end{example}

\begin{example}[तापमान के अंतर]\label{ex:g7:negatives:temperature}
एक दिन में तापमान $-6$ डिग्री से $+9$ डिग्री तक गया। बढ़त यह
\omterm{ex:g1:subtraction:difference}{अंतर} है
\[
9 - (-6) = 9 + 6 = 15 \text{ डिग्री}.
\]
संख्या रेखा पर: $-6$ से $0$ तक $6$ डिग्री, और $0$ से $9$ तक
$9$ डिग्री और।
\end{example}

\section{पूरे समतल में निर्देशांक}

\begin{definition}[चिह्नों वाले निर्देशांक]\label{def:g7:negatives:coords}
मूल बिंदु $O$ पर काटती हुई दो \omterm{def:g6:lines:perp}{लंब} अंकित अक्षों के साथ समतल के हर
बिंदु को दो सापेक्ष निर्देशांक $(x, y)$ मिल जाते हैं: ऋण $x$
यानी खड़ी अक्ष के बाईं ओर, ऋण $y$ यानी आड़ी अक्ष के नीचे।
\end{definition}

\section{समस्या: वह शून्यवाँ वर्ष जो कभी था ही नहीं}

\begin{problem}[{सप्ताहांत समस्या --- इतिहास की समय-रेखा पर
सापेक्ष संख्याएँ: युग-सीमा के पार अवधियाँ, खगोलविदों की
तरकीब, और शुद्ध-परिवर्तन का सिद्धांत}]
\label{pb:g7:negatives:1}
इस अध्याय की संख्या रेखा का असली दुनिया में एक मशहूर जुड़वाँ है:
इतिहास की समय-रेखा, जिसमें ईसा-पूर्व के वर्ष बाईं ओर और उसके
बाद के वर्ष दाईं ओर फैले हैं। पर इतिहासकारों की रेखा में एक जाल
छिपा है जिसने कई गणनाएँ बिगाड़ी हैं: \emph{कोई शून्यवाँ वर्ष है
ही नहीं} --- ईसा-पूर्व $1$ के तुरंत बाद ईसा के बाद का $1$ आ जाता
है। यह समस्या तापमान, लिफ़्ट और बैंक-खातों की गणना करती है, फिर
इतिहास के टूटे \omterm{def:g1:counting:zero}{शून्य} की मरम्मत वैसे ही करती है जैसे खगोलविद
करते हैं: \omterm{def:g7:negatives:def}{सापेक्ष संख्याओं} से।

\medskip
\noindent\textbf{भाग I --- शुरुआती अभ्यास, और पहला जाल।}
\begin{enumerate}
  \item मों ब्लां की चोटी $+4\,808$ m की ऊँचाई पर है; मृत सागर
        का \omterm{def:g5:solids:def}{किनारा} $-430$ m पर। दोनों की ऊँचाई का \omterm{ex:g1:subtraction:difference}{अंतर}
        निकालो।
  \item किसी बैंक-खाते में $-45$ यूरो हैं (ओवरड्राफ़्ट)। उसका
        मालिक $120$ यूरो जमा करता है, $60$ यूरो का बिल चुकाता
        है, फिर $15$ यूरो जमा करता है। शेष को जोड़ों की एक ही
        कड़ी के रूप में लिखो और निकालो। शुरू के $-45$ को ठीक
        $0$ पर लाने के लिए एक ही बार में कितना जमा करना पड़ता?
  \item जूलियस सीज़र का जन्म ईसा-पूर्व $100$ में हुआ और मृत्यु
        ईसा-पूर्व $44$ में। ईसा-पूर्व के वर्षों को ऋण संख्याएँ
        मानकर (ईसा-पूर्व $100$ को $-100$, ईसा-पूर्व $44$ को
        $-44$), $(-44) - (-100)$ निकालो: सीज़र कितने साल जिया?
  \item सम्राट ऑगस्टस का जन्म ईसा-पूर्व $63$ में हुआ और मृत्यु
        ईसा के बाद $14$ में। वही विधि $14 - (-63) = 77$ साल
        देती है --- पर इतिहासकार कहते हैं कि उसकी मृत्यु $76$ की
        उम्र में हुई। उनकी आपत्ति: गिनती $77$ ऐसे वर्ष से होकर
        चलती है जो कभी था ही नहीं। कौन-सा?
  \item खगोलविद समय-रेखा की मरम्मत नाम बदलकर करते हैं: ईसा के
        बाद के वर्ष अपनी संख्या रखते हैं ($+14$, $+14$ ही रहता
        है), पर ईसा-पूर्व $1$ को $0$, ईसा-पूर्व $2$ को $-1$, और
        आम तौर पर ईसा-पूर्व $n$ को $-(n - 1)$ बना दिया जाता है।
        ईसा-पूर्व $63$ को बदलो, और प्रश्न 4 की गणना खगोलविदों की
        संख्याओं से दोबारा करो। क्या अब उत्तर इतिहासकारों को
        सही लगता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- युगों के पार गणना।}
अब से हर गणना में खगोलविदों वाला नामकरण इस्तेमाल करो, और अंत में
ईसा-पूर्व/ईसा के बाद में वापस बदल लो ($0$ या ऋण $-m$ नाम वाला
वर्ष इतिहासकारों का ईसा-पूर्व $m + 1$ है)।
\begin{enumerate}[resume]
  \item खगोलविदों की संख्याओं में बदलो: ईसा-पूर्व $1$;
        ईसा-पूर्व $10$; ईसा के बाद $476$; और ईसा-पूर्व $753$,
        जब कहा जाता है कि रोम बसा था।
  \item रोम के बसने (ईसा-पूर्व $753$) से पश्चिमी साम्राज्य के
        पतन (ईसा के बाद $476$) तक कितने साल बीते?
  \item कोई पुच्छल तारा हर $76$ साल बाद लौटता है और ईसा-पूर्व
        $12$ में दिखा था। उसके अगले तीन दर्शनों के, और
        ईसा-पूर्व $12$ से ठीक पहले वाले दर्शन के, इतिहासकारों
        वाले वर्ष बताओ।
  \item ईसा-पूर्व $5$ से ईसा के बाद $5$ तक कितने साल बीते?
        (लुभावना उत्तर $10$ है; समय-रेखा कुछ और कहती है।)
  \item ईसा-पूर्व $3$ में जन्मा बच्चा किस वर्ष (ईसा-पूर्व या
        ईसा के बाद) दस साल का हुआ?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- शुद्ध-परिवर्तन का सिद्धांत।}
\begin{enumerate}[resume]
  \item रेखा पर दो संख्याओं के बीच की \emph{दूरी} बड़ी में से
        छोटी घटाने पर मिलती है --- और हमेशा धन होती है।
        $-7$ और $-2$ के बीच की दूरी निकालो, फिर
        $-3.5$ और $4.5$ के बीच की।
  \item एक लिफ़्ट मंज़िल $0$ से शुरू होकर $+3$,
        $-5$, $+2$, $-1$, $+4$ चालें करती है। चतुराई से समूह
        बनाकर आख़िरी मंज़िल निकालो (\cref{met:g7:negatives:add},
        जहाँ हो सके वहाँ पहले \omterm{def:g7:negatives:def}{विपरीत संख्याएँ})। चालें दूसरे क्रम
        में करने से मंज़िल बदल जाती? क्यों?
  \item एक हफ़्ते में दिन-दर-दिन तापमान $+2$, $-4$, $+1$,
        $0$, $-3$, $+5$, $-2$ डिग्री बदलता है।
        \emph{शुद्ध परिवर्तन} निकालो। हफ़्ता $-2$ डिग्री से शुरू
        हुआ हो, तो कहाँ ख़त्म हुआ? सिद्धांत लिखो: आख़िरी मान $=$
        शुरुआती मान $+$ (सारे परिवर्तनों का \omterm{def:g1:addition:def}{योगफल})।
  \item किसी पैदल यात्री की चाल उसे ठीक उसी ऊँचाई पर वापस ले आती
        है जहाँ से वह चली थी। उसकी ऊँचाई के सारे परिवर्तनों का
        \omterm{def:g1:addition:def}{योगफल} क्या होना चाहिए, और क्यों? उसके छह दर्ज
        परिवर्तनों में से पाँच हैं $+4$, $-7$, $+2$, $+3$, $-5$
        (सैकड़ों मीटर में): छठा निकालो।
  \item आख़िरी सवाल, फिर रोम में। एक इतिहासकार गिनता है:
        ``गणतंत्र ईसा-पूर्व $509$ से ईसा-पूर्व $27$ तक चला, फिर
        साम्राज्य ईसा के बाद $476$ तक; कुल मिलाकर ईसा-पूर्व
        $509$ से ईसा के बाद $476$ तक, यानी
        $509 + 476 = 985$ साल का रोमन राज्य।'' गणना खगोलविदों की
        संख्याओं से दोबारा करो। सही कुल क्या है --- और इस बार भी
        किस वर्ष ने ग़लती करवाई?
\end{enumerate}
\end{problem}
