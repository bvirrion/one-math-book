\chapter{अक्षर-व्यंजक}\label{ch:g7:literal}

गणना में लिखा कोई अक्षर या तो ऐसी संख्या के लिए खड़ा होता है जो अभी
पता नहीं है, या फिर एक साथ \emph{सारी} संख्याओं के लिए: $P = 4c$
लिखते ही हर वर्ग का \omterm{def:g6:measure:perimeter}{परिमाप} एक ही सूत्र में आ जाता है। यह अध्याय
लिखने के तौर-तरीक़े, अक्षरों की जगह मान रखना, और आसान व्यंजकों को
सरल करना सिखाता है --- यहीं से बीजगणित शुरू होता है, जो
\cref{ch:g8:equations} में आगे बढ़ता है।

\section{अक्षरों से लिखना}

\begin{definition}[अक्षर-व्यंजक]\label{def:g7:literal:def}
\emph{अक्षर-व्यंजक}\index{अक्षर-व्यंजक} ऐसी गणना है जिसमें एक या
अधिक अक्षर हों और हर अक्षर किसी संख्या के लिए खड़ा हो। लिखने के
तौर-तरीक़े:
\begin{itemize}
  \item अक्षर या कोष्ठक से पहले $\times$ का चिह्न छोड़ दिया जाता है:
        $3 \times a = 3a$, \ $a \times b = ab$, \
        $2 \times (x + 5) = 2(x + 5)$;
  \item पर दो संख्याओं के बीच वह रहता है: $3 \times 5$ को $35$ नहीं
        लिखा जा सकता!
  \item $a \times a$ को $a^2$ लिखते हैं (``$a$ का वर्ग''),
        और $a \times a \times a = a^3$;
  \item $1a$ को $a$ लिखते हैं, और $0a$ को $0$।
\end{itemize}
\end{definition}

\begin{example}\label{ex:g7:literal:write}
लिखावट सरल करो:
\[
5 \times x + 3 \times y = 5x + 3y,
\qquad
a \times 7 = 7a \ \text{(संख्या पहले)},
\qquad
x \times x \times 4 = 4x^2 .
\]
\end{example}

\begin{example}[सूत्र भी अक्षर-व्यंजक हैं]\label{ex:g7:literal:formulas}
लंबाई $L$ और चौड़ाई $w$ वाले आयत के लिए: \omterm{def:g6:measure:perimeter}{परिमाप} $P = 2(L + w)$,
\omterm{def:g6:measure:area}{क्षेत्रफल} $A = Lw$। एक ही सूत्र अनगिनत आयतों को एक साथ समेट लेता
है --- अक्षरों का पूरा मतलब यही है।
\end{example}

\begin{tikzpicture}[scale=0.8]
  \draw[thick] (0,0) rectangle (4.4,2.0);
  \node[below, font=\small] at (2.2,0) {$L$};
  \node[left, font=\small] at (0,1.0) {$w$};
  \node[font=\small] at (2.2,1.0) {$A = Lw$};
  \node[above, font=\small, omDef] at (2.2,2.05) {$P = 2(L + w)$};
\end{tikzpicture}

संख्याएँ मिल जाती हैं।}
\end{omfigure}

\section{मान रखना}

\begin{method}[व्यंजक का मान निकालना]\label{met:g7:literal:substitute}
अक्षरों के दिए हुए मानों पर किसी व्यंजक का मान निकालने के लिए:
\begin{enumerate}
  \item व्यंजक को छिपे हुए $\times$ चिह्न वापस डालकर फिर से लिखो:
        $3a + a^2$ यानी $3 \times a + a \times a$;
  \item हर अक्षर की जगह उसका मान रखो, ज़रूरत हो तो उसके चारों ओर
        कोष्ठक लगाकर;
  \item \cref{ch:g7:priorities} की प्राथमिकताओं का ध्यान रखते हुए
        गणना करो।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:literal:substitute}
$x = 5$ पर $E = 3x + x^2$ का मान निकालो:
\[
E = 3 \times 5 + 5 \times 5 = 15 + 25 = 40 .
\]
$a = 4$ और $b = 0.5$ पर $F = 2(a + 3b)$ का मान निकालो:
\[
F = 2 \times (4 + 3 \times 0.5) = 2 \times (4 + 1.5) = 2 \times 5.5
= 11 .
\]
\end{example}

\section{व्यंजकों को सरल करना}

\begin{proposition}[समान पदों को मिलाना]\label{prop:g7:literal:reduce}
जिन पदों का अक्षर वाला हिस्सा एक जैसा हो, उन्हें उनके संख्या वाले
हिस्से जोड़कर मिलाया जा सकता है (यह उल्टा पढ़ा हुआ वितरण नियम है,
\cref{thm:g7:priorities:distributivity}):
\[
5x + 3x = (5 + 3)x = 8x,
\qquad
7a - 2a = 5a .
\]
अलग-अलग अक्षर वाले पदों को (जैसे $3x$ और $5y$, या $x$ और $x^2$)
आपस में नहीं मिलाया जा सकता।
\end{proposition}

\begin{example}\label{ex:g7:literal:reduce}
पदों को छाँटते हुए $E = 4x + 7 + 3x + 2$ को सरल करो:
\[
E = (4x + 3x) + (7 + 2) = 7x + 9 .
\]
ध्यान दो: $7x + 9$, $16x$ \emph{नहीं} है! $x = 1$ पर जाँचो:
$4 + 7 + 3 + 2 = 16$ और $7 \times 1 + 9 = 16$ मिल जाते हैं, पर
$16x$ भी तब $16$ ही होता --- $x = 2$ पर जाँचो: असली व्यंजक से
$8 + 7 + 6 + 2 = 23$, और $7 \times 2 + 9 = 23$, जबकि
$16 \times 2 = 32$। सरल रूप $7x + 9$ सही है, $16x$ ग़लत।
\end{example}

\begin{method}[किसी समानता की जाँच]\label{met:g7:literal:test}
यह देखने के लिए कि दो व्यंजक हर मान पर बराबर हो सकते हैं या नहीं:
\begin{enumerate}
  \item दोनों में एक ही मान रखो, कम से कम दो बार (जैसे $x = 2$ और
        $x = 10$; $0$ और $1$ से बचो, वे बहुत सारे \omterm{ex:g1:subtraction:difference}{अंतर} छिपा लेते
        हैं);
  \item अगर एक बार भी उत्तर अलग निकले तो व्यंजक बराबर \emph{नहीं}
        हैं --- एक प्रति-उदाहरण किसी भी आम दावे को गिरा देता है;
  \item अगर उत्तर मिल जाएँ तो समानता सिर्फ़ \emph{संभव} लगती है:
        प्रमाण के लिए बीजगणित चाहिए (फैलाना, सरल करना)।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:literal:test}
क्या $2(x + 3)$ हर $x$ के लिए $2x + 6$ के बराबर है? फैलाने से यह
सिद्ध हो जाता है: $2(x+3) = 2x + 2 \times 3 = 2x + 6$। क्या
$(x + 1)^2$, $x^2 + 1$ के बराबर है? $x = 2$ पर जाँचो:
$(2+1)^2 = 9$ पर $2^2 + 1 = 5$। नहीं --- और एक प्रति-उदाहरण
``नहीं'' का पूरा प्रमाण है।
\end{example}

\section{व्यंजक बनाना}

\begin{example}[स्थिति से व्यंजक तक]\label{ex:g7:literal:produce}
एक टैक्सी $4$ यूरो और उसके ऊपर हर किलोमीटर के $2$ यूरो लेती है।
$k$ किलोमीटर की सवारी का दाम है
\[
p = 4 + 2k \quad\text{यूरो}.
\]
$k = 7$ पर: $p = 4 + 14 = 18$ यूरो। व्यंजक \emph{ही} आम उत्तर है;
मान रखते ही हर ख़ास उत्तर निकल आता है।
\end{example}

\begin{example}[संख्या के खेल]\label{ex:g7:literal:trick}
``कोई संख्या सोचो, उसे दुगुना करो, $10$ जोड़ो, $2$ से भाग दो, अपनी
संख्या घटा दो।'' उस संख्या को $n$ कहो और क़दमों पर चलो:
\[
n \ \to\ 2n \ \to\ 2n + 10 \ \to\ \frac{2n + 10}{2} = n + 5 \ \to\
n + 5 - n = 5 .
\]
$n$ चाहे जो रहा हो, हर किसी को $5$ मिलता है --- अक्षर इस जादू का
राज़ खोल देते हैं।
\end{example}

\section{समस्या: तीलियों की भविष्यवाणी}

\begin{problem}[{सप्ताहांत समस्या --- बढ़ते ढर्रे और उनका $n$-वाँ
पद: एक ही ढर्रे के तीन सूत्र, और वह मशहूर ढर्रा जो छठे क़दम पर
टूट जाता है}]
\label{pb:g7:literal:1}
तीलियों को वर्गों की एक क़तार में सजाओ: पहले एक वर्ग, फिर दो, फिर
तीन\,\dots\ सौवीं आकृति में कितनी तीलियाँ लगेंगी? एक-एक करके
गिनने में पूरा सप्ताहांत निकल जाएगा; \emph{$n$ वाला सूत्र} हर
आकृति का जवाब एक साथ पल भर में दे देता है --- अक्षर ठीक इसीलिए हैं
(\cref{ex:g7:literal:formulas})। पर सावधान: यह समस्या एक ऐसे नामी
ढर्रे पर ख़त्म होती है जो पाँच क़दम तक ईमानदारी से चलता है और फिर
उस पर भरोसा करने वाले हर किसी को धोखा दे जाता है।

\medskip
\noindent\textbf{भाग I --- वर्गों की एक क़तार।}
आकृति $1$ तीलियों का एक वर्ग है ($4$ तीलियाँ); उसके बाद हर आकृति
एक और वर्ग जोड़ती है, जो पिछले वर्ग के साथ एक भुजा बाँटता है।
\begin{enumerate}
  \item आकृति $1$ से $4$ तक बनाओ और उनकी तीलियाँ गिनो।
  \item समझाओ कि हर नया वर्ग ठीक $3$ ज़्यादा तीलियाँ माँगता है,
        और इसी से बिना किसी सूत्र के आकृति $10$ की गिनती बताओ।
  \item सूत्र का कारण बताओ: आकृति $n$ को $3n + 1$ तीलियाँ चाहिए।
        (वह अकेला ``$+1$'' कहाँ से आता है?) सवाल 1 से उसे जाँचो।
  \item आमिर दूसरी तरह सोचता है: ``पहले वर्ग के लिए $4$ तीलियाँ,
        फिर बाक़ी $n - 1$ वर्गों में हर एक के लिए $3$'', यानी
        $4 + 3(n - 1)$। उसके व्यंजक को फैलाकर सरल करो
        (\cref{prop:g7:literal:reduce}): क्या वह सवाल 3 से सहमत
        है?
  \item लेना तीसरी तरह गिनती है: ``$2n$ आड़ी तीलियाँ, और $n + 1$
        खड़ी।'' उसका सूत्र लिखो, उसे सरल करो, और नतीजा निकालो।
        फिर शुरू वाले सवाल का जवाब दो: आकृति $100$ के लिए कितनी
        तीलियाँ?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- त्रिभुज, मेज़ें, और उल्टी दिशा में
चलना।}
\begin{enumerate}[resume]
  \item तीलियों के त्रिभुजों की एक क़तार (हर नया त्रिभुज पिछले पर
        टिका हुआ): $1$, $2$, $3$ त्रिभुजों की तीलियाँ गिनो, फिर
        $n$ त्रिभुजों के लिए सूत्र $2n + 1$ का कारण बताओ।
  \item उल्टी दिशा में: ठीक $49$ तीलियों से क़तार में कितने
        त्रिभुज बन सकते हैं? (सूत्र को क़दम दर क़दम उलटो।)
  \item दावत की वर्गाकार मेज़ों की हर ख़ाली भुजा पर एक मेहमान
        बैठता है। जब $n$ मेज़ें एक लंबी क़तार में सटा दी जाती
        हैं, तो समझाओ कि बैठने का सूत्र $2n + 2$ क्यों है, और उसे
        $1$, $2$, $3$ मेज़ों पर जाँचो।
  \item एक दावत में $30$ मेहमान आने हैं। एक क़तार के लिए कितनी
        मेज़ें चाहिए? खानसामा उन्हीं मेज़ों को सात-सात की
        \emph{दो} क़तारों में बाँट देता है: अब कितने मेहमान बैठ
        पाएँगे? सूत्र से यह फ़ायदा समझाओ।
  \item बिंदुओं का ढर्रा: आकृति $n$ बिंदुओं की एक L है, $n$
        बिंदु ऊँची और $n$ बिंदु चौड़ी (कोना एक ही बार गिना हुआ)।
        आकृति $1$, $2$, $3$ के बिंदु गिनो, सूत्र निकालो, और जो
        संख्याएँ मिलें उनका नाम बताओ। (एक के भीतर एक रखकर ये L
        मिलकर एक वर्ग ढक देती हैं --- यह कहानी
        \cref{pb:g8:equations:1} में आगे चलती है।)
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- भरोसा, जाँच, प्रमाण।}
\begin{enumerate}[resume]
  \item एक सहपाठी का दावा है कि भाग I की आकृति $n$ को
        $3(n + 2) - 5$ तीलियाँ चाहिए। फैलाकर और सरल करके इस दावे
        का फ़ैसला करो --- जाँचने की ज़रूरत नहीं।
  \item ``वर्ग करने से संख्याएँ बड़ी होती हैं, इसलिए हर संख्या
        $n$ के लिए $n^2 \geq n$।'' इस दावे को $n = 2$, $3$, $10$
        पर जाँचो; फिर ऐसी संख्या ढूँढ़ो जिस पर यह टूट जाए। तीन
        सफल जाँचों ने क्या सिद्ध किया, और अकेली नाकामी क्या
        सिद्ध करती है (\cref{met:g7:literal:test})?
  \item एक वृत्त पर बिंदु लगाओ और उनके बीच \emph{हर} \omterm{def:g6:lines:objects}{रेखाखंड}
        खींचो, फिर वृत्त के भीतर बने क्षेत्र गिनो। यह $2$, $3$,
        $4$ और $5$ बिंदुओं के लिए करो (बिंदु असमान दूरी पर रखो,
        ताकि कोई तीन \omterm{def:g6:lines:objects}{रेखाखंड} भीतर एक ही बिंदु पर न कटें)।
        गिनतियाँ लिख लो। यह ढर्रा किस सूत्र की ओर इशारा करता है?
  \item अब $6$ बिंदु लो, बड़े और असमान दूरी पर बनाकर, और बहुत
        ध्यान से गिनो (छोटे-छोटे क्षेत्र बहुत हैं)। तुम्हें कितने
        क्षेत्र मिले --- और उस अटकल का क्या हुआ?
  \item दो वाक्यों में सीख: दुगुना होने वाली अटकल को \emph{सिद्ध}
        करने के लिए क्या चाहिए था, और भाग I तथा II के तीली-सूत्र
        कभी ऐसे ख़तरे में क्यों नहीं थे? एक सिद्ध भविष्यवाणी के
        साथ जश्न मनाओ: $2\,026$ वर्गों की क़तार के लिए कितनी
        तीलियाँ?
\end{enumerate}
\end{problem}
