\chapter{त्रिभुज और कोण}\label{ch:g7:triangles}

क्या तुम $10$, $3$ और $2$ भुजाओं का त्रिभुज बना सकते हो? और
$80^\circ$, $70^\circ$ तथा $50^\circ$ कोणों वाला? यह अध्याय दोनों
सवालों का जवाब समतल ज्यामिति के दो सबसे काम के तथ्यों से देता है:
त्रिभुज असमिका और कोणों का $180^\circ$ \omterm{def:g1:addition:def}{योगफल} --- और साथ में वह
कोण-शब्दावली जो कटती तथा \omterm{def:g6:lines:perp}{समांतर} रेखाओं के साथ आती है।

\section{कटती रेखाओं के आसपास के कोण}

\begin{definition}[कोणों के जोड़े]\label{def:g7:triangles:pairs}
\begin{itemize}
  \item दो कोण \emph{पूरक} तब होते हैं जब उनके माप जुड़कर
        $90^\circ$ बनाएँ, और \emph{संपूरक} तब जब वे जुड़कर
        $180^\circ$ बनाएँ।
  \item जब दो रेखाएँ कटती हैं, तो कटान-बिंदु के पार एक-दूसरे के
        सामने पड़ने वाले कोण \emph{शीर्षाभिमुख
        कोण}\index{शीर्षाभिमुख कोण} कहलाते हैं।
  \item जब कोई रेखा दो और रेखाओं को काटती है, तो दोनों कटानों पर
        एक ही जगह बैठे कोण \emph{संगत कोण} कहलाते हैं; और दोनों
        रेखाओं के बीच, काटने वाली रेखा की उलटी-उलटी तरफ़ पड़ने वाले
        कोण \emph{एकांतर कोण}\index{एकांतर कोण}।
\end{itemize}
\end{definition}

\begin{theorem}[कोणों की बराबरी]\label{thm:g7:triangles:equalities}
\begin{enumerate}
  \item \omterm{def:g7:triangles:pairs}{शीर्षाभिमुख कोण} बराबर होते हैं।
  \item अगर दो रेखाएँ \omterm{def:g6:lines:perp}{समांतर} हों, तो संगत कोण बराबर होते हैं, और
        \omterm{def:g7:triangles:pairs}{एकांतर कोण} भी बराबर होते हैं। उलटे तरफ़ से: संगत (या
        एकांतर) कोणों का बराबर होना रेखाओं को \omterm{def:g6:lines:perp}{समांतर} होने पर मजबूर
        कर देता है।
\end{enumerate}
\end{theorem}

\begin{proof}[1 का प्रमाण]
किसी रेखा की एक ही तरफ़ पड़े दो कोण $\widehat{1}$ और $\widehat{2}$
मिलकर एक ऋजु कोण बनाते हैं: $\widehat 1 + \widehat 2 = 180^\circ$।
$\widehat 1$ का \omterm{def:g7:triangles:pairs}{शीर्षाभिमुख कोण} $\widehat 3$ भी उसी कारण से
$\widehat 2 + \widehat 3 = 180^\circ$ मानता है। इसलिए
$\widehat 1 = 180^\circ - \widehat 2 = \widehat 3$। बात 2 इस स्तर
पर मान ली जाती है।
\end{proof}

\section{त्रिभुज के कोणों का योग}

\begin{theorem}[कोणों का योग]\label{thm:g7:triangles:sum}
प्रत्येक त्रिभुज में तीनों कोण जुड़कर $180^\circ$ बनाते हैं।
\end{theorem}

\begin{proof}
मान लो $ABC$ एक त्रिभुज है। $A$ से होकर $(BC)$ के \omterm{def:g6:lines:perp}{समांतर} रेखा
खींचो। \omterm{def:g5:solids:def}{शीर्ष} $A$ पर तीन कोण इस \omterm{def:g6:lines:perp}{समांतर} रेखा के साथ क़तार में बैठते
हैं और एक ऋजु कोण, $180^\circ$, बनाते हैं: बीच वाला $\widehat{A}$
है, और बाहर के दोनों $\widehat B$ तथा $\widehat C$ के \omterm{def:g7:triangles:pairs}{एकांतर कोण}
हैं (रेखाएँ \omterm{def:g6:lines:perp}{समांतर} हैं!), इसलिए वे $\widehat B$ और $\widehat C$ के
बराबर हैं। यानी
$\widehat B + \widehat A + \widehat C = 180^\circ$।
\end{proof}

\begin{example}\label{ex:g7:triangles:sum}
किसी त्रिभुज के दो कोण $67^\circ$ और $58^\circ$ के हैं। तीसरा कोण
होगा
\[
180 - (67 + 58) = 180 - 125 = 55^\circ .
\]
याद रखने लायक़ ख़ास स्थितियाँ: \omterm{def:g6:shapes:triangles}{समबाहु त्रिभुज} का प्रत्येक कोण
$180 \div 3 = 60^\circ$ होता है; \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} के दोनों न्यून कोण
पूरक होते हैं (वे जुड़कर $180 - 90 = 90^\circ$ बनाते हैं); और
\omterm{def:g6:shapes:triangles}{समद्विबाहु त्रिभुज} के आधार वाले कोण बराबर होते हैं, इसलिए प्रत्येक
$\frac{180 - \text{शिखर कोण}}{2}$ होता है।
\end{example}

\begin{example}[दो समकोण वाला त्रिभुज होता ही नहीं]\label{ex:g7:triangles:tworight}
जिस त्रिभुज के दो कोण $90^\circ$ के हों वह अकेले उन्हीं में
$180^\circ$ ख़र्च कर देगा, और तीसरे के लिए $0^\circ$ बचेगा ---
असंभव। कोणों का \omterm{def:g1:addition:def}{योगफल} एक साथ बहुत सारे त्रिभुजों को मना कर देता है।
\end{example}

\section{त्रिभुज असमिका}

\begin{theorem}[त्रिभुज असमिका]\label{thm:g7:triangles:inequality}
प्रत्येक त्रिभुज में हर भुजा बाक़ी दो के \omterm{def:g1:addition:def}{योगफल} से छोटी होती है:
भुजाओं $a$, $b$, $c$ वाले त्रिभुज के लिए,
\[
a < b + c .
\]
उलटे तरफ़ से: तीन लंबाइयों में अगर \emph{सबसे बड़ी} बाक़ी दो के
\omterm{def:g1:addition:def}{योगफल} से छोटी हो, तो उनसे त्रिभुज हमेशा बनाया जा सकता है। और अगर
सबसे बड़ी बाक़ी दो के \omterm{def:g1:addition:def}{योगफल} के \emph{बराबर} हो, तो ``त्रिभुज'' चपटा
हो जाता है: तीनों \omterm{def:g5:solids:def}{शीर्ष} एक ही रेखा पर आ जाते हैं।
\end{theorem}
\admitted

\begin{example}\label{ex:g7:triangles:inequality}
क्या भुजाएँ $10$, $3$ और $2$ हो सकती हैं? सबसे बड़ी को जाँचो: क्या
$10 < 3 + 2 = 5$? नहीं --- ऐसा कोई त्रिभुज है ही नहीं। परकार इसका
कारण दिखा देता है: $10$ लंबे \omterm{def:g6:lines:objects}{रेखाखंड} के दोनों सिरों से खींचे गए
$3$ और $2$ \omterm{def:g3:shapes:shapes}{त्रिज्या} के चाप कभी मिलते ही नहीं।

भुजाएँ $7$, $5$ और $4$? सबसे बड़ी: $7 < 5 + 4 = 9$: हाँ, त्रिभुज
बनता है (सबसे बड़ी भुजा जाँच लेना काफ़ी है)।
\end{example}

\begin{method}[त्रिभुज का होना और उसकी रचना]\label{met:g7:triangles:construct}
तीन लंबाइयाँ दी हों तो:
\begin{enumerate}
  \item सबसे बड़ी की तुलना बाक़ी दो के \omterm{def:g1:addition:def}{योगफल} से करो; अगर वह
        सचमुच छोटी न हो तो रुक जाओ: त्रिभुज नहीं बनेगा;
  \item नहीं तो \cref{met:g6:shapes:construct} की तरह रचना करो
        (आधार \omterm{def:g6:lines:objects}{रेखाखंड}, परकार के दो चाप);
  \item कोणों से कोई रचना करने के बाद मन ही मन जाँच लो कि दिए हुए
        कोणों का योग $180^\circ$ से कम है --- और तीसरा कोण मिलकर
        ठीक $180^\circ$ बनाता है।
\end{enumerate}
\end{method}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g7:triangles:1}
इनके पूरक ($90^\circ$ तक) और संपूरक ($180^\circ$ तक) बताओ:
$30^\circ$; \ $45^\circ$; \ $72^\circ$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:2}
दो रेखाएँ कटती हैं; बने चार कोणों में से एक $35^\circ$ का है।
बाक़ी तीनों के माप बताओ, और हर एक का कारण भी।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:3}
उस त्रिभुज का तीसरा कोण निकालो जिसके पहले दो कोण इतने हैं:
(a) $40^\circ$ और $60^\circ$; \
(b) $90^\circ$ और $28^\circ$; \
(c) $75^\circ$ और $75^\circ$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:4}
एक \omterm{def:g6:shapes:triangles}{समद्विबाहु त्रिभुज} का शिखर कोण $40^\circ$ है। उसके दोनों आधार
कोण निकालो। एक दूसरे \omterm{def:g6:shapes:triangles}{समद्विबाहु त्रिभुज} का आधार कोण $70^\circ$ है:
उसका शिखर कोण निकालो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:5}
इनमें से कौन-कौन सी तिकड़ी किसी त्रिभुज की भुजाएँ हो सकती है?
त्रिभुज असमिका से कारण बताओ (सबसे बड़ी भुजा देखो!):
\[
(6,\ 8,\ 12); \qquad
(5,\ 5,\ 11); \qquad
(4,\ 9,\ 13); \qquad
(7,\ 7,\ 7).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:6}
किसी त्रिभुज के कोण $x$, $2x$ और $3x$ हैं। $x$ और तीनों कोण
निकालो। यह किस तरह का त्रिभुज है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:triangles:7}
$BC = 6$ cm, $\widehat{ABC} = 50^\circ$ और
$\widehat{ACB} = 60^\circ$ वाला त्रिभुज $ABC$ बनाओ ($B$ और $C$ पर
चाँदा रखकर)। बनाने से पहले $\widehat{BAC}$ का माप बताओ, फिर अपनी
आकृति पर जाँचो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:8}
दो \omterm{def:g6:lines:perp}{समांतर} रेखाओं को एक तीसरी रेखा काटती है और पहले कटान पर
(काटने वाली रेखा और एक \omterm{def:g6:lines:perp}{समांतर} रेखा के बीच) $54^\circ$ का कोण बनता
है। स्थिति का चित्र बनाओ और बने आठों कोणों के माप बताओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:9}
किसी त्रिभुज की दो भुजाएँ $8$ cm और $3$ cm की हैं। तीसरी भुजा किन
दो मानों के बीच होनी चाहिए? एक ऐसी पूर्ण-संख्या लंबाई बताओ जो चलती
है और एक जो नहीं चलती।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:triangles:10}
किसी त्रिभुज $ABC$ में $\widehat A = 2\widehat B$ और
$\widehat C = 90^\circ$ है। $\widehat A$ और $\widehat B$ निकालो।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:triangles:11}
किसी भी \omterm{def:g4:geometry:polygon}{चतुर्भुज} $ABCD$ को एक विकर्ण के साथ काटकर दो त्रिभुजों में
बाँटा जा सकता है, और तब उसके कोण पढ़े जा सकते हैं। इससे सिद्ध करो
कि प्रत्येक \omterm{def:g4:geometry:polygon}{चतुर्भुज} के चारों कोण जुड़कर $360^\circ$ बनाते हैं, और
फिर \omterm{def:g4:geometry:polygon}{पंचभुज} (पाँच भुजाएँ) के कोणों का \omterm{def:g1:addition:def}{योगफल} निकालो।
\end{exercise}

\section{समस्या: मुहल्ले का एक चक्कर}

\begin{problem}[{सप्ताहांत समस्या --- किसी भी बहुभुज के कोण:
भीतर $(n-2) \times 180^\circ$, बाहर हमेशा $360^\circ$ का घूमना,
और सिर्फ़ तीन सम आकार ही फ़र्श क्यों ढक पाते हैं}]
\label{pb:g7:triangles:1}
\Cref{exo:g7:triangles:11} ने एक \omterm{def:g4:geometry:polygon}{चतुर्भुज} को दो त्रिभुजों में
काटकर $360^\circ$ पाया था। यह समस्या उसी सोच को कितनी भी भुजाओं
वाले बहुभुजों तक ले जाती है, फिर एक दूसरा, बिलकुल अलग प्रमाण खोजती
है --- आकृति के चारों ओर \emph{चलकर} और घुमावों को जोड़कर --- और
आख़िर में तुम्हारे स्नानघर के फ़र्श पर पहुँचती है: सारे सम बहुभुजों
में से ठीक तीन ही समतल को बिना खाली जगह और बिना ऊपर चढ़े ढक सकते
हैं, और मधुमक्खियों ने अपना चुन लिया है।

\medskip
\noindent\textbf{भाग I --- भीतर के कोण।}
\begin{enumerate}
  \item किसी \omterm{def:g4:geometry:polygon}{षट्भुज} (छह भुजाएँ) के एक \omterm{def:g5:solids:def}{शीर्ष} से उस \omterm{def:g5:solids:def}{शीर्ष} को छोड़ते
        सारे विकर्ण खींचो। \omterm{def:g4:geometry:polygon}{षट्भुज} कितने त्रिभुजों में कटता है, और
        उसके \emph{सारे} भीतरी कोणों का \omterm{def:g1:addition:def}{योगफल} कितना है
        (\cref{thm:g7:triangles:sum})?
  \item आम बात कहो: $n$ भुजाओं वाले किसी \omterm{def:g4:geometry:polygon}{बहुभुज} के एक \omterm{def:g5:solids:def}{शीर्ष} से
        विकर्णों का पंखा कितने त्रिभुज बनाता है? इससे भीतरी कोणों
        के \omterm{def:g1:addition:def}{योगफल} का सूत्र निकालो, और उसे $n = 3$, $4$, $5$, $6$
        पर जाँचो।
  \item एक दशभुज ($10$ भुजाएँ) और एक द्वादशभुज ($12$ भुजाएँ) के
        भीतरी कोणों का \omterm{def:g1:addition:def}{योगफल} निकालो।
  \item \emph{सम} \omterm{def:g4:geometry:polygon}{बहुभुज} में सारे भीतरी कोण बराबर होते हैं।
        \omterm{def:g6:shapes:triangles}{समबाहु त्रिभुज}, वर्ग, सम \omterm{def:g4:geometry:polygon}{पंचभुज}, सम \omterm{def:g4:geometry:polygon}{षट्भुज} और सम \omterm{def:g4:geometry:polygon}{अष्टभुज}
        का भीतरी कोण निकालो।
  \item बिना किसी सूत्र के समझाओ कि \omterm{def:g4:geometry:polygon}{बहुभुज} में एक \omterm{def:g5:solids:def}{शीर्ष} बढ़ते ही
        कोणों का \omterm{def:g1:addition:def}{योगफल} ठीक $180^\circ$ क्यों बढ़ जाता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- मुहल्ले का चक्कर।}
किसी बहुभुजाकार मुहल्ले की सीमा पर हमेशा आगे की ओर चलो। हर कोने
पर तुम किसी कोण से \emph{मुड़ते} हो --- उस कोने का \emph{बहिष्कोण};
और मुड़ना तब तक चलता है जब तक तुम अपनी शुरुआती जगह पर, शुरुआती
दिशा में मुँह किए, वापस नहीं आ जाते।
\begin{enumerate}[resume]
  \item जिस कोने का भीतरी कोण $\widehat A$ है, वहाँ तुम कितना
        मुड़ते हो? (भीतरी और बाहरी कोण संपूरक होते हैं।)
        $40^\circ$, $60^\circ$, $80^\circ$ कोणों वाले त्रिभुज के
        तीनों घुमाव निकालो।
  \item सवाल 6 के तीनों घुमाव जोड़ो। पूरे चक्कर के बारे में कौन-सी
        बात दिखती है?
  \item समझाओ कि \emph{किसी भी} \omterm{def:g4:geometry:polygon}{बहुभुज} --- तीन भुजाओं वाला हो या
        तीस --- के पूरे चक्कर के घुमाव जुड़कर हमेशा ठीक
        $360^\circ$ क्यों बनते हैं। (वापस पहुँचने तक, सारे घुमाव
        मिलाकर, तुम्हारी नाक ने क्या किया?)
  \item भीतरी कोणों वाला सूत्र दूसरी बार निकालो: $n$ कोनों में से
        हर एक पर भीतरी कोण $+$ घुमाव $= 180^\circ$; सारे कोनों पर
        जोड़ो और सवाल 8 का इस्तेमाल करो।
  \item सम \omterm{def:g4:geometry:polygon}{बहुभुज} में सारे घुमाव बराबर होते हैं, इसलिए प्रत्येक
        बहिष्कोण $\frac{360^\circ}{n}$ होता है। इससे तुरंत जवाब
        दो: किस सम \omterm{def:g4:geometry:polygon}{बहुभुज} के भीतरी कोण $150^\circ$ के होते हैं?
        और \emph{किसी भी} सम \omterm{def:g4:geometry:polygon}{बहुभुज} के भीतरी कोण $155^\circ$ के
        क्यों नहीं हो सकते?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- फ़र्श ढकना।}
ढके हुए फ़र्श के प्रत्येक बिंदु के चारों ओर, वहाँ मिलने वाली टाइलों
के कोने जुड़कर ठीक $360^\circ$ बनाने चाहिए --- न कोई ख़ाली जगह, न
कोई चढ़ाव (\cref{pb:g6:lines:1})।
\begin{enumerate}[resume]
  \item \omterm{def:g6:shapes:triangles}{समबाहु त्रिभुज}, वर्ग और सम \omterm{def:g4:geometry:polygon}{षट्भुज} --- तीनों के लिए: ढकाई
        के एक कोने-बिंदु पर कितनी टाइलें मिलती हैं? हर बार
        $360^\circ$ जाँचो।
  \item दिखाओ कि सम \omterm{def:g4:geometry:polygon}{पंचभुज} फ़र्श नहीं ढक सकते: तीन कोने मिलकर
        कितना बनाते हैं, और चार कितना बनाते?
  \item दिखाओ कि छह से \emph{ज़्यादा} भुजाओं वाला कोई सम \omterm{def:g4:geometry:polygon}{बहुभुज}
        फ़र्श नहीं ढक सकता: हर कोने पर कम से कम तीन टाइलें मिलनी
        ही चाहिए (हर कोण $180^\circ$ से छोटा है), और $135^\circ$
        या उससे बड़े तीन कोनों का क्या होता है? सम ढकने वालों की
        पूरी सूची बताओ।
  \item स्नानघर के फ़र्श अक्सर सम अष्टभुजों के साथ छोटे वर्ग
        मिलाते हैं। जाँचो कि यह कोना चलता है:
        $135^\circ + 135^\circ + 90^\circ$। एक और जाना-पहचाना
        मेल हर कोने पर एक वर्ग, एक सम \omterm{def:g4:geometry:polygon}{षट्भुज} और एक सम द्वादशभुज
        रखता है: उसे भी जाँचो (द्वादशभुज का भीतरी कोण सवाल 10 के
        तरीक़े से निकल आता है)।
  \item मधुमक्खियों की पसंद: छत्ते के ख़ाने सम \omterm{def:g4:geometry:polygon}{षट्भुज} होते हैं।
        उनके कोने जाँचो ($3$ कोण), और एक वाक्य में --- रानी दीदो
        की खोज (\cref{pb:g6:measure:1}) की मदद से --- समझाओ कि
        तीनों संभव सम टाइलों में से \omterm{def:g4:geometry:polygon}{षट्भुज} ही उतने शहद के लिए
        सबसे कम मोम क्यों माँगता है।
\end{enumerate}
\end{problem}
