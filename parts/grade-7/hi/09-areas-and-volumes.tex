\chapter{क्षेत्रफल और आयतन}\label{ch:g7:areas}

पिछले साल हमने आयत, \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} और डिब्बे नापे थे
(\cref{ch:g6:measure})। अब एक कैंची से --- यहाँ से एक टुकड़ा काटो,
वहाँ चिपका दो --- वही सोच \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज}, \emph{सारे} त्रिभुजों
और चक्र का \omterm{def:g6:measure:area}{क्षेत्रफल} दे देती है, और \omterm{def:g7:areas:prism}{प्रिज़्म} तथा \omterm{def:g7:areas:prism}{बेलन} का \omterm{def:g6:measure:volume}{आयतन} भी।

\section{समांतर चतुर्भुज का क्षेत्रफल}

\begin{theorem}[समांतर चतुर्भुज]\label{thm:g7:areas:parallelogram}
जिस \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} की एक भुजा $b$ लंबी हो (उसका \emph{आधार}) और उस
भुजा तथा सामने वाली भुजा के बीच की दूरी $h$ हो (उसकी \emph{ऊँचाई}),
उसका \omterm{def:g6:measure:area}{क्षेत्रफल} है
\[
A = b \times h .
\]
\end{theorem}

\begin{proof}[कैंची से प्रमाण]
एक तरफ़ बाहर निकलता \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} काटकर दूसरी तरफ़ चिपका दो: \omterm{def:g7:central:parallelogram}{समांतर
चतुर्भुज} उतने ही \omterm{def:g6:measure:area}{क्षेत्रफल} वाला $b \times h$ का आयत बन जाता है।
\end{proof}

ऊँचाई दोनों आधारों के बीच की \emph{\omterm{def:g6:lines:perp}{लंब}} दूरी होती है, तिरछी भुजा की
लंबाई नहीं! बहुत तिरछे \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} की भुजाएँ लंबी और \omterm{def:g6:measure:area}{क्षेत्रफल}
छोटा हो सकता है।
\end{remark}

\section{त्रिभुज का क्षेत्रफल}

\begin{theorem}[त्रिभुज]\label{thm:g7:areas:triangle}
जिस त्रिभुज का आधार $b$ और उससे जुड़ी ऊँचाई $h$ हो (सामने वाले \omterm{def:g5:solids:def}{शीर्ष}
से आधार-रेखा तक की \omterm{def:g6:lines:perp}{लंब} दूरी), उसका \omterm{def:g6:measure:area}{क्षेत्रफल} है
\[
A = \frac{b \times h}{2} .
\]
\end{theorem}

\begin{proof}
त्रिभुज की दो नक़लें, जिनमें से एक किसी भुजा के मध्यबिंदु के चारों
ओर \omterm{def:g3:division:half}{आधा} चक्कर घुमा दी गई हो, मिलकर आधार $b$ और ऊँचाई $h$ वाला \omterm{def:g7:central:parallelogram}{समांतर
चतुर्भुज} बना देती हैं (यह \omterm{def:g7:central:def}{केंद्रीय सममिति} ही काम कर रही है,
\cref{prop:g7:central:preserved})। त्रिभुज उसका \omterm{def:g3:division:half}{आधा} है:
$\frac{bh}{2}$।
\end{proof}

\omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} बनाते हैं: त्रिभुज का \omterm{def:g6:measure:area}{क्षेत्रफल}
$\frac{b\times h}{2}$ है। तीनों भुजाओं में से कोई भी आधार बन सकती है
--- अपनी \emph{अपनी} ऊँचाई के साथ।}
\end{omfigure}

\begin{example}\label{ex:g7:areas:triangle}
किसी त्रिभुज का आधार $9$ cm है और उससे जुड़ी ऊँचाई $4$ cm नापती
है:
\[
A = \frac{9 \times 4}{2} = \frac{36}{2} = 18 \text{ cm}^2 .
\]
\emph{\omterm{def:g6:shapes:triangles}{समकोण}} त्रिभुज में \omterm{def:g6:shapes:triangles}{समकोण} बनाती दोनों भुजाएँ ही आधार और ऊँचाई
हैं: इससे \cref{prop:g6:measure:formulas} वाला सूत्र वापस मिल जाता है।
\end{example}

\section{चक्र का क्षेत्रफल}

\begin{theorem}[चक्र]\label{thm:g7:areas:disk}
$r$ \omterm{def:g3:shapes:shapes}{त्रिज्या} वाले चक्र का \omterm{def:g6:measure:area}{क्षेत्रफल} है
\[
A = \pi r^2 .
\]
\end{theorem}

\begin{proof}[प्रमाण का विचार]
चक्र को बहुत सारी पतली, बराबर फाँकों में काटो और उन्हें एक के बाद
एक सिर-से-पूँछ रख दो: वे लगभग $r$ ऊँचाई का एक \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} बना
देती हैं, जिसका आधार आधी परिधि यानी $\pi r$ है
(\cref{prop:g6:measure:circle})। उसका \omterm{def:g6:measure:area}{क्षेत्रफल}
$\pi r \times r = \pi r^2$ के पास होता है, और फाँकें जितनी पतली
होती जाएँ, यह उतना ही ठीक होता जाता है। पूरा प्रमाण समाकलन गणित
माँगता है, जो हाई-स्कूल खंड में है।
\end{proof}

{\small चक्र को फाँकों में काटकर उन्हें खोल देना: लगभग $\pi r$
आधार (आधी सीमा) और $r$ ऊँचाई वाला एक \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज}।}
\end{omfigure}

\begin{example}\label{ex:g7:areas:disk}
एक गोल मेज़ की सतह की \omterm{def:g3:shapes:shapes}{त्रिज्या} $60$ cm है। उसका \omterm{def:g6:measure:area}{क्षेत्रफल}
$\pi \times 60^2 = 3600\pi \approx 11\,310$ cm$^2$ है, यानी
$1.1$ m$^2$ से थोड़ा ज़्यादा। \omterm{ex:g1:subtraction:difference}{अंतर} पर ध्यान दो: \emph{\omterm{def:g6:measure:perimeter}{परिमाप}}
$2\pi r$ में $r$ आता है, और \emph{\omterm{def:g6:measure:area}{क्षेत्रफल}} $\pi r^2$ में $r^2$।
\end{example}

\section{प्रिज़्म और बेलन}

\begin{definition}[प्रिज़्म, बेलन]\label{def:g7:areas:prism}
\emph{प्रिज़्म}\index{प्रिज़्म} वह \omterm{def:g5:solids:def}{ठोस} है जिसके दो एक जैसे \omterm{def:g6:lines:perp}{समांतर}
बहुभुजाकार \omterm{def:g5:solids:def}{फलक} (\emph{आधार}) आयतों से जुड़े हों; और
\emph{बेलन}\index{बेलन} वही है, बस आधार चक्र होते हैं।
\emph{ऊँचाई} दोनों आधारों के बीच की दूरी है।
\end{definition}

\begin{theorem}[आयतन]\label{thm:g7:areas:volume}
आधार का \omterm{def:g6:measure:area}{क्षेत्रफल} $B$ और ऊँचाई $h$ वाले \omterm{def:g7:areas:prism}{प्रिज़्म} या \omterm{def:g7:areas:prism}{बेलन} के लिए:
\[
V = B \times h .
\]
\end{theorem}
\admitted

\begin{example}\label{ex:g7:areas:volume}
एक तंबू करवट लेटा हुआ \omterm{def:g7:areas:prism}{प्रिज़्म} है: उसके आधार ऐसे त्रिभुज हैं जिनका
आधार $2$ m और ऊँचाई $1.5$ m है, और तंबू $3$ m लंबा है।
\begin{enumerate}
  \item आधार का \omterm{def:g6:measure:area}{क्षेत्रफल}: $B = \frac{2 \times 1.5}{2} = 1.5$ m$^2$।
  \item \omterm{def:g6:measure:volume}{आयतन}: $V = B \times h = 1.5 \times 3 = 4.5$ m$^3$।
\end{enumerate}
$4$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या} और $11$ cm ऊँचाई वाला एक डिब्बा:
$V = \pi \times 4^2 \times 11 = 176\pi \approx 553$ cm$^3$, यानी
मोटे तौर पर \omterm{def:g3:division:half}{आधा} लीटर।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g7:areas:1}
$8$ cm आधार और $5$ cm ऊँचाई वाले \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} का \omterm{def:g6:measure:area}{क्षेत्रफल}
निकालो; और $6.5$ m आधार तथा $4$ m ऊँचाई वाले दूसरे का भी।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:areas:2}
किसी \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} की भुजाएँ $10$ cm और $6$ cm हैं, और $10$ cm
वाले आधार से जुड़ी ऊँचाई $4$ cm नापती है। उसका \omterm{def:g6:measure:area}{क्षेत्रफल} निकालो।
उत्तर $60$ cm$^2$ क्यों नहीं है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:areas:3}
$12$ cm आधार और $7$ cm ऊँचाई वाले त्रिभुज का \omterm{def:g6:measure:area}{क्षेत्रफल} निकालो; $9$
और $6$ की समकोण-भुजाओं वाले \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} का; और $5.5$ m आधार तथा
$2$ m ऊँचाई वाले त्रिभुज का।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:areas:4}
$5$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या} वाले चक्र का \omterm{def:g6:measure:area}{क्षेत्रफल} निकालो, फिर $10$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या}
वाले आधे चक्र का ($\pi$ के साथ ठीक मान, फिर इकाई तक गोल)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:areas:5}
$3$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या} वाले चक्र का \omterm{def:g6:measure:perimeter}{परिमाप} \emph{और} \omterm{def:g6:measure:area}{क्षेत्रफल} निकालो।
दोनों उत्तरों में से कौन cm में है और कौन cm$^2$ में?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:areas:6}
$24$ cm$^2$ आधार-क्षेत्रफल और $10$ cm ऊँचाई वाले \omterm{def:g7:areas:prism}{प्रिज़्म} का \omterm{def:g6:measure:volume}{आयतन}
निकालो; और $3$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या} तथा $8$ cm ऊँचाई वाले \omterm{def:g7:areas:prism}{बेलन} का (ठीक मान,
फिर गोल किया हुआ)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:areas:7}
किसी त्रिभुज का \omterm{def:g6:measure:area}{क्षेत्रफल} $28$ cm$^2$ और आधार $8$ cm है। उससे जुड़ी
ऊँचाई निकालो, पहले समीकरण लिखकर।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:areas:8}
कोई भी त्रिभुज बनाओ और आधार-ऊँचाई के तीनों जोड़े ध्यान से नापो।
प्रत्येक जोड़े के लिए $\frac{b \times h}{2}$ निकालो: तीनों उत्तर
मिलने चाहिए (नापने की ग़लती छोड़कर)। ऐसा क्यों?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:areas:9}
समलंब आकार के एक बग़ीचे को एक विकर्ण से दो त्रिभुजों में बाँटा जा
सकता है, जिनकी ऊँचाई एक ही, $h = 20$ m, है और आधार $30$ m तथा
$18$ m हैं। उसका कुल \omterm{def:g6:measure:area}{क्षेत्रफल} निकालो। क्या तुम समलंब के लिए कोई आम
सूत्र अटकल सकते हो?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:areas:10}
$6$ cm \omterm{def:g3:shapes:shapes}{त्रिज्या} वाले एक बेलनाकार फूलदान में $15$ cm ऊँचाई तक पानी
है। सारा पानी एक ख़ाली डिब्बे में उँडेल दिया जाता है जिसका आधार
$18$ cm $\times$ $12$ cm है। डिब्बे में पानी की ऊँचाई कितनी बनेगी?
(\omterm{def:g6:measure:volume}{आयतन} वही, आधार अलग।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:areas:11}
$30$ cm \omterm{def:g4:geometry:circle}{व्यास} वाला एक पिज़्ज़ा $9$ यूरो का है; $40$ cm \omterm{def:g4:geometry:circle}{व्यास} वाला
$14$ यूरो का। दोनों के \omterm{def:g6:measure:area}{क्षेत्रफल} और $100$ cm$^2$ का दाम निकालो।
कौन-सा पिज़्ज़ा ज़्यादा फ़ायदे का सौदा है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:areas:12}
किसी \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज} की \omterm{def:g6:shapes:triangles}{समकोण} बनाती दोनों भुजाएँ $6$ cm और $8$ cm की
हैं, और उसका कर्ण $10$ cm का। पहले उन दोनों भुजाओं से उसका
\omterm{def:g6:measure:area}{क्षेत्रफल} निकालो, फिर उसी \omterm{def:g6:measure:area}{क्षेत्रफल} से \emph{कर्ण} से जुड़ी ऊँचाई
निकालो।
\end{exercise}

\section{समस्या: ग़ायब हो जाने वाला वर्ग}

\begin{problem}[{सप्ताहांत समस्या --- वह मशहूर ``प्रमाण'' कि
$64 = 65$, जिसका भेद क्षेत्रफल और ढलान खोल देते हैं, और मिठाई में
पिज़्ज़े तथा छल्ले}]
\label{pb:g7:areas:1}
एक जादूगर $8 \times 8$ के वर्ग को चार टुकड़ों में काटता है, उन्हें
इधर-उधर सरकाता है, और उन्हीं से $5 \times 13$ का आयत जोड़ देता है।
दर्शक गिनते हैं: $8 \times 8 = 64$, पर $5 \times 13 = 65$ --- एक
वर्ग इकाई हवा में से पैदा हो गई! इसी अध्याय के क्षेत्रफल-सूत्र ठीक
वे औज़ार हैं जो इस चाल को पकड़ लेते हैं। फिर यह समस्या ईमानदार
क्षेत्रफल-तर्क को काम पर लगाती है: समलंबों पर, तिरछी ढेरियों पर,
पिज़्ज़े के हिसाब-किताब पर --- और आख़िर में जादूगर के पास लौटती है,
एक और भी अजीब खेल के लिए।

\medskip
\noindent\textbf{भाग I --- चाल, दिखाई भी और खोली भी।}
$8 \times 8$ वाले वर्ग के चारों टुकड़े ये हैं: $8$ और $3$ की
समकोण-भुजाओं वाले दो \omterm{def:g6:shapes:triangles}{समकोण त्रिभुज}, और $5$ तथा $3$ की \omterm{def:g6:lines:perp}{समांतर}
भुजाओं और $5$ ऊँचाई वाले दो \omterm{def:g6:shapes:triangles}{समकोण} समलंब।
\begin{enumerate}
  \item असली वर्ग का \omterm{def:g6:measure:area}{क्षेत्रफल} और बताए गए आयत का \omterm{def:g6:measure:area}{क्षेत्रफल}
        निकालो। एक पंक्ति में गड़बड़ी बताओ।
  \item चारों टुकड़ों में से प्रत्येक का \omterm{def:g6:measure:area}{क्षेत्रफल} निकालो
        (\cref{thm:g7:areas:triangle}; समलंबों के लिए उन्हें काट
        लो या \cref{exo:g7:areas:9} इस्तेमाल करो)।
  \item चारों \omterm{def:g6:measure:area}{क्षेत्रफल} जोड़ो। दोनों आकृतियों में से --- वर्ग या
        आयत --- टुकड़े सचमुच कौन-सी भर सकते हैं?
  \item आयत वाली सजावट में एक त्रिभुज का कर्ण ($8$ चलकर $3$
        चढ़ता हुआ) किसी समलंब की तिरछी किनार ($5$ चलकर $2$ चढ़ती
        हुई) को एक ही सीधे ``विकर्ण'' में आगे बढ़ाता हुआ माना
        जाता है। इस जोड़ को जाँचो: क्या $8$ में $3$ और $5$ में
        $2$ एक ही ढलान हैं (\cref{thm:g7:prop:cross})?
  \item बताओ पैंसठवाँ वर्ग कहाँ छिपा है: उस झूठे विकर्ण के साथ
        बनी ख़ाली जगह की शक्ल क्या है, उसका ठीक-ठीक \omterm{def:g6:measure:area}{क्षेत्रफल}
        क्या है, और आँख उसे क्यों नहीं पकड़ पाती? (उतने \omterm{def:g6:measure:area}{क्षेत्रफल}
        की एक झिरी आयत के क़रीब $13$ इकाई लंबे विकर्ण पर फैलाई
        जाए तो औसतन कितनी चौड़ी होगी?)
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- ईमानदार क्षेत्रफल-तर्क।}
\begin{enumerate}[resume]
  \item $4$ cm की दूरी पर दो \omterm{def:g6:lines:perp}{समांतर रेखाएँ} खींचो, उनमें से एक पर
        $6$ cm का आधार लो, और उसी आधार पर तीन अलग-अलग त्रिभुज
        बनाओ जिनके शिखर दूसरी रेखा के अलग-अलग बिंदुओं पर हों।
        तीनों \omterm{def:g6:measure:area}{क्षेत्रफल} निकालो। शिखर अपनी रेखा पर चाहे जहाँ सरके,
        तीनों बराबर क्यों रहते हैं?
  \item समलंब का सूत्र आम तौर पर सिद्ध करो: $B$ और $b$ \omterm{def:g6:lines:perp}{समांतर}
        भुजाओं तथा $h$ ऊँचाई वाला समलंब, एक विकर्ण के साथ काटने
        पर, एक ही ऊँचाई $h$ वाले दो त्रिभुज देता है। नतीजा
        निकालो:
        \[
        A = \frac{(B + b) \times h}{2} .
        \]
  \item इसे $B = 9$ cm, $b = 5$ cm, $h = 4$ cm वाले समलंब पर
        लगाओ।
  \item बढ़ई यही सूत्र भेस बदलकर इस्तेमाल करते हैं: \omterm{def:g6:measure:area}{क्षेत्रफल} $=$
        (दोनों \omterm{def:g6:lines:perp}{समांतर} भुजाओं का औसत) $\times$ ऊँचाई। समझाओ कि यह
        वही नियम क्यों है, और सवाल 8 को इसी तरह दोबारा निकालो।
  \item ताश के पत्तों की सीधी ढेरी को तिरछा कर दो। समझाओ कि \omterm{def:g6:measure:volume}{आयतन}
        क्यों नहीं बदला, और इससे \omterm{def:g7:areas:prism}{प्रिज़्म} के सूत्र
        $V = B \times h$ (\cref{thm:g7:areas:volume}) की ऊँचाई
        के बारे में क्या पता चलता है: तिरछी ढेरी के लिए कौन-सी
        ऊँचाई लेनी होगी --- तिरछी लंबाई या खड़ी वाली? (एक विमा
        ऊपर, \cref{exo:g7:areas:2} से तुलना करो।)
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- छल्ले, पिज़्ज़े, और जादूगर की
वापसी।}
\begin{enumerate}[resume]
  \item $3$ m \omterm{def:g3:shapes:shapes}{त्रिज्या} वाला एक गोल तालाब $5$ m \omterm{def:g3:shapes:shapes}{त्रिज्या} वाले एक
        गोल लॉन के बीचोंबीच है। घास के छल्ले का \omterm{def:g6:measure:area}{क्षेत्रफल} निकालो
        ($\pi$ के साथ ठीक मान, फिर m$^2$ तक गोल)।
  \item उस अकेले चक्र की \omterm{def:g3:shapes:shapes}{त्रिज्या} निकालो जिसका \omterm{def:g6:measure:area}{क्षेत्रफल} ठीक उसी
        छल्ले के \omterm{def:g6:measure:area}{क्षेत्रफल} के बराबर हो। (अब तुम्हारे पास जो तीन
        \omterm{def:g3:shapes:shapes}{त्रिज्याएँ} हैं वे एक मशहूर तिकड़ी बनाती हैं, जिसकी कहानी
        \cref{ch:g8:pythagoras} में है।)
  \item ज़्यादा पिज़्ज़ा कौन-सा है: $40$ cm \omterm{def:g4:geometry:circle}{व्यास} वाला एक
        पिज़्ज़ा, या $28$ cm \omterm{def:g4:geometry:circle}{व्यास} वाले दो? निकालो और तुलो।
  \item किसी पिज़्ज़े का \omterm{def:g6:measure:area}{क्षेत्रफल} \emph{दुगुना} करने के लिए उसकी
        \omterm{def:g3:shapes:shapes}{त्रिज्या} मोटे तौर पर कितने गुना बढ़नी चाहिए? उम्मीदवार
        $1.4$ को जाँचो (याद रखो कि \omterm{def:g6:measure:area}{क्षेत्रफल} मापनी-गुणक के
        \emph{वर्ग} के साथ बढ़ता है, \cref{pb:g6:propdata:1}),
        और बताओ कि ठीक-ठीक वह गुणक --- वह संख्या जिसका वर्ग $2$
        है --- कहाँ सरकारी तौर पर आती है
        (\cref{ex:g8:pythagoras:diagonal})।
  \item जादूगर लौटता है: अब वह $13 \times 13$ के वर्ग को काटता
        है, जिसमें $5$, $8$, $13$ वही भूमिकाएँ निभाते हैं जो
        पहले $3$, $5$, $8$ निभा रहे थे, और उनसे $8 \times 21$ का
        आयत जोड़ देता है। दोनों \omterm{def:g6:measure:area}{क्षेत्रफल} निकालो: इस बार क्या
        ग़ायब हुआ? संख्याएँ $3, 5, 8, 13, 21$
        \cref{pb:g7:fractions:1} वाली श्रेणी के लगातार पद हैं ---
        जादूगर के नफ़े-नुक़सान का ढर्रा बताओ, और सीख भी: \omterm{def:g6:measure:area}{क्षेत्रफल}
        कभी झूठ क्यों नहीं बोलते?
\end{enumerate}
\end{problem}
