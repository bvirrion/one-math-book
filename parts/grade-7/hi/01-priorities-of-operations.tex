\chapter{संक्रियाओं की प्राथमिकता}\label{ch:g7:priorities}

$3 + 4 \times 5$ कितना है? बाएँ से दाएँ निकालो तो $35$ मिलता है;
पहले गुणा करो तो $23$। गणित को एक ही उत्तर चाहिए, इसलिए गणनाओं
के अपने यातायात-नियम हैं --- प्राथमिकताएँ। यह अध्याय उन्हें बताता
है और बीजगणित की पहली सर्वसमिका, वितरण नियम, सामने लाता है।

\section{प्राथमिकता के नियम}

\begin{theorem}[प्राथमिकता के नियम]\label{thm:g7:priorities:rules}
कोष्ठक न हों तो किसी गणना में:
\begin{enumerate}
  \item गुणा और भाग, जोड़ और घटाव से \emph{पहले} किए जाते हैं;
  \item एक ही स्तर की संक्रियाएँ बाएँ से दाएँ की जाती हैं।
\end{enumerate}
कोष्ठक सब पर भारी हैं: कोष्ठकों के भीतर वाला पहले निकलता है, और
सबसे भीतर वाले से शुरू करके।
\end{theorem}

\begin{example}\label{ex:g7:priorities:basic}
हर पंक्ति में एक कदम, और (मन-ही-मन) जो निकाला जा रहा है उसे रेखांकित करते हुए:
\begin{align*}
3 + 4 \times 5 &= 3 + 20 = 23
&&\text{(पहले गुणा)}\\
20 - 8 + 2 &= 12 + 2 = 14
&&\text{(एक ही स्तर: बाएँ से दाएँ)}\\
18 \div 3 \times 2 &= 6 \times 2 = 12
&&\text{(एक ही स्तर: बाएँ से दाएँ)}\\
(3 + 4) \times 5 &= 7 \times 5 = 35
&&\text{(पहले कोष्ठक)।}
\end{align*}
जालों पर ध्यान दो: $20 - 8 + 2$, $20 - 10$ \emph{नहीं} है, और
$18 \div 3 \times 2$, $18 \div 6$ \emph{नहीं} है।
\end{example}

\begin{example}[कोष्ठक के भीतर कोष्ठक]\label{ex:g7:priorities:nested}
\begin{align*}
50 - 2 \times \bigl(3 + (10 - 4)\bigr)
&= 50 - 2 \times (3 + 6) && \text{(सबसे भीतरी कोष्ठक)}\\
&= 50 - 2 \times 9 && \text{(बचा हुआ कोष्ठक)}\\
&= 50 - 18 && \text{(गुणा)}\\
&= 32 .
\end{align*}
\end{example}

\begin{remark}[भिन्न की रेखा में कोष्ठक छिपे होते हैं]
\omterm{def:g6:fractions:def}{भिन्न} की रेखा अपने \omterm{def:g4:fractions:def}{अंश} और हर को ऐसे बाँध देती है मानो हर एक के
इर्द-गिर्द अनदेखे कोष्ठक हों:
\[
\frac{12 + 8}{4} = \frac{20}{4} = 5,
\qquad\text{न कि}\quad 12 + \frac{8}{4} = 14 .
\]
\end{remark}

\begin{method}[सुरक्षित गणना]\label{met:g7:priorities:safe}
\begin{enumerate}
  \item पहले होने वाली संक्रिया पहचानो (सबसे भीतरी कोष्ठक, फिर
        $\times$ और $\div$, फिर $+$ और $-$);
  \item \emph{केवल वही} निकालो, और नतीजे को जगह पर रखकर पूरा
        व्यंजक दोबारा लिखो;
  \item तब तक दोहराओ जब तक एक ही संख्या बचे --- हर पंक्ति में एक
        संक्रिया, कोई छोटा रास्ता नहीं।
\end{enumerate}
\end{method}

\section{वितरण नियम}

\begin{theorem}[वितरण नियम]\label{thm:g7:priorities:distributivity}
सारी संख्याओं $k$, $a$, $b$ के लिए:
\[
k \times (a + b) = k \times a + k \times b,
\qquad
k \times (a - b) = k \times a - k \times b .
\]
\end{theorem}

\begin{proof}[चित्र से उपपत्ति]
चौड़ाई $k$ के उस आयत का \omterm{def:g6:measure:area}{क्षेत्रफल} गिनो जो लंबाइयों $a$ और $b$ के
दो छोटे आयतों में बँटा है: पूरा लेने पर वह $k \times (a+b)$ है;
दो हिस्सों में $k \times a + k \times b$। वही आयत, वही
\omterm{def:g6:measure:area}{क्षेत्रफल}।
\end{proof}

\begin{omfigure}
\begin{tikzpicture}[scale=0.75]
  \fill[omDef!14] (0,0) rectangle (4.2,1.8);
  \fill[omThm!14] (4.2,0) rectangle (6.4,1.8);
  \draw[thick] (0,0) rectangle (6.4,1.8);
  \draw[thick] (4.2,0) -- (4.2,1.8);
  \node[below, font=\small] at (2.1,0) {$a$};
  \node[below, font=\small] at (5.3,0) {$b$};
  \node[left, font=\small] at (0,0.9) {$k$};
  \node[font=\small] at (2.1,0.9) {$k \times a$};
  \node[font=\small] at (5.3,0.9) {$k \times b$};
\end{tikzpicture}

{\small क्षेत्रफलों के रूप में वितरण नियम: बड़ा आयत
$k \times (a + b)$ दोनों छोटे आयतों का \omterm{def:g1:addition:def}{योगफल} है।}
\end{omfigure}

\begin{example}[दोनों दिशाएँ]\label{ex:g7:priorities:distributivity}
\emph{फैलाना} (बाएँ से दाएँ) मानसिक गणित आसान कर देता है:
\[
7 \times 103 = 7 \times (100 + 3) = 700 + 21 = 721,
\qquad
6 \times 98 = 6 \times (100 - 2) = 600 - 12 = 588 .
\]
\emph{गुणनखंडन} (दाएँ से बाएँ) साझा गुणनखंड बाहर निकाल लेता है:
\[
23 \times 12 + 23 \times 8 = 23 \times (12 + 8) = 23 \times 20 = 460 .
\]
\end{example}

\begin{example}[शब्दों को गणना में बदलना]\label{ex:g7:priorities:words}
``$8$ और $5$ तथा $3$ के \omterm{def:g2:mult:def}{गुणनफल} का \omterm{def:g1:addition:def}{योगफल}'' है $8 + 5 \times 3 =
23$। ``$5$ का, $8$ और $3$ के \omterm{def:g1:addition:def}{योगफल} से \omterm{def:g2:mult:def}{गुणनफल}'' है
$5 \times (8 + 3) = 55$: ``\dots\ के \omterm{def:g1:addition:def}{योगफल} से \omterm{def:g2:mult:def}{गुणनफल}'' शब्द
\omterm{def:g1:addition:def}{योगफल} के इर्द-गिर्द कोष्ठक लगा देते हैं।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g7:priorities:1}
हर पंक्ति में एक संक्रिया लेकर निकालो:
\[
5 + 3 \times 6, \qquad
(5 + 3) \times 6, \qquad
30 - 12 \div 4, \qquad
(30 - 12) \div 4 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:2}
निकालो:
\[
24 - 6 + 2, \qquad
24 \div 6 \times 2, \qquad
24 - 6 \times 2, \qquad
24 \div (6 \times 2).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:3}
कदम-दर-कदम निकालो:
\[
7 + 3 \times (10 - 6), \qquad
40 - (2 + 3) \times 4, \qquad
60 \div \bigl(2 \times (7 - 4)\bigr).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:4}
भिन्नें निकालो (अनदेखे कोष्ठक!):
\[
\frac{18 + 6}{8}, \qquad
\frac{30}{7 - 2}, \qquad
\frac{5 \times 8}{4 + 6} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:5}
वितरण नियम से मन-ही-मन निकालो:
\[
8 \times 102, \qquad
25 \times 99, \qquad
14 \times 12 + 14 \times 8, \qquad
37 \times 45 - 37 \times 35 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:6}
गणना एक ही पंक्ति में लिखो, जहाँ ज़रूरी हो वहाँ कोष्ठक लगाकर, फिर
निकालो: ``$50$ का और $6$ तथा $7$ के \omterm{def:g2:mult:def}{गुणनफल} का \omterm{ex:g1:subtraction:difference}{अंतर}'';
``$4$ और $5$ के \omterm{def:g1:addition:def}{योगफल} का, $10$ और $8$ के \omterm{ex:g1:subtraction:difference}{अंतर} से
\omterm{def:g2:mult:def}{गुणनफल}''।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:priorities:7}
अमीर ने $2.50$-$2.50$ की $3$ कॉपियाँ और $1.20$-$1.20$ की $2$
क़लमें ख़रीदीं। उसका कुल एक ही व्यंजक में लिखो, फिर निकालो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:priorities:8}
$4 + 2 \times 5 - 1$ में कोष्ठक ऐसे लगाओ कि नतीजा हो: (a) $13$;
(b) $29$; (c) $24$। (इनमें से एक को कोई कोष्ठक चाहिए ही नहीं ---
कौन-सा?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:priorities:9}
क्या $12 + 4 \times k$ और $16 \times k$ हर संख्या $k$ के लिए एक
ही हैं? $k = 1$ और $k = 2$ पर आज़माओ, और निष्कर्ष निकालो।
प्राथमिकता की कौन-सी ग़लती एक से दूसरे तक ले जाती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:priorities:10}
सिनेमा का एक टिकट $9$ यूरो का है, पर $n$ लोगों का समूह एकमुश्त
$12$ यूरो और प्रति व्यक्ति $6$ यूरो देता है।
\begin{enumerate}
  \item समूह के दाम का व्यंजक लिखो, और $n = 5$ के लिए निकालो।
  \item कितने लोगों से समूह वाली दर $n$ अलग-अलग टिकटों से सस्ती
        पड़ने लगती है? ($n$ के कुछ मान आज़माओ।)
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:priorities:11}
अंकों $1$, $2$, $3$, $4$ में से हर एक को ठीक एक बार, चारों
संक्रियाएँ और मनचाहे कोष्ठक लेकर, $24$, $10$ और $1$ के बराबर
व्यंजक लिखो।
\end{exercise}

\section{समस्या: मिस्री लिपिक की तरह गुणा करना}

\begin{problem}[{सप्ताहांत समस्या --- वितरण नियम से निकली मानसिक
गणित की तरकीबें, और चार हज़ार साल से चली आ रही दुगुना-आधा वाली
गुणा-विधि}]
\label{pb:g7:priorities:1}
कैलकुलेटरों से बहुत पहले --- और विद्यालय में पहाड़े रटाए जाने से
भी पहले --- मिस्र के लिपिक और रूसी किसान किन्हीं दो संख्याओं का
गुणा केवल \emph{दुगुना करने, \omterm{def:g3:division:half}{आधा} करने और जोड़ने} से कर लेते थे।
उनका तरीक़ा जादू जैसा लगता है; असल में वह पूरी ताक़त से काम करता
वितरण नियम (\cref{thm:g7:priorities:distributivity}) है। यह
समस्या पहले तुम्हारा मानसिक गणित तेज़ करती है, फिर वह पुरानी
विधि सिखाती है, और फिर बोनट खोलकर दिखाती है कि वह चलती क्यों है।

\medskip
\noindent\textbf{भाग I --- न निकालने की कला।}
\begin{enumerate}
  \item \cref{ex:g7:priorities:distributivity} की तरह फैलाकर
        मन-ही-मन निकालो:
        $7 \times 102$ और $9 \times 98$।
  \item गुणनखंडन से मन-ही-मन निकालो:
        $17 \times 13 + 17 \times 7$ और
        $45 \times 101 - 45$।
  \item ``$5$ से गुणा'' वाली तरकीब: $5$ से गुणा करने के लिए $10$
        से गुणा करो और \omterm{def:g3:division:half}{आधा} कर दो। $5 \times 87$ इसी तरह निकालो,
        और समझाओ कि यह तरकीब जायज़ क्यों है।
  \item दो और तरकीबें, जिन्हें पक्का करके इस्तेमाल करना है:
        $25$ से गुणा करने के लिए $100$ से गुणा करके $4$ का भाग
        दो ($25 \times 32$ निकालो); $99$ से गुणा करने के लिए
        $100$ से गुणा करके एक बार घटा दो ($99 \times 47$ निकालो)।
  \item एक चेतावनी: वितरण नियम किसी गुणनखंड को \emph{\omterm{def:g1:addition:def}{योगफल}} पर
        फैलाता है, \omterm{def:g2:mult:def}{गुणनफल} पर कभी नहीं।
        $8 \times (5 + 3)$ की $8 \times 5 + 8 \times 3$ से तुलना
        करो, और फिर $8 \times (5 \times 3)$ की
        $(8 \times 5) \times (8 \times 3)$ से: कौन-सी जोड़ी मिलती
        है, कौन-सी नहीं?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- दुगुना-आधा वाली विधि।}
$13 \times 24$ के लिए विधि यह है। दो स्तंभ लिखो। बायाँ स्तंभ:
$13$ से शुरू करो और \emph{\omterm{def:g3:division:half}{आधा}} करते जाओ, \omterm{def:g3:division:remainder}{शेषफल} छोड़ते हुए,
$1$ तक: \ $13,\ 6,\ 3,\ 1$। दायाँ स्तंभ: $24$ से शुरू करो और
उतनी ही बार \emph{दुगुना} करो: \ $24,\ 48,\ 96,\ 192$। जिस
पंक्ति की \emph{बाईं} संख्या सम हो उसे काट दो, और बची हुई दाईं
संख्याएँ जोड़ लो।
\begin{enumerate}[resume]
  \item $13 \times 24$ के लिए विधि चलाओ: कौन-सी पंक्ति कटती है,
        क्या \omterm{def:g1:addition:def}{योगफल} बचता है, और क्या वह $13 \times 24$ के बराबर
        है?
  \item इसी विधि से $21 \times 17$ निकालो ($21$ को \omterm{def:g3:division:half}{आधा} करो,
        $17$ को दुगुना), और उत्तर आम गुणा से जाँचो।
  \item इसी विधि से $16 \times 25$ निकालो। कितनी पंक्तियाँ बचती
        हैं, और $1$, $2$, $4$, $8$, $16, \dots$ से शुरू होने वाले
        बाएँ स्तंभ में ख़ास क्या है?
  \item अब \emph{क्यों} चलती है, एक-एक कदम। हर बराबरी को वितरण
        नियम से पक्का करो:
        \[
        13 \times 24 = (12 + 1) \times 24
        = 12 \times 24 + 24 = 6 \times 48 + 24 .
        \]
        सार: बाएँ को \omterm{def:g3:division:half}{आधा} और दाएँ को दुगुना करने से \omterm{def:g2:mult:def}{गुणनफल} वही
        रहता है --- पर बाईं संख्या \emph{विषम} हो, तो दाईं संख्या
        की एक प्रति अलग हो जाती है और उसे जमा कर लेना पड़ता है।
        सारणी की पहली पंक्ति कौन-सी संख्या जमा करती है?
  \item सारणी में नीचे तक हिसाब चलाते जाओ:
        $6 \times 48 = 3 \times 96$, फिर
        $3 \times 96 = 1 \times 192 + 96$, फिर
        $1 \times 192 = 192$। रास्ते में जमा हुआ सब इकट्ठा करो,
        और समझाओ कि जमा हुई संख्याएँ ठीक उन्हीं पंक्तियों की दाईं
        संख्याएँ क्यों हैं जिनकी बाईं संख्या विषम है --- यानी वे
        पंक्तियाँ जिन्हें विधि रखती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- बोनट के नीचे: दुगुनों के
\omterm{def:g1:addition:def}{योगफल}।}
\begin{enumerate}[resume]
  \item प्रश्न 6 में बची हुई पंक्तियाँ $24$ के $1$, $4$ और $8$
        दुगुनों के बराबर थीं: विधि ने चुपचाप
        $13 \times 24 = (1 + 4 + 8) \times 24$ लिख दिया। जाँचो कि
        $1 + 4 + 8 = 13$, और पता करो कि प्रश्न 7 में $21$ के लिए
        विधि ने कौन-से दुगुने ($1$, $2$, $4$, $8$, $16$) रखे।
  \item $25$ और $42$ को दुगुनों की सूची $1, 2, 4, 8, 16, 32$ की
        संख्याओं के \omterm{def:g1:addition:def}{योगफल} के रूप में लिखो, हर एक को ज़्यादा से
        ज़्यादा एक बार लेकर।
  \item समझाओ कि \emph{हर} पूर्ण संख्या ऐसे क्यों लिखी जा सकती
        है: जो सबसे बड़ी दुगुनी संख्या समा जाए वह लो, घटाओ,
        दोहराओ --- यह रुकना क्यों पड़ेगा? $100$ को ऐसे \omterm{def:g1:addition:def}{योगफल} के
        रूप में लिखो।
  \item मिस्र के लिपिक चार हज़ार साल पहले ठीक यही करते थे:
        $19 \times 35$ निकालने के लिए वे $35$ की दुगुनी सारणी
        बनाते ($1 \to 35$, $2 \to 70$, $4 \to 140$,
        $8 \to 280$, $16 \to 560$), $19$ बनाने वाली पंक्तियाँ
        चुनते, और जोड़ लेते। उनके तरीक़े से गणना करो।
  \item आख़िरी सवाल: आधुनिक संगणक, जो $0$ और $1$ से गिनते हैं,
        असल में लिपिकों जैसे ही गुणा करते हैं --- दुगुनों और
        जोड़ों से। $13 \times 24$ के लिए संगणक कौन-से तीन जोड़
        करेगा? (हर संख्या दुगुनों में क्यों बँट जाती है, और
        इसका $0$ तथा $1$ से क्या नाता है, यह घातों वाले अध्याय
        की सप्ताहांत समस्या है: \cref{pb:g8:powers:1}।)
\end{enumerate}
\end{problem}
