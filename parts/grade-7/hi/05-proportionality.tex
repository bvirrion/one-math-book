\chapter{समानुपात}\label{ch:g7:prop}

पिछले साल \omterm{def:g6:propdata:def}{समानुपाती} सारणियाँ मिली थीं (\cref{ch:g6:propdata}); यह
अध्याय \omterm{def:g7:prop:table}{समानुपात} को एक पूरा औज़ार बना देता है: चौथा मान निकालने का
तिरछा नियम, प्रतिशत, और नक़्शों तथा नमूनों की \omterm{def:g7:prop:scale}{मापनी}। सबसे मशहूर
\omterm{def:g7:prop:table}{समानुपात} --- चाल --- \cref{ch:g8:speed} में पढ़ा जाएगा।

\section{सारणियाँ पहचानना और पूरी करना}

\begin{definition}[समानुपात सारणी]\label{def:g7:prop:table}
दो पंक्तियों वाली कोई सारणी \emph{समानुपात सारणी}\index{समानुपात}
तब होती है जब दूसरी पंक्ति की संख्याएँ पहली पंक्ति की संख्याओं को
एक ही तयशुदा संख्या --- \emph{गुणांक} --- से गुणा करने पर मिलें।
यही बात दूसरे शब्दों में: सारे स्तंभों के \omterm{def:g3:division:remainder}{भागफल}
$\frac{\text{नीचे}}{\text{ऊपर}}$ बराबर हों।
\end{definition}

\begin{example}\label{ex:g7:prop:recognize}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|ccc}
ऊपर & $4$ & $10$ & $14$ \\
\hline
नीचे & $6$ & $15$ & $21$ \\
\end{tabular}
\end{center}
\omterm{def:g3:division:remainder}{भागफल}: $\frac64 = 1.5$, $\frac{15}{10} = 1.5$,
$\frac{21}{14} = 1.5$: \omterm{def:g6:propdata:def}{समानुपाती}, गुणांक $1.5$।
\end{example}

\begin{theorem}[तिरछा नियम]\label{thm:g7:prop:cross}
जिस \omterm{def:g7:prop:table}{समानुपात सारणी} के स्तंभ $\begin{pmatrix} a \\ b
\end{pmatrix}$ और $\begin{pmatrix} c \\ d \end{pmatrix}$ हों, उसमें
तिरछे \omterm{def:g2:mult:def}{गुणनफल} बराबर होते हैं:
\[
a \times d = b \times c .
\]
इससे तीन मानों से चौथा मान निकाला जा सकता है:
$d = \dfrac{b \times c}{a}$।
\end{theorem}

\begin{proof}
मान लो गुणांक $k$ है: $b = ka$ और $d = kc$। तब
\[
a \times d = a \times kc = k \times ac,
\qquad
b \times c = ka \times c = k \times ac :
\]
दोनों तिरछे \omterm{def:g2:mult:def}{गुणनफल} $k \times ac$ के बराबर हैं। $ad = bc$ को $a$ से
भाग देने पर $d$ वाला सूत्र मिल जाता है।
\end{proof}

\begin{example}[चौथा समानुपाती]\label{ex:g7:prop:fourth}
एक जैसी $7$ किताबों का वज़न $2.8$ kg है, तो ऐसी $12$ किताबों का
वज़न कितना होगा? सारणी और तिरछा नियम:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cc}
किताबें & $7$ & $12$ \\
\hline
kg & $2.8$ & $m$ \\
\end{tabular}
\end{center}
\[
7 \times m = 2.8 \times 12
\quad\Longrightarrow\quad
m = \frac{2.8 \times 12}{7} = \frac{33.6}{7} = 4.8 \text{ kg}.
\]
मात्रक से जाँच: एक किताब का वज़न $0.4$ kg, बारह का $4.8$ kg।
\end{example}

\section{प्रतिशत}

\begin{method}[प्रतिशत लगाना और निकालना]\label{met:g7:prop:percent}
\begin{enumerate}
  \item $t\,\%$ \emph{लगाना}: $\frac{t}{100}$ से गुणा करो
        ($350$ का $12\,\%$ है $0.12 \times 350 = 42$);
  \item कोई हिस्सा कितने प्रतिशत है, यह \emph{निकालना}:
        $\frac{\text{हिस्सा}}{\text{पूरा}}$ निकालो और उसे $100$ के
        ऊपर लिखो ($60$ में से $21$:
        $\frac{21}{60} = \frac{35}{100} = 35\,\%$);
  \item हमेशा पूछो: \emph{किसका प्रतिशत?} किसे पूरा माना गया है,
        यह मायने रखता है।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:prop:percent}
एक स्कूल में $400$ में से $180$ विद्यार्थी लड़कियाँ हैं। प्रतिशत:
\[
\frac{180}{400} = \frac{180 \div 4}{400 \div 4} = \frac{45}{100}
= 45\,\% .
\]
दूसरी तरफ़: $55\,\%$ लड़के कितने हैं? $0.55 \times 400 = 220$।
जाँच: $180 + 220 = 400$।
\end{example}

\section{मापनी}

\begin{definition}[मापनी]\label{def:g7:prop:scale}
$\frac{1}{n}$ \emph{मापनी}\index{मापनी} ($1 : n$ लिखी जाने वाली) के
किसी नक़्शे या नमूने पर नक़्शे की दूरियाँ असली दूरियों के \omterm{def:g6:propdata:def}{समानुपाती}
होती हैं, गुणांक $\frac1n$ के साथ:
\[
\text{नक़्शे की दूरी} = \frac{1}{n} \times \text{असली दूरी},
\]
और दोनों \emph{एक ही मात्रक} में नापी जाती हैं।
\end{definition}

\begin{example}\label{ex:g7:prop:scale}
$1 : 50\,000$ \omterm{def:g7:prop:scale}{मापनी} वाले नक़्शे पर दो शहर $7$ cm दूर हैं। असली दूरी:
\[
7 \times 50\,000 = 350\,000 \text{ cm} = 3\,500 \text{ m} = 3.5
\text{ km}.
\]
उलटी तरफ़, $12$ km लंबी पैदल राह नक़्शे पर इतनी नापती है:
$12$ km $= 1\,200\,000$ cm, इसलिए $1\,200\,000 \div 50\,000 = 24$ cm।
\end{example}

\end{axis}
\end{tikzpicture}

{\small \omterm{def:g7:prop:scale}{मापनी} एक \omterm{def:g7:prop:table}{समानुपात} है: नक़्शे की दूरियाँ असली दूरियों के
सामने रखने पर मूलबिंदु से जाती एक सीधी रेखा पर बैठती हैं (यहाँ
$1$ cm $\leftrightarrow$ $4$ km; बिंदुदार रेखाएँ $7$ cm
$\leftrightarrow$ $28$ km पढ़ती हैं)।}
\end{omfigure}

\begin{remark}
\omterm{def:g7:prop:table}{समानुपात} की सबसे तेज़ जाँच आलेख है (\cref{ch:g6:propdata}):
\emph{मूलबिंदु से जाती} सीधी रेखा पर पड़े बिंदु \omterm{def:g6:propdata:def}{समानुपाती} राशियाँ
बताते हैं; जो सीधी रेखा मूलबिंदु से नहीं जाती (जैसे तयशुदा भाड़े
वाली टैक्सी का किराया) वह \omterm{def:g7:prop:table}{समानुपात} \emph{नहीं} है, भले ही वह सीधी
रेखा हो। \cref{ch:g9:linfunc} के एकघाती फलन इसे ठीक-ठीक कहेंगे।
\end{remark}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:prop:8}

\begin{exercise}[$\star\star$]\label{exo:g7:prop:9}

        निकालो।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:prop:10}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:prop:11}
एक फ़ोटोकॉपी मशीन काग़ज़ों को उनके आकार के $80\,\%$ तक छोटा कर देती

\begin{enumerate}

\end{enumerate}
\end{exercise}

\section{समस्या: एक डंडे से पृथ्वी नापना}

\begin{problem}[{सप्ताहांत समस्या --- परछाइयाँ एक समानुपात हैं, और
एरातोस्थनीज़ ने 240 ईसा पूर्व में पृथ्वी की परिधि कैसे नाप ली}]
\label{pb:g7:prop:1}
बाईस सदी पहले, बिना दूरबीन, बिना उपग्रह और बिना कैलकुलेटर के,
सिकंदरिया के पुस्तकालयाध्यक्ष ने पृथ्वी नाप ली --- एक डंडे, एक
कुएँ, ऊँटों के एक क़ाफ़िले और इसी अध्याय के गणित से। यह समस्या उनके
क़दमों पर चलती है: पहले परछाइयों का \omterm{def:g7:prop:table}{समानुपात}, फिर वह मशहूर गणना, और
आख़िर में यह कि उनका उत्तर इंसानी \omterm{def:g7:prop:scale}{मापनी} पर कैसा दिखता है।

\medskip
\noindent\textbf{भाग I --- एक डंडे की परछाईं।}
सूरज इतनी दूर है कि उसकी किरणें हम तक एक-दूसरे के \omterm{def:g6:lines:perp}{समांतर} पहुँचती
हैं। इसलिए किसी एक पल पर सारी खड़ी चीज़ें और उनकी परछाइयाँ
\omterm{def:g6:propdata:def}{समानुपाती} होती हैं।
\begin{enumerate}
  \item दोपहर में $1$ m के एक खड़े डंडे की परछाईं $0.4$ m पड़ती
        है, और एक पेड़ की परछाईं $3.2$ m नापती है। पेड़ कितना ऊँचा
        है (\cref{thm:g7:prop:cross})?
  \item उसी पल एक मीनार की परछाईं $26$ m है: मीनार कितनी ऊँची है?
        और $1.5$ m ऊँचे बच्चे की परछाईं कितनी लंबी होगी?
  \item एक वाक्य में समझाओ कि ये सारी गणनाएँ दिन के \emph{एक ही
        पल} पर ही क्यों सही होती हैं।
  \item कहते हैं थेल्स ने बड़े \omterm{def:g5:solids:def}{पिरामिड} को उस पल का इंतज़ार करके
        नापा जब \emph{उनकी अपनी परछाईं ठीक उनकी ऊँचाई जितनी लंबी
        थी}। उस पल \omterm{def:g7:prop:table}{समानुपात} गुणांक क्या है, और अगर \omterm{def:g5:solids:def}{पिरामिड} की
        परछाईं का सिरा उसके शिखर के ठीक नीचे वाले ज़मीनी बिंदु से
        $147$ m दूर है तो \omterm{def:g5:solids:def}{पिरामिड} कितना ऊँचा है?
  \item भाग I का सार: किसी एक पल पर डंडे, पेड़, मीनार और \omterm{def:g5:solids:def}{पिरामिड}
        --- सबके लिए कौन-सी राशि एक जैसी रहती है
        (\cref{def:g7:prop:table})?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- एरातोस्थनीज़ की गणना।}
एरातोस्थनीज़ दो बातें जानते थे। स्येने नगर में, ग्रीष्म संक्रांति
की दोपहर, सूरज \emph{ठीक सिर के ऊपर} रहता था: धूप गहरे से गहरे
कुओं की तली तक पहुँच जाती थी, और खड़े डंडों की परछाईं बनती ही नहीं
थी। ठीक उसी पल सिकंदरिया में, जो $800$ km सीधे उत्तर है, एक खड़े
डंडे की परछाईं \emph{बनती} थी, और सूरज की किरणें खड़ी दिशा से
$7.2^\circ$ का कोण बनाती थीं।
\begin{enumerate}[resume]
  \item समझाओ कि ये दोनों अवलोकन मिलकर क्यों सिद्ध करते हैं कि
        स्येने की ज़मीन और सिकंदरिया की ज़मीन \omterm{def:g6:lines:perp}{समांतर} नहीं हैं ---
        यानी पृथ्वी की सतह मुड़ी हुई है। (चपटी पृथ्वी पर \omterm{def:g6:lines:perp}{समांतर}
        किरणें दो डंडों के साथ क्या करतीं?)
  \item \omterm{def:g6:lines:perp}{समांतर} सूरज-किरणों के साथ गोल पृथ्वी का चित्र बनाओ: उसमें
        सिकंदरिया वाला $7.2^\circ$ पृथ्वी के केंद्र पर फिर से
        दिखता है, दोनों नगरों की दिशाओं के बीच के कोण की तरह।
        चित्र बनाओ और ख़ुद को समझाओ (साफ़ कारण
        \cref{ch:g7:triangles} के बराबर कोणों के जोड़ों से मिलता
        है)।
  \item $7.2^\circ$ एक पूरे चक्कर का कितना भाग है?
  \item स्येने से सिकंदरिया तक का चाप ($800$ km) $7.2^\circ$ के
        बराबर है; पूरी परिधि $360^\circ$ के। \omterm{def:g7:prop:table}{समानुपात} बनाओ और
        पृथ्वी की परिधि निकालो।
  \item $\pi \approx 3.14$ लेकर (\cref{prop:g6:measure:circle})
        पृथ्वी का \omterm{def:g4:geometry:circle}{व्यास} निकालो, और उसे नज़दीकी सौ किलोमीटर तक गोल
        करो। (आज का मान $12\,742$ km है --- 240 ईसा पूर्व में एक
        डंडे वाला आदमी कितने पास पहुँचा?)
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- इंसानी \omterm{def:g7:prop:scale}{मापनी} पर पृथ्वी।}
\begin{enumerate}[resume]
  \item एक भूगोलक $1 : 40\,000\,000$ \omterm{def:g7:prop:scale}{मापनी} पर बना है। सवाल 9 में
        मिली परिधि से दिखाओ कि भूगोलक की परिधि ठीक $1$ m है। उसका
        \omterm{def:g4:geometry:circle}{व्यास} कितना है, नज़दीकी सेंटीमीटर तक?
  \item मों ब्लां क़रीब $4.8$ km ऊँचा है। $4.8$ km को सेंटीमीटर
        में बदलो, $40\,000\,000$ से भाग दो, और भूगोलक पर उस पहाड़
        की ऊँचाई मिलीमीटर में बताओ। इस उत्तर से पता चलता है कि
        पृथ्वी सचमुच कितनी ``ऊबड़-खाबड़'' है?
  \item एक पैदल यात्री रोज़ $40$ km चलता है। इस रफ़्तार से
        एरातोस्थनीज़ की पूरी परिधि नापने में कितने दिन लगेंगे ---
        और मोटे तौर पर वह कितने साल हुए?
  \item साँस लेने लायक़ वायुमंडल ज़मीन से क़रीब पहले $10$ km में ही
        सिमटा है। यह पृथ्वी की \omterm{def:g3:shapes:shapes}{त्रिज्या} (क़रीब $6\,370$ km) का
        कितने प्रतिशत है (\cref{met:g7:prop:percent})? प्रतिशत को
        दशमलव के एक अंक तक गोल करो।
  \item इतिहासकार मानते हैं कि एरातोस्थनीज़ ने जो मान बताया था वह
        आज के मात्रकों में क़रीब $42\,000$ km रहा होगा।
        $40\,000$ km को सही मान मानकर उनकी ग़लती का प्रतिशत
        निकालो। डंडों, कुओं और \omterm{def:g7:prop:table}{समानुपात} के बारे में एक वाक्य में
        नतीजा लिखो।
\end{enumerate}
\end{problem}
