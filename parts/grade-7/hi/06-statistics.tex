\chapter{आँकड़ों को सँवारना}\label{ch:g7:stats}

कुछ भी गिनने-निकालने से पहले आँकड़े जुटाने, गिनने और दिखाने पड़ते
हैं। यह अध्याय बिखरी हुई देखी-सुनी बातों की सूची को एक साफ़ सारणी
या चित्र में बदलना सिखाता है: \omterm{def:g7:stats:frequency}{गिनती}, \omterm{def:g7:stats:frequency}{आवृत्ति}, श्रेणियों में बँटे
आँकड़े, पट्टी-आलेख और वृत्त-आलेख। आँकड़ों को एक ही संख्या (माध्य,
माध्यिका) में समेटना \cref{ch:g9:statproba} का विषय है।

\section{गिनती और आवृत्ति}

\begin{definition}[गिनती और आवृत्ति]\label{def:g7:stats:frequency}
किसी सर्वेक्षण के प्रत्येक संभव मान की \emph{गिनती}\index{गिनती}
यह है कि वह कितनी बार आया; और उसकी \emph{आवृत्ति}\index{आवृत्ति}
गिनती को कुल अवलोकनों की संख्या से भाग देने पर मिलती है:
\[
\text{आवृत्ति} = \frac{\text{गिनती}}{\text{कुल}} .
\]
आवृत्तियाँ \omterm{def:g6:fractions:def}{भिन्न}, दशमलव या प्रतिशत में लिखी जा सकती हैं, और उनका
योग हमेशा $1$ (यानी $100\,\%$) होता है।
\end{definition}

\begin{example}\label{ex:g7:stats:frequency}
एक कक्षा के $25$ विद्यार्थियों का मनपसंद खेल, कच्ची सूची से सारणी
तक:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cccc|c}
खेल & फ़ुटबॉल & तैराकी & टेनिस & जूडो & कुल \\
\hline
\omterm{def:g7:stats:frequency}{गिनती} & $10$ & $6$ & $5$ & $4$ & $25$ \\
\omterm{def:g7:stats:frequency}{आवृत्ति} & $0.40$ & $0.24$ & $0.20$ & $0.16$ & $1$ \\
\end{tabular}
\end{center}
फ़ुटबॉल के लिए: $\frac{10}{25} = \frac{40}{100} = 0.40 = 40\,\%$।
\omterm{def:g7:stats:frequency}{आवृत्ति} वाली पंक्ति का योग $1$ आना ही चाहिए --- एक पक्की, मुफ़्त
जाँच।
\end{example}

\begin{remark}[आवृत्ति क्यों?]
अलग-अलग आकार के समूहों की गिनतियाँ आपस में नहीं तुलनी चाहिए: $25$
की कक्षा में $10$ फ़ुटबॉल-प्रेमी ($40\,\%$) उस साल के $120$
विद्यार्थियों में से $12$ ($10\,\%$) से ज़्यादा जोश दिखाते हैं।
\omterm{def:g7:stats:frequency}{आवृत्ति} सबको एक ही \omterm{def:g7:prop:scale}{मापनी} पर ले आती है।
\end{remark}

\section{आँकड़ों को श्रेणियों में बाँटना}

\begin{example}[श्रेणियाँ]\label{ex:g7:stats:classes}
$20$ विद्यार्थियों की ऊँचाइयाँ (cm में) $148$ से $172$ तक फैली
हैं। हर ऊँचाई अलग लिखने पर सारणी में लगभग उतने ही स्तंभ बन जाएँगे
जितने विद्यार्थी; इसके बजाय उन्हें \emph{श्रेणियों} में बाँटो:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|ccc|c}
ऊँचाई (cm) & $\intco{145}{155}$ & $\intco{155}{165}$ &
$\intco{165}{175}$ & कुल \\
\hline
\omterm{def:g7:stats:frequency}{गिनती} & $5$ & $9$ & $6$ & $20$ \\
\omterm{def:g7:stats:frequency}{आवृत्ति} & $0.25$ & $0.45$ & $0.30$ & $1$ \\
\end{tabular}
\end{center}
प्रत्येक ऊँचाई ठीक एक ही श्रेणी में गिनी जाती है ($155$,
$\intco{155}{165}$ में जाता है, दोनों में नहीं --- इसीलिए अंतराल
एक तरफ़ से खुले रखे गए हैं)। बाँटने से ब्योरा कुछ खोता है, पर
पढ़ने में आसानी आ जाती है।
\end{example}

\section{आलेख}

\begin{method}[आलेख चुनना और बनाना]\label{met:g7:stats:charts}
\begin{enumerate}
  \item \emph{पट्टी-आलेख}: प्रत्येक मान या श्रेणी के लिए एक पट्टी,
        जिसकी ऊँचाई \omterm{def:g7:stats:frequency}{गिनती} (या \omterm{def:g7:stats:frequency}{आवृत्ति}) के \omterm{def:g6:propdata:def}{समानुपाती} हो। मानों की
        आपसी तुलना के लिए अच्छा।
  \item \emph{वृत्त-आलेख}: प्रत्येक मान के लिए एक फाँक, जिसका कोण
        \omterm{def:g7:stats:frequency}{आवृत्ति} के \omterm{def:g6:propdata:def}{समानुपाती} हो --- पूरा वृत्त ($360^\circ$) कुल
        को दिखाता है, इसलिए
        \[
        \text{कोण} = \text{आवृत्ति} \times 360^\circ .
        \]
        पूरे में किसका कितना हिस्सा है, यह दिखाने के लिए अच्छा।
  \item दोनों में: शीर्षक, नाम, और या तो अंकन या आलेख पर लिखे हुए
        प्रतिशत।
\end{enumerate}
\end{method}

\begin{example}\label{ex:g7:stats:pie}
\cref{ex:g7:stats:frequency} के खेलों के कोण: फ़ुटबॉल
$0.40 \times 360 = 144^\circ$; तैराकी $0.24 \times 360 =
86.4^\circ$; टेनिस $72^\circ$; जूडो $57.6^\circ$। जाँच:
$144 + 86.4 + 72 + 57.6 = 360$।
\end{example}

\begin{example}[परखकर पढ़ना]\label{ex:g7:stats:critical}
जिस आलेख की खड़ी धुरी $0$ से नहीं, $9$ से शुरू होती है, उसमें $10$
वाली पट्टी $9.1$ वाली से दस गुना ऊँची लगती है --- संख्याएँ ईमानदार
हैं, फिर भी तस्वीर धोखा देती है। किसी भी आलेख के सामने पहला काम:
देखो धुरी कहाँ से शुरू होती है और एक अंकन कितने का है
(\cref{met:g6:propdata:read})।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g7:stats:1}
$20$ विद्यार्थियों के अंक ये रहे: $12$, $15$, $9$, $12$, $14$,
$9$, $12$, $15$, $11$, $12$, $9$, $14$, $12$, $11$, $15$, $9$, $12$,
$14$, $11$, $12$। गिनतियों की सारणी बनाओ (प्रत्येक अंक का एक स्तंभ)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:2}
\cref{exo:g7:stats:1} की सारणी में \omterm{def:g7:stats:frequency}{आवृत्ति} वाली पंक्ति जोड़ो
($20$ की भिन्नों के रूप में, फिर प्रतिशत में)। योग जाँचो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:3}
$50$ परिवारों के एक सर्वेक्षण में $30$ के पास एक कार है।
``एक कार'' की \omterm{def:g7:stats:frequency}{आवृत्ति} दशमलव में और प्रतिशत में क्या है? एक दूसरे
सर्वेक्षण में $90$ में से $45$ परिवारों के पास एक कार है: किस
सर्वेक्षण में अनुपात ज़्यादा है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:4}
$15$ कुत्तों के वज़न (kg): $8$, $23$, $31$, $12$, $9$, $27$, $35$,
$14$, $22$, $18$, $29$, $11$, $25$, $33$, $16$। उन्हें
$\intco{5}{15}$, $\intco{15}{25}$, $\intco{25}{35}$,
$\intco{35}{45}$ श्रेणियों में बाँटो और गिनतियाँ बताओ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:5}
इस सारणी का पट्टी-आलेख बनाओ:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{l|cccc}
पालतू & $0$ & $1$ & $2$ & $3$ \\
\hline
परिवार & $8$ & $12$ & $6$ & $4$ \\
\end{tabular}
\end{center}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:6}
\cref{exo:g7:stats:5} की सारणी के लिए प्रत्येक मान की \omterm{def:g7:stats:frequency}{आवृत्ति} और
वृत्त-आलेख का कोण निकालो (जाँच: कुल $360^\circ$)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:stats:7}
मनपसंद ऋतुओं का एक वृत्त-आलेख दिखाता है: गर्मी $162^\circ$, वसंत
$90^\circ$, पतझड़ $54^\circ$, जाड़ा $54^\circ$। हर ऋतु को कितने
प्रतिशत लोगों ने चुना? अगर $60$ लोगों ने जवाब दिया, तो गर्मी
कितनों ने चुनी?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:8}
कक्षा A में $30$ में से $12$ विद्यार्थी पैदल स्कूल आते हैं; कक्षा
B में $40$ में से $14$। किस कक्षा में पैदल आने वालों का अनुपात
ज़्यादा है? गिनतियों से नहीं, आवृत्तियों से कारण बताओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:9}
एक पत्रिका महीने की बिक्री का पट्टी-आलेख छापती है: $9\,800$, फिर
$10\,000$, फिर $10\,100$, और खड़ी धुरी $9\,700$ से शुरू होती है।
बताओ पाठक को क्या दिखता है, और $0$ से शुरू होने वाली ईमानदार धुरी
इसकी जगह क्या दिखाती।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:stats:10}
एक आवृत्ति-सारणी में तीन मान हैं, जिनकी आवृत्तियाँ $0.35$, $0.4$
और $f$ हैं। $f$ निकालो। अगर कुल \omterm{def:g7:stats:frequency}{गिनती} $80$ है तो तीनों गिनतियाँ
बताओ।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:stats:11}
एक स्कूल में $55\,\%$ विद्यार्थी लड़कियाँ हैं। लड़कियों में से
$40\,\%$ भोजनालय में खाती हैं; लड़कों में से $60\,\%$। $400$
विद्यार्थियों में से कितने भोजनालय में खाते हैं? (चारों समूहों के
आकार क़दम दर क़दम निकालो।) यह कुल मिलाकर कितने प्रतिशत हुआ?
\end{exercise}

\section{समस्या: आँकड़े कैसे झूठ बोलते हैं --- और उन्हें कैसे पकड़ें}

\begin{problem}[{सप्ताहांत समस्या --- गिनती बनाम आवृत्ति, वह
सर्वेक्षण जिसने पूरे देश को धोखा दिया, और दो अस्पतालों का
विरोधाभास}]
\label{pb:g7:stats:1}
संख्याएँ झूठ नहीं बोलतीं, पर उनसे भरमाया जा सकता है: एक बड़ी \omterm{def:g7:stats:frequency}{गिनती}
छोटी \omterm{def:g7:stats:frequency}{आवृत्ति} को छिपा सकती है, एक विशाल सर्वेक्षण बेकार हो सकता है,
और --- सबसे अजीब बात --- कोई अस्पताल मरीज़ों की \emph{हर} श्रेणी
में अपने प्रतिद्वंद्वी से जीतकर भी कुल आँकड़ों में हार सकता है। इस
समस्या में तीनों जाल खुलते हैं, और हथियार सिर्फ़ इसी अध्याय की
गिनतियाँ और आवृत्तियाँ हैं (\cref{def:g7:stats:frequency})।

\medskip
\noindent\textbf{भाग I --- \omterm{def:g7:stats:frequency}{गिनती} \omterm{def:g7:stats:frequency}{आवृत्ति} नहीं होती।}
\begin{enumerate}
  \item स्कूल A अपने $400$ रस-डिब्बों में से $120$ का पुनर्चक्रण
        करता है; स्कूल B अपने $250$ में से $90$ का। किस स्कूल की
        पुनर्चक्रित डिब्बों की \emph{\omterm{def:g7:stats:frequency}{गिनती}} बड़ी है? किसकी
        \emph{\omterm{def:g7:stats:frequency}{आवृत्ति}} बड़ी है? पुनर्चक्रण का इनाम किस स्कूल को
        मिलना चाहिए?
  \item एक नई समय-सारणी पर राय ली जाती है: $30$ में से $18$
        शिक्षकों को वह पसंद है, और $600$ में से $210$
        विद्यार्थियों को। दोनों आवृत्तियाँ निकालो, फिर पूरे $630$
        लोगों में \omterm{def:g7:stats:frequency}{आवृत्ति}। मिली-जुली \omterm{def:g7:stats:frequency}{आवृत्ति} विद्यार्थियों वाली
        \omterm{def:g7:stats:frequency}{आवृत्ति} के इतने पास क्यों बैठती है?
  \item छपे हुए एक वृत्त-आलेख की फाँकों पर $45\,\%$, $30\,\%$,
        $20\,\%$ और $10\,\%$ लिखा है। और कोई जानकारी लिए बिना
        तुम्हें कैसे पता चलता है कि कहीं ग़लती हुई है?
  \item $40$ अवलोकनों की एक आवृत्ति-सारणी आधी मिट गई है: गिनतियाँ
        $14$, $10$, $?$, $?$ और आवृत्तियाँ $0.35$, $0.25$,
        $0.15$, $?$। मिटी हुई प्रविष्टियाँ फिर से बनाओ।
  \item एक सर्वेक्षण की आवृत्तियाँ $0.40$, $0.35$ और $0.25$
        निकलती हैं। वृत्त-आलेख के तीनों कोण निकालो
        (\cref{met:g7:stats:charts}) और जाँचो कि वे वृत्त पूरा
        कर देते हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- वह सर्वेक्षण जिसने पूरे देश को धोखा
दिया।}
सन 1936 में एक अमेरिकी पत्रिका ने टेलीफ़ोन-निर्देशिकाओं और
गाड़ियों के पंजीकरण से लिए गए पतों पर एक करोड़ मतपत्र भेजे, $24$
लाख जवाब पाए, और लैंडन नाम के उम्मीदवार की भारी जीत की भविष्यवाणी
की। जॉर्ज गैलप नाम के एक युवा सांख्यिकीविद् ने सिर्फ़ क़रीब पचास
हज़ार लोगों से पूछा --- ऐसे चुने हुए कि वे पूरी आबादी से मिलते-जुलते
हों --- और उलटी भविष्यवाणी की। चुनाव रूज़वेल्ट ने भारी \omterm{ex:g1:subtraction:difference}{अंतर} से
जीता।
\begin{enumerate}[resume]
  \item भेजे गए मतपत्रों में से कितनी \omterm{def:g7:stats:frequency}{आवृत्ति} वापस आई?
  \item सन 1936 में टेलीफ़ोन और गाड़ियाँ अमीरी की निशानी थीं। एक
        या दो वाक्यों में समझाओ कि पत्रिका का नमूना इतना विशाल
        होकर भी क्यों डूबना ही था --- और यह सवाल 1 का कौन-सा सबक
        पूरे देश की \omterm{def:g7:prop:scale}{मापनी} पर दोहराता है।
  \item एक खिलौना-नमूना। किसी शहर में $1\,000$ अमीर मतदाता हैं,
        जिनमें से $70\,\%$ L को चाहते हैं, और $9\,000$ साधारण
        मतदाता, जिनमें से $30\,\%$ L को। शहर में L का असली समर्थन
        निकालो। एक ``टेलीफ़ोन-निर्देशिका सर्वेक्षण'' $500$ अमीर
        और $500$ साधारण मतदाताओं तक पहुँचता है: वह कितने समर्थन
        की भविष्यवाणी करेगा?
  \item चुपचाप बिछा जाल: किसी कंपनी के $200$ ग्राहक नाख़ुश हैं और
        $800$ ख़ुश। वह सबसे राय पूछती है; नाख़ुशों में से
        $40\,\%$ जवाब देते हैं, पर ख़ुशों में से सिर्फ़ $10\,\%$।
        दोनों तरह के जवाबों की संख्या निकालो, और
        \emph{जवाब देने वालों में} नाख़ुशी की \omterm{def:g7:stats:frequency}{आवृत्ति}। उसे असली
        \omterm{def:g7:stats:frequency}{आवृत्ति} से तुलो।
  \item सवाल 8 वाले सर्वेक्षण को सुधारो: $1\,000$ बातचीत उतनी ही
        रखो, पर उन्हें शहर के असली अनुपात में बाँटो। सुधरा हुआ
        सर्वेक्षण अब क्या भविष्यवाणी करता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- दो अस्पतालों का विरोधाभास।}
दो अस्पताल अपने ठीक हुए मरीज़ों की गिनतियाँ, बीमारी की गंभीरता के
हिसाब से बाँटकर, छापते हैं:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|cc}
 & हल्के मामले & गंभीर मामले \\
\hline
अस्पताल A & $100$ में से $90$ ठीक & $400$ में से $280$ ठीक \\
अस्पताल B & $400$ में से $340$ ठीक & $100$ में से $65$ ठीक \\
\end{tabular}
\end{center}
\begin{enumerate}[resume]
  \item अस्पताल A के ठीक होने की \omterm{def:g7:stats:frequency}{आवृत्ति} निकालो: हल्के मामलों
        में, गंभीर मामलों में, और कुल मिलाकर (पूरे $500$ मरीज़ों
        पर)।
  \item वही तीनों गणनाएँ अस्पताल B के लिए। हल्के मामलों में कौन
        जीतता है? गंभीर मामलों में?
  \item दोनों की कुल आवृत्तियाँ तुलो। जो अजीब बात हुई है उसे
        साफ़-साफ़ लिखो।
  \item चाल समझाओ: दोनों अस्पतालों के मरीज़ों के \emph{मिश्रण}
        तुलो, और बताओ कि कुल \omterm{def:g7:stats:frequency}{आवृत्ति} दोनों श्रेणियों की
        आवृत्तियों से धोखा कैसे कर सकती है। अगर तुम्हारा मामला
        गंभीर हो तो किस अस्पताल को चुनना चाहिए?
  \item एक अख़बार की सुर्खी है: ``अस्पताल B ठीक करने में आगे:
        $81\,\%$ बनाम $74\,\%$।'' संपादक के नाम वह दो वाक्यों की
        चिट्ठी लिखो जो यह पूरी समस्या तुम्हें सिखा चुकी है ---
        एक वाक्य इस पर कि सुर्खी क्या छिपा रही है, और एक आम सीख
        पर (एक जैसी चीज़ों को ही तुलो, और भार का ध्यान रखो)।
\end{enumerate}
\end{problem}
