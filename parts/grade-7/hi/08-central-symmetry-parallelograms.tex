\chapter{केंद्रीय सममिति और समांतर चतुर्भुज}\label{ch:g7:central}

किसी आकृति को एक बिंदु के चारों ओर \omterm{def:g3:division:half}{आधा} चक्कर घुमा दो: यही
\emph{\omterm{def:g7:central:def}{केंद्रीय सममिति}} है, \cref{ch:g6:symmetry} के परावर्तन के बाद
इस किताब का दूसरा रूपांतरण। इसकी सबसे चमकीली आकृति \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज}
है --- वह \omterm{def:g4:geometry:polygon}{चतुर्भुज} जो अपने ही विकर्णों के कटान-बिंदु के बारे में
\omterm{def:g7:central:def}{सममित} होता है।

\section{केंद्रीय सममिति}

\begin{definition}[किसी बिंदु के बारे में सममित]\label{def:g7:central:def}
किसी बिंदु $M$ का, किसी बिंदु $O$ के बारे में,
\emph{सममित}\index{केंद्रीय सममिति} बिंदु वह $M'$ है जिसके लिए $O$,
$[MM']$ का \emph{मध्यबिंदु} हो। सममित आकृति वह है जो मूल आकृति को
$O$ के चारों ओर \omterm{def:g3:division:half}{आधा} चक्कर ($180^\circ$) घुमाने पर बनती है।
\end{definition}

देखो)।}
\end{omfigure}

\begin{method}[किसी बिंदु का सममित बिंदु बनाना]\label{met:g7:central:construct}
$O$ के बारे में $M$ का \omterm{def:g7:central:def}{सममित} बिंदु $M'$ बनाने के लिए:
\begin{enumerate}
  \item रेखा $(MO)$ खींचो और उसे $O$ के आगे बढ़ाओ;
  \item $MO$ नापो (पैमाने या परकार से);
  \item बढ़ी हुई रेखा पर $M'$ इस तरह रखो कि $OM' = OM$ हो।
\end{enumerate}
जाल वाले काग़ज़ पर: $M$ से $O$ तक आड़े और खड़े क़दम गिनो, और
\emph{उतने ही} क़दम $O$ से आगे चलकर $M'$ तक पहुँचो।
\end{method}

\begin{example}[जाल वाले काग़ज़ पर]\label{ex:g7:central:grid}
अगर $M$, $O$ से $3$ ख़ाने बाईं ओर और $1$ ख़ाना ऊपर है, तो उसका
\omterm{def:g7:central:def}{सममित} बिंदु $M'$, $O$ से $3$ ख़ाने \emph{दाईं} ओर और $1$ ख़ाना
\emph{नीचे} होगा: \omterm{def:g7:central:def}{केंद्रीय सममिति} दोनों दिशाएँ एक साथ उलट देती है
(परावर्तन सिर्फ़ एक उलटता है)।
\end{example}

\begin{proposition}[आधा चक्कर क्या-क्या बचाए रखता है]\label{prop:g7:central:preserved}
\omterm{def:g7:central:def}{केंद्रीय सममिति} लंबाइयाँ, कोण, \omterm{def:g6:measure:perimeter}{परिमाप} और \omterm{def:g6:measure:area}{क्षेत्रफल} बचाए रखती है;
किसी रेखा का प्रतिबिंब उसके \emph{\omterm{def:g6:lines:perp}{समांतर}} रेखा होती है, किसी
\omterm{def:g6:lines:objects}{रेखाखंड} का प्रतिबिंब उतनी ही लंबाई का \omterm{def:g6:lines:perp}{समांतर} \omterm{def:g6:lines:objects}{रेखाखंड}, और किसी वृत्त
का प्रतिबिंब उतनी ही \omterm{def:g3:shapes:shapes}{त्रिज्या} का वृत्त (जिसका केंद्र पहले केंद्र का
\omterm{def:g7:central:def}{सममित} बिंदु होता है)।
\end{proposition}
\admitted

\begin{remark}
परावर्तन से तुलना करो (\cref{prop:g6:symmetry:preserved}): दोनों
लंबाइयाँ और कोण बचाए रखते हैं; पर परावर्तन आकृति को पलट देता है
(अक्षर $b$ से $d$ बन जाता है), जबकि \omterm{def:g3:division:half}{आधा} चक्कर उसे पढ़ने लायक़ ही
रखता है --- बस उलटा ($b$ से $q$)। और परावर्तन के उलट, \omterm{def:g3:division:half}{आधा} चक्कर
प्रत्येक रेखा को उसी के \omterm{def:g6:lines:perp}{समांतर} रेखा पर भेजता है।
\end{remark}

\begin{definition}[सममिति केंद्र]\label{def:g7:central:center}
कोई बिंदु $O$ किसी आकृति का \emph{सममिति केंद्र}\index{सममिति केंद्र}
तब होता है जब $O$ के चारों ओर \omterm{def:g3:division:half}{आधा} चक्कर उस आकृति को ठीक उसी पर भेज
दे। जैसे अक्षर S, N, Z का एक सममिति केंद्र होता है; वृत्त का भी
होता है (उसका केंद्र); त्रिभुज का कभी नहीं होता।
\end{definition}

\section{समांतर चतुर्भुज}

\begin{definition}[समांतर चतुर्भुज]\label{def:g7:central:parallelogram}
\emph{समांतर चतुर्भुज}\index{समांतर चतुर्भुज} वह \omterm{def:g4:geometry:polygon}{चतुर्भुज} है जिसके
विकर्ण अपने साझे मध्यबिंदु पर कटते हैं। तब वही कटान-बिंदु आकृति का
\omterm{def:g7:central:center}{सममिति केंद्र} होता है: प्रत्येक \omterm{def:g5:solids:def}{शीर्ष} सामने वाले \omterm{def:g5:solids:def}{शीर्ष} का आधे चक्कर
वाला प्रतिबिंब होता है।
\end{definition}

\begin{theorem}[समांतर चतुर्भुज के गुण]\label{thm:g7:central:properties}
केंद्र $O$ वाले \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} $ABCD$ में:
\begin{enumerate}
  \item आमने-सामने की भुजाएँ \omterm{def:g6:lines:perp}{समांतर} होती हैं:
        $(AB) \parallel (CD)$ और $(BC) \parallel (AD)$;
  \item आमने-सामने की भुजाओं की लंबाई बराबर होती है: $AB = CD$ और
        $BC = AD$;
  \item आमने-सामने के कोण बराबर होते हैं: $\widehat A = \widehat C$
        और $\widehat B = \widehat D$।
\end{enumerate}
\end{theorem}

\begin{proof}
$O$ के चारों ओर \omterm{def:g3:division:half}{आधा} चक्कर $A$ को $C$ पर और $B$ को $D$ पर भेजता है
(परिभाषा: $O$ दोनों विकर्णों का मध्यबिंदु है)। इसलिए वह \omterm{def:g6:lines:objects}{रेखाखंड}
$[AB]$ को \omterm{def:g6:lines:objects}{रेखाखंड} $[CD]$ पर भेजता है; और
\cref{prop:g7:central:preserved} से प्रतिबिंब $[AB]$ के \omterm{def:g6:lines:perp}{समांतर} तथा
उतनी ही लंबाई का होता है: बातें 1 और 2। वह $A$ वाले कोण को $C$ वाले
कोण पर भी भेजता है, और कोण बचे रहते हैं: बात 3।
\end{proof}

\begin{theorem}[समांतर चतुर्भुज पहचानना]\label{thm:g7:central:recognize}
कोई \omterm{def:g4:geometry:polygon}{चतुर्भुज} (जिसकी भुजाएँ आपस में न कटती हों) तभी \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज}
हो जाता है जब इनमें से \emph{कोई एक} बात सही हो:
\begin{enumerate}
  \item उसके विकर्णों का मध्यबिंदु एक ही हो (यही परिभाषा है);
  \item उसकी आमने-सामने की भुजाएँ दो-दो करके \omterm{def:g6:lines:perp}{समांतर} हों;
  \item आमने-सामने की दो भुजाएँ \omterm{def:g6:lines:perp}{समांतर} \emph{और} बराबर लंबाई की
        हों।
\end{enumerate}
\end{theorem}
\admitted

\begin{example}\label{ex:g7:central:recognize}
$ABCD$ में $(AB) \parallel (CD)$ और $AB = CD = 5$ cm है। कसौटी 3
लागू होती है: $ABCD$ एक \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} है --- यानी बिना और कुछ नापे
हमें यह भी पता चल गया कि $AD = BC$, $(AD) \parallel (BC)$, और उसके
विकर्ण एक-दूसरे को बीचोंबीच काटते हैं।
\end{example}

\begin{remark}[ख़ास समांतर चतुर्भुज]
\cref{ch:g6:shapes} वाले आयत, \omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज} और वर्ग ठीक वही \omterm{def:g7:central:parallelogram}{समांतर
चतुर्भुज} हैं जिनमें एक गुण और है: बराबर विकर्ण (आयत), \omterm{def:g6:lines:perp}{लंब} विकर्ण
(\omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज}), दोनों (वर्ग)। \cref{ex:g6:shapes:recognize} में वादा
किया गया कोण वाला तर्क अब चल पड़ता है: \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} में लगातार
आने वाले कोण संपूरक होते हैं (\omterm{def:g6:lines:perp}{समांतर} रेखाओं के साथ एकांतर तथा संगत
कोण, \cref{thm:g7:triangles:equalities}), इसलिए एक \omterm{def:g6:shapes:triangles}{समकोण} चारों को
\omterm{def:g6:shapes:triangles}{समकोण} बना देता है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g7:central:1}
जाल वाले काग़ज़ पर $O$ रखो, फिर $M$ को $O$ से चार ख़ाने दाईं ओर और
एक ख़ाना ऊपर। $O$ के बारे में $M$ का \omterm{def:g7:central:def}{सममित} बिंदु $M'$ बनाओ, और $O$
से होकर जाती \emph{खड़ी जाल-रेखा} के बारे में भी $M$ का \omterm{def:g7:central:def}{सममित} बिंदु
बनाओ। दोनों प्रतिबिंबों की तुलना करो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:2}
एक त्रिभुज $ABC$ और उसके बाहर एक बिंदु $O$ बनाओ। $O$ के चारों ओर
आधे चक्कर से उसका प्रतिबिंब $A'B'C'$ बनाओ। लंबाइयों $A'B'$ और $AB$
के बारे में तुम क्या कह सकते हो? और रेखाओं $(A'B')$ तथा $(AB)$ के
बारे में?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:3}
कैलकुलेटर के परदे पर लिखे जाने वाले किन अंकों ($0$ से $9$) का एक
\omterm{def:g7:central:center}{सममिति केंद्र} होता है? और H, A, N, S, T, Z, O में से किन बड़े अक्षरों
का?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:4}
$KLMN$ केंद्र $O$ वाला एक \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} है, जिसमें $KL = 7$ cm,
$LM = 4$ cm और $\widehat K = 115^\circ$ है। कारण सहित बताओ: $MN$,
$NK$, $\widehat M$, और $[LN]$ का मध्यबिंदु।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:5}
अपने विकर्णों से एक \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} $ABCD$ बनाओ: $AC = 8$ cm और
$BD = 5$ cm, जो अपने मध्यबिंदुओं पर $60^\circ$ का कोण बनाते हुए
कटते हैं। (पहले विकर्ण खींचो!)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g7:central:6}
किसी \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} का एक कोण $115^\circ$ का है। लगातार आने वाले
कोणों के संपूरक होने वाली टिप्पणी से बाक़ी तीनों कोण बताओ।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:7}
$A(1, 1)$, $B(4, 2)$ और बिंदु $O(2.5, 2.5)$ लगाओ। $O$ के चारों ओर
आधे चक्कर से बने प्रतिबिंबों $A'$ और $B'$ के निर्देशांक निकालो
(\cref{ex:g7:central:grid} वाली ख़ाने गिनने की तरकीब से)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:8}
$ABCD$ एक ऐसा \omterm{def:g4:geometry:polygon}{चतुर्भुज} है जिसमें $AB = CD$ है। क्या इतने से वह
\omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} बन जाता है? अगर नहीं, तो एक प्रति-उदाहरण बनाओ (एक
\omterm{def:g6:shapes:triangles}{समद्विबाहु} समलंब मदद करेगा), और बताओ कि शर्त में और क्या जोड़ना
पड़ेगा।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:9}
मान लो $ABC$ एक त्रिभुज है और $O$, $[BC]$ का मध्यबिंदु। $O$ के बारे
में $A$ का \omterm{def:g7:central:def}{सममित} बिंदु $A'$ बनाओ। दिखाओ कि $ABA'C$ एक \omterm{def:g7:central:parallelogram}{समांतर
चतुर्भुज} है। (\cref{thm:g7:central:recognize} की कौन-सी कसौटी यहाँ
मुफ़्त में मिल जाती है?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g7:central:10}
सही या ग़लत, कारण के साथ: ``जिस \omterm{def:g4:geometry:polygon}{चतुर्भुज} के आमने-सामने के कोण
दो-दो करके बराबर हों वह \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} होता है''; ``जिस \omterm{def:g7:central:parallelogram}{समांतर
चतुर्भुज} के विकर्ण \omterm{def:g6:lines:perp}{लंब} हों वह \omterm{def:g6:shapes:quadrilaterals}{समचतुर्भुज} होता है''; ``जिस \omterm{def:g7:central:parallelogram}{समांतर
चतुर्भुज} का एक कोण \omterm{def:g6:shapes:triangles}{समकोण} हो वह आयत होता है''।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:g7:central:11}
मान लो $ABCD$ कोई भी \omterm{def:g4:geometry:polygon}{चतुर्भुज} है और $I$, $J$, $K$, $L$ उसकी भुजाओं
$[AB]$, $[BC]$, $[CD]$, $[DA]$ के मध्यबिंदु। कई उदाहरण बनाओ।
$IJKL$ हमेशा क्या दिखता है? (मध्यबिंदु प्रमेय वाला प्रमाण
\cref{ch:g8:midpoints} में आएगा --- और सदिशों वाला प्रमाण
हाई-स्कूल खंड में।)
\end{exercise}

\section{समस्या: केक, माध्यिका, और वह दूसरा केंद्र जो हो ही नहीं
सकता}

\begin{problem}[{सप्ताहांत समस्या --- काम पर लगे आधे चक्कर: केक की
इंसाफ़ वाली कटाई, माध्यिकाओं की सीमा, और क्यों किसी आकृति के दो
सममिति केंद्र नहीं हो सकते}]
\label{pb:g7:central:1}
\omterm{def:g3:division:half}{आधा} चक्कर सारे रूपांतरणों में सबसे सीधा-सादा लगता है --- फिर भी वह
केक को सिद्ध रूप से बराबर हिस्सों में काट देता है, त्रिभुज की
माध्यिकाएँ नाप देता है, और शुद्ध गणित का एक नन्हा मोती छिपाए बैठा
है: किसी घिरी हुई आकृति का एक \omterm{def:g7:central:center}{सममिति केंद्र} हो सकता है, पर
\emph{दो कभी नहीं}। इसका प्रमाण भाग I में की गई एक खोज पर टिका है:
एक के बाद एक किए गए दो आधे चक्कर मिलकर एक सरकाव बन जाते हैं।

\medskip
\noindent\textbf{भाग I --- आधे चक्कर की क़वायद।}
\begin{enumerate}
  \item जाल वाले काग़ज़ पर मूलबिंदु $O(0, 0)$ को केंद्र मानो।
        \cref{ex:g7:central:grid} वाली ख़ाने गिनने की तरकीब से
        $O$ के चारों ओर आधे चक्कर पर $M(3, 1)$, $N(-2, 4)$ और
        $P(0, -5)$ के प्रतिबिंब बताओ। $(x, y)$ के लिए आम नियम
        क्या है?
  \item अब केंद्र $C(2, 1)$ है। $M(5, 3)$ का प्रतिबिंब निकालो,
        और यह आम नुस्ख़ा जाँचो: प्रतिबिंब का प्रत्येक निर्देशांक
        \emph{केंद्र के दुगुने में से बिंदु का निर्देशांक घटाकर}
        मिलता है।
  \item ताश के पत्ते ऐसे बनाए जाते हैं कि उलटे भी वैसे ही पढ़े
        जाएँ --- यानी \omterm{def:g7:central:def}{केंद्रीय सममिति}, ताकि किसी भी खिलाड़ी को
        पत्ता ``उलटा'' न दिखे। समझाओ कि जिस पत्ते पर
        \emph{विषम} संख्या में निशान हों, उनमें से एक निशान ठीक
        पत्ते के केंद्र पर क्यों होना ही चाहिए।
  \item एक के बाद एक दो आधे चक्कर: केंद्र $O_1(2, 0)$ और
        $O_2(5, 0)$ लो। बिंदु $A(1, 1)$ को $O_1$ के चारों ओर आधे
        चक्कर से भेजो, फिर उस प्रतिबिंब को $O_2$ के चारों ओर आधे
        चक्कर से। शुरुआत और अंत की तुलना करो। यही $B(0, 3)$ के
        साथ दोहराओ। दोनों आधे चक्कर मिलकर कौन-सा एक, सीधा-सादा
        रूपांतरण बन गए?
  \item सवाल 4 वाले सरकाव को दूरी $O_1 O_2$ के सामने नापो, और
        अपनी खोज लिखो। (\cref{pb:g6:symmetry:1} के दो \omterm{def:g6:lines:perp}{समांतर}
        दर्पणों से तुलना करो: वही घटना, नए खिलाड़ी।)
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- केक और माध्यिकाएँ।}
\begin{enumerate}[resume]
  \item मान लो $O$ किसी \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} का केंद्र है। समझाओ कि
        $O$ से होकर जाती कोई भी रेखा सीमा से जिन दो बिंदुओं पर
        मिलती है वे $O$ के बारे में \omterm{def:g7:central:def}{सममित} क्यों होते हैं, और वह
        रेखा \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} को ऐसे दो टुकड़ों में क्यों बाँटती
        है जो एक-दूसरे की ठीक आधे चक्कर वाली नक़लें हैं --- यानी
        बराबर \omterm{def:g6:measure:area}{क्षेत्रफल} के।
  \item हलवाई का प्रमेय: आयताकार केक उसके केंद्र से होकर जाती
        \emph{किसी भी} सीधी कटाई से बिलकुल बराबर बँट जाता है।
        और आगे: मान लो आयताकार केक में एक आयताकार छेद है (किसी
        भी नाप का, कहीं भी, चाहे तिरछा ही)। वह एक सीधी कटाई बताओ
        जिससे दोनों हिस्सों में केक बराबर आए --- और सवाल 6 से
        उसका कारण दो।
  \item किसी \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} $ABCD$ के तीन \omterm{def:g5:solids:def}{शीर्ष} $A(0, 0)$,
        $B(6, 1)$ और $C(8, 5)$ हैं। $D$ और केंद्र $O$ के
        निर्देशांक निकालो।
  \item \Cref{exo:g7:central:9} ने त्रिभुज $ABC$ और $[BC]$ के
        मध्यबिंदु $O$ से \omterm{def:g7:central:parallelogram}{समांतर चतुर्भुज} $ABA'C$ बनाया था, जहाँ
        $A'$, $O$ के बारे में $A$ का \omterm{def:g7:central:def}{सममित} बिंदु है। उसे और
        त्रिभुज $ABA'$ में त्रिभुज असमिका
        (\cref{thm:g7:triangles:inequality}) को मिलाकर
        \emph{माध्यिका की सीमा} सिद्ध करो: माध्यिका $[AO]$ के
        लिए
        \[
        AO < \frac{AB + AC}{2} .
        \]
        (ध्यान दो कि $AA' = 2\,AO$ और $BA' = AC$।)
  \item जिस त्रिभुज में $AB = 5$ cm और $AC = 7$ cm है, उसमें $A$
        से खींची माध्यिका की लंबाई किन दो मानों के बीच होनी
        चाहिए? ($ABA'$ में त्रिभुज असमिका दोनों तरफ़ से लगाओ।)
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- किसी आकृति के कितने केंद्र हो सकते
हैं?}
\begin{enumerate}[resume]
  \item प्रत्येक आकृति के बारे में बताओ कि उसका \omterm{def:g7:central:center}{सममिति केंद्र} है
        या नहीं, और कहाँ है: एक \omterm{def:g6:lines:objects}{रेखाखंड}; एक पूरी रेखा; एक \omterm{def:g6:shapes:triangles}{समबाहु
        त्रिभुज}; एक वर्ग; एक वृत्त; और अक्षर S तथा Z।
  \item \omterm{def:g6:shapes:triangles}{समबाहु त्रिभुज} वाला अपना दावा सिद्ध करो: त्रिभुज को उसी
        पर भेजने वाला \omterm{def:g3:division:half}{आधा} चक्कर तीनों \omterm{def:g5:solids:def}{शीर्षों} के जोड़े बनाएगा ---
        पर तीन विषम है, इसलिए कोई एक \omterm{def:g5:solids:def}{शीर्ष} अपने ही ऊपर भेजा
        जाएगा। इससे उस \omterm{def:g5:solids:def}{शीर्ष} को क्या होना पड़ेगा, और वह असंभव
        क्यों है?
  \item सवाल 12 को आम बनाओ: समझाओ कि \omterm{def:g3:numbers:evenodd}{विषम संख्या} में \omterm{def:g5:solids:def}{शीर्षों}
        वाले \emph{किसी भी} \omterm{def:g4:geometry:polygon}{बहुभुज} का \omterm{def:g7:central:center}{सममिति केंद्र} क्यों नहीं
        होता।
  \item वह मोती: मान लो किसी आकृति के \emph{दो} अलग-अलग \omterm{def:g7:central:center}{सममिति
        केंद्र} $O_1$ और $O_2$ हैं। एक के बाद एक दोनों आधे चक्कर
        करो और सवाल 5 वाली खोज इस्तेमाल करो: कौन-सी गति आकृति को
        ठीक उसी पर भेजेगी? समझाओ कि काग़ज़ के एक पन्ने पर समा
        जाने वाली आकृति इससे क्यों नहीं बच सकती, और नतीजा
        निकालो।
  \item सवाल 14 का नतीजा एक झिरी छोड़ जाता है: \emph{बिना सीमा
        वाली} आकृतियाँ। किसी अनंत पट्टी-नमूने पर जाँचो --- जैसे
        पैरों के निशानों की कभी न ख़त्म होने वाली क़तार: बाएँ,
        दाएँ, बाएँ, दाएँ\,\dots\ --- कि उसके (कम से कम) दो
        अलग-अलग \omterm{def:g7:central:center}{सममिति केंद्र} हैं, और वह स्थानांतरण पहचानो
        जिसकी भविष्यवाणी सवाल 14 ने की थी।
\end{enumerate}
\end{problem}
