\chapter{La resta: primeros pasos}\label{ch:g1:subtraction}

Siete globos y dos salen volando: ¿cuántos quedan? Quitar es
\emph{restar}, y se escribe con el signo $-$. Además, la resta tiene una
segunda cara: también responde a «¿cuántos más?».

\section{Quitar}

\begin{definition}[La resta]\label{def:g1:subtraction:def}
\emph{Restar} es quitar y contar lo que queda. Escribimos
\[
7 - 2 = 5
\]
y se lee: «siete menos dos es igual a cinco».
\end{definition}

\begin{example}[Contar hacia atrás]\label{ex:g1:subtraction:countback}
Cuando se quita poco, cuenta hacia atrás en la recta: para $11 - 3$, di
«once» y da tres pasos atrás --- «diez, nueve, ocho». Así,
$11 - 3 = 8$.
\end{example}

\section{¿Cuántos más?}

\begin{example}[La diferencia]\label{ex:g1:subtraction:difference}
«Nina tiene $9$ canicas y Tomás tiene $6$. ¿Cuántas tiene Nina de más?»
No se quita nada y, sin embargo, es una resta: empareja las canicas
y cuenta las de Nina que se quedan sin pareja:
\[
9 - 6 = 3 .
\]
Nina tiene $3$ más. El resultado de una resta se llama
\emph{diferencia}\index{diferencia}.
\end{example}

\begin{method}[Contar hacia delante]\label{met:g1:subtraction:countup}
Para hallar una \omterm{ex:g1:subtraction:difference}{diferencia}, lo más cómodo suele ser contar hacia \emph{delante}:
para $12 - 9$, sal del $9$ y sube hasta el $12$: «diez, once,
doce» --- tres pasos. Así, $12 - 9 = 3$. (Sube pasando por el $10$ cuando
el hueco es mayor: para $14 - 8$, del $8$ al $10$ hay $2$ y del $10$ al $14$
hay $4$: \omterm{ex:g1:subtraction:difference}{diferencia} $6$.)
\end{method}

\section{La suma y la resta son gemelas}

\begin{example}[Familias de operaciones]\label{ex:g1:subtraction:family}
Los números $3$, $4$ y $7$ forman una pequeña familia:
\[
3 + 4 = 7, \qquad
4 + 3 = 7, \qquad
7 - 4 = 3, \qquad
7 - 3 = 4 .
\]
Si sabes una operación de la familia, las otras tres salen gratis.
La resta deshace la \omterm{def:g1:addition:def}{suma}: después de $+4$, hacer $-4$ te devuelve al principio.
\end{example}

\begin{example}[Comprobar]\label{ex:g1:subtraction:check}
¿Has calculado $13 - 5 = 8$? Compruébalo con la \omterm{def:g1:addition:def}{suma} gemela:
$8 + 5 = 13$. \checkmark\ Una resta es correcta justo cuando la \omterm{def:g1:addition:def}{suma}
de vuelta funciona.
\end{example}

\begin{example}[¿Qué operación?]\label{ex:g1:subtraction:choose}
«$6$ pájaros en el cable y $4$ salen volando»: se quita, $6 - 4 = 2$.
«$6$ pájaros y llegan $4$ más»: se junta, $6 + 4 = 10$.
«$6$ gorriones y $4$ petirrojos: ¿cuántos gorriones hay de más?»:
\omterm{ex:g1:subtraction:difference}{diferencia}, $6 - 4 = 2$. ¡Dibuja la historia si dudas!
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:1}
Dibuja, tacha y calcula: $6 - 2$; \quad $8 - 5$; \quad
$10 - 4$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:2}
Calcula contando hacia atrás: $9 - 3$; \quad $12 - 2$; \quad
$15 - 4$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:3}
Cuenta hacia delante para hallar la \omterm{ex:g1:subtraction:difference}{diferencia}: $11 - 8$; \quad $13 - 9$; \quad
$16 - 12$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:4}
Escribe las cuatro operaciones de la familia del $2$, el $6$ y el $8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:5}
Calcula y comprueba con la \omterm{def:g1:addition:def}{suma} gemela: $14 - 6$; \quad
$17 - 9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:6}
Completa los huecos:
\[
10 - \;?\; = 7, \qquad
\;?\; - 5 = 5, \qquad
12 - \;?\; = 12 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:7}
«En el bol hay $15$ cerezas. Los niños se comen $8$. ¿Cuántas
quedan?» Escribe la resta y la frase de respuesta.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:8}
Para cada historia, elige $+$ o $-$ y calcula: «$9$ patos en el
estanque y $5$ se van nadando»; «$9$ patos y llegan $5$ ocas»;
«Ali tiene $9$ años y su hermana $5$: ¿cuántos años es Ali mayor?».
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:subtraction:9}
Mamá pata tenía $12$ huevos. Han salido algunos pollitos; ahora quedan
$12 - ?$ huevos y ella cuenta $7$ huevos sin abrir. ¿De cuántos huevos
han salido pollitos? (¿De qué familia de operaciones se trata?)
\end{exercise}
