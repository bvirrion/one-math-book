\chapter{Figuras y espacio}\label{ch:g1:shapes}

Las ventanas son rectángulos, las ruedas son círculos, los tejados esconden
triángulos: las figuras están por todas partes. En este capítulo aprendemos a
reconocerlas contando lados y esquinas, y a decir con precisión \emph{dónde}
están las cosas: a la izquierda, a la derecha, encima, debajo, entre.

\section{Las cuatro figuras básicas}

\begin{definition}[Cuadrado, rectángulo, triángulo, círculo]\label{def:g1:shapes:def}
\begin{itemize}
  \item el \emph{triángulo}\index{triángulo} tiene $3$ lados y $3$ esquinas;
  \item el \emph{cuadrado}\index{cuadrado} tiene $4$ lados iguales y $4$ esquinas;
  \item el \emph{rectángulo}\index{rectángulo} tiene $4$ lados y $4$ esquinas, con
        los lados opuestos iguales (un cuadrado estirado);
  \item el \emph{círculo}\index{círculo} es perfectamente redondo: ni lados ni esquinas.
\end{itemize}
\end{definition}

\begin{example}[Una figura girada sigue siendo la misma]\label{ex:g1:shapes:tilted}
Un cuadrado apoyado sobre una esquina sigue siendo un cuadrado --- cuatro
lados iguales, cuatro esquinas --- aunque parezca una cometa. Girar una
figura no cambia lo que es.
\end{example}

\begin{example}[Caza de figuras]\label{ex:g1:shapes:hunt}
En clase: la pizarra es un rectángulo, el reloj un círculo, una
servilleta doblada puede ser un triángulo y algunas ventanas son cuadrados. Los
objetos reales no son figuras perfectas, pero sus caras se les parecen mucho
(los cuerpos y sus caras vuelven en el \cref{ch:g2:shapes}).
\end{example}

\section{¿Dónde está?}

\begin{definition}[Palabras de posición]\label{def:g1:shapes:position}
Para decir dónde está algo: \emph{izquierda} / \emph{derecha},
\emph{encima} / \emph{debajo}, \emph{sobre} / \emph{bajo},
\emph{delante} / \emph{detrás}, \emph{entre},
\emph{dentro} / \emph{fuera}.
\end{definition}

\section{Caminos en la cuadrícula}

\begin{method}[Describir un camino]\label{met:g1:shapes:path}
En papel cuadriculado, un camino es una lista de pasos:
$\to$ (una casilla a la derecha), $\leftarrow$ (a la izquierda), $\uparrow$ (arriba),
$\downarrow$ (abajo).
\begin{enumerate}
  \item para seguir un camino: sitúate en la casilla marcada y obedece
        las flechas una a una;
  \item para describir un camino: recórrelo mentalmente y escribe
        las flechas en orden;
  \item dos caminos distintos pueden unir las mismas dos casillas --- unos
        más cortos y otros más largos.
\end{enumerate}
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g1:shapes:1}
Para cada figura, cuenta los lados y las esquinas y dile su nombre: una figura
con $3$ esquinas; una figura redonda sin esquinas; una figura con $4$
lados iguales.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:2}
Busca en casa: dos rectángulos, un círculo, un triángulo y un cuadrado.
(¿Puertas? ¿platos? ¿señales de tráfico?)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:3}
¿Verdadero o falso? «Un cuadrado girado sobre una esquina deja de ser un
cuadrado.» Explícalo con el \cref{ex:g1:shapes:tilted}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:4}
Dibuja en papel cuadriculado: un cuadrado de $3$ casillas por $3$; un rectángulo de
$5$ por $2$; un triángulo que use una diagonal de la cuadrícula.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:5}
Pon tres juguetes en fila. Describe la fila con palabras de posición:
¿cuál está entre los otros dos? ¿Cuál está a la izquierda?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:6}
Dibuja un círculo. Dibuja una cruz \emph{dentro}, una estrella
\emph{fuera} y un punto \emph{sobre} el propio círculo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:7}
En papel cuadriculado, marca una casilla de salida. Sigue:
$\to\ \uparrow\ \to\ \uparrow\ \to$. Marca la llegada. Después escribe
el camino que vuelve directamente a la salida.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:8}
¿Cuántas esquinas hay en total: dos triángulos y un cuadrado? ¿Y un círculo
y dos rectángulos?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:shapes:9}
Coge $12$ cerillas. ¿Puedes formar el contorno de un cuadrado usándolas
\emph{todas}, con cerillas enteras y el mismo número en cada lado? ¿Cuántas
cerillas por lado? ¿Podrías hacerlo con $10$ cerillas?
\end{exercise}
