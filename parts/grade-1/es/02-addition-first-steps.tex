\chapter{Suma: primeros pasos}\label{ch:g1:addition}

Tres canicas rojas y dos azules: ¿cuántas hay en total? Juntar y contar
es \emph{sumar} --- la primera operación, con sus dos signos famosos
$+$ y $=$.

\section{Juntar}


\begin{definition}[Suma]\label{def:g1:addition:def}
\emph{Sumar} es juntar y contar todo. Escribimos
\[
3 + 2 = 5
\]
y leemos: ``tres más dos es igual a cinco''. El resultado de una suma
se llama la \emph{suma}\index{suma}.
\end{definition}

\begin{example}[El orden no importa]\label{ex:g1:addition:order}
$2 + 3$ y $3 + 2$ juntar las mismas canicas: ambas forman $5$.
para calcular $2 + 9$, es más rápido empezar desde el número mayor:
diga "nueve", luego cuente con dos --- "diez, once". Entonces
$2 + 9 = 11$.
\end{example}

\section{Los diez amigos}

\begin{example}[Complementos a 10]\label{ex:g1:addition:friends}
Dos números que hacen $10$ juntos son "diez amigos":
\[
1 + 9, \quad 2 + 8, \quad 3 + 7, \quad 4 + 6, \quad 5 + 5 .
\]
Diez dedos los muestran todos: levanta $3$ dedos, y el $7$ doblado
los unos son el amigo. Saberse de memoria a los diez amigos hace que muchos
adiciones rápidamente.
\end{example}

\end{tikzpicture}

{\small El cuadro de diez muestra cada par de diez amigos de un vistazo:

\begin{method}[Cruzando el diez]\label{met:g1:addition:crossten}
para calcular $8 + 5$:
\begin{enumerate}
  \item tomar de la $5$ que $8$ necesita alcanzar $10$: eso es
        $2$ (sus diez amigos);
  \item $8 + 2 = 10$, y $3$ restos de la $5$;
  \item $10 + 3 = 13$. Entonces $8 + 5 = 13$.
\end{enumerate}
pasando por $10$ se vuelve duro \omterm{def:g1:addition:def}{suma} en dos fáciles.
\end{method}

\begin{example}[Dobles]\label{ex:g1:addition:doubles}
Vale la pena aprenderse los dobles de memoria:
\[
1+1=2, \quad 2+2=4, \quad 3+3=6, \quad 4+4=8, \quad 5+5=10,
\]
\[
6+6=12, \quad 7+7=14, \quad 8+8=16, \quad 9+9=18, \quad 10+10=20 .
\]
Los casi dobles vienen gratis: $6 + 7$ es uno mas que $6 + 6$, entonces $13$.
\end{example}

\section{Pequeños problemas}

\begin{example}\label{ex:g1:addition:problem}
``Lea tiene $7$ pegatinas. ella consigue $6$ más. ¿Cuántos ahora?
Obtener más significa agregar: $7 + 6 = 13$ (hasta las diez:
$7 + 3 = 10$, entonces $+3$). Lea tiene $13$ pegatinas. Termina siempre con
una pequeña frase!
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star $]\label{exo:g1:addition:1}
Dibuja las canicas y calcula: $4 + 3$; \quad $5 + 2$; \quad
$6 + 4$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g1:addition:2}
Calcular partiendo del número mayor: $3 + 8$; \quad
$2 + 12$; \quad $4 + 9$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g1:addition:3}
Dale los diez amigo de: $7$; \ $4$; \ $9$; \ $5$; \ $1$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g1:addition:4}
Calcular hasta la decena (\cref{met:g1:addition:crossten}):
$9 + 4$; \quad $8 + 7$; \quad $6 + 5$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g1:addition:5}
Recita los dobles y luego calcula usándolos: $7 + 8$; \quad
$5 + 6$; \quad $9 + 8$.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g1:addition:6}
Completa los agujeros:
\[
6 + \;?\; = 10, \qquad
\;?\; + 5 = 12, \qquad
10 + \;?\; = 17 .
\]
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g1:addition:7}
``En el autobús hay $8$ niños; $5$ los adultos suben. ¿Cómo
¿Hay mucha gente en el autobús?'' Escribe la \omterm{def:g1:addition:def}{suma} y la respuesta.
frase.
\end{exercise}

\begin{exercise}[$\star $]\label{exo:g1:addition:8}
Tom tiene $6$ coches rojos y $7$ autos azules. Mía tiene $10$ coches. quien tiene
¿más autos?
\end{exercise}

\begin{exercise}[$\star\star $]\label{exo:g1:addition:9}
Espectáculo de tres dados $4$, $6$ y $5$. ¿Cuál es el total? (Agrega dos dados
primero --- ¡elige los dos que hacen diez amigos!)
\end{exercise}
