\chapter{Medir y medir el tiempo}\label{ch:g1:measure}

¿Qué lápiz es más largo? ¿Qué día es hoy? ¿Cuánto cuesta un caramelo?
En este capítulo comparamos longitudes y las medimos en centímetros,
ponemos en orden los días de la semana, leemos las horas en punto en el reloj
y contamos monedas pequeñas.

\section{Más largo y más corto}

\begin{method}[Comparar longitudes]\label{met:g1:measure:compare}
Para comparar dos longitudes, alinéalas \emph{por un mismo extremo}:
\begin{enumerate}
  \item pon los dos objetos uno al lado del otro, empezando en la misma línea;
  \item el que llega más lejos es el \emph{más largo} y el otro es el
        \emph{más corto};
  \item si los dos llegan al mismo sitio, miden \emph{lo mismo}.
\end{enumerate}
Hacer coincidir el punto de partida es todo el secreto --- comparar desde
salidas desiguales engaña a la vista.
\end{method}

\begin{definition}[El centímetro]\label{def:g1:measure:cm}
Para decir \emph{cuánto} mide, medimos con una regla en
\emph{centímetros} (cm). Pon el $0$ de la regla al principio del
objeto y lee el número donde termina.
\end{definition}

\section{Los días de la semana}

\begin{example}[Los siete días]\label{ex:g1:measure:days}
La semana tiene $7$ días, siempre en el mismo orden:
\[
\text{lunes, martes, miércoles, jueves, viernes, sábado,
domingo.}
\]
Después del domingo vuelve a empezar el lunes. \emph{Ayer} es el día
anterior y \emph{mañana} el día siguiente: si hoy es miércoles,
ayer fue martes y mañana será jueves.
\end{example}

\section{Las horas en punto en el reloj}

\begin{method}[Leer una hora en punto]\label{met:g1:measure:clock}
Cuando la aguja larga apunta justo hacia arriba, al $12$, es una hora
en punto: lee el número al que apunta la aguja \emph{corta}. Aguja corta en
el $3$ y aguja larga en el $12$: son las $3$ en punto.
\end{method}

\section{Monedas pequeñas}

\begin{example}[Formar una cantidad]\label{ex:g1:measure:coins}
Con monedas de $1$, $2$ y $5$ se pueden formar muchas cantidades de varias
maneras. Para formar $7$:
\[
5 + 2, \qquad
5 + 1 + 1, \qquad
2 + 2 + 2 + 1 .
\]
Empezar por las monedas más grandes suele ser lo más rápido y usa
menos monedas.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g1:measure:1}
Alinea tres lápices por un extremo. ¿Cuál es el más largo?
¿Y el más corto?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:2}
Mide con tu regla: el lado largo de este libro; tu dedo
meñique; una cuchara. Responde en centímetros enteros.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:3}
Di el día: después del lunes; antes del domingo; dos días después del viernes.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:4}
Hoy es jueves. ¿Qué día fue ayer? ¿Qué día será mañana?
¿Qué día será dentro de tres días?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:5}
Dibuja un reloj que marque las $6$ en punto. ¿Dónde está la aguja corta? ¿Y
la aguja larga?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:6}
Lee la hora en punto: aguja corta en el $5$ y larga en el $12$; aguja
corta en el $11$ y larga en el $12$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:7}
Forma $8$ con monedas de $1$, $2$ y $5$ de dos maneras distintas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:8}
Leo tiene una moneda de $5$ y tres monedas de $2$. ¿Cuánto dinero tiene
en total?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:measure:9}
Un caramelo cuesta $3$. Zoe paga con una moneda de $5$. ¿Cuánto le
devuelven? (Cuenta hacia delante del $3$ al $5$.)
\end{exercise}
