\chapter{Optellen: eerste stappen}\label{ch:g1:addition}

Drie rode knikkers en twee blauwe knikkers: hoeveel zijn dat samen?
Samenvoegen en het geheel tellen, dat is \emph{optellen} --- de eerste
bewerking, met haar twee beroemde tekens $+$ en $=$.

\section{Samenvoegen}

\begin{definition}[Optellen]\label{def:g1:addition:def}
\emph{Optellen} is samenvoegen en alles tellen. We schrijven
\[
3 + 2 = 5
\]
en lezen: ``drie plus twee is vijf''. De uitkomst van een optelling
heet de \emph{som}\index{som}.
\end{definition}

  \node[font=\large] at (3.0,0) {$+$};
  \foreach \i in {0,1} \fill[omDef] (3.8+\i*1.0,0) circle (0.3);
  \node[font=\large] at (5.8,0) {$=$};
  \foreach \i in {0,1,2} \fill[omThm] (6.6+\i*1.0,0) circle (0.3);

  \node[below, font=\small] at (5.6,-0.7)
    {$3 + 2 = 5$: drie en twee maken vijf};
\end{tikzpicture}

{\small Een optelling in plaatjes: de twee groepjes samen vormen één
groepje van vijf.}
\end{omfigure}

\begin{example}[De volgorde maakt niets uit]\label{ex:g1:addition:order}
$2 + 3$ en $3 + 2$ voegen dezelfde knikkers samen: beide maken $5$. Om
$2 + 9$ uit te rekenen, begin je sneller bij het grootste getal: zeg
``negen'' en tel er twee bij --- ``tien, elf''. Dus $2 + 9 = 11$.
\end{example}

\section{De vriendjes van tien}

\begin{example}[Aanvullen tot 10]\label{ex:g1:addition:friends}
Twee getallen die samen $10$ maken, zijn ``vriendjes van tien'':
\[
1 + 9, \quad 2 + 8, \quad 3 + 7, \quad 4 + 6, \quad 5 + 5 .
\]
Tien vingers laten ze allemaal zien: steek $3$ vingers op, en de $7$
gebogen vingers zijn het vriendje. Wie de vriendjes van tien uit het
hoofd kent, telt veel sommen razendsnel op.
\end{example}

  \draw[thick] (0,0) grid[step=1] (5,2);
  \draw[very thick] (0,0) rectangle (5,2);
  \foreach \i in {0,1,2} \fill[omDef] (\i+0.5,1.5) circle (0.3);
  \foreach \i in {3,4} \draw[omThm, thick] (\i+0.5,1.5) circle (0.3);
  \foreach \i in {0,...,4} \draw[omThm, thick] (\i+0.5,0.5) circle (0.3);
  \node[right, font=\small, align=left] at (5.6,1.0)
    {een \emph{tienraam}: $10$ vakjes.\\
     $3$ vol, $7$ leeg: $3 + 7 = 10$};
\end{tikzpicture}

{\small Het tienraam laat elk paar vriendjes van tien in één oogopslag
zien: wat gevuld is en wat leeg blijft, maken samen altijd $10$.}
\end{omfigure}

\begin{method}[Over het tiental heen]\label{met:g1:addition:crossten}
Om $8 + 5$ uit te rekenen:
\begin{enumerate}
  \item haal uit de $5$ wat de $8$ nodig heeft om $10$ te worden: dat
        is $2$ (zijn vriendje van tien);
  \item $8 + 2 = 10$, en van de $5$ blijft $3$ over;
  \item $10 + 3 = 13$. Dus $8 + 5 = 13$.
\end{enumerate}
Via $10$ rekenen maakt van één moeilijke \omterm{def:g1:addition:def}{som} twee makkelijke.
\end{method}

\begin{example}[Dubbels]\label{ex:g1:addition:doubles}
De dubbels zijn het waard om uit het hoofd te leren:
\[
1+1=2, \quad 2+2=4, \quad 3+3=6, \quad 4+4=8, \quad 5+5=10,
\]
\[
6+6=12, \quad 7+7=14, \quad 8+8=16, \quad 9+9=18, \quad 10+10=20 .
\]
De bijna-dubbels krijg je er gratis bij: $6 + 7$ is één meer dan
$6 + 6$, dus $13$.
\end{example}

\section{Kleine verhaaltjes}

\begin{example}\label{ex:g1:addition:problem}
``Lea heeft $7$ stickers. Ze krijgt er $6$ bij. Hoeveel heeft ze nu?''
Erbij krijgen is optellen: $7 + 6 = 13$ (via tien: $7 + 3 = 10$, dan
$+3$). Lea heeft $13$ stickers. Sluit altijd af met een antwoordzin!
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g1:addition:1}
Teken de knikkers en reken uit: $4 + 3$; \quad $5 + 2$; \quad
$6 + 4$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:2}
Reken uit en begin bij het grootste getal: $3 + 8$; \quad
$2 + 12$; \quad $4 + 9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:3}
Geef het vriendje van tien van: $7$; \ $4$; \ $9$; \ $5$; \ $1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:4}
Reken over het tiental heen (\cref{met:g1:addition:crossten}):
$9 + 4$; \quad $8 + 7$; \quad $6 + 5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:5}
Zeg de dubbels op en reken er dan mee: $7 + 8$; \quad
$5 + 6$; \quad $9 + 8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:6}
Vul de gaten in:
\[
6 + \;?\; = 10, \qquad
\;?\; + 5 = 12, \qquad
10 + \;?\; = 17 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:7}
``In de bus zitten $8$ kinderen; er stappen $5$ volwassenen in.
Hoeveel mensen zitten er in de bus?'' Schrijf de optelling en de
antwoordzin.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:8}
Tom heeft $6$ rode auto's en $7$ blauwe auto's. Mia heeft $10$ auto's.
Wie heeft er meer?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:addition:9}
Drie dobbelstenen laten $4$, $6$ en $5$ zien. Wat is het totaal? (Tel
eerst twee dobbelstenen op --- kies de twee die vriendjes van tien
zijn!)
\end{exercise}
