\chapter{Getallen tot 100}\label{ch:g1:tens}

Een grote stapel één voor één tellen duurt eindeloos. De grote truc:
bundel de voorwerpen in pakjes van \emph{tien} en tel de pakjes. Zo
zijn de getallen tot $100$ opgebouwd --- en daarom schrijf je ze met
twee cijfers.

\section{Bundels van tien}

\begin{definition}[Tientallen en eenheden]\label{def:g1:tens:def}
Een \emph{tiental} is een bundel van tien eenheden. Het getal $34$
betekent $3$ tientallen en $4$ eenheden:
\[
34 = 10 + 10 + 10 + 4 .
\]
Het linkercijfer telt de bundels, het rechtercijfer de losse eenheden.
\end{definition}

      ({0.16*\i},0) -- ({0.16*\i},1.7);
    \draw[omExo, thick, rounded corners=3pt]
      (-0.22,0.62) rectangle (1.66,1.05);
    \end{scope}
  }
  % vier losse stokjes
  \foreach \i in {0,...,3} \draw[omThm, very thick]
    (7.7+0.4*\i,0) -- (7.7+0.4*\i,1.7);
  \node[right, font=\small] at (9.7,0.85)
    {$3$ bundels en $4$ stokjes: $34$};
\end{tikzpicture}

{\small Vierendertig stokjes: drie bundels van tien (elk met een
elastiekje) en vier losse stokjes. Je hoeft niet één voor één tot $34$
te tellen!}
\end{omfigure}

\begin{example}[De getallen lezen]\label{ex:g1:tens:reading}
$20$ is twee tientallen (twintig), $30$ drie tientallen
(dertig)\,\dots\ tot $90$ (negentig) en $100$ (honderd: tien
tientallen!). Daartussenin: $47$ lees je ``zevenenveertig'' --- in het
Nederlands noem je eerst de $7$ eenheden en dan de $4$ tientallen,
precies omgekeerd aan de schrijfwijze.
\end{example}

{\small De plaatswaardetabel van $34$: de $3$ staat op de plaats van de

jaar een kolom erbij!}
\end{omfigure}

\begin{example}[Dezelfde cijfers, andere getallen]\label{ex:g1:tens:place}
$27$ en $72$ gebruiken dezelfde cijfers, maar zijn heel verschillend:
$27$ is $2$ tientallen en $7$ eenheden; $72$ is $7$ tientallen en $2$
eenheden --- veel groter! De \emph{plaats} van een cijfer bepaalt wat
het telt.
\end{example}

\section{Het honderdveld}

\begin{omfigure}
\begin{tikzpicture}[scale=0.52]
  \foreach \r in {0,...,9} {
    \foreach \c in {0,...,9} {
      \pgfmathtruncatemacro{\n}{\r*10+\c+1}
      \draw[gray!50] (\c,-\r) +(-0.5,-0.5) rectangle +(0.5,0.5);
      \node[font=\footnotesize] at (\c,-\r) {\n};
    }
  }
  \draw[omThm, very thick] (3,-4) +(-0.5,-0.5) rectangle +(0.5,0.5);
  \draw[omDef, very thick] (3,-5) +(-0.5,-0.5) rectangle +(0.5,0.5);
\end{tikzpicture}

{\small Het honderdveld: elke rij is één tiental. Eén stap naar rechts
is $+1$; één stap naar \emph{beneden} is $+10$ (van de rode $44$
recht naar beneden naar de blauwe $54$).}
\end{omfigure}

\begin{method}[Bewegen op het veld]\label{met:g1:tens:grid}
\begin{enumerate}
  \item $+1$ / $-1$: één stap naar rechts / naar links;
  \item $+10$ / $-10$: één stap naar beneden / naar boven --- alleen
        het cijfer van de tientallen verandert: $37 + 10 = 47$;
  \item tellen met tientallen: $10, 20, 30, \dots$ --- volg de laatste
        kolom van het veld recht naar beneden.
\end{enumerate}
\end{method}

\begin{example}[Tellen met 2 en met 5]\label{ex:g1:tens:skip}
Met $2$: $2, 4, 6, 8, 10, 12, \dots$ (de even getallen). Met $5$:
$5, 10, 15, 20, 25, \dots$ --- ze eindigen op $5$ of $0$ en vormen een
mooi patroon op het veld. Zo tellen bereidt het vermenigvuldigen voor
(\cref{ch:g2:mult}).
\end{example}

\section{Grotere getallen vergelijken}

\begin{method}[Vergelijken met tientallen]\label{met:g1:tens:compare}
Wie meer tientallen heeft, wint: $61 > 58$, want $6$ tientallen zijn
meer dan $5$ tientallen --- ook al is $8$ meer dan $1$! Alleen als de
tientallen gelijk zijn, beslissen de eenheden: $43 < 47$.
\end{method}

\begin{example}\label{ex:g1:tens:compare}
Zet $35$, $53$, $29$ op volgorde: eerst de tientallen ($2$, $3$, $5$):
\[
29 < 35 < 53 .
\]
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g1:tens:1}
Hoeveel tientallen en eenheden zitten er in: $26$; \ $40$; \ $73$;
\ $99$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:2}
Schrijf het getal: $5$ tientallen en $2$ eenheden; \ $8$ tientallen;
\ $1$ tiental en $9$ eenheden; \ $10$ tientallen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:3}
Schrijf in cijfers: drieënzestig; \ veertig; \ vijfentachtig; \
honderd.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:4}
Reken uit door je op het honderdveld te verplaatsen: $23 + 10$; \quad
$67 + 10$; \quad $45 - 10$; \quad $80 - 10$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:5}
Tel verder: $10, 20, 30, ?, ?, ?$. \quad
$5, 10, 15, ?, ?, ?$. \quad $12, 14, 16, ?, ?, ?$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:6}
Neem over en vul in met $<$ of $>$:
\[
34 \;?\; 43, \qquad
58 \;?\; 61, \qquad
70 \;?\; 69, \qquad
99 \;?\; 100 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:7}
Zet op volgorde, van klein naar groot: $46$; \ $19$; \ $64$; \ $91$;
\ $41$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:8}
Een tuinier bundelt $52$ bloemen in bosjes van tien. Hoeveel volle
bosjes krijgt hij, en hoeveel losse bloemen blijven over?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:tens:9}
Begin op het honderdveld bij $36$; ga twee stappen naar beneden en één
stap naar links. Waar kom je uit? Welke berekening heb je zo gemaakt?
\end{exercise}
