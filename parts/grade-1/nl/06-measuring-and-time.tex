\chapter{Meten en tijd}\label{ch:g1:measure}

Welk potlood is langer? Welke dag is het? Hoeveel kost een snoepje? In
dit hoofdstuk vergelijken we lengtes en meten we ze in centimeters,
zetten we de dagen van de week op volgorde, lezen we hele uren op de
klok en tellen we kleine muntjes.

\section{Langer en korter}

\begin{method}[Lengtes vergelijken]\label{met:g1:measure:compare}
Om twee lengtes te vergelijken, leg je ze naast elkaar met \emph{één
uiteinde gelijk}:
\begin{enumerate}
  \item leg beide voorwerpen naast elkaar, met hetzelfde beginpunt;
  \item wat verder komt, is \emph{langer}, het andere is
        \emph{korter};
  \item komen ze even ver, dan zijn ze \emph{even lang}.
\end{enumerate}
Gelijk beginnen is het hele geheim --- vergelijken vanaf scheve
beginpunten bedriegt het oog.
\end{method}

\begin{definition}[De centimeter]\label{def:g1:measure:cm}
Om te zeggen \emph{hoe} lang iets is, meten we met een lat in
\emph{centimeters} (cm). Leg de $0$ van de lat bij het begin van het
voorwerp en lees het getal aan het einde af.
\end{definition}

\section{De dagen van de week}

\begin{example}[De zeven dagen]\label{ex:g1:measure:days}
De week heeft $7$ dagen, altijd in dezelfde volgorde:
\[
\text{maandag, dinsdag, woensdag, donderdag, vrijdag, zaterdag,
zondag.}
\]
Na zondag begint het weer bij maandag. \emph{Gisteren} is de dag
ervoor, \emph{morgen} de dag erna: als het vandaag woensdag is, was het
gisteren dinsdag en is het morgen donderdag.
\end{example}

\section{Hele uren op de klok}

\begin{method}[Een heel uur lezen]\label{met:g1:measure:clock}
Als de lange wijzer recht naar boven wijst, naar de $12$, dan is het een
heel uur: lees het getal waar de \emph{korte} wijzer naar wijst. Korte
wijzer op de $3$, lange wijzer op de $12$: het is $3$ uur.
\end{method}

\section{Kleine muntjes}

\begin{example}[Een bedrag maken]\label{ex:g1:measure:coins}
Met muntjes van $1$, $2$ en $5$ kun je veel bedragen op verschillende
manieren maken. Om $7$ te maken:
\[
5 + 2, \qquad
5 + 1 + 1, \qquad
2 + 2 + 2 + 1 .
\]
Met de grootste muntjes beginnen gaat meestal het snelst, en kost de
minste muntjes.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g1:measure:1}
Leg drie potloden naast elkaar met één uiteinde gelijk. Welk potlood is
het langst? Welk is het kortst?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:2}
Meet met je lat: de lange zijde van dit boek; je pink; een lepel.
Antwoord in hele centimeters.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:3}
Noem de dag: na maandag; voor zondag; twee dagen na vrijdag.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:4}
Vandaag is het donderdag. Welke dag was het gisteren? Welke dag is het
morgen? Welke dag is het over drie dagen?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:5}
Teken een klok die $6$ uur aanwijst. Waar staat de korte wijzer? Waar
staat de lange wijzer?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:6}
Lees het hele uur: korte wijzer op $5$, lange wijzer op $12$; korte
wijzer op $11$, lange wijzer op $12$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:7}
Maak $8$ met muntjes van $1$, $2$ en $5$, op twee verschillende
manieren.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:8}
Leo heeft een muntje van $5$ en drie muntjes van $2$. Hoeveel geld
heeft hij samen?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:measure:9}
Een snoepje kost $3$. Zoë betaalt met een muntje van $5$. Hoeveel
wisselgeld krijgt ze terug? (Tel door van $3$ naar $5$.)
\end{exercise}
