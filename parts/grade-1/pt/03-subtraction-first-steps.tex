\chapter{Subtração: primeiros passos}\label{ch:g1:subtraction}

Sete balões, dois voam para longe: quantos ficam? Tirar é
\emph{subtrair}, e escreve-se com o sinal $-$. E a subtração tem uma
segunda cara: ela também responde ``quantos a mais?''.

\section{Tirar}

\begin{definition}[Subtração]\label{def:g1:subtraction:def}
\emph{Subtrair} é tirar e contar o que ficou. Escrevemos
\[
7 - 2 = 5
\]
e lemos: ``sete menos dois é igual a cinco''.
\end{definition}

\begin{example}[Contar para trás]\label{ex:g1:subtraction:countback}
Para tirar pouco, conte para trás na reta: $11 - 3$: diga ``onze'' e
dê três passos para trás --- ``dez, nove, oito''. Logo
$11 - 3 = 8$.
\end{example}

\section{Quantos a mais?}

\begin{example}[A diferença]\label{ex:g1:subtraction:difference}
``Nina tem $9$ bolinhas e Tomás tem $6$. Quantas Nina tem a mais?''
Nada foi tirado, e ainda assim é uma subtração: forme pares com as
bolinhas e conte as de Nina que ficaram sem par:
\[
9 - 6 = 3 .
\]
Nina tem $3$ a mais. O resultado de uma subtração chama-se
\emph{diferença}\index{diferença}.
\end{example}

\begin{method}[Contar para cima]\label{met:g1:subtraction:countup}
Para achar uma \omterm{ex:g1:subtraction:difference}{diferença}, contar \emph{para cima} costuma ser o mais
fácil: em $12 - 9$, comece no $9$ e suba até o $12$: ``dez, onze,
doze'' --- três passos. Logo $12 - 9 = 3$. (Suba passando pelo $10$
quando a distância for maior: $14 - 8$: do $8$ ao $10$ são $2$, do
$10$ ao $14$ são $4$: \omterm{ex:g1:subtraction:difference}{diferença} $6$.)
\end{method}

\section{Adição e subtração são gêmeas}

\begin{example}[Famílias de fatos]\label{ex:g1:subtraction:family}
Os números $3$, $4$ e $7$ formam uma pequena família:
\[
3 + 4 = 7, \qquad
4 + 3 = 7, \qquad
7 - 4 = 3, \qquad
7 - 3 = 4 .
\]
Saber um fato da família dá os outros três de graça. A subtração
desfaz a adição: depois de $+4$, fazer $-4$ traz você de volta.
\end{example}

\begin{example}[Conferir]\label{ex:g1:subtraction:check}
Calculou $13 - 5 = 8$? Confira com a adição gêmea: $8 + 5 = 13$.
\checkmark\ Uma subtração está certa exatamente quando somar de
volta funciona.
\end{example}

\begin{example}[Qual operação?]\label{ex:g1:subtraction:choose}
``$6$ passarinhos no fio, $4$ voam para longe'': tirar,
$6 - 4 = 2$. ``$6$ passarinhos e chegam mais $4$'': juntar,
$6 + 4 = 10$. ``$6$ pardais e $4$ sabiás: quantos pardais a mais?'':
\omterm{ex:g1:subtraction:difference}{diferença}, $6 - 4 = 2$. Na dúvida, desenhe a história!
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:1}
Desenhe, risque e calcule: $6 - 2$; \quad $8 - 5$; \quad
$10 - 4$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:2}
Conte para trás para calcular: $9 - 3$; \quad $12 - 2$; \quad
$15 - 4$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:3}
Conte para cima para achar a \omterm{ex:g1:subtraction:difference}{diferença}: $11 - 8$; \quad
$13 - 9$; \quad $16 - 12$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:4}
Escreva os quatro fatos da família de $2$, $6$ e $8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:5}
Calcule e depois confira com a adição gêmea: $14 - 6$; \quad
$17 - 9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:6}
Complete os buracos:
\[
10 - \;?\; = 7, \qquad
\;?\; - 5 = 5, \qquad
12 - \;?\; = 12 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:7}
``Há $15$ cerejas na tigela. As crianças comem $8$. Quantas ficam?''
Escreva a subtração e a frase de resposta.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:8}
Para cada história, escolha $+$ ou $-$ e depois calcule: ``$9$ patos
no lago, $5$ vão embora nadando''; ``$9$ patos, chegam $5$
gansos''; ``Ali tem $9$ anos, a irmã tem $5$: quantos anos Ali tem a
mais?''.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:subtraction:9}
A mamãe pata tinha $12$ ovos. Alguns chocaram; agora restam
$12 - ?$ ovos e ela conta $7$ ovos ainda fechados. Quantos ovos
chocaram? (De que família de fatos se trata?)
\end{exercise}
