\chapter{Medidas e tempo}\label{ch:g1:measure}

Qual lápis é mais comprido? Que dia é hoje? Quanto custa uma bala?
Neste capítulo comparamos comprimentos e os medimos em centímetros,
colocamos os dias da semana em ordem, lemos as horas cheias no
relógio e contamos moedinhas.

\section{Mais comprido e mais curto}

\begin{method}[Comparar comprimentos]\label{met:g1:measure:compare}
Para comparar dois comprimentos, alinhe-os \emph{por uma das
pontas}:
\begin{enumerate}
  \item ponha os dois objetos lado a lado, partindo da mesma linha;
  \item o que chegar mais longe é o \emph{mais comprido}; o outro é o
        \emph{mais curto};
  \item se os dois chegarem ao mesmo lugar, têm o \emph{mesmo
        comprimento}.
\end{enumerate}
Todo o segredo está em alinhar o começo --- comparar a partir de
pontas tortas engana o olho.
\end{method}

\begin{definition}[O centímetro]\label{def:g1:measure:cm}
Para dizer \emph{quanto} é comprido, medimos com uma régua em
\emph{centímetros} (cm). Coloque o $0$ da régua no começo do objeto
e leia o número no fim dele.
\end{definition}

\section{Os dias da semana}

\begin{example}[Os sete dias]\label{ex:g1:measure:days}
A semana tem $7$ dias, sempre na mesma ordem:
\[
\text{segunda, terça, quarta, quinta, sexta, sábado, domingo.}
\]
Depois de domingo, a segunda começa de novo. \emph{Ontem} é o dia de
antes, \emph{amanhã} é o dia de depois: se hoje é quarta, ontem foi
terça e amanhã é quinta.
\end{example}

\section{Horas cheias no relógio}

\begin{method}[Ler uma hora cheia]\label{met:g1:measure:clock}
Quando o ponteiro grande aponta direto para cima, para o $12$, é uma
hora cheia: leia o número para onde aponta o ponteiro
\emph{pequeno}. Ponteiro pequeno no $3$, ponteiro grande no $12$:
são $3$ horas.
\end{method}

\section{Moedinhas}

\begin{example}[Formar uma quantia]\label{ex:g1:measure:coins}
Com moedas de $1$, $2$ e $5$, muitas quantias podem ser formadas de
várias maneiras. Para formar $7$:
\[
5 + 2, \qquad
5 + 1 + 1, \qquad
2 + 2 + 2 + 1 .
\]
Começar pelas moedas maiores costuma ser o caminho mais rápido, e é
o que usa menos moedas.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g1:measure:1}
Alinhe três lápis por uma das pontas. Qual é o mais comprido? Qual é
o mais curto?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:2}
Meça com a sua régua: o lado maior deste livro; o seu dedo mindinho;
uma colher. Responda em centímetros inteiros.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:3}
Diga o dia: depois de segunda; antes de domingo; dois dias depois de
sexta.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:4}
Hoje é quinta. Que dia foi ontem? Que dia é amanhã? Que dia será
daqui a três dias?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:5}
Desenhe um relógio marcando $6$ horas. Onde fica o ponteiro pequeno?
E o ponteiro grande?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:6}
Leia a hora cheia: ponteiro pequeno no $5$, ponteiro grande no $12$;
ponteiro pequeno no $11$, ponteiro grande no $12$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:7}
Forme $8$ com moedas de $1$, $2$ e $5$, de duas maneiras diferentes.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:measure:8}
Léo tem uma moeda de $5$ e três moedas de $2$. Quanto dinheiro ele
tem ao todo?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:measure:9}
Uma bala custa $3$. Zoe paga com uma moeda de $5$. Quanto ela recebe
de troco? (Conte para cima, de $3$ até $5$.)
\end{exercise}
