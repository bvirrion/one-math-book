\chapter{Números até 100}\label{ch:g1:tens}

Contar uma pilha grande um por um demora uma eternidade. O grande
truque: agrupar os objetos em pacotes de \emph{dez} e contar os
pacotes. É assim que os números até $100$ são construídos --- e por
isso eles se escrevem com dois algarismos.

\section{Pacotes de dez}

\begin{definition}[Dezenas e unidades]\label{def:g1:tens:def}
Uma \emph{dezena} é um pacote de dez unidades. O número $34$ quer
dizer $3$ dezenas e $4$ unidades:
\[
34 = 10 + 10 + 10 + 4 .
\]
O algarismo da esquerda conta os pacotes; o da direita, as unidades
soltas.
\end{definition}

  \foreach \b in {0,1,2} {
    \begin{scope}[xshift=\b*2.4cm]
    \foreach \i in {0,...,9} \draw[omDef, very thick]
      ({0.16*\i},0) -- ({0.16*\i},1.7);
    \draw[omExo, thick, rounded corners=3pt]
      (-0.22,0.62) rectangle (1.66,1.05);
    \end{scope}
  }
  % quatro palitos soltos
  \foreach \i in {0,...,3} \draw[omThm, very thick]
    (7.7+0.4*\i,0) -- (7.7+0.4*\i,1.7);
  \node[right, font=\small] at (9.7,0.85)
    {$3$ pacotes e $4$ palitos: $34$};
\end{tikzpicture}

{\small Trinta e quatro palitos: três pacotes de dez (cada um com uma
tira) e quatro palitos soltos. Não é preciso contar até $34$ um por
um!}
\end{omfigure}

\begin{example}[Ler os números]\label{ex:g1:tens:reading}
$20$ são duas dezenas (vinte), $30$ são três dezenas
(trinta)\,\dots\ até $90$ (noventa) e $100$ (cem: dez dezenas!). No
meio: $47$ lê-se ``quarenta e sete'' --- quarenta pelas $4$ dezenas,
sete pelas unidades.
\end{example}

  \node[font=\small] at (1.1,1.2) {dezenas};
  \node[font=\small] at (3.3,1.2) {unidades};

  \node[font=\large, omThm] at (3.3,0.4) {$4$};
\end{tikzpicture}

{\small O quadro de posições do $34$: o $3$ fica na casa das dezenas
e o $4$ na casa das unidades. Esse quadrinho ganha uma coluna nova a
cada ano!}
\end{omfigure}

\begin{example}[Mesmos algarismos, números diferentes]\label{ex:g1:tens:place}
$27$ e $72$ usam os mesmos algarismos, mas são bem diferentes: $27$
são $2$ dezenas e $7$ unidades; $72$ são $7$ dezenas e $2$ unidades
--- muito maior! A \emph{casa} de um algarismo decide o que ele
conta.
\end{example}

\section{O quadro de cem}

\begin{omfigure}
\begin{tikzpicture}[scale=0.52]
  \foreach \r in {0,...,9} {
    \foreach \c in {0,...,9} {
      \pgfmathtruncatemacro{\n}{\r*10+\c+1}
      \draw[gray!50] (\c,-\r) +(-0.5,-0.5) rectangle +(0.5,0.5);
      \node[font=\footnotesize] at (\c,-\r) {\n};
    }
  }
  \draw[omThm, very thick] (3,-4) +(-0.5,-0.5) rectangle +(0.5,0.5);
  \draw[omDef, very thick] (3,-5) +(-0.5,-0.5) rectangle +(0.5,0.5);
\end{tikzpicture}

{\small O quadro de cem: cada linha é uma dezena. Um passo para a
direita é $+1$; um passo \emph{para baixo} é $+10$ (do $44$ vermelho
direto para o $54$ azul).}
\end{omfigure}

\begin{method}[Andar no quadro]\label{met:g1:tens:grid}
\begin{enumerate}
  \item $+1$ / $-1$: um passo para a direita / para a esquerda;
  \item $+10$ / $-10$: um passo para baixo / para cima --- só muda o
        algarismo das dezenas: $37 + 10 = 47$;
  \item contar de dez em dez: $10, 20, 30, \dots$ --- é descer pela
        última coluna do quadro.
\end{enumerate}
\end{method}

\begin{example}[Contar de 2 em 2 e de 5 em 5]\label{ex:g1:tens:skip}
De $2$ em $2$: $2, 4, 6, 8, 10, 12, \dots$ (os números pares). De
$5$ em $5$: $5, 10, 15, 20, 25, \dots$ --- terminam em $5$ ou $0$ e
formam um desenho bonito no quadro. Contar de tantos em tantos
prepara a multiplicação (\cref{ch:g2:mult}).
\end{example}

\section{Comparar números maiores}

\begin{method}[Comparar pelas dezenas]\label{met:g1:tens:compare}
Ganha quem tem mais dezenas: $61 > 58$ porque $6$ dezenas vencem $5$
dezenas --- mesmo que $8$ seja maior que $1$! Só quando as dezenas
são iguais é que as unidades decidem: $43 < 47$.
\end{method}

\begin{example}\label{ex:g1:tens:compare}
Ordene $35$, $53$, $29$: primeiro as dezenas ($2$, $3$, $5$):
\[
29 < 35 < 53 .
\]
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g1:tens:1}
Quantas dezenas e quantas unidades há em: $26$; \ $40$; \ $73$; \
$99$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:2}
Escreva o número: $5$ dezenas e $2$ unidades; \ $8$ dezenas; \ $1$
dezena e $9$ unidades; \ $10$ dezenas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:3}
Escreva em algarismos: sessenta e três; \ quarenta; \ oitenta e
cinco; \ cem.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:4}
Calcule andando no quadro de cem: $23 + 10$; \quad $67 + 10$;
\quad $45 - 10$; \quad $80 - 10$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:5}
Continue: $10, 20, 30, ?, ?, ?$. \quad
$5, 10, 15, ?, ?, ?$. \quad $12, 14, 16, ?, ?, ?$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:6}
Copie e complete com $<$ ou $>$:
\[
34 \;?\; 43, \qquad
58 \;?\; 61, \qquad
70 \;?\; 69, \qquad
99 \;?\; 100 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:7}
Ordene do menor para o maior: $46$; \ $19$; \ $64$; \ $91$; \
$41$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:8}
Um jardineiro amarra $52$ flores em ramos de dez. Quantos ramos
completos ele faz, e quantas flores ficam soltas?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:tens:9}
No quadro de cem, comece no $36$; desça dois passos e ande um passo
para a esquerda. Onde você chega? Que conta você acabou de fazer?
\end{exercise}
