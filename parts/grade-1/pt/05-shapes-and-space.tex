\chapter{Formas e espaço}\label{ch:g1:shapes}

Janelas são retângulos, rodas são círculos, telhados escondem
triângulos: as formas estão em toda parte. Neste capítulo aprendemos
a reconhecê-las contando lados e cantos, e a dizer com precisão
\emph{onde} as coisas estão: à esquerda, à direita, acima, abaixo,
entre.

\section{As quatro formas básicas}

\begin{definition}[Quadrado, retângulo, triângulo, círculo]\label{def:g1:shapes:def}
\begin{itemize}
  \item o \emph{triângulo}\index{triângulo} tem $3$ lados e $3$ cantos;
  \item o \emph{quadrado}\index{quadrado} tem $4$ lados iguais e $4$ cantos;
  \item o \emph{retângulo}\index{retângulo} tem $4$ lados e $4$ cantos, com os
        lados opostos iguais (um quadrado esticado);
  \item o \emph{círculo}\index{círculo} é perfeitamente redondo: sem lados, sem cantos.
\end{itemize}
\end{definition}

\begin{example}[Forma inclinada continua sendo forma]\label{ex:g1:shapes:tilted}
Um quadrado apoiado num canto continua sendo um quadrado --- quatro
lados iguais, quatro cantos --- mesmo que pareça uma pipa. Girar uma
forma não muda o que ela é.
\end{example}

\begin{example}[Caça às formas]\label{ex:g1:shapes:hunt}
Na sala de aula: o quadro é um retângulo, o relógio é um círculo, um
guardanapo dobrado pode ser um triângulo, algumas janelas são
quadradas. Os objetos de verdade não são formas perfeitas, mas as
suas faces chegam perto delas (os sólidos e as suas faces voltam no
\cref{ch:g2:shapes}).
\end{example}

\section{Onde está?}

\begin{definition}[Palavras de posição]\label{def:g1:shapes:position}
Para dizer onde algo está: \emph{à esquerda} / \emph{à direita},
\emph{acima} / \emph{abaixo}, \emph{sobre} / \emph{sob},
\emph{à frente de} / \emph{atrás de}, \emph{entre},
\emph{dentro} / \emph{fora}.
\end{definition}

\section{Caminhos na malha quadriculada}

\begin{method}[Descrever um caminho]\label{met:g1:shapes:path}
No papel quadriculado, um caminho é uma lista de passos:
$\to$ (um quadradinho para a direita), $\leftarrow$ (para a
esquerda), $\uparrow$ (para cima), $\downarrow$ (para baixo).
\begin{enumerate}
  \item Para seguir um caminho: comece no quadradinho marcado e
        obedeça às setas uma a uma;
  \item para descrever um caminho: percorra-o de cabeça e escreva as
        setas em ordem;
  \item dois caminhos diferentes podem ligar os mesmos dois
        quadradinhos --- uns mais curtos, outros mais longos.
\end{enumerate}
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g1:shapes:1}
Para cada forma, conte os lados e os cantos e dê o nome dela: uma
forma com $3$ cantos; uma forma redonda sem cantos; uma forma com
$4$ lados iguais.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:2}
Encontre em casa: dois retângulos, um círculo, um triângulo, um
quadrado. (Portas? pratos? placas de trânsito?)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:3}
Verdadeiro ou falso? ``Um quadrado apoiado num canto deixa de ser um
quadrado.'' Explique usando o \cref{ex:g1:shapes:tilted}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:4}
Desenhe no papel quadriculado: um quadrado de $3$ quadradinhos por
$3$; um retângulo de $5$ por $2$; um triângulo usando uma diagonal
da malha.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:5}
Coloque três brinquedos em fila. Descreva a fila com palavras de
posição: qual está entre os outros? Qual está à esquerda?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:6}
Desenhe um círculo. Desenhe uma cruz \emph{dentro} dele, uma estrela
\emph{fora} dele e um ponto \emph{sobre} a própria linha do círculo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:7}
No papel quadriculado, marque um quadradinho de saída. Siga:
$\to\ \uparrow\ \to\ \uparrow\ \to$. Marque a chegada. Depois
escreva o caminho que volta direto para a saída.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:8}
Quantos cantos ao todo: dois triângulos e um quadrado? Um círculo e
dois retângulos?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:shapes:9}
Pegue $12$ palitos de fósforo. Você consegue contornar um quadrado
usando \emph{todos} eles, com palitos inteiros e a mesma quantidade
em cada lado? Quantos palitos por lado? E daria para fazer isso com
$10$ palitos?
\end{exercise}
