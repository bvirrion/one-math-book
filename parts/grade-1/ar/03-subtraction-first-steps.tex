\chapter{الطرح: الخطوات الأولى}\label{ch:g1:subtraction}

سبعة بالونات، طار منها اثنان: كم بقي؟ الأخذ من الشيء هو
\emph{الطرح}، ويُكتب بالإشارة $-$. وللطرح وجه ثانٍ: فهو يجيب
أيضًا عن سؤال «بكم يزيد؟».

\section{الأخذ من الشيء}

\begin{definition}[الطرح]\label{def:g1:subtraction:def}
\emph{الطرح} هو أن نأخذ من الشيء ونعدّ ما بقي. نكتب
\[
7 - 2 = 5
\]
ونقرأ: «سبعة ناقص اثنين يساوي خمسة».
\end{definition}

\begin{example}[العدّ إلى الوراء]\label{ex:g1:subtraction:countback}
في الطرح الصغير، عُدّ إلى الوراء على المستقيم: في $11 - 3$ قل
«أحد عشر»، ثم ثلاث خطوات إلى الوراء --- «عشرة، تسعة، ثمانية». إذن
$11 - 3 = 8$.
\end{example}

\section{بكم يزيد؟}

\begin{example}[الفرق]\label{ex:g1:subtraction:difference}
«عند نور $9$ كرات وعند عمر $6$. بكم تزيد نور؟»
لم يُؤخذ شيء، ومع ذلك هي عملية طرح: زوِّج بين الكرات، ثم عُدّ كرات
نور التي بلا شريك:
\[
9 - 6 = 3 .
\]
تزيد نور بمقدار $3$. ويسمّى ناتج الطرح
\emph{الفرق}\index{فرق}.
\end{example}

\begin{method}[العدّ إلى الأمام]\label{met:g1:subtraction:countup}
لإيجاد \omterm{ex:g1:subtraction:difference}{الفرق}، غالبًا ما يكون العدّ \emph{إلى الأمام} أسهل:
في $12 - 9$ اِبدأ من $9$ واصعد إلى $12$: «عشرة، أحد عشر، اثنا
عشر» --- ثلاث خطوات. إذن $12 - 9 = 3$. (واصعد عبر $10$ في الفجوات
الأكبر: في $14 - 8$، من $8$ إلى $10$ مقدارها $2$، ومن $10$ إلى $14$
مقدارها $4$: \omterm{ex:g1:subtraction:difference}{فالفرق} $6$.)
\end{method}

\section{الجمع والطرح توأمان}

\begin{example}[عائلات الأعداد]\label{ex:g1:subtraction:family}
الأعداد $3$ و $4$ و $7$ تكوّن عائلة صغيرة:
\[
3 + 4 = 7, \qquad
4 + 3 = 7, \qquad
7 - 4 = 3, \qquad
7 - 3 = 4 .
\]
ومعرفة واحدة من هذه الحقائق تعطي الثلاث الأخرى مجانًا. الطرح يُلغي
الجمع: بعد $+4$، يعيدك $-4$ إلى حيث كنت.
\end{example}

\begin{example}[التحقّق]\label{ex:g1:subtraction:check}
حسبت $13 - 5 = 8$؟ تحقّق بالجمع التوأم:
$8 + 5 = 13$. \checkmark\ عملية الطرح صحيحة إذا نجح الجمع
العكسي، ولا تكون صحيحة إلا حينئذٍ.
\end{example}

\begin{example}[أي عملية؟]\label{ex:g1:subtraction:choose}
«$6$ طيور على السلك، طار منها $4$»: أخذٌ، $6 - 4 = 2$.
«$6$ طيور، ثم وصل $4$ غيرها»: ضمٌّ، $6 + 4 = 10$.
«$6$ عصافير و $4$ يمامات: بكم تزيد العصافير؟»:
فرقٌ، $6 - 4 = 2$. اُرسم القصة إذا ترددت!
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:1}
اُرسم واشطب واحسب: $6 - 2$؛ \quad $8 - 5$؛ \quad
$10 - 4$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:2}
احسب بالعدّ إلى الوراء: $9 - 3$؛ \quad $12 - 2$؛ \quad
$15 - 4$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:3}
عُدّ إلى الأمام لإيجاد \omterm{ex:g1:subtraction:difference}{الفرق}: $11 - 8$؛ \quad $13 - 9$؛ \quad
$16 - 12$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:4}
اُكتب حقائق العائلة الأربع للأعداد $2$ و $6$ و $8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:5}
احسب، ثم تحقّق بالجمع التوأم: $14 - 6$؛ \quad
$17 - 9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:6}
أكمل الفراغات:
\[
10 - \;?\; = 7, \qquad
\;?\; - 5 = 5, \qquad
12 - \;?\; = 12 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:7}
«في الطبق $15$ حبة كرز، أكل الأطفال $8$ منها. كم بقي؟» اُكتب
عملية الطرح وجملة الجواب.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:subtraction:8}
لكل قصة، اختر $+$ أو $-$، ثم احسب: «$9$ بطات في البركة، سبحت
$5$ بعيدًا»؛ «$9$ بطات، ثم وصلت $5$ إوزّات»؛
«عمر علي $9$ سنوات وعمر أخته $5$: بكم يكبرها علي؟».
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:subtraction:9}
كان عند البطة الأم $12$ بيضة. فقس بعضها، فبقي منها $12 - ?$ بيضة،
وهي تعدّ $7$ بيضات لم تفقس. كم بيضة فقست؟ (وأي عائلة أعداد
هذه؟)
\end{exercise}
