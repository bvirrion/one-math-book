\chapter{الجمع: الخطوات الأولى}\label{ch:g1:addition}

ثلاث كرات حمراء وكرتان زرقاوان: كم صارت كلها؟ ضمُّ الأشياء وعدُّ
\omterm{def:g1:addition:def}{المجموع} كله هو \emph{الجمع} --- أول عملية، ولها إشارتان
مشهورتان $+$ و $=$.

\section{الضمّ معًا}

\begin{definition}[الجمع]\label{def:g1:addition:def}
\emph{الجمع} هو أن نضمّ الأشياء ونعدّها كلها. نكتب
\[
3 + 2 = 5
\]
ونقرأ: «ثلاثة زائد اثنين يساوي خمسة». ويسمّى ناتج الجمع
\emph{المجموع}\index{مجموع}.
\end{definition}

\begin{example}[الترتيب لا يهمّ]\label{ex:g1:addition:order}
$2 + 3$ و $3 + 2$ يضمّان الكرات نفسها: كلاهما يعطي $5$. ولحساب
$2 + 9$ الأسرع أن تبدأ بالعدد الأكبر: قل «تسعة»، ثم عُدّ اثنين ---
«عشرة، أحد عشر». إذن
$2 + 9 = 11$.
\end{example}

\section{أصدقاء العشرة}

\begin{example}[متمّمات العشرة]\label{ex:g1:addition:friends}
العددان اللذان يعطيان $10$ معًا «صديقان في العشرة»:
\[
1 + 9, \quad 2 + 8, \quad 3 + 7, \quad 4 + 6, \quad 5 + 5 .
\]
وأصابع اليدين العشر تُظهرها كلها: اِرفع $3$ أصابع، فتكون الأصابع
المطويّة $7$، وهي الصديق. وحفظ أصدقاء العشرة عن ظهر قلب يجعل كثيرًا
من عمليات الجمع سريعة.
\end{example}

\end{omfigure}

\begin{method}[تجاوز العشرة]\label{met:g1:addition:crossten}
لحساب $8 + 5$:
\begin{enumerate}
  \item خذ من $5$ ما يحتاجه $8$ ليبلغ $10$، وهو
        $2$ (صديقه في العشرة)؛
  \item فيصير $8 + 2 = 10$، ويبقى $3$ من العدد $5$؛
  \item ثم $10 + 3 = 13$. إذن $8 + 5 = 13$.
\end{enumerate}
المرور بالعدد $10$ يحوّل مجموعًا صعبًا إلى مجموعين سهلين.
\end{method}

\begin{example}[الأضعاف]\label{ex:g1:addition:doubles}
الأضعاف جديرة بالحفظ عن ظهر قلب:
\[
1+1=2, \quad 2+2=4, \quad 3+3=6, \quad 4+4=8, \quad 5+5=10,
\]
\[
6+6=12, \quad 7+7=14, \quad 8+8=16, \quad 9+9=18, \quad 10+10=20 .
\]
والقريب من الضعف يأتي مجانًا: $6 + 7$ أكبر بواحد من $6 + 6$، أي $13$.
\end{example}

\section{مسائل صغيرة}

\begin{example}\label{ex:g1:addition:problem}
«عند لينا $7$ ملصقات، فحصلت على $6$ أخرى. كم صار عندها؟»
الحصول على المزيد يعني الجمع: $7 + 6 = 13$ (بالمرور بالعشرة:
$7 + 3 = 10$، ثم $+3$). صار عند لينا $13$ ملصقًا. واختم دائمًا
بجملة صغيرة تجيب عن السؤال!
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g1:addition:1}
اُرسم الكرات ثم احسب: $4 + 3$؛ \quad $5 + 2$؛ \quad
$6 + 4$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:2}
احسب بادئًا بالعدد الأكبر: $3 + 8$؛ \quad
$2 + 12$؛ \quad $4 + 9$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:3}
أعطِ صديق العشرة لكل من: $7$؛ \ $4$؛ \ $9$؛ \ $5$؛ \ $1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:4}
احسب بالمرور بالعشرة (\cref{met:g1:addition:crossten}):
$9 + 4$؛ \quad $8 + 7$؛ \quad $6 + 5$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:5}
اِسرد الأضعاف، ثم احسب بها: $7 + 8$؛ \quad
$5 + 6$؛ \quad $9 + 8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:6}
أكمل الفراغات:
\[
6 + \;?\; = 10, \qquad
\;?\; + 5 = 12, \qquad
10 + \;?\; = 17 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:7}
«في الحافلة $8$ أطفال، فصعد إليها $5$ كبار. كم شخصًا في
الحافلة؟» اُكتب عملية الجمع وجملة الجواب.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:addition:8}
عند عمر $6$ سيارات حمراء و $7$ زرقاء. وعند مريم $10$ سيارات. من
عنده سيارات أكثر؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:addition:9}
ثلاثة أحجار نرد تُظهر $4$ و $6$ و $5$. كم \omterm{def:g1:addition:def}{المجموع}؟ (اجمع حجرين
أولًا --- واختر الاثنين اللذين يكوّنان صديقَي العشرة!)
\end{exercise}
