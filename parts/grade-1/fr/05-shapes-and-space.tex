\chapter{Figures et espace}\label{ch:g1:shapes}

Les fenêtres sont des rectangles, les roues des cercles, les toits
cachent des triangles~: les figures sont partout. Ce chapitre apprend
à les reconnaître en comptant côtés et sommets, et à dire
précisément \emph{où} se trouvent les choses~: à gauche, à
droite, au-dessus, en dessous, entre.

\section{Les quatre figures de base}

\begin{definition}[Carré, rectangle, triangle, cercle]\label{def:g1:shapes:def}
\begin{itemize}
  \item le \emph{triangle}\index{triangle} a $3$ côtés et $3$ sommets~;
  \item le \emph{carré}\index{carré} a $4$ côtés égaux et $4$ sommets~;
  \item le \emph{rectangle}\index{rectangle} a $4$ côtés et $4$ sommets, avec les
        côtés opposés égaux (un carré étiré)~;
  \item le \emph{cercle}\index{cercle} est parfaitement rond~: pas de côté, pas
        de sommet.
\end{itemize}
\end{definition}

\begin{example}[Les figures penchées restent des figures]\label{ex:g1:shapes:tilted}
Un carré posé sur un sommet reste un carré --- quatre côtés
égaux, quatre sommets --- même s'il ressemble à un losange.
Tourner une figure ne change pas ce qu'elle est.
\end{example}

\begin{example}[Chasse aux figures]\label{ex:g1:shapes:hunt}
Dans la classe~: le tableau est un rectangle, l'horloge un cercle,
une serviette pliée peut être un triangle, certaines fenêtres
sont des carrés. Les objets réels ne sont pas des figures
parfaites, mais leurs faces s'en rapprochent (les solides et leurs
faces reviennent dans \cref{ch:g2:shapes}).
\end{example}

\section{Où est-ce~?}

\begin{definition}[Mots de position]\label{def:g1:shapes:position}
Pour dire où se trouve quelque chose~: \emph{à gauche} /
\emph{à droite}, \emph{au-dessus} / \emph{en dessous},
\emph{sur} / \emph{sous}, \emph{devant} / \emph{derrière},
\emph{entre}, \emph{à l'intérieur} / \emph{à l'extérieur}.
\end{definition}

\section{Chemins sur un quadrillage}

\begin{method}[Décrire un chemin]\label{met:g1:shapes:path}
Sur du papier quadrillé, un chemin est une liste de pas~:
$\to$ (une case à droite), $\leftarrow$ (à gauche), $\uparrow$
(vers le haut), $\downarrow$ (vers le bas).
\begin{enumerate}
  \item Pour suivre un chemin~: partir de la case marquée et obéir
        aux flèches une par une~;
  \item pour décrire un chemin~: le parcourir dans la tête et
        écrire les flèches dans l'ordre~;
  \item deux chemins différents peuvent relier les mêmes deux
        cases --- certains plus courts, d'autres plus longs.
\end{enumerate}
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g1:shapes:1}
Pour chaque figure, compte les côtés et les sommets et nomme-la~:
une figure avec $3$ sommets~; une figure ronde sans sommet~; une
figure avec $4$ côtés égaux.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:2}
Trouve à la maison~: deux rectangles, un cercle, un triangle, un
carré. (Portes~? assiettes~? panneaux de signalisation~?)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:3}
Vrai ou faux~? \guillemotleft{} Un carré posé sur un sommet
n'est plus un carré. \guillemotright{} Explique avec
\cref{ex:g1:shapes:tilted}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:4}
Dessine sur du papier quadrillé~: un carré de $3$ cases sur
$3$~; un rectangle de $5$ sur $2$~; un triangle en utilisant une
diagonale du quadrillage.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:5}
Place trois jouets en rangée. Décris la rangée avec des mots de
position~: lequel est entre les autres~? Lequel est à gauche~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:6}
Dessine un cercle. Dessine une croix \emph{à l'intérieur}, une
étoile \emph{à l'extérieur}, et un point \emph{sur} le cercle
lui-même.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:7}
Sur du papier quadrillé, marque une case de départ. Suis~:
$\to\ \uparrow\ \to\ \uparrow\ \to$. Marque l'arrivée. Puis
écris le chemin qui revient tout droit au départ.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:shapes:8}
Combien de sommets en tout~: deux triangles et un carré~? Un
cercle et deux rectangles~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:shapes:9}
Prends $12$ allumettes. Peux-tu dessiner le contour d'un carré en
utilisant \emph{toutes} les allumettes, avec des allumettes entières
et le même nombre sur chaque côté~? Combien d'allumettes par
côté~? Pourrait-on le faire avec $10$ allumettes~?
\end{exercise}
