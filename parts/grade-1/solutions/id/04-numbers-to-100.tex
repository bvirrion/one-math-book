\section*{Bab \ref{ch:g1:tens} --- Bilangan sampai 100}

\begin{solution}{exo:g1:tens:1}
$26$: $2$ puluhan, $6$ satuan. $40$: $4$ puluhan, $0$ satuan. $73$:
$7$ puluhan, $3$ satuan. $99$: $9$ puluhan, $9$ satuan.
\end{solution}

\begin{solution}{exo:g1:tens:2}
$52$; \ $80$; \ $19$; \ $100$.
\end{solution}

\begin{solution}{exo:g1:tens:3}
$63$; \ $40$; \ $85$; \ $100$.
\end{solution}

\begin{solution}{exo:g1:tens:4}
$23 + 10 = 33$; \quad $67 + 10 = 77$; \quad $45 - 10 = 35$; \quad
$80 - 10 = 70$.
\end{solution}

\begin{solution}{exo:g1:tens:5}
$10, 20, 30, 40, 50, 60$. \quad $5, 10, 15, 20, 25, 30$. \quad
$12, 14, 16, 18, 20, 22$.
\end{solution}

\begin{solution}{exo:g1:tens:6}
$34 < 43$; \ $58 < 61$; \ $70 > 69$; \ $99 < 100$.
\end{solution}

\begin{solution}{exo:g1:tens:7}
$19 < 41 < 46 < 64 < 91$.
\end{solution}

\begin{solution}{exo:g1:tens:8}
$52$ adalah $5$ puluhan dan $2$ satuan: $5$ ikat penuh dan $2$ bunga
lepas.
\end{solution}

\begin{solution}{exo:g1:tens:9}
Dari $36$, dua langkah ke bawah berarti $+20$ ($\to 56$), satu
langkah ke kiri berarti $-1$ ($\to 55$). Kamu mendarat di $55$.
Perhitungannya adalah $36 + 20 - 1$, yaitu $36 + 19$.

\end{solution}
