\section*{Bab \ref{ch:g1:addition} --- Penjumlahan: Langkah Pertama}

\begin{solution}{exo:g1:addition:1}
$4 + 3 = 7$; \quad $5 + 2 = 7$; \quad $6 + 4 = 10$.
\end{solution}

\begin{solution}{exo:g1:addition:2}
$3 + 8 = 11$ (mulai dari $8$, bilang maju $3$); \quad
$2 + 12 = 14$; \quad $4 + 9 = 13$.
\end{solution}

\begin{solution}{exo:g1:addition:3}
Sahabat sepuluh: $7 \to 3$; \ $4 \to 6$; \ $9 \to 1$; \ $5 \to 5$;
\ $1 \to 9$.
\end{solution}

\begin{solution}{exo:g1:addition:4}
$9 + 4 = 13$ ($9 + 1 = 10$, lalu $+3$); \quad
$8 + 7 = 15$ ($8 + 2 = 10$, lalu $+5$); \quad
$6 + 5 = 11$ ($6 + 4 = 10$, lalu $+1$).
\end{solution}

\begin{solution}{exo:g1:addition:5}
$7 + 8 = 15$ (satu lebihnya dari $7 + 7 = 14$); \quad
$5 + 6 = 11$ (satu lebihnya dari $5 + 5$); \quad
$9 + 8 = 17$ (satu kurangnya dari $9 + 9 = 18$).
\end{solution}

\begin{solution}{exo:g1:addition:6}
$6 + 4 = 10$; \quad $7 + 5 = 12$; \quad $10 + 7 = 17$.
\end{solution}

\begin{solution}{exo:g1:addition:7}
$8 + 5 = 13$. Ada $13$ orang di dalam bus.
\end{solution}

\begin{solution}{exo:g1:addition:8}
Tono punya $6 + 7 = 13$ mobil, Mia punya $10$: mobil Tono lebih banyak.
\end{solution}

\begin{solution}{exo:g1:addition:9}
Jumlahkan dulu sahabat sepuluh $4$ dan $6$: $4 + 6 = 10$, lalu
$10 + 5 = 15$. \omterm{def:g1:addition:def}{Jumlahnya} $15$.

\end{solution}
