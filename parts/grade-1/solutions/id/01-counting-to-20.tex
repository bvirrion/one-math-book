\section*{Bab \ref{ch:g1:counting} --- Membilang sampai 20}

\begin{solution}{exo:g1:counting:1}
Jawabannya bergantung pada namanya; dua tangan punya $10$ jari; dua
ekor kucing punya $8$ kaki.
\end{solution}

\begin{solution}{exo:g1:counting:2}
$7$; \ $12$; \ $20$; \ $0$. Dengan kata: empat; lima belas; delapan belas.
\end{solution}

\begin{solution}{exo:g1:counting:3}
$8, 9, 10, 11, 12, 13$. Mundur: $15, 14, 13, 12, 11, 10$.
\end{solution}

\begin{solution}{exo:g1:counting:4}
Tepat sebelum $10$: $9$. Tepat sesudah $16$: $17$. Di antara $12$ dan
$14$: $13$.
\end{solution}

\begin{solution}{exo:g1:counting:5}
$6 < 9$; \ $14 > 8$; \ $11 < 13$; \ $20 > 19$.
\end{solution}

\begin{solution}{exo:g1:counting:6}
Memasangkan $6$ lingkaran dengan $8$ tanda silang menyisakan $2$ tanda
silang tanpa pasangan: tanda silang lebih banyak, lebihnya $2$.
\end{solution}

\begin{solution}{exo:g1:counting:7}
$2 < 7 < 10 < 12 < 17$.
\end{solution}

\begin{solution}{exo:g1:counting:8}
Kedua: Budi. Keempat: Dani. Dani ada di urutan ke-$4$.
\end{solution}

\begin{solution}{exo:g1:counting:9}
Tiga langkah sesudah $13$: $16$. Tiga langkah sebelum $13$: $10$.

\end{solution}
