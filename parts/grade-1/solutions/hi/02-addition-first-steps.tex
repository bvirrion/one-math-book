\section*{अध्याय \ref{ch:g1:addition} --- जोड़: पहले कदम}

\begin{solution}{exo:g1:addition:1}
$4 + 3 = 7$; \quad $5 + 2 = 7$; \quad $6 + 4 = 10$।
\end{solution}

\begin{solution}{exo:g1:addition:2}
$3 + 8 = 11$ ($8$ से शुरू, $3$ आगे गिनो); \quad
$2 + 12 = 14$; \quad $4 + 9 = 13$।
\end{solution}

\begin{solution}{exo:g1:addition:3}
दस के साथी: $7 \to 3$; \ $4 \to 6$; \ $9 \to 1$; \ $5 \to 5$;
\ $1 \to 9$।
\end{solution}

\begin{solution}{exo:g1:addition:4}
$9 + 4 = 13$ ($9 + 1 = 10$, फिर $+3$); \quad
$8 + 7 = 15$ ($8 + 2 = 10$, फिर $+5$); \quad
$6 + 5 = 11$ ($6 + 4 = 10$, फिर $+1$)।
\end{solution}

\begin{solution}{exo:g1:addition:5}
$7 + 8 = 15$ ($7 + 7 = 14$ से एक ज़्यादा); \quad
$5 + 6 = 11$ ($5 + 5$ से एक ज़्यादा); \quad
$9 + 8 = 17$ ($9 + 9 = 18$ से एक कम)।
\end{solution}

\begin{solution}{exo:g1:addition:6}
$6 + 4 = 10$; \quad $7 + 5 = 12$; \quad $10 + 7 = 17$।
\end{solution}

\begin{solution}{exo:g1:addition:7}
$8 + 5 = 13$। बस में $13$ लोग हैं।
\end{solution}

\begin{solution}{exo:g1:addition:8}
टॉम के पास $6 + 7 = 13$ कारें, मिया के पास $10$: टॉम के पास ज़्यादा।
\end{solution}

\begin{solution}{exo:g1:addition:9}
पहले दस के साथी $4$ और $6$ जोड़ो: $4 + 6 = 10$, फिर
$10 + 5 = 15$। कुल $15$।
\end{solution}
