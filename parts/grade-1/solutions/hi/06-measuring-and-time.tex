\section*{अध्याय \ref{ch:g1:measure} --- मापना और समय}

\begin{solution}{exo:g1:measure:1}
उत्तर पेंसिलों पर निर्भर करता है; एक सिरे से मिलाने पर जो सबसे
आगे पहुँचे वह सबसे लंबी, जो सबसे कम पहुँचे वह सबसे छोटी।
\end{solution}

\begin{solution}{exo:g1:measure:2}
उत्तर अलग-अलग; हर बार मापपट्टी का $0$ वस्तु की शुरुआत पर रखकर
उसके सिरे पर पूरा सेंटीमीटर पढ़ा जाता है।
\end{solution}

\begin{solution}{exo:g1:measure:3}
सोमवार के बाद: मंगलवार। रविवार से पहले: शनिवार। शुक्रवार के दो
दिन बाद: रविवार।
\end{solution}

\begin{solution}{exo:g1:measure:4}
बीता कल: बुधवार। आने वाला कल: शुक्रवार। तीन दिन बाद: रविवार।
\end{solution}

\begin{solution}{exo:g1:measure:5}
$6$ बजे छोटी सुई $6$ पर होती है और लंबी सुई सीधे ऊपर
$12$ पर।
\end{solution}

\begin{solution}{exo:g1:measure:6}
छोटी सुई $5$ पर: $5$ बजे। छोटी सुई $11$ पर: $11$ बजे।
\end{solution}

\begin{solution}{exo:g1:measure:7}
$8$ के लिए, जैसे $5 + 2 + 1$ या $2 + 2 + 2 + 2$ (और भी सही
उत्तर हैं)।
\end{solution}

\begin{solution}{exo:g1:measure:8}
$5 + 2 + 2 + 2 = 11$। लियो के पास कुल $11$ हैं।
\end{solution}

\begin{solution}{exo:g1:measure:9}
$3$ से $5$ तक $2$ है: ज़ोई को $2$ वापस मिलते हैं।

\end{solution}
