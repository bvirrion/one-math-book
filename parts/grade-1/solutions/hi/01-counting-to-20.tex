\section*{अध्याय \ref{ch:g1:counting} --- 20 तक गिनती}

\begin{solution}{exo:g1:counting:1}
उत्तर नाम पर निर्भर करता है; दो हाथों में $10$ उँगलियाँ होती हैं;
दो बिल्लियों के $8$ पैर होते हैं।
\end{solution}

\begin{solution}{exo:g1:counting:2}
$7$; \ $12$; \ $20$; \ $0$। शब्दों में: चार; पंद्रह; अठारह।
\end{solution}

\begin{solution}{exo:g1:counting:3}
$8, 9, 10, 11, 12, 13$। उलटी गिनती: $15, 14, 13, 12, 11, 10$।
\end{solution}

\begin{solution}{exo:g1:counting:4}
$10$ से ठीक पहले: $9$। $16$ के ठीक बाद: $17$। $12$ और
$14$ के बीच: $13$।
\end{solution}

\begin{solution}{exo:g1:counting:5}
$6 < 9$; \ $14 > 8$; \ $11 < 13$; \ $20 > 19$।
\end{solution}

\begin{solution}{exo:g1:counting:6}
$6$ वृत्तों के साथ $8$ क्रॉस जोड़ने पर $2$ क्रॉस बिना जोड़ीदार
बचते हैं: क्रॉस ज़्यादा हैं, $2$ से।
\end{solution}

\begin{solution}{exo:g1:counting:7}
$2 < 7 < 10 < 12 < 17$।
\end{solution}

\begin{solution}{exo:g1:counting:8}
दूसरे: बॉब। चौथे: डैन। डैन स्थान $4$ पर है।
\end{solution}

\begin{solution}{exo:g1:counting:9}
$13$ के तीन कदम बाद: $16$। $13$ से तीन कदम पहले: $10$।
\end{solution}
