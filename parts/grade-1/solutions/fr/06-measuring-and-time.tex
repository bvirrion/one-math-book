\section*{Chapitre \ref{ch:g1:measure} --- Mesures et temps}

\begin{solution}{exo:g1:measure:1}
Les réponses dépendent des crayons~; le plus long dépasse le
plus loin quand ils sont alignés à une extrémité, le plus
court dépasse le moins.
\end{solution}

\begin{solution}{exo:g1:measure:2}
Les réponses varient~; chacune se lit en plaçant le $0$ de la
règle au début de l'objet et en lisant le centimètre entier à
son extrémité.
\end{solution}

\begin{solution}{exo:g1:measure:3}
Après lundi~: mardi. Avant dimanche~: samedi. Deux jours après
vendredi~: dimanche.
\end{solution}

\begin{solution}{exo:g1:measure:4}
Hier~: mercredi. Demain~: vendredi. Dans trois jours~: dimanche.
\end{solution}

\begin{solution}{exo:g1:measure:5}
À $6$~heures, la petite aiguille pointe vers le $6$ et la grande
aiguille pointe vers le haut, sur le $12$.
\end{solution}

\begin{solution}{exo:g1:measure:6}
Petite aiguille sur $5$~: $5$~heures. Petite aiguille sur $11$~:
$11$~heures.
\end{solution}

\begin{solution}{exo:g1:measure:7}
Pour $8$, par exemple $5 + 2 + 1$ ou $2 + 2 + 2 + 2$ (d'autres
réponses correctes existent).
\end{solution}

\begin{solution}{exo:g1:measure:8}
$5 + 2 + 2 + 2 = 11$. Léo a $11$.
\end{solution}

\begin{solution}{exo:g1:measure:9}
De $3$ à $5$ en montant, c'est $2$~: Zoé reçoit $2$.
\end{solution}
