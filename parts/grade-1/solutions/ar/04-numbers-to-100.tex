\section*{الفصل \ref{ch:g1:tens} --- الأعداد حتى 100}

\begin{solution}{exo:g1:tens:1}
$26$: $2$ عشرات و $6$ آحاد. $40$: $4$ عشرات و $0$ آحاد. $73$: $7$
عشرات و $3$ آحاد. $99$: $9$ عشرات و $9$ آحاد.
\end{solution}

\begin{solution}{exo:g1:tens:2}
$52$؛ \ $80$؛ \ $19$؛ \ $100$.
\end{solution}

\begin{solution}{exo:g1:tens:3}

\end{solution}

\begin{solution}{exo:g1:tens:4}
$23 + 10 = 33$؛ \quad $67 + 10 = 77$؛ \quad $45 - 10 = 35$؛ \quad
$80 - 10 = 70$.
\end{solution}

\begin{solution}{exo:g1:tens:6}

\end{solution}

\begin{solution}{exo:g1:tens:7}

\end{solution}

\begin{solution}{exo:g1:tens:8}

مفردة.
\end{solution}

\begin{solution}{exo:g1:tens:9}
من $36$، خطوتان نزولًا تعني $+20$ ($\to 56$)، وخطوة يسارًا تعني
$-1$ ($\to 55$). فتصل إلى $55$. والعملية هي
$36 + 20 - 1$، أي $36 + 19$.
\end{solution}
