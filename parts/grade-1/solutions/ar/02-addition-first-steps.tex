\section*{الفصل \ref{ch:g1:addition} --- الجمع: الخطوات الأولى}

\begin{solution}{exo:g1:addition:1}
$4 + 3 = 7$؛ \quad $5 + 2 = 7$؛ \quad $6 + 4 = 10$.
\end{solution}

\begin{solution}{exo:g1:addition:2}
$3 + 8 = 11$ (اِبدأ من $8$ وعُدّ $3$)؛ \quad
$2 + 12 = 14$؛ \quad $4 + 9 = 13$.
\end{solution}

\begin{solution}{exo:g1:addition:3}
أصدقاء العشرة: $7 \to 3$؛ \ $4 \to 6$؛ \ $9 \to 1$؛ \ $5 \to 5$؛
\ $1 \to 9$.
\end{solution}

\begin{solution}{exo:g1:addition:4}
$9 + 4 = 13$ ($9 + 1 = 10$، ثم $+3$)؛ \quad
$8 + 7 = 15$ ($8 + 2 = 10$، ثم $+5$)؛ \quad
$6 + 5 = 11$ ($6 + 4 = 10$، ثم $+1$).
\end{solution}

\begin{solution}{exo:g1:addition:5}
$7 + 8 = 15$ (أكبر بواحد من $7 + 7 = 14$)؛ \quad
$5 + 6 = 11$ (أكبر بواحد من $5 + 5$)؛ \quad
$9 + 8 = 17$ (أقل بواحد من $9 + 9 = 18$).
\end{solution}

\begin{solution}{exo:g1:addition:6}
$6 + 4 = 10$؛ \quad $7 + 5 = 12$؛ \quad $10 + 7 = 17$.
\end{solution}

\begin{solution}{exo:g1:addition:7}
$8 + 5 = 13$. في الحافلة $13$ شخصًا.
\end{solution}

\begin{solution}{exo:g1:addition:8}
عند عمر $6 + 7 = 13$ سيارة، وعند مريم $10$: فعمر عنده أكثر.
\end{solution}

\begin{solution}{exo:g1:addition:9}
اجمع صديقَي العشرة $4$ و $6$ أولًا: $4 + 6 = 10$، ثم
$10 + 5 = 15$. \omterm{def:g1:addition:def}{المجموع} $15$.
\end{solution}
