\section*{الفصل \ref{ch:g1:counting} --- العدّ حتى 20}

\begin{solution}{exo:g1:counting:1}
الجواب يختلف باختلاف الاسم؛ في اليدين $10$ أصابع، وفي القطتين
$8$ أرجل.
\end{solution}

\begin{solution}{exo:g1:counting:2}
$7$؛ \ $12$؛ \ $20$؛ \ $0$. وبالحروف: أربعة؛ خمسة عشر؛ ثمانية عشر.
\end{solution}

\begin{solution}{exo:g1:counting:3}
$8, 9, 10, 11, 12, 13$. وإلى الوراء: $15, 14, 13, 12, 11, 10$.
\end{solution}

\begin{solution}{exo:g1:counting:4}
قبل $10$ مباشرة: $9$. وبعد $16$ مباشرة: $17$. وبين $12$
و $14$: $13$.
\end{solution}

\begin{solution}{exo:g1:counting:5}
$6 < 9$؛ \ $14 > 8$؛ \ $11 < 13$؛ \ $20 > 19$.
\end{solution}

\begin{solution}{exo:g1:counting:6}
تزويج $6$ دوائر مع $8$ علامات × يترك $2$ منها بلا
شريك: فالعلامات أكثر بمقدار $2$.
\end{solution}

\begin{solution}{exo:g1:counting:7}
$2 < 7 < 10 < 12 < 17$.
\end{solution}

\begin{solution}{exo:g1:counting:8}
الثاني: بلال. والرابع: داود. وداود في المرتبة $4$.
\end{solution}

\begin{solution}{exo:g1:counting:9}
ثلاث خطوات بعد $13$: $16$. وثلاث خطوات قبل $13$: $10$.
\end{solution}
