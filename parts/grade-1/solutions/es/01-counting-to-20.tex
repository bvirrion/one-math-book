\section*{Capítulo \ref{ch:g1:counting} --- Contar hasta 20}

\begin{solution}{exo:g1:counting:1}
Depende del nombre; dos manos tienen $10$ dedos y dos gatos
tienen $8$ patas.
\end{solution}

\begin{solution}{exo:g1:counting:2}
$7$; \ $12$; \ $20$; \ $0$. Con letras: cuatro; quince; dieciocho.
\end{solution}

\begin{solution}{exo:g1:counting:3}
$8, 9, 10, 11, 12, 13$. Hacia atrás: $15, 14, 13, 12, 11, 10$.
\end{solution}

\begin{solution}{exo:g1:counting:4}
Justo antes del $10$: $9$. Justo después del $16$: $17$. Entre el $12$ y el
$14$: $13$.
\end{solution}

\begin{solution}{exo:g1:counting:5}
$6 < 9$; \ $14 > 8$; \ $11 < 13$; \ $20 > 19$.
\end{solution}

\begin{solution}{exo:g1:counting:6}
Al emparejar $6$ círculos con $8$ cruces quedan $2$ cruces sin
pareja: hay más cruces, $2$ más.
\end{solution}

\begin{solution}{exo:g1:counting:7}
$2 < 7 < 10 < 12 < 17$.
\end{solution}

\begin{solution}{exo:g1:counting:8}
El segundo: Bruno. El cuarto: Diego. Diego ocupa el puesto $4$.
\end{solution}

\begin{solution}{exo:g1:counting:9}
Tres pasos después del $13$: $16$. Tres pasos antes del $13$: $10$.
\end{solution}
