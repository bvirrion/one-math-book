\section*{Capítulo \ref{ch:g1:counting} --- Contar hasta 20}

\begin{solution}{exo:g1:counting:1}
Las respuestas dependen del nombre; dos manos tienen $10$ dedos; dos gatos
tener $8$ piernas.
\end{solution}

\begin{solution}{exo:g1:counting:2}
$7$; \ $12$; \ $20$; \ $0$. En palabras: cuatro; quince; dieciocho.
\end{solution}

\begin{solution}{exo:g1:counting:3}
$8, 9, 10, 11, 12, 13$. Hacia atrás: $15, 14, 13, 12, 11, 10$.
\end{solution}

\begin{solution}{exo:g1:counting:4}
justo antes $10$: $9$. Justo después $16$: $17$. Entre $12$ y
$14$: $13$.
\end{solution}

\begin{solution}{exo:g1:counting:5}
$6 < 9$; \ $14 > 8$; \ $11 < 13$; \ $20 > 19$.
\end{solution}

\begin{solution}{exo:g1:counting:6}
Emparejamiento $6$ círculos con $8$ cruza deja $2$ cruza sin
compañero: los cruces son más, por $2$.
\end{solution}

\begin{solution}{exo:g1:counting:7}
$2 < 7 < 10 < 12 < 17$.
\end{solution}

\begin{solution}{exo:g1:counting:8}
Segundo: Bob. Cuarto: Dan. Dan está en su lugar $4$.
\end{solution}

\begin{solution}{exo:g1:counting:9}
Tres pasos después $13$: $16$. tres pasos antes $13$: $10$.
\end{solution}
