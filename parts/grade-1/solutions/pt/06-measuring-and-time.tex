\section*{Capítulo \ref{ch:g1:measure} --- Medidas e tempo}

\begin{solution}{exo:g1:measure:1}
As respostas dependem dos lápis; alinhados por uma das pontas, o
mais comprido é o que chega mais longe e o mais curto é o que chega
menos longe.
\end{solution}

\begin{solution}{exo:g1:measure:2}
As respostas variam; cada medida é lida colocando o $0$ da régua no
começo do objeto e lendo o centímetro inteiro no fim dele.
\end{solution}

\begin{solution}{exo:g1:measure:3}
Depois de segunda: terça. Antes de domingo: sábado. Dois dias depois
de sexta: domingo.
\end{solution}

\begin{solution}{exo:g1:measure:4}
Ontem: quarta. Amanhã: sexta. Daqui a três dias: domingo.
\end{solution}

\begin{solution}{exo:g1:measure:5}
Às $6$ horas o ponteiro pequeno aponta para o $6$ e o ponteiro
grande aponta para cima, para o $12$.
\end{solution}

\begin{solution}{exo:g1:measure:6}
Ponteiro pequeno no $5$: $5$ horas. Ponteiro pequeno no $11$: $11$
horas.
\end{solution}

\begin{solution}{exo:g1:measure:7}
Para $8$, por exemplo $5 + 2 + 1$ ou $2 + 2 + 2 + 2$ (há outras
respostas corretas).
\end{solution}

\begin{solution}{exo:g1:measure:8}
$5 + 2 + 2 + 2 = 11$. Léo tem $11$.
\end{solution}

\begin{solution}{exo:g1:measure:9}
De $3$ até $5$ vão $2$: Zoe recebe $2$ de troco.
\end{solution}
