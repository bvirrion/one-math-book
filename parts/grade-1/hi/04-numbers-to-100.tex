\chapter{100 तक की संख्याएँ}\label{ch:g1:tens}

बड़े ढेर को एक-एक करके गिनने में बहुत समय लगता है। बड़ी तरकीब यह
है: वस्तुओं को \emph{दस} के बंडलों में बाँधो, और बंडल गिनो। $100$
तक की संख्याएँ इसी तरह बनी हैं --- और इसीलिए वे दो अंकों में
लिखी जाती हैं।

\section{दस के बंडल}

\begin{definition}[दहाई और इकाई]\label{def:g1:tens:def}
एक \emph{दहाई} दस इकाइयों का बंडल है। संख्या $34$ का मतलब है $3$
दहाई और $4$ इकाई:
\[
34 = 10 + 10 + 10 + 4 .
\]
बायाँ अंक बंडल गिनता है, दायाँ अंक खुली इकाइयाँ।
\end{definition}

\begin{omfigure}
\begin{tikzpicture}[scale=0.75]
  % दस तीलियों के तीन बंडल, हर एक पट्टी से बँधा
  \foreach \b in {0,1,2} {
    \begin{scope}[xshift=\b*2.4cm]
    \foreach \i in {0,...,9} \draw[omDef, very thick]
      ({0.16*\i},0) -- ({0.16*\i},1.7);
    \draw[omExo, thick, rounded corners=3pt]
      (-0.22,0.62) rectangle (1.66,1.05);
    \end{scope}
  }
  % चार खुली तीलियाँ
  \foreach \i in {0,...,3} \draw[omThm, very thick]
    (7.7+0.4*\i,0) -- (7.7+0.4*\i,1.7);
  \node[right, font=\small] at (9.7,0.85)
    {$3$ बंडल और $4$ तीलियाँ: $34$};
\end{tikzpicture}

{\small चौंतीस तीलियाँ: दस-दस के तीन बंडल (हर एक पट्टी से बँधा)
और चार खुली तीलियाँ। $34$ तक एक-एक करके गिनने की ज़रूरत नहीं!}
\end{omfigure}

\begin{example}[संख्याएँ पढ़ना]\label{ex:g1:tens:reading}
$20$ दो दहाई है (बीस), $30$ तीन दहाई (तीस)\,\dots\ और ऐसे ही
$90$ (नब्बे) तथा $100$ (सौ: दस दहाई!) तक। बीच में:
$47$ को ``सैंतालीस'' पढ़ते हैं --- चालीस उसकी $4$ दहाइयों का,
सात इकाइयों का।
\end{example}

\begin{omfigure}
\begin{tikzpicture}[scale=0.9]
  \draw (0,0) grid[xstep=2.2, ystep=0.8] (4.4,1.6);
  \node[font=\small] at (1.1,1.2) {दहाई};
  \node[font=\small] at (3.3,1.2) {इकाई};
  \node[font=\large, omDef] at (1.1,0.4) {$3$};
  \node[font=\large, omThm] at (3.3,0.4) {$4$};
\end{tikzpicture}

{\small $34$ की स्थान-सारणी: $3$ दहाई के खाने में बैठा है, $4$
इकाई के खाने में। इस छोटी सारणी में हर साल एक नया खाना जुड़ता
जाएगा!}
\end{omfigure}

\begin{example}[एक ही अंक, अलग संख्याएँ]\label{ex:g1:tens:place}
$27$ और $72$ में अंक वही हैं, पर संख्याएँ बहुत अलग: $27$ यानी
$2$ दहाई और $7$ इकाई; $72$ यानी $7$ दहाई और $2$ इकाई --- कहीं
ज़्यादा बड़ी! अंक का \emph{स्थान} तय करता है कि वह क्या गिन रहा है।
\end{example}

\section{सौ का जाल}

\begin{omfigure}
\begin{tikzpicture}[scale=0.52]
  \foreach \r in {0,...,9} {
    \foreach \c in {0,...,9} {
      \pgfmathtruncatemacro{\n}{\r*10+\c+1}
      \draw[gray!50] (\c,-\r) +(-0.5,-0.5) rectangle +(0.5,0.5);
      \node[font=\footnotesize] at (\c,-\r) {\n};
    }
  }
  \draw[omThm, very thick] (3,-4) +(-0.5,-0.5) rectangle +(0.5,0.5);
  \draw[omDef, very thick] (3,-5) +(-0.5,-0.5) rectangle +(0.5,0.5);
\end{tikzpicture}

{\small सौ का जाल: हर पंक्ति एक दहाई है। दाईं ओर एक कदम
$+1$ है; \emph{नीचे} एक कदम $+10$ (लाल $44$ से सीधे नीचे
नीले $54$ तक)।}
\end{omfigure}

\begin{method}[जाल पर चलना]\label{met:g1:tens:grid}
\begin{enumerate}
  \item $+1$ / $-1$: दाईं / बाईं ओर एक कदम;
  \item $+10$ / $-10$: नीचे / ऊपर एक कदम --- केवल दहाई का अंक
        बदलता है: $37 + 10 = 47$;
  \item दस-दस गिनना: $10, 20, 30, \dots$ --- जाल के आख़िरी
        स्तंभ में सीधे नीचे उतरते जाओ।
\end{enumerate}
\end{method}

\begin{example}[2-2 और 5-5 गिनना]\label{ex:g1:tens:skip}
$2$-$2$ करके: $2, 4, 6, 8, 10, 12, \dots$ (सम संख्याएँ)।
$5$-$5$ करके: $5, 10, 15, 20, 25, \dots$ --- ये $5$ या $0$ पर
ख़त्म होती हैं और जाल पर सुंदर प्रतिरूप बनाती हैं। ऐसे छलाँग लगाकर
गिनना गुणा की तैयारी है (\cref{ch:g2:mult})।
\end{example}

\section{बड़ी संख्याओं की तुलना}

\begin{method}[दहाइयों से तुलना]\label{met:g1:tens:compare}
जिसमें ज़्यादा दहाई, वही बड़ी: $61 > 58$, क्योंकि $6$ दहाई $5$
दहाई पर भारी हैं --- चाहे $8$, $1$ से ज़्यादा हो! दहाई बराबर हों
तभी इकाई फ़ैसला करती है: $43 < 47$।
\end{method}

\begin{example}\label{ex:g1:tens:compare}
$35$, $53$, $29$ को क्रम में लगाओ: पहले दहाई देखो ($2$, $3$, $5$):
\[
29 < 35 < 53 .
\]
\end{example}

\section{अभ्यास}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:2}
संख्या लिखो: $5$ दहाई और $2$ इकाई; \ $8$ दहाई; \ $1$ दहाई और
$9$ इकाई; \ $10$ दहाई।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:3}
अंकों में लिखो: तिरसठ; \ चालीस; \ पचासी; \ सौ।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:4}
सौ के जाल पर चलकर निकालो: $23 + 10$; \quad $67 + 10$;
\quad $45 - 10$; \quad $80 - 10$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:5}
आगे बढ़ाओ: $10, 20, 30, ?, ?, ?$। \quad
$5, 10, 15, ?, ?, ?$। \quad $12, 14, 16, ?, ?, ?$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:6}
उतार कर $<$ या $>$ से पूरा करो:
\[
34 \;?\; 43, \qquad
58 \;?\; 61, \qquad
70 \;?\; 69, \qquad
99 \;?\; 100 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:7}
सबसे छोटी से सबसे बड़ी के क्रम में लिखो: $46$; \ $19$; \ $64$; \ $91$; \
$41$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g1:tens:8}
एक माली $52$ फूलों को दस-दस के गुच्छों में बाँधता है। कितने पूरे
गुच्छे बनेंगे, और कितने फूल खुले बचेंगे?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g1:tens:9}
सौ के जाल पर $36$ से शुरू करो; दो कदम नीचे और एक कदम बाईं ओर
चलो। कहाँ पहुँचे? अभी-अभी कौन-सी गणना की?
\end{exercise}
