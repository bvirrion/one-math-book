\chapter{Differentiaalvergelijkingen}\label{ch:b2:diffeq}

Bachelorjaar 1 loste de lineaire vergelijkingen op die een formule
toelaten. Dit hoofdstuk levert wat formules niet kunnen: de
\emph{stelling van Cauchy--Lipschitz} --- bestaan en eenduidigheid
voor $y' = f(t, y)$ --- bewezen met de vastepuntsstelling van
Banach, precies zoals in \cref{ch:b2:metric} beloofd; en daarna de
volledige theorie van de \emph{lineaire stelsels} $X' = A(t)X +
B(t)$, met de \omterm{ex:b2:nvs:matrixexp}{matrixexponentiaal} en de \omterm{def:b2:diffeq:wronskian}{wronskiaan} als
rekenmotoren.

\section{De stelling van Cauchy--Lipschitz}

\begin{theorem}[Cauchy--Lipschitz, versie met globale lipschitzvoorwaarde]\label{thm:b2:diffeq:cauchylipschitz}
Zij $I$ een segment en $f \colon I \times \R^n \to \R^n$ \omterm{def:b2:metric:continuity}{continu} en
\emph{lipschitz in de tweede veranderlijke}, uniform in de eerste:
$\norm{f(t, y) - f(t, z)} \leq k\,\norm{y - z}$ voor alle $t \in
I$. Dan heeft voor elke $(t_0, y_0) \in I \times \R^n$ het
beginwaardeprobleem\index{stelling van Cauchy--Lipschitz}
\[
y' = f(t, y), \qquad y(t_0) = y_0
\]
precies één oplossing $y \colon I \to \R^n$ van klasse $C^1$.
\end{theorem}

\begin{proof}
\emph{Herformulering.} Een \omterm{def:b2:metric:continuity}{continue} $y$ lost het probleem op dan en
slechts dan als zij voldoet aan de integraalvergelijking
\[
y(t) = y_0 + \int_{t_0}^{t} f\bigl(s, y(s)\bigr)\,\dd s
=: T(y)(t)
\]
(de hoofdstelling van de integraalrekening, in beide richtingen;
een \omterm{def:b2:metric:continuity}{continue} oplossing van de integraalvergelijking is automatisch
$C^1$).

\emph{Een contractie, na hernormeren.} Schat op de \omterm{def:b2:nvs:banach}{banachruimte} $E
= C(I, \R^n)$ met de \emph{gewogen} \omterm{def:b2:nvs:norm}{norm}
\[
N(y) = \sup_{t \in I}\; \eu^{-2k\abs{t - t_0}}\,\norm{y(t)} ,
\]
(equivalent met de supremumnorm: het gewicht is op het segment $I$
naar boven en naar onder begrensd, dus $E$ blijft \omterm{def:b2:metric:complete}{volledig}) voor
$y, z \in E$ en, zeg, $t \geq t_0$:
\[
\norm{T(y)(t) - T(z)(t)}
\leq \int_{t_0}^{t} k\,\norm{y(s) - z(s)}\,\dd s
\leq k\,N(y - z)\int_{t_0}^{t} \eu^{2k(s - t_0)}\dd s
\leq \frac{N(y-z)}{2}\,\eu^{2k(t - t_0)} .
\]
Vermenigvuldigen met $\eu^{-2k(t - t_0)}$ en het supremum nemen
(het geval $t < t_0$ is \omterm{def:b2:quadratic:adjoint}{symmetrisch}): $N\bigl(T(y) - T(z)\bigr)
\leq \frac12 N(y - z)$: $T$ is een $\frac12$-contractie van de
volledige ruimte $(E, N)$. De vastepuntsstelling van Banach
(\cref{thm:b2:metric:banach}) levert een uniek vast punt: de
unieke oplossing.
\end{proof}

\begin{remark}
Is $f$ slechts van klasse $C^1$ (lokaal lipschitz), dan geldt de
stelling \emph{lokaal}, met een maximale oplossing op een maximaal
\omterm{def:b2:metric:topology}{open} interval; oplossingen kunnen in eindige tijd exploderen ($y' =
y^2$, $y(0) = 1$: $y(t) = \frac{1}{1-t}$, verdwenen in $t = 1$). De
globale lipschitzvoorwaarde is wat het hele segment koopt. Twee
gevolgen die het onthouden waard zijn: oplossingskrommen van een
differentiaalvergelijking met lipschitzveld \emph{snijden elkaar
nooit}; en de nulfunctie is de enige oplossing van een lineaire
homogene vergelijking die ergens verdwijnt.
\end{remark}

\begin{example}[Eenduidigheid is een stelling: een veld dat lekt]\label{ex:b2:diffeq:leaky}
Beschouw $y' = 2\sqrt{\abs y}$ met $y(0) = 0$. De nulfunctie lost
haar op; en ook
\[
y(t) = \begin{cases} 0 & t \leq 0,\\ t^2 & t \geq 0,
\end{cases}
\]
die van klasse $C^1$ is (beide stukken hebben afgeleide $0$ in het
lijmpunt) en voldoet aan $y'(t) = 2t = 2\sqrt{t^2}$ voor $t > 0$
--- sterker nog, het opstijgen uitstellen geeft een oplossing voor
\emph{elk} losmoment $c \geq 0$: oneindig veel oplossingen door
dezelfde beginwaarden. Geen tegenspraak met
\cref{thm:b2:diffeq:cauchylipschitz}: nabij $y = 0$ is
\[
\frac{\abs{2\sqrt y - 2\sqrt z}}{\abs{y - z}}
= \frac{2}{\sqrt y + \sqrt z} \longrightarrow +\infty ,
\]
het veld is niet lipschitz in $y$, en de stelling zwijgt. Het
inzicht om te onthouden: de fysische lezing is een emmer die
onder de zwaartekracht leegloopt, achterstevoren afgespeeld ---
vanuit de lege toestand kan men niet zeggen wanneer het vullen
begon; het determinisme van differentiaalvergelijkingen is precies
de lipschitzvoorwaarde, geen natuurwet.
\end{example}

\section{Lineaire stelsels}

\begin{theorem}[Structuur van de lineaire stelsels]\label{thm:b2:diffeq:linear}
Zijn $A \colon I \to \mathcal{M}_n(\R)$ en $B \colon I \to \R^n$
\omterm{def:b2:metric:continuity}{continu} op een interval $I$. Dan heeft voor elke $(t_0, X_0)$ het
probleem
\[
X' = A(t)X + B(t), \qquad X(t_0) = X_0
\]
precies één oplossing op \emph{heel} $I$. De oplossingen van het
homogene stelsel ($B = 0$) vormen een vectorruimte $\mathcal{S}_H$
van dimensie precies $n$, en de evaluatie $X \mapsto X(t_0)$ is een
isomorfisme $\mathcal{S}_H \to \R^n$; algemene oplossing $=$
particuliere $+$ homogene.
\end{theorem}

\begin{proof}
Op elk segment $J \subseteq I$ dat $t_0$ bevat is $f(t, X) = A(t)X
+ B(t)$ \omterm{def:b2:metric:continuity}{continu}, en lipschitz in $X$ met constante $k = \sup_J
\vertiii{A(t)}$ (eindig: \omterm{def:b2:metric:continuity}{continu} op een segment):
\cref{thm:b2:diffeq:cauchylipschitz} is van toepassing op $J$;
laten we $J$ heel $I$ uitputten, dan lijmt de eenduidigheid de
oplossingen aaneen tot één oplossing op $I$. De lineariteit van de
oplossingsverzameling en van de evaluatieafbeelding is duidelijk;
de evaluatie is bijectief wegens het bestaan (surjectief) en de
eenduidigheid (injectief): $\dim \mathcal{S}_H = n$. De affiene
structuur is woordelijk het argument uit bachelorjaar 1.
\end{proof}

\begin{example}[Het evaluatie-isomorfisme, concreet]\label{ex:b2:diffeq:evaliso}
Voor $y'' + y = 0$, opgevat als het stelsel $X' = \begin{pmatrix} 0
& 1\\ -1 & 0\end{pmatrix}X$ met $X = (y, y')$: de stelling zegt dat
de oplossingsruimte een vlak is en dat $X \mapsto X(0) = (y(0),
y'(0))$ een isomorfisme op $\R^2$ is. De oplossingen $\cos$ en
$\sin$ evalueren tot $(1, 0)$ en $(0, 1)$ --- de canonieke basis
van $\R^2$ --- dus vormen zij een basis van de oplossingsruimte, en
\emph{elke} oplossing is
\[
y(t) = y(0)\cos t + y'(0)\sin t ,
\]
met de coëfficiënten rechtstreeks van de beginwaarden afgelezen,
zonder enig lineair stelsel op te lossen. Het inzicht om te
onthouden: het \omterm{def:b2:diffeq:wronskian}{fundamentaalstelsel} kiezen waarvan de beginwaarden
de canonieke basis vormen (hier $\cos, \sin$) is precies de
kolommen van $\eu^{tA}$ kiezen; het evaluatie-isomorfisme is de
reden waarom beginvoorwaarden banen parametriseren --- de
meetkundige inhoud van ``deterministische dynamica'' voor lineaire
vergelijkingen.
\end{example}

\begin{definition}[Wronskiaan]\label{def:b2:diffeq:wronskian}
Voor oplossingen $X_1, \dots, X_n$ van het homogene stelsel is de
\emph{wronskiaan}\index{wronskiaan} $W(t) = \det\bigl(X_1(t),
\dots, X_n(t)\bigr)$. Volgens het isomorfisme hierboven verdwijnt
$W$ ofwel identiek (de familie is afhankelijk) ofwel nooit (een
\emph{fundamentaalstelsel}\index{fundamentaalstelsel});
kwantitatief is $W' = \operatorname{tr}\bigl(A(t)\bigr) W$, dus
\[
W(t) = W(t_0)\,\exp\Bigl(\int_{t_0}^{t}
\operatorname{tr} A(s)\,\dd s\Bigr)
\quad \text{(formule van Liouville)}.
\]
\end{definition}

\begin{example}[Liouville nagegaan op een euler-vergelijking]\label{ex:b2:diffeq:eulerwronskian}
Op $\intoo{0}{\infty}$ heeft de vergelijking $t^2y'' + ty' - y = 0$
de oplossingen $y_1(t) = t$ en $y_2(t) = \frac1t$ (substitueer).
Hun \omterm{def:b2:diffeq:wronskian}{wronskiaan}:
\[
W(t) = \det\begin{pmatrix} t & \tfrac1t\\[2pt]
1 & -\tfrac{1}{t^2}\end{pmatrix}
= -\frac1t - \frac1t = -\frac2t ,
\]
nooit nul: een \omterm{def:b2:diffeq:wronskian}{fundamentaalstelsel}. Ga nu Liouville na: in
genormaliseerde vorm $y'' + \frac1t\,y' - \frac{1}{t^2}\,y = 0$
heeft de begeleidende matrix $A(t) = \begin{pmatrix} 0 & 1\\
\frac{1}{t^2} & -\frac1t\end{pmatrix}$ spoor $-\frac1t$, dus
\[
W(t) = W(1)\exp\Bigl(-\int_1^t\frac{\dd s}{s}\Bigr)
= -2\,\eu^{-\ln t} = -\frac2t . \checkmark
\]
Het inzicht om te onthouden: Liouville voorspelt de \emph{vorm} van
de \omterm{def:b2:diffeq:wronskian}{wronskiaan} nog voordat er één oplossing bekend is --- hier dat
$W$ gelijk moet zijn aan $\frac{c}{t}$; dat is wat de methode van
de orderverlaging aandrijft (\cref{prop:b2:diffeq:secondorder}),
waar het kennen van $y_1$ en van de vorm van de \omterm{def:b2:diffeq:wronskian}{wronskiaan} $y_2$
met één kwadratuur vastlegt.
\end{example}

\begin{proof}[Bewijs van de formule van Liouville]
$W(t) = \det M(t)$ met $M' = AM$. Differentiëren we de \omterm{def:b2:linalg:det}{determinant}
als multilineaire functie van de kolommen, dan is
\[
W'(t) = \sum_j \det(X_1, \dots, X_j', \dots, X_n)
= \sum_j \det(X_1, \dots, AX_j, \dots, X_n) .
\]
Nu is de afbeelding $(C_1, \dots, C_n) \mapsto \sum_j \det(C_1,
\dots, AC_j, \dots, C_n)$ $n$-lineair en alternerend (met twee
gelijke kolommen $C_i = C_k$ verdwijnen de termen $j \notin \{i,
k\}$ ronduit, en de termen $j = i$ en $j = k$ heffen elkaar
paarsgewijs op na één kolomverwisseling): volgens de
eenduidigheidsstelling (\cref{thm:b2:linalg:detspace}) is zij $c
\cdot \det$, met $c$ afgelezen op de canonieke kolommen: $c =
\sum_j \det(e_1, \dots, Ae_j, \dots, e_n) = \sum_j a_{jj} =
\operatorname{tr} A$. Bijgevolg is $W' =
\operatorname{tr}\bigl(A(t)\bigr)W$: een scalaire lineaire
differentiaalvergelijking, opgelost door de formule uit
bachelorjaar 1.
\end{proof}

\section{Constante coëfficiënten: de matrixexponentiaal}

\begin{theorem}\label{thm:b2:diffeq:matrixexp}
Voor $A \in \mathcal{M}_n(\R)$ (of $\C$) voldoet de exponentiaal
$\eu^{tA} = \sum_k \frac{(tA)^k}{k!}$
(\cref{ex:b2:nvs:matrixexp}) hieraan: $t \mapsto \eu^{tA}$ is van
klasse $C^1$ (zelfs $C^\infty$) met\index{matrixexponentiaal}
\[
\frac{\dd}{\dd t}\,\eu^{tA} = A\,\eu^{tA} = \eu^{tA}A ,
\qquad
\eu^{(s+t)A} = \eu^{sA}\,\eu^{tA},
\qquad
(\eu^{A})^{-1} = \eu^{-A} ;
\]
en $\eu^{A + B} = \eu^A\eu^B$ \emph{wanneer} $AB = BA$. Het
beginwaardeprobleem $X' = AX$, $X(0) = X_0$ heeft de unieke
oplossing $X(t) = \eu^{tA}X_0$; met een bronterm geldt de formule
van de \emph{variatie van de constanten}:
\[
X(t) = \eu^{(t - t_0)A}X_0 + \int_{t_0}^{t} \eu^{(t-s)A}B(s)\,\dd
s .
\]
\end{theorem}

\begin{proof}
\emph{\omterm{def:b2:diffcalc:differential}{Differentieerbaarheid}:} de reeks $\sum \frac{t^kA^k}{k!}$ en
haar termsgewijs afgeleide reeks $\sum \frac{t^{k-1}A^k}{(k-1)!} =
A\sum \frac{(tA)^{k-1}}{(k-1)!}$ convergeren normaal op elk segment
(\omterm{def:b2:nvs:norm}{normen} $\leq \frac{(\abs t\,\vertiii A)^k}{k!}$): differentieer
term voor term (\cref{thm:b2:funcseq:seriestransfer}, vectorwaardig).
De twee volgorden $A\eu^{tA}$ en $\eu^{tA}A$ stemmen overeen omdat
elke partiaalsom met $A$ commuteert.

\emph{Groepswet:} voor commuterende $A, B$ herschikt het
\omterm{thm:b2:series:fubini}{Cauchy-product} van de twee exponentiële reeksen zich volgens het
binomium precies als in \cref{ex:b2:series:exp} (de \omterm{def:b2:series:def}{absolute
convergentie} in de banachalgebra rechtvaardigt dit): $\eu^{A+B} =
\eu^A\eu^B$; met $B = sA$ geeft dit de groepswet met één parameter,
en $B = -A$ de inverse.

\emph{Beginwaardeprobleem:} $X(t) = \eu^{tA}X_0$ lost het op
(differentieer); eenduidigheid volgens
\cref{thm:b2:diffeq:linear}. Variatie van de constanten: zet $Y(t)
= \eu^{-tA}X(t)$; differentiëren geeft $Y' = \eu^{-tA}(X' - AX) =
\eu^{-tA}B(t)$; integreer van $t_0$ tot $t$ en vermenigvuldig weer
met $\eu^{tA}$.
\end{proof}

\begin{method}[$\eu^{tA}$ berekenen]\label{met:b2:diffeq:computeexp}
Herleid $A$ (\cref{ch:b2:reduction}): is $A = PDP^{-1}$ met $D$
diagonaal, dan is $\eu^{tA} = P\,\eu^{tD}P^{-1}$ met $\eu^{tD}$
diagonaal met elementen $\eu^{t\lambda_i}$; gebruik in het algemeen
Dunford $A = D + N$ (commuterend): $\eu^{tA} = \eu^{tD}\,\eu^{tN}$
met $\eu^{tN}$ een \emph{veelterm} in $t$ (de nilpotentie kapt de
reeks af). Complexe \omterm{def:b2:reduction:eigen}{eigenwaarden} groeperen zich tot blokken van het
type rotatie maal exponentiaal (\cref{exo:b2:diffeq:5}).
\end{method}

\begin{remark}[Klassieke valkuilen]
\emph{(i) $\eu^{A+B} \neq \eu^A\eu^B$ zonder commutatie:} neem $A =
\begin{pmatrix} 0 & 1\\ 0 & 0\end{pmatrix}$ en $B =
\begin{pmatrix} 0 & 0\\ 1 & 0\end{pmatrix}$. Dan is $\eu^A = I +
A$ en $\eu^B = I + B$ (nilpotentie), dus
\[
\eu^A\eu^B = \begin{pmatrix} 2 & 1\\ 1 & 1\end{pmatrix},
\qquad\text{terwijl}\qquad
\eu^{A+B} = \cosh(1)\,I + \sinh(1)\,(A + B)
= \begin{pmatrix} \cosh 1 & \sinh 1\\ \sinh 1 & \cosh 1
\end{pmatrix},
\]
met $(A+B)^2 = I$; en $\cosh 1 \approx 1.54 \neq 2$. De groepswet
van \cref{thm:b2:diffeq:matrixexp} draagt een echte hypothese.
\emph{(ii) Niet-lineaire intuïtie op lineair terrein:} de
oplossingen van een \emph{lineair} stelsel met \omterm{def:b2:metric:continuity}{continue}
coëfficiënten leven op het hele interval
(\cref{thm:b2:diffeq:linear}) --- explodeert een
kandidaat-oplossing binnen $I$, dan was de vergelijking niet
lineair of is de berekening fout; omgekeerd mag men voor
niet-lineaire vergelijkingen nooit globaliteit beloven zonder
argument ($y' = y^2$). \emph{(iii) Delen door de onbekende:} bij
het scheiden van de veranderlijken in $y' = y(1-y)$ verdwijnen
stilzwijgend de constante oplossingen $0$ en $1$ --- precies die
welke de fasenlijn ordenen (\cref{exo:b2:diffeq:3}); som constante
oplossingen eerst op. \emph{(iv) Beginwaarden leggen vectoren vast,
geen scalairen:} een scalaire vergelijking van orde $n$ heeft $n$
voorwaarden nodig ($y, y', \dots$ in $t_0$); alleen $y(t_0)$
opleggen laat een familie met $n-1$ parameters over, een klassieke
bron van ``verloren'' constanten.
\end{remark}

\begin{example}[Een $3\times3$-exponentiaal via Dunford]\label{ex:b2:diffeq:dunford3}
Los $X' = AX$ op voor $A = \begin{pmatrix} 2 & 1 & 0\\ 0 & 2 & 0\\
0 & 0 & 3\end{pmatrix}$. Dunford per blok: $A = D + N$ met $D =
\operatorname{diag}(2, 2, 3)$ en $N = E_{12}$, die commuteren ($N$
leeft binnen het blok bij \omterm{def:b2:reduction:eigen}{eigenwaarde} $2$) en met $N^2 = 0$:
\[
\eu^{tA} = \eu^{tD}\,\eu^{tN}
= \begin{pmatrix}
\eu^{2t} & t\,\eu^{2t} & 0\\
0 & \eu^{2t} & 0\\
0 & 0 & \eu^{3t}
\end{pmatrix} .
\]
De algemene oplossing leest men kolom voor kolom af: $X(t) =
\bigl(\eu^{2t}(x_0 + ty_0),\ \eu^{2t}y_0,\ \eu^{3t}z_0\bigr)$.
Verstandscontroles: in $t = 0$ is de matrix $I$; haar \omterm{def:b2:linalg:det}{determinant}
is $\eu^{7t} = \eu^{t\operatorname{tr}A}$, zoals Liouville eist; en
de factor $t$ verschijnt precies waar de \omterm{def:b2:reduction:eigen}{eigenwaarde} $2$ defect is.
Het inzicht om te onthouden: veeltermen maal exponentiëlen zijn
geen gok die men uit het hoofd leert --- zij zijn de afgekapte
reeks $\eu^{tN}$, en hun graad wordt door de nilpotentie-index
begrensd, nooit meer.
\end{example}

\begin{method}[$X' = AX + B(t)$ oplossen, van begin tot eind]\label{met:b2:diffeq:solvelinear}
\begin{enumerate}
  \item Spectrum van $A$; dan $\eu^{tA}$ volgens
        \cref{met:b2:diffeq:computeexp} (diagonaliseer; of Dunford
        als in \cref{ex:b2:diffeq:dunford3}; of een veeltermtruc
        zoals $A^2 = -I$).
  \item Een particuliere oplossing: de variatie van de constanten
        $\int_{t_0}^t\eu^{(t-s)A}B(s)\dd s$ werkt altijd; voor
        $B$ van exponentieel-veeltermtype is een aanzet van
        dezelfde vorm (met verhoogde graad bij \omterm{pb:b2:diffeq:1}{resonantie},
        \cref{exo:b2:diffeq:10}) sneller.
  \item Algemene oplossing $= \eu^{(t-t_0)A}X_0 +$ particuliere;
        pas de beginwaarden \emph{als laatste} aan, op de
        volledige formule.
  \item Verstandscontroles: $X(t_0)$ klopt; de groei van het
        homogene deel stemt overeen met de reële delen van de
        \omterm{def:b2:reduction:eigen}{eigenwaarden} (\cref{exo:b2:diffeq:8}); en de $\det$ van
        een fundamentaalmatrix gehoorzaamt aan Liouville.
\end{enumerate}
\end{method}

\begin{example}[Een fasenportret]\label{ex:b2:diffeq:phaseportrait}
$X' = AX$ met $A = \begin{pmatrix} 0 & 1\\ -1 & 0\end{pmatrix}$:
$A^2 = -I$, dus splitst de reeks zich als
\[
\eu^{tA} = (\cos t)\,I + (\sin t)\,A
= \begin{pmatrix} \cos t & \sin t\\ -\sin t & \cos t
\end{pmatrix} :
\]
de banen zijn cirkels die met de wijzers van de klok mee worden
doorlopen --- de harmonische oscillator $x'' + x = 0$ in de kledij
van de eerste orde. \omterm{def:b2:reduction:eigen}{Eigenwaarden} $\pm\iu$ op de imaginaire as: een
\emph{centrum}. Algemener beslissen de reële delen van de
\omterm{def:b2:reduction:eigen}{eigenwaarden} van $A$ over de groei of het verval van $\norm{X(t)}$
(\cref{exo:b2:diffeq:8}).
\end{example}

niet-lineair is: nabij een evenwicht gedraagt een glad

linearisering, waarvan het spoor-determinantvlak het portret
klasseert. De weekendopgave werkt het verhaal van de oscillator

de wiskunde achter schokdempers, wisselstroomkringen en spectrale
kloven tegelijk. Het volume van bachelorjaar 3 keert terug met de
kwalitatieve theorie (stromingen, stabiliteit, eerste integralen)
op variëteiten.
\end{remark}

\section{Tweede orde met variabele coëfficiënten}

\begin{proposition}\label{prop:b2:diffeq:secondorder}
De vergelijking $y'' + a(t)y' + b(t)y = c(t)$ ($a, b, c$ \omterm{def:b2:metric:continuity}{continu} op
$I$) is het stelsel $X' = A(t)X + B(t)$ voor $X = (y, y')$: er
bestaan oplossingen, en zij zijn uniek op heel $I$ voor
willekeurige beginwaarden $(y(t_0), y'(t_0))$; de homogene
oplossingen vormen een vlak. Is één homogene oplossing $y_1$ bekend
die nergens verdwijnt, dan vindt men een tweede onafhankelijke
oplossing door \emph{orderverlaging}: de substitutie $y = y_1 z$
maakt van de homogene vergelijking een vergelijking van eerste orde
voor $z'$, die met kwadraturen wordt opgelost.
\end{proposition}

\begin{proof}
De vorm als stelsel en \cref{thm:b2:diffeq:linear} geven alles wat
structureel is. Orderverlaging: substitueren we $y = y_1z$, dan is
\[
y_1 z'' + (2y_1' + a y_1)z' + \underbrace{(y_1'' + ay_1' +
by_1)}_{=\,0}\,z = 0 :
\]
een lineaire vergelijking van eerste orde in $u = z'$, oplosbaar
met de formule uit bachelorjaar 1; $u$ integreren geeft $z$ en dus
$y_2 = y_1 z$, onafhankelijk van $y_1$ zodra $z$ niet constant is.
\end{proof}

\begin{example}\label{ex:b2:diffeq:lowering}
$t^2y'' - 2y = 0$ op $\intoo{0}{\infty}$: $y_1 = t^2$ is een
oplossing. Substitueer $y = t^2z$: uit $y' = t^2z' + 2tz$ en $y'' =
t^2z'' + 4tz' + 2z$ volgt
\[
t^2y'' - 2y = t^4 z'' + 4t^3z' = 0,
\qquad\text{dat wil zeggen}\qquad \frac{z''}{z'} = -\frac4t :
\]
$z' = t^{-4}$ (op een constante na), $z = -\frac{1}{3t^3}$, en $y_2
= t^2z = -\frac{1}{3t}$. Algemene oplossing: $y = \alpha t^2 +
\frac{\beta}{t}$.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b2:diffeq:1}
Los $X' = AX$, $X(0) = (1, 0)^{\mathsf T}$ op voor $A =
\begin{pmatrix} 1 & 1\\ 0 & 2\end{pmatrix}$ (diagonaliseer) en $A =
\begin{pmatrix} 2 & 1\\ 0 & 2 \end{pmatrix}$ (Dunford).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:diffeq:2}
Welke beginwaardeproblemen hebben volgens
\cref{thm:b2:diffeq:cauchylipschitz} unieke globale oplossingen op
$\R$? $y' = \sin(ty)$; $\;y' = y^2$; $\;y' = \abs y$. Los het
laatste expliciet op met $y(0) = 0$ en met $y(0) = 1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:diffeq:3}
Bewijs dat twee verschillende maximale oplossingen van $y' =
f(t,y)$ ($f$ lipschitz in $y$) nooit op hetzelfde tijdstip dezelfde
waarde aannemen, en leid af dat oplossingen van $y' = y(1 - y)$ die
in $\intoo{0}{1}$ beginnen, voor altijd in $\intoo{0}{1}$ blijven.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:4}
Bereken $\eu^{tA}$ voor $A = \begin{pmatrix} 3 & 1\\ -1 &
1\end{pmatrix}$ \emph{(Dunford: $(A - 2I)^2 = 0$)}, en los $X' = AX
+ \begin{pmatrix} \eu^{2t}\\ 0\end{pmatrix}$, $X(0) = 0$, op met de
variatie van de constanten.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:5}
Bewijs voor $A = \begin{pmatrix} \alpha & -\beta\\ \beta &
\alpha\end{pmatrix}$ dat $\eu^{tA} =
\eu^{\alpha t}\begin{pmatrix} \cos\beta t & -\sin\beta t\\
\sin\beta t & \cos\beta t\end{pmatrix}$ --- spiraalvormige banen
--- op twee manieren: via de reeks (schrijf $A = \alpha I + \beta
J$, $J^2 = -I$), en via de complexe identificatie $z' = (\alpha +
\iu\beta)z$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:6}
(Lemma van Gronwall) Zij $u$ \omterm{def:b2:metric:continuity}{continu} en niet-negatief met $u(t)
\leq C + k\int_{t_0}^{t} u(s)\,\dd s$ op $\intco{t_0}{T}$. Bewijs
dat $u(t) \leq C\,\eu^{k(t - t_0)}$ \emph{(differentieer $v(t) =
\eu^{-kt}\int_{t_0}^t u$)}. Leid opnieuw de eenduidigheid in
Cauchy--Lipschitz af, en de \omterm{def:b2:metric:continuity}{continue} afhankelijkheid $\norm{y(t) -
z(t)} \leq \norm{y_0 - z_0}\,\eu^{k\abs{t - t_0}}$ voor twee
oplossingen met verschillende beginwaarden.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:7}
Bepaal, wetend dat $y_1(t) = \frac{\sin t}{t}$ de vergelijking
$ty'' + 2y' + ty = 0$ op $\intoo{0}{\pi}$ oplost, een tweede
onafhankelijke oplossing door orderverlaging, en geef de algemene
oplossing.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffeq:8}
Zij $A \in \mathcal{M}_n(\C)$ met alle \omterm{def:b2:reduction:eigen}{eigenwaarden} van (strikt)
negatief reëel deel. Bewijs dat elke oplossing van $X' = AX$ naar
$0$ streeft als $t \to +\infty$, met een exponentieel tempo:
$\norm{X(t)} \leq C\,\eu^{-\alpha t}$ voor een zekere $\alpha > 0$.
\emph{(Trigonaliseer; behandel het driehoekige stelsel vanaf de
laatste rij naar boven, of gebruik Dunford: $\eu^{tA} =
\eu^{tD}\eu^{tN}$ met $\norm{\eu^{tD}} \leq \eu^{-\alpha' t}$ en
$\eu^{tN}$ een veelterm in $t$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffeq:9}
(Geen ontsnapping in eindige tijd bij lineaire groei) Neem aan dat
$f$ \omterm{def:b2:metric:continuity}{continu} is met $\norm{f(t, y)} \leq a\norm y + b$ op
$\intco{0}{\infty} \times \R^n$, en lokaal lipschitz in $y$. Bewijs
met Gronwall (\cref{exo:b2:diffeq:6}) op de integraalvorm dat de
maximale oplossingen globaal zijn (gedefinieerd op heel
$\intco{0}{\infty}$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:diffeq:10}
Los $y'' - 3y' + 2y = \eu^{t}$ op: homogene oplossingen, dan een
particuliere oplossing van de vorm $\alpha t\,\eu^{t}$ \emph{(waarom
faalt de naïeve gok $\alpha\eu^t$?)}; algemene oplossing en de
oplossing met $y(0) = y'(0) = 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:11}
Bereken $\eu^{tA}$ voor het jordanblok
\[
A = \begin{pmatrix} \lambda & 1 & 0\\ 0 & \lambda & 1\\
0 & 0 & \lambda\end{pmatrix},
\]
en beschrijf alle oplossingen van $X' = AX$: exponentiëlen maal
veeltermvectoren, met graden tot en met $2$. Waar komt de
veeltermgraad vandaan?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffeq:12}
(Periodieke aandrijving, periodiek antwoord) Zij $A \in
\mathcal{M}_n(\R)$ en $B \colon \R \to \R^n$ \omterm{def:b2:metric:continuity}{continu} en
$T$-periodiek.
\begin{enumerate}
  \item Toon aan dat een oplossing van $X' = AX + B(t)$
        $T$-periodiek is dan en slechts dan als $X(T) = X(0)$
        \emph{(vergelijk $X(\cdot + T)$ met $X$)}.
  \item Toon aan dat de \omterm{def:b2:reduction:eigen}{eigenwaarden} van $\eu^{TA}$ de getallen
        $\eu^{T\lambda}$ zijn met $\lambda \in
        \operatorname{Sp}A$ \emph{(trigonaliseer over $\C$)}. Leid
        af: ligt geen enkele \omterm{def:b2:reduction:eigen}{eigenwaarde} van $A$ in
        $\frac{2\iu\pi}{T}\Z$, dan is $I - \eu^{TA}$
        inverteerbaar.
  \item Bewijs onder die hypothese dat het stelsel precies één
        $T$-periodieke oplossing heeft, met
        \[
        X(0) = \bigl(I - \eu^{TA}\bigr)^{-1}
        \int_0^{T}\eu^{(T-s)A}B(s)\,\dd s .
        \]
        Waarmee stemt het uitgesloten geval overeen voor de
        harmonische oscillator? (De weekendopgave antwoordt:
        \omterm{pb:b2:diffeq:1}{resonantie}.)
\end{enumerate}
\end{exercise}

\section{Probleem: trillingen, resonantie en de
vergelijkingsstellingen van Sturm}

\begin{problem}\label{pb:b2:diffeq:1}
Eén vergelijking regeert de mechanische en de elektrische wereld:
\[
x'' + 2\zeta\omega\,x' + \omega^2 x = F(t),
\qquad \omega > 0,\ \zeta \geq 0 .
\]
Deze opgave bestudeert haar \omterm{def:b2:metric:complete}{volledig} --- via de klassering van de
vlakke lineaire stelsels in het spoor-determinantvlak, de drie
dempingsregimes, het antwoord in stationaire toestand op een
periodieke aandrijving met haar
\emph{resonantiepiek}\index{resonantie} en de resonantiecatastrofe
--- en verlaat daarna de constante coëfficiënten voor de
\emph{scheidings- en vergelijkingsstellingen van
Sturm}\index{vergelijkingsstelling van Sturm}, die de nulpunten van
de oplossingen van $y'' + q(t)y = 0$ beheersen zonder ook maar
enige formule.

\medskip
\noindent\textbf{Deel I --- Het spoor-determinantvlak.} Zij $A \in
\mathcal M_2(\R)$, $\tau = \operatorname{tr}A$, $\delta = \det A$
en $\Delta = \tau^2 - 4\delta$.
\begin{enumerate}
  \item Toon aan dat de \omterm{def:b2:reduction:eigen}{eigenwaarden} van $A$ gelijk zijn aan
        $\frac{\tau\pm\sqrt\Delta}{2}$ en klasseer: twee reële
        \omterm{def:b2:reduction:eigen}{eigenwaarden} van tegengesteld teken dan en slechts dan als
        $\delta < 0$; reële \omterm{def:b2:reduction:eigen}{eigenwaarden} van hetzelfde teken dan en
        slechts dan als $\delta > 0$ en $\Delta \geq 0$ (teken van
        $\tau$); een paar niet-reële toegevoegde \omterm{def:b2:reduction:eigen}{eigenwaarden} dan
        en slechts dan als $\Delta < 0$ (reëel deel $\frac\tau2$).
  \item (Zadelpunt, $\delta < 0$) Schrijf met \omterm{def:b2:reduction:eigen}{eigenwaarden} $\mu < 0
        < \lambda$ en \omterm{def:b2:reduction:eigen}{eigenvectoren} $v_\pm$ de algemene oplossing
        en beschrijf de banen: twee stabiele en twee instabiele
        halfrechten, en alle overige banen asymptotisch aan
        beide. Waarom kan geen enkele oplossing behalve $0$ op
        heel $\R$ begrensd blijven?
  \item (Knopen, $\delta > 0$, $\Delta > 0$) Toon voor $\mu <
        \lambda < 0$ aan dat elke oplossing ongelijk aan nul naar
        $0$ streeft, en dat alle banen behalve die op de snelle as
        \emph{rakend aan de trage eigenrichting} aankomen
        \emph{(vergelijk $\eu^{\mu t}$ en $\eu^{\lambda t}$)}.
  \item (Spiralen en centra, $\Delta < 0$) Beschrijf, met de
        \omterm{def:b2:reduction:eigen}{eigenwaarden} geschreven als $\alpha \pm \iu\beta$, de banen
        met behulp van \cref{exo:b2:diffeq:5} (na een reële
        basisverandering, in die algemeenheid aangenomen of
        bewezen voor de stelsels van Deel II, die hieronder worden
        gebruikt): spiralen die convergeren voor $\alpha =
        \frac\tau2 < 0$, divergeren voor $\tau > 0$, en gesloten
        krommen (centrum) voor $\tau = 0$.
  \item (Grensgevallen) Toon voor een dubbele \omterm{def:b2:reduction:eigen}{eigenwaarde} ($\Delta
        = 0$) aan dat $\eu^{tA} = \eu^{\lambda t}(I + tN)$ met $N
        = A - \lambda I$ nilpotent, en onderscheid de ster ($N =
        0$) van de oneigenlijke knoop ($N \neq 0$). Vat Deel I
        samen in het spoor-determinantbeeld uit de figuur van dit
        hoofdstuk.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De gedempte oscillator.} Nu is $F =
0$: $x'' + 2\zeta\omega x' + \omega^2x = 0$, dat wil zeggen $X' =
AX$ met $A = \begin{pmatrix} 0 & 1\\ -\omega^2 &
-2\zeta\omega\end{pmatrix}$.
\begin{enumerate}[resume]
  \item Bereken $\tau, \delta, \Delta$ en plaats de drie regimes in
        het spoor-determinantvlak: \emph{onderkritisch gedempt} $0
        < \zeta < 1$ (stabiele spiraal), \emph{kritisch gedempt}
        $\zeta = 1$ (dubbele \omterm{def:b2:reduction:eigen}{eigenwaarde}), \emph{overkritisch
        gedempt} $\zeta > 1$ (stabiele knoop); $\zeta = 0$ is het
        centrum.
  \item Los de drie regimes expliciet op:
        \[
        \zeta < 1:\ \eu^{-\zeta\omega t}\bigl(a\cos\omega_d t
        + b\sin\omega_dt\bigr),\ \omega_d =
        \omega\sqrt{1-\zeta^2};
        \qquad
        \zeta = 1:\ (a + bt)\,\eu^{-\omega t};
        \]
        $\zeta > 1$: twee reële exponentiëlen. Definieer de
        pseudoperiode $\frac{2\pi}{\omega_d}$ en toon aan dat de
        verhouding van opeenvolgende maxima van $\abs x$ de
        constante $\eu^{-2\pi\zeta/\sqrt{1-\zeta^2}}$ is (het
        logaritmische decrement).
  \item (Het principe van de deurdranger) Voor $\zeta \geq 1$ wordt
        het verval bestuurd door de traagste \omterm{def:b2:reduction:eigen}{eigenwaarde}
        $\lambda_{\mathrm{traag}} = -\omega\bigl(\zeta -
        \sqrt{\zeta^2-1}\bigr)$. Toon aan dat
        $\abs{\lambda_{\mathrm{traag}}} = \frac{\omega}{\zeta +
        \sqrt{\zeta^2 - 1}}$ een \emph{dalende} functie van $\zeta
        \geq 1$ is: de kritische demping $\zeta = 1$ geeft de
        snelste niet-oscillerende terugkeer naar rust.
  \item (Energie) Zij $E(t) = \frac12x'^2 + \frac12\omega^2x^2$.
        Bewijs dat $E' = -2\zeta\omega\,x'^2 \leq 0$, en leid af
        dat de vergelijking voor $\zeta > 0$ geen periodieke
        oplossing heeft behalve nul \emph{(een periode zou $E$
        constant afdwingen, en dus $x' \equiv 0$)}.
  \item Leg in twee zinnen uit waarom het centrum $\zeta = 0$
        \emph{structureel breekbaar} is: elke $\zeta > 0$, hoe
        klein ook, vernietigt de periodiciteit --- en waar dat in
        het spoor-determinantvlak te zien is (de lijn van de centra
        heeft een leeg inwendige).
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Gedwongen trillingen en \omterm{pb:b2:diffeq:1}{resonantie}.}
Nu is $F(t) = F\cos(\gamma t)$ met $F, \gamma > 0$.
\begin{enumerate}[resume]
  \item ($\zeta > 0$: de stationaire toestand) Zoek $x_p =
        \Re\bigl(z\,\eu^{\iu\gamma t}\bigr)$: toon aan dat
        \[
        z = \frac{F}{\omega^2 - \gamma^2 +
        2\iu\zeta\omega\gamma},
        \qquad
        A(\gamma) := \abs z = \frac{F}{\sqrt{(\omega^2 -
        \gamma^2)^2 + 4\zeta^2\omega^2\gamma^2}} ,
        \]
        en schrijf $x_p = A(\gamma)\cos(\gamma t - \varphi)$ met
        $\tan\varphi =
        \frac{2\zeta\omega\gamma}{\omega^2-\gamma^2}$.
  \item Toon aan dat \emph{elke} oplossing gelijk is aan $x_p$ plus
        een overgangsverschijnsel uit Deel II, dat naar $0$
        streeft: wat de beginwaarden ook zijn, het systeem
        vergrendelt zich op de stationaire toestand --- amplitude
        $A(\gamma)$, fase-achterstand $\varphi$.
  \item (De resonantiekromme) Maximaliseer $A$: toon aan dat
        $A(\gamma)$ een inwendig maximum heeft dan en slechts dan
        als $\zeta < \frac{1}{\sqrt2}$, in
        \[
        \gamma_* = \omega\sqrt{1 - 2\zeta^2},
        \qquad
        A(\gamma_*) =
        \frac{F}{2\zeta\omega^2\sqrt{1-\zeta^2}} ,
        \]
        en dat de piek voor kleine $\zeta$ het statische antwoord
        $A(0) = \frac F{\omega^2}$ versterkt met de factor $\approx
        \frac{1}{2\zeta}$.
  \item ($\zeta = 0$, buiten \omterm{pb:b2:diffeq:1}{resonantie}) Toon voor $\gamma \neq
        \omega$ aan dat de oplossing met $x(0) = x'(0) = 0$ gelijk
        is aan
        \[
        x(t) = \frac{F}{\omega^2 -
        \gamma^2}\bigl(\cos\gamma t - \cos\omega t\bigr)
        = \frac{2F}{\omega^2-\gamma^2}
        \sin\frac{(\omega-\gamma)t}{2}
        \sin\frac{(\omega+\gamma)t}{2} :
        \]
        begrensd, met \emph{zwevingen} --- een snelle trilling
        onder een trage omhullende --- wanneer $\gamma$ dicht bij
        $\omega$ ligt.
  \item ($\zeta = 0$, \omterm{pb:b2:diffeq:1}{resonantie}) Toon voor $\gamma = \omega$ aan
        dat $x_p(t) = \frac{F}{2\omega}\,t\sin(\omega t)$ een
        oplossing is, en vind haar terug als de limiet van vraag 14
        voor $\gamma \to \omega$: de amplitude groeit voor altijd
        lineair --- de resonantiecatastrofe.
  \item (Verband met Fourier) Een algemene periodieke aandrijving
        ontbindt in harmonischen (het hoofdstuk over
        \omterm{def:b2:fourier:coefficients}{Fourierreeksen}); wegens de lineariteit is de stationaire
        toestand de som van de harmonische antwoorden. Welke
        harmonische resoneert bij een ongedempte oscillator met
        frequentie $\omega = 3$, aangedreven door een blokgolf van
        het type uit \cref{exo:b2:fourier:1} (harmonischen op alle
        oneven gehele getallen)? Eén zin over waarom ingenieurs
        blokgolven vrezen.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De stellingen van Sturm.} Beschouw
$y'' + q(t)\,y = 0$ op een interval $I$, met $q$ \omterm{def:b2:metric:continuity}{continu}. (Elke
vergelijking $y'' + ay' + by = 0$ herleidt tot deze normaalvorm met
de substitutie $y = u\exp\bigl(-\frac12\int a\bigr)$; vraag 21
toont een variant van de truc in actie.)
\begin{enumerate}[resume]
  \item Toon voor twee oplossingen $y_1, y_2$ aan dat de \omterm{def:b2:diffeq:wronskian}{wronskiaan}
        $W = y_1y_2' - y_1'y_2$ \emph{constant} is, en nul dan en
        slechts dan als de oplossingen evenredig zijn; en dat een
        oplossing die niet nul is, alleen \emph{enkelvoudige,
        geïsoleerde} nulpunten heeft.
  \item (Scheidingsstelling van Sturm) Zijn $y_1, y_2$
        onafhankelijke oplossingen en $a < b$ twee opeenvolgende
        nulpunten van $y_1$. Bewijs dat $y_2$ precies eenmaal
        verdwijnt in $\intoo{a}{b}$ \emph{(evalueer de constante
        $W$ in $a$ en $b$: daar is $W = y_1'y_2$, en $y_1'(a)$ en
        $y_1'(b)$ hebben tegengesteld teken)}: de nulpunten van
        onafhankelijke oplossingen interlaceren.
  \item (Vergelijkingsstelling van Sturm) Zij $q_1 \leq q_2$ op
        $I$, $y \neq 0$ met $y'' + q_1y = 0$, $z \neq 0$ met $z'' +
        q_2z = 0$, en $a < b$ opeenvolgende nulpunten van $y$. Toon
        aan dat $z$ verdwijnt in $\intcc{a}{b}$ --- strikt
        daarbinnen als $q_1 < q_2$ ergens op $\intoo ab$
        \emph{(geldt $z \neq 0$ op $\intoo ab$, bestudeer dan $(yz'
        - y'z)' = (q_1 - q_2)yz$ met vaste tekens voor $y$ en $z$
        en vergelijk de randwaarden)}.
  \item Leid de \emph{grenzen op de afstand} af: geldt $0 < m^2
        \leq q(t) \leq M^2$ op $I$, dan voldoen twee opeenvolgende
        nulpunten $a < b$ van een oplossing ongelijk aan nul van
        $y'' + qy = 0$ aan
        \[
        \frac{\pi}{M} \;\leq\; b - a \;\leq\; \frac{\pi}{m}
        \]
        \emph{(vergelijk met $u'' + M^2u = 0$ en $u'' + m^2u = 0$,
        waarvan de nulpunten op afstand $\frac\pi M$ en $\frac\pi
        m$ liggen)}. Ga dit na op de harmonische oscillator.
  \item Vorm $ty'' + 2y' + ty = 0$ (\cref{exo:b2:diffeq:7}) om met
        $u = ty$ tot $u'' + u = 0$, vind onmiddellijk haar
        oplossingen $\frac{\sin t}t$ en $\frac{\cos t}{t}$ terug, en
        besluit dat de nulpunten van elke oplossing ongelijk aan nul
        precies op afstand $\pi$ liggen: het wereldbeeld van Sturm
        --- nulpunten worden bestuurd door de coëfficiënt $q$, met
        of zonder formules.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Duhamel en de grens van de
begrensdheid.}
\begin{enumerate}[resume]
  \item (Duhamel voor de oscillator) Toon aan dat voor \omterm{def:b2:metric:continuity}{continue} $F$
        de oplossing van $x'' + \omega^2x = F(t)$ met $x(0) = x'(0)
        = 0$ gelijk is aan
        \[
        x(t) = \frac1\omega\int_0^t\sin\bigl(\omega(t -
        s)\bigr)F(s)\,\dd s ,
        \]
        en leid daaruit de resonante oplossing van vraag 15 opnieuw
        af met $F(s) = F\cos(\omega s)$ \emph{(product in som
        omzetten)}.
  \item ($\zeta > 0$: begrensde ingang, begrensde uitgang) Toon aan
        dat voor $\zeta > 0$ en \emph{elke} begrensde \omterm{def:b2:metric:continuity}{continue} $F$
        elke oplossing van de gedempte vergelijking begrensd is op
        $\intco{0}{\infty}$ \emph{(variatie van de constanten plus
        het exponentiële verval $\vertiii{\eu^{tA}} \leq
        C\eu^{-\alpha t}$ uit \cref{exo:b2:diffeq:8})}.
  \item ($\zeta = 0$) Toon aan dat zonder demping een begrensde
        periodieke aandrijving alle oplossingen begrensd houdt
        \emph{behalve} precies in \omterm{pb:b2:diffeq:1}{resonantie} ($\gamma = \omega$,
        vraag 15 tegenover vraag 14): demping is wat van de grens
        van de begrensdheid \omterm{def:b2:funcseq:def}{uniforme} stabiliteit maakt.
  \item Synthese. In telkens één zin: (i) hoe het
        spoor-determinantvlak de Delen I--II ordent en waar de
        aandrijving (Deel III) het verlaat; (ii) de fysische
        betekenis van $\gamma_*$, $A(\gamma_*)$ en de factor
        $\frac1{2\zeta}$; (iii) wat de stellingen van Sturm zeggen
        wat expliciete formules niet kunnen; (iv) welke twee
        resultaten uit deze opgave de rest van het boek
        stilzwijgend zal hergebruiken (wronskianen die volgens
        Liouville constant zijn; stabiliteit bij begrensde ingang).
\end{enumerate}
\end{problem}
