\chapter{Hermitische vormen}\label{ch:b2:hermitian}

Complexe vectorruimten hebben hun eigen meetkunde van het
inproduct, met één wending: lineariteit in de ene veranderlijke,
\emph{geconjugeerde} lineariteit in de andere. De beloning voor het
aanvaarden van die wending is een spectraaltheorie die nog schoner
is dan de reële --- \omterm{def:b2:hermitian:adjoint}{hermitische} endomorfismen hebben reële
\omterm{def:b2:reduction:eigen}{eigenwaarden}, \omterm{def:b2:hermitian:adjoint}{unitaire} endomorfismen hebben \omterm{def:b2:reduction:eigen}{eigenwaarden} van
modulus $1$, en beide diagonaliseren in orthonormale basissen. Dit
korte hoofdstuk speelt het euclidische programma van
\cref{ch:b2:quadratic} over $\C$.

\section{Hermitische inproducten}

\begin{definition}\label{def:b2:hermitian:def}
Een \emph{hermitisch inproduct}\index{hermitisch inproduct} op een
complexe vectorruimte $E$ is een afbeelding
$\langle\cdot,\cdot\rangle \colon E \times E \to \C$ die lineair is
in de tweede veranderlijke, geconjugeerd \omterm{def:b2:quadratic:adjoint}{symmetrisch} ($\langle y,
x\rangle = \conj{\langle x, y\rangle}$ --- en dus geconjugeerd
lineair in de eerste veranderlijke) en positief definiet ($\langle
x, x\rangle > 0$ voor $x \neq 0$). Het standaardvoorbeeld op
$\C^n$:
\[
\langle x, y \rangle = \sum_{i=1}^{n} \conj{x_i}\, y_i ;
\]
op \omterm{def:b2:metric:continuity}{continue} functies, $\langle f, g\rangle = \int_a^b \conj
f\,g$. \omterm{def:b2:nvs:norm}{Norm}: $\norm x = \sqrt{\langle x,x\rangle}$; een
eindigdimensionale complexe ruimte die hiermee is uitgerust, heet
een \emph{hermitische ruimte}\index{hermitische ruimte}.
\end{definition}

\begin{example}[Eerste berekeningen]\label{ex:b2:hermitian:firstcomp}
Neem in $\C^2$ de vectoren $x = (1+\iu,\ 2-\iu)$ en $y = (\iu,\
1)$. Dan is
\[
\norm x^2 = \abs{1+\iu}^2 + \abs{2-\iu}^2 = 2 + 5 = 7,
\qquad
\norm y^2 = 1 + 1 = 2,
\]
en, met conjugatie van het eerste argument,
\[
\langle x, y\rangle
= \conj{(1+\iu)}\,\iu + \conj{(2-\iu)}\cdot1
= (1-\iu)\iu + (2+\iu)
= (\iu + 1) + (2 + \iu) = 3 + 2\iu .
\]
Cauchy--Schwarz klopt: $\abs{\langle x, y\rangle}^2 = 9 + 4 = 13
\leq 14 = \norm x^2\norm y^2$ --- dicht bij gelijkheid, omdat $x$
dicht bij een veelvoud van $y$ ligt. Merk ook op dat $\langle y,
x\rangle = \conj{3 + 2\iu} = 3 - 2\iu$: geconjugeerde symmetrie in
actie, en de reden waarom $\langle x, x\rangle$ altijd reëel is.
\end{example}

\begin{theorem}[Cauchy--Schwarz, complex]\label{thm:b2:hermitian:cs}
$\abs{\langle x, y\rangle} \leq \norm x \norm y$, met gelijkheid
dan en slechts dan als $x, y$ lineair afhankelijk zijn; en
$\norm\cdot$ is een \omterm{def:b2:nvs:norm}{norm}. Bovendien bestaan er orthonormale
basissen (Gram--Schmidt verloopt woordelijk hetzelfde), met
\[
x = \sum_i \langle e_i, x\rangle\, e_i,
\qquad
\norm x^2 = \sum_i \abs{\langle e_i, x\rangle}^2 .
\]
\end{theorem}

\begin{proof}
Voor $y \neq 0$ en $t \in \C$: $0 \leq \norm{x - ty}^2 = \norm x^2
- 2\Re\bigl(\conj t\langle y, x\rangle\bigr) + \abs t^2\norm y^2$.
Kies $t = \frac{\langle y, x\rangle}{\norm y^2}$:
\[
0 \leq \norm x^2 - \frac{\abs{\langle y, x\rangle}^2}{\norm y^2},
\]
wat de ongelijkheid is; gelijkheid dwingt $x = ty$ af.
Driehoeksongelijkheid, \omterm{def:b2:metric:complete}{volledig}:
\[
\norm{x + y}^2 = \norm x^2 + 2\,\Re\langle x, y\rangle
+ \norm y^2
\leq \norm x^2 + 2\,\abs{\langle x, y\rangle} + \norm y^2
\leq \bigl(\norm x + \norm y\bigr)^2 ,
\]
met $\Re z \leq \abs z$ en vervolgens Cauchy--Schwarz;
homogeniteit en scheiding zijn onmiddellijk, dus $\norm\cdot$ is
een \omterm{def:b2:nvs:norm}{norm}. Gram--Schmidt: als in het reële geval, met de conjugaten
op de plaatsen die de definitie voorschrijft (let op de afspraak:
onze producten zijn geconjugeerd lineair in de \emph{eerste} sleuf,
zodat de coördinaten $\langle e_i, x\rangle$ zijn; het volgende
voorbeeld doorloopt het algoritme eenmaal \omterm{def:b2:metric:complete}{volledig}).
\end{proof}

\begin{example}[Complexe Gram--Schmidt, volledig doorlopen]\label{ex:b2:hermitian:gramschmidt}
Orthonormaliseer de basis $v_1 = (1, \iu)$, $v_2 = (0, 1)$ van
$\C^2$. Eerste vector: $\norm{v_1}^2 = \abs1^2 + \abs\iu^2 = 2$,
dus $e_1 = \frac{1}{\sqrt2}(1, \iu)$. Projecteer $v_2$ --- met de
conjugatie in de eerste sleuf:
\[
\langle e_1, v_2\rangle
= \frac{1}{\sqrt2}\bigl(\conj{1}\cdot0 +
\conj{\iu}\cdot1\bigr)
= \frac{-\iu}{\sqrt2} ,
\qquad
v_2 - \langle e_1, v_2\rangle e_1
= (0,1) + \frac{\iu}{2}\,(1, \iu)
= \Bigl(\frac\iu2,\ \frac12\Bigr) .
\]
De \omterm{def:b2:nvs:norm}{norm} hiervan is $\sqrt{\frac14 + \frac14} = \frac{1}{\sqrt2}$:
$e_2 = \frac{1}{\sqrt2}(\iu, 1)$. Controle: $\langle e_1,
e_2\rangle = \frac12(\conj1\cdot\iu + \conj\iu\cdot1) =
\frac12(\iu - \iu) = 0$. Coördinaten van $v_2$ in de nieuwe basis:
$v_2 = \langle e_1, v_2\rangle e_1 + \langle e_2, v_2\rangle e_2$
met $\langle e_2, v_2\rangle = \frac{1}{\sqrt2}$ --- let op de
volgorde: $\langle v_2, e_1\rangle$ zou de \emph{geconjugeerde}
coëfficiënt geven. Het inzicht om te onthouden: het algoritme is
woordelijk het euclidische; de enige valkuil is waar de conjugatie
valt, en het berekenen van $\norm{v_2 - \text{proj}}^2 > 0$
gebruikt stilzwijgend de positiviteit --- het axioma dat de hele
meetkunde laat werken.
\end{example}

\section{Adjunct, hermitische en unitaire endomorfismen}

\begin{definition}\label{def:b2:hermitian:adjoint}
De \emph{\omterm{def:b2:quadratic:adjoint}{adjunct}} $u^*$ van $u \in \mathcal{L}(E)$ is gedefinieerd
door $\langle u^*(x), y\rangle = \langle x, u(y)\rangle$; in een
orthonormale basis is $\operatorname{Mat}(u^*) = \conj{A}^{\mathsf
T} =: A^{\dagger}$ (de geconjugeerde \omterm{def:b2:linalg:transpose}{getransponeerde}) --- immers,
is $B = (b_{ij})$ de matrix van $u^*$ in de orthonormale basis
$(e_i)$, dan is $b_{ij} = \langle e_i, u^*(e_j)\rangle$, en de
definiërende identiteit geeft
\[
\conj{b_{ij}}
= \langle u^*(e_j), e_i\rangle
= \langle e_j, u(e_i)\rangle = a_{ji} ,
\qquad\text{dus}\qquad
B = \conj{A}^{\mathsf T} .
\] En $u$ heet
\emph{hermitisch}\index{hermitisch endomorfisme} wanneer $u^* = u$
($A^\dagger = A$), \emph{unitair}\index{unitaire groep} wanneer
$u^*u = \mathrm{id}$ ($A^\dagger A = I$: de groep $U(n)$), en
\emph{normaal} wanneer $u^*u = uu^*$.
\end{definition}

\begin{proposition}\label{prop:b2:hermitian:eigenvalues}
\omterm{def:b2:reduction:eigen}{Eigenwaarden} van een \omterm{def:b2:hermitian:adjoint}{hermitisch endomorfisme} zijn \emph{reëel};
\omterm{def:b2:reduction:eigen}{eigenwaarden} van een \omterm{def:b2:hermitian:adjoint}{unitair} endomorfisme hebben \emph{modulus
$1$}; in beide gevallen staan de \omterm{def:b2:reduction:eigen}{eigenruimten} bij verschillende
\omterm{def:b2:reduction:eigen}{eigenwaarden} loodrecht op elkaar.
\end{proposition}

\begin{proof}
\omterm{def:b2:hermitian:adjoint}{Hermitisch}, $u(x) = \lambda x$, $x \neq 0$:
\[
\lambda \norm x^2 = \langle x, u(x)\rangle = \langle u(x), x\rangle
= \conj{\langle x, u(x)\rangle} = \conj\lambda\,\norm x^2 ,
\]
dus $\lambda \in \R$. \omterm{def:b2:hermitian:adjoint}{Unitair}: $\norm{u(x)} = \norm x$ (uit $u^*u =
\mathrm{id}$), dus $\abs\lambda\norm x = \norm x$. Orthogonaliteit
(\omterm{def:b2:hermitian:adjoint}{hermitisch} geval): $\lambda\langle x, y\rangle = \langle u(x),
y\rangle = \langle x, u(y)\rangle = \mu\langle x, y\rangle$ voor
\omterm{def:b2:reduction:eigen}{eigenvectoren} met reële $\lambda \neq \mu$. \omterm{def:b2:hermitian:adjoint}{Unitair} geval,
\omterm{def:b2:metric:complete}{volledig}: voor $u(x) = \lambda x$, $u(y) = \mu y$ met $\lambda \neq
\mu$ (beide van modulus $1$) is
\[
\langle x, y\rangle = \langle u(x), u(y)\rangle
= \conj\lambda\mu\,\langle x, y\rangle ,
\]
en $\conj\lambda\mu = \frac{\mu}{\lambda} \neq 1$: de factor is
niet $1$, dus $\langle x, y\rangle = 0$.
\end{proof}

\begin{example}[Een antihermitische matrix, gediagonaliseerd]\label{ex:b2:hermitian:skewexample}
$A = \begin{pmatrix} 0 & -2\\ 2 & 0\end{pmatrix}$ voldoet aan
$A^\dagger = A^{\mathsf T} = -A$: \omterm{def:b2:hermitian:adjoint}{antihermitisch} (en ook reëel
antisymmetrisch --- over $\R$ heeft zij helemaal geen
\omterm{def:b2:reduction:eigen}{eigenwaarden}). \omterm{def:b2:reduction:charpoly}{Karakteristieke veelterm} $X^2 + 4$: \omterm{def:b2:reduction:eigen}{eigenwaarden}
$\pm2\iu$, zuiver imaginair, zoals \cref{exo:b2:hermitian:9} in het
algemeen voorspelt. \omterm{def:b2:reduction:eigen}{Eigenvectoren}: $(A - 2\iu I)v = 0$ geeft $v_1 =
\frac{1}{\sqrt2}(1, \iu)$, en $v_2 = \frac{1}{\sqrt2}(1, -\iu)$
voor $-2\iu$; zij staan loodrecht op elkaar:
\[
\langle v_1, v_2\rangle
= \tfrac12\bigl(\conj{1}\cdot1 +
\conj{\iu}\cdot(-\iu)\bigr)
= \tfrac12(1 - 1) = 0 .
\]
Dus $A = U\operatorname{diag}(2\iu, -2\iu)\,U^\dagger$ met $U =
(v_1\ v_2)$ \omterm{def:b2:hermitian:adjoint}{unitair}. Het inzicht om te onthouden: $H = -\iu A =
\begin{pmatrix} 0 & 2\iu\\ -2\iu & 0\end{pmatrix}$ is \omterm{def:b2:hermitian:adjoint}{hermitisch}
met het reële spectrum $\{\pm2\}$ en \emph{dezelfde} \omterm{def:b2:reduction:eigen}{eigenvectoren}
--- de bijectie $u \mapsto \iu u$ tussen \omterm{def:b2:hermitian:adjoint}{hermitische} en
\omterm{def:b2:hermitian:adjoint}{antihermitische} endomorfismen (\cref{exo:b2:hermitian:9}), matrix
voor matrix gezien; over $\R$ is diezelfde $A$ een
draaivermenigvuldiging zonder enige \omterm{def:b2:reduction:eigen}{eigenvector}, en pas de overgang
naar $\C$ onthult haar normaalvorm.
\end{example}

\begin{theorem}[Hermitische spectraalstelling]\label{thm:b2:hermitian:spectral}
Elk \omterm{def:b2:hermitian:adjoint}{hermitisch endomorfisme} van een \omterm{def:b2:hermitian:def}{hermitische ruimte} bezit een
orthonormale basis van \omterm{def:b2:reduction:eigen}{eigenvectoren} (met reële \omterm{def:b2:reduction:eigen}{eigenwaarden}):
$A^\dagger = A$ impliceert $A = U D U^{\dagger}$ met $U \in U(n)$
en $D$ reëel diagonaal.\index{spectraalstelling!hermitische}
\end{theorem}

\begin{proof}
Over $\C$ valt de \omterm{def:b2:reduction:charpoly}{karakteristieke veelterm} in lineaire factoren
uiteen: er bestaat een \omterm{def:b2:reduction:eigen}{eigenvector} $e_1$ (\cref{ch:b2:reduction})
--- geen compactheidsargument nodig, een voordeel van $\C$.
Normaliseer haar. Haar orthogonale complement $F = e_1^\perp$ is
stabiel: voor $x \perp e_1$ is
\[
\langle e_1, u(x)\rangle = \langle u(e_1), x\rangle
= \lambda_1\langle e_1, x\rangle = 0
\]
($\lambda_1$ reëel). De beperking is \omterm{def:b2:hermitian:adjoint}{hermitisch}; pas inductie naar
de dimensie toe en plak aaneen. In detail: de beperking $u|_F$ is
een endomorfisme van de \omterm{def:b2:hermitian:def}{hermitische ruimte} $F$ (dimensie $n - 1$)
met $\langle u|_F(x), y\rangle = \langle x, u|_F(y)\rangle$, geërfd
van $u$; de inductiehypothese levert een orthonormale basis $(e_2,
\dots, e_n)$ van $F$ uit \omterm{def:b2:reduction:eigen}{eigenvectoren}, en $(e_1, e_2, \dots, e_n)$
is orthonormaal in $E$ ($e_1 \perp F$) en bestaat uit
\omterm{def:b2:reduction:eigen}{eigenvectoren} van $u$. Matricieel vertaald: de kolommen van $U$
zijn de $e_i$, $U^\dagger U = I$ drukt hun orthonormaliteit uit, en
$AU = UD$ verzamelt de eigenwaardevergelijkingen, waaruit $A =
UDU^\dagger$ met $D$ reëel diagonaal
(\cref{prop:b2:hermitian:eigenvalues}).
\end{proof}

\begin{example}\label{ex:b2:hermitian:pauli}
$A = \begin{pmatrix} 0 & -\iu\\ \iu & 0\end{pmatrix}$ is \omterm{def:b2:hermitian:adjoint}{hermitisch}
($A^\dagger = A$): \omterm{def:b2:reduction:eigen}{eigenwaarden} uit $\chi_A = X^2 - 1$: $\pm 1$
(reëel, zoals beloofd), met orthonormale \omterm{def:b2:reduction:eigen}{eigenvectoren}
$\frac{1}{\sqrt2}(1, \iu)^{\mathsf T}$ en $\frac{1}{\sqrt2}(1,
-\iu)^{\mathsf T}$. (Natuurkundigen kennen $A$ als een
Pauli-matrix; dat \omterm{def:b2:hermitian:adjoint}{hermitische} spectra reëel zijn, is de reden
waarom kwantumobservabelen door \omterm{def:b2:hermitian:adjoint}{hermitische} operatoren worden
gemodelleerd.)
\end{example}

\begin{example}[Een positief definiete hermitische matrix, uitgewerkt]\label{ex:b2:hermitian:pdrun}
$A = \begin{pmatrix} 2 & 1-\iu\\ 1+\iu & 3\end{pmatrix}$:
\omterm{def:b2:hermitian:adjoint}{hermitisch}, want de diagonaal is reëel en de elementen buiten de
diagonaal zijn elkaars complex geconjugeerde. \omterm{def:b2:reduction:charpoly}{Karakteristieke
veelterm}:
\[
(2-\lambda)(3-\lambda) - \abs{1-\iu}^2
= \lambda^2 - 5\lambda + 4
= (\lambda - 1)(\lambda - 4) :
\]
spectrum $\{1, 4\}$, reëel en positief --- $A$ is positief
definiet. \omterm{def:b2:reduction:eigen}{Eigenvectoren}: voor $\lambda = 4$ geeft het stelsel $(A -
4I)v = 0$ dat $v_4 = (1 - \iu,\ 2)$ (controleer de tweede rij:
$(1+\iu)(1-\iu) - 2 = 0$); voor $\lambda = 1$: $v_1 = (1 - \iu,\
-1)$. Orthogonaliteit, met de conjugatie in de eerste sleuf:
\[
\langle v_4, v_1\rangle
= \conj{(1-\iu)}\,(1-\iu) + \conj{2}\,(-1)
= 2 - 2 = 0 . \checkmark
\]
Normaliseren ($\norm{v_4}^2 = 2 + 4 = 6$, $\norm{v_1}^2 = 2 + 1 =
3$) geeft de \omterm{def:b2:hermitian:adjoint}{unitaire} $U = \bigl(\frac{v_4}{\sqrt6}\
\frac{v_1}{\sqrt3}\bigr)$ met $A =
U\operatorname{diag}(4,1)U^\dagger$. Het inzicht om te onthouden:
de lezing volgens Rayleigh is onmiddellijk --- op de eenheidssfeer
van $\C^2$ doorloopt $\langle x, Ax\rangle$ precies $\intcc{1}{4}$,
aangenomen in de twee \omterm{def:b2:reduction:eigen}{eigenvectoren}; dit is de kiem voor $n = 2$
van de theorie van Courant en Fischer die in de weekendopgave wordt
opgebouwd. Ga ook buiten de \omterm{def:b2:reduction:eigen}{eigenvectoren} na dat de vorm reëel is:
in $x = (1, \iu)$ is
\[
Ax = \bigl(2 + (1-\iu)\iu,\ (1+\iu) + 3\iu\bigr)
= (3 + \iu,\ 1 + 4\iu),
\]
\[
\langle x, Ax\rangle = \conj{1}\,(3+\iu) +
\conj{\iu}\,(1+4\iu)
= (3 + \iu) + (-\iu)(1 + 4\iu) = 3 + \iu - \iu + 4 = 7 \in
\R ,
\]
zoals het bewijsmechanisme van
\cref{prop:b2:hermitian:eigenvalues} (geconjugeerde symmetrie tegen
$A^\dagger = A$) voor elke $x$ garandeert.
\end{example}

\begin{example}[Een unitaire matrix gediagonaliseerd]\label{ex:b2:hermitian:unitary}
$U = \frac{1}{\sqrt2}\begin{pmatrix} 1 & \iu\\ \iu & 1
\end{pmatrix}$ (\omterm{def:b2:hermitian:adjoint}{unitair} volgens \cref{exo:b2:hermitian:2}). Haar
\omterm{def:b2:reduction:charpoly}{karakteristieke veelterm} is $\bigl(X - \frac{1}{\sqrt2}\bigr)^2 +
\frac12$, met nulpunten
\[
\lambda_\pm = \frac{1 \pm \iu}{\sqrt2} = \eu^{\pm\iu\pi/4},
\]
van modulus $1$ zoals \cref{prop:b2:hermitian:eigenvalues}
beloofde, en met orthonormale \omterm{def:b2:reduction:eigen}{eigenvectoren} $\frac{1}{\sqrt2}(1,
\pm1)$. Dus $U = V\operatorname{diag}(\eu^{\iu\pi/4},
\eu^{-\iu\pi/4})V^\dagger$: in de juiste basis is $U$ een paar
vlakke rotaties over $\pm\frac\pi4$ --- een reële rotatiematrix
heeft geen reële \omterm{def:b2:reduction:eigen}{eigenvectoren}, maar over $\C$ valt zij uiteen in
twee scalairen van modulus $1$. Het inzicht om te onthouden:
\omterm{def:b2:hermitian:adjoint}{hermitische} spectra leven op de reële rechte, \omterm{def:b2:hermitian:adjoint}{unitaire} spectra op
de eenheidscirkel; beide zijn schaduwen van dezelfde normaliteit,
en de Cayley-transformatie van \cref{exo:b2:hermitian:6} beeldt het
ene beeld op het andere af.
\end{example}

\begin{example}[De Cayley-transformatie, berekend]\label{ex:b2:hermitian:cayleyrun}
Voer \cref{exo:b2:hermitian:6} uit op $H = \begin{pmatrix} 0 & 1\\
1 & 0\end{pmatrix}$ (\omterm{def:b2:hermitian:adjoint}{hermitisch}, spectrum $\{1, -1\}$, orthonormale
\omterm{def:b2:reduction:eigen}{eigenvectoren} $\frac{1}{\sqrt2}(1, \pm1)$). In de eigenbasis is
alles scalair: de transformatie $\lambda \mapsto \frac{\lambda -
\iu}{\lambda + \iu}$ zendt
\[
1 \longmapsto \frac{1 - \iu}{1 + \iu} = -\iu,
\qquad
-1 \longmapsto \frac{-1 - \iu}{-1 + \iu} = \iu ,
\]
(vermenigvuldig met de geconjugeerde van de noemer), dus $U = (H -
\iu I)(H + \iu I)^{-1}$ is de \omterm{def:b2:hermitian:adjoint}{unitaire} matrix met \omterm{def:b2:reduction:eigen}{eigenwaarden}
$\mp\iu$ op diezelfde \omterm{def:b2:reduction:eigen}{eigenvectoren}:
\[
U = \frac{1}{2}\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
\begin{pmatrix} -\iu & 0\\ 0 & \iu\end{pmatrix}
\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
= \begin{pmatrix} 0 & -\iu\\ -\iu & 0\end{pmatrix} .
\]
Controle: $U^\dagger U = I$, en $1 \notin \operatorname{Sp}U =
\{\pm\iu\}$, zoals de theorie belooft. Het inzicht om te onthouden:
de reële rechte wordt afgebeeld op de eenheidscirkel zonder het
punt $1$ --- \omterm{def:b2:reduction:eigen}{eigenwaarde} voor \omterm{def:b2:reduction:eigen}{eigenwaarde} is de
Cayley-transformatie de möbiusafbeelding
$\frac{\lambda-\iu}{\lambda+\iu}$, en matrices volgen eenvoudigweg
hun spectrum.
\end{example}

\begin{remark}[Klassieke valkuilen]
\emph{(i) Waar de streep valt:} dit boek conjugeert de
\emph{eerste} sleuf, zodat de coördinaten $\langle e_i, x\rangle$
zijn en $\langle\lambda x, y\rangle = \conj\lambda\langle x,
y\rangle$; veel teksten conjugeren in plaats daarvan de tweede
sleuf --- vertaal dus voordat je formules vergelijkt, of tekens van
$\iu$ gaan stilzwijgend mis. \emph{(ii) Complexe polarisatie is
sterker:} geldt over $\C$ dat $\langle x, u(x)\rangle = 0$ voor
\emph{alle} $x$, dan is $u = 0$ (werk $x + y$ en $x + \iu y$ uit:
zowel het reële als het imaginaire deel van $\langle x,
u(y)\rangle$ verdwijnt); over $\R$ faalt dit --- de rotatie over
$\frac\pi2$ voldoet overal aan $\langle x, u(x)\rangle = 0$.
Bijgevolg dwingt, alleen over $\C$, de uitspraak ``$\langle x,
u(x)\rangle \in \R$ voor alle $x$'' al af dat $u$ \omterm{def:b2:hermitian:adjoint}{hermitisch} is.
\emph{(iii) Reëel normaal is niet \omterm{def:b2:reduction:diag}{diagonaliseerbaar}:} de matrix van
\cref{ex:b2:hermitian:skewexample} is normaal maar heeft geen reële
\omterm{def:b2:reduction:eigen}{eigenwaarde}; \omterm{def:b2:hermitian:adjoint}{unitaire} diagonalisatie is een stelling \emph{over}
$\C$, en over $\R$ krijgt men slechts blokreducties.
\emph{(iv) Unitariteit nagaan:} $U^\dagger U = I$ betekent dat de
\emph{kolommen} orthonormaal zijn voor het \omterm{def:b2:hermitian:adjoint}{hermitische} product ---
$UU^{\mathsf T}$ testen, of de conjugatie in de kolomproducten
vergeten, zijn de twee klassieke manieren om een verkeerde matrix
goed te keuren.
\end{remark}

\begin{example}[Isometrieën zijn precies de unitaire afbeeldingen]\label{ex:b2:hermitian:isometry}
Normbehoud lijkt zwakker dan unitariteit, maar over $\C$ is het dat
niet: geldt $\norm{u(x)} = \norm x$ voor alle $x$, dan is $u^*u =
\mathrm{id}$. Immers, $v = u^*u - \mathrm{id}$ is \omterm{def:b2:hermitian:adjoint}{hermitisch} en
voldoet aan $\langle x, v(x)\rangle = \norm{u(x)}^2 - \norm x^2 =
0$ voor elke $x$; volgens de complexe polarisatie (valkuil (ii)
hierboven) is een afbeelding met identiek verdwijnende
``diagonaal'' nul: $v = 0$. Concreet verloopt de polarisatie als
\[
0 = \langle x + y, v(x+y)\rangle
= \langle x, v(y)\rangle + \langle y, v(x)\rangle,
\qquad
0 = \langle x + \iu y, v(x + \iu y)\rangle
= \iu\langle x, v(y)\rangle - \iu\langle y, v(x)\rangle ,
\]
en de twee regels samen dwingen $\langle x, v(y)\rangle = 0$ af
voor alle $x, y$. Het inzicht om te onthouden: hierdoor kan
``\omterm{def:b2:hermitian:adjoint}{unitair}'' louter door lengten te meten worden nagegaan ---
starheid die het hoofdstuk over Fourierreeksen zal uitbuiten, waar
het behoud van de energie $\norm f_2$ (Parseval) hetzelfde is als
het behoud van alle inproducten van coëfficiënten.
\end{example}

\begin{remark}[Vooruitblik binnen dit volume]
De \omterm{def:b2:hermitian:adjoint}{hermitische} machinerie die hier wordt gebouwd, wordt vrijwel
onmiddellijk verbruikt. Het hoofdstuk over Fourierreeksen
\emph{is} \omterm{def:b2:hermitian:adjoint}{hermitische} meetkunde in oneindige dimensie: de
exponentiëlen $(e_n)$ vormen een orthonormale familie voor $\langle
f, g\rangle = \frac{1}{2\pi}\int\conj fg$, de ongelijkheid van
Bessel is de projectieschatting uit \cref{thm:b2:hermitian:cs} van
dit hoofdstuk, en Parseval is haar gelijkheid in de limiet. De
eindige Fouriertransformatie (\cref{exo:b2:hermitian:10}) duikt
telkens op wanneer een convolutie gediagonaliseerd moet worden. En
de weekendopgave van dit hoofdstuk --- Courant--Fischer, Weyl,
interlacering --- levert de stabiliteit van \omterm{def:b2:reduction:eigen}{eigenwaarden} waarop het
hoofdstuk over differentiaalvergelijkingen zich beroept wanneer het
beweert dat kleine verstoringen van een systeem zijn frequenties
slechts weinig verplaatsen. Terugkijkend is alles hier de complexe
spiegel van het hoofdstuk over \omterm{def:b2:quadratic:def}{kwadratische vormen}: houd de twee
woordenboeken naast elkaar ($A^{\mathsf T} \leftrightarrow
A^\dagger$, orthogonaal $\leftrightarrow$ \omterm{def:b2:hermitian:adjoint}{unitair}, Rayleigh reëel
in beide).
\end{remark}

\begin{remark}[Normale endomorfismen]
Over $\C$ luidt de definitieve uitspraak: $u$ is \emph{\omterm{def:b2:hermitian:adjoint}{unitair}
\omterm{def:b2:reduction:diag}{diagonaliseerbaar} dan en slechts dan als $u$ normaal is} ($u^*u =
uu^*$) --- waarmee \omterm{def:b2:hermitian:adjoint}{hermitische}, \omterm{def:b2:hermitian:adjoint}{unitaire} en \omterm{def:b2:hermitian:adjoint}{antihermitische}
afbeeldingen in één klap gedekt zijn. Het bewijs is een aangename
versterking van het argument hierboven
(\cref{exo:b2:hermitian:8}). Over $\R$ levert normaliteit daarentegen
alleen blokdiagonalisatie op (rotatieblokken): de complexe
meetkunde is werkelijk eenvoudiger.
\end{remark}

\begin{remark}[Waar dit wordt gebruikt]
De \omterm{def:b2:hermitian:adjoint}{hermitische} spectraaltheorie is de wiskunde van de
kwantummechanica: observabelen worden gemodelleerd door
\omterm{def:b2:hermitian:adjoint}{hermitische} operatoren (reële spectra = meetbare waarden), de
tijdsevolutie door \omterm{def:b2:hermitian:adjoint}{unitaire} operatoren (normbehoud = behoud van
waarschijnlijkheid). Binnen dit boek rust het hoofdstuk over
Fourierreeksen op de orthonormaliteit van de exponentiëlen --- een
uitspraak over een \omterm{def:b2:hermitian:def}{hermitisch inproduct} --- en de diagonalisatie
van circulante matrices (\cref{exo:b2:hermitian:10}) is de eindige
Fouriertransformatie. De weekendopgave ontwikkelt de variationele
rekening van \omterm{def:b2:reduction:eigen}{eigenwaarden} (Courant--Fischer, Weyl, interlacering),
het dagelijks brood van de numerieke wiskunde en de wiskundige
fysica; het volume van bachelorjaar 3 breidt haar uit tot \omterm{def:b2:metric:compact}{compacte}
\omterm{def:b2:quadratic:adjoint}{zelftoegevoegde} operatoren op hilbertruimten.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b2:hermitian:1}
Bereken op $\C^2$ de grootheden $\langle x, y\rangle$, $\norm x$,
$\norm y$ voor $x = (1, \iu)$, $y = (\iu, 1)$; staan zij loodrecht
op elkaar? Geef een orthonormale basis die $\frac{x}{\norm x}$
bevat.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:hermitian:2}
Welke zijn \omterm{def:b2:hermitian:adjoint}{hermitisch}? \omterm{def:b2:hermitian:adjoint}{Unitair}? Normaal?
\[
\begin{pmatrix} 1 & \iu\\ -\iu & 2 \end{pmatrix},
\qquad
\frac{1}{\sqrt2}\begin{pmatrix} 1 & \iu\\ \iu & 1\end{pmatrix},
\qquad
\begin{pmatrix} 0 & 1\\ 0 & 0 \end{pmatrix}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:hermitian:3}
Bewijs dat een matrix $A \in \mathcal{M}_n(\C)$ zich op precies één
manier laat schrijven als $A = H + \iu K$ met $H, K$ \omterm{def:b2:hermitian:adjoint}{hermitisch}
\emph{(de ``reële en imaginaire delen'' $H = \frac{A +
A^\dagger}{2}$, $K = \frac{A - A^\dagger}{2\iu}$)}, en dat $A$
normaal is dan en slechts dan als $H$ en $K$ commuteren.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:4}
Diagonaliseer in een orthonormale basis: $A = \begin{pmatrix} 2 &
\iu\\ -\iu & 2\end{pmatrix}$, en bereken $A^k$ voor $k \in \N$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:5}
Bewijs dat $U(n)$ \omterm{def:b2:metric:compact}{compact} is en dat de eigenwaardeafbeelding
surjectief is: elke $\lambda$ van modulus $1$ treedt op bij een
zekere \omterm{def:b2:hermitian:adjoint}{unitaire} matrix. Bewijs dat $\det U \in \mathbb{U}$ (de
eenheidscirkel) voor $U \in U(n)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:6}
(Cayley-transformatie) Zij $H$ \omterm{def:b2:hermitian:adjoint}{hermitisch}. Bewijs dat $H + \iu I$
inverteerbaar is en dat $U = (H - \iu I)(H + \iu I)^{-1}$ \omterm{def:b2:hermitian:adjoint}{unitair}
is, met $1 \notin \operatorname{Sp}(U)$. \emph{(Werk spectraal: op
een eigenbasis van $H$ is alles scalair.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:7}
Bewijs voor \omterm{def:b2:hermitian:adjoint}{hermitische} positief definiete $A$ ($\langle x,
Ax\rangle > 0$ voor $x \neq 0$) dat $\operatorname{Sp}(A)
\subseteq \intoo{0}{\infty}$, dat $A = B^2$ voor een \omterm{def:b2:hermitian:adjoint}{hermitische}
positief definiete $B$, en dat $\det A > 0$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:hermitian:8}
(Spectraalstelling voor normale endomorfismen) Zij $u$ normaal op
een \omterm{def:b2:hermitian:def}{hermitische ruimte}.
\begin{enumerate}
  \item Bewijs dat $\norm{u(x)} = \norm{u^*(x)}$ voor alle $x$, en
        leid af dat $\ker(u - \lambda) = \ker(u^* -
        \conj\lambda)$.
  \item Bewijs dat de \omterm{def:b2:reduction:eigen}{eigenruimten} van $u$ bij verschillende
        \omterm{def:b2:reduction:eigen}{eigenwaarden} orthogonaal zijn, en dat het orthogonale
        complement van een \omterm{def:b2:reduction:eigen}{eigenruimte} $u$-stabiel is.
  \item Besluit met inductie dat $u$ \omterm{def:b2:hermitian:adjoint}{unitair} \omterm{def:b2:reduction:diag}{diagonaliseerbaar} is;
        en omgekeerd.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:hermitian:9}
Een endomorfisme heet \emph{\omterm{def:b2:hermitian:adjoint}{antihermitisch}} wanneer $u^* = -u$.
Bewijs dat zijn \omterm{def:b2:reduction:eigen}{eigenwaarden} zuiver imaginair zijn, dat $u \mapsto
\iu u$ een bijectie is van de \omterm{def:b2:hermitian:adjoint}{hermitische} naar de \omterm{def:b2:hermitian:adjoint}{antihermitische}
endomorfismen, en dat \omterm{def:b2:hermitian:adjoint}{antihermitische} endomorfismen \omterm{def:b2:hermitian:adjoint}{unitair}
\omterm{def:b2:reduction:diag}{diagonaliseerbaar} zijn (\cref{exo:b2:hermitian:8}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:10}
(De eindige Fouriertransformatie) Zij $S$ de cyclische
verschuiving van $\C^n$: $S(x_0, x_1, \dots, x_{n-1}) = (x_{n-1},
x_0, \dots, x_{n-2})$, en $\omega = \eu^{2\iu\pi/n}$.
\begin{enumerate}
  \item Toon aan dat $S$ \omterm{def:b2:hermitian:adjoint}{unitair} is en dat de vectoren $f_k =
        \frac{1}{\sqrt n}\bigl(1, \omega^k, \omega^{2k}, \dots,
        \omega^{(n-1)k}\bigr)$, $0 \leq k < n$, een orthonormale
        basis van \omterm{def:b2:reduction:eigen}{eigenvectoren} vormen: $Sf_k = \omega^{-k}
        f_k$.
  \item Leid af dat elke \emph{circulante} matrix $C =
        \sum_{j=0}^{n-1} c_jS^j$ normaal is, gediagonaliseerd
        door dezelfde basis, met \omterm{def:b2:reduction:eigen}{eigenwaarden} $\widehat c(k) =
        \sum_j c_j\,\omega^{-jk}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:11}
Zij $P$ een idempotent ($P^2 = P$) endomorfisme van een
\omterm{def:b2:hermitian:def}{hermitische ruimte}. Bewijs dat $P$ de \emph{orthogonale} projectie
op $\operatorname{im} P$ is dan en slechts dan als $P^* = P$. Geef
de matrix van de orthogonale projectie op $\C v$ ($\norm v = 1$),
en op een deelruimte met orthonormale basis $(v_1, \dots, v_k)$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:hermitian:12}
(Spectrale projectoren via interpolatie) Zij $A$ \omterm{def:b2:hermitian:adjoint}{hermitisch} met
\emph{verschillende} \omterm{def:b2:reduction:eigen}{eigenwaarden} $\lambda_1, \dots, \lambda_p$ en
eigenruimteontbinding $E = \bigoplus_i E_i$. Definieer de
Lagrange-veeltermen $L_i(X) = \prod_{j\neq i}\frac{X -
\lambda_j}{\lambda_i - \lambda_j}$. Bewijs dat $P_i = L_i(A)$ de
orthogonale projectie op $E_i$ is, dat $P_iP_j = 0$ voor $i \neq
j$, dat $\sum_i P_i = I$ en dat $A = \sum_i \lambda_iP_i$ (de
\emph{spectrale ontbinding}); druk $f(A)$ voor een willekeurige
veelterm $f$ uit in de $P_i$.
\end{exercise}

\section{Probleem: Courant--Fischer, Weyl en de rekening met
eigenwaarden}

\begin{problem}\label{pb:b2:hermitian:1}
De \omterm{def:b2:reduction:eigen}{eigenwaarden} van een \omterm{def:b2:hermitian:adjoint}{hermitische} matrix zijn niet zomaar
nulpunten van een veelterm: het zijn oplossingen van
\emph{optimalisatieproblemen}. Dat variationele standpunt ---
\omterm{pb:b2:hermitian:1}{Rayleigh-quotiënten} en de \emph{min-maxstelling van
Courant--Fischer}\index{stelling van Courant--Fischer} --- maakt
\omterm{def:b2:reduction:eigen}{eigenwaarden} vergelijkbaar, stabiel en berekenbaar, en deze opgave
oogst haar klassieke gewassen: de \emph{verstoringsongelijkheden
van Weyl}\index{ongelijkheid van Weyl}, de \emph{interlacering van
Cauchy}, de spoorongelijkheden van Schur en Ky Fan, de
monotonie van de matrixvierkantswortel, en het spectrum van de
discrete laplaciaan. Overal zijn $A, B, E$ \omterm{def:b2:hermitian:adjoint}{hermitisch} op $E =
\C^n$ met \omterm{def:b2:reduction:eigen}{eigenwaarden} in dalende volgorde opgesomd, $\lambda_1(A)
\geq \dots \geq \lambda_n(A)$, en is $R_A(x) = \frac{\langle x,
Ax\rangle}{\langle x, x\rangle}$ voor $x \neq 0$ het
\emph{Rayleigh-quotiënt}\index{Rayleigh-quotiënt}.

\medskip
\noindent\textbf{Deel I --- \omterm{pb:b2:hermitian:1}{Rayleigh-quotiënten} en min-max.} Leg
een orthonormale eigenbasis $(e_1, \dots, e_n)$ vast met $Ae_i =
\lambda_ie_i$.
\begin{enumerate}
  \item Toon aan dat $R_A(x)$ reëel is en dat
        \[
        \lambda_n \leq R_A(x) \leq \lambda_1
        \qquad (x \neq 0),
        \]
        waarbij beide grenzen worden aangenomen: $\lambda_1 =
        \max R_A$, $\lambda_n = \min R_A$.
  \item Toon aan dat de kritieke punten van $R_A$ precies de
        \omterm{def:b2:reduction:eigen}{eigenvectoren} van $A$ zijn \emph{(werk $t \mapsto R_A(x +
        tv)$ uit in $t = 0$ voor willekeurige $v$, en vervang $v$
        vervolgens door $\iu v$)}.
  \item Zij $V_k = \operatorname{Vect}(e_1, \dots, e_k)$ en $W_k =
        \operatorname{Vect}(e_k, \dots, e_n)$. Toon aan dat
        \[
        \min_{x \in V_k\setminus\{0\}} R_A(x) = \lambda_k
        = \max_{x \in W_k\setminus\{0\}} R_A(x) .
        \]
  \item Bewijs de \emph{stelling van Courant--Fischer}: voor $1
        \leq k \leq n$ is
        \[
        \lambda_k = \max_{\dim V = k}\;\min_{x \in
        V\setminus\{0\}} R_A(x)
        = \min_{\dim W = n-k+1}\;\max_{x \in
        W\setminus\{0\}} R_A(x)
        \]
        \emph{(voor elke $V$ van dimensie $k$ is $V \cap W_k \neq
        \{0\}$ volgens Grassmann, dus $\min_V R_A \leq
        \lambda_k$; vraag 3 toont dat de grens wordt
        aangenomen)}.
  \item (Monotonie) Schrijf $A \leq B$ wanneer $B - A$ positief
        semidefiniet is. Leid uit vraag 4 af: uit $A \leq B$
        volgt $\lambda_k(A) \leq \lambda_k(B)$ voor elke $k$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De ongelijkheden van Weyl.}
\begin{enumerate}[resume]
  \item Toon aan dat deelruimten $V, W \subseteq \C^n$ met $\dim V
        + \dim W > n$ elkaar niet-triviaal snijden, en
        veralgemeen: $\dim(V_1 \cap V_2 \cap V_3) \geq \dim V_1 +
        \dim V_2 + \dim V_3 - 2n$.
  \item Bewijs de \emph{ongelijkheid van Weyl}: voor $i + j - 1
        \leq n$ is
        \[
        \lambda_{i+j-1}(A + B) \leq \lambda_i(A) +
        \lambda_j(B)
        \]
        \emph{(snijd de deelruimten $W_i(A)$, $W_j(B)$ en
        $V_{i+j-1}(A+B)$ uit vraag 3 en tel dimensies)}.
  \item Definieer $\vertiii{E}_2 = \max_{\norm x = 1}\norm{Ex}$ en
        toon aan dat $\vertiii E_2 = \max_k\abs{\lambda_k(E)}$
        voor \omterm{def:b2:hermitian:adjoint}{hermitische} $E$. Leid de
        \emph{verstoringsstelling van Weyl} af:
        \[
        \bigl|\lambda_k(A + E) - \lambda_k(A)\bigr| \leq
        \vertiii{E}_2
        \qquad (1 \leq k \leq n) :
        \]
        elke \omterm{def:b2:reduction:eigen}{eigenwaarde} is een $1$-lipschitzfunctie van de
        matrix.
  \item (Verstoringen van rang één) Zij $P$ \omterm{def:b2:hermitian:adjoint}{hermitisch} positief
        semidefiniet van rang $1$. Toon aan dat
        \[
        \lambda_k(A) \leq \lambda_k(A + P) \leq
        \lambda_{k-1}(A) \qquad (2 \leq k \leq n),
        \]
        samen met $\lambda_1(A) \leq \lambda_1(A+P)$: de nieuwe
        \omterm{def:b2:reduction:eigen}{eigenwaarden} \emph{interlaceren} de oude.
  \item Ga vraag 8 numeriek na: $A = \begin{pmatrix} 2 &
        \iu\\ -\iu & 2\end{pmatrix}$ (\cref{exo:b2:hermitian:4}:
        spectrum $\{3, 1\}$) en $E = \begin{pmatrix} 0 & 1\\ 1 &
        0\end{pmatrix}$ (spectrum $\{1, -1\}$): bereken het
        spectrum van $A + E$ en beide leden van de ongelijkheid.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Interlacering en
spoorongelijkheden.}
\begin{enumerate}[resume]
  \item (Interlacering van Cauchy) Zij $B$ de leidende
        hoofdondermatrix van $A$ van formaat $(n-1)\times(n-1)$.
        Bewijs dat
        \[
        \lambda_{k+1}(A) \leq \lambda_k(B) \leq \lambda_k(A)
        \qquad (1 \leq k \leq n-1)
        \]
        \emph{(beschouw $\C^{n-1} \subseteq \C^n$; daarop is
        $R_B$ de beperking van $R_A$; pas Courant--Fischer op
        beide niveaus toe)}.
  \item Herhaal: voor een hoofdondermatrix $B$ van formaat $n -
        m$ geldt $\lambda_{k+m}(A) \leq \lambda_k(B) \leq
        \lambda_k(A)$.
  \item (Schur) Zij $d_1 \geq d_2 \geq \dots \geq d_n$ de
        gesorteerde diagonaalelementen van $A$. Bewijs voor elke
        $k$:
        \[
        \sum_{i=1}^{k} d_i \leq \sum_{i=1}^{k}\lambda_i(A),
        \]
        met gelijkheid bij $k = n$ (het spoor)
        \emph{(de $k$ gekozen diagonaalelementen vormen een
        hoofdondermatrix van formaat $k\times k$; begrens haar
        spoor met vraag 12)}.
  \item (Ky Fan) Bewijs dat
        \[
        \sum_{i=1}^{k}\lambda_i(A)
        = \max\Bigl\{\sum_{i=1}^{k}\langle x_i, Ax_i\rangle
        : (x_1, \dots, x_k) \text{ orthonormaal}\Bigr\} .
        \]
  \item Ga de vragen 11 en 13 na op
        \[
        A = \begin{pmatrix} 2 & 1 & 0\\ 1 & 2 & 1\\
        0 & 1 & 2\end{pmatrix}
        \qquad
        \bigl(\operatorname{Sp} = \{2 - \sqrt2,\ 2,\
        2+\sqrt2\}\bigr)
        \]
        tegenover haar leidende blok van formaat $2\times2$
        (spectrum $\{1, 3\}$) en haar diagonaal.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De ordening van Loewner.} $A \leq B$
betekent nog altijd dat $B - A$ positief semidefiniet is; alle
matrices in dit deel zijn \omterm{def:b2:hermitian:adjoint}{hermitisch}.
\begin{enumerate}[resume]
  \item Toon aan: uit $A \leq B$ volgt $a_{ii} \leq b_{ii}$ voor
        alle $i$, $\operatorname{tr} A \leq \operatorname{tr} B$,
        en $C^\dagger AC \leq C^\dagger BC$ voor \emph{elke}
        complexe matrix $C$.
  \item Toon aan dat kwadrateren \emph{niet} monotoon is: ga voor
        \[
        A = \begin{pmatrix} 1 & 0\\ 0 & 0\end{pmatrix},
        \qquad
        B = \begin{pmatrix} 2 & 1\\ 1 & 1\end{pmatrix},
        \]
        na dat $0 \leq A \leq B$ maar $A^2 \not\leq B^2$.
  \item Bewijs dat de vierkantswortel \emph{wel} monotoon is: uit
        $0 \leq A \leq B$ volgt $\sqrt A \leq \sqrt B$ \emph{(zij
        $\mu$ een \omterm{def:b2:reduction:eigen}{eigenwaarde} van $\sqrt B - \sqrt A$ met
        \omterm{def:b2:reduction:eigen}{eigenvector} $v$ van \omterm{def:b2:nvs:norm}{norm} $1$; bereken $\langle v, (B -
        A)v\rangle = \mu\bigl(\langle v, \sqrt B\,v\rangle +
        \langle v, \sqrt A\,v\rangle\bigr)$ en bespreek)}.
  \item Bewijs dat inverteren antitoon is op positief definiete
        matrices: uit $0 < A \leq B$ volgt $B^{-1} \leq A^{-1}$
        \emph{(congrueer met $A^{-1/2}$ om te herleiden tot $I
        \leq M \Rightarrow M^{-1} \leq I$, wat in een spectrale
        basis scalair is)}.
  \item Zijn $A, B$ positief definiet. Toon aan dat de
        \omterm{def:b2:reduction:eigen}{eigenwaarden} van $AB$ (in het algemeen niet \omterm{def:b2:hermitian:adjoint}{hermitisch}!)
        reëel en positief zijn, en dat
        \[
        \lambda_{\max}(AB) \leq
        \lambda_{\max}(A)\,\lambda_{\max}(B)
        \]
        \emph{(conjugeer met $\sqrt A$: $AB \sim \sqrt
        A\,B\sqrt A$)}.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- De discrete laplaciaan, uitgewerkt.}
Zij $T_n$ de tridiagonale matrix van formaat $n \times n$ met $2$
op de diagonaal en $-1$ op de twee aangrenzende diagonalen.
\begin{enumerate}[resume]
  \item Ga met $\theta_k = \frac{k\pi}{n+1}$ na dat de vectoren
        $v_k = \bigl(\sin(j\theta_k)\bigr)_{1\leq j\leq n}$
        voldoen aan $T_nv_k = (2 - 2\cos\theta_k)\,v_k$
        \emph{(identiteit die een product in een som omzet;
        controleer de randrijen $j = 1, n$)}. Besluit:
        \[
        \operatorname{Sp}(T_n) = \Bigl\{4\sin^2
        \frac{k\pi}{2(n+1)} : 1 \leq k \leq n\Bigr\},
        \]
        alle enkelvoudig, alle positief: $T_n$ is positief
        definiet.
  \item (Een potentiaal) Klem voor een reële diagonaalmatrix $D =
        \operatorname{diag}(d_1, \dots, d_n)$ het spectrum in:
        voor elke $k$ geldt
        \[
        \lambda_k(T_n) + \min_i d_i \;\leq\;
        \lambda_k(T_n + D) \;\leq\; \lambda_k(T_n) + \max_i
        d_i .
        \]
  \item Ga de interlacering van Cauchy tussen $T_3$ en $T_2$
        expliciet na (spectra $\{2 \pm \sqrt2, 2\}$ en $\{1,
        3\}$), en interpreteer: $T_2$ is $T_3$ met één uiteinde
        van het pad verwijderd.
  \item Toon aan dat de extreme \omterm{def:b2:reduction:eigen}{eigenwaarden} voor $n \to \infty$
        voldoen aan
        \[
        \lambda_{\min}(T_n) = 4\sin^2\frac{\pi}{2(n+1)}
        \sim \frac{\pi^2}{(n+1)^2},
        \qquad
        \lambda_{\max}(T_n) \to 4 ,
        \]
        zodat het conditiegetal $\kappa_n =
        \lambda_{\max}/\lambda_{\min}$ groeit als
        $\frac{4(n+1)^2}{\pi^2}$: het discretiseren van een
        tweede afgeleide op een steeds fijner rooster is
        intrinsiek slecht geconditioneerd.
  \item Synthese. In telkens één zin: (i) waarom de variationele
        karakterisering, en niet de \omterm{def:b2:reduction:charpoly}{karakteristieke veelterm},
        \omterm{def:b2:reduction:eigen}{eigenwaarden} stabiel maakt (vragen 8--9); (ii) welke
