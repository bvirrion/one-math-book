\chapter{Reductie van endomorfismen}\label{ch:b2:reduction}

Om een endomorfisme te doorgronden, zoekt men de richtingen die het
alleen maar uitrekt. Dit hoofdstuk bouwt het gereedschap ---
\omterm{def:b2:reduction:eigen}{eigenwaarden}, karakteristieke en \omterm{def:b2:reduction:polyu}{minimale veeltermen}, het
kernontbindingslemma --- en oogst de vruchten: criteria voor
diagonalisatie en trigonalisatie, Cayley--Hamilton, de
\omterm{thm:b2:reduction:dunford}{Dunford-ontbinding}, en het berekenen van machten en exponentiëlen
waarop \cref{ch:b2:diffeq} zal teren. Overal is $E$ een
eindigdimensionale $K$-vectorruimte ($K = \R$ of $\C$) en $u \in
\mathcal{L}(E)$, met $n = \dim E$.

\section{Eigenwaarden en eigenvectoren}

\begin{definition}\label{def:b2:reduction:eigen}
Een $\lambda \in K$ heet een \emph{eigenwaarde}\index{eigenwaarde}
van $u$ wanneer $u(x) = \lambda x$ voor zekere $x \neq 0$ (een
\emph{eigenvector}); de \emph{eigenruimte} is $E_\lambda(u) =
\ker(u - \lambda\,\mathrm{id})$. De verzameling van de eigenwaarden
is het \emph{spectrum}\index{spectrum} $\operatorname{Sp}(u)$. Een
deelruimte $F$ heet \emph{stabiel} wanneer $u(F) \subseteq F$;
eigenruimten zijn stabiel, en op een stabiele deelruimte is het
geïnduceerde endomorfisme $u|_F$ gedefinieerd.
\end{definition}

\begin{theorem}[Onafhankelijkheid van de eigenruimten]\label{thm:b2:reduction:independence}
\omterm{def:b2:reduction:eigen}{Eigenvectoren} bij paarsgewijs verschillende \omterm{def:b2:reduction:eigen}{eigenwaarden} vormen een
vrije familie; gelijkwaardig: de som van de \omterm{def:b2:reduction:eigen}{eigenruimten}
$E_{\lambda_1} + \dots + E_{\lambda_r}$ (met verschillende
$\lambda_i$) is direct. In het bijzonder heeft $u$ hoogstens $n$
\omterm{def:b2:reduction:eigen}{eigenwaarden}.
\end{theorem}

\begin{proof}
Met inductie naar $r$. Stel $x_1 + \dots + x_r = 0$ met $x_i \in
E_{\lambda_i}$, en neem de uitspraak voor $r - 1$ aan. Pas $u$ toe
en trek $\lambda_r$ maal de relatie af:
\[
\sum_{i=1}^{r-1} (\lambda_i - \lambda_r)\, x_i = 0 ,
\]
dus geeft de inductiehypothese $(\lambda_i - \lambda_r)x_i = 0$,
dat wil zeggen $x_i = 0$ voor $i < r$, en daarna $x_r = 0$. Een
directe som van ruimten ongelijk aan nul in een ruimte van dimensie
$n$ heeft hoogstens $n$ termen.
\end{proof}

\begin{definition}[Karakteristieke veelterm]\label{def:b2:reduction:charpoly}
$\chi_u(X) = \det(X\,\mathrm{id} - u)$\index{karakteristieke
veelterm} --- in elke basis berekend als $\det(XI_n - A)$, een
monische veelterm van graad $n$, invariant onder gelijkvormigheid
(\cref{thm:b2:linalg:detrules}). Haar nulpunten in $K$ zijn precies
de \omterm{def:b2:reduction:eigen}{eigenwaarden} ($\lambda$ \omterm{def:b2:reduction:eigen}{eigenwaarde} $\iff u -
\lambda\,\mathrm{id}$ niet injectief $\iff \chi_u(\lambda) = 0$),
en
\[
\chi_u(X) = X^n - (\operatorname{tr} u)\, X^{n-1} + \dots +
(-1)^n \det u .
\]
De \emph{algebraïsche multipliciteit}\index{algebraïsche
multipliciteit} $m_\lambda$ van een \omterm{def:b2:reduction:eigen}{eigenwaarde} is haar
multipliciteit als nulpunt van $\chi_u$; de \emph{meetkundige
multipliciteit}\index{meetkundige multipliciteit} is $\dim
E_\lambda$, en er geldt $1 \leq \dim E_\lambda \leq m_\lambda$.
\end{definition}

\begin{proof}[Bewijs van de genoemde feiten]
De uitspraken over de coëfficiënten: ontwikkel $\det(XI - A)$ met
de permutatieformule; de identieke permutatie draagt $\prod_i (X -
a_{ii}) = X^n - (\sum a_{ii})X^{n-1} + \dots$ bij, en elke andere
permutatie houdt hoogstens $n - 2$ diagonaalplaatsen vast en draagt
dus graad $\leq n - 2$ bij: de twee hoogste coëfficiënten zijn
zoals beweerd; en $X = 0$ geeft de constante term $\det(-A) =
(-1)^n\det A$.

Meetkundig $\leq$ algebraïsch: zij $d = \dim E_\lambda$ en vul een
basis van $E_\lambda$ aan tot een basis van $E$; de matrix van $u$
is dan blokbovendriehoekig met linksboven het blok $\lambda I_d$,
zodat $\chi_u(X) = (X - \lambda)^d\, \chi_{\text{(onderste
blok)}}(X)$: de multipliciteit van $\lambda$ is minstens $d$.
\end{proof}

\begin{example}[Dezelfde $\chi$, andere meetkunde]\label{ex:b2:reduction:samechi}
De matrices
\[
\begin{pmatrix}2 & 0\\ 0 & 2\end{pmatrix}
\qquad\text{en}\qquad
\begin{pmatrix}2 & 1\\ 0 & 2\end{pmatrix}
\]
hebben dezelfde \omterm{def:b2:reduction:charpoly}{karakteristieke veelterm} $(X - 2)^2$, hetzelfde
spoor, dezelfde \omterm{def:b2:linalg:det}{determinant} en hetzelfde \omterm{def:b2:reduction:eigen}{spectrum} --- en toch zijn
zij niet gelijkvormig: bij de eerste heeft $E_2$ dimensie $2$
(\omterm{def:b2:reduction:charpoly}{meetkundige multipliciteit} $2$), bij de tweede dimensie $1$. De
\omterm{def:b2:reduction:charpoly}{karakteristieke veelterm} ziet alleen de \omterm{def:b2:reduction:charpoly}{algebraïsche
multipliciteiten}; de dimensies van de \omterm{def:b2:reduction:eigen}{eigenruimten} zijn de fijnere
invariant, en de \omterm{def:b2:reduction:polyu}{minimale veelterm} beslist ($X - 2$ tegenover $(X -
2)^2$). Moraal voor elke bespreking van diagonaliseerbaarheid:
$\chi$ stelt de kandidatenlijst op, maar de kernen brengen de
stemmen uit.
\end{example}

\begin{definition}[Diagonaliseerbaar, trigonaliseerbaar]\label{def:b2:reduction:diag}
$u$ heet \emph{diagonaliseerbaar}\index{diagonaliseerbaar} wanneer
$E$ een basis van \omterm{def:b2:reduction:eigen}{eigenvectoren} bezit (voor een matrix: gelijkvormig
met een diagonaalmatrix), en
\emph{trigonaliseerbaar}\index{trigonaliseerbaar} wanneer haar
matrix in een zekere basis bovendriehoekig is.
\end{definition}

\begin{theorem}[Criteria voor diagonaliseerbaarheid]\label{thm:b2:reduction:diagcrit}
De volgende uitspraken zijn gelijkwaardig:
\begin{enumerate}
  \item $u$ is \omterm{def:b2:reduction:diag}{diagonaliseerbaar};
  \item $E = \bigoplus_{\lambda \in \operatorname{Sp} u} E_\lambda$;
  \item $\chi_u$ valt over $K$ in lineaire factoren uiteen en $\dim
        E_\lambda = m_\lambda$ voor elke \omterm{def:b2:reduction:eigen}{eigenwaarde};
  \item (voldoende, niet nodig) $\chi_u$ heeft $n$ verschillende
        nulpunten in $K$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1 $\iff$ 2): een basis van \omterm{def:b2:reduction:eigen}{eigenvectoren} valt uiteen in basissen
van de $E_\lambda$, en omgekeerd geeft het aaneenschakelen van
basissen van de directe termen een basis van $E$
(\cref{thm:b2:reduction:independence} maakt de som direct; de
gelijkheid van dimensies maakt haar alles).

(2 $\iff$ 3): in de diagonale basis valt $\chi_u = \prod (X -
\lambda)^{\dim E_\lambda}$ uiteen, met kloppende multipliciteiten.
Omgekeerd, stel dat $\chi_u$ uiteenvalt met overal $\dim E_\lambda
= m_\lambda$; dan heeft de directe som van de \omterm{def:b2:reduction:eigen}{eigenruimten} (direct
volgens \cref{thm:b2:reduction:independence}) dimensie
\[
\sum_{\lambda}\dim E_\lambda = \sum_{\lambda} m_\lambda =
\deg\chi_u = n ,
\]
waarbij de middelste gelijkheid geldt omdat de graad van een
veelterm die in lineaire factoren uiteenvalt de som van haar
nulpuntmultipliciteiten is: de som is heel $E$. Merk op waar elke
hypothese werd gebruikt: het uiteenvallen vulde de graad, de
gelijkheid van multipliciteiten vulde de dimensies.

(4 $\Rightarrow$ 1): $n$ verschillende \omterm{def:b2:reduction:eigen}{eigenwaarden} leveren $n$
onafhankelijke \omterm{def:b2:reduction:eigen}{eigenvectoren}
(\cref{thm:b2:reduction:independence}): een basis.
\end{proof}

\begin{method}[Diagonaliseerbaarheid beslissen]\label{met:b2:reduction:decide}
Test in de praktijk in deze volgorde --- elke stap kan het werk al
afmaken. (1) Dient zich een annihilerende veelterm aan die in
lineaire factoren met enkelvoudige nulpunten uiteenvalt ($u^2 =
\mathrm{id}$, $u^2 = u$, $u^k = \mathrm{id}$)? Zo ja: dan is $u$
\omterm{def:b2:reduction:diag}{diagonaliseerbaar}, zonder rekenwerk
(\cref{cor:b2:reduction:minpolycrit} hieronder). (2) Bereken
$\chi_u$; heeft zij $n$ verschillende nulpunten in $K$, dan is $u$
\omterm{def:b2:reduction:diag}{diagonaliseerbaar} (\cref{thm:b2:reduction:diagcrit}~(4)). (3) Zo
niet, vergelijk dan, uitsluitend voor elk meervoudig nulpunt
$\lambda$, de waarde $\dim\ker(u - \lambda\,\mathrm{id})$ met de
multipliciteit $m_\lambda$: elk tekort doodt de
diagonaliseerbaarheid, en overal gelijkheid bewijst haar. Bereken
nooit de \omterm{def:b2:reduction:eigen}{eigenruimten} van enkelvoudige nulpunten (hun dimensie is
gedwongen $1$), en trigonaliseer nooit alleen om te beslissen.
\end{method}

\begin{example}[Diagonalisatie aan het werk]\label{ex:b2:reduction:diagwork}
$A = I + J = \begin{psmallmatrix}2 & 1 & 1\\ 1 & 2 & 1\\ 1 & 1 &
2\end{psmallmatrix}$, met $J$ de matrix vol enen: uit
$\operatorname{Sp}(J) = \{3, 0\}$
(\cref{ex:b2:linalg:onesmatrix}) volgt $\operatorname{Sp}(A) = \{4,
1\}$, met \omterm{def:b2:reduction:eigen}{eigenruimten} $\R(1,1,1)$ en het vlak $\{x + y + z = 0\}$:
dimensies $1 + 2 = 3$, dus \omterm{def:b2:reduction:diag}{diagonaliseerbaar}
(\cref{thm:b2:reduction:diagcrit}~(2)). Machten zonder ook maar
één basisovergangsmatrix: met $\Pi = J/3$ de projector op
$\R(1,1,1)$ geldt
\[
A = 4\,\Pi + 1\cdot(I - \Pi)
\quad\Longrightarrow\quad
A^k = 4^k\,\Pi + (I - \Pi)
= \frac{4^k - 1}{3}\,J + I .
\]
(Controle voor $k = 1$: $\frac{4-1}3 J + I = A$.) Het inzicht om te
onthouden: zodra de \omterm{def:b2:reduction:eigen}{eigenruimten} zichtbaar zijn, berekenen
spectrale \emph{projectoren} machten sneller dan $PDP^{-1}$ ooit
zal doen --- en de formule toont de dynamica: $A^k$ groeit als
$4^k$ langs $(1,1,1)$ en blijft op het orthogonale vlak staan waar
het staat.
\end{example}

\begin{theorem}[Trigonalisatie]\label{thm:b2:reduction:trigonalization}
$u$ is \omterm{def:b2:reduction:diag}{trigonaliseerbaar} over $K$ dan en slechts dan als $\chi_u$
over $K$ in lineaire factoren uiteenvalt. In het bijzonder is elk
endomorfisme van een $\C$-vectorruimte
\omterm{def:b2:reduction:diag}{trigonaliseerbaar}.\index{trigonalisatie}
\end{theorem}

\begin{proof}
($\Rightarrow$) De \omterm{def:b2:reduction:charpoly}{karakteristieke veelterm} van een driehoeksmatrix
is $\prod(X - t_{ii})$: die valt uiteen.

($\Leftarrow$) Inductie naar $n$. Omdat $\chi_u$ uiteenvalt, heeft
zij een nulpunt $\lambda$: kies een \omterm{def:b2:reduction:eigen}{eigenvector} $e_1$. In een basis
die met $e_1$ begint, is de matrix $\begin{pmatrix} \lambda &
\ast\\ 0 & B\end{pmatrix}$, en $\chi_u = (X - \lambda)\chi_B$: dus
valt ook $\chi_B$ uiteen. De inductiehypothese toegepast op de
$(n-1) \times (n-1)$-matrix $B$ geeft een inverteerbare $Q$ met
$Q^{-1}BQ$ bovendriehoekig; de hele matrix conjugeren met
$\begin{pmatrix}1 & 0\\ 0 & Q\end{pmatrix}$ maakt haar
driehoekig.
\end{proof}

\begin{example}[Met de hand trigonaliseren]\label{ex:b2:reduction:trigwork}
$B = \begin{pmatrix}3 & -1\\ 1 & 1\end{pmatrix}$: hier is $\chi_B =
X^2 - 4X + 4 = (X - 2)^2$, en $\ker(B - 2I) =
\ker\begin{psmallmatrix}1 & -1\\ 1 & -1\end{psmallmatrix}$ is de
rechte opgespannen door $e_1' = (1, 1)$: één \omterm{def:b2:reduction:eigen}{eigenwaarde}, een
eendimensionale \omterm{def:b2:reduction:eigen}{eigenruimte} --- niet \omterm{def:b2:reduction:diag}{diagonaliseerbaar}, maar wel
\omterm{def:b2:reduction:diag}{trigonaliseerbaar} (\cref{thm:b2:reduction:trigonalization}). Vul de
basis aan met $e_2' = (1, 0)$ en reken:
\[
u(e_1') = (2, 2) = 2e_1',
\qquad
u(e_2') = (3, 1) = 1\cdot e_1' + 2\, e_2' ,
\]
zodat de matrix in de basis $(e_1', e_2')$ gelijk is aan $T =
\begin{psmallmatrix}2 & 1\\ 0 & 2\end{psmallmatrix}$. Het inzicht
om te onthouden: de diagonaal van $T$ lag vast (beide elementen
moeten de dubbele \omterm{def:b2:reduction:eigen}{eigenwaarde} $2$ zijn); alleen het hoekelement
hing van de keuze van $e_2'$ af, en door $e_2'$ te herschalen kan
het elke waarde ongelijk aan nul krijgen --- de hardnekkige ``$1$''
is de schaduw van het nilpotente deel dat Dunford zal isoleren.
\end{example}

\section{Veeltermen van een endomorfisme}

\begin{definition}\label{def:b2:reduction:polyu}
Zet voor $P = \sum a_k X^k \in K[X]$ de afbeelding $P(u) = \sum a_k
u^k \in \mathcal{L}(E)$. De afbeelding $P \mapsto P(u)$ is een
morfisme van algebra's $K[X] \to \mathcal{L}(E)$
(\cref{def:b2:structures:algebra}); haar kern $\{P : P(u) = 0\}$ is
een \omterm{def:b2:structures:ideal}{ideaal} van $K[X]$, ongelijk aan nul (de familie
$(\mathrm{id}, u, \dots, u^{n^2})$ is afhankelijk in het
$n^2$-dimensionale $\mathcal{L}(E)$), en wordt dus \omterm{def:b2:structures:generated}{voortgebracht}
door een unieke monische veelterm $\mu_u$: de \emph{minimale
veelterm}\index{minimale veelterm}
(\cref{thm:b2:structures:principal}).
\end{definition}

\begin{proposition}\label{prop:b2:reduction:minpoly}
\begin{enumerate}
  \item $P(u) = 0 \iff \mu_u \mid P$; de \omterm{def:b2:reduction:eigen}{eigenwaarden} van $u$ zijn
        nulpunten van elke annihilerende veelterm, en de nulpunten
        van $\mu_u$ zijn \emph{precies} de \omterm{def:b2:reduction:eigen}{eigenwaarden}.
  \item Is $F$ stabiel, dan geldt $\mu_{u|_F} \mid \mu_u$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) De deelbaarheid is de definitie van een voortbrenger. Geldt
$u(x) = \lambda x$ met $x \neq 0$, dan is $0 = P(u)(x) =
P(\lambda)x$, dus $P(\lambda) = 0$: \omterm{def:b2:reduction:eigen}{eigenwaarden} zijn nulpunten van
annihilerende veeltermen, in het bijzonder van $\mu_u$. Omgekeerd,
is $\lambda$ een nulpunt, dan is $\mu_u = (X - \lambda)Q$ met
$Q(u) \neq 0$ (de graad van $\mu_u$ is minimaal): kies $y$ met
$Q(u)(y) \neq 0$; dan toont $(u - \lambda)(Q(u)(y)) = \mu_u(u)(y) =
0$ de \omterm{def:b2:reduction:eigen}{eigenvector} $Q(u)(y)$.

(2) Er geldt $\mu_u(u|_F) = \mu_u(u)|_F = 0$; pas nu (1) toe op
$u|_F$.
\end{proof}

\begin{example}[Minimale veeltermen met de hand gevonden]\label{ex:b2:reduction:minpolyhand}
De \omterm{def:b2:reduction:polyu}{minimale veelterm} wordt gevonden door achtereenvolgende graden
te proberen. Voor de matrix $J \in \mathcal{M}_3(\R)$ vol enen:
$J \neq \lambda I$ (graad $1$ valt af), en $J^2 = 3J$, dus
\[
\mu_J = X^2 - 3X = X(X - 3) :
\]
graad $2$, uiteenvallend, met enkelvoudige nulpunten --- dus is $J$
\omterm{def:b2:reduction:diag}{diagonaliseerbaar} met \omterm{def:b2:reduction:eigen}{spectrum} $\{0, 3\}$
(\cref{cor:b2:reduction:minpolycrit} hieronder), wat
\cref{ex:b2:linalg:onesmatrix} bevestigt zonder ook maar één
\omterm{def:b2:linalg:det}{determinant}. Voor de verwisselingsmatrix $A$ uit
\cref{ex:b2:reduction:projectorswork}: uit $A \neq \pm I$ en $A^2 =
I$ volgt $\mu_A = X^2 - 1$. In beide gevallen is het patroon
hetzelfde: raad uit de structuur een identiteit van lage graad
(rang één dwingt $J^2 = (\operatorname{tr}J)\,J$ af; een involutie
dwingt $A^2 = I$ af) en ga daarna na dat geen echte deler
annihileert. \omterm{def:b2:reduction:polyu}{Minimale veeltermen} worden meestal \emph{gevonden},
niet uit $\chi$ berekend.
\end{example}

\begin{theorem}[Kernontbindingslemma]\label{thm:b2:reduction:kernels}
Is $P = P_1 P_2 \cdots P_r$ met paarsgewijs onderling ondeelbare
$P_i$, dan geldt\index{kernontbindingslemma}
\[
\ker P(u) = \ker P_1(u) \oplus \dots \oplus \ker P_r(u),
\]
en de projecties op de termen zijn veeltermen in $u$.
\end{theorem}

\begin{proof}
Het volstaat het geval $r = 2$ te behandelen en daarna te
induceren. Bézout in $K[X]$ (\cref{thm:b2:structures:principal})
geeft $U P_1 + V P_2 = 1$, dus voor elke $x$
\[
x = \underbrace{U(u)P_1(u)(x)}_{=:\,x_2}
+ \underbrace{V(u)P_2(u)(x)}_{=:\,x_1}.
\]
Ligt $x \in \ker P(u)$, dan is $P_2(u)(x_2) = U(u)\,P(u)(x) = 0$
(veeltermen in $u$ commuteren), dus $x_2 \in \ker P_2(u)$, en
symmetrisch $x_1 \in \ker P_1(u)$: de som vult $\ker P(u)$; beide
termen liggen in $\ker P(u)$ omdat $P_i \mid P$. Directheid: uit $x
\in \ker P_1(u) \cap \ker P_2(u)$ volgt $x = U(u)P_1(u)x +
V(u)P_2(u)x = 0$. De formules voor $x_1$ en $x_2$ tonen de
projecties als $V(u)P_2(u)$ en $U(u)P_1(u)$.
\end{proof}

\begin{example}[Het kernlemma met expliciete projectoren]\label{ex:b2:reduction:projectorswork}
Zij $A = \begin{psmallmatrix}0 & 1 & 0\\ 1 & 0 & 0\\ 0 & 0 &
1\end{psmallmatrix}$ (verwissel de eerste twee coördinaten). Dan is
$A^2 = I$: de veelterm $X^2 - 1 = (X - 1)(X + 1)$ annihileert $A$,
haar factoren zijn onderling ondeelbaar, en Bézout is expliciet:
\[
\frac{1}{2}(X + 1) - \frac12(X - 1) = 1 .
\]
Volgens het bewijs van \cref{thm:b2:reduction:kernels} zijn de
projecties op $\ker(A - I)$ en $\ker(A + I)$ de veeltermen in $A$
\[
\pi_+ = \frac{A + I}{2} = \frac12\begin{pmatrix}
1 & 1 & 0\\ 1 & 1 & 0\\ 0 & 0 & 2\end{pmatrix},
\qquad
\pi_- = \frac{I - A}{2} = \frac12\begin{pmatrix}
1 & -1 & 0\\ -1 & 1 & 0\\ 0 & 0 & 0\end{pmatrix}.
\]
Controle: $\pi_+ + \pi_- = I$, $\pi_+\pi_- = 0$, $\pi_\pm^2 =
\pi_\pm$, en de beelden zijn het vlak $\{x = y\}$ (symmetrische
vectoren, \omterm{def:b2:reduction:eigen}{eigenwaarde} $1$) en de rechte $\R(1, -1, 0)$
(antisymmetrisch, \omterm{def:b2:reduction:eigen}{eigenwaarde} $-1$). Het kernlemma is geen
uitspraak over bestaan alleen: de coëfficiënten van Bézout
\emph{zijn} de formules voor de projectoren.
\end{example}

\begin{example}[Projectoren berekenen ook de exponentiële]\label{ex:b2:reduction:expprojectors}
Dezelfde verwisselingsmatrix, één opbrengst verder. Omdat $A =
\pi_+ - \pi_-$ met projectoren die in algebraïsche zin orthogonaal
zijn ($\pi_+\pi_- = 0$), voldoet elke macht aan $A^k = \pi_+ +
(-1)^k\pi_-$, en de exponentiële reeks hergroepeert zich per
projector:
\[
\eu^{tA} = \sum_k \frac{t^k}{k!}\bigl(\pi_+ +
(-1)^k\pi_-\bigr)
= \eu^{t}\,\pi_+ + \eu^{-t}\,\pi_- =
\begin{pmatrix}
\cosh t & \sinh t & 0\\
\sinh t & \cosh t & 0\\
0 & 0 & \eu^{t}
\end{pmatrix}.
\]
(Controle voor $t = 0$: de identiteit; afgeleide in $0$: $A$.) De
eigenontbinding zet een matrixreeks om in twee scalaire reeksen ---
precies het mechanisme dat \cref{ch:b2:diffeq} op elk
\omterm{def:b2:reduction:diag}{diagonaliseerbaar} stelsel zal laten lopen, en de reden dat
hyperbolische functies symmetrische koppelingen besturen.
\end{example}

\begin{corollary}[Diagonaliseerbaarheid via de minimale veelterm]\label{cor:b2:reduction:minpolycrit}
$u$ is \omterm{def:b2:reduction:diag}{diagonaliseerbaar} $\iff$ $\mu_u$ valt over $K$ uiteen met
\emph{enkelvoudige} nulpunten $\iff$ een zekere annihilerende
veelterm van $u$ valt uiteen met enkelvoudige nulpunten.
\end{corollary}

\begin{proof}
Geldt $P(u) = 0$ met $P = \prod_{i}(X - \lambda_i)$ (verschillende
$\lambda_i$), dan geeft het lemma $E = \ker P(u) = \bigoplus_i
\ker(u - \lambda_i)$: een directe som van \omterm{def:b2:reduction:eigen}{eigenruimten}, dus is $u$
\omterm{def:b2:reduction:diag}{diagonaliseerbaar} (\cref{thm:b2:reduction:diagcrit}). Omgekeerd
wordt een \omterm{def:b2:reduction:diag}{diagonaliseerbare} $u$ gedood door $\prod_{\lambda \in
\operatorname{Sp}u}(X - \lambda)$ (die doodt elke \omterm{def:b2:reduction:eigen}{eigenruimte}), en
die veelterm valt uiteen met enkelvoudige nulpunten; en $\mu_u$
deelt haar terwijl beide dezelfde nulpunten hebben
(\cref{prop:b2:reduction:minpoly}): $\mu_u$ is precies dat product.
\end{proof}

\begin{example}\label{ex:b2:reduction:projections}
Projecties voldoen aan $p^2 = p$: geannihileerd door $X(X-1)$, dat
uiteenvalt met enkelvoudige nulpunten --- dus \omterm{def:b2:reduction:diag}{diagonaliseerbaar} met
\omterm{def:b2:reduction:eigen}{spectrum} $\subseteq \{0, 1\}$, en $E = \ker p \oplus \ker(p -
\mathrm{id})$: de meetkundige analyse uit bachelorjaar 1, in één
regel opnieuw bewezen. Symmetrieën ($s^2 = \mathrm{id}$,
\omterm{def:b2:linalg:annihilator}{annihilator} $X^2 - 1$): \omterm{def:b2:reduction:diag}{diagonaliseerbaar} zodra
$\operatorname{char} K \neq 2$, met \omterm{def:b2:reduction:eigen}{spectrum} $\subseteq \{\pm 1\}$.
Een endomorfisme met $u^3 = u^2$ en $u^2 \neq u$: geannihileerd
door $X^2(X - 1)$, en \emph{niet} noodzakelijk \omterm{def:b2:reduction:diag}{diagonaliseerbaar}
--- het criterium ziet dat (het dubbele nulpunt $0$ moet worden
getest: \omterm{def:b2:reduction:diag}{diagonaliseerbaar} dan en slechts dan als bovendien $\ker
u^2 = \ker u$).
\end{example}

\begin{example}[Het lichaam beslist: een rotatie in $\R^3$]\label{ex:b2:reduction:rotation3d}
Zij $R$ de kwartslag om de $z$-as:
\[
R = \begin{pmatrix}
0 & -1 & 0\\
1 & 0 & 0\\
0 & 0 & 1
\end{pmatrix},
\qquad
\chi_R = (X - 1)(X^2 + 1).
\]
Over $\R$: de enige \omterm{def:b2:reduction:eigen}{eigenwaarde} is $1$, met als \omterm{def:b2:reduction:eigen}{eigenruimte} de as
$\R e_3$ --- één rechte van vaste vectoren en geen verdere
reductie: $R$ is in $\mathcal{M}_3(\R)$ noch \omterm{def:b2:reduction:diag}{diagonaliseerbaar} noch
\omterm{def:b2:reduction:diag}{trigonaliseerbaar} ($\chi_R$ valt niet uiteen). Over $\C$: drie
verschillende \omterm{def:b2:reduction:eigen}{eigenwaarden} $1, \iu, -\iu$, dus is $R$
\omterm{def:b2:reduction:diag}{diagonaliseerbaar}, met \omterm{def:b2:reduction:eigen}{eigenvectoren} $e_3$ en $e_1 \mp \iu e_2$. De
meetkunde was in de algebra al hoorbaar: rotaties in het vlak
hebben geen reële invariante richtingen, en de complexe
\omterm{def:b2:reduction:eigen}{eigenwaarden} $\pm\iu$ van modulus $1$ bewaren de hoek
($\pm\frac\pi2$) die de reële matrix alleen door coördinaten te
mengen kan uitdrukken.
\end{example}

\begin{example}[Minimaal tegenover karakteristiek]\label{ex:b2:reduction:minvschar}
Voor $D = \operatorname{diag}(2, 2, 3)$ is $\chi_D = (X - 2)^2(X -
3)$ maar $\mu_D = (X - 2)(X - 3)$, want $(D - 2I)(D - 3I) = 0$ (ga
het op de canonieke basis na), terwijl geen van beide factoren $D$
alleen doodt. Voor het verschuivingsblok $N =
\begin{psmallmatrix}0 & 1\\ 0 & 0\end{psmallmatrix} \oplus (3)$,
dat wil zeggen $N' = \begin{psmallmatrix}0 & 1 & 0\\ 0 & 0 & 0\\ 0
& 0 & 3\end{psmallmatrix}$, geldt $\chi_{N'} = X^2(X - 3)$
\emph{én} $\mu_{N'} = X^2(X - 3)$ --- het dubbele nulpunt is hier
werkelijk nodig, omdat $N'$ aan de kernkant niet \omterm{def:b2:reduction:diag}{diagonaliseerbaar}
is ($N'e_2 = e_1 \neq 0$). Vuistregel: $\mu$ en $\chi$ hebben
dezelfde nulpunten (\cref{prop:b2:reduction:minpoly}); de
multipliciteit in $\mu$ meet de grootte van het grootste
nilpotente blok, die in $\chi$ de totale dimensie van de
karakteristieke deelruimte.
\end{example}

\begin{theorem}[Cayley--Hamilton]\label{thm:b2:reduction:cayleyhamilton}
Er geldt $\chi_u(u) = 0$; bijgevolg $\mu_u \mid \chi_u$ en $\deg
\mu_u \leq n$.\index{stelling van Cayley--Hamilton}
\end{theorem}

\begin{proof}
Leg $x \neq 0$ vast en zij $d$ maximaal zodanig dat $(x, u(x),
\dots, u^{d-1}(x))$ vrij is; schrijf
\[
u^d(x) = -a_0 x - a_1 u(x) - \dots - a_{d-1}u^{d-1}(x),
\]
en zet $P_x = X^d + a_{d-1}X^{d-1} + \dots + a_0$, zodat $P_x(u)(x)
= 0$. Vul de vrije familie aan tot een basis van $E$: daarin heeft
$u$ de blokvorm $\begin{pmatrix} C & \ast\\ 0 & D\end{pmatrix}$,
waarbij $C$ de begeleidende matrix van $P_x$ is, waarvan de
\omterm{def:b2:reduction:charpoly}{karakteristieke veelterm} $P_x$ is (ontwikkel $\det(XI - C)$ langs de
eerste kolom, met inductie naar $d$). Bijgevolg is $\chi_u = P_x
\cdot \chi_D$, en
\[
\chi_u(u)(x) = \chi_D(u)\bigl(P_x(u)(x)\bigr) = 0 .
\]
Het argument geldt voor elke $x$: dus $\chi_u(u) = 0$.
\end{proof}

\begin{example}[Cayley--Hamilton aan het werk]\label{ex:b2:reduction:chwork}
$A = \begin{pmatrix} 1 & 2\\ 3 & 4\end{pmatrix}$: hier is $\chi_A =
X^2 - 5X - 2$, dus $A^2 = 5A + 2I$. Elke macht van $A$ klapt in
elkaar tot een combinatie van $I$ en $A$:
\[
A^4 = (5A + 2I)^2 = 25A^2 + 20A + 4I = 145A + 54I =
\begin{pmatrix} 199 & 290\\ 435 & 634\end{pmatrix},
\]
en de inverse krijgen we er gratis bij: uit $A(A - 5I) = 2I$ volgt
\[
A^{-1} = \tfrac12(A - 5I)
= \begin{pmatrix} -2 & 1\\ 3/2 & -1/2\end{pmatrix}.
\]
Het inzicht om te onthouden: Cayley--Hamilton perst de hele algebra
$K[A]$ samen tot $\operatorname{Vect}(I, A, \dots, A^{n-1})$ ---
$\dim K[A] = \deg\mu_A \leq n$, hoe hoog de machten ook zijn die je
nodig hebt.
\end{example}

\begin{remark}[Klassieke valkuilen]\label{rem:b2:reduction:pitfalls}
(i) \omterm{def:b2:reduction:eigen}{Eigenwaarden} tellen niet op: $\operatorname{Sp}(A + B)$ is niet
$\operatorname{Sp}A + \operatorname{Sp}B$, en een som van
\omterm{def:b2:reduction:diag}{diagonaliseerbare} matrices hoeft niet \omterm{def:b2:reduction:diag}{diagonaliseerbaar} te zijn ---
$\begin{psmallmatrix}1 & 1\\ 0 & 0\end{psmallmatrix} +
\begin{psmallmatrix}0 & 0\\ 0 & 1\end{psmallmatrix} =
\begin{psmallmatrix}1 & 1\\ 0 & 1\end{psmallmatrix}$ is een som van
twee \omterm{def:b2:reduction:diag}{diagonaliseerbare} matrices (elk met verschillende
\omterm{def:b2:reduction:eigen}{eigenwaarden}) en is niet \omterm{def:b2:reduction:diag}{diagonaliseerbaar}; alleen
\emph{commuterende} families gedragen zich netjes
(\cref{exo:b2:reduction:9}). (ii) ``$\chi_u$ valt uiteen'' is een
hypothese over het \emph{lichaam}: een rotatie in het vlak heeft
$\chi = X^2 - 2\cos\theta\,X + 1$, die over $\C$ wel en over $\R$
niet uiteenvalt --- \omterm{def:b2:reduction:diag}{diagonaliseerbaar} in $\mathcal{M}_2(\C)$, niet
\omterm{def:b2:reduction:diag}{trigonaliseerbaar} in $\mathcal{M}_2(\R)$. (iii) De ongelijkheid
loopt van meetkundig naar algebraïsch, nooit andersom; alleen
$\dim E_\lambda \geq 1$ nagaan bewijst niets over
diagonaliseerbaarheid. (iv) $\mu_u$ is niet $\chi_u$: gelijkheid
geldt precies wanneer elke \omterm{def:b2:reduction:eigen}{eigenwaarde} één enkele blokketen heeft
(bijvoorbeeld bij begeleidende matrices, de weekendopgave van dit
hoofdstuk); $\chi$ gebruiken waar $\mu$ nodig is, blaast elke
machtsberekening op. (v) De $d$ en $\nu$ van Dunford zijn
veeltermen in $u$ --- een ontbinding $u = d' + \nu'$ met de juiste
eigenschappen maar met $d'\nu' \neq \nu'd'$ is \emph{geen} Dunford
en is nooit eenduidig.
\end{remark}

\begin{remark}[Waar dit hoofdstuk wordt gebruikt]
De reductie is het werkpaard van de rest van het boek: machten en
exponentiëlen van matrices drijven de lineaire
differentiaalstelsels van \cref{ch:b2:diffeq}; de spectraalstelling
van \cref{ch:b2:quadratic} is diagonalisatie, orthogonaal gemaakt;
genererende functies (\cref{ch:b2:genfun}) leiden de asymptotiek
van de recurrenties uit de weekendopgave van dit hoofdstuk
analytisch opnieuw af. In het volume van bachelorjaar 3 loopt
hetzelfde programma in oneindige dimensie: de spectraaltheorie van
compacte zelftoegevoegde operatoren, waar rijen van \omterm{def:b2:reduction:eigen}{eigenwaarden} de
plaats van eindige spectra innemen, en de theorie van
Perron--Frobenius voor positieve matrices, die verklaart
\emph{waarom} de dominante \omterm{def:b2:reduction:eigen}{eigenwaarden} van telproblemen positief
en enkelvoudig zijn.
\end{remark}

\section{Nilpotenten en de Dunford-ontbinding}

\begin{proposition}[Nilpotente endomorfismen]\label{prop:b2:reduction:nilpotent}
Voor $u$ waarvan $\chi_u$ uiteenvalt zijn de volgende uitspraken
gelijkwaardig: $u^n = 0$; $u^k = 0$ voor zekere $k$;
$\operatorname{Sp}(u) = \{0\}$; $\chi_u = X^n$; $u$ is
\omterm{def:b2:reduction:diag}{trigonaliseerbaar} met nuldiagonaal. Een \omterm{prop:b2:reduction:nilpotent}{nilpotent endomorfisme}
heeft $\mu_u = X^{\text{(nilpotentie-index)}}$, en de index is
$\leq n$.\index{nilpotent endomorfisme}
\end{proposition}

\begin{proof}
Uit $u^k = 0$ volgt dat elke \omterm{def:b2:reduction:eigen}{eigenwaarde} een nulpunt van $X^k$ is:
\omterm{def:b2:reduction:eigen}{spectrum} $\{0\}$ (niet leeg zodra $\chi$ uiteenvalt --- over $\C$
altijd). Dan is $\chi_u = X^n$ (alle nulpunten zijn nul) en geeft
Cayley--Hamilton $u^n = 0$; trigonalisatie
(\cref{thm:b2:reduction:trigonalization}) zet nullen op de
diagonaal (de diagonaal draagt de \omterm{def:b2:reduction:eigen}{eigenwaarden}). Omgekeerd, zij $A$
strikt bovendriehoekig: $a_{ij} = 0$ voor $j \leq i$. We tonen met
inductie aan dat
\[
(A^k)_{ij} = 0 \qquad \text{zodra } j \leq i + k - 1,
\]
dat wil zeggen: elke macht duwt het nulgebied één diagonaal hoger.
Voor $k = 1$ is dat de hypothese. Voor de stap:
\[
(A^{k+1})_{ij} = \sum_{\ell} (A^k)_{i\ell}\,a_{\ell j} ,
\]
en elke term verdwijnt: ofwel $\ell \leq i + k - 1$ (de eerste
factor is $0$ volgens de inductiehypothese), ofwel $\ell \geq i +
k$, en dan doodt $j \leq i + k \leq \ell$ de tweede factor. Bij $k
= n$ geldt de voorwaarde $j \leq i + n - 1$ voor alle $i, j \leq
n$: dus $A^n = 0$. De \omterm{def:b2:reduction:polyu}{minimale veelterm} deelt $X^n$, en de
annihilatie definieert de index.
\end{proof}

\begin{theorem}[Dunford-ontbinding]\label{thm:b2:reduction:dunford}
Neem aan dat $\chi_u$ over $K$ uiteenvalt (automatisch voor $K =
\C$). Dan bestaat er een \emph{uniek} paar $(d, \nu)$ met
\[
u = d + \nu, \qquad d \text{ diagonaliseerbaar}, \quad \nu
\text{ nilpotent}, \quad d\nu = \nu d ,
\]
en bovendien zijn $d$ en $\nu$ veeltermen in
$u$.\index{Dunford-ontbinding}
\end{theorem}

\begin{proof}
\emph{Bestaan.} Schrijf $\chi_u = \prod_{i=1}^{r} (X -
\lambda_i)^{m_i}$ (verschillende $\lambda_i$) en zet $N_i = \ker(u
- \lambda_i)^{m_i}$, de \emph{karakteristieke deelruimten}. Volgens
Cayley--Hamilton en het kernlemma
(\cref{thm:b2:reduction:kernels}) is
\[
E = N_1 \oplus \dots \oplus N_r ,
\]
met projecties $\pi_i$ die veeltermen in $u$ zijn; elke $N_i$ is
stabiel (veeltermen in $u$ commuteren met $u$). Definieer $d =
\sum_i \lambda_i \pi_i$: een veelterm in $u$, \omterm{def:b2:reduction:diag}{diagonaliseerbaar}
(zij werkt op $N_i$ als $\lambda_i$, zodat $E$ in haar \omterm{def:b2:reduction:eigen}{eigenruimten}
uiteenvalt). Dan is $\nu = u - d$ een veelterm in $u$ (en
commuteert dus met $d$), en op elke $N_i$ werkt zij als $u -
\lambda_i$, waar $(u - \lambda_i)^{m_i} = 0$: dus $\nu^{\max m_i} =
0$ op elke term, en $\nu$ is nilpotent.

\emph{Eenduidigheid.} Zij $u = d' + \nu'$ een tweede zulk paar.
Omdat $d'$ en $\nu'$ met elkaar commuteren, commuteren zij met $u =
d' + \nu'$ en dus met elke veelterm in $u$ --- in het bijzonder met
$d$ en $\nu$. Dan is $d - d'$ \omterm{def:b2:reduction:diag}{diagonaliseerbaar} (twee commuterende
\omterm{def:b2:reduction:diag}{diagonaliseerbare} afbeeldingen zijn simultaan \omterm{def:b2:reduction:diag}{diagonaliseerbaar}:
\cref{exo:b2:reduction:9}) en gelijk aan $\nu' - \nu$, dat
nilpotent is: geldt $\nu^k = 0$ en $\nu'^{k'} = 0$, dan staat de
commutatie de binomiale ontwikkeling toe:
\[
(\nu' - \nu)^{k + k' - 1}
= \sum_{j=0}^{k+k'-1}\binom{k + k' - 1}{j}
\,\nu'^{\,j}\,(-\nu)^{k + k' - 1 - j} ,
\]
waarin elke term sterft: ofwel $j \geq k'$ (eerste factor nul),
ofwel $k + k' - 1 - j \geq k$ (tweede factor nul), en één van beide
geldt altijd. Een \omterm{def:b2:reduction:diag}{diagonaliseerbare} nilpotent is nul (haar \omterm{def:b2:reduction:eigen}{spectrum}
is $\{0\}$ en zij is in een zekere basis diagonaal): dus $d = d'$
en $\nu = \nu'$.
\end{proof}

\begin{example}[Machten en exponentiëlen]\label{ex:b2:reduction:powers}
$A = \begin{pmatrix} 3 & 1\\ -1 & 1\end{pmatrix}$: hier is $\chi_A
= X^2 - 4X + 4 = (X-2)^2$, één \omterm{def:b2:reduction:eigen}{eigenwaarde} $2$, met een \omterm{def:b2:reduction:eigen}{eigenruimte}
van dimensie $1$: niet \omterm{def:b2:reduction:diag}{diagonaliseerbaar}. Dunford: $D = 2I$, $N = A
- 2I = \begin{pmatrix} 1 & 1\\ -1 & -1\end{pmatrix}$, met $N^2 =
0$. Dan is
\[
A^k = (2I + N)^k = 2^k I + k\,2^{k-1} N ,
\qquad
\eu^{tA} = \eu^{2t}(I + tN),
\]
volgens het binomium voor commuterende termen, respectievelijk de
exponentiële reeks (\cref{ch:b2:diffeq}) gesplitst over
commuterende termen. Reductie maakt van matrixdynamica scalaire
dynamica.
\end{example}

\begin{remark}[Vooruitblik binnen dit volume]\label{rem:b2:reduction:perspectives}
De reductie is een knooppunt; let op deze vier spaken. In
\cref{ch:b2:nvs} maken aangepaste normen van ``alle \omterm{def:b2:reduction:eigen}{eigenwaarden}
hebben modulus $< 1$'' de uitspraak ``een zekere operatornorm is $<
1$'', zodat spectra de convergentie van machten en reeksen gaan
besturen. In \cref{ch:b2:diffeq} wordt het recept van
\cref{ex:b2:reduction:powers} de algemene oplossing van $X' = AX$:
Dunford splitst $\eu^{tA}$ in blokken van het type veelterm maal
exponentiële, en de stabiliteit is af te lezen aan de reële delen
van de \omterm{def:b2:reduction:eigen}{eigenwaarden}. In \cref{ch:b2:quadratic} dwingt een
inproduct af wat lineaire algebra alleen niet kan: symmetrische
matrices worden \emph{orthogonaal} \omterm{def:b2:reduction:diag}{diagonaliseerbaar}, zonder enig
nilpotent deel. En in \cref{ch:b2:genfun} keert de asymptotiek van
de dominante \omterm{def:b2:reduction:eigen}{eigenwaarde} uit de weekendopgave van dit hoofdstuk
analytisch terug, als de kleinste singulariteit van een genererende
functie --- twee talen voor één groeisnelheid.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b2:reduction:2}
Toon op twee manieren aan dat $C = \begin{pmatrix} 1 & 1\\ 0 &
1\end{pmatrix}$ niet \omterm{def:b2:reduction:diag}{diagonaliseerbaar} is: via de \omterm{def:b2:reduction:eigen}{eigenruimten} en

\begin{exercise}[$\star$]\label{exo:b2:reduction:3}
Zij $u$ zodanig dat $u^2 - 5u + 6\,\mathrm{id} = 0$. Bewijs dat $u$
\omterm{def:b2:reduction:diag}{diagonaliseerbaar} is, bepaal de mogelijke spectra, en bereken $u^k$

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:reduction:4}
Zij $u$ \omterm{def:b2:reduction:diag}{diagonaliseerbaar} en $F$ een stabiele deelruimte. Bewijs
dat $u|_F$ \omterm{def:b2:reduction:diag}{diagonaliseerbaar} is \emph{(beperk een annihilerende
veelterm met enkelvoudige nulpunten die uiteenvalt)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:reduction:5}
(Fibonacci) Zij $A = \begin{pmatrix} 1 & 1\\ 1 & 0\end{pmatrix}$.
Diagonaliseer $A$ over $\R$ en leid daaruit de formule van Binet af
voor de rij van Fibonacci ($F_0 = 0$, $F_1 = 1$, $F_{n+1} = F_n +
F_{n-1}$):
\[
F_n = \frac{\varphi^n - \psi^n}{\sqrt 5},
\qquad \varphi = \frac{1 + \sqrt5}{2},\ \psi = \frac{1 -
\sqrt5}{2}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:reduction:6}
Zij $u \in \mathcal{L}(E)$ met $u^2$ \omterm{def:b2:reduction:diag}{diagonaliseerbaar} en $u$
inverteerbaar ($K = \C$). Bewijs dat $u$ \omterm{def:b2:reduction:diag}{diagonaliseerbaar} is. Geef
een tegenvoorbeeld wanneer $u$ niet inverteerbaar is.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:reduction:7}
Bereken de \omterm{thm:b2:reduction:dunford}{Dunford-ontbinding}, $A^k$ en $\eu^{tA}$ voor
\[
A = \begin{pmatrix} 2 & 1 & 0\\ 0 & 2 & 1\\ 0 & 0 & 2
\end{pmatrix}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:reduction:8}
Zij $A \in \mathcal{M}_n(\C)$ met $A^k = I$ voor zekere $k \geq 1$.
Bewijs dat $A$ \omterm{def:b2:reduction:diag}{diagonaliseerbaar} is en dat haar \omterm{def:b2:reduction:eigen}{eigenwaarden}
$k$-de eenheidswortels zijn. Leid af dat een inverteerbare complexe
matrix van eindige orde die gelijkvormig is met een driehoeksmatrix
met diagonaal vol enen, de identiteit is.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:reduction:9}
(Simultane diagonalisatie) Zijn $u, v$ \omterm{def:b2:reduction:diag}{diagonaliseerbaar} en
commuterend. Bewijs dat zij \emph{simultaan} \omterm{def:b2:reduction:diag}{diagonaliseerbaar}
zijn: een zekere basis diagonaliseert beide. \emph{(Elke
\omterm{def:b2:reduction:eigen}{eigenruimte} van $u$ is $v$-stabiel; diagonaliseer daar de
beperkingen van $v$, met \cref{exo:b2:reduction:4}.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:reduction:10}
Zij $u \in \mathcal{L}(\C^n)$. Bewijs dat $u$ \omterm{def:b2:reduction:diag}{diagonaliseerbaar} is
dan en slechts dan als elke $u$-stabiele deelruimte een
$u$-stabiel complement bezit. \emph{(Voor $\Leftarrow$: pas de
eigenschap toe op $F = \sum_\lambda E_\lambda(u)$, de som van alle
\omterm{def:b2:reduction:eigen}{eigenruimten}; was een stabiel complement $G$ ongelijk aan nul, dan
zou het trigonaliseren van $u|_G$ een \omterm{def:b2:reduction:eigen}{eigenvector} van $u$ binnen
$G$ opleveren --- in strijd met $G \cap F = \{0\}$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:reduction:11}
(Spectraalstraal in het klein, een voorproefje van de analyse die
komt) Zij $A \in \mathcal{M}_2(\C)$ met beide \omterm{def:b2:reduction:eigen}{eigenwaarden} van
modulus $< 1$. Bewijs dat $A^k \to 0$ element voor element als $k
\to \infty$. \emph{(Trigonaliseer: $A = P(T)P^{-1}$ met $T$
bovendriehoekig; bereken $T^k$ expliciet --- onderscheid gelijke en
verschillende \omterm{def:b2:reduction:eigen}{eigenwaarden} --- en schat af.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:reduction:12}
Zij $u \in \mathcal{L}(\C^n)$ met $\operatorname{rk} u = 1$ ($n
\geq 2$). Toon aan dat $\chi_u = X^{n-1}(X - \operatorname{tr} u)$,
en dat $u$ \omterm{def:b2:reduction:diag}{diagonaliseerbaar} is dan en slechts dan als
$\operatorname{tr} u \neq 0$. \emph{(Herinner uit
\cref{exo:b2:linalg:5} dat $u^2 = (\operatorname{tr} u)\,u$.)}
\end{exercise}

\section{Probleem: Lineaire recurrenties en begeleidende matrices}

Een lineaire recurrentie $u_{n+k} = a_{k-1}u_{n+k-1} + \dots + a_0
u_n$ is een vermomde matrixmacht, en de reductie zet haar om in
gesloten formules, groeisnelheden en foutschattingen. Deze
weekendopgave bouwt het woordenboek op --- begeleidende matrices
aan de ene kant, de verschuivingsoperator op de ruimte van rijen
aan de andere ---, bewijst de \emph{hoofdstelling over lineaire
recurrenties} (de algemene oplossing is $\sum_i
Q_i(n)\lambda_i^n$ over de nulpunten van de \omterm{def:b2:reduction:charpoly}{karakteristieke
veelterm}) en besteedt de opbrengst aan de diofantische benadering
van $\sqrt2$, aan het tellen van wandelingen en woorden, en aan een
ring van gekoppelde rijen die alleen door simultane diagonalisatie
te ontwarren is.

\begin{problem}[{Weekendopgave --- de hoofdstelling over lineaire
recurrenties}]
\label{pb:b2:reduction:1}
Leg $k \geq 1$ vast, scalairen $a_0, \dots, a_{k-1} \in \C$ met
$a_0 \neq 0$, de monische veelterm $P = X^k - a_{k-1}X^{k-1} -
\dots - a_1 X - a_0$, en de recurrentie
\[
(\mathcal R)\colon\quad u_{n+k} = a_{k-1}u_{n+k-1} + \dots +
a_1 u_{n+1} + a_0 u_n \qquad (n \geq 0).
\]
De \emph{begeleidende matrix} van $P$ is
\[
C =
\begin{pmatrix}
0 & 1 & & \\
 & \ddots & \ddots & \\
 & & 0 & 1\\
a_0 & a_1 & \cdots & a_{k-1}
\end{pmatrix}
\in \mathcal{M}_k(\C).
\]

\medskip
\noindent\textbf{Deel I --- Het woordenboek van de begeleidende
matrix.}
\begin{enumerate}
  \item Toon aan dat een rij $(u_n)$ aan $(\mathcal R)$ voldoet dan
        en slechts dan als de vectoren $v_n = (u_n, u_{n+1}, \dots,
        u_{n+k-1})^{\mathsf T}$ voldoen aan $v_{n+1} = Cv_n$, en
        dus $v_n = C^n v_0$.
  \item Bewijs dat $\chi_C = P$ (ontwikkel $\det(XI - C)$ langs de
        eerste kolom en induceer naar $k$), en vervolgens dat ook
        $\mu_C = P$ \emph{(ga over op $C^{\mathsf T}$, waarvoor
        $e_1$ cyclisch is, en merk op dat een matrix en haar
        \omterm{def:b2:linalg:transpose}{getransponeerde} dezelfde \omterm{def:b2:reduction:polyu}{minimale veelterm} hebben)}.
  \item Toon aan dat voor elk nulpunt $\lambda$ van $P$ de vector
        $(1, \lambda, \dots, \lambda^{k-1})^{\mathsf T}$ de
        \omterm{def:b2:reduction:eigen}{eigenruimte} van $C$ bij $\lambda$ opspant; leid af dat
        \emph{elke} \omterm{def:b2:reduction:eigen}{eigenruimte} van $C$ dimensie $1$ heeft, en dat
        $C$ \omterm{def:b2:reduction:diag}{diagonaliseerbaar} is dan en slechts dan als $P$ $k$
        verschillende nulpunten heeft.
  \item Neem aan dat $P$ verschillende nulpunten $\lambda_1, \dots,
        \lambda_k$ heeft. Toon aan dat de meetkundige rijen
        $(\lambda_i^n)_n$ een basis vormen van de oplossingsruimte
        van $(\mathcal R)$, zodat elke oplossing $u_n = \sum_i
        c_i\lambda_i^n$ is voor eenduidig bepaalde constanten
        $c_i$.
  \item Los volledig op: $u_{n+2} = u_{n+1} + 6u_n$, $u_0 = 1$,
        $u_1 = 8$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De verschuivingsoperator en de
hoofdstelling.} Zij $\mathcal{S}$ de $\C$-vectorruimte van alle
complexe rijen en $S \in \mathcal{L}(\mathcal{S})$ de
verschuiving, $S\bigl((u_n)_n\bigr) = (u_{n+1})_n$.
\begin{enumerate}[resume]
  \item Toon aan dat de oplossingsverzameling van $(\mathcal R)$
        gelijk is aan $\ker P(S)$, en dat zij precies dimensie $k$
        heeft \emph{(stuur een oplossing naar haar
        beginwaarden)}.
  \item Leg uit waarom het kernontbindingslemma
        (\cref{thm:b2:reduction:kernels}) zonder enige wijziging
        op $S$ toepasbaar is op het oneindigdimensionale
        $\mathcal{S}$, en schrijf de resulterende ontbinding van
        $\ker P(S)$ op voor $P = \prod_{i=1}^{r}(X -
        \lambda_i)^{m_i}$ (verschillende $\lambda_i$, alle
        ongelijk aan nul omdat $a_0 \neq 0$).
  \item Toon voor $\lambda \neq 0$ en $m \geq 1$ aan dat
        \[
        \ker\,(S - \lambda\,\mathrm{id})^m
        = \bigl\{\,\bigl(Q(n)\,\lambda^n\bigr)_n : Q \in
        \C_{m-1}[X]\,\bigr\},
        \]
        van dimensie $m$. \emph{(Bereken $(S -
        \lambda)\bigl(Q(n)\lambda^n\bigr) =
        \lambda^{n+1}(\Delta Q)(n)$ met $\Delta Q = Q(X + 1) -
        Q(X)$, en gebruik dat $\Delta$ de graad verlaagt; begrens
        voor de dimensie door $m$ via de beginwaarden.)}
  \item (De hoofdstelling over lineaire recurrenties) Besluit: is
        $P = \prod_{i=1}^{r}(X - \lambda_i)^{m_i}$ met
        verschillende $\lambda_i$ die alle ongelijk aan nul zijn,
        dan zijn de oplossingen van $(\mathcal R)$ precies de
        rijen
        \[
        u_n = \sum_{i=1}^{r} Q_i(n)\,\lambda_i^n,
        \qquad Q_i \in \C_{m_i - 1}[X],
        \]
        met eenduidig bepaalde veeltermen $Q_i$.
  \item Los volledig op: $u_{n+2} = 4u_{n+1} - 4u_n$, $u_0 = 1$,
        $u_1 = 0$, en controleer het antwoord op $u_2$.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Dominante nulpunten en diofantische
opbrengsten.}
\begin{enumerate}[resume]
  \item Neem aan dat de nulpunten enkelvoudig zijn met
        $\abs{\lambda_1} > \abs{\lambda_i}$ voor $i \geq 2$, en
        dat $u_n = \sum_i c_i \lambda_i^n$ met $c_1 \neq 0$. Toon
        aan dat $u_n \sim c_1\lambda_1^n$ en $u_{n+1}/u_n \to
        \lambda_1$.
  \item (Pell) Definieer $a_{n+1} = a_n + 2b_n$, $b_{n+1} = a_n +
        b_n$, met $a_0 = b_0 = 1$. Toon aan dat $q(a, b) = a^2 -
        2b^2$ voldoet aan $q(a_{n+1}, b_{n+1}) = -q(a_n, b_n)$, en
        dus $a_n^2 - 2b_n^2 = (-1)^{n+1}$; breng dit in verband
        met de \omterm{def:b2:linalg:det}{determinant} van $M = \begin{psmallmatrix}1 & 2\\ 1
        & 1\end{psmallmatrix}$.
  \item Leid de foutschatting
        \[
        \Bigl|\frac{a_n}{b_n} - \sqrt2\Bigr|
        = \frac{1}{b_n\,(a_n + \sqrt2\,b_n)}
        \leq \frac{1}{2b_n^2}
        \]
        af, en toon aan dat zij meetkundig afneemt met factor $3 -
        2\sqrt2$ \emph{(bepaal de \omterm{def:b2:reduction:eigen}{eigenwaarden} van $M$ en de groei
        van $b_n$)}.
  \item (Algemene groei) Bewijs uit vraag 9: (a) voldoet elk
        nulpunt aan $\abs{\lambda_i} \leq \rho$, dan is
        $\abs{u_n} \leq C\,n^{m-1}\rho^n$ met $m = \max_i m_i$;
        (b) is er precies één nulpunt $\lambda_1$ van maximale
        modulus en is $Q_1 \neq 0$, dan geldt $u_{n+1}/u_n \to
        \lambda_1$ --- ga dit na op de oplossing van vraag 10.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Wandelingen en woorden tellen.} Voor
een eindige graaf met knopenverzameling $\{1, \dots, N\}$ heeft de
\emph{adjacentiematrix} $A$ als elementen $A_{ij} = 1$ wanneer
$ij$ een rib is, en anders $0$.
\begin{enumerate}[resume]
  \item Bewijs dat $(A^n)_{ij}$ het aantal wandelingen van lengte
        $n$ van $i$ naar $j$ is (rijen van $n$ ribben, waarbij elke
        stap langs een rib loopt).
  \item (De driehoek) Voor de volledige graaf op $3$ knopen is $A
        = J - I$: toon met het \omterm{def:b2:reduction:eigen}{spectrum} van $J$
        (\cref{ex:b2:linalg:onesmatrix}) aan dat
        \[
        (A^n)_{ii} = \frac{2^n + 2(-1)^n}{3},
        \qquad
        (A^n)_{ij} = \frac{2^n - (-1)^n}{3} \quad (i \neq j),
        \]
        en controleer beide voor $n = 2$ door de wandelingen op te
        sommen.
  \item (Woorden zonder $11$) Zij $w_n$ het aantal binaire woorden
        van lengte $n$ zonder twee opeenvolgende enen. Codeer de
        woorden naar hun laatste letter om een overdrachtsmatrix
        te krijgen, toon aan dat $w_{n+2} = w_{n+1} + w_n$, leid
        af dat $w_n = F_{n+2}$ (Fibonacci,
        \cref{exo:b2:reduction:5}), en geef de groeisnelheid
        $\lim w_{n+1}/w_n$.
  \item (Het pad) Toon voor de padgraaf $1 - 2 - 3$ aan dat de
        \omterm{def:b2:reduction:eigen}{eigenwaarden} van $A$ gelijk zijn aan $\sqrt2, 0, -\sqrt2$
        met \omterm{def:b2:reduction:eigen}{eigenvectoren} $(1, \pm\sqrt2, 1)$ en $(1, 0, -1)$, en
        leid af dat het aantal wandelingen van lengte $n$ van het
        ene uiteinde naar het andere gelijk is aan
        $\bigl((\sqrt2)^n + (-\sqrt2)^n\bigr)/4$: nul voor oneven
        $n$, en $2^{\,n/2 - 1}$ voor even $n$. Controleer dit voor
        $n = 4$.
  \item (Spoorformule) Toon aan dat het totale aantal gesloten
        wandelingen van lengte $n$ (over alle beginpunten) gelijk
        is aan $\operatorname{tr}(A^n) = \sum_i \lambda_i^n$, en
        ga dit na op de driehoek.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Een ring van rijen: simultane
diagonalisatie.} Leg $k \geq 3$ vast, zij $\omega =
\eu^{2\iu\pi/k}$, en zij $W \in \mathcal{M}_k(\C)$ de cyclische
verschuiving: $W e_i = e_{i+1}$ (indices modulo $k$, kolommen
genummerd $0, \dots, k-1$).
\begin{enumerate}[resume]
  \item Toon aan dat $W^{\mathsf T}$ de begeleidende matrix van
        $X^k - 1$ is, leid af dat $\chi_W = \mu_W = X^k - 1$, en
        dat $W$ \omterm{def:b2:reduction:diag}{diagonaliseerbaar} is met de $k$ enkelvoudige
        \omterm{def:b2:reduction:eigen}{eigenwaarden} $\omega^j$ en \omterm{def:b2:reduction:eigen}{eigenvectoren} $f_j = (1,
        \omega^{-j}, \omega^{-2j}, \dots, \omega^{-(k-1)j})^{\mathsf
        T}$.
  \item Een \emph{circulante} matrix is $C = c_0 I + c_1 W + \dots
        + c_{k-1}W^{k-1}$. Toon aan dat alle circulante matrices
        commuteren, dat de basis $(f_0, \dots, f_{k-1})$ ze
        \emph{alle} tegelijk diagonaliseert, en dat de
        \omterm{def:b2:reduction:eigen}{eigenwaarden} van $C$ gelijk zijn aan $\widehat
        c(\omega^j) = \sum_m c_m \omega^{jm}$, $j = 0, \dots,
        k-1$.
  \item Leid af dat $\det C = \prod_{j=0}^{k-1} \widehat
        c(\omega^j)$, en ga na dat $k = 3$ de factorisatie van
        \cref{exo:b2:linalg:8} teruggeeft.
  \item (Het halssnoergemiddelde) Zij $x^{(n+1)} = Mx^{(n)}$ met
        $M = \frac12(W + W^{-1})$: elk van $k$ getallen die in een
        ring liggen wordt vervangen door het gemiddelde van zijn
        twee buren. Toon aan dat de \omterm{def:b2:reduction:eigen}{eigenwaarden} van $M$ gelijk
        zijn aan $\cos(2\pi j/k)$, en dat de coëfficiënt van
        $x^{(0)}$ op $f_0$ het gemiddelde $\frac1k\sum_m
        x^{(0)}_m$ is \emph{(tel de coördinaten van de $f_j$ op)}.
  \item Besluit: voor oneven $k$ convergeert $x^{(n)}$ naar de
        constante vector met als waarde het gemiddelde van de
        beginwaarden; geef voor $k = 4$ de \omterm{def:b2:reduction:eigen}{eigenwaarde} die voor de
        niet-convergentie verantwoordelijk is, en de exacte
        hindernis (een coëfficiënt van het alternerende gemiddelde
        die nul moet zijn).
  \item (Synthese) In telkens één zin: hoe de begeleidende matrix
        de analyse van $(\mathcal R)$ omzet in reductie; waar het
        kernontbindingslemma geen eindige dimensie nodig had;
        waarom dominante \omterm{def:b2:reduction:eigen}{eigenwaarden} de groeisnelheden en de
        diofantische fout besturen; waarom machten van de
        adjacentiematrix wandelingen tellen; en wat commuterende
        matrices opleveren. Noem de twee toppen: de hoofdstelling
        over lineaire recurrenties, en --- voor de positieve
        matrices van Deel IV, in het volume van bachelorjaar 3 ---
        de stelling van Perron--Frobenius.
\end{enumerate}
\end{problem}
