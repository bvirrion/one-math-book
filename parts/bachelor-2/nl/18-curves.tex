\chapter{Krommen}\label{ch:b2:curves}

De meetkunde wordt nu \omterm{def:b2:diffcalc:differential}{differentiaal}. Een kromme is een punt dat
door de ruimte beweegt; de analyse geeft ons haar snelheid en
versnelling, en de meetkunde vraagt wat intrinsiek is ---
onafhankelijk van hoe snel wij de baan doorlopen. De antwoorden
zijn de \emph{\omterm{def:b2:curves:length}{booglengte}}, die de baan zelf meet, en de
\emph{\omterm{thm:b2:curves:frenet2d}{kromming}}, die meet hoe zij buigt. In dimensie $3$ meet een
tweede invariant, de \emph{\omterm{thm:b2:curves:frenet3d}{torsie}}, hoe de kromme uit haar vlak
draait. Het boekhoudkundige hulpmiddel voor dit alles is het
bewegende \emph{frenetstelsel}.

\section{Geparametriseerde bogen}

\begin{definition}[Geparametriseerde boog]\label{def:b2:curves:arc}
Een \emph{geparametriseerde boog}\index{geparametriseerde boog} van
klasse $\mathcal{C}^k$ ($k \geq 1$) is een afbeelding $\gamma
\colon I \to \R^n$ van klasse $\mathcal{C}^k$ op een interval $I$.
Een punt $\gamma(t)$ heet \emph{regulier} wanneer $\gamma'(t) \neq
0$, en de boog heet \emph{regulier} wanneer al haar punten dat
zijn. De rechte door $\gamma(t)$ met richting $\gamma'(t)$ is de
\emph{raaklijn}\index{raaklijn} in een regulier punt.
\end{definition}

\begin{definition}[Parameterverandering]\label{def:b2:curves:reparam}
Een \emph{parameterverandering}\index{parameterverandering} van
klasse $\mathcal{C}^k$ is een $\mathcal{C}^k$-diffeomorfisme
$\theta \colon J \to I$ tussen intervallen (met $\theta' \neq 0$
overal). De bogen $\gamma$ en $\gamma \circ \theta$ heten
\emph{equivalent}; een \emph{meetkundige boog}\index{meetkundige
boog} (of kromme) is een equivalentieklasse. Begrippen die
invariant zijn onder parameterverandering --- de baan, de \omterm{def:b2:curves:arc}{raaklijn},
de \omterm{def:b2:curves:length}{booglengte}, de \omterm{thm:b2:curves:frenet2d}{kromming} --- heten \emph{meetkundig}.
\end{definition}

\begin{remark}
De baan alleen legt de \omterm{def:b2:curves:reparam}{meetkundige boog} niet vast: de
parametriseringen $t \mapsto (\cos t, \sin t)$ op $[0, 2\pi]$ en op
$[0, 4\pi]$ hebben hetzelfde beeld, maar doorlopen de cirkel
eenmaal en tweemaal. Een \omterm{def:b2:curves:reparam}{meetkundige boog} onthoudt de
multipliciteit en de oriëntatie van het doorlopen, niet de
snelheid.
\end{remark}

\begin{example}[Snelheid verandert niets
meetkundigs]\label{ex:b2:curves:speedinvariance}
Parametriseer de eenheidscirkel met
\[
\gamma(t) = (\cos t^2,\ \sin t^2),
\qquad t \in \intcc0{\sqrt{2\pi}} .
\]
De snelheid $\norm{\gamma'(t)} = 2t$ groeit lineair, en toch is
\[
L = \int_0^{\sqrt{2\pi}}2t\,\dd t = 2\pi ,
\]
dezelfde \omterm{def:b2:curves:length}{lengte} als bij constante snelheid --- zoals
\cref{thm:b2:curves:lengthinv} belooft, via de
\omterm{def:b2:curves:reparam}{parameterverandering} $t \mapsto t^2$. De \omterm{def:b2:curves:arc}{raaklijn}, de \omterm{thm:b2:curves:frenet2d}{kromming}
berekend met \cref{prop:b2:curves:kappaformula} en elke andere
meetkundige grootheid stemmen eveneens overeen; alleen $t = 0$
verdient een blik, waar $\gamma'(0) = 0$ \emph{deze
parametrisering} irregulier maakt hoewel de baan een volmaakte
cirkel is. Meetkundige uitspraken verdragen slechte
parametriseringen slecht: parametriseer eerst opnieuw, besluit
daarna.
\end{example}

\begin{example}\label{ex:b2:curves:cusp}
De boog $\gamma(t) = (t^2, t^3)$ is $\mathcal{C}^\infty$ maar niet
regulier: $\gamma'(0) = (0, 0)$. Haar baan, de semikubische
parabool $y^2 = x^3$, heeft een \emph{keerpunt} in de oorsprong: de
gladheid van de parametrisering verhindert geen meetkundige
singulariteit waar de snelheid verdwijnt. Daarom is de
regulariteitshypothese $\gamma' \neq 0$ geen opsmuk.
\end{example}

\section{Booglengte}

\begin{definition}[Booglengte]\label{def:b2:curves:length}
Zij $\gamma \colon [a, b] \to \R^n$ een $\mathcal{C}^1$-boog. Haar
\emph{lengte}\index{booglengte} is
\[
L(\gamma) = \int_a^b \norm{\gamma'(t)}\, \dd t ,
\]
waarbij $\norm{\cdot}$ de euclidische \omterm{def:b2:nvs:norm}{norm} is. De
\emph{booglengtefunctie}\index{booglengtefunctie} met basispunt
$t_0$ is $s(t) = \int_{t_0}^t \norm{\gamma'(u)}\,\dd u$.
\end{definition}

\begin{theorem}[De lengte is meetkundig; karakterisering met
veelhoeken]\label{thm:b2:curves:lengthinv}
\leavevmode
\begin{enumerate}
\item Is $\theta \colon [c, d] \to [a, b]$ een
$\mathcal{C}^1$-parameterverandering, dan is $L(\gamma \circ
\theta) = L(\gamma)$.
\item $L(\gamma)$ is het supremum van de \omterm{def:b2:curves:length}{lengten} van de
ingeschreven veelhoeken:
\[
L(\gamma)
= \sup\Bigl\{\, \sum_{i=1}^{m} \norm{\gamma(t_i) -
\gamma(t_{i-1})} \;:\; a = t_0 < t_1 < \dots < t_m = b \,\Bigr\}.
\]
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} Met de substitutie $t = \theta(u)$ (volume van
bachelorjaar 1, geldig omdat $\theta$ monotoon en $\mathcal{C}^1$
is) is
\[
\int_c^d \norm{(\gamma\circ\theta)'(u)}\,\dd u
= \int_c^d \norm{\gamma'(\theta(u))}\,\abs{\theta'(u)}\,\dd u
= \int_a^b \norm{\gamma'(t)}\,\dd t ,
\]
waarbij we $(\gamma\circ\theta)' = \theta'\cdot
(\gamma'\circ\theta)$ gebruikten en, als $\theta$ daalt, het teken
van $\theta'$ door het omkeren van de grenzen wordt opgeslorpt.

\emph{2.} Voor elke onderverdeling is $\gamma(t_i) - \gamma(t_{i-1})
= \int_{t_{i-1}}^{t_i}\gamma'(t)\,\dd t$, dus volgens de
driehoeksongelijkheid voor integralen $\norm{\gamma(t_i) -
\gamma(t_{i-1})} \leq \int_{t_{i-1}}^{t_i}\norm{\gamma'}$: elke
veelhoek is korter dan $L(\gamma)$, dus $\sup \leq L(\gamma)$.

Voor de omgekeerde ongelijkheid, zij $\varepsilon > 0$. Omdat
$\gamma'$ \omterm{def:b2:metric:continuity}{continu} is op het \omterm{def:b2:metric:compact}{compacte} $[a,b]$, is zij uniform
\omterm{def:b2:metric:continuity}{continu}: er is een $\delta > 0$ met $\norm{\gamma'(t) -
\gamma'(u)} \leq \varepsilon$ zodra $\abs{t - u} \leq \delta$. Neem
een onderverdeling met stap $\leq \delta$. Op elk stuk geldt, voor
$t \in [t_{i-1}, t_i]$,
\[
\gamma(t_i) - \gamma(t_{i-1})
= \int_{t_{i-1}}^{t_i}\gamma'(t)\,\dd t
= (t_i - t_{i-1})\,\gamma'(t_{i-1}) + R_i,
\qquad
\norm{R_i} \leq \varepsilon\,(t_i - t_{i-1}),
\]
want $R_i = \int_{t_{i-1}}^{t_i}(\gamma'(t) -
\gamma'(t_{i-1}))\,\dd t$. Bijgevolg is
\[
\norm{\gamma(t_i) - \gamma(t_{i-1})}
\geq (t_i - t_{i-1})\norm{\gamma'(t_{i-1})}
- \varepsilon (t_i - t_{i-1}) .
\]
Sommeren, en $\sum (t_i - t_{i-1})\norm{\gamma'(t_{i-1})}$
vergelijken met $\int_a^b\norm{\gamma'}$ (een Riemannsom van de
\omterm{def:b2:metric:continuity}{continue} functie $\norm{\gamma'}$, die voor voldoend kleine
$\delta$ opnieuw wegens de uniforme \omterm{def:b2:metric:continuity}{continuïteit} binnen
$\varepsilon(b - a)$ van de integraal ligt), geeft een veelhoek met
\omterm{def:b2:curves:length}{lengte} $\geq L(\gamma) - 2\varepsilon(b - a)$. Met $\varepsilon \to
0$ volgt de bewering.
\end{proof}

\begin{example}[Archimedes en de ingeschreven
veelhoeken]\label{ex:b2:curves:archimedes}
Voor de eenheidscirkel heeft de ingeschreven regelmatige
$n$-hoek \omterm{def:b2:curves:length}{lengte} $L_n = 2n\sin\frac\pi n$, en de ontwikkeling $\sin
x = x - \frac{x^3}6 + O(x^5)$ geeft
\[
L_n = 2\pi - \frac{\pi^3}{3n^2} + O\Bigl(\frac1{n^4}\Bigr) :
\]
de veelhoeklengten van \cref{thm:b2:curves:lengthinv} convergeren
\emph{kwadratisch}. Numeriek: $L_6 = 6$ (de zeshoek, die de ruwe
schatting $\pi > 3$ geeft), terwijl $L_{96} = 192\sin\frac{\pi}{96}
\approx 6.28206$ tegenover $2\pi \approx 6.28319$ --- de fout
$0.00113$ stemt overeen met de voorspelde $\pi^3/(3\cdot96^2)
\approx 0.00112$. Daarom kon Archimedes, door de zeshoek vijf maal
te verdubbelen tot $96$ zijden, $\pi$ met de hand tot op drie
cijfers insluiten: elke verdubbeling deelt de fout door vier. Het
supremum in de karakterisering met veelhoeken wordt niet alleen in
de limiet bereikt; het wordt \emph{snel} bereikt, omdat een gladde
kromme zich pas op tweede orde van haar koorden losmaakt.
\end{example}

\begin{theorem}[Parametrisering naar booglengte]\label{thm:b2:curves:arclength}
Zij $\gamma \colon I \to \R^n$ een reguliere $\mathcal{C}^k$-boog
($k \geq 1$). De \omterm{def:b2:curves:length}{booglengtefunctie} $s$ is een
$\mathcal{C}^k$-diffeomorfisme van $I$ op een interval $J$, en
$\tilde\gamma = \gamma \circ s^{-1}$ voldoet overal aan
$\norm{\tilde\gamma'} = 1$. Op een translatie van de parameter en
de oriëntatie na is deze parametrisering \emph{naar \omterm{def:b2:curves:length}{booglengte}} (of
\emph{met eenheidssnelheid}) uniek.
\end{theorem}

\begin{proof}
$s'(t) = \norm{\gamma'(t)} > 0$ en $s'$ is $\mathcal{C}^{k-1}$ (de
samenstelling van de $\mathcal{C}^{k-1}$-afbeelding $\gamma'$ met
de \omterm{def:b2:nvs:norm}{norm}, glad buiten $0$), dus is $s$ van klasse $\mathcal{C}^k$,
strikt stijgend, een bijectie op $J = s(I)$, en haar inverse is
$\mathcal{C}^k$ volgens de stelling van de inverse functie in
dimensie $1$ (volume van bachelorjaar 1). Dan is
\[
\tilde\gamma'(\sigma)
= \frac{\gamma'(t)}{s'(t)}
= \frac{\gamma'(t)}{\norm{\gamma'(t)}},
\qquad t = s^{-1}(\sigma),
\]
een eenheidsvector. Is $\hat\gamma = \gamma\circ\theta$ een andere
parametrisering met eenheidssnelheid, dan is $\abs{\theta'} = 1$,
dus $\theta' = \pm 1$ constant (\omterm{def:b2:metric:continuity}{continuïteit}), dat wil zeggen
$\theta(u) = \pm u + c$.
\end{proof}

\begin{remark}
De \omterm{def:b2:curves:length}{booglengte} is de parameter die de meetkunde van de dynamica
scheidt. Een baan kan met elk snelheidsprofiel worden doorlopen ---
de fysica van de beweging --- maar elke vraag die invariant is
onder parametrisering (vorm, buiging, osculatie) heeft één
canonieke klok: de afgelegde afstand. Daarom worden alle
krommingsformules hieronder \emph{gedefinieerd} bij
eenheidssnelheid en daarna met de kettingregel \emph{vertaald} naar
willekeurige parametriseringen: de vertaalfactoren zijn machten van
$v = s'$, en die correct bijhouden is de hele inhoud van
\cref{prop:b2:curves:kappaformula}.
\end{remark}

\begin{example}[Cirkel en schroeflijn]\label{ex:b2:curves:helix}
Voor de cirkel $\gamma(t) = (R\cos t, R\sin t)$ is $\norm{\gamma'}
= R$, dus $s = Rt$ en is de \omterm{def:b2:curves:length}{lengte} van een volle omwenteling $2\pi
R$. Voor de schroeflijn $\gamma(t) = (a\cos t,\ a\sin t,\ bt)$ met
$a > 0$ is $\norm{\gamma'(t)} = \sqrt{a^2 + b^2}$ constant: de
schroeflijn wordt met constante snelheid doorlopen, en $s =
t\sqrt{a^2 + b^2}$.
\end{example}

\begin{example}[Booglengte in
poolcoördinaten]\label{ex:b2:curves:polarlength}
Een poolkromme $r = r(\theta)$ is de boog $\gamma(\theta) =
(r\cos\theta,\ r\sin\theta)$, met
\[
\gamma'(\theta) = (r'\cos\theta - r\sin\theta,\
r'\sin\theta + r\cos\theta),
\qquad
\norm{\gamma'(\theta)}^2 = r'^2 + r^2
\]
(de kruistermen heffen elkaar op): het lengte-element in
poolcoördinaten is $\sqrt{r^2 + r'^2}\,\dd\theta$. Voor de
cardioïde $r = 1 + \cos\theta$:
\[
r^2 + r'^2 = (1 + \cos\theta)^2 + \sin^2\theta = 2 +
2\cos\theta = 4\cos^2\tfrac\theta2,
\]
en op $\intcc{-\pi}{\pi}$ is de cosinus van de halve hoek
niet-negatief, dus
\[
L = \int_{-\pi}^{\pi}2\cos\tfrac\theta2\,\dd\theta
= \Bigl[4\sin\tfrac\theta2\Bigr]_{-\pi}^{\pi}
= 4 - (-4) = 8 :
\]
net als bij de cycloïdeboog van \cref{exo:b2:curves:1} heeft een
kromme die uit cirkels is opgebouwd een rationale \omterm{def:b2:curves:length}{lengte}, zonder
enige $\pi$. De ontbinding met de halve hoek is de standaardtruc
voor de \omterm{def:b2:curves:length}{lengten} van krommen die uit cirkels ontstaan; faalt zij (de
ellips), dan is de \omterm{def:b2:curves:length}{lengte} een werkelijk nieuwe functie --- een
elliptische integraal, buiten de elementaire gesloten vormen.
\end{example}

\section{Kromming in het vlak}

Overal in deze paragraaf zijn de bogen $\mathcal{C}^2$ en regulier
in het georiënteerde euclidische vlak. Wij parametriseren naar
\omterm{def:b2:curves:length}{booglengte} en schrijven $T(s) = \tilde\gamma'(s)$ voor de
eenheidsraakvector, en $N(s)$ voor de eenheidsvector die er direct
loodrecht op staat (de rotatie van $T$ over $+\pi/2$).

\begin{theorem}[Formules van Frenet in het vlak]\label{thm:b2:curves:frenet2d}
Zij $\tilde\gamma$ een $\mathcal{C}^2$-boog met eenheidssnelheid in
het georiënteerde vlak. Er bestaat een \omterm{def:b2:metric:continuity}{continue} functie $\kappa$,
de (algebraïsche) \emph{kromming}\index{kromming}, zodanig dat
\[
T'(s) = \kappa(s)\, N(s),
\qquad
N'(s) = -\kappa(s)\, T(s).
\]
\end{theorem}

\begin{proof}
Omdat $\norm{T(s)}^2 = 1$ voor alle $s$, geeft het differentiëren
van het inproduct $2\langle T'(s), T(s)\rangle = 0$: $T'(s)$ staat
loodrecht op $T(s)$ en is dus collineair met $N(s)$ (dimensie $2$);
schrijf $T'(s) = \kappa(s) N(s)$ met $\kappa(s) = \langle T'(s),
N(s)\rangle$, \omterm{def:b2:metric:continuity}{continu}. Evenzo is $N' \perp N$, dus $N' = \lambda
T$; en het differentiëren van $\langle T, N\rangle = 0$ geeft
$\langle T', N\rangle + \langle T, N'\rangle = \kappa + \lambda =
0$.
\end{proof}

\begin{definition}\label{def:b2:curves:curvature}
Is $\kappa(s) \neq 0$, dan is de
\emph{kromtestraal}\index{kromtestraal} $R(s) = 1/\abs{\kappa(s)}$
en het \emph{krommingsmiddelpunt}\index{krommingsmiddelpunt}
$\tilde\gamma(s) + \frac{1}{\kappa(s)} N(s)$; de cirkel met dat
middelpunt en die straal $R(s)$ is de \emph{osculerende
cirkel}\index{osculerende cirkel}, de beste cirkelvormige
benadering van de kromme in $\tilde\gamma(s)$.
\end{definition}

\begin{example}[De osculerende cirkel van de
exponentiële functie]\label{ex:b2:curves:oscexp}
Voor $y = \eu^x$ in het punt $(0, 1)$: $f'(0) = f''(0) = 1$, dus
volgens de grafiekformule hieronder
\[
\kappa(0) = \frac{1}{(1 + 1)^{3/2}} = \frac1{2\sqrt2},
\qquad R = 2\sqrt2 .
\]
De eenheidsraakvector is $T = \frac{(1, 1)}{\sqrt2}$, de directe
normaal $N = \frac{(-1, 1)}{\sqrt2}$, en het
\omterm{def:b2:curves:curvature}{krommingsmiddelpunt} is
\[
(0, 1) + 2\sqrt2\cdot\frac{(-1, 1)}{\sqrt2} = (-2,\ 3) :
\]
de \omterm{def:b2:curves:curvature}{osculerende cirkel} heeft vergelijking $(x + 2)^2 + (y - 3)^2 =
8$. Als controle van de bewering ``beste cirkelvormige
benadering'': de cirkelvergelijking naar $y$ oplossen nabij $(0,1)$
en ontwikkelen geeft $y = 1 + x + \frac{x^2}2 + O(x^3)$ --- precies
de tweede-ordeontwikkeling van Taylor van $\eu^x$. De \omterm{def:b2:curves:curvature}{osculerende
cirkel} past bij de waarde, de helling \emph{en} de tweede
afgeleide; een gewone raakcirkel past alleen bij de eerste twee.
\end{example}

\begin{example}[De evolute van een cirkel is haar
middelpunt]\label{ex:b2:curves:circleevolute}
Voor de cirkel met straal $R$, tegen de wijzers van de klok in
doorlopen, is $\kappa = 1/R$ en wijst $N$ naar het middelpunt, dus
is het \omterm{def:b2:curves:curvature}{krommingsmiddelpunt} $\tilde\gamma + \frac1\kappa N$ voor
elke $s$ het middelpunt van de cirkel: de \omterm{def:b2:curves:curvature}{osculerende cirkel} van
een cirkel is de cirkel zelf, en de meetkundige plaats van de
\omterm{def:b2:curves:curvature}{krommingsmiddelpunten} klapt ineen tot één punt. Dit ontaarde geval
ijkt \cref{exo:b2:curves:6}: daar is de snelheid van de evolute
$-\frac{\kappa'}{\kappa^2}N$, die identiek verdwijnt precies
wanneer $\kappa$ constant is.
\end{example}

\begin{proposition}[Kromming in een willekeurige
parametrisering]\label{prop:b2:curves:kappaformula}
Voor een reguliere $\mathcal{C}^2$-vlakke boog $\gamma(t) = (x(t),
y(t))$ is
\[
\kappa(t)
= \frac{x'(t)\,y''(t) - y'(t)\,x''(t)}
       {\bigl(x'(t)^2 + y'(t)^2\bigr)^{3/2}} ,
\]
in het bijzonder $\kappa = \dfrac{y''}{(1 + y'^2)^{3/2}}$ voor een
grafiek $y = f(x)$.
\end{proposition}

\begin{proof}
Schrijf $v(t) = \norm{\gamma'(t)} = s'(t)$, zodat $\gamma' = vT$
(de gegevens bij eenheidssnelheid samengesteld met $s$).
Differentiëren geeft
\[
\gamma'' = v'T + v\,T'\cdot s' = v'T + v^2\kappa N .
\]
Neem nu de \omterm{def:b2:linalg:det}{determinant} (in de canonieke georiënteerde basis) van
$(\gamma', \gamma'')$: omdat $\det(T, T) = 0$ en $\det(T, N) = 1$,
is
\[
\det(\gamma', \gamma'') = \det(vT,\ v'T + v^2\kappa N)
= v^3\kappa .
\]
Het linkerlid is $x'y'' - y'x''$, en $v^3 = (x'^2 + y'^2)^{3/2}$.
Het geval van de grafiek is de parametrisering $t \mapsto (t,
f(t))$.
\end{proof}

\begin{example}[Cirkel, rechte, parabool]\label{ex:b2:curves:kappaexamples}
Een rechte heeft $\kappa = 0$ (en omgekeerd: $T' = 0$ betekent dat
$T$ constant is, dus $\tilde\gamma(s) = \tilde\gamma(0) + sT$, een
rechte). De cirkel met straal $R$, tegen de wijzers van de klok in
doorlopen, heeft $\kappa = 1/R$: met $\gamma(t) = (R\cos t, R\sin
t)$ geeft de formule $\kappa = R^2/R^3$. Voor de parabool $y =
x^2/2$: $\kappa(x) = 1/(1 + x^2)^{3/2}$, maximaal in de top --- de
parabool buigt het scherpst waar zij omkeert.
\end{example}

\begin{remark}[Klassieke valkuilen rond de kromming]
(i) De algebraïsche \omterm{thm:b2:curves:frenet2d}{kromming} van een vlakke boog verandert van
teken wanneer de oriëntatie van de boog of van het vlak wordt
omgekeerd: alleen $\abs\kappa$ en $R = 1/\abs\kappa$ zijn zuiver
meetkundig. Een cirkel die met de wijzers van de klok mee wordt
doorlopen, heeft $\kappa = -1/R$. (ii) De grafiekformule $\kappa =
f''/(1 + f'^2)^{3/2}$ kiest stilzwijgend de parametrisering naar
$x$; haar toepassen op een kromme die nabij het punt geen grafiek
is (verticale \omterm{def:b2:curves:arc}{raaklijn}) is de klassieke blunder. (iii) In een punt
waar $\gamma' = 0$ is niets gedefinieerd --- noch $T$ noch $\kappa$
--- en kan de baan werkelijk breken
(\cref{ex:b2:curves:cusp}); ga altijd de regulariteit na voordat je
de eenheidsraakvector differentieert. (iv) In de ruimte is $\kappa
= \norm{T'} \geq 0$ per afspraak: er valt geen teken verkeerd te
doen, maar ook geen teken uit te buiten --- informatie van het type
buigpunt verhuist naar de \omterm{thm:b2:curves:frenet3d}{torsie}. (v) Ten slotte is $\kappa$ een
afgeleide \emph{naar de \omterm{def:b2:curves:length}{booglengte}}: voor een parametrisering
zonder eenheidssnelheid is het vergeten van de factor $v^3$ in
\cref{prop:b2:curves:kappaformula} de meest voorkomende fout in de
praktijk.
\end{remark}

\begin{theorem}[De kromming legt de kromme vast]\label{thm:b2:curves:fundamental}
Zij $\kappa \colon J \to \R$ \omterm{def:b2:metric:continuity}{continu}. Er bestaat een
$\mathcal{C}^2$-boog met eenheidssnelheid in het vlak met \omterm{thm:b2:curves:frenet2d}{kromming}
$\kappa$, en hij is uniek op een directe isometrie na (een rotatie
gevolgd door een translatie).
\end{theorem}

\begin{proof}
\emph{Bestaan.} Houd $s_0 \in J$ vast en zet $\varphi(s) =
\int_{s_0}^s \kappa(u)\,\dd u$, en dan
\[
\tilde\gamma(s) = \Bigl(\int_{s_0}^s \cos\varphi(u)\,\dd u,\
\int_{s_0}^s \sin\varphi(u)\,\dd u\Bigr).
\]
Dan is $T(s) = (\cos\varphi(s), \sin\varphi(s))$ een
eenheidsvector, $N(s) = (-\sin\varphi, \cos\varphi)$, en
\[
T'(s) = \varphi'(s)\,(-\sin\varphi, \cos\varphi)
= \kappa(s)\,N(s) :
\]
de boog heeft eenheidssnelheid en \omterm{thm:b2:curves:frenet2d}{kromming} $\kappa$.

\emph{Eenduidigheid.} Zijn $\gamma_1, \gamma_2$ bogen met
eenheidssnelheid en dezelfde \omterm{thm:b2:curves:frenet2d}{kromming}. Elke eenheidsraakvector laat
zich tot een hoekfunctie liften, met een expliciete constructie:
beschouw $T_j$ als het complexe getal $z_j = a_j + \iu b_j$ van
modulus $1$, kies $\varphi_j(0)$ met $z_j(0) =
\eu^{\iu\varphi_j(0)}$, en zet
\[
\varphi_j(s) = \varphi_j(0) + \int_0^s\det\bigl(T_j,
T_j'\bigr)(u)\,\dd u .
\]
Uit $\abs{z_j} = 1$ volgt $\operatorname{Re}(\conj{z_j}\,z_j') =
0$, dus $\conj{z_j}\,z_j' = \iu\det(T_j, T_j') = \iu\,\varphi_j'$,
dat wil zeggen $z_j' = \iu\varphi_j'z_j$; dan is
\[
\bigl(z_j\,\eu^{-\iu\varphi_j}\bigr)'
= \eu^{-\iu\varphi_j}\bigl(z_j' - \iu\varphi_j'z_j\bigr) = 0,
\]
zodat overal $z_j = \eu^{\iu\varphi_j}$: $T_j = (\cos\varphi_j,
\sin\varphi_j)$ met $\varphi_j$ van klasse $\mathcal C^1$.
Bovendien is $\det(T_j, T_j') = \det(T_j, \kappa N_j) = \kappa$,
dus $\varphi_j' = \kappa$. Bijgevolg is $\varphi_2 = \varphi_1 + c$
voor een constante $c$: $T_2$ is $T_1$ gedraaid over de vaste hoek
$c$, dus integreren geeft $\gamma_2 = \rho(\gamma_1) + w$, waarbij
$\rho$ de rotatie over de hoek $c$ is en $w$ een constante vector.
\end{proof}

\begin{remark}
Dit is het eendimensionale prototype van een
\emph{hoofdstelling van de meetkunde}: een \omterm{def:b2:metric:complete}{volledig} stelsel lokale
invarianten (hier één functie) klasseert het object op een starre
beweging na. De driedimensionale versie hieronder heeft twee
invarianten nodig.
\end{remark}

\begin{example}[Constante kromming betekent
cirkel]\label{ex:b2:curves:constantkappa}
Neem $\kappa \equiv \kappa_0 > 0$ in de bestaansformule: $\varphi(s)
= \kappa_0 s$ en
\[
\tilde\gamma(s) = \Bigl(\frac{\sin\kappa_0s}{\kappa_0},\
\frac{1 - \cos\kappa_0s}{\kappa_0}\Bigr) :
\]
de cirkel met straal $1/\kappa_0$ en middelpunt $(0, 1/\kappa_0)$,
met eenheidssnelheid doorlopen. Volgens de eenduidigheidshelft van
de stelling is \emph{elke} boog met eenheidssnelheid en constante
\omterm{thm:b2:curves:frenet2d}{kromming} $\kappa_0$ een stuk van een cirkel met straal
$1/\kappa_0$ (of een rechte als $\kappa_0 = 0$) --- de omkering van
de berekening in \cref{ex:b2:curves:kappaexamples}, en het vlakke
geval van \cref{exo:b2:curves:9}.
\end{example}

\begin{example}[Een kromme uit haar kromming
reconstrueren]\label{ex:b2:curves:reconstruction}
Welke kromme met eenheidssnelheid heeft \omterm{def:b2:curves:curvature}{kromtestraal} $R(s) = 1 +
s^2$? Het bestaansbewijs volgen met $\kappa(s) = \frac1{1+s^2}$ en
$s_0 = 0$: $\varphi(s) = \arctan s$, dus
\[
T(s) = (\cos\arctan s,\ \sin\arctan s)
= \Bigl(\frac{1}{\sqrt{1+s^2}},\
\frac{s}{\sqrt{1+s^2}}\Bigr),
\]
en integreren geeft
\[
\tilde\gamma(s) = \Bigl(\ln\bigl(s + \sqrt{1 + s^2}\bigr),\
\sqrt{1 + s^2} - 1\Bigr).
\]
Zetten we $x = \ln(s + \sqrt{1+s^2})$, dat wil zeggen $s = \sinh
x$, dan is de tweede coördinaat $\cosh x - 1$: de kromme is de
\emph{kettinglijn} $y = \cosh x - 1$. Hiermee is de cirkel rond met
\cref{exo:b2:curves:3}, waar wij $R = \cosh^2 x = 1 + \sinh^2 x = 1
+ s^2$ rechtstreeks berekenden: de hoofdstelling waarborgt dat de
kettinglijn de \emph{enige} kromme met dit krommingsprofiel is, op
een directe isometrie na.
\end{example}

\begin{remark}[Waar de kromming hierna wordt gebruikt]
De ontbinding $\gamma'' = v'T + v^2\kappa N$ die in het bewijs van
\cref{prop:b2:curves:kappaformula} werd verkregen, is de kinematica
van elke gekromde beweging: tangentiële tegenover centripetale
versnelling. De \omterm{thm:b2:curves:frenet2d}{kromming} keert terug bij oppervlakken
(\cref{ch:b2:surfaces}) via de \omterm{thm:b2:curves:frenet2d}{kromming} van krommen die erop
getekend zijn, en de rekenkunde van de \omterm{pb:b2:curves:1}{omhullenden} uit de
weekendopgave van dit hoofdstuk --- evoluten, caustieken --- is de
meetkundige optica van golffronten. Het volume van bachelorjaar 3
neemt het intrinsieke standpunt weer op voor deelvariëteiten van
$\R^n$.
\end{remark}

\section{Het frenetstelsel in de ruimte}

Zij nu $\tilde\gamma \colon J \to \R^3$ een $\mathcal{C}^3$-boog
met eenheidssnelheid die \emph{biregulier} is: $T'(s) \neq 0$ voor
alle $s$. Dan definieert $\kappa(s) = \norm{T'(s)} > 0$ de
\emph{\omterm{thm:b2:curves:frenet2d}{kromming}} (geen teken in de ruimte: er is geen bevoorrechte
oriëntatie van de normaal), en zetten wij:
\[
N(s) = \frac{T'(s)}{\kappa(s)}
\quad\text{(hoofdnormaal)},
\qquad
B(s) = T(s) \wedge N(s)
\quad\text{(binormaal)} ,
\]
zodat $(T, N, B)$ een direct orthonormaal stelsel is, het
\emph{frenetstelsel}. Het vlak door $\tilde\gamma(s)$ opgespannen
door $T, N$ is het \emph{osculerende vlak}.

\begin{theorem}[Formules van Frenet in de ruimte]\label{thm:b2:curves:frenet3d}
Er bestaat een \omterm{def:b2:metric:continuity}{continue} functie $\tau$, de
\emph{torsie}\index{torsie}, met
\[
T' = \kappa N, \qquad
N' = -\kappa T + \tau B, \qquad
B' = -\tau N .
\]
\end{theorem}

\begin{proof}
De eerste formule is de definitie van $N$. Elk van de vectoren $T,
N, B$ heeft constante \omterm{def:b2:nvs:norm}{norm} $1$ en zij staan paarsgewijs loodrecht
op elkaar; het differentiëren van de zes betrekkingen $\langle X,
Y\rangle = \delta_{XY}$ toont dat de matrix van $(T', N', B')$ in
de basis $(T, N, B)$ antisymmetrisch is: immers $\langle X',
Y\rangle + \langle X, Y'\rangle = 0$ en $\langle X', X\rangle = 0$.
Haar element op plaats $(N, T)$ is $\langle N', T\rangle = -\langle
N, T'\rangle = -\kappa$, en het element $\langle T', B\rangle =
\kappa\langle N, B\rangle = 0$ in de $(T, B)$-kolom. Het overige
vrije element $\tau = \langle N', B\rangle$ een naam geven levert
precies de drie getoonde formules: de antisymmetrie vult $\langle
B', N\rangle = -\tau$ en $\langle B', T\rangle = 0$ aan. De
\omterm{def:b2:metric:continuity}{continuïteit} van $\tau = \langle N', B\rangle$ is duidelijk, omdat
$N'$ en $B$ \omterm{def:b2:metric:continuity}{continu} zijn ($\tilde\gamma$ is $\mathcal{C}^3$, dus $N
= T'/\kappa$ is $\mathcal{C}^1$).
\end{proof}

\begin{example}[De vector van Darboux]\label{ex:b2:curves:darboux}
De drie formules van Frenet klappen samen tot één. Zet $\omega(s) =
\tau\,T + \kappa\,B$ (de \emph{vector van Darboux}). Met $B \wedge
T = N$, $T \wedge N = B$ en $N \wedge B = T$:
\[
\omega \wedge T = \kappa\,N = T', \qquad
\omega \wedge N = \tau\,B - \kappa\,T = N', \qquad
\omega \wedge B = -\tau\,N = B' :
\]
elke stelselvector evolueert als $X' = \omega \wedge X$, de
kinematische handtekening van een \emph{ogenblikkelijke rotatie}
met hoeksnelheidsvector $\omega$. Het stelsel draait met snelheid
$\norm\omega = \sqrt{\kappa^2 + \tau^2}$ om de bewegende as
$\omega$; de \omterm{thm:b2:curves:frenet2d}{kromming} is de component van het draaien om de
binormaal, de \omterm{thm:b2:curves:frenet3d}{torsie} de component om de raakvector. Voor de
schroeflijn is $\omega$ een constante vector langs de as van de
cilinder --- en precies daarom precedeert het stelsel van de
schroeflijn gelijkmatig. De antisymmetrie van de frenetmatrix, in
\cref{exo:b2:curves:7} uitgebuit, is de matrixvorm van dit ene
meetkundige feit.
\end{example}

\begin{proposition}[De torsie meet de vlakheid]\label{prop:b2:curves:torsion}
Een bireguliere boog ligt in een vlak dan en slechts dan als $\tau
\equiv 0$; in dat geval is het vlak het (constante) osculerende
vlak.
\end{proposition}

\begin{proof}
Is $\tau \equiv 0$, dan is $B' = 0$, dus is $B$ een constante
eenheidsvector $B_0$, en
\[
\frac{\dd}{\dd s}\langle \tilde\gamma(s), B_0\rangle
= \langle T(s), B_0\rangle = 0 :
\]
$\langle \tilde\gamma, B_0\rangle$ is constant, dus ligt de boog in
een vlak loodrecht op $B_0$. Omgekeerd, ligt de boog in een vlak
$P$, dan zijn $T$ en $T'$ (en dus $N$) voor alle $s$ evenwijdig aan
de richting van $P$; dus is $B = T \wedge N$ een van de twee
eenheidsnormalen van $P$, en omdat zij \omterm{def:b2:metric:continuity}{continu} is, is zij constant;
dan dwingt $0 = B' = -\tau N$ met $N \neq 0$ af dat $\tau \equiv
0$.
\end{proof}

\begin{example}[Een gekantelde cirkel heeft torsie nul]
De boog $\gamma(t) = \bigl(\cos t,\ \tfrac{\sin t}{\sqrt2},\
\tfrac{\sin t}{\sqrt2}\bigr)$ ligt in het vlak $y = z$ en is de
eenheidscirkel van dat vlak (controle: $\norm{\gamma(t)} = 1$, en
de orthonormale basis $(1,0,0)$, $(0, \tfrac1{\sqrt2},
\tfrac1{\sqrt2})$ van het vlak vertoont de standaardparametrisering).
Zonder enige berekening met Frenet voorspelt
\cref{prop:b2:curves:torsion} dat $\tau \equiv 0$, en moet de vaste
binormaal de eenheidsnormaal $\pm(0, \tfrac1{\sqrt2},
-\tfrac1{\sqrt2})$ van het vlak zijn. De \omterm{thm:b2:curves:frenet3d}{torsie} meet niet het
``gekanteld zijn in de ruimte''; zij meet het \emph{verlaten} van
een vlak. Alleen de $b$ ongelijk aan nul van de schroeflijn
hieronder brengt echte \omterm{thm:b2:curves:frenet3d}{torsie} voort.
\end{example}

\begin{example}[De schroeflijn]\label{ex:b2:curves:helixfrenet}
Voor de schroeflijn $\gamma(t) = (a\cos t, a\sin t, bt)$ met $a >
0$ berekenden wij $s = ct$ met $c = \sqrt{a^2 + b^2}$. Dan is
\[
T = \frac1c(-a\sin t,\ a\cos t,\ b),
\qquad
T' \cdot \frac{\dd t}{\dd s}
= \frac{1}{c^2}(-a\cos t, -a\sin t, 0),
\]
dus $\kappa = a/c^2 = a/(a^2 + b^2)$ en $N = (-\cos t, -\sin t,
0)$: de hoofdnormaal wijst horizontaal naar de as. Vervolgens is $B
= T \wedge N = \frac1c(b\sin t, -b\cos t, a)$, en $B' \frac{\dd
t}{\dd s} = \frac{b}{c^2}(\cos t, \sin t, 0) = -\tau N$ geeft
\[
\boxed{\ \kappa = \frac{a}{a^2 + b^2}, \qquad
\tau = \frac{b}{a^2 + b^2}. \ }
\]
Beide invarianten zijn constant --- en men kan omgekeerd aantonen
dat de enige bireguliere krommen met constante $\kappa > 0$ en
constante $\tau$ de schroeflijnen zijn (cirkels wanneer $\tau =
0$). Let op de tekens: $b > 0$ geeft een rechtsdraaiende
schroeflijn met positieve \omterm{thm:b2:curves:frenet3d}{torsie}.
\end{example}

\begin{example}[Kromming en torsie zonder booglengte; de
gedraaide kubiek]\label{ex:b2:curves:twistedcubic}
Opnieuw parametriseren naar \omterm{def:b2:curves:length}{booglengte} is meestal onmogelijk in
gesloten vorm, dus moeten de invarianten uit de ruwe afgeleiden
worden gehaald. Schrijf $v = \norm{\gamma'} = s'$; dan is $\gamma'
= vT$ en, zoals in het bewijs van
\cref{prop:b2:curves:kappaformula},
\[
\gamma'' = v'T + v^2\kappa N,
\qquad
\gamma' \wedge \gamma'' = v^3\kappa\,(T \wedge N) =
v^3\kappa\,B .
\]
\omterm{def:b2:nvs:norm}{Normen} nemen ($\kappa \geq 0$ in de ruimte) geeft
\[
\kappa = \frac{\norm{\gamma' \wedge \gamma''}}{v^3} .
\]
$\gamma''$ nog eenmaal differentiëren en $N' = v(-\kappa T + \tau
B)$ omzetten (kettingregel via $s$): de enige $B$-component komt
uit de laatste term,
\[
\gamma''' = \bigl(v'' - v^3\kappa^2\bigr)T +
\bigl(v'v\kappa + (v^2\kappa)'\bigr)N + v^3\kappa\tau\,B,
\]
zodat, gepaard met $\gamma' \wedge \gamma'' = v^3\kappa B$,
\[
\det(\gamma', \gamma'', \gamma''') = \langle\gamma' \wedge
\gamma'',\ \gamma'''\rangle = v^6\kappa^2\tau,
\qquad\text{dat wil zeggen}\qquad
\tau = \frac{\det(\gamma', \gamma'',
\gamma''')}{\norm{\gamma' \wedge \gamma''}^2} .
\]
Toepassing op de \emph{gedraaide kubiek} $\gamma(t) = (t,\ t^2,\
t^3)$ in $t = 0$: $\gamma' = (1, 0, 0)$, $\gamma'' = (0, 2, 0)$,
$\gamma''' = (0, 0, 6)$, dus $v = 1$,
\[
\gamma' \wedge \gamma'' = (0, 0, 2), \qquad \kappa(0) = 2,
\qquad
\det(\gamma', \gamma'', \gamma''') = 12, \qquad \tau(0) =
\frac{12}{4} = 3 .
\]
Het inzicht om te onthouden: beide formules zijn verhoudingen
waarin de snelheid $v$ precies in de nodige graad wegvalt ---
$\kappa$ schaalt als een tweede afgeleide per eenheid \omterm{def:b2:curves:length}{lengte},
$\tau$ als het gemengde volume van drie afgeleiden per kwadraat van
een oppervlakte --- en \emph{daarom} zijn zij meetkundig terwijl
$\gamma''$ zelf dat niet is.
\end{example}

\begin{remark}[Hoofdstelling voor ruimtekrommen]
Net als in het vlak legt het paar $(\kappa, \tau)$ met $\kappa > 0$
een bireguliere boog vast op een directe isometrie van $\R^3$ na:
de formules van Frenet vormen een lineair differentiaalstelsel voor
het stelsel $(T, N, B)$, waarop de theorie van Cauchy--Lipschitz
uit \cref{ch:b2:diffeq} van toepassing is; de orthonormaliteit van
het oplossingsstelsel blijft bewaard omdat de coëfficiëntenmatrix
antisymmetrisch is (hetzelfde argument met de Gram-matrix als in
\cref{exo:b2:curves:7}), en de kromme wordt teruggevonden door $T$
te integreren. Wij laten de bijzonderheden aan de lezer over als
een stevige maar leerzame oefening.
\end{remark}

\section{Lokale studie: ligging ten opzichte van de raaklijn}

\begin{proposition}[Lokale gedaante in een regulier
punt]\label{prop:b2:curves:local}
Zij $\gamma$ een vlakke boog van klasse $\mathcal{C}^k$ in $t_0$,
met $p$ de kleinste index waarvoor $\gamma^{(p)}(t_0) \neq 0$ en
$q$ de kleinste index $> p$ waarvoor $\gamma^{(q)}(t_0)$ niet
collineair is met $\gamma^{(p)}(t_0)$ (aangenomen dat beide
bestaan, $q \leq k$). In de basis $(u, v) = (\gamma^{(p)}(t_0),
\gamma^{(q)}(t_0))$ met oorsprong $\gamma(t_0)$ geeft
Taylor--Young de coördinaten
\[
X(t) \sim \frac{(t - t_0)^p}{p!},
\qquad
Y(t) \sim \frac{(t - t_0)^q}{q!} .
\]
Het lokale beeld hangt alleen van de pariteiten van $p$ en $q$ af:
\begin{center}
\small
\begin{tabular}{c c p{5.6cm}}
$p$ oneven, $q$ even & \emph{gewoon punt} & de kromme kruist geen
rechte, blijft aan één kant van de \omterm{def:b2:curves:arc}{raaklijn}\\
$p$ oneven, $q$ oneven & \emph{buigpunt} & de kromme kruist haar
\omterm{def:b2:curves:arc}{raaklijn}\\
$p$ even, $q$ oneven & \emph{keerpunt van de eerste soort} & beide
takken aan weerszijden van de \omterm{def:b2:curves:arc}{raaklijn}\\
$p$ even, $q$ even & \emph{keerpunt van de tweede soort} & beide
takken aan dezelfde kant\\
\end{tabular}
\end{center}
\end{proposition}

\begin{proof}
Taylor--Young van orde $q$ (de functie $\gamma$ is
$\mathcal{C}^q$ nabij $t_0$):
\[
\gamma(t) - \gamma(t_0)
= \sum_{j=p}^{q} \frac{(t-t_0)^j}{j!}\,\gamma^{(j)}(t_0)
+ o\bigl((t-t_0)^q\bigr).
\]
Wegens de keuze van $p$ en $q$ is elke $\gamma^{(j)}(t_0)$ met $p
\leq j < q$ collineair met $u$; de componenten in de basis $(u, v)$
verzamelen geeft $X(t) = \frac{(t-t_0)^p}{p!}(1 + o(1))$ en $Y(t) =
\frac{(t-t_0)^q}{q!}(1 + o(1))$. De tekentabel van $X$ en $Y$ voor
$t \gtrless t_0$ --- precies bestuurd door de pariteiten --- geeft
de vier beelden: is bijvoorbeeld $p$ even, dan is $X > 0$ aan beide
kanten (beide takken vertrekken in de richting $+u$: een keerpunt),
en de kant van de \omterm{def:b2:curves:arc}{raaklijn} ($\operatorname{sign} Y$) wisselt als
$q$ oneven is.
\end{proof}

\begin{example}
Voor $\gamma(t) = (t^2, t^3)$ in $t_0 = 0$
(\cref{ex:b2:curves:cusp}): $\gamma'' (0)= (2, 0)$, $\gamma'''(0) =
(0, 6)$, dus $p = 2$, $q = 3$: een keerpunt van de eerste soort,
het vertrouwde beeld van de semikubische parabool. Voor $\gamma(t)
= (t, t^3)$ in $0$: $p = 1$, $q = 3$: buigpunt --- de derdegraads
kromme kruist haar \omterm{def:b2:curves:arc}{raaklijn}.
\end{example}

\begin{remark}[Vooruitblik binnen dit volume]
Krommen voeden de volgende hoofdstukken op drie manieren. Op een
oppervlak getekend definiëren zij zijn raakvlakken en zijn eerste
fundamentaalvorm (\cref{ch:b2:surfaces}), en hun \omterm{def:b2:curves:length}{lengten} worden
berekend door de omgevende metriek te beperken --- het hoofdstuk
hierna is grotendeels dit hoofdstuk, gerelativeerd. De rekenkunde
van de \omterm{pb:b2:curves:1}{omhullenden} uit de weekendopgave ontmoet dubbele integralen
in \cref{ch:b2:multint}, waar de oppervlakte van de \omterm{pb:b2:curves:1}{astroïde}
opnieuw wordt berekend met de formule van Green
(\cref{exo:b2:multint:5}) --- één kromme, twee theorieën,
overeenstemmende antwoorden. En het stelsel van Frenet gebruikte al
de lineaire differentiaalvergelijkingen van \cref{ch:b2:diffeq}
(bestaan, eenduidigheid en het argument voor het behoud van de
orthogonaliteit uit \cref{exo:b2:curves:7}): de hoofdstelling voor
krommen is een stelling over differentiaalvergelijkingen in
meetkundige kledij.
\end{remark}

\begin{remark}[Methode: de lokale studie uitvoeren]
In de praktijk is de klassering een routine van vier stappen.
\emph{Eén:} differentieer in $t_0$ tot de eerste afgeleide
ongelijk aan nul verschijnt: haar index is $p$, haar waarde de
vector $u$. \emph{Twee:} blijf differentiëren tot een afgeleide
verschijnt die niet collineair is met $u$: index $q$, vector $v$.
\emph{Drie:} lees de pariteiten $(p, q)$ in de tabel. \emph{Vier:}
teken: de kromme vertrekt langs $+u$ als $p$ oneven is (langs $u$
en dan weer terug langs $u$ als $p$ even is), aan de kant van $v$
die het teken van $Y$ voorschrijft. Twee waarschuwingen. Het
stelsel $(u, v)$ is in het algemeen \emph{niet} orthonormaal --- de
tabel beschrijft liggingen ten opzichte van de \omterm{def:b2:curves:arc}{raaklijn}, geen
hoeken of afstanden, dus lees geen \omterm{thm:b2:curves:frenet2d}{kromming} van het beeld af. En
tussenliggende afgeleiden die collineair zijn met $u$ zijn tussen
de rangen $p$ en $q$ toegestaan (zij verschuiven alleen de
ontwikkeling van $X$); wat \emph{niet} mag gebeuren, is stoppen bij
de eerste afgeleide ongelijk aan nul en $q = p + 1$ gokken: voor
$\gamma(t) = (t^2, t^4 + t^5)$ is de naïeve gok $q = 3$ verkeerd,
want $\gamma^{(3)}(0)$ is nog altijd collineair met $\gamma''(0)$
--- en dat is precies \cref{exo:b2:curves:5}.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b2:curves:1}
Bereken de \omterm{def:b2:curves:length}{lengte} van één boog van de cycloïde $\gamma(t) = (t -
\sin t,\ 1 - \cos t)$, $t \in [0, 2\pi]$. \emph{(Gebruik $1 - \cos
t = 2\sin^2(t/2)$.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:curves:2}
Bereken de \omterm{thm:b2:curves:frenet2d}{kromming} van de ellips $\gamma(t) = (a\cos t,\ b\sin t)$
($a > b > 0$) en lokaliseer de punten van maximale en minimale
\omterm{thm:b2:curves:frenet2d}{kromming}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:curves:3}
Toon aan dat de \omterm{def:b2:curves:length}{booglengte} van de grafiek van $f(x) = \cosh x$ over
$[0, x]$ gelijk is aan $\sinh x$, en bereken de \omterm{thm:b2:curves:frenet2d}{kromming} van deze
kromme (de \emph{kettinglijn}). Ga na dat $R(x) = 1/\kappa(x) =
\cosh^2 x$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:curves:4}
(Logaritmische spiraal) Zij $\gamma(t) = e^{t}(\cos t,\ \sin t)$,
$t \in \R$. Toon aan dat de hoek tussen $\gamma(t)$ en $\gamma'(t)$
constant is, bereken de \omterm{def:b2:curves:length}{booglengte} van $\gamma$ op $(-\infty, 0]$
(eindig!) en de \omterm{thm:b2:curves:frenet2d}{kromming}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:curves:5}
Bepaal $p$, $q$ en de lokale gedaante (gewoon punt, buigpunt,
keerpunt) van $\gamma(t) = (t^2,\ t^4 + t^5)$ in $t = 0$, en van
$\gamma(t) = (t^3,\ t^4)$ in $t = 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:curves:6}
Zij $\gamma$ een vlakke boog met eenheidssnelheid en $\kappa(s) >
0$ voor alle $s$, en zij $c(s) = \gamma(s) + \frac{1}{\kappa(s)}N(s)$
het \omterm{def:b2:curves:curvature}{krommingsmiddelpunt} (de kromme $c$ is de \emph{evolute}). Toon
aan, aangenomen dat $\kappa$ van klasse $\mathcal{C}^1$ is, dat
$c'(s) = -\frac{\kappa'(s)}{\kappa(s)^2}N(s)$: de evolute raakt de
normaallijnen van $\gamma$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:curves:7}
Zij $A(s)$ een \omterm{def:b2:metric:continuity}{continue} familie antisymmetrische $3 \times
3$-matrices en $F' = F A$ een matrixoplossing met $F(s_0)$
orthogonaal. Toon aan dat $F(s)$ orthogonaal is voor alle $s$.
\emph{(Differentieer $G = F F^{\mathsf T}$ en gebruik de
eenduidigheid in Cauchy--Lipschitz.)} Leg het belang voor het
stelsel van Frenet uit.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:curves:8}
(Totale \omterm{thm:b2:curves:frenet2d}{kromming} van een gesloten convexe kromme) Zij
$\tilde\gamma$ een gesloten vlakke $\mathcal{C}^2$-boog met
eenheidssnelheid en \omterm{def:b2:curves:length}{lengte} $L$ (dus $\tilde\gamma(s + L) =
\tilde\gamma(s)$), eenmaal tegen de wijzers van de klok in
doorlopen. Leg met de hoekfunctie $\varphi$ waarvoor $T =
(\cos\varphi, \sin\varphi)$ uit \cref{thm:b2:curves:fundamental}
uit waarom $\varphi(L) - \varphi(0)$ een veelvoud van $2\pi$ is, en
toon aan dat $\int_0^L \kappa(s)\,\dd s = \varphi(L) -
\varphi(0)$. (Voor een cirkel met straal $R$: $\int \kappa =
\frac1R \cdot 2\pi R = 2\pi$. De stelling van de draaiende
\omterm{def:b2:curves:arc}{raaklijnen} beweert de waarde $2\pi$ voor elke enkelvoudige gesloten
kromme; dat hoef je niet te bewijzen.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:curves:9}
Toon aan dat een bireguliere ruimtekromme met constante $\kappa >
0$ en $\tau = 0$ (een boog van) een cirkel met straal $1/\kappa$
is. \emph{(Gebruik \cref{prop:b2:curves:torsion} en toon dan aan
dat het middelpunt $\gamma + \frac1\kappa N$ constant is.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:curves:10}
Bereken de \omterm{def:b2:curves:length}{booglengte} van de parabool $y = x^2/2$ over $\intcc0a$
en toon aan dat zij gelijk is aan
\[
\tfrac12\Bigl(a\sqrt{1 + a^2} + \ln\bigl(a + \sqrt{1 +
a^2}\bigr)\Bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:curves:11}
Zij $\gamma$ een reguliere $\mathcal C^2$-boog in $\R^n$ waarvan
alle \omterm{def:b2:curves:arc}{raaklijnen} door een vast punt $P$ gaan. Bewijs dat de baan van
$\gamma$ in een rechte lijn bevat is. \emph{(Parametriseer naar
\omterm{def:b2:curves:length}{booglengte}, schrijf $\gamma(s) + \lambda(s)T(s) = P$ en
differentieer.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:curves:12}
(Hoofdstelling voor ruimtekrommen) Zijn $\kappa > 0$ en $\tau$
\omterm{def:b2:metric:continuity}{continue} functies op een interval $J$. Voer het programma uit van
de opmerking na \cref{ex:b2:curves:helixfrenet}: (a) toon aan dat
het lineaire stelsel $F' = FA(s)$, met $A(s)$ de antisymmetrische
frenetmatrix opgebouwd uit $\kappa, \tau$ en $F(s_0)$ een direct
orthonormaal stelsel, een unieke globale oplossing heeft die een
direct orthonormaal stelsel blijft; (b) construeer een bireguliere
kromme met eenheidssnelheid, \omterm{thm:b2:curves:frenet2d}{kromming} $\kappa$ en \omterm{thm:b2:curves:frenet3d}{torsie} $\tau$;
(c) bewijs de eenduidigheid op een directe isometrie van $\R^3$ na.
\end{exercise}

\section{Probleem: omhullenden --- de astroïde, twee evoluten en
een caustiek}

\begin{problem}[{Weekendopgave --- de machine van de omhullenden
en vier klassieke krommen}]\label{pb:b2:curves:1}
Een familie rechten met één parameter dekt het vlak gewoonlijk niet
gelijkmatig: de rechten hopen zich op langs een kromme die aan alle
raakt, hun \emph{omhullende}\index{omhullende}. Lichtstralen maken
\omterm{pb:b2:curves:1}{omhullenden} zichtbaar als \emph{caustieken} --- de heldere kromme
met keerpunten in een kop koffie. Deze opgave bouwt de algemene
machine van de \omterm{pb:b2:curves:1}{omhullenden} en laat haar dan vier keer lopen: de
glijdende ladder (\omterm{pb:b2:curves:1}{astroïde}), de normalen van de parabool en van de
cycloïde (evoluten, met aan het eind de slinger van Huygens) en de
caustiek van het koffiekopje (\omterm{pb:b2:curves:1}{nefroïde}). Overal duidt $D_t$ de
rechte met vergelijking $a(t)\,x + b(t)\,y = c(t)$ aan, waarbij $a,
b, c$ van klasse $\mathcal C^2$ zijn met $(a(t), b(t)) \neq (0,0)$,
en is $\Delta(t) = a(t)b'(t) - a'(t)b(t)$.

\medskip
\noindent\textbf{Deel I --- De machine van de \omterm{pb:b2:curves:1}{omhullenden}.}
\begin{enumerate}
  \item Neem aan dat $\Delta(t) \neq 0$. Toon aan dat het
        \emph{karakteristieke stelsel}
        \[
        \begin{cases} a(t)\,x + b(t)\,y = c(t)\\
        a'(t)\,x + b'(t)\,y = c'(t)\end{cases}
        \]
        een unieke oplossing $E(t) = (x(t), y(t))$ heeft, gegeven
        door $x = \dfrac{cb' - c'b}{\Delta}$, $y = \dfrac{ac' -
        a'c}{\Delta}$.
  \item Neem bovendien aan dat $E$ nabij $t$ van klasse $\mathcal
        C^1$ is met $E'(t) \neq 0$. Toon, door de eerste
        vergelijking van het stelsel te differentiëren, aan dat
        $a(t)\,x'(t) + b(t)\,y'(t) = 0$, en besluit dat de kromme
        $E$ door een punt van $D_t$ gaat \emph{met de richting
        van} $D_t$: de familie raakt aan $E$, die haar
        \emph{\omterm{pb:b2:curves:1}{omhullende}} wordt genoemd.
  \item Verstandscontrole: de \omterm{def:b2:curves:arc}{raaklijnen} van de parabool $y =
        x^2/2$ in de punten $(t, t^2/2)$ zijn $tx - y = t^2/2$. Ga
        na dat de machine van de \omterm{pb:b2:curves:1}{omhullenden} de parabool zelf
        teruggeeft.
  \item (\omterm{pb:b2:curves:1}{Omhullende} van de normalen) Zij $\gamma$ van
        eenheidssnelheid met $\kappa(s) \neq 0$. De normaallijn in
        $\gamma(s)$ is $\{M : \langle M - \gamma(s), T(s) \rangle =
        0\}$. Toon aan dat haar karakteristieke stelsel $\langle M
        - \gamma(s), N(s)\rangle = 1/\kappa(s)$ afdwingt, en dus
        dat het karakteristieke punt het \omterm{def:b2:curves:curvature}{krommingsmiddelpunt} is:
        \emph{de \omterm{pb:b2:curves:1}{omhullende} van de normalen is de evolute}, waarmee
        \cref{exo:b2:curves:6} wordt teruggevonden. Ga na dat
        $\Delta(s) = \kappa(s)$.
  \item Twee ontaardingen. Toon voor de waaier $D_\theta : x\cos
        \theta + y\sin\theta = 0$ aan dat het karakteristieke punt
        voor elke $\theta$ de oorsprong is (de ``\omterm{pb:b2:curves:1}{omhullende}'' klapt
        ineen tot een punt, en $E' = 0$: vraag 2 is niet van
        toepassing). Toon voor een familie evenwijdige rechten
        ($a, b$ constant) aan dat $\Delta \equiv 0$ en dat het
        karakteristieke stelsel in het algemeen strijdig is: geen
        \omterm{pb:b2:curves:1}{omhullende}.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De glijdende ladder en de \omterm{pb:b2:curves:1}{astroïde}.}
Een segment van \omterm{def:b2:curves:length}{lengte} $1$ glijdt met het ene uiteinde $P_t =
(\cos t, 0)$ op de vloer en het andere $Q_t = (0, \sin t)$ tegen de
muur, $t \in \intoo0{\pi/2}$.
\begin{enumerate}[resume]
  \item Toon aan dat de rechte $(P_tQ_t)$ vergelijking $x\sin t +
        y\cos t = \sin t\cos t$ heeft, en dat de machine van de
        \omterm{pb:b2:curves:1}{omhullenden} het karakteristieke punt
        \[
        E(t) = (\cos^3 t,\ \sin^3 t)
        \]
        geeft: de \emph{astroïde}\index{astroïde}, met impliciete
        vergelijking $x^{2/3} + y^{2/3} = 1$ (door symmetrie tot de
        andere kwadranten uitgebreid).
  \item Toon aan dat $E'(0) = 0$ en, met de lokale klassering
        (\cref{prop:b2:curves:local}), dat de \omterm{pb:b2:curves:1}{astroïde} in $(1, 0)$
        een keerpunt van de eerste soort heeft --- en evenzo in
        haar vier punten op de assen.
  \item Bereken $\norm{E'(t)} = \tfrac32\abs{\sin 2t}$ en leid af
        dat de totale \omterm{def:b2:curves:length}{lengte} van de \omterm{pb:b2:curves:1}{astroïde} $6$ is.
  \item Waar raakt de ladder de \omterm{pb:b2:curves:1}{astroïde}? Toon aan dat $E(t) = P_t
        + \sin^2 t\,(Q_t - P_t)$: het raakpunt verdeelt de ladder
        in de verhouding $\sin^2 t : \cos^2 t$, en veegt haar
        terwijl de ladder glijdt van het ene uiteinde naar het
        andere.
  \item Bereken de oppervlakte die de \omterm{pb:b2:curves:1}{astroïde} omsluit: toon aan
        dat de oppervlakte in het eerste kwadrant
        $3\int_0^{\pi/2}\sin^4 t\cos^2 t\,\dd t$ is, evalueer de
        integraal door lineariseren ($\sin^2 2t = \tfrac{1 - \cos
        4t}2$), en besluit dat de totale oppervlakte $3\pi/8$ is.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De evolute van de parabool.} Zij
$\gamma(t) = (t, t^2/2)$.
\begin{enumerate}[resume]
  \item Toon aan dat de normaallijn in $\gamma(t)$ vergelijking $x
        + t\,y = t + t^3/2$ heeft.
  \item Laat de machine van de \omterm{pb:b2:curves:1}{omhullenden} lopen: toon aan dat de
        \omterm{pb:b2:curves:1}{omhullende} van de normalen
        \[
        E(t) = \Bigl(-t^3,\ 1 + \tfrac32 t^2\Bigr)
        \]
        is, met impliciete vergelijking $x^2 = \tfrac8{27}(y -
        1)^3$: een semikubische parabool.
  \item Kruiscontrole met vraag 4: bereken het
        \omterm{def:b2:curves:curvature}{krommingsmiddelpunt} $\gamma(t) + \frac1{\kappa(t)}N(t)$
        uit $\kappa(t) = (1 + t^2)^{-3/2}$
        (\cref{ex:b2:curves:kappaexamples}) en vind hetzelfde punt
        terug.
  \item Toon aan dat de evolute in $(0, 1)$ een keerpunt van de
        eerste soort heeft, het \omterm{def:b2:curves:curvature}{krommingsmiddelpunt} in de top ---
        het punt waar $\kappa$ extremaal is, zoals de formule $c' =
        -\frac{\kappa'}{\kappa^2}N$ uit \cref{exo:b2:curves:6}
        voorspelt.
  \item Hoeveel normalen van de parabool gaan door een gegeven
        punt $(x_0, y_0)$? Toon aan dat het antwoord bestuurd wordt
        door de derdegraadsvergelijking $\tfrac{t^3}2 + (1 -
        y_0)\,t - x_0 = 0$; behandel het geval op de as $x_0 = 0$
        \omterm{def:b2:metric:complete}{volledig} (één normaal voor $y_0 < 1$, drie voor $y_0 > 1$),
        en interpreteer de evolute als de overgangskromme.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De caustiek van het koffiekopje.}
Evenwijdige stralen met richting $(1, 0)$ treffen de binnenkant van
de spiegelende cirkel $x^2 + y^2 = 1$; de straal die $P_\theta =
(\cos\theta, \sin\theta)$ raakt, weerkaatst volgens de
weerkaatsingswet.
\begin{enumerate}[resume]
  \item Verantwoord met de spiegelsymmetrie in de normaal (de
        straal) dat de weerkaatste richting $v = u - 2\langle u,
        n\rangle n$ is met $u = (1,0)$, $n = (\cos\theta,
        \sin\theta)$, en bereken $v = -(\cos2\theta,
        \sin2\theta)$.
  \item Toon aan dat de weerkaatste straal op de rechte
        \[
        x\sin 2\theta - y\cos 2\theta = \sin\theta
        \]
        ligt.
  \item Laat de machine van de \omterm{pb:b2:curves:1}{omhullenden} lopen ($\Delta = 2$):
        toon aan dat de caustiek
        \[
        E(\theta) = \Bigl(\tfrac{3\cos\theta -
        \cos3\theta}4,\ \tfrac{3\sin\theta -
        \sin3\theta}4\Bigr)
        \]
        is, de \emph{nefroïde}\index{nefroïde}.
  \item Bereken $E'(\theta) = \tfrac32\sin\theta\,
        (\cos2\theta, \sin2\theta)$; ga na dat de raakrichting de
        richting van de weerkaatste straal is (vraag 16),
        lokaliseer de twee keerpunten $(\pm\tfrac12, 0)$, en toon
        aan dat de weerkaatste straal de as $y = 0$ snijdt in $x =
        \frac1{2\cos\theta}$ --- zodat bijna axiale stralen in $x =
        \tfrac12$ samenkomen: de brandpuntsafstand $R/2$ van een
        spiegel met straal $R$.
  \item Toon aan dat de \omterm{pb:b2:curves:1}{nefroïde} totale \omterm{def:b2:curves:length}{lengte} $6$ heeft en dat
        nabij $\theta = 0$
        \[
        E(\theta) - \bigl(\tfrac12, 0\bigr) =
        \bigl(\tfrac34\theta^2 + o(\theta^2),\ \theta^3 +
        o(\theta^3)\bigr) :
        \]
        een keerpunt van de eerste soort, wijzend langs de as.
  \item Leg in één alinea uit waarom de caustiek helder is: door
        elk punt net buiten de caustiek gaan twee weerkaatste
        stralen, en in elk punt erop zijn de stralen ``oneindig
        geconcentreerd'' (de afbeelding $(\theta, \text{afstand
        langs de straal}) \mapsto \R^2$ heeft precies op de
        \omterm{pb:b2:curves:1}{omhullende} een kritiek punt).
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Huygens: de cycloïde is haar eigen
evolute.} Zij $\gamma(t) = (t - \sin t,\ 1 - \cos t)$, $t \in
\intoo0{2\pi}$, één boog van de cycloïde.
\begin{enumerate}[resume]
  \item Bereken $\kappa(t) = -\dfrac1{4\sin(t/2)}$ en het
        \omterm{def:b2:curves:curvature}{krommingsmiddelpunt}; toon aan dat de evolute
        \[
        c(t) = (t + \sin t,\ \cos t - 1)
        \]
        is, en dat de substitutie $t = u + \pi$ haar vertoont als
        de oorspronkelijke cycloïde, getransleerd over $(\pi,
        -2)$: \emph{de evolute van een cycloïde is een congruente
        cycloïde} (Huygens).
  \item Ga na dat de \omterm{def:b2:curves:curvature}{kromtestraal} in de top $t = \pi$ gelijk is aan
        $4$, de helft van de \omterm{def:b2:curves:length}{lengte} $8$ van één boog
        (\cref{exo:b2:curves:1}); lokaliseer het keerpunt van de
        evolute recht onder de top, op afstand $4$.
  \item (De eigenschap van het koord) Zij $\gamma$ van
        eenheidssnelheid met $\kappa > 0$, $\kappa$ van klasse
        $\mathcal C^1$ en $R = 1/\kappa$ strikt monotoon. Toon met
        $c' = R'N$ aan dat de \omterm{def:b2:curves:length}{booglengte} van de evolute tussen
        $c(s_0)$ en $c(s_1)$ gelijk is aan $\abs{R(s_1) -
        R(s_0)}$. Interpreteer: een strak koord dat van de evolute
        wordt afgewikkeld en bij aanvang \omterm{def:b2:curves:length}{lengte} $R(s_0)$ heeft,
        laat zijn vrije uiteinde de oorspronkelijke kromme
        beschrijven --- zodat een slinger die tussen twee
        cycloïdale wangen van de klok van Huygens zwaait, een
        cycloïde beschrijft.
  \item Synthese. De machine van Deel I bracht een \omterm{pb:b2:curves:1}{astroïde}, een
        semikubische parabool, een \omterm{pb:b2:curves:1}{nefroïde} en een cycloïde voort.
        Geef voor elk van de vier families in één zin aan waar de
        hypothesen $\Delta \neq 0$ en $E' \neq 0$ golden of
        faalden, en welk meetkundig verschijnsel (keerpunt,
        brandpunt, ontaarding) elk falen van $E' \neq 0$
        aankondigde. Waar moeten de extrema van de \omterm{thm:b2:curves:frenet2d}{kromming} op de
        \omterm{pb:b2:curves:1}{omhullende} van de normalen verschijnen, en waarom?
\end{enumerate}
\end{problem}
