\chapter{Discrete toevalsveranderlijken}\label{ch:b2:randomvar}

\omterm{def:b2:randomvar:law}{Toevalsveranderlijken} ordenen kansberekeningen rond functies in
plaats van rond \omterm{def:b2:proba:space}{gebeurtenissen}. Op \omterm{def:b2:structures:countable}{aftelbare} ruimten wordt de
theorie aangedreven door de \omterm{def:b2:series:summable}{sommeerbare families} van
\cref{ch:b2:series}: de \omterm{def:b2:randomvar:expectation}{verwachtingswaarde} is de som van een
familie geïndexeerd door de \omterm{def:b2:proba:space}{uitkomstenruimte}, en al haar
eigenschappen --- lineariteit, overdracht, de productformule voor
\omterm{def:b2:proba:independence}{onafhankelijke} veranderlijken --- zijn stellingen over \omterm{def:b2:series:summable}{sommeerbare
families}. Het hoofdstuk bewijst de sleutelongelijkheden van Markov,
Chebyshev, Cauchy--Schwarz en Jensen, en eindigt met de klassieke
\omterm{def:b2:randomvar:law}{verdelingen} en de zwakke wet van de grote aantallen, waarvan het
bewijs twee regels lang is zodra Chebyshev beschikbaar is.

\section{Toevalsveranderlijken en hun verdelingen}

\begin{definition}[Discrete toevalsveranderlijke; verdeling]\label{def:b2:randomvar:law}
Zij $(\Omega, \P)$ een \omterm{def:b2:structures:countable}{aftelbare} \omterm{def:b2:proba:space}{kansruimte}. Een
\emph{toevalsveranderlijke}\index{toevalsveranderlijke} is een
afbeelding $X \colon \Omega \to E$ ($E$ een willekeurige
verzameling; \emph{reële} toevalsveranderlijke wanneer $E = \R$).
Haar \emph{verdeling}\index{verdeling!van een toevalsveranderlijke}
(of \emph{kansverdeling}\index{kansverdeling}) is de \omterm{def:b2:proba:space}{kansmaat}
$\P_X$ op de \omterm{def:b2:structures:countable}{aftelbare verzameling} $X(\Omega)$ gedefinieerd door
\[
\P_X(\{x\}) = \P(X = x)
= \P\bigl(\{\omega : X(\omega) = x\}\bigr) .
\]
\end{definition}

\begin{example}[De klassieke verdelingen]\label{ex:b2:randomvar:laws}
\leavevmode
\begin{itemize}
\item \emph{Bernoulli} $\mathcal{B}(p)$: $X \in \{0, 1\}$, $\P(X =
1) = p$. De indicator van een \omterm{def:b2:proba:space}{gebeurtenis}.
\item \emph{Binomiaal} $\mathcal{B}(n, p)$: $\P(X = k) = \binom nk
p^k(1-p)^{n-k}$, $0 \leq k \leq n$: het aantal successen in $n$
\omterm{def:b2:proba:independence}{onafhankelijke} bernoulli-experimenten (bovenbouwvolume; hieronder
opnieuw bewezen via sommen van \omterm{def:b2:proba:independence}{onafhankelijke} veranderlijken).
\item \emph{Meetkundig} $\mathcal{G}(p)$: $\P(X = k) =
(1-p)^{k-1}p$, $k \in \N^*$: de rang van het eerste succes
(\cref{ex:b2:proba:geometric}).
\item \emph{Poisson} $\mathcal{P}(\lambda)$: $\P(X = k) =
e^{-\lambda}\frac{\lambda^k}{k!}$, $k \in \N$ --- een \omterm{def:b2:proba:space}{kansmaat}
wegens de exponentiële reeks. De \omterm{def:b2:randomvar:law}{verdeling} van de zeldzame
\omterm{def:b2:proba:space}{gebeurtenissen} (\cref{ch:b2:genfun}).
\end{itemize}
\end{example}

\begin{remark}[Welke verdeling modelleert wat]
De vier \omterm{def:b2:randomvar:law}{verdelingen} beantwoorden vier oervragen: Bernoulli, ``is
het gebeurd?''; binomiaal, ``hoe vaak in $n$ pogingen?'';
meetkundig, ``hoe lang tot de eerste keer?''; Poisson, ``hoeveel
\omterm{def:b2:proba:space}{gebeurtenissen} bij een gegeven tempo, wanneer er veel pogingen zijn
die elk onwaarschijnlijk zijn?''. De vraag herkennen is negen
tiende van het modelleren: sommen van indicatoren wijzen op de
binomiale, wachttijden op de meetkundige, en tellingen van zeldzame
\omterm{def:b2:proba:space}{gebeurtenissen} op de \omterm{def:b2:randomvar:law}{poisson-verdeling} --- waarbij de overgang van
binomiaal naar Poisson door de \omterm{def:b2:randomvar:law}{verdeling} van de zeldzame
\omterm{def:b2:proba:space}{gebeurtenissen} in \cref{ch:b2:genfun} precies wordt gemaakt.
\end{remark}

\begin{proposition}[Geheugenloosheid van de meetkundige
verdeling]\label{prop:b2:randomvar:memoryless}
Is $X \sim \mathcal{G}(p)$, dan geldt voor alle $m, n \in \N$
\[
\P(X > m + n \mid X > m) = \P(X > n) ,
\]
en de meetkundige \omterm{def:b2:randomvar:law}{verdelingen} zijn de enige \omterm{def:b2:randomvar:law}{verdelingen} op $\N^*$
met deze eigenschap.
\end{proposition}

\begin{proof}
De meetkundige gewichten sommeren geeft $\P(X > n) = (1-p)^n$.
Bijgevolg is
\[
\P(X > m + n \mid X > m)
= \frac{\P(X > m + n)}{\P(X > m)}
= \frac{(1-p)^{m+n}}{(1-p)^m} = (1-p)^n = \P(X > n).
\]
Omgekeerd, voldoet $G(n) = \P(X > n)$ aan $G(m + n) = G(m)G(n)$ met
$G(0) = 1$, dan is $G(n) = G(1)^n$ met inductie; $q = G(1) \in
\intco{0}{1}$, en ofwel is $q = 0$ ofwel is de \omterm{def:b2:randomvar:law}{verdeling}
$\mathcal{G}(1 - q)$: $\P(X = k) = G(k-1) - G(k) = q^{k-1}(1 -
q)$.
\end{proof}

\begin{example}[Geen enkel getal is ooit ``aan de beurt'']
Gooi met een dobbelsteen en wacht op een zes: de wachttijd is $X
\sim \mathcal G(1/6)$. De geheugenloosheid zegt dat na $10$
vruchteloze worpen de \emph{resterende} wachttijd $X - 10$, gegeven
$X > 10$, opnieuw $\mathcal G(1/6)$ is: de voorwaardelijke
verwachte wachttijd is nog altijd $6$ worpen, precies als bij het
begin. De dobbelsteen onthoudt niets, en geen zes is ooit ``aan de
beurt'' --- de drogreden van de gokker is het geloof dat de
voorwaardelijke \omterm{def:b2:randomvar:law}{verdeling} verschoven zou moeten zijn. Omgekeerd
zegt de eenduidigheidshelft van de propositie dat deze
onverschilligheid meetkundige wachttijden \emph{karakteriseert}:
elke wachttijd waarvan de voorspelling zich nooit bijstelt, is
meetkundig. Echte wachtrijen en levensduren stellen zich meestal
wél bij, en precies zo merkt men dat zij niet meetkundig zijn.
\end{example}

\section{Verwachtingswaarde}

\begin{definition}[Verwachtingswaarde]\label{def:b2:randomvar:expectation}
Een reële \omterm{def:b2:randomvar:law}{toevalsveranderlijke} $X$ op $(\Omega, \P)$ \emph{heeft
een verwachtingswaarde} wanneer de familie
$\bigl(X(\omega)\,\P(\{\omega\})\bigr)_{\omega \in \Omega}$
\omterm{def:b2:series:summable}{sommeerbaar} is (\cref{ch:b2:series}); haar
\emph{verwachtingswaarde}\index{verwachtingswaarde} is dan
\[
\E(X) = \sum_{\omega \in \Omega} X(\omega)\,\P(\{\omega\}) .
\]
\end{definition}

\begin{theorem}[Overdrachtsstelling]\label{thm:b2:randomvar:transfer}
$X$ heeft een \omterm{def:b2:randomvar:expectation}{verwachtingswaarde} dan en slechts dan als de familie
$\bigl(x\,\P(X = x)\bigr)_{x \in X(\Omega)}$ \omterm{def:b2:series:summable}{sommeerbaar} is, en dan
is
\[
\E(X) = \sum_{x \in X(\Omega)} x\,\P(X = x) .
\]
Algemener heeft voor $f \colon X(\Omega) \to \R$ de veranderlijke
$f(X)$ een \omterm{def:b2:randomvar:expectation}{verwachtingswaarde} dan en slechts dan als $\sum_x
\abs{f(x)}\,\P(X = x) < \infty$, en dan is $\E(f(X)) = \sum_x
f(x)\,\P(X = x)$.
\end{theorem}

\begin{proof}
Partitioneer $\Omega$ in de niveauverzamelingen $\Omega_x = \{X =
x\}$, $x \in X(\Omega)$. Volgens de stelling over sommeren in
pakketten voor \omterm{def:b2:series:summable}{sommeerbare families} (\cref{ch:b2:series}) is de
familie $(X(\omega)\P(\{\omega\}))_\omega$ \omterm{def:b2:series:summable}{sommeerbaar} dan en
slechts dan als elk pakket dat is (automatisch: $\sum_{\omega \in
\Omega_x}\abs{x}\P(\{\omega\}) = \abs x\,\P(X = x)$) \emph{en} de
familie van pakketsommen $\bigl(x\,\P(X = x)\bigr)_x$ \omterm{def:b2:series:summable}{sommeerbaar}
is --- en dan stemmen de totale sommen overeen. Voor $f(X)$: pas de
bewezen uitspraak toe op de veranderlijke $Y = f \circ X$, waarvan
de niveauverzamelingen $\{Y = y\} = \bigsqcup_{x : f(x) = y}\{X =
x\}$ zijn; een tweede sommatie in pakketten zet $\sum_y y\,\P(Y =
y)$ om in $\sum_x f(x)\,\P(X = x)$, waarbij de pakketten nu de
waarden $x$ groeperen naar hun beeld $f(x)$, en de absolute
\omterm{def:b2:series:summable}{sommeerbaarheid} van de ene familie gelijkwaardig is met die van de
andere.
\end{proof}

\begin{theorem}[Eigenschappen van de verwachtingswaarde]\label{thm:b2:randomvar:linearity}
Op de verzameling \omterm{def:b2:randomvar:law}{toevalsveranderlijken} met \omterm{def:b2:randomvar:expectation}{verwachtingswaarde}:
\begin{enumerate}
\item (Lineariteit) $\E(aX + bY) = a\,\E(X) + b\,\E(Y)$.
\item (Positiviteit en monotonie) $X \geq 0 \Rightarrow \E(X) \geq
0$; $X \leq Y \Rightarrow \E(X) \leq \E(Y)$; en $\abs{\E(X)} \leq
\E(\abs X)$.
\item (Dominatie) Is $\abs X \leq Z$ en heeft $Z$ een
\omterm{def:b2:randomvar:expectation}{verwachtingswaarde}, dan heeft $X$ er ook een.
\end{enumerate}
\end{theorem}

\begin{proof}
Alle zijn eigenschappen van sommen van \omterm{def:b2:series:summable}{sommeerbare families}
(\cref{ch:b2:series}): de lineariteit van de som, de positiviteit
term voor term, en het dominatiecriterium voor de \omterm{def:b2:series:summable}{sommeerbaarheid}.
(Merk op dat de lineariteit onmiddellijk is op de \emph{definitie}
over $\Omega$, terwijl zij op de overdrachtsformule onhandig zou
zijn --- één voordeel van $\E$ stroomopwaarts te definiëren.)
\end{proof}

\begin{example}\label{ex:b2:randomvar:expectations}
$X \sim \mathcal{B}(n, p)$: schrijf $X = X_1 + \dots + X_n$ als een
som van bernoulli-indicatoren en gebruik de lineariteit: $\E(X) =
np$ --- geen binomiaalcoëfficiënten nodig. $X \sim
\mathcal{G}(p)$: $\E(X) = \sum_{k\geq1}k(1-p)^{k-1}p =
p\cdot\frac{1}{(1 - (1-p))^2} = \frac1p$, door de meetkundige reeks
binnen haar schijf te differentiëren (\cref{ch:b2:powerseries}). $X
\sim \mathcal{P}(\lambda)$: $\E(X) = \sum_{k\geq1}k
e^{-\lambda}\frac{\lambda^k}{k!} = \lambda
e^{-\lambda}\sum_{j\geq0}\frac{\lambda^j}{j!} = \lambda$.
\end{example}

\begin{example}[Overdracht in actie]\label{ex:b2:randomvar:transferex}
Bereken voor $X \sim \mathcal P(\lambda)$ de waarde
$\E\bigl(\frac1{1+X}\bigr)$ --- de \omterm{def:b2:randomvar:law}{verdeling} van $\frac1{1+X}$
zelf is onhandig, maar de overdracht vraagt er nooit naar:
\[
\E\Bigl(\frac1{1+X}\Bigr)
= \sum_{k\geq0}\frac{1}{k+1}\,\eu^{-\lambda}
\frac{\lambda^k}{k!}
= \frac{\eu^{-\lambda}}{\lambda}\sum_{k\geq0}
\frac{\lambda^{k+1}}{(k+1)!}
= \frac{\eu^{-\lambda}}{\lambda}\bigl(\eu^\lambda - 1\bigr)
= \frac{1 - \eu^{-\lambda}}{\lambda} .
\]
Twee lessen. Rekenkundig: een verschoven exponentiële reeks
herkennen is het hele werk --- de overdracht herleidt
\omterm{def:b2:randomvar:expectation}{verwachtingswaarden} van $f(X)$ tot het hanteren van reeksen.
Structureel: de naïeve invulwaarde zou $\frac1{1 + \E X} = \frac1{1
+ \lambda}$ zijn, en het echte antwoord is groter,
\[
\frac{1 - \eu^{-\lambda}}{\lambda} \geq
\frac{1}{1 + \lambda},
\]
precies zoals de ongelijkheid van Jensen voor de convexe functie $t
\mapsto \frac1{1+t}$ eist. \omterm{def:b2:randomvar:expectation}{Verwachtingswaarden} van \omterm{def:b2:affine:convex}{convexe} beelden
liggen boven de naïeve invulwaarde, en de overdracht plus een
reekscontrole maakt de abstracte ongelijkheid concreet.
\end{example}

\begin{theorem}[Onafhankelijkheid en producten]\label{thm:b2:randomvar:product}
\omterm{def:b2:randomvar:law}{Toevalsveranderlijken} $X, Y$ heten \emph{\omterm{def:b2:proba:independence}{onafhankelijk}} wanneer
$\P(X = x, Y = y) = \P(X = x)\P(Y = y)$ voor alle $x, y$ ---
gelijkwaardig: wanneer de \omterm{def:b2:proba:space}{gebeurtenissen} $\{X \in A\}$ en $\{Y \in
B\}$ \omterm{def:b2:proba:independence}{onafhankelijk} zijn voor alle $A, B$. Zijn $X$ en $Y$
\omterm{def:b2:proba:independence}{onafhankelijke} reële veranderlijken met \omterm{def:b2:randomvar:expectation}{verwachtingswaarden}, dan
heeft $XY$ een \omterm{def:b2:randomvar:expectation}{verwachtingswaarde} en is
\[
\E(XY) = \E(X)\,\E(Y) .
\]
\end{theorem}

\begin{proof}
De gelijkwaardigheid van de twee formuleringen volgt door de
\omterm{def:b2:funcseq:def}{puntsgewijze} identiteit over $(x, y) \in A \times B$ te sommeren
(tweemaal $\sigma$-additiviteit). Voor het product: de dubbele
familie $\bigl(xy\,\P(X = x)\P(Y = y)\bigr)_{(x,y)}$ is
\omterm{def:b2:series:summable}{sommeerbaar}, want volgens Fubini voor families
(\cref{ch:b2:series}) is
\[
\sum_{x, y}\abs x \abs y\,\P(X{=}x)\P(Y{=}y)
= \Bigl(\sum_x \abs x \P(X{=}x)\Bigr)
\Bigl(\sum_y \abs y \P(Y{=}y)\Bigr) < \infty ;
\]
wegens de \omterm{def:b2:proba:independence}{onafhankelijkheid} is deze familie precies
$\bigl(xy\,\P(X = x, Y = y)\bigr)$, waarvan de som $\E(XY)$ is
volgens de overdracht toegepast op de veranderlijke $(X, Y) \mapsto
xy$; Fubini evalueert de ongetekende som opnieuw als het product
$\E(X)\E(Y)$.
\end{proof}

\begin{example}[Producten, met en zonder
onafhankelijkheid]\label{ex:b2:randomvar:productcontrast}
Gooi met twee eerlijke dobbelstenen. Is $Y$ de tweede dobbelsteen
(\omterm{def:b2:proba:independence}{onafhankelijk} van de eerste), dan is $\E(XY) = \E(X)\E(Y) =
3.5^2 = 12.25$. Is daarentegen $Y = X$ (het ``product'' van een
dobbelsteen met zichzelf), dan is
\[
\E(X^2) = \frac{1 + 4 + 9 + 16 + 25 + 36}{6} = \frac{91}{6}
\approx 15.17 \neq 12.25 :
\]
dezelfde marginale \omterm{def:b2:randomvar:law}{verdelingen} in beide scenario's, verschillende
gezamenlijke \omterm{def:b2:randomvar:law}{verdelingen}, verschillende \omterm{def:b2:randomvar:expectation}{verwachtingswaarden} van het
product. De moraal, het onthouden waard: $\E(XY)$ is een
functionaal van het \emph{paar}, niet van de twee marginalen --- en
het gat $\E(X^2) - \E(X)^2 \approx 2.92$ is volgens
König--Huygens precies de \omterm{def:b2:randomvar:variance}{variantie} $\frac{35}{12}$ van de
dobbelsteen.
\end{example}

\section{Variantie, covariantie en de klassieke ongelijkheden}

\begin{definition}[Momenten, variantie]\label{def:b2:randomvar:variance}
$X$ heeft een \emph{moment van orde 2} wanneer $X^2$ een
\omterm{def:b2:randomvar:expectation}{verwachtingswaarde} heeft (dan ook $X$, wegens de dominatie: $\abs X
\leq \frac{1 + X^2}{2}$). Haar \emph{variantie}\index{variantie} en
\emph{standaardafwijking}\index{standaardafwijking} zijn dan
\[
V(X) = \E\bigl((X - \E(X))^2\bigr)
= \E(X^2) - \E(X)^2 ,
\qquad
\sigma(X) = \sqrt{V(X)} ,
\]
(de tweede vorm --- de formule van \emph{König--Huygens} --- door
het kwadraat uit te werken en de lineariteit te gebruiken:
\[
\E\bigl((X - \E X)^2\bigr)
= \E\bigl(X^2 - 2X\,\E X + \E(X)^2\bigr)
= \E(X^2) - 2\,\E(X)^2 + \E(X)^2 ,
\]
waarbij de middelste term gebruikt dat $\E X$ een constante is).
Voor $X, Y$ met tweede momenten is de
\emph{covariantie}\index{covariantie}
\[
\operatorname{Cov}(X, Y)
= \E\bigl((X - \E X)(Y - \E Y)\bigr)
= \E(XY) - \E(X)\E(Y) .
\]
\end{definition}

\begin{theorem}[Gereedschapskist van de variantie]\label{thm:b2:randomvar:variancerules}
Voor veranderlijken met tweede momenten:
\begin{enumerate}
\item $V(aX + b) = a^2\,V(X)$;
\item $V(X + Y) = V(X) + V(Y) + 2\operatorname{Cov}(X, Y)$, en
algemener
\[
V\Bigl(\sum_{i=1}^n X_i\Bigr)
= \sum_{i=1}^n V(X_i)
+ 2\sum_{i < j}\operatorname{Cov}(X_i, X_j) ;
\]
\item zijn $X, Y$ \omterm{def:b2:proba:independence}{onafhankelijk}, dan is $\operatorname{Cov}(X, Y) =
0$ (het omgekeerde is onwaar), zodat de \omterm{def:b2:randomvar:variance}{varianties} van
\omterm{def:b2:proba:independence}{onafhankelijke} veranderlijken optellen.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1} en \emph{2} zijn het uitwerken van kwadraten plus de
lineariteit; de producten $X_iX_j$ hebben \omterm{def:b2:randomvar:expectation}{verwachtingswaarden}
volgens Cauchy--Schwarz hieronder (of wegens $\abs{X_iX_j} \leq
\frac{X_i^2 + X_j^2}{2}$). \emph{3} is
\cref{thm:b2:randomvar:product} toegepast op de gecentreerde
veranderlijken. Een standaardtegenvoorbeeld voor het omgekeerde:
$X$ uniform op $\{-1, 0, 1\}$ en $Y = X^2$ zijn ongecorreleerd
($\E(XY) = \E(X^3) = 0 = \E X \cdot \E Y$) maar duidelijk
afhankelijk.
\end{proof}

\begin{theorem}[Ongelijkheden van Markov en
Chebyshev]\label{thm:b2:randomvar:markov}
\leavevmode
\begin{enumerate}
\item (Markov) Heeft $X \geq 0$ een \omterm{def:b2:randomvar:expectation}{verwachtingswaarde}, dan is voor
elke $a > 0$
\[
\P(X \geq a) \leq \frac{\E(X)}{a} .
\]
\item (Chebyshev) Heeft $X$ een tweede moment, dan is voor elke
$\varepsilon > 0$
\[
\P\bigl(\abs{X - \E(X)} \geq \varepsilon\bigr)
\leq \frac{V(X)}{\varepsilon^2} .
\]
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} Puntsgewijs is $a\,\mathbf{1}_{X \geq a} \leq X$ (op de
\omterm{def:b2:proba:space}{gebeurtenis} is het linkerlid $a \leq X$; erbuiten $0 \leq X$). Neem
\omterm{def:b2:randomvar:expectation}{verwachtingswaarden}: $a\,\P(X \geq a) \leq \E(X)$ wegens de
monotonie en $\E(\mathbf{1}_A) = \P(A)$.
\emph{2.} Pas Markov toe op de niet-negatieve veranderlijke $(X -
\E X)^2$ op niveau $a = \varepsilon^2$: de \omterm{def:b2:proba:space}{gebeurtenis} $\{(X - \E
X)^2 \geq \varepsilon^2\}$ is precies $\{\abs{X - \E X} \geq
\varepsilon\}$.
\end{proof}

\begin{example}[Ongecorreleerd maar aan elkaar
vastgelijmd]\label{ex:b2:randomvar:sumdiff}
Gooi met twee eerlijke dobbelstenen, $X$ en $Y$ \omterm{def:b2:proba:independence}{onafhankelijk}, en
zet $S = X + Y$, $D = X - Y$. Wegens de bilineariteit van de
\omterm{def:b2:randomvar:variance}{covariantie} is
\[
\operatorname{Cov}(S, D) = V(X) - V(Y) +
\operatorname{Cov}(Y, X) - \operatorname{Cov}(X, Y) = V(X) -
V(Y) = 0 :
\]
som en verschil zijn ongecorreleerd. \omterm{def:b2:proba:independence}{Onafhankelijk}? Zeker niet: $S
= 12$ dwingt $D = 0$ af, terwijl $\P(D = 0) = \frac16$ is zonder
voorwaarde. De correlatie toetst alleen het \emph{lineaire} deel
van een afhankelijkheid; hier wordt de afhankelijkheid gedragen
door de voorwaarde dat $S$ en $D$ dezelfde pariteit hebben,
onzichtbaar voor de \omterm{def:b2:randomvar:variance}{covariantie}. (Voor dit paar had de nulcovariantie
$V(X) = V(Y)$ nodig: gelijke \omterm{def:b2:randomvar:law}{verdelingen}, niet de
\omterm{def:b2:proba:independence}{onafhankelijkheid}, deden het werk.)
\end{example}

\begin{example}[Wanneer Markov exact is]\label{ex:b2:randomvar:markovsharp}
De ongelijkheid van Markov is een gelijkheid precies wanneer er
niets verloren gaat in de grens $a\,\mathbf 1_{X\geq a} \leq X$: de
veranderlijke moet alleen de waarden $0$ en $a$ aannemen. Concreet,
is $\P(X = a) = \pi$ en $\P(X = 0) = 1 - \pi$, dan is $\E(X) =
a\pi$ en
\[
\P(X \geq a) = \pi = \frac{\E(X)}{a} .
\]
Een realistische lezing: in een bevolking waar het gemiddelde
vermogen $100$ is en het vermogen ofwel $0$ ofwel $10^6$ bedraagt,
is het aandeel miljonairs precies $10^{-4}$ --- de grens van
Markov, exact bereikt door een maximale ongelijkheid. Zodra $X$
zich over tussenliggende waarden uitspreidt, is de grens strikt,
vaak wild; maar zoals het extreme geval toont, kan uit het
gemiddelde alleen geen betere ongelijkheid worden gehaald.
\end{example}

\begin{example}[Chebyshev is scherp --- zonder verdere
hypothesen]\label{ex:b2:randomvar:chebsharp}
Houd $\varepsilon > 0$ en $q \in \intoc01$ vast, en laat $X$ de
waarden $\pm\varepsilon$ elk met kans $\frac q2$ aannemen en $0$
met kans $1 - q$. Dan is $\E(X) = 0$, $V(X) = q\varepsilon^2$ en
\[
\P\bigl(\abs{X - \E X} \geq \varepsilon\bigr) = q
= \frac{V(X)}{\varepsilon^2} :
\]
gelijkheid in Chebyshev. De ongelijkheid kan dus niet worden
verbeterd met alleen de \omterm{def:b2:randomvar:variance}{variantie} --- het verval $1/\varepsilon^2$
is de exacte prijs van informatie over het tweede moment. Sneller
verval vereist sterkere hypothesen: begrensdheid van de
veranderlijke koopt \emph{exponentiële} concentratie, zoals
\cref{exo:b2:randomvar:7} voorproeft en de weekendopgave van dit
hoofdstuk stelselmatig ontwikkelt.
\end{example}

\begin{theorem}[Cauchy--Schwarz en Jensen]\label{thm:b2:randomvar:jensen}
\leavevmode
\begin{enumerate}
\item (Cauchy--Schwarz) Hebben $X, Y$ tweede momenten, dan heeft
$XY$ een \omterm{def:b2:randomvar:expectation}{verwachtingswaarde} en is $\E(XY)^2 \leq
\E(X^2)\,\E(Y^2)$; bijgevolg $\operatorname{Cov}(X,Y)^2 \leq
V(X)V(Y)$.
\item (Jensen) Is $\varphi \colon I \to \R$ \omterm{def:b2:affine:convex}{convex} op een interval
dat $X(\Omega)$ bevat, en hebben $X$ en $\varphi(X)$
\omterm{def:b2:randomvar:expectation}{verwachtingswaarden}, dan is
\[
\varphi\bigl(\E(X)\bigr) \leq \E\bigl(\varphi(X)\bigr) .
\]
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} \omterm{def:b2:series:summable}{Sommeerbaarheid} van $XY$: $\abs{XY} \leq \frac{X^2 +
Y^2}2$. De afbeelding $(X, Y) \mapsto \E(XY)$ is een positieve
\omterm{def:b2:quadratic:def}{symmetrische bilineaire vorm} op de ruimte van veranderlijken met
tweede momenten, dus is de abstracte ongelijkheid van
Cauchy--Schwarz uit \cref{ch:b2:quadratic} van toepassing
(positief \emph{semi}definiet volstaat voor de ongelijkheid). Haar
toepassen op de gecentreerde veranderlijken geeft de grens voor de
\omterm{def:b2:randomvar:variance}{covariantie}.

\emph{2.} Ten eerste ligt $m = \E(X)$ in $I$: $I$ is een interval
dat alle waarden van $X$ bevat, en de \omterm{def:b2:randomvar:expectation}{verwachtingswaarde} is
monotoon, dus ligt $m$ tussen $\inf X(\Omega)$ en $\sup
X(\Omega)$. Volgens de stelling over de steunrechte voor convexe
functies (\cref{ch:b2:realfun}) bestaan er $\alpha, \beta$ met
$\varphi(t) \geq \alpha t + \beta$ voor alle $t \in I$ en
$\varphi(m) = \alpha m + \beta$. Dan is puntsgewijs op $\Omega$
$\varphi(X) \geq \alpha X + \beta$; \omterm{def:b2:randomvar:expectation}{verwachtingswaarden} nemen geeft
\[
\E\bigl(\varphi(X)\bigr) \geq \alpha\,\E(X) + \beta
= \varphi\bigl(\E(X)\bigr). \qedhere
\]
\end{proof}

\begin{example}
Jensen met $\varphi(t) = t^2$ geeft $\E(X)^2 \leq \E(X^2)$ --- de
positiviteit van de \omterm{def:b2:randomvar:variance}{variantie}; met $\varphi(t) = 1/t$ op
$\intoo{0}{\infty}$: $\frac{1}{\E X} \leq \E\bigl(\frac1X\bigr)$
--- het harmonisch gemiddelde ligt onder het rekenkundig
gemiddelde, nu in toevallige gedaante.
\end{example}

\begin{remark}[Klassieke valkuilen]
(i) $\E(XY) = \E(X)\E(Y)$ \emph{vereist} \omterm{def:b2:proba:independence}{onafhankelijkheid} (of
minstens een nulcovariantie): met $Y = X$ krijgt men $\E(X^2) \neq
\E(X)^2$ zodra $V(X) > 0$. (ii) Evenzo is $V(X + X) = 4V(X)$, niet
$2V(X)$: \omterm{def:b2:randomvar:variance}{varianties} tellen alleen op over \omterm{def:b2:proba:independence}{onafhankelijke} (of
ongecorreleerde) termen. (iii) $\E(f(X))$ is niet $f(\E(X))$; voor
\omterm{def:b2:affine:convex}{convexe} $f$ vertelt Jensen zelfs de richting van de fout, als in
\cref{ex:b2:randomvar:transferex}. (iv) Het bestaan is een echte
hypothese: voor de Sint-Petersburgveranderlijke $X = 2^K$ met $\P(K
= k) = 2^{-k}$ ($k \geq 1$) is
\[
\sum_{k\geq1}2^k\cdot2^{-k} = \sum_{k\geq1}1 = \infty :
\]
$X$ is bijna zeker eindig en heeft toch geen \omterm{def:b2:randomvar:expectation}{verwachtingswaarde}, en
er bestaat geen eerlijke inzet voor het spel. De \omterm{def:b2:series:summable}{sommeerbaarheid} in
de definitie van $\E$ is geen boekhoudkundige muggenzifterij ---
het is waar zware staarten worden opgespoord. (v) Ten slotte heeft
de overdrachtsstelling \emph{absolute} \omterm{def:b2:series:summable}{sommeerbaarheid} nodig
voordat enige herschikking van de som over de waarden geoorloofd is
(\cref{ch:b2:series}).
\end{remark}


\begin{example}[Chebyshev op honderd worpen]\label{ex:b2:randomvar:hundredtosses}
Voor $X \sim \mathcal B(100, \frac12)$: $\E X = 50$, $V(X) = 25$.
Chebyshev met $\varepsilon = 6$:
\[
\P(45 \leq X \leq 55) = \P(\abs{X - 50} < 6)
\geq 1 - \frac{25}{36} \approx 0.31 ,
\]
terwijl de exacte binomiale som $\approx 0.73$ geeft. De
gewaarborgde $31\%$ ligt ver van de waarheid, maar zij vergde
\emph{alleen} het gemiddelde en de \omterm{def:b2:randomvar:variance}{variantie} --- hetzelfde
certificaat geldt woordelijk voor elke veranderlijke met $\E = 50$
en $V = 25$, hoe exotisch ook, en
\cref{ex:b2:randomvar:chebsharp} toont dat een zekere zulke
veranderlijke haar verzadigt. Universaliteit heeft een prijs; is de
\omterm{def:b2:randomvar:law}{verdeling} werkelijk binomiaal, dan dichten de exponentiële
gereedschappen van de weekendopgave het grootste deel van het gat.
\end{example}

\begin{example}[De correlatie van een deel met zijn
geheel]\label{ex:b2:randomvar:partwhole}
Hoe sterk is voor \omterm{def:b2:proba:independence}{onafhankelijke}, identiek verdeelde $X, Y$ met
\omterm{def:b2:randomvar:variance}{variantie} $\sigma^2 > 0$ één term gecorreleerd met de som $S = X +
Y$? Bereken
\[
\operatorname{Cov}(X, S) = \operatorname{Cov}(X, X) +
\operatorname{Cov}(X, Y) = \sigma^2 + 0 = \sigma^2,
\qquad V(S) = 2\sigma^2,
\]
zodat de correlatiecoëfficiënt
\[
\rho(X, S) = \frac{\operatorname{Cov}(X,
S)}{\sigma(X)\,\sigma(S)}
= \frac{\sigma^2}{\sigma\cdot\sigma\sqrt2}
= \frac{1}{\sqrt2} \approx 0.707
\]
is, wat de gemeenschappelijke \omterm{def:b2:randomvar:law}{verdeling} ook is --- dobbelstenen,
munten, poisson-tellingen. Met $n$ termen geeft dezelfde berekening
$\rho(X_1, S_n) = 1/\sqrt n$: de invloed van elke afzonderlijke
term op het totaal verdunt als een vierkantswortel, wat de schaduw
in de correlatie is van de schaal $\sqrt n$ van de fluctuaties.
Cauchy--Schwarz waarborgt altijd $\abs\rho \leq 1$; hier wordt de
grens precies bereikt in het ontaarde geval $n = 1$ en vervalt zij
daarna voorspelbaar.
\end{example}

\begin{example}[Gewogen ongelijkheid tussen rekenkundig en
meetkundig gemiddelde uit Jensen]\label{ex:b2:randomvar:amgm}
Laat $Y$ de positieve waarden $a_1, \dots, a_k$ aannemen met kansen
$\lambda_1, \dots, \lambda_k$. De functie $-\ln$ is \omterm{def:b2:affine:convex}{convex} op
$\intoo0\infty$, dus geeft Jensen dat $-\ln\E(Y) \leq \E(-\ln Y)$,
dat wil zeggen
\[
a_1^{\lambda_1}a_2^{\lambda_2}\cdots a_k^{\lambda_k}
\;\leq\; \lambda_1a_1 + \lambda_2a_2 + \dots + \lambda_ka_k :
\]
de gewogen ongelijkheid tussen het rekenkundig en het meetkundig
gemiddelde, met gelijkheid dan en slechts dan als $Y$ constant is.
Gelijke gewichten $\lambda_i = \frac1k$ geven de klassieke versie
terug. De kansrekening heeft stilletjes een zuiver algebraïsche
stelling bewezen: een \omterm{def:b2:randomvar:law}{kansverdeling} kiezen is slechts een
boekhoudkundig hulpmiddel voor \omterm{def:b2:affine:convex}{convexe} combinaties --- opnieuw het
barycentrische standpunt van \cref{ch:b2:affine}, nu met Jensen als
motor.
\end{example}

\section{De zwakke wet van de grote aantallen}

\begin{theorem}[Zwakke wet van de grote aantallen]\label{thm:b2:randomvar:lln}
Zij $(X_k)_{k \geq 1}$ een rij paarsgewijs \omterm{def:b2:proba:independence}{onafhankelijke}
\omterm{def:b2:randomvar:law}{toevalsveranderlijken} met dezelfde \omterm{def:b2:randomvar:law}{verdeling}, die een tweede moment
toelaat; schrijf $m = \E(X_1)$ en $S_n = X_1 + \dots + X_n$. Dan is
voor elke $\varepsilon > 0$
\[
\P\Bigl(\,\Bigl|\frac{S_n}{n} - m\Bigr| \geq \varepsilon\Bigr)
\;\leq\; \frac{V(X_1)}{n\,\varepsilon^2}
\xrightarrow[n \to \infty]{} 0 .
\]
\end{theorem}

\begin{proof}
Wegens de lineariteit is $\E(S_n/n) = m$; volgens
\cref{thm:b2:randomvar:variancerules} (de paarsgewijze
\omterm{def:b2:proba:independence}{onafhankelijkheid} doodt de \omterm{def:b2:randomvar:variance}{covarianties}) is $V(S_n) = n\,V(X_1)$,
dus $V(S_n/n) = V(X_1)/n$. De ongelijkheid van Chebyshev toegepast
op $S_n/n$ geeft de grens.
\end{proof}

\begin{remark}
Dit is de stelling die de kans met de frequentie verbindt: voor
$X_k$ de indicator van een \omterm{def:b2:proba:space}{gebeurtenis} $A$ in \omterm{def:b2:proba:independence}{onafhankelijke}
herhalingen is $S_n/n$ de waargenomen frequentie van $A$, en de wet
van de grote aantallen zegt dat zij zich rond $\P(A)$ concentreert
met tempo $\frac{p(1-p)}{n\varepsilon^2}$. De \emph{sterke} wet
($S_n/n \to m$ bijna zeker) is een stelling voor bachelorjaar 3 ---
haar bewijs bij vierde momenten is nochtans binnen bereik: zie
\cref{exo:b2:randomvar:9}, dat Borel--Cantelli op de grens van het
type Chebyshev laat lopen. Dezelfde schatting van Chebyshev dreef
het bewijs met de veeltermen van Bernstein van de
benaderingsstelling van Weierstrass in \cref{ch:b2:funcseq} aan ---
de telhulpstelling daar \emph{was} de zwakke wet van de grote
aantallen in vermomming.
\end{remark}

\begin{example}[Vijftig plaatjes verzamelen]
\label{ex:b2:randomvar:couponnumeric}
De plaatjesverzamelaar van \cref{exo:b2:randomvar:3} met $n = 50$
verschillende speeltjes: het verwachte totaal is
\[
\E(T_{50}) = 50\,H_{50} = 50\sum_{k=1}^{50}\frac1k
\approx 50 \times 4.499 \approx 225
\]
dozen --- viereneenhalf maal de naïeve schatting $50$. De
harmonische groei is het hele verhaal: de eerste $25$ speeltjes
komen in ongeveer $50\ln2 \approx 35$ dozen binnen, terwijl het
\emph{laatste} speeltje alleen al gemiddeld $50$ dozen kost (een
meetkundige wachttijd met parameter $\frac1{50}$).
Voltooiingsproblemen worden door hun eindspel beheerst, en daarom
vindt \cref{exo:b2:randomvar:12} fluctuaties van de orde $n$ --- de
grootte van die laatste meetkundige wachttijd --- rond het
gemiddelde $n\ln n$.
\end{example}

\begin{example}[Hoe groot moet $n$ zijn?]\label{ex:b2:randomvar:llnnumeric}
Om de waargenomen frequentie tot op $\varepsilon = 0.01$ van
$\P(A)$ vast te pinnen met een betrouwbaarheid van $95\%$, eist de
grens van Chebyshev
\[
\frac{p(1-p)}{n\varepsilon^2} \leq \frac{1}{4n\varepsilon^2}
\leq 0.05,
\qquad\text{dat wil zeggen}\qquad
n \geq \frac{1}{4\cdot0.05\cdot(0.01)^2} = 50\,000 .
\]
De afhankelijkheid is bruut in $\varepsilon$ (kwadratisch) en mild
in de betrouwbaarheid (lineair in $1/\alpha$). Beide kenmerken zijn
eigenschappen van de \emph{grens}, niet van de waarheid: de
exponentiële ongelijkheden van de weekendopgave verlagen de prijs
van betrouwbaarheid van $1/\alpha$ tot $\ln(1/\alpha)$ ---
dezelfde specificatie zal daar ongeveer $18\,500$ waarnemingen
kosten --- terwijl de schaal $1/\varepsilon^2$ echt en
onverbeterlijk is. Weten welk deel van een grens los zit, is even
nuttig als de grens zelf.
\end{example}

\begin{remark}[Vooruitblik binnen dit volume]
Vooruit voedt alles hier \cref{ch:b2:genfun}: de
\omterm{def:b2:randomvar:expectation}{verwachtingswaarde} $\E(t^X)$ van één listige functie van $X$ pakt
de hele \omterm{def:b2:randomvar:law}{verdeling} in een machtreeks, momenten worden afgeleiden in
$1$, en identiteiten van het type Wald voor toevallige sommen
dragen de theorie van de vertakkingsprocessen; de productstelling
voor \omterm{def:b2:proba:independence}{onafhankelijke} veranderlijken wordt de multiplicativiteit van

exponentiële moment uit de weekendopgave van dit hoofdstuk Markov
toegepast op $\eu^{tX}$ --- één ongelijkheid, verscherpt met één
goede verandering van veranderlijke, over drie hoofdstukken heen.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:1}
Bereken $\E(X)$ en $V(X)$ voor $X \sim \mathcal{B}(n, p)$ (via
indicatoren), $X \sim \mathcal{P}(\lambda)$ (toon aan dat $V(X) =
\lambda$) en $X \sim \mathcal{G}(p)$ (toon aan dat $V(X) =
\frac{1-p}{p^2}$; gebruik $\E(X(X-1))$ en de tweede afgeleide van
de meetkundige reeks).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:2}
Zijn $X \sim \mathcal{P}(\lambda)$ en $Y \sim \mathcal{P}(\mu)$
\omterm{def:b2:proba:independence}{onafhankelijk}. Toon aan dat $X + Y \sim \mathcal{P}(\lambda + \mu)$
(convolutie van de gewichten; binomium), en dat de voorwaardelijke
\omterm{def:b2:randomvar:law}{verdeling} van $X$ gegeven $X + Y = n$ binomiaal
$\mathcal{B}\bigl(n, \frac{\lambda}{\lambda + \mu}\bigr)$ is.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:3}
(Plaatjesverzamelaar, \omterm{def:b2:randomvar:expectation}{verwachtingswaarde}) Een ontbijtgraanmerk
verstopt in elke doos uniform een van $n$ verschillende speeltjes.
Zij $T_n$ het aantal dozen dat nodig is om alle $n$ speeltjes te
verzamelen. Schrijf $T_n$ als een som van \omterm{def:b2:proba:independence}{onafhankelijke}
meetkundige veranderlijken (de tijd tot een \emph{nieuw} speeltje
wanneer er nog $k$ ontbreken) en toon aan dat
\[
\E(T_n) = n\sum_{k=1}^{n}\frac{1}{k} \sim n\ln n
\]
(equivalent volgens de vergelijking van reeks en integraal uit
\cref{ch:b2:comparison}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:4}
Zij $X \geq 0$ met gehele waarden. Bewijs de \emph{staartformule}
\[
\E(X) = \sum_{n=1}^{\infty} \P(X \geq n)
\]
(wanneer een van beide leden eindig is), door $X =
\sum_{n\geq1}\mathbf{1}_{X \geq n}$ te schrijven en de sommaties te
verwisselen (Fubini voor niet-negatieve families). Vind $\E(X) =
\frac1p$ voor de meetkundige \omterm{def:b2:randomvar:law}{verdeling} terug.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:5}
(Steekproeven zonder teruglegging zijn sterker geconcentreerd) Een
urne bevat $N$ ballen, waarvan $M$ wit. Trek er $n \leq N$ zonder
teruglegging en laat $X$ de witte tellen (\emph{hypergeometrische}
\omterm{def:b2:randomvar:law}{verdeling}). Toon met indicatoren $X = \sum_{i=1}^n Y_i$, waarbij
$Y_i$ de $i$-de trekking is, aan dat elke $Y_i$ bernoulli met
parameter $p = M/N$ is (symmetrie!), besluit dat $\E(X) = np$
precies als met teruglegging, en toon aan dat
$\operatorname{Cov}(Y_i, Y_j) = -\frac{p(1-p)}{N-1} < 0$ voor $i
\neq j$, zodat $V(X) = np(1-p)\frac{N - n}{N - 1} \leq np(1-p)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:6}
Zij $X$ met een tweede moment. Toon aan dat $c \mapsto \E\bigl((X -
c)^2\bigr)$ precies in $c = \E(X)$ minimaal is, met minimum $V(X)$.
Toon vervolgens aan dat $\P(X = \E(X)) = 1$ dan en slechts dan als
$V(X) = 0$. \emph{(Voor het tweede punt: is $V(X) = 0$, gebruik dan
Chebyshev met $\varepsilon = 1/n$ en de monotone \omterm{def:b2:metric:continuity}{continuïteit},
\cref{thm:b2:proba:continuity}.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:7}
(Concentratie verslaat Markov) Zij $S_n \sim \mathcal{B}(n,
\frac12)$ (het aantal keer kop in $n$ eerlijke worpen). Vergelijk
de grenzen gegeven door Markov ($\P(S_n \geq \frac{3n}{4})$), door
Chebyshev, en door de exponentiële methode (Chernoff):
\[
\P\Bigl(S_n \geq \frac{3n}4\Bigr)
\leq \E\bigl(e^{tS_n}\bigr)e^{-3nt/4}
= \Bigl(\frac{1 + e^t}{2}\Bigr)^n e^{-3nt/4}
\quad (t > 0),
\]
en optimaliseer $t$ om een exponentieel kleine grens te krijgen.
\emph{(In $t = \ln 3$: grens $\bigl(2\cdot 3^{-3/4}\bigr)^n \approx
(0.877)^n$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:8}
(Weierstrass opnieuw, kansrekenkundig) Zij $f \colon [0,1] \to \R$
\omterm{def:b2:metric:continuity}{continu} en $S_n \sim \mathcal{B}(n, x)$. Toon aan dat de veelterm
van Bernstein $B_nf(x) = \sum_{k=0}^n f\bigl(\frac kn\bigr)\binom
nk x^k(1-x)^{n-k}$ gelijk is aan
$\E\bigl[f\bigl(\frac{S_n}{n}\bigr)\bigr]$, en leid de schatting
$\abs{B_nf(x) - f(x)} \leq \omega_f(\delta) + \frac{2\norm
f_\infty}{4n\delta^2}$ uit \cref{ch:b2:funcseq} opnieuw af in deze
kansrekenkundige taal (splits op $\bigl|\frac{S_n}{n} - x\bigr|
\geq \delta$ en gebruik Chebyshev).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:9}
(Sterke wet bij vierde momenten) Zij $(X_k)$ \omterm{def:b2:proba:independence}{onafhankelijk},
identiek verdeeld en gecentreerd ($\E X_1 = 0$), met $\E(X_1^4) <
\infty$. Toon, door $\E(S_n^4)$ uit te werken en de overlevende
termen te tellen (alleen termen $\E(X_i^4)$ en $\E(X_i^2X_j^2)$ met
$i \neq j$), aan dat $\E(S_n^4) \leq C n^2$ voor een constante $C$.
Leid af dat $\sum_n \P\bigl(\abs{S_n/n} \geq \varepsilon\bigr) <
\infty$ voor elke $\varepsilon > 0$ (Markov op orde 4) en besluit
met Borel--Cantelli (\cref{thm:b2:proba:borelcantelli}) dat $S_n/n
\to 0$ bijna zeker, in de volgende passende formulering: de
\omterm{def:b2:proba:space}{gebeurtenis} $\bigcap_{j}\bigcup_N\bigcap_{n \geq N}\{\abs{S_n/n} <
\frac1j\}$ heeft kans $1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:10}
Er wordt met twee eerlijke dobbelstenen geworpen; zij $M$ de
grootste van de twee uitkomsten. Toon met de staartformule van
\cref{exo:b2:randomvar:4} (eindige versie) aan dat
\[
\E(M) = \sum_{k=1}^{6}\P(M \geq k)
= 6 - \sum_{j=0}^5\Bigl(\frac j6\Bigr)^2 = \frac{161}{36}
\approx 4.47 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:11}
Zij $F_n$ het aantal vaste punten van een uniform willekeurige
permutatie van $\{1, \dots, n\}$ ($n \geq 2$). Bereken, door $F_n =
\sum_i\mathbf 1_{\sigma(i) = i}$ te schrijven, dat $\E(F_n) = 1$ en
$\operatorname{Cov}(\mathbf 1_{\sigma(i)=i}, \mathbf
1_{\sigma(j)=j}) = \frac1{n^2(n-1)}$ voor $i \neq j$, en besluit
dat $V(F_n) = 1$: gemiddeld blijft één letter vast, met \omterm{def:b2:randomvar:variance}{variantie}
precies $1$, wat $n$ ook is.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:12}
(Plaatjesverzamelaar, concentratie) Toon in de situatie van
\cref{exo:b2:randomvar:3} aan dat
\[
V(T_n) = \sum_{k=1}^n\frac{1 - k/n}{(k/n)^2}
\leq n^2\sum_{k=1}^n\frac{1}{k^2} \leq \frac{\pi^2}{6}n^2,
\]
met de \omterm{def:b2:proba:independence}{onafhankelijkheid} van de meetkundige fasen en $V(\mathcal
G(p)) = \frac{1-p}{p^2}$ (\cref{exo:b2:randomvar:1}; de waarde
$\pi^2/6$ is \cref{ex:b2:fourier:basel}). Leid met Chebyshev af dat
$\dfrac{T_n}{n\ln n} \to 1$ \emph{in kans}: de totale tijd van de
verzamelaar is $n\ln n$, op fluctuaties van de orde $n$ na.
\end{exercise}

\section{Probleem: de gereedschapskist van de concentratie, van
Markov tot Hoeffding}

\begin{problem}[{Weekendopgave --- exponentiële concentratie met
de hand, en hoeveel mensen een peiling moet bevragen}]
\label{pb:b2:randomvar:1}
De ongelijkheid van Markov kost één moment en koopt een verval
$1/a$; Chebyshev kost twee momenten en koopt $1/\varepsilon^2$ ---
en \cref{ex:b2:randomvar:chebsharp} toont dat dit alles is wat die
momenten kunnen kopen. Deze opgave beklimt de rest van de ladder:
de exponentiële methode (Chernoff) met haar \emph{exacte} tempo
voor muntworpen, de ongelijkheid van Hoeffding voor alle begrensde
veranderlijken, en de opbrengst --- expliciete, eerlijke
steekproefomvangen voor peilingen, verkiezingsuitspraken en het
testen van munten.\index{concentratieongelijkheden}\index{ongelijkheid
van Hoeffding}\index{grens van Chernoff} Overal is $S_n \sim
\mathcal B(n, p)$ een som van $n$ \omterm{def:b2:proba:independence}{onafhankelijke}
bernoulli-veranderlijken en $\widehat p_n = S_n/n$ de empirische
frequentie.

\medskip
\noindent\textbf{Deel I --- IJken op de eerlijke munt.} Hier is $p
= \frac12$ en $a \in \intoo{\frac12}{1}$.
\begin{enumerate}
  \item Markov op niveau $an$: toon aan dat $\P(S_n \geq an) \leq
        \frac1{2a}$, een grens die niet eens naar $0$ streeft. Waar
        verliest Markov zo veel?
  \item Chebyshev: toon met de symmetrie van de eerlijke binomiale
        \omterm{def:b2:randomvar:law}{verdeling} om $n/2$ aan dat
        \[
        \P(S_n \geq an) = \tfrac12\,
        \P\bigl(\abs{S_n - \tfrac n2} \geq n(a -
        \tfrac12)\bigr)
        \leq \frac{1}{8n(a - 1/2)^2},
        \]
        dat wil zeggen $\frac2n$ in $a = \frac34$: eindelijk
        polynomiaal verval.
  \item (Chernoff, algemeen niveau) Bereken $\E(\eu^{tS_n}) =
        \bigl(\frac{1 + \eu^t}2\bigr)^n$ en optimaliseer $\P(S_n
        \geq an) \leq \E(\eu^{tS_n})\eu^{-tan}$ over $t > 0$: toon
        aan dat de optimale $t$ gelijk is aan
        $\ln\frac{a}{1-a}$ en dat
        \[
        \P(S_n \geq an) \leq \eu^{-n\,I(a)},
        \qquad
        I(a) = \ln 2 + a\ln a + (1-a)\ln(1-a) > 0 .
        \]
        Ga na dat $a = \frac34$ de grens $\bigl(2\cdot3^{-3/4}\bigr)^n$
        van \cref{exo:b2:randomvar:7} teruggeeft.
  \item (De exponent is exact) Zij $k = an$ een geheel getal.
        Bewijs uit het feit dat $\binom nk a^k(1-a)^{n-k}$ de
        grootste van de $n + 1$ termen van een \omterm{def:b2:randomvar:law}{kansverdeling} is,
        dat $\binom nk \geq \frac{\eu^{nH(a)}}{n+1}$ met $H(a) =
        -a\ln a - (1-a)\ln(1-a)$, en leid de bijpassende ondergrens
        af:
        \[
        \P(S_n \geq an) \geq \binom{n}{an}2^{-n}
        \geq \frac{\eu^{-n\,I(a)}}{n + 1} .
        \]
  \item Zet de drie grenzen in een tabel voor $n = 100$, $a =
        \frac34$: Markov $\frac23$, Chebyshev $0.02$, Chernoff
        $\approx 2.1\cdot10^{-6}$ (de werkelijke waarde is $\approx
        2.8\cdot10^{-7}$). De moraal, in één zin?
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De ongelijkheid van Hoeffding.}
\begin{enumerate}[resume]
  \item (Geval van Rademacher) Bewijs voor $\varepsilon = \pm1$ met
        elk kans $\frac12$ dat
        \[
        \E(\eu^{t\varepsilon}) = \cosh t \leq \eu^{t^2/2}
        \qquad (t \in \R)
        \]
        door de twee reeksen term voor term te vergelijken ($(2k)!
        \geq 2^kk!$).
  \item Leid af, voor \omterm{def:b2:proba:independence}{onafhankelijke} rademacher-veranderlijken
        $\varepsilon_1, \dots, \varepsilon_n$ en elke $s > 0$:
        \[
        \P\Bigl(\sum_{i=1}^n\varepsilon_i \geq s\Bigr)
        \leq \eu^{-s^2/(2n)} .
        \]
  \item Vertaal naar eerlijke munten ($X_i =
        \frac{1+\varepsilon_i}2$): $\P\bigl(\widehat p_n -
        \tfrac12 \geq \delta\bigr) \leq \eu^{-2n\delta^2}$, en de
        tweezijdige versie met een factor $2$.
  \item (Hulpstelling van Hoeffding) Zij $X \in \intcc01$ met $\E X
        = p$, en $\psi(t) = \ln\E(\eu^{tX})$. Verantwoord dat
        $\psi$ tweemaal \omterm{def:b2:diffcalc:differential}{differentieerbaar} is met
        \[
        \psi''(t) = \E_t(X^2) - \E_t(X)^2, \qquad
        \E_t(Y) := \frac{\E(Y\eu^{tX})}{\E(\eu^{tX})},
        \]
        een \emph{\omterm{def:b2:randomvar:variance}{variantie}} van een herwogen veranderlijke die nog
        altijd waarden in $\intcc01$ aanneemt; begrens haar door
        $\frac14$ (het minimaliteitsargument van
        \cref{exo:b2:randomvar:6}) en besluit met Taylor:
        \[
        \E\bigl(\eu^{t(X - p)}\bigr) \leq \eu^{t^2/8} .
        \]
  \item (Ongelijkheid van Hoeffding) Leid voor \omterm{def:b2:proba:independence}{onafhankelijke} $X_i
        \in \intcc01$ met gemeenschappelijk gemiddelde $p$ af dat
        \[
        \P\bigl(\abs{\widehat p_n - p} \geq \delta\bigr)
        \leq 2\,\eu^{-2n\delta^2}
        \qquad (\delta > 0).
        \]
  \item Vergelijk het tempo $\frac{p(1-p)}{n\delta^2}$ van
        Chebyshev met $2\eu^{-2n\delta^2}$ van Hoeffding: welke
        hypothese vereist elk, en vanaf welke $n$ (ruwweg) wint de
        exponentiële grens bij $\delta = 0.03$, $p = \frac12$?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Hoeveel mensen moet een peiling
bevragen?} Een peiling bevraagt $n$ \omterm{def:b2:proba:independence}{onafhankelijke}, uniform gekozen
kiezers; elk antwoordt eerlijk; $p$ is de werkelijke score en
$\widehat p_n$ het cijfer van de peiling.
\begin{enumerate}[resume]
  \item Toon aan dat de peiling tot op $\pm\delta$ nauwkeurig is
        met betrouwbaarheid $1 - \alpha$ (dat wil zeggen
        $\P(\abs{\widehat p_n - p} \geq \delta) \leq \alpha$) zodra
        \[
        n \;\geq\; \frac{\ln(2/\alpha)}{2\,\delta^2} .
        \]
  \item Bereken de vereiste $n$ voor de gebruikelijke specificatie
        ``drie punten, vijfennegentig procent'' ($\delta = 0.03$,
        $\alpha = 0.05$): $n \geq 2050$; en voor één punt: $n \geq
        18\,445$. Merk het opvallende feit op --- en verklaar het
        --- dat het antwoord de omvang van de bevolking niet
        bevat.
  \item Doe vraag 13 opnieuw met Chebyshev ($V(X_1) = p(1-p) \leq
        \frac14$): $n \geq \frac1{4\alpha\delta^2} = 5556$ bij drie
        punten. Merk op dat steekproeven \emph{zonder} teruglegging
        alleen maar helpen (\cref{exo:b2:randomvar:5}: de \omterm{def:b2:randomvar:variance}{variantie}
        krimpt met $\frac{N-n}{N-1}$).
  \item (Een verkiezing uitroepen) De werkelijke score van een
        kandidaat is $p = 0.52$. Hoeveel kiezers moeten worden
        bevraagd opdat $\P(\widehat p_n \leq \tfrac12) \leq 0.01$?
        Toon aan dat $n \geq \frac{\ln 100}{2\cdot(0.02)^2} \approx
        5757$ --- een nek-aan-nekrace uitroepen kost veel meer dan
        een score schatten.
  \item Wat de wiskunde \emph{niet} dekt: som de aannames van het
        model op (\omterm{def:b2:proba:independence}{onafhankelijke} uniforme steekproeven, eerlijke
        antwoorden, vaste $p$), en leg in een korte alinea uit
        waarom echte peilingsfouten door \emph{vertekening} worden
        beheerst (niet-uniforme steekproeven, weigeringen), die
        geen enkele verhoging van $n$ verkleint.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Scherper en goedkoper.}
\begin{enumerate}[resume]
  \item (Mediaan van gemiddelden: exponentieel verval uit twee
        momenten) Verdeel een budget van $km$ waarnemingen over $k$
        \omterm{def:b2:proba:independence}{onafhankelijke} groepen van $m$; zijn $\widehat p^{(1)},
        \dots, \widehat p^{(k)}$ de groepsgemiddelden en $M$ hun
        mediaan. Kies $m$ zo dat elke groep voldoet aan
        $\P(\abs{\widehat p^{(i)} - p} \geq \delta) \leq \frac18$
        (Chebyshev: $m \geq \frac2{\delta^2}$ volstaat). Toon aan
        dat als $\abs{M - p} \geq \delta$, dan minstens $k/2$
        groepen fout zitten, en leid af dat
        \[
        \P(\abs{M - p} \geq \delta)
        \leq \binom{k}{\lceil k/2\rceil}\Bigl(\frac18
        \Bigr)^{k/2}
        \leq 2^k\cdot 8^{-k/2} = 2^{-k/2} :
        \]
        exponentiële concentratie met niets meer dan \omterm{def:b2:randomvar:variance}{varianties}.
  \item (Paley--Zygmund) Bewijs voor $X \geq 0$ met een tweede
        moment dat $\P(X > 0) \geq \dfrac{\E(X)^2}{\E(X^2)}$
        \emph{(Cauchy--Schwarz op $X\mathbf 1_{X>0}$)}: het
        gereedschap in de omgekeerde richting --- momenten kunnen
        \omterm{def:b2:proba:space}{gebeurtenissen} ook \emph{afdwingen}.
  \item (Pinsker, lichte versie) Toon aan dat $I(a) \geq 2\bigl(a -
        \tfrac12\bigr)^2$ op $\intoo{\frac12}1$ \emph{(het verschil
        verdwijnt tot op tweede orde in $\frac12$ en zijn tweede
        afgeleide is $\frac1{a(1-a)} - 4 \geq 0$)}: de exacte
        exponent van Chernoff verslaat altijd de kwadratische van
        Hoeffding.
  \item Ontwikkel $I\bigl(\tfrac12 + \delta\bigr) = 2\delta^2 +
        O(\delta^4)$ en combineer met vraag 4: voor kleine
        afwijkingen is de exponent $2n\delta^2$ van Hoeffding
        asymptotisch \emph{exact} --- geen enkele methode kan hem
        met meer dan polynomiale factoren verslaan.
  \item Stel de tabel van de gereedschapskist op: geef voor Markov,
        Chebyshev, de grens met het vierde moment van
        \cref{exo:b2:randomvar:9}, Hoeffding en Chernoff met
        exponent $I$ telkens in één regel: de vereiste hypothese,
        het verkregen verval, en de vraag in deze opgave waar zij
        het scherpst was.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Dividenden.}
\begin{enumerate}[resume]
  \item (Een munt testen) Een munt is ofwel eerlijk ofwel scheef
        met $p = 0.55$. Je werpt haar $n$ maal en verklaart haar
        ``scheef'' wanneer $\widehat p_n > 0.525$. Toon aan dat
        beide foutkansen hoogstens $\eu^{-2n(0.025)^2}$ zijn, en
        dat $n \geq 3685$ worpen beide onder $1\%$ waarborgen.
  \item (Zeldzame \omterm{def:b2:proba:space}{gebeurtenissen} hebben een variantiebewuste grens
        nodig) Zij $p = 0.01$ en neem de relatieve specificatie
        $\delta = p/2 = 0.005$, $\alpha = 0.05$. Vergelijk de
        steekproefomvangen die Hoeffding ($n \approx 74\,000$) en
        Chebyshev met de echte \omterm{def:b2:randomvar:variance}{variantie} $p(1-p)$ ($n \approx
        7920$) eisen: de variantieblinde exponentiële grens verliest
        van het bescheiden tweede moment. Formuleer de moraal, en
        waar het ontbrekende gereedschap (een variantiebewuste
        exponentiële grens; de poisson-benadering van
        \cref{ch:b2:genfun}) vandaan zal komen.
  \item (Sterke wet voor munten) Bewijs uit $\sum_n
        2\eu^{-2n\delta^2} < \infty$ en Borel--Cantelli
        (\cref{thm:b2:proba:borelcantelli}) dat $\widehat p_n \to
        p$ bijna zeker voor \omterm{def:b2:proba:independence}{onafhankelijke} muntworpen: formuleer de
        bijna-zekere \omterm{def:b2:proba:space}{gebeurtenis} als
        $\bigcap_j\bigcup_N\bigcap_{n\geq N} \{\abs{\widehat p_n -
        p} < \tfrac1j\}$ zoals in \cref{exo:b2:randomvar:9}, en
        besluit. (De begrensdheid vervangt het vierde moment dat
        daar werd gebruikt.)
  \item Synthese. In vijf zinnen: wat elke sport van de ladder
        (momenten één, twee, vier; begrensd exponentieel; exacte
        exponent) kost en koopt; waarom $2050$ mensen bevragen
        volstaat voor een land van welke omvang ook; en welke van
        deze grenzen het volume van bachelorjaar 3 tot de exacte
        constanten van de centrale limietstelling zal aanscherpen.
\end{enumerate}
\end{problem}
