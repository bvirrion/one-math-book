\chapter{Kwadratische vormen}\label{ch:b2:quadratic}

Een \omterm{def:b2:quadratic:def}{kwadratische vorm} is de algebraïsche schaduw van een meetkunde:
de delen met signatuur nul maken vlak, de positieve delen krommen
de ene kant op en de negatieve de andere. Dit hoofdstuk herleidt
elke reële \omterm{def:b2:quadratic:def}{kwadratische vorm} tot een som van $\pm$-kwadraten
(Gauss), bewijst dat de aantallen tekens intrinsiek zijn
(Sylvester), en bekroont daarna de euclidische meetkunde met de
\emph{spectraalstelling}: \omterm{def:b2:quadratic:adjoint}{symmetrische} endomorfismen
diagonaliseren in orthonormale basissen --- de meest gebruikte
stelling van de toegepaste lineaire algebra.

\section{Bilineaire en kwadratische vormen}

\begin{definition}\label{def:b2:quadratic:def}
Een \emph{\omterm{def:b2:quadratic:adjoint}{symmetrische} bilineaire vorm} op een reële vectorruimte
$E$ is een bilineaire $\varphi \colon E \times E \to \R$ met
$\varphi(x, y) = \varphi(y, x)$; de bijbehorende \emph{kwadratische
vorm}\index{kwadratische vorm} is $q(x) = \varphi(x, x)$. De vorm
$\varphi$ wordt uit $q$ teruggewonnen door
\emph{polarisatie}\index{polarisatie-identiteit}:
\[
\varphi(x, y) = \tfrac12\bigl(q(x + y) - q(x) - q(y)\bigr).
\]
In een basis $(e_i)$ is de \emph{matrix} van $\varphi$ de
\omterm{def:b2:quadratic:adjoint}{symmetrische} $B = (\varphi(e_i, e_j))$, met $q(x) = X^{\mathsf T} B
X$; een basisovergang met matrix $P$ vervangt $B$ door $P^{\mathsf
T} B P$ (\emph{congruentie}\index{congruente matrices} --- geen
gelijkvormigheid!). De \emph{rang} van $q$ is $\operatorname{rk}
B$ (invariant: congruentie vermenigvuldigt met inverteerbare
matrices).
\end{definition}

\begin{example}
Op $\R^2$ heeft $q(x, y) = x^2 + 4xy + y^2$ de matrix
$\begin{pmatrix} 1 & 2\\ 2 & 1\end{pmatrix}$. Een inproduct is
precies een \omterm{def:b2:quadratic:def}{symmetrische bilineaire vorm} waarvan de \omterm{def:b2:quadratic:def}{kwadratische
vorm} positief definiet is; dit hoofdstuk bestudeert het algemene
geval, met onbepaald teken.
\end{example}

\begin{example}[Congruentie in actie]\label{ex:b2:quadratic:congruence}
Neem $q(x, y) = x^2 + 4xy + y^2$ (matrix $B = \begin{pmatrix} 1 &
2\\ 2 & 1\end{pmatrix}$) en de nieuwe basis $e_1' = (1, 1)$, $e_2'
= (1, -1)$, dat wil zeggen $P = \begin{pmatrix} 1 & 1\\ 1 &
-1\end{pmatrix}$. Dan is
\[
P^{\mathsf T}BP
= \begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
\begin{pmatrix} 1 & 2\\ 2 & 1\end{pmatrix}
\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
= \begin{pmatrix} 6 & 0\\ 0 & -2\end{pmatrix} :
\]
in de coördinaten $(u, v)$ langs de nieuwe basis is $q = 6u^2 -
2v^2$ --- controle: $x = u + v$ en $y = u - v$ geven $x^2 + 4xy +
y^2 = 6u^2 - 2v^2$ rechtstreeks. Merk op dat de nieuwe
diagonaalelementen $6, -2$ \emph{niet} de \omterm{def:b2:reduction:eigen}{eigenwaarden} $3, -1$ van
$B$ zijn: \omterm{def:b2:quadratic:def}{congruentie} herschaalt, en alleen gelijkvormigheid
bewaart spectra --- maar de tekens komen overeen, zoals de stelling
van Sylvester eist. (De basis is hier orthogonaal maar niet
orthonormaal; haar normaliseren met $\frac{1}{\sqrt2}$ zou de
diagonaal door $2$ delen en de \omterm{def:b2:reduction:eigen}{eigenwaarden} teruggeven.)
\end{example}

\begin{example}[Gram-determinanten meten oppervlakte]\label{ex:b2:quadratic:gramarea}
Voor $v_1, v_2$ in een euclidische ruimte verpakt de
Gram-matrix $G = \bigl(\langle v_i, v_j\rangle\bigr)$ de lengten
en de hoek; haar \omterm{def:b2:linalg:det}{determinant} verpakt de \emph{oppervlakte}:
\[
\det G = \norm{v_1}^2\norm{v_2}^2 - \langle v_1, v_2\rangle^2
= \norm{v_1}^2\norm{v_2}^2\bigl(1 - \cos^2\theta\bigr)
= \bigl(\norm{v_1}\,\norm{v_2}\sin\theta\bigr)^2 ,
\]
het kwadraat van de oppervlakte van het parallellogram op $v_1,
v_2$ --- en Cauchy--Schwarz is precies de uitspraak $\det G \geq
0$. Uitgewerkt voorbeeld: $v_1 = (1, 2, 2)$ en $v_2 = (2, 1, -2)$
in $\R^3$:
\[
G = \begin{pmatrix} 9 & 0\\ 0 & 9 \end{pmatrix},
\qquad
\det G = 81 :
\]
de vectoren zijn orthogonaal van lengte $3$ en spannen een
parallellogram (hier een vierkant) op met oppervlakte $\sqrt{81} =
9$. Het inzicht om te onthouden: er werd geen kruisproduct en geen
magie van dimensie $3$ gebruikt --- $\sqrt{\det G}$ meet
$k$-dimensionaal volume in \emph{elke} dimensie, het startpunt van
Deel I van de weekendopgave en van de oppervlakte-integralen
verderop in dit volume.
\end{example}

\section{Gauss-reductie en de inertie van Sylvester}

\begin{theorem}[Gauss-reductie]\label{thm:b2:quadratic:gauss}
Elke \omterm{def:b2:quadratic:def}{kwadratische vorm} $q$ op een eindigdimensionale reële ruimte
laat zich schrijven als\index{Gauss-reductie}
\[
q = \sum_{i=1}^{s} \ell_i^2 - \sum_{j=1}^{t} m_j^2 ,
\]
waarbij $\ell_1, \dots, \ell_s, m_1, \dots, m_t$ lineair
onafhankelijke lineaire vormen zijn; gelijkwaardig: in een zekere
basis is de matrix van $q$ diagonaal met elementen $+1$ ($s$ keer),
$-1$ ($t$ keer) en $0$.
\end{theorem}

\begin{proof}
Inductie naar het aantal veranderlijken, in coördinaten: $q(x_1,
\dots, x_n)$.

\emph{Geval 1: er komt een kwadraat voor}, zeg de coëfficiënt $a$
van $x_1^2$ is ongelijk aan nul. Groepeer alle termen met $x_1$ en
maak het kwadraat af:
\[
q = a\Bigl(x_1 + \frac{1}{a}\,\lambda(x_2, \dots,
x_n)\Bigr)^{\!2} + q_1(x_2, \dots, x_n),
\]
waarbij $\lambda$ lineair is en $q_1$ kwadratisch in de overige
veranderlijken: één onafhankelijke vorm is afgesplitst (zij bevat
$x_1$, de andere niet), de inductie is op $q_1$ van toepassing, en
de $\pm$-tekens komen uit het teken van $a$ na herschaling met
$\sqrt{\abs a}$.

\emph{Geval 2: geen kwadraat, maar wel een kruisterm}, zeg
$b\,x_1x_2$ met $b \neq 0$. Gebruik de identiteit
\[
x_1x_2 = \tfrac14\bigl((x_1 + x_2)^2 - (x_1 - x_2)^2\bigr)
\]
na groepering: met $q = b\,x_1x_2 + x_1\alpha + x_2\beta + q_2$
(waarbij $\alpha, \beta, q_2$ in de overige veranderlijken staan)
gaat men na dat
\[
q = \frac{b}{4}\Bigl[\Bigl(x_1 + x_2 + \frac{\alpha +
\beta}{b}\Bigr)^{2} - \Bigl(x_1 - x_2 + \frac{\beta -
\alpha}{b}\Bigr)^{2}\Bigr] + \widetilde q ,
\]
met $\widetilde q$ vrij van $x_1, x_2$: twee onafhankelijke vormen
zijn afgesplitst, en de inductie maakt het af.

Onafhankelijkheid van de verzamelde vormen: orden de partijen zoals
zij ontstonden. De vormen van de eerste partij bevatten $x_1$
(geval 1) of $x_1, x_2$ (geval 2); alle latere vormen zijn vrij van
die veranderlijken. Stel dat een lineaire combinatie van alle
verzamelde vormen verdwijnt. Lezen we de coëfficiënt van $x_1$ (en
$x_2$), dan draagt alleen de eerste partij bij, en binnen die
partij zijn de ene of twee vormen zichtbaar onafhankelijk ($\ell$
alleen; of $\ell \pm m$ met onafhankelijke $\ell, m$): de
coëfficiënten van de eerste partij verdwijnen dus. Haal die partij
weg en herhaal: met inductie langs de partijen verdwijnen alle
coëfficiënten --- de hele familie is vrij, met de driehoekigheid
expliciet gemaakt.
\end{proof}

\begin{theorem}[Traagheidswet van Sylvester]\label{thm:b2:quadratic:sylvester}
Het paar $(s, t)$ in \cref{thm:b2:quadratic:gauss} hangt alleen van
$q$ af en niet van de reductie: het is de
\emph{signatuur}\index{signatuur van een kwadratische vorm} van
$q$. Bovendien is
\[
s = \max\{\dim F : q|_F \text{ positief definiet}\},
\]
en \omterm{def:b2:quadratic:adjoint}{symmetrisch} voor $t$.
\end{theorem}

\begin{proof}
Zij $q = \sum_{i \leq s}\ell_i^2 - \sum_{j\leq t} m_j^2$ en zij
$F_+$ het opspansel van de vectoren waarop de $(\ell_i)$ als
coördinaten werken --- concreet: vul de onafhankelijke familie
$(\ell_1, \dots, \ell_s, m_1, \dots, m_t)$ aan tot een basis van de
\omterm{def:b2:linalg:dual}{duale ruimte} $E^*$, en zij $(u_1, \dots, u_n)$ de basis van $E$
waarvan dit de coördinaatvormen zijn (de antiduale basis:
$\ell_i(u_k) = \delta_{ik}$ voor $k \leq s$, en de latere vormen
verdwijnen op de eerdere vectoren). Zet $F_+ =
\operatorname{Vect}(u_1, \dots, u_s)$: voor $x = \sum_{i\leq
s}x_iu_i \in F_+$ is
\[
\ell_i(x) = x_i,
\qquad m_j(x) = 0,
\qquad\text{dus}\qquad
q(x) = \sum_{i\leq s}x_i^2 > 0 \quad (x \neq 0) :
\]
$q|_{F_+}$ is dus positief definiet en het maximum hierboven is
$\geq s$. Omgekeerd, zij $F$ een deelruimte met $q|_F$ positief
definiet, en $G = \{x : \ell_1(x) = \dots = \ell_s(x) = 0\}$, van
codimensie $\leq s$; op $G$ is $q(x) = -\sum m_j^2 \leq 0$. Dan is
$F \cap G = \{0\}$ (een vector daar ongelijk aan nul zou $q > 0$ en
$q \leq 0$ hebben), dus $\dim F \leq \dim E - \dim G \leq s$.
Bijgevolg is het maximum voor \emph{elke} reductie gelijk aan $s$:
$s$ is intrinsiek, en $t = \operatorname{rk} q - s$ eveneens.
\end{proof}

\begin{example}\label{ex:b2:quadratic:gaussexample}
$q(x, y, z) = xy + yz + zx$ (geen kwadraten). Met $x_1 = x$ en $x_2
= y$: $q = xy + z(x + y)$, en de identiteit met twee kwadraten
geeft
\[
q = \tfrac14(x + y + 2z)^2 - \tfrac14(x - y)^2 - z^2 ,
\]
(werk uit ter controle). Drie onafhankelijke vormen: signatuur $(1,
2)$, rang $3$. Eén positieve richting, twee negatieve: de
meetkunde van de ``lichtkegel'' van deze vorm.
\end{example}

\begin{example}[Een ontaarde vorm, volledig herleid]\label{ex:b2:quadratic:degenerate}
$q(x, y, z) = xy + yz$ op $\R^3$: geen kwadraten, dus geval 2 met
de groepering $q = y(x + z)$. De identiteit met twee kwadraten op
het product van de onafhankelijke vormen $y$ en $x + z$ geeft
\[
q = \frac14\bigl(y + x + z\bigr)^2 -
\frac14\bigl(y - x - z\bigr)^2 .
\]
De twee lineaire vormen $y + x + z$ en $y - x - z$ zijn
onafhankelijk (hun verschil is $2(x+z)$, hun som $2y$), zodat
Sylvester aflezen kan: signatuur $(1, 1)$, rang $2$ ---
\emph{ontaard}. De kern van de polaire vorm vindt men door
$\varphi(v, \cdot) = 0$ op te lossen: met matrix
$\frac12\begin{pmatrix} 0&1&0\\ 1&0&1\\ 0&1&0\end{pmatrix}$ is de
kern $\{y = 0,\ x + z = 0\} = \R\,(1, 0, -1)$, de richting waarin
$q$ niets ziet. Het inzicht om te onthouden: rangtekort verschijnt
bij Gauss als ``de veranderlijken raken op'' --- de reductie
leverde uit drie dimensies maar twee kwadraten, en de ontbrekende
dimensie is precies de kern.
\end{example}

\begin{example}[Eén vorm, twee wegen naar de signatuur]\label{ex:b2:quadratic:tworoads}
$q(x, y, z) = 2x^2 + 2y^2 + 2z^2 + 2xy + 2yz$, met matrix
$\begin{pmatrix} 2 & 1 & 0\\ 1 & 2 & 1\\ 0 & 1 & 2
\end{pmatrix}$. \emph{Weg 1, Gauss:} maak de kwadraten op volgorde
af,
\[
q = 2\Bigl(x + \frac y2\Bigr)^{\!2} + \frac32 y^2 + 2yz + 2z^2
= 2\Bigl(x + \frac y2\Bigr)^{\!2}
+ \frac32\Bigl(y + \frac{2z}{3}\Bigr)^{\!2} + \frac43 z^2 :
\]
drie positieve kwadraten op onafhankelijke vormen, signatuur $(3,
0)$: positief definiet. \emph{Weg 2, \omterm{def:b2:reduction:eigen}{eigenwaarden}:} de matrix is de
tridiagonale $2I + N$ met $N$ de buurmatrix; haar \omterm{def:b2:reduction:eigen}{eigenwaarden} zijn
$2 + \sqrt2$, $2$ en $2 - \sqrt2$ (controleer de \omterm{def:b2:reduction:eigen}{eigenvectoren} $(1,
\pm\sqrt2, 1)$ en $(1, 0, -1)$), alle positief: hetzelfde oordeel,
volgens \cref{cor:b2:quadratic:principalaxes}. Gauss is sneller;
\omterm{def:b2:reduction:eigen}{eigenwaarden} zeggen meer (zij geven de hoofdassen en de extreme
waarden van $q$ op de bol). Het inzicht om te onthouden: de
positieve spillen $2, \frac32, \frac43$ van Gauss zijn precies de
verhoudingen $\frac{\Delta_k}{\Delta_{k-1}}$ van de leidende
hoofdminoren ($\Delta_1 = 2$, $\Delta_2 = 3$, $\Delta_3 = 4$) --- de
weekendopgave bewijst dit in het algemeen.
\end{example}

\begin{method}[Een signatuur berekenen: drie routes]\label{met:b2:quadratic:signature}
\begin{enumerate}
  \item \emph{Gauss} (werkt altijd, met de hand het snelst): maak
        de kwadraten op volgorde af, met geval 2 zodra er geen
        kwadraat beschikbaar is; tel de tekens. Ga na dat de
        verzamelde lineaire vormen onafhankelijk zijn --- minder
        vormen dan veranderlijken betekent een kern
        (\cref{ex:b2:quadratic:degenerate}).
  \item \emph{Leidende minoren} (voor definietheidstests): alle
        $\Delta_k > 0$ dan en slechts dan als positief definiet
        (\cref{exo:b2:quadratic:8}); de spillen
        $\Delta_k/\Delta_{k-1}$ geven zelfs de coëfficiënten van
        Gauss (weekendopgave). Deze route faalt stilzwijgend zodra
        een $\Delta_k = 0$: val dan terug op route 1.
  \item \emph{\omterm{def:b2:reduction:eigen}{Eigenwaarden}} (het meest informatief, het duurst):
        de tekens van het spectrum
        (\cref{cor:b2:quadratic:principalaxes}); zij leveren ook
        de hoofdassen en de extreme waarden van $q$ op de
        eenheidsbol. Verkies deze route wanneer de
        eigenstructuur toch al nodig is.
\end{enumerate}
\end{method}

\section{De spectraalstelling}

Zij nu $E$ euclidisch (met inproduct $\langle\cdot,\cdot\rangle$,
volume van bachelorjaar 1).

\begin{definition}[Adjunct; symmetrische endomorfismen]\label{def:b2:quadratic:adjoint}
Voor $u \in \mathcal{L}(E)$ is de \emph{adjunct}\index{adjunct}
$u^*$ het unieke endomorfisme met
\[
\langle u(x), y\rangle = \langle x, u^*(y)\rangle
\qquad (x, y \in E);
\]
in een orthonormale basis is $\operatorname{Mat}(u^*) =
\operatorname{Mat}(u)^{\mathsf T}$. En $u$ heet
\emph{symmetrisch}\index{symmetrisch endomorfisme}
(zelftoegevoegd) wanneer $u^* = u$ --- gelijkwaardig: wanneer haar
matrix in een orthonormale basis symmetrisch is.
\end{definition}

\begin{proof}[Bestaan en eenduidigheid van de adjunct]
Bij vaste $y$ is de vorm $x \mapsto \langle u(x), y\rangle$
lineair, en dus (eindige dimensie) van de gedaante $\langle x,
z_y\rangle$ voor een unieke $z_y$ --- waarbij de afbeelding $y
\mapsto z_y =: u^*(y)$ wegens de eenduidigheid lineair is. De
identificatie van de matrix: lees $\langle u(e_i), e_j\rangle$ op
beide manieren.
\end{proof}

\begin{example}[De adjunct hangt van het inproduct af]\label{ex:b2:quadratic:weightedadjoint}
Neem op $\R^2$ het \emph{gewogen} inproduct $\langle x, y\rangle_D
= x_1y_1 + 2x_2y_2$ (matrix $D = \operatorname{diag}(1,2)$) en $u$
met matrix $A = \begin{pmatrix} 0 & 1\\ 0 & 0\end{pmatrix}$ in de
canonieke basis. Uit $\langle u(x), y\rangle_D = (Ax)^{\mathsf T}Dy
= x^{\mathsf T}(A^{\mathsf T}D)y$ en $\langle x, u^*(y)\rangle_D =
x^{\mathsf T}(DA^*)y$ volgt dat de matrix van de \omterm{def:b2:quadratic:adjoint}{adjunct}
\[
A^* = D^{-1}A^{\mathsf T}D
= \begin{pmatrix} 1 & 0\\ 0 & \tfrac12\end{pmatrix}
\begin{pmatrix} 0 & 0\\ 1 & 0\end{pmatrix}
\begin{pmatrix} 1 & 0\\ 0 & 2\end{pmatrix}
= \begin{pmatrix} 0 & 0\\ \tfrac12 & 0\end{pmatrix}
\neq A^{\mathsf T}
\]
is. Controle op $x = (1,0)$ en $y = (0,1)$:
\[
\langle u(x), y\rangle_D = \langle (0,0), y\rangle_D = 0,
\quad
\langle x, u^*(y)\rangle_D
= \langle(1,0), (0,\tfrac12)\rangle_D = 0 ;
\]
en op $x = (0,1)$, $y = (1,0)$:
\[
\langle u(x), y\rangle_D = \langle(1,0),(1,0)\rangle_D = 1,
\quad
\langle x, u^*(y)\rangle_D =
\langle(0,1),(0,\tfrac12)\rangle_D = 1 .
\] Het inzicht om te onthouden:
``$\operatorname{Mat}(u^*) = \operatorname{Mat}(u)^{\mathsf T}$''
is een uitspraak over \emph{orthonormale} basissen alleen; in het
algemeen komt de metriek $D$ ertussen, precies als bij de simultane
reductie van de weekendopgave.
\end{example}

\begin{theorem}[Spectraalstelling]\label{thm:b2:quadratic:spectral}
Zij $u$ een \omterm{def:b2:quadratic:adjoint}{symmetrisch endomorfisme} van een euclidische ruimte
$E$. Dan bezit $E$ een \emph{orthonormale basis van \omterm{def:b2:reduction:eigen}{eigenvectoren}}
van $u$; alle \omterm{def:b2:reduction:eigen}{eigenwaarden} zijn reëel, en de \omterm{def:b2:reduction:eigen}{eigenruimten} bij
verschillende \omterm{def:b2:reduction:eigen}{eigenwaarden} zijn orthogonaal. Matrixvorm: elke
reële \omterm{def:b2:quadratic:adjoint}{symmetrische} matrix $A$ schrijft zich als
\[
A = P\,D\,P^{\mathsf T},
\qquad P \text{ orthogonaal } (P^{\mathsf T} P = I),\ D
\text{ diagonaal}.\index{spectraalstelling}
\]
\end{theorem}

\begin{proof}
\emph{Er bestaat een \omterm{def:b2:reduction:eigen}{eigenvector}.} De functie $x \mapsto \langle
u(x), x\rangle$ is \omterm{def:b2:metric:continuity}{continu} op de eenheidssfeer $S$ van $E$, die
\omterm{def:b2:metric:compact}{compact} is (eindige dimensie, \cref{thm:b2:nvs:finitedim}): zij
neemt haar maximum $\lambda$ aan in een zekere $a \in S$. Wij
beweren dat $u(a) = \lambda a$. Voor elke $y \perp a$ met $\norm y
= 1$ en $t \in \R$ ligt de vector $x_t = \frac{a + ty}{\sqrt{1 +
t^2}}$ op $S$ (want $\norm{a + ty}^2 = 1 + t^2$ volgens Pythagoras);
de gemaximaliseerde functie uitwerken geeft
\[
g(t) = \langle u(x_t), x_t\rangle
= \frac{\langle u(a), a\rangle + 2t\langle u(a), y\rangle +
t^2\langle u(y), y\rangle}{1 + t^2}
\]
(de symmetrie van $u$ voegde de twee kruistermen samen: $\langle
u(a), y\rangle = \langle a, u(y)\rangle = \langle u(y), a\rangle$).
$g$ is een differentieerbare functie van $t$ met een maximum in $t
= 0$; de quotiëntregel in $0$ geeft
\[
g'(0) = \frac{2\langle u(a), y\rangle\cdot 1 - \langle u(a),
a\rangle\cdot 0}{1} = 2\langle u(a), y\rangle = 0 .
\]
Dus staat $u(a)$ loodrecht op het hele hypervlak $a^\perp$: $u(a)
\in (a^{\perp})^{\perp} = \R a$, dat wil zeggen $u(a) = \mu a$; en
$\mu = \langle u(a), a\rangle = \lambda$.

\emph{Inductie.} Het orthogonale complement $F = a^\perp$ is
$u$-stabiel: voor $x \perp a$ is $\langle u(x), a\rangle = \langle
x, u(a)\rangle = \lambda\langle x, a\rangle = 0$. De beperking
$u|_F$ is \omterm{def:b2:quadratic:adjoint}{symmetrisch} voor het geïnduceerde inproduct; met
inductie naar de dimensie heeft $F$ een orthonormale eigenbasis;
zet $a$ ervoor.

\emph{Aanvullingen.} De \omterm{def:b2:reduction:eigen}{eigenwaarden} zijn de reële getallen
$\langle u(e), e\rangle$ op de eigenbasis. Orthogonaliteit van de
\omterm{def:b2:reduction:eigen}{eigenruimten}: uit $u(x) = \lambda x$ en $u(y) = \mu y$ volgt
$\lambda\langle x, y\rangle = \langle u(x), y\rangle = \langle x,
u(y)\rangle = \mu\langle x, y\rangle$, dus $\langle x, y\rangle =
0$ zodra $\lambda \neq \mu$. Matrixvorm: de kolommen van $P$ zijn
de orthonormale eigenbasis.
\end{proof}

\begin{example}[Een volledige spectrale doorloop]\label{ex:b2:quadratic:fullrun}
Diagonaliseer $A = \begin{pmatrix} 1 & 2\\ 2 & 1\end{pmatrix}$
orthogonaal. \omterm{def:b2:reduction:charpoly}{Karakteristieke veelterm} $(1 - \lambda)^2 - 4$:
\omterm{def:b2:reduction:eigen}{eigenwaarden} $3$ en $-1$. \omterm{def:b2:reduction:eigen}{Eigenvectoren}: $(A - 3I)v = 0$ geeft $v_1
= \frac{1}{\sqrt2}(1,1)$; $(A + I)v = 0$ geeft $v_2 =
\frac{1}{\sqrt2}(1,-1)$ --- orthogonaal, zoals
\cref{thm:b2:quadratic:spectral} zonder rekenwerk garandeert. Met
$P = (v_1\ v_2)$ (een rotatie over $\frac\pi4$):
\[
P^{\mathsf T}AP = \begin{pmatrix} 3 & 0\\ 0 & -1
\end{pmatrix},
\qquad
x^2 + 4xy + y^2 = 3u^2 - v^2
\quad\text{in het gedraaide assenstelsel} .
\]
De vorm van \cref{exo:b2:quadratic:1} is dus van hyperbooltype:
signatuur $(1,1)$, in overeenstemming met haar \omterm{thm:b2:quadratic:gauss}{Gauss-reductie} $(x +
2y)^2 - 3y^2$ --- andere kwadraten, dezelfde signatuur, zoals
Sylvester eist. Het inzicht om te onthouden: Gauss gaf het antwoord
sneller, maar de spectrale route meldt bovendien dat $q$ op de
eenheidscirkel precies $\intcc{-1}{3}$ doorloopt, aangenomen langs
$v_2$ en $v_1$: het extra werk koopt meetkunde.
\end{example}

\begin{corollary}[Hoofdassen; positiviteitstests]\label{cor:b2:quadratic:principalaxes}
\begin{enumerate}
  \item Elke \omterm{def:b2:quadratic:def}{kwadratische vorm} $q$ op een euclidische ruimte
        diagonaliseert in een zekere \emph{orthonormale} basis:
        $q(x) = \sum_i \lambda_i x_i^2$ met $\lambda_i$ de
        \omterm{def:b2:reduction:eigen}{eigenwaarden} van de \omterm{def:b2:quadratic:adjoint}{symmetrische} matrix van $q$; de
        signatuur telt de positieve en de negatieve \omterm{def:b2:reduction:eigen}{eigenwaarden}.
  \item Een \omterm{def:b2:quadratic:adjoint}{symmetrische} matrix is positief semidefiniet
        (respectievelijk definiet) dan en slechts dan als al haar
        \omterm{def:b2:reduction:eigen}{eigenwaarden} $\geq 0$ (respectievelijk $> 0$) zijn; en dan
        zijn de extreme waarden van het Rayleigh-quotiënt
        \[
        \min_{\norm x = 1} \langle Ax, x\rangle = \lambda_{\min},
        \qquad
        \max_{\norm x = 1} \langle Ax, x\rangle = \lambda_{\max} .
        \]
\end{enumerate}
\end{corollary}

\begin{proof}
(1) Schrijf $q(x) = \langle A x, x\rangle$ met $A$ \omterm{def:b2:quadratic:adjoint}{symmetrisch} (de
matrix van $q$ in een orthonormale basis); diagonaliseer $A$ met de
spectraalstelling: voor $x = \sum x_ie_i$ in de orthonormale
eigenbasis is
\[
q(x) = \Bigl\langle \sum_i \lambda_ix_ie_i,\ \sum_j
x_je_j\Bigr\rangle = \sum_i \lambda_i x_i^2
\]
(de orthonormaliteit doodt de kruistermen). De $\pm$-tekens van de
$\lambda_i$ tellen volgens Sylvester de signatuur: elke coördinaat
met $\sqrt{\abs{\lambda_i}}$ herschalen levert een \omterm{thm:b2:quadratic:gauss}{Gauss-reductie}
met onafhankelijke vormen.

(2) In de eigenbasis is $\langle Ax, x\rangle = \sum \lambda_i
x_i^2$, ingeklemd tussen $\lambda_{\min}\norm x^2$ en
$\lambda_{\max}\norm x^2$, met gelijkheid in de bijbehorende
\omterm{def:b2:reduction:eigen}{eigenvectoren}; de positiviteit van alle \omterm{def:b2:reduction:eigen}{eigenwaarden} is dus
gelijkwaardig met die van de vorm.
\end{proof}

\begin{example}\label{ex:b2:quadratic:spectralexample}
$A = \begin{pmatrix} 2 & 1\\ 1 & 2 \end{pmatrix}$: \omterm{def:b2:reduction:eigen}{eigenwaarden} $3$
(\omterm{def:b2:reduction:eigen}{eigenvector} $\frac{1}{\sqrt2}(1,1)$) en $1$
($\frac{1}{\sqrt2}(1,-1)$). De \omterm{def:b2:quadratic:def}{kwadratische vorm} $2x^2 + 2xy +
2y^2$ wordt $3X^2 + Y^2$ in het gedraaide orthonormale
assenstelsel: de hoofdassen van een ellips, uitgerekend. De
\omterm{thm:b2:quadratic:gauss}{Gauss-reductie} bereikt ook een diagonale vorm, maar alleen de
spectraalstelling bereikt haar \emph{zonder lengten te
vervormen}.
\end{example}

\begin{example}[Een ellips volledig geïdentificeerd]\label{ex:b2:quadratic:ellipserun}
Welke kromme is $5x^2 + 4xy + 2y^2 = 6$? De matrix
$\begin{pmatrix} 5 & 2\\ 2 & 2\end{pmatrix}$ heeft \omterm{def:b2:reduction:charpoly}{karakteristieke
veelterm} $\lambda^2 - 7\lambda + 6 = (\lambda - 1)(\lambda - 6)$:
\omterm{def:b2:reduction:eigen}{eigenwaarden} $1$ en $6$, beide positief --- dus een ellips.
Orthonormale \omterm{def:b2:reduction:eigen}{eigenvectoren}: voor $\lambda = 1$ lost men
$\begin{pmatrix} 4 & 2\\ 2 & 1\end{pmatrix}v = 0$ op: $v_1 =
\frac{1}{\sqrt5}(1, -2)$; voor $\lambda = 6$: $v_2 =
\frac{1}{\sqrt5}(2, 1)$. In de gedraaide coördinaten $(X, Y)$ langs
$(v_2, v_1)$ wordt de vergelijking
\[
6X^2 + Y^2 = 6,
\qquad\text{dat wil zeggen}\qquad
X^2 + \frac{Y^2}{6} = 1 :
\]
halve assen $1$ (langs $v_2$) en $\sqrt6$ (langs $v_1$). Het
inzicht om te onthouden: de ruwe gedaante was gratis --- $\det = 6
> 0$ en een positief spoor kondigen een ellips aan nog voordat er
één \omterm{def:b2:reduction:eigen}{eigenvector} is berekend --- maar alleen de spectraalstelling
levert de richtingen en lengten van de assen, dat wil zeggen de
werkelijke meetkunde.
\end{example}

\begin{example}[Extrema op de bol, van het spectrum afgelezen]\label{ex:b2:quadratic:sphereextremes}
Wat zijn de extreme waarden van $q(x,y,z) = 2xy + 2yz + 2zx$ op de
eenheidsbol? Haar matrix (de matrix met enen buiten de diagonaal
uit \cref{exo:b2:quadratic:2}) heeft \omterm{def:b2:reduction:eigen}{eigenwaarden} $2$ en $-1$
(dubbel), dus geeft \cref{cor:b2:quadratic:principalaxes}~(2):
\[
\max_{\norm v = 1} q(v) = 2
\ \text{ in } v = \tfrac{1}{\sqrt3}(1,1,1),
\qquad
\min_{\norm v = 1} q(v) = -1
\ \text{ op de cirkel } x + y + z = 0 .
\]
Geen analyse, geen multiplicatoren van Lagrange: de
spectraalstelling lost deze optimalisatie onder nevenvoorwaarde
rechtstreeks op --- en geeft de maximalisator erbij. Het inzicht om
te onthouden: vergelijk dit met de methode van de multiplicatoren
uit het hoofdstuk over differentiaalrekening, die met meer werk
dezelfde kritieke punten vindt; voor \emph{kwadratische}
doelfuncties op bollen zijn spectra de koninklijke weg (de
weekendopgave van het hermitische hoofdstuk bouwt de hele theorie
van Courant en Fischer op deze waarneming).
\end{example}

\begin{remark}[Klassieke valkuilen]
\emph{(i) \omterm{def:b2:quadratic:def}{Congruentie} is geen gelijkvormigheid:} een basisovergang
voor een vorm werkt als $P^{\mathsf T}BP$, niet als $P^{-1}BP$;
\omterm{def:b2:reduction:eigen}{eigenwaarden} zijn \emph{geen} invarianten van een \omterm{def:b2:quadratic:def}{kwadratische vorm}
($I$ en $4I$ zijn congruent via $P = 2I$) --- alleen hun tekens
zijn dat (Sylvester). Spreek pas van de \omterm{def:b2:reduction:eigen}{eigenwaarden} van een vorm
zodra er een inproduct is vastgelegd. \emph{(ii) Positieve
elementen bewijzen niets:} $\begin{pmatrix} 1 & 2\\ 2 &
1\end{pmatrix}$ heeft louter positieve elementen en toch signatuur
$(1,1)$ ($\det = -3$); omgekeerd kan een positief definiete matrix
negatieve elementen buiten de diagonaal hebben
(\cref{ex:b2:quadratic:tworoads} verschoven: $2I - N$ werkt
evengoed). Gebruik \cref{met:b2:quadratic:signature}.
\emph{(iii) Afhankelijke kwadraten:} $q = \ell_1^2 - \ell_2^2$
schrijven zegt niets zodra $\ell_1, \ell_2$ evenredig zijn --- $x^2
+ 2xy + y^2 = (x+y)^2$ heeft rang $1$ en niet $2$; ga altijd de
onafhankelijkheid na voordat je de signatuur afleest. \emph{(iv)
Extrema op de bol zonder \omterm{def:b2:metric:compact}{compactheid}:} de grenzen van Rayleigh in
\cref{cor:b2:quadratic:principalaxes} worden aangenomen omdat de
bol \omterm{def:b2:metric:compact}{compact} is; op de \omterm{def:b2:metric:topology}{open} bal of op de hele ruimte heeft een
onbepaalde vorm noch maximum noch minimum.
\end{remark}

\begin{remark}[Waar dit wordt gebruikt]
De spectraalstelling is het meest geëxporteerde resultaat van dit
boek: de statistiek diagonaliseert er covariantiematrices mee
(hoofdcomponentenanalyse), de mechanica haalt er de normale modi
van trillingen uit (de simultane reductie van de weekendopgave), de
numerieke wiskunde bouwt er de ontbindingen van Cholesky en in
singuliere waarden op (dezelfde opgave), en het volgende hoofdstuk
brengt haar over naar complexe hermitische ruimten. Het volume van
bachelorjaar 3 bewijst haar oneindigdimensionale gedaante voor
\omterm{def:b2:metric:compact}{compacte} \omterm{def:b2:quadratic:adjoint}{zelftoegevoegde} operatoren, waar het compactheidsargument
uit het eindigdimensionale bewijs het hele verhaal wordt.
\end{remark}

\begin{remark}[Vooruitblik binnen dit volume]
\omterm{def:b2:quadratic:def}{Kwadratische vormen} lopen in drie vermommingen door de rest van
Boek 4. Als \emph{hessianen}: het hoofdstuk over
differentiaalrekening klasseert kritieke punten naar de signatuur
van de tweede-ordevorm, zodat de invariantie van Sylvester juist
datgene is wat van ``zadelpunt'' een welgedefinieerd woord maakt.
Als \emph{energieën}: de oscillatoren van het hoofdstuk over
differentiaalvergelijkingen dragen de kwadratische energie
$\frac12x'^2 + \frac12\omega^2x^2$, en de simultane reductie van de
weekendopgave van dit hoofdstuk is precies het uithalen van de
normale modi. Als \emph{meetkunde}: de kegelsneden van dit
hoofdstuk groeien uit tot de kwadrieken van de meetkundige
hoofdstukken, waar de tweede fundamentaalvorm van een oppervlak ---
een \omterm{def:b2:quadratic:def}{kwadratische vorm} op elk raakvlak --- met haar signatuur
beslist of het oppervlak als een schaal dan wel als een zadel
kromt. Het hermitische hoofdstuk, hierna, speelt de hele partituur
over $\C$ opnieuw.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b2:quadratic:1}
Herleid met Gauss en geef rang en signatuur:
\[
q_1(x,y) = x^2 + 4xy + y^2,
\qquad
q_2(x,y,z) = x^2 + 2y^2 + 3z^2 + 2xy + 2yz .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:quadratic:2}
Diagonaliseer $A = \begin{pmatrix} 0 & 1 & 1\\ 1 & 0 & 1\\ 1 & 1 &
0\end{pmatrix}$ orthogonaal (de \omterm{def:b2:reduction:eigen}{eigenwaarden} komen uit de
berekening van \cref{ch:b2:reduction}; maak nu de basis
orthonormaal) en herleid de vorm $q(x,y,z) = 2xy + 2yz + 2zx$ tot
hoofdassen.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:quadratic:3}
Bewijs dat $u^{**} = u$, dat $(u \circ v)^* = v^* \circ u^*$, en
dat $\ker u^* = (\operatorname{im} u)^{\perp}$. Leid af dat
$\operatorname{rk} u^* = \operatorname{rk} u$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:4}
Zij $A$ reëel \omterm{def:b2:quadratic:adjoint}{symmetrisch} met $A^3 = A$. Bewijs dat $A^2$ de matrix
van een orthogonale projectie is. Verbind algemener de spectrale
ontbindingen van $A$ en $P(A)$ voor een veelterm $P$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:5}
Bewijs dat $O(n) = \{P : P^{\mathsf T}P = I\}$ een \omterm{def:b2:metric:compact}{compacte}
deelverzameling van $\mathcal{M}_n(\R)$ is \emph{(gesloten:
origineel van $\{I\}$ onder een \omterm{def:b2:metric:continuity}{continue} afbeelding; begrensd: de
kolommen zijn eenheidsvectoren)}. Is zij \omterm{def:b2:metric:connected}{samenhangend}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:6}
(Vierkantswortel) Zij $A$ \omterm{def:b2:quadratic:adjoint}{symmetrisch} positief semidefiniet.
Construeer een \omterm{def:b2:quadratic:adjoint}{symmetrische} positief semidefiniete $B$ met $B^2 =
A$, en bewijs dat zij \emph{uniek} is \emph{(bestaan: neem in een
spectrale basis de vierkantswortels van de \omterm{def:b2:reduction:eigen}{eigenwaarden};
eenduidigheid: een kandidaat $B$ commuteert met $A = B^2$ en
bewaart dus haar \omterm{def:b2:reduction:eigen}{eigenruimten} --- herleid op elk daarvan tot het
scalaire geval)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:7}
Bewijs voor reële \omterm{def:b2:quadratic:adjoint}{symmetrische} $A$ dat $\vertiii{A}_2 :=
\sup_{\norm x_2 = 1}\norm{Ax}_2 = \max_i \abs{\lambda_i}$ (de
spectraalstraal), en bereken $\vertiii{A}_2$ voor $A =
\begin{pmatrix} 1 & 2\\ 2 & 1\end{pmatrix}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:quadratic:8}
(Criterium van Sylvester) Zij $A$ reëel \omterm{def:b2:quadratic:adjoint}{symmetrisch} met leidende
hoofdminoren $\Delta_1, \dots, \Delta_n$ (de \omterm{def:b2:linalg:det}{determinanten} van de
blokken linksboven). Bewijs dat $A$ positief definiet is dan en
slechts dan als alle $\Delta_k > 0$. \emph{(Voor $\Rightarrow$:
beperkingen van een definiete vorm zijn definiet, en de \omterm{def:b2:linalg:det}{determinant}
van een positief definiete matrix --- het product van haar
\omterm{def:b2:reduction:eigen}{eigenwaarden} --- is positief. Voor $\Leftarrow$: inductie naar $n$;
het blok linksboven van formaat $n-1$ is positief definiet,
diagonaliseer de vorm op die deelruimte en maak het kwadraat in de
laatste veranderlijke af; het teken van het laatste
diagonaalelement wordt door $\det A = \Delta_n > 0$ bestuurd.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:quadratic:9}
(Courant--Fischer, tweede \omterm{def:b2:reduction:eigen}{eigenwaarde}) Zij $u$ \omterm{def:b2:quadratic:adjoint}{symmetrisch} met
\omterm{def:b2:reduction:eigen}{eigenwaarden} $\lambda_1 \geq \lambda_2 \geq \dots \geq \lambda_n$.
Bewijs dat
\[
\lambda_2 = \min_{\substack{H \text{ hypervlak}}}\;
\max_{\substack{x \in H,\ \norm x = 1}} \langle u(x), x\rangle .
\]
\emph{(Voor $\leq$: elk hypervlak ontmoet het vlak opgespannen door
de twee bovenste \omterm{def:b2:reduction:eigen}{eigenvectoren}. Voor $\geq$: kies $H =
(e_1)^{\perp}$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:10}
Bepaal rang en signatuur van $q(x_1, \dots, x_n) = \sum_{i < j}
x_ix_j$ op $\R^n$ ($n \geq 2$), op twee manieren: met de
algebraïsche identiteit $2q = \bigl(\sum x_i\bigr)^2 - \sum x_i^2$
samen met de beperking van $q$ tot het hypervlak $\sum x_i = 0$; en
door de \omterm{def:b2:reduction:eigen}{eigenwaarden} van haar matrix $\frac12(J - I)$ te berekenen,
waarbij $J$ de matrix vol enen is.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:11}
Zijn $A$ en $B$ reëel \omterm{def:b2:quadratic:adjoint}{symmetrisch} met $B$ positief semidefiniet.
Bewijs dat
\[
\lambda_{\min}(A)\operatorname{tr} B
\;\leq\;
\operatorname{tr}(AB)
\;\leq\;
\lambda_{\max}(A)\operatorname{tr} B .
\]
\emph{(Schrijf $B = C^{\mathsf T}C$ en $\operatorname{tr}(AB) =
\sum_i \langle A c_i, c_i\rangle$ over de kolommen $c_i$ van
$C^{\mathsf T}$.)} In het bijzonder is $\operatorname{tr}(AB) \geq
0$ zodra beide positief semidefiniet zijn.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:quadratic:12}
Beschouw op $E = \mathcal{M}_n(\R)$ de afbeelding $q(M) =
\operatorname{tr}(M^2)$.
\begin{enumerate}
  \item Toon aan dat $q$ een \omterm{def:b2:quadratic:def}{kwadratische vorm} is met polaire vorm
        $\varphi(M, N) = \operatorname{tr}(MN)$.
  \item Toon aan dat de \omterm{def:b2:quadratic:adjoint}{symmetrische} en de antisymmetrische
        matrices $\varphi$-orthogonale deelruimten vormen waarop
        $q$ respectievelijk positief definiet en negatief definiet
        is \emph{(bereken $\operatorname{tr}(M^2)$ element voor
        element in elk geval)}.
  \item Besluit: $q$ heeft signatuur $\bigl(\frac{n(n+1)}{2},
        \frac{n(n-1)}{2}\bigr)$ en rang $n^2$.
\end{enumerate}
\end{exercise}

\section{Probleem: Cholesky, Hadamard en de polaire ontbinding}

\begin{problem}\label{pb:b2:quadratic:1}
De spectraalstelling is een microscoop; deze opgave gebruikt haar
als fabriek. Uit Gram-matrices maken we de
\emph{Cholesky-factorisatie}\index{Cholesky-factorisatie} (en
herkennen we de spillen van Gauss als verhoudingen van minoren),
bewijzen we daarna de \emph{ongelijkheid van
Hadamard}\index{ongelijkheid van Hadamard} voor \omterm{def:b2:linalg:det}{determinanten},
bouwen we de \emph{polaire ontbinding}\index{polaire ontbinding} $A
= QS$ en de \emph{ontbinding in singuliere waarden}, klasseren we
de kegelsneden in het vlak, en eindigen we met de simultane
reductie van twee vormen --- de stelling achter de normale modi van
trillingen. Overal is $E = \R^n$ met haar standaardinproduct.

\medskip
\noindent\textbf{Deel I --- Gram-matrices en Cholesky.} Voor
vectoren $v_1, \dots, v_n \in E$ is hun \emph{Gram-matrix} $G =
\bigl(\langle v_i, v_j\rangle\bigr)_{i,j}$.
\begin{enumerate}
  \item Toon aan dat $G$ \omterm{def:b2:quadratic:adjoint}{symmetrisch} positief semidefiniet is, en
        positief definiet dan en slechts dan als $(v_1, \dots,
        v_n)$ lineair onafhankelijk is \emph{(bereken $X^{\mathsf
        T}GX$)}.
  \item Toon omgekeerd aan dat elke \omterm{def:b2:quadratic:adjoint}{symmetrische} positief
        semidefiniete $A$ een Gram-matrix is: $A = C^{\mathsf T}C$
        voor zekere $C$ (gebruik de vierkantswortel van
        \cref{exo:b2:quadratic:6}), met $C$ inverteerbaar dan en
        slechts dan als $A$ definiet is.
  \item Leid af dat een positief semidefiniete $A$ voor alle $i, j$
        voldoet aan $\abs{a_{ij}} \leq \sqrt{a_{ii}\,a_{jj}}$
        \emph{(beperk tot twee coördinaten)} --- de ongelijkheid
        van Cauchy--Schwarz, matricieel herlezen.
  \item (Cholesky) Zij $A$ positief definiet. Bewijs dat er een
        \emph{unieke} bovendriehoekige $T$ met positieve
        diagonaalelementen bestaat zodanig dat
        \[
        A = T^{\mathsf T}\,T
        \]
        \emph{(bestaan: pas Gram--Schmidt toe op vectoren die $A$
        als Gram-matrix realiseren; eenduidigheid: geldt
        $T_1^{\mathsf T}T_1 = T_2^{\mathsf T}T_2$, toon dan aan dat
        $T_1T_2^{-1}$ orthogonaal en driehoekig is met positieve
        diagonaal, en dus $I$)}.
  \item Toon aan dat de leidende hoofdminoren voldoen aan $\Delta_k
        = (t_{11}\cdots t_{kk})^2$, en leid af dat de spillen die
        de \omterm{thm:b2:quadratic:gauss}{Gauss-reductie} van een positief definiete vorm in de
        natuurlijke volgorde van de veranderlijken oplevert,
        gelijk zijn aan
        \[
        d_k = \frac{\Delta_k}{\Delta_{k-1}}
        \qquad (\Delta_0 = 1) :
        \]
        de minoren van het criterium van Sylvester
        (\cref{exo:b2:quadratic:8}) en de spillen van Gauss zijn
        dezelfde gegevens. Controleer dit op
        \cref{ex:b2:quadratic:tworoads}.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De ongelijkheid van Hadamard.}
\begin{enumerate}[resume]
  \item Zij $A$ positief definiet. Bewijs dat
        \[
        \det A \leq a_{11}\,a_{22}\cdots a_{nn}
        \]
        \emph{(normaliseer: $B = DAD$ met $D =
        \operatorname{diag}(a_{ii}^{-1/2})$ heeft diagonaal vol
        enen; begrens $\det B = \prod \mu_i$ met de ongelijkheid
        tussen rekenkundig en meetkundig gemiddelde tegen
        $\operatorname{tr} B = n$)}.
  \item Toon aan dat gelijkheid geldt dan en slechts dan als $A$
        diagonaal is.
  \item Leid de \emph{ongelijkheid van Hadamard} af: voor elke
        reële vierkante matrix $M$ met kolommen $c_1, \dots, c_n$
        geldt
        \[
        \abs{\det M} \leq \prod_{i=1}^{n}\norm{c_i}_2 ,
        \]
        met gelijkheid (voor inverteerbare $M$) dan en slechts dan
        als de kolommen paarsgewijs orthogonaal zijn \emph{(pas de
        vragen 6--7 toe op $M^{\mathsf T}M$)}.
  \item Meetkundige en combinatorische opbrengsten: interpreteer
        vraag 8 als ``het volume van een parallellepipedum is
        hoogstens het product van zijn ribbelengten''; en toon aan
        dat een matrix met alle elementen in $\intcc{-1}{1}$
        voldoet aan $\abs{\det M} \leq n^{n/2}$. (Matrices die deze
        grens bereiken --- hadamardmatrices --- bestaan voor $n =
        1, 2$ en voor vele veelvouden van $4$; of zij voor
        \emph{alle} veelvouden van $4$ bestaan, is een beroemd open
        probleem.)
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- \omterm{pb:b2:quadratic:1}{Polaire ontbinding} en singuliere
waarden.}
\begin{enumerate}[resume]
  \item Zij $A$ inverteerbaar. Toon aan dat $A^{\mathsf T}A$
        positief definiet is, en dat
        \[
        S = \sqrt{A^{\mathsf T}A}
        \quad\text{(de vierkantswortel van
        \cref{exo:b2:quadratic:6})},
        \qquad Q = AS^{-1}
        \]
        een ontbinding $A = QS$ geven met $Q$ orthogonaal en $S$
        positief definiet.
  \item Bewijs dat deze ontbinding van een inverteerbare $A$
        eenduidig is.
  \item Breid het bestaan uit tot willekeurige $A$: kies
        $\varepsilon_k \to 0$ met $A + \varepsilon_k I$
        inverteerbaar, schrijf $A + \varepsilon_kI = Q_kS_k$, en
        gebruik de \omterm{def:b2:metric:compact}{compactheid} van $O(n)$
        (\cref{exo:b2:quadratic:5}) om $Q_k \to Q$ te extraheren;
        toon aan dat $S_k = Q_k^{\mathsf T}(A + \varepsilon_kI)$
        naar een positief semidefiniete $S$ convergeert met $A =
        QS$ en $S = \sqrt{A^{\mathsf T}A}$. Waar faalt de
        eenduidigheid voor singuliere $A$?
  \item (Ontbinding in singuliere waarden) Leid af dat elke reële
        vierkante $A$ zich laat schrijven als
        \[
        A = U\,\Sigma\,V^{\mathsf T},
        \qquad U, V \in O(n),\quad
        \Sigma = \operatorname{diag}(\sigma_1, \dots,
        \sigma_n),\ \sigma_i \geq 0 ,
        \]
        waarbij de $\sigma_i$ (de \emph{singuliere waarden}) de
        \omterm{def:b2:reduction:eigen}{eigenwaarden} van $\sqrt{A^{\mathsf T}A}$ zijn.
  \item Drie gevolgen: $\vertiii{A}_2 = \sigma_{\max}$ voor
        \emph{elke} reële $A$ (wat \cref{exo:b2:quadratic:7}
        veralgemeent); $\abs{\det A} = \sigma_1\cdots\sigma_n$; en
        het beeld van de eenheidsbol onder een inverteerbare $A$ is
        een ellipsoïde met halve assen $\sigma_1, \dots, \sigma_n$
        langs de kolommen van $U$.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Kegelsneden, met de
spectraalstelling.} Een kegelsnede in het vlak is de
nulverzameling van $f(x) = q(x) + \langle b, x\rangle + c$, met $q
\neq 0$ een \omterm{def:b2:quadratic:def}{kwadratische vorm} met matrix $A$, $b \in \R^2$ en $c
\in \R$.
\begin{enumerate}[resume]
  \item Herleid $f$ met een rotatie (hoofdassen,
        \cref{cor:b2:quadratic:principalaxes}) gevolgd door een
        translatie, en klasseer de mogelijke niet-lege,
        niet-ontaarde gedaanten naar de signatuur van $q$: ellips
        ($\det A > 0$), hyperbool ($\det A < 0$), parabool ($\det A
        = 0$, rang $1$, met de lineaire term niet geabsorbeerd).
  \item Voer de reductie \omterm{def:b2:metric:complete}{volledig} uit voor
        \[
        x^2 + 4xy + y^2 + 2x - 2y = 4 :
        \]
        gedraaide coördinaten, herleide vergelijking, aard en
        middelpunt van de kegelsnede.
  \item (Centrale kegelsneden) Neem aan dat $\det A \neq 0$. Toon
        aan dat het \emph{middelpunt} $x_0 = -\frac12 A^{-1}b$ is,
        en dat de \omterm{def:b2:quadratic:def}{congruentie} met $\begin{pmatrix} I & x_0\\ 0 &
        1\end{pmatrix}$ van de $3\times3$-matrix $\widetilde Q =
        \begin{pmatrix} A & b/2 \\ b^{\mathsf T}/2 &
        c\end{pmatrix}$ oplevert dat
        \[
        \det\widetilde Q = f(x_0)\,\det A :
        \]
        de centrale kegelsnede is ontaard (een punt of twee
        rechten) precies wanneer $\det\widetilde Q = 0$.
  \item Ga vraag 17 na op het voorbeeld van vraag 16: bereken
        $x_0$, $f(x_0)$ en $\det\widetilde Q$, en besluit opnieuw
        dat de kegelsnede een niet-ontaarde hyperbool is.
  \item (Een bundel kwadrieken) Klasseer voor $\lambda \in \R$ het
        oppervlak
        \[
        x^2 + y^2 + z^2 + 2\lambda(xy + yz + zx) = 1
        \]
        naar de \omterm{def:b2:reduction:eigen}{eigenwaarden} van haar matrix \emph{(structuur met
        enen: \omterm{def:b2:reduction:eigen}{eigenwaarden} $1 + 2\lambda$ en $1 - \lambda$
        dubbel)}: bol of ellipsoïde, cilinder, paar vlakken, en
        hyperboloïden van één en van twee bladen, naargelang
        $\lambda$.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Twee vormen tegelijk: simultane
reductie.}
\begin{enumerate}[resume]
  \item Zij $q$ positief definiet en $q'$ een willekeurige
        \omterm{def:b2:quadratic:def}{kwadratische vorm} op $E$. Bewijs dat er een basis van $E$
        bestaat die orthonormaal is voor $q$ en orthogonaal voor
        $q'$: daarin is $q = \sum x_i^2$ en $q' = \sum \mu_i
        x_i^2$ \emph{(gebruik $q$ als inproduct en pas de
        spectraalstelling toe op het endomorfisme dat $q'$
        voorstelt)}.
  \item Matrixvorm: voor $A$ positief definiet en $B$ \omterm{def:b2:quadratic:adjoint}{symmetrisch}
        bestaat er een inverteerbare $P$ met $P^{\mathsf T}AP = I$
        en $P^{\mathsf T}BP = \operatorname{diag}(\mu_1, \dots,
        \mu_n)$, waarbij de $\mu_i$ de nulpunten van $\det(B - \mu
        A) = 0$ zijn.
  \item Voer dit \omterm{def:b2:metric:complete}{volledig} uit voor
        \[
        A = \begin{pmatrix} 2 & 1\\ 1 & 1 \end{pmatrix},
        \qquad
        B = \begin{pmatrix} 0 & 1\\ 1 & 0 \end{pmatrix} :
        \]
        de veralgemeende \omterm{def:b2:reduction:eigen}{eigenwaarden} $\mu_\pm$, en vectoren die
        beide vormen diagonaliseren.
  \item Toon aan dat de positieve definietheid niet mag vervallen:
        voor
        \[
        A = \begin{pmatrix} 1 & 0\\ 0 & -1 \end{pmatrix},
        \qquad
        B = \begin{pmatrix} 0 & 1\\ 1 & 0 \end{pmatrix},
        \]
        diagonaliseert geen enkele basis beide vormen \emph{(zou
        $P$ ze beide diagonaliseren, dan zou $\det(B - \mu A)$ in
        lineaire factoren met reële nulpunten uiteenvallen; reken
        haar uit)}.
  \item Toon aan dat de $\mu_i$ uit vraag 21 de \omterm{def:b2:reduction:eigen}{eigenwaarden} van
        $A^{-1}B$ zijn, en dat $A^{-1}B$, hoewel in het algemeen
        niet \omterm{def:b2:quadratic:adjoint}{symmetrisch}, altijd \omterm{def:b2:reduction:diag}{diagonaliseerbaar} is met reële
        \omterm{def:b2:reduction:eigen}{eigenwaarden} \emph{(conjugeer met $\sqrt A$)}.
  \item Synthese. In telkens één zin: (i) de ene stelling waarop
        elk deel steunde; (ii) welke resultaten uit de Delen
        I--III overleven voor positief \emph{semi}definiete
        matrices, en welke definietheid nodig hebben; (iii) het
        fysische systeem waarvan de vragen 20--22 de kleine
        trillingen diagonaliseren (kinetische en potentiële energie
        als de twee vormen), en wat de $\mu_i$ daar betekenen.
\end{enumerate}
\end{problem}
