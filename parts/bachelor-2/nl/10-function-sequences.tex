\chapter{Rijen en reeksen van functies}\label{ch:b2:funcseq}

Wanneer functies naar een functie convergeren, welke eigenschappen
overleven dan de overgang naar de limiet? \omterm{def:b2:funcseq:def}{Puntsgewijze convergentie}
bewaart vrijwel niets; \emph{\omterm{def:b2:funcseq:def}{uniforme}} convergentie --- convergentie
in de supnorm --- bewaart de \omterm{def:b2:metric:continuity}{continuïteit}, integralen op segmenten
en, met één kanttekening, afgeleiden. Dit hoofdstuk bewijst de drie
overdrachtsstellingen, hun versies voor reeksen, en bekroont ze met
de benaderingsstelling van Weierstrass, bewezen met de fraaie
kansveeltermen van Bernstein.

\section{Puntsgewijze en uniforme convergentie}

\begin{definition}\label{def:b2:funcseq:def}
Zijn $f_n, f \colon X \to \R$ (of $\C$, of een genormeerde ruimte)
met $X$ een willekeurige verzameling. Dan convergeert $(f_n)$
\emph{puntsgewijs}\index{puntsgewijze convergentie} naar $f$
wanneer $f_n(x) \to f(x)$ voor elke $x$, en
\emph{uniform}\index{uniforme convergentie} wanneer
\[
\norm{f_n - f}_\infty = \sup_{x \in X}\, \abs{f_n(x) - f(x)}
\xrightarrow[n \to \infty]{} 0 .
\]
Uniform impliceert puntsgewijs; op $C(\intcc{a}{b})$ is uniforme
convergentie precies de convergentie in de \omterm{def:b2:nvs:banach}{banachruimte}
$\bigl(C(\intcc{a}{b}), \norm\cdot_\infty\bigr)$ van
\cref{ch:b2:nvs}.
\end{definition}

\begin{example}\label{ex:b2:funcseq:xn}
Op $\intcc{0}{1}$ convergeert $f_n(x) = x^n$ puntsgewijs naar de
\emph{discontinue} limiet $f = \mathbf{1}_{\{1\}}$; de convergentie
is niet \omterm{def:b2:funcseq:def}{uniform}: $\norm{f_n - f}_\infty \geq f_n\bigl(1 -
\tfrac1n\bigr) = (1 - \tfrac1n)^n \to \eu^{-1} \neq 0$. Op
$\intcc{0}{a}$ met $a < 1$ is zij \emph{wel} \omterm{def:b2:funcseq:def}{uniform} ($\sup = a^n
\to 0$): uniformiteit is evenzeer een eigenschap van het domein als
van de rij.
\end{example}

\begin{example}[Twee limieten die weigeren te commuteren]\label{ex:b2:funcseq:doublelimit}
Het hele hoofdstuk gaat over het verwisselen van limieten, dus hier
is het kleinst mogelijke falen. Zij $a_{n,m} = \dfrac{n}{n+m}$ voor
$n, m \geq 1$. Dan is
\[
\lim_{m\to\infty}\Bigl(\lim_{n\to\infty}a_{n,m}\Bigr)
= \lim_{m\to\infty} 1 = 1,
\qquad
\lim_{n\to\infty}\Bigl(\lim_{m\to\infty}a_{n,m}\Bigr)
= \lim_{n\to\infty} 0 = 0 :
\]
beide herhaalde limieten bestaan en zij verschillen. Elke
overdrachtsstelling van dit hoofdstuk is een vergunning om twee
limieten te verwisselen --- $\lim_n$ met $\lim_{x\to a}$
(\omterm{def:b2:metric:continuity}{continuïteit}), met $\int$ (integratie), met $\frac{\dd}{\dd x}$
(differentiatie) --- en de \omterm{def:b2:funcseq:def}{uniforme convergentie} is precies het
tarief dat de verwisseling legaal maakt. Het inzicht om te
onthouden: zodra een ``bewijs'' stilzwijgend twee limietbewerkingen
omwisselt, is dit tweeregelige rooster het tegenvoorbeeld dat men
ertegenover moet zetten; de glijdende bulten van
\cref{exo:b2:funcseq:2} zijn hetzelfde verschijnsel in het kostuum
van een integraal.
\end{example}

\section{De drie overdrachtsstellingen}

\begin{theorem}[Continuïteit]\label{thm:b2:funcseq:continuity}
Is elke $f_n$ \omterm{def:b2:metric:continuity}{continu} in $a$ en geldt $f_n \to f$ \omterm{def:b2:funcseq:def}{uniform} op een
omgeving van $a$, dan is $f$ \omterm{def:b2:metric:continuity}{continu} in $a$. Een \omterm{def:b2:funcseq:def}{uniforme} limiet
van \omterm{def:b2:metric:continuity}{continue} functies is dus \omterm{def:b2:metric:continuity}{continu}.
\end{theorem}

\begin{proof}
Het $3\varepsilon$-argument dat al in
\cref{thm:b2:metric:rncomplete} is gebruikt: kies $n$ met
$\norm{f_n - f}_\infty \leq \varepsilon$, en daarna $\delta$ uit de
\omterm{def:b2:metric:continuity}{continuïteit} van $f_n$ in $a$; voor $\abs{x - a} \leq \delta$ is
\[
\abs{f(x) - f(a)} \leq \abs{f(x) - f_n(x)} + \abs{f_n(x) - f_n(a)}
+ \abs{f_n(a) - f(a)} \leq 3\varepsilon . \qedhere
\]
\end{proof}

\begin{example}[Uniformiteit faalt precies waar de limiet breekt]\label{ex:b2:funcseq:threshold}
Zij op $\intcc{0}{2}$ de rij $f_n(x) = \dfrac{x^n}{1 + x^n}$. De
\omterm{def:b2:funcseq:def}{puntsgewijze} limiet is een functie in drie stukken:
\[
f(x) = \begin{cases} 0 & 0 \leq x < 1,\\[2pt]
\tfrac12 & x = 1,\\[2pt]
1 & 1 < x \leq 2, \end{cases}
\]
discontinu in $1$, dus kan de convergentie volgens
\cref{thm:b2:funcseq:continuity} op $\intcc{0}{2}$ niet \omterm{def:b2:funcseq:def}{uniform}
zijn. Op de gesloten stukken die de drempel vermijden is zij dat
wel: voor $0 \leq x \leq a < 1$ is
\[
\sup_{\intcc{0}{a}}\abs{f_n - 0}
= \frac{a^n}{1 + a^n} \leq a^n \to 0 ,
\]
en voor $1 < b \leq x \leq 2$ is
\[
\sup_{\intcc{b}{2}}\abs{f_n - 1}
= \frac{1}{1 + b^n} \leq b^{-n} \to 0 ,
\]
beide suprema berekend met de monotonie van $u \mapsto
\frac{u}{1+u}$ en van $x \mapsto x^n$. Het inzicht om te onthouden:
het falen van de uniformiteit is gelokaliseerd bij de
discontinuïteit van de limiet --- dezelfde meetkunde als in
\cref{ex:b2:funcseq:xn}, en de reden dat de discipline ``\omterm{def:b2:funcseq:def}{uniform} op
elk segment binnenin'' het hele hoofdstuk terugkeert.
\end{example}

\begin{theorem}[Integratie op een segment]\label{thm:b2:funcseq:integration}
Geldt $f_n \to f$ \omterm{def:b2:funcseq:def}{uniform} op $\intcc{a}{b}$, met $f_n$ stuksgewijs
\omterm{def:b2:metric:continuity}{continu} (en $f$ eveneens), dan is
\[
\int_a^b f_n \longrightarrow \int_a^b f .
\]
\end{theorem}

\begin{proof}
De lineariteit en de driehoeksongelijkheid voor integralen geven
\[
\Bigl|\int_a^b f_n - \int_a^b f\Bigr|
= \Bigl|\int_a^b (f_n - f)\Bigr|
\leq \int_a^b\abs{f_n - f}
\leq (b - a)\,\norm{f_n - f}_\infty \longrightarrow 0 .
\]
De lengtefactor $(b - a)$ is de plaats waar de \omterm{def:b2:metric:compact}{compactheid} van het
segment binnenkomt: op \omterm{def:b2:metric:compact}{niet-compacte} intervallen levert dezelfde
schatting de nutteloze grens $\infty\cdot0$ op, en het besluit
faalt zonder dominatie werkelijk --- de vlakke bulten $f_n =
\frac1n\mathbf 1_{\intcc{0}{n}}$ convergeren \omterm{def:b2:funcseq:def}{uniform} naar $0$ op
$\intco{0}{\infty}$ en houden toch $\int f_n = 1$ vast (zie de
opmerking over valkuilen hieronder), en de glijdende bulten uit de
opmerking van \cref{ch:b2:integration} doen hetzelfde met
\omterm{def:b2:funcseq:def}{puntsgewijze convergentie}; \omterm{def:b2:funcseq:def}{uniforme convergentie} beheerst hoogten,
nooit breedten.
\end{proof}

\begin{theorem}[Differentiatie]\label{thm:b2:funcseq:differentiation}
Zij $f_n$ van klasse $C^1$ op een interval $I$, waarbij $(f_n')$
\emph{\omterm{def:b2:funcseq:def}{uniform}} op $I$ (of op elk segment van $I$) naar zekere $g$
convergeert en $(f_n(x_0))$ in één punt $x_0$ convergeert. Dan
convergeert $(f_n)$ (\omterm{def:b2:funcseq:def}{uniform} op segmenten) naar een
$C^1$-functie $f$, en geldt $f' = g$: men mag de limiet
differentiëren.
\end{theorem}

\begin{proof}
Definieer $f(x) = \lim f_n(x_0) + \int_{x_0}^x g$: dat is
legitiem, want $g$ is \omterm{def:b2:metric:continuity}{continu} --- zij is immers de \emph{\omterm{def:b2:funcseq:def}{uniforme}}
limiet op segmenten van de \omterm{def:b2:metric:continuity}{continue} $f_n'$, zodat
\cref{thm:b2:funcseq:continuity} van toepassing is, en de
integraal van een \omterm{def:b2:metric:continuity}{continue} functie is welgedefinieerd met, volgens
de hoofdstelling van de integraalrekening,
\[
f'(x) = g(x) \qquad (x \in I) :
\]
de kandidaat-limiet is dus \emph{per constructie} $C^1$ met de
juiste afgeleide, nog voordat er iets over convergentie is
bewezen. Opnieuw met de hoofdstelling is $f_n(x) = f_n(x_0) +
\int_{x_0}^x f_n'$; aftrekken geeft
\[
\abs{f_n(x) - f(x)} \leq \abs{f_n(x_0) - \lim f_n(x_0)}
+ \abs{x - x_0}\,\norm{f_n' - g}_{\infty} ,
\]
wat op elk segment \omterm{def:b2:funcseq:def}{uniform} naar $0$ gaat. En $f$ is per constructie
$C^1$ met $f' = g$.
\end{proof}

\begin{example}[Waarom de hypothese op de afgeleiden zit]\label{ex:b2:funcseq:sqrtabs}
Zij $F_n(x) = \sqrt{x^2 + \frac1n}$ op $\R$. Elke $F_n$ is $C^1$
(zelfs $C^\infty$), en de convergentie naar $\abs x$ is op heel
$\R$ \omterm{def:b2:funcseq:def}{uniform}:
\[
0 \leq F_n(x) - \abs x
= \frac{(x^2 + \frac1n) - x^2}{\sqrt{x^2+\frac1n} + \abs x}
= \frac{1/n}{\sqrt{x^2 + \frac1n} + \abs x}
\leq \frac{1/n}{1/\sqrt n} = \frac{1}{\sqrt n} .
\]
En toch is de limiet $\abs x$ in $0$ niet differentieerbaar: de
\omterm{def:b2:funcseq:def}{uniforme convergentie} van de \emph{functies}, hoe snel ook,
draagt geen differentieerbaarheid over. Het falen is aan de
afgeleiden zichtbaar:
\[
F_n'(x) = \frac{x}{\sqrt{x^2 + \frac1n}}
\longrightarrow \begin{cases} 1 & x > 0,\\ 0 & x = 0,\\
-1 & x < 0, \end{cases}
\]
een discontinue \omterm{def:b2:funcseq:def}{puntsgewijze} limiet, zodat $(F_n')$ bij $0$ niet
\omterm{def:b2:funcseq:def}{uniform} kan convergeren (opnieuw
\cref{thm:b2:funcseq:continuity}). Het inzicht om te onthouden:
\cref{thm:b2:funcseq:differentiation} veronderstelt bewust de
\omterm{def:b2:funcseq:def}{uniforme convergentie} van de $f_n'$ en niet die van de $f_n$ ---
en dit voorbeeld is de reden.
\end{example}

\section{Reeksen van functies}

\begin{definition}\label{def:b2:funcseq:series}
Een functiereeks $\sum u_n$ convergeert puntsgewijs of \omterm{def:b2:funcseq:def}{uniform}
wanneer haar partiële sommen dat doen. Zij convergeert
\emph{normaal}\index{normale convergentie} (op $X$) wanneer $\sum
\norm{u_n}_\infty < \infty$. Normale convergentie impliceert
\omterm{def:b2:funcseq:def}{uniforme convergentie} (in de \omterm{def:b2:nvs:banach}{banachruimte} van de begrensde
functies: \cref{thm:b2:nvs:absoluteconvergence}), die op haar beurt
\omterm{def:b2:funcseq:def}{puntsgewijze convergentie} impliceert; beide implicaties zijn
strikt.
\end{definition}

\begin{example}[Eén reeks, drie oordelen]\label{ex:b2:funcseq:threemodes}
Neem $u_n(x) = \frac{x^n}{n}$ op $\intco{0}{1}$.
\emph{Puntsgewijs:} zij convergeert voor elke $x \in \intco01$
(vergelijking met de meetkundige reeks). \emph{\omterm{def:b2:funcseq:series}{Normaal} op
$\intcc{0}{a}$ met $a < 1$:} $\norm{u_n}_{\infty,\intcc0a} =
\frac{a^n}{n}$, \omterm{def:b2:series:summable}{sommeerbaar}. \emph{Niet \omterm{def:b2:funcseq:series}{normaal} op
$\intco{0}{1}$:} $\norm{u_n}_{\infty,\intco01} = \frac1n$, en
$\sum\frac1n$ divergeert. \emph{Zelfs niet \omterm{def:b2:funcseq:def}{uniform} op
$\intco{0}{1}$:} de restsom verzet zich bij $1$,
\[
R_N(x) = \sum_{n>N}\frac{x^n}{n}
\geq \sum_{n=N+1}^{2N}\frac{x^n}{n}
\geq \frac{N\,x^{2N}}{2N} = \frac{x^{2N}}{2}
\xrightarrow[x\to1^-]{} \frac12 ,
\]
dus $\sup_{\intco01}\abs{R_N} \geq \frac12$ voor elke $N$. Het
inzicht om te onthouden: alle vier de oordelen bestaan vreedzaam
naast elkaar --- de som $-\ln(1-x)$ is \omterm{def:b2:metric:continuity}{continu} op $\intco{0}{1}$
omdat de \omterm{def:b2:metric:continuity}{continuïteit} alleen uniformiteit \emph{bij elk punt}
nodig heeft, dat wil zeggen op de segmenten $\intcc0a$; aan de rand
exploderen is het goed recht van de som.
\end{example}

\begin{theorem}[Overdracht voor reeksen]\label{thm:b2:funcseq:seriestransfer}
Convergeert $\sum u_n$ \omterm{def:b2:funcseq:def}{uniform} (bijvoorbeeld \omterm{def:b2:funcseq:series}{normaal}) op de
betreffende verzameling, dan gaat de \omterm{def:b2:metric:continuity}{continuïteit} van alle $u_n$ in
$a$ over op de som; mag de integratie op een segment term voor term
worden uitgevoerd; en is, wanneer $\sum u_n(x_0)$ convergeert
terwijl $\sum u_n'$ \omterm{def:b2:funcseq:def}{uniform} op segmenten convergeert, de som van
klasse $C^1$ met afgeleide $\sum u_n'$.
\end{theorem}

\begin{proof}
Alles is de bijbehorende stelling toegepast op de partiële sommen
$S_N = \sum_{n\leq N}u_n$, die eindige sommen van functies met de
betreffende regelmaat zijn. \omterm{def:b2:metric:continuity}{Continuïteit}: elke $S_N$ is \omterm{def:b2:metric:continuity}{continu} in
$a$ en $S_N \to \sum u_n$ \omterm{def:b2:funcseq:def}{uniform}:
\cref{thm:b2:funcseq:continuity}. Integratie: op het segment is
\[
\int_a^b \sum_{n\geq0} u_n
= \lim_N \int_a^b S_N
= \lim_N \sum_{n=0}^{N}\int_a^b u_n
= \sum_{n\geq0}\int_a^b u_n
\]
volgens \cref{thm:b2:funcseq:integration} (eerste gelijkheid) en
de lineariteit van de integraal (tweede). Differentiatie: de $S_N$
zijn $C^1$, $S_N(x_0)$ convergeert, en $S_N' = \sum_{n\leq N}u_n'$
convergeert \omterm{def:b2:funcseq:def}{uniform} op segmenten:
\cref{thm:b2:funcseq:differentiation} geeft dan dat de som $C^1$ is
met afgeleide $\lim S_N' = \sum u_n'$.
\end{proof}

\begin{remark}[Klassieke valkuilen]
Vier valstrikken, alle in tentamenwerk gezien. \emph{(i) Half
gecontroleerde suprema:} $f_n$ langs een goedgekozen rij $x_n$
evalueren begrenst $\norm{f_n - f}_\infty$ alleen van
\emph{onderen} --- genoeg om de uniformiteit te weerleggen (zoals
in \cref{ex:b2:funcseq:xn}), nooit om haar te bewijzen; om haar te
bewijzen moet men het supremum begrenzen met een berekening die
voor \emph{alle} $x$ geldig is. \emph{(ii) Uniformiteit op de
verkeerde verzameling:} normale of \omterm{def:b2:funcseq:def}{uniforme convergentie} geldt vaak
op elke $\intcc{-a}{a}$ of $\intco\delta\infty$ maar faalt op de
\omterm{def:b2:metric:topology}{open} vereniging; dat is geen bezwaar --- \omterm{def:b2:metric:continuity}{continuïteit} en
differentieerbaarheid zijn lokaal, dus levert de discipline segment
voor segment uit \cref{ex:b2:funcseq:zeta} ze op de hele \omterm{def:b2:metric:topology}{open}
verzameling. \emph{(iii) Integreren over niet-segmenten:}
\cref{thm:b2:funcseq:integration} is een uitspraak over segmenten;
op $\intco0\infty$ verhindert \omterm{def:b2:funcseq:def}{uniforme convergentie} niet dat massa
naar oneindig ontsnapt ($f_n = \frac1n\mathbf 1_{\intcc{0}{n}}$
convergeert \omterm{def:b2:funcseq:def}{uniform} naar $0$ terwijl $\int f_n = 1$) --- gebruik
daar de gedomineerde convergentie. \emph{(iv) De limiet
differentiëren:} \cref{ex:b2:funcseq:sqrtabs}; de hypothese over
afgeleiden staat op $(f_n')$, en geen enkele
convergentiesnelheid van $(f_n)$ kan haar vervangen.
\end{remark}

\begin{example}[De $\zeta$-functie van Riemann]\label{ex:b2:funcseq:zeta}
$\zeta(s) = \sum_{n\geq1} n^{-s}$ convergeert \omterm{def:b2:funcseq:series}{normaal} op elke
halfrechte $\intco{a}{+\infty}$ met $a > 1$ ($\norm{n^{-s}}_\infty
= n^{-a}$, \omterm{def:b2:series:summable}{sommeerbaar}): dus is $\zeta$ \omterm{def:b2:metric:continuity}{continu} op
$\intoo{1}{+\infty}$; en termsgewijs differentiëren (ook de
afgeleide reeks $\sum -\ln n\; n^{-s}$ convergeert \omterm{def:b2:funcseq:series}{normaal} op
$\intco{a}{\infty}$) maakt $\zeta$ van klasse $C^1$ --- en, na
itereren, $C^\infty$ --- met $\zeta'(s) = -\sum \frac{\ln n}{n^s}$.
Let op de discipline: de \omterm{def:b2:funcseq:series}{normale convergentie} wordt op
\emph{deel}halfrechten nagegaan, nooit op de \omterm{def:b2:metric:topology}{open} $\intoo{1}{\infty}$
zelf, waar zij faalt.
\end{example}

\begin{example}[Een logaritmische reeks, tot het eind uitgewerkt]\label{ex:b2:funcseq:logseries}
Zij $F(x) = \sum_{n\geq1} \frac{\eu^{-nx}}{n}$ op
$\intoo{0}{\infty}$. Elke term is op $\intco{\delta} \infty$
begrensd door $\frac{\eu^{-n\delta}}{n} \leq \eu^{-n\delta}$, een
convergente meetkundige reeks: dus \omterm{def:b2:funcseq:series}{normale convergentie} op elke
$\intco\delta\infty$, en $F$ is \omterm{def:b2:metric:continuity}{continu} op $\intoo{0}{\infty}$. De
afgeleide reeks $\sum -\eu^{-nx}$ convergeert eveneens \omterm{def:b2:funcseq:series}{normaal} op
$\intco\delta\infty$ ($\norm{\eu^{-nx}}_{\infty,\intco\delta\infty}
= \eu^{-n\delta}$), dus is $F$ van klasse $C^1$ met een
meetkundige afgeleide:
\[
F'(x) = -\sum_{n\geq1}\eu^{-nx}
= \frac{-\eu^{-x}}{1 - \eu^{-x}}
= \frac{-1}{\eu^{x} - 1} .
\]
Itereren maakt $F$ van klasse $C^\infty$. Het integreren van $F'$
(zowel $F$ als $x \mapsto -\ln(1 - \eu^{-x})$ verdwijnt bij
$+\infty$ en beide hebben dezelfde afgeleide op $\intoo0\infty$)
geeft
\[
F(x) = -\ln\bigl(1 - \eu^{-x}\bigr),
\]
de logaritmische reeks in $t = \eu^{-x}$. Het inzicht om te
onthouden: als $x \to 0^+$ is $F(x) = -\ln(x + O(x^2)) =
\ln\frac1x + O(x)$ --- de reeks divergeert logaritmisch aan de
rand, precies zoals de harmonische reeks die zij in $x = 0$ wordt;
\omterm{def:b2:funcseq:series}{normale convergentie} op $\intco\delta\infty$ maar niet op
$\intoo0\infty$ is het symptoom.
\end{example}

\begin{method}[Uniforme convergentie bewijzen of weerleggen]\label{met:b2:funcseq:uniform}
Voor $f_n \to f$ puntsgewijs op $X$:
\begin{enumerate}
  \item Bereken of begrens $\norm{f_n - f}_\infty$: bestudeer de
        functie $x \mapsto \abs{f_n(x) - f(x)}$ (afgeleide,
        monotonie) om haar maximum te lokaliseren; een grens die
        voor alle $x$ geldt en naar $0$ gaat, bewijst de
        uniformiteit.
  \item Om haar te \emph{weerleggen}: geef punten $x_n$ met
        $\abs{f_n(x_n) - f(x_n)} \not\to 0$ (vaak volgt $x_n$ de
        bewegende bult, zoals in \cref{exo:b2:funcseq:1}); of
        beroep je in contrapositie op een overdrachtsstelling ---
        een discontinue limiet van \omterm{def:b2:metric:continuity}{continue} functies
        (\cref{ex:b2:funcseq:threshold}), of $\int f_n \not\to
        \int f$ op een segment.
  \item Probeer bij reeksen eerst de \omterm{def:b2:funcseq:series}{normale convergentie}
        ($\sum\sup\abs{u_n} < \infty$); faalt zij globaal, test
        haar dan op de segmenten die ertoe doen
        (\cref{ex:b2:funcseq:threemodes}); faalt zij overal, dan
        kan de \omterm{def:b2:funcseq:def}{uniforme convergentie} nog steeds gelden via de
        grens voor de restsom van een alternerende reeks
        (\cref{exo:b2:funcseq:4}) of via partiële sommatie.
\end{enumerate}
\end{method}

\section{De benaderingsstelling van Weierstrass}

\begin{theorem}[Weierstrass, via Bernstein]\label{thm:b2:funcseq:weierstrass}
Elke \omterm{def:b2:metric:continuity}{continue} $f \colon \intcc{0}{1} \to \R$ is een \omterm{def:b2:funcseq:def}{uniforme} limiet
van veeltermen --- expliciet: van haar
\emph{Bernsteinveeltermen}\index{Bernsteinveeltermen}\index{benaderingsstelling
van Weierstrass}
\[
B_n(f)(x) = \sum_{k=0}^{n} f\Bigl(\frac kn\Bigr)\binom nk x^k
(1-x)^{n-k} .
\]
\end{theorem}

\begin{proof}
Leg $x \in \intcc{0}{1}$ vast en zet $p_k(x) = \binom nk x^k(1 -
x)^{n-k}$. Drie binomiale identiteiten, verkregen door $(x + y)^n$
en haar twee afgeleiden naar $x$ in $y = 1 - x$ te evalueren:
\[
\sum_k p_k = 1,
\qquad
\sum_k k\,p_k = nx,
\qquad
\sum_k k(k-1) p_k = n(n-1)x^2 .
\]
In detail: $(x+y)^n = \sum_k\binom nk x^ky^{n-k}$ in $y = 1-x$ is
de eerste; differentiëren naar $x$ geeft
\[
n(x+y)^{n-1} = \sum_k k\binom nk x^{k-1}y^{n-k} ,
\]
en daarna met $x$ vermenigvuldigen en $y = 1 - x$ zetten geeft de
tweede; twee keer differentiëren en met $x^2$ vermenigvuldigen
geeft de derde. Werk $(k - nx)^2 = k(k-1) + k(1 - 2nx) + n^2x^2$
uit en combineer de drie:
\[
\sum_k (k - nx)^2 p_k
= n(n-1)x^2 + nx(1 - 2nx) + n^2x^2
= nx(1 - x) \leq \frac n4 ,
\]
de \emph{variantie-identiteit}.

Schat nu af, met $\sum p_k = 1$:
\[
\abs{B_n(f)(x) - f(x)}
\leq \sum_{k} \Bigl| f\Bigl(\frac kn\Bigr) - f(x)\Bigr|\, p_k(x)
= \Sigma_{\text{dichtbij}} + \Sigma_{\text{veraf}} ,
\]
gesplitst naargelang $\abs{\frac kn - x} \leq \delta$ of niet. Bij
gegeven $\varepsilon > 0$ levert de uniforme \omterm{def:b2:metric:continuity}{continuïteit} van $f$
(Heine) een $\delta$ met $\Sigma_{\text{dichtbij}} \leq
\varepsilon$. Voor de verre som, met $M = \norm f_\infty$: met de
variantie-identiteit en de teltruc van Chebyshev is
\[
\Sigma_{\text{veraf}} \leq 2M \sum_{\abs{k - nx} > n\delta} p_k
\leq 2M\,\frac{\sum_k (k - nx)^2 p_k}{n^2\delta^2}
\leq \frac{2M}{4 n \delta^2} \cdot 1
= \frac{M}{2n\delta^2}
\xrightarrow[n\to\infty]{} 0 ,
\]
\omterm{def:b2:funcseq:def}{uniform} in $x$. Dus $\norm{B_n(f) - f}_\infty \leq \varepsilon +
\frac{M}{2n\delta^2} \leq 2\varepsilon$ voor grote $n$.
\end{proof}

\begin{remark}
Met een affiene substitutie geldt de stelling op elk segment
$\intcc{a}{b}$. Zij faalt op $\R$ (een \omterm{def:b2:funcseq:def}{uniforme} limiet van
veeltermen op $\R$ is een veelterm: \cref{exo:b2:funcseq:8}). De
kanstheoretische lezing --- $B_n(f)(x)$ is de verwachtingswaarde
van $f$ in een binomiaal gemiddelde, en de variantiegrens is de
ongelijkheid van Chebyshev --- wordt in \cref{ch:b2:genfun}
eerlijk gemaakt.
\end{remark}

\begin{remark}[Waar dit wordt gebruikt]
De benaderingsstelling van Weierstrass is de dichtheidsstelling van
de klassieke analyse: zij maakt $C(\intcc ab)$ separabel, laat toe
integraalidentiteiten alleen op veeltermen te controleren
(momentenproblemen), en ligt onder de goniometrische versie die in
het fourierhoofdstuk met de kern van Fejér wordt bewezen. De
weekendopgave van dit hoofdstuk haalt de kwantitatieve inhoud uit
het bewijs van Bernstein --- convergentiesnelheden bestuurd door de
continuïteitsmodulus --- en isoleert daarna wat het bewijs werkelijk
deed werken, in de stelling van Korovkin: positiviteit plus drie
testfuncties. Het volume van bachelorjaar 3 veralgemeent de
dichtheidsuitspraak tot willekeurige deelalgebra's
(Stone--Weierstrass) en tot \omterm{def:b2:metric:compact}{compacte} ruimten.
\end{remark}

\begin{example}[Veelhoekige benadering, met een snelheid]\label{ex:b2:funcseq:polygonal}
Zij $f$ $L$-Lipschitz op $\intcc{0}{1}$ en $I_nf$ de stuksgewijs
affiene interpolant in de knooppunten $\frac kn$. Op een cel
$\intcc{\frac kn}{\frac{k+1}n}$ liggen zowel $f(x)$ als $I_nf(x)$
tussen de extreme waarden die een $L$-Lipschitz-functie bij de twee
knoopwaarden kan aannemen, dus voor $x$ in de cel geldt met $x_k =
\frac kn$
\[
\abs{I_nf(x) - f(x)}
\leq \abs{I_nf(x) - f(x_k)} + \abs{f(x_k) - f(x)}
\leq L\,\abs{x - x_k} + L\,\abs{x - x_k}
\leq \frac{2L}{n}
\]
(de interpolant is op de cel zelf $L$-Lipschitz: haar helling is
een differentiequotiënt van $f$). Bijgevolg is $\norm{I_nf -
f}_\infty \leq \frac{2L}{n}$: de veelhoekige benadering van
Lipschitz-functies convergeert met snelheid $\frac1n$ ---
\emph{sneller} dan de $\frac{1}{\sqrt n}$ van Bernstein voor
dezelfde klasse (weekendopgave, Deel II). Het inzicht om te
onthouden: de veelhoek interpoleert maar is niet glad, Bernstein is
glad maar traag; tussen de regelmaat van de benadering en haar
snelheid bestaat geen gratis lunch --- een afweging die de
verzadigingsresultaten van de weekendopgave precies maken.
\end{example}

\begin{remark}[Vooruitblik binnen dit volume]
De \omterm{def:b2:funcseq:def}{uniforme convergentie} is vanaf hier het werkpaard van dit boek.
Het hoofdstuk over machtreeksen draait \omterm{def:b2:metric:complete}{volledig} op \omterm{def:b2:funcseq:series}{normale

stelling daar is een bijzonder geval van de overdrachtsstellingen

sprongen), en zijn Fejér-middelen slagen dankzij dezelfde
$3\varepsilon$-mechaniek die \cref{thm:b2:funcseq:continuity}
bewees. Het hoofdstuk over differentiaalvergelijkingen definieert

\cref{thm:b2:funcseq:seriestransfer} toegepast op de

hoofdstuk.
\end{remark}

(Een \omterm{def:b2:metric:continuity}{continue}, nergens differentieerbare functie --- begeleid) Zij
$\varphi$ de afstand tot het dichtstbijzijnde gehele getal
($1$-periodiek, $\norm{\varphi}_\infty = \frac12$,
$1$-Lipschitz) en
\[
W(x) = \sum_{n=0}^{\infty} \Bigl(\frac{3}{4}\Bigr)^{\!n}
\varphi(4^n x) .
\]
Bewijs: (a) $W$ is \omterm{def:b2:metric:continuity}{continu} op $\R$ (\omterm{def:b2:funcseq:series}{normale convergentie}); (b) voor
elke $x$ en elke $m$ voldoet het differentiequotiënt, met $h_m =

(de termen $n > m$ verdwijnen wegens de periodiciteit; de term $n =
m$ draagt precies $3^m$ bij; de termen $n < m$ worden door de
Lipschitz-eigenschap begrensd). Besluit dat $W$ nergens
differentieerbaar is.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:funcseq:12}
(Een integraalvergelijking van Volterra met reeksen) Definieer voor
$f \in C(\intcc{0}{1})$ de afbeelding $Tf(x) = \int_0^x f(t)\,\dd
t$.
\begin{enumerate}
  \item Toon met inductie aan dat voor $n \geq 1$
        \[
        T^n f(x) = \int_0^x \frac{(x -
        t)^{n-1}}{(n-1)!}\,f(t)\,\dd t,
        \qquad
        \norm{T^n f}_\infty \leq \frac{\norm f_\infty}{n!} .
        \]
  \item Leid af dat $S = \sum_{n\geq0} T^n f$ \omterm{def:b2:funcseq:series}{normaal} convergeert
        op $\intcc{0}{1}$ en de integraalvergelijking $S = f + TS$
        oplost.
  \item Ga na dat $S(x) = f(x) + \int_0^x \eu^{x-t}f(t)\,\dd t$
        dezelfde vergelijking oplost, en bewijs de eenduidigheid
        van de \omterm{def:b2:metric:continuity}{continue} oplossingen \emph{(is $S = TS$, dan
        $\norm{S}_\infty \leq \norm{T^nS}_\infty \to 0$)}: besluit
        tot de gesloten vorm van de som.
\end{enumerate}
\end{exercise}

\section{Probleem: Benaderingssnelheden en de stelling van Korovkin}

\begin{problem}\label{pb:b2:funcseq:1}
Het bewijs van Bernstein van \cref{thm:b2:funcseq:weierstrass}
verbergt twee schatten. Ten eerste is het \emph{kwantitatief}: hoe
snel $B_nf \to f$ gaat, wordt bestuurd door de
continuïteitsmodulus van $f$, met de scherpe snelheid aangenomen
door $\abs{x - \frac12}$. Ten tweede is het \emph{structureel}: het
enige wat ertoe deed, was dat $B_n$ een positieve lineaire operator
is die zich netjes gedraagt op $1$, $x$ en $x^2$ --- die
waarneming, geïsoleerd, is de \emph{stelling van
Korovkin}\index{stelling van Korovkin}. Deze opgave bewijst beide,
en sluit af met de exacte asymptotiek van Voronovskaja. Overal is
$f \in C(\intcc{0}{1})$, $M = \norm f_\infty$, $p_k(x) = \binom nk
x^k(1-x)^{n-k}$, en staat $e_j$ voor $x \mapsto x^j$.

\medskip
\noindent\textbf{Deel I --- De operator van Bernstein.}
\begin{enumerate}
  \item Toon aan dat $B_n$ lineair en \emph{positief} is ($f \geq 0
        \Rightarrow B_nf \geq 0$), dus monotoon ($f \leq g
        \Rightarrow B_nf \leq B_ng$), met $\norm{B_nf}_\infty \leq
        \norm f_\infty$, en dat $B_nf$ in beide eindpunten met $f$
        samenvalt.
  \item Leid de identiteiten $B_n e_0 = e_0$, $B_n e_1 = e_1$ en
        $B_n e_2 = e_2 + \frac{e_1 - e_2}{n}$ opnieuw af
        \emph{(differentieer $(x + y)^n$ tweemaal en zet $y = 1 -
        x$)}.
  \item Leid de variantie-identiteit $\sum_k \bigl(\frac kn -
        x\bigr)^2 p_k(x) = \frac{x(1-x)}{n}$ af, en met
        Cauchy--Schwarz de grens voor het eerste moment
        \[
        \sum_{k=0}^{n}\Bigl|\frac kn - x\Bigr|\,p_k(x)
        \leq \sqrt{\frac{x(1-x)}{n}} \leq \frac{1}{2\sqrt n} .
        \]
  \item Toon aan dat $B_nf \geq f$ op $\intcc{0}{1}$ zodra $f$
        convex is \emph{(de eindige ongelijkheid van Jensen voor de
        gewichten $p_k(x)$)}.
  \item (De telgrens van Chebyshev, herformuleerd) Toon voor
        $\delta > 0$ aan dat
        \[
        \sum_{\abs{k/n - x} > \delta} p_k(x)
        \leq \frac{x(1-x)}{n\delta^2}
        \leq \frac{1}{4n\delta^2} ,
        \]
        en geef de kanstheoretische lezing: $B_nf(x)$ middelt $f$
        over een binomiaal steekproefgemiddelde dat zich rond $x$
        concentreert.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Snelheden: de continuïteitsmodulus.}
Zet voor $\delta > 0$ de waarde $\omega(\delta) = \sup\{\abs{f(s) -
f(t)} : s, t \in \intcc{0}{1},\ \abs{s - t} \leq \delta\}$.
\begin{enumerate}[resume]
  \item Toon aan: $\omega$ is eindig en niet-dalend,
        $\omega(\delta) \to 0$ als $\delta \to 0^+$ (Heine),
        $\omega$ is subadditief ($\omega(\delta_1 + \delta_2) \leq
        \omega(\delta_1) + \omega(\delta_2)$), en
        $\omega(\lambda\delta) \leq (1 + \lambda)\,\omega(\delta)$
        voor alle $\lambda > 0$.
  \item Bewijs de hoofdschatting, voor elke $\delta > 0$:
        \[
        \abs{B_nf(x) - f(x)}
        \leq \sum_k \omega\Bigl(\Bigl|\frac kn -
        x\Bigr|\Bigr)p_k(x)
        \leq \Bigl(1 + \frac1\delta\sum_k\Bigl|\frac kn -
        x\Bigr|p_k(x)\Bigr)\,\omega(\delta) .
        \]
  \item Kies $\delta = n^{-1/2}$ en besluit tot de
        \emph{kwantitatieve stelling van Weierstrass}:
        \[
        \norm{B_nf - f}_\infty \leq
        \frac32\,\omega\Bigl(\frac{1}{\sqrt n}\Bigr)
        \xrightarrow[n\to\infty]{} 0 .
        \]
  \item Leid de snelheden af: $\norm{B_nf - f}_\infty \leq
        \frac{3L}{2\sqrt n}$ voor $L$-Lipschitz $f$, en $\leq
        \frac32 C n^{-\alpha/2}$ voor $\alpha$-höldercontinue $f$
        ($\abs{f(s) - f(t)} \leq C\abs{s-t}^\alpha$).
  \item (Het scherpe voorbeeld --- een binomiale identiteit) Bewijs
        voor $m \geq 1$ dat
        \[
        \sum_{k=m+1}^{2m} (k - m)\binom{2m}{k}
        = \frac{m}{2}\binom{2m}{m},
        \qquad\text{dus}\qquad
        \sum_{k=0}^{2m}\abs{k - m}\binom{2m}{k}
        = m\binom{2m}{m}
        \]
        \emph{(gebruik $k\binom{2m}k = 2m\binom{2m-1}{k-1}$ en de
        symmetrie van de binomiaalrij, die $\sum_{j=m}^{2m-1}
        \binom{2m-1}{j} = 2^{2m-2}$ geeft)}.
  \item Leid voor $f(t) = \abs{t - \frac12}$ de exacte waarde en
        haar asymptotiek af (centrale binomiaalcoëfficiënt,
        \cref{ex:b2:comparison:centralbinomial}):
        \[
        B_{2m}f\Bigl(\frac12\Bigr) - f\Bigl(\frac12\Bigr)
        = \frac{\binom{2m}{m}}{2\cdot4^{m}}
        \;\sim\; \frac{1}{2\sqrt{\pi m}} :
        \]
        de snelheid $\omega(n^{-1/2})$ uit vraag 8 wordt (op een
        constante na) aangenomen --- voor louter \omterm{def:b2:metric:continuity}{continue} $f$ is
        de $n^{-1/2}$ van Bernstein eerlijk.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De stelling van Korovkin.} Zij
$(L_n)$ een rij \emph{positieve lineaire} operatoren van
$C(\intcc{0}{1})$ naar zichzelf zodanig dat $L_ne_j \to e_j$
\omterm{def:b2:funcseq:def}{uniform} voor $j = 0, 1, 2$.
\begin{enumerate}[resume]
  \item Toon aan dat een positieve lineaire $L$ monotoon is en
        puntsgewijs aan $\abs{Lf} \leq L\abs f$ voldoet.
  \item Toon aan: bij elke $\varepsilon > 0$ is er een $\delta > 0$
        zodanig dat voor \emph{alle} $s, x \in \intcc{0}{1}$
        \[
        \abs{f(s) - f(x)} \leq \varepsilon +
        \frac{2M}{\delta^2}(s - x)^2
        \]
        \emph{(behandel $\abs{s - x} \leq \delta$ met Heine en
        $\abs{s-x} > \delta$ met de grove grens $2M$)}.
  \item Leg $x$ vast, pas $L_n$ toe op de ongelijkheid van vraag 13
        in de veranderlijke $s$, en leid af dat
        \[
        \abs{L_nf(x) - f(x)\,L_ne_0(x)}
        \leq \varepsilon\,L_ne_0(x) + \frac{2M}{\delta^2}
        \bigl(L_ne_2(x) - 2x\,L_ne_1(x) + x^2 L_ne_0(x)\bigr).
        \]
  \item Toon aan dat $\sup_x \bigl(L_ne_2(x) - 2x\,L_ne_1(x) + x^2
        L_ne_0(x)\bigr) \to 0$, en stel daarna de \emph{stelling
        van Korovkin} samen: $L_nf \to f$ \omterm{def:b2:funcseq:def}{uniform} voor
        \emph{elke} $f \in C(\intcc{0}{1})$.
  \item Ga na dat $(B_n)$ aan de hypothesen van Korovkin voldoet:
        Weierstrass voor de derde keer, uit drie monomen.
  \item Zij $I_n$ de operator van de stuksgewijs affiene
        interpolatie in de knooppunten $\frac kn$. Toon aan dat
        $I_n$ positief en lineair is, dat $I_ne_0 = e_0$ en
        $I_ne_1 = e_1$, en dat $\norm{I_ne_2 - e_2}_\infty =
        \frac{1}{4n^2}$ \emph{(op elke cel is de fout van de
        affiene interpolatie van $t^2$ gelijk aan $(t - a)(b -
        t)$)}. Besluit met Korovkin: de veelhoekige interpolanten
        convergeren \omterm{def:b2:funcseq:def}{uniform} voor elke \omterm{def:b2:metric:continuity}{continue} $f$.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Opbrengsten: dichtheid, momenten,
afgeleiden.}
\begin{enumerate}[resume]
  \item Toon aan dat de veeltermen met \emph{rationale}
        coëfficiënten dicht liggen in $\bigl(C(\intcc{0}{1}),
        \norm\cdot_\infty\bigr)$: deze \omterm{def:b2:nvs:banach}{banachruimte} is dus
        separabel.
  \item (De momenten bepalen de functie) Zij $f \in
        C(\intcc{0}{1})$ met $\int_0^1 f(t)\,t^n \dd t = 0$ voor
        elke $n \in \N$. Toon aan dat $\int_0^1 f P = 0$ voor elke
        veelterm, daarna dat $\int_0^1 f^2 = 0$, en ten slotte dat
        $f = 0$.
  \item Bewijs de identiteit voor de afgeleide
        \[
        (B_nf)'(x) = n\sum_{k=0}^{n-1}\Bigl(
        f\Bigl(\frac{k+1}{n}\Bigr) -
        f\Bigl(\frac kn\Bigr)\Bigr)\,
        \binom{n-1}{k}x^k(1-x)^{n-1-k}
        \]
        \emph{(differentieer $p_k$ en herindexeer --- een
        Abelsommatie)}.
  \item Neem aan dat $f$ van klasse $C^1$ is. Toon met de
        middelwaardestelling in elke toename en een vergelijking
        met $B_{n-1}(f')$ aan dat $(B_nf)' \to f'$ \omterm{def:b2:funcseq:def}{uniform} op
        $\intcc{0}{1}$. Leid af: voor $f \in C^1$ bestaan er
        veeltermen die naar $f$ convergeren \emph{samen met} hun
        afgeleiden.
  \item Neem aan dat $f$ van klasse $C^2$ is. Toon met
        Taylor--Lagrange in $x$ aan dat
        \[
        \abs{B_nf(x) - f(x)} \leq
        \frac{\norm{f''}_\infty}{2}\cdot\frac{x(1-x)}{n}
        \leq \frac{\norm{f''}_\infty}{8n} :
        \]
        gladheid tilt de snelheid op van $n^{-1/2}$ naar
        $n^{-1}$.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Verzadiging: de stelling van
Voronovskaja.}
\begin{enumerate}[resume]
  \item Bewijs de identiteit voor het vierde moment
        \[
        \sum_k (k - nx)^4 p_k(x)
        = nx(1-x)\bigl(1 + 3(n-2)x(1-x)\bigr) \leq n^2
        \quad (n \geq 1)
        \]
        \emph{(werk $k^4$ uit in dalende faculteiten $k(k-1)\cdots$
        en gebruik de differentiatietruc van vraag 2 nog twee
        keer)}.
  \item (Voronovskaja) Zij $f$ van klasse $C^2$ en $x \in
        \intcc{0}{1}$. Bewijs, met $f(t) = f(x) + f'(x)(t-x) +
        \frac{f''(x)}2(t-x)^2 + \eta(t)(t-x)^2$ waarbij $\eta$
        begrensd is en $\eta(t) \to 0$ als $t \to x$, dat
        \[
        n\bigl(B_nf(x) - f(x)\bigr)
        \xrightarrow[n\to\infty]{}
        \frac{x(1-x)}{2}\,f''(x)
        \]
        \emph{(splits de $\eta$-som bij $\abs{t - x} \leq \delta$;
        beheers het verre deel met vraag 23)}. De fout van vraag
        22 is dus exact in orde \emph{én} in constante: $B_n$
        \emph{verzadigt} bij $\frac1n$, hoe glad $f$ ook is ---
        vergelijk \cref{exo:b2:funcseq:7}.
  \item Synthese. In telkens één zin: (i) wat de positiviteit
        alleen al opleverde (de Delen I en III); (ii) waar de
        \omterm{def:b2:metric:compact}{compactheid} van $\intcc{0}{1}$ in elk deel binnenkwam;
        (iii) waarom drie testfuncties in de stelling van Korovkin
        volstaan; (iv) de afweging die Bernstein maakt (een robuuste
        $n^{-1/2}$ voor ruwe $f$, maar een plafond van $\frac1n$
        voor gladde $f$), en welk hoofdstuk van dit boek hetzelfde
        spel met goniometrische veeltermen zal spelen.
\end{enumerate}
\end{problem}
