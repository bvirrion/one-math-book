\chapter{Integratie}\label{ch:b2:integration}

Het volume van bachelorjaar 1 bouwde de integraal op een segment.
Dit hoofdstuk breidt haar uit tot willekeurige intervallen
(oneigenlijke integralen, met de volledige
vergelijkingsgereedschapskist) en bestudeert daarna integralen die
van een \emph{parameter} afhangen --- \omterm{def:b2:metric:continuity}{continuïteit} en
differentiëren onder het integraalteken --- aangedreven door de
stelling van de gedomineerde convergentie, het enige resultaat van
dit hoofdstuk dat op vertrouwen wordt aangenomen. De
$\Gamma$-functie dient als doorlopend voorbeeld en als poort naar de
helft van de speciale functies van de wiskunde.

\section{Integralen op een willekeurig interval}

\begin{definition}\label{def:b2:integration:improper}
Zij $f$ stuksgewijs \omterm{def:b2:metric:continuity}{continu} op $\intco{a}{b}$ ($b \in \R$ of
$+\infty$). De integraal
\emph{convergeert}\index{oneigenlijke integraal} wanneer $\lim_{x
\to b^-} \int_a^x f$ bestaat; men schrijft dan $\int_a^b f$ voor de
limiet. (Evenzo op $\intoc{a}{b}$, en op $\intoo{a}{b}$ door in een
inwendig punt te splitsen --- de keuze doet er dankzij Chasles niet
toe.) De integraal \emph{convergeert \omterm{def:b2:series:def}{absoluut}} wanneer $\int_a^b
\abs f$ convergeert; \omterm{def:b2:series:def}{absolute convergentie} impliceert convergentie,
via het cauchycriterium:
\[
\Bigl| \int_x^{y} f \Bigr| \leq \int_x^{y} \abs f
\]
en de \omterm{def:b2:metric:complete}{volledigheid} van $\R$ (de primitieve heeft de
cauchy-eigenschap). In detail: zij $F(x) = \int_a^x f$ en $G(x) =
\int_a^x \abs f$. Convergeert $\int^b\abs f$, dan heeft $G$ een
limiet in $b^-$, dus is er bij elke $\varepsilon > 0$ een $c < b$
met $G(y) - G(x) \leq \varepsilon$ zodra $c \leq x \leq y < b$; de
formule hierboven draagt die cauchy-eigenschap over op $F$. Voor
elke rij $x_n \to b^-$ vormen de waarden $F(x_n)$ dan een cauchyrij
reële getallen, convergent wegens de \omterm{def:b2:metric:complete}{volledigheid}, en het
vervlechten van twee zulke rijen laat zien dat de limiet voor alle
rijen dezelfde is: $F$ heeft dus een limiet in $b^-$.
\end{definition}

\begin{theorem}[Vergelijkingsgereedschap voor positieve functies]\label{thm:b2:integration:comparison}
Voor $f, g \geq 0$ stuksgewijs \omterm{def:b2:metric:continuity}{continu} op $\intco{a}{b}$:
\begin{enumerate}
  \item $\int_a^b f$ \omterm{def:b2:integration:improper}{convergeert} dan en slechts dan als de
        primitieve $x \mapsto \int_a^x f$ begrensd is;
  \item is $f \leq g$, dan dwingt de convergentie van $\int g$ die
        van $\int f$ af; de divergentie draagt de andere kant op
        over;
  \item is $f \sim g$ bij $b$, dan hebben de twee integralen
        dezelfde aard;
  \item de referentieschalen: bij $+\infty$ \omterm{def:b2:integration:improper}{convergeert}
        $\int^{\infty} \frac{\dd t}{t^\alpha}$ dan en slechts dan
        als $\alpha > 1$, en $\int^\infty \frac{\dd t}{t(\ln
        t)^\beta}$ dan en slechts dan als $\beta > 1$; bij een
        eindig eindpunt $b$ \omterm{def:b2:integration:improper}{convergeert} $\int^b \frac{\dd t}{(b -
        t)^\alpha}$ dan en slechts dan als $\alpha < 1$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) De primitieve $F(x) = \int_a^x f$ is niet-dalend (want $f \geq
0$). Is zij begrensd, dan is $\ell = \sup_{x < b}F$ eindig en $F(x)
\to \ell$: bij gegeven $\varepsilon > 0$ is er een $F(x_0) > \ell -
\varepsilon$, en de monotonie sluit $F(x) \in \intoc{\ell -
\varepsilon}{\ell}$ in voor $x_0 \leq x < b$. Is zij onbegrensd,
dan is $F \to +\infty$: divergentie.

(2) Uit $f \leq g$ volgt $\int_a^x f \leq \int_a^x g$ voor alle
$x$; \omterm{def:b2:integration:improper}{convergeert} $\int^b g$, dan is het rechterlid begrensd en dus
ook het linkerlid, en besluit (1). Contrapositie draagt de
divergentie de andere kant op over.

(3) Uit $f \sim g$ bij $b$ volgt het bestaan van een $c < b$ met
\[
\tfrac12\,g(t) \;\leq\; f(t) \;\leq\; 2\,g(t)
\qquad (c \leq t < b) :
\]
volgens (2), in beide richtingen toegepast op $\intco{c}{b}$,
hebben de twee integralen dezelfde aard; het beginstuk
$\intcc{a}{c}$ is een eigenlijke integraal en verandert niets.

(4) Expliciete primitieven: voor $\alpha \neq 1$ en $\beta \neq 1$
is
\[
\int_c^x \frac{\dd t}{t^\alpha}
= \frac{x^{1-\alpha} - c^{1-\alpha}}{1 - \alpha},
\qquad
\int_c^x \frac{\dd t}{t(\ln t)^\beta}
= \frac{(\ln x)^{1-\beta} - (\ln c)^{1-\beta}}{1 - \beta},
\]
met logaritmen in de uitgezonderde gevallen: begrensd als $x \to
+\infty$ precies wanneer $\alpha > 1$ respectievelijk $\beta > 1$.
Bij een eindig eindpunt herleidt de substitutie $u = b - t$ alles
tot de schaal $\int_0 u^{-\alpha}\,\dd u$, begrensd dan en slechts
dan als $\alpha < 1$. Pas telkens (1) toe.
\end{proof}

\begin{example}[Twee opwarmers, tot het eind uitgewerkt]\label{ex:b2:integration:warmups}
\emph{(a)} $\displaystyle\int_0^1 \ln t\,\dd t$: de integrand
explodeert in $0^+$, maar daar is $\abs{\ln t} =
o\bigl(t^{-1/2}\bigr)$ (logaritmen verliezen van machten) en
$\int_0 t^{-1/2}$ \omterm{def:b2:integration:improper}{convergeert}: dus \omterm{def:b2:series:def}{absolute convergentie}. De
waarde, met partiële integratie op $\intcc{\varepsilon}{1}$:
\[
\int_\varepsilon^1 \ln t\,\dd t
= \bigl[t\ln t - t\bigr]_\varepsilon^1
= -1 - \varepsilon\ln\varepsilon + \varepsilon
\xrightarrow[\varepsilon\to0^+]{} -1 .
\]
\emph{(b)} $\displaystyle\int_0^\infty \frac{\ln t}{1 +
t^2}\,\dd t$: problemen aan beide uiteinden, dus splits bij $1$.
Bij $0$ is $\abs{\ln t}$ integreerbaar als in (a); bij $\infty$ is
$\frac{\ln t}{1+t^2} = o(t^{-3/2})$: dus \omterm{def:b2:series:def}{absoluut} convergent. De
substitutie $t = \frac1u$ beeldt $\intoo{0}{1}$ af op
$\intoo{1}{\infty}$ en
\[
\int_0^1 \frac{\ln t}{1+t^2}\,\dd t
= \int_1^{\infty} \frac{-\ln u}{1 + u^{-2}}\cdot
\frac{\dd u}{u^2}
= -\int_1^\infty \frac{\ln u}{1+u^2}\,\dd u :
\]
de twee helften heffen elkaar op, en de integraal is $0$. Het
inzicht om te onthouden: de symmetrie onder $t \mapsto \frac1t$ is
een bladzijde rekenwerk waard --- dezelfde truc dreef al
\cref{exo:b2:integration:3} aan.
\end{example}

\begin{example}[Eén waarde, drie integralen]\label{ex:b2:integration:onecos}
Bestudeer $I = \displaystyle\int_0^{\infty} \frac{1 - \cos
t}{t^2}\,\dd t$. Bij $0$ is $1 - \cos t \sim \frac{t^2}2$, dus
laat de integrand zich \omterm{def:b2:metric:continuity}{continu} voortzetten met de waarde $\frac12$
--- helemaal geen singulariteit. Bij $\infty$ is $0 \leq \frac{1 -
\cos t}{t^2} \leq \frac{2}{t^2}$: \omterm{def:b2:series:def}{absolute convergentie}
(\cref{thm:b2:integration:comparison}). De waarde: integreer
partieel op $\intcc{\varepsilon}{M}$ met $u = 1 - \cos t$ en $v' =
t^{-2}$:
\[
\int_\varepsilon^M \frac{1 - \cos t}{t^2}\,\dd t
= \Bigl[-\frac{1 - \cos t}{t}\Bigr]_\varepsilon^M
+ \int_\varepsilon^M \frac{\sin t}{t}\,\dd t .
\]
De haak verdwijnt aan beide uiteinden ($\frac{1 -
\cos\varepsilon}{\varepsilon} \sim \frac\varepsilon2$; begrensde
teller bij $M$), en de integraal gaat naar de waarde van Dirichlet
$\frac\pi2$ (\cref{exo:b2:integration:10}): dus $I = \frac\pi2$.
Het inzicht om te onthouden: met $1 - \cos t = 2\sin^2\frac t2$ en
$u = \frac t2$ is
\[
I = \int_0^\infty \frac{2\sin^2 u}{(2u)^2}\;2\,\dd u
= \int_0^\infty \Bigl(\frac{\sin u}{u}\Bigr)^{\!2}\dd u :
\]
de drie klassiekers $\int_0^\infty\frac{\sin t}{t}\dd t$,
$\int_0^\infty\bigl(\frac{\sin t}{t}\bigr)^2\dd t$
(\cref{exo:b2:integration:11}) en $I$ delen alle de waarde
$\frac\pi2$, doorgegeven met partiële integratie en substitutie ---
en alleen de eerste is halfconvergent: de partiële integratie ruilde
de \omterm{def:b2:series:def}{absolute convergentie} in voor een eenvoudiger integrand.
\end{example}

\begin{example}[Een halfconvergente integraal]\label{ex:b2:integration:sint}
$\displaystyle\int_1^{\infty} \frac{\sin t}{t}\,\dd t$
\omterm{def:b2:integration:improper}{convergeert}: integreer partieel,
\[
\int_1^x \frac{\sin t}{t}\dd t
= \Bigl[\frac{-\cos t}{t}\Bigr]_1^x - \int_1^x \frac{\cos
t}{t^2}\dd t ,
\]
waarbij de haak een limiet heeft en de laatste integraal \omterm{def:b2:series:def}{absoluut}
\omterm{def:b2:integration:improper}{convergeert} ($\abs{\cos t}/t^2 \leq t^{-2}$). Maar niet \omterm{def:b2:series:def}{absoluut}:
uit $\abs{\sin t} \geq \sin^2 t$ volgt
\[
\int_1^x \frac{\abs{\sin t}}{t}\,\dd t
\;\geq\; \int_1^x \frac{\sin^2t}{t}\,\dd t
= \underbrace{\int_1^x \frac{\dd t}{2t}}_{=\ \frac12\ln x
\ \to\ \infty}
\;-\; \underbrace{\int_1^x \frac{\cos 2t}{2t}\,\dd
t}_{\text{convergent}} ,
\]
waarbij de laatste integraal \omterm{def:b2:integration:improper}{convergeert} dankzij \emph{dezelfde}
partiële integratie als hierboven (met $\sin 2t$ in de haak): een
divergent stuk min een convergent stuk divergeert. Dus \omterm{def:b2:integration:improper}{convergeert}
$\int_1^\infty\frac{\sin t}{t}\dd t$ zonder \omterm{def:b2:series:def}{absoluut} te
convergeren --- de integraaltegenhanger van de alternerende reeks,
waarbij de partiële integratie de rol van de alternerende test
speelt.
\end{example}

\section{De convergentiestelling}

\begin{theorem}[Gedomineerde convergentie]\label{thm:b2:integration:dominated}
Zij $(f_n)$ stuksgewijs \omterm{def:b2:metric:continuity}{continu} op een interval $I$, puntsgewijs
convergerend naar een stuksgewijs \omterm{def:b2:metric:continuity}{continue} $f$, en neem aan dat er
een vaste integreerbare $\varphi \geq 0$ is ($\int_I \varphi <
\infty$) met\index{stelling van de gedomineerde convergentie}
\[
\abs{f_n(t)} \leq \varphi(t) \qquad (t \in I,\ n \in \N).
\]
Dan convergeren alle $\int_I f_n$ en $\int_I f$ \omterm{def:b2:series:def}{absoluut}, en geldt
\[
\int_I f_n \xrightarrow[n \to \infty]{} \int_I f .
\]
\end{theorem}
\admitted

\begin{remark}
Het eerlijke bewijs hoort bij de lebesgue-integratietheorie van
bachelorjaar 3; de uitspraak wordt vanaf nu echter voortdurend
gebruikt. De hypothese van de \emph{dominatie} is het hele punt:
puntsgewijze convergentie alleen volstaat niet ($f_n =
n\,\mathbf{1}_{\intoo{0}{1/n}}$, glijdende bulten: $\int f_n = 1
\not\to 0 = \int f$). De stelling geldt ook voor een \omterm{def:b2:metric:continuity}{continue}
parameter ($f_\lambda$ met $\lambda \to \lambda_0$), via de
karakterisering van limieten met rijen.
\end{remark}

\begin{example}[Een gaussische limiet, met dominatie]\label{ex:b2:integration:wallislimit}
Bereken $\displaystyle\lim_{n\to\infty} I_n$ met $I_n =
\int_0^\infty \Bigl(1 + \frac{t^2}{n}\Bigr)^{\!-n}\dd t$.
Puntsgewijs geldt $(1 + t^2/n)^n \to \eu^{t^2}$ (de limiet van de
samengestelde interest), dus gaan de integranden naar
$\eu^{-t^2}$. Dominatie: de rij $n \mapsto (1 + u/n)^n$ is voor $u
\geq 0$ niet-dalend (de ongelijkheid tussen rekenkundig en
meetkundig gemiddelde op de $n + 1$ factoren $1, 1 + \frac un,
\dots, 1 + \frac un$ geeft $(1 + \frac u{n+1})^{n+1} \geq (1 +
\frac un)^n$), dus voor $n \geq 2$:
\[
\Bigl(1 + \frac{t^2}{n}\Bigr)^{\!-n}
\leq \Bigl(1 + \frac{t^2}{2}\Bigr)^{\!-2},
\]
een integreerbare dominant ($\sim 4t^{-4}$ bij oneindig). De
gedomineerde convergentie geeft
\[
I_n \xrightarrow[n\to\infty]{} \int_0^\infty \eu^{-t^2}\dd t
= \frac{\sqrt\pi}{2}
\]
(de gaussische integraal uit \cref{exo:b2:integration:8}).
Sluitende controle: de substitutie $t = \sqrt n\tan\theta$ berekent
$I_n$ exact, $I_n = \sqrt n\int_0^{\pi/2}\cos^{2n-2}\theta\,
\dd\theta = \sqrt n\,W_{2n-2}$, en de asymptotiek van Wallis $W_m
\sim \sqrt{\pi/(2m)}$ (\cref{lem:b2:comparison:wallis}) geeft
opnieuw $\sqrt n\,W_{2n-2} \to \frac{\sqrt\pi}2$: de twee pijlers
van dit hoofdstuk en het vorige zijn het eens.
\end{example}

\begin{example}[Gedomineerde convergentie met continue parameter]
\label{ex:b2:integration:arctanlimit}
Bereken
\[
\lim_{x\to+\infty}\int_0^\infty
\frac{\arctan(xt)}{1+t^2}\,\dd t .
\]
Voor elke $t > 0$ gaat $\arctan(xt) \to \frac\pi2$ als $x \to
\infty$; en de dominatie
\[
\Bigl|\frac{\arctan(xt)}{1+t^2}\Bigr|
\leq \frac{\pi/2}{1+t^2},
\qquad\text{integreerbaar en onafhankelijk van } x,
\]
geldt voor alle $x$. Volgens de vorm met \omterm{def:b2:metric:continuity}{continue} parameter van
\cref{thm:b2:integration:dominated} (karakterisering met rijen:
test langs elke $x_n \to \infty$) is
\[
\int_0^\infty\frac{\arctan(xt)}{1+t^2}\,\dd t
\xrightarrow[x\to+\infty]{}
\frac\pi2\int_0^\infty\frac{\dd t}{1+t^2}
= \frac{\pi^2}{4} .
\]
Het inzicht om te onthouden: het ene punt $t = 0$, waar de
puntsgewijze limiet $0$ is in plaats van $\frac\pi2$, verandert
niets --- de limietfunctie komt alleen via haar integraal binnen,
een van de stille zegeningen van de stelling.
\end{example}

\section{Integralen met een parameter}

\begin{theorem}[Continuïteit onder het integraalteken]\label{thm:b2:integration:continuity}
Zij $f \colon A \times I \to \R$ ($A$ een \omterm{def:b2:metric:def}{metrische ruimte}, $I$ een
interval) met: $t \mapsto f(x, t)$ stuksgewijs \omterm{def:b2:metric:continuity}{continu} voor elke
$x$; $x \mapsto f(x, t)$ \omterm{def:b2:metric:continuity}{continu} voor elke $t$; en een
\emph{dominatie} $\abs{f(x,t)} \leq \varphi(t)$ (met $\varphi$
integreerbaar op $I$ en onafhankelijk van $x$). Dan is
\[
F(x) = \int_I f(x, t)\,\dd t
\]
gedefinieerd en \omterm{def:b2:metric:continuity}{continu} op $A$.\index{parameterintegraal}
\end{theorem}

\begin{proof}
Gedefinieerdheid: de dominatie geeft \omterm{def:b2:series:def}{absolute convergentie}.
\omterm{def:b2:metric:continuity}{Continuïteit} in $x_0$: voor elke rij $x_n \to x_0$ convergeren de
functies $g_n(t) = f(x_n, t)$ puntsgewijs naar $f(x_0, t)$
(\omterm{def:b2:metric:continuity}{continuïteit} in $x$) onder de vaste dominant $\varphi$: de
gedomineerde convergentie geeft $F(x_n) \to F(x_0)$; besluit met
de karakterisering van de \omterm{def:b2:metric:continuity}{continuïteit} met rijen
(\cref{def:b2:metric:continuity}).
\end{proof}

\begin{theorem}[Differentiëren onder het integraalteken]\label{thm:b2:integration:leibnizrule}
Zij $f \colon J \times I \to \R$ ($J$ een interval van parameters)
met: $t \mapsto f(x,t)$ integreerbaar op $I$ voor elke $x$; $x
\mapsto f(x,t)$ van klasse $C^1$ voor elke $t$, waarbij de partiële
afgeleide $\frac{\partial f}{\partial x}$ stuksgewijs \omterm{def:b2:metric:continuity}{continu} in
$t$ is en gedomineerd wordt: $\bigl|\frac{\partial f}{\partial
x}(x,t)\bigr| \leq \psi(t)$ met $\psi$ integreerbaar. Dan is $F(x)
= \int_I f(x,t)\dd t$ van klasse $C^1$ op $J$ en geldt
\[
F'(x) = \int_I \frac{\partial f}{\partial x}(x, t)\,\dd t .
\]
\end{theorem}

\begin{proof}
Leg $x$ vast en zij $h_n \to 0$. De differentiequotiënten
\[
\frac{F(x + h_n) - F(x)}{h_n}
= \int_I \frac{f(x + h_n, t) - f(x, t)}{h_n}\,\dd t
\]
hebben integranden die puntsgewijs naar $\frac{\partial
f}{\partial x}(x, t)$ convergeren en door $\psi(t)$ worden
gedomineerd: de middelwaardeongelijkheid in $x$ bij vaste $t$ geeft
namelijk
\[
\Bigl|\frac{f(x + h_n, t) - f(x,t)}{h_n}\Bigr|
\leq \sup_{\xi} \Bigl|\frac{\partial f}{\partial x}(\xi, t)\Bigr|
\leq \psi(t) .
\]
De gedomineerde convergentie geeft dan als limiet van de
quotiënten $\int_I \frac{\partial f}{\partial x}(x,t)\dd t$: dus
is $F$ differentieerbaar met de aangekondigde afgeleide, en die is
\omterm{def:b2:metric:continuity}{continu} volgens \cref{thm:b2:integration:continuity} toegepast op
$\frac{\partial f}{\partial x}$.
\end{proof}

\begin{example}[Een parameterintegraal getoetst aan een formule]\label{ex:b2:integration:arctanparam}
Zij $F(x) = \displaystyle\int_0^\infty \frac{\dd t}{t^2 + x}$ voor
$x > 0$. Op elke $\intcc{a}{b} \subset \intoo{0}{\infty}$ wordt de
integrand gedomineerd door $\frac{1}{t^2 + a}$, integreerbaar en
onafhankelijk van $x$: dus is $F$ \omterm{def:b2:metric:continuity}{continu}
(\cref{thm:b2:integration:continuity}). Hier laat de stelling zich
aan een expliciete waarde toetsen:
\[
F(x) = \Bigl[\frac{1}{\sqrt x}\arctan\frac{t}{\sqrt
x}\Bigr]_0^\infty = \frac{\pi}{2\sqrt x} ,
\]
zichtbaar \omterm{def:b2:metric:continuity}{continu}. Differentieer nu onder de integraal: de
afgeleide naar $x$, namelijk $-\frac{1}{(t^2+x)^2}$, wordt op
$\intcc ab$ gedomineerd door $\frac{1}{(t^2+a)^2}$, integreerbaar,
zodat \cref{thm:b2:integration:leibnizrule} geeft
\[
F'(x) = -\int_0^\infty \frac{\dd t}{(t^2 + x)^2}
\qquad\text{terwijl}\qquad
F'(x) = -\frac{\pi}{4}\,x^{-3/2} ,
\]
zodat we gratis een nieuwe integraal hebben \emph{berekend}:
$\int_0^\infty\frac{\dd t}{(t^2+x)^2} = \frac{\pi}{4x^{3/2}}$. Het
inzicht om te onthouden: het differentiëren van een bekende
\omterm{thm:b2:integration:continuity}{parameterintegraal} is een fabriek van nieuwe formules --- itereren
levert $\int_0^\infty\frac{\dd t}{(t^2+1)^n}$ voor elke $n$, zonder
enige goniometrische substitutie.
\end{example}

\begin{method}[Een oneigenlijke integraal bestuderen]\label{met:b2:integration:study}
Gegeven $\int_a^b f$:
\begin{enumerate}
  \item Lokaliseer het probleem: noem de eindpunten (of de
        inwendige punten) waar $f$ onbegrensd is of het interval
        oneindig, en splits zo dat elk stuk precies één lastig
        uiteinde heeft.
  \item Heeft $f$ bij dat uiteinde een vast teken, zoek dan een
        equivalent en vergelijk met de referentieschalen van
        \cref{thm:b2:integration:comparison}.
  \item Oscilleert $f$, test dan eerst $\abs f$ (\omterm{def:b2:series:def}{absolute
        convergentie}). Divergeert $\int\abs f$, probeer dan
        partiële integratie om de oscillatie voor afname in te
        ruilen, zoals in \cref{ex:b2:integration:sint};
        ondergrenzen als $\abs{\sin t} \geq \sin^2t$ sporen echte
        halfconvergentie op.
  \item Voor een waarde en niet alleen de aard: partiële
        integratie, substitutie, of een parameter (differentieer
        een eenvoudiger integraal, zoals in
        \cref{ex:b2:integration:arctanparam} en
        \cref{ex:b2:integration:laplace}).
  \item Controles op elke berekende waarde: teken en ruwe grootte
        tegen een grove grens ($\int_0^\infty \eu^{-t^2}\dd t \in
        \intoo{0}{1 + \int_1^\infty \eu^{-t}}$, dus
        $\frac{\sqrt\pi}{2} \approx 0.886$ is plausibel); en
        consistentie onder schaling ($t \mapsto \lambda t$ moet
        beide leden op dezelfde manier herschalen --- de snelste
        detector van een verloren factor).
\end{enumerate}
\end{method}

\begin{remark}[Klassieke valkuilen]
Drie terugkerende fouten. \emph{(i) Dominanten die van de parameter
afhangen:} de dominatie $\abs{f(x,t)} \leq \varphi(t)$ moet uniform
in $x$ zijn op de beschouwde verzameling; zij geldt meestal op
segmenten $\intcc ab$ maar niet globaal --- voor
$\int_0^\infty\eu^{-xt}\dd t$ bestaat er geen integreerbare
dominant die voor alle $x > 0$ geldig is, en toch volstaat
domineren op $x \geq a > 0$ om op de hele \omterm{def:b2:metric:topology}{open} halfrechte te
werken, want \omterm{def:b2:metric:continuity}{continuïteit} en afgeleiden zijn lokale begrippen.
\emph{(ii) Integranden met wisselend teken vergelijken:} de
vergelijkingsgereedschapskist is voor \emph{niet-negatieve}
functies; uit $\abs f \leq g$ met $\int g$ divergent volgt niets
--- $\int_1^\infty\frac{\sin t}t\,\dd t$ \omterm{def:b2:integration:improper}{convergeert} hoewel elke
vergelijking met $\frac1t$ faalt. \emph{(iii) De helft van het
probleem vergeten:} bestudeer op $\intoo{0}{\infty}$ altijd beide
uiteinden apart; $\int_0^\infty\frac{\dd t}{t}$ divergeert aan
\emph{beide} kanten, en een splitsing die convergent lijkt, kan
stilzwijgend twee oneindigheden tegen elkaar wegstrepen. De veilige
reflex is de checklist van \cref{met:b2:integration:study}.
\end{remark}

\begin{example}[Een grensgeval van Bertrand, tot op het cijfer]\label{ex:b2:integration:bertrandboundary}
De schaal $\int^\infty\frac{\dd t}{t(\ln t)^\beta}$ uit
\cref{thm:b2:integration:comparison} ligt precies op de rand van de
machtschalen; haar grensgevallen verdienen één volledige
berekening. Voor $\beta = 2$:
\[
\int_\eu^{\infty}\frac{\dd t}{t(\ln t)^2}
= \Bigl[-\frac{1}{\ln t}\Bigr]_\eu^{\infty}
= 0 - (-1) = 1 ,
\]
een convergente integraal met een aangenaam exacte waarde; terwijl
voor $\beta = 1$
\[
\int_\eu^{x}\frac{\dd t}{t\ln t}
= \bigl[\ln\ln t\bigr]_\eu^{x} = \ln\ln x
\longrightarrow \infty ,
\]
divergent --- maar zo traag dat het bereiken van $\ln\ln x = 10$ de
waarde $x = \eu^{\eu^{10}} \approx 10^{9566}$ vergt. Het inzicht om
te onthouden: tussen ``elke macht $t^{-1-\varepsilon}$
\omterm{def:b2:integration:improper}{convergeert}'' en ``$t^{-1}$ divergeert'' woont een oneindige ladder
van logaritmische schalen, die elkaar steeds verder verfijnen; de
substitutie $u = \ln t$ klapt elke sport op de vorige, en daarom
weerspiegelen de criteria van Bertrand die van Riemann één niveau
hoger.
\end{example}

\begin{remark}[Vooruitblik binnen dit volume]
Het gereedschap van dit hoofdstuk staat op het punt overal op te
duiken. De gedomineerde convergentie is de motor achter de
benaderende eenheden van het volgende hoofdstuk (glijdende kernen,
zowel Bernstein als Fejér); de \omterm{def:b2:metric:continuity}{continuïteit} en het differentiëren
onder het integraalteken leveren het rekenen met
fouriercoëfficiënten in het fourierhoofdstuk op, waar elke $c_n(f)$
een vermomde \omterm{thm:b2:integration:continuity}{parameterintegraal} is. De $\Gamma$-functie keert twee
keer terug: in het hoofdstuk over meervoudige integralen, waar een
dubbelintegraal eindelijk de Beta--Gamma-formule van Euler \omterm{def:b2:metric:complete}{volledig}
bewijst, en in de kanshoofdstukken, waar integralen van
$\Gamma$-type de standaarddichtheden normaliseren en hun momenten
berekenen. En de halfconvergente $\int\frac{\sin t}{t}$ duikt op
als de constante van Gibbs in het fourierhoofdstuk --- dezelfde
integraal, die de overshoot van de partiële sommen bij een sprong
meet.
\end{remark}

\begin{definition}[De $\Gamma$-functie]\label{def:b2:integration:gamma}
Voor $x > 0$:\index{Gamma-functie}
\[
\Gamma(x) = \int_0^{\infty} t^{x-1}\,\eu^{-t}\,\dd t ,
\]
convergent aan beide uiteinden ($t^{x-1}$ is integreerbaar bij
$0^+$ zodra $x > 0$; exponentiële afname bij $\infty$).
\end{definition}

\begin{theorem}\label{thm:b2:integration:gammaprops}
$\Gamma$ is \omterm{def:b2:metric:continuity}{continu} op $\intoo{0}{+\infty}$, voldoet aan de
functionaalvergelijking
\[
\Gamma(x + 1) = x\,\Gamma(x),
\qquad \Gamma(1) = 1,
\qquad\text{dus}\qquad \Gamma(n + 1) = n! ,
\]
en is van klasse $C^1$ (zelfs $C^\infty$) met $\Gamma'(x) =
\int_0^\infty t^{x-1}\eu^{-t}\ln t\,\dd t$.
\end{theorem}

\begin{proof}
Functionaalvergelijking: integreer partieel op
$\intcc{\varepsilon}{M}$ en laat de uiteinden gaan: $\int
t^{x}\eu^{-t} = [-t^x\eu^{-t}] + x\int t^{x-1}\eu^{-t}$, waarbij de
randtermen verdwijnen --- immers $\varepsilon^x\eu^{-\varepsilon}
\to 0$ als $\varepsilon \to 0^+$ omdat $x > 0$, en $M^x\eu^{-M} \to
0$ als $M \to \infty$ omdat de exponentiële elke macht verslaat; en
beide afgekapte integralen convergeren naar hun oneigenlijke
waarden dankzij de convergentie die in
\cref{def:b2:integration:gamma} is vastgesteld. Verder is
$\Gamma(1) = \int \eu^{-t} = 1$; inductie geeft de faculteit.

\omterm{def:b2:metric:continuity}{Continuïteit} op $\intcc{a}{b} \subset \intoo{0}{\infty}$: domineer
$t^{x-1}\eu^{-t}$ door $\varphi(t) = (t^{a-1} + t^{b-1})\eu^{-t}$,
integreerbaar en onafhankelijk van $x \in \intcc{a}{b}$:
\cref{thm:b2:integration:continuity} is dus op elk zulk segment van
toepassing, en daarmee op de hele halfrechte.
Differentieerbaarheid: de afgeleide naar $x$, te weten
$t^{x-1}\eu^{-t}\ln t$, wordt op $\intcc{a}{b}$ gedomineerd door
$(t^{a-1} + t^{b-1})\eu^{-t}\,\abs{\ln t}$, nog altijd
integreerbaar: \cref{thm:b2:integration:leibnizrule}; itereren
geeft alle afgeleiden (elke stap voegt een macht van $\ln t$ toe,
en dat is onschadelijk).
\end{proof}

\begin{example}[Faculteiten van halve getallen]\label{ex:b2:integration:halffactorial}
De functionaalvergelijking en $\Gamma\bigl(\frac12\bigr) =
\sqrt\pi$ (één substitutie verwijderd van
\cref{exo:b2:integration:8}: zet $t = u^2$ in de definiërende
integraal) brengen alle waarden in halve getallen voort:
\[
\Gamma\Bigl(\frac32\Bigr) = \frac12\,\Gamma\Bigl(\frac12\Bigr)
= \frac{\sqrt\pi}{2},
\qquad
\Gamma\Bigl(\frac52\Bigr) = \frac32\cdot\frac{\sqrt\pi}{2}
= \frac{3\sqrt\pi}{4},
\qquad
\Gamma\Bigl(\frac72\Bigr) = \frac{15\sqrt\pi}{8} .
\]
Omdat $\Gamma(n+1) = n!$, mag men gerust zeggen dat ``$\frac12! =
\frac{\sqrt\pi}{2} \approx 0.886$'': de faculteit is
geïnterpoleerd, en de interpolerende kromme duikt tussen $0! = 1$
en $1! = 1$ onder $1$ (haar minimum $\approx 0.8856$ in $x \approx
1.4616$ past bij het convexiteitsbeeld uit Deel I van de
weekendopgave). Het inzicht om te onthouden: niets in de integraal
$\int_0^\infty t^{x-1}\eu^{-t}\dd t$ bevoorrecht de gehele getallen
--- de discreetheid van de faculteit was een toevalligheid van het
tellen, en $\sqrt\pi$ is wat er tussen $1$ en $1$ woont.
\end{example}

\begin{remark}[Waar $\Gamma$ vandaan verder gaat]
De weekendopgave van dit hoofdstuk bouwt de hele calculus van
Euler rond $\Gamma$ op: de \omterm{pb:b2:integration:1}{Beta-functie}, haar recursies met
partiële integratie, de Wallis-integralen als waarden van Beta, en
de limietformule van Gauss. Het hoofdstuk over meervoudige
integralen bewijst de Beta--Gamma-formule van Euler voor alle
argumenten met een dubbelintegraal; de kanshoofdstukken ontmoeten
$\Gamma$ opnieuw bij de normalisatie van de meest voorkomende
dichtheden en bij de momenten van wachttijden. Het volume van
bachelorjaar 3 bouwt $\Gamma$ op lebesgue-fundamenten opnieuw op,
bewijst de eenduidigheidsstelling van Bohr en Mollerup, en breidt
de formule van Stirling met gedomineerde convergentie uit van de
gehele getallen naar de reële halfrechte.
\end{remark}

\begin{example}[Een klassieke berekening door te differentiëren]\label{ex:b2:integration:laplace}
Zij voor $x \in \R$ de waarde $F(x) = \int_0^{\infty}
\eu^{-t^2}\cos(xt)\,\dd t$ (\omterm{def:b2:series:def}{absoluut} convergent, gedomineerd door
$\eu^{-t^2}$). Volgens
\cref{thm:b2:integration:leibnizrule} (met dominatie van de
afgeleide naar $x$ door $t\,\eu^{-t^2}$, integreerbaar):
\[
F'(x) = -\int_0^\infty t\,\eu^{-t^2}\sin(xt)\,\dd t
= \Bigl[\frac{\eu^{-t^2}}{2}\sin(xt)\Bigr]_0^\infty
- \frac x2\int_0^\infty \eu^{-t^2}\cos(xt)\,\dd t
= -\frac x2\,F(x),
\]
(partiële integratie met $u' = t\eu^{-t^2}$). De
differentiaalvergelijking $F' = -\frac x2 F$ integreert tot $F(x) =
F(0)\,\eu^{-x^2/4}$: de integraal van gaussisch type reproduceert
zichzelf. De constante $F(0) = \int_0^\infty \eu^{-t^2}\dd t =
\frac{\sqrt\pi}{2}$ wordt in \cref{exo:b2:integration:8} berekend
--- en nog eens, met een dubbelintegraal, in
\cref{ch:b2:multint}.
\end{example}

\begin{exercise}[$\star$]\label{exo:b2:integration:1}
Bepaal de aard van $\displaystyle\int_0^1 \frac{\dd
t}{\sqrt{t(1-t)}}$, $\;\displaystyle\int_1^\infty \frac{\ln
t}{t^2}\dd t$ en $\;\displaystyle\int_0^\infty \frac{\dd t}{1 +
t^2\sin^2 t}$ \emph{(vergelijk met het divergente gedrag van
harmonisch type bij $t = n\pi$)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:integration:2}
Bereken $\displaystyle\int_0^\infty t^n \eu^{-\lambda t}\,\dd t$
($\lambda > 0$) met $\Gamma$, en $\displaystyle\int_0^1 (\ln t)^n
\dd t$ met de substitutie $t = \eu^{-u}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:integration:3}
Bewijs dat $\displaystyle\int_0^{\infty} \frac{\dd t}{(1 + t^2)(1 +
t^x)}$ voor elke $x \in \R$ welgedefinieerd en onafhankelijk van
$x$ is. \emph{(Substitueer $t \mapsto \frac1t$ en middel de twee
uitdrukkingen.)} Wat is haar waarde?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:4}
(Bertrand-integralen bij een eindig eindpunt) Voor welke $(\alpha,
\beta)$ \omterm{def:b2:integration:improper}{convergeert} $\displaystyle\int_0^{1/2} \frac{\dd
t}{t^\alpha\,\abs{\ln t}^\beta}$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:5}
Zij $F(x) = \displaystyle\int_0^{\infty} \frac{\eu^{-xt}}{1 +
t^2}\,\dd t$ voor $x \geq 0$. Bewijs dat $F$ \omterm{def:b2:metric:continuity}{continu} is op
$\intco{0}{\infty}$ en van klasse $C^2$ op $\intoo{0}{\infty}$,
dat daar $F'' + F = \frac1x$ geldt, en dat $F(x) \to 0$ als $x \to
+\infty$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:6}
(Frullani) Zij $f$ \omterm{def:b2:metric:continuity}{continu} op $\intco{0}{+\infty}$ met een eindige
limiet $f(\infty)$ bij $+\infty$. Bewijs dat voor $a, b > 0$ geldt
\[
\int_0^{\infty} \frac{f(at) - f(bt)}{t}\,\dd t
= \bigl(f(0) - f(\infty)\bigr)\,\ln\frac ba .
\]
\emph{(Substitueer op $\intcc{\varepsilon}{M}$ in elk stuk en
hergroepeer tot $\int_{a\varepsilon}^{b\varepsilon} -
\int_{aM}^{bM}$ van $\frac{f(u)}u\,\dd u$; sluit in met de
\omterm{def:b2:metric:continuity}{continuïteit} in $0$ en de limiet bij $\infty$.)} Bereken
$\int_0^\infty \frac{\eu^{-t} - \eu^{-2t}}{t}\dd t$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:7}
Verantwoord en bereken $\lim_{n\to\infty} \displaystyle\int_0^n
\Bigl(1 - \frac tn\Bigr)^{\!n} t^{x-1}\,\dd t$ voor $x > 0$
\emph{(gedomineerde convergentie met $\varphi(t) =
\eu^{-t}t^{x-1}$, met $(1 - t/n)^n \leq \eu^{-t}$; de limiet is
$\Gamma(x)$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:8}
(De gaussische integraal met een parametertruc) Zet voor $x \geq 0$
\[
G(x) = \Bigl(\int_0^x \eu^{-t^2}\dd t\Bigr)^{\!2},
\qquad
H(x) = \int_0^1 \frac{\eu^{-x^2(1+t^2)}}{1 + t^2}\,\dd t .
\]
Bewijs dat $G' + H' = 0$ (differentieer $H$ onder de integraal en
substitueer $u = xt$ in de resulterende integraal), leid af dat
$G(x) + H(x) = \frac\pi4$ voor alle $x$, en besluit dat
\[
\int_0^{\infty} \eu^{-t^2}\,\dd t = \frac{\sqrt\pi}{2} .
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:9}
Bewijs dat $\Gamma$ \emph{logaritmisch convex} is: $\ln\Gamma$ is
convex op $\intoo{0}{\infty}$. \emph{(Cauchy--Schwarz voor
integralen toegepast op $t^{(x+y)/2 - 1}\eu^{-t} =
\bigl(t^{x-1}\eu^{-t}\bigr)^{1/2} \bigl(t^{y-1}\eu^{-t}\bigr)^{1/2}$
geeft $\Gamma\bigl(\frac{x+y}{2}\bigr)^2 \leq \Gamma(x)\Gamma(y)$;
combineer met de \omterm{def:b2:metric:continuity}{continuïteit} en \cref{exo:b2:realfun:8}.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:10}
(Integraal van Dirichlet) Zet $F(x) = \displaystyle\int_0^{\infty}
\frac{\sin t}{t}\,\eu^{-xt}\,\dd t$ voor $x > 0$.
\begin{enumerate}
  \item Verantwoord $F'(x) = -\frac{1}{1 + x^2}$ (differentieer
        onder de integraal; bereken $\int_0^\infty \eu^{-xt}\sin
        t\,\dd t$ met twee partiële integraties).
  \item Bewijs dat $F(x) \to 0$ als $x \to +\infty$ en leid af dat
        $F(x) = \frac\pi2 - \arctan x$.
  \item Neem de \omterm{def:b2:metric:continuity}{continuïteit} van $F$ in $0^+$ aan (een stelling van
        het type Abel) en besluit tot de waarde van de
        halfconvergente integraal:
        \[
        \int_0^{\infty} \frac{\sin t}{t}\,\dd t = \frac{\pi}{2}.
        \]
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:11}
Verantwoord de convergentie van $\displaystyle\int_0^\infty
\Bigl(\frac{\sin t}{t}\Bigr)^{\!2}\dd t$, en bereken haar daarna
met één partiële integratie en \cref{exo:b2:integration:10}:
\[
\int_0^\infty \Bigl(\frac{\sin t}{t}\Bigr)^{\!2}\dd t
= \frac{\pi}{2} .
\]
(Dezelfde waarde als $\int_0^\infty \frac{\sin t}{t}\dd t$ --- maar
deze keer is de convergentie \omterm{def:b2:series:def}{absoluut}.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:12}
(De gaussische staart) Zet voor $x > 0$ de waarde $T(x) =
\displaystyle \int_x^\infty \eu^{-t^2}\dd t$.
\begin{enumerate}
  \item Schrijf $\eu^{-t^2} = \frac{1}{-2t}\cdot(-2t\,\eu^{-t^2})$
        en integreer tweemaal partieel om
        \[
        T(x) = \eu^{-x^2}\Bigl(\frac{1}{2x} -
        \frac{1}{4x^3}\Bigr) +
        \frac34\int_x^\infty \frac{\eu^{-t^2}}{t^4}\,\dd t
        \]
        te verkrijgen.
  \item Begrens de restterm: $0 \leq \frac34\int_x^\infty
        t^{-4}\eu^{-t^2}\dd t \leq \frac{3}{8x^5}\,\eu^{-x^2}$, en
        leid de insluiting
        \[
        \eu^{-x^2}\Bigl(\frac{1}{2x} - \frac{1}{4x^3}\Bigr)
        \leq T(x) \leq \frac{\eu^{-x^2}}{2x},
        \qquad\text{dus}\qquad
        T(x) \sim \frac{\eu^{-x^2}}{2x} \quad (x \to +\infty)
        \]
        af.
  \item Waarom kan de volledige alternerende reeks die men door de
        partiële integraties te herhalen krijgt, voor vaste $x$
        nooit convergeren? \emph{(Vergelijk de groei van de
        coëfficiënten $1\cdot3\cdots(2k-1)$ met de machten
        $(2x^2)^k$.)}
\end{enumerate}
\end{exercise}

\section{Probleem: Eulers integralen --- Beta, Gamma en de
limietformule van Gauss}

\begin{problem}\label{pb:b2:integration:1}
De $\Gamma$-functie uit \cref{def:b2:integration:gamma} is de ene
helft van Eulers calculus van integralen; de andere helft is de
\emph{Beta-functie}\index{Beta-functie}
\[
B(x, y) = \int_0^1 t^{x-1}(1 - t)^{y-1}\,\dd t .
\]
Deze opgave ontwikkelt het paar $(\Gamma, B)$ uitsluitend met het
gereedschap van dit hoofdstuk --- partiële integratie, substitutie,
gedomineerde convergentie --- en mondt uit in de
\emph{Beta--Gamma-formule van Euler} $B(x,y) =
\frac{\Gamma(x)\Gamma(y)} {\Gamma(x+y)}$ op de halve getallen en in
de \emph{limietformule van Gauss}\index{limietformule van Gauss}
voor $\Gamma$. Onderweg duiken de Wallis-integralen van
\cref{lem:b2:comparison:wallis} op als waarden van Beta, en valt de
verdubbelingsformule van Legendre eruit.

\medskip
\noindent\textbf{Deel I --- De fijne structuur van $\Gamma$.}
\begin{enumerate}
  \item Herhaal waarom $\Gamma(x) = \int_0^\infty
        t^{x-1}\eu^{-t}\dd t$ precies voor $x > 0$ \omterm{def:b2:integration:improper}{convergeert}, en
        toon aan dat
        \[
        \Gamma(x) \sim \frac1x \qquad (x \to 0^+)
        \]
        \emph{(functionaalvergelijking plus de \omterm{def:b2:metric:continuity}{continuïteit} van
        $\Gamma$ in $1$)}.
  \item Bewijs dat $\Gamma\bigl(\tfrac12\bigr) = \sqrt\pi$
        \emph{(substitueer $t = u^2$ en beroep je op
        \cref{exo:b2:integration:8})}, en leid af dat $\int_\R
        \eu^{-u^2/2}\dd u = \sqrt{2\pi}$.
  \item Toon met inductie aan dat voor $n \in \N$
        \[
        \Gamma\Bigl(n + \frac12\Bigr)
        = \frac{(2n)!}{4^n\,n!}\,\sqrt\pi .
        \]
  \item Verantwoord $\Gamma''(x) = \int_0^\infty
        t^{x-1}\eu^{-t}(\ln t)^2\dd t > 0$, en leid af dat
        $\Gamma$ strikt convex is, een uniek minimum aanneemt in
        een zekere $x_0 \in \intoo{1}{2}$ \emph{($\Gamma(1) =
        \Gamma(2) = 1$ en Rolle)}, daalt op $\intoo{0}{x_0}$ en
        stijgt op $\intoo{x_0}{\infty}$.
  \item Toon aan dat $\Gamma$ elke macht verslaat: voor elke $k \in
        \N$ is $x^k = o\bigl(\Gamma(x)\bigr)$ als $x \to +\infty$
        \emph{(klem $x$ tussen gehele getallen in en gebruik
        $\Gamma(n+1) = n!$ met de monotonie uit vraag 4)}.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De \omterm{pb:b2:integration:1}{Beta-functie}, met partiële
integratie.}
\begin{enumerate}[resume]
  \item Toon aan dat $B(x,y)$ precies voor $x > 0$ en $y > 0$
        \omterm{def:b2:integration:improper}{convergeert}, en dat $B(x,y) = B(y,x)$.
  \item Bereken $B(x, 1) = \frac1x$, en bewijs met partiële
        integratie dat voor $x, y > 0$
        \[
        B(x, y+1) = \frac{y}{x}\,B(x+1, y) .
        \]
  \item Leid uit de splitsing $t^{x-1}(1-t)^{y-1} = t^{x}(1-t)^{y-1}
        + t^{x-1}(1-t)^{y}$ af dat $B(x,y) = B(x+1,y) + B(x,y+1)$,
        en combineer dit met vraag 7 tot de
        \emph{afdalingsrelaties}
        \[
        B(x, y+1) = \frac{y}{x+y}\,B(x,y),
        \qquad
        B(x+1, y) = \frac{x}{x+y}\,B(x,y) .
        \]
  \item Leid af dat voor gehele $m, n \geq 1$
        \[
        B(m, n) = \frac{(m-1)!\,(n-1)!}{(m+n-1)!}
        = \frac{1}{(m+n-1)\binom{m+n-2}{m-1}} .
        \]
  \item Bewijs \emph{de formule van Euler met één geheel argument}:
        voor elke $x > 0$ en $n \in \N^*$ geldt
        \[
        B(x, n) = \frac{\Gamma(x)\,\Gamma(n)}{\Gamma(x + n)}
        \]
        \emph{(inductie naar $n$: beide leden zijn $\frac1x$ bij $n
        = 1$ en voldoen aan dezelfde afdalingsrelatie)}.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- De Wallis-integralen als waarden van
Beta.}
\begin{enumerate}[resume]
  \item Substitueer $t = \sin^2\theta$ om de goniometrische vorm
        \[
        B(x, y) = 2\int_0^{\pi/2}
        \sin^{2x-1}\theta\,\cos^{2y-1}\theta\,\dd\theta
        \]
        te verkrijgen.
  \item Leid af dat $W_n = \frac12\,B\bigl(\frac{n+1}2,
        \frac12\bigr)$ voor de Wallis-integraal $W_n =
        \int_0^{\pi/2}\sin^n \theta\,\dd\theta$, en vind de
        recurrentie $W_n = \frac{n-1}{n}W_{n-2}$ uit
        \cref{lem:b2:comparison:wallis} terug uit de
        afdalingsrelaties van vraag 8 alleen.
  \item Bereken $B\bigl(\frac12, \frac12\bigr) = 2W_0 = \pi$ en
        toets dit aan $\Gamma\bigl(\frac12\bigr)^2/\Gamma(1)$: de
        formule van Euler geldt dus in $\bigl(\frac12,
        \frac12\bigr)$.
  \item Leid uit de recurrentie de gesloten vorm $W_{2n} =
        \frac\pi2\, \frac{(2n)!}{4^n(n!)^2}$ af, en ga na dat
        \[
        B\Bigl(n + \frac12, \frac12\Bigr)
        = \frac{\Gamma\bigl(n + \frac12\bigr)\Gamma\bigl(
        \frac12\bigr)}{\Gamma(n+1)} .
        \]
        Besluit met inductie en de afdalingsrelaties dat de formule
        van Euler $B(x,y) =
        \frac{\Gamma(x)\Gamma(y)}{\Gamma(x+y)}$ geldt zodra $2x$
        en $2y$ positieve gehele getallen zijn.
  \item Substitueer $u = \frac{t}{1-t}$ om de derde klassieke vorm
        \[
        B(x,y) = \int_0^\infty
        \frac{u^{x-1}}{(1+u)^{x+y}}\,\dd u
        \]
        te verkrijgen, en ga het geval $x = y = \frac12$
        rechtstreeks na ($u = v^2$ herleidt het tot
        $\int_0^\infty\frac{2\,\dd v}{1+v^2}$).
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De limietformule van Gauss.}
\begin{enumerate}[resume]
  \item Bewijs voor $x > 0$ en $n \in \N^*$ met $n$
        opeenvolgende partiële integraties dat
        \[
        \int_0^n \Bigl(1 - \frac tn\Bigr)^{\!n} t^{x-1}\,\dd t
        = \frac{n!\;n^x}{x(x+1)\cdots(x+n)} .
        \]
  \item Besluit met \cref{exo:b2:integration:7} (gedomineerde
        convergentie) tot de \emph{limietformule van Gauss}:
        \[
        \Gamma(x) = \lim_{n\to\infty}
        \frac{n!\;n^x}{x(x+1)\cdots(x+n)}
        \qquad (x > 0).
        \]
  \item Toon door logaritmen te nemen aan dat voor $x > 0$
        \[
        \ln\Gamma(x) = -\ln x - \gamma x +
        \sum_{k=1}^{\infty}\Bigl(\frac xk -
        \ln\Bigl(1 + \frac xk\Bigr)\Bigr),
        \]
        waarbij $\gamma$ de constante van Euler is
        (\cref{ex:b2:comparison:harmonic}); verantwoord de
        convergentie van de reeks \emph{(de algemene term is $\sim
        \frac{x^2}{2k^2}$)}.
  \item Gebruik de formule van Gauss in $x = \frac12$ en de
        asymptotiek van de centrale binomiaalcoëfficiënt
        $\binom{2n}{n} \sim \frac{4^n}{\sqrt{\pi n}}$
        (\cref{ex:b2:comparison:centralbinomial}) om
        $\Gamma\bigl(\frac12\bigr) = \sqrt\pi$ opnieuw te
        berekenen: de constante van Stirling en de gaussische
        integraal zijn hetzelfde getal in twee vermommingen.
  \item Ga na dat de formule van Gauss de functionaalvergelijking
        opnieuw bewijst: besluit uit de exacte identiteit
        \[
        \frac{n!\,n^{x+1}}{(x+1)\cdots(x+n+1)}
        = \frac{n!\,n^{x}}{x(x+1)\cdots(x+n)}\cdot
        \frac{n\,x}{x+n+1}
        \]
        opnieuw tot $\Gamma(x+1) = x\,\Gamma(x)$. (De formule van
        Gauss legt $\Gamma$ \omterm{def:b2:metric:complete}{volledig} vast; het volume van
        bachelorjaar 3 bewijst de scherpere stelling van Bohr en
        Mollerup: de functionaalvergelijking plus de logaritmische
        convexiteit pinnen $\Gamma$ al vast.)
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Opbrengsten.}
\begin{enumerate}[resume]
  \item Toon voor $a > 0$ aan dat $\int_0^\infty \eu^{-t^a}\dd t =
        \Gamma\bigl(1 + \frac1a\bigr)$, en bereken de limiet als $a
        \to +\infty$ met gedomineerde convergentie \emph{(de
        puntsgewijze limiet is $\mathbf 1_{t < 1}$; domineer door
        $1$ op $\intoc{0}{1}$ en daarbuiten door $\eu^{-t^2}$, voor
        $a \geq 2$)}. Toets het antwoord aan de \omterm{def:b2:metric:continuity}{continuïteit} van
        $\Gamma$.
  \item Toon voor $n \geq 1$ aan dat
        \[
        \int_0^1 \frac{\dd t}{\sqrt{1 - t^n}}
        = \frac1n\,B\Bigl(\frac1n, \frac12\Bigr),
        \]
        en vind de waarden $2$ ($n = 1$) en $\frac\pi2$ ($n = 2$)
        terug. (Voor $n = 4$ is dit de \emph{constante van de
        lemniscaat}, die geen elementaire gesloten vorm heeft; haar
        verhaal hoort bij de theorie van de elliptische
        integralen.)
  \item (Momenten) Toon voor $x > 0$ en $k \in \N$ aan dat
        \[
        \frac{1}{\Gamma(x)}\int_0^\infty
        t^{k}\,t^{x-1}\eu^{-t}\,\dd t
        = \frac{\Gamma(x+k)}{\Gamma(x)}
        = x(x+1)\cdots(x+k-1),
        \]
        de stijgende faculteit; ga na dat $x = 1$ de waarde $k!$
        geeft. (In de kanshoofdstukken is dit het $k$-de moment van
        een standaarddichtheid voor wachttijden.)
  \item Bewijs de Beta-identiteit, geldig voor \emph{alle} $x >
        0$:
        \[
        B(x, x) = 2^{1-2x}\,B\Bigl(x, \frac12\Bigr)
        \]
        \emph{(substitueer $t = \frac{1+s}2$, buit de symmetrie in
        $s$ uit, en zet daarna $s = \sqrt v$)}. Leid daaruit voor
        $2x \in \N^*$ de \emph{verdubbelingsformule van Legendre}
        \[
        \Gamma(x)\,\Gamma\Bigl(x + \frac12\Bigr)
        = 2^{1-2x}\,\sqrt\pi\;\Gamma(2x)
        \]
        af, en ga haar met vraag 3 rechtstreeks na in $x = n$.
        (Voor algemene $x$ volgt zij uit dezelfde identiteit zodra
        de formule van Euler voor alle argumenten bekend is --- het
        bewijs met de dubbelintegraal in het hoofdstuk over
        meervoudige integralen.)
  \item Synthese. In telkens één zin: (i) waar de partiële
        integratie heel Deel II droeg; (ii) waar de gedomineerde
        convergentie in de Delen IV en V binnenkwam; (iii) welke
        asymptotische ingrediënten uit het vergelijkingshoofdstuk
        zijn ingevoerd; (iv) wat er nu van de formule van Euler
        $B(x,y) = \Gamma(x)\Gamma(y)/\Gamma(x+y)$ bewezen is, en
        wat er voor de dubbelintegraal overblijft.
\end{enumerate}
\end{problem}
