\chapter{Verzamelingen en structuren}\label{ch:b2:structures}

Dit openingshoofdstuk slijpt de grondslagen uit het volume van
bachelorjaar 1 tot dagelijks gereedschap: het rekenen met
verzamelingen en quotiënten, het vergelijken van oneindige

\omterm{def:b2:structures:countable}{aftelbaarheid} ligt zowel aan de topologie als aan de kansrekening
ten grondslag.

\section{Verzamelingen, afbeeldingen, quotiënten}

We gebruiken zonder verdere plichtplegingen de taal van
verzamelingen, afbeeldingen, equivalentierelaties en orderelaties
uit het volume van bachelorjaar 1. Twee punten verdienen een
zorgvuldige formulering.

\begin{proposition}[Beelden en originelen van families]\label{prop:b2:structures:images}
Zij $f \colon E \to F$, en zijn $(A_i)_{i \in I}$, $(B_j)_{j \in J}$
families van deelverzamelingen van $E$ respectievelijk $F$. Dan
geldt
\[
f^{-1}\Bigl(\bigcup_j B_j\Bigr) = \bigcup_j f^{-1}(B_j),
\qquad
f^{-1}\Bigl(\bigcap_j B_j\Bigr) = \bigcap_j f^{-1}(B_j),
\qquad
f^{-1}(F \setminus B) = E \setminus f^{-1}(B),
\]
\[
f\Bigl(\bigcup_i A_i\Bigr) = \bigcup_i f(A_i),
\qquad
f\Bigl(\bigcap_i A_i\Bigr) \subseteq \bigcap_i f(A_i)
\quad (\text{gelijkheid voor injectieve } f).
\]
\end{proposition}

\begin{proof}
Elke identiteit is niets anders dan het uitschrijven van de
definities; bijvoorbeeld $x \in f^{-1}(\bigcap B_j) \iff f(x) \in
B_j$ voor alle $j$ $\iff x \in f^{-1}(B_j)$ voor alle $j$. De
identiteiten voor beelden, het falen van de gelijkheid bij de
doorsnede en de reparatie via injectiviteit zijn in het volume van
bachelorjaar 1 voor twee verzamelingen bewezen; voor families
verloopt het argument woordelijk hetzelfde.
\end{proof}

\begin{example}[Waar de insluiting voor beelden strikt is]\label{ex:b2:structures:imagestrict}
Neem $f \colon \R \to \R$, $f(x) = x^2$, met $A_1 =
\intcc{-1}{0}$ en $A_2 = \intcc{0}{1}$. Dan is
\[
f(A_1 \cap A_2) = f(\{0\}) = \{0\},
\qquad
f(A_1) \cap f(A_2) = \intcc{0}{1} \cap \intcc{0}{1} =
\intcc{0}{1} :
\]
de insluiting van \cref{prop:b2:structures:images} is zo strikt
als maar kan --- de twee originelen $\pm x$ van eenzelfde waarde
liggen in verschillende $A_i$. Injectiviteit is precies wat die
splitsing verbiedt, en daarom voldoen originelen (die nooit punten
samenvoegen) onvoorwaardelijk aan alle vier de identiteiten,
terwijl beelden die over doorsneden kwijtraken. Vuistregel voor
het hele boek: duw \emph{originelen} onbekommerd door
verzamelingsbewerkingen heen; ga met beelden voorzichtig om.
\end{example}

\begin{definition}[Quotiëntverzameling]\label{def:b2:structures:quotient}
Zij $\mathcal{R}$ een equivalentierelatie op $E$. De
\emph{quotiëntverzameling}\index{quotiëntverzameling}
$E/\mathcal{R}$ is de verzameling van de equivalentieklassen; de
surjectie $\pi \colon E \to E/\mathcal{R}$, $x \mapsto
\mathrm{cl}(x)$, heet de \emph{canonieke
projectie}\index{canonieke projectie}.

\emph{Universele eigenschap (factorisatie):} is $f \colon E \to F$
\emph{verenigbaar} met $\mathcal{R}$ (dat wil zeggen $x
\mathbin{\mathcal{R}} y \implies f(x) = f(y)$), dan bestaat er
precies één afbeelding $\overline f \colon E/\mathcal{R} \to F$
met $f = \overline f \circ \pi$.
\end{definition}

\begin{proof}[Bewijs van de universele eigenschap]
Eenduidigheid: de eis $f = \overline f \circ \pi$ luidt
\[
\overline f\bigl(\mathrm{cl}(x)\bigr) = f(x)
\qquad (x \in E),
\]
en omdat $\pi$ surjectief is, is elk element van $E/\mathcal{R}$
van de vorm $\mathrm{cl}(x)$: alle waarden van $\overline f$
liggen dus vast. Bestaan: neem de formule hierboven als
\emph{definitie} van $\overline f$; zij is ondubbelzinnig juist
dankzij de verenigbaarheid --- is $\mathrm{cl}(x) =
\mathrm{cl}(y)$, dan geldt $x \mathbin{\mathcal{R}} y$, dus $f(x)
= f(y)$ en de twee kandidaat-waarden vallen samen --- en zij
factoriseert $f$ per constructie. Merk de taakverdeling op:
surjectiviteit van $\pi$ levert de eenduidigheid, verenigbaarheid
levert het bestaan.
\end{proof}

\begin{example}
$\Z/n\Z$ is het quotiënt van $\Z$ naar congruentie modulo $n$; de
controles op welgedefinieerdheid uit het volume van bachelorjaar 1
waren telkens een toepassing van de universele eigenschap.
Quotiënten maken van ``constructies op representanten die
verenigbaar zijn'' echte afbeeldingen --- daar leunen we hieronder
voortdurend op.
\end{example}

\section{Aftelbaarheid en kardinaliteit}

\begin{definition}[Gelijkmachtigheid, aftelbaarheid]\label{def:b2:structures:countable}
Twee verzamelingen heten \emph{gelijkmachtig}\index{gelijkmachtigheid}
wanneer er een bijectie tussen beide bestaat. Een verzameling heet
\emph{aftelbaar}\index{aftelbare verzameling} wanneer zij
gelijkmachtig is met $\N$ (sommige auteurs rekenen de eindige
verzamelingen mee; wij zeggen \emph{hoogstens aftelbaar} voor
``eindig of aftelbaar'').
\end{definition}

\begin{proposition}[Stabiliteitseigenschappen]\label{prop:b2:structures:countablestable}
\begin{enumerate}
  \item Elke oneindige deelverzameling van $\N$ is \omterm{def:b2:structures:countable}{aftelbaar}; een
        verzameling is hoogstens \omterm{def:b2:structures:countable}{aftelbaar} dan en slechts dan als
        zij in $\N$ injecteert, dan en slechts dan als zij leeg is
        of een surjectief beeld van $\N$.
  \item $\N \times \N$ is \omterm{def:b2:structures:countable}{aftelbaar}; een product van twee hoogstens
        \omterm{def:b2:structures:countable}{aftelbare verzamelingen} is hoogstens \omterm{def:b2:structures:countable}{aftelbaar}.
  \item Een hoogstens \omterm{def:b2:structures:countable}{aftelbare} vereniging van hoogstens \omterm{def:b2:structures:countable}{aftelbare
        verzamelingen} is hoogstens \omterm{def:b2:structures:countable}{aftelbaar}.
  \item $\Z$ en $\Q$ zijn \omterm{def:b2:structures:countable}{aftelbaar}.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Som een oneindige $A \subseteq \N$ op met herhaalde minima:
$a_0 = \min A$, $a_{k+1} = \min\,(A \setminus \{a_0, \dots,
a_k\})$ (niet leeg, want $A$ is oneindig); de afbeelding $k
\mapsto a_k$ is strikt stijgend, injectief en surjectief op $A$
(elke $a \in A$ overtreft slechts eindig veel elementen van $A$ en
wordt dus bereikt). Injecteert $E$ via $\varphi$ in $\N$, dan is
$E$ \omterm{def:b2:structures:countable}{gelijkmachtig} met $\varphi(E) \subseteq \N$: eindig of
\omterm{def:b2:structures:countable}{aftelbaar}. Is $s \colon \N \to E$ surjectief, dan injecteert $x
\mapsto \min s^{-1}(\{x\})$ de verzameling $E$ in $\N$.

(2) De afbeelding $(p, q) \mapsto 2^p(2q + 1) - 1$ is een bijectie
$\N^2 \to \N$ (elk positief geheel getal heeft precies één
splitsing $2^p m$ met $m$ oneven, wegens de eenduidige
priemfactorisatie). Voor producten: stel injecties samen.

(3) Zijn verzamelingen $E_n$ gegeven met surjecties $s_n \colon \N
\to E_n$ (onschadelijk wanneer een $E_n$ eindig is: herhaal
waarden), dan is $(n, k) \mapsto s_n(k)$ een surjectie van het
\omterm{def:b2:structures:countable}{aftelbare} $\N^2$ op $\bigcup E_n$.

(4) $\Z = \N \cup (-\N^*)$: een \omterm{def:b2:structures:countable}{aftelbare} vereniging. En $\Q$ is
een surjectief beeld van $\Z \times \N^*$ (de breukafbeelding),
dus hoogstens \omterm{def:b2:structures:countable}{aftelbaar}, en oneindig.
\end{proof}

\begin{example}[Een koppelfunctie, doorgerekend]\label{ex:b2:structures:pairing}
De bijectie $(p, q) \mapsto 2^p(2q + 1) - 1$ uit het bewijs
verdient het om aan het werk gezien te worden. Haar eerste
waarden:
\[
\begin{array}{c|ccccc}
 & q = 0 & q = 1 & q = 2 & q = 3 & q = 4\\
\hline
p = 0 & 0 & 2 & 4 & 6 & 8\\
p = 1 & 1 & 5 & 9 & 13 & 17\\
p = 2 & 3 & 11 & 19 & 27 & 35\\
p = 3 & 7 & 23 & 39 & 55 & 71
\end{array}
\]
Rij $p$ verzamelt de getallen $n$ waarvoor $n + 1$ precies door
$2^p$ deelbaar is: elk natuurlijk getal komt precies één keer
voor. Decoderen is even expliciet als coderen: voor $n = 43$
ontbinden we $n + 1 = 44 = 2^2\cdot 11 = 2^2(2\cdot5 + 1)$, dus
$(p, q) = (2, 5)$. Het inzicht om te onthouden: bewijzen over
\omterm{def:b2:structures:countable}{aftelbaarheid} zijn vaak vermomde \emph{algoritmen} --- hier
``haal de factoren $2$ eruit''.
\end{example}

\begin{example}[De algebraïsche getallen zijn aftelbaar]\label{ex:b2:structures:algebraic}
Een complex getal heet \emph{algebraïsch} wanneer het een
nulpunt is van een veelterm ongelijk aan nul met rationale
coëfficiënten. De verzameling $\overline\Q$ van de algebraïsche
getallen is \omterm{def:b2:structures:countable}{aftelbaar}: de veeltermen van graad $\leq d$ over $\Q$
injecteren in $\Q^{d+1}$, een eindig product van \omterm{def:b2:structures:countable}{aftelbare
verzamelingen}
(\cref{prop:b2:structures:countablestable}~(2)); de vereniging
over $d$ somt de veeltermen ongelijk aan nul op als $P_0, P_1,
P_2, \dots$; elke $P_k$ heeft eindig veel nulpunten; en
\[
\overline\Q = \bigcup_{k \in \N}\ \{\text{nulpunten van } P_k\}
\]
is een \omterm{def:b2:structures:countable}{aftelbare} vereniging van eindige verzamelingen
(\cref{prop:b2:structures:countablestable}~(3)), oneindig omdat
zij $\Q$ bevat. Samen met de overaftelbaarheid van $\R$
(\cref{thm:b2:structures:cantor} hieronder) bewijst dit ---
zonder er ook maar één aan te wijzen --- dat er transcendente
getallen bestaan en dat zij een overaftelbare meerderheid vormen:
het telargument van Cantor uit 1874, bestaan louter op grond van
kardinaliteit.
\end{example}

\begin{theorem}[Cantor; overaftelbaarheid van $\R$]\label{thm:b2:structures:cantor}
\begin{enumerate}
  \item Voor elke verzameling $E$ bestaat er geen surjectie $E \to
        \mathcal{P}(E)$.
  \item $\R$ is \emph{niet} \omterm{def:b2:structures:countable}{aftelbaar}.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) is in het volume van bachelorjaar 1 bewezen (met de
diagonaalverzameling $D = \{x : x \notin f(x)\}$).

(2) Stel dat $(x_n)_{n \in \N}$ heel $\R$ opsomt. Bouw
geneste segmenten $I_0 \supseteq I_1 \supseteq \dots$ met
$\abs{I_n} = 3^{-n}$ en $x_n \notin I_n$: verdeel het huidige
segment in drie gesloten derden; minstens één derde vermijdt $x_n$
(een punt raakt hoogstens twee van de drie). De stelling van de
geneste segmenten (aanliggende eindpunten) levert $\ell \in
\bigcap_n I_n$; maar $\ell = x_N$ voor zekere $N$, terwijl $x_N
\notin I_N$: tegenspraak.
\end{proof}

\begin{theorem}[Cantor--Bernstein]\label{thm:b2:structures:cantorbernstein}
Injecteert $E$ in $F$ en $F$ in $E$, dan zijn $E$ en $F$
\omterm{def:b2:structures:countable}{gelijkmachtig}.\index{stelling van Cantor--Bernstein}
\end{theorem}

\begin{proof}
Zijn $f \colon E \to F$ en $g \colon F \to E$ injecties. Volg voor
elk punt (van $E$ of van $F$) zijn \emph{voorouderketen} van
opeenvolgende originelen, $x \mapsto g^{-1}(x) \mapsto
f^{-1}(g^{-1}(x)) \mapsto \dots$ --- elke stap is gedefinieerd
zolang het huidige punt in het beeld van de betreffende injectie
ligt, en is dan eenduidig wegens injectiviteit. Er zijn drie
elkaar uitsluitende lotgevallen: de keten stopt in een punt van $E
\setminus g(F)$ (\emph{oorsprong in $E$}), zij stopt in een punt
van $F \setminus f(E)$ (\emph{oorsprong in $F$}), of zij stopt
nooit. Dit verdeelt $E = E_E \cup E_F \cup E_\infty$ en $F = F_E
\cup F_F \cup F_\infty$ naar de oorsprong.

Merk nu op dat $f$ de verzameling $E_E$ \emph{op} $F_E$ afbeeldt:
de keten van $f(x)$ is die van $x$ met één stap ervoor, dus de
oorsprongen komen overeen; en elke $y \in F_E$ heeft een keten met
minstens één stap (haar oorsprong ligt in $E$), zodat $y = f(x)$
met $x \in E_E$. Hetzelfde argument geeft bijecties $f \colon
E_\infty \to F_\infty$ en $g \colon F_F \to E_F$. Aan elkaar
geplakt is
\[
h(x) =
\begin{cases}
f(x) & \text{als } x \in E_E \cup E_\infty,\\
g^{-1}(x) & \text{als } x \in E_F,
\end{cases}
\]
een bijectie van $E$ op $F = F_E \cup F_\infty \cup F_F$: zij is
stuksgewijs bijectief en de drie stukken van het doel zijn
disjunct.
\end{proof}

\begin{example}\label{ex:b2:structures:cbexample}
$\intoo{0}{1}$ en $\intcc{0}{1}$ zijn \omterm{def:b2:structures:countable}{gelijkmachtig}: de identiteit
injecteert de ene kant op, $x \mapsto \frac{x + 1}{3}$ de andere;
de stelling fabriceert vervolgens de (noodzakelijk discontinue)
bijectie. Evenzo zijn $\R$, $\intoo{0}{1}$ (via bijecties van het
type $\tanh$) en $\mathcal{P}(\N)$ (binaire ontwikkelingen,
\cref{exo:b2:structures:3}) alle \omterm{def:b2:structures:countable}{gelijkmachtig}: ``de machtigheid
van het continuüm''.
\end{example}

\begin{example}[Het segment en het vierkant]\label{ex:b2:structures:squareseg}
$\intcc{0}{1}$ en $\intcc{0}{1}^2$ zijn \omterm{def:b2:structures:countable}{gelijkmachtig} ---
dimensie is onzichtbaar voor kardinaliteit. De ene injectie is
triviaal: $x \mapsto (x, 0)$. Stuur voor de andere het paar $(x,
y)$ naar het reële getal waarvan de decimalen die van $x$ en $y$
afwisselen,
\[
(0.x_1x_2x_3\dots,\ 0.y_1y_2y_3\dots)
\;\longmapsto\; 0.x_1y_1x_2y_2x_3y_3\dots,
\]
waarbij we per coördinaat de ontwikkeling kiezen die niet op
louter $9$'s eindigt: met die afspraak leggen de cijfers van het
beeld die van $x$ en $y$ vast, dus is de afbeelding injectief
(surjectief hoeft zij niet te zijn --- beelden hebben bijvoorbeeld
nooit vanaf zeker moment alleen maar $9$'s op de oneven posities
--- en dat is geen bezwaar). Cantor--Bernstein
(\cref{thm:b2:structures:cantorbernstein}) stelt daaruit een
echte bijectie samen. Continuïteit is uiteraard hopeloos: een
continue bijectie tussen beide bestaat niet --- de hoofdstukken
over metrische ruimten leggen uit waarom (samenhang onderscheidt
de rechte van het vlak, \cref{ch:b2:metric}).
\end{example}

\section{Groepen}

\begin{definition}[Voortgebrachte ondergroep; orde]\label{def:b2:structures:generated}
Zij $G$ een groep en $A \subseteq G$. De door $A$
\emph{voortgebrachte} ondergroep, genoteerd $\langle A \rangle$,
is de kleinste ondergroep die $A$ bevat --- concreet: alle
eindige producten van elementen van $A$ en hun inversen. Een groep
heet \emph{cyclisch}\index{cyclische groep} wanneer zij door één
element wordt voortgebracht: $\langle a\rangle = \{a^k : k \in
\Z\}$. De \emph{orde}\index{orde!van een element} van $a \in G$ is
$\operatorname{ord}(a) = \abs{\langle a \rangle}$ (eventueel
oneindig); is zij eindig, dan is zij de kleinste $n \geq 1$ met
$a^n = e$, en geldt $a^k = e \iff \operatorname{ord}(a) \mid k$.
\end{definition}

\begin{proof}[Bewijs van de karakterisering van de orde]
Geldt $a^m = e$ voor zekere $m \geq 1$, zij dan $n \geq 1$ de
kleinste met $a^n = e$. De elementen $e, a, \dots, a^{n-1}$ zijn
paarsgewijs verschillend (uit $a^{i} = a^{j}$ met $0 \leq i < j <
n$ volgt $a^{j-i} = e$, in strijd met de minimaliteit), en elke
$a^k$ herleidt zich via de deling met rest $k = nq + r$ tot een
van hen: $\langle a\rangle$ telt precies $n$ elementen, en $a^k =
a^r = e \iff r = 0 \iff n \mid k$. Is geen enkele macht triviaal,
dan zijn alle $a^k$ ($k \in \Z$) verschillend (hetzelfde
delingsargument) en is de \omterm{def:b2:structures:generated}{orde} oneindig.
\end{proof}

\begin{theorem}[Lagrange]\label{thm:b2:structures:lagrange}
Zij $G$ een eindige groep en $H$ een ondergroep. Dan deelt $\abs
H$ het getal $\abs G$.\index{stelling van Lagrange} In het
bijzonder deelt de \omterm{def:b2:structures:generated}{orde} van elk element $\abs G$, en geldt
$a^{\abs G} = e$ voor alle $a \in G$.
\end{theorem}

\begin{proof}
De relatie $x \sim y \iff x^{-1}y \in H$ is een equivalentie
(reflexief omdat $e \in H$; symmetrisch via inversen; transitief
via producten). De klasse van $x$ is de \emph{linkernevenklasse}
$xH = \{xh : h \in H\}$, en $h \mapsto xh$ is een bijectie $H \to
xH$ (met inverse $y \mapsto x^{-1}y$): alle klassen tellen $\abs
H$ elementen. De klassen verdelen $G$ (de algemene
partitiestelling uit het volume van bachelorjaar 1), dus $\abs G =
\abs H \times (\text{aantal nevenklassen})$. Voor een element:
pas dit toe op $H = \langle a\rangle$; dan is $a^{\abs G} =
(a^{\operatorname{ord} a})^{\abs G / \operatorname{ord} a} = e$.
\end{proof}

\begin{example}[Nevenklassen in actie: $A_3$ binnen $\mathfrak{S}_3$]\label{ex:b2:structures:cosets}
Neem $G = \mathfrak{S}_3$ (\omterm{def:b2:structures:generated}{orde} $6$) en $H = A_3 =
\{\mathrm{id},\ (1\,2\,3),\ (1\,3\,2)\}$. De linkernevenklassen
zijn
\[
H = \{\mathrm{id},\ (1\,2\,3),\ (1\,3\,2)\},
\qquad
(1\,2)H = \{(1\,2),\ (2\,3),\ (1\,3)\} :
\]
twee klassen van drie elementen die $G$ verdelen, precies zoals
de telling $\abs G = \abs H \times (\text{aantal nevenklassen})$
voorschrijft --- en zichtbaar is dit de opsplitsing in even en
oneven permutaties. Merk op dat $(1\,3)H = (1\,2)H$ hoewel
$(1\,3) \neq (1\,2)$: nevenklassen zijn \emph{klassen}, niet
gelabeld door hun representanten, en $x^{-1}y \in H$ is de enige
legitieme vergelijking. Dit beeld van twee klassen is het
algemene beeld voor de signatuur: $A_n$ en zijn enige
begeleidende nevenklasse halveren $\mathfrak{S}_n$, en zo telt de
weekendopgave de bereikbare puzzelstanden.
\end{example}

\begin{example}\label{ex:b2:structures:lagrangeapp}
Twee onmiddellijke opbrengsten. \emph{Groepen van priemorde zijn
\omterm{def:b2:structures:generated}{cyclisch}:} is $\abs G = p$ priem en $a \neq e$, dan deelt
$\operatorname{ord}(a)$ het getal $p$ en is zij niet $1$, dus is
zij $p$: $\langle a\rangle = G$. \emph{Het ondergroepentralie van
$\Z/12\Z$:} volgens \cref{prop:b2:structures:cyclic} hieronder is
er precies één ondergroep per deler van $12$ --- van \omterm{def:b2:structures:generated}{orde} $1, 2,
3, 4, 6, 12$, achtereenvolgens \omterm{def:b2:structures:generated}{voortgebracht} door $\overline 0$,
$\overline 6$, $\overline 4$, $\overline 3$, $\overline 2$,
$\overline 1$. Tot slot een waarschuwing: de \emph{omkering} van
Lagrange faalt in het algemeen --- $A_4$ heeft \omterm{def:b2:structures:generated}{orde} $12$ maar
geen ondergroep van \omterm{def:b2:structures:generated}{orde} $6$, zoals we in de weekendopgave van
dit hoofdstuk bewijzen (\cref{pb:b2:structures:1}, vraag 14).
Lagrange beperkt de mogelijke \omterm{def:b2:structures:generated}{orden}; hij belooft ze niet.
\end{example}

\begin{proposition}[Cyclische groepen]\label{prop:b2:structures:cyclic}
Zij $G = \langle a \rangle$ \omterm{def:b2:structures:generated}{cyclisch} van \omterm{def:b2:structures:generated}{orde} $n$.
\begin{enumerate}
  \item $G$ is isomorf met $(\Z/n\Z, +)$, via $\overline k \mapsto
        a^k$.
  \item Elke ondergroep van $G$ is \omterm{def:b2:structures:generated}{cyclisch}; bij elke deler $d
        \mid n$ hoort precies één ondergroep van \omterm{def:b2:structures:generated}{orde} $d$, namelijk
        $\langle a^{n/d}\rangle$.
  \item $a^k$ brengt $G$ voort dan en slechts dan als $\gcd(k, n)
        = 1$: $G$ heeft $\varphi(n)$ voortbrengers (de functie van
        Euler\index{totientfunctie van Euler}).
\end{enumerate}
\end{proposition}

\begin{proof}
(1) De afbeelding $k \mapsto a^k$ van $\Z$ op $G$ is verenigbaar
met de congruentie modulo $n$ ($a^{k} = a^{k'} \iff n \mid k -
k'$, volgens de karakterisering van de \omterm{def:b2:structures:generated}{orde}); de universele
eigenschap (\cref{def:b2:structures:quotient}) levert een
welgedefinieerd bijectief morfisme vanuit $\Z/n\Z$.

(2) Zij $H \leq G$ niet triviaal en $m$ de kleinste $\geq 1$ met
$a^m \in H$. Deling met rest geeft $H = \langle a^m\rangle$ (voor
$a^k \in H$ dwingt $k = mq + r$ dat $a^r \in H$, dus $r = 0$), en
$m \mid n$ (deel $n$ door $m$: $a^{n \bmod m} \in H$). Dan is
$\abs H = n/m$; met $m = n/d$ wordt elke deler $d$ gerealiseerd.
Eenduidigheid: elke ondergroep van \omterm{def:b2:structures:generated}{orde} $d$ is volgens het
voorgaande van de vorm $\langle a^m \rangle$ met $n/m = d$ ---
dus ligt $m = n/d$ vast en daarmee de ondergroep.

(3) We beweren dat $\operatorname{ord}(a^k) = \frac{n}{\gcd(k,
n)}$. Schrijf $d = \gcd(k, n)$. Voor elke $m \geq 1$ geeft de
karakterisering van de \omterm{def:b2:structures:generated}{orde} uit \cref{def:b2:structures:generated}
de keten van equivalenties
\[
(a^k)^m = e
\iff n \mid km
\iff \frac{n}{d} \,\Big|\, \frac{k}{d}\,m
\iff \frac{n}{d} \,\Big|\, m ,
\]
waarbij de laatste stap op het lemma van Gauss steunt, want
$\frac nd$ en $\frac kd$ zijn onderling ondeelbaar. De kleinste
zulke $m$ is $\frac nd$: $\operatorname{ord}(a^k) = \frac
n{\gcd(k,n)}$, en dat is gelijk aan $n$ precies wanneer $\gcd(k,
n) = 1$. Er zijn $\varphi(n)$ zulke klassen $k$ modulo $n$.
\end{proof}

\section{De symmetrische groep}

\begin{definition}\label{def:b2:structures:sn}
$\mathfrak{S}_n$ is de groep van de permutaties van $\intint{1}{n}$
(\omterm{def:b2:structures:generated}{orde} $n!$). Een \emph{cykel}\index{cykel}
$(a_1\,a_2\,\cdots\,a_k)$ beeldt $a_1 \mapsto a_2 \mapsto \dots
\mapsto a_k \mapsto a_1$ af en laat al het overige vast; $k$ is
haar \emph{lengte}, en een $2$-cykel heet een
\emph{transpositie}\index{transpositie}. Twee cykels zijn
\emph{disjunct} wanneer hun dragers (de niet-vaste punten) dat
zijn.
\end{definition}

\begin{theorem}[Ontbinding in cykels]\label{thm:b2:structures:cycles}
Elke permutatie $\sigma \neq \mathrm{id}$ is een product van
paarsgewijs disjuncte \omterm{def:b2:structures:sn}{cykels}, eenduidig op de volgorde van de
factoren na. Disjuncte \omterm{def:b2:structures:sn}{cykels} commuteren, en
$\operatorname{ord}(\sigma)$ is het kleinste gemene veelvoud van
de lengten.
\end{theorem}

\begin{proof}
Beschouw op de drager van $\sigma$ de ``baanrelatie'': $x \sim y$
zodra $y = \sigma^k(x)$ voor zekere $k \in \Z$ --- een
equivalentierelatie. Elke klasse $\{x, \sigma(x), \dots,
\sigma^{k-1}(x)\}$ (eindig, zodat de iteraties terugkeren --- de
eerste herhaling moet wegens injectiviteit $x$ zelf zijn) draagt
de \omterm{def:b2:structures:sn}{cykel} $(x\ \sigma(x)\ \cdots\ \sigma^{k-1}(x))$, en $\sigma$ is
het product van deze \omterm{def:b2:structures:sn}{cykels}: op elke baan werkt alleen de
bijbehorende \omterm{def:b2:structures:sn}{cykel}. Eenduidigheid: elke ontbinding in disjuncte
\omterm{def:b2:structures:sn}{cykels} reproduceert precies de banen (de \omterm{def:b2:structures:sn}{cykel} door $x$ moet $(x\
\sigma(x)\ \cdots)$ zijn). Disjuncte \omterm{def:b2:structures:sn}{cykels} commuteren omdat zij
disjuncte punten verplaatsen; de uitspraak over de \omterm{def:b2:structures:generated}{orde} volgt
omdat $\sigma^m = \mathrm{id}$ dan en slechts dan als de $m$-de
macht van elke \omterm{def:b2:structures:sn}{cykel} dat is (disjunctheid), dan en slechts dan als
elke lengte $m$ deelt.
\end{proof}

\begin{example}[Cykeltype als telling]\label{ex:b2:structures:cycletype}
Hoeveel permutaties van $\mathfrak{S}_9$ hebben cykeltype $(4, 3,
2)$ --- één $4$-cykel, één $3$-cykel, één \omterm{def:b2:structures:sn}{transpositie}? Kies de
dragers en de cyclische volgorden:
\[
\frac{9!}{4\cdot 3\cdot 2}
= \frac{362\,880}{24} = 15\,120 :
\]
zet de negen symbolen op een rij ($9!$ manieren), haak de eerste
vier, de volgende drie en de laatste twee in \omterm{def:b2:structures:sn}{cykels}, en deel door
de rotaties binnen elk haakje ($4$, $3$ en $2$ stuks) die dezelfde
permutatie opleveren. (De cykellengten zijn hier verschillend,
dus is er geen verdere deling nodig; bij gelijke lengten zou men
ook nog door de permutaties van de gelijke haakjes moeten delen.)
Elke zulke permutatie heeft \omterm{def:b2:structures:generated}{orde} $\operatorname{lcm}(4,3,2) = 12$
en signatuur $(-1)^3(-1)^2(-1)^1 = +1$
(\cref{thm:b2:structures:cycles} en de signatuurstelling
hieronder). Eén partitie van $9$, één conjugatieklasse, één
telling --- de combinatoriek van $\mathfrak{S}_n$ is de
rekenkunde van de partities.
\end{example}

\begin{theorem}[Signatuur]\label{thm:b2:structures:signature}
Er bestaat precies één groepsmorfisme $\varepsilon \colon
\mathfrak{S}_n \to \{\pm 1\}$ (voor $n \geq 2$) dat op
\omterm{def:b2:structures:sn}{transposities} de waarde $-1$ aanneemt: de
\emph{signatuur}\index{signatuur}. Bovendien is
$\varepsilon(\sigma) = (-1)^{I(\sigma)}$, waarbij $I(\sigma)$ het
aantal \emph{inversies} is (paren $i < j$ met $\sigma(i) >
\sigma(j)$), heeft een $k$-cykel signatuur $(-1)^{k-1}$, en heeft
de \emph{alternerende groep} $A_n = \ker\varepsilon$ \omterm{def:b2:structures:generated}{orde}
$\frac{n!}{2}$.
\end{theorem}

\begin{proof}
\emph{Bestaan.} Zet voor $\sigma \in \mathfrak{S}_n$
\[
\varepsilon(\sigma)
= \prod_{1 \leq i < j \leq n}
\frac{\sigma(j) - \sigma(i)}{j - i} .
\]
De absolute waarden van de factoren vermenigvuldigen tot $1$ (de
ongeordende paren $\{\sigma(i), \sigma(j)\}$ doorlopen alle
paren), dus $\varepsilon(\sigma) = (-1)^{I(\sigma)} \in
\{\pm1\}$. Morfisme: voor $\sigma, \tau$ is
\[
\varepsilon(\sigma\tau)
= \prod_{i<j} \frac{\sigma(\tau(j)) - \sigma(\tau(i))}{j - i}
= \prod_{i<j} \frac{\sigma(\tau(j)) - \sigma(\tau(i))}{\tau(j) -
\tau(i)} \cdot \prod_{i<j} \frac{\tau(j) - \tau(i)}{j - i}
= \varepsilon(\sigma)\,\varepsilon(\tau),
\]
waarbij het middelste product gelijk is aan
$\varepsilon(\sigma)$ na herindexering langs de paren
$\{\tau(i), \tau(j)\}$ (elk ongeordend paar komt één keer voor,
en teller en noemer wisselen samen van teken). Een \omterm{def:b2:structures:sn}{transpositie}
$\tau = (a\,b)$ met $a < b$ heeft een oneven aantal inversies;
precies geteld zijn de geïnverteerde paren $(i, j)$, $i < j$, met
$\tau(i) > \tau(j)$:
\[
(a, j) \ \text{voor } a < j < b, \qquad
(i, b) \ \text{voor } a < i < b, \qquad
(a, b) \ \text{zelf},
\]
dat wil zeggen $(b - a - 1) + (b - a - 1) + 1 = 2(b - a) - 1$
stuks, dus oneven. (Alternatief: reken $(1\,2)$ rechtstreeks na,
met één inversie, en conjugeer --- geconjugeerden hebben dezelfde
signatuur, want $\varepsilon$ is een morfisme naar een abelse
groep.) Bijgevolg is $\varepsilon((a\,b)) = (-1)^{2(b-a)-1} = -1$.

\emph{Eenduidigheid.} \omterm{def:b2:structures:sn}{Transposities} brengen $\mathfrak{S}_n$
voort (elke \omterm{def:b2:structures:sn}{cykel} voldoet aan $(a_1\cdots a_k) =
(a_1\,a_k)(a_1\,a_{k-1})\cdots(a_1\,a_2)$, en
\cref{thm:b2:structures:cycles} maakt het af); een morfisme naar
$\{\pm1\}$ ligt vast door zijn waarden op voortbrengers.

\emph{Gevolgen.} De cykelidentiteit hierboven schrijft een
$k$-cykel als $k - 1$ \omterm{def:b2:structures:sn}{transposities}: signatuur $(-1)^{k-1}$. Voor
$A_n$: het morfisme $\varepsilon$ is surjectief (\omterm{def:b2:structures:sn}{transposities}
bestaan zodra $n \geq 2$), en de twee ``nevenklassen'' $A_n$ en
$(1\,2)A_n$ zijn \omterm{def:b2:structures:countable}{gelijkmachtig} en verdelen $\mathfrak{S}_n$ (het
argument van Lagrange): $\abs{A_n} = \frac{n!}{2}$.
\end{proof}

\begin{example}\label{ex:b2:structures:snexample}
$\sigma = \begin{pmatrix} 1&2&3&4&5&6\\ 3&6&5&4&1&2 \end{pmatrix}
= (1\,3\,5)(2\,6)$: \omterm{def:b2:structures:generated}{orde} $\operatorname{lcm}(3,2) = 6$, signatuur
$(-1)^{2}\cdot(-1)^{1} = -1$. De signatuur is de snelste
pariteitstest op schuifelingen --- en de motor achter de
determinant in \cref{ch:b2:linalg}.
\end{example}

\begin{example}[Drie wegen naar één teken]\label{ex:b2:structures:threeroads}
Zij $\sigma \in \mathfrak{S}_5$ de permutatie die $1, 2, 3, 4, 5$
naar $3, 5, 4, 1, 2$ stuurt. \emph{Via \omterm{def:b2:structures:sn}{cykels}:} $1 \mapsto 3
\mapsto 4 \mapsto 1$ en $2 \mapsto 5 \mapsto 2$, dus $\sigma =
(1\,3\,4)(2\,5)$ en $\varepsilon(\sigma) = (-1)^{2}(-1)^{1} =
-1$. \emph{Via inversies:} in de waardenlijst $3, 5, 4, 1, 2$
zijn de paren buiten volgorde $(3,1)$, $(3,2)$, $(5,4)$, $(5,1)$,
$(5,2)$, $(4,1)$, $(4,2)$: zeven stuks, en $(-1)^7 = -1$.
\emph{Via \omterm{def:b2:structures:sn}{transposities}:} $\sigma = (1\,4)(1\,3)(2\,5)$, drie
factoren, $(-1)^3 = -1$. Drie berekeningen, één pariteit: de
eenduidigheid in \cref{thm:b2:structures:signature} garandeert
dat geen enkel boekhoudschema ze ooit kan laten botsen --- en
juist dat maakt $\varepsilon$ bruikbaar als invariant (zie de
weekendopgave).
\end{example}

\begin{remark}[Waar de signatuur vandaan verder gaat]
De signatuur is het zaad van drie latere oogsten: zij bouwt de
determinant en haar productregel in \cref{ch:b2:linalg}; zij
levert pariteitsinvarianten voor combinatorische puzzels (de
weekendopgave van dit hoofdstuk lost er de vijftienpuzzel mee op);
en de alternerende groepen $A_n$ die zij definieert worden
centraal in het volume van bachelorjaar 3, waar hun enkelvoudigheid
voor $n \geq 5$ verklaart waarom vergelijkingen van graad $5$ niet
met wortelvormen op te lossen zijn.
\end{remark}

\section{Ringen, idealen, quotiënten}

\begin{definition}[Ideaal]\label{def:b2:structures:ideal}
Zij $A$ een commutatieve ring. Een \emph{ideaal}\index{ideaal} $I
\subseteq A$ is een additieve ondergroep zodanig dat $a x \in I$
voor alle $a \in A$ en $x \in I$. Kernen van ringmorfismen zijn
idealen; er geldt $I = A$ dan en slechts dan als $1 \in I$, dan en
slechts dan als $I$ een eenheid bevat. Het door $x$
\emph{\omterm{def:b2:structures:generated}{voortgebrachte}} ideaal is $xA = \{xa\}$ (een
\emph{hoofdideaal}).
\end{definition}

\begin{theorem}[{Idealen van $\Z$ en van $K[X]$}]\label{thm:b2:structures:principal}
Elk \omterm{def:b2:structures:ideal}{ideaal} van $\Z$ is $n\Z$ voor een unieke $n \in \N$; elk
\omterm{def:b2:structures:ideal}{ideaal} van $K[X]$ ($K$ een lichaam) is $P\,K[X]$ voor een unieke
monische (of nul) $P$. Bijgevolg bestaan er in beide ringen
grootste gemene delers met Bézout-relaties: $x\Z + y\Z =
\gcd(x,y)\Z$, en evenzo voor veeltermen.
\end{theorem}

\begin{proof}
Voor $\Z$ is dit de ondergroepenstelling uit het volume van
bachelorjaar 1 (een \omterm{def:b2:structures:ideal}{ideaal} is in het bijzonder een ondergroep, en
$n\Z$ is een \omterm{def:b2:structures:ideal}{ideaal}). Voor $K[X]$: zij $I \neq \{0\}$ een \omterm{def:b2:structures:ideal}{ideaal}
en $P \in I$ ongelijk aan nul van minimale graad, monisch
genormeerd. Voor $F \in I$ geeft de deling met rest $F = PQ + R$
dat $R = F - PQ \in I$ met $\deg R < \deg P$: de minimaliteit
dwingt $R = 0$, dus $I = P\,K[X]$. Eenduidigheid: twee monische
voortbrengers delen elkaar. De uitspraken van Bézout zijn de
gelijkheid van het \omterm{def:b2:structures:ideal}{ideaal} $x\Z + y\Z$ (respectievelijk zijn
veeltermtegenhanger) met het hoofdideaal van de grootste gemene
deler --- precies de definitie van de ggd die in bachelorjaar 1
gebruikt werd, nu herkend als een uitspraak over idealen.
\end{proof}

\begin{example}[Een ggd van veeltermen, langs twee wegen]\label{ex:b2:structures:polygcd}
Bereken $\gcd(X^3 - 1,\ X^2 - 1)$ in $\Q[X]$. \emph{Via Euclides:}
\[
X^3 - 1 = X\,(X^2 - 1) + (X - 1),
\qquad
X^2 - 1 = (X + 1)(X - 1) + 0 ,
\]
dus de ggd is $X - 1$, en terugsubstitueren geeft de
Bézout-relatie
\[
X - 1 = 1\cdot(X^3 - 1) - X\cdot(X^2 - 1).
\]
\emph{Via idealen:} het \omterm{def:b2:structures:ideal}{ideaal} $(X^3 - 1)\Q[X] + (X^2 - 1)\Q[X]$
is een hoofdideaal (\cref{thm:b2:structures:principal}); het
bevat $X - 1$ (zie de formule hierboven) en het ligt in $(X -
1)\Q[X]$ (beide voortbrengers verdwijnen in $1$ en zijn dus
veelvouden van $X - 1$): de monische voortbrenger is $X - 1$. Het
inzicht om te onthouden: het idealenstandpunt herkent de ggd
\emph{zonder te delen} --- de gemeenschappelijke nulpunten
lokaliseren het \omterm{def:b2:structures:ideal}{ideaal}, en Euclides bevestigt het slechts.
\end{example}

\begin{definition}[Quotiëntring $\Z/n\Z$, opnieuw bekeken]\label{def:b2:structures:quotientring}
Voor een \omterm{def:b2:structures:ideal}{ideaal} $I$ van $A$ is de relatie $x \sim y \iff x - y \in
I$ een equivalentie die verenigbaar is met $+$ en $\times$; de
\omterm{def:b2:structures:quotient}{quotiëntverzameling} $A/I$ erft een ringstructuur --- de
\emph{quotiëntring}\index{quotiëntring} --- die van $\pi \colon A
\to A/I$ een morfisme met kern $I$ maakt. Voor $A = \Z$ en $I =
n\Z$ is dit de $\Z/n\Z$ uit het volume van bachelorjaar 1, nu met
haar universele eigenschap: elk morfisme dat $I$ doodt,
factoriseert over $A/I$.
\end{definition}

\begin{theorem}[Chinese reststelling, ringvorm]\label{thm:b2:structures:crt}
Als $\gcd(m, n) = 1$, dan is de afbeelding
\[
\Z/mn\Z \longrightarrow \Z/m\Z \times \Z/n\Z,
\qquad
\overline{x} \longmapsto (x \bmod m,\; x \bmod n)
\]
een ringisomorfisme.\index{Chinese reststelling} Bijgevolg geldt
$\varphi(mn) = \varphi(m)\varphi(n)$ voor onderling ondeelbare $m,
n$, en
\[
\varphi(n) = n \prod_{p \mid n} \Bigl(1 - \frac 1p\Bigr)
\quad (p \text{ priem}).
\]
\end{theorem}

\begin{proof}
De afbeelding is een welgedefinieerd ringmorfisme (de
verenigbaarheden zijn onmiddellijk). Injectiviteit: uit $x \equiv
0$ modulo $m$ en modulo $n$ met $\gcd(m,n) = 1$ volgt $mn \mid x$
(Gauss). Surjectiviteit: beide leden tellen $mn$ elementen, dus
volstaat de injectiviteit (gelijke eindige aantallen) --- of
expliciet: uit een Bézout-relatie $um + vn = 1$ gaat de klasse van
\[
x = b\,um + a\,vn
\]
naar $(a \bmod m,\ b \bmod n)$, want $vn = 1 - um \equiv 1
\pmod m$ maakt $x \equiv a \pmod m$, en symmetrisch modulo $n$
--- het recept dat numeriek wordt uitgevoerd in
\cref{ex:b2:structures:crtinverse}. Eenheden komen overeen met
paren eenheden (de eenheden van een productring zijn juist de
paren eenheden), zodat $\varphi(mn) = \varphi(m)\varphi(n)$. Voor
een priemmacht is $\varphi(p^k) = p^k - p^{k-1}$ (de
niet-eenheden modulo $p^k$ zijn de veelvouden van $p$); de
multiplicativiteit zet daaruit de productformule in elkaar.
\end{proof}

\begin{example}[Het Chinese isomorfisme omkeren]\label{ex:b2:structures:crtinverse}
Neem $m = 8$ en $n = 9$. De inverse van het isomorfisme wordt
expliciet gemaakt door de twee \emph{idempotenten}: zoek $u \equiv
1 \pmod 8$, $u \equiv 0 \pmod 9$ en $v \equiv 0 \pmod 8$, $v
\equiv 1 \pmod 9$. Uit $u = 9k \equiv 1 \pmod 8$ volgt $k \equiv
1$, dus $u = 9$; uit $v = 8k \equiv 1 \pmod 9$ volgt $-k \equiv
1$, $k \equiv 8$, dus $v = 64$. De klasse van $x = 9a + 64b$
modulo $72$ is dan de unieke oplossing van $x \equiv a \pmod 8$,
$x \equiv b \pmod 9$: voor $a = 3$ en $b = 5$ krijgt men $27 +
320 = 347 \equiv 59 \pmod{72}$ --- precies de tussenwaarde die
substitutie oplevert in \cref{exo:b2:structures:8}. Het inzicht
om te onthouden: $u$ en $v$ voldoen modulo $72$ aan $u + v \equiv
1$, $uv \equiv 0$, $u^2 \equiv u$ en $v^2 \equiv v$; zij zijn de
beelden van $(1, 0)$ en $(0, 1)$, en elke Chinese ontbinding is
uiteindelijk een ontbinding van $1$ in orthogonale idempotenten.
\end{example}

\begin{theorem}[Euler; Fermat opnieuw]\label{thm:b2:structures:euler}
De eenheden van $\Z/n\Z$ vormen een groep van \omterm{def:b2:structures:generated}{orde} $\varphi(n)$;
bijgevolg geldt voor $\gcd(a, n) = 1$
\[
a^{\varphi(n)} \equiv 1 \pmod n
\qquad (\text{de stelling van Euler\index{stelling van Euler}}),
\]
en de kleine stelling van Fermat is het geval $n = p$ priem, nu
één regel verwijderd van Lagrange.
\end{theorem}

\begin{proof}
De inverteerbare klassen zijn precies die van de gehele getallen
die onderling ondeelbaar zijn met $n$ (volume van bachelorjaar 1):
$\varphi(n)$ stuks, en zij vormen een groep onder de
vermenigvuldiging. Lagrange (\cref{thm:b2:structures:lagrange}):
elk element tot de macht de \omterm{def:b2:structures:generated}{orde} van de groep is de eenheid.
\end{proof}

\begin{example}[Een eenhedengroep zonder voortbrenger]\label{ex:b2:structures:units15}
De groep $(\Z/15\Z)^*$ telt $\varphi(15) = \varphi(3)\varphi(5)
= 8$ elementen. Is zij \omterm{def:b2:structures:generated}{cyclisch}? Bereken de \omterm{def:b2:structures:generated}{orden} met behulp van
het Chinese isomorfisme $(\Z/15\Z)^* \simeq (\Z/3\Z)^* \times
(\Z/5\Z)^*$ (een eenheid modulo $15$ is een paar eenheden): de
factoren hebben \omterm{def:b2:structures:generated}{orde} $2$ en $4$, dus deelt de \omterm{def:b2:structures:generated}{orde} van elk element
$\operatorname{lcm}(2, 4) = 4 < 8$ --- geen enkel element brengt
voort. Concreet:
\[
2^4 = 16 \equiv 1, \qquad
4^2 = 16 \equiv 1, \qquad
7^4 \equiv 1, \qquad
11^2 = 121 \equiv 1, \qquad
14^2 \equiv 1 \pmod{15} :
\]
\omterm{def:b2:structures:generated}{orden} $4, 2, 4, 2, 2$ en nooit $8$. Vergelijk dit met
\cref{exo:b2:structures:10}: $(\Z/p\Z)^*$ \emph{is} \omterm{def:b2:structures:generated}{cyclisch} voor
priem $p$, omdat de eenhedengroep daar in een lichaam ligt. De
stelling van Euler blijft gelden met exponent $\varphi(15) = 8$,
maar de werkelijke universele exponent is hier $4$ --- Euler geeft
een bovengrens, niet altijd de scherpste.
\end{example}

\begin{definition}[Algebra]\label{def:b2:structures:algebra}
Een \emph{$K$-algebra}\index{algebra} is een $K$-vectorruimte $A$
met een ringstructuur waarvan de vermenigvuldiging
$K$-bilineair is. Voorbeelden: $K[X]$, $\mathcal{M}_n(K)$,
$\mathcal{L}(E)$, functieruimten $\mathcal{F}(X, K)$, en $\C$ als
$\R$-algebra. Morfismen van algebra's zijn lineaire ringmorfismen;
de \emph{evaluatie} $P \mapsto P(u)$ van $K[X]$ naar
$\mathcal{L}(E)$ (of $\mathcal{M}_n(K)$) is het centrale
voorbeeld, en de motor van \cref{ch:b2:reduction}.
\end{definition}

\begin{example}[Een evaluatiemorfisme en zijn kern]\label{ex:b2:structures:evaluation}
Neem $A = \begin{pmatrix}0 & 1\\ 0 & 0\end{pmatrix}$ en de
evaluatie $\varepsilon_A \colon \R[X] \to \mathcal{M}_2(\R)$, $P
\mapsto P(A)$. Omdat $A^2 = 0$ is
\[
P(A) = P(0)\,I + P'(0)\,A =
\begin{pmatrix} P(0) & P'(0)\\ 0 & P(0)\end{pmatrix},
\]
(alleen de constante en de lineaire term van $P$ overleven).
Bijgevolg is $\ker\varepsilon_A = \{P : P(0) = P'(0) = 0\} =
X^2\,\R[X]$: een hoofdideaal, precies zoals
\cref{thm:b2:structures:principal} voorspelt, \omterm{def:b2:structures:generated}{voortgebracht} door
de monische $X^2$ van kleinste graad in de kern --- de
\emph{minimale veelterm} van $A$, hoofdrolspeler in
\cref{ch:b2:reduction}. Het beeld is de tweedimensionale
commutatieve \omterm{def:b2:structures:algebra}{algebra} $\{aI + bA\}$: evaluatiemorfismen krimpen het
oneindigdimensionale $\R[X]$ tot kleine, berekenbare \omterm{def:b2:structures:algebra}{algebra}'s.
\end{example}

\begin{remark}[Vooruitblik: drie melodieën om op te letten]\label{rem:b2:structures:perspectives}
Drie structurele ideeën uit dit hoofdstuk keren in het hele
volume terug, telkens zwaarder georkestreerd. \emph{Factoriseren
over een quotiënt} (\cref{def:b2:structures:quotient}): het bouwt
hier $\Z/n\Z$, het definieert afbeeldingen op oplossingsruimten
van lineaire stelsels in \cref{ch:b2:linalg}, en het ligt stil
onder elk argument van het type ``welgedefinieerd op klassen''.
\emph{Invarianten}: de signatuur is een morfisme naar $\{\pm1\}$
waar geen enkele toegestane zet omheen kan --- dezelfde logica
geeft de productregel van de determinant
(\cref{ch:b2:linalg}), de invariantie van het spoor onder
gelijkvormigheid, en de behouden grootheden van
\cref{ch:b2:diffeq}. \emph{Tellen tegen een structuur in}:
Lagrange telt via nevenklassen, dimensie telt via basissen
(\cref{ch:b2:linalg}), multipliciteit telt via veeltermgraden
(\cref{ch:b2:reduction}); zodra een grens wonderbaarlijk lijkt,
doet ergens een partitie of een gradering het telwerk.
\end{remark}

\begin{remark}[Klassieke valkuilen]\label{rem:b2:structures:pitfalls}
Vier klassiekers. (i) Van een afbeelding op een quotiënt moet
worden nagegaan dat zij \emph{welgedefinieerd} is:
``$\overline x \mapsto$ (formule in $x$)'' is alleen legitiem
wanneer de formule constant is op de klassen --- de
verenigbaarheid uit \cref{def:b2:structures:quotient}, geen
formaliteit. (ii) $\operatorname{ord}(ab) =
\operatorname{lcm}(\operatorname{ord}a, \operatorname{ord}b)$ is
in het algemeen \emph{onjuist}, zelfs voor commuterende elementen
($a$ en $a^{-1}$); \cref{exo:b2:structures:4} geeft de juiste
uitspraak onder commutatie en onderlinge ondeelbaarheid, en
disjuncte \omterm{def:b2:structures:sn}{cykels} de juiste versie voor permutaties. (iii)
\omterm{def:b2:structures:countable}{Aftelbaarheid} overleeft \omterm{def:b2:structures:countable}{aftelbare} \emph{verenigingen} en eindige
\emph{producten}, maar geen \omterm{def:b2:structures:countable}{aftelbare} producten: $\{0,1\}^{\N}$
is overaftelbaar (\cref{exo:b2:structures:3}) hoewel elke factor
twee elementen heeft. (iv) Cantor--Bernstein heeft alleen
injecties in beide richtingen nodig, maar de bijectie die de
stelling bouwt is doorgaans discontinu en niet expliciet ---
verwacht er geen formule voor
(\cref{ex:b2:structures:cbexample}).
\end{remark}

\begin{remark}[Waar dit hoofdstuk wordt gebruikt]
Vrijwel overal. De signatuur bouwt determinanten
(\cref{ch:b2:linalg}); het evaluatiemorfisme $P \mapsto P(u)$ en
de hoofdidealen van $K[X]$ leveren de minimale veeltermen en de
kernontbindingen van \cref{ch:b2:reduction}; \omterm{def:b2:structures:countable}{aftelbaarheid} is het
toneel waarop \cref{ch:b2:proba} speelt (kansrekening op
\omterm{def:b2:structures:countable}{aftelbare} ruimten) en de reden dat de topologie telkens weer
\omterm{def:b2:structures:countable}{aftelbare} dichte verzamelingen produceert (\cref{ch:b2:metric}).
De quotiëntconstructie $A/I$ wordt in het volume van bachelorjaar
3 opnieuw ingezet om lichamen $K[X]/(P)$ te bouwen en daaruit de
Galoistheorie: de universele eigenschap die hier bewezen is,
wordt daar woordelijk gebruikt.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b2:structures:1}
Welke van de volgende verzamelingen zijn \omterm{def:b2:structures:countable}{aftelbaar}? De verzameling
van de eindige deelverzamelingen van $\N$; de verzameling van

$\tau = (2\,3\,7)$. Bereken $\sigma\tau$ en $\tau\sigma$ in
disjuncte \omterm{def:b2:structures:sn}{cykels}, de \omterm{def:b2:structures:generated}{orden} en signaturen van alle vier de

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:structures:3}
Construeer expliciete injecties die aantonen dat

van de binaire rijen paarsgewijs \omterm{def:b2:structures:countable}{gelijkmachtig} zijn \emph{(binaire
ontwikkelingen in beide richtingen; Cantor--Bernstein neemt het
ongemak van de dubbele voorstelling weg)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:structures:4}
Zij $G$ een groep en zijn $a, b \in G$ commuterende elementen van
eindige, onderling ondeelbare \omterm{def:b2:structures:generated}{orden} $m$ en $n$. Bewijs dat
$\operatorname{ord}(ab) = mn$. Toon met een voorbeeld in
$\mathfrak{S}_3$ aan dat de commutatie onmisbaar is.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:structures:5}
Zij $G$ een eindige groep van even \omterm{def:b2:structures:generated}{orde}. Bewijs dat $G$ een
element van \omterm{def:b2:structures:generated}{orde} $2$ bevat. \emph{(Koppel elk element aan zijn
inverse en tel de elementen die aan zichzelf gekoppeld zijn.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:structures:6}
Bewijs dat $A_n$ ($n \geq 3$) door de $3$-cykels wordt
\omterm{def:b2:structures:generated}{voortgebracht}. \emph{(Een product van twee \omterm{def:b2:structures:sn}{transposities} is een
$3$-cykel of een product van twee $3$-cykels.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:structures:7}
Bepaal alle groepsmorfismen: van $(\Q, +)$ naar $(\Z, +)$; van
$(\Z/n\Z, +)$ naar $(\Z/m\Z, +)$ \emph{(tel ze: $\gcd(m,n)$)}; van
$(\Q, +)$ naar $(\Q_+^*, \times)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:structures:8}
Bereken met de Chinese reststelling $\varphi(360)$, bepaal alle
$x$ met $x \equiv 3 \pmod 8$, $x \equiv 5 \pmod 9$ en $x \equiv 2
\pmod 5$, en bereken de laatste twee cijfers van $3^{2026}$
\emph{(Euler modulo $100$; let op: werk modulo $4$ en modulo
$25$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:structures:9}
Bewijs dat een eindig integriteitsgebied een lichaam is. Leid
daaruit (opnieuw) af dat $\Z/n\Z$ een lichaam is dan en slechts
dan als $n$ priem is.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:structures:10}
(Een klassieker) Zij $K$ een lichaam en $G$ een \emph{eindige}
ondergroep van $(K^*, \times)$. Bewijs dat $G$ \omterm{def:b2:structures:generated}{cyclisch} is.
\emph{Aanwijzing: zij $m$ de maximale \omterm{def:b2:structures:generated}{orde} onder de elementen van
$G$; toon aan dat de \omterm{def:b2:structures:generated}{orde} van elk element $m$ deelt (gebruik
\cref{exo:b2:structures:4} op geschikte onderling ondeelbare
delen), zodat heel $G$ aan $x^m = 1$ voldoet; tel vervolgens de
nulpunten van $X^m - 1$.} In het bijzonder is $(\Z/p\Z)^*$
\omterm{def:b2:structures:generated}{cyclisch}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:structures:11}
Bewijs dat de groep $(\Q, +)$ niet \omterm{def:b2:structures:generated}{cyclisch} is, en sterker nog:
dat zij niet eens eindig \omterm{def:b2:structures:generated}{voortgebracht} is. Bewijs anderzijds dat
elke eindig \omterm{def:b2:structures:generated}{voortgebrachte} ondergroep van $(\Q, +)$ \omterm{def:b2:structures:generated}{cyclisch} is.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:structures:12}
(Criterium van Dedekind) Bewijs dat elke oneindige verzameling een
\omterm{def:b2:structures:countable}{aftelbare} deelverzameling bevat, en leid daaruit af dat een
verzameling $E$ oneindig is dan en slechts dan als zij
\omterm{def:b2:structures:countable}{gelijkmachtig} is met een echte deelverzameling van zichzelf.
\emph{(Voor de rechtstreekse implicatie: verschuif een \omterm{def:b2:structures:countable}{aftelbare}
deelverzameling één stap; voor de omkering: denk aan het
duivenhokprincipe.)}
\end{exercise}

\section{Probleem: De vijftienpuzzel}

De vijftienpuzzel is een bakje van $4 \times 4$ met vijftien
schuivende plaatjes, genummerd van $1$ tot $15$, en één lege plek;
een zet schuift een plaatje dat aan de lege plek grenst daarin. In
de jaren 1890 maakte Sam Loyd de puzzel beroemd door \$1000 uit te
loven aan wie de plaatjes $14$ en $15$ zou verwisselen en alle
overige plaatjes op hun plaats zou terugzetten. Niemand heeft dat
geld ooit opgehaald, en deze weekendopgave bewijst beide helften
van de reden: de signatuur uit
\cref{thm:b2:structures:signature} verbiedt Loyds verwisseling,
en --- de moeilijkere, constructieve helft --- \emph{alles} wat de
signatuur toelaat is werkelijk oplosbaar. De volledige uitspraak
is de stelling van Johnson en Story (1879).

\begin{problem}[{Weekendopgave --- de oplosbaarheidsstelling van
Johnson en Story}]
\label{pb:b2:structures:1}
Nummer de vakjes van $1$ tot $16$ in leesvolgorde (van links naar
rechts, van boven naar beneden), zodat vakje $k$ in rij $i$ en
kolom $j$ ligt met $k = 4(i - 1) + j$. Vakje $16$ (rechtsonder) is
het \emph{thuis} van de lege plek; we behandelen de lege plek als
een zestiende plaatje, geschreven als $b$ en vereenzelvigd met het
getal $16$. Een \emph{stand} is een bijectie $\sigma \colon
\intint1{16} \to \intint1{16}$, vakje $\mapsto$ inhoud; de
\emph{opgeloste} stand is $\sigma = \mathrm{id}$. Overal is
$\varepsilon$ de signatuur uit
\cref{thm:b2:structures:signature}, en twee vakjes heten
\emph{buren} wanneer zij een rib van het bakje delen.

\medskip
\noindent\textbf{Deel I --- Standen, zetten, signaturen.}
\begin{enumerate}
  \item Verantwoord dat de standen precies de elementen van
        $\mathfrak{S}_{16}$ zijn, zodat er $16! =
        20\,922\,789\,888\,000$ van zijn, en dat het aantal
        toegestane zetten vanuit een gegeven stand $2$, $3$ of $4$
        bedraagt, naargelang de lege plek in een hoek, aan een
        rand of in het inwendige ligt.
  \item Zij $\sigma$ een stand, $p = \sigma^{-1}(16)$ het vakje
        van de lege plek en $c$ een vakje dat aan $p$ grenst. Toon
        aan dat het schuiven van het plaatje van $c$ naar $p$ de
        stand $\sigma' = \sigma \circ \tau$ oplevert met $\tau =
        (p\ c)$, en leid af dat elke zet de signatuur omklapt:
        $\varepsilon(\sigma') = -\varepsilon(\sigma)$.
  \item Kleur het bakje als een schaakbord: $\chi(k) = (-1)^{i+j}$
        voor het vakje $k$ in rij $i$ en kolom $j$. Toon aan dat
        elke zet $\chi(\text{vakje van de lege plek})$ omklapt, en
        leid af dat een reeks zetten die de lege plek naar haar
        beginvakje terugbrengt, een even lengte heeft.
  \item Toon aan dat
        \[
        I(\sigma) = \varepsilon(\sigma)\,
        \chi\bigl(\sigma^{-1}(16)\bigr)
        \]
        onder elke toegestane zet invariant is, en bereken
        $I(\mathrm{id})$.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Loyds prijsvraag: de invariant aan
het werk.}
\begin{enumerate}[resume]
  \item Loyds stand $\sigma_L$ stemt met de opgeloste stand
        overeen, behalve dat de vakjes $14$ en $15$ de plaatjes
        $15$ en $14$ dragen. Bereken $I(\sigma_L)$ en besluit dat
        geen enkele reeks zetten $\sigma_L$ met de opgeloste stand
        verbindt: Loyds \$1000 heeft nooit gevaar gelopen.
  \item Toon aan dat precies de helft van alle standen aan $I =
        +1$ voldoet: $\abs{\{\sigma : I(\sigma) = +1\}} = 16!/2$.
        \emph{(Koppel, bij vast vakje voor de lege plek, de
        standen aan elkaar door met één vaste \omterm{def:b2:structures:sn}{transpositie} van
        twee andere vakjes samen te stellen.)}
  \item Toon aan dat elke zet door een toegestane zet ongedaan
        wordt gemaakt, dat ``$\sigma'$ is vanuit $\sigma$
        bereikbaar met toegestane zetten'' een equivalentierelatie
        is, en dat de klasse $R$ van de opgeloste stand voldoet
        aan $R \subseteq \{I = +1\}$. Besluit dat er minstens twee
        klassen zijn.
  \item Neem aan dat de lege plek thuis is: $\sigma(16) = 16$.
        Toon aan dat $I(\sigma) = \varepsilon(\rho)$, waarbij
        $\rho \in \mathfrak{S}_{15}$ de beperking van $\sigma$ tot
        de vakjes $1, \dots, 15$ is, en dat elke stand met
        toegestane zetten naar een stand met de lege plek thuis
        kan worden gebracht. Besluit: om $R = \{I = +1\}$ te
        bewijzen volstaat het elke \emph{even} permutatie van de
        vijftien niet-thuisvakjes te realiseren met een reeks
        zetten die begint en eindigt met de lege plek thuis.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Rondgangen van de lege plek en de
programmagroep.} Een \emph{programma} is een eindige reeks
toegestane zetten, gestart vanuit een stand met de lege plek
thuis, waarvan de eindstand de lege plek opnieuw thuis heeft. Het
\emph{effect} ervan is de permutatie $\pi$ van de vakjes
gedefinieerd door: de inhoud van vakje $x$ eindigt in vakje
$\pi(x)$.
\begin{enumerate}[resume]
  \item Toon aan dat een programma dat vanuit $\sigma$ wordt
        uitgevoerd in $\sigma \circ \pi^{-1}$ eindigt; dat twee
        programma's na elkaar hun effecten samenstellen; en dat de
        verzameling $H$ van alle effecten een ondergroep van
        $\mathfrak{S}_{15}$ is (permutaties van de vakjes $1,
        \dots, 15$) die in de alternerende groep $A_{15}$ ligt.
  \item (De elementaire rondgang) Schuif, met de lege plek thuis,
        de lege plek rond het blok van $2 \times 2$ rechtsonder:
        vakjes $16 \to 12 \to 11 \to 15 \to 16$. Toon aan dat het
        effect de $3$-cykel $(11\ 12\ 15)$ is, en dat de
        omgekeerde rondgang $(11\ 15\ 12)$ geeft. Beide liggen in
        $H$.
  \item (De grote rondgang) Ga na dat
        \[
        16 \to 15 \to 14 \to 13 \to 9 \to 5 \to 1 \to 2 \to 3
        \to 4 \to 8 \to 7 \to 6 \to 10 \to 11 \to 12 \to 16
        \]
        een gesloten wandeling door alle zestien vakjes is
        (uitsluitend stappen tussen buren), en dat het effect
        ervan de $15$-cykel
        \[
        \zeta = (15\ 12\ 11\ 10\ 6\ 7\ 8\ 4\ 3\ 2\ 1\ 5\ 9\ 13\
        14)
        \]
        is. Schrijf $x_0 = 15$, $x_1 = 12$, $x_2 = 11$, \dots,
        $x_{14} = 14$ voor haar cyclische volgorde, en ga na dat
        de omgekeerde elementaire rondgang uit vraag 10 precies
        $(x_0\ x_1\ x_2)$ is.
  \item Bewijs de conjugatieformule in elke $\mathfrak{S}_n$: voor
        een permutatie $g$ en een $3$-cykel geldt
        \[
        g\,(a\ b\ c)\,g^{-1} = \bigl(g(a)\ g(b)\ g(c)\bigr),
        \]
        en merk op dat $H$, als groep, gesloten is onder conjugatie
        door haar eigen elementen.
  \item Leid af dat $H$ alle vijftien \emph{opeenvolgende}
        $3$-cykels van de grote rondgang bevat:
        \[
        s_t = (x_t\ x_{t+1}\ x_{t+2}) \qquad (t \in \Z/15\Z,
        \text{ indices modulo } 15).
        \]
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De alternerende groep voortbrengen.}
\begin{enumerate}[resume]
  \item (Lemma A) Zijn $s$ en $t$ $3$-cykels waarvan de dragers
        precies twee punten gemeen hebben, zeg de dragers $\{a, b,
        c\}$ en $\{b, c, d\}$. Toon aan dat, na eventueel $s$ of
        $t$ door zijn inverse te vervangen (wat aan de
        \omterm{def:b2:structures:generated}{voortgebrachte} ondergroep niets verandert), het product
        $st$ een dubbele \omterm{def:b2:structures:sn}{transpositie} is; toon aan dat $A_4$ geen
        ondergroep van \omterm{def:b2:structures:generated}{orde} $6$ bevat \emph{(een ondergroep van
        index $2$ bevat elk kwadraat; tel de $3$-cykels onder de
        kwadraten)}; en besluit dat $\langle s, t\rangle$ de
        volledige alternerende groep op de vier letters $\{a, b,
        c, d\}$ is.
  \item (Lemma B) Zij $X$ een verzameling van $k \geq 4$ letters,
        $w \notin X$, en zij $G$ een ondergroep van een
        $\mathfrak{S}_n$ die elke even permutatie van $X$ bevat en
        bovendien één $3$-cykel $(u\ v\ w)$ met $u, v \in X$. Toon
        aan dat er voor alle verschillende $a, b \in X$ een
        \emph{even} permutatie $g$ van $X$ bestaat met $g(u) = a$
        en $g(v) = b$, en leid af dat $(a\ b\ w) \in G$.
  \item Leid af dat de groep $G$ uit Lemma B elke even permutatie
        van $X \cup \{w\}$ bevat \emph{(gebruik
        \cref{exo:b2:structures:6}: de $3$-cykels brengen voort)}.
        Bewijs vervolgens, door de Lemma's A en B aaneen te
        schakelen langs de opeenvolgende $3$-cykels $s_0, s_1,
        \dots, s_{12}$ uit vraag 13, dat $\langle s_0, \dots,
        s_{12}\rangle = A_{15}$.
  \item Besluit dat $H = A_{15}$: \emph{elke even herschikking van
        de vijftien plaatjes is met een programma te bereiken}, en
        $H$ telt $15!/2 = 653\,837\,184\,000$ elementen.
  \item (De stelling van Johnson en Story, 1879) Zet de vragen 6,
        7, 8 en 17 aaneen: de standen die vanuit de opgeloste
        stand bereikbaar zijn, zijn \emph{precies} de $16!/2 =
        10\,461\,394\,944\,000$ standen met $I = +1$; en de
        bereikbaarheid heeft precies \emph{twee} klassen, die van
        de opgeloste stand en die van Loyds $\sigma_L$. \emph{(Voor
        het tweede punt: hernoem de plaatjes $14$ en $15$ en toon
        aan dat $\sigma \mapsto (14\ 15) \circ \sigma$ reeksen
        zetten op reeksen zetten afbeeldt en $\{I = +1\}$ met $\{I
        = -1\}$ verwisselt.)}
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Criteria, varianten en het uitzicht
van bovenaf.}
\begin{enumerate}[resume]
  \item (Het praktische criterium) Lees de vijftien plaatjes in de
        leesvolgorde van hun vakjes, sla de lege plek over, en zij
        $N$ het aantal inversies van deze lijst; zij $r$ de rij
        van de lege plek, geteld vanaf \emph{onderen}. Toon aan
        dat $I(\sigma) = (-1)^{N + r + 1}$, zodat $\sigma$
        oplosbaar is dan en slechts dan als $N + r$ oneven is.
  \item (Groepsacties) Een \emph{actie} van een groep $G$ op een
        verzameling $X$ is een afbeelding $G \times X \to X$, $(g,
        x) \mapsto g \cdot x$, met $e \cdot x = x$ en $g \cdot (h
        \cdot x) = (gh) \cdot x$; de \emph{baan} van $x$ is $G
        \cdot x$, en de actie heet \emph{vrij} wanneer uit $g
        \cdot x = x$ volgt dat $g = e$. Toon aan dat $h \cdot
        \sigma = \sigma \circ h^{-1}$ een vrije actie van $H$ op
        de verzameling van de standen met de lege plek thuis
        definieert, dat haar banen precies de klassen van
        wederzijdse bereikbaarheid met programma's zijn, en haal
        uit het aantal banen terug dat deze standen in precies
        $15!\,/\,\abs H = 2$ klassen uiteenvallen.
  \item (De hindernis bij $3 \times 3$) Toon aan dat het bord van
        $3 \times 3$ \emph{geen} gesloten wandeling toelaat die
        elk vakje precies één keer bezoekt: de strategie met de
        grote rondgang uit Deel III faalt voor de achtpuzzel.
        \emph{(Kleur de negen vakjes als een schaakbord.)}
  \item (De reparatie) Op het bord van $3 \times 3$ met de vakjes
        $1$ tot $9$ in leesvolgorde en thuis $9$: bereken de
        effecten van de rondgang langs de rand $9 \to 8 \to 7 \to
        4 \to 1 \to 2 \to 3 \to 6 \to 9$ (een $7$-cykel $\zeta'$
        die het midden $5$ vasthoudt) en van de hoekrondgang $9
        \to 6 \to 5 \to 8 \to 9$ (een $3$-cykel door het midden).
        Bewijs, door de laatste met de machten van $\zeta'$ te
        conjugeren en de Lemma's A en B aaneen te schakelen, dat
        de programmagroep van de achtpuzzel heel $A_8$ is, en dus
        dat precies $9!/2 = 181\,440$ van de $9! = 362\,880$
        standen oplosbaar zijn.
  \item (Een arm bord) Zij het bord nu één enkele \omterm{def:b2:structures:sn}{cykel} van $n
        \geq 4$ vakjes met $n - 1$ plaatjes. Toon aan dat de
        cyclische volgorde van de plaatjes invariant is, dat elke
        bereikbaarheidsklasse precies $n(n - 1)$ standen telt
        \emph{(de klassen zijn de banen van een \omterm{def:b2:structures:generated}{cyclische groep}
        van \omterm{def:b2:structures:generated}{orde} $\operatorname{lcm}(n, n-1) = n(n-1)$)}, en dat
        er $(n - 2)!$ klassen zijn --- voor $n \geq 5$ veel meer
        dan $2$: op een smal bord vangt de pariteitsinvariant
        vrijwel niets, en heerst de meetkunde.
  \item Twee uitspraken volgens het criterium van vraag 19: het
        volledig omgekeerde bakje (plaatjes $15, 14, \dots, 1$ in
        de vakjes $1$ tot $15$, lege plek thuis) en het bakje met
        de lege plek in vakje $1$, gevolgd door de plaatjes $15,
        14, \dots, 1$ in de vakjes $2$ tot $16$. Welk van beide is
        oplosbaar?
  \item (Synthese) Het bewijs rust op twee onafhankelijke pijlers:
        een \emph{invariant} ($I$, gebouwd op het
        signatuurmorfisme) die aantoont dat hoogstens de helft van
        de standen bereikbaar is, en een stelling over
        \emph{expliciete voortbrenging} ($H = A_{15}$) die
        aantoont dat minstens de helft dat is. Zeg in telkens één
        zin waar de volgende zaken zijn binnengekomen: de
        morfisme-eigenschap van $\varepsilon$; de stelling van
        Lagrange; de voortbrenging van $A_n$ door $3$-cykels;
        conjugatie. Formuleer het metaprincipe in één regel.
\end{enumerate}
\end{problem}
