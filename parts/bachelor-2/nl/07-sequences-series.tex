\chapter{Rijen en reeksen}\label{ch:b2:series}

De theorie van de getallenreeksen (volume van bachelorjaar 1) rijpt
hier in drie richtingen: reeksen met waarden in \omterm{def:b2:nvs:banach}{banachruimten}, waar
de \omterm{def:b2:series:def}{absolute convergentie} het werk doet; de fijnere tests voor reële
reeksen (Abelsommatie); en \emph{\omterm{def:b2:series:summable}{sommeerbare families}} --- sommatie
losgemaakt van de volgorde van de termen --- met de stelling van
Fubini voor dubbelsommen en het \omterm{thm:b2:series:fubini}{Cauchy-product}. Deze gereedschappen
dragen alle hoofdstukken over functiereeksen die nog komen.

\section{Reeksen in genormeerde ruimten}

\begin{definition}\label{def:b2:series:def}
Voor een rij $(u_n)$ in een genormeerde ruimte $E$ convergeert de
reeks $\sum u_n$ wanneer haar partiële sommen convergeren; zij
convergeert \emph{absoluut}\index{absolute convergentie!in een
banachruimte} wanneer $\sum \norm{u_n} < \infty$. In een
\emph{\omterm{def:b2:nvs:banach}{banachruimte}} impliceert absolute convergentie de convergentie
(\cref{thm:b2:nvs:absoluteconvergence}); in een niet-volledige
ruimte kan dit falen (\cref{exo:b2:series:9}).
\end{definition}

\begin{example}\label{ex:b2:series:neumann}
In $\mathcal{M}_n(K)$ (of in $\mathcal{L}_c(E)$ met $E$ een
\omterm{def:b2:nvs:banach}{banachruimte}): voor $\vertiii A < 1$ convergeert de
\emph{neumannreeks} $\sum A^k$ \omterm{def:b2:series:def}{absoluut} naar $(I - A)^{-1}$
(bewezen in \cref{exo:b2:nvs:5}); en $\sum \frac{A^k}{k!}$
convergeert voor elke $A$ \omterm{def:b2:series:def}{absoluut} naar $\eu^A$
(\cref{ex:b2:nvs:matrixexp}). Meetkundige en exponentiële reeksen
met operatorwaarden gedragen zich als hun scalaire modellen --- en
dat is de hele bedoeling van het banachkader.
\end{example}

\section{Abelsommatie}

\begin{theorem}[Sommatie en test van Abel]\label{thm:b2:series:abel}
(Partiële sommatie) Voor scalairen $a_n$ en vectoren $b_n$, met
$B_n = \sum_{k=0}^{n} b_k$, geldt\index{Abelsommatie}
\[
\sum_{n=0}^{N} a_n b_n
= a_N B_N - \sum_{n=0}^{N-1} (a_{n+1} - a_n) B_n .
\]
(Test van Abel) Is $(a_n)$ een reële rij die dalend naar $0$ gaat en
zijn de partiële sommen $B_n$ \emph{begrensd} (in een
\omterm{def:b2:nvs:banach}{banachruimte}), dan convergeert $\sum a_n b_n$.
\end{theorem}

\begin{proof}
De identiteit, stap voor stap: schrijf met $B_{-1} = 0$ de term
$b_n = B_n - B_{n-1}$ en splits,
\[
\sum_{n=0}^{N} a_nb_n
= \sum_{n=0}^{N} a_nB_n - \sum_{n=0}^{N} a_nB_{n-1}
= \sum_{n=0}^{N} a_nB_n - \sum_{n=0}^{N-1} a_{n+1}B_{n} ,
\]
waarbij de tweede som is herindexeerd met $n \mapsto n + 1$ (de
term met $B_{-1}$ verdwijnt); het samennemen van het
gemeenschappelijke bereik $0 \leq n \leq N-1$ laat $a_NB_N$ over
plus $\sum_{n\leq N-1}(a_n - a_{n+1})B_n$: de gevraagde formule.
Het is de discrete partiële integratie, met $(B_n)$ als primitieve
van $(b_n)$ en het verschil $a_{n+1} - a_n$ als afgeleide van
$(a_n)$. Voor de test, met $\norm{B_n} \leq M$: de randterm $a_N
B_N \to 0$; en de reeks $\sum (a_n - a_{n+1})B_n$ convergeert
\omterm{def:b2:series:def}{absoluut}, want
\[
\sum_n \norm{(a_{n+1} - a_n)B_n} \leq M \sum_n (a_n - a_{n+1})
= M a_0 < \infty
\]
(telescoperen, met $a_n \downarrow 0$). Beide stukken van de
identiteit convergeren, en dus ook $\sum a_n b_n$.
\end{proof}

\begin{example}\label{ex:b2:series:sinn}
$\sum \frac{\sin n}{n}$ convergeert: $a_n = \frac1n \downarrow 0$
en $B_n = \sum_{k=1}^{n} \sin k$ is begrensd --- immers $B_n =
\Im\sum_{k \leq n} \eu^{\iu k} = \Im\,\frac{\eu^{\iu}(\eu^{\iu n} -
1)}{\eu^{\iu} - 1}$, van modulus $\leq \frac{2}{\abs{\eu^{\iu} -
1}}$. Zij convergeert \emph{niet} \omterm{def:b2:series:def}{absoluut} ($\abs{\sin n} \geq
\sin^2 n = \frac{1 - \cos 2n}{2}$, en $\sum \frac{1 - \cos 2n}{2n}$
divergeert omdat $\sum \frac{\cos 2n}{n}$ met dezelfde test van
Abel convergeert terwijl $\sum \frac{1}{2n}$ divergeert). De test
voor alternerende reeksen is het bijzondere geval $b_n = (-1)^n$.
\end{example}

\begin{example}[Abel op de convergentiecirkel]\label{ex:b2:series:abelboundary}
Voor welke complexe $z$ met $\abs z = 1$ convergeert $\sum_{n \geq
1} \frac{z^n}{n}$? In $z = 1$ is het de harmonische reeks: die
divergeert. Voor $z \neq 1$ op de cirkel is de test van Abel van
toepassing met $a_n = \frac1n \downarrow 0$ en $b_n = z^n$,
waarvan de partiële sommen onafhankelijk van $N$ begrensd zijn:
\[
\Bigl|\sum_{n=1}^{N} z^n\Bigr|
= \Bigl|\frac{z(z^N - 1)}{z - 1}\Bigr|
\leq \frac{2}{\abs{z - 1}} .
\]
Dus convergent --- al is het nooit \omterm{def:b2:series:def}{absoluut} ($\sum\frac1n$). Eén
reeks, een cirkel van gedragingen: divergentie in één enkel punt,
en overal elders halve convergentie. Dit is het standaardgedrag van
machtreeksen op de rand (\cref{ch:b2:powerseries}), hier met blote
handen aangetroffen; in $z = -1$ vindt men de alternerende
harmonische reeks terug, en in $z = \eu^{\iu\theta}$ zijn het
reële en het imaginaire deel de reeksen $\sum\frac{\cos
n\theta}{n}$ en $\sum\frac{\sin n\theta}{n}$ uit
\cref{exo:b2:series:4}.
\end{example}

\begin{example}[Een alternerende reeks met een valstrik]\label{ex:b2:series:trap}
Convergeert $\sum_{n\geq2} \dfrac{(-1)^n}{\sqrt n + (-1)^n}$? De
tekens wisselen en de termen gaan naar $0$ --- en toch is de test
voor alternerende reeksen \emph{niet} van toepassing: de moduli
$\frac{1}{\sqrt n + (-1)^n}$ dalen niet (zij springen bij elke
oneven $n$ omhoog). Ontwikkel in plaats daarvan:
\[
\frac{(-1)^n}{\sqrt n + (-1)^n}
= \frac{(-1)^n}{\sqrt n}\cdot
\frac{1}{1 + \frac{(-1)^n}{\sqrt n}}
= \frac{(-1)^n}{\sqrt n} - \frac{1}{n}
+ O\Bigl(\frac{1}{n^{3/2}}\Bigr).
\]
Het eerste stuk convergeert (de alternerende test, eerlijk
toegepast op $\frac1{\sqrt n}\downarrow0$), het derde convergeert
\omterm{def:b2:series:def}{absoluut} --- maar het middelste stuk is de divergente harmonische
reeks: de som \emph{divergeert} naar $-\infty$. Het inzicht om te
onthouden: faalt de monotonie, ontwikkel dan tot elk stuk ofwel
\omterm{def:b2:series:def}{absoluut} convergent is ofwel een schoon testgeval; de verborgen
$-\frac1n$ is voor tekentellen onzichtbaar.
\end{example}

\begin{remark}[Klassieke valkuilen]\label{rem:b2:series:pitfalls}
(i) ``De termen gaan naar $0$'' bewijst niets: de harmonische reeks
divergeert. (ii) De alternerende test vraagt \emph{dalende} moduli
--- \cref{ex:b2:series:trap} is het canonieke tegenvoorbeeld en de
derde reeks van \cref{exo:b2:series:1} de oefening. (iii)
Voorwaardelijk convergente reeksen mogen niet worden herschikt
(\cref{ex:b2:series:rearrange}), en hun \omterm{thm:b2:series:fubini}{Cauchy-producten} kunnen
divergeren: voor het kwadraat van $\sum\frac{(-1)^n}{\sqrt{n+1}}$
voldoen de diagonaaltermen aan
\[
\abs{c_k} = \sum_{m=0}^{k}
\frac{1}{\sqrt{(m+1)(k-m+1)}}
\geq (k+1)\cdot\frac{2}{k+2} \longrightarrow 2 \neq 0
\]
(elke factor is volgens de ongelijkheid tussen rekenkundig en
meetkundig gemiddelde hoogstens $\frac{k+2}2$), dus divergeert
$\sum c_k$ --- \omterm{def:b2:series:def}{absolute convergentie} van minstens één factor
(\cref{exo:b2:series:8}) is geen luxe. (iv) \omterm{def:b2:series:summable}{Sommeerbaarheid} gaat
per definitie over \emph{absolute} grenzen: een voorwaardelijk
\omterm{def:b2:series:summable}{sommeerbare familie} bestaat niet.
\end{remark}

\section{Sommeerbare families}

\begin{definition}\label{def:b2:series:summable}
Zij $I$ een \omterm{def:b2:structures:countable}{aftelbare} indexverzameling. Een familie $(u_i)_{i \in
I}$ van \emph{niet-negatieve reële getallen} heet
\emph{sommeerbaar}\index{sommeerbare familie} wanneer de eindige
partiële sommen begrensd zijn; haar som is
\[
\sum_{i \in I} u_i = \sup_{F \subseteq I \text{ eindig}} \sum_{i
\in F} u_i \in \intcc{0}{+\infty} .
\]
Een familie reële of complexe getallen (of vectoren uit een
\omterm{def:b2:nvs:banach}{banachruimte}) heet sommeerbaar wanneer $(\norm{u_i})$ dat is; haar
som wordt dan gedefinieerd door te splitsen in een positief en een
negatief (of een reëel en een imaginair) deel --- gelijkwaardig:
als de gemeenschappelijke waarde van $\sum_{n} u_{\sigma(n)}$ over
alle opsommingen $\sigma$ van $I$ (zie hieronder).
\end{definition}

\begin{method}[Een test kiezen]\label{met:b2:series:choosing}
Tegenover $\sum u_n$, in deze volgorde: (1) geldt $u_n \not\to 0$,
dan divergentie, klaar. (2) Hebben de termen een vast teken,
vergelijk dan: zoek een equivalent (\cref{ch:b2:comparison}) en
plaats het op de kaart van Riemann en Bertrand. (3) Wisselen de
tekens met \emph{dalende} moduli, gebruik dan de alternerende test;
zijn de moduli niet monotoon, ontwikkel de term dan tot elk stuk
\omterm{def:b2:series:def}{absoluut} convergent is of een schoon testgeval
(\cref{ex:b2:series:trap}). (4) Is het tekenpatroon oscillerend
maar gestructureerd ($\sin n\theta$, $\eu^{\iu n\theta}$,
matrixmachten), gebruik dan de test van Abel met begrensde partiële
sommen. (5) \omterm{def:b2:series:def}{Absolute convergentie} is altijd de moeite waard om als
eerste na te gaan: zij is sterker, ongevoelig voor de volgorde, en
zij opent \omterm{thm:b2:series:fubini}{Cauchy-producten} en Fubini.
\end{method}

\begin{example}[Sommeerbaarheid door diagonalen te tellen]\label{ex:b2:series:diagonalcount}
Voor welke $s > 0$ is de familie $\bigl((m + n)^{-s}\bigr)_{m, n
\geq 1}$ \omterm{def:b2:series:summable}{sommeerbaar}? Groepeer de eindige partiële sommen naar de
diagonalen $m + n = k$: de diagonaal $k$ draagt $k - 1$ paren, elk
met bijdrage $k^{-s}$, dus worden de eindige sommen precies
begrensd door (en uitgeput door)
\[
\sum_{k \geq 2} \frac{k - 1}{k^{s}} ,
\]
een reeks met positieve termen die equivalent is met $k^{1-s}$:
\omterm{def:b2:series:summable}{sommeerbaar} dan en slechts dan als $s - 1 > 1$, dat wil zeggen $s >
2$. De tweedimensionale index vreet een volle macht op: een vlak
van termen is ``één dimensie divergenter'' dan een rechte --- de
teltmeetkunde van de indexverzameling, niet de grootte van de
afzonderlijke termen, beslist over de \omterm{def:b2:series:summable}{sommeerbaarheid}. (Dezelfde
telling laat zien dat $\bigl((m^2 + n^2)^{-1}\bigr)$ niet
\omterm{def:b2:series:summable}{sommeerbaar} is: op de diagonaal $m + n = k$ is elke term minstens
$k^{-2}$, en $(k-1)\cdot k^{-2}$ sommeert als de harmonische
reeks.)
\end{example}

\begin{theorem}[Sommeerbaarheid en volgorde]\label{thm:b2:series:rearrangement}
\begin{enumerate}
  \item Voor niet-negatieve families is de som invariant onder elke
        opsomming: $\sum_{i} u_i = \sum_{n=0}^{\infty}
        u_{\sigma(n)}$ voor elke bijectie $\sigma \colon \N \to
        I$.
  \item Een reële of complexe reeks $\sum u_n$ is
        \emph{commutatief convergent} (elke herschikking
        convergeert, met dezelfde som) dan en slechts dan als zij
        \omterm{def:b2:series:def}{absoluut} convergeert.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Elke partiële som $\sum_{n \leq N} u_{\sigma(n)}$ is een
eindige partiële som van de familie (dus $\leq$ het supremum); en
elke eindige $F$ ligt in een zekere $\{\sigma(0), \dots,
\sigma(N)\}$ (dus is het supremum $\leq$ de limiet van de reeks).
De twee grenzen sluiten op elkaar aan.

(2) Geldt $\sum\abs{u_n} < \infty$: kies bij een herschikking
$\sigma$ en $\varepsilon > 0$ een $N$ met $\sum_{n >
N}\abs{u_n} \leq \varepsilon$; voorbij de rang waar $\sigma$
$\intint{0}{N}$ heeft uitgeput, verschillen de herschikte partiële
sommen hoogstens $\varepsilon$ van de oorspronkelijke limiet:
dezelfde som. Geldt $\sum \abs{u_n} = \infty$ terwijl $\sum u_n$
convergeert (reëel geval; het complexe volgt coördinaatsgewijs),
dan divergeren het positieve en het negatieve deel beide, en kan
men zo herschikken dat men elke voorgeschreven limiet bereikt ---
de stelling van Riemann, uitgevoerd in \cref{exo:b2:series:5} ---
zodat de commutatieve convergentie faalt.
\end{proof}

\begin{example}[Een herschikking op heterdaad betrapt]\label{ex:b2:series:rearrange}
De alternerende harmonische reeks heeft som
$\sum_{n\geq1}\frac{(-1)^{n-1}}{n} = \ln 2$ (volume van
bachelorjaar 1). Herschik haar als ``één positieve, twee
negatieve'':
\[
1 - \frac12 - \frac14 + \frac13 - \frac16 - \frac18 + \frac15 -
\cdots
\]
Groeperen we per blok van drie, dan is
\[
\frac{1}{2k-1} - \frac{1}{4k-2} - \frac1{4k}
= \frac{1}{4k-2} - \frac{1}{4k}
= \frac12\Bigl(\frac{1}{2k-1} - \frac1{2k}\Bigr),
\]
zodat de herschikte reeks naar $\frac12\ln 2$ convergeert --- de
helft van de oorspronkelijke som, met precies dezelfde termen.
\omterm{def:b2:series:def}{Niet-absoluut} convergente reeksen onthouden de volgorde van hun
termen; \omterm{def:b2:series:summable}{sommeerbare families} zijn juist die welke dat niet doen.
\end{example}

\begin{example}[Groeperen mag, ontgroeperen niet]\label{ex:b2:series:grouping}
Opeenvolgende termen van een \emph{convergente} reeks groeperen
verandert de som nooit: de gegroepeerde partiële sommen vormen een
deelrij van de oorspronkelijke. De omgekeerde bewerking is
verboden:
\[
(1 - 1) + (1 - 1) + (1 - 1) + \cdots = 0 + 0 + \cdots = 0,
\]
terwijl het ontgroepeerde $1 - 1 + 1 - 1 + \cdots$ divergeert (de
partiële sommen oscilleren tussen $1$ en $0$). Ontgroeperen is
alleen legitiem onder een compenserende hypothese --- bijvoorbeeld
dat de termen naar $0$ gaan bij begrensde bloklengten: dan drijven
de oorspronkelijke sommen tussen twee gegroepeerde partiële sommen
hoogstens een som van begrensd veel $o(1)$-termen weg, en draagt de
convergentie terug over. Dat is precies de voorwaarde waaronder de
blokberekening van \cref{ex:b2:series:rearrange} een bewijs is en
geen goocheltruc.
\end{example}

\begin{theorem}[Fubini voor families; Cauchy-producten]\label{thm:b2:series:fubini}
Zij $(u_{m,n})_{(m,n) \in \N^2}$ een \omterm{def:b2:series:summable}{sommeerbare} dubbele familie
(dat wil zeggen $\sup_F \sum_F \abs{u_{m,n}} < \infty$). Dan
geldt\index{stelling van Fubini voor reeksen}
\[
\sum_{(m,n)} u_{m,n}
= \sum_{m=0}^{\infty}\Bigl(\sum_{n=0}^{\infty} u_{m,n}\Bigr)
= \sum_{n=0}^{\infty}\Bigl(\sum_{m=0}^{\infty} u_{m,n}\Bigr)
= \sum_{k=0}^{\infty}\Bigl(\sum_{m+n=k} u_{m,n}\Bigr),
\]
waarbij alle binnenste reeksen (\omterm{def:b2:series:def}{absoluut}) convergeren. In het
bijzonder: convergeren $\sum a_m$ en $\sum b_n$ \omterm{def:b2:series:def}{absoluut}, dan
convergeert hun \emph{Cauchy-product}\index{Cauchy-product}
\omterm{def:b2:series:def}{absoluut}, met
\[
\Bigl(\sum_m a_m\Bigr)\Bigl(\sum_n b_n\Bigr)
= \sum_{k=0}^{\infty} c_k,
\qquad
c_k = \sum_{m=0}^{k} a_m b_{k-m} .
\]
\end{theorem}

\begin{proof}
\emph{Niet-negatief geval.} Elke groepering (naar rijen, kolommen
of diagonalen) berekent hetzelfde supremum: elke eindige
verzameling paren ligt in een eindig blok rijen (wat elke
gegroepeerde som van onderen door eindige partiële sommen en van
boven door het totaal begrenst), en de monotone convergentie van de
partiële sommen doet de rest --- concreet, voor rijen: uit
$\sum_{m \leq M}\sum_{n \leq N} u_{m,n} \leq S$ volgt, eerst met $N
\to \infty$ en dan met $M \to \infty$, dat $\sum_m \sum_n u_{m,n}
\leq S$; omgekeerd ligt elke eindige $F$ in zo'n rechthoek, dus $S
\leq \sum_m\sum_n u_{m,n}$. Voor diagonalen: dezelfde twee grenzen
met driehoeken in plaats van rechthoeken.

\emph{Algemeen geval.} Splits in een positief en een negatief (een
reëel en een imaginair) deel, elk een \omterm{def:b2:series:summable}{sommeerbare} niet-negatieve
familie; de vier groeperingen stemmen op elk deel overeen en dus
ook op het verschil; de \omterm{def:b2:series:def}{absolute convergentie} van de binnenste
reeksen komt uit het niet-negatieve geval toegepast op
$\abs{u_{m,n}}$.

\emph{\omterm{thm:b2:series:fubini}{Cauchy-product}.} De familie $u_{m,n} = a_m b_n$ is
\omterm{def:b2:series:summable}{sommeerbaar}: de eindige partiële sommen van $\abs{a_mb_n}$ worden
begrensd door $\bigl(\sum\abs{a_m}\bigr)\bigl(\sum\abs{b_n}\bigr)$.
Rijen geven $\bigl(\sum a_m\bigr)\bigl(\sum b_n\bigr)$; diagonalen
geven $\sum_k c_k$.
\end{proof}

\begin{example}[De exponentiële identiteit, eerlijk]\label{ex:b2:series:exp}
Voor $a, b \in \C$ (of commuterende matrices):
\[
\Bigl(\sum_m \frac{a^m}{m!}\Bigr)\Bigl(\sum_n
\frac{b^n}{n!}\Bigr)
= \sum_k \sum_{m+n=k} \frac{a^m b^n}{m!\,n!}
= \sum_k \frac{(a + b)^k}{k!},
\]
volgens het binomium op elke diagonaal: dus $\eu^a \eu^b =
\eu^{a+b}$ --- de functionaalvergelijking van $\exp$, uitsluitend
uit de reeks afgeleid. (De commutatie wordt in de binomiale stap
gebruikt; voor niet-commuterende matrices faalt de identiteit
werkelijk, \cref{ch:b2:diffeq}.)
\end{example}

\begin{example}[Cauchy-producten als rekenmiddel]\label{ex:b2:series:squaresum}
Uit de meetkundige reeks en de formule $\sum_{n \geq 1} nz^n =
\frac{z}{(1-z)^2}$ ($\abs z < 1$) van \cref{exo:b2:series:2} maakt
nog één \omterm{thm:b2:series:fubini}{Cauchy-product} het tweede moment af. Vermenigvuldig
$\sum_m mz^m$ met $\sum_n z^n$: de diagonaalcoëfficiënt is
$\sum_{m=0}^k m = \frac{k(k+1)}2$, dus
\[
\frac{z}{(1-z)^3} = \sum_{k\geq0}\frac{k(k+1)}{2}\,z^k ,
\]
en de identiteit $n^2 = 2\cdot\frac{n(n+1)}2 - n$ zet in elkaar
\[
\sum_{n\geq1} n^2z^n = \frac{2z}{(1-z)^3} - \frac{z}{(1-z)^2}
= \frac{z(1+z)}{(1-z)^3} .
\]
Bij $z = \frac12$: $\sum_{n\geq1}\frac{n^2}{2^n} =
\frac{\frac12\cdot\frac32}{\frac18} = 6$ --- een gesloten waarde
zonder ook maar ergens te differentiëren, alleen \omterm{def:b2:series:def}{absoluut}
convergente reeksen die als veeltermen worden vermenigvuldigd.
Dezelfde ketting van identiteiten berekent elke $\sum n^dz^n$, en
wie kansrekening kent, herkent het tweede factoriële moment van de
meetkundige verdeling (\cref{ch:b2:genfun}).
\end{example}

\begin{example}[Een dubbelsom uitgerekend]\label{ex:b2:series:doublesum}
Zij voor reële $s > 1$ de waarde $\zeta(s) = \sum_{n\geq1}
n^{-s}$. Delers tellen met een dubbele sommatie --- de familie
$(m^{-s}n^{-s})$ over $(m,n) \in (\N^*)^2$ is \omterm{def:b2:series:summable}{sommeerbaar} (een
product van convergente positieve reeksen) --- en groeperen naar
het product $q = mn$ geeft
\[
\zeta(s)^2 = \sum_{m,n} \frac{1}{(mn)^s}
= \sum_{q=1}^{\infty} \frac{d(q)}{q^s},
\]
waarbij $d(q)$ het aantal delers van $q$ is. \omterm{def:b2:series:summable}{Sommeerbare families}
maken van combinatoriek analyse.
\end{example}

\begin{example}[Een berekening met Fubini: $\sum_n (\zeta(n) - 1) = 1$]\label{ex:b2:series:zetasum}
Voor gehele $n \geq 2$ is $\zeta(n) - 1 = \sum_{k \geq 2} k^{-n}$.
De dubbele familie $(k^{-n})_{k, n \geq 2}$ is \omterm{def:b2:series:summable}{sommeerbaar}: sommeer
eerst de meetkundige kolommen,
\[
\sum_{k\geq2}\sum_{n\geq2} \frac{1}{k^n}
= \sum_{k\geq2} \frac{1/k^2}{1 - 1/k}
= \sum_{k\geq2} \frac{1}{k(k-1)} = 1
\]
(telescoperen), en alle termen zijn positief, dus staat
\cref{thm:b2:series:fubini} toe in plaats daarvan naar rijen te
sommeren:
\[
\sum_{n\geq2}\bigl(\zeta(n) - 1\bigr) = 1 .
\]
De oneindig vele $\zeta$-waarden, elk transcendent ogend, hebben
staarten die precies tot $1$ optellen. Het inzicht om te onthouden:
heeft een dubbelsom positieve termen, bereken haar dan in de
volgorde die in elkaar klapt --- hier zijn de kolommen meetkundig,
de rijen raadselachtig, en Fubini brengt de klap over.
\end{example}

\begin{example}[De meetkundige reeks lost een vergelijking op]\label{ex:b2:series:neumannsolve}
Los in de \omterm{def:b2:nvs:banach}{banachruimte} $\bigl(C(\intcc01),
\norm\cdot_\infty\bigr)$ de vergelijking $x - K(x) = y$ op, waarbij
$K(f)$ de constante functie $\frac12\int_0^1 f$ is. De \omterm{thm:b2:nvs:continuouslinear}{operatornorm}
voldoet aan $\vertiii K \leq \frac12 < 1$, dus is de neumannreeks
van toepassing (\cref{ex:b2:series:neumann}): $x = \sum_{n\geq0}
K^n(y)$. Bereken de iteraties: $K(y) = \frac12\int_0^1 y$ (een
constante), en $K$ toepassen op een constante $c$ geeft $\frac
c2$, dus $K^n(y) = \frac{1}{2^{n-1}}\cdot\frac12\int_0^1 y$ voor
$n \geq 1$. Het sommeren van de meetkundige constanten geeft
\[
x = y + \Bigl(\int_0^1 y\Bigr)
\sum_{n\geq1}\frac{1}{2^n} = y + \int_0^1 y .
\]
Controle: $x - K(x) = y + \int y - \frac12\bigl(\int y + \int
y\bigr) = y$. Een oneindige reeks, een eindig antwoord en een
controle van één regel --- de meetkundige reeks is een
inversiealgoritme, niet louter een uitspraak over convergentie.
\end{example}

\begin{example}[Telescoperen met breuksplitsing]\label{ex:b2:series:telescope}
Exact sommeren is zeldzaam; telescoperen is de voornaamste
leverancier ervan. Splits
\[
\frac{1}{n(n+1)(n+2)}
= \frac{1}{2}\Bigl(\frac{1}{n(n+1)} -
\frac{1}{(n+1)(n+2)}\Bigr),
\]
(na te gaan door op één noemer te brengen), zodat de partiële
sommen in elkaar klappen:
\[
\sum_{n=1}^{N}\frac{1}{n(n+1)(n+2)}
= \frac12\Bigl(\frac{1}{1\cdot2} -
\frac{1}{(N+1)(N+2)}\Bigr)
\longrightarrow \frac14 .
\]
Hetzelfde patroon --- schrijf de term als $c(u_n - u_{n+1})$ voor
een expliciete $(u_n)$ --- loste \cref{exo:b2:series:10}
(arctangenten) op en berekent elke $\sum\frac{1}{n(n+1)\cdots(n +
k)} = \frac{1}{k\cdot k!}$. Bestaat er op dit niveau een exacte
som, dan zit er meestal een telescoop in de term verborgen.
\end{example}

\begin{remark}[Vooruitblik binnen dit volume]\label{rem:b2:series:perspectives}
Drie hoofdstukken hierna zijn rechtstreekse klanten. Voor
\cref{ch:b2:funcseq}: normale convergentie van $\sum f_n$ is
\omterm{def:b2:series:def}{absolute convergentie} van $\sum\norm{f_n}_\infty$ in de
\omterm{def:b2:nvs:banach}{banachruimte} $\bigl(C, \norm\cdot_\infty\bigr)$ ---
\cref{thm:b2:nvs:absoluteconvergence} in een ander kostuum. Voor
\cref{ch:b2:powerseries}: binnen de convergentieschijf is alles
\omterm{def:b2:series:def}{absoluut} en \omterm{def:b2:series:summable}{sommeerbaar}, zodat \omterm{thm:b2:series:fubini}{Cauchy-producten} en herschikkingen
vrij spel hebben (en dat is waarom machtreeksen zich als
veeltermen vermenigvuldigen); op de rand neemt de test van Abel het
over (\cref{ex:b2:series:abelboundary}). Voor
\cref{ch:b2:genfun}: kansgenererende functies zijn machtreeksen
waarvan alle bewerkingen --- producten voor sommen van
onafhankelijke variabelen, dubbelsommen voor samengestelde
verdelingen --- door \cref{thm:b2:series:fubini} worden
gelegitimeerd. \omterm{def:b2:series:summable}{Sommeerbare families} zijn de juridische afdeling van
de analyse die komt.
\end{remark}

\begin{remark}[Waar dit hoofdstuk wordt gebruikt]
Alles met een oneindige som gaat hier langs: machtreeksen
(\cref{ch:b2:powerseries}) zijn vermomde \omterm{def:b2:series:summable}{sommeerbare families},
fouriercoëfficiënten worden met \omterm{thm:b2:series:fubini}{Cauchy-producten} vermenigvuldigd en
door Parseval herschikt (\cref{ch:b2:fourier}), en
kansgenererende functies (\cref{ch:b2:genfun}) zijn de stelling
van Fubini toegepast op verwachtingswaarden. Het volume van
bachelorjaar 3 absorbeert de \omterm{def:b2:series:summable}{sommeerbare families} in de
lebesgue-integratie over de telmaat --- waar
\cref{thm:b2:series:fubini} een bijzonder geval van de stelling van
Fubini en Tonelli wordt.
\end{remark}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:b2:series:1}
Bepaal de aard van $\sum \dfrac{\cos n}{n}$, $\;\sum
\dfrac{(-1)^n}{\ln n}$ en $\;\sum \dfrac{(-1)^n}{n^{3/4} + \cos n}$
\emph{(ontwikkel zoals bij de valstrik uit bachelorjaar 1: de
alternerende test heeft monotonie nodig)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:series:2}
Bewijs dat voor $\abs z < 1$ geldt $\sum_{n\geq1} n z^{n} =
\dfrac{z}{(1-z)^2}$, via het \omterm{thm:b2:series:fubini}{Cauchy-product} van $\sum z^n$ met
zichzelf.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:3}
(Lemma van het type Kronecker) Zij $\sum b_n$ een convergente reële
reeks. Bewijs met Abelsommatie dat $\dfrac{1}{n}\sum_{k=1}^{n}
k\,b_k \to 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:4}
Onderzoek de convergentie van $\sum \dfrac{\sin(n\theta)}{n^\alpha}$
($\theta \in \R$, $\alpha > 0$): voor welke $(\theta, \alpha)$ is
zij \omterm{def:b2:series:def}{absoluut} convergent, halfconvergent of divergent?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:5}
(Herschikking van Riemann) Zij $\sum u_n$ een convergente maar niet
\omterm{def:b2:series:def}{absoluut} convergente reële reeks, en $\ell \in \R$. Bewijs dat een
zekere herschikking van $\sum u_n$ naar $\ell$ convergeert.
\emph{(Toon aan dat beide deelreeksen van positieve en van
negatieve termen divergeren; wissel dan gulzig af: neem positieve
termen tot je $\ell$ overschrijdt, dan negatieve tot je eronder
duikt, enzovoort; de termen gaan naar $0$, wat convergentie naar
$\ell$ afdwingt.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:6}
Bewijs dat de familie
$\Bigl(\dfrac{x^{m+n}}{m!\,n!}\Bigr)_{(m,n)\in\N^2}$ voor elke $x
\in \R$ \omterm{def:b2:series:summable}{sommeerbaar} is, en leid de identiteit $(\eu^x)^2 =
\eu^{2x}$ opnieuw af door de dubbelsom langs de diagonalen $m + n =
k$ te groeperen.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:7}
Bewijs dat de familie $\bigl(\frac{1}{m^2 n^2}\bigr)_{m,n \geq 1}$
\omterm{def:b2:series:summable}{sommeerbaar} is, en dat het groeperen naar de grootste gemene deler
geeft: met $q = \gcd(m,n)$,
\[
\zeta(2)^2 = \sum_{q\geq1} \frac{1}{q^4}
\sum_{\substack{a,b \geq 1\\ \gcd(a,b)=1}} \frac{1}{a^2b^2}
= \zeta(4) \cdot S,
\]
waarbij $S = \sum_{\gcd(a,b)=1} \frac{1}{a^2b^2}$: leid af dat $S =
\zeta(2)^2/\zeta(4)$. \emph{(Elk paar $(m,n)$ schrijft zich
eenduidig als $(qa, qb)$ met $\gcd(a,b) = 1$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:8}
(Stelling van Abel over producten, lichte versie) Neem aan dat
$\sum a_n$ \omterm{def:b2:series:def}{absoluut} convergeert en $\sum b_n$ convergeert. Bewijs
dat hun \omterm{thm:b2:series:fubini}{Cauchy-product} $\sum c_n$ convergeert, met $\sum c_n =
(\sum a_n)(\sum b_n)$. \emph{(Schrijf $C_N = \sum_{k\leq N} c_k =
\sum_n a_n B_{N-n}$ met $B$ de partiële sommen van $b$; splits
naargelang $n \leq N/2$ of niet, met de begrensdheid van $(B_m)$ en
de absolute staart van $(a_n)$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:9}
Geef in de (niet-volledige) ruimte $E$ van de reële rijen die vanaf
zeker moment nul zijn, met de supnorm, een \omterm{def:b2:series:def}{absoluut} convergente
reeks die in $E$ niet convergeert. \emph{(Probeer $u_n = 2^{-n}
e_n$ met $(e_n)$ de canonieke rijen.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:10}
Ga de identiteit $\arctan(n+1) - \arctan(n) =
\arctan\dfrac{1}{n^2 + n + 1}$ na, en leid daaruit de exacte waarde
van
\[
\sum_{n=1}^{\infty} \arctan\frac{1}{n^2 + n + 1}
\]
af.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:11}
Bepaal de aard (met equivalenten) van
\[
\sum_n \bigl(\sqrt{n+1} - \sqrt n\bigr)^{\alpha}
\ (\alpha > 0), \qquad
\sum_n \Bigl(1 - \cos\frac1n\Bigr), \qquad
\sum_n \Bigl(\eu - \Bigl(1 + \frac1n\Bigr)^{\!n}\Bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:12}
Zij $(a_n)$ positief en \emph{dalend} met $\sum a_n$ convergent.
Bewijs dat $n\,a_n \to 0$ \emph{(begrens $n a_{2n}$ door een
staart)}. Toon aan dat het omgekeerde faalt en dat de monotonie
onmisbaar is, met expliciete tegenvoorbeelden.
\end{exercise}

\section{Probleem: Eulers $\zeta(2) = \pi^2/6$, via Cauchys
cotangenssom}

Eulers beroemdste identiteit, $1 + \frac14 + \frac19 + \cdots =
\frac{\pi^2}6$, laat een \omterm{def:b2:metric:complete}{volledig} elementair bewijs toe, van
Cauchy: de formule van de Moivre levert een veelterm waarvan de
nulpunten de getallen $\cot^2\frac{k\pi}{2n+1}$ zijn, Vieta somt
die nulpunten exact op, en de insluiting $\cot^2\theta <
\frac{1}{\theta^2} < 1 + \cot^2\theta$ perst de partiële sommen van
$\sum\frac1{k^2}$ tussen twee expliciete rationale grenzen. We
voeren het bewijs \omterm{def:b2:metric:complete}{volledig} uit, halen met dezelfde methode $\zeta(4)
= \frac{\pi^4}{90}$ binnen, en brengen daarna de hele grens tussen
convergentie en divergentie in kaart met de reeksen van Bertrand
--- en bewijzen dat er op die grens helemaal geen traagste
convergente reeks ligt.

\begin{problem}[{Weekendopgave --- $\zeta(2) = \pi^2/6$ en het
panorama van Bertrand}]
\label{pb:b2:series:1}
Overal is $n \geq 1$ en $\theta_k = \dfrac{k\pi}{2n+1}$ voor $k =
1, \dots, n$; merk op dat $0 < \theta_k < \frac\pi2$.

\medskip
\noindent\textbf{Deel I --- De cotangensidentiteit.}
\begin{enumerate}
  \item Bewijs de formule van de Moivre $(\cos\theta +
        \iu\sin\theta)^m = \cos m\theta + \iu\sin m\theta$ ($m \in
        \N$), en leid daaruit voor $m = 2n + 1$ af dat
        \[
        \sin\bigl((2n{+}1)\theta\bigr) = \sum_{j=0}^{n}
        (-1)^j\binom{2n+1}{2j+1}\cos^{2(n-j)}\theta\,
        \sin^{2j+1}\theta .
        \]
  \item Leid af dat voor $\theta \in \intoo{0}{\frac\pi2}$
        \[
        \sin\bigl((2n{+}1)\theta\bigr) =
        \sin^{2n+1}\theta\; P_n(\cot^2\theta),
        \qquad
        P_n(x) = \sum_{j=0}^{n}
        (-1)^j\binom{2n+1}{2j+1}x^{\,n-j},
        \]
        een veelterm van graad $n$ met kopcoëfficiënt $2n + 1$.
  \item Toon aan dat $x_k = \cot^2\theta_k$, $k = 1, \dots, n$,
        $n$ \emph{verschillende} nulpunten van $P_n$ zijn --- en
        dus alle nulpunten.
  \item Bewijs met Vieta de exacte identiteit
        \[
        \sum_{k=1}^{n} \cot^2\frac{k\pi}{2n+1}
        = \frac{n(2n-1)}{3}.
        \]
  \item Leid ook $\displaystyle\sum_{k=1}^{n}
        \frac{1}{\sin^2\theta_k} = \frac{2n(n+1)}{3}$ af.
  \item Bewijs de insluiting $\cot^2\theta < \dfrac1{\theta^2} <
        \dfrac{1}{\sin^2\theta}$ voor $\theta \in
        \intoo{0}{\frac\pi2}$ \emph{(uit $\sin\theta < \theta <
        \tan\theta$)}.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- De insluiting sluit: de stelling van
Euler.}
\begin{enumerate}[resume]
  \item Stel vast, door vraag 6 over $k = 1, \dots, n$ met $\theta
        = \theta_k$ te sommeren, dat
        \[
        \frac{n(2n-1)}{3} \;<\;
        \frac{(2n+1)^2}{\pi^2}\sum_{k=1}^{n}\frac1{k^2}
        \;<\; \frac{2n(n+1)}{3}.
        \]
  \item Besluit (\emph{de stelling van Euler, met het bewijs van
        Cauchy}):
        \[
        \zeta(2) = \sum_{k=1}^{\infty}\frac{1}{k^2} =
        \frac{\pi^2}{6}.
        \]
  \item Haal een snelheid uit de insluiting: toon aan dat
        \[
        \Bigl|\sum_{k=1}^{n}\frac1{k^2} - \frac{\pi^2}6\Bigr|
        = O\Bigl(\frac1n\Bigr),
        \]
        in overeenstemming met de exacte staart $\sum_{k>n}k^{-2}
        = \frac1n - \frac{1}{2n^2} + O(n^{-3})$ uit
        \cref{exo:b2:comparison:11}.
  \item Laat de machine één verdieping hoger draaien: toon met de
        tweede symmetrische functie van Vieta van $P_n$ aan dat
        \[
        \sum_{k=1}^{n}\cot^4\theta_k =
        \Bigl(\frac{n(2n-1)}3\Bigr)^{\!2} -
        \frac{2n(2n-1)(2n-2)(2n-3)}{60}
        \;\sim\; \frac{8n^4}{45},
        \]
        en sluit in met $\cot^4 < \theta^{-4} < (1 + \cot^2)^2$
        om $\zeta(4) = \dfrac{\pi^4}{90}$ te verkrijgen.
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Opbrengsten.}
\begin{enumerate}[resume]
  \item Leid uit $\zeta(2) = \frac{\pi^2}6$ af dat
        \[
        \sum_{k\geq0}\frac{1}{(2k+1)^2} = \frac{\pi^2}{8},
        \qquad
        \sum_{k\geq1}\frac{(-1)^{k-1}}{k^2} = \frac{\pi^2}{12}.
        \]
  \item Combineer met \cref{exo:b2:series:7}: bereken $S =
        \sum_{\gcd(a,b)=1}\frac{1}{a^2b^2} =
        \frac{\zeta(2)^2}{\zeta(4)} = \frac52$, en interpreteer
        $\frac{1}{\zeta(2)} = \frac{6}{\pi^2} \approx 0.608$ als
        de dichtheid van de onderling ondeelbare paren (formuleer
        de heuristiek eerlijk: de rigoureuze telling is een zaak
        voor het volume van bachelorjaar 3).
  \item (Gecertificeerde versnelling) De staartformule van vraag 9
        geeft $\sum_{k\leq n}k^{-2} + \frac1n - \frac1{2n^2} =
        \frac{\pi^2}6 + O(n^{-3})$. Vergelijk het werk dat zes
        cijfers van $\zeta(2)$ kosten: rechtstreeks sommeren
        tegenover de gecorrigeerde som bij $n = 100$ (waar de fout
        $1.7\cdot10^{-7}$ is).
  \item Ga vraag 4 met de hand na bij $n = 1$ en $n = 2$ (de
        waarden $\cot^2\frac\pi3 = \frac13$ en $\cot^2\frac\pi5 +
        \cot^2\frac{2\pi}5 = 2$), met $\cos\frac\pi5 =
        \frac{1+\sqrt5}4$ of met een numerieke evaluatie.
  \item Bewijs de bijbehorende identiteit
        \[
        \sum_{k=1}^{n}\tan^2\frac{k\pi}{2n+1} = n(2n+1)
        \]
        \emph{(de getallen $\tan^2\theta_k$ zijn de nulpunten van
        de omgekeerde veelterm $x^nP_n(1/x)$)}, en ga haar na bij
        $n = 1$.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- Het panorama van Bertrand.} Beschouw
voor $\alpha, \beta \in \R$ de \emph{reeks van Bertrand}
\[
\sum_{n \geq 3} \frac{1}{n^{\alpha}(\ln n)^{\beta}} .
\]
\begin{enumerate}[resume]
  \item Toon aan dat de reeks voor $\alpha > 1$ convergeert, wat
        $\beta$ ook is \emph{(vergelijk met
        $n^{-(1+\alpha)/2}$)}.
  \item Toon aan dat zij voor $\alpha < 1$ divergeert, wat $\beta$
        ook is.
  \item Voor $\alpha = 1$: bewijs met de vergelijking van reeks en
        integraal (\cref{thm:b2:comparison:seriesintegral}) voor
        $f(t) = \frac{1}{t(\ln t)^\beta}$ dat de reeks
        convergeert dan en slechts dan als $\beta > 1$.
  \item Herhaal de grensbepaling: toon aan dat
        $\sum\frac{1}{n\ln n\,\ln\ln n}$ divergeert terwijl
        $\sum\frac{1}{n\ln n\,(\ln\ln n)^2}$ convergeert.
  \item Twee valstrikken: bepaal de aard van
        \[
        \sum_n \frac{1}{n^{1 + 1/\ln n}}
        \qquad\text{en}\qquad
        \sum_n \frac{1}{n^{1 + 1/\ln\ln n}}
        \]
        \emph{(bereken $n^{1/\ln n}$ exact; vergelijk
        $n^{1/\ln\ln n}$ met elke macht van $\ln n$)}.
  \item (Er is geen traagste convergente reeks) Zij $\sum a_n$ een
        willekeurige convergente reeks met $a_n > 0$, en $R_n =
        \sum_{k \geq n}a_k$ haar staarten. Bewijs dat $\sum
        \frac{a_n}{\sqrt{R_n}}$ nog steeds \emph{convergeert}
        \emph{(vergelijk met de telescopische
        $2(\sqrt{R_n} - \sqrt{R_{n+1}})$)}, hoewel
        $\frac{a_n/\sqrt{R_n}}{a_n} \to \infty$: elke convergente
        reeks wordt strikt gedomineerd door een andere convergente
        reeks. De grens van de convergentie is geen kromme maar
        een mistbank.
\end{enumerate}

\medskip
\noindent\textbf{Deel V --- Kruiscontroles en synthese.}
\begin{enumerate}[resume]
  \item (Condensatie van Cauchy) Bewijs: voor positieve dalende
        $(a_n)$ convergeert $\sum a_n$ dan en slechts dan als
        $\sum 2^k a_{2^k}$ convergeert. Leid er de grens van vraag
        18 opnieuw uit af.
  \item (De prijs van traagheid) Begrens voor $\sum\frac1{n(\ln
        n)^2}$ de staart door een integraal en toon aan dat
        sommeren tot $N = 10^6$ nog altijd een fout groter dan
        $0.07$ overlaat: convergentie die de theorie certificeert,
        kan numeriek onbruikbaar zijn --- vergelijk dit met vraag
        13.
  \item Klasseer (met verantwoording van één regel):
        $\sum\frac1{n\ln n}$, $\sum\frac1{n^{1.01}}$,
        $\sum\frac{(\ln n)^{100}}{n^{1.001}}$ en
        $\sum\frac1{n(\ln n)(\ln\ln n)^{3}}$.
  \item (Synthese) In telkens één zin: hoe de Moivre een
        goniometrische identiteit in een veelterm met berekenbare
        nulpuntsommen veranderde; waar de insluiting \emph{exacte}
        identiteiten in de eindpunten nodig had in plaats van
        equivalenten; welk gereedschap uit \cref{ch:b2:comparison}
        Deel IV aandreef; en wat vraag 21 zegt over de droom van
        een ``universele vergelijkingstest''. Noem de twee toppen:
        Eulers $\zeta(2) = \frac{\pi^2}6$ (en de verdieping
        erboven, $\zeta(4) = \frac{\pi^4}{90}$), en de
        classificatie van Bertrand. Vermeld waar $\zeta(2)$
        opnieuw zal worden bewezen: met Parseval in
        \cref{ch:b2:fourier} --- één stelling, twee
        beschavingen.
\end{enumerate}
\end{problem}
