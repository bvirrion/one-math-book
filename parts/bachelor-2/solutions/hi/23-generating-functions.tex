\section*{अध्याय \ref{ch:b2:genfun} --- प्रायिकता जनक फलन}

\begin{solution}{exo:b2:genfun:1}
निष्पक्ष पासा: $G(t) = \frac16(t + t^2 + \dots + t^6) = \frac t6(1 + t
+ \dots + t^5)$। यदि दो निष्पक्ष पासों का योग $\{2, \dots, 12\}$ पर एकसमान
होता, तो
\[
G(t)^2 = \frac{t^2}{36}\,h(t)^2
= \frac{t^2}{11}\sum_{k=0}^{10}t^k ,
\qquad h(t) = 1 + t + \dots + t^5 .
\]
अब $h$ विषम घात $5$ का वास्तविक बहुपद है, अतः उसका कोई वास्तविक मूल
है (मध्यमान प्रमेय; मूर्त रूप में $h(-1) = 0$), इसलिए $h^2$ का कोई वास्तविक
मूल है। पर $\sum_{k=0}^{10}t^k$ का कोई नहीं है: वह $t \geq 0$ के लिए धनात्मक है, और $t < 0$ के
लिए वह $\frac{t^{11} - 1}{t - 1}$ के बराबर है, जो दो ऋणात्मक संख्याओं का विभाग है।
विरोधाभास --- अर्थात् दो निष्पक्ष पासों का योग कभी एकसमान नहीं होता
(जैसा पासा-योगों का जाना-पहचाना त्रिभुजीय \omterm{def:b2:randomvar:law}{बंटन} पुष्ट करता है)।
\end{solution}

\begin{solution}{exo:b2:genfun:2}
\emph{द्विपद:} $G(t) = (1 - p + pt)^n$, $G'(t) = np(1 - p +
pt)^{n-1}$, $G''(t) = n(n-1)p^2(1 - p + pt)^{n-2}$, अतः
\[
\E(X) = G'(1) = np,
\qquad
V(X) = G''(1) + G'(1) - G'(1)^2
= n(n-1)p^2 + np - n^2p^2 = np(1-p).
\]
\emph{ज्यामितीय} ($q = 1 - p$): $G(t) = \frac{pt}{1 - qt}$, अतः $G'(t) = \frac{p}{(1 - qt)^2}$ तथा $G''(t) = \frac{2pq}{(1 -
qt)^3}$; $t = 1$ पर ($1 - q = p$ बरतकर):
\[
\E(X) = \frac{p}{p^2} = \frac1p,
\qquad
V(X) = \frac{2q}{p^2} + \frac1p - \frac{1}{p^2}
= \frac{2q + p - 1}{p^2}
= \frac{q}{p^2} ,
\]
जो कम मेहनत के साथ \cref{exo:b2:randomvar:1} से मेल खाता है।
\end{solution}

\begin{solution}{exo:b2:genfun:3}
नहीं, भिन्न भारण के साथ भी नहीं। मान लीजिए $X, Y$ $\{1, \dots, 6\}$ पर ऐसे \omterm{def:b2:randomvar:law}{नियम} हैं
जिनका योग एकसमान है। तब $G_X(t) = t\,a(t)$ तथा $G_Y(t) = t\,b(t)$, जहाँ $a, b$ \emph{अधिक से अधिक}
घात $5$ के वास्तविक बहुपद हैं --- और उनकी घातें जुड़कर $10$ बननी ही
चाहिए (योग $12$ तक धनात्मक प्रायिकता के साथ पहुँचता है), अतः $\deg a = \deg b = 5$, और
दोनों विषम। \cref{exo:b2:genfun:1} की तरह,
\[
a(t)\,b(t) = \frac{1}{11}\sum_{k=0}^{10}t^k
\]
बाएँ पक्ष पर कोई वास्तविक मूल बाध्य कर देता (हर विषम-घात वास्तविक
बहुपद के पास एक होता है) और दाएँ पक्ष पर कोई नहीं। अतः दो \omterm{def:b2:proba:independence}{स्वतंत्र}
पासों का कोई भी भारण --- बराबर हो या न हो --- एकसमान योग उत्पन्न
नहीं करता।
\end{solution}

\begin{solution}{exo:b2:genfun:4}
$G_N(s) = e^{\lambda(s-1)}$ तथा $G_X(t) = 1 - p + pt$ के साथ \cref{thm:b2:genfun:compound} से:
\[
G_S(t) = e^{\lambda(1 - p + pt - 1)} = e^{\lambda p(t - 1)} :
\]
$S \sim \mathcal{P}(\lambda p)$। छोड़ी गई गिनती $D = N - S$ उन्हीं वस्तुओं को गिनती है जो प्रायिकता $1 - p$
के साथ रखी गईं, अतः उसी परिकलन से $D \sim \mathcal{P}(\lambda(1 - p))$। स्वतंत्रता, सीधे: $j, k \in \N$ के
लिए
\begin{align*}
\P(S = j,\ D = k)
&= \P(N = j + k)\,\binom{j+k}{j}p^jq^k
= e^{-\lambda}\frac{\lambda^{j+k}}{(j+k)!}\,
\frac{(j+k)!}{j!\,k!}\,p^jq^k\\
&= \Bigl(e^{-\lambda p}\frac{(\lambda p)^j}{j!}\Bigr)
\Bigl(e^{-\lambda q}\frac{(\lambda q)^k}{k!}\Bigr)
\end{align*}
जहाँ $q = 1 - p$: अर्थात् संयुक्त \omterm{def:b2:randomvar:law}{नियम} $\mathcal{P}(\lambda p) \otimes \mathcal{P}(\lambda q)$ के रूप में गुणनखंडित हो जाता
है। यादृच्छिक रूप से बाँटा गया कोई प्वासों प्रवाह \emph{\omterm{def:b2:proba:independence}{स्वतंत्र}}
प्वासों प्रवाह देता है --- एक छोटा चमत्कार, जिसका क़तार-सिद्धांत में
लगातार उपयोग होता है।
\end{solution}

\begin{solution}{exo:b2:genfun:5}
क्रमागत चित्तों के बीच प्रतीक्षा-समय \omterm{def:b2:proba:independence}{स्वतंत्र} ज्यामितीय $\mathcal{G}(p)$ चर हैं
(स्मृतिहीनता: हर चित्त के बाद खेल फिर से शुरू हो जाता है), अतः $T_r = W_1 + \dots + W_r$,
और गुणनात्मकता (\cref{thm:b2:genfun:product}) देती है
\[
G_{T_r}(t) = \Bigl(\frac{pt}{1 - qt}\Bigr)^{r},
\qquad
\E(T_r) = r\,\E(W_1) = \frac rp,
\qquad
V(T_r) = r\,V(W_1) = \frac{rq}{p^2}
\]
($q = 1 - p$; \omterm{def:b2:randomvar:variance}{प्रसरण} स्वतंत्रता से जुड़ जाते हैं)। प्रसार: व्यापकीकृत
द्विपद श्रेणी (\cref{ch:b2:powerseries}) से $(1 - qt)^{-r} = \sum_{m\geq0}
\binom{m + r - 1}{r - 1}q^mt^m$, अतः $p^rt^r(1 - qt)^{-r}$ में $t^n$ का गुणांक ($m = n - r$ के
साथ)
\[
\P(T_r = n) = \binom{n-1}{r-1}p^r(1-p)^{n-r},
\qquad n \geq r ,
\]
है --- अर्थात् \emph{ऋणात्मक द्विपद} \omterm{def:b2:randomvar:law}{नियम}; और साहचर्यिक रूप से:
$r$-वाँ चित्त उछाल $n$ पर तभी गिरता है जब पिछले $r - 1$ चित्त अपने
स्थान पहले $n - 1$ उछालों में चुनें।
\end{solution}

\begin{solution}{exo:b2:genfun:6}
$m = 1\cdot\frac38 + 2\cdot\frac38 + 3\cdot\frac18 = \frac{3 + 6 +
3}{8} = \frac32 > 1$: अधिक्रांतिक। \omterm{ex:b2:powerseries:fibonacci}{जनक फलन}

G(t) = \frac{1 + 3t + 3t^2 + t^3}{8} = \frac{(1 + t)^3}{8} ,
\]
है, अतः अचल बिंदु $(1 + t)^3 = 8t$ हल करते हैं, अर्थात् $t^3 + 3t^2 - 5t + 1

t^3 + 3t^2 - 5t + 1 = (t - 1)\bigl(t^2 + 4t - 1\bigr),

और $t^2 + 4t - 1 = 0$ $t = -2 \pm \sqrt5$ देता है। $\intco{0}{1}$ में मूल $\sqrt5 - 2 \approx 0.236$ है: \cref{thm:b2:genfun:extinction} से,

q = \sqrt 5 - 2 .
\]
(एक सुखद जाँच: संतति \omterm{def:b2:randomvar:law}{नियम} $3$ \omterm{def:b2:proba:independence}{स्वतंत्र} निष्पक्ष सिक्कों का \omterm{def:b2:randomvar:law}{नियम}
है, $Z_1 \sim \mathcal{B}(3, \frac12)$।)
\end{solution}

\begin{solution}{exo:b2:genfun:7}
\emph{\omterm{def:b2:randomvar:expectation}{प्रत्याशा}।} पहले $\E(Z_n) = m^n$: \cref{thm:b2:genfun:compound} से $\E(Z_{n+1}) = \E(Z_n)\,m$, और $\E(Z_0) = 1$। कुल $\bigl(Z_n(\omega)\P(\{\omega\})
\bigr)_{n, \omega}$
अऋणात्मक है, अतः कुलों के लिए फ़ूबिनी बिना किसी शर्त के लागू होती
है:
\[
\E(Y) = \sum_{n=0}^{\infty}\E(Z_n)
= \sum_{n=0}^\infty m^n = \frac{1}{1 - m} < \infty
\]
(विशेष रूप से $Y$ लगभग निश्चित रूप से परिमित है: जो अवक्रांतिक
स्थिति में निश्चित विलोपन से संगत है)।

\emph{फलनिक समीकरण।} जनसंख्या को पूर्वज के बच्चों के अनुसार अपघटित
कीजिए: यदि पूर्वज के $Z_1 = k$ बच्चे हों, तो कुल संतति $Y = 1 + Y_1 + \dots + Y_k$ है, जहाँ $Y_i$
$i$-वें बच्चे के वंश की कुल संतति है --- और $Y_i$ $Y$ की \omterm{def:b2:proba:independence}{स्वतंत्र}
प्रतिलिपियाँ हैं, जो $Z_1$ से \omterm{def:b2:proba:independence}{स्वतंत्र} हैं (भिन्न वंश असंयुक्त,
\omterm{def:b2:proba:independence}{स्वतंत्र} प्रजनन-घटनाएँ बरतते हैं)। \cref{thm:b2:genfun:compound} की तरह $Z_1$ पर प्रतिबंधित
करने पर:
\[
H(t) = \E\bigl(t^Y\bigr)
= t\sum_{k=0}^\infty \P(Z_1 = k)\,H(t)^k
= t\,G\bigl(H(t)\bigr),
\]
जहाँ गुणक $t$ स्वयं पूर्वज का लेखा रखता है। (द्विआधारी शाखन के
\omterm{def:b2:randomvar:law}{नियम} $p_0 = 1 - p$, $p_2 = p$ के लिए $H$ में यह द्विघात समीकरण स्पष्ट रूप से हल
और प्रसारित किया जा सकता है --- \cref{ch:b2:powerseries} की \omterm{ex:b2:powerseries:catalan}{कातालान संख्याएँ} कुल-वृक्ष
गिनती हैं।)
\end{solution}

\begin{solution}{exo:b2:genfun:8}
मान लीजिए $R > 1$ त्रिज्या है और $t \in \intoo{1}{R}$ स्थिर कीजिए। तब $\E(t^X) = G(t) < \infty$, और अऋणात्मक
चर $t^X$ पर स्तर $t^n$ पर लगाई गई मार्कोव असमिका (\cref{thm:b2:randomvar:markov}):
\[
\P(X \geq n) = \P\bigl(t^X \geq t^n\bigr)
\leq \frac{G(t)}{t^n} = C\rho^n,
\qquad C = G(t),\quad \rho = \frac1t \in \intoo{0}{1}.
\]
\emph{विलोम:} यदि $\P(X \geq n) \leq C\rho^n$, तो $p_n \leq
\P(X \geq n) \leq C\rho^n$, अतः $\abs t < \frac1\rho$ के लिए श्रेणी $\sum p_n\abs t^n$
अभिसारी गुणोत्तर श्रेणी $C\sum(\rho\abs t)^n$ से प्रभावित है: त्रिज्या कम से कम
$\frac1\rho > 1$ है। \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} की त्रिज्या और पुच्छ का गुणोत्तर क्षय एक ही
गुणधर्म के दो चेहरे हैं।
\end{solution}

\begin{solution}{exo:b2:genfun:9}
$p_k^{(n)} = \P(X_n = k)$, $p_k = \P(X = k)$ लिखिए।

\emph{स्थिति $k = 0$।} $t \in \intoo{0}{1}$ तथा $\sum_j q_j \leq 1$ वाले किसी भी \omterm{def:b2:randomvar:law}{नियम} $(q_j)$ के लिए:
\[
\Bigl|\,q_0 - \sum_j q_jt^j\Bigr|
= \sum_{j \geq 1} q_j t^j
\leq \sum_{j\geq1}t^j = \frac{t}{1 - t} .
\]
इसलिए
\[
\abs{p_0^{(n)} - p_0}
\leq \frac{2t}{1 - t}
+ \abs{G_{X_n}(t) - G_X(t)} .
\]
$\varepsilon > 0$ दिया हो, तो $\frac{2t}{1-t} <
\frac\varepsilon2$ वाला $t$ चुनिए, फिर ऐसा $n_0$ कि अंतिम पद $n \geq n_0$
के लिए $<
\frac\varepsilon2$ हो: अतः $p_0^{(n)} \to p_0$।

\emph{आगमन-चरण।} मान लीजिए $j < k$ के लिए $p_j^{(n)} \to p_j$। \emph{खिसके हुए} फलनों
\[
g_n(t) = \frac{G_{X_n}(t) - p^{(n)}_0}{t}
= \sum_{j\geq0} p^{(n)}_{j+1}t^j,
\qquad
g(t) = \frac{G_X(t) - p_0}{t} ,
\]
पर विचार कीजिए, जो उप-प्रायिकता अनुक्रमों $(p^{(n)}_{j+1})_j$ के \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} हैं (कुल
द्रव्यमान $\leq 1$, और $k = 0$ वाले तर्क ने इतना ही बरता था)। नियत $t \in \intoo{0}{1}$ के
लिए परिकल्पना तथा स्थिति $k = 0$ से $g_n(t)
\to g(t)$। $g_n$ पर $k =
0$ वाला तर्क लगाने पर
$p_1^{(n)} \to p_1$ मिलता है; और खिसकाव को $k$ बार दोहराने पर हर $k$ के लिए $p_k^{(n)} \to p_k$
मिल जाता है। (यह लेवी की संततता प्रमेय का विविक्त, प्रारंभिक
उदाहरण है, जिसका व्यापक रूप --- अभिलक्षणिक फलनों के लिए --- वर्ष 3
का मील का पत्थर है।)
\end{solution}

\begin{solution}{exo:b2:genfun:10}
\omterm{def:b2:funcseq:def}{बिंदुवार}, $\frac{1 + (-1)^X}{2}$ $X$ के सम होने पर $1$ के बराबर है और विषम होने पर
$0$, अतः \omterm{def:b2:randomvar:expectation}{प्रत्याशा} लेने पर (अंतरण),
\[
\P(X \text{ सम}) = \frac{1 + \E\bigl((-1)^X\bigr)}2 =
\frac{1 + G_X(-1)}2 .
\]
प्वासों: $\lambda$ के बढ़ने के साथ $\frac{1 + \eu^{-2\lambda}}2 \to \frac12$। द्विपद: $\frac{1 + (1 - 2p)^n}2$। दोनों स्थितियों में
$G_X(-1) \to 0$ कहता है कि $X$ की सम-विषमता कोई निष्पक्ष सिक्का बन जाती है:
\omterm{def:b2:randomvar:law}{नियम} बहुत-से पूर्णांकों पर फैल जाता है और अपनी सम-विषमता भूल जाता
है।
\end{solution}

\begin{solution}{exo:b2:genfun:11}
$t + \dots + t^6 = t\,\frac{1 - t^6}{1 - t}$ तथा $1 - t^6 =
(1 - t)(1 + t)(1 + t + t^2)(1 - t + t^2)$, जिससे बताया गया गुणनखंडन मिल जाता है। पहले पासे के
लिए $(1 + t)(1 + t + t^2) = 1 +
2t + 2t^2 + t^3$, अतः $\frac{t(1+t)(1+t+t^2)}6 = \frac{t +
2t^2 + 2t^3 + t^4}6$: फलक $\{1, 2, 2, 3, 3, 4\}$। दूसरे के लिए प्रसार करने पर
\[
(1 + 2t + 2t^2 + t^3)(1 - t + t^2)^2 = 1 + t^2 + t^3 + t^4 +
t^5 + t^7,
\]
अतः $\frac{t(1+t)(1+t+t^2)(1-t+t^2)^2}6 = \frac{t + t^3 + t^4
+ t^5 + t^6 + t^8}6$: फलक $\{1, 3, 4, 5, 6, 8\}$। दोनों जनक फलनों का गुणनफल छहों गुणनखंडों को $\bigl(\frac{t(1+t)(1+t+t^2)(1-t+t^2)}6
\bigr)^2$
में फिर समूहित कर देता है, जो मानक पासे के फलन का वर्ग है: अर्थात्
सिशरमान युग्म का कुल \omterm{def:b2:randomvar:law}{नियम} ठीक मानक है --- और \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} ऐसे सारे
पुनःसमूहन वर्गीकृत कर देते हैं।
\end{solution}

\begin{solution}{exo:b2:genfun:12}
मान लीजिए $A$ तथा $B$ क्रमशः ``कोई चालू चित्त नहीं'' तथा ``एक चालू
चित्त'' से शुरू होने वाली शेष अवधि के \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} हैं। एक उछाल ख़र्च
होता है, फिर: अवस्था $0$ से पट अवस्था $0$ पर लौटा देता है और चित्त
अवस्था $1$ पर ले जाता है; अवस्था $1$ से चित्त खेल समाप्त कर देता है
और पट अवस्था $0$ पर लौटा देता है:
\[
A(t) = t\bigl(q\,A(t) + p\,B(t)\bigr),
\qquad
B(t) = t\bigl(p + q\,A(t)\bigr).
\]
प्रतिस्थापित करने पर: $A(1 - qt) = pt\,B = pt(pt + qtA)$, अतः
\[
G_T(t) = A(t) = \frac{p^2t^2}{1 - qt - pq\,t^2} .
\]
$t = 1$ पर हर $1 - q - pq = p(1 - q) = p^2$ है: $G_T(1) = 1$, अर्थात् खेल लगभग निश्चित रूप से समाप्त
होता है (जैसा \cref{exo:b2:proba:6} ने पुनरावृत्ति से दिखाया)। $1$ पर लघुगणकीय
अवकलन: $D(t) = 1 - qt - pqt^2$, $D'(1) = -q - 2pq$ के साथ $\E(T) = 2 - \frac{D'(1)}{D(1)}$:
\[
\E(T) = 2 + \frac{q + 2pq}{p^2} = \frac{2p^2 + q + 2pq}{p^2}
= \frac{1 + p}{p^2},
\]
जो $p = \frac12$ के लिए $6$ है।
\end{solution}

\begin{solution}{pb:b2:genfun:1}
\textbf{1.} $t \in \intoo01$ के लिए $G_n = G
\circ G_{n-1}$ पर शृंखला नियम $G_n'(t) =
G'\bigl(G_{n-1}(t)\bigr)G_{n-1}'(t)$ देता है। $t \to 1^-$ होने पर
$G_{n-1}(t) \uparrow 1$, और $G'$ अनह्रासमान है जिसकी $1$ पर बायीं सीमा $m$ है, अतः पहला
गुणक $m$ की ओर जाता है; और आगमन से दूसरा $m^{n-1}$ की ओर। \cref{thm:b2:genfun:moments} से $\E(Z_n) = G_n'(1^-) = m^n$।

\textbf{2.} एक बार और अवकलन करने पर
\[
G_n'' = G''(G_{n-1})\,(G_{n-1}')^2 +
G'(G_{n-1})\,G_{n-1}'',
\]
और $t \to 1^-$ लेने पर: $a_n = G_n''(1)$, $a_1 = G''(1)$ के साथ $a_n = G''(1)m^{2(n-1)} + m\,
a_{n-1}$। $m \neq
1$ के लिए आगमन से जाँचा जाता
है कि $a_n = G''(1)\,m^{n-1}
\frac{m^n - 1}{m - 1}$ (पुनरावृत्ति $m\cdot G''(1)m^{n-2}\frac{m^{n-1}-1}{m-1}$ में $G''(1)m^{2n-2}$ जोड़ती है, और $m^{n-1} + \frac{m^{n-1}-1}{m-1} = \frac{m^n - 1}{m-1}$); $m = 1$ के लिए
$a_n = a_{n-1} + G''(1) = n\,G''(1)$।

\textbf{3.} $V(Z_n) = a_n + m^n - m^{2n}$ तथा $G''(1) =
\sigma^2 + m^2 - m$। $m \neq 1$ के लिए टुकड़ा $(m^2 -
m)m^{n-1}\frac{m^n-1}{m-1} = m^n(m^n - 1)$ ठीक $m^n -
m^{2n}$ को काट देता
है, और $V(Z_n) =
\sigma^2m^{n-1}\frac{m^n-1}{m-1}$ बच रहता है। $m = 1$ के लिए: $V(Z_n) =
nG''(1) = n\sigma^2$।

\textbf{4.} $Z_n$ कोई अऋणात्मक पूर्णांक चर है, अतः मार्कोव (\cref{thm:b2:randomvar:markov}) से
$\P(Z_n > 0) = \P(Z_n \geq 1) \leq \E(Z_n) = m^n$। $m < 1$ के लिए यह गुणोत्तर रूप से --- और \omterm{def:b2:series:summable}{योग्य} रूप से --- क्षय
करता है, अतः बोरेल--कांतेली यह भी दे देती है कि केवल परिमित रूप से
कई पीढ़ियाँ अरिक्त हैं, जो फिर से विलोपन ही है।

\textbf{5.} कोशी--श्वार्ज़: $\E(Z_n)^2 = \E(Z_n\mathbf
1_{Z_n>0})^2 \leq \E(Z_n^2)\,\P(Z_n > 0)$। प्रश्न 3 तथा $m < 1$ के साथ:
\[
\E(Z_n^2) = V(Z_n) + m^{2n}
\leq \frac{\sigma^2m^{n-1}}{1-m} + m^{2n},
\]
अतः $m^{2n}$ को इस परिबंध से भाग देकर और $m^n$ से सरल करके,
\[
\P(Z_n > 0) \geq \frac{m^n}{\frac{\sigma^2}{m(1-m)} + m^n}
\geq \Bigl(\frac{\sigma^2}{m(1-m)} + 1\Bigr)^{-1}m^n ,
\]
जहाँ हर में $m^n \leq 1$ का उपयोग हुआ। प्रश्न 4 के साथ: $\P(Z_n > 0) \asymp m^n$।

\textbf{6.} $G(t) = q\sum_k(pt)^k = \frac{q}{1 - pt}$, और $m
= G'(1) = \frac{pq}{(1-p)^2} = \frac pq$। $p
< \frac12$ के लिए अवक्रांतिक, $p = \frac12$ के लिए
क्रांतिक, $p
> \frac12$ के लिए अधिक्रांतिक।

\textbf{7.} $G(t) = t$ का अर्थ है $pt^2 - t + q = 0$, जिसके मूल $\frac{1 \pm \abs{p - q}}{2p}$ हैं, अर्थात् $1$
तथा $\frac qp =
\frac1m$। विलोपन प्रायिकता $\intcc01$ का सबसे छोटा अचल बिंदु है (\cref{thm:b2:genfun:extinction}): $m \leq 1$
होने पर $q_{\mathrm{ext}} = 1$, और $m >
1$ होने पर $\frac1m$।

\textbf{8.} $m \neq 1$ के लिए $p = \frac m{m+1}$, $q =
\frac1{m+1}$ के साथ: यदि $q_n = \frac{m^n - 1}{m^{n+1} - 1}$, तो
\[
1 - p\,q_n = \frac{(m+1)(m^{n+1} - 1) - m(m^n - 1)}
{(m+1)(m^{n+1} - 1)} = \frac{m^{n+2} - 1}{(m+1)(m^{n+1} -
1)},
\]
अतः $q_{n+1} = \frac{q}{1 - pq_n} = \frac{m^{n+1} -
1}{m^{n+2} - 1}$; और आधार स्थिति $q_0 = 0$ टिकती है। $m = 1$ के लिए: $G(t) = \frac1{2 - t}$ तथा $q_{n+1} = \frac1{2 -
\frac{n}{n+1}} = \frac{n+1}{n+2}$,
जहाँ $q_0 = 0$।

\textbf{9.} $1 - q_n = \frac{m^n(m - 1)}{m^{n+1} - 1}$। $m < 1$ के लिए हर $-1$ की ओर जाता है: $1 - q_n \sim (1 -
m)\,m^n$। $m > 1$ के लिए:
\[
q_{\mathrm{ext}} - q_n = \frac1m - \frac{m^n - 1}{m^{n+1} -
1} = \frac{m - 1}{m\,(m^{n+1} - 1)} \sim \frac{m -
1}{m^{2}}\;m^{-n} .
\]
और $t = \frac
qp$ पर मूल्यांकित $G'(t) = \frac{pq}{(1 - pt)^2}$ (जहाँ $1 - pt = 1 - q = p$) $G'(q_{\mathrm{ext}})
= \frac qp = \frac1m$ देता है: अर्थात् प्रेक्षित
अनुपात $m^{-1}$ ठीक आकर्षक अचल बिंदु पर अवकलज है।

\textbf{10.} $p = \frac12$ के लिए: $G''(t) = \frac{1/4}{(1 -
t/2)^3}$, अतः $G''(1) = 2$ तथा $\sigma^2 = G''(1) + m - m^2 =
2$। \omterm{def:b2:multint:exact}{संवृत रूप} $1 - q_n = \frac1{n+1}$ देता
है: अर्थात् उत्तरजीविता प्रायिकता $1/n$ की तरह क्षय करती है --- इतनी
धीमे कि वह \omterm{def:b2:series:summable}{योग्य} नहीं, जबकि कोई भी अवक्रांतिक दर होती।

\textbf{11.} आगमन: $G_1(t) = \frac1{2-t}$ $n = 1$ के लिए सूत्र से मेल खाता है, और
\[
G(G_n(t)) = \cfrac{1}{2 - \cfrac{n - (n-1)t}{n+1 - nt}}
= \frac{n + 1 - nt}{2(n+1) - 2nt - n + (n-1)t}
= \frac{n+1 - nt}{n + 2 - (n+1)t} .
\]
तब
\[
\frac{G_n(t) - q_n}{1 - q_n}
= (n+1)\,\Bigl(\frac{n - (n-1)t}{n+1 - nt} -
\frac{n}{n+1}\Bigr)
= \frac{t}{n + 1 - nt}
= \frac{\frac{t}{n+1}}{1 - \frac{n}{n+1}t} ,
\]
जो $\N^*$ पर ज्यामितीय \omterm{def:b2:randomvar:law}{नियम} $\mathcal G\bigl(\frac1{n+1}
\bigr)$ का \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} है (\cref{ex:b2:genfun:classical}): उत्तरजीविता दी
हो तो $\P(Z_n = k \mid Z_n > 0) =
\frac1{n+1}\bigl(\frac n{n+1}\bigr)^{k-1}$, और सप्रतिबंध माध्य $n + 1$। बिना प्रतिबंधन का माध्य $1 = \E(Z_n)$
किसी लुप्त होती उत्तरजीविता प्रायिकता तथा रैखिक रूप से बढ़ते
सप्रतिबंध आकार का गुणनफल है।

\textbf{12.} यदि वंश पीढ़ी $n$ पर विलुप्त हो जाए, तो $Y = Z_0 + \dots + Z_{n-1}$ परिमित
है; और यदि वह कभी विलुप्त न हो, तो $Y \geq \sum_n 1 = \infty$। अतः $\{Y < \infty\}$ विलोपन \omterm{def:b2:proba:space}{घटना} है और
$\P(Y < \infty) = q_{\mathrm{ext}}$। \cref{exo:b2:genfun:7} में $H(t) = tG(H(t))$ की व्युत्पत्ति --- जहाँ पूर्वज गुणक $t$ का योगदान
करता है, और उसके बच्चे $Y$ की \omterm{def:b2:proba:independence}{स्वतंत्र} प्रतिलिपियाँ स्थापित करते
हैं जिनकी गिनती $G$ के द्वारा होती है --- केवल
\cref{thm:b2:genfun:compound} बरतती है, जो हर व्यवस्था में वैध है।

\textbf{13.} $G(s) = \frac{1 + s^2}2$ के साथ समीकरण $tH^2 - 2H + t = 0$ पढ़ा जाता है, अतः $H = \frac{1 - \sqrt{1 - t^2}}{t}$ ($H(0) = 0$
वाला मूल)। कातालान श्रेणी
$C(x) = \frac{1 - \sqrt{1 - 4x}}{2x}$
(\cref{ex:b2:powerseries:catalan}) से तुलना करने पर: $H(t) = \frac
t2\,C\bigl(\frac{t^2}4\bigr) =
\sum_{k\geq0}C_k\,\frac{t^{2k+1}}{2^{2k+1}}$, अर्थात् $\P(Y =
2k+1) = C_k2^{-2k-1}$। जाँचें: $\P(Y = 1) = C_0/2 = \frac12$ (पूर्वज के
कोई बच्चा नहीं); $\P(Y = 3) = C_1/8 = \frac18$ (दो बच्चे, दोनों निःसंतान: $\frac12\cdot\frac12\cdot
\frac12$)।

\textbf{14.} $\intoo01$ पर $H = tG(H)$ का अवकलन करके और $t \to 1^-$ लेकर (एकदिष्ट सीमाएँ,
जैसी \cref{thm:b2:genfun:moments} में): $H'(1)\bigl(1 - G'(H(1))\bigr)
= G(H(1))$। अवक्रांतिक स्थिति में $H(1) = 1$ तथा $\E(Y) =
H'(1) = \frac1{1 - m}$। क्रांतिक
स्थिति में $G'(1) = 1$ बायें गुणक को लुप्त कर देता है जबकि दायाँ पक्ष $1$
है: अतः कोई परिमित $H'(1)$ विद्यमान नहीं हो सकता, इसलिए $\E(Y) = \infty$ --- फिर
भी $\P(Y <
\infty) = q = 1$।

\textbf{15.} \cref{ex:b2:comparison:centralbinomial} से $C_k = \frac1{k+1}\binom{2k}k \sim
\frac{4^k}{\sqrt\pi\,k^{3/2}}$, अतः
\[
\P(Y = 2k+1) = \frac{C_k}{2\cdot4^{k}} \sim
\frac1{2\sqrt\pi\,k^{3/2}} .
\]
पुच्छ जोड़ने पर ($\int_K^\infty
k^{-3/2}\dd k = 2K^{-1/2}$ से ऊपर तथा नीचे तुलना करके): $\P(Y > 2K)
\asymp K^{-1/2}$, अर्थात्
$\P(Y > n) \asymp n^{-1/2}$ --- अपरिमित माध्य वाली कोई भारी पुच्छ, जो प्रश्न 14 को
परिमाणित कर देती है।

\textbf{16.} दोनों क्रांतिक वस्तुएँ --- निष्पक्ष चहलक़दमी का
वापसी-समय (\cref{ch:b2:proba} की सप्ताहांत समस्या) तथा क्रांतिक कुल संतति ---
लगभग निश्चित रूप से परिमित हैं पर अपरिमित माध्य वाली, और उनके
स्थानीय नियमों का घातांक $-3/2$ तथा पुच्छों का घातांक $-1/2$ है। यह कोई
संयोग नहीं: किसी कुल-वृक्ष का बच्चा-दर-बच्चा अन्वेषण कोई $\pm1$ पथ
उत्पन्न करता है (हर जन्म पर एक पग ऊपर, हर मृत्यु पर एक नीचे), जो
ठीक कोई निष्पक्ष चहलक़दमी है, और $Y$ कोई प्रथम-अभिगमन समय बन जाता
है। क्रांतिकता का अर्थ है शून्य बहाव: प्रक्रम सदा विलोपन तथा
विस्फोट दोनों के कगार पर रहता है, और शून्य-बहाव वाली यादृच्छिकता के
$\sqrt{}$-मापक्रम उतार-चढ़ाव ठीक यही घातांक उत्पन्न करते हैं।

\textbf{17.} $G''(t) = \sum_{n\geq2}n(n-1)p_nt^{n-2}$ के पद अऋणात्मक हैं, अतः वह $\intco01$ पर अनह्रासमान है
और उसकी सीमा $G''(1) = \sigma^2$ परिमित है (क्रांतिकता $\E Z_1(Z_1 - 1) = \sigma^2$ बना देती है); और जिस
अनह्रासमान फलन की सीमा सीमांत मान के बराबर हो वह $1$ पर \omterm{def:b2:metric:continuity}{संतत} है।
बिंदु $1$ पर समाकल शेषफल वाली टेलर:
\[
G(t) = 1 + (t - 1) + \int_1^t(t - s)G''(s)\,\dd s
= t + \frac{G''(1)}2(1-t)^2 + o\bigl((1-t)^2\bigr),
\]
क्योंकि $s \to 1^-$ होने पर $G''(s) = G''(1) + o(1)$।

\textbf{18.} साझे हर पर लाने पर $h(t) =
\frac{G(t) - t}{(1 - G(t))(1 - t)}$। प्रश्न 17 से अंश $b(1-t)^2 + o((1-t)^2)$ है और
$1 - G(t) = (1 -
t)\bigl(1 - b(1-t) + o(1-t)\bigr)$, अतः $h(t) \to b$।

\textbf{19.} $t = q_j$ तथा $G(q_j) =
q_{j+1}$ पर $h$ की परिभाषा से: $\frac1{1 - q_{j+1}} - \frac1{1-q_j} = h(q_j)$; और $j = 0$ से
($q_0 = 0$) जोड़ने पर प्रदर्शित रूप मिल जाता है। चूँकि क्रांतिक प्रक्रम
मिट जाता है, $q_j \uparrow 1$, अतः $h(q_j) \to
b$ और चेज़ारो माध्य $\frac1n\sum_{j<n}h(q_j) \to b$: $\frac1{1-q_n} \sim bn$, अर्थात्
\[
\P(Z_n > 0) \sim \frac1{bn} = \frac{2}{\sigma^2 n} .
\]

\textbf{20.} ज्यामितीय क्रांतिक स्थिति: $\sigma^2 = 2$ (प्रश्न 10), अतः
कोल्मोगोरोव $1 - q_n \sim \frac1n$ का अनुमान लगाता है --- और सटीक मान $\frac1{n+1}$ है।

\textbf{21.} चूँकि $Z_n\mathbf 1_{Z_n > 0} = Z_n$, $\E(Z_n
\mid Z_n > 0) = \frac{\E(Z_n)}{\P(Z_n > 0)} = \frac1{1 -
q_n} \sim \frac{\sigma^2n}2$। ज्यामितीय स्थिति में यह $n + 1$ है, जो
प्रश्न 11 से ठीक-ठीक मेल खाता है ($\sigma^2 = 2$)। क्रांतिक चित्र: विलोपन
निश्चित है, माध्य आकार $1$ पर जमा हुआ है, और विरल बचे हुए वंशों का
आकार रैखिक रूप से बढ़ता है --- हर गुणक दूसरे को संतुलित करता हुआ।

\textbf{22.} $\mathcal P(\lambda)$ संतति के लिए $G(t) =
\eu^{\lambda(t-1)}$, और विलोपन प्रायिकता $q = \eu^{\lambda(q-1)}$ का सबसे
छोटा मूल है। संख्यात्मक रूप से: $\lambda = 1.5$ $q \approx 0.417$ देता है ($q \mapsto
\eu^{1.5(q-1)}$ दोहराइए:
$0, 0.223, 0.312, 0.356, \dots \to 0.4172$); $\lambda = 2$ $q \approx 0.203$ देता है। अतः एक सूचक रोगी प्रायिकता $58\%$ ($\lambda = 1.5$) या
$80\%$ ($\lambda = 2$) के साथ किसी बड़े प्रकोप को जन्म देता है --- संभावित,
निश्चित नहीं। $q_0 = 0$ से पुनरावृत्ति \emph{सबसे छोटे} मूल तक इसलिए
अभिसरित होती है कि $G$ अनह्रासमान है: आगमन से किसी भी अचल बिंदु
$r$ के लिए $q_n \leq r$, और $(q_n)$ बढ़ता है (वह $\P(Z_n = 0)$ है), अतः उसकी सीमा सब
से नीचे कोई अचल बिंदु है।

\textbf{23.} $k$ पूर्वज \omterm{def:b2:proba:independence}{स्वतंत्र} कुल-वृक्ष स्थापित करते हैं, और
पूर्ण विलोपन $k$ \omterm{def:b2:proba:independence}{स्वतंत्र} विलोपन \omterm{def:b2:proba:space}{घटनाओं} का प्रतिच्छेद है:
प्रायिकता $q^k$। $\lambda = 1.5$ के लिए: प्रकोप प्रायिकता $1 - q^k \geq 0.99$ के लिए $q^k \leq 0.01$ चाहिए,
अर्थात् $k \geq
\frac{\ln 0.01}{\ln 0.417} \approx 5.3$: यानी छह आरंभिक रोगी प्रकोप को $99\%$ निश्चित बना देते
हैं।

\textbf{24.} \emph{$G'(q) < 1$:} $G - \mathrm{id}$ \omterm{def:b2:affine:convex}{उत्तल} है और $q$ तथा $1$ पर लुप्त होता
है, अतः वह $\intcc
q1$ पर $\leq 0$ है; और यदि $G'(q) = 1$ होता, तो $q$ पर स्पर्शी (जिसे
उत्तलता $G$ के नीचे रखती है) $\intcc q1$ पर $G(t) \geq t$ को बाध्य कर देती,
इसलिए वहाँ $G \equiv \mathrm{id}$, जो सारे गुणांक $p_n$ ($n \geq 2$) मार देता और $m > 1$ के
विरुद्ध पड़ता। \emph{गुणोत्तर अभिसरण:} सब $n$ के लिए $q_n < q$ (आगमन,
$G$ वर्धमान), और माध्यमान प्रमेय $c_n \in \intoo{q_n}q$ के साथ $q - q_{n+1} =
G'(c_n)(q - q_n)$ देती है, अतः
$G'(c_n)
\leq G'(q) < 1$ तथा $q - q_n \leq q\,G'(q)^n$। \emph{साथी प्रक्रम:} $\widehat G(t) = G(qt)/q =
\sum_kp_kq^{k-1}t^k$ के गुणांक अऋणात्मक हैं और
$\widehat G(1) = G(q)/q = 1$: अर्थात् कोई \omterm{ex:b2:powerseries:fibonacci}{जनक फलन}; और उसका माध्य $\widehat
G'(1) = G'(q) < 1$ है: यानी
अवक्रांतिक। ज्यामितीय कुल: $G(t) =
\frac{q}{1-pt}$, $q_{\mathrm{ext}} = \frac qp$, और
\[
\widehat G(t) = \frac pq\cdot\frac{q}{1 - p\frac qp t}
= \frac{p}{1 - qt} :
\]
अर्थात् $p$ तथा $q$ की अदला-बदली वाला ज्यामितीय संतति \omterm{def:b2:randomvar:law}{नियम} ---
अपनी विलोपन \omterm{def:b2:proba:space}{घटना} पर देखा गया अधिक्रांतिक प्रक्रम दर्पण में
अवक्रांतिक प्रक्रम है।

\textbf{25.} सारणी: $m < 1$: $q = 1$, $\P(Z_n > 0)
\asymp m^n$ (प्रश्न 4 से 5), $\E(Y) = \frac1{1-m}$, और परिबद्ध
सप्रतिबंध माध्य वाली बची हुई पीढ़ियाँ। $m = 1$: $q = 1$, $\P(Z_n > 0) \sim \frac2{\sigma^2n}$
(कोल्मोगोरोव), $\P(Y > n) \asymp n^{-1/2}$ के साथ $\E(Y) = \infty$, और $\sim \frac{\sigma^2n}2$ आकार के उत्तरजीवी। $m > 1$:
$q < 1$ सबसे छोटा अचल बिंदु है, $q - q_n = O(G'(q)^n)$, वृद्धि $\E(Z_n) = m^n$, और मरने पर
प्रतिबंधित करने पर प्रक्रम अवक्रांतिक साथी है (प्रश्न 24)। औज़ार:
जनक फलनों के संयोजन ने जनसंख्या-पुनरावृत्ति को फलन-पुनरावृत्ति में
बदल दिया; उत्तलता ने अचल बिंदुओं की ज्यामिति नियत कर दी; $1^-$ पर
टेलर ने आघूर्ण-परिकल्पनाओं को स्थानीय प्रसारों में बदला; और चेज़ारो
औसतन ने किसी दूरबीनी योग से कोल्मोगोरोव का $1/n$ निकाल लिया। वर्ष 3
का खंड मार्टिंगेल $Z_n/
m^n$ जोड़ता है --- जिसकी लगभग-निश्चित सीमा $\E(Z_n) = m^n$
को प्रक्षेप-दर-प्रक्षेप वृद्धि दर में परिष्कृत कर देती है --- तथा
याग्लोम की प्रमेय, जो प्रश्न 11 में देखी गई सप्रतिबंध ज्यामिति के
पीछे का सीमा नियम है।
\end{solution}
