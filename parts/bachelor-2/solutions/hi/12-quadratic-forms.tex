\end{solution}

\begin{solution}{exo:b2:quadratic:3}
$u^{**} = u$: सभी $y$ के लिए $\langle u^{**}x, y\rangle = \langle x, u^*y\rangle =
\langle ux, y\rangle$। $(uv)^* = v^*u^*$: $\langle
uvx, y\rangle = \langle vx, u^*y\rangle = \langle x,
v^*u^*y\rangle$। केंद्रक: $y \in \ker u^* \iff \langle x, u^*y
\rangle = 0\ \forall x \iff \langle u(x), y\rangle = 0\ \forall x
\iff y \perp \operatorname{im} u$। कोटियाँ: $\dim\ker u^* = n -
\operatorname{rk} u$ (लांबिक
पूरक), अतः कोटि--शून्यता से $\operatorname{rk} u^* = \operatorname{rk} u$ --- अर्थात् परिवर्त-कोटि प्रमेय का
यूक्लिडीय अवतार।
\end{solution}

\begin{solution}{exo:b2:quadratic:4}
वर्णक्रमीय: $A = PDP^{\mathsf T}$, $D$ विकर्ण जिसकी प्रविष्टियाँ $\lambda_i$ $\lambda_i^3 = \lambda_i$ पूरा करती हैं:
$\lambda_i \in
\{-1, 0, 1\}$। तब $D^2$ विकर्ण सहित $A^2 = PD^2P^{\mathsf T}$, जिसकी प्रविष्टियाँ $0/1$ हैं: अतः $A^2$ \omterm{def:b2:quadratic:adjoint}{सममित} और
वर्गसम है ($(A^2)^2 = A^4 = A\cdot A^3 = A^2$) --- और \omterm{def:b2:quadratic:adjoint}{सममित} वर्गसम $=$ लांबिक प्रक्षेप होता है (वह $\ker A$
के अनुदिश $\ker(A^2 - I) = \ker(A-I)\oplus\ker(A+I)$ पर प्रक्षेप है, और वर्णक्रमीय प्रमेय से ये लांबिक हैं)।

सामान्य रूप में $P(A) = P\!\left(\text{विकर्ण}\right)$: $P(A)$ के \omterm{def:b2:reduction:eigen}{अभिलक्षणिक सदिश} वही हैं
और \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $P(\lambda_i)$ --- अर्थात् \omterm{def:b2:reduction:diag}{विकर्णनीय} स्तर पर ``वर्णक्रमीय
प्रतिचित्रण''।
\end{solution}

P^{\mathsf T}P$ (बहुपद प्रविष्टियाँ) के लिए $O(n) = g^{-1}(\{I\})$। परिबद्ध: $P

इकाई सदिश है, अतः सारी प्रविष्टियाँ $\intcc{-1}{1}$ में हैं। $\mathcal{M}_n(\R) \simeq \R^{n^2}$ में संवृत और परिबद्ध:

\begin{solution}{exo:b2:quadratic:6}
\emph{अस्तित्व:} $D =
\operatorname{diag}(\lambda_i)$, $\lambda_i \geq 0$ सहित $A = PDP^{\mathsf T}$; $\sqrt D =
\operatorname{diag}(\sqrt{\lambda_i})$ के साथ $B =
P\sqrt D P^{\mathsf T}$ रखिए: यह \omterm{def:b2:quadratic:adjoint}{सममित},
धनात्मक अर्धनिश्चित है और $B^2 = A$।

\emph{अद्वितीयता:} मान लीजिए $B$ \omterm{def:b2:quadratic:adjoint}{सममित} धनात्मक अर्धनिश्चित है और $B^2 = A$। $B$
$A$ के साथ क्रमविनिमेय है; अतः $B$ हर \omterm{def:b2:reduction:eigen}{अभिलक्षणिक समष्टि} $E_\lambda(A)$ को सुरक्षित
रखता है ($Ax = \lambda x$ के लिए: $A(Bx) = BAx = \lambda Bx$)। $E_\lambda(A)$ पर $B$ का प्रतिबंधन \omterm{def:b2:quadratic:adjoint}{सममित} धनात्मक
अर्धनिश्चित है जिसका वर्ग $\lambda\,\mathrm{id}$ है; उसके \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\mu$ $\mu^2
= \lambda$, $\mu \geq 0$
पूरा करते हैं: $\mu = \sqrt\lambda$ --- अतः वह प्रतिबंधन, जो अकेले \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\sqrt\lambda$ के
साथ \omterm{def:b2:reduction:diag}{विकर्णनीय} है, $\sqrt\lambda\,\mathrm{id}$ \emph{ही} है। और चूँकि $E =
\bigoplus E_\lambda(A)$, $B$ निर्धारित हो
जाता है: $B = \sqrt A$।
\end{solution}

\begin{solution}{exo:b2:quadratic:7}
किसी प्रसामान्य लांबिक अभिलक्षणिक आधार में $\norm{Ax}_2^2 = \sum \lambda_i^2
x_i^2 \leq (\max_i \lambda_i^2)\norm x_2^2$, और संगत \omterm{def:b2:reduction:eigen}{अभिलक्षणिक सदिश}
पर समता: $\vertiii A_2 = \max\abs{\lambda_i}$। दिए गए आव्यूह के लिए: \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $3, -1$ (\cref{ex:b2:quadratic:spectralexample} का जुड़वाँ):
$\vertiii A_2 = 3$।
\end{solution}

\begin{solution}{exo:b2:quadratic:8}
($\Rightarrow$) ऊपरी-बायाँ $k \times k$ खंड $A_k$ उस (निश्चित) रूप के पहले $k$ आधार सदिशों के
विस्तार तक प्रतिबंधन का आव्यूह है: अतः धनात्मक निश्चित, इसलिए उसके
\omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} धनात्मक हैं और $\Delta_k = \det A_k > 0$।

($\Leftarrow$) $n$ पर आगमन; $n = 1$ स्पष्ट है। मान लीजिए सभी $\Delta_k > 0$। आगमन से $A_{n-1}$ धनात्मक
निश्चित है: अर्थात् $F = \operatorname{Vect}(e_1, \dots,
e_{n-1})$ तक प्रतिबंधित रूप $q$ निश्चित है। $q|_F$ का विकर्णन
कीजिए (गाउस): $q|_F = \sum y_i^2$ सहित निर्देशांक $y_1, \dots, y_{n-1}$। पूरी समष्टि में अंतिम चर में वर्ग
पूरा करने पर
\[
q = \sum_{i=1}^{n-1} \bigl(y_i + c_i x_n\bigr)^2 + c\,x_n^2
\]
उपयुक्त अचरों के साथ (क्रॉस पदों को वर्गों में समो लीजिए)। यह समानयन
चिह्नांक $(n-1 + \epsilon, \cdot)$ दिखा देता है, जहाँ $\epsilon$ $c$ का चिह्न-योगदान है; और \omterm{def:b2:linalg:det}{सारणिक}
\omterm{def:b2:quadratic:def}{सर्वांगसमता} के अंतर्गत विकर्ण गुणांकों के गुणनफल का चिह्न बनाए रखता है
($\det(P^{\mathsf T}AP) = (\det P)^2\det A$): अतः $\Delta_n
> 0$ $c > 0$ को बाध्य कर देता है। इसलिए $q$ स्वतंत्र फलनिकों के
$n$ वर्गों का योग है: अर्थात् धनात्मक निश्चित।
\end{solution}

\begin{solution}{exo:b2:quadratic:9}
मान लीजिए $(e_1, \dots, e_n)$ $\lambda_1
\geq \dots \geq \lambda_n$ के लिए कोई प्रसामान्य लांबिक अभिलक्षणिक आधार है।

\emph{न्यूनतम-महत्तम $\lambda_2 \leq$:} किसी भी अधिसमतल $H$ के लिए $V = \operatorname{Vect}(e_1, e_2)$ की विमा $2$ है
और $\dim(H \cap V) \geq 1$ (ग्रासमान): कोई इकाई $x \in H \cap V$ चुनिए, $x = ae_1 + be_2$, $a^2 + b^2 = 1$:
\[
\langle u(x), x\rangle = \lambda_1 a^2 + \lambda_2 b^2 \geq
\lambda_2 :
\]
अतः हर अधिसमतल का महत्तम $\geq \lambda_2$ है।

\emph{$\geq$:} $H = e_1^{\perp}$ के लिए हर इकाई $x = \sum_{i\geq2}
x_ie_i \in H$ के लिए $\langle u(x), x\rangle = \sum_{i \geq 2}
\lambda_i x_i^2 \leq \lambda_2$, और $e_2$ पर वह प्राप्त होता
है: अर्थात् इस अधिसमतल का महत्तम ठीक $\lambda_2$ है। इसलिए $H$ पर न्यूनतम $\lambda_2$ है।
\end{solution}

\begin{solution}{exo:b2:quadratic:10}
\emph{बीजीय रास्ता:} $2q = \bigl(\sum x_i\bigr)^2 - \sum
x_i^2$। अधिसमतल $H : \sum x_i = 0$ (विमा $n -
1$) पर $q = -\frac12\sum x_i^2$ ऋणात्मक निश्चित
है; और रेखा $\R(1, \dots, 1)$ पर $q(t, \dots, t) = \binom n2 t^2 > 0$। जिस उपसमष्टि पर $q$ धनात्मक निश्चित है वह $H$ से
तुच्छ रूप से मिलती है, अतः उसकी विमा $\leq 1$ है: सिल्वेस्टर (\cref{thm:b2:quadratic:sylvester}) से $s = 1$, और
$H$ से $t \geq n-1$; और कोटि $\leq n$ चिह्नांक $(1, n-1)$, कोटि $n$ को बाध्य कर देती है।

\emph{वर्णक्रमीय रास्ता:} आव्यूह $\frac12(J - I)$ है; $J$ के \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $n$ ($(1,\dots,1)$
पर) और $0$ ($H$ पर) हैं, अतः $\frac12(J-I)$ के \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\frac{n-1}{2}$ (एक बार) और $-\frac12$
($n-1$ बार) हैं: अर्थात् एक धनात्मक, $n-1$ ऋणात्मक --- \cref{cor:b2:quadratic:principalaxes} के अनुसार वही
चिह्नांक।
\end{solution}

\begin{solution}{exo:b2:quadratic:11}
$B = C^{\mathsf T}C$ लिखिए ($C =
\sqrt B$ के द्वारा \cref{exo:b2:quadratic:6})। तब $C^{\mathsf T}$ के स्तंभों $c_1, \dots, c_n$ के साथ:
\[
\operatorname{tr}(AB) = \operatorname{tr}(AC^{\mathsf T}C)
= \operatorname{tr}(CAC^{\mathsf T})
= \sum_{i=1}^n \langle A c_i, c_i\rangle .
\]
\cref{cor:b2:quadratic:principalaxes}~(2) के अनुसार हर पद $\lambda_{\min}(A)\norm{c_i}^2$ और $\lambda_{\max}(A)\norm{c_i}^2$ के बीच पड़ता है, और $\sum\norm{c_i}^2 =
\operatorname{tr}(C^{\mathsf T}C)^{\vphantom1} =
\operatorname{tr} B$: अतः दोहरी
असमिका निकल आती है। और यदि $A$ भी धनात्मक अर्धनिश्चित हो, तो $\lambda_{\min}(A) \geq 0$:
$\operatorname{tr}(AB) \geq 0$।
\end{solution}

\begin{solution}{exo:b2:quadratic:12}
\begin{enumerate}
  \item $\varphi(M, N) = \operatorname{tr}(MN)$ द्विरैखिक और \omterm{def:b2:quadratic:adjoint}{सममित} है ($\operatorname{tr}(MN) =
        \operatorname{tr}(NM)$), और $\varphi(M, M) = q(M)$: अतः $q$ $\varphi$ का \omterm{def:b2:quadratic:def}{द्विघात
        रूप} है।
  \item \omterm{def:b2:quadratic:adjoint}{सममित} $S$ और प्रतिसममित $K$ के लिए: $\operatorname{tr}(SK) =
        \operatorname{tr}\bigl((SK)^{\mathsf T}\bigr) =
        \operatorname{tr}(K^{\mathsf T}S^{\mathsf T}) =
        -\operatorname{tr}(KS) = -\operatorname{tr}(SK)$, अतः $\varphi(S, K) = 0$: अर्थात् दोनों
        उपसमष्टियाँ $\varphi$-लांबिक हैं। प्रविष्टि-दर-प्रविष्टि $\operatorname{tr}
        (M^2) = \sum_{i,j} m_{ij}m_{ji}$: \omterm{def:b2:quadratic:adjoint}{सममित}
        $M$ के लिए यह $\sum m_{ij}^2 > 0$ है ($M \neq 0$); और प्रतिसममित $M$ के लिए $-\sum m_{ij}^2 < 0$।
  \item विमाओं $\frac{n(n+1)}2$ तथा $\frac{n(n-1)}2$ के साथ $\mathcal M_n(\R) = S_n \oplus A_n$; और इस $\varphi$-लांबिक विभाजन के
        अनुकूल \omterm{thm:b2:quadratic:gauss}{गाउस समानयन} $q$ को $\frac{n(n+1)}2$ धनात्मक और $\frac{n(n-1)}2$ ऋणात्मक वर्गों के
        रूप में लिख देता है: अतः चिह्नांक $\bigl(\frac{n(n+1)}2, \frac{n(n-1)}2\bigr)$ (सिल्वेस्टर), कोटि $n^2$:
        यानी रूप अनपभ्रष्ट है।
\end{enumerate}
\end{solution}

\begin{solution}{pb:b2:quadratic:1}
\textbf{1.} अंतर्गुणन की सममितता से $G$ \omterm{def:b2:quadratic:adjoint}{सममित} है, और
\[
X^{\mathsf T}GX = \sum_{i,j}x_ix_j\langle v_i, v_j\rangle
= \Bigl\|\sum_i x_iv_i\Bigr\|^2 \geq 0 ,
\]
जिसमें समता तभी जब $\sum x_iv_i = 0$: अतः $G$ निश्चित है तभी जब एकमात्र शून्य संचय
तुच्छ हो, अर्थात् तभी जब कुल स्वतंत्र हो।

\textbf{2.} $B = \sqrt A$ (\cref{exo:b2:quadratic:6}) के साथ: $A
= B^2 = B^{\mathsf T}B$, जो $B$ के स्तंभों का ग्राम आव्यूह है;
$C = B$ लीजिए। और $X^{\mathsf T}AX = \norm{CX}^2$, अतः $A$ निश्चित है तभी जब $X \neq 0$ के लिए $CX \neq 0$, अर्थात् तभी
जब $C$ प्रतिलोमनीय हो।

\textbf{3.} रूप का $\operatorname{Vect}(e_i, e_j)$ तक प्रतिबंधन आव्यूह $\begin{pmatrix} a_{ii} & a_{ij}\\ a_{ij} & a_{jj}
\end{pmatrix}$ वाला है, जो अब भी धनात्मक
अर्धनिश्चित है: अतः उसका \omterm{def:b2:linalg:det}{सारणिक} (उसके अऋणात्मक अभिलक्षणिक मानों का
गुणनफल) $\geq 0$ है: $a_{ij}^2 \leq a_{ii}a_{jj}$। यह किसी ग्राम-साक्षात्कार के सदिशों $v_i, v_j$ के लिए
कोशी--श्वार्ज़ ही है।

\textbf{4.} \emph{अस्तित्व:} $A$ को किसी स्वतंत्र कुल $(v_1, \dots, v_n)$ का ग्राम आव्यूह
लिखिए (प्रश्न 1--2)। ग्राम--श्मिट कोई प्रसामान्य लांबिक $(e_1, \dots, e_n)$ उत्पन्न करता
है जहाँ
\[
v_k = \sum_{i \leq k} t_{ik}\,e_i,
\qquad t_{kk} = \bigl\| v_k - \operatorname{proj}_{k-1}v_k
\bigr\| > 0 ,
\]
अतः $T = (t_{ik})$ धनात्मक विकर्ण वाला ऊपरी त्रिभुजाकार है, और
\[
a_{jk} = \langle v_j, v_k\rangle
= \sum_i t_{ij}t_{ik} = (T^{\mathsf T}T)_{jk} .
\]
\emph{अद्वितीयता:} यदि $T_1^{\mathsf T}T_1 = T_2^{\mathsf T}T_2$ हो तो $R = T_1T_2^{-1}$ $R^{\mathsf T}R = I$ पूरा करता है: अर्थात् $R$
लांबिक है, और साथ ही धनात्मक विकर्ण वाला ऊपरी त्रिभुजाकार भी (ऐसों का
गुणनफल)। तब $R^{-1} = R^{\mathsf T}$ एक साथ ऊपरी (ऊपरी का प्रतिलोम) और निचला (ऊपरी का
\omterm{def:b2:linalg:transpose}{परिवर्त}) त्रिभुजाकार है: अतः विकर्ण; और लांबिक विकर्ण आव्यूह की
प्रविष्टियाँ $\pm1$ होती हैं, तथा धनात्मकता $R = I$ को बाध्य कर देती है: $T_1 = T_2$।

\textbf{5.} $i, j \leq k$ के लिए $(T^{\mathsf T}T)_{ij} = \sum_m
t_{mi}t_{mj}$ में केवल $m \leq \min(i,j) \leq k$ आते हैं: अतः $A$ का अग्र $k\times k$ खंड
$T_k^{\mathsf T}T_k$ है, जहाँ $T_k$ $T$ का अग्र खंड है। इसलिए $\Delta_k = (\det T_k)^2
= (t_{11}\cdots t_{kk})^2$। अब धनात्मक निश्चित रूप के
स्वाभाविक क्रम वाले \omterm{thm:b2:quadratic:gauss}{गाउस समानयन} में कभी कोई शून्य वर्ग-गुणांक नहीं मिलता
(धुरी-अंक क्रमशः समानीत धनात्मक निश्चित खंडों की विकर्ण प्रविष्टियाँ हैं):
वह $\ell_k = x_k + (\text{पद, जिनमें: }
x_{k+1}, \dots)$ सहित $q
= \sum_k d_k\ell_k^2$ उत्पन्न करता है, अर्थात् एकघाती
त्रिभुजाकार $L$ सहित $A = L^{\mathsf T}DL$; तब $T = \sqrt D\,L$ कोई चोलेस्की गुणनखंड है, अतः
अद्वितीयता से $t_{kk}^2 = d_k$ और
\[
d_k = \frac{(t_{11}\cdots t_{kk})^2}
{(t_{11}\cdots t_{k-1,k-1})^2}
= \frac{\Delta_k}{\Delta_{k-1}} .
\]
\cref{ex:b2:quadratic:tworoads} पर: $\Delta_1, \Delta_2,
\Delta_3 = 2, 3, 4$, और धुरी-अंक $2, \frac32, \frac43$ थे।

\textbf{6.} हर $a_{ii} = e_i^{\mathsf T}Ae_i > 0$। $D =
\operatorname{diag}(a_{ii}^{-1/2})$ और $B = DAD$ लीजिए: यह (\omterm{def:b2:quadratic:def}{सर्वांगसमता} से) धनात्मक निश्चित
है, जहाँ $b_{ii} = 1$, अतः $\operatorname{tr}
B = n$। उसके \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\mu_i > 0$ समांतर--गुणोत्तर माध्य से
\[
\det B = \prod_i\mu_i
\leq \Bigl(\frac{\sum\mu_i}{n}\Bigr)^{\!n} = 1 ,
\]
पूरा करते हैं, और $\det B = (\det D)^2\det A = \dfrac{\det A}{\prod
a_{ii}}$: अतः $\det A \leq \prod a_{ii}$।

\textbf{7.} समांतर--गुणोत्तर माध्य समता तभी होती है जब सारे $\mu_i$ बराबर हों
($1$ के); और जिस \omterm{def:b2:quadratic:adjoint}{सममित} आव्यूह का एकमात्र \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $1$ हो वह $PIP^{\mathsf T} = I$ है।
अतः समता तभी जब $B = I$, अर्थात् तभी जब $i \neq j$ के लिए $a_{ij} = 0$: यानी $A$ विकर्ण।

\textbf{8.} यदि $M$ अपभ्रष्ट हो तो दोनों पक्ष $\geq 0 =
\abs{\det M}$ हैं। अन्यथा $A = M^{\mathsf T}M$ धनात्मक
निश्चित है जहाँ $a_{ii} = \norm{c_i}^2$ और $\det A = (\det
M)^2$: अतः प्रश्न 6 $(\det M)^2 \leq \prod\norm{c_i}^2$ देता है। समता तभी जब $A = M^{\mathsf T}M$
विकर्ण हो (प्रश्न 7), अर्थात् तभी जब स्तंभ जोड़े-जोड़े में लांबिक हों।

\textbf{9.} $\abs{\det M}$ स्तंभों से बने समांतर षट्फलक का आयतन है: और वह आयतन
अधिकतम भुजाओं की लंबाइयों के गुणनफल जितना होता है, तथा समता ठीक आयताकार
डिब्बों के लिए। यदि $\abs{m_{ij}} \leq 1$ हो तो $\norm{c_i} \leq \sqrt n$, अतः $\abs{\det M} \leq n^{n/2}$। (इसे प्राप्त करने के लिए
प्रविष्टियाँ $\pm1$ वाले लांबिक स्तंभ चाहिए: अर्थात् कोई आदामार आव्यूह।)

\textbf{10.} $X \neq 0$ ($A$ प्रतिलोमनीय) के लिए $X^{\mathsf T}A^{\mathsf T}AX = \norm{AX}^2 > 0$: अतः $A^{\mathsf T}A$ धनात्मक निश्चित
है। उसका वर्गमूल $S$ धनात्मक निश्चित है (\omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\sqrt{\lambda_i} > 0$), इसलिए
प्रतिलोमनीय, और $Q = AS^{-1}$
\[
Q^{\mathsf T}Q = S^{-1}A^{\mathsf T}AS^{-1}
= S^{-1}S^2S^{-1} = I :
\]
पूरा करता है: अर्थात् लांबिक $Q$ तथा धनात्मक निश्चित $S$ सहित $A = QS$।

\textbf{11.} यदि $A = QS = Q'S'$ हो तो $S'^{\,2} =
S'^{\mathsf T}Q'^{\mathsf T}Q'S' = A^{\mathsf T}A = S^2$; और एक ही वर्ग वाले दो धनात्मक अर्धनिश्चित
आव्यूह मेल खाते हैं (\cref{exo:b2:quadratic:6}): अतः $S' = S$, और फिर $Q' = AS^{-1} =
Q$।

\textbf{12.} $\det(A + \varepsilon I)$ $\varepsilon$ में अशून्य बहुपद है: उसके परिमित मूल हैं, अतः कोई
अनुक्रम $\varepsilon_k \to 0$ उनसे बच जाता है। $A + \varepsilon_kI =
Q_kS_k$ लिखिए (प्रश्न 10)। $O(n)$ \omterm{def:b2:metric:compact}{संहत} है (\cref{exo:b2:quadratic:5}):
अतः किसी उपअनुक्रम से $Q_{\varphi(k)}
\to Q \in O(n)$ मिलता है। तब
\[
S_{\varphi(k)} = Q_{\varphi(k)}^{\mathsf T}
\bigl(A + \varepsilon_{\varphi(k)}I\bigr)
\longrightarrow Q^{\mathsf T}A =: S,
\]
जो ऐसों की सीमा होने से \omterm{def:b2:quadratic:adjoint}{सममित} धनात्मक अर्धनिश्चित है (संवृत शर्तें), और
$A = QS$। इसके अतिरिक्त $S^2 = S^{\mathsf T}S =
A^{\mathsf T}QQ^{\mathsf T}A = A^{\mathsf T}A$, अतः अद्वितीयता से $S =
\sqrt{A^{\mathsf T}A}$। अपभ्रष्ट $A$ के लिए $S$
अपभ्रष्ट है और $Q$ अद्वितीय नहीं: उसे $(\operatorname{im} S)^{\perp}$ पर मनमाने ढंग से बदला जा सकता
है --- चरम स्थिति $A = 0$, जहाँ हर लांबिक $Q$ काम कर जाता है।

\textbf{13.} $S = P\Sigma P^{\mathsf T}$ का विकर्णन कीजिए (वर्णक्रमीय प्रमेय), $\Sigma = \operatorname{diag}(\sigma_i)$, जहाँ $\sigma_i \geq 0$ $S =
\sqrt{A^{\mathsf T}A}$
के \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} हैं। तब
\[
A = QS = (QP)\,\Sigma\,P^{\mathsf T} = U\Sigma V^{\mathsf T},
\qquad U = QP,\ V = P \in O(n) .
\]

\textbf{14.} $S$ के किसी शीर्ष \omterm{def:b2:reduction:eigen}{अभिलक्षणिक सदिश} पर समता के साथ $\norm{Ax}^2 = x^{\mathsf T}S^2x \leq
\sigma_{\max}^2\norm x^2$: $\vertiii A_2 = \sigma_{\max}$
--- \omterm{def:b2:quadratic:adjoint}{सममित} $A$ के लिए $S = \sqrt{A^2}$ के \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\abs{\lambda_i}$ हैं, जिससे \cref{exo:b2:quadratic:7} पुनः मिल
जाता है। \omterm{def:b2:linalg:det}{सारणिक}: $\abs{\det A} =
\abs{\det U}\det\Sigma\abs{\det V} = \sigma_1\cdots\sigma_n$। गोलक: $\norm y = 1$ के साथ $x = Vy$ लिखने पर $Ax = U\Sigma y$ के $U$ के
स्तंभों वाले प्रसामान्य लांबिक तंत्र में निर्देशांक $z_i = \sigma_iy_i$ होते हैं: अतः
प्रतिबिंब $\{\sum z_i^2/\sigma_i^2 =
1\}$ है, यानी अर्ध-अक्ष $\sigma_i$ वाला दीर्घवृत्तज।

\textbf{15.} मुख्य अक्षों तक घूर्णन (\cref{cor:b2:quadratic:principalaxes}) $f$ को $\lambda_1X^2 + \lambda_2Y^2 + \beta_1X + \beta_2Y + c$ में बदल देता है,
जहाँ $\lambda_1\lambda_2 = \det A$। यदि $\det A \neq 0$ हो, तो स्थानांतरण $X \mapsto X -
\frac{\beta_1}{2\lambda_1}$ (और वैसे ही $Y$) से रैखिक पद समा
जाते हैं: $\lambda_1X'^2
+ \lambda_2Y'^2 = c'$। $\det A > 0$ (समान चिह्न) के लिए: कोई दीर्घवृत्त (उचित चिह्न वाला
$c'$), कोई बिंदु, अथवा रिक्त। $\det
A < 0$ के लिए: कोई अतिपरवलय ($c' \neq 0$) अथवा दो
प्रतिच्छेदी रेखाएँ। और यदि कोटि $1$ के साथ $\det A = 0$ (मान लीजिए $\lambda_2 = 0 \neq
\lambda_1$): $\lambda_1X'^2 + \beta_2Y + c''$,
जो $\beta_2 \neq 0$ होने पर परवलय है; अन्यथा दो समांतर रेखाएँ, एक रेखा, या रिक्त।
अनपभ्रष्ट आकार: दीर्घवृत्त, अतिपरवलय, परवलय --- और उन पर $\det A$ के चिह्न का
शासन है।

\textbf{16.} द्विघात भाग $x^2 + 4xy + y^2$ का आव्यूह $\begin{pmatrix}1 & 2\\ 2 & 1\end{pmatrix}$ है, जिसके \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $3$
और $-1$ हैं तथा प्रसामान्य लांबिक दिशाएँ $\frac{1}{\sqrt2}(1,1)$, $\frac{1}{\sqrt2}(1,-1)$। घूर्णित निर्देशांकों
$u =
\frac{x+y}{\sqrt2}$ में $v = \frac{x-y}{\sqrt2}$: $x^2 + y^2 = u^2
+ v^2$, $2xy = u^2 - v^2$, अतः रूप $3u^2 - v^2$ है, और $2x
- 2y = 2\sqrt2\,v$। समीकरण बन जाता है
\[
3u^2 - v^2 + 2\sqrt2\,v = 4
\quad\Longleftrightarrow\quad
3u^2 - \bigl(v - \sqrt2\bigr)^2 = 2 :
\]
अर्थात् कोई अतिपरवलय, जिसका केंद्र $(u, v) = (0, \sqrt2)$ पर है, यानी $(x, y) =
(1, -1)$, और अक्ष घूर्णित
तंत्र के अनुदिश।

\textbf{17.} जब भी $Ax_0 = -\frac b2$, अर्थात् $x_0 = -\frac12A^{-1}b$, तब $f(x) = (x - x_0)^{\mathsf T}A(x - x_0) + f(x_0)$: $f$ का प्रवणक ठीक वहीं लुप्त
होता है ($x_0$ सममिति का केंद्र है)। $M = \begin{pmatrix} I & x_0\\ 0 &
1\end{pmatrix}$ के साथ:
\[
M^{\mathsf T}\widetilde QM
= \begin{pmatrix}
A & Ax_0 + \frac b2\\[2pt]
\bigl(Ax_0 + \frac b2\bigr)^{\mathsf T} &
x_0^{\mathsf T}Ax_0 + b^{\mathsf T}x_0 + c
\end{pmatrix}
= \begin{pmatrix} A & 0\\ 0 & f(x_0)\end{pmatrix},
\]
और $\det M = 1$: $\det\widetilde Q = f(x_0)\det A$। केंद्रित समीकरण $q(X) = -f(x_0)$ पढ़ी जाती है: $f(x_0) = 0$ के लिए वह $q(X) = 0$ पर
पतित हो जाती है (चिह्नांक $(1,1)$ होने पर केंद्र से गुज़रती दो रेखाएँ, और
$q$ के निश्चित होने पर अकेला बिंदु $x_0$); और $f(x_0) \neq 0$ के लिए शांकव सच्चा
दीर्घवृत्त या अतिपरवलय है।

\textbf{18.} $A^{-1} = -\frac13\begin{pmatrix} 1 & -2\\ -2 &
1\end{pmatrix}$, $\frac b2 = (1, -1)$: $x_0 =
-A^{-1}\frac b2 = (1, -1)$, जैसा प्रश्न 16 में मिला था। $f(x_0) =
q(1,-1) + 2 + 2 - 4 = (1 - 4 + 1) + 0 = -2 \neq 0$, और $\det\widetilde Q = f(x_0)\det A = (-2)(-3) = 6 \neq 0$:
अर्थात् अनपभ्रष्ट; $\det A = -3 < 0$: अतिपरवलय --- और वास्तव में केंद्रित समीकरण $3u^2 - (v - \sqrt2)^2 = -f(x_0) = 2$
प्रश्न 16 से मेल खाती है।

\textbf{19.} आव्यूह $(1-\lambda)I + \lambda J$ है: \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $1 + 2\lambda$ (दिशा $(1,1,1)$) और $1 -
\lambda$ (दोहरा,
$x + y + z = 0$ पर)। स्थितियाँ:
\begin{itemize}
  \item $-\frac12 < \lambda < 1$: सारे \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} धनात्मक: अर्थात् $(1,1,1)$ के परितः घूर्णज
        दीर्घवृत्तज ($\lambda = 0$ के लिए गोला);
  \item $\lambda = 1$: $q = (x+y+z)^2$: समीकरण दो समांतर समतल $x + y + z = \pm1$ देती है;
  \item $\lambda = -\frac12$: \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $0, \frac32,
        \frac32$: अक्ष $(1,1,1)$ वाला वृत्तीय बेलन;
  \item $\lambda > 1$: चिह्नांक $(1, 2)$: द्वि-पर्ण अतिपरवलयज;
  \item $\lambda < -\frac12$: चिह्नांक $(2, 1)$: एक-पर्ण अतिपरवलयज।
\end{itemize}

\textbf{20.} $q$ का ध्रुवी रूप $E$ पर कोई अंतर्गुणन $\langle\cdot,\cdot\rangle_q$ है। स्थिर $x$ के
लिए $y \mapsto
\varphi'(x, y)$ ($q'$ का ध्रुवी रूप) रैखिक है, अतः किसी अद्वितीय $z_x$ के लिए $\langle z_x, y\rangle_q$ के
बराबर; $u(x) := z_x$ रैखिक है (अद्वितीयता से), और $\langle u(x), y\rangle_q =
\varphi'(x,y) = \varphi'(y,x) = \langle u(y), x\rangle_q$: अर्थात् यूक्लिडीय समष्टि
$(E,
\langle\cdot,\cdot\rangle_q)$ में $u$ \omterm{def:b2:quadratic:adjoint}{सममित} है। वर्णक्रमीय प्रमेय (\cref{thm:b2:quadratic:spectral}) $q$-प्रसामान्य लांबिक
अभिलक्षणिक आधार $(\varepsilon_i)$, $u(\varepsilon_i) =
\mu_i\varepsilon_i$ दे देती है: उसमें $q(x) = \sum x_i^2$ और $q'(x) =
\langle u(x), x\rangle_q = \sum\mu_ix_i^2$।

\textbf{21.} मान लीजिए $P$ उस आधार का आव्यूह है: \omterm{def:b2:quadratic:def}{सर्वांगसमता} $P^{\mathsf T}AP = I$ और $P^{\mathsf T}BP =
\operatorname{diag}(\mu_i)$
देती है। तब
\[
\det(B - \mu A) = \det(P^{-\mathsf T})
\det\bigl(\operatorname{diag}(\mu_i) - \mu I\bigr)
\det(P^{-1})
= (\det P)^{-2}\prod_i(\mu_i - \mu) :
\]
अर्थात् $\mu_i$ कुलक $\det(B - \mu A)$ के मूल हैं।

\textbf{22.} $\det(B - \mu A) =
\det\begin{pmatrix} -2\mu & 1-\mu\\ 1-\mu & -\mu\end{pmatrix}
= 2\mu^2 - (1-\mu)^2 = \mu^2 + 2\mu - 1$: मूल $\mu_\pm = -1
\pm \sqrt2$। $(B - \mu_\pm A)v = 0$ हल करने पर: $v_\pm = (1 -
\mu_\pm,\ 2\mu_\pm)$ (मूलों पर दूसरी पंक्ति की
सर्वसमिका $(1-\mu)^2 =
2\mu^2$ इसकी पुष्टि करती है)। $A$-मानदंड सुथरे निकलते हैं: $q_A(v_\pm) = 2(1 + \mu_\pm^2)$, और
$\mu_+ + \mu_- = -2$, $\mu_+\mu_- = -1$ का उपयोग करके $\varphi_A(v_+, v_-) = 0$ जाँचा जा सकता है। आधार $\Bigl(\frac{v_+}{\sqrt{2(1 +
\mu_+^2)}}, \frac{v_-}{\sqrt{2(1+\mu_-^2)}}\Bigr)$ $A$ के लिए
प्रसामान्य लांबिक है और $B$ का विकर्णन प्रविष्टियों $\mu_\pm$ के साथ करता है।

\textbf{23.} यदि कोई प्रतिलोमनीय $P$ दोनों रूपों का विकर्णन करता, तो
प्रश्न 21 का परिकलन $\det(B - \mu A) = (\det
P)^{-2}\prod(d_{2i} - \mu d_{1i})$ देता, अर्थात् वास्तविक रैखिक गुणनखंडों में बँटा
कोई वास्तविक बहुपद। परंतु यहाँ
\[
\det(B - \mu A) = \det\begin{pmatrix} -\mu & 1\\ 1 &
\mu\end{pmatrix} = -\mu^2 - 1 ,
\]
घात $2$ का है और उसका कोई वास्तविक मूल नहीं: विरोधाभास। (लोरेंत्स
चिह्नांक $A$ बिना किसी वास्तविक अक्ष के $B$-``घूर्णन'' की अनुमति देता है।)

\textbf{24.} धनात्मक निश्चित $A^{1/2} = \sqrt A$ के साथ $A^{-1}B = A^{-1/2}\bigl(A^{-1/2}BA^{-1/2}\bigr)
A^{1/2}$ (\cref{exo:b2:quadratic:6}): अतः $A^{-1}B$ \emph{\omterm{def:b2:quadratic:adjoint}{सममित}}
$A^{-1/2}BA^{-1/2}$ के सदृश है, इसलिए वास्तविक अभिलक्षणिक मानों के साथ \omterm{def:b2:reduction:diag}{विकर्णनीय}। और $\det(B - \mu A) = \det A\cdot
\det(A^{-1}B - \mu I)$:
अर्थात् प्रश्न 21 के कुलक-मूल $\mu_i$ ठीक $A^{-1}B$ के \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} हैं।

\textbf{25.} (क) हर भाग वर्णक्रमीय प्रमेय पर टिका रहा: वर्गमूल के द्वारा
(चोलेस्की, ध्रुवीय), अभिलक्षणिक मानों के परिबंधों के द्वारा (आदामार),
मुख्य अक्षों के द्वारा (शांकव), और $q$-अनुकूलित रूप के द्वारा (युगपत
समानयन)। (ख) ग्राम-साक्षात्कार, आदामार और \omterm{pb:b2:quadratic:1}{ध्रुवीय अपघटन} अर्धनिश्चित
दुनिया में भी बचे रहते हैं; जबकि चोलेस्की की अद्वितीयता, धुरी-अंक वाला
सूत्र और युगपत समानयन को निश्चितता चाहिए (प्रश्न 12 और 23 ठीक-ठीक दिखा
देते हैं कि वे कैसे विफल होते हैं)। (ग) युग्मित छोटे दोलन: गतिज ऊर्जा
(धनात्मक निश्चित) और स्थितिज ऊर्जा दो \omterm{def:b2:quadratic:def}{द्विघात रूप} हैं; प्रश्न 20--22 का
आधार निकाय की \emph{प्रसामान्य विधाएँ} है, और वहाँ $\mu_i$ कोणीय आवृत्तियों
के वर्ग हैं।
\end{solution}
