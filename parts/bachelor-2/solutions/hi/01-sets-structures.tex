\section*{अध्याय \ref{ch:b2:structures} --- समुच्चय और संरचनाएँ}

\begin{solution}{exo:b2:structures:1}
\emph{$\N$ के परिमित उपसमुच्चय:} \omterm{def:b2:structures:countable}{गणनीय} --- $\intint{0}{n}$ के उपसमुच्चयों का
समुच्चय परिमित है, और परिमित उपसमुच्चय इन्हीं का $n$ पर \omterm{def:b2:structures:countable}{गणनीय} सम्मिलन
बनाते हैं (\cref{prop:b2:structures:countablestable}~(3)); अनंत इसलिए कि उसमें सभी एकल समुच्चय हैं।

\emph{$\N$ के सभी उपसमुच्चय:} \omterm{def:b2:structures:countable}{गणनीय} नहीं, कैंटर की प्रमेय से (\cref{thm:b2:structures:cantor}~(1),
$E = \N$ के साथ)।

\emph{$\R \setminus \Q$:} \omterm{def:b2:structures:countable}{गणनीय} नहीं --- अन्यथा $\R = \Q \cup
(\R\setminus\Q)$ दो \omterm{def:b2:structures:countable}{गणनीय} समुच्चयों का सम्मिलन
होता, जो \cref{thm:b2:structures:cantor}~(2) के विरुद्ध है।

\emph{$\Q$ पर बहुपद:} \omterm{def:b2:structures:countable}{गणनीय} --- घात $\leq n$ के बहुपद $\Q^{n+1}$ में एकैकी रूप से
जाते हैं (\omterm{def:b2:structures:countable}{गणनीय} समुच्चयों के परिमित गुणन), और फिर $n$ पर सम्मिलन ले
लीजिए।

\emph{अंततः शून्य होने वाले द्विआधारी अनुक्रम:} \omterm{def:b2:structures:countable}{गणनीय} --- वे $\N$ के
परिमित उपसमुच्चयों (अपने वाहक) के साथ एकैकी आच्छादक संगति में हैं।
\end{solution}

\begin{solution}{exo:b2:structures:2}
अवयव-दर-अवयव परिकलन कीजिए, पहले दाहिने गुणनखंड को लगाते हुए। $\sigma\tau$ $1 \mapsto \sigma(1) = 4$,
$\;2 \mapsto \sigma(3)
= 5$, $\;3 \mapsto \sigma(7) = 7$, $\;4 \mapsto \sigma(4) = 2$, $\;5
\mapsto \sigma(5) = 3$, $\;6 \mapsto \sigma(6) = 1$, $\;7 \mapsto
\sigma(2) = 6$ भेजता है:
\[
\sigma\tau = (1\,4\,2\,5\,3\,7\,6),
\]
अर्थात् एक $7$-चक्र। इसी प्रकार $\tau\sigma$ $1 \mapsto \tau(4) = 4$, $\;2 \mapsto \tau(6) = 6$, $\;3 \mapsto \tau(5) = 5$, $\;4 \mapsto
\tau(2) = 3$, $\;5 \mapsto \tau(3) = 7$,
$\;6 \mapsto \tau(1) = 1$, $\;7 \mapsto \tau(7) = 2$ भेजता है:
\[
\tau\sigma = (1\,4\,3\,5\,7\,2\,6),
\]
यह भी $7$-चक्र है (जैसी अपेक्षा थी: $\sigma\tau$ और $\tau\sigma$ संयुग्मी हैं, अतः
उनका चक्र-प्रकार एक ही है)।

कोटियाँ और चिह्न: $\sigma$ का चक्र-प्रकार $(4,2)$ है: \omterm{def:b2:structures:generated}{कोटि} $\operatorname{lcm}(4,2) = 4$, चिह्न $(-1)^3(-1)^1 = +1$; $\tau$
एक $3$-चक्र है: \omterm{def:b2:structures:generated}{कोटि} $3$, चिह्न $+1$; दोनों गुणनफल $7$-चक्र हैं: \omterm{def:b2:structures:generated}{कोटि}
$7$, चिह्न $(-1)^6 = +1$।

$\sigma^{2026}$: $2026 = 4 \times 506 + 2$, अतः $\sigma^{2026} =
\sigma^2 = (1\,2)(4\,6)$ ($4$-चक्र का वर्ग लीजिए; \omterm{def:b2:structures:sn}{पार्यय} वर्ग होकर लुप्त हो
जाता है)।
\end{solution}

\begin{solution}{exo:b2:structures:3}
$\{0,1\}^{\N} \to \mathcal{P}(\N)$: कोई अनुक्रम अपने वाहक पर जाता है --- यह एकैकी आच्छादक है (सूचक
फलन), किसी प्रमेय की आवश्यकता नहीं।

$\{0,1\}^{\N} \to \intcc{0}{1}$: आधार $3$ वाला प्रतिचित्रण $(a_n) \mapsto
\sum 2a_n 3^{-n-1}$ एकैकी है (दो भिन्न अनुक्रम पहली बार
\omterm{def:b2:structures:generated}{कोटि} $N$ पर भिन्न होते हैं; पुच्छ $2\cdot
3^{-N-1}$ जितने अंतर की भरपाई नहीं कर
सकते, क्योंकि $\sum_{n > N} 2\cdot 3^{-n-1} = 3^{-N-1} <
2\cdot3^{-N-1}$)।

$\intcc{0}{1} \to \{0,1\}^{\N}$: द्विआधारी प्रसार लीजिए, और (मान लीजिए) वह प्रसार चुनिए जो अंत तक
सब $1$ न हो: यह एकैकी है।

अंतिम दो एकैकी प्रतिचित्रणों पर कैंटर--बर्नस्टाइन (\cref{thm:b2:structures:cantorbernstein}) लगाने से $\intcc{0}{1}$
और $\{0,1\}^{\N}$ \omterm{def:b2:structures:countable}{समशक्त} हो जाते हैं, अतः तीनों समुच्चय \omterm{def:b2:structures:countable}{समशक्त} हैं।
\end{solution}

\begin{solution}{exo:b2:structures:4}
मान लीजिए $c = ab = ba$ और $d = \operatorname{ord}(c)$। पहले $c^{mn} =
a^{mn} b^{mn} = e$ (क्रमविनिमेयता घात को बाँटने देती है),
अतः $d
\mid mn$। विलोमतः $c^d = e$ से $a^d = b^{-d}$ मिलता है; यह अवयव $\langle a\rangle \cap \langle b\rangle$ में पड़ता है, जिसकी
\omterm{def:b2:structures:generated}{कोटि} $m$ और $n$ दोनों को विभाजित करती है (प्रत्येक \omterm{def:b2:structures:generated}{चक्रीय समूह} में
लाग्रांज), अतः वह तुच्छ है: $a^d = b^d = e$, जिससे $m \mid d$ और $n \mid d$; और सहअभाज्यता से
$mn \mid d$। अतः $d = mn$।

$\mathfrak{S}_3$ में: लीजिए $a = (1\,2)$ (\omterm{def:b2:structures:generated}{कोटि} $2$) और $b =
(1\,2\,3)$ (\omterm{def:b2:structures:generated}{कोटि} $3$), जिनकी कोटियाँ
सहअभाज्य हैं परंतु जो क्रमविनिमेय नहीं हैं: $ab =
(2\,3)$ की \omterm{def:b2:structures:generated}{कोटि} $2 \neq 6$ है ---
वास्तव में $\mathfrak{S}_3$ में \omterm{def:b2:structures:generated}{कोटि} $6$ का कोई अवयव है ही नहीं। क्रमविनिमेयता
अनिवार्य है।
\end{solution}

\begin{solution}{exo:b2:structures:5}
प्रत्येक $x \in G$ को $x^{-1}$ के साथ युग्मित कीजिए। $x \neq x^{-1}$ वाले युग्म $\{x, x^{-1}\}$ दो-दो
अवयवों के हैं और अपने सम्मिलन का विभाजन करते हैं; शेष अवयव ठीक वे हैं
जिनके लिए $x = x^{-1}$, अर्थात् $x^2 =
e$। चूँकि $\abs G$ सम है और दो-अवयवी युग्म सम
संख्या में अवयव ढकते हैं, समुच्चय $\{x : x^2 = e\}$ की गणनसंख्या सम है; उसमें $e$
है, अतः उसमें कम से कम एक और अवयव $x \neq
e$ भी है --- अर्थात् \omterm{def:b2:structures:generated}{कोटि} $2$ का
अवयव।
\end{solution}

\begin{solution}{exo:b2:structures:6}
$A_n$ का प्रत्येक अवयव सम संख्या में पार्ययों का गुणनफल है (\cref{thm:b2:structures:signature}:
पार्ययों में अपघटित कीजिए; चिह्न $+1$ होने से गिनती सम रहती है)। अतः
इतना ही पर्याप्त है कि दो पार्ययों के प्रत्येक गुणनफल को $3$-चक्रों से
लिखा जाए:
\[
(a\,b)(a\,c) = (a\,c\,b),
\qquad
(a\,b)(c\,d) = (a\,c\,b)(a\,c\,d) \quad (\text{भिन्न } a,b,c,d),
\]
(मान रखकर जाँच लीजिए), और $(a\,b)(a\,b) = \mathrm{id}$। अतः $3$-चक्र $A_n$ को \omterm{def:b2:structures:generated}{जनित} करते हैं।
\end{solution}

\begin{solution}{exo:b2:structures:7}
\emph{$(\Q,+) \to (\Z,+)$:} केवल शून्य समाकारिता। किसी भी $x$ तथा हर $n \geq 1$ के लिए $f(x) = n f\bigl(\frac xn\bigr)$
$\Z$ में $n$ से विभाज्य है; और हर $n$ से विभाज्य एकमात्र पूर्णांक $0$
है, अतः सभी $x$ के लिए $f(x) = 0$।

\emph{$(\Z/n\Z, +) \to (\Z/m\Z, +)$:} समाकारिता $c = f(\overline 1)$ से निर्धारित होती है, जिसे $n c \equiv 0 \pmod m$ पूरा करना
चाहिए, अर्थात् $c$ $\frac{m}{\gcd(m,n)}$ का गुणज हो; ऐसे वर्ग $\gcd(m,n)$ हैं, और हर चुनाव
वास्तव में समाकारिता देता भी है (सार्वत्रिक गुणधर्म से $k \mapsto kc$ को $\Z/n\Z$ से
होकर गुणनखंडित कीजिए)।

\emph{$(\Q, +) \to (\Q_+^*, \times)$:} केवल तुच्छ समाकारिता। यदि $f(x) = y$, तो हर $n$ के लिए $y = f(n \cdot \frac xn) =
f(\frac xn)^n$ $\Q_+^*$
में $n$-वीं घात है। परंतु कोई परिमेय $y \neq 1$ सभी $n$ के लिए $n$-वीं घात
नहीं हो सकता: $y$ में कोई अभाज्य अशून्य घातांक $v$ के साथ आता है, और
$n > \abs
v$ के लिए $n \nmid v$ ($n$-वीं घातों के घातांक अनन्य गुणनखंडन से $n$-के
गुणज होते हैं, अर्थात् $n$ के गुणज)। अतः $f \equiv 1$।
\end{solution}

\begin{solution}{exo:b2:structures:8}
$360 = 2^3 \cdot 3^2 \cdot 5$:
$\varphi(360) = 360\bigl(1 - \tfrac12\bigr)\bigl(1 -
\tfrac13\bigr)\bigl(1 - \tfrac15\bigr) = 360 \cdot \tfrac12 \cdot
\tfrac23 \cdot \tfrac45 = 96$.

निकाय: मापांक $8, 9, 5$ परस्पर सहअभाज्य हैं, कुल $360$। $x
\equiv 3 \pmod 8$ और $x \equiv 5 \pmod 9$ से: $3 +
8k \equiv 5 \pmod 9$
सहित $x = 3 + 8k$, अर्थात् $-k \equiv 2$, $k \equiv -2 \equiv 7
\pmod 9$: $x \equiv 3 + 56 = 59 \pmod{72}$। फिर $59 + 72\ell \equiv
2 \pmod 5$: $4 + 2\ell \equiv 2$, $2\ell \equiv 3 \equiv 8$, $\ell
\equiv 4 \pmod 5$: $x \equiv 59 + 288 = 347 \pmod{360}$।

$3^{2026}$ के अंतिम दो अंक: $4$ के सापेक्ष, $3^{2026} = 9^{1013} \equiv
1$। $25$ के सापेक्ष: $\varphi(25) = 20$ और $2026 = 20\cdot101 + 6$,
अतः $3^{2026} \equiv 3^6 = 729 \equiv 4 \pmod{25}$। $x \equiv 1
\pmod 4$, $x \equiv 4 \pmod{25}$ हल कीजिए: $x = 4 + 25k \equiv 1 \pmod 4$ से $k \equiv 1 \pmod 4$ मिलता है: $x \equiv 29 \pmod{100}$। अंतिम दो अंक
$29$ हैं।
\end{solution}

\begin{solution}{exo:b2:structures:9}
मान लीजिए $A$ एक परिमित पूर्णांकीय प्रांत है और $a \in A$, $a \neq 0$। प्रतिचित्रण
$x \mapsto ax$ एकैकी है ($ax = ay \implies a(x - y) = 0
\implies x = y$, क्योंकि शून्य भाजक नहीं हैं); और परिमित समुच्चय का
स्वयं में एकैकी प्रतिचित्रण आच्छादक होता है (प्रथम वर्ष का खंड,
कबूतरखाना तुल्यता)। अतः किसी $b$ के लिए $1 = ab$: हर अशून्य अवयव
प्रतिलोमनीय है, अर्थात् $A$ एक क्षेत्र है।

$\Z/n\Z$: यदि $n$ अभाज्य है तो वह पूर्णांकीय प्रांत है ($n \mid ab
\implies n \mid a$ या $n \mid b$,
यूक्लिड की प्रमेयिका), परिमित है, अतः क्षेत्र है; और यदि $n = rs$ भाज्य है
तो $\overline r\,\overline s =
\overline 0$ शून्य भाजक दिखा देता है।
\end{solution}

\begin{solution}{exo:b2:structures:10}
मान लीजिए $m = \max\{\operatorname{ord}(x) : x \in G\}$, जो $a$ पर प्राप्त होता है।

\emph{दावा: प्रत्येक $x \in G$ की \omterm{def:b2:structures:generated}{कोटि} $m$ को विभाजित करती है।} मान लीजिए
किसी $x$ की \omterm{def:b2:structures:generated}{कोटि} $q$ ऐसी है कि $q \nmid m$: तब कोई अभाज्य घात $p^k$ $q$ को
विभाजित करती है पर $m$ को नहीं। $p \nmid m'$ और $j
< k$ के साथ $m = p^j m'$ लिखिए। अवयव
$a^{p^j}$ की \omterm{def:b2:structures:generated}{कोटि} $m'$ है; अवयव $x^{q/p^k}$ की \omterm{def:b2:structures:generated}{कोटि} $p^k$ है; ये कोटियाँ सहअभाज्य
हैं और दोनों अवयव क्रमविनिमेय हैं ($G \subseteq K^*$ अबेली है), अतः \cref{exo:b2:structures:4} से उनके
गुणनफल की \omterm{def:b2:structures:generated}{कोटि} $p^k m' > p^j m'
= m$ हुई: यह महत्तमता के विरुद्ध है।

अतः सभी $x \in G$ $x^m = 1$ को संतुष्ट करते हैं: बहुपद $X^m - 1$ के क्षेत्र $K$ में कम
से कम $\abs G$ मूल हैं, जिससे $\abs G \leq m$ (घात $m$ के अशून्य बहुपद के अधिकतम $m$
मूल होते हैं, प्रथम वर्ष का खंड)। परंतु लाग्रांज से $m = \operatorname{ord}(a) \leq \abs G$। अतः $m = \abs G$, और
गणनसंख्या $m =
\abs G$ वाला $\langle a \rangle$ पूरा $G$ ही है: अर्थात् \omterm{def:b2:structures:generated}{चक्रीय}।

$K = \Z/p\Z$ के लिए: $(\Z/p\Z)^*$ $K^*$ का परिमित उपसमूह है, अतः \omterm{def:b2:structures:generated}{चक्रीय} है (\omterm{def:b2:structures:generated}{कोटि} $p - 1$)।
\end{solution}

\begin{solution}{exo:b2:structures:11}
\emph{\omterm{def:b2:structures:generated}{चक्रीय} नहीं:} उपसमूह $\langle \frac pq\rangle$ $\frac pq$ के पूर्णांक गुणजों से बना है, और
(लघुतम रूप में) उन सबका हर $q$ को विभाजित करता है; अतः उसमें $\frac{1}{2q}$ नहीं
आता। कोई एक जनक $\Q$ के असीमित हरों तक नहीं पहुँच सकता।

\emph{परिमिततः \omterm{def:b2:structures:generated}{जनित} भी नहीं:} $\frac{p_1}{q_1}, \dots, \frac{p_k}{q_k}$ से \omterm{def:b2:structures:generated}{जनित} उपसमूह उन परिमेय संख्याओं
से बना है जिनके हर $Q = q_1 \cdots q_k$ को विभाजित करते हैं (पूर्णांक संचयों का हर $Q$
को विभाजित करता है): उसमें $\frac{1}{2Q}$ नहीं आता।

\emph{परिमिततः \omterm{def:b2:structures:generated}{जनित} उपसमूह \omterm{def:b2:structures:generated}{चक्रीय} होते हैं:} उपर्युक्त $Q$ के साथ,
उपसमूह $H = \langle \frac{p_1}{q_1}, \dots,
\frac{p_k}{q_k}\rangle$ $\frac{1}{Q}\Z$ में समाहित है। प्रतिचित्रण $x
\mapsto Qx$ $\frac1Q\Z$ से $\Z$ पर
तुल्याकारिता है, जो $H$ को $\Z$ के किसी उपसमूह पर ले जाती है, और वह
किसी $n$ के लिए $n\Z$ है (प्रथम वर्ष का खंड): अतः $H = \frac{n}{Q}\Z$ \omterm{def:b2:structures:generated}{चक्रीय} है, जो
$\frac nQ$ से \omterm{def:b2:structures:generated}{जनित} है।
\end{solution}

\begin{solution}{exo:b2:structures:12}
\emph{एक \omterm{def:b2:structures:countable}{गणनीय} उपसमुच्चय।} मान लीजिए $E$ अनंत है। $a_0,
a_1, a_2, \dots$ की रचना आगमन से
कीजिए: $E$ रिक्त नहीं है, अतः $a_0 \in E$ चुनिए; और यदि $a_0, \dots, a_n$ चुने जा चुके हैं
तो $E \setminus \{a_0, \dots,
a_n\}$ रिक्त नहीं है ($E$ परिमित नहीं है), वहाँ से $a_{n+1}$ चुनिए। रचना से
ही $a_n$ परस्पर भिन्न हैं, अतः $A = \{a_n : n
\in \N\}$ $E$ का \omterm{def:b2:structures:countable}{गणनीय} उपसमुच्चय है।

\emph{अनंत $\implies$ अपने उचित उपसमुच्चय के \omterm{def:b2:structures:countable}{समशक्त}।} $f \colon E \to E \setminus \{a_0\}$ को इस प्रकार
परिभाषित कीजिए: $f(a_n) = a_{n+1}$, और $x \notin A$ के लिए $f(x) = x$। यह एकैकी है (दोनों टुकड़े
एकैकी हैं और उनके प्रतिबिंब असंयुक्त), तथा $E \setminus
\{a_0\}$ पर आच्छादक भी: हर $a_{n+1}$
तक पहुँचा जाता है, हर $x \notin A$ तक भी। अतः $E$ उचित उपसमुच्चय $E \setminus \{a_0\}$ के \omterm{def:b2:structures:countable}{समशक्त}
है।

\emph{विलोम।} यदि $E$ परिमित है और $g \colon E \to F$ $F \neq E$ सहित $F \subseteq E$ पर एकैकी आच्छादक
प्रतिचित्रण है, तो $g$ $E$ का स्वयं में ऐसा एकैकी प्रतिचित्रण है जो
आच्छादक नहीं, और यह कबूतरखाना सिद्धांत के विरुद्ध है (प्रथम वर्ष का
खंड: परिमित समुच्चय का स्वयं में एकैकी प्रतिचित्रण एकैकी आच्छादक होता
है)। अतः अपने उचित उपसमुच्चय के \omterm{def:b2:structures:countable}{समशक्त} समुच्चय अनंत होता है।
\end{solution}

\textbf{2.} सरकने के बाद खाने $p$ में $c$ की पुरानी वस्तु आ जाती है
और खाने $c$ में रिक्त; शेष सभी खाने अछूते रहते हैं: $\sigma'(p) = \sigma(c)$, $\sigma'(c) = \sigma(p) = 16$, और
अन्यत्र $\sigma' = \sigma$। यह ठीक $\sigma' = \sigma

\textbf{3.} सटे हुए खाने ठीक एक ही निर्देशांक में एक पग से भिन्न होते

$\chi(\text{रिक्त का खाना})$ पलट जाता है। रिक्त के किसी बंद पथ पर $\chi$
हर चाल के साथ एक बार पलटता है और अपने आरंभिक मान पर लौट आता है: अतः
चालों की संख्या सम है।

\textbf{4.} प्रश्न 2 और 3 से, एक चाल $I(\sigma) = \varepsilon(\sigma)\chi(\sigma^{-1}(16))$ के दोनों गुणनखंड पलट देती

\textbf{6.} कोई खाना $p$ स्थिर कीजिए तथा $p$ से भिन्न दो अन्य खाने

\textbf{8.} यदि $\sigma(16) = 16$, तो $\sigma$ खानों $1, \dots, 15$ का क्रमचय करता है; इस प्रतिबंधन

\textbf{14.} प्रतिलोम लेने की छूट के साथ मान लीजिए $s = (a\ b\ c)$ और $t =
(b\ c\ d)$ ($\{a,b,c\}$ पर

\[
st \colon a \mapsto b,\quad b \mapsto a,\quad c \mapsto d,\quad
d \mapsto c, \qquad\text{अर्थात्}\quad st = (a\ b)(c\ d),

अर्थात् दोहरा \omterm{def:b2:structures:sn}{पार्यय}। उपसमूह $G = \langle s, t\rangle$ चार अक्षरों के सम क्रमचयों से बना

भी, अतः $6 \mid \abs G$ (लाग्रांज, \cref{thm:b2:structures:lagrange}, दोनों \omterm{def:b2:structures:generated}{चक्रीय} उपसमूहों पर लगाकर)। यदि
$A_4$ में \omterm{def:b2:structures:generated}{कोटि} $6$ का कोई उपसमूह $K$ होता, तो उसका सूचकांक $2$

के लिए सहसमुच्चय केवल $K$ तथा $gK$ हैं, अतः सहसमुच्चय $g^2K$ या तो
$K$ है या $gK$, और $g^2K =
gK$ से $g \in K$ आवश्यक हो जाता। अतः हर वर्ग $K$ में
पड़ता। परंतु हर $3$-चक्र $\gamma$ एक वर्ग है, $\gamma = (\gamma^2)^2$, और $A_4$ में आठ
$3$-चक्र हैं: $8 > 6$, जो विरोधाभास है। अतः $\abs G = 12$: $G = A_4$।

\textbf{15.} $u \mapsto a$, $v \mapsto b$ को $X$ के एकैकी आच्छादक प्रतिचित्रण $g_0$ तक

भेजता है। $X$ के बाहर तत्समक से बढ़ाइए: इस प्रकार एक सम क्रमचय $g \in

जिसमें $g(w) = w$ का उपयोग हुआ।

\textbf{16.} $X \cup \{w\}$ का प्रत्येक $3$-चक्र $G$ में पड़ता है: जिनका वाहक

$(b\ a\ w)$, और दोनों प्रश्न 15 से मिल जाते हैं। \cref{exo:b2:structures:6} से $X \cup \{w\}$ के $(k+1)$-अवयवी
समुच्चय के $3$-चक्र उसका एकांतर समूह \omterm{def:b2:structures:generated}{जनित} करते हैं, अतः $G$ में $X \cup \{w\}$
का हर सम क्रमचय है। \emph{शृंखलाबद्ध करना:} मान लीजिए $G = \langle s_0, \dots,
s_{12}\rangle$। $s_0 = (x_0\ x_1\ x_2)$ और
$s_1 = (x_1\ x_2\ x_3)$ पर प्रमेयिका A लगाने से (उनके वाहक $\{x_1, x_2\}$ साझा करते हैं) $X_4 = \{x_0, x_1, x_2, x_3\}$ के
सभी सम क्रमचय मिल जाते हैं। यदि $G$ में $X_m = \{x_0, \dots,
x_{m-1}\}$ ($4 \leq m \leq 14$) के सभी सम क्रमचय
हैं, तो $s_{m-2} = (x_{m-2}\
x_{m-1}\ x_m)$ में $u = x_{m-2}, v = x_{m-1} \in X_m$ तथा नया अक्षर $w = x_m$ है: प्रमेयिका B और पहला भाग
मिलकर $X_{m+1}$ के सभी सम क्रमचय दे देते हैं। $m = 14$ तक आगमन: $G \supseteq
A_{15}$ (सभी
पंद्रह खानों के सम क्रमचय), और $G
\subseteq A_{15}$ क्योंकि हर $s_t$ सम है: अतः $\langle s_0, \dots,
s_{12}\rangle = A_{15}$।

\textbf{17.} प्रश्न 13 और 16: $A_{15} = \langle s_0, \dots,
s_{12}\rangle \subseteq H$; प्रश्न 9: $H \subseteq A_{15}$। अतः \omterm{def:b2:structures:generated}{कोटि} $15!/2 = 653\,837\,184\,000$ वाला
$H = A_{15}$: पंद्रह गोटियों का हर सम पुनर्विन्यास किसी कार्यक्रम का प्रभाव है।

\textbf{18.} प्रश्न 8 ने $R = \{I = +1\}$ को इस पर घटा दिया था कि हर सम $\rho \in \mathfrak{S}_{15}$ किसी
कार्यक्रम से साकार हो: यह प्रश्न 17 से हो चुका। प्रश्न 6 के साथ
मिलाकर, $\abs R = 16!/2 =
10\,461\,394\,944\,000$। \emph{दो वर्ग:} $t_0 = (14\
15)$ को \emph{वस्तुओं} पर क्रिया करने
दीजिए: $\varphi(\sigma) = t_0 \circ
\sigma$। $\sigma$ से चली वैध चाल $\varphi(\sigma)$ से भी वैध चाल है (रिक्त का खाना
अपरिवर्तित रहता है: $(t_0\sigma)^{-1}(16) = \sigma^{-1}(t_0(16)) = \sigma^{-1}(16)$, और सरकाया गया खाना वही रहता है), तथा $\varphi(\sigma \circ \tau)
= \varphi(\sigma) \circ \tau$:
अर्थात् $\varphi$ चाल-शृंखलाओं को चाल-शृंखलाओं पर एकैकी आच्छादक रूप से भेजता
है (वह अंतर्वलन है)। वह $I$ पलट देता है: $\varepsilon(t_0\sigma) = -\varepsilon(\sigma)$, और रिक्त का खाना वही
रहता है। अतः $\varphi$ $\mathrm{id}$ के वर्ग $R = \{I = +1\}$ को $\varphi(\mathrm{id})
= \sigma_L$ के वर्ग पर एकैकी आच्छादक
रूप से भेजता है, जो इसलिए पूरा $\{I = -1\}$ है: ठीक दो वर्ग। यही
जॉनसन--स्टोरी प्रमेय है।

\textbf{19.} खानों को पठन-क्रम में अंकित कीजिए और मान लीजिए $k = 4(i -
1) + j$
रिक्त का खाना है। $\sigma$ के व्युत्क्रम गिनिए (ऐसे खाना-युग्म $x < y$ जिनके
लिए $\sigma(x) > \sigma(y)$): दो गोटी-खानों के युग्म $N$ देते हैं; रिक्त वाले युग्मों में:
रिक्त के बाद के सभी खानों में गोटियाँ $< 16$ हैं, और हर एक व्युत्क्रमित
है ($16 - k$ युग्म), जबकि उससे पहले के खाने कभी व्युत्क्रमित नहीं होते।
अतः $\varepsilon(\sigma) = (-1)^{N + 16 - k} = (-1)^{N + k}$। चूँकि $k = 4(i-1) + j \equiv j \pmod 2$,
\[
I(\sigma) = (-1)^{N + j}\,(-1)^{i + j} = (-1)^{N + i}
= (-1)^{N + r + 1}
\]
जिसमें $i = 5 - r$ का उपयोग हुआ। प्रश्न 18 से $\sigma$ हल्य है यदि और केवल यदि
$I(\sigma) = +1$, अर्थात् यदि और केवल यदि $N + r$ विषम है। जाँचिए: हल किया हुआ, $N = 0$,
$r
= 1$: विषम, अतः हल्य; लॉयड, $N = 1$, $r = 1$: सम, अतः अहल्य।

\textbf{20.} \emph{क्रिया:} $e \cdot \sigma = \sigma \circ
\mathrm{id} = \sigma$ और $g \cdot (h \cdot \sigma) = \sigma
\circ h^{-1} \circ g^{-1} = \sigma \circ (gh)^{-1} = (gh) \cdot
\sigma$; और $\sigma \circ h^{-1}$ फिर से रिक्त-घर
विन्यास है ($h$ खाने $16$ को अचर छोड़ता है)। \emph{मुक्त:} $\sigma \circ
h^{-1} = \sigma$ से $h^{-1} = \mathrm{id}$
मिलता है ($\sigma^{-1}$ से संयोजन कीजिए)। \emph{कक्षाएँ = कार्यक्रम वर्ग:}
प्रश्न 9 कहता है कि $\sigma$ से कार्यक्रमों द्वारा प्राप्य विन्यास ठीक
$\pi \in H$ वाले $\sigma \circ \pi^{-1}$ हैं: अर्थात् कक्षा $H
\cdot \sigma$। \emph{गणना:} मुक्तता से $h \mapsto h \cdot
\sigma$
एकैकी हो जाता है, अतः हर कक्षा में $\abs H = 15!/2$ अवयव हैं; इसलिए $15!$
रिक्त-घर विन्यास $15!\,/\,(15!/2) = 2$ कक्षाओं में बँटते हैं --- अर्थात् दो
जॉनसन--स्टोरी वर्गों की रिक्त-घर छाया।

\textbf{21.} शतरंजी रंगाई के लिए $3 \times 3$ जालक द्विभाजित है: पथ का हर पग

\textbf{22.} \emph{परिमाप-भ्रमण} $9 \to 8 \to 7 \to 4 \to 1
\to 2 \to 3 \to 6 \to 9$ (सभी पग सटे हुए; लंबाई $8$, जो
सम है): $c_1 = 8, c_2 = 7, c_3 =
4, c_4 = 1, c_5 = 2, c_6 = 3, c_7 = 6$ के साथ प्रश्न 11 वाले लेखे से प्रभाव
\[
\zeta' = (8\ 6\ 3\ 2\ 1\ 4\ 7),
\]
है, अर्थात् केंद्र $5$ को अचर छोड़ने वाला $7$-चक्र ($7$ की वस्तु $8$
पर जाती है, $4$ की $7$ पर, $1$ की $4$ पर, $2$ की $1$ पर, $3$ की
$2$ पर, $6$ की $3$ पर, और $8$ की $6$ पर)। \emph{कोने-भ्रमण}
$9 \to 6 \to
5 \to 8 \to 9$: प्रभाव $(6\ 8\ 5)$ ($5$ की वस्तु $6$ पर जाती है, $8$ की $5$ पर,

संयुग्मन (प्रश्न 12):
\[
\zeta'^{\,t}\,(6\ 8\ 5)\,\zeta'^{-t}
= (y_{t+1}\ y_t\ 5) =: T_t \in H_{3\times3},

क्योंकि $\zeta'$ $5$ को अचर छोड़ता है। $T_0 = (y_1\ y_0\ 5)$ और $T_1 = (y_2\ y_1\ 5)$ के वाहक ठीक $\{y_1, 5\}$
साझा करते हैं: अतः प्रमेयिका A $\{y_0, y_1, y_2, 5\}$ के सभी सम क्रमचय दे देती है। फिर
प्रमेयिका B से $T_2 = (y_3\ y_2\ 5)$ $y_3$ को जोड़ लेता है (उसके अक्षर $y_2, 5$ वर्तमान

समूह सम क्रमचयों से ही बना है (प्रश्न 9 का तर्क पट पर निर्भर नहीं
करता)। अतः $H_{3\times3} = A_8$; और प्रश्न 6, 8, 18 का तर्क --- जो पट पर निर्भर नहीं
करता --- दिखाता है कि प्राप्य विन्यास ठीक वे हैं जिनके लिए $I = +1$:
अर्थात् $9!$ का आधा, यानी $181\,440$।

\textbf{23.} \omterm{def:b2:structures:sn}{चक्र} के चारों ओर खानों को $0, \dots, n-1$ अंकित कीजिए। कोई चाल रिक्त
को उसके दो पड़ोसियों में से एक के साथ बदल देती है। रिक्त के ठीक बाद से
आरंभ करके गोटियों को \omterm{def:b2:structures:generated}{चक्रीय} क्रम में पढ़िए: इससे $n - 1$ गोटियों को
सूचीबद्ध करने वाला एक शब्द $w$ मिलता है। रिक्त को एक पग आगे बढ़ाने से
$(p, w)$ के स्थान पर $(p + 1, \rho w)$ आ जाता है, जहाँ $p$ रिक्त का खाना है और $\rho$ शब्द
को एक स्थान \omterm{def:b2:structures:generated}{चक्रीय} रूप से घुमाता है; पीछे की चाल इसका प्रतिलोम है।
अतः गोटियों का \emph{\omterm{def:b2:structures:generated}{चक्रीय}} क्रम (घूर्णन को छोड़कर शब्द) अपरिवर्त्य
है। $(p, w)$ का प्राप्य वर्ग प्रतिचित्रण $g \colon (p, w) \mapsto (p+1,
\rho w)$ की कक्षा है, जो दोनों \omterm{def:b2:structures:generated}{चक्रीय}
समूहों के गुणन (अर्थात् $\Z/n\Z$ के स्थानांतरण और $n-1$ शब्द-स्थानों के
घूर्णन) में \omterm{def:b2:structures:generated}{कोटि} $\operatorname{lcm}(n, n-1) =
n(n-1)$ का अवयव है, और लघुत्तम समापवर्त्य $n(n-1)$ इसलिए है
कि $\gcd(n, n-1) = 1$: अतः प्रत्येक वर्ग में ठीक $n(n-1)$ विन्यास हैं, और सबकी मालाएँ
एक ही हैं। वर्ग: $n!\,/\,\bigl(n(n-1)\bigr) = (n-2)!$। $n \geq 5$ के लिए $(n-2)! > 2$: सम-विषमता वाला अपरिवर्त्य
(जो अधिक से अधिक दो वर्ग देता है) यहाँ लगभग पूरी बाधा के प्रति अंधा
है; और $4
\times 4$ पट की समृद्धि --- जहाँ सम-विषमता \emph{एकमात्र} बाधा है
--- सच्चा ज्यामितीय तथ्य है, औपचारिक नहीं।

\textbf{24.} दोनों पटों में गोटियाँ पूरी तरह उलटे क्रम में हैं, अतः
दोनों स्थितियों में $N = \binom{15}{2} = 105$ (गोटियों का हर युग्म व्युत्क्रमित है)।
\emph{रिक्त घर में:} $r = 1$, $N + r = 106$ सम: अतः अहल्य। \emph{रिक्त खाने $1$
में:} रिक्त ऊपरी पंक्ति में है, $r = 4$, $N + r = 109$ विषम: अतः हल्य। जो दो पट
केवल छेद की जगह में भिन्न हैं, वे दीवार के आमने-सामने पड़ जाते हैं।
