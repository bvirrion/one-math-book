\section*{अध्याय \ref{ch:b2:proba} --- गणनीय समष्टियों पर प्रायिकता}

\begin{solution}{exo:b2:proba:1}
सममिति से: खींचने का क्रम गेंदों $1$ तथा $2$ पर कोई एकसमान
यादृच्छिक सापेक्ष क्रम प्रेरित करता है, अतः
$\P(1 \text{ पहले आती है, फिर } 2) =
\frac12$। औपचारिक रूप से: किसी
खींच-अनुक्रम में गेंदों $1$ तथा $2$ की स्थितियाँ बदल देना
(समसंभावी) परिणामों का कोई एकैकी आच्छादन है जो \omterm{def:b2:proba:space}{घटना} को उसके पूरक
से अदल-बदल देता है। गेंदों $1, \dots, k$ में: इन $k$ गेंदों का सापेक्ष क्रम
$k!$ क्रमों में एकसमान है, और गेंद $1$ उनमें से $(k-1)!$ में पहली है:
प्रायिकता $\frac{(k-1)!}{k!} = \frac1k$।
\end{solution}

\begin{solution}{exo:b2:proba:2}
$\frac{1}{k(k+1)} = \frac1k - \frac1{k+1}$, अतः $\sum_{k\geq1} p_k$ दूरबीन की तरह सिकुड़कर $1$ हो जाता है: यानी कोई
\omterm{def:b2:proba:space}{प्रायिकता माप}। सम परिणाम:
\[
\P(2\N^*) = \sum_{j=1}^{\infty}\frac{1}{2j(2j+1)}
= \sum_{j=1}^{\infty}\Bigl(\frac{1}{2j} - \frac{1}{2j+1}\Bigr)
= \frac12 - \frac13 + \frac14 - \frac15 + \cdots
\]
यह एकांतर हरात्मक श्रेणी ही है, जिसका पहला पद हटा दिया गया है और
चिह्न पलट दिए गए हैं: चूँकि $\ln 2 = 1 - \frac12 + \frac13 -
\frac14 + \cdots$ (\cref{ch:b2:series}),
\[
\P(2\N^*) = -\bigl(\ln 2 - 1\bigr) = 1 - \ln 2 \approx 0.307 .
\]
\end{solution}

\begin{solution}{exo:b2:proba:3}
मान लीजिए $S$ = रोगी, $+$ = धनात्मक जाँच। विभाजन $\{S, S^c\}$ के साथ बेज़
(\cref{thm:b2:proba:bayes}):
\[
\P(S \mid +)
= \frac{0.99 \times 10^{-4}}
       {0.99 \times 10^{-4} + 0.01 \times 0.9999}
= \frac{0.000099}{0.000099 + 0.009999}
\approx 0.0098 ,
\]
जो $1\%$ से नीचे है। यद्यपि जाँच ``99\% सटीक'' है, धनात्मक परिणाम
आपको लगभग $99\%$ संभावना के साथ स्वस्थ ही छोड़ देता है: विशाल स्वस्थ
बहुमत में से मिथ्या धनात्मक अत्यल्प रोगी अल्पमत के सच्चे धनात्मकों
को डुबो देते हैं। विरल स्थितियों के लिए छँटाई-जाँचें सदा इसी
आधार-दर परिकलन के द्वारा पढ़ी जानी चाहिए।
\end{solution}

\begin{solution}{exo:b2:proba:4}
$\Omega$ पर \omterm{def:b2:funcseq:def}{बिंदुवार}: $\omega \in \bigcup A_i$ तभी जब कोई गुणनखंड $1 - \mathbf{1}_{A_i}(\omega)$ लुप्त हो, अतः
\[
\mathbf{1}_{\bigcup A_i}
= 1 - \prod_{i=1}^n\bigl(1 - \mathbf{1}_{A_i}\bigr)
= \sum_{\emptyset \neq J \subseteq \{1,\dots,n\}}
(-1)^{\abs J + 1}\prod_{i \in J}\mathbf{1}_{A_i} ,
\]
जो गुणनफल खोलकर और $1$ को पार ले जाकर मिलता है। अब $\prod_{i\in J}\mathbf{1}_{A_i} = \mathbf{1}_{\bigcap_{i \in J}
A_i}$, और भारों
$\P(\{\omega\})$ के सामने योग लेने पर --- जो वैध है: परिमित रूप से कई परिबद्ध
पद, और हर कुल \omterm{def:b2:series:summable}{योग्य} --- हर सूचक अपनी \omterm{def:b2:proba:space}{घटना} की प्रायिकता में बदल जाता
है, जिससे सूत्र मिल जाता है।
\end{solution}

\begin{solution}{exo:b2:proba:5}
मान लीजिए $A_i$ = ``चिट्ठी $i$ सही लिफ़ाफ़े में है''। आकार $k$ के
$J$ के लिए $\P\bigl(\bigcap_{i\in J}A_i\bigr) =
\frac{(n-k)!}{n!}$ (जहाँ $k$ चिट्ठियाँ नियत कीजिए और बाक़ी का क्रमचय)।
समावेश--अपवर्जन से,
\[
\P\Bigl(\bigcup A_i\Bigr)
= \sum_{k=1}^n (-1)^{k+1}\binom nk \frac{(n-k)!}{n!}
= \sum_{k=1}^n \frac{(-1)^{k+1}}{k!} ,
\]
अतः
\[
\P(\text{कोई मिलान नहीं})
= 1 - \P\Bigl(\bigcup A_i\Bigr)
= \sum_{k=0}^{n}\frac{(-1)^k}{k!}
\xrightarrow[n\to\infty]{} e^{-1} \approx 0.368 .
\]
ठीक एक मिलान: ठीक एक अचल बिंदु वाला कोई क्रमचय नियत चिट्ठी के
चुनाव ($n$ तरीक़े) तथा शेष $n - 1$ के किसी \emph{विन्यास-भंग}
(कोई-मिलान-नहीं व्यवस्था) से निर्धारित होता है; विन्यास-भंगों की
संख्या के लिए $D_{n-1} = (n-1)!\sum_{k=0}^{n-1}\frac{(-1)^k}{k!}$ लिखने पर (जो पहला भाग है, $(n-1)!$ से मापित),
\[
\P(\text{ठीक एक मिलान})
= \frac{n\,D_{n-1}}{n!}
= \frac{D_{n-1}}{(n-1)!}
= \sum_{k=0}^{n-1}\frac{(-1)^k}{k!}
\xrightarrow[n\to\infty]{} e^{-1} :
\]
अर्थात् सीमा में ``कोई मिलान नहीं'' और ``ठीक एक मिलान'' समान रूप
से संभावित हैं, और दोनों की प्रायिकता $e^{-1}$ है।
\end{solution}

\begin{solution}{exo:b2:proba:6}
आरंभ पर प्रतिबंधित कीजिए (शृंखला नियम / \cref{thm:b2:proba:bayes}):
\begin{itemize}
\item पहला उछाल पट (प्रायिकता $1 - p$): खेल नए सिरे से शुरू हो जाता
है; $n$ से अधिक चलने का अर्थ है वहाँ से $n - 1$ से अधिक चलना: योगदान
$(1-p)\,q_{n-1}$;
\item पहले उछाल चिपट (प्रायिकता $p(1-p)$): दो उछालों के बाद फिर से आरंभ:
योगदान $p(1-p)\,q_{n-2}$;
\item पहले उछाल चिचि: खेल समाप्त हो चुका है ($n$ उछालों के भीतर,
$n
\geq 2$): योगदान $0$।
\end{itemize}
इसलिए $q_n = (1-p)q_{n-1} + p(1-p)q_{n-2}$। अभिलक्षणिक समीकरण $r^2 = (1-p)r + p(1-p)$ के मूल
\[
r_\pm = \frac{(1-p) \pm \sqrt{(1-p)^2 + 4p(1-p)}}{2},
\]
हैं, जहाँ $\abs{r_\pm} < 1$: वस्तुतः बहुपद $\chi(r) = r^2 -
(1-p)r - p(1-p)$ $\chi(1) = 1 - (1-p) - p(1-p) = p^2 >
0$ तथा $\chi(-1) = 1 + (1-p) - p(1-p) > 0$ पूरा करता है, जबकि
$\chi(0) =
-p(1-p) < 0$: अर्थात् एक मूल $\intoo{-1}{0}$ में, एक $\intoo{0}{1}$ में। अतः $q_n = \alpha r_+^n + \beta r_-^n \to 0$। घटनाएँ ``खेल
$n$ से अधिक चलता है'' घटकर ``खेल कभी समाप्त नहीं होता'' तक जाती
हैं; एकदिष्ट संततता (\cref{thm:b2:proba:continuity}) देती है
$\P(\text{कभी समाप्त नहीं होता}) = \lim q_n = 0$: अर्थात् खेल लगभग निश्चित
रूप से समाप्त होता है।
\end{solution}

\begin{solution}{exo:b2:proba:7}
\emph{$\P(R_n) = 1/n$:} पहली $n$ खींचों में अंतिम खींच की $n$ सापेक्ष
स्थितियों में से हर एक समान रूप से संभावित है (सापेक्ष क्रम की
एकसमानता), और $R_n$ वह \omterm{def:b2:proba:space}{घटना} है कि वह सबसे बड़ी है: प्रायिकता $1/n$।

\emph{अपरिमित रूप से कई कीर्तिमान:} $\sum_n \P(R_n) = \sum 1/n =
\infty$ और $R_n$ \omterm{def:b2:proba:independence}{स्वतंत्र} हैं (मान
लिया गया), अतः बोरेल--कांतेली~2 (\cref{thm:b2:proba:borelcantelli}) देती है $\P(\limsup R_n) = 1$: कीर्तिमान
लगभग निश्चित रूप से कभी बंद नहीं होते --- पर वे लघुगणकीय रूप से
विरल होते जाते हैं।

\emph{लगातार कीर्तिमान:} स्वतंत्रता से,
\[
\sum_n \P(R_n \cap R_{n+1})
= \sum_n \frac{1}{n(n+1)} < \infty ,
\]
अतः बोरेल--कांतेली~1 लागू होती है: लगभग निश्चित रूप से केवल परिमित
रूप से कई बार किसी कीर्तिमान के तुरंत बाद दूसरा कीर्तिमान आता है।
प्रमेयिका के दोनों आधे भाग साथ-साथ काम करते हैं: अपरिमित रूप से कई
कीर्तिमान, पर (लगभग निश्चित रूप से) अंततः कभी लगातार दो नहीं।
\end{solution}

\begin{solution}{exo:b2:proba:8}
\emph{पहला भाग:} स्वतंत्रता के साथ $\sum \P(A_n) = \sum\frac{1}{n+1} = \infty$: बोरेल--कांतेली~2 $\P(\limsup A_n) = 1$
देती है। हर अलग-अलग $A_n$ उत्तरोत्तर असंभावित है, फिर भी लगभग हर
$\omega$ उनमें से अपरिमित रूप से कई में है।

\emph{स्वतंत्रता के बिना प्रतिउदाहरण:} \cref{exo:b2:proba:2} के भारों $p_k = \frac{1}{k(k+1)}$ के साथ
$\Omega = \N^*$ लीजिए, और $A_n = \{k \in \N^* : k \geq n\}$। तब
\[
\P(A_n) = \sum_{k \geq n}\Bigl(\frac1k - \frac1{k+1}\Bigr)
= \frac1n ,
\qquad
\sum_n \P(A_n) = \infty ,
\]
पर $A_n$ ह्रासमान हैं, अतः $\limsup_n A_n = \bigcap_n A_n =
\emptyset$: $\P(\limsup A_n) = 0$। जब घटनाएँ समष्टि के किसी
सिकुड़ते भाग पर ढेर हो जाएँ, तब अकेली $\sum\P(A_n)$ की अपसारिता कुछ भी
आश्वस्त नहीं करती --- स्वतंत्रता ही वह है जो इस साज़िश को रोकती है।
\end{solution}

\begin{solution}{exo:b2:proba:9}
$q = 1 - p$ के साथ पहला चित्त कोटि $2j + 1$ पर प्रायिकता $q^{2j}p$ के साथ गिरता है,
अतः
\[
\P(\text{विषम कोटि}) = \sum_{j\geq0}q^{2j}p
= \frac{p}{1 - q^2} = \frac{1}{1 + q} .
\]
निष्पक्ष सिक्के के लिए: $\frac1{1 + 1/2} = \frac23$। (विश्वसनीयता जाँच: विषम कोटियाँ अधिक
संभावित होनी चाहिए, क्योंकि कोटि $1$ पहले आती है --- और वस्तुतः सदा
$\frac1{1+q} > \frac12$।)
\end{solution}

\begin{solution}{exo:b2:proba:10}
घटनाएँ $B_N = \bigcap_{n=1}^N A_n^c$ घटकर $\bigcap_nA_n^c$ तक जाती हैं, और पूरकों की स्वतंत्रता से $\P(B_N) = \prod_{n=1}^N(1 - p_n)$;
एकदिष्ट संततता (\cref{thm:b2:proba:continuity}) प्रदर्शित सीमा दे देती है। लघुगणक लेने पर,
$\prod(1 - p_n) > 0$ तभी जब $\sum-\ln(1 -
p_n) < \infty$। यदि $\sum p_n < \infty$ हो, तो $p_n \to 0$ और $-\ln(1 - p_n) \sim p_n$: लघु-श्रेणी अभिसरित
होती है। और यदि $\sum p_n = \infty$ हो, तो $-\ln(1 - p_n) \geq p_n$ अपसारिता को बाध्य कर देता है, अतः
गुणनफल $0$ है। यह बोरेल--कांतेली~2 से मेल खाता है: $\sum p_n = \infty$ के लिए न
केवल $\P(\text{कोई }A_n\text{ घटित नहीं होती}) = 0$, बल्कि
लगभग निश्चित रूप से अपरिमित रूप से कई $A_n$ घटित होती हैं।
\end{solution}

\begin{solution}{exo:b2:proba:11}
मान लीजिए पहली ख़ाली मिलने वाली डिब्बी $A$ है, और दूसरी डिब्बी में
$k$ हैं। इसका अर्थ है: पहली $2n - k$ पहुँचों में ठीक $n$ $A$ तक गईं और
$n - k$ $B$ तक (किसी क्रम में), और पहुँच संख्या $2n - k + 1$ फिर $A$ तक गई, जहाँ
वह ख़ाली मिली। पहुँचें \omterm{def:b2:proba:independence}{स्वतंत्र} निष्पक्ष चुनाव हैं, अतः इस \omterm{def:b2:proba:space}{घटना} की
प्रायिकता $\binom{2n-k}{n}2^{-(2n-k)}\cdot\frac12$ है; दुगुना करने पर (ख़ाली डिब्बी कोई भी हो सकती है)
\[
\P(\text{दूसरी डिब्बी में }k) = \binom{2n-k}{n}\,2^{-(2n-k)} .
\]
$n = 1$ के लिए: $k = 1$ $\binom11 2^{-1} = \frac12$ देता है और $k
= 0$ $\binom21 2^{-2} = \frac12$ देता है: कुल $1$,
जैसा होना ही चाहिए।
\end{solution}

\begin{solution}{exo:b2:proba:12}
(क) यदि सब $n$ के लिए $\P(\{n\}) = c$ हो, तो $\sigma$-योज्यता $1 = \sum_nc$ को बाध्य कर देती
है: जो असंभव है, चाहे $c = 0$ हो (योग $0$) या $c > 0$ (योग अपरिमित)। $\N$
पर कोई एकसमान प्रायिकता नहीं है।

(ख) यदि $A \cap B = \emptyset$ तथा $d(A)$, $d(B)$ विद्यमान हों, तो $\abs{(A \sqcup B)\cap\intint1n} = \abs{A\cap\intint1n} +
\abs{B\cap\intint1n}$, अतः $d(A \sqcup B) = d(A) + d(B)$: अर्थात्
ऐसे युग्मों पर परिमित योज्यता। हर एकल समुच्चय का गणना-फलन अंततः
अचर है, अतः घनत्व $0$, जबकि $d(\N^*) = 1$। यदि $d$ $\sigma$-योज्य होता, तो $\N^* =
\bigsqcup_k\{k\}$
$1 = \sum_k 0 = 0$ देता: घनत्व परिमित रूप से योज्य है पर $\sigma$-योज्य नहीं ---
अभिगृहीत में सार है।

(ग) $A = \bigcup_{k\geq0}\intint{4^k}{2\cdot4^k - 1}$ लीजिए ($4^k$ से $2\cdot4^k - 1$ तक के खंड)। $n = 2\cdot4^K -
1$ पर गणना $\sum_{k\leq K}4^k \sim \frac43 4^K$ है, जो
अनुपात $\to \frac23$ देती है; $n = 4^{K+1} - 1$ पर गणना अपरिवर्तित रहती है, जो अनुपात
$\to \frac13$ देती है। अनुपात सीमाओं $\frac13$ तथा $\frac23$ के बीच दोलन करता है: कोई
घनत्व नहीं।
\end{solution}

\begin{solution}{pb:b2:proba:1}
\textbf{1.} \omterm{def:b2:curves:length}{लंबाई} $n$ का कोई पथ अपने ऊपर-पगों के समुच्चय से
निर्धारित होता है; $k$ पर समाप्त होने का अर्थ है $u - (n - u) = k$ वाले $u$
ऊपर-पग तथा $n - u$ नीचे-पग, अर्थात् $u = \frac{n+k}2$: यह तभी संभव है जब $n + k$ सम हो
और $\abs k \leq n$, और $\binom{n}{(n+k)/2}$ तरीक़ों से। हर विशिष्ट पथ $n$ उछालों पर निष्पक्ष
गुणनफल माप का एक बिंदु है: प्रायिकता $2^{-n}$। इसलिए $\P(S_n = k) = N_n(k)2^{-n}$।

\textbf{2.} $S_n$ की सम-विषमता $n$ जैसी है, अतः $S_{2n+1} \neq
0$; और $u_n = N_{2n}(0)4^{-n} = \binom{2n}n4^{-n}$। मान:
$u_1 = \frac12$, $u_2 = \frac6{16} = \frac38$, $u_3 =
\frac{20}{64} = \frac5{16}$।

\textbf{3.} $\dfrac{u_n}{u_{n-1}} =
\dfrac{\binom{2n}n}{4\binom{2n-2}{n-1}} =
\dfrac{(2n)(2n-1)}{4n^2} = \dfrac{2n-1}{2n} < 1$: ह्रासमान। \cref{ex:b2:comparison:centralbinomial} से $\binom{2n}n \sim
\frac{4^n}{\sqrt{\pi n}}$, अतः $u_n \sim \frac1{\sqrt{\pi n}}
\to 0$, और $\sum
n^{-1/2}$ से तुलना
करने पर $\sum u_n$ अपसरित हो जाता है।

\textbf{4.} $0$ को छूने वाला $1$ से $k$ तक का कोई पथ दिया हो, तो
उसके आरंभिक खंड को ($0$ की \emph{पहली} भेंट तक) क्षैतिज अक्ष से
परावर्तित कीजिए: परिणाम $-1$ से $k$ तक का कोई पथ है, और यह
संक्रिया अंतर्वलन है --- $-1$ से $k \geq 1$ तक के हर पथ को $0$ पार करना
ही पड़ता है, और उसके आरंभिक खंड को वापस परावर्तित करने पर मूल पथ
पुनः मिल जाता है। इसलिए छूने वाले पथों की संख्या $N_{n-1}(k + 1)$ है ($-1$ से
$k$ तक विस्थापन $k + 1$ है)। $0$ से $k$ तक का कोई पथ जो समय $0$
के बाद $>
0$ बना रहता है, किसी ऊपर-पग से शुरू होता है और फिर $0$
को छुए बिना $n - 1$ पगों में $1$ से $k$ तक जाता है: ऐसे $N_{n-1}(k-1) - N_{n-1}(k+1)$ पथ हैं।

\textbf{5.} $\binom{n-1}{m-1} = \frac mn\binom nm$ तथा $\binom{n-1}{m} =
\frac{n-m}n\binom nm$ का उपयोग करते हुए $m = \frac{n+k}2$ के साथ:
\[
\frac{N_{n-1}(k-1) - N_{n-1}(k+1)}{N_n(k)}
= \frac{\binom{n-1}{m-1} - \binom{n-1}{m}}{\binom nm}
= \frac{m - (n - m)}{n} = \frac kn .
\]
$n = 3$, $k = 1$ के लिए: $N_3(1) = 3$ पथ ($++-$, $+-+$, $-++$), जिनमें से केवल $++-$
धनात्मक बना रहता है ($+-+$ समय $2$ पर $0$ पर लौट आता है): तीन में
से एक, और $\frac kn = \frac13$।

\textbf{6.} सममिति से प्रायिकता $2\P(S_i > 0\ \forall
i \leq 2n)$ है। अंत्यबिंदु $2k$ पर योग लेकर
और प्रश्न 4 का उपयोग करके ($n$ की जगह $2n$ रखकर):
\[
\P(S_i > 0\ \forall i) = 2^{-2n}\sum_{k\geq1}
\bigl(N_{2n-1}(2k-1) - N_{2n-1}(2k+1)\bigr)
= 2^{-2n}\,N_{2n-1}(1),
\]
जो कोई दूरबीनी योग है। अब $N_{2n-1}(1) = \binom{2n-1}{n}$ तथा $2\binom{2n-1}n = \binom{2n}n$ (पास्काल), अतः प्रदर्शित
प्रायिकता $2\cdot2^{-2n}\binom{2n-1}n =
\binom{2n}n4^{-n} = u_n$ है।

\textbf{7.} घटनाएँ $D_n = \{S_i \neq 0,\ i \leq 2n\}$ घटती हैं, और उनका प्रतिच्छेद ``कभी कोई
वापसी नहीं'' है; एकदिष्ट संततता तथा प्रश्न 6 से
$\P(\text{कोई वापसी नहीं}) = \lim u_n =
0$: अर्थात् चहलक़दमी लगभग निश्चित रूप
से लौटती है। इसके अतिरिक्त $f_n = \P(D_{n-1})
- \P(D_n) = u_{n-1} - u_n$, और प्रश्न 3 से
\[
u_{n-1} - u_n = u_n\Bigl(\frac{2n}{2n-1} - 1\Bigr) =
\frac{u_n}{2n-1};
\qquad
\sum_{n\geq1}f_n = u_0 - \lim u_n = 1 .
\]

\textbf{8.} $2n\,f_n = \frac{2n}{2n-1}u_n \geq u_n$, और $\sum u_n = \infty$ (प्रश्न 3): अतः श्रेणी $\sum 2nf_n$ अपसरित हो जाती
है। पहली वापसी निश्चित है पर उसका कोई परिमित माध्य प्रतीक्षा-समय
नहीं --- अर्थात् चहलक़दमी \emph{शून्य पुनरावर्ती} है, उस शब्दावली
में जो \cref{ch:b2:randomvar} देगा।

\textbf{9.} \omterm{def:b2:proba:space}{घटना} ``कम से कम $k$ वापसियाँ'' उन \omterm{def:b2:proba:space}{घटनाओं} का $0 < n_1 < \dots < n_k$ पर
असंयुक्त \omterm{def:b2:structures:countable}{गणनीय} संघ है कि ``पहली $k$ वापसियाँ ठीक समयों $2n_1, \dots, 2n_k$ पर
होती हैं''। ऐसी कोई \omterm{def:b2:proba:space}{घटना} असंयुक्त उछाल-खंडों $\intint1{2n_1}$, $\intint{2n_1+1}{2n_2}$, \dots पर
निर्भर $k$ \omterm{def:b2:proba:space}{घटनाओं} का प्रतिच्छेद है, जहाँ हर खंड किसी नई चहलक़दमी से
माँगता है कि वह ठीक निर्धारित पगों के बाद अपनी पहली वापसी करे;
खंडों की स्वतंत्रता से उसकी प्रायिकता $f_{n_1}f_{n_2-n_1}\cdots
f_{n_k-n_{k-1}}$ है। पुलिंदों से योग लेने
पर (\cref{ch:b2:series}, और सारे पद अऋणात्मक):
\[
\P(\text{कम से कम }k\text{ वापसियाँ})
= \Bigl(\sum_{n\geq1}f_n\Bigr)^{\!k} = 1^k = 1 .
\]
घटनाएँ $k$ में घटती हैं, अतः एकदिष्ट संततता से
$\P(\text{अपरिमित रूप से कई वापसियाँ}) = 1$: यानी पुनरावृत्ति।

\textbf{10.} प्रश्न 9 से चहलक़दमी $0$ से दूर अपरिमित रूप से कई
भ्रमण करती है। हर भ्रमण का पहला पग कोई नया सिक्का है, जो उससे पहले
की हर चीज़ से \omterm{def:b2:proba:independence}{स्वतंत्र} है: पहले $m$ भ्रमणों के नीचे की ओर शुरू होने
की प्रायिकता $2^{-m}$ है। $1$ तक पहुँचने के लिए चहलक़दमी को केवल एक
ऊपर की ओर शुरू होने वाला भ्रमण चाहिए ($<0$ से उसे $1$ तक पहुँचने से
पहले $0$ से गुज़रना ही पड़ता है, क्योंकि पग $\pm1$ हैं), अतः हर $m$
के लिए $\P(\text{कभी नहीं पहुँचती }1)
\leq 2^{-m}$: अर्थात् चहलक़दमी लगभग
निश्चित रूप से $1$ पर पहुँचती है। (लगभग निश्चित रूप से परिमित)
पहुँच-समय पर अपघटन करने पर वहाँ से फिर आरंभ की गई चहलक़दमी $1$ से
शुरू होने वाली कोई नई चहलक़दमी है: अतः आगमन से वह लगभग निश्चित रूप
से हर $k \geq 1$ पर पहुँचती है, और सममिति से हर $k \leq -1$ पर। अंत में, $k$
की पहली भेंट पर फिर आरंभ करने पर उस नई चहलक़दमी पर प्रश्न 9 लागू
होता है: अर्थात् लगभग निश्चित रूप से हर स्थल पर अपरिमित बार जाया
जाता है।

\textbf{11.} घटनाएँ $A_n = \{S_{2n} = 0\}$ स्वतंत्रता से बहुत दूर हैं (समय $2n$ पर
$0$ पर होना समय $2n + 2$ पर $0$ पर होने को $u_{n+1}$ से कहीं अधिक संभावित बना
देता है), अतः बोरेल--कांतेली~2 उपलब्ध नहीं है, और वस्तुतः भाग II
का सारा काम उसकी जगह लेना ही था। दूसरी दिशा को कोई स्वतंत्रता नहीं
चाहिए: \emph{यदि} $\sum\P(A_n)$ अभिसरित हो, तो बोरेल--कांतेली~1 लगभग निश्चित
रूप से परिमित रूप से कई वापसियाँ दे देती है। नीचे की हर क्षणिकता
उपपत्ति का इंजन यही निष्पत्ति है।

\textbf{12.} समय $2n$ पर वापसी के लिए $n$ ऊपर- तथा $n$ नीचे-पग
चाहिए: $\P(S_{2n} = 0) = \binom{2n}np^nq^n =
u_n(4pq)^n$, और $p \neq
\frac12$ के लिए $4pq = 1 - (p - q)^2 < 1$। चूँकि $u_n \leq 1$, श्रेणी $\sum\P(S_{2n} = 0)$ गुणोत्तर
$\sum(4pq)^n$ से प्रभावित है: अर्थात् अभिसारी। बोरेल--कांतेली~1 से
$\P(S_{2n} = 0 \text{ अपरिमित बार}) =
0$: अर्थात् लगभग निश्चित रूप से परिमित
रूप से कई वापसियाँ।

\textbf{13.} सम $n + k$ के लिए $\P(S_n = k) =
\binom{n}{\frac{n+k}2}p^{\frac{n+k}2}q^{\frac{n-k}2}$; द्विपद गुणांक अधिक से अधिक
केंद्रीय गुणांक जितना है, और $p^{\frac{n+k}2}q^{\frac{n-k}2} = (pq)^{n/2}(p/q)^{k/2}$, जिससे बताया गया परिबंध $\leq
2^n(pq)^{n/2}(p/q)^{k/2} = (4pq)^{n/2}(p/q)^{k/2}$
मिलता है, जो $n$ में \omterm{def:b2:series:summable}{योग्य} है क्योंकि $\sqrt{4pq} < 1$। बोरेल--कांतेली~1: स्थल
$k$ पर लगभग निश्चित रूप से परिमित बार जाया जाता है; और $k \in
\Z$ पर
अपवाद-रूप शून्य \omterm{def:b2:proba:space}{घटनाओं} का संघ भी शून्य है (\omterm{def:b2:structures:countable}{गणनीय} उप-योज्यता)। अतः
लगभग निश्चित रूप से हर स्थल पर परिमित बार जाया जाता है, इसलिए
पूर्णांक अनुक्रम $(S_n)$ हर परिबद्ध खिड़की को सदा के लिए छोड़ देता है:
$\abs{S_n} \to \infty$।

\textbf{14.} $\P(S_n \neq 0,\ 1 \leq n \leq 200) = u_{100} =
\binom{200}{100}4^{-100} \approx \frac1{\sqrt{100\pi}}
\approx 0.056$: अर्थात् बीस में से एक से अधिक संभावना कि $200$
निष्पक्ष उछाल कभी बराबरी पर न आएँ। क्षय $1/\sqrt{\pi n}$ कष्टदायक रूप से धीमा
है: बराबरी की निश्चितता (प्रश्न 7) बहुत लंबे बराबरी-रहित खिंचावों
के अनुकूल है --- भाग IV की चापज्या परिघटनाओं का पहला स्वाद।

\textbf{15.} पहले पग पर प्रतिबंधित कीजिए। यदि $X_1 = +1$, तो $T_1 = 1$, और $f_1 = \frac12$
मेल खाता है। यदि $X_1 = -1$, तो चहलक़दमी को $-1$ से $1$ तक चढ़ना पड़ता है;
खंड-अपघटन से, समय $2n$ पर पहली बार $0$ पर लौटना इस प्रकार बँटता है:
एक पग नीचे, फिर $-1$ से शुरू होने वाली कोई नई चहलक़दमी जो $2n - 1$ पगों
में पहली बार $0$ तक पहुँचे --- तुल्य रूप से कोई नई चहलक़दमी जो
पहली बार $+1$ तक पहुँचे --- अथवा ऊपर की ओर वाली सममित घटना। दोनों
चिह्न बराबर योगदान देते हैं:
\[
f_n = 2\cdot\tfrac12\,\P(T_1 = 2n - 1) = \P(T_1 = 2n-1) .
\]
इसलिए $\P(T_1 < \infty) = \sum f_n = 1$, जबकि प्रश्न 7 से $\sum_n(2n -
1)f_n = \sum_n u_n = \infty$: अर्थात् चहलक़दमी लगभग निश्चित
रूप से $1$ तक पहुँचती है, पर अपरिमित माध्य समय में।

\textbf{16.} $\{M_n \geq k\}$ का अंतिम मान $S_n = m$ के अनुसार विभाजन कीजिए। $m \geq k$ के
लिए शर्त $M_n \geq k$ स्वतः पूरी है। $m < k$ के लिए पथ को स्तर $k$ की उसकी
\emph{पहली} भेंट के बाद परावर्तित कीजिए: यह $\{M_n \geq k, S_n = m\}$ तथा $\{S_n = 2k - m\}$ के बीच कोई
एकैकी आच्छादन है ($2k - m > k$ पर समाप्त होने वाला हर पथ $k$ पर जाता है;
वापस परावर्तित करना प्रतिलोम है)। इसलिए
\[
\P(M_n \geq k)
= \sum_{m > k}\P(S_n = m) + \P(S_n = k)
+ \sum_{m < k}\P(S_n = 2k - m)
= 2\P(S_n > k) + \P(S_n = k).
\]

\textbf{17.} $k = 1$ वाले सम समय $2n$ पर: $\P(S_{2n} = 1)
= 0$ तथा $\P(S_{2n} > 1) = \P(S_{2n} \geq 2)$, अतः
\[
\P(M_{2n} \geq 1) = 2\P(S_{2n} \geq 2)
= \P(S_{2n} \geq 2) + \P(S_{2n} \leq -2)
= 1 - u_n .
\]
इस प्रकार $\P(S_i \leq 0\ \forall i \leq 2n) = u_n$: अर्थात् चहलक़दमी पहले $2n$ पगों में उतनी ही बार आगे
नहीं होती जितनी बार वह बराबरी पर नहीं आती (प्रश्न 6) --- दो
बिलकुल भिन्न घटनाएँ, जिन्हें एक ही $u_n$ ढो रहा है।

\textbf{18.} $\{L_{2n} = 2k\} = \{S_{2k} = 0\} \cap
\{\text{उछालों } 2k+1, \dots, 2n \text{ की चहलक़दमी में कोई शून्य
नहीं}\}$। दोनों घटनाएँ असंयुक्त उछाल-खंडों पर निर्भर हैं, अतः
वे \omterm{def:b2:proba:independence}{स्वतंत्र} हैं; पहली की प्रायिकता $u_k$ है, और दूसरी की $u_{n-k}$, जो
नई $(2n-2k)$-पग चहलक़दमी पर प्रश्न 6 लगाने से मिलती है। इसलिए $\P(L_{2n} = 2k) = u_ku_{n-k}$।
चूँकि $L_{2n}$ ठीक मान $0, 2, \dots, 2n$ लेता है, ये प्रायिकताएँ $1$ तक जुड़ती हैं:
$\sum_{k=0}^nu_ku_{n-k} = 1$, यानी किसी प्रायिकतात्मक विभाजन से मिली एक द्विपद सर्वसमिका।

\textbf{19.} सममिति तत्काल है: $u_ku_{n-k} =
u_{n-k}u_k$। चूँकि $u_j$ $j$ में घटता है,
गुणनफल $u_ku_{n-k}$ केंद्रीय $k$ के लिए सबसे छोटा और छोरों $k \in \{0, n\}$ पर सबसे
बड़ा है, जहाँ वह $u_n$ के बराबर है; परिमाणात्मक रूप से बीच में $u_ku_{n-k} \approx \frac1{\pi\sqrt{k(n-k)}}$,
जबकि किनारों पर $u_n \approx \frac1{\sqrt{\pi n}}$। $n = 5$ के लिए: $\P(L_{10} = 0) = \P(L_{10} = 10) =
u_5 = \frac{63}{256} \approx 0.246$, जबकि $\P(L_{10} = 4) =
u_2u_3 = \frac38\cdot\frac5{16} = \frac{15}{128} \approx
0.117$। किसी लंबे
निष्पक्ष खेल में अंतिम बराबरी सबसे अधिक संभावना से बिलकुल आरंभ में
या बिलकुल अंत में होती है: सिक्के में कोई पक्षपात न होने पर भी
प्रायः एक खिलाड़ी विशाल खिंचावों तक आगे रहता है।

\textbf{20.} चित्र यह है: समय $n$ पर चहलक़दमी मापक्रम $\sqrt n$ पर जीती
है (प्रश्न 3 का द्विपद फैलाव --- $u_n \sim
1/\sqrt{\pi n}$ केंद्रीय शिखर की ऊँचाई है);
वह प्रायिकता $1$ के साथ $0$ पर अपरिमित बार लौटती है (भाग II), फिर
भी वापसियों के बीच के प्रतीक्षा-समय का माध्य अपसारी है (प्रश्न 8),
और इसीलिए अकेले भ्रमण किसी भी क्षितिज का धनात्मक अंश घेर सकते हैं;
इसके अनुरूप किसी $2n$-पग खेल की अंतिम बराबरी फैली हुई है और उसके
चरम मान सर्वाधिक संभावित हैं (प्रश्न 18 तथा 19), और कभी आगे न होने
की प्रायिकता वही धीरे-धीरे क्षय होती $u_n$ है जो कभी बराबरी न होने
की है (प्रश्न 17)। सीमा में निश्चितता, हर परिमित क्षितिज पर दृढ़ता:
निष्पक्ष चहलक़दमी यही है।

\textbf{21.} $\{S_{2n} = 0\}$ ($n \geq 1$) का पहली वापसी-समय $2k$, $1 \leq k \leq n$, के अनुसार
विभाजन कीजिए: पहले $2k$ उछालों का खंड कोई पहली वापसी साकार करता है,
शेष $2n - 2k$ उछाल किसी नई चहलक़दमी की वापसी साकार करते हैं, और दोनों
खंड \omterm{def:b2:proba:independence}{स्वतंत्र} हैं: $u_n = \sum_{k=1}^nf_ku_{n-k}$। दोनों श्रेणियों $U(x) = \sum u_nx^n$, $F(x) = \sum f_nx^n$ की त्रिज्या $\geq
1$
है (गुणांक $\intcc01$ में हैं), और \omterm{thm:b2:series:fubini}{कोशी गुणनफल} (\cref{ch:b2:powerseries}) $0 \leq x < 1$ के लिए देता
है
\[
U(x) - 1 = \sum_{n\geq1}\Bigl(\sum_{k=1}^n
f_ku_{n-k}\Bigr)x^n = F(x)\,U(x),
\qquad\text{अर्थात्}\qquad
U(x)\bigl(1 - F(x)\bigr) = 1 .
\]

\textbf{22.} $x \uparrow 1$ होने पर $U(x)$ तथा $F(x)$ बढ़ते हैं (अऋणात्मक गुणांक);
हर आंशिक योग $\sum_{n\leq
N}u_n$ $\sum_{n\leq N}u_nx^n \leq U(x)$ की कोई सीमा है, अतः $U(x) \uparrow \sum u_n \in \intoc0{+\infty}$, और इसी प्रकार $F(x) \uparrow f = \sum f_n$।
यदि $\sum u_n = \infty$: $1 -
F(x) = 1/U(x) \to 0$, अतः $f = 1$। यदि $\sum u_n = S < \infty$: $1 - f = 1/S > 0$, अतः $f < 1$। जाँचें:
निष्पक्ष चहलक़दमी, $\sum u_n =
\infty$ तथा $f = 1$ (प्रश्न 3, 7); पक्षपाती चहलक़दमी,
$\sum
u_n(4pq)^n < \infty$ और उसके अनुरूप $f = 1 -
1/\sum_{n\geq0}u_n(4pq)^n < 1$, जो वापसियों की संख्या के लगभग निश्चित
परिमित होने (प्रश्न 12) से संगत है।

\textbf{23.} $\Z^2$ चहलक़दमी के चारों पगों $(\pm1, 0), (0, \pm1)$ के लिए $U = X + Y$ तथा $V = X - Y$ की
वृद्धियाँ ये हैं: $(1,0)$ के लिए $(+,+)$, $(0,1)$ के लिए $(+,-)$, $(0,-1)$ के लिए $(-,+)$,
$(-1,0)$ के लिए $(-,-)$ --- और चिह्नों का हर युग्म प्रायिकता $\frac14 = \frac12\cdot\frac12$ के साथ:
अर्थात् दोनों निर्देशांक चहलक़दमियाँ $(U_n)$ तथा $(V_n)$ $\Z$ पर \omterm{def:b2:proba:independence}{स्वतंत्र}
निष्पक्ष चहलक़दमियाँ हैं। चूँकि $S^{(2)}_{2n} = (0,0)$ तभी जब $U_{2n} = 0$ तथा $V_{2n} = 0$,
\[
\P\bigl(S^{(2)}_{2n} = (0,0)\bigr) = u_n^2 \sim \frac1{\pi
n}, \qquad \sum_nu_n^2 = \infty .
\]
प्रश्न 21 की नवीकरण सर्वसमिका तथा प्रश्न 22 का द्विभाजन कुछ भी
एकविमीय नहीं बरतते (केवल पहली वापसी पर अपघटन तथा असंयुक्त-खंड
स्वतंत्रता), अतः $\sum u_n^{(2)} = \infty$ $f^{(2)} =
1$ दे देता है, और प्रश्न 9 का तर्क उसे और
आगे बढ़ा देता है: $\Z^2$ पर चहलक़दमी लगभग निश्चित रूप से मूल बिंदु पर
अपरिमित बार लौटती है।

\textbf{24.} मान लिए गए परिबंध $\P(S^{(3)}_{2n} = 0)
\leq Cn^{-3/2}$ के साथ श्रेणी अभिसरित होती है,
और बोरेल--कांतेली~1 लगभग निश्चित रूप से परिमित रूप से कई वापसियाँ
दे देती है: अर्थात् $\Z^3$ पर चहलक़दमी क्षणिक है (और घातांक $-d/2$ वाला
वही परिबंध हर $d \geq 3$ को साध लेता है)। कुल मिलाकर: \emph{पोया की
प्रमेय} --- \omterm{pb:b2:proba:1}{सरल यादृच्छिक चहलक़दमी} $\Z$ तथा $\Z^2$ पर पुनरावर्ती है,
और $d \geq 3$ के लिए $\Z^d$ पर क्षणिक। कोई पिया हुआ आदमी घर का रास्ता पा
लेता है; कोई पिया हुआ पक्षी शायद न पाए।

\textbf{25.} पथ गिनने तथा परावर्तन ने ठीक-ठीक नियम उत्पन्न किए
($u_n$, मतपत्र प्रमेय, $f_n$, अधिकतम, अंतिम शून्य); एकदिष्ट संततता ने
हर सीमांत कथन (``कम से कम एक बार लौटती है'', ``अपरिमित बार'') को
परिमित-क्षितिज प्रायिकताओं की किसी सीमा में बदल दिया;
असंयुक्त-खंड स्वतंत्रता ने नवीकरण अपघटन चलाए (प्रश्न 9, 18, 21)
--- वह मार्कोव गुणधर्म का \omterm{def:b2:structures:countable}{गणनीय} कंकाल है; बोरेल--कांतेली~1
क्षणिकता का हथियार रही (प्रश्न 12 से 13, 24), जिसे कोई स्वतंत्रता
नहीं चाहिए थी; और नवीकरण सर्वसमिका ने सब कुछ द्विभाजन $\sum u_n = \infty \iff$
पुनरावृत्ति में व्यवस्थित कर दिया। अकेला \omterm{def:b2:powerseries:analytic}{वैश्लेषिक} निवेश स्थानीय
आकलन $u_n \sim 1/\sqrt{\pi
n}$ है: उसका वर्ग $1/(\pi n)$ अब भी अपसरित होता है (विमा $2$,
पुनरावर्ती), जबकि $n^{-3/2}$ अभिसरित होता है (विमा $3$, क्षणिक) --- अंततः
पोया की प्रमेय $\sum n^{-d/2}$ की अपसारिता के विषय में एक कथन है।
\end{solution}
