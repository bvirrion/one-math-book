\section*{अध्याय \ref{ch:b2:series} --- अनुक्रम और श्रेणियाँ}

\begin{solution}{exo:b2:series:1}
$\sum\frac{\cos n}{n}$: $a_n = \frac1n$ और $b_n =
\cos n$ के साथ आबेल की जाँच, जिसके आंशिक योग परिबद्ध हैं
(गुणोत्तर योग का वास्तविक भाग, जैसा \cref{ex:b2:series:sinn} में): अतः अभिसारी (\omterm{def:b2:series:def}{निरपेक्ष रूप
से} नहीं, उसी $\cos^2$ युक्ति से)।

$\sum \frac{(-1)^n}{\ln n}$ ($n \geq 2$): एकांतर जाँच, $\frac{1}{\ln n} \downarrow 0$: अभिसारी; \omterm{def:b2:series:def}{निरपेक्ष रूप से} नहीं ($\ln n
\leq n$)।

$\sum \frac{(-1)^n}{n^{3/4} + \cos n}$: प्रसार कीजिए,
\[
\frac{(-1)^n}{n^{3/4} + \cos n}
= \frac{(-1)^n}{n^{3/4}}\cdot
\frac{1}{1 + \frac{\cos n}{n^{3/4}}}
= \frac{(-1)^n}{n^{3/4}}
- \frac{(-1)^n\cos n}{n^{3/2}} + O\Bigl(\frac{1}{n^{9/4}}\Bigr).
\]
पहली श्रेणी: एकांतर, अभिसारी। दूसरी: \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसारी ($\frac{1}{n^{3/2}}$
पैमाना)। तीसरी: \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसारी। कुल मिलाकर: अभिसारी।
\end{solution}

\begin{solution}{exo:b2:series:2}
$\sum_{m\geq0} z^m$ का अपने साथ \omterm{thm:b2:series:fubini}{कोशी गुणनफल} (दोनों $\abs z < 1$ के लिए \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसारी):
विकर्ण गुणांक $c_k =
\sum_{m=0}^{k} 1 = k + 1$ है, अतः
\[
\frac{1}{(1-z)^2} = \sum_{k\geq0} (k+1)z^k .
\]
$z$ से गुणा करके पुनः सूचीबद्ध करने पर: $\sum_{n \geq 1} n z^n =
\frac{z}{(1-z)^2}$।
\end{solution}

\begin{solution}{exo:b2:series:3}
मान लीजिए $B_n = \sum_{k \leq n} b_k \to B$। $a_k = k$ के साथ \omterm{thm:b2:series:abel}{आबेल योग}:
\[
\sum_{k=1}^{n} k\,b_k = n B_n - \sum_{k=1}^{n-1} B_k
\quad\Longrightarrow\quad
\frac1n \sum_{k=1}^{n} k b_k = B_n - \frac{1}{n}\sum_{k=1}^{n-1}
B_k .
\]
अभिसारी $(B_k)$ के चेज़ारो माध्य उसकी सीमा $B$ पर जाते हैं (प्रथम वर्ष का
खंड), अतः दाहिना पक्ष $B - B = 0$ की ओर जाता है।
\end{solution}

\begin{solution}{exo:b2:series:4}
यदि $\theta \in \pi\Z$: तो सारे पद लुप्त हैं --- तुच्छ रूप से अभिसारी। मान लीजिए $\theta \notin \pi\Z$।

$\alpha > 1$: \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसारी ($n^{-\alpha}$ से प्रभुत्व)।

$0 < \alpha \leq 1$: आबेल की जाँच लागू होती है ($a_n = n^{-\alpha}
\downarrow 0$; $\sin n\theta$ के आंशिक योग $\frac{1}{\abs{\sin(\theta/2)}}$ से परिबद्ध,
गुणोत्तर योग): अतः अभिसारी। \omterm{def:b2:series:def}{निरपेक्ष रूप से} नहीं: $\abs{\sin n\theta} \geq \sin^2 n\theta = \frac{1 -
\cos 2n\theta}{2}$, और $\sum \frac{1 - \cos 2n\theta}{2n^\alpha}$ अपसरित
होती है ($\sum n^{-\alpha}$ अपसरित; $2\theta
\notin 2\pi\Z$ होने पर $\sum
\frac{\cos 2n\theta}{n^\alpha}$ आबेल से अभिसरित; और अपवर्जित स्थिति
$2\theta \in 2\pi\Z$ का अर्थ $\theta \in \pi\Z$ है, जो पहले ही निपट चुका)। अतः अर्ध-अभिसारी।
\end{solution}

\begin{solution}{exo:b2:series:5}
मान लीजिए $p_1, p_2, \dots$ $(u_n)$ के अऋणात्मक पद क्रम में हैं और $q_1, q_2, \dots$ ऋणात्मक पद। $\sum p_k$
और $\sum
q_k$ दोनों अपसरित होते हैं: यदि उनमें से एक अभिसरित होता, तो दूसरा
अभिसारी $\sum u_n$ में से उसे घटाने पर मिलता, अतः वह भी अभिसरित होता --- और तब
$\sum \abs{u_n} = \sum p_k - \sum q_k$ अभिसरित हो जाता, जो परिकल्पना के विरुद्ध है। साथ ही $u_n \to 0$ ($\sum u_n$
अभिसरित होती है)।

लालची पुनर्विन्यास: चालू योग के $\ell$ से पहली बार आगे निकलने तक धनात्मक पद
$p_1, p_2, \dots$ लीजिए (यह संभव है: $\sum p_k =
+\infty$); फिर योग के $\ell$ से नीचे गिरने तक ऋणात्मक पद
(संभव: $\sum q_k = -\infty$); और यही सदा दोहराइए (हर चरण परिमित है, और हर पद ठीक एक बार
प्रयुक्त होता है: अर्थात् सच्चा पुनर्विन्यास)। हर बदलाव के बाद चालू योग
की $\ell$ से दूरी अधिकतम अंतिम प्रयुक्त पद जितनी है; और $m$-वें बदलाव पर
प्रयुक्त पदों का सूचकांक $\to \infty$ है तथा $u_n \to 0$, अतः चालू योग $\ell$ पर अभिसरित
होते हैं।
\end{solution}

\begin{solution}{exo:b2:series:6}
योग्यता: $\frac{\abs x^{m+n}}{m!n!}$ के परिमित आंशिक योग $\bigl(\sum_m
\frac{\abs x^m}{m!}\bigr)^2 = \eu^{2\abs x}$ से परिबद्ध हैं। विकर्ण समूहन
(\cref{thm:b2:series:fubini}):
\[
(\eu^{x})^2 = \sum_{m,n} \frac{x^{m+n}}{m!\,n!}
= \sum_{k=0}^{\infty} x^k \sum_{m+n=k} \frac{1}{m!\,n!}
= \sum_k \frac{x^k}{k!}\sum_{m=0}^{k}\binom km
= \sum_k \frac{(2x)^k}{k!} = \eu^{2x} .
\]
\end{solution}

\begin{solution}{exo:b2:series:7}
योग्यता: गुणनफल कुल के रूप में $\zeta(2)^2$ से परिबद्ध (\cref{thm:b2:series:fubini} का कोशी-गुणनफल वाला
तर्क)। प्रतिचित्रण $(q, a, b) \mapsto (qa, qb)$, जो $\gcd(a, b) = 1$ वाले त्रिकों से युग्मों $(m, n)$ पर जाता है,
एकैकी आच्छादक है ($q = \gcd(m,n)$ रखिए)। \omterm{def:b2:series:summable}{योग्य कुल} को तदनुसार समूहबद्ध करने पर
(सूचकांक समुच्चय का एक विभाजन --- जो \omterm{def:b2:structures:countable}{गणनीय} वर्गों में विभाजन पर लगाए गए
\cref{thm:b2:series:rearrangement}/\cref{thm:b2:series:fubini} के अनुसार \omterm{def:b2:series:summable}{योग्य} कुलों के लिए वैध है):
\[
\zeta(2)^2 = \sum_{q} \sum_{\gcd(a,b)=1} \frac{1}{q^4 a^2 b^2}
= \zeta(4)\, S,
\qquad\text{अतः}\qquad
S = \frac{\zeta(2)^2}{\zeta(4)} .
\]
(\cref{ch:b2:fourier} से मान $\zeta(2) = \frac{\pi^2}{6}$, $\zeta(4) = \frac{\pi^4}{90}$ लेने पर: $S = \frac{5}{2}$।)
\end{solution}

\begin{solution}{exo:b2:series:8}
मान लीजिए $A = \sum a_n$ (निरपेक्ष), $B_m = \sum_{k\leq m} b_k \to B$, जो $M$ से परिबद्ध है। तब
\[
C_N = \sum_{k=0}^{N} c_k = \sum_{n=0}^{N} a_n B_{N-n}
\]
($a$ के सूचकांक के अनुसार इकट्ठा कीजिए)। लिखिए
\[
C_N - AB = \sum_{n=0}^{N} a_n (B_{N-n} - B) - B\sum_{n > N} a_n .
\]
अंतिम पद $0$ की ओर जाता है। योग को $n = \lfloor N/2
\rfloor$ पर बाँटिए: $n \leq N/2$ के लिए $N - n \geq N/2$, अतः
$\abs{B_{N-n} - B}
\leq \varepsilon_N := \sup_{m \geq N/2}\abs{B_m - B} \to 0$, और यह भाग $\leq \varepsilon_N \sum\abs{a_n}$ है; और $n > N/2$ के लिए $\abs{B_{N-n} - B} \leq 2M$, अतः यह भाग $\leq 2M \sum_{n >
N/2} \abs{a_n} \to 0$ है।
इसलिए $C_N \to AB$।
\end{solution}

\begin{solution}{exo:b2:series:9}
$u_n = 2^{-n} e_n$ लीजिए ($e_n$ वह अनुक्रम है जिसमें स्थान $n$ पर अकेला $1$ है)। तब $\sum \norm{u_n}_\infty = \sum 2^{-n} <
\infty$:
अर्थात् \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसारी। परंतु आंशिक योग $S_N =
(1, \tfrac12, \dots, 2^{-N}, 0, \dots)$ को अनुक्रम $(2^{-n})_n$ पर
अभिसरित होना पड़ता, जो अंततः शून्य \emph{नहीं} है: अतः $E$ के बाहर। और
$E$ के भीतर $(S_N)$ बिना सीमा का कोशी है ($\norm{S_N
- x}_\infty \geq 2^{-N-1}$ किसी भी अंततः शून्य $x$ की
मदद नहीं करता: कोटि $K$ के आगे लुप्त होने वाले किसी भी $x \in E$ के लिए $N > K$
के लिए $\norm{S_N - x} \geq
2^{-K-1}$): अतः श्रेणी $E$ में अभिसरित नहीं होती। \cref{thm:b2:nvs:absoluteconvergence} को ठीक पूर्णता
ही चाहिए।
\end{solution}

\begin{solution}{exo:b2:series:10}
$\arctan(n+1) - \arctan n$ और $\arctan\frac{1}{n^2+n+1}$ दोनों $\intoo{0}{\frac\pi2}$ में हैं, और स्पर्शज्या का योग-सूत्र देता है
\[
\tan\bigl(\arctan(n{+}1) - \arctan n\bigr)
= \frac{(n+1) - n}{1 + n(n+1)} = \frac{1}{n^2 + n + 1} :
\]
अर्थात् ऐसे अंतराल में स्पर्शज्याएँ बराबर हैं जहाँ $\tan$ एकैकी है, अतः
सर्वसमिका सत्य है। क्रमिक निरसन से,
\[
\sum_{n=1}^{N}\arctan\frac{1}{n^2+n+1}
= \arctan(N{+}1) - \arctan 1 \xrightarrow[N\to\infty]{}
\frac\pi2 - \frac\pi4 = \frac\pi4 .
\]

\end{solution}

\begin{solution}{exo:b2:series:11}
पहली: $\sqrt{n+1} - \sqrt n = \frac{1}{\sqrt{n+1} + \sqrt n}
\sim \frac{1}{2\sqrt n}$, अतः पद $\sim
2^{-\alpha}n^{-\alpha/2}$ हैं: अभिसरण तभी जब $\frac\alpha2 > 1$, अर्थात् $\alpha > 2$। दूसरी: $1 - \cos\frac1n \sim
\frac{1}{2n^2}$:
अभिसरित। तीसरी: $\bigl(1 + \frac1n\bigr)^n =
\eu^{\,n\ln(1 + 1/n)} = \eu^{\,1 - \frac1{2n} + O(n^{-2})} =
\eu\bigl(1 - \frac{1}{2n} + O(n^{-2})\bigr)$, अतः
\[
\eu - \Bigl(1 + \frac1n\Bigr)^{\!n} \sim \frac{\eu}{2n} :
\]
धनात्मक पद जो हरात्मक श्रेणी के किसी गुणज के समतुल्य हैं: अपसरित।
\end{solution}

\begin{solution}{exo:b2:series:12}
एकदिष्टता से $n\,a_{2n} \leq a_{n+1} + a_{n+2} + \dots +
a_{2n} = S_{2n} - S_n \to 0$ (अभिसारी श्रेणी के लिए कोशी कसौटी)। अतः $2n\,a_{2n} \to 0$, और $(2n{+}1)\,
a_{2n+1} \leq (2n{+}1)a_{2n} = \frac{2n+1}{2n}\,(2n\,a_{2n}) \to
0$:
$(na_n)$ के दोनों उपअनुक्रम $0$ की ओर जाते हैं, अतः $na_n \to 0$।

\emph{विलोम विफल:} $a_n = \frac1{n\ln n}$ धनात्मक ह्रासमान है और $na_n = \frac1{\ln n} \to 0$, फिर भी $\sum a_n$ अपसरित
होती है (बर्ट्रांड सीमांत, \cref{pb:b2:series:1}, प्रश्न 18)। \emph{एकदिष्टता अनिवार्य:}
$n$ के $2$ की घात होने पर $a_n = \frac1n$ लीजिए और अन्यथा $a_n = 2^{-n}$: तब $\sum a_n \leq
\sum_k 2^{-k} + \sum_n 2^{-n} < \infty$, परंतु $2$
की घातों पर $na_n = 1$: अतः $na_n \not\to 0$।
\end{solution}

\begin{solution}{pb:b2:series:1}
\textbf{1.} $m$ पर आगमन: $m = 0$ के लिए दोनों पक्ष $1$ हैं; और चरण में $\cos\theta + \iu\sin\theta$
से गुणा किया जाता है तथा योग-सूत्र $\cos(m\theta + \theta) =
\cos m\theta\cos\theta - \sin m\theta\sin\theta$, $\sin(m\theta
+ \theta) = \sin m\theta\cos\theta + \cos m\theta\sin\theta$ प्रयुक्त होते हैं। इसके
बदले $m = 2n+1$ के साथ द्विपद प्रमेय से प्रसार करके काल्पनिक भाग इकट्ठा करने पर
($\iu\sin\theta$ की विषम घातें, $\iu^{2j+1} = (-1)^j\iu$ के साथ):
\[
\sin\bigl((2n{+}1)\theta\bigr) = \sum_{j=0}^{n}(-1)^j
\binom{2n+1}{2j+1}\cos^{2(n-j)}\theta\,\sin^{2j+1}\theta .
\]

\textbf{2.} $\intoo0{\frac\pi2}$ पर $\sin\theta \neq 0$: हर पद में से $\sin^{2n+1}\theta$ बाहर निकालिए, जिससे $\bigl(\frac{\cos^2\theta}{\sin^2\theta}\bigr)^{n-j} =
(\cot^2\theta)^{n-j}$ बचता
है: अर्थात् $P_n(x) =
\sum_j(-1)^j\binom{2n+1}{2j+1}x^{n-j}$ के साथ प्रदर्शित सर्वसमिका। उसका घात-$n$ गुणांक $j = 0$ का
$\binom{2n+1}{1} = 2n + 1 \neq 0$ है।

\textbf{3.} $\theta_k = \frac{k\pi}{2n+1}$ पर: $\sin\bigl((2n{+}1)\theta_k\bigr) = \sin k\pi = 0$ जबकि $\sin^{2n+1}\theta_k \neq 0$, अतः $P_n(\cot^2\theta_k) = 0$। $\theta_k$ $\intoo0{\frac\pi2}$ में यथार्थतः बढ़ते
हैं, जहाँ $\cot^2$ यथार्थतः घटता है: अतः मान $x_k =
\cot^2\theta_k$ परस्पर भिन्न हैं ---
अर्थात् घात-$n$ के बहुपद के $n$ भिन्न मूल, इसलिए वही सब।

\textbf{4.} वियेत: मूलों का योग $x^{n-1}$- और $x^n$-गुणांकों के अनुपात का ऋण
होता है:
\[
\sum_{k=1}^{n}\cot^2\theta_k =
\frac{\binom{2n+1}{3}}{\binom{2n+1}{1}}
= \frac{(2n+1)(2n)(2n-1)/6}{2n+1} = \frac{n(2n-1)}{3}.
\]

\textbf{5.} $\frac{1}{\sin^2\theta} = 1 + \cot^2\theta$: योग लेने पर $n + \frac{n(2n-1)}3 = \frac{3n + 2n^2 - n}{3} =
\frac{2n(n+1)}{3}$।

\textbf{6.} $\intoo{0}{\frac\pi2}$ पर: $\sin\theta < \theta <
\tan\theta$ (प्रथम वर्ष का खंड)। व्युत्क्रम लेने से क्रम उलट
जाता है: $\cot\theta < \frac1\theta < \frac1{\sin\theta}$, और (सब धनात्मक होने से) वर्ग करने पर $\cot^2\theta < \frac1{\theta^2} <
\frac1{\sin^2\theta}$।

\textbf{7.} प्रश्न 6 को $\theta = \theta_k$ पर $k \leq
n$ पर जोड़िए, प्रश्न 4 और 5 तथा $\frac1{\theta_k^2} =
\frac{(2n+1)^2}{k^2\pi^2}$ का
उपयोग करते हुए:
\[
\frac{n(2n-1)}3 < \frac{(2n+1)^2}{\pi^2}\sum_{k=1}^n\frac1{k^2}
< \frac{2n(n+1)}3 .
\]

\textbf{8.} $\frac{\pi^2}{(2n+1)^2}$ से गुणा कीजिए:
\[
\frac{\pi^2}{3}\cdot\frac{n(2n-1)}{(2n+1)^2}
< \sum_{k=1}^{n}\frac1{k^2} <
\frac{\pi^2}{3}\cdot\frac{2n(n+1)}{(2n+1)^2}.
\]
दोनों परिबंध $\frac{\pi^2}3\cdot\frac12 = \frac{\pi^2}6$ की ओर जाते हैं (परिमेय भिन्न $\frac12$ की ओर जाती हैं)।
आंशिक योग बढ़ते हैं, अतः वे अभिसरित होते हैं, और निचोड़ $\zeta(2) =
\frac{\pi^2}6$ दे देता है:
यही ऑयलर की प्रमेय है, कोशी की उपपत्ति से।

\textbf{9.} आंशिक योग बढ़कर $\zeta(2) =
\frac{\pi^2}6$ तक जाते हैं, अतः $0 \leq \frac{\pi^2}6 - \sum_{k\leq n}k^{-2}$; और प्रश्न 8 का
निचला परिबंध देता है
\[
\frac{\pi^2}6 - \sum_{k\leq n}\frac1{k^2}
\leq \frac{\pi^2}6 - \frac{\pi^2}3\cdot\frac{n(2n-1)}{(2n+1)^2}
= \frac{\pi^2}6\cdot\frac{(2n+1)^2 - (4n^2 -
2n)}{(2n+1)^2}
= \frac{\pi^2}6\cdot\frac{6n + 1}{(2n+1)^2}
= O\Bigl(\frac1n\Bigr),
\]
जो \cref{exo:b2:comparison:11} की यथार्थ पुच्छ $\frac1n - \frac1{2n^2} + O(n^{-3})$ से मेल खाता है।

\textbf{10.} मूलों का दूसरा प्रारंभिक सममित फलन $\sigma_2 = \frac{\binom{2n+1}5}{\binom{2n+1}1} =
\frac{(2n)(2n-1)(2n-2)(2n-3)}{120}$ है, अतः
\[
\sum_k\cot^4\theta_k = \sigma_1^2 - 2\sigma_2
= \Bigl(\frac{n(2n-1)}3\Bigr)^2 -
\frac{2n(2n-1)(2n-2)(2n-3)}{60}
\sim \frac{4n^4}9 - \frac{4n^4}{15} = \frac{8n^4}{45}.
\]
$\cot^4\theta < \theta^{-4} < (1 + \cot^2\theta)^2 =
1 + 2\cot^2\theta + \cot^4\theta$ को निचोड़कर और योग लेकर: दोनों बाहरी योग $\frac{8n^4}{45}(1 + o(1))$ हैं (जोड़ा गया $n + 2\sigma_1 =
O(n^2)$
नगण्य है), जबकि बीच वाला $\frac{(2n+1)^4}{\pi^4}\sum_{k\leq n}k^{-4}$ है। अतः
\[
\sum_{k\leq n}\frac1{k^4} \longrightarrow
\pi^4\cdot\frac{8/45}{16} = \frac{\pi^4}{90}.
\]

\textbf{11.} $\zeta(2)$ को सम-विषमता के अनुसार बाँटने पर: $\sum_{\text{सम}}
= \sum_j\frac{1}{(2j)^2} = \frac14\zeta(2) = \frac{\pi^2}{24}$,
अतः $\sum_{\text{विषम}} = \zeta(2) - \frac{\pi^2}{24} =
\frac{\pi^2}8$। एकांतर: $\sum_k\frac{(-1)^{k-1}}{k^2} =
\sum_{\text{विषम}} - \sum_{\text{सम}} = \frac{\pi^2}8 -
\frac{\pi^2}{24} = \frac{\pi^2}{12}$
(पुनःसमूहन को निरपेक्ष अभिसरण न्यायोचित ठहराता है, \cref{thm:b2:series:rearrangement})।

\textbf{12.} $S = \frac{\zeta(2)^2}{\zeta(4)} =
\frac{(\pi^2/6)^2}{\pi^4/90} = \frac{90}{36} = \frac52$। अनुमान: \cref{exo:b2:series:7} की सर्वसमिका $\zeta(2)^2 = \zeta(4)S$ कहती है कि $\gcd$ बाहर
निकालने से युग्म सहअभाज्य युग्मों में पुनर्मानकित हो जाते हैं; अतः
व्युत्क्रम $\frac1{\zeta(2)} = \frac6{\pi^2} \approx 0.608$ सभी युग्मों में सहअभाज्य युग्मों के घनत्व का स्वाभाविक
उम्मीदवार है --- और यह $\lim_N \frac{1}{N^2}\#\{(m,n) \leq N :
\gcd = 1\}$ के बारे में कथन है जिसकी ईमानदार उपपत्ति
(त्रुटि पदों सहित) तृतीय वर्ष के खंड की है।

\textbf{13.} सीधे योग की त्रुटि $\sim \frac1n$ है: छह अंकों के लिए लगभग $10^6$ पद
चाहिए। और सुधारे हुए योग $\sum_{k\leq n}k^{-2} + \frac1n - \frac1{2n^2}$ की त्रुटि $O(n^{-3})$ है: $n = 100$ पर वह $1.6449339\dots$ निकलता है
जबकि $\frac{\pi^2}6 = 1.6449341\dots$ --- अर्थात् त्रुटि $1.7\cdot10^{-7}$, यानी सौ पदों से सात अंक। अनंतस्पर्शी
सुधार कच्चे धैर्य को परिमाण की चार कोटियों से हरा देते हैं।

\textbf{14.} $n = 1$: $P_1(x) = 3x - 1$, मूल $\frac13$, और वास्तव में $\cot^2\frac\pi3 = \bigl(\frac1{\sqrt3}\bigr)^2 =
\frac13 = \frac{1\cdot1}3$। $n = 2$: सूत्र $\frac{2\cdot3}3 = 2$ की
भविष्यवाणी करता है; और $\cos\frac\pi5 = \frac{1 +
\sqrt5}{4}$ के साथ $\cot^2 36^\circ \approx 1.894$ तथा $\cot^2 72^\circ \approx 0.106$ निकलते हैं: योग $2.000$।

\textbf{15.} संख्याएँ $\tan^2\theta_k = \frac1{x_k}$ $Q(x) = x^nP_n\bigl(\frac1x\bigr) =
\sum_{j=0}^n(-1)^j\binom{2n+1}{2j+1}x^j$ के मूल हैं ($x_k$ अशून्य हैं)। $Q$ पर
वियेत: अग्र गुणांक $(-1)^n$ है (पद $j = n$), अगला $(-1)^{n-1}\binom{2n+1}{2n-1} =
(-1)^{n-1}\binom{2n+1}{2}$, अतः
\[
\sum_{k=1}^n\tan^2\theta_k =
-\frac{(-1)^{n-1}\binom{2n+1}2}{(-1)^n} = \binom{2n+1}2\cdot
\frac{2}{2n+1}\cdot\frac{2n+1}{2}
= n(2n+1).
\]
$n = 1$ पर जाँच: $\tan^2\frac\pi3 = 3 = 1\cdot3$।

\textbf{16.} मान लीजिए $\gamma = \frac{1+\alpha}2 \in
\intoo{1}{\alpha}$। तब $\frac{n^{-\alpha}(\ln
n)^{-\beta}}{n^{-\gamma}} = n^{\gamma - \alpha}(\ln n)^{-\beta}
\to 0$ ($n$ की कोई ऋणात्मक घात $\ln n$ की हर घात
को हरा देती है), अतः अंततः पद $\gamma > 1$ के साथ $\leq n^{-\gamma}$ हैं: अर्थात् किसी रीमान
श्रेणी से तुलना करके अभिसरण।

\textbf{17.} मान लीजिए $\gamma = \frac{1+\alpha}2 \in
\intoo{\alpha}{1}$: अब $\frac{n^{-\gamma}}{n^{-\alpha}(\ln
n)^{-\beta}} = n^{\alpha-\gamma}(\ln n)^{\beta} \to 0$, अतः अंततः पद $\gamma < 1$ के साथ $\geq n^{-\gamma}$ हैं:
अर्थात् अपसरण।

\textbf{18.} $f(t) = \frac1{t(\ln t)^\beta}$ धनात्मक और \omterm{def:b2:metric:continuity}{संतत} है, तथा बड़े $t$ के लिए ह्रासमान (उसके
लघुगणक का अवकलज अंततः $-\frac1t\bigl(1 + \frac{\beta}{\ln t}\bigr) < 0$ है)। प्रतिअवकलज: $\beta \neq 1$ के लिए
$\int^x f = \frac{(\ln x)^{1-\beta}}{1-\beta} + \text{अचर}$, जिसकी परिमित सीमा तभी होती है जब $\beta > 1$; और $\beta = 1$ के लिए
$\int^x f = \ln\ln x \to \infty$। \cref{thm:b2:comparison:seriesintegral} के अनुसार श्रेणी और समाकल की प्रकृति एक ही है: अतः अभिसरण तभी
जब $\beta > 1$।

\textbf{19.} वही जाँच: $\frac{\dd}{\dd t}\ln\ln\ln t =
\frac{1}{t\ln t\,\ln\ln t}$, और $\ln\ln\ln t \to \infty$: अपसरण। और $-\frac1{\ln\ln t} \to
0$ के साथ $\frac{\dd}{\dd t}\Bigl(-\frac1{\ln\ln t}\Bigr)
= \frac{1}{t\ln t\,(\ln\ln t)^2}$: अभिसरण।

\textbf{20.} पहली: $n^{1/\ln n} = \eu^{\ln n/\ln n} = \eu$, अतः पद ठीक $\frac{1}{\eu\,n}$ हैं: अर्थात् हरात्मक श्रेणी का
गुणज, इसलिए \emph{अपसारी} --- घातांक $1 +
\frac1{\ln n}$ $1$ तक बहुत तेज़ी से रेंग जाता
है। दूसरी: $n^{1/\ln\ln n}
= \eu^{\ln n/\ln\ln n}$, और अंततः $\frac{\ln n}{\ln\ln n} \geq
2\ln\ln n$, अतः $n^{1/\ln\ln n} \geq (\ln n)^2$: पद $\leq \frac1{n(\ln n)^2}$ हैं, जो अभिसारी बर्ट्रांड
श्रेणी है (प्रश्न 18): अर्थात् \emph{अभिसारी}। सीमांत इन दोनों घातांकों
के ठीक बीच से गुज़रता है।

\textbf{21.} $R_n \downarrow 0$ और
\[
\sqrt{R_n} - \sqrt{R_{n+1}} = \frac{R_n -
R_{n+1}}{\sqrt{R_n} + \sqrt{R_{n+1}}} =
\frac{a_n}{\sqrt{R_n} + \sqrt{R_{n+1}}} \geq
\frac{a_n}{2\sqrt{R_n}},
\]
अतः $\sum_n \frac{a_n}{\sqrt{R_n}} \leq 2\sum_n(\sqrt{R_n} -
\sqrt{R_{n+1}}) = 2\sqrt{R_1} < \infty$ (क्रमिक निरसन)। फिर भी $\frac{a_n/\sqrt{R_n}}{a_n} = \frac1{\sqrt{R_n}} \to \infty$: नई श्रेणी अनंत गुना बड़ी होते हुए भी
अभिसरित होती है। कोई भी अभिसारी श्रेणी सबसे धीमी नहीं होती; किसी भी
स्थिर कुल से की गई तुलना-जाँचें कभी \omterm{def:b2:metric:complete}{पूर्ण} नहीं हो सकतीं।

\textbf{22.} ह्रासमान धनात्मक $(a_n)$ के लिए $2$ की क्रमागत घातों के बीच के
पदों को समूहबद्ध कीजिए:
\[
2^{k}a_{2^{k+1}} \leq \sum_{n=2^k}^{2^{k+1}-1} a_n \leq
2^ka_{2^k} .
\]
$k$ पर योग लेने पर: यदि $\sum 2^ka_{2^k}$ अभिसरित हो तो $\sum a_n$ के आंशिक योग परिबद्ध हैं
(अभिसारी); और यदि $\sum a_n$ अभिसरित हो तो $\sum_k 2^{k+1}a_{2^{k+1}} \leq 2\sum_n a_n <
\infty$। $a_n = \frac1{n(\ln n)^\beta}$ के लिए: $2^ka_{2^k} =
\frac{1}{(k\ln 2)^\beta}$, और $\sum k^{-\beta}$
अभिसरित होती है तभी जब $\beta > 1$: अर्थात् प्रश्न 18 का वही सीमांत, बिना
समाकलों के।

\textbf{23.} समाकल-तुलना से
\[
\sum_{n > N}\frac{1}{n(\ln n)^2} \geq
\int_{N+1}^{\infty}\frac{\dd t}{t(\ln t)^2} =
\frac{1}{\ln(N+1)},
\]
जो $N = 10^6$ पर $\approx 0.0724$ है: अर्थात् दस लाख पदों के बाद भी पुच्छ $0.07$ से बड़ी रहती
है --- श्रेणी अभिसरित तो होती है, पर कोई भी सीधा योग उसका मान कभी नहीं
दिखा पाएगा। इसकी तुलना प्रश्न 13 से कीजिए, जहाँ एक ही अनंतस्पर्शी सुधार
ने सौ पदों से सात अंक ख़रीद लिए: कोई श्रेणी \emph{कैसे} अभिसरित होती है,
यह जानना उसके अभिसरित होने भर को जानने से अधिक क़ीमती है।

\textbf{24.} $\sum\frac1{n\ln n}$: अपसरित ($\alpha = 1$, $\beta = 1$, प्रश्न 18)। $\sum\frac1{n^{1.01}}$: अभिसरित (रीमान,
$\alpha > 1$)। $\sum\frac{(\ln n)^{100}}{n^{1.001}}$: अभिसरित ($\alpha = 1.001 > 1$, $\beta = -100$, प्रश्न 16)। $\sum\frac1{n\ln n(\ln\ln n)^3}$: अभिसरित (प्रश्न 19 का
ढर्रा: प्रतिअवकलज $-\frac12(\ln\ln t)^{-2}$, परिमित सीमा)।

\textbf{25.} डी म्वाव्र $\sin(2n{+}1)
\theta_k$ के लुप्त होने को $\cot^2\theta_k$ पर किसी बहुपद के लुप्त
होने में बदल देता है, और वियेत वे यथार्थ मूल-योग पढ़ लेता है जिनका अकेला
विश्लेषण केवल आकलन कर सकता था (प्रश्न 1--5)। निचोड़ को दोनों ओर $\frac{n(2n-1)}3$ और
$\frac{2n(n+1)}3$ के \emph{यथार्थ} मान चाहिए थे --- समतुल्य तो प्रश्न को ही मान लेते,
क्योंकि पूरी बात अचर $\frac{\pi^2}6$ की ही है (प्रश्न 7--8)। भाग IV पूरा-का-पूरा \cref{ch:b2:comparison}
की श्रेणी--समाकल तुलना पर चला, जिसमें वर्गीकरण का काम लघुगणकीय
प्रतिअवकलजों ने किया (प्रश्न 18--19)। और प्रश्न 21 सार्वत्रिक
तुलना-जाँच का स्वप्न तोड़ देता है: हर अभिसारी श्रेणी के नीचे एक और, अनंत
गुना धीमी श्रेणी पड़ी है --- बर्ट्रांड जैसे पैमाने सीमांत का मानचित्र
उत्तरोत्तर बारीक बनाते हैं पर उस तक कभी नहीं पहुँचते। शिखर: ऑयलर का $\zeta(2)
= \frac{\pi^2}6$
अपनी ऊपरी मंज़िल $\zeta(4) =
\frac{\pi^4}{90}$ सहित (प्रश्न 8, 10), और बर्ट्रांड वर्गीकरण (प्रश्न
16--18); $\zeta(2)$ \cref{ch:b2:fourier} में लौटता है, जहाँ पारसेवाल की सर्वसमिका उसे आरे-दाँत
तरंग की फूरिये श्रेणी से एक ही पंक्ति में फिर सिद्ध कर देती है --- एक
अचर, दो सभ्यताएँ।
\end{solution}
