\section*{अध्याय \ref{ch:b2:multint} --- रेखा समाकल और बहुक समाकल}

\begin{solution}{exo:b2:multint:1}
(क) खंड $\gamma(t) = (t, t)$, $t \in [0,1]$:
\[
\int_\gamma y^2\dd x + x\dd y
= \int_0^1 (t^2 + t)\,\dd t = \frac13 + \frac12 = \frac56 .
\]
(ख) परवलय $\gamma(t) = (t, t^2)$:
\[
\int_0^1 \bigl(t^4\cdot 1 + t\cdot 2t\bigr)\dd t
= \frac15 + \frac23 = \frac{13}{15} .
\]
दोनों मान भिन्न हैं, अतः समाकल पथ पर निर्भर है: यानी रूप \omterm{def:b2:multint:exact}{यथातथ}
\emph{नहीं} है --- और इसके अनुरूप $P_y = 2y \neq 1 = Q_x$, अतः वह संवृत भी नहीं है।
\end{solution}

\begin{solution}{exo:b2:multint:2}
$P = 2xy + y^3$, $Q = x^2 + 3xy^2 + 1$: $P_y = 2x + 3y^2 = Q_x$, जो $\R^2$ पर संवृत है। ऐसा $f$ खोजिए कि $f_x = P$: $f = x^2y + xy^3
+ g(y)$;
फिर $f_y = x^2 + 3xy^2 + g'(y) = Q$ $g'(y) = 1$ को बाध्य कर देता है, कहिए $g(y) = y$। अतः
\[
f(x, y) = x^2y + xy^3 + y
\]
कोई \omterm{def:b2:multint:exact}{विभव} है ($\R^2$ तारकाकार है, अतः प्वांकारे की प्रमेयिका से कोई
\omterm{def:b2:multint:exact}{विभव} होना ही था --- पर उसे दिखा देना अधिक तेज़ है)। \cref{thm:b2:multint:ftc} से, $(0,0)$
से $(1,2)$ तक किसी भी चाप के लिए:
\[
\int_\gamma\omega = f(1, 2) - f(0, 0) = 2 + 8 + 2 = 12 .
\]
\end{solution}

\begin{solution}{exo:b2:multint:3}
प्रांत $\{0 \leq x \leq 1,\ x^2 \leq y \leq x\}$ है। \emph{पहले $y$:}
\[
\int_0^1\!\int_{x^2}^{x}(x + y)\,\dd y\,\dd x
= \int_0^1\Bigl(x(x - x^2)
+ \frac{x^2 - x^4}{2}\Bigr)\dd x
= \int_0^1\Bigl(\frac{3x^2}{2} - x^3 - \frac{x^4}{2}\Bigr)\dd x
= \frac12 - \frac14 - \frac1{10} = \frac{3}{20} .
\]
\emph{पहले $x$:} ऊँचाई $y \in [0, 1]$ पर फाँक $y \leq x \leq
\sqrt y$ है, अतः
\[
\int_0^1\!\int_{y}^{\sqrt y}(x + y)\,\dd x\,\dd y
= \int_0^1\Bigl(\frac{y - y^2}{2}
+ y(\sqrt y - y)\Bigr)\dd y
= \frac14 - \frac16 + \frac25 - \frac13 = \frac{3}{20} .
\]
\end{solution}

\begin{solution}{exo:b2:multint:4}
\emph{पहला समाकल।} चक्रिका $D_R$ पर, ध्रुवीय निर्देशांकों में:
\[
\iint_{D_R}\frac{\dd x\,\dd y}{(1 + x^2 + y^2)^2}
= \int_0^{2\pi}\!\!\int_0^R
\frac{\rho\,\dd\rho\,\dd\alpha}{(1 + \rho^2)^2}
= 2\pi\Bigl[-\frac{1}{2(1 + \rho^2)}\Bigr]_0^R
= \pi\Bigl(1 - \frac{1}{1 + R^2}\Bigr)
\xrightarrow[R \to \infty]{} \pi .
\]
\emph{दूसरा समाकल।} चतुर्थांश चक्रिका $0 \leq \alpha \leq
\frac\pi2$, $0 \leq \rho \leq 1$ है, और $xy =
\rho^2\cos\alpha\sin\alpha$:
\[
\iint_{D'}xy\,\dd x\,\dd y
= \int_0^{\pi/2}\!\!\cos\alpha\sin\alpha\,\dd\alpha
\int_0^1 \rho^3\,\dd\rho
= \frac12\cdot\frac14 = \frac18 .
\]
\end{solution}

\begin{solution}{exo:b2:multint:5}
$x = \cos^3 t$, $y = \sin^3 t$ के साथ \cref{cor:b2:multint:area} से: $x' = -3\cos^2 t\sin t$, $y' = 3\sin^2 t\cos t$, अतः
\[
xy' - yx' = 3\cos^4 t\sin^2 t + 3\sin^4 t\cos^2 t
= 3\sin^2 t\cos^2 t = \frac{3}{4}\sin^2 2t
= \frac{3}{8}(1 - \cos 4t) .
\]
इसलिए
\[
A = \frac12\int_0^{2\pi}\frac38(1 - \cos 4t)\,\dd t
= \frac{3}{16}\cdot 2\pi = \frac{3\pi}{8} .
\]
(\omterm{pb:b2:curves:1}{ऐस्ट्रॉयड} \omterm{def:b2:multint:domain}{क्षेत्रफल} $\pi$ की इकाई चक्रिका में समा जाता है; उसके
चार-उभयाग्र वाले तारे के आकार के लिए $\pi$ का तीन-आठवाँ भाग
विश्वसनीय है।)
\end{solution}

\begin{solution}{exo:b2:multint:6}
\emph{चट्टा लगाकर:} इकाई चक्रिका $D$ के हर $(x, y)$ के ऊपर $z$ $x^2 + y^2$ से
$1$ तक चलता है:
\[
V = \iint_D \bigl(1 - x^2 - y^2\bigr)\dd x\,\dd y
= \int_0^{2\pi}\!\!\int_0^1 (1 - \rho^2)\rho\,\dd\rho\,\dd\alpha
= 2\pi\Bigl(\frac12 - \frac14\Bigr) = \frac\pi2 .
\]
\emph{काटकर:} ऊँचाई $z \in [0, 1]$ पर फाँक चक्रिका $x^2 + y^2 \leq z$ है, जिसका \omterm{def:b2:multint:domain}{क्षेत्रफल}
$\pi z$ है:
\[
V = \int_0^1 \pi z\,\dd z = \frac\pi2 .
\]
\end{solution}

\begin{solution}{exo:b2:multint:7}
गोलीय निर्देशांकों में आयतन-अवयव
$r^2\cos\varphi\,\dd r\,\dd\theta\,\dd\varphi$
है (\cref{ex:b2:multint:spherical}), और कोणीय भाग $4\pi$ तक समाकलित होता है ($\theta$ से $2\pi$,
$\int_{-\pi/2}^{\pi/2}\cos = 2$)। खोल का आयतन
\[
\int_a^b 4\pi r^2\,\dd r = \frac43\pi\bigl(b^3 - a^3\bigr).
\]
है। दूसरे समाकल के लिए समाकल्य $1/r$ केवल $r$ पर निर्भर करता है:
\[
\iiint_B \frac{\dd x\,\dd y\,\dd z}{r}
= \int_0^R 4\pi r^2\cdot\frac1r\,\dd r
= 4\pi\,\frac{R^2}{2} = 2\pi R^2 .
\]
(समाकल्य मूल बिंदु पर फट पड़ता है, पर हानिरहित ढंग से: $r^2/r = r$ \omterm{def:b2:metric:continuity}{संतत}
है --- खोलों $\varepsilon \leq r
\leq R$ पर समाकल $\varepsilon \to 0$ होने पर अभिसरण करता है, और कथन का
ठीक-ठीक अर्थ यही है। इस प्रकार का परिकलन न्यूटन की उस प्रमेय की ओर
पहला पग है कि कोई समांगी गोलक अपने केंद्र पर रखे बिंदु-द्रव्यमान की
तरह आकर्षित करता है।)
\end{solution}

\begin{solution}{exo:b2:multint:8}
समाकल्य $g(t; x, y) = xP(tx, ty) + yQ(tx, ty)$ $(x, y)$ में $\mathcal{C}^1$ है, $t$ में \omterm{def:b2:metric:continuity}{संतत} है, और उसके आंशिक अवकलज
$[0,1] \times U$ पर \omterm{def:b2:metric:continuity}{संतत} हैं; समाकल चिह्न के भीतर अवकलन (\cref{ch:b2:integration}, जो \omterm{def:b2:metric:compact}{संहत}
$t$-अंतराल $[0,1]$ पर लगाया गया है, जहाँ प्रभुत्व स्वतः मिल जाता है)
देता है
\[
f_x(x, y)
= \int_0^1 \bigl(P(tx, ty) + tx\,P_x(tx, ty)
+ ty\,Q_x(tx, ty)\bigr)\dd t .
\]
संवृतता $Q_x = P_y$ का उपयोग करने पर:
\[
tx\,P_x(tx, ty) + ty\,P_y(tx, ty)
= t\,\frac{\dd}{\dd t}\bigl[P(tx, ty)\bigr] ,
\]
अतः समाकल्य $P(tx, ty) + t\frac{\dd}{\dd t}P(tx, ty) =
\frac{\dd}{\dd t}\bigl[t\,P(tx, ty)\bigr]$ है और
\[
f_x(x, y) = \Bigl[t\,P(tx, ty)\Bigr]_0^1 = P(x, y) .
\]
सममित रूप से $f_y = Q$ (वही परिकलन, जहाँ $P_y = Q_x$ दूसरी तरह बरता गया)।
ध्यान दीजिए कि परिकल्पना कहाँ प्रवेश करती है: $f$ खंड $[0, M]$ के
अनुदिश समाकलन से परिभाषित होता है, और वह खंड $U$ में ठीक इसलिए
पड़ता है कि $U$ तारकाकार है।
\end{solution}

\begin{solution}{exo:b2:multint:9}
पट्टी $[0, A] \times [0, \infty)$ पर फलन $(x, y)
\mapsto e^{-xy}\sin x$ किसी भोले अर्थ में $y =
\infty$ तक \omterm{def:b2:funcseq:def}{एकसमान रूप से}
निरपेक्षतः समाकलनीय नहीं है, पर हर पुनरावृत्त समाकल अभिसरण करता
है और उनकी समता $[0, A]
\times [0, B]$ पर फ़ूबिनी तथा किसी सीमा $B \to \infty$ से निकलती है
(पुच्छ $\int_0^A\int_B^\infty e^{-xy}\abs{\sin x}\,\dd y\,\dd x \leq
\int_0^A \frac{e^{-Bx}\abs{\sin x}}{x}\dd x \leq \int_0^A e^{-Bx}
\dd x\to 0$, जहाँ $\abs{\sin x} \leq x$ का उपयोग हुआ)।

\emph{पहले $y$:} $x
> 0$ के लिए $\int_0^\infty e^{-xy}\,\dd y = \frac1x$, अतः पहला समाकल $\int_0^A \frac{\sin x}{x}\,\dd x$ है।

\emph{पहले $x$:} खंडशः दो बार समाकलन (या $\int_0^A e^{(i - y)x}\dd x$ का काल्पनिक भाग लेना)
देता है
\[
\int_0^A e^{-xy}\sin x\,\dd x
= \frac{1 - e^{-Ay}(y\sin A + \cos A)}{1 + y^2} .
\]
$[0, \infty)$ पर $y$ में समाकलन करने पर पद $\int_0^\infty
\frac{\dd y}{1 + y^2} = \frac\pi2$ अलग हो जाता है:
\[
\int_0^A \frac{\sin x}{x}\,\dd x
= \frac{\pi}{2}
- \int_0^\infty e^{-Ay}\,\frac{y\sin A + \cos A}{1 + y^2}\,\dd y .
\]
शेष $A \to \infty$ होने पर $\int_0^\infty e^{-Ay}\frac{y +
1}{1 + y^2}\dd y \leq \int_0^\infty e^{-Ay}\cdot\frac{1+y}{1+y^2}
\,\dd y \to 0$ से परिबद्ध है (प्रभावी अभिसरण, या कच्चा
परिबंध $\frac{1 + y}{1 + y^2} \leq \frac32$, जो $\frac{3}{2A}$ देता है)। अतः $\int_0^\infty\frac{\sin x}{x}\dd x =
\frac\pi2$ --- वही मान जो \cref{ch:b2:integration} में
किसी प्राचल-समाकल के अवकलन से मिला था; यहाँ उसकी जगह फ़ूबिनी काम कर
देती है।
\end{solution}

\begin{solution}{exo:b2:multint:10}
\omterm{def:b2:curves:length}{चाप-लंबाई} $s \in [0, 2\pi]$ से प्राचलन कीजिए, अतः $x'^2 + y'^2 = 1$, और इस तरह स्थानांतरित
कीजिए कि $\int_0^{2\pi} x(s)\,\dd s = 0$ हो जाए। \cref{cor:b2:multint:area} से,
\[
2A = \oint x\,\dd y - y\,\dd x
= \int_0^{2\pi}\bigl(xy' - yx'\bigr)\dd s .
\]
आवर्त पर $\oint y\,\dd x$ का खंडशः समाकलन करने पर (सीमांत पद आवर्तिता से कट
जाते हैं) $-\int yx' = \int y'x$, अतः वस्तुतः $2A = 2\int_0^{2\pi}xy'\,\dd s$। तब $2xy' \leq x^2 + y'^2$ देता है
\[
2A \leq \int_0^{2\pi}\bigl(x^2 + y'^2\bigr)\dd s
= \int_0^{2\pi} x^2 + \int_0^{2\pi}\bigl(1 - x'^2\bigr)
= 2\pi - \int_0^{2\pi}\bigl(x'^2 - x^2\bigr)\dd s .
\]
विर्टिंगर असमिका (\cref{ch:b2:fourier} के अभ्यास: शून्य माध्य वाले $2\pi$-आवर्ती
$\mathcal{C}^1$ फलन के लिए $\int x^2 \leq \int x'^2$) अंतिम समाकल को अऋणात्मक बना देती है:
$A \leq \pi$। समता के लिए विर्टिंगर में समता ($x(s) =
a\cos s + b\sin s$) तथा $2xy' \leq x^2 + y'^2$ में समता
(\omterm{def:b2:funcseq:def}{बिंदुवार} $y' = x$) चाहिए, जो $y = a\sin s - b\cos s + c$ को बाध्य कर देता है: अर्थात् वक्र
इकाई वृत्त है (उपयुक्त रूप से केंद्रित)। और चूँकि \omterm{def:b2:curves:length}{लंबाई} $L$ का
कोई वक्र मापन से \omterm{def:b2:curves:length}{लंबाई} $2\pi$ तक लाया जा सकता है, व्यापक कथन $A \leq
\frac{L^2}{4\pi}$
है: दिए गए परिमाप वाले सारे संवृत वक्रों में वृत्त सबसे बड़ा
\omterm{def:b2:multint:domain}{क्षेत्रफल} घेरता है।
\end{solution}

\begin{solution}{exo:b2:multint:11}
गोलीय निर्देशांकों में $z = r\sin\varphi$ तथा $\dd x\,\dd
y\,\dd z = r^2\cos\varphi\,\dd r\,\dd\theta\,\dd\varphi$:
\[
\iiint_B z^2
= \int_0^R r^4\,\dd r\int_0^{2\pi}\dd\theta
\int_{-\pi/2}^{\pi/2}\sin^2\varphi\cos\varphi\,\dd\varphi
= \frac{R^5}5\cdot2\pi\cdot
\Bigl[\frac{\sin^3\varphi}3\Bigr]_{-\pi/2}^{\pi/2}
= \frac{4\pi R^5}{15}.
\]
निर्देशांकों के क्रमचय के अंतर्गत गोलक की सममिति से $\iiint_B x^2 = \iiint_B y^2 = \iiint_B z^2$, अतः $\iiint_B(x^2
+ y^2 + z^2) = 3\cdot\frac{4\pi R^5}{15} = \frac{4\pi R^5}5$।
खोल-जाँच: $\int_0^R r^2\cdot 4\pi r^2\,\dd r = \frac{4\pi
R^5}5$ --- समाकल्य $r^2$ त्रिज्या $r$ के गोले पर अचर है,
जिसका \omterm{def:b2:multint:domain}{क्षेत्रफल} $4\pi r^2$ है।
\end{solution}

\begin{solution}{exo:b2:multint:12}
$(x_i, y_i)$ से $(x_{i+1}, y_{i+1})$ तक की भुजा का $\gamma(t) = \bigl((1-t)x_i + tx_{i+1},\ (1-t)y_i +
ty_{i+1}\bigr)$ से प्राचलन कीजिए। $\frac12\oint(x\,\dd y - y\,\dd x)$ में उसका
योगदान
\[
\frac12\int_0^1\Bigl(\bigl((1-t)x_i +
tx_{i+1}\bigr)(y_{i+1} - y_i) - \bigl((1-t)y_i +
ty_{i+1}\bigr)(x_{i+1} - x_i)\Bigr)\dd t ,
\]
है, और चूँकि $\int_0^1\bigl((1-t)u + tv\bigr)\dd t = \frac{u +
v}2$, यह
\[
\frac14\Bigl((x_i + x_{i+1})(y_{i+1} - y_i) - (y_i +
y_{i+1})(x_{i+1} - x_i)\Bigr)
= \frac12\bigl(x_iy_{i+1} - x_{i+1}y_i\bigr),
\]
के बराबर है, क्योंकि मिश्रित पद कट जाते हैं। $m$ भुजाओं पर जोड़ने
पर \cref{cor:b2:multint:area} से फ़ीता सूत्र मिल जाता है। त्रिभुज $(0,0), (1,0), (0,1)$: $\frac12\bigl((0\cdot0 - 1\cdot0) +
(1\cdot1 - 0\cdot0) + (0\cdot0 - 0\cdot1)\bigr) = \frac12$, जो सही
\omterm{def:b2:multint:domain}{क्षेत्रफल} है।
\end{solution}

\begin{solution}{pb:b2:multint:1}
\textbf{1.} गोलक $B_n(R)$ का वर्णन पुनरावृत्त सीमाओं $-R \leq x_n \leq R$, फिर $\abs{x_{n-1}} \leq \sqrt{R^2 -
x_n^2}$,
इत्यादि से होता है; $n$ एकचर समाकलों में से हर एक में $x_i = Ru_i$
प्रतिस्थापित करने पर हर एक $R$ से गुणित हो जाता है और सीमाएँ $B_n(1)$
की सीमाओं पर भेज दी जाती हैं: $v_n(R) = R^n\,v_n(1) =
V_nR^n$।

\textbf{2.} $x_n = t$ के अनुदिश काटने पर: $B_n(1)$ की फाँक गोलक $B_{n-1}\bigl(\sqrt{1 - t^2}\bigr)$ है, अतः
प्रश्न 1 से,
\[
V_n = \int_{-1}^1 v_{n-1}\bigl(\sqrt{1 - t^2}\bigr)\,\dd t
= V_{n-1}\int_{-1}^{1}(1 - t^2)^{\frac{n-1}2}\,\dd t .
\]

\textbf{3.} $\intcc{-\pi/2}{\pi/2}$ पर $t = \sin\theta$, $\dd t =
\cos\theta\,\dd\theta$ तथा $(1 - t^2)^{\frac{n-1}2} =
\cos^{n-1}\theta$ के साथ:
\[
\int_{-1}^1(1 - t^2)^{\frac{n-1}2}\dd t
= \int_{-\pi/2}^{\pi/2}\cos^n\theta\,\dd\theta
= 2\int_0^{\pi/2}\cos^n\theta\,\dd\theta = 2W_n,
\]
और $\theta \mapsto \frac\pi2 - \theta$ $W_n$ के ज्या तथा कोज्या रूपों की अदला-बदली कर देता है।

\textbf{4.} $\sin^n = \sin^{n-2}(1 - \cos^2)$ लिखिए और $\int\sin^{n-2}\cos\cdot\cos$ का खंडशः समाकलन कीजिए ($v =
\frac{\sin^{n-1}}{n-1}$):
\[
W_n = W_{n-2} - \frac{W_n}{n-1}
\quad\Longrightarrow\quad
W_n = \frac{n-1}{n}W_{n-2}.
\]
इसलिए $nW_nW_{n-1} = (n-1)W_{n-1}W_{n-2}$: अनुक्रम $(nW_nW_{n-1})$ अचर है और $1\cdot W_1W_0 =
1\cdot\frac\pi2$ के बराबर, अतः $W_nW_{n-1} = \frac{\pi}{2n}$।

\textbf{5.} प्रश्न 2 तथा 3 दो बार $V_n = 2W_nV_{n-1}$ देते हैं:
\[
V_n = 2W_n\cdot 2W_{n-1}\,V_{n-2}
= 4\,\frac{\pi}{2n}\,V_{n-2} = \frac{2\pi}n\,V_{n-2}.
\]

\textbf{6.} $V_0 = 1$ से: $V_{2k} = \frac{2\pi}{2k}V_{2k-2}
= \frac\pi kV_{2k-2}$, अतः आगमन से $V_{2k} = \frac{\pi^k}{k!}$। $V_1 = 2$ से: $V_{2k+1} =
\frac{2\pi}{2k+1}V_{2k-1}$, अतः
\[
V_{2k+1} = 2\prod_{j=1}^k\frac{2\pi}{2j+1}
= \frac{2^{k+1}\pi^k}{1\cdot3\cdot5\cdots(2k+1)} .
\]

\textbf{7.} $V_1 = 2$, $V_2 = \pi \approx 3.142$, $V_3 =
\frac{4\pi}3 \approx 4.189$, $V_4 = \frac{\pi^2}2 \approx
4.935$, $V_5 = \frac{8\pi^2}{15} \approx 5.264$, $V_6 =
\frac{\pi^3}6 \approx 5.168$, $V_7 = \frac{16\pi^3}{105}
\approx 4.725$। एक-पग अनुपात $V_n/V_{n-1} = 2W_n$
है, और $(W_n)$ ह्रासमान है (\omterm{def:b2:funcseq:def}{बिंदुवार} $\sin^n \leq \sin^{n-1}$)। अब $2W_5 = 2\cdot\frac45\cdot\frac23 =
\frac{16}{15} > 1$ जबकि $2W_6 =
2\cdot\frac56\cdot\frac34\cdot\frac12\cdot\frac\pi2 =
\frac{5\pi}{16} < 1$:
अनुपात $n = 5$ तक $1$ से अधिक हैं और $n = 6$ से आगे $1$ से कम ---
अर्थात् $(V_n)$ अपने अधिकतम $V_5$ तक बढ़ता है और फिर घटता है।

\textbf{8.} $n \geq 13 > 4\pi$ के लिए: $V_n/V_{n-2} = 2\pi/n <
\tfrac12$, अतः किसी नियत अचर के साथ $V_{n} \leq C\cdot 2^{-n/2}$; और
बेहतर, किसी भी $q > 0$ के लिए बड़े $n$ हेतु $2\pi/n < q^2$, अतः $V_n/q^n \to 0$: क्षय हर
गुणोत्तर अनुक्रम को हरा देता है। $\sum V_n$ का अभिसरण अनुपात $V_n/V_{n-2} \to 0$ से
निकलता है (किसी कोटि से आगे गुणोत्तर श्रेणी से तुलना कीजिए)।

\textbf{9.} $\sum_{k\geq0}V_{2k}x^{2k} =
\sum_{k\geq0}\frac{(\pi x^2)^k}{k!} = \eu^{\pi x^2}$, अर्थात् चरघातांकी श्रेणी (\cref{ch:b2:powerseries}), जो हर $x$ के लिए
अभिसारी है। $x = 1$ पर: $\sum_kV_{2k} = \eu^\pi \approx
23.14$।

\textbf{10.} घन $\intcc{-R}R^n$ पर समाकल्य गुणनफल $\prod_i\eu^{-x_i^2}$ है, अतः पुनरावृत्त समाकल
गुणनखंडित हो जाता है: $\bigl(\int_{-R}^R\eu^{-t^2}\dd t\bigr)^n$। $R \to \infty$ लेने पर और
$\int_\R\eu^{-t^2}\dd t = \sqrt\pi$
(\cref{ex:b2:multint:polar}) का उपयोग करने पर: $I_n = \pi^{n/2}$।

\textbf{11.} $t = u^2$ $\Gamma(\tfrac12) =
\int_0^\infty t^{-1/2}\eu^{-t}\dd t =
2\int_0^\infty\eu^{-u^2}\dd u = \sqrt\pi$ देता है। $\Gamma(s+1) = s\Gamma(s)$ को दोहराने पर: $\Gamma(k+1) = k!\,\Gamma(1) =
k!$, और
\[
\Gamma\Bigl(k + \frac32\Bigr)
= \Bigl(k + \frac12\Bigr)\Bigl(k - \frac12\Bigr)\cdots
\frac12\cdot\Gamma\Bigl(\frac12\Bigr)
= \frac{(2k+1)(2k-1)\cdots1}{2^{k+1}}\,\sqrt\pi .
\]

\textbf{12.} $F_n = \pi^{n/2}/\Gamma(\frac n2 + 1)$ रखिए। चूँकि $\Gamma(\frac n2 + 1) = \frac n2\,\Gamma(\frac n2) =
\frac n2\,\Gamma(\frac{n-2}2 + 1)$, हमें $F_n =
\frac{2\pi}nF_{n-2}$ मिलता है: यानी वही
पुनरावृत्ति जो $V_n$ है (प्रश्न 5)। आधार: $F_1 = \sqrt\pi/\Gamma(\frac32) =
\sqrt\pi/(\frac{\sqrt\pi}2) = 2 = V_1$ तथा $F_2 =
\pi/\Gamma(2) = \pi = V_2$। आगमन से सब
$n$ के लिए $V_n = F_n$; प्रश्न 11 इसे वापस प्रश्न 6 के दोनों संवृत रूपों
में बदल देता है।

\textbf{13.} $r = \sqrt t$ के साथ: $\int_0^\infty
\eu^{-r^2}r^{n-1}\dd r = \frac12\int_0^\infty
t^{\frac n2 - 1}\eu^{-t}\dd t = \frac12\Gamma(\frac n2)$. इसलिए
\[
n\,V_n\int_0^\infty\eu^{-r^2}r^{n-1}\dd r
= V_n\cdot\frac n2\,\Gamma\Bigl(\frac n2\Bigr)
= V_n\,\Gamma\Bigl(\frac n2 + 1\Bigr) = \pi^{n/2} = I_n .
\]
दोनों पक्ष सिद्ध हो जाने पर सर्वसमिका को गाउस समाकल के खोल-अपघटन
के रूप में \emph{पढ़ा} जा सकता है: त्रिज्या $r$ का गोला \omterm{def:b2:multint:domain}{क्षेत्रफल}
$nV_nr^{n-1}$ ढोता है, और गाउसी भार $\eu^{-r^2}$ खोलों पर समाकलित होता है।

\textbf{14.} $s_0 = V_1 = 2$ ($0$-गोला दो बिंदु हैं), $s_1 = 2V_2 = 2\pi$, $s_2 = 3V_3 = 4\pi$, $s_3 = 4V_4 =
2\pi^2$; और $\int_0^R s_{n-1}r^{n-1}\dd r = V_nR^n =
v_n(R)$:
अर्थात् \omterm{def:b2:multint:domain}{क्षेत्रफल} आयतन का त्रिज्य अवकलज है।

\textbf{15.} $n = 2k$ के लिए स्टर्लिंग (\cref{thm:b2:comparison:stirling}) $k! \sim
\sqrt{2\pi k}\,(k/\eu)^k$ देता है, अतः
\[
V_{2k} = \frac{\pi^k}{k!}
\sim \frac{(\pi\eu/k)^k}{\sqrt{2\pi k}}
= \frac1{\sqrt{\pi n}}\Bigl(\frac{2\pi\eu}n\Bigr)^{n/2}
\qquad (n = 2k).
\]
विषम $n$ के लिए: $V_{2k+1} = 2W_{2k+1}V_{2k} \leq 2V_{2k}$, अतः वही अति-गुणोत्तर क्षय-परिबंध टिके रहते
हैं (किसी गुणक $2$ तथा घातांक में एक के खिसकाव तक) --- यानी हर
$q >
0$ के लिए $V_n = o(q^n)$।

\textbf{16.} $V_2/4 = \pi/4 \approx 0.785$; $V_3/8 = \pi/6
\approx 0.524$; $V_{10}/2^{10} = \frac{\pi^5}{120\cdot1024}
\approx 0.0025$। सामान्यतः $\frac{V_n/2^n}{V_{n-2}/2^{n-2}}
= \frac{2\pi}{4n} = \frac{\pi}{2n} \to 0$: अनुपात
(अति-गुणोत्तर रूप से) $0$ की ओर जाता है। अंतर्लिखित गोलक लुप्त
होता हुआ अंश घेरता है: घन का आयतन उसके कोनों में पलायन कर जाता है।

\textbf{17.} प्रश्न 1 से त्रिज्या $1 -
\varepsilon$ के भीतरी \omterm{pb:b2:multint:1}{गोलक का आयतन} $V_n(1-\varepsilon)^n$
है, अतः बाहरी खोल अंश $1 - (1 - \varepsilon)^n \to 1$ ढोता है। $n = 100$ के लिए $\varepsilon = 0.01$: $(0.99)^{100} =
\eu^{100\ln0.99} \approx \eu^{-1.005} \approx 0.366$: अर्थात्
गोलक का लगभग $63\%$ भाग उसके पृष्ठ से $1\%$ के भीतर पड़ता है।

\textbf{18.} $(W_n)$ घटता है, अतः $W_n \leq W_{n-1} \leq
W_{n-2} = \frac{n}{n-1}W_n$: दबाव से $W_{n-1}/W_n \to 1$। $W_nW_{n-1} = \frac\pi{2n}$ से गुणा करने
पर: $W_n^2 \sim
\frac\pi{2n}$, अर्थात् $W_n \sim \sqrt{\pi/(2n)}$। निचला परिबंध: $W_n^2 \geq W_nW_{n+1} = \frac{\pi}{2(n+1)}$, अतः हर $n$ के लिए $W_n \geq
\sqrt{\pi/(2(n+1))}$।

\textbf{19.} अंश: $1 - x^2 \leq \eu^{-x^2}$ $(1 -
x^2)^{\frac{n-1}2} \leq \eu^{-(n-1)x^2/2}$ देता है, और $a =
\frac{n-1}2$ के साथ,
\[
\int_\delta^1(1 - x^2)^{\frac{n-1}2}\dd x
\leq \int_\delta^\infty\eu^{-ax^2}\dd x
\leq \int_\delta^\infty\frac x\delta\,\eu^{-ax^2}\dd x
= \frac{\eu^{-a\delta^2}}{2a\delta}
= \frac{\eu^{-(n-1)\delta^2/2}}{(n-1)\delta}.
\]
हर: प्रश्न 18 से $2W_n \geq \sqrt{2\pi/(n+1)}$। भाग देने पर प्रदर्शित परिबंध मिल जाता है।
$\delta =
s/\sqrt{n-1}$ के लिए वह $n$ में \omterm{def:b2:funcseq:def}{एकसमान रूप से} $\sqrt{\tfrac{n+1}{2\pi(n-1)}}\;\eu^{-s^2/2}/s = O\bigl(
\eu^{-s^2/2}/s\bigr)$ बन जाता है: अर्थात्
मध्यम रूप से बड़े $s$ के लिए पट्टी $\abs{x_1} \leq s/\sqrt{n-1}$ के बाहर लगभग कोई आयतन नहीं
है --- और सममिति से वही हर दिशा के लिए टिकता है।

\textbf{20.} दोनों कथन साथ-साथ रह सकते हैं क्योंकि वे एक ही बिंदु
के भिन्न निर्देशांकों का वर्णन करते हैं। $B_n(1)$ के लगभग हर बिंदु का
\omterm{def:b2:nvs:norm}{मानदंड} $1$ के निकट है (प्रश्न 17: गोले के पास त्रिज्य संकेंद्रण),
फिर भी उसका हर $n$ निर्देशांक छोटा है, कोटि $1/\sqrt n$ का (प्रश्न 19),
और यह संगत है क्योंकि आकार $1/\sqrt n$ वाले $n$ निर्देशांकों का \omterm{def:b2:nvs:norm}{मानदंड}
कोटि $1$ का होता है। ऊँची-विमीय आयतन वहीं संकेंद्रित होता है जहाँ
सारे निर्देशांक \omterm{def:b2:nvs:norm}{मानदंड} का बजट बराबर-बराबर बाँटते हैं --- गोले के
पास, पर हर निर्देशांक-अक्ष के ध्रुव से दूर।

\textbf{21.} $\Delta_n$ को $x_n = t \in \intcc01$ पर काटिए: फाँक $\{x' \in \R^{n-1} : x_i \geq 0,\ \sum x_i \leq 1 -
t\} = (1-t)\Delta_{n-1}$ है, जिसका आयतन
समघातता से $(1-t)^{n-1}\operatorname{vol}(\Delta_{n-1})$ है। अतः
\[
\operatorname{vol}(\Delta_n) =
\operatorname{vol}(\Delta_{n-1})\int_0^1(1 - t)^{n-1}\dd t =
\frac{\operatorname{vol}(\Delta_{n-1})}{n}
\quad\Longrightarrow\quad
\operatorname{vol}(\Delta_n) = \frac1{n!}\,.
\]

\textbf{22.} $2^n$ चिह्न-अष्टांश $C_n$ को $\Delta_n$ की $2^n$ प्रतियों में काट
देते हैं (निर्देशांक अतिसमतलों पर नगण्य अतिव्यापन के साथ): $\operatorname{vol}(C_n) = \frac{2^n}{n!}$।
यदि $\sum\abs{x_i} \leq 1$, तो $\sum x_i^2 \leq
\bigl(\sum\abs{x_i}\bigr)^2 \leq 1$: $C_n \subseteq B_n(1)$; और $B_n(1) \subseteq \intcc{-1}1^n$, क्योंकि $\abs{x_i} \leq
\norm x$। इसलिए $\frac{2^n}{n!} \leq V_n \leq 2^n$ ---
जो प्रश्न 15 से संगत है, जो $V_n$ को क्रमगुणित तथा गुणोत्तर मापों
के बीच रखता है।

\textbf{23.} इकाई चक्रिका के $(x, y)$ के लिए $B_4(1)$ की फाँक $(z, w)$-समतल में
त्रिज्या $\sqrt{1 - x^2 - y^2}$ की चक्रिका है, जिसका \omterm{def:b2:multint:domain}{क्षेत्रफल} $\pi(1 - x^2 - y^2)$ है। ध्रुवीय
निर्देशांकों में:
\[
V_4 = \iint_{x^2+y^2\leq1}\pi(1 - x^2 - y^2)\,\dd x\,\dd y
= \pi\int_0^{2\pi}\!\!\int_0^1(1 - \rho^2)\rho\,
\dd\rho\,\dd\alpha
= \pi\cdot2\pi\cdot\frac14 = \frac{\pi^2}2 ,
\]
जो प्रश्न 6 से मेल खाता है।

\textbf{24.} प्रायिकता आयतन-अनुपात $\dfrac{V_{20}}{2^{20}} = \dfrac{\pi^{10}}{10!\cdot2^{20}}
\approx \dfrac{0.0258}{1\,048\,576} \approx
2.5\cdot10^{-8}$ है। पहली सफलता तक खींचों की
संख्या उसके व्युत्क्रम की कोटि की है, लगभग $4\cdot10^7$: जो
अस्वीकृति-प्रतिचयक चक्रिका के लिए ख़ूब चला था ($\pi/4$ सफलताएँ), वह
विमा $20$ में बेकार है --- एक पंक्ति में विमीयता का अभिशाप।

\textbf{25.} पहले रास्ते (भाग I तथा II) ने बरता: पुनरावृत्त समाकल का
फ़ूबिनी-प्रकार का कटान, हर निर्देशांक में एकचर प्रतिस्थापन
(समघातता), और \omterm{pb:b2:multint:1}{वालिस समाकल} --- अर्थात् विशुद्ध एकचर कलन तथा आगमन।
दूसरे रास्ते (भाग III) ने बरता: $I_n$ की गुणनफल संरचना के लिए
फ़ूबिनी, \cref{ex:b2:multint:polar} के गाउस समाकल के द्वारा ध्रुवीय चर-परिवर्तन, और $\Gamma$
फलन का फलनिक समीकरण। वे $V_n =
\pi^{n/2}/\Gamma(\frac n2 + 1)$ पर मिलते हैं, जहाँ स्टर्लिंग (\cref{thm:b2:comparison:stirling})
सूत्र को अनंतस्पर्शियों में बदल देता है। वर्ष 3 का खंड यह सब लेबेग
समाकल पर फिर से खड़ा करता है: वहाँ फ़ूबिनी तथा चर-परिवर्तन व्यापक
समाकलनीय फलनों के लिए प्रमेय हैं, गोलीय निर्देशांक हर विमा में
विद्यमान हैं, और वही गोलक-आयतन गुणनफल-माप तथा स्टर्लिंग वाली
समस्याओं के हल किए गए लाभांश के रूप में फिर प्रकट होते हैं --- जहाँ
हमारे हाथ से बनाए दबावों की जगह प्रभावी अभिसरण ले लेता है।
\end{solution}
