\section*{Hoofdstuk \ref{ch:b2:randomvar} --- Discrete toevalsveranderlijken}

\begin{solution}{exo:b2:randomvar:1}
\emph{Binomiaal:} $X = \sum_{i=1}^n X_i$ met \omterm{def:b2:proba:independence}{onafhankelijke}
bernoulli-veranderlijken $X_i$; $V(X_i) = \E(X_i^2) - \E(X_i)^2 = p
- p^2$, en \omterm{def:b2:randomvar:variance}{varianties} van \omterm{def:b2:proba:independence}{onafhankelijke} veranderlijken tellen op
(\cref{thm:b2:randomvar:variancerules}):
\[
\E(X) = np, \qquad V(X) = np(1-p) .
\]
\emph{Poisson:} $\E\bigl(X(X-1)\bigr) =
\sum_{k\geq2}k(k-1)e^{-\lambda}\frac{\lambda^k}{k!} = \lambda^2
e^{-\lambda}\sum_{j\geq0}\frac{\lambda^j}{j!} = \lambda^2$, dus
\[
V(X) = \E(X^2) - \E(X)^2
= \lambda^2 + \lambda - \lambda^2 = \lambda .
\]
\emph{Meetkundig} ($q = 1 - p$): $\sum_{k\geq0}q^k =
\frac{1}{1-q}$ tweemaal binnen de schijf differentiëren
(\cref{ch:b2:powerseries}) geeft $\sum_{k\geq2}k(k-1)q^{k-2} =
\frac{2}{(1-q)^3}$, dus
\[
\E\bigl(X(X-1)\bigr) = pq\sum_{k\geq2}k(k-1)q^{k-2}
= \frac{2q}{p^2},
\qquad
V(X) = \frac{2q}{p^2} + \frac1p - \frac{1}{p^2}
= \frac{q}{p^2} = \frac{1-p}{p^2} .
\]
\end{solution}

\begin{solution}{exo:b2:randomvar:2}
\emph{Som:} voor $n \in \N$ geldt, wegens de disjunctheid en de
\omterm{def:b2:proba:independence}{onafhankelijkheid},
\[
\P(X + Y = n)
= \sum_{k=0}^n \P(X = k)\P(Y = n - k)
= e^{-(\lambda + \mu)}\frac{1}{n!}
\sum_{k=0}^n \binom nk \lambda^k\mu^{n-k}
= e^{-(\lambda+\mu)}\frac{(\lambda + \mu)^n}{n!}
\]
volgens het binomium: $X + Y \sim \mathcal{P}(\lambda + \mu)$.
\emph{Voorwaardelijke \omterm{def:b2:randomvar:law}{verdeling}:} voor $0 \leq k \leq n$ is
\[
\P(X = k \mid X + Y = n)
= \frac{\P(X = k)\P(Y = n - k)}{\P(X + Y = n)}
= \binom nk
\Bigl(\frac{\lambda}{\lambda+\mu}\Bigr)^{k}
\Bigl(\frac{\mu}{\lambda+\mu}\Bigr)^{n-k} ,
\]
de binomiale \omterm{def:b2:randomvar:law}{verdeling} $\mathcal{B}\bigl(n,
\frac{\lambda}{\lambda+\mu}\bigr)$: gegeven het totale aantal
``kiest'' elke \omterm{def:b2:proba:space}{gebeurtenis} \omterm{def:b2:proba:independence}{onafhankelijk} haar eerste bron met een
kans evenredig aan haar tempo.
\end{solution}

\begin{solution}{exo:b2:randomvar:3}
Wanneer er nog $k$ speeltjes ontbreken, brengt elke nieuwe doos met
kans $\frac kn$ een nieuw speeltje, \omterm{def:b2:proba:independence}{onafhankelijk} van het verleden:
de wachttijd $W_k$ tot het volgende nieuwe speeltje is meetkundig
$\mathcal{G}\bigl(\frac kn\bigr)$, met $\E(W_k) = \frac nk$, en
$T_n = W_n + W_{n-1} + \dots + W_1$ (de eerste doos geeft altijd
een nieuw speeltje: $W_n = 1$, in overeenstemming met $\E = n/n$).
Wegens de lineariteit is
\[
\E(T_n) = \sum_{k=1}^n \frac nk = n\sum_{k=1}^n\frac1k
\sim n\ln n ,
\]
met $\sum_{k\leq n}\frac1k = \ln n + \gamma + o(1)$
(\cref{ch:b2:comparison}). De laatste paar speeltjes verzamelen is
wat kost: de helft van de dozen gaat naar de laatste handvol.
\end{solution}

\begin{solution}{exo:b2:randomvar:4}
Puntsgewijs is $X(\omega) = \#\{n \geq 1 : X(\omega) \geq n\} =
\sum_{n\geq1}\mathbf{1}_{X \geq n}(\omega)$. De dubbele familie
$\bigl(\mathbf{1}_{X \geq n}(\omega)\,\P(\{\omega\})\bigr)_{n,
\omega}$ is niet-negatief, dus is Fubini voor families
(\cref{ch:b2:series}) onvoorwaardelijk van toepassing: eerst in $n$
sommeren geeft $\E(X)$, eerst in $\omega$ sommeren geeft $\sum_n
\P(X \geq n)$; de twee zijn tegelijk eindig en gelijk. Voor $X \sim
\mathcal{G}(p)$: $\P(X \geq n) = q^{n-1}$ ($q = 1-p$), dus $\E(X) =
\sum_{n\geq1}q^{n-1} = \frac{1}{1 - q} = \frac1p$.
\end{solution}

\begin{solution}{exo:b2:randomvar:5}
\emph{Symmetrie:} de $i$-de getrokken bal is een uniform
willekeurige bal uit de urne (elke van de $N$ ballen heeft evenveel
kans om op positie $i$ van de trekvolgorde te belanden), dus $\P(Y_i
= 1) = \frac MN = p$ en $\E(X) = np$ wegens de lineariteit --- geen
\omterm{def:b2:proba:independence}{onafhankelijkheid} nodig.

\emph{\omterm{def:b2:randomvar:variance}{Covariantie}:} voor $i \neq j$ is $\E(Y_iY_j) =
\P(\text{trekkingen } i, j \text{ beide wit}) =
\frac{M(M-1)}{N(N-1)}$ (geordende paren verschillende posities
krijgen uniform een geordend paar verschillende ballen). Bijgevolg
is
\[
\operatorname{Cov}(Y_i, Y_j)
= \frac{M(M-1)}{N(N-1)} - \frac{M^2}{N^2}
= \frac{M(N - M)}{N^2}\cdot\frac{-1}{N-1}
= -\frac{p(1-p)}{N-1} < 0 :
\]
een witte bal trekken maakt wit schaarser voor de andere
trekkingen. Volgens \cref{thm:b2:randomvar:variancerules} is
\[
V(X) = np(1-p) + n(n-1)\Bigl(-\frac{p(1-p)}{N-1}\Bigr)
= np(1-p)\,\frac{N - n}{N - 1} \leq np(1-p) :
\]
steekproeven zonder teruglegging hebben hetzelfde gemiddelde maar
een \emph{kleinere} \omterm{def:b2:randomvar:variance}{variantie} dan met teruglegging (gelijkheid
alleen voor $n = 1$), waarbij de negatieve correlaties als
stabilisator werken. Voor $n = N$ verdwijnt de \omterm{def:b2:randomvar:variance}{variantie}: de
telling is dan deterministisch.
\end{solution}

\begin{solution}{exo:b2:randomvar:6}
Uitwerken rond $m = \E(X)$:
\[
\E\bigl((X - c)^2\bigr)
= \E\bigl((X - m)^2\bigr) + 2(m - c)\,\E(X - m) + (m - c)^2
= V(X) + (m - c)^2 ,
\]
minimaal precies in $c = m$ met waarde $V(X)$ --- de
\omterm{def:b2:randomvar:expectation}{verwachtingswaarde} is de beste constante voorspeller in
kwadratisch gemiddelde.

Is $\P(X = m) = 1$, dan verdwijnt $(X - m)^2$ met kans $1$, dus
$V(X) = 0$ (de definiërende familie heeft nultermen behalve op een
nulverzameling). Omgekeerd, is $V(X) = 0$, dan geeft Chebyshev
(\cref{thm:b2:randomvar:markov}) dat $\P\bigl(\abs{X - m} \geq
\frac1n\bigr) \leq n^2\,V(X) = 0$ voor elke $n$; de \omterm{def:b2:proba:space}{gebeurtenissen}
$\bigl\{\abs{X - m} \geq \frac1n\bigr\}$ stijgen naar $\{X \neq
m\}$, dus levert de monotone \omterm{def:b2:metric:continuity}{continuïteit}
(\cref{thm:b2:proba:continuity}) dat $\P(X \neq m) = 0$.
\end{solution}

\begin{solution}{exo:b2:randomvar:7}
$\E(S_n) = \frac n2$ en $V(S_n) = \frac n4$.
\emph{Markov:} $\P\bigl(S_n \geq \frac{3n}4\bigr) \leq
\frac{n/2}{3n/4} = \frac23$ --- een constante grens, nutteloos voor
grote $n$.
\emph{Chebyshev:} de \omterm{def:b2:proba:space}{gebeurtenis} impliceert $\abs{S_n - \frac n2}
\geq \frac n4$, dus is de kans $\leq \frac{n/4}{(n/4)^2} =
\frac4n$ --- zij vervalt, maar slechts polynomiaal.
\emph{Chernoff:} wegens de \omterm{def:b2:proba:independence}{onafhankelijkheid} is $\E(e^{tS_n}) =
\prod_{i=1}^n\E(e^{tX_i}) = \bigl(\frac{1 + e^t}{2}\bigr)^n$, en
Markov toegepast op $e^{tS_n} \geq e^{3nt/4}$ geeft voor elke $t >
0$
\[
\P\Bigl(S_n \geq \frac{3n}4\Bigr)
\leq \Bigl(\frac{1 + e^t}{2}\Bigr)^n e^{-3nt/4}
= \exp\Bigl(n\bigl(\ln\tfrac{1 + e^t}{2} - \tfrac{3t}4\bigr)\Bigr).
\]
Minimaliseer de exponent: $\frac{\dd}{\dd t}\ln\frac{1+e^t}{2} =
\frac{e^t}{1 + e^t} = \frac34$ in $e^t = 3$, dat wil zeggen $t =
\ln 3$, wat
\[
\P\Bigl(S_n \geq \frac{3n}4\Bigr)
\leq \Bigl(\frac{4}{2}\Bigr)^n 3^{-3n/4}
= \bigl(2 \cdot 3^{-3/4}\bigr)^n \approx (0.877)^n
\]
geeft, exponentieel klein. De hiërarchie Markov $\to$ Chebyshev
$\to$ Chernoff is de standaardladder: elke sport past Markov toe op
een sneller groeiende functie van de veranderlijke.
\end{solution}

\begin{solution}{exo:b2:randomvar:8}
Volgens de overdrachtsstelling
(\cref{thm:b2:randomvar:transfer}) toegepast op
$f\bigl(\frac{S_n}{n}\bigr)$ met $S_n \sim \mathcal{B}(n, x)$:
\[
\E\Bigl[f\Bigl(\frac{S_n}{n}\Bigr)\Bigr]
= \sum_{k=0}^n f\Bigl(\frac kn\Bigr)\binom nk x^k(1-x)^{n-k}
= B_nf(x) .
\]
Houd $\delta > 0$ vast en splits $\abs{f(S_n/n) - f(x)}$ op de
\omterm{def:b2:proba:space}{gebeurtenis} $D = \bigl\{\abs{\frac{S_n}{n} - x} \geq
\delta\bigr\}$: buiten $D$ is het verschil hoogstens de
continuïteitsmodulus $\omega_f(\delta) = \sup_{\abs{s -
t}\leq\delta}\abs{f(s) - f(t)}$; op $D$ hoogstens $2\norm
f_\infty$. \omterm{def:b2:randomvar:expectation}{Verwachtingswaarden} nemen en Chebyshev gebruiken met
$V\bigl(\frac{S_n}{n}\bigr) = \frac{x(1-x)}{n} \leq \frac{1}{4n}$:
\[
\abs{B_nf(x) - f(x)}
\leq \E\,\abs{f(S_n/n) - f(x)}
\leq \omega_f(\delta)
+ 2\norm f_\infty\,\P(D)
\leq \omega_f(\delta) + \frac{2\norm f_\infty}{4n\delta^2} .
\]
De uniforme \omterm{def:b2:metric:continuity}{continuïteit} van $f$ op $[0, 1]$ maakt
$\omega_f(\delta) \to 0$: kies $\delta$ en dan $n$, en $B_nf \to f$
uniform --- de benaderingsstelling van Weierstrass uit
\cref{ch:b2:funcseq}, waarvan de ``telhulpstelling'' nu herkenbaar
is als de ongelijkheid van Chebyshev voor de binomiale \omterm{def:b2:randomvar:law}{verdeling}.
\end{solution}

\begin{solution}{exo:b2:randomvar:9}
Werk $S_n^4 = \sum_{i,j,k,l}X_iX_jX_kX_l$ uit en neem
\omterm{def:b2:randomvar:expectation}{verwachtingswaarden}. Wegens de \omterm{def:b2:proba:independence}{onafhankelijkheid} en het centreren
verdwijnt elke term met een index die precies eenmaal voorkomt
($\E(X_i) = 0$ zondert zich af). Overlevende termen: de $n$
diagonale termen $\E(X_i^4)$, en de termen die twee paren gelijke
indices koppelen, $\E(X_i^2X_j^2) = \E(X_1^2)^2$ voor $i \neq j$,
die $3n(n-1)$ maal voorkomen: kies het ongeordende paar waarden
($\binom n2$ manieren), en dan de $\frac{4!}{2!\,2!} = 6$ manieren
om hen over de vier plaatsen te verdelen --- $6\binom n2 =
3n(n-1)$. Bijgevolg is, met $\E(X_1^2)^2 \leq \E(X_1^4)$ (Jensen of
Cauchy--Schwarz),
\[
\E(S_n^4) = n\,\E(X_1^4) + 3n(n-1)\,\E(X_1^2)^2
\leq C n^2,
\qquad C = 4\,\E(X_1^4) .
\]
Markov op orde 4:
\[
\P\Bigl(\Bigl|\frac{S_n}{n}\Bigr| \geq \varepsilon\Bigr)
= \P\bigl(S_n^4 \geq n^4\varepsilon^4\bigr)
\leq \frac{Cn^2}{n^4\varepsilon^4}
= \frac{C}{n^2\varepsilon^4} ,
\]
een \omterm{def:b2:series:summable}{sommeerbare} reeks. Volgens Borel--Cantelli~1
(\cref{thm:b2:proba:borelcantelli}) heeft voor elke $j$ de
\omterm{def:b2:proba:space}{gebeurtenis} $B_j = \limsup_n\bigl\{\abs{S_n/n} \geq
\frac1j\bigr\}$ kans $0$, dus $\P\bigl(\bigcup_j B_j\bigr) = 0$
wegens de \omterm{def:b2:structures:countable}{aftelbare} subadditiviteit. Op het complement --- met kans
$1$ --- is er voor elke $j$ een $N$ met $\abs{S_n/n} < \frac1j$
voor alle $n \geq N$: precies $S_n/n \to 0$. De sterke wet van de
grote aantallen geldt onder een vierde moment; die hypothese
schrappen (de stelling van Kolmogorov) is werk voor bachelorjaar 3.
\end{solution}

\begin{solution}{exo:b2:randomvar:10}
$\P(M \leq k) = \bigl(\frac k6\bigr)^2$ (beide dobbelstenen
hoogstens $k$, \omterm{def:b2:proba:independence}{onafhankelijk}), dus $\P(M \geq k) = 1 -
\bigl(\frac{k-1}6\bigr)^2$ en
\[
\E(M) = \sum_{k=1}^6\P(M \geq k)
= 6 - \frac{0 + 1 + 4 + 9 + 16 + 25}{36}
= 6 - \frac{55}{36} = \frac{161}{36} \approx 4.47 ,
\]
ruim boven het gemiddelde $3.5$ van één dobbelsteen, zoals het een
maximum betaamt.
\end{solution}

\begin{solution}{exo:b2:randomvar:11}
Met $I_i = \mathbf 1_{\sigma(i) = i}$: $\P(\sigma(i) = i) =
\frac{(n-1)!}{n!} = \frac1n$, dus $\E(F_n) = n\cdot\frac1n = 1$.
Voor $i \neq j$: $\P(\sigma(i) = i, \sigma(j) = j) =
\frac{(n-2)!}{n!} = \frac1{n(n-1)}$, en dus
\[
\operatorname{Cov}(I_i, I_j) = \frac1{n(n-1)} - \frac1{n^2}
= \frac{1}{n^2(n-1)} .
\]
Volgens de gereedschapskist van de \omterm{def:b2:randomvar:variance}{variantie}
(\cref{thm:b2:randomvar:variancerules}) is
\[
V(F_n) = n\cdot\frac1n\Bigl(1 - \frac1n\Bigr)
+ n(n-1)\cdot\frac1{n^2(n-1)}
= 1 - \frac1n + \frac1n = 1 .
\]
Gemiddelde $1$, \omterm{def:b2:randomvar:variance}{variantie} $1$, onafhankelijk van $n$ --- in
overeenstemming met de poisson-limiet van het probleem van de
overeenkomsten (\cref{exo:b2:proba:5}).
\end{solution}

\begin{solution}{exo:b2:randomvar:12}
$T_n = \sum_{k=1}^nG_k$, waarbij $G_k \sim \mathcal G(k/n)$ de tijd
is tot een nieuw speeltje wanneer er $k$ ontbreken, en de fasen
\omterm{def:b2:proba:independence}{onafhankelijk} zijn. Bijgevolg is
\[
V(T_n) = \sum_{k=1}^n\frac{1 - k/n}{(k/n)^2}
\leq \sum_{k=1}^n\frac{n^2}{k^2}
\leq \frac{\pi^2}6\,n^2 ,
\]
volgens \cref{ex:b2:fourier:basel}. Met $\E(T_n) = nH_n$ en $H_n =
\sum_1^n\frac1k$ (\cref{exo:b2:randomvar:3}) geeft Chebyshev voor
$\varepsilon > 0$
\[
\P\bigl(\abs{T_n - nH_n} \geq \varepsilon\,n\ln n\bigr)
\leq \frac{\pi^2n^2/6}{\varepsilon^2n^2\ln^2 n}
= \frac{\pi^2}{6\,\varepsilon^2\ln^2n}
\xrightarrow[n\to\infty]{} 0 .
\]
Omdat $H_n \sim \ln n$, toont delen door $n\ln n$ dat $T_n/(n\ln n)
\to 1$ in kans: de fluctuaties van $T_n$ zijn van de orde $n$,
verwaarloosbaar tegenover het gemiddelde $n\ln n$.
\end{solution}

\begin{solution}{pb:b2:randomvar:1}
\textbf{1.} $\E(S_n) = \frac n2$ en Markov
(\cref{thm:b2:randomvar:markov}) geven $\P(S_n \geq an) \leq
\frac{n/2}{an} = \frac1{2a}$. Markov kent alleen het gemiddelde:
zij kan een veranderlijke die zich rond $n/2$ concentreert niet
onderscheiden van een die tussen $0$ en $n$ is uitgesmeerd, en
beprijst de staart dus alsof alle massa daar zou kunnen zitten.

\textbf{2.} De eerlijke binomiale \omterm{def:b2:randomvar:law}{verdeling} is \omterm{def:b2:quadratic:adjoint}{symmetrisch} om $n/2$
($S_n$ en $n - S_n$ hebben dezelfde \omterm{def:b2:randomvar:law}{verdeling}), dus zijn met $x =
n(a - \frac12) > 0$ de twee \omterm{def:b2:proba:space}{gebeurtenissen} $\{S_n - \frac n2 \geq
x\}$ en $\{S_n - \frac n2 \leq -x\}$ disjunct en even
waarschijnlijk: $\P(S_n \geq an) = \frac12\P(\abs{S_n - \frac n2}
\geq x)$. Chebyshev met $V(S_n) = \frac n4$:
\[
\P(S_n \geq an)
\leq \frac12\cdot\frac{n/4}{n^2(a - 1/2)^2}
= \frac1{8n(a - 1/2)^2},
\]
wat $\frac2n$ is in $a = \frac34$.

\textbf{3.} Wegens de \omterm{def:b2:proba:independence}{onafhankelijkheid} en de productstelling is
$\E(\eu^{tS_n}) = \bigl(\E \eu^{tX_1}\bigr)^n = \bigl(\frac{1 +
\eu^t}2\bigr)^n$. Markov toegepast op $\eu^{tS_n}$:
\[
\P(S_n \geq an) \leq \eu^{-tan}\Bigl(\frac{1 +
\eu^t}2\Bigr)^{\!n} = \exp\Bigl(n\bigl(\ln\tfrac{1 +
\eu^t}2 - ta\bigr)\Bigr).
\]
De afgeleide van de exponent naar $t$ is $\frac{\eu^t}{1 + \eu^t} -
a$, die verdwijnt in $\eu^t = \frac a{1-a}$, dat wil zeggen $t^* =
\ln\frac a{1-a} > 0$; daar is $\frac{1 + \eu^{t^*}}2 =
\frac1{2(1-a)}$ en is de exponent gelijk aan
\[
n\Bigl(-\ln 2 - \ln(1-a) - a\ln\frac a{1-a}\Bigr)
= -n\bigl(\ln2 + a\ln a + (1-a)\ln(1-a)\bigr) = -n\,I(a),
\]
met $I(\frac12) = 0$ en $I'(a) = \ln\frac a{1-a} > 0$ op
$\intoo{\frac12}1$: $I(a) > 0$. In $a = \frac34$: $\eu^{-I(3/4)} =
\frac12(\tfrac34)^{-3/4}(\tfrac14)^{-1/4} = 2\cdot3^{-3/4}$, de
grens van \cref{exo:b2:randomvar:7}.

\textbf{4.} De $n + 1$ getallen $\binom nja^j(1-a)^{n-j}$ sommeren
tot $1$, en het grootste is dat in $j = k = an$ (de modus van
$\mathcal B(n, a)$ is hier $\floor{(n+1)a} = k$). Een maximum van
$n + 1$ getallen die tot $1$ sommeren, is minstens $\frac1{n+1}$:
\[
\binom nk a^k(1-a)^{n-k} \geq \frac1{n+1}
\quad\Longrightarrow\quad
\binom nk \geq \frac{a^{-an}(1-a)^{-n(1-a)}}{n+1}
= \frac{\eu^{nH(a)}}{n+1}.
\]
Bijgevolg is $\P(S_n \geq an) \geq \binom{n}{an}2^{-n} \geq
\eu^{n(H(a) - \ln2)}/(n+1) = \eu^{-nI(a)}/(n+1)$: op de
polynomiale factor $n + 1$ na is de exponent van Chernoff de
waarheid.

\textbf{5.} $n = 100$, $a = \frac34$: Markov $\frac23$; Chebyshev
$\frac2{100} = 0.02$; Chernoff $(2\cdot3^{-3/4})^{100} =
\eu^{-100\,I(3/4)} \approx 2.1\cdot10^{-6}$, tegenover de exacte
$2.8\cdot10^{-7}$. Moraal: elk moment aan informatie deelt de grens
polynomiaal; het exponentiële moment verandert haar \emph{aard}.

\textbf{6.} $\cosh t = \sum_{k\geq0}\frac{t^{2k}}{(2k)!}$ en
$\eu^{t^2/2} = \sum_{k\geq0}\frac{t^{2k}}{2^kk!}$; de bewering
volgt term voor term uit $(2k)! \geq 2^kk!$, dat met inductie
geldt: $(2k)! = 2k(2k-1)\cdot(2k-2)! \geq 2k\cdot 2^{k-1}(k-1)! =
2^kk!\cdot(2k-1) \geq 2^kk!$.

\textbf{7.} Wegens de \omterm{def:b2:proba:independence}{onafhankelijkheid} is
$\E\bigl(\eu^{t\sum\varepsilon_i} \bigr) = (\cosh t)^n \leq
\eu^{nt^2/2}$, dus geeft Markov dat $\P(\sum\varepsilon_i \geq s)
\leq \eu^{nt^2/2 - ts}$; minimaliseren in $t = s/n$ levert
$\eu^{-s^2/(2n)}$.

\textbf{8.} Met $X_i = \frac{1 + \varepsilon_i}2$ is $\widehat p_n
- \frac12 = \frac1{2n}\sum\varepsilon_i$, dus $\{\widehat p_n -
\frac12 \geq \delta\} = \{\sum\varepsilon_i \geq 2n\delta\}$ en
geeft vraag 7 de grens $\eu^{-(2n\delta)^2/(2n)} =
\eu^{-2n\delta^2}$. De symmetrische \omterm{def:b2:proba:space}{gebeurtenis} heeft dezelfde
grens, waaruit de factor $2$ voor $\abs{\widehat p_n - \frac12}
\geq \delta$.

\textbf{9.} $\E(\eu^{tX}) = \sum_x\eu^{tx}\P(X = x)$ is een reeks
van gladde functies van $t$ waarvan de termsgewijze afgeleiden op
elk \omterm{def:b2:metric:compact}{compact} $t$-interval worden gedomineerd door
$\eu^{\abs t}\P(X = x)$ (want $0 \leq x \leq 1$): volgens de
differentiatiestelling voor normaal convergente reeksen
(\cref{thm:b2:funcseq:differentiation}) is zij tweemaal
\omterm{def:b2:diffcalc:differential}{differentieerbaar}, en de quotiëntregel geeft $\psi' = \E_t(X)$ en
$\psi'' = \E_t(X^2) - \E_t(X)^2$, waarbij $\E_t$ de
\omterm{def:b2:randomvar:expectation}{verwachtingswaarde} voor de herwogen gewichten
$\eu^{tx}\P(X{=}x)/\E(\eu^{tX})$ is --- niet-negatief, sommerend
tot $1$, gedragen door dezelfde waarden $x \in \intcc01$. Een
\omterm{def:b2:randomvar:variance}{variantie} van een veranderlijke met waarden in $\intcc01$ is
hoogstens $\frac14$: volgens \cref{exo:b2:randomvar:6} is zij
$\min_c\E_t((X - c)^2) \leq \E_t\bigl((X - \tfrac12)^2\bigr) \leq
\tfrac14$. Taylor met integraalrest, met $\psi(0) = 0$ en $\psi'(0)
= p$:
\[
\psi(t) = tp + \int_0^t(t - s)\,\psi''(s)\,\dd s
\leq tp + \frac{t^2}2\cdot\frac14,
\]
dat wil zeggen $\E(\eu^{t(X - p)}) \leq \eu^{t^2/8}$ voor alle
reële $t$.

\textbf{10.} Wegens de \omterm{def:b2:proba:independence}{onafhankelijkheid} is $\E\bigl(\eu^{t(S_n -
np)}\bigr) \leq \eu^{nt^2/8}$; Markov en de optimalisatie $t =
4\delta$ geven
\[
\P(\widehat p_n - p \geq \delta)
\leq \eu^{nt^2/8 - tn\delta}\Big|_{t = 4\delta}
= \eu^{-2n\delta^2};
\]
dit toepassen op de veranderlijken $1 - X_i$ (ook in $\intcc01$)
begrenst de andere staart, waaruit de tweezijdige
$2\eu^{-2n\delta^2}$.

\textbf{11.} Chebyshev heeft alleen een tweede moment nodig en
geeft $\frac{p(1-p)}{n\delta^2}$; Hoeffding heeft
\emph{begrensdheid} nodig en geeft $2\eu^{-2n\delta^2}$. In $p =
\frac12$, $\delta = 0.03$ zijn de grenzen (bij benadering)
$\frac{278}{n}$ tegenover $2\eu^{-0.0018n}$; zij kruisen elkaar
rond $n \approx 1200$, waarna de exponentiële grens wint, en dat
overtuigend ($n = 5000$: $0.056$ tegenover $2.5\cdot10^{-4}$).

\textbf{12.} Volgens Hoeffding (vraag 10) is $\P(\abs{\widehat p_n
- p} \geq \delta) \leq 2\eu^{-2n\delta^2} \leq \alpha$ zodra
$2n\delta^2 \geq \ln\frac2\alpha$, dat wil zeggen $n \geq
\frac{\ln(2/\alpha)}{2\delta^2}$.

\textbf{13.} $\delta = 0.03$, $\alpha = 0.05$: $n \geq \frac{\ln
40}{2\cdot0.0009} \approx 2049.4$: $2050$ mensen. Voor $\delta =
0.01$: $n \geq \frac{\ln40}{0.0002} \approx 18\,445$. De omvang van
de bevolking komt nooit voor omdat elke bevraagde kiezer als een
verse trekking uit $\mathcal B(p)$ wordt gemodelleerd: de
moeilijkheid van de peiling is de \omterm{def:b2:randomvar:variance}{variantie} van een munt, niet de
grootte van het land. De marge halveren kost viermaal de
steekproef --- de wet $1/\delta^2$.

\textbf{14.} Chebyshev: $\P(\abs{\widehat p_n - p} \geq \delta)
\leq \frac{p(1-p)}{n\delta^2} \leq \frac1{4n\delta^2} \leq \alpha$
voor $n \geq \frac1{4\alpha\delta^2}$, dat wil zeggen $5556$ bij
drie punten --- ongeveer $2.7$ maal de eis van Hoeffding. Zonder
teruglegging wordt de \omterm{def:b2:randomvar:variance}{variantie} met $\frac{N - n}{N-1} < 1$
vermenigvuldigd (\cref{exo:b2:randomvar:5}), dus kan dezelfde $n$
het alleen beter doen: de berekening met teruglegging is de
behoudende.

\textbf{15.} $\{\widehat p_n \leq \frac12\} \subseteq \{\widehat
p_n - 0.52 \leq -0.02\}$, dus geeft de eenzijdige grens van
Hoeffding dat $\P(\widehat p_n \leq \tfrac12) \leq
\eu^{-2n(0.02)^2} \leq 0.01$ zodra $n \geq \frac{\ln
100}{2\cdot0.0004} \approx 5756.5$: $5757$ kiezers. De kosten
schalen met het omgekeerde kwadraat van de \emph{voorsprong}, niet
van de gewenste nauwkeurigheid: nek-aan-nekraces zijn duur.

\textbf{16.} Gebruikt: de steekproef wordt uniform en \omterm{def:b2:proba:independence}{onafhankelijk}
uit het kiezerskorps getrokken; elke bevraagde antwoordt, en
eerlijk; en $p$ beweegt niet tijdens de peiling. Echte peilingen
schenden alle drie: bereikbare en bereidwillige respondenten vormen
geen uniforme steekproef (selectie- en non-responsvertekening), en
antwoorden kunnen onwaar of onstabiel zijn. Dat zijn
\emph{vertekeningsfouten}: zij verschuiven $\E(\widehat p_n)$ weg
van $p$ met een hoeveelheid die niet van $n$ afhangt, dus verkleint
geen enkele steekproefomvang hen --- de wiskunde van dit deel
beheerst alleen de fluctuatieterm.

\textbf{17.} Chebyshev voor één groep van omvang $m$:
$\P(\abs{\widehat p^{(i)} - p} \geq \delta) \leq
\frac{1}{4m\delta^2} \leq \frac18$ voor $m \geq \frac2{\delta^2}$.
Zitten minder dan $k/2$ groepen fout, dan liggen meer dan $k/2$ van
de waarden $\widehat p^{(i)}$ in het \omterm{def:b2:metric:topology}{open} interval $\intoo{p -
\delta}{p + \delta}$, en hun mediaan dus ook; bijgevolg dwingt
$\{\abs{M - p} \geq \delta\}$ minstens $\lceil k/2\rceil$ fouten
onder $k$ \omterm{def:b2:proba:independence}{onafhankelijke} groepen af. De somgrens over de $\binom
k{\lceil k/2\rceil}$ mogelijke verzamelingen foute groepen geeft
\[
\P(\abs{M - p} \geq \delta)
\leq \binom{k}{\lceil k/2\rceil}
\Bigl(\frac18\Bigr)^{k/2}
\leq 2^k\,8^{-k/2} = 2^{-k/2} :
\]
exponentieel verval in het aantal groepen, gekocht met niets dan
\omterm{def:b2:randomvar:variance}{varianties} --- nuttig precies wanneer de termen onbegrensd zijn en
Hoeffding niet beschikbaar is.

\textbf{18.} Cauchy--Schwarz (\cref{thm:b2:randomvar:jensen}):
\[
\E(X) = \E(X\,\mathbf 1_{X>0})
\leq \sqrt{\E(X^2)}\sqrt{\E(\mathbf 1_{X>0}^2)}
= \sqrt{\E(X^2)\,\P(X > 0)} ;
\]
kwadrateer en deel.

\textbf{19.} Zij $h(a) = I(a) - 2(a - \tfrac12)^2$. Dan is
$h(\tfrac12) = 0$, verdwijnt $h'(a) = \ln\frac a{1-a} - 4(a -
\tfrac12)$ in $\tfrac12$, en is
\[
h''(a) = \frac1a + \frac1{1-a} - 4 = \frac{1}{a(1-a)} - 4
\geq 0
\]
omdat $a(1-a) \leq \frac14$. Dus stijgt $h'$ vanaf $0$ op
$\intco{\frac12}1$, waardoor $h' \geq 0$ en $h \geq 0$: $I(a) \geq
2(a - \tfrac12)^2$.

\textbf{20.} $I(\tfrac12) = I'(\tfrac12) = 0$, en $I''(a) =
\frac1{a(1-a)}$ geeft $I''(\tfrac12) = 4$, terwijl
$I'''(\tfrac12) = 0$ (de functie is \omterm{def:b2:quadratic:adjoint}{symmetrisch} om $\tfrac12$), dus
$I(\tfrac12 + \delta) = 2\delta^2 + O(\delta^4)$. Vraag 4 begrenst
de werkelijke staart dan van \emph{onder} door $\eu^{-n(2\delta^2 +
O(\delta^4))}/(n+1)$: voor kleine $\delta$ is de exponent
$2n\delta^2$ van Hoeffding asymptotisch exact --- alleen
verbeteringen polynomiaal in $n$ zijn mogelijk.

\textbf{21.} Markov: één moment, verval $1/a$, alleen nuttig als
motor achter de andere (vraag 1 toont haar vlak). Chebyshev: twee
momenten, verval $\frac{V}{n\delta^2}$, scherp zonder verdere
hypothesen (\cref{ex:b2:randomvar:chebsharp}), en het beste
gereedschap bij vraag 23. Vierde moment
(\cref{exo:b2:randomvar:9}): verval $C/n^2$, net genoeg
\omterm{def:b2:series:summable}{sommeerbaarheid} voor een sterke wet. Hoeffding: begrensde
veranderlijken, verval $2\eu^{-2n\delta^2}$, het werkpaard van Deel
III. Chernoff met het exacte tempo $I(a)$: volledige exponentiële
momenten, een onverslaanbare exponent (vragen 4, 20), het
ijkpunt voor al het overige.

\textbf{22.} Is de munt eerlijk: $\P(\widehat p_n > 0.525) \leq
\P(\widehat p_n - \tfrac12 \geq 0.025) \leq \eu^{-2n(0.025)^2}$. Is
$p = 0.55$: $\P(\widehat p_n \leq 0.525) \leq \P(\widehat p_n -
0.55 \leq -0.025) \leq \eu^{-2n(0.025)^2}$. Beide fouten liggen
onder $0.01$ wanneer $2n(0.025)^2 \geq \ln 100$, dat wil zeggen $n
\geq 3684.2$: $3685$ worpen. (Hypothesen die $2.5$ punten uit
elkaar liggen onderscheiden kost wat schatten tot op $\pm2.5$
punten kost.)

\textbf{23.} Hoeffding: $n \geq \frac{\ln 40}{2(0.005)^2} \approx
73\,778$. Chebyshev met de echte \omterm{def:b2:randomvar:variance}{variantie} $p(1-p) = 0.0099$: $n
\geq \frac{0.0099}{0.05\cdot(0.005)^2} = 7920$ --- negen maal
goedkoper. De exponent $2n\delta^2$ van Hoeffding beprijst de
\omterm{def:b2:randomvar:variance}{variantie} op haar slechtste geval $\frac14$, absurd pessimistisch
wanneer $p = 0.01$; het bescheiden tweede moment weet beter. Het
ontbrekende gereedschap is een variantiebewuste exponentiële grens
(de ongelijkheid van Bernstein, bachelorjaar 3) --- of, voor
zeldzame \omterm{def:b2:proba:space}{gebeurtenissen}, de poisson-benadering bewezen in
\cref{ch:b2:genfun}, die op de natuurlijke relatieve schaal
werkt.

\textbf{24.} Houd $\delta > 0$ vast: $\sum_n 2\eu^{-2n\delta^2} <
\infty$ (een reeks van meetkundig type), dus geeft
Borel--Cantelli~1 (\cref{thm:b2:proba:borelcantelli}) dat
$\P(\abs{\widehat p_n - p} \geq \delta \text{ oneindig vaak}) = 0$,
dat wil zeggen dat de \omterm{def:b2:proba:space}{gebeurtenis} $E_j =
\bigcup_N\bigcap_{n\geq N}\{\abs{\widehat p_n - p} < \tfrac1j\}$
voor elke $j$ kans $1$ heeft. De \omterm{def:b2:structures:countable}{aftelbare} doorsnede
$\bigcap_jE_j$ heeft nog altijd kans $1$ (subadditiviteit op de
complementen), en daarop is $\widehat p_n \to p$: de sterke wet van
de grote aantallen voor muntworpen, waarbij de begrensdheid de rol
speelt die het vierde moment in \cref{exo:b2:randomvar:9} speelde.

\textbf{25.} Eén moment koopt een vlakke grens; twee kopen
$1/(n\delta^2)$, en niet meer (het voorbeeld van de scherpte); vier
kopen $1/n^2$, genoeg om tot een bijna-zekere wet te telescoperen;
begrensdheid koopt $\eu^{-2n\delta^2}$; en het volledige
exponentiële moment koopt het exacte tempo $I$, dat geen enkele
methode verslaat. $2050$ mensen bevragen volstaat voor elk land
omdat de fluctuatie van de steekproef door de \omterm{def:b2:randomvar:variance}{variantie} van de munt
wordt bestuurd, niet door de omvang van de bevolking --- de
prijskaartjes $1/\delta^2$ en $\ln(1/\alpha)$ zijn universeel. De
centrale limietstelling van het volume van bachelorjaar 3 vervangt
deze ongelijkheden op de schaal $\sqrt n$ door een exacte
limietverdeling met expliciete constanten --- waarmee elke grens
van deze opgave in een asymptotische gelijkheid verandert.
\end{solution}
