\section*{Hoofdstuk \ref{ch:b2:multint} --- Lijnintegralen en meervoudige integralen}

\begin{solution}{exo:b2:multint:1}
(a) Segment $\gamma(t) = (t, t)$, $t \in [0,1]$:
\[
\int_\gamma y^2\dd x + x\dd y
= \int_0^1 (t^2 + t)\,\dd t = \frac13 + \frac12 = \frac56 .
\]
(b) Parabool $\gamma(t) = (t, t^2)$:
\[
\int_0^1 \bigl(t^4\cdot 1 + t\cdot 2t\bigr)\dd t
= \frac15 + \frac23 = \frac{13}{15} .
\]
De twee waarden verschillen, dus hangt de integraal van het pad af:
de vorm is \emph{niet} \omterm{def:b2:multint:exact}{exact} --- daarmee in overeenstemming is $P_y
= 2y \neq 1 = Q_x$, zodat zij niet eens \omterm{def:b2:multint:exact}{gesloten} is.
\end{solution}

\begin{solution}{exo:b2:multint:2}
$P = 2xy + y^3$, $Q = x^2 + 3xy^2 + 1$: $P_y = 2x + 3y^2 = Q_x$,
\omterm{def:b2:multint:exact}{gesloten} op $\R^2$. Zoek $f$ met $f_x = P$: $f = x^2y + xy^3 +
g(y)$; dan dwingt $f_y = x^2 + 3xy^2 + g'(y) = Q$ af dat $g'(y) =
1$, zeg $g(y) = y$. Dus is
\[
f(x, y) = x^2y + xy^3 + y
\]
een \omterm{def:b2:multint:exact}{potentiaal} ($\R^2$ is stervormig, dus moest er volgens het
lemma van Poincaré een \omterm{def:b2:multint:exact}{potentiaal} bestaan --- maar hem aanwijzen
gaat sneller). Volgens \cref{thm:b2:multint:ftc} is voor elke boog
van $(0,0)$ naar $(1,2)$
\[
\int_\gamma\omega = f(1, 2) - f(0, 0) = 2 + 8 + 2 = 12 .
\]
\end{solution}

\begin{solution}{exo:b2:multint:3}
Het gebied is $\{0 \leq x \leq 1,\ x^2 \leq y \leq x\}$. Eerst
$y$:
\[
\int_0^1\!\int_{x^2}^{x}(x + y)\,\dd y\,\dd x
= \int_0^1\Bigl(x(x - x^2)
+ \frac{x^2 - x^4}{2}\Bigr)\dd x
= \int_0^1\Bigl(\frac{3x^2}{2} - x^3 - \frac{x^4}{2}\Bigr)\dd x
= \frac12 - \frac14 - \frac1{10} = \frac{3}{20} .
\]
Eerst $x$: de schijf op hoogte $y \in [0, 1]$ is $y \leq x \leq
\sqrt y$, dus
\[
\int_0^1\!\int_{y}^{\sqrt y}(x + y)\,\dd x\,\dd y
= \int_0^1\Bigl(\frac{y - y^2}{2}
+ y(\sqrt y - y)\Bigr)\dd y
= \frac14 - \frac16 + \frac25 - \frac13 = \frac{3}{20} .
\]
\end{solution}

\begin{solution}{exo:b2:multint:4}
\emph{Eerste integraal.} Op de schijf $D_R$, in poolcoördinaten:
\[
\iint_{D_R}\frac{\dd x\,\dd y}{(1 + x^2 + y^2)^2}
= \int_0^{2\pi}\!\!\int_0^R
\frac{\rho\,\dd\rho\,\dd\alpha}{(1 + \rho^2)^2}
= 2\pi\Bigl[-\frac{1}{2(1 + \rho^2)}\Bigr]_0^R
= \pi\Bigl(1 - \frac{1}{1 + R^2}\Bigr)
\xrightarrow[R \to \infty]{} \pi .
\]
\emph{Tweede integraal.} De kwartschijf is $0 \leq \alpha \leq
\frac\pi2$, $0 \leq \rho \leq 1$, en $xy =
\rho^2\cos\alpha\sin\alpha$:
\[
\iint_{D'}xy\,\dd x\,\dd y
= \int_0^{\pi/2}\!\!\cos\alpha\sin\alpha\,\dd\alpha
\int_0^1 \rho^3\,\dd\rho
= \frac12\cdot\frac14 = \frac18 .
\]
\end{solution}

\begin{solution}{exo:b2:multint:5}
Volgens \cref{cor:b2:multint:area} met $x = \cos^3 t$, $y = \sin^3
t$: $x' = -3\cos^2 t\sin t$, $y' = 3\sin^2 t\cos t$, dus
\[
xy' - yx' = 3\cos^4 t\sin^2 t + 3\sin^4 t\cos^2 t
= 3\sin^2 t\cos^2 t = \frac{3}{4}\sin^2 2t
= \frac{3}{8}(1 - \cos 4t) .
\]
Bijgevolg is
\[
A = \frac12\int_0^{2\pi}\frac38(1 - \cos 4t)\,\dd t
= \frac{3}{16}\cdot 2\pi = \frac{3\pi}{8} .
\]
(De \omterm{pb:b2:curves:1}{astroïde} past in de eenheidsschijf met \omterm{def:b2:multint:domain}{oppervlakte} $\pi$; drie
achtste van $\pi$ is aannemelijk voor haar stervorm met vier
keerpunten.)
\end{solution}

\begin{solution}{exo:b2:multint:6}
\emph{Stapelen:} boven elke $(x, y)$ van de eenheidsschijf $D$
loopt $z$ van $x^2 + y^2$ tot $1$:
\[
V = \iint_D \bigl(1 - x^2 - y^2\bigr)\dd x\,\dd y
= \int_0^{2\pi}\!\!\int_0^1 (1 - \rho^2)\rho\,\dd\rho\,\dd\alpha
= 2\pi\Bigl(\frac12 - \frac14\Bigr) = \frac\pi2 .
\]
\emph{In schijven snijden:} de schijf op hoogte $z \in [0, 1]$ is
de schijf $x^2 + y^2 \leq z$, met \omterm{def:b2:multint:domain}{oppervlakte} $\pi z$:
\[
V = \int_0^1 \pi z\,\dd z = \frac\pi2 .
\]
\end{solution}

\begin{solution}{exo:b2:multint:7}
In bolcoördinaten is het volume-element
$r^2\cos\varphi\,\dd r\,\dd\theta\,\dd\varphi$
(\cref{ex:b2:multint:spherical}), en het hoekgedeelte integreert
tot $4\pi$ ($2\pi$ uit $\theta$, $\int_{-\pi/2}^{\pi/2}\cos = 2$).
Het volume van de schil is
\[
\int_a^b 4\pi r^2\,\dd r = \frac43\pi\bigl(b^3 - a^3\bigr).
\]
Voor de tweede integraal hangt de integrand $1/r$ alleen van $r$
af:
\[
\iiint_B \frac{\dd x\,\dd y\,\dd z}{r}
= \int_0^R 4\pi r^2\cdot\frac1r\,\dd r
= 4\pi\,\frac{R^2}{2} = 2\pi R^2 .
\]
(De integrand explodeert in de oorsprong, maar onschadelijk:
$r^2/r = r$ is \omterm{def:b2:metric:continuity}{continu} --- de integraal over de schillen
$\varepsilon \leq r \leq R$ convergeert als $\varepsilon \to 0$, en
dat is de precieze zin van de uitspraak. Dit soort berekening is de
eerste stap naar de stelling van Newton dat een homogene bol
aantrekt als een puntmassa in haar middelpunt.)
\end{solution}

\begin{solution}{exo:b2:multint:8}
De integrand $g(t; x, y) = xP(tx, ty) + yQ(tx, ty)$ is
$\mathcal{C}^1$ in $(x, y)$ en \omterm{def:b2:metric:continuity}{continu} in $t$, met partiële
afgeleiden die \omterm{def:b2:metric:continuity}{continu} zijn op $[0,1] \times U$; differentiatie
onder het integraalteken (\cref{ch:b2:integration}, toegepast op
het \omterm{def:b2:metric:compact}{compacte} $t$-interval $[0,1]$, waar de dominatie vanzelf geldt)
geeft
\[
f_x(x, y)
= \int_0^1 \bigl(P(tx, ty) + tx\,P_x(tx, ty)
+ ty\,Q_x(tx, ty)\bigr)\dd t .
\]
Met de geslotenheid $Q_x = P_y$:
\[
tx\,P_x(tx, ty) + ty\,P_y(tx, ty)
= t\,\frac{\dd}{\dd t}\bigl[P(tx, ty)\bigr] ,
\]
dus is de integrand $P(tx, ty) + t\frac{\dd}{\dd t}P(tx, ty) =
\frac{\dd}{\dd t}\bigl[t\,P(tx, ty)\bigr]$ en
\[
f_x(x, y) = \Bigl[t\,P(tx, ty)\Bigr]_0^1 = P(x, y) .
\]
\omterm{def:b2:quadratic:adjoint}{Symmetrisch} is $f_y = Q$ (dezelfde berekening met $P_y = Q_x$ in de
andere richting gebruikt). Merk op waar de hypothese binnenkomt:
$f$ is gedefinieerd door langs het segment $[0, M]$ te integreren,
dat precies in $U$ ligt omdat $U$ stervormig is.
\end{solution}

\begin{solution}{exo:b2:multint:9}
Op de strook $[0, A] \times [0, \infty)$ is de functie $(x, y)
\mapsto e^{-xy}\sin x$ niet in naïeve zin uniform \omterm{def:b2:series:def}{absoluut}
integreerbaar tot $y = \infty$, maar elke herhaalde integraal
convergeert en hun gelijkheid volgt uit Fubini op $[0, A] \times
[0, B]$ plus een limiet $B \to \infty$ (de staart
$\int_0^A\int_B^\infty e^{-xy}\abs{\sin x}\,\dd y\,\dd x \leq
\int_0^A \frac{e^{-Bx}\abs{\sin x}}{x}\dd x \leq \int_0^A e^{-Bx}
\dd x\to 0$, met $\abs{\sin x} \leq x$).

\emph{Eerst $y$:} $\int_0^\infty e^{-xy}\,\dd y = \frac1x$ voor $x
> 0$, dus is de eerste integraal $\int_0^A \frac{\sin x}{x}\,\dd
x$.

\emph{Eerst $x$:} twee partiële integraties (of het imaginaire deel
van $\int_0^A e^{(i - y)x}\dd x$ nemen) geven
\[
\int_0^A e^{-xy}\sin x\,\dd x
= \frac{1 - e^{-Ay}(y\sin A + \cos A)}{1 + y^2} .
\]
Naar $y$ integreren over $[0, \infty)$ zondert de term
$\int_0^\infty \frac{\dd y}{1 + y^2} = \frac\pi2$ af:
\[
\int_0^A \frac{\sin x}{x}\,\dd x
= \frac{\pi}{2}
- \int_0^\infty e^{-Ay}\,\frac{y\sin A + \cos A}{1 + y^2}\,\dd y .
\]
De rest wordt begrensd door $\int_0^\infty e^{-Ay}\frac{y + 1}{1 +
y^2}\dd y \leq \int_0^\infty e^{-Ay}\cdot\frac{1+y}{1+y^2} \,\dd y
\to 0$ als $A \to \infty$ (gedomineerde convergentie, of de ruwe
grens $\frac{1 + y}{1 + y^2} \leq \frac32$ die $\frac{3}{2A}$
geeft). Bijgevolg is $\int_0^\infty\frac{\sin x}{x}\dd x =
\frac\pi2$ --- dezelfde waarde als die in
\cref{ch:b2:integration} werd verkregen door een
\omterm{thm:b2:integration:continuity}{parameterintegraal} te differentiëren; hier doet Fubini het werk.
\end{solution}

\begin{solution}{exo:b2:multint:10}
Parametriseer naar \omterm{def:b2:curves:length}{booglengte} $s \in [0, 2\pi]$, zodat $x'^2 +
y'^2 = 1$, en transleer zo dat $\int_0^{2\pi} x(s)\,\dd s = 0$.
Volgens \cref{cor:b2:multint:area} is
\[
2A = \oint x\,\dd y - y\,\dd x
= \int_0^{2\pi}\bigl(xy' - yx'\bigr)\dd s .
\]
$\oint y\,\dd x$ partieel integreren over de periode (de randtermen
heffen elkaar op wegens de periodiciteit) geeft $-\int yx' = \int
y'x$, dus is in feite $2A = 2\int_0^{2\pi}xy'\,\dd s$. Dan geeft
$2xy' \leq x^2 + y'^2$ dat
\[
2A \leq \int_0^{2\pi}\bigl(x^2 + y'^2\bigr)\dd s
= \int_0^{2\pi} x^2 + \int_0^{2\pi}\bigl(1 - x'^2\bigr)
= 2\pi - \int_0^{2\pi}\bigl(x'^2 - x^2\bigr)\dd s .
\]
De ongelijkheid van Wirtinger (oefeningen van
\cref{ch:b2:fourier}: voor een $2\pi$-periodieke
$\mathcal{C}^1$-functie met gemiddelde nul is $\int x^2 \leq \int
x'^2$) maakt de laatste integraal niet-negatief: $A \leq \pi$.
Gelijkheid vereist gelijkheid in Wirtinger ($x(s) = a\cos s + b\sin
s$) en in $2xy' \leq x^2 + y'^2$ ($y' = x$ puntsgewijs), wat $y =
a\sin s - b\cos s + c$ afdwingt: de kromme is de eenheidscirkel
(passend gecentreerd). Omdat een kromme met \omterm{def:b2:curves:length}{lengte} $L$ tot \omterm{def:b2:curves:length}{lengte}
$2\pi$ herschaalt, luidt de algemene uitspraak $A \leq
\frac{L^2}{4\pi}$: van alle \omterm{def:b2:multint:exact}{gesloten} krommen met gegeven omtrek
omsluit de cirkel de grootste \omterm{def:b2:multint:domain}{oppervlakte}.
\end{solution}

\begin{solution}{exo:b2:multint:11}
In bolcoördinaten is $z = r\sin\varphi$ en $\dd x\,\dd y\,\dd z =
r^2\cos\varphi\,\dd r\,\dd\theta\,\dd\varphi$:
\[
\iiint_B z^2
= \int_0^R r^4\,\dd r\int_0^{2\pi}\dd\theta
\int_{-\pi/2}^{\pi/2}\sin^2\varphi\cos\varphi\,\dd\varphi
= \frac{R^5}5\cdot2\pi\cdot
\Bigl[\frac{\sin^3\varphi}3\Bigr]_{-\pi/2}^{\pi/2}
= \frac{4\pi R^5}{15}.
\]
Wegens de symmetrie van de bol onder het permuteren van de
coördinaten is $\iiint_B x^2 = \iiint_B y^2 = \iiint_B z^2$, dus
$\iiint_B(x^2 + y^2 + z^2) = 3\cdot\frac{4\pi R^5}{15} = \frac{4\pi
R^5}5$. Controle met schillen: $\int_0^R r^2\cdot 4\pi r^2\,\dd r =
\frac{4\pi R^5}5$ --- de integrand $r^2$ is constant op de bol met
straal $r$, die \omterm{def:b2:multint:domain}{oppervlakte} $4\pi r^2$ heeft.
\end{solution}

\begin{solution}{exo:b2:multint:12}
Parametriseer de zijde van $(x_i, y_i)$ naar $(x_{i+1}, y_{i+1})$
met $\gamma(t) = \bigl((1-t)x_i + tx_{i+1},\ (1-t)y_i +
ty_{i+1}\bigr)$. Haar bijdrage aan $\frac12\oint(x\,\dd y - y\,\dd
x)$ is
\[
\frac12\int_0^1\Bigl(\bigl((1-t)x_i +
tx_{i+1}\bigr)(y_{i+1} - y_i) - \bigl((1-t)y_i +
ty_{i+1}\bigr)(x_{i+1} - x_i)\Bigr)\dd t ,
\]
en omdat $\int_0^1\bigl((1-t)u + tv\bigr)\dd t = \frac{u + v}2$, is
dit gelijk aan
\[
\frac14\Bigl((x_i + x_{i+1})(y_{i+1} - y_i) - (y_i +
y_{i+1})(x_{i+1} - x_i)\Bigr)
= \frac12\bigl(x_iy_{i+1} - x_{i+1}y_i\bigr),
\]
waarbij de kruistermen wegvallen. Sommeren over de $m$ zijden geeft
volgens \cref{cor:b2:multint:area} de veterformule. Driehoek
$(0,0), (1,0), (0,1)$: $\frac12\bigl((0\cdot0 - 1\cdot0) +
(1\cdot1 - 0\cdot0) + (0\cdot0 - 0\cdot1)\bigr) = \frac12$, de
juiste \omterm{def:b2:multint:domain}{oppervlakte}.
\end{solution}

\begin{solution}{pb:b2:multint:1}
\textbf{1.} De bol $B_n(R)$ wordt beschreven door de herhaalde
grenzen $-R \leq x_n \leq R$, dan $\abs{x_{n-1}} \leq \sqrt{R^2 -
x_n^2}$, enzovoort; $x_i = Ru_i$ substitueren in elk van de $n$
integralen in één veranderlijke vermenigvuldigt elke met $R$ en
beeldt de grenzen af op die van $B_n(1)$: $v_n(R) = R^n\,v_n(1) =
V_nR^n$.

\textbf{2.} In schijven snijden langs $x_n = t$: de schijf van
$B_n(1)$ is de bol $B_{n-1}\bigl(\sqrt{1 - t^2}\bigr)$, dus is
volgens vraag 1
\[
V_n = \int_{-1}^1 v_{n-1}\bigl(\sqrt{1 - t^2}\bigr)\,\dd t
= V_{n-1}\int_{-1}^{1}(1 - t^2)^{\frac{n-1}2}\,\dd t .
\]

\textbf{3.} Met $t = \sin\theta$, $\dd t = \cos\theta\,\dd\theta$
en $(1 - t^2)^{\frac{n-1}2} = \cos^{n-1}\theta$ op
$\intcc{-\pi/2}{\pi/2}$:
\[
\int_{-1}^1(1 - t^2)^{\frac{n-1}2}\dd t
= \int_{-\pi/2}^{\pi/2}\cos^n\theta\,\dd\theta
= 2\int_0^{\pi/2}\cos^n\theta\,\dd\theta = 2W_n,
\]
en $\theta \mapsto \frac\pi2 - \theta$ verwisselt de sinus- en de
cosinusvorm van $W_n$.

\textbf{4.} Schrijf $\sin^n = \sin^{n-2}(1 - \cos^2)$ en integreer
$\int\sin^{n-2}\cos\cdot\cos$ partieel ($v =
\frac{\sin^{n-1}}{n-1}$):
\[
W_n = W_{n-2} - \frac{W_n}{n-1}
\quad\Longrightarrow\quad
W_n = \frac{n-1}{n}W_{n-2}.
\]
Bijgevolg is $nW_nW_{n-1} = (n-1)W_{n-1}W_{n-2}$: de rij
$(nW_nW_{n-1})$ is constant, gelijk aan $1\cdot W_1W_0 =
1\cdot\frac\pi2$, dus $W_nW_{n-1} = \frac{\pi}{2n}$.

\textbf{5.} De vragen 2--3 geven $V_n = 2W_nV_{n-1}$, tweemaal:
\[
V_n = 2W_n\cdot 2W_{n-1}\,V_{n-2}
= 4\,\frac{\pi}{2n}\,V_{n-2} = \frac{2\pi}n\,V_{n-2}.
\]

\textbf{6.} Uit $V_0 = 1$: $V_{2k} = \frac{2\pi}{2k}V_{2k-2} =
\frac\pi kV_{2k-2}$, dus $V_{2k} = \frac{\pi^k}{k!}$ met inductie.
Uit $V_1 = 2$: $V_{2k+1} = \frac{2\pi}{2k+1}V_{2k-1}$, dus
\[
V_{2k+1} = 2\prod_{j=1}^k\frac{2\pi}{2j+1}
= \frac{2^{k+1}\pi^k}{1\cdot3\cdot5\cdots(2k+1)} .
\]

\textbf{7.} $V_1 = 2$, $V_2 = \pi \approx 3.142$, $V_3 =
\frac{4\pi}3 \approx 4.189$, $V_4 = \frac{\pi^2}2 \approx 4.935$,
$V_5 = \frac{8\pi^2}{15} \approx 5.264$, $V_6 = \frac{\pi^3}6
\approx 5.168$, $V_7 = \frac{16\pi^3}{105} \approx 4.725$. De
verhouding met één stap is $V_n/V_{n-1} = 2W_n$, en $(W_n)$ is
dalend ($\sin^n \leq \sin^{n-1}$ puntsgewijs). Nu is $2W_5 =
2\cdot\frac45\cdot\frac23 = \frac{16}{15} > 1$ terwijl $2W_6 =
2\cdot\frac56\cdot\frac34\cdot\frac12\cdot\frac\pi2 =
\frac{5\pi}{16} < 1$: de verhoudingen zijn groter dan $1$ tot en
met $n = 5$ en kleiner dan $1$ vanaf $n = 6$ --- $(V_n)$ stijgt tot
haar maximum $V_5$ en daalt daarna.

\textbf{8.} Voor $n \geq 13 > 4\pi$ is $V_n/V_{n-2} = 2\pi/n <
\tfrac12$, dus $V_{n} \leq C\cdot 2^{-n/2}$ met een vaste
constante; sterker nog, voor elke $q > 0$ is $2\pi/n < q^2$ voor
grote $n$, dus $V_n/q^n \to 0$: het verval verslaat elke
meetkundige rij. De convergentie van $\sum V_n$ volgt uit de
verhouding $V_n/V_{n-2} \to 0$ (vanaf een zekere rang met een
meetkundige reeks vergelijken).

\textbf{9.} $\sum_{k\geq0}V_{2k}x^{2k} = \sum_{k\geq0}\frac{(\pi
x^2)^k}{k!} = \eu^{\pi x^2}$, de exponentiële reeks
(\cref{ch:b2:powerseries}), convergent voor elke $x$. In $x = 1$:
$\sum_kV_{2k} = \eu^\pi \approx 23.14$.

\textbf{10.} Op de kubus $\intcc{-R}R^n$ is de integrand het
product $\prod_i\eu^{-x_i^2}$, dus ontbindt de herhaalde integraal:
$\bigl(\int_{-R}^R\eu^{-t^2}\dd t\bigr)^n$. Met $R \to \infty$ en
$\int_\R\eu^{-t^2}\dd t = \sqrt\pi$
(\cref{ex:b2:multint:polar}): $I_n = \pi^{n/2}$.

\textbf{11.} $t = u^2$ geeft $\Gamma(\tfrac12) = \int_0^\infty
t^{-1/2}\eu^{-t}\dd t = 2\int_0^\infty\eu^{-u^2}\dd u = \sqrt\pi$.
$\Gamma(s+1) = s\Gamma(s)$ herhalen: $\Gamma(k+1) = k!\,\Gamma(1) =
k!$, en
\[
\Gamma\Bigl(k + \frac32\Bigr)
= \Bigl(k + \frac12\Bigr)\Bigl(k - \frac12\Bigr)\cdots
\frac12\cdot\Gamma\Bigl(\frac12\Bigr)
= \frac{(2k+1)(2k-1)\cdots1}{2^{k+1}}\,\sqrt\pi .
\]

\textbf{12.} Zet $F_n = \pi^{n/2}/\Gamma(\frac n2 + 1)$. Omdat
$\Gamma(\frac n2 + 1) = \frac n2\,\Gamma(\frac n2) = \frac
n2\,\Gamma(\frac{n-2}2 + 1)$, is $F_n = \frac{2\pi}nF_{n-2}$:
dezelfde recursie als voor $V_n$ (vraag 5). Basisgevallen: $F_1 =
\sqrt\pi/\Gamma(\frac32) = \sqrt\pi/(\frac{\sqrt\pi}2) = 2 = V_1$
en $F_2 = \pi/\Gamma(2) = \pi = V_2$. Met inductie is $V_n = F_n$
voor alle $n$; vraag 11 zet dit terug om in de twee gesloten vormen
van vraag 6.

\textbf{13.} Met $r = \sqrt t$: $\int_0^\infty \eu^{-r^2}r^{n-1}\dd
r = \frac12\int_0^\infty t^{\frac n2 - 1}\eu^{-t}\dd t =
\frac12\Gamma(\frac n2)$. Bijgevolg is
\[
n\,V_n\int_0^\infty\eu^{-r^2}r^{n-1}\dd r
= V_n\cdot\frac n2\,\Gamma\Bigl(\frac n2\Bigr)
= V_n\,\Gamma\Bigl(\frac n2 + 1\Bigr) = \pi^{n/2} = I_n .
\]
Omdat beide leden bewezen zijn, kan de identiteit worden
\emph{gelezen} als de ontbinding in schillen van de integraal van
Gauss: de sfeer met straal $r$ draagt \omterm{def:b2:multint:domain}{oppervlakte} $nV_nr^{n-1}$, en
het gaussische gewicht $\eu^{-r^2}$ wordt over de schillen
geïntegreerd.

\textbf{14.} $s_0 = V_1 = 2$ (de $0$-sfeer bestaat uit twee
punten), $s_1 = 2V_2 = 2\pi$, $s_2 = 3V_3 = 4\pi$, $s_3 = 4V_4 =
2\pi^2$; en $\int_0^R s_{n-1}r^{n-1}\dd r = V_nR^n = v_n(R)$: de
\omterm{def:b2:multint:domain}{oppervlakte} is de radiale afgeleide van het volume.

\textbf{15.} Voor $n = 2k$ geeft Stirling
(\cref{thm:b2:comparison:stirling}) dat $k! \sim \sqrt{2\pi
k}\,(k/\eu)^k$, dus
\[
V_{2k} = \frac{\pi^k}{k!}
\sim \frac{(\pi\eu/k)^k}{\sqrt{2\pi k}}
= \frac1{\sqrt{\pi n}}\Bigl(\frac{2\pi\eu}n\Bigr)^{n/2}
\qquad (n = 2k).
\]
Voor oneven $n$: $V_{2k+1} = 2W_{2k+1}V_{2k} \leq 2V_{2k}$, dus
gelden dezelfde grenzen van supermeetkundig verval (op een factor
$2$ en een verschuiving van één in de exponent na) --- voor elke $q
> 0$ is $V_n = o(q^n)$.

\textbf{16.} $V_2/4 = \pi/4 \approx 0.785$; $V_3/8 = \pi/6 \approx
0.524$; $V_{10}/2^{10} = \frac{\pi^5}{120\cdot1024} \approx
0.0025$. In het algemeen is $\frac{V_n/2^n}{V_{n-2}/2^{n-2}} =
\frac{2\pi}{4n} = \frac{\pi}{2n} \to 0$: de verhouding streeft
(supermeetkundig) naar $0$. De ingeschreven bol neemt een
verdwijnend aandeel in: het volume van de kubus verhuist naar haar
hoeken.

\textbf{17.} Volgens vraag 1 heeft de binnenste bol met straal $1 -
\varepsilon$ volume $V_n(1-\varepsilon)^n$, dus draagt de
buitenschil het aandeel $1 - (1 - \varepsilon)^n \to 1$. Voor $n =
100$ en $\varepsilon = 0.01$: $(0.99)^{100} = \eu^{100\ln0.99}
\approx \eu^{-1.005} \approx 0.366$: ongeveer $63\%$ van de bol
ligt binnen $1\%$ van haar oppervlak.

\textbf{18.} $(W_n)$ daalt, dus $W_n \leq W_{n-1} \leq W_{n-2} =
\frac{n}{n-1}W_n$: inklemmen geeft $W_{n-1}/W_n \to 1$.
Vermenigvuldigen met $W_nW_{n-1} = \frac\pi{2n}$: $W_n^2 \sim
\frac\pi{2n}$, dat wil zeggen $W_n \sim \sqrt{\pi/(2n)}$.
Ondergrens: $W_n^2 \geq W_nW_{n+1} = \frac{\pi}{2(n+1)}$, dus $W_n
\geq \sqrt{\pi/(2(n+1))}$ voor elke $n$.

\textbf{19.} Teller: $1 - x^2 \leq \eu^{-x^2}$ geeft $(1 -
x^2)^{\frac{n-1}2} \leq \eu^{-(n-1)x^2/2}$, en met $a =
\frac{n-1}2$ is
\[
\int_\delta^1(1 - x^2)^{\frac{n-1}2}\dd x
\leq \int_\delta^\infty\eu^{-ax^2}\dd x
\leq \int_\delta^\infty\frac x\delta\,\eu^{-ax^2}\dd x
= \frac{\eu^{-a\delta^2}}{2a\delta}
= \frac{\eu^{-(n-1)\delta^2/2}}{(n-1)\delta}.
\]
Noemer: $2W_n \geq \sqrt{2\pi/(n+1)}$ volgens vraag 18. Delen geeft
de getoonde grens. Voor $\delta = s/\sqrt{n-1}$ wordt zij
$\sqrt{\tfrac{n+1}{2\pi(n-1)}}\;\eu^{-s^2/2}/s = O\bigl(
\eu^{-s^2/2}/s\bigr)$, uniform in $n$: buiten de plak $\abs{x_1}
\leq s/\sqrt{n-1}$ zit vrijwel geen volume, voor matig grote $s$
--- en wegens de symmetrie geldt hetzelfde in elke richting.

\textbf{20.} De twee uitspraken bestaan naast elkaar omdat zij
verschillende coördinaten van hetzelfde punt beschrijven. Vrijwel
elk punt van $B_n(1)$ heeft een \omterm{def:b2:nvs:norm}{norm} dicht bij $1$ (vraag 17:
radiale concentratie nabij de sfeer), en toch is elk van zijn $n$
coördinaten klein, van de orde $1/\sqrt n$ (vraag 19), wat
verenigbaar is omdat $n$ coördinaten van grootte $1/\sqrt n$ een
\omterm{def:b2:nvs:norm}{norm} van de orde $1$ hebben. Het volume in hoge dimensie
concentreert zich waar alle coördinaten het normbudget gelijk
delen --- nabij de sfeer, maar ver van elke pool op een
coördinaatas.

\textbf{21.} Snijd $\Delta_n$ in schijven bij $x_n = t \in
\intcc01$: de schijf is $\{x' \in \R^{n-1} : x_i \geq 0,\ \sum x_i
\leq 1 - t\} = (1-t)\Delta_{n-1}$, met volume
$(1-t)^{n-1}\operatorname{vol}(\Delta_{n-1})$ wegens de
homogeniteit. Dus
\[
\operatorname{vol}(\Delta_n) =
\operatorname{vol}(\Delta_{n-1})\int_0^1(1 - t)^{n-1}\dd t =
\frac{\operatorname{vol}(\Delta_{n-1})}{n}
\quad\Longrightarrow\quad
\operatorname{vol}(\Delta_n) = \frac1{n!}\,.
\]

\textbf{22.} De $2^n$ tekenorthanten snijden $C_n$ in $2^n$ kopieën
van $\Delta_n$ (met verwaarloosbare overlappingen op de
coördinaathypervlakken): $\operatorname{vol}(C_n) =
\frac{2^n}{n!}$. Geldt $\sum\abs{x_i} \leq 1$, dan is $\sum x_i^2
\leq \bigl(\sum\abs{x_i}\bigr)^2 \leq 1$: $C_n \subseteq B_n(1)$;
en $B_n(1) \subseteq \intcc{-1}1^n$ omdat $\abs{x_i} \leq \norm x$.
Bijgevolg is $\frac{2^n}{n!} \leq V_n \leq 2^n$ --- in
overeenstemming met vraag 15, die $V_n$ tussen de faculteits- en de
meetkundige schaal plaatst.

\textbf{23.} Voor $(x, y)$ in de eenheidsschijf is de schijf van
$B_4(1)$ de schijf met straal $\sqrt{1 - x^2 - y^2}$ in het $(z,
w)$-vlak, met \omterm{def:b2:multint:domain}{oppervlakte} $\pi(1 - x^2 - y^2)$. In poolcoördinaten:
\[
V_4 = \iint_{x^2+y^2\leq1}\pi(1 - x^2 - y^2)\,\dd x\,\dd y
= \pi\int_0^{2\pi}\!\!\int_0^1(1 - \rho^2)\rho\,
\dd\rho\,\dd\alpha
= \pi\cdot2\pi\cdot\frac14 = \frac{\pi^2}2 ,
\]
in overeenstemming met vraag 6.

\textbf{24.} De kans is de volumeverhouding $\dfrac{V_{20}}{2^{20}}
= \dfrac{\pi^{10}}{10!\cdot2^{20}} \approx
\dfrac{0.0258}{1\,048\,576} \approx 2.5\cdot10^{-8}$. Het aantal
trekkingen tot de eerste treffer is van de orde van het omgekeerde,
ongeveer $4\cdot10^7$: een verwerpende steekproef die voor de
schijf prachtig werkte ($\pi/4$ van de treffers) is nutteloos in
dimensie $20$ --- de vloek van de dimensionaliteit in één regel.

\textbf{25.} Weg één (Delen I--II) gebruikte: het in schijven
snijden van de herhaalde integraal in de stijl van Fubini, de
substitutie in één veranderlijke in elke coördinaat (homogeniteit)
en de \omterm{pb:b2:multint:1}{integralen van Wallis} --- zuivere analyse in één
veranderlijke plus inductie. Weg twee (Deel III) gebruikte: Fubini
voor de productstructuur van $I_n$, de verandering naar
poolcoördinaten via de integraal van Gauss uit
\cref{ex:b2:multint:polar}, en de functionaalvergelijking van de
$\Gamma$-functie. Zij ontmoeten elkaar in $V_n =
\pi^{n/2}/\Gamma(\frac n2 + 1)$, waarbij Stirling
(\cref{thm:b2:comparison:stirling}) de formule in asymptotiek
omzet. Het volume van bachelorjaar 3 bouwt dit alles opnieuw op met
de integraal van Lebesgue: daar zijn Fubini en de verandering van
veranderlijken stellingen voor algemene integreerbare functies,
bestaan bolcoördinaten in elke dimensie, en duiken dezelfde volumes
van de bol op als de uitgewerkte dividenden van de opgaven over
productmaten en Stirling --- met de gedomineerde convergentie in de
plaats van onze zelfgemaakte insluitingen.
\end{solution}
