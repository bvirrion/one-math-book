\section*{Hoofdstuk \ref{ch:b2:surfaces} --- Oppervlakken}

\begin{solution}{exo:b2:surfaces:1}
Met $\sigma(\theta, r) = (r\cos\theta,\ r\sin\theta,\ r)$:
\[
\sigma_\theta = (-r\sin\theta,\ r\cos\theta,\ 0),
\qquad
\sigma_r = (\cos\theta,\ \sin\theta,\ 1),
\]
onafhankelijk voor $r > 0$. Het \omterm{def:b2:surfaces:tangent}{raakvlak} in $M_0 =
\sigma(\theta_0, r_0)$ gaat door $M_0$ met richtingen
$\sigma_\theta, \sigma_r$. Nu is $M_0 - O = r_0(\cos\theta_0,
\sin\theta_0, 1) = r_0\,\sigma_r(\theta_0, r_0)$ zelf een
raakrichting: de top $O$ ligt in het \omterm{def:b2:surfaces:tangent}{raakvlak}. (Dit is het algemene
gedrag van kegels: zij worden beschreven door rechten door de top,
en een \omterm{def:b2:surfaces:tangent}{raakvlak} bevat de beschrijvende rechte door het raakpunt.)
\end{solution}

\begin{solution}{exo:b2:surfaces:2}
Pas \cref{prop:b2:surfaces:gradient} toe op $f(x,y,z) =
\frac{x^2}{a^2} + \frac{y^2}{b^2} + \frac{z^2}{c^2}$: $\nabla
f(x_0,y_0,z_0) = 2\bigl(\frac{x_0}{a^2}, \frac{y_0}{b^2},
\frac{z_0}{c^2}\bigr) \neq 0$ op het oppervlak. Het \omterm{def:b2:surfaces:tangent}{raakvlak} is
\[
\frac{x_0}{a^2}(x - x_0) + \frac{y_0}{b^2}(y - y_0)
+ \frac{z_0}{c^2}(z - z_0) = 0,
\qquad\text{dat wil zeggen}\qquad
\frac{x_0\,x}{a^2} + \frac{y_0\,y}{b^2} + \frac{z_0\,z}{c^2} = 1,
\]
waarbij gebruikt is dat $(x_0, y_0, z_0)$ aan de vergelijking van
de ellipsoïde voldoet --- de regel ``splits de kwadraten'' die de
uitdrukking $\langle M_0, M \rangle = R^2$ van de bol
veralgemeent.
\end{solution}

\begin{solution}{exo:b2:surfaces:3}
$\sigma_u = (-v\sin u,\ v\cos u,\ a)$ en $\sigma_v = (\cos u,\
\sin u,\ 0)$, dus
\[
E = v^2 + a^2, \qquad F = 0, \qquad G = 1,
\qquad
\sqrt{EG - F^2} = \sqrt{v^2 + a^2} > 0
\]
(de helicoïde is overal regulier, ook op haar as $v = 0$).
\omterm{def:b2:surfaces:area}{Oppervlakte} van het stuk:
\[
\mathcal{A} = \int_0^{2\pi}\!\!\int_0^1
\sqrt{v^2 + a^2}\;\dd v\,\dd u
= 2\pi\cdot\frac12\Bigl[v\sqrt{v^2 + a^2}
+ a^2\ln\bigl(v + \sqrt{v^2 + a^2}\bigr)\Bigr]_0^1 ,
\]
dat wil zeggen $\mathcal{A} = \pi\Bigl(\sqrt{1 + a^2} +
a^2\ln\frac{1 + \sqrt{1 + a^2}}{a}\Bigr)$.
\end{solution}

\begin{solution}{exo:b2:surfaces:4}
Afgeleiden:
\[
\sigma_\theta = \bigl(-(R + r\cos\psi)\sin\theta,\
(R + r\cos\psi)\cos\theta,\ 0\bigr),
\qquad
\sigma_\psi = \bigl(-r\sin\psi\cos\theta,\
-r\sin\psi\sin\theta,\ r\cos\psi\bigr).
\]
Dan is
\[
E = (R + r\cos\psi)^2,
\qquad
F = r\sin\psi\cos\psi\,(R + r\cos\psi)
\bigl(\sin\theta\cos\theta - \sin\theta\cos\theta\bigr) = 0,
\qquad
G = r^2 ,
\]
dus $\sqrt{EG - F^2} = r(R + r\cos\psi) \geq r(R - r) > 0$: overal
regulier. \omterm{def:b2:surfaces:area}{Oppervlakte}:
\[
\mathcal{A}
= \int_0^{2\pi}\!\!\int_0^{2\pi} r(R + r\cos\psi)
\,\dd\theta\,\dd\psi
= 2\pi r \Bigl(2\pi R + r\int_0^{2\pi}\cos\psi\,\dd\psi\Bigr)
= 4\pi^2 R r ,
\]
waarbij de term met $\cos\psi$ tot nul integreert --- de stelling
van Pappus: \omterm{def:b2:surfaces:area}{oppervlakte} $=$ (\omterm{def:b2:curves:length}{lengte} van de gedraaide cirkel)
$\times$ (afstand afgelegd door haar middelpunt).
\end{solution}

\begin{solution}{exo:b2:surfaces:5}
Voor $\sigma(x, y) = (x, y, f(x,y))$: $\sigma_x = (1, 0, f_x)$,
$\sigma_y = (0, 1, f_y)$, dus $E = 1 + f_x^2$, $F = f_xf_y$, $G = 1
+ f_y^2$ en
\[
EG - F^2 = (1 + f_x^2)(1 + f_y^2) - f_x^2f_y^2
= 1 + f_x^2 + f_y^2 ,
\]
wat de aangekondigde oppervlakteformule geeft. Voor $f =
\frac12(x^2 + y^2)$ op de eenheidsschijf: $1 + f_x^2 + f_y^2 = 1 +
x^2 + y^2$, en in poolcoördinaten ($x = \rho\cos\alpha$, $y =
\rho\sin\alpha$, jacobiaan $\rho$, \cref{ch:b2:multint}):
\[
\mathcal{A}
= \int_0^{2\pi}\!\!\int_0^1 \sqrt{1 + \rho^2}\;\rho
\,\dd\rho\,\dd\alpha
= 2\pi\Bigl[\tfrac13(1 + \rho^2)^{3/2}\Bigr]_0^1
= \frac{2\pi}{3}\bigl(2\sqrt2 - 1\bigr).
\]
\end{solution}

\begin{solution}{exo:b2:surfaces:6}
Uit de berekening van \cref{ex:b2:surfaces:spherearea}: $E =
R^2\cos^2\varphi$, $F = 0$, $G = R^2$, dus volgens
\cref{prop:b2:surfaces:curvelength}
\[
L = \int_a^b R\sqrt{\cos^2\varphi(t)\,\theta'(t)^2
+ \varphi'(t)^2}\;\dd t .
\]
Zijn de eindpunten $(\theta_0, \varphi_1)$ en $(\theta_0,
\varphi_2)$ met $\varphi_1 < \varphi_2$. Voor elke verbindende
kromme is
\[
L \geq \int_a^b R\,\abs{\varphi'(t)}\,\dd t
\geq R\,\Bigl|\int_a^b \varphi'(t)\,\dd t\Bigr|
= R\,(\varphi_2 - \varphi_1),
\]
waarbij de niet-negatieve term $\cos^2\varphi\,\theta'^2$ wordt
weggelaten en de driehoeksongelijkheid voor integralen wordt
gebruikt. De meridiaanboog $\theta \equiv \theta_0$, met $\varphi$
stijgend van $\varphi_1$ tot $\varphi_2$, heeft \omterm{def:b2:curves:length}{lengte} precies
$R(\varphi_2 - \varphi_1)$: zij is de kortste. (Meridianen zijn
grootcirkels; dit is het eerste, elementaire geval van het feit dat
de geodeten van de bol grootcirkels zijn.)
\end{solution}

\begin{solution}{exo:b2:surfaces:7}
Houd een op $S$ getekende kromme $\gamma$ vast en zij $g(t) =
\norm{\gamma(t) - \Omega}^2$. Dan is $g'(t) = 2\langle \gamma'(t),\
\gamma(t) - \Omega\rangle$. De normaallijn in $M = \gamma(t)$ gaat
volgens de hypothese door $\Omega$, dus is $\gamma(t) - \Omega$ een
\emph{normaalvector}, loodrecht op het \omterm{def:b2:surfaces:tangent}{raakvlak} en in het bijzonder
op de snelheid $\gamma'(t)$ (\cref{prop:b2:surfaces:velocity}): $g'
= 0$, en $g$ is constant langs elke getekende kromme.

Nu is de verzameling $S_c = \{M \in S : \norm{M - \Omega}^2 = c\}$
gesloten in $S$; zij is ook \omterm{def:b2:metric:topology}{open} in $S$: nabij elk van haar punten
is $S$ een reguliere grafiek, dus wordt elk nabij punt van $S$
ermee verbonden door een getekende kromme (een gelift segment),
waarlangs $g$ constant is. Omdat $S$ \omterm{def:b2:metric:connected}{samenhangend} is en $S_c$ voor
de juiste $c$ niet leeg, is $S = S_c \subseteq$ de bol met
middelpunt $\Omega$ en straal $\sqrt c$ (\cref{ch:b2:metric}: het
argument met de \omterm{def:b2:metric:connected}{samenhang}).
\end{solution}

\begin{solution}{exo:b2:surfaces:8}
Schrijf $\Phi(u', v') = (u, v)$. Volgens de kettingregel is
\[
\tilde\sigma_{u'} = \frac{\partial u}{\partial u'}\sigma_u
+ \frac{\partial v}{\partial u'}\sigma_v,
\qquad
\tilde\sigma_{v'} = \frac{\partial u}{\partial v'}\sigma_u
+ \frac{\partial v}{\partial v'}\sigma_v ,
\]
met de partiële afgeleiden van $\sigma$ geëvalueerd in
$\Phi(u',v')$. Het kruisproduct bilineair uitwerken en $\sigma_u
\wedge \sigma_u = \sigma_v \wedge \sigma_v = 0$, $\sigma_v \wedge
\sigma_u = -\sigma_u \wedge \sigma_v$ gebruiken geeft
\[
\tilde\sigma_{u'} \wedge \tilde\sigma_{v'}
= \Bigl(\frac{\partial u}{\partial u'}
\frac{\partial v}{\partial v'}
- \frac{\partial v}{\partial u'}
\frac{\partial u}{\partial v'}\Bigr)\,
\sigma_u \wedge \sigma_v
= \det J_\Phi \cdot (\sigma_u \wedge \sigma_v)\circ\Phi .
\]
\omterm{def:b2:nvs:norm}{Normen} nemen geeft de identiteit. Vervolgens geeft de formule voor
de verandering van veranderlijken (\cref{ch:b2:multint}), toegepast
op de afbeelding $\Phi$ op $K' = \Phi^{-1}(K)$:
\[
\iint_{K'} \norm{\tilde\sigma_{u'} \wedge \tilde\sigma_{v'}}
\,\dd u'\dd v'
= \iint_{K'} \norm{\sigma_u \wedge \sigma_v}\circ\Phi\;
\abs{\det J_\Phi}\,\dd u'\dd v'
= \iint_K \norm{\sigma_u \wedge \sigma_v}\,\dd u\,\dd v :
\]
de twee parametriseringen kennen aan hetzelfde stuk oppervlak
dezelfde \omterm{def:b2:surfaces:area}{oppervlakte} toe.
\end{solution}

\begin{solution}{exo:b2:surfaces:9}
In de bolkaart is $z = R\sin\varphi$, en de zone stemt overeen met
$\varphi_1 \leq \varphi \leq \varphi_2$ met $z_i = R\sin\varphi_i$.
Met het oppervlakte-element $R^2\cos\varphi\,\dd\theta\,\dd\varphi$
(\cref{ex:b2:surfaces:spherearea}):
\[
\mathcal A = \int_{\varphi_1}^{\varphi_2}\!\!\int_0^{2\pi}
R^2\cos\varphi\,\dd\theta\,\dd\varphi
= 2\pi R^2(\sin\varphi_2 - \sin\varphi_1)
= 2\pi R\,(z_2 - z_1) .
\]
Het resultaat hangt alleen van de hoogte $z_2 - z_1$ af: schijven
van gelijke dikte dragen gelijke \omterm{def:b2:surfaces:area}{oppervlakten}, of zij nu bij de
evenaar of bij de pool worden gesneden --- de hoedendoosstelling
van Archimedes, en de reden waarom de manteloppervlakte van de
omgeschreven cilinder ($2\pi R \cdot 2R = 4\pi R^2$) gelijk is aan
de \omterm{def:b2:surfaces:area}{oppervlakte} van de bol.
\end{solution}

\begin{solution}{exo:b2:surfaces:10}
$\sigma_\theta = (-r\sin\theta,\ r\cos\theta,\ 0)$ en $\sigma_z =
(r'\cos\theta,\ r'\sin\theta,\ 1)$, dus
\[
\sigma_\theta \wedge \sigma_z
= (r\cos\theta,\ r\sin\theta,\ -r\,r') ,
\]
een normaalvector in $M = (r\cos\theta, r\sin\theta, z)$. De
normaallijn is
\[
t \mapsto \bigl(r(1 + t)\cos\theta,\ r(1 + t)\sin\theta,\
z - t\,r\,r'\bigr),
\]
die in $t = -1$ het punt $(0,\ 0,\ z + r(z)\,r'(z))$ bereikt: elke
normaallijn snijdt de as, op hoogte $z + rr'$. (Dit is de
driedimensionale reden waarom de omwentelingssymmetrie in het
normaalveld overleeft.)
\end{solution}

\begin{solution}{exo:b2:surfaces:11}
$\sigma_u = (-\sin u, \cos u, 0)$, $\sigma_v = (0, 0, 1)$: $E = 1$,
$F = 0$, $G = 1$. Volgens \cref{prop:b2:surfaces:curvelength} is de
\omterm{def:b2:curves:length}{lengte} van een getekende kromme $\int\sqrt{u'^2 + v'^2}\,\dd t$ ---
de \omterm{def:b2:curves:length}{lengte} van haar parameterschaduw $(u(t), v(t))$ in het vlak: de
kaart is een lokale isometrie (het afrollen van de cilinder). De
schroeflijn $t \mapsto (\cos t, \sin t, ct)$, $t \in
\intcc0{2\pi}$, heeft als schaduw het segment van $(0,0)$ naar
$(2\pi, 2\pi c)$, met \omterm{def:b2:curves:length}{lengte} $2\pi\sqrt{1 + c^2}$. Elke getekende
kromme van $(1,0,0)$ naar $(1, 0, 2\pi c)$ die één volle
omwenteling maakt, heeft een \omterm{def:b2:metric:continuity}{continue} schaduw die $(0, 0)$ met
$(2\pi, 2\pi c)$ verbindt, met vlakke \omterm{def:b2:curves:length}{lengte} $\geq$ die van het
rechte segment; omdat de \omterm{def:b2:curves:length}{lengten} overeenstemmen, is de schroeflijn
de kortste.
\end{solution}

\begin{solution}{exo:b2:surfaces:12}
De functie $g(M) = \norm{M}^2$ is \omterm{def:b2:metric:continuity}{continu} op het \omterm{def:b2:metric:compact}{compacte} $S$, dus
neemt zij haar maximum aan in een zeker punt $M_0$. Voor elke op
$S$ getekende kromme $\gamma$ met $\gamma(0) = M_0$ heeft de
functie $t \mapsto \norm{\gamma(t)}^2$ een maximum in $t = 0$, dus
verdwijnt haar afgeleide $2\langle\gamma'(0), M_0\rangle$: $M_0$
staat loodrecht op elke raakvector, dat wil zeggen normaal op $S$
in $M_0$. Omdat ook $\nabla f(M_0) \neq 0$ de normaallijn richt
(\cref{prop:b2:surfaces:gradient}), zijn $\nabla f(M_0)$ en
$\vect{OM_0}$ collineair. Voor de ellipsoïde betekent radialiteit
\[
\Bigl(\frac{x_0}{a^2}, \frac{y_0}{b^2},
\frac{z_0}{c^2}\Bigr) = \mu\,(x_0, y_0, z_0) :
\]
elke coördinaat voldoet aan $x_0(\frac1{a^2} - \mu) = 0$,
enzovoort; omdat $a^{-2}, b^{-2}, c^{-2}$ verschillend zijn, is
hoogstens één coördinaat ongelijk aan nul, en de oplossingen op het
oppervlak zijn de zes uiteinden van de assen $(\pm a, 0, 0)$, $(0,
\pm b, 0)$, $(0, 0, \pm c)$. De verste punten zijn $(\pm a, 0, 0)$,
op afstand $a = \max(a,b,c)$.
\end{solution}

\begin{solution}{pb:b2:surfaces:1}
\textbf{1.} Uitwerken, met de symmetrie van $A$ ($t^{\mathsf
T}\!APY = (At)^{\mathsf T}PY$), geeft
\[
q(PY + t) = Y^{\mathsf T}P^{\mathsf T}\!APY +
2\,(At + b)^{\mathsf T}PY + \bigl(t^{\mathsf T}\!At +
2b^{\mathsf T}t + c\bigr),
\]
wat het getoonde drietal is. $A' = P^{\mathsf T}\!AP = P^{-1}AP$ is
gelijkvormig met $A$: dezelfde \omterm{def:b2:reduction:charpoly}{karakteristieke veelterm}, hetzelfde
spectrum, dezelfde rang en signatuur. Ten slotte is $\{q = 0\} =
\{sq = 0\}$ voor $s \neq 0$, dus hoort bij de verzameling alleen de
vergelijking \emph{op een scalair na}; schalen met $s$
vermenigvuldigt alle \omterm{def:b2:reduction:eigen}{eigenwaarden} met $s$.

\textbf{2.} De spectraalstelling levert een $P \in O(3)$ met
$P^{\mathsf T}\!AP = \operatorname{diag}(\lambda_1, \lambda_2,
\lambda_3)$; vraag 1 met $t = 0$ maakt van de vergelijking
$\sum\lambda_iy_i^2 + 2\sum\beta_iy_i + c = 0$, met $\beta =
P^{\mathsf T}b$.

\textbf{3.} Voor $\lambda_i \neq 0$: $\lambda_iy_i^2 +
2\beta_iy_i = \lambda_i\bigl(y_i +
\frac{\beta_i}{\lambda_i}\bigr)^2 - \frac{\beta_i^2}{\lambda_i}$;
de translatie $z_i = y_i + \beta_i/\lambda_i$ (en $z_i = y_i$ voor
$i > r$) geeft
\[
\sum_{i \leq r}\lambda_iz_i^2 + 2\sum_{i > r}\beta_iz_i +
c'' = 0, \qquad
c'' = c - \sum_{i\leq r}\frac{\beta_i^2}{\lambda_i}.
\]

\textbf{4.} $q(2\Omega - X) = (2\Omega - X)^{\mathsf T} A(2\Omega -
X) + 2b^{\mathsf T}(2\Omega - X) + c$; uitwerken en $q(X)$
aftrekken laat de kwadratische termen wegvallen en geeft
\[
q(2\Omega - X) - q(X) = -4\,\langle A\Omega + b,\ X\rangle +
4\,\langle A\Omega + b,\ \Omega\rangle .
\]
Is $A\Omega + b = 0$, dan verdwijnt dit identiek: de puntspiegeling
bewaart $q$ en dus de \omterm{pb:b2:surfaces:1}{kwadriek}. Omgekeerd zegt ``$q(2\Omega - X) =
q(X)$ voor alle $X$'' dat de \omterm{def:b2:affine:subspace}{affiene} functie hierboven op heel
$\R^3$ verdwijnt, wat afdwingt dat haar lineaire deel $A\Omega + b$
nul is. Middelpunten $=$ oplossingen van $A\Omega = -b$: een
niet-lege verzameling dan en slechts dan als $b \in
\operatorname{im}A$ (een \omterm{def:b2:affine:subspace}{affiene deelruimte} met richting $\ker A$),
en één enkel punt dan en slechts dan als $A$ inverteerbaar is.

\textbf{5.} Signatuur $(3,0)$ (alle $\lambda_i > 0$): $\delta > 0$
geeft $\frac{x^2}{a^2} + \frac{y^2}{b^2} + \frac{z^2}{c^2} = 1$ met
$a = \sqrt{\delta/\lambda_1}$, enzovoort --- een ellipsoïde;
$\delta = 0$: het enkele punt $O$; $\delta < 0$: leeg. Signatuur
$(2,1)$ ($\lambda_1, \lambda_2 > 0 > \lambda_3$): $\delta > 0$:
$\frac{x^2}{a^2} + \frac{y^2}{b^2} - \frac{z^2}{c^2} = 1$, de
eenbladige hyperboloïde; $\delta = 0$: de kegel $\frac{x^2}{a^2} +
\frac{y^2}{b^2} = \frac{z^2}{c^2}$; $\delta < 0$:
$\frac{z^2}{c^2} - \frac{x^2}{a^2} - \frac{y^2}{b^2} = 1$, de
tweebladige hyperboloïde ($\abs z \geq c$: twee componenten).

\textbf{6.} Het kwadratische deel heeft matrix $A$ met diagonaal
$1$ en elementen $2$ buiten de diagonaal: $A = 2J - I$. Omdat $J$
spectrum $\{3, 0, 0\}$ heeft (\omterm{def:b2:reduction:eigen}{eigenvector} $(1,1,1)$ bij $3$), heeft
$A$ spectrum $\{5, -1, -1\}$, met de \omterm{def:b2:reduction:eigen}{eigenwaarde} $5$ gedragen door
$\R(1,1,1)$. In de gedraaide coördinaten: $5u^2 - v^2 - w^2 = 1$,
dat wil zeggen $\frac{u^2}{1/5} - v^2 - w^2 = 1$: een tweebladige
hyperboloïde, en wel een omwentelingshyperboloïde (gelijke
\omterm{def:b2:reduction:eigen}{eigenwaarden} $-1$) om de as $\R(1,1,1)$.

\textbf{7.} Het oppervlak is $\bigl(\frac xa - \frac
zc\bigr)\bigl(\frac xa + \frac zc\bigr) = \bigl(1 - \frac
yb\bigr)\bigl(1 + \frac yb\bigr)$. Definieer voor $(\lambda : \mu)
\neq (0:0)$ de rechte
\[
D_{\lambda:\mu} :\quad
\lambda\Bigl(\frac xa - \frac zc\Bigr) = \mu\Bigl(1 - \frac
yb\Bigr), \qquad
\mu\Bigl(\frac xa + \frac zc\Bigr) = \lambda\Bigl(1 + \frac
yb\Bigr)
\]
(twee onafhankelijke \omterm{def:b2:affine:subspace}{affiene} vergelijkingen: een rechte). De twee
vergelijkingen vermenigvuldigen toont dat elk punt van
$D_{\lambda:\mu}$ op het oppervlak ligt wanneer $\lambda\mu \neq
0$; de gevallen $\lambda = 0$ of $\mu = 0$ gaat men rechtstreeks na
(bijvoorbeeld $\lambda = 0$: $y = b$, $\frac xa = -\frac zc$, wat
aan de vergelijking voldoet). De tweede familie
$D'_{\lambda:\mu}$ verwisselt de twee factoren in het rechterlid.

\textbf{8.} Houd $M$ op het oppervlak vast. De voorwaarden voor $M
\in D_{\lambda:\mu}$ vormen een homogeen lineair
$2\times2$-stelsel in $(\lambda, \mu)$ waarvan de \omterm{def:b2:linalg:det}{determinant}
\[
\Bigl(\frac{x^2}{a^2} - \frac{z^2}{c^2}\Bigr) - \Bigl(1 -
\frac{y^2}{b^2}\Bigr) = 0
\]
is, precies omdat $M$ op de \omterm{pb:b2:surfaces:1}{kwadriek} ligt: er bestaat een
niet-triviale oplossing $(\lambda : \mu)$. De
coëfficiëntenmatrix is nooit nul (dat zou $1 - \frac yb = 1 +
\frac yb = 0$ afdwingen), dus is haar rang $1$ en is de oplossing
op een schaalfactor na uniek: precies één rechte van de familie
gaat door $M$. Hetzelfde geldt voor de tweede familie, en de twee
rechten zijn verschillend (in $(a, 0, 0)$ zijn zij $\{x = a,\
\frac zc = \frac yb\}$ en $\{x = a,\ \frac zc = -\frac yb\}$): de
eenbladige hyperboloïde is dubbel beschreven.

\textbf{9.} Zij $M = (x, y, z)$ op de hyperboloïde, $\rho^2 =
\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1 + \frac{z^2}{c^2} \geq 1$.
Het punt $N = (x,\ y,\ \varepsilon c\rho)$ met $\varepsilon$ het
teken van $z$ voldoet aan $\frac{x^2}{a^2} + \frac{y^2}{b^2} -
\frac{(c\rho)^2}{c^2} = 0$: $N \in C$, en
\[
d(M, C) \leq \abs{z - \varepsilon c\rho}
= c\bigl(\rho - \sqrt{\rho^2 - 1}\bigr)
= \frac{c}{\rho + \sqrt{\rho^2 - 1}} .
\]
Gaat $\norm M \to \infty$, dan gaat $\rho \to \infty$ (alle drie de
coördinaten worden door veelvouden van $\rho$ begrensd), dus $d(M,
C) \to 0$. De doorsnede $x = a$: $\frac{y^2}{b^2} -
\frac{z^2}{c^2} = 0$, het paar snijdende rechten van vraag 8 --- en
evenzo in $x = -a$.

\textbf{10.} Met $\lambda_3 = 0$ laat vraag 3 $\lambda_1z_1^2 +
\lambda_2z_2^2 + 2\beta_3z_3 + c'' = 0$ over. In de eigenbasis is
$\operatorname{im}A = \operatorname{Vect}(e_1, e_2)$, dus bestaan
er volgens vraag 4 middelpunten dan en slechts dan als $\beta_3 =
0$. Is $\beta_3 \neq 0$, dan verwijdert de translatie $z_3 \mapsto
z_3 - c''/(2\beta_3)$ de constante en blijft $\lambda_1z_1^2 +
\lambda_2z_2^2 + 2\beta_3z_3 = 0$ over, dat wil zeggen $z_3 = px^2
+ qy^2$ na hernoemen: een \emph{elliptische paraboloïde} als
$\lambda_1\lambda_2 > 0$, een \emph{hyperbolische paraboloïde} als
$\lambda_1\lambda_2 < 0$ --- en inderdaad geen middelpunt. Is
$\beta_3 = 0$, dan bevat de vergelijking $\lambda_1z_1^2 +
\lambda_2z_2^2 + c'' = 0$ de veranderlijke $z_3$ niet: de \omterm{pb:b2:surfaces:1}{kwadriek}
is een cilinder boven de bijbehorende vlakke kegelsnede ---
elliptische cilinder, rechte of lege verzameling als
$\lambda_1\lambda_2 > 0$; hyperbolische cilinder of paar snijdende
vlakken als $\lambda_1\lambda_2 < 0$ --- met een hele rechte
middelpunten $\{(z_1^*, z_2^*)\} \times \R$.

\textbf{11.} $xy - z = 0$. De substitutie $x = \frac{u +
v}{\sqrt2}$, $y = \frac{u - v}{\sqrt2}$ (rotatie over $\pi/4$)
geeft $xy = \frac{u^2 - v^2}2$, zodat de vergelijking $z =
\frac12(u^2 - v^2)$ wordt: een hyperbolische paraboloïde --- het
zadel van de figuur, op de factor $\frac12$ na.

\textbf{12.} De rechte $\{x = x_0,\ z = x_0y\}$ (geparametriseerd
met $y$) ligt duidelijk op $z = xy$, en $\{y = y_0,\ z = xy_0\}$
eveneens; door $(x_0, y_0, x_0y_0)$ gaan zij beide. Eenduidigheid:
blijft $t \mapsto (x_0 + tv_1, y_0 + tv_2, z_0 + tv_3)$ op het
oppervlak, dan geeft de coëfficiënt van $t^2$ in $(x_0 +
tv_1)(y_0 + tv_2) - z_0 - tv_3$ dat $v_1v_2 = 0$, dus $v_1 = 0$ of
$v_2 = 0$, wat in een van de twee families belandt: door elk punt
één rechte van elke familie --- de tweede dubbel beschreven
\omterm{pb:b2:surfaces:1}{kwadriek}.

\textbf{13.} De kwadraten afmaken: $(x - 1)^2 + (y + 2)^2 = 2$,
zonder enige voorwaarde op $z$: een rechte cirkelvormige cilinder
met straal $\sqrt2$ en als as de verticale rechte $\{(1, -2, z) : z
\in \R\}$ --- een rechte middelpunten, zoals vraag 10 voorspelt.

\textbf{14.} Met $\lambda_2 = \lambda_3 = 0$ is de herleide
vergelijking $\lambda_1z_1^2 + 2\beta_2z_2 + 2\beta_3z_3 + c'' =
0$. Een rotatie van het kernvlak $(z_2, z_3)$ lijnt de lineaire vorm
uit: $2\beta_2z_2 + 2\beta_3z_3 = 2\beta w$ met $\beta =
\sqrt{\beta_2^2 + \beta_3^2}$. Is $\beta \neq 0$, transleer dan $w$
om $c''$ op te slorpen: $\lambda_1z_1^2 + 2\beta w = 0$, een
\emph{parabolische cilinder}; is $\beta = 0$, dan geeft
$\lambda_1z_1^2 = -c''$ twee evenwijdige vlakken ($c'' \lambda_1 <
0$), een dubbel vlak ($c'' = 0$) of de lege verzameling. Voor $(x +
y)^2 = z$: met $u = \frac{x + y}{\sqrt2}$ luidt de vergelijking $z
= 2u^2$: een parabolische cilinder, invariant onder translaties
langs $(1, -1, 0)$.

\textbf{15.} Elke \omterm{pb:b2:surfaces:1}{kwadriek} wordt door een rotatie plus translaties
gebracht op een van: (rang 3) ellipsoïde, punt, lege verzameling,
eenbladige hyperboloïde, kegel, tweebladige hyperboloïde; (rang 2)
elliptische paraboloïde, hyperbolische paraboloïde, elliptische
cilinder, rechte, lege verzameling, hyperbolische cilinder, paar
snijdende vlakken; (rang 1) parabolische cilinder, paar
evenwijdige vlakken, dubbel vlak, lege verzameling. Identificeren
wij de drie lege varianten als verschillende \emph{\omterm{def:b2:affine:subspace}{affiene} types}
van vergelijkingen, dan komt de telling op zeventien; daarvan zijn
er negen echte oppervlakken: ellipsoïde, de twee hyperboloïden, de
kegel, de twee paraboloïden en de drie cilinders.

\textbf{16.} Algoritme. (i) Lees $(A, b, c)$ af; bereken de
\omterm{def:b2:reduction:charpoly}{karakteristieke veelterm} van $A$, haar \omterm{def:b2:reduction:eigen}{eigenwaarden} (reëel, volgens
de spectraalstelling) en $r = \operatorname{rank}A$. (ii) Los
$A\Omega = -b$ op (eliminatie van Gauss): oplosbaar of niet ---
middelpunten of niet. (iii) Is zij oplosbaar, transleer dan naar
een middelpunt: de vergelijking wordt $\sum\lambda_iz_i^2 + c'' =
0$ met $c'' = c + \langle b, \Omega\rangle$; sorteer naar $r$, de
signatuur en het teken van $c''$ met de vragen 5, 10 en 14. (iv) Is
zij niet oplosbaar ($r \leq 2$), draai en herleid dan als in de
vragen 10 en 14: paraboloïde ($r = 2$) of parabolische cilinder ($r
= 1$), elliptisch of hyperbolisch naargelang het teken van
$\lambda_1\lambda_2$. Elke stap is een eindige berekening:
nulpunten van een derdegraadsveelterm met reële nulpunten, rangen,
lineaire stelsels.

\textbf{17.} $A$ heeft diagonaal $1$ en $-1$ buiten de diagonaal:
$A = 2I - J$, met spectrum $\{2 - 3,\ 2,\ 2\} = \{-1, 2, 2\}$,
waarbij $-1$ op $\R(1,1,1)$ zit. Signatuur $(2,1)$, $b = 0$,
rechterlid $\delta = 1 > 0$: $2u^2 + 2v^2 - w^2 = 1$, een
eenbladige omwentelingshyperboloïde om de as $\R(1,1,1)$.

\textbf{18.} De kwadraten afmaken: $(x-1)^2 + (y+2)^2 - (z-1)^2 +
(-1 - 4 + 1 + 4) = 0$, dat wil zeggen
\[
(x-1)^2 + (y+2)^2 = (z-1)^2 :
\]
de constante verdween --- een rechte cirkelvormige \emph{kegel} met
top (en uniek middelpunt) $(1, -2, 1)$ en as evenwijdig met $Oz$.

\textbf{19.} Het kwadratische deel $x^2 + 4xy + y^2$ heeft matrix
$\begin{psmallmatrix}1 & 2\\ 2 & 1\end{psmallmatrix}$ (in het
$xy$-vlak), met \omterm{def:b2:reduction:eigen}{eigenwaarden} $3$ (op $(1,1)$) en $-1$ (op
$(1,-1)$): met $u = \frac{x+y}{\sqrt2}$ en $v =
\frac{x-y}{\sqrt2}$ is het gelijk aan $3u^2 - v^2$, en de \omterm{pb:b2:surfaces:1}{kwadriek}
is
\[
z = \tfrac32u^2 - \tfrac12v^2 :
\]
een hyperbolische paraboloïde ($A$ heeft rang $2$ en $b$ heeft een
component langs $\ker A = \R e_z$: geen middelpunt).

\textbf{20.} Een starre beweging transformeert de gegevens van de
vergelijking als $A \mapsto P^{\mathsf T}\!AP$ (dezelfde
\omterm{def:b2:reduction:eigen}{eigenwaarden}), en de normaalvormen hebben geen resterende vrijheid
behalve het permuteren van coördinaten en het vermenigvuldigen van
de hele vergelijking met een scalair ($> 0$ om de schrijfwijze te
bewaren): twee centrale normaalvormen vallen op een isometrie na
samen dan en slechts dan als de lijsten coëfficiënten op een
permutatie en een gemeenschappelijke positieve factor na
overeenstemmen --- voor de ellipsoïde dan en slechts dan als de
halve assen $(a, b, c)$ overeenstemmen. Affien mag men bovendien
elke coördinaat afzonderlijk schalen ($z_i \mapsto
z_i\sqrt{\abs{\lambda_i}}$), wat de \omterm{def:b2:reduction:eigen}{eigenwaarden} uitwist en alleen
hun tekens overlaat: elke ellipsoïde wordt $u^2 + v^2 + w^2 = 1$,
de bol. De traagheidsstelling van Sylvester
(\cref{thm:b2:quadratic:sylvester}) waarborgt dat de signatuur elke
inverteerbare lineaire verandering overleeft: de \omterm{def:b2:affine:subspace}{affiene} types van
vraag 15 zijn werkelijk verschillend.

\textbf{21.} Parametriseer het vlak affien: $M = P + su + tv$. Dan
is $q(P + su + tv)$ een veelterm van graad $\leq 2$ in $(s, t)$
(werk de \omterm{def:b2:quadratic:def}{kwadratische vorm} bilineair uit), dus is de doorsnede $\{q
= 0\}$ een kegelsnede van het vlak, mogelijk ontaard. Voor $z = xy$
en het vlak $z = c$: $xy = c$, een hyperbool voor $c \neq 0$, en
voor $c = 0$ de twee coördinaatassen --- het paar beschrijvende
rechten door de oorsprong.

\textbf{22.} \emph{Ellipsoïde}: begrensd, bevat geen rechte.
\emph{Tweebladige hyperboloïde} $\frac{z^2}{c^2} - \frac{x^2}{a^2}
- \frac{y^2}{b^2} = 1$: beperk tot $p + tv$; de coëfficiënt van
$t^2$, $\frac{v_3^2}{c^2} - \frac{v_1^2}{a^2} - \frac{v_2^2}{b^2}$,
moet verdwijnen, dus $v_3 \neq 0$ (anders $v = 0$); de coëfficiënt
van $t$ geeft $\frac{p_3v_3}{c^2} = \frac{p_1v_1}{a^2} +
\frac{p_2v_2}{b^2}$, en Cauchy--Schwarz levert
\[
\frac{p_3^2}{c^2}
= \frac{c^2}{v_3^2}\Bigl(\frac{p_1v_1}{a^2} +
\frac{p_2v_2}{b^2}\Bigr)^2
\leq \frac{c^2}{v_3^2}\Bigl(\frac{p_1^2}{a^2} +
\frac{p_2^2}{b^2}\Bigr)\frac{v_3^2}{c^2}
= \frac{p_1^2}{a^2} + \frac{p_2^2}{b^2},
\]
dus is de constante term $\leq 0 \neq 1$: geen rechte.
\emph{Elliptische paraboloïde} $z = \frac{x^2}{a^2} +
\frac{y^2}{b^2}$: de coëfficiënt van $t^2$ dwingt $v_1 = v_2 = 0$
af, en dan is de vergelijking lineair en niet constant in $t$: geen
rechte. \emph{Kegel} $x^2 + y^2 = z^2$: een rechte erop voldoet aan
$q(p) = q(v) = B(p, v) = 0$ voor de lorentzvorm; gelijkheid in de
vlakke Cauchy--Schwarz $\abs{p_1v_1 + p_2v_2} = \abs{p_3v_3} =
\sqrt{(p_1^2 + p_2^2)(v_1^2 + v_2^2)}$ dwingt $(p_1, p_2) \parallel
(v_1, v_2)$ af en vervolgens $p = kv$: alle rechten gaan door de
top --- één familie. \emph{Cilinders}: voor de elliptische en de
parabolische cilinder dwingt de coëfficiënt van $t^2$ af dat $v_1 =
v_2 = 0$ (alleen de beschrijvende rechten); voor de hyperbolische
cilinder $\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1$ leidt $\frac{v_1}a
= \pm \frac{v_2}b$ met $v_2 \neq 0$ via de coëfficiënt van $t$ tot
$\frac{p_1^2}{a^2} = \frac{p_2^2}{b^2}$, in tegenspraak met de
constante term $1$: opnieuw alleen de verticale beschrijvende
rechten. De dubbel beschreven \omterm{pb:b2:surfaces:1}{kwadrieken} zijn dus precies de
eenbladige hyperboloïde en de hyperbolische paraboloïde.

\textbf{23.} Gauss: $x^2 - 2xy - 2zx = (x - y - z)^2 - y^2 - z^2 -
2yz$, dus
\[
q = (x - y - z)^2 - 4yz = (x - y - z)^2 + (y - z)^2 - (y +
z)^2 :
\]
drie onafhankelijke kwadraten met tekens $(+, +, -)$ --- signatuur
$(2, 1)$, overeenkomend met vraag 17, en zonder enige berekening
van \omterm{def:b2:reduction:eigen}{eigenwaarden}. Gauss verliest de metrische gegevens: de nieuwe
coördinaten zijn niet orthonormaal, dus de \omterm{def:b2:reduction:eigen}{eigenwaarden} (de vorm
van de hyperboloïde, haar assen en hun \omterm{def:b2:curves:length}{lengten}) zijn weg; alleen
het \omterm{def:b2:affine:subspace}{affiene} type blijft over.

\textbf{24.} Zijn $p$, $n$, $z$ de aantallen positieve, negatieve
en verdwijnende \omterm{def:b2:reduction:eigen}{eigenwaarden}, met $p + n + z = 3$. De regel van
Descartes begrenst $p$ door het aantal $V$ tekenwisselingen van
$\chi_A$, en $n$ door het aantal $V'$ tekenwisselingen van
$\chi_A(-\lambda)$; bovendien brengt elk paar opeenvolgende
coëfficiënten ongelijk aan nul in precies één van de twee
veeltermen een wisseling voort, dus $V + V' \leq 3 - z$
(verdwijnende nulpunten zijn zichtbaar als verdwijnende laatste
coëfficiënten). Dan is $p + n = 3 - z \geq V + V' \geq p + n$:
gelijkheid, dus $p = V$ precies --- de tekens van de \omterm{def:b2:reduction:eigen}{eigenwaarden}
zijn af te lezen. Voor $\chi_A(\lambda) = \lambda^3 - 3\lambda^2 -
9\lambda - 5$: de tekens $+,-,-,-$ geven $V = 1$, en
$\chi_A(-\lambda) = -\lambda^3 - 3\lambda^2 + 9\lambda - 5$ heeft
tekens $-,-,+,-$: $V' = 2$. Signatuur $(1, 2)$ --- in
overeenstemming met de exacte ontbinding $\chi_A = (\lambda -
5)(\lambda + 1)^2$ van vraag 6.

\textbf{25.} Algoritme: diagonaliseer het kwadratische deel
orthonormaal (spectraalstelling: dit is de enige \omterm{def:b2:powerseries:analytic}{analytisch} diepe
stap, en zij is wat de klassering \emph{euclidisch} maakt); los
$A\Omega = -b$ op om over centrale tegenover parabolische types te
beslissen en om het lineaire deel weg te transleren waar dat kan
(\omterm{def:b2:affine:subspace}{affiene} meetkunde); lees het type af uit rang, signatuur en de
constante (Sylvester waarborgt dat dit invarianten zijn); telt
alleen het \omterm{def:b2:affine:subspace}{affiene} type, dan vervangt de reductie van Gauss de
spectraalstelling, ten koste van de metrische informatie. In
$\R^2$ klasseert dezelfde machine de kegelsneden: ellips,
hyperbool, parabool, plus paren rechten, een rechte, een punt en
lege verzamelingen. In $\R^n$ verandert er niets dan de
boekhouding: de types worden geïndexeerd door de signatuur van $A$,
de ligging van $b$ ten opzichte van $\operatorname{im}A$ en één
constante --- waarbij de omrande matrix van $(A, b, c)$ met
$(n{+}2)$ rijen een \omterm{def:b2:metric:compact}{compacte} invariant levert.
\end{solution}
