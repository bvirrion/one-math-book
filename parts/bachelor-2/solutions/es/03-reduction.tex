\section*{Capítulo \ref{ch:b2:reduction} --- Reducción de endomorfismos}

\begin{solution}{exo:b2:reduction:1}
$A$: $\chi_A = X^2 - 2X - 3 = (X - 3)(X + 1)$. \omterm{def:b2:reduction:eigen}{Vectores propios}:
para $3$, $(1,1)$; para $-1$, $(1,-1)$. Así pues, $P =
\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}$ da $P^{-1}AP =
\operatorname{diag}(3, -1)$.

$B = J - I$, donde $J$ es la matriz de unos. $J$ tiene rango $1$, con
$Jv = 3v$ para $v = (1,1,1)$ y $Jw = 0$ sobre el plano $x + y + z =
0$: el \omterm{def:b2:reduction:eigen}{espectro} de $B$ es $\{2, -1\}$, con \omterm{def:b2:reduction:eigen}{subespacios propios}
$\operatorname{Vect}(1,1,1)$ (de dimensión $1$) y $\{x + y + z = 0\}$
(de dimensión $2$, con base $(1,-1,0), (1,0,-1)$). La matriz $P$ con
esas tres columnas da $P^{-1}BP = \operatorname{diag}(2, -1, -1)$.
\end{solution}

\begin{solution}{exo:b2:reduction:2}
\emph{Por \omterm{def:b2:reduction:eigen}{subespacios propios}:} $\chi_C = (X-1)^2$, con único
\omterm{def:b2:reduction:eigen}{valor propio} $1$; $\ker(C - I) = \ker\begin{pmatrix} 0&1\\
0&0\end{pmatrix}$ es la recta $\operatorname{Vect}(e_1)$, de
dimensión $1 < 2 = m_1$, luego no es \omterm{def:b2:reduction:diag}{diagonalizable}
(\cref{thm:b2:reduction:diagcrit}).

\emph{Por el \omterm{def:b2:reduction:polyu}{polinomio mínimo}:} $\mu_C$ divide a $(X-1)^2$ y $C
\neq I$, luego $\mu_C = (X-1)^2$: una raíz doble, así que no es
\omterm{def:b2:reduction:diag}{diagonalizable} (\cref{cor:b2:reduction:minpolycrit}).
\end{solution}

\begin{solution}{exo:b2:reduction:3}
$X^2 - 5X + 6 = (X-2)(X-3)$: escindido con raíces simples, luego $u$
es \omterm{def:b2:reduction:diag}{diagonalizable} (\cref{cor:b2:reduction:minpolycrit}), con
$\operatorname{Sp}(u) \subseteq \{2, 3\}$. \omterm{def:b2:reduction:eigen}{Espectros} posibles:
$\{2\}$ ($u = 2\,\mathrm{id}$), $\{3\}$ ($u = 3\,\mathrm{id}$) o
$\{2, 3\}$.

Potencias: busquemos $u^k = a_k\,\mathrm{id} + b_k\,u$. Sobre los
\omterm{def:b2:reduction:eigen}{subespacios propios} esto se lee $2^k = a_k + 2b_k$ y $3^k = a_k +
3b_k$; resolviendo, $b_k = 3^k - 2^k$ y $a_k = 3\cdot2^k - 2\cdot
3^k$:
\[
u^k = (3\cdot 2^k - 2\cdot 3^k)\,\mathrm{id} + (3^k - 2^k)\, u .
\]
(Válido para los tres \omterm{def:b2:reduction:eigen}{espectros}: las identidades se cumplen \omterm{def:b2:reduction:eigen}{valor
propio} a \omterm{def:b2:reduction:eigen}{valor propio}.)
\end{solution}

\begin{solution}{exo:b2:reduction:4}
$u$ es \omterm{def:b2:reduction:diag}{diagonalizable}: $P = \prod_{\lambda}(X - \lambda)$ sobre el
\omterm{def:b2:reduction:eigen}{espectro} anula $u$, es escindido y tiene raíces simples. Entonces
$P(u|_F) = P(u)|_F = 0$: la restricción está anulada por un polinomio
escindido con raíces simples, luego es \omterm{def:b2:reduction:diag}{diagonalizable}
(\cref{cor:b2:reduction:minpolycrit}).
\end{solution}

\begin{solution}{exo:b2:reduction:5}
$\chi_A = X^2 - X - 1$, con raíces $\varphi$ y $\psi$ (distintas):
\omterm{def:b2:reduction:diag}{diagonalizable}, con \omterm{def:b2:reduction:eigen}{vectores propios} $(\varphi, 1)$ y $(\psi,
1)$. La recurrencia da $\begin{pmatrix} F_{n+1}\\ F_n \end{pmatrix} =
A^n \begin{pmatrix}1\\ 0\end{pmatrix}$. Descompongamos $(1, 0)$ sobre
los \omterm{def:b2:reduction:eigen}{vectores propios}: $(1,0) = \frac{1}{\varphi -
\psi}\bigl((\varphi, 1) - (\psi, 1)\bigr)$ con $\varphi - \psi =
\sqrt5$. Aplicar $A^n$ multiplica cada componente propia por la
potencia $n$-ésima de su \omterm{def:b2:reduction:eigen}{valor propio}; leyendo la segunda
coordenada:
\[
F_n = \frac{\varphi^n - \psi^n}{\sqrt 5} .
\]
(Comprobación: para $n = 1$ se obtiene $\frac{\varphi -
\psi}{\sqrt5} = 1$.)
\end{solution}

\begin{solution}{exo:b2:reduction:6}
Sea $P = \prod_i (X - \mu_i)$ un polinomio que anula $u^2$, escindido
con raíces simples $\mu_i$ (el \omterm{def:b2:reduction:eigen}{espectro} de $u^2$). Como $u$ es
invertible, $0$ no es \omterm{def:b2:reduction:eigen}{valor propio} de $u^2$ (pues $\det u^2 = (\det
u)^2 \neq 0$), así que todos los $\mu_i \neq 0$. Entonces
\[
Q(X) = \prod_i (X^2 - \mu_i) = \prod_i (X - \sqrt{\mu_i})(X +
\sqrt{\mu_i})
\]
anula $u$: en efecto, $Q(u) = \prod_i (u^2 - \mu_i\,\mathrm{id}) =
P(u^2) = 0$. Sus raíces $\pm \sqrt{\mu_i}$ (raíces cuadradas
complejas) son distintas dos a dos porque los $\mu_i$ son distintos y
no nulos ($\sqrt{\mu_i} = -\sqrt{\mu_j}$ daría $\mu_i = \mu_j$).
Escindido y con raíces simples: $u$ es \omterm{def:b2:reduction:diag}{diagonalizable}.

Contraejemplo sin invertibilidad: $u = \begin{pmatrix} 0 & 1\\ 0 &
0\end{pmatrix}$: $u^2 = 0$ es \omterm{def:b2:reduction:diag}{diagonalizable} y $u$ no lo es.
\end{solution}

\begin{solution}{exo:b2:reduction:7}
$A = 2I + N$ con $N$ el desplazamiento ($N e_2 = e_1$, $Ne_3 = e_2$),
$N^3 = 0$ y $N^2 = E_{13}$: esta \emph{es} la \omterm{thm:b2:reduction:dunford}{descomposición de
Dunford} ($2I$ diagonal, $N$ nilpotente, y conmutan; la unicidad hace
que sea la única). Binomio con términos que conmutan:
\[
A^k = 2^k I + k 2^{k-1} N + \binom k2 2^{k-2} N^2
= \begin{pmatrix}
2^k & k2^{k-1} & \binom k2 2^{k-2}\\
0 & 2^k & k2^{k-1}\\
0 & 0 & 2^k
\end{pmatrix},
\]
\[
\eu^{tA} = \eu^{2t}\Bigl(I + tN + \frac{t^2}{2}N^2\Bigr)
= \eu^{2t}\begin{pmatrix}
1 & t & t^2/2\\
0 & 1 & t\\
0 & 0 & 1
\end{pmatrix}.
\]
\end{solution}

\begin{solution}{exo:b2:reduction:8}
$X^k - 1$ anula $A$ y se escinde sobre $\C$ con las $k$ raíces
distintas $\eu^{2\iu\pi j/k}$: $A$ es \omterm{def:b2:reduction:diag}{diagonalizable}
(\cref{cor:b2:reduction:minpolycrit}) y sus \omterm{def:b2:reduction:eigen}{valores propios}, raíces
de $X^k - 1$, son raíces $k$-ésimas de la unidad.

Si además $A$ es semejante a una matriz triangular con diagonal de
unos, todos los \omterm{def:b2:reduction:eigen}{valores propios} valen $1$, y $A$, \omterm{def:b2:reduction:diag}{diagonalizable} con
único \omterm{def:b2:reduction:eigen}{valor propio} $1$, vale $P\,I\,P^{-1} = I$.
\end{solution}

\begin{solution}{exo:b2:reduction:9}
Escribamos $E = \bigoplus_\lambda E_\lambda(u)$
(\cref{thm:b2:reduction:diagcrit}). Cada $E_\lambda(u)$ es estable
por $v$: para $x \in E_\lambda$, $u(v(x)) = v(u(x)) = \lambda v(x)$.
La restricción de $v$ a $E_\lambda(u)$ es \omterm{def:b2:reduction:diag}{diagonalizable}
(\cref{exo:b2:reduction:4}): elíjase una base de $E_\lambda(u)$
formada por \omterm{def:b2:reduction:eigen}{vectores propios} de $v$. Concatenando esas bases sobre
todos los $\lambda$ se obtiene una base de $E$ cuyos vectores son
\omterm{def:b2:reduction:eigen}{vectores propios} de \emph{ambos}, de $u$ (por pertenecer a
$E_\lambda(u)$) y de $v$ (por construcción).
\end{solution}

\begin{solution}{exo:b2:reduction:10}
($\Rightarrow$) Sea $u$ \omterm{def:b2:reduction:diag}{diagonalizable} y $F$ estable. Entonces
$u|_F$ es \omterm{def:b2:reduction:diag}{diagonalizable} (\cref{exo:b2:reduction:4}): $F$ tiene una
base de \omterm{def:b2:reduction:eigen}{vectores propios} que se extiende, dentro de cada
\omterm{def:b2:reduction:eigen}{subespacio propio} global $E_\lambda$, a una base de $E_\lambda$
(teorema de la base incompleta dentro de $E_\lambda$, partiendo de la
parte de la base de $F$ que allí se aloja; obsérvese que $F =
\bigoplus_\lambda (F \cap E_\lambda)$ porque $u|_F$ es
\omterm{def:b2:reduction:diag}{diagonalizable}). Los vectores añadidos generan un suplementario
estable (cada uno está en algún $E_\lambda$, luego su envoltura es
estable por $u$).

($\Leftarrow$) Sea $F = \sum_\lambda E_\lambda(u)$ (subespacio
estable) y $G$ un suplementario estable. Si $G \neq \{0\}$, entonces
$\chi_{u|_G}$ se escinde sobre $\C$, así que $u|_G$ tiene un
\omterm{def:b2:reduction:eigen}{vector propio} $x \in G$
(\cref{thm:b2:reduction:trigonalization}, o directamente la
existencia de una raíz); pero todo \omterm{def:b2:reduction:eigen}{vector propio} de $u$ está en $F$,
luego $x \in F \cap G = \{0\}$: contradicción. Por tanto $G = \{0\}$
y $E = F$: los \omterm{def:b2:reduction:eigen}{subespacios propios} llenan $E$, es decir, $u$ es
\omterm{def:b2:reduction:diag}{diagonalizable}.
\end{solution}

\begin{solution}{exo:b2:reduction:11}
Trigonalicemos: $A = PTP^{-1}$, $T = \begin{pmatrix} \lambda & c\\ 0
& \mu\end{pmatrix}$, con $\abs\lambda, \abs\mu < 1$. Entonces $A^k =
PT^kP^{-1}$, y basta con que $T^k \to 0$.

\emph{\omterm{def:b2:reduction:eigen}{Valores propios} distintos:} por inducción,
\[
T^k = \begin{pmatrix}
\lambda^k & c\,\dfrac{\lambda^k - \mu^k}{\lambda - \mu}\\[4pt]
0 & \mu^k
\end{pmatrix},
\]
y cada entrada tiende a $0$ (pues $\abs{\lambda}^k, \abs\mu^k \to
0$).

\emph{\omterm{def:b2:reduction:eigen}{Valores propios} iguales ($\mu = \lambda$):} $T = \lambda I +
cE_{12}$ y $T^k = \lambda^k I + k\lambda^{k-1}cE_{12}$; la entrada
$k\lambda^{k-1} \to 0$, ya que $\abs\lambda < 1$ (lo geométrico gana
a lo polinómico). En ambos casos $T^k \to 0$ entrada a entrada, luego
$A^k = PT^kP^{-1} \to 0$ (multiplicar por las matrices fijas $P,
P^{-1}$ es continuo en las entradas: cada entrada del producto es una
combinación lineal fija).
\end{solution}

\begin{solution}{exo:b2:reduction:12}
$\ker u$ tiene dimensión $n - 1$ (teorema del rango), de modo que $0$
es un \omterm{def:b2:reduction:eigen}{valor propio} de \omterm{def:b2:reduction:charpoly}{multiplicidad geométrica} $n - 1$, y
$\chi_u$ es divisible por $X^{n-1}$
(\cref{def:b2:reduction:charpoly}: geométrica $\leq$ algebraica).
Escribamos $\chi_u = X^{n-1}(X - \alpha)$; como el coeficiente de
$X^{n-1}$ es $-\operatorname{tr} u$, resulta $\alpha =
\operatorname{tr} u$: $\chi_u = X^{n-1}(X - \operatorname{tr} u)$.

Si $\operatorname{tr} u \neq 0$: el \omterm{def:b2:reduction:eigen}{valor propio} $\operatorname{tr}
u$ es raíz de $\chi_u$, luego lleva asociado un \omterm{def:b2:reduction:eigen}{vector propio}; los
\omterm{def:b2:reduction:eigen}{subespacios propios} de $0$ y de $\operatorname{tr} u$ tienen
dimensiones $n - 1$ y $\geq 1$, que suman $\geq n$: llenan $E$ y $u$
es \omterm{def:b2:reduction:diag}{diagonalizable} (\cref{thm:b2:reduction:diagcrit}). Si
$\operatorname{tr} u = 0$: por el~\cref{exo:b2:linalg:5}, $u^2 =
(\operatorname{tr} u)u = 0$ con $u \neq 0$, así que $u$ es un
nilpotente no nulo, y un nilpotente \omterm{def:b2:reduction:diag}{diagonalizable} es nulo
(\cref{prop:b2:reduction:nilpotent}): no es \omterm{def:b2:reduction:diag}{diagonalizable}.
\end{solution}

\begin{solution}{pb:b2:reduction:1}
\textbf{1.} Las $k - 1$ primeras coordenadas de $Cv_n$ son
$u_{n+1}, \dots, u_{n+k-1}$ (la superdiagonal desplaza), y la
última es $a_0 u_n + \dots + a_{k-1}u_{n+k-1}$. Así pues,
$v_{n+1} = Cv_n$ se cumple para todo $n$ si y solo si coinciden
las últimas coordenadas para todo $n$, es decir, si y solo si se
cumple $(\mathcal R)$. Iterando, $v_n = C^nv_0$.

\textbf{2.} Desarrollemos $D_k(X) = \det(XI_k - C)$ por la primera
columna: las dos entradas no nulas son $X$ (posición $(1,1)$) y
$-a_0$ (posición $(k,1)$). El primer menor tiene la forma de
$D_{k-1}$ para los coeficientes $a_1, \dots, a_{k-1}$; el segundo
menor es triangular superior con diagonal $-1$, de \omterm{def:b2:linalg:det}{determinante}
$(-1)^{k-1}$, y le corresponde el signo $(-1)^{k+1}$ por la
posición. La inducción sobre $k$ (caso base $k = 1$: $X - a_0$) da
\[
D_k(X) = X\bigl(X^{k-1} - a_{k-1}X^{k-2} - \dots - a_1\bigr) -
a_0 = P(X).
\]
Para $\mu_C$: como $Q(C^{\mathsf T}) = Q(C)^{\mathsf T}$ para todo
polinomio, $C$ y $C^{\mathsf T}$ tienen los mismos \omterm{def:b2:linalg:annihilator}{anuladores} y,
por tanto, el mismo \omterm{def:b2:reduction:polyu}{polinomio mínimo}. Para $C^{\mathsf T}$: las
columnas dan $C^{\mathsf T}e_1 = e_2$, \dots, $C^{\mathsf
T}e_{k-1} = e_k$, de modo que $(e_1, C^{\mathsf T}e_1, \dots,
(C^{\mathsf T})^{k-1}e_1)$ es la base canónica, que es libre. Un
polinomio $Q \neq 0$ de grado $< k$ cumple entonces $Q(C^{\mathsf
T})e_1 \neq 0$ (es una combinación no trivial de vectores de la
base): $\deg\mu \geq k$. Y como $\mu \mid \chi = P$ con $\deg P =
k$, resulta $\mu_C = P$.

\textbf{3.} Para $v = (1, \lambda, \dots, \lambda^{k-1})^{\mathsf
T}$: las filas $1$ a $k-1$ de $Cv$ dan $\lambda, \lambda^2, \dots,
\lambda^{k-1}$, es decir, $\lambda$ por las $k - 1$ primeras
entradas de $v$; la última fila da $\sum_m a_m\lambda^m =
\lambda^k - P(\lambda) = \lambda^k = \lambda\cdot\lambda^{k-1}$.
Luego $Cv = \lambda v$. Recíprocamente, las ecuaciones $(Cx)_i =
\lambda x_i$ para $i < k$ se leen $x_{i+1} = \lambda x_i$: todo
\omterm{def:b2:reduction:eigen}{vector propio} es proporcional a $v$, y por tanto todo
\omterm{def:b2:reduction:eigen}{subespacio propio} tiene dimensión exactamente $1$. Es
\omterm{def:b2:reduction:diag}{diagonalizable} si y solo si las dimensiones de los
\omterm{def:b2:reduction:eigen}{subespacios propios} suman $k$
(\cref{thm:b2:reduction:diagcrit}), si y solo si hay $k$
\omterm{def:b2:reduction:eigen}{valores propios} distintos, si y solo si $P$ tiene $k$ raíces
distintas (los \omterm{def:b2:reduction:eigen}{valores propios} son las raíces de $\chi_C = P$).

\textbf{4.} Cada $(\lambda_i^n)_n$ es solución de $(\mathcal R)$:
$\lambda_i^{n+k} = \lambda_i^n\,\lambda_i^k = \lambda_i^n\sum_m
a_m\lambda_i^m$. Libertad: una combinación nula $\sum_i
c_i\lambda_i^n = 0$ para $n = 0, \dots, k-1$ es un sistema de
Vandermonde (\cref{exo:b2:linalg:11}) en los $c_i$: todos los $c_i
= 0$. El espacio de soluciones tiene dimensión $k$ (pregunta 6,
cuya demostración es elemental e independiente): $k$ soluciones
libres forman una base, y las coordenadas son únicas.

\textbf{5.} $P = X^2 - X - 6 = (X - 3)(X + 2)$: solución general
$u_n = A\,3^n + B(-2)^n$. Condiciones iniciales: $A + B = 1$, $3A -
2B = 8$, luego $A = 2$, $B = -1$:
\[
u_n = 2\cdot 3^n - (-2)^n .
\]
(Comprobación: $u_2 = u_1 + 6u_0 = 14$ y $2\cdot9 - 4 = 14$.)

\textbf{6.} $P(S)\bigl((u_n)\bigr)$ es la sucesión $n \mapsto
u_{n+k} - a_{k-1}u_{n+k-1} - \dots - a_0u_n$: se anula si y solo si
se cumple $(\mathcal R)$, de modo que el conjunto de soluciones es
$\ker P(S)$, un subespacio. La aplicación $\ker P(S) \to \C^k$, $u
\mapsto (u_0, \dots, u_{k-1})$, es lineal, inyectiva (la
recurrencia determina $u_{k}, u_{k+1}, \dots$ a partir de los $k$
primeros valores, por inducción) y sobreyectiva (defínase $u_n$
recursivamente a partir de cualesquiera datos iniciales): la
dimensión es $k$.

\textbf{7.} La demostración del~\cref{thm:b2:reduction:kernels}
solo usa la identidad de Bézout en $\C[X]$ y el hecho de que los
polinomios en un endomorfismo fijo conmutan. Ninguna de las dos
cosas menciona la dimensión del espacio ambiente: el lema vale
literalmente para $S \in \mathcal{L}(\mathcal{S})$. Por tanto,
\[
\ker P(S) = \bigoplus_{i=1}^{r}
\ker\,(S - \lambda_i\,\mathrm{id})^{m_i}.
\]

\textbf{8.} Para $Q \in \C[X]$, la sucesión $(S -
\lambda)\bigl(Q(n)\lambda^n\bigr)_n$ tiene término $n$-ésimo
$Q(n{+}1)\lambda^{n+1} - \lambda Q(n)\lambda^n =
\lambda^{n+1}(\Delta Q)(n)$, con $\Delta Q = Q(X{+}1) - Q(X)$ de
grado $\deg Q - 1$ (los términos dominantes se cancelan).
Iterando, $(S - \lambda)^m\bigl(Q(n)\lambda^n\bigr) =
\bigl(\lambda^{n+m}(\Delta^m Q)(n)\bigr)_n$, y $\Delta^m Q = 0$
cuando $\deg Q \leq m - 1$: el conjunto del miembro derecho está
contenido en el núcleo. Es un subespacio de dimensión $m$: las
sucesiones $(n^j\lambda^n)_n$, $0 \leq j < m$, son libres, ya que
$\sum_j c_j n^j\lambda^n = 0$ para todo $n$ obliga (dividiendo por
$\lambda^n \neq 0$) a que el polinomio $\sum_j c_jX^j$ se anule en
todo $n \in \N$ y por tanto sea nulo. Recíprocamente, $\dim\ker(S
- \lambda)^m \leq m$: desarrollando $(S - \lambda)^m = \sum_j
\binom mj(-\lambda)^{m-j}S^j$, la ecuación $(S - \lambda)^m u = 0$
es una recurrencia lineal de orden $m$ (con coeficiente director
$1$), así que $u$ queda determinada por $u_0, \dots, u_{m-1}$ como
en la pregunta 6. La igualdad de dimensiones concluye.

\textbf{9.} Combinando las preguntas 7 y 8: toda solución se
descompone de manera única como suma de elementos de los $\ker(S -
\lambda_i)^{m_i}$, es decir, $u_n = \sum_i Q_i(n)\lambda_i^n$ con
$\deg Q_i \leq m_i - 1$; los $Q_i$ son únicos porque la
descomposición es directa y, dentro de cada sumando, los
coeficientes de $Q_i$ son las coordenadas en la base
$(n^j\lambda_i^n)_j$ (pregunta 8). Comprobación de sensatez sobre
las dimensiones: $\sum_i m_i = k$.

\textbf{10.} $P = X^2 - 4X + 4 = (X - 2)^2$: las soluciones son
$(a + bn)2^n$. Datos iniciales: $a = 1$ y $2(a + b) = 0$, luego $b
= -1$:
\[
u_n = (1 - n)\,2^n .
\]
Comprobación: $u_2 = 4u_1 - 4u_0 = -4$, y $(1 - 2)\cdot4 = -4$.

\textbf{11.} Escribamos $u_n = \lambda_1^n\bigl(c_1 +
\sum_{i\geq2} c_i(\lambda_i/\lambda_1)^n\bigr)$; cada cociente
tiene módulo $< 1$, así que el paréntesis tiende a $c_1 \neq 0$:
$u_n \sim c_1\lambda_1^n$. En particular $u_n \neq 0$ para $n$
grande, y
\[
\frac{u_{n+1}}{u_n} =
\lambda_1\,\frac{c_1 + o(1)}{c_1 + o(1)} \longrightarrow
\lambda_1 .
\]

\textbf{12.} Calculemos:
\[
q(a_{n+1}, b_{n+1}) = (a_n + 2b_n)^2 - 2(a_n + b_n)^2
= -a_n^2 + 2b_n^2 = -q(a_n, b_n).
\]
Con $q(a_0, b_0) = 1 - 2 = -1$: $a_n^2 - 2b_n^2 = (-1)^{n+1}$.
Estructuralmente: $q(a, b) = (a - \sqrt2\,b)(a + \sqrt2\,b)$, y la
aplicación lineal $M$ multiplica el factor $a + \sqrt2 b$ por $1 +
\sqrt2$ y el factor $a - \sqrt2 b$ por $1 - \sqrt2$ (calcúlese:
$a_{n+1} + \sqrt2 b_{n+1} = (1 + \sqrt2)(a_n + \sqrt2 b_n)$); el
producto queda multiplicado en cada paso por $(1 + \sqrt2)(1 -
\sqrt2) = -1 = \det M$.

\textbf{13.} Como $a_n^2 - 2b_n^2 = (a_n - \sqrt2 b_n)(a_n +
\sqrt2 b_n) = (-1)^{n+1}$,
\[
\Bigl|\frac{a_n}{b_n} - \sqrt2\Bigr|
= \frac{\abs{a_n^2 - 2b_n^2}}{b_n(a_n + \sqrt2 b_n)}
= \frac{1}{b_n(a_n + \sqrt2 b_n)} \leq \frac1{2b_n^2},
\]
usando $a_n \geq b_n \geq 1$ (por inducción, ambas crecen), de
donde $a_n + \sqrt2 b_n \geq (1 + \sqrt2)b_n \geq 2b_n$.
\omterm{def:b2:reduction:eigen}{Valores propios} de $M$: $\chi_M = X^2 - 2X - 1$, con raíces $1 \pm
\sqrt2$; como $(a_0, b_0)$ tiene componente no nula sobre el
\omterm{def:b2:reduction:eigen}{vector propio} dominante (todas sus entradas son positivas), $b_n
\sim c(1 + \sqrt2)^n$ con $c > 0$ (pregunta 11). Por tanto, el
error es $\asymp (1 + \sqrt2)^{-2n} = (3 + 2\sqrt2)^{-n}$:
decrecimiento geométrico de razón $1/(3 + 2\sqrt2) = 3 - 2\sqrt2
\approx 0.172$.

\textbf{14.} (a) De la pregunta 9: $\abs{u_n} \leq \sum_i
\abs{Q_i(n)}\abs{\lambda_i}^n \leq \bigl(\sum_i
\abs{Q_i(n)}\bigr)\rho^n$, y cada $\abs{Q_i(n)} \leq C_i n^{m_i -
1} \leq C_i n^{m-1}$ para $n \geq 1$: basta sumar las constantes.
(b) Sea $\rho' = \max_{i \geq 2}\abs{\lambda_i} <
\abs{\lambda_1}$, $d = \deg Q_1$ y $c \neq 0$ su coeficiente
director. Entonces $u_n = Q_1(n)\lambda_1^n + R_n$ con $\abs{R_n}
\leq Cn^{m-1}\rho'^n$, y
\[
\frac{R_n}{Q_1(n)\lambda_1^n} = O\Bigl(n^{m-1-d}
\bigl(\rho'/\abs{\lambda_1}\bigr)^n\Bigr) \longrightarrow 0
\]
(lo geométrico gana a lo polinómico). Así pues,
\[
u_n \sim Q_1(n)\,\lambda_1^n \sim c\,n^d\lambda_1^n,
\qquad
\frac{u_{n+1}}{u_n} \longrightarrow \lambda_1
\quad\text{(pues } Q_1(n{+}1)/Q_1(n) \to 1\text{)}.
\]
Comprobación sobre la pregunta 10: para $u_n = (1-n)2^n$ el
cociente es
\[
\frac{(-n)2^{n+1}}{(1-n)2^n} = 2\,\frac{-n}{1-n}
\longrightarrow 2 = \lambda_1 .
\]

\textbf{15.} Por inducción sobre $n$. Para $n = 1$, $A_{ij}$
cuenta los caminos de longitud $1$. Paso inductivo: un camino de
longitud $n + 1$ de $i$ a $j$ es un camino de longitud $n$ de $i$
a algún vértice $\ell$ seguido de una arista $\ell j$:
\[
\#\{\text{caminos}\} = \sum_{\ell} (A^n)_{i\ell}A_{\ell j} =
(A^{n+1})_{ij}.
\]

\textbf{16.} $J = 3\Pi$, donde $\Pi = J/3$ es la proyección sobre
$\operatorname{Vect}(1,1,1)$ paralelamente al plano $x + y + z =
0$ ($\Pi^2 = \Pi$ porque $J^2 = 3J$). Entonces $A = J - I = 2\Pi -
(I - \Pi)$ y, al ser $\Pi$ e $I - \Pi$ proyecciones
complementarias,
\[
A^n = 2^n\,\Pi + (-1)^n (I - \Pi),
\qquad\text{es decir,}\qquad
(A^n)_{ij} = \frac{2^n}3 + (-1)^n\Bigl(\delta_{ij} -
\frac13\Bigr),
\]
lo que da las dos fórmulas del enunciado. Para $n = 2$: en la
diagonal, $(4 + 2)/3 = 2$ (los caminos $i \to \ell \to i$ por cada
uno de los dos vecinos $\ell$); fuera de la diagonal, $(4 - 1)/3 =
1$ (el único camino $i \to \ell \to j$ pasando por el tercer
vértice).

\textbf{17.} Sean $w_n^{(0)}$ y $w_n^{(1)}$ el número de palabras
admisibles de longitud $n$ terminadas en $0$ y en $1$,
respectivamente. Al añadir una letra: un $0$ puede seguir a
cualquier cosa, y un $1$ solo a un $0$:
\[
\begin{pmatrix} w_{n+1}^{(0)}\\ w_{n+1}^{(1)}\end{pmatrix}
= \begin{pmatrix} 1 & 1\\ 1 & 0\end{pmatrix}
\begin{pmatrix} w_n^{(0)}\\ w_n^{(1)}\end{pmatrix}.
\]
Sumando, $w_{n+2} = w_{n+1} + w_n$ (o bien: condiciónese sobre la
primera letra). Con $w_1 = 2$ y $w_2 = 3$ resulta $w_n = F_{n+2}$
por inducción ($F_3 = 2$, $F_4 = 3$, misma recurrencia).
Crecimiento: las raíces de $X^2 - X - 1$ son $\varphi >
\abs\psi$ (\cref{exo:b2:reduction:5}) y la componente en $\varphi$
es no nula (los $w_n$ son positivos y $\psi^n \to 0$), así que la
pregunta 11 da $w_{n+1}/w_n \to \varphi = \frac{1 + \sqrt5}2$.

\textbf{18.} $A = \begin{psmallmatrix} 0&1&0\\ 1&0&1\\
0&1&0\end{psmallmatrix}$. Comprobemos:
\[
A(1, \pm\sqrt2, 1)^{\mathsf T}
= (\pm\sqrt2, 2, \pm\sqrt2)^{\mathsf T}
= \pm\sqrt2\,(1, \pm\sqrt2, 1)^{\mathsf T},
\qquad
A(1, 0, -1)^{\mathsf T} = 0 :
\]
\omterm{def:b2:reduction:eigen}{valores propios} $\sqrt2, -\sqrt2, 0$ ($= 2\cos\frac\pi4,
2\cos\frac{3\pi}4, 2\cos\frac\pi2$). Descompóngase $e_1$ en la
base de \omterm{def:b2:reduction:eigen}{vectores propios} y léase la tercera coordenada, o úsese la
simetría: con $v_\pm = (1, \pm\sqrt2, 1)$ y $v_0 = (1, 0, -1)$ se
comprueba que $e_1 = \frac14 v_+ + \frac14 v_- + \frac12 v_0$,
luego para $n \geq 1$
\[
(A^n)_{13} = \Bigl(\tfrac14(\sqrt2)^n v_+ +
\tfrac14(-\sqrt2)^n v_- + 0\Bigr)_{\!3}
= \frac{(\sqrt2)^n + (-\sqrt2)^n}{4},
\]
que vale cero para $n$ impar (el grafo es bipartito: los extremos
están a distancia par) y $2\cdot 2^{n/2}/4 = 2^{n/2 - 1}$ para $n$
par. Para $n = 4$: $2^{1} = 2$, que corresponde a los dos caminos
$1\,2\,1\,2\,3$ y $1\,2\,3\,2\,3$.

\textbf{19.} Los caminos cerrados de longitud $n$ que parten de
$i$ son $(A^n)_{ii}$; sumando sobre $i$ se obtiene
$\operatorname{tr}(A^n)$. Trigonalizando $A$ (sobre $\C$), $A^n$
es triangular con diagonal $\lambda_i^n$: $\operatorname{tr}(A^n)
= \sum_i\lambda_i^n$. En el triángulo: $\operatorname{tr}(A^n) =
3\,\frac{2^n + 2(-1)^n}3 = 2^n + 2(-1)^n = 2^n + (-1)^n +
(-1)^n$, correspondiente al \omterm{def:b2:reduction:eigen}{espectro} $\{2, -1, -1\}$, en
coherencia con la pregunta 16.

\textbf{20.} Las columnas de $W^{\mathsf T}$: $W^{\mathsf T}e_i =
e_{i-1}$ para $i \geq 1$ y $W^{\mathsf T}e_0 = e_{k-1}$;
reetiquetando en el orden $e_0, e_1, \dots$, esto es exactamente
la matriz compañera de $X^k - 1$ ($a_0 = 1$ y los demás $a_m =
0$). Por la pregunta 2, $\chi_{W} = \chi_{W^{\mathsf T}} = X^k - 1
= \mu_{W}$. Las raíces $\omega^j$ ($j = 0, \dots, k-1$) son las
$k$ raíces $k$-ésimas distintas de la unidad: $W$ es
\omterm{def:b2:reduction:diag}{diagonalizable} (pregunta 3, o \cref{exo:b2:reduction:8}: $W^k =
I$). \omterm{def:b2:reduction:eigen}{Vectores propios}: $Wf_j = \sum_m \omega^{-jm}e_{m+1} =
\sum_{m'}\omega^{-j(m'-1)}e_{m'} = \omega^j f_j$.

\textbf{21.} Las circulantes son polinomios en $W$, y los
polinomios en una matriz fija conmutan entre sí. Cada $f_j$ es
\omterm{def:b2:reduction:eigen}{vector propio} de toda potencia: $W^m f_j = \omega^{jm}f_j$, luego
\[
Cf_j = \sum_m c_m\omega^{jm} f_j = \widehat c(\omega^j)\,f_j :
\]
la base $(f_0, \dots, f_{k-1})$ (libre: Vandermonde en los
$\omega^{-j}$ distintos, \cref{exo:b2:linalg:11}) diagonaliza
todas las circulantes a la vez, con los \omterm{def:b2:reduction:eigen}{valores propios}
indicados.

\textbf{22.} El \omterm{def:b2:linalg:det}{determinante} es el producto de los \omterm{def:b2:reduction:eigen}{valores propios}
(diagonalícese): $\det C = \prod_{j}\widehat c(\omega^j)$. Para $k
= 3$, con $c_0 = a$, $c_1 = b$, $c_2 = c$ y $\omega = j =
\eu^{2\iu\pi/3}$:
\[
\det C = (a + b + c)(a + bj + cj^2)(a + bj^2 + cj^4),
\]
y $j^4 = j$: exactamente la factorización del~\cref{exo:b2:linalg:8}.

\textbf{23.} $M = \frac12(W + W^{-1})$ es una circulante ($W^{-1}
= W^{k-1}$), con \omterm{def:b2:reduction:eigen}{valores propios} $\frac12(\omega^j + \omega^{-j})
= \cos\frac{2\pi j}k$ sobre la misma base $f_j$. Coordenadas:
escribamos $x^{(0)} = \sum_j \alpha_j f_j$. Las coordenadas de
$f_j$ suman $\sum_m \omega^{-jm}$, que vale $k$ para $j = 0$ y $0$
en los demás casos (suma geométrica de razón $\omega^{-j} \neq
1$). Sumando las coordenadas de $x^{(0)}$: $\sum_m x^{(0)}_m =
\alpha_0\,k$, luego $\alpha_0 = \frac1k\sum_m x^{(0)}_m$, la
media.

\textbf{24.} $x^{(n)} = M^nx^{(0)} = \sum_j
\alpha_j\cos^n\bigl(\tfrac{2\pi j}k\bigr)f_j$. Para $k$ impar,
$\abs{\cos(2\pi j/k)} < 1$ para todo $j \neq 0$ (el ángulo nunca
vale $0$ ni $\pi$), luego todos los términos salvo el de $j = 0$
tienden a $0$: $x^{(n)} \to \alpha_0 f_0$, el vector constante
igual a la media; promediar en un anillo impar iguala. Para $k =
4$ los \omterm{def:b2:reduction:eigen}{valores propios} son $1, 0, -1, 0$: el término $j = 2$,
$\alpha_2(-1)^nf_2$ con $f_2 = (1, -1, 1, -1)^{\mathsf T}$, oscila
indefinidamente. La obstrucción es la media alternada:
multiplicando las coordenadas de $x^{(0)}$ por $(-1)^m$ y sumando,
el mismo cálculo de suma geométrica da $\sum_m (-1)^mx^{(0)}_m =
4\alpha_2$; el proceso converge si y solo si $x^{(0)}_0 -
x^{(0)}_1 + x^{(0)}_2 - x^{(0)}_3 = 0$, y entonces converge a la
media.

\textbf{25.} La matriz compañera convierte una recurrencia escalar
de orden $k$ en una recurrencia vectorial de primer orden, de modo
que las fórmulas cerradas pasan a ser enunciados sobre $C^n$,
terreno propio de la reducción (preguntas 1--5). El lema de
descomposición en núcleos es álgebra polinómica pura (Bézout más
conmutación), así que parte $\ker P(S)$ aunque $\mathcal{S}$ tenga
dimensión infinita (preguntas 7--9). Los \omterm{def:b2:reduction:eigen}{valores propios}
dominantes gobiernan el crecimiento porque cualquier otra
contribución es geométricamente despreciable tras normalizar, y
esa es también la razón de que el error de Pell decrezca como el
cuadrado de la raíz dominante (preguntas 11--14). Las potencias de
la matriz de adyacencia cuentan caminos porque la multiplicación
de matrices suma sobre los vértices intermedios, de modo que los
\omterm{def:b2:reduction:eigen}{espectros} cuentan caminos cerrados (preguntas 15--19). Las
matrices que conmutan comparten una base de \omterm{def:b2:reduction:eigen}{vectores propios}, y
entonces una única base de Fourier diagonaliza de un golpe toda el
álgebra de las circulantes (preguntas 20--24). Cumbres: el teorema
fundamental de las recurrencias lineales (pregunta 9); y, para las
matrices no negativas, la razón de que raíces dominantes como
$\varphi$ o $1 + \sqrt2$ sean automáticamente reales, positivas y
simples es el teorema de Perron--Frobenius, demostrado en el
volumen del tercer año.
\end{solution}
