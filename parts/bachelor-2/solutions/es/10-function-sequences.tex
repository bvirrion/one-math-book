\section*{Capítulo \ref{ch:b2:funcseq} --- Sucesiones y series de
funciones}

\begin{solution}{exo:b2:funcseq:1}
$f_n(x) = \frac{x}{1 + nx}$: límite \omterm{def:b2:funcseq:def}{puntual} $0$ sobre
$\intcc{0}{1}$. \omterm{def:b2:funcseq:def}{Uniformemente}: $f_n$ crece sobre $\intcc{0}{1}$
(derivada $\frac{1}{(1+nx)^2} > 0$), luego $\norm{f_n}_\infty =
f_n(1) = \frac{1}{1+n} \to 0$: \emph{uniforme} sobre $\intcc{0}{1}$.

$g_n(x) = nx\,\eu^{-nx^2}$: límite \omterm{def:b2:funcseq:def}{puntual} $0$ (la exponencial
gana). Supremo: $g_n' = n\eu^{-nx^2}(1 - 2nx^2)$ se anula en $x_n =
\frac{1}{\sqrt{2n}}$, donde $g_n(x_n) = \sqrt{\frac n2}\,\eu^{-1/2}
\to \infty$: no es uniforme sobre $\intcc{0}{1}$; pero sí lo es
sobre $\intco{\delta}{1}$, ya que allí $g_n(x) \leq
n\,\eu^{-n\delta^2} \to 0$.

$h_n(x) = x^n(1 - x^n)$: límite \omterm{def:b2:funcseq:def}{puntual} $0$ sobre $\intcc{0}{1}$
(por ambos factores; y en $x = 1$, $h_n = 0$). Supremo: con $u = x^n
\in \intcc{0}{1}$, $u(1-u) \leq \frac14$ se alcanza en $u =
\frac12$, es decir, en $x = 2^{-1/n} \in \intoo{0}{1}$:
$\norm{h_n}_\infty = \frac14 \not\to 0$, luego no es uniforme sobre
$\intcc{0}{1}$; sí lo es sobre $\intcc{0}{a}$ ($\sup \leq a^n \to
0$).
\end{solution}

\begin{solution}{exo:b2:funcseq:2}
$\int_0^1 nx\,\eu^{-nx^2}\dd x = \bigl[-\tfrac12
\eu^{-nx^2}\bigr]_0^1 = \frac{1 - \eu^{-n}}{2} \to \frac12$,
mientras que $\int_0^1 \lim g_n = 0$. No hay contradicción: el~\cref{thm:b2:funcseq:integration} exige convergencia
\emph{uniforme} sobre el segmento, cosa que aquí falla (la joroba de
altura $\sim\sqrt n$ se desliza hacia $0$).
\end{solution}

\begin{solution}{exo:b2:funcseq:3}
\omterm{def:b2:funcseq:series}{Convergencia normal} sobre $\intcc{-1}{1}$: $\norm{x^n/n^2}_\infty =
\frac{1}{n^2}$, \omterm{def:b2:series:summable}{sumable}, luego $S$ es allí \omterm{def:b2:metric:continuity}{continua}
(\cref{thm:b2:funcseq:seriestransfer}).

Derivada: la serie derivada $\sum \frac{x^{n-1}}{n}$ converge
\omterm{def:b2:funcseq:series}{normalmente} sobre cada $\intcc{-a}{a}$ con $a < 1$ ($\sup =
\frac{a^{n-1}}{n}$): $S$ es de clase $C^1$ sobre $\intoo{-1}{1}$
con
\[
S'(x) = \sum_{n\geq1} \frac{x^{n-1}}{n} = \frac1x \sum_{n\geq1}
\frac{x^n}{n} = -\frac{\ln(1 - x)}{x}
\qquad (0 < \abs x < 1),
\]
siendo la última identidad la serie logarítmica del primer año
(vuelta a deducir honestamente en el~\cref{ch:b2:powerseries}).
\end{solution}

\begin{solution}{exo:b2:funcseq:4}
Para $x > 0$ fijo, la serie es alternada con $\frac{1}{n + x}
\downarrow 0$: hay \omterm{def:b2:funcseq:def}{convergencia puntual}, y la cota del resto
$\abs{R_N(x)} \leq \frac{1}{N + 1 + x} \leq \frac{1}{N+1}$ es
\emph{uniforme} sobre $\intco{\delta}{\infty}$ (de hecho, sobre
$\intoo{0}{\infty}$): \omterm{def:b2:funcseq:def}{convergencia uniforme}. (No normal:
$\norm{\frac{(-1)^n}{n+x}}_\infty = \frac{1}{n + \delta}$,
divergente.) La \omterm{def:b2:metric:continuity}{continuidad} se sigue del~\cref{thm:b2:funcseq:seriestransfer}.

Ecuación funcional: reindexemos $F(x + 1)$ con $m = n + 1$:
\[
F(x+1) = \sum_{n\geq0} \frac{(-1)^n}{n + 1 + x}
= \sum_{m\geq1}\frac{(-1)^{m-1}}{m+x} ,
\]
de modo que, aislando el término $m = 0$ de $F(x)$,
\[
F(x) + F(x+1)
= \frac{1}{x} + \sum_{m\geq1}
\frac{(-1)^m + (-1)^{m-1}}{m+x} = \frac1x .
\]
\end{solution}

\begin{solution}{exo:b2:funcseq:5}
Pongamos $g_n = f_n - f \geq 0$ (decreciente en $n$ por hipótesis, y
con límite \omterm{def:b2:funcseq:def}{puntual} $0$); cada $g_n$ es \omterm{def:b2:metric:continuity}{continua}. Fijemos
$\varepsilon > 0$ y sea $U_n = \{x : g_n(x) < \varepsilon\}$:
\omterm{def:b2:metric:topology}{abierto} (imagen recíproca de un \omterm{def:b2:metric:topology}{abierto}), creciente ($g_{n+1} \leq
g_n$) y recubridor de $K$ (\omterm{def:b2:funcseq:def}{convergencia puntual}). Por
Borel--Lebesgue (\cref{thm:b2:metric:borellebesgue}), un número
finito de ellos $U_{n_1} \subseteq \dots \subseteq U_{n_k}$ recubre
$K$: luego $K = U_{n_k}$, es decir, $\norm{g_{n_k}}_\infty \leq
\varepsilon$, y por monotonía $\norm{g_n}_\infty \leq \varepsilon$
para todo $n \geq n_k$: \omterm{def:b2:funcseq:def}{convergencia uniforme}. (La monotonía es
esencial: las jorobas deslizantes del~\cref{exo:b2:funcseq:2}
convergen \omterm{def:b2:funcseq:def}{puntualmente} sobre un \omterm{def:b2:metric:compact}{compacto} sin uniformidad.)
\end{solution}

\begin{solution}{exo:b2:funcseq:6}
Sustituyamos $u = nx$:
\[
\int_0^1 \frac{n f(x)}{1 + n^2x^2}\dd x
= \int_0^n \frac{f(u/n)}{1 + u^2}\,\dd u .
\]
Los integrandos $h_n(u) = \frac{f(u/n)}{1+u^2}\mathbf{1}_{u \leq n}$
convergen \omterm{def:b2:funcseq:def}{puntualmente} a $\frac{f(0)}{1+u^2}$ (\omterm{def:b2:metric:continuity}{continuidad} de $f$
en $0$) y están dominados por $\frac{\norm f_\infty}{1 + u^2}$,
integrable sobre $\intco{0}{\infty}$: la convergencia dominada
(\cref{thm:b2:integration:dominated}) da el límite
\[
\int_0^\infty \frac{f(0)}{1 + u^2}\dd u = \frac{\pi}{2} f(0) .
\]
(Los núcleos se concentran en $0$: una identidad aproximada.)
\end{solution}

\begin{solution}{exo:b2:funcseq:7}
Para $f(x) = x^2$, usemos la segunda familia de identidades
binomiales de la demostración: $\sum_k k^2 p_k = n(n-1)x^2 + nx$.
Por tanto
\[
B_n(f)(x) = \sum_k \frac{k^2}{n^2}\,p_k
= \frac{n(n-1)x^2 + nx}{n^2}
= x^2 + \frac{x(1 - x)}{n} :
\]
$B_n(f) - f = \frac{x(1-x)}{n}$, de \omterm{def:b2:nvs:norm}{norma} del supremo $\frac{1}{4n}
= O\bigl(\frac1n\bigr)$, como se predijo.
\end{solution}

\begin{solution}{exo:b2:funcseq:8}
Para $\varepsilon = 1$ existe $N$ con $\norm{P_n - P_m}_{\infty,
\R} \leq 1$ para $m, n \geq N$. Un polinomio acotado sobre $\R$ es
constante (uno no constante tiende a $\pm\infty$): $P_n - P_m =
c_{n,m}$, constantes. Así, para $n \geq N$: $P_n = P_N + c_n$ con
$c_n = P_n(0) - P_N(0)$ convergente (\omterm{def:b2:funcseq:def}{convergencia puntual} en $0$).
Por tanto $f = \lim P_n = P_N + \lim c_n$: un polinomio.
\end{solution}

\begin{solution}{exo:b2:funcseq:9}
(a) $\norm{(3/4)^n\varphi(4^n\cdot)}_\infty = \frac12 (3/4)^n$:
\omterm{def:b2:funcseq:series}{convergencia normal}, luego $W$ es \omterm{def:b2:metric:continuity}{continua}
(\cref{thm:b2:funcseq:seriestransfer}).

(b) Fijemos $x$ y $m$; elijamos el signo de $h_m = \pm\frac12
4^{-m}$ de modo que el segmento $\intcc{4^mx}{4^m(x + h_m)}$ (de
longitud $\frac12$) no contenga ningún semientero, con lo que
$\varphi$ resulta afín de pendiente $\pm1$ sobre él (es posible: un
intervalo de longitud $\frac12$ corta a lo sumo un punto
semientero; tómese el lado que lo evite).

Para $n > m$: $4^n h_m = \pm\frac12 4^{n-m}$ es un entero y $\varphi$
es $1$-periódica, así que el término $n$-ésimo de la diferencia se
anula.

Para $n = m$: $\abs{\varphi(4^m x + 4^m h_m) - \varphi(4^m x)} =
\abs{4^m h_m} = \frac12$ ($\varphi$ es afín de pendiente $\pm 1$
sobre el segmento), de modo que el término aporta exactamente
$(3/4)^m \cdot \frac{1/2}{\abs{h_m}} = (3/4)^m\,4^m = 3^m$ al
cociente.

Para $n < m$: como $\varphi$ es $1$-lipschitziana,
$\bigl|(3/4)^n\bigl(\varphi(4^nx + 4^nh_m) -
\varphi(4^nx)\bigr)\bigr| \leq (3/4)^n 4^n\abs{h_m} =
3^n\abs{h_m}$: cada uno aporta a lo sumo $3^n$ al cociente.

Por tanto,
\[
\Bigl|\frac{W(x + h_m) - W(x)}{h_m}\Bigr|
\geq 3^m - \sum_{n=0}^{m-1} 3^n
= 3^m - \frac{3^m - 1}{2} = \frac{3^m + 1}{2}
\longrightarrow \infty .
\]
Si $W$ fuera derivable en $x$, todo cociente incremental a lo largo
de $h_m \to 0$ convergería a $W'(x)$: contradicción. $W$ es
\omterm{def:b2:metric:continuity}{continua} en todas partes y no derivable en ninguna.
\end{solution}

\begin{solution}{exo:b2:funcseq:10}
\omterm{def:b2:funcseq:def}{Puntualmente}: para $x \in \intco{0}{1}$ la serie es geométrica de
razón $-x$,
\[
\sum_{n\geq0}(-1)^n x^n(1-x) = \frac{1-x}{1+x},
\]
y en $x = 1$ todos los términos se anulan: suma $0 =
\frac{1-1}{2}$, coherente; la suma es \omterm{def:b2:metric:continuity}{continua} sobre
$\intcc{0}{1}$. Uniformidad: el resto es una cola geométrica,
\[
\abs{R_N(x)} = \frac{x^{N+1}(1-x)}{1+x} \leq x^{N+1}(1 - x)
\leq \max_{\intcc01} t^{N+1}(1-t)
= \frac{1}{N+2}\Bigl(\frac{N+1}{N+2}\Bigr)^{\!N+1}
\leq \frac{1}{N+2} \to 0 ,
\]
uniformemente en $x$. No normal: $\norm{u_n}_\infty = \max x^n(1-x)
= \frac{1}{n+1}\bigl(\frac{n}{n+1}\bigr)^n \sim \frac{1}{\eu\,n}$, y
$\sum \frac1{\eu n}$ diverge. Contraste: $\sum x^n(1-x)$ tiene sumas
parciales $1 - x^{N+1}$, que convergen \omterm{def:b2:funcseq:def}{puntualmente} a la función
\emph{discontinua} $\mathbf 1_{\intco01}$: por el~\cref{thm:b2:funcseq:continuity}, esa convergencia no puede ser
uniforme sobre $\intcc{0}{1}$.
\end{solution}

\begin{solution}{exo:b2:funcseq:11}
El límite $f$ es \omterm{def:b2:metric:continuity}{continuo} (\cref{thm:b2:funcseq:continuity}).
Entonces
\[
\abs{f_n(x_n) - f(x)}
\leq \abs{f_n(x_n) - f(x_n)} + \abs{f(x_n) - f(x)}
\leq \norm{f_n - f}_\infty + \abs{f(x_n) - f(x)} ,
\]
y ambos términos tienden a $0$ (\omterm{def:b2:funcseq:def}{convergencia uniforme};
\omterm{def:b2:metric:continuity}{continuidad} de $f$ en $x$). Contraejemplo con mera
\omterm{def:b2:funcseq:def}{convergencia puntual}: $g_n(x) = nx\,\eu^{-nx^2} \to 0$
\omterm{def:b2:funcseq:def}{puntualmente} sobre $\intcc{0}{1}$, con las $g_n$ y el límite
\omterm{def:b2:metric:continuity}{continuos}, y sin embargo en $x_n = \frac{1}{\sqrt{2n}} \to 0$:
\[
g_n(x_n) = \sqrt{\frac n2}\,\eu^{-1/2} \longrightarrow +\infty
\neq 0 = f(0) .
\]
\end{solution}

\begin{solution}{exo:b2:funcseq:12}
\begin{enumerate}
  \item Inducción. El caso $n = 1$ es la definición. Supuesta la
        fórmula para $n$, pongamos $g(x) = \int_0^x
        \frac{(x-t)^n}{n!} f(t)\dd t$. Para un integrando \omterm{def:b2:metric:continuity}{continuo}
        en $(x,t)$ y de clase $C^1$ en $x$, la
        \omterm{thm:b2:integration:continuity}{integral con parámetro} de límite variable se deriva como
        \[
        g'(x) = \frac{(x-x)^n}{n!}f(x)
        + \int_0^x \frac{(x-t)^{n-1}}{(n-1)!}f(t)\dd t
        = T^nf(x)
        \]
        (sepárese $g(x+h) - g(x)$ en la franja $\int_x^{x+h}$, que
        es $O(h\cdot\sup)$ y cuyo integrando se anula en $t = x$
        como $h^n$, y la integral fija del incremento en $x$,
        tratada con la desigualdad del valor medio y la
        \omterm{def:b2:metric:continuity}{continuidad}). Además $(T^{n+1}f)' = T^nf$ (teorema
        fundamental del cálculo) y $g(0) = T^{n+1}f(0) = 0$: dos
        primitivas de $T^nf$ que se anulan en $0$ coinciden, luego
        $T^{n+1}f = g$. La cota:
        \[
        \abs{T^nf(x)} \leq \norm f_\infty
        \int_0^x \frac{(x-t)^{n-1}}{(n-1)!}\dd t
        = \norm f_\infty\,\frac{x^n}{n!}
        \leq \frac{\norm f_\infty}{n!} .
        \]
  \item $\sum_n \norm{T^nf}_\infty \leq \eu\,\norm f_\infty$:
        \omterm{def:b2:funcseq:series}{convergencia normal} y, por tanto, uniforme; $S$ es
        \omterm{def:b2:metric:continuity}{continua}. Las sumas parciales cumplen $S_N = f + T
        S_{N-1}$, y $T$ es $1$-lipschitziana para
        $\norm\cdot_\infty$ ($\abs{Tg(x)} \leq x\norm g_\infty$):
        haciendo $N \to \infty$ en ambos miembros resulta $S = f +
        TS$.
  \item Pongamos $V(x) = f(x) + \eu^x\int_0^x \eu^{-t}f(t)\dd t$.
        Entonces $V - f$ es de clase $C^1$ con $(V-f)'(x) =
        \eu^x\int_0^x\eu^{-t}f + f(x) = V(x)$, y $(TV)' = V$ con $(V
        - f)(0) = TV(0) = 0$: luego $V - f = TV$, es decir, $V$
        resuelve la ecuación. Unicidad: si $S_1, S_2$ son soluciones
        \omterm{def:b2:metric:continuity}{continuas}, $D = S_1 - S_2$ cumple $D = TD$, de donde $D =
        T^nD$ para todo $n$ y $\norm D_\infty \leq \frac{\norm
        D_\infty}{n!} \to 0$: $D = 0$. Por tanto
        \[
        \sum_{n\geq0} T^nf(x) = f(x) +
        \int_0^x \eu^{x-t}f(t)\,\dd t .
        \]
        (La serie $\sum T^n$ es una serie geométrica de operadores:
        un primer sabor de la resolvente $(\mathrm{Id} - T)^{-1}$,
        desarrollada en el volumen del tercer año.)
\end{enumerate}
\end{solution}

\begin{solution}{pb:b2:funcseq:1}
\textbf{1.} La linealidad es clara a partir de la fórmula.
Positividad: los pesos $p_k(x) \geq 0$, luego $f \geq 0$ obliga a
$B_nf \geq 0$; la monotonía se sigue aplicándolo a $g - f$. Cota:
de $\pm f \leq \norm f_\infty$ resulta $\pm B_nf \leq \norm
f_\infty B_ne_0 = \norm f_\infty$. Extremos: $p_k(0) = \mathbf
1_{k=0}$ y $p_k(1) = \mathbf 1_{k=n}$, de modo que $B_nf(0) =
f(0)$ y $B_nf(1) = f(1)$.

\textbf{2.} Derivemos $(x+y)^n = \sum_k\binom nk x^ky^{n-k}$ en
$x$, multipliquemos por $x$ y pongamos $y = 1 - x$:
\[
nx = \sum_k k\,p_k(x) ;
\]
dos veces, multiplicando por $x^2$: $n(n-1)x^2 = \sum_k
k(k-1)p_k(x)$. Por tanto $B_ne_0 = 1$ (teorema del binomio),
$B_ne_1(x) = \frac{nx}{n} = x$ y
\[
B_ne_2(x) = \frac{\sum_k k^2p_k}{n^2}
= \frac{n(n-1)x^2 + nx}{n^2}
= x^2 + \frac{x(1-x)}{n} .
\]

\textbf{3.} Desarrollando:
\[
\sum_k\Bigl(\frac kn - x\Bigr)^{\!2} p_k
= B_ne_2(x) - 2x\,B_ne_1(x) + x^2
= \frac{x(1-x)}{n} .
\]
Cauchy--Schwarz con la separación $\abs{k/n - x}\sqrt{p_k} \cdot
\sqrt{p_k}$:
\[
\sum_k\Bigl|\frac kn - x\Bigr| p_k
\leq \Bigl(\sum_k\Bigl(\frac kn - x\Bigr)^2
p_k\Bigr)^{\!1/2}
= \sqrt{\frac{x(1-x)}{n}} \leq \frac{1}{2\sqrt n},
\]
usando $x(1-x) \leq \frac14$.

\textbf{4.} Los pesos $p_k(x)$ son no negativos, suman $1$ y tienen
baricentro $\sum_k \frac kn p_k(x) = x$ (pregunta 2). La
desigualdad de Jensen finita para la $f$ convexa (inducción desde
la definición con dos puntos, volumen del primer año) da
\[
f(x) = f\Bigl(\sum_k \frac kn\,p_k\Bigr)
\leq \sum_k f\Bigl(\frac kn\Bigr)p_k = B_nf(x) .
\]

\textbf{5.} Sobre $\{k : \abs{k/n - x} > \delta\}$ se tiene
$\bigl(\frac{k/n - x}{\delta}\bigr)^2 > 1$, luego
\[
\sum_{\abs{k/n-x}>\delta} p_k
\leq \frac{1}{\delta^2}\sum_k\Bigl(\frac kn -
x\Bigr)^2p_k
= \frac{x(1-x)}{n\delta^2} \leq \frac{1}{4n\delta^2} .
\]
Lectura: $p_k(x)$ es la ley de la frecuencia muestral $S_n/n$ de
$n$ lanzamientos de una moneda de sesgo $x$; su media es $x$, su
varianza $\frac{x(1-x)}n \to 0$, y la fórmula anterior es la
desigualdad de Chebyshev: la masa se concentra en $x$, de modo que
promediar $f$ contra ella reproduce $f(x)$ en el límite (el~\cref{ch:b2:genfun} oficializa el vocabulario).

\textbf{6.} $\omega \leq 2M < \infty$; la monotonía es clara
(supremo sobre un conjunto mayor). Heine: una $f$ \omterm{def:b2:metric:continuity}{continua} sobre un
\omterm{def:b2:metric:compact}{compacto} es uniformemente continua, que es exactamente decir
$\omega(\delta) \to 0$ cuando $\delta \to 0^+$. Subaditividad: si
$\abs{s - t} \leq \delta_1 + \delta_2$, el punto $u$ del segmento
$\intcc st$ a distancia $\min(\delta_1, \abs{s-t})$ de $s$ cumple
$\abs{s-u} \leq \delta_1$ y $\abs{u-t} \leq \delta_2$, y
$\abs{f(s)-f(t)} \leq \abs{f(s)-f(u)} + \abs{f(u)-f(t)}$. Iterando,
$\omega(p\delta) \leq p\,\omega(\delta)$ para $p \in \N^*$; y para
$\lambda > 0$, con $p = \lceil\lambda\rceil \leq 1 + \lambda$:
$\omega(\lambda\delta) \leq \omega(p\delta) \leq p\,\omega(\delta)
\leq (1+\lambda)\omega(\delta)$.

\textbf{7.} Como $\sum p_k = 1$:
\[
\abs{B_nf(x) - f(x)}
= \Bigl|\sum_k\bigl(f(k/n) - f(x)\bigr)p_k\Bigr|
\leq \sum_k\omega\bigl(\abs{k/n - x}\bigr)p_k .
\]
Para cada $k$, la pregunta 6 con $\lambda = \abs{k/n - x}/\delta$
da $\omega(\abs{k/n-x}) \leq \bigl(1 +
\frac{\abs{k/n-x}}\delta\bigr)\omega(\delta)$; sumando contra los
$p_k$ se obtiene la estimación maestra.

\textbf{8.} Insertemos la cota de la pregunta 3:
\[
\abs{B_nf(x) - f(x)} \leq \Bigl(1 +
\frac{1}{2\delta\sqrt n}\Bigr)\omega(\delta),
\]
uniformemente en $x$; con $\delta = n^{-1/2}$ el paréntesis vale
$\frac32$: $\norm{B_nf - f}_\infty \leq \frac32\omega(n^{-1/2}) \to
0$ por la pregunta 6 (Heine). Esto vuelve a demostrar el~\cref{thm:b2:funcseq:weierstrass} con una velocidad.

\textbf{9.} Ser $L$-lipschitziana significa $\omega(\delta) \leq
L\delta$: velocidad $\frac{3L}{2\sqrt n}$. Ser $\alpha$-hölderiana
significa $\omega(\delta) \leq C\delta^\alpha$: velocidad
$\frac{3C}{2} n^{-\alpha/2}$.

\textbf{10.} Usando $k\binom{2m}k = 2m\binom{2m-1}{k-1}$:
\[
\sum_{k=m+1}^{2m}k\binom{2m}k
= 2m\sum_{j=m}^{2m-1}\binom{2m-1}{j}
= 2m\cdot 2^{2m-2},
\]
porque $j \mapsto 2m-1-j$ biyecta $\{m,\dots,2m-1\}$ sobre
$\{0,\dots,m-1\}$, de modo que la suma es la mitad de $2^{2m-1}$.
Además, $\sum_{k=m+1}^{2m}\binom{2m}k = \frac{2^{2m} -
\binom{2m}m}{2}$ (misma simetría). Por tanto,
\[
\sum_{k=m+1}^{2m}(k-m)\binom{2m}k
= m\,2^{2m-1} - m\,\frac{2^{2m} - \binom{2m}m}{2}
= \frac m2\binom{2m}m .
\]
La sustitución $k \mapsto 2m-k$ envía los términos con $k < m$
sobre los de $k > m$ (mismos binomiales, mismo $\abs{k-m}$): la
suma en valor absoluto es el doble de la suma de un lado, es decir,
$m\binom{2m}m$.

\textbf{11.} En $x = \frac12$, $p_k(\tfrac12) = \binom{2m}k2^{-2m}$
y $f(\tfrac12) = 0$:
\[
B_{2m}f\Bigl(\frac12\Bigr)
= \sum_k\Bigl|\frac{k}{2m} - \frac12\Bigr|
\binom{2m}k 2^{-2m}
= \frac{2^{-2m}}{2m}\,m\binom{2m}m
= \frac{\binom{2m}m}{2\cdot4^m}
\sim \frac{1}{2\sqrt{\pi m}}
\]
por el~\cref{ex:b2:comparison:centralbinomial}. Como aquí
$\omega_f(\delta) = \delta$ (la función es $1$-lipschitziana y la
cota se alcanza), la pregunta 8 predice a lo sumo
$\frac32(2m)^{-1/2}$: el error verdadero $\frac{1}{2\sqrt{\pi m}}$
tiene exactamente el orden $n^{-1/2}$; la velocidad es óptima salvo
la constante.

\textbf{12.} De $f \leq g$ resulta $g - f \geq 0$, luego $L(g-f)
\geq 0$, es decir, $Lf \leq Lg$. Y de $-\abs f \leq f \leq \abs f$:
$-L\abs f \leq Lf \leq L\abs f$, es decir, $\abs{Lf} \leq L\abs f$.

\textbf{13.} Por Heine, elíjase $\delta$ con $\abs{f(s)-f(x)} \leq
\varepsilon$ siempre que $\abs{s-x} \leq \delta$. Si $\abs{s - x} >
\delta$, entonces $\frac{(s-x)^2}{\delta^2} > 1$ y $\abs{f(s)-f(x)}
\leq 2M \leq \frac{2M}{\delta^2}(s-x)^2$. En ambos casos vale la
cota anunciada.

\textbf{14.} Fijemos $x$; la pregunta 13 dice, como funciones de
$s$:
\[
-\varepsilon e_0 - \frac{2M}{\delta^2}q_x
\;\leq\; f - f(x)e_0
\;\leq\; \varepsilon e_0 + \frac{2M}{\delta^2}q_x,
\qquad q_x = e_2 - 2x\,e_1 + x^2e_0 .
\]
Apliquemos el operador lineal monótono $L_n$ (pregunta 12) y
evaluemos en $x$:
\[
\abs{L_nf(x) - f(x)L_ne_0(x)}
\leq \varepsilon L_ne_0(x)
+ \frac{2M}{\delta^2}\bigl(L_ne_2(x) - 2xL_ne_1(x)
+ x^2L_ne_0(x)\bigr) .
\]

\textbf{15.} Escribamos $\alpha_j = L_ne_j - e_j$, de modo que
$\norm{\alpha_j}_\infty \to 0$. Como $e_2(x) - 2xe_1(x) + x^2e_0(x)
= 0$:
\[
L_ne_2(x) - 2xL_ne_1(x) + x^2L_ne_0(x)
= \alpha_2(x) - 2x\,\alpha_1(x) + x^2\alpha_0(x),
\]
de \omterm{def:b2:nvs:norm}{norma} del supremo a lo sumo $\norm{\alpha_2} +
2\norm{\alpha_1} + \norm{\alpha_0} \to 0$. Además $L_ne_0 \to e_0$
\omterm{def:b2:funcseq:def}{uniformemente}, luego $L_ne_0 \leq 2$ para $n$ grande, y
$\abs{f(x)}\abs{L_ne_0(x) - 1} \leq M\norm{\alpha_0} \to 0$.
Ensamblando con la pregunta 14: para $n$ grande y uniformemente en
$x$,
\[
\abs{L_nf(x) - f(x)} \leq 2\varepsilon +
\frac{2M}{\delta^2}\,o(1) + M\,o(1) \leq 3\varepsilon :
\]
$L_nf \to f$ \omterm{def:b2:funcseq:def}{uniformemente}; el teorema de Korovkin.

\textbf{16.} $B_ne_0 = e_0$ y $B_ne_1 = e_1$ exactamente, y
$\norm{B_ne_2 - e_2}_\infty = \max_x\frac{x(1-x)}{n} = \frac{1}{4n}
\to 0$ (pregunta 2): se aplica Korovkin y Weierstrass sale por
tercera vez.

\textbf{17.} $I_nf$ es lineal en $f$ (lo son los valores nodales) y,
sobre cada celda, el interpolante afín de valores nodales no
negativos es no negativo: es positivo. $I_ne_0 = e_0$ e $I_ne_1 =
e_1$ porque una función afín coincide con su propio interpolante.
Sobre una celda $\intcc ab$ (con $b - a = \frac1n$), el interpolante
afín de $e_2$ es $L(t) = (a+b)t - ab$, y
\[
L(t) - t^2 = (t-a)(b-t) \in
\intcc{0}{\tfrac{(b-a)^2}{4}} ,
\]
con máximo en el punto medio: $\norm{I_ne_2 - e_2}_\infty =
\frac{1}{4n^2} \to 0$. Korovkin: $I_nf \to f$ \omterm{def:b2:funcseq:def}{uniformemente} para
toda $f$ \omterm{def:b2:metric:continuity}{continua}; aproximación poligonal, sin necesidad de
ninguna otra estimación.

\textbf{18.} Dados $f$ y $\varepsilon$: Weierstrass proporciona un
polinomio $P = \sum_{j=0}^d a_jx^j$ con $\norm{f - P}_\infty \leq
\frac\varepsilon2$; sustituir cada $a_j$ por un racional $b_j$ con
$\abs{a_j - b_j} \leq \frac{\varepsilon}{2(d+1)}$ mueve la \omterm{def:b2:nvs:norm}{norma} del
supremo sobre $\intcc{0}{1}$ a lo sumo $\frac\varepsilon2$. El
conjunto de los polinomios de coeficientes racionales es una unión
\omterm{def:b2:structures:countable}{numerable} (sobre $d$) de \omterm{def:b2:structures:countable}{conjuntos numerables} y, por tanto,
\omterm{def:b2:structures:countable}{numerable}, y es denso: $C(\intcc{0}{1})$ es separable.

\textbf{19.} Por linealidad, $\int_0^1 fP = 0$ para todo polinomio
$P$. Elijamos polinomios $P_n \to f$ \omterm{def:b2:funcseq:def}{uniformemente}
(Weierstrass):
\[
\Bigl|\int_0^1 f^2\Bigr|
= \Bigl|\int_0^1 f\,(f - P_n)\Bigr|
\leq \norm f_\infty\,\norm{f - P_n}_\infty
\longrightarrow 0 ,
\]
luego $\int_0^1 f^2 = 0$. Si $f(x_0) \neq 0$, la \omterm{def:b2:metric:continuity}{continuidad} da $f^2
\geq c > 0$ sobre un subintervalo, en contra de la anulación de la
integral: $f = 0$. En consecuencia, dos funciones \omterm{def:b2:metric:continuity}{continuas} con los
mismos momentos $\int f t^n$ coinciden.

\textbf{20.} Con $p_{n,k}(x) = \binom nk x^k(1-x)^{n-k}$ y los
convenios $p_{n-1,-1} = p_{n-1,n} = 0$, la regla del producto y las
identidades $k\binom nk = n\binom{n-1}{k-1}$, $(n-k)\binom nk =
n\binom{n-1}{k}$ dan
\[
p_{n,k}'(x) = n\bigl(p_{n-1,k-1}(x) - p_{n-1,k}(x)\bigr) .
\]
Sumando contra $f(k/n)$ y desplazando el índice en la primera suma
(\omterm{thm:b2:series:abel}{sumación de Abel}):
\[
(B_nf)'(x) = n\sum_{j=0}^{n-1}\Bigl(f\Bigl(\frac{j+1}n\Bigr)
- f\Bigl(\frac jn\Bigr)\Bigr)p_{n-1,j}(x) .
\]

\textbf{21.} Por el teorema del valor medio, $f(\frac{j+1}n) -
f(\frac jn) = \frac1n f'(\xi_j)$ con $\xi_j \in
\intoo{j/n}{(j+1)/n}$, de modo que $(B_nf)'(x) = \sum_j
f'(\xi_j)\,p_{n-1,j}(x)$. El nodo $\frac{j}{n-1}$ también está en
$\intcc{j/n}{(j+1)/n}$ (ambas desigualdades se reducen a $j \leq
n-1$), luego $\abs{\xi_j - \frac j{n-1}} \leq \frac1n$ y
\[
\bigl|(B_nf)'(x) - B_{n-1}(f')(x)\bigr|
\leq \sum_j\Bigl|f'(\xi_j) -
f'\Bigl(\frac{j}{n-1}\Bigr)\Bigr| p_{n-1,j}(x)
\leq \omega_{f'}\Bigl(\frac1n\Bigr) \longrightarrow 0
\]
\omterm{def:b2:funcseq:def}{uniformemente}. Como $B_{n-1}(f') \to f'$ \omterm{def:b2:funcseq:def}{uniformemente} (el~\cref{thm:b2:funcseq:weierstrass} aplicado a la función \omterm{def:b2:metric:continuity}{continua}
$f'$), la desigualdad triangular da $(B_nf)' \to f'$
\omterm{def:b2:funcseq:def}{uniformemente}. Los polinomios $P_n = B_nf$ convergen entonces a $f$
en el sentido $C^1$.

\textbf{22.} Taylor--Lagrange en $x$: $f(\frac kn) - f(x) =
f'(x)(\frac kn - x) + \frac{f''(\xi_k)}2(\frac kn - x)^2$. Sumando
contra los $p_k$, el término lineal muere (pregunta 2):
\[
\abs{B_nf(x) - f(x)}
\leq \frac{\norm{f''}_\infty}{2}\sum_k\Bigl(\frac kn -
x\Bigr)^2p_k
= \frac{\norm{f''}_\infty}{2}\cdot\frac{x(1-x)}{n}
\leq \frac{\norm{f''}_\infty}{8n} .
\]

\textbf{23.} Dos derivaciones más de $(x+y)^n$ dan los momentos
factoriales, con $n_{(j)} = n(n-1)\cdots(n-j+1)$:
\[
\sum_k k_{(j)}\,p_k = n_{(j)}\,x^j \qquad (j = 3, 4),
\]
y las identidades $k^3 = k_{(3)} + 3k_{(2)} + k$ y $k^4 = k_{(4)} +
6k_{(3)} + 7k_{(2)} + k$ los convierten en momentos de potencias:
\[
\sum_k k^3p_k = n_{(3)}x^3 + 3n_{(2)}x^2 + nx,
\qquad
\sum_k k^4p_k = n_{(4)}x^4 + 6n_{(3)}x^3 + 7n_{(2)}x^2 + nx .
\]
Desarrollando $(k - nx)^4$ y agrupando (un cálculo paciente pero
puramente mecánico con los cuatro momentos de potencias):
\[
\sum_k(k-nx)^4p_k = nx(1-x)\bigl(1 + 3(n-2)x(1-x)\bigr) .
\]
Con $x(1-x) \leq \frac14$, el miembro derecho es a lo sumo $\frac
n4\bigl(1 + \frac{3n}4\bigr) = \frac{3n^2}{16} + \frac n4 \leq n^2$
para $n \geq 1$.

\textbf{24.} La forma de Peano de Taylor en $x$ define $\eta(t) =
\frac{f(t) - f(x) - f'(x)(t-x) - \frac12f''(x)(t-x)^2} {(t-x)^2}$
para $t \neq x$, con $\eta(x) = 0$: por Taylor--Lagrange, $\eta(t) =
\frac12\bigl(f''(\xi) - f''(x)\bigr)$ para cierto $\xi$ entre $t$ y
$x$, de modo que $\abs\eta \leq \norm{f''}_\infty$ y $\eta(t) \to 0$
cuando $t \to x$ (\omterm{def:b2:metric:continuity}{continuidad} de $f''$). Sumando el desarrollo
contra los $p_k$ y usando las preguntas 2--3:
\[
n\bigl(B_nf(x) - f(x)\bigr)
= \frac{x(1-x)}{2}f''(x)
+ n\sum_k\eta\Bigl(\frac kn\Bigr)\Bigl(\frac kn -
x\Bigr)^2p_k .
\]
Dado $\varepsilon$, elíjase $\delta$ con $\abs\eta \leq \varepsilon$
sobre $\abs{t - x}\leq\delta$. Parte cercana: a lo sumo
$\varepsilon\,n\cdot\frac{x(1-x)}n \leq \varepsilon$. Parte lejana:
con $C = \norm{f''}_\infty$ y la pregunta 23,
\[
n\,C\sum_{\abs{k/n-x}>\delta}\Bigl(\frac kn -
x\Bigr)^2p_k
\leq \frac{nC}{\delta^2}\sum_k\Bigl(\frac kn -
x\Bigr)^4p_k
= \frac{nC}{\delta^2 n^4}\sum_k(k-nx)^4p_k
\leq \frac{C}{\delta^2 n} \longrightarrow 0 .
\]
Por tanto, $n(B_nf(x) - f(x)) \to \frac{x(1-x)}2f''(x)$: el teorema
de Voronovskaya. Para $f = e_2$ esto es exacto en todo $n$
(\cref{exo:b2:funcseq:7}): el techo de $\frac1n$ es real.

\textbf{25.} (i) La positividad convirtió desigualdades
\omterm{def:b2:funcseq:def}{puntuales} en desigualdades entre operadores: dio la cota de la
\omterm{def:b2:nvs:norm}{norma}, Jensen, Chebyshev y todo Korovkin; la linealidad sola no
demuestra nada aquí. (ii) La \omterm{def:b2:metric:compact}{compacidad} entró por Heine (preguntas
6 y 13), por la acotación de $f$ y por el hecho mismo de que la
\omterm{def:b2:nvs:norm}{norma} $\norm\cdot_\infty$ sea finita. (iii) Tres funciones de
prueba bastan porque la positividad reduce todo a controlar $L_n$
sobre la única familia $(s-x)^2 = e_2 - 2xe_1 + x^2e_0$, cuya
envoltura es la de $e_0, e_1, e_2$. (iv) Bernstein converge a la
velocidad honesta $\omega(n^{-1/2})$ para toda $f$ \omterm{def:b2:metric:continuity}{continua}
(óptima, pregunta 11) pero satura en $\frac1n$ para $f$ suave
(pregunta 24); el capítulo de Fourier ejecuta el mismo programa
para funciones periódicas con el núcleo de Fejér, otro operador
positivo con las mismas virtudes y la misma modestia.
\end{solution}
