\section*{Capítulo \ref{ch:b2:diffeq} --- Ecuaciones diferenciales}

\begin{solution}{exo:b2:diffeq:1}
Primera matriz: \omterm{def:b2:reduction:eigen}{valores propios} $1, 2$, con \omterm{def:b2:reduction:eigen}{vectores propios}
$(1,0)$ y $(1,1)$. Descompóngase $X_0 = (1,0) = 1\cdot(1,0) +
0\cdot(1,1)$: la solución es
\[
X(t) = \eu^{t}\begin{pmatrix}1\\ 0\end{pmatrix} .
\]
(El vector inicial es él mismo un \omterm{def:b2:reduction:eigen}{vector propio}.)

Segunda: $A = 2I + N$, $N = E_{12}$, $N^2 = 0$: $\eu^{tA} =
\eu^{2t}(I + tN)$, luego
\[
X(t) = \eu^{2t}\begin{pmatrix} 1 & t\\ 0 & 1\end{pmatrix}
\begin{pmatrix}1\\ 0\end{pmatrix}
= \eu^{2t}\begin{pmatrix}1\\ 0\end{pmatrix}.
\]
\end{solution}

\begin{solution}{exo:b2:diffeq:2}
$y' = \sin(ty)$: $\abs{\sin(ty) - \sin(tz)} \leq \abs t\,\abs{y -
z}$; es \omterm{def:b2:metric:continuity}{lipschitziana} en $y$ \omterm{def:b2:funcseq:def}{uniformemente} sobre todo segmento de
tiempos: soluciones globales únicas sobre $\R$ (aplíquese el teorema
sobre cada segmento).

$y' = y^2$: solo es localmente \omterm{def:b2:metric:continuity}{lipschitziana}; no hay teorema global, y
en efecto $y(0) = 1$ explota en $t = 1$.

$y' = \abs y$: $\abs\cdot$ es $1$-lipschitziana: existencia y unicidad
globales. Con $y(0) = 0$: $y \equiv 0$ (¡unicidad!). Con $y(0) = 1$:
$y$ permanece positiva (no puede cruzar la solución nula), luego $y' =
y$: $y = \eu^t$.
\end{solution}

\begin{solution}{exo:b2:diffeq:3}
Si $y(t_1) = z(t_1)$ en algún instante, entonces $y$ y $z$ resuelven
el mismo problema de Cauchy en $t_1$: la unicidad fuerza $y = z$
sobre su intervalo común; soluciones distintas nunca se encuentran.

Para $y' = y(1-y)$: las constantes $0$ y $1$ son soluciones. Una
solución que arranque en $\intoo{0}{1}$ nunca puede alcanzar $0$ ni
$1$ (chocaría con una solución constante): permanece en
$\intoo{0}{1}$ y, por tanto, es global (acotada, sin explosión; por
ejemplo, por el criterio del~\cref{exo:b2:diffeq:9}, o porque el
campo vectorial está acotado sobre la banda en la que queda
atrapada).
\end{solution}

\begin{solution}{exo:b2:diffeq:4}
$(A - 2I)^2 = \begin{pmatrix}1 & 1\\ -1 & -1\end{pmatrix}^2 = 0$:
Dunford con $D = 2I$ y $N = A - 2I$:
\[
\eu^{tA} = \eu^{2t}\,(I + tN)
= \eu^{2t}\begin{pmatrix} 1 + t & t\\ -t & 1 - t\end{pmatrix}.
\]
Variación de las constantes con $B(s) = (\eu^{2s}, 0)^{\mathsf T}$:
\[
X(t) = \int_0^t \eu^{(t-s)A}B(s)\,\dd s
= \eu^{2t}\int_0^t \begin{pmatrix} 1 + (t-s)\\ -(t-s)
\end{pmatrix}\dd s
= \eu^{2t}\begin{pmatrix} t + \frac{t^2}{2}\\[2pt] -\frac{t^2}{2}
\end{pmatrix},
\]
usando $\eu^{(t-s)A}B(s) = \eu^{2(t-s)}(I + (t-s)N)\,(\eu^{2s},
0)^{\mathsf T} = \eu^{2t}\bigl(1 + (t-s),\, -(t-s)\bigr)^{\mathsf
T}$. (Comprobación: $X(0) = 0$; y $X' - AX = (\eu^{2t}, 0)^{\mathsf
T}$ por derivación.)
\end{solution}

\begin{solution}{exo:b2:diffeq:5}
\emph{Vía la serie:} $A = \alpha I + \beta J$ con $J =
\begin{pmatrix}0 & -1\\ 1 & 0\end{pmatrix}$ y $J^2 = -I$; los dos
sumandos conmutan, luego $\eu^{tA} = \eu^{\alpha t}\,\eu^{\beta tJ}$,
y la serie de $\eu^{\beta t J}$ se escinde según las potencias pares
e impares en $\cos(\beta t)I + \sin(\beta t)J$: la matriz de rotación
por homotecia enunciada.

\emph{Vía los complejos:} identifíquese $(x, y) \in \R^2$ con $z = x +
\iu y$; el sistema $X' = AX$ se lee $z' = (\alpha + \iu\beta)z$, cuya
solución $z(t) = \eu^{\alpha t}\eu^{\iu\beta t}z_0$ es exactamente la
espiral: módulo $\eu^{\alpha t}$ y argumento avanzando a velocidad
$\beta$.
\end{solution}

\begin{solution}{exo:b2:diffeq:6}
Sea $v(t) = \eu^{-k(t - t_0)}\int_{t_0}^t u$. Entonces
\[
v'(t) = \eu^{-k(t-t_0)}\Bigl(u(t) - k\int_{t_0}^t u\Bigr)
\leq C\,\eu^{-k(t-t_0)},
\]
por la hipótesis. Integrando de $t_0$ a $t$ (con $v(t_0) = 0$): $v(t)
\leq \frac{C}{k}\bigl(1 - \eu^{-k(t - t_0)}\bigr)$, es decir,
$k\int_{t_0}^t u \leq C\bigl(\eu^{k(t-t_0)} - 1\bigr)$; y
reinyectando esto en la hipótesis: $u(t) \leq C\eu^{k(t-t_0)}$.

Unicidad y dependencia: dos soluciones $y, z$ de la ecuación integral
cumplen
\[
\norm{y(t) - z(t)} \leq \norm{y_0 - z_0} +
k\int_{t_0}^{t}\norm{y - z},
\]
y Gronwall con $C = \norm{y_0 - z_0}$ da la cota exponencial; con $C
= 0$ se obtiene la unicidad.
\end{solution}

\begin{solution}{exo:b2:diffeq:7}
Sustitúyase $y = y_1 z$ con $y_1 = \frac{\sin t}{t}$: la fórmula
general de bajada del orden (\cref{prop:b2:diffeq:secondorder}) da,
para $u = z'$,
\[
y_1 u' + \Bigl(2y_1' + \frac{2}{t}\,y_1\Bigr)u = 0
\]
(con la ecuación normalizada como $y'' + \frac2t y' + y = 0$).
Calculemos $2y_1' + \frac2t y_1 = 2\,\frac{t\cos t - \sin t}{t^2} +
\frac{2\sin t}{t^2} = \frac{2\cos t}{t}$: así pues,
\[
\frac{u'}{u} = -\frac{2\cos t}{t}\cdot\frac{t}{\sin t}
= -2\cot t
\quad\Longrightarrow\quad
u = \frac{1}{\sin^2 t}
\quad (\text{salvo constante}),
\]
y $z = -\cot t$, lo que da $y_2 = y_1 z = -\frac{\cos t}{t}$.
Solución general sobre $\intoo{0}{\pi}$:
\[
y(t) = \alpha\,\frac{\sin t}{t} + \beta\,\frac{\cos t}{t} .
\]
(Son las funciones esféricas de Bessel de orden cero.)
\end{solution}

\begin{solution}{exo:b2:diffeq:8}
Dunford: $A = D + N$ que conmutan, con $D$ \omterm{def:b2:reduction:diag}{diagonalizable} y los mismos
\omterm{def:b2:reduction:eigen}{valores propios}, y $N$ nilpotente, de modo que
\[
\eu^{tA} = \eu^{tD}\,\eu^{tN},
\qquad
\eu^{tN} = \sum_{k < n} \frac{t^kN^k}{k!}
\ \text{(un polinomio matricial en } t).
\]
Sea $-2\alpha = \max_i \Re\lambda_i < 0$. En una base que
diagonalice $D$, $\vertiii{\eu^{tD}} \leq \eu^{-2\alpha t}$ (las
entradas $\eu^{t\lambda_i}$ tienen módulo $\eu^{t\Re\lambda_i}$); y
las \omterm{def:b2:nvs:norm}{normas} en bases distintas difieren en constantes. Por tanto,
\[
\norm{X(t)} \leq \vertiii{\eu^{tA}}\,\norm{X_0}
\leq C'\,\eu^{-2\alpha t}\,(1 + t)^{n-1}\,\norm{X_0}
\leq C\,\eu^{-\alpha t}\norm{X_0} ,
\]
absorbiendo el polinomio en un factor exponencial ($\eu^{-\alpha
t}(1+t)^{n-1} \to 0$, luego está acotado).
\end{solution}

\begin{solution}{exo:b2:diffeq:9}
Sea $y$ una solución maximal sobre $\intco{0}{T}$, con $T \leq
\infty$, y supongamos $T < \infty$. La forma integral da, para $t <
T$,
\[
\norm{y(t)} \leq \norm{y_0} + \int_0^t \bigl(a\norm{y(s)} +
b\bigr)\dd s
\leq \bigl(\norm{y_0} + bT\bigr) + a\int_0^t\norm{y(s)}\,\dd s ,
\]
y Gronwall acota $\norm{y(t)} \leq (\norm{y_0} + bT)\,\eu^{aT} =: M$
sobre $\intco{0}{T}$: la solución permanece en una bola \omterm{def:b2:metric:compact}{compacta}.
Entonces $y' = f(t, y)$ está acotada cerca de $T$, luego $y$ es
\omterm{def:b2:metric:continuity}{lipschitziana} cerca de $T$ y se extiende de manera \omterm{def:b2:metric:continuity}{continua} a $T$
(criterio de Cauchy); resolver el problema de Cauchy en $(T, y(T))$
prolonga $y$ más allá de $T$, en contradicción con la maximalidad.
Por tanto, $T = \infty$: no hay escape en tiempo finito con
crecimiento lineal.
\end{solution}

\begin{solution}{exo:b2:diffeq:10}
Raíces características de $r^2 - 3r + 2$: $1$ y $2$, luego las
soluciones homogéneas son $a\eu^t + b\eu^{2t}$. La conjetura
$\alpha\eu^t$ falla porque $\eu^t$ ya resuelve la ecuación homogénea
(la raíz $r = 1$ ``resuena'' con el segundo miembro). Con $y = \alpha
t\,\eu^t$: $y' = \alpha(1 + t)\eu^t$, $y'' = \alpha(2 + t)\eu^t$, y
\[
y'' - 3y' + 2y = \alpha\eu^t\bigl(2 + t - 3 - 3t + 2t\bigr)
= -\alpha\,\eu^t :
\]
$\alpha = -1$, $y_p = -t\,\eu^t$. Solución general: $y = a\eu^t +
b\eu^{2t} - t\eu^t$. Datos iniciales $y(0) = y'(0) = 0$: $a + b = 0$
y $a + 2b - 1 = 0$, luego $b = 1$ y $a = -1$:
\[
y(t) = \eu^{2t} - (1 + t)\,\eu^{t} .
\]

\end{solution}

\begin{solution}{exo:b2:diffeq:11}
$A = \lambda I + N$ con $N = E_{12} + E_{23}$: $N^2 = E_{13}$, $N^3 =
0$, y $\lambda I$ conmuta con $N$:
\[
\eu^{tA} = \eu^{\lambda t}\Bigl(I + tN +
\frac{t^2}{2}N^2\Bigr)
= \eu^{\lambda t}\begin{pmatrix}
1 & t & \frac{t^2}{2}\\
0 & 1 & t\\
0 & 0 & 1
\end{pmatrix}.
\]
Soluciones: $X(t) = \eu^{\lambda t}\bigl(X_0 + tNX_0 +
\frac{t^2}2N^2X_0\bigr)$; cada componente es $\eu^{\lambda t}$ por un
polinomio de grado $\leq 2$. La cota del grado es el índice de
nilpotencia menos uno: la serie de $\eu^{tN}$ se trunca en $N^2$.
\end{solution}

\begin{solution}{exo:b2:diffeq:12}
\begin{enumerate}
  \item Si $X(T) = X(0)$, entonces $Y(t) = X(t + T)$ resuelve $Y' =
        AY + B(t + T) = AY + B(t)$ con $Y(0) = X(0)$: la unicidad
        (\cref{thm:b2:diffeq:linear}) da $Y = X$, es decir, $X$ es
        $T$-periódica. El recíproco es trivial.
  \item Trigonalícese sobre $\C$: $A = PT'P^{-1}$ con $T'$
        triangular superior de diagonal $(\lambda_i)$. Toda potencia
        de una matriz triangular es triangular con diagonal
        $(\lambda_i^k)$, luego $\eu^{TA} = P\eu^{TT'}P^{-1}$ es
        triangular en la misma base con diagonal
        $(\eu^{T\lambda_i})$: esos son los \omterm{def:b2:reduction:eigen}{valores propios}.
        Entonces $I - \eu^{TA}$ es invertible si y solo si
        $\eu^{T\lambda} \neq 1$ para todo \omterm{def:b2:reduction:eigen}{valor propio}, es decir, si
        y solo si $T\lambda \notin 2\iu\pi\Z$, que es la hipótesis
        enunciada.
  \item Variación de las constantes: $X(T) = \eu^{TA}X(0) +
        \int_0^T\eu^{(T-s)A}B(s)\dd s$, de modo que $X(T) = X(0)$ se
        lee
        \[
        \bigl(I - \eu^{TA}\bigr)X(0) =
        \int_0^{T}\eu^{(T-s)A}B(s)\,\dd s ,
        \]
        que tiene una única solución $X(0)$ bajo la hipótesis de
        invertibilidad: exactamente una solución $T$-periódica. Para
        el oscilador armónico ($\lambda = \pm\iu\omega$), el caso
        excluido es $\omega T \in 2\pi\Z$: una excitación cuyo
        periodo es múltiplo del periodo propio; la \omterm{pb:b2:diffeq:1}{resonancia}, como
        cuantifica el problema de fin de semana.
\end{enumerate}
\end{solution}

\begin{solution}{pb:b2:diffeq:1}
\textbf{1.} El \omterm{def:b2:reduction:charpoly}{polinomio característico} es $\lambda^2 - \tau\lambda +
\delta$, de raíces $\frac{\tau\pm\sqrt\Delta}{2}$. Si $\delta < 0$,
entonces $\Delta = \tau^2 - 4\delta > 0$ y las dos raíces reales
tienen producto $\delta < 0$: signos opuestos. Si $\delta > 0$ y
$\Delta \geq 0$: raíces reales de producto $> 0$ y suma $\tau$, luego
ambas del signo de $\tau$. Si $\Delta < 0$: par conjugado $\alpha \pm
\iu\beta$ con $\alpha = \frac\tau2$ y $\beta =
\frac{\sqrt{-\Delta}}2$.

\textbf{2.} $X(t) = a\,\eu^{\mu t}v_- + b\,\eu^{\lambda t}v_+$. Las
órbitas con $a = 0$ (o con $b = 0$) recorren la recta propia
inestable (o la estable); todas las demás tienen $\norm X \to \infty$
en ambos sentidos del tiempo, asintóticas a $\R v_+$ cuando $t \to
+\infty$ y a $\R v_-$ cuando $t \to -\infty$: la imagen de la silla.
La acotación sobre todo $\R$ fuerza $b = 0$ (si no, explosión en
$+\infty$) y $a = 0$ (en $-\infty$): solo el origen.

\textbf{3.} Con $\mu < \lambda < 0$, ambas exponenciales decaen:
$X(t) \to 0$. Si $b \neq 0$, sáquese factor $\eu^{\lambda t}$:
\[
X(t) = \eu^{\lambda t}\bigl(b\,v_\lambda + a\,\eu^{(\mu -
\lambda)t}v_\mu\bigr),
\qquad \eu^{(\mu-\lambda)t} \to 0 :
\]
la dirección de $X(t)$ tiende a $\R v_\lambda$, la dirección propia
\emph{lenta}; todas las órbitas salvo el eje rápido llegan tangentes
a ella (el panel derecho de los retratos de fases del capítulo).

\textbf{4.} En la base en la que $A = \begin{pmatrix} \alpha &
-\beta\\ \beta & \alpha\end{pmatrix}$ (\cref{exo:b2:diffeq:5}; para
los sistemas del oscilador de la parte II esta forma se alcanza
mediante un cambio de base real explícito), la solución es
$\eu^{\alpha t}$ por una rotación de ángulo $\beta t$: espirales
logarítmicas, contractivas cuando $\alpha = \frac\tau2 < 0$,
expansivas cuando $\tau > 0$, y curvas cerradas (elipses en las
coordenadas originales) cuando $\tau = 0$: el centro.

\textbf{5.} $\Delta = 0$ da el \omterm{def:b2:reduction:eigen}{valor propio} doble $\lambda =
\frac\tau2$; por Cayley--Hamilton
(\cref{thm:b2:reduction:cayleyhamilton}), $(A - \lambda I)^2 = 0$, de
modo que $N = A - \lambda I$ es nilpotente, conmuta con $\lambda I$ y
$\eu^{tA} = \eu^{\lambda t}(I + tN)$. Si $N = 0$: $A = \lambda I$ y
todas las semirrectas son órbitas (nodo estrella). Si $N \neq 0$:
$X(t) = \eu^{\lambda t}(X_0 + tNX_0)$, y para $NX_0 \neq 0$ la
dirección converge a la única dirección propia
$\operatorname{im}N$: nodo impropio. Esto completa la imagen
traza--determinante.

\textbf{6.} $\tau = -2\zeta\omega$, $\delta = \omega^2 > 0$, $\Delta
= 4\omega^2(\zeta^2 - 1)$. Así pues: $0 < \zeta < 1$ da $\Delta < 0$
y $\tau < 0$: foco estable; $\zeta = 1$: $\Delta = 0$: nodo estable
degenerado; $\zeta > 1$: $\Delta > 0$, $\tau < 0$, $\delta > 0$: nodo
estable; $\zeta = 0$: $\tau = 0$, $\delta > 0$: centro. Un viaje
vertical por el plano a la altura $\delta = \omega^2$.

\textbf{7.} Raíces $r = -\zeta\omega \pm \omega\sqrt{\zeta^2-1}$.
Para $\zeta < 1$: $r = -\zeta\omega \pm \iu\omega_d$ con $\omega_d =
\omega\sqrt{1-\zeta^2}$:
\[
x(t) = \eu^{-\zeta\omega t}\bigl(a\cos\omega_dt +
b\sin\omega_dt\bigr)
= R\,\eu^{-\zeta\omega t}\cos(\omega_dt - \varphi) .
\]
Para $\zeta = 1$: $x = (a + bt)\eu^{-\omega t}$. Para $\zeta > 1$: $x
= a\eu^{r_-t} + b\eu^{r_+t}$, con ambos ritmos negativos. En el caso
subamortiguado, los máximos sucesivos de $\abs x$ ocurren en
instantes separados por el seudoperiodo $\frac{2\pi}{\omega_d}$
(misma fase del coseno), y su cociente es
$\eu^{-\zeta\omega\cdot2\pi/\omega_d} =
\eu^{-2\pi\zeta/\sqrt{1-\zeta^2}}$: el decremento logarítmico, un
medidor de amortiguamiento legible en un osciloscopio.

\textbf{8.} Racionalizando,
\[
\abs{\lambda_{\mathrm{lento}}} = \omega\bigl(\zeta -
\sqrt{\zeta^2-1}\bigr)
= \frac{\omega}{\zeta + \sqrt{\zeta^2 - 1}} ,
\]
cuyo denominador crece con $\zeta \geq 1$: el ritmo de decaimiento es
máximo en $\zeta = 1$, donde vale $\omega$. Una puerta
sobreamortiguada se cierra sin dar un portazo, pero \emph{despacio};
el amortiguamiento crítico es el óptimo del ingeniero.

\textbf{9.} $E' = x'x'' + \omega^2xx' = x'\bigl(-2\zeta\omega x' -
\omega^2x\bigr) + \omega^2xx' = -2\zeta\omega\,x'^2 \leq 0$. Si $x$
fuera periódica y no constante, $E$ sería periódica y no creciente,
luego constante, lo que forzaría $x' \equiv 0$: $x$ constante, y
entonces $\omega^2x = 0$, o sea, $x \equiv 0$. Por tanto, para $\zeta
> 0$ la única solución periódica es el reposo: el amortiguamiento
mata todo \omterm{def:b2:structures:sn}{ciclo}.

\textbf{10.} El centro $\zeta = 0$ vive sobre la recta $\tau = 0$ del
plano traza--determinante, un conjunto de interior vacío: una
perturbación arbitrariamente pequeña de la matriz (cualquier
amortiguamiento físico) desplaza $\tau$ fuera de cero y convierte las
órbitas cerradas en espirales. La periodicidad del oscilador sin
amortiguamiento es, pues, un fenómeno al filo de la navaja, no
robusto.

\textbf{11.} Sustituyendo $x_p = \Re(z\eu^{\iu\gamma t})$ en la
ecuación:
\[
\bigl(-\gamma^2 + 2\iu\zeta\omega\gamma +
\omega^2\bigr)z = F
\quad\Longrightarrow\quad
z = \frac{F}{\omega^2 - \gamma^2 + 2\iu\zeta\omega\gamma},
\]
de modo que $x_p = \abs z\cos(\gamma t - \varphi)$ con $\varphi =
\arg(\omega^2 - \gamma^2 + 2\iu\zeta\omega\gamma)$, es decir,
$\tan\varphi = \frac{2\zeta\omega\gamma}{\omega^2 - \gamma^2}$, y el
$A(\gamma) = \abs z$ enunciado.

\textbf{12.} La diferencia de dos soluciones resuelve la ecuación
homogénea, que para $\zeta > 0$ decae a $0$ (pregunta 7): toda
solución es $x_p$ más un transitorio que se desvanece en el infinito.
El régimen estacionario es un atractor global: las condiciones
iniciales se olvidan y solo quedan $A(\gamma)$ y el desfase
$\varphi$.

\textbf{13.} Minimícese $g(u) = (\omega^2 - u)^2 +
4\zeta^2\omega^2u$ sobre $u = \gamma^2 \geq 0$: $g'(u) = -2(\omega^2
- u) + 4\zeta^2\omega^2 = 0$ en $u = \omega^2(1 - 2\zeta^2)$,
interior si y solo si $\zeta < \frac{1}{\sqrt2}$. Allí,
\[
g(u_*) = 4\zeta^4\omega^4 + 4\zeta^2\omega^4(1 - 2\zeta^2)
= 4\zeta^2\omega^4(1 - \zeta^2),
\qquad
A(\gamma_*) = \frac{F}{2\zeta\omega^2\sqrt{1 - \zeta^2}} .
\]
Frente a la respuesta estática $A(0) = \frac{F}{\omega^2}$: la
amplificación es $\frac{1}{2\zeta\sqrt{1-\zeta^2}} \approx
\frac{1}{2\zeta}$ para $\zeta$ pequeño; un sistema poco amortiguado
cerca de $\gamma_* \approx \omega$ multiplica la entrada por cien
cuando $\zeta = 0.005$.

\textbf{14.} La $x$ enunciada cumple $x(0) = x'(0) = 0$ y
\[
x'' + \omega^2 x = \frac{F(\omega^2 -
\gamma^2)\cos\gamma t}{\omega^2 - \gamma^2} = F\cos\gamma t
\]
(las partes en $\cos\omega t$ se cancelan). La forma de producto se
sigue de $\cos p - \cos q = 2\sin\frac{q+p}{2}\sin\frac{q-p}2$ con $p
= \gamma t$ y $q = \omega t$. Para $\gamma$ cerca de $\omega$, el
factor $\sin\frac{(\omega-\gamma)t}2$ es una envolvente lenta que
modula la oscilación rápida $\sin\frac{(\omega+\gamma)t}2$: batidos,
de amplitud $\frac{2F}{\abs{\omega^2-\gamma^2}}$; grande, pero
acotada.

\textbf{15.} Para $x_p = \frac{F}{2\omega}t\sin\omega t$:
\[
x_p'' = \frac{F}{2\omega}\bigl(2\omega\cos\omega t -
\omega^2t\sin\omega t\bigr)
= F\cos\omega t - \omega^2x_p :
\]
es solución. Y a $t$ fijo, haciendo $\gamma \to \omega$ en la
pregunta 14:
\[
\frac{2F\sin\frac{(\omega-\gamma)t}2
\sin\frac{(\omega+\gamma)t}{2}}
{(\omega-\gamma)(\omega+\gamma)}
\longrightarrow
\frac{2F\cdot\frac{(\omega-\gamma)t}2\big/(\omega-\gamma)
\cdot\sin\omega t}{2\omega}
= \frac{F\,t\sin\omega t}{2\omega} .
\]
La amplitud crece linealmente sin cota: la catástrofe de
\omterm{pb:b2:diffeq:1}{resonancia}, la razón por la que los soldados rompen el paso sobre los
puentes.

\textbf{16.} La onda cuadrada lleva armónicos en todas las
frecuencias impares $n = 1, 3, 5, \dots$; por linealidad, cada
armónico $n$ resulta amplificado por la respuesta del oscilador en
$\gamma = n$. Para $\omega = 3$, el tercer armónico entra en
\omterm{pb:b2:diffeq:1}{resonancia} exactamente. Los ingenieros temen las entradas cuadradas
(y en diente de sierra) porque excitan \emph{todos} los armónicos
impares a la vez: sea cual sea la frecuencia propia de la estructura,
algún armónico la está esperando.

\textbf{17.} $W' = y_1y_2'' - y_1''y_2 = -qy_1y_2 + qy_1y_2 = 0$: $W$
es constante (Liouville con una matriz compañera de traza nula). Que
$W = 0$ en un punto hace que los datos iniciales de $y_2$ sean
proporcionales a los de $y_1$, luego $y_2$ es proporcional a $y_1$
(unicidad); y $W \neq 0$ si y solo si son independientes. Si $y(t_0)
= y'(t_0) = 0$, entonces $y \equiv 0$ (unicidad): una solución no
nula tiene ceros simples, y un cero simple es aislado ($y'$ tiene
signo fijo en las cercanías).

\textbf{18.} Entre ceros consecutivos $a < b$, $y_1$ conserva un
signo; digamos $y_1 > 0$ sobre $\intoo ab$: entonces $y_1'(a) > 0$ e
$y_1'(b) < 0$ (ceros simples). Evaluando la constante $W = y_1y_2' -
y_1'y_2$ en $a$ y en $b$:
\[
W = -y_1'(a)\,y_2(a) = -y_1'(b)\,y_2(b) ,
\]
de modo que $y_2(a)$ e $y_2(b)$ tienen signos opuestos ($W \neq 0$
impide que ninguno se anule): $y_2$ se anula en $\intoo ab$ (valores
intermedios). Y no puede anularse dos veces ahí: dos ceros de $y_2$
flanquearían un cero de $y_1$ por el mismo argumento con los papeles
intercambiados, en contradicción con que $a$ y $b$ sean consecutivos.
Luego hay exactamente un cero: entrelazamiento.

\textbf{19.} Supongamos que $z$ no tiene ceros en $\intoo ab$;
sustituyendo $y$ y $z$ por sus opuestas si hace falta, supongamos $y
> 0$ y $z > 0$ sobre $\intoo ab$. Póngase $\varphi = yz' - y'z$:
$\varphi' = yz'' - y''z = (q_1 - q_2)\,yz \leq 0$ sobre $\intoo ab$,
luego $\varphi$ es no creciente. Pero $\varphi(a) = -y'(a)z(a) \leq
0$ (pues $y'(a) > 0$ y $z(a) \geq 0$) y $\varphi(b) = -y'(b)z(b) \geq
0$ (pues $y'(b) < 0$ y $z(b) \geq 0$): una función no creciente que
va de $\leq 0$ a $\geq 0$ se anula idénticamente, así que $(q_1 -
q_2)yz \equiv 0$ sobre $\intoo ab$. Si $q_1 < q_2$ en algún punto de
$\intoo ab$, esto es absurdo (allí $y, z > 0$): $z$ debe anularse
estrictamente dentro. En general (con $q_1 \leq q_2$), o bien $z$ se
anula en $\intoo ab$, o bien $\varphi \equiv 0$ fuerza a $z$ a ser
proporcional a $y$, que se anula en $a$ y en $b$: en todos los casos,
$z$ tiene un cero en $\intcc ab$.

\textbf{20.} Cota superior: compárese $y$ (de coeficiente $q \geq
m^2$) con $u(t) = \sin(m(t - a))$ (de coeficiente $m^2 \leq q$, de
modo que $y$ hace el papel de $z$ en la pregunta 19): si $y$ no
tuviera ceros en $\intoc{a}{a + \pi/m}$, los ceros $a$ y $a +
\frac\pi m$ de $u$ serían consecutivos con $y \neq 0$ entre ellos, en
contradicción con la pregunta 19; luego los ceros consecutivos de $y$
distan $\leq \frac\pi m$. Cota inferior: si dos ceros consecutivos $a
< b$ de $y$ cumplieran $b - a < \frac\pi M$, entonces $z(t) =
\sin(M(t-a))$ (de coeficiente $M^2 \geq q$) tendría que anularse en
$\intcc ab \subset \intoo{a}{a + \pi/M} \cup\{a\}$, donde su único
cero es el propio $a$; pero la pregunta 19 aplicada sobre $\intoo
ab$ con estrictez en los extremos da un cero en $\intcc ab$, y $z >
0$ sobre $\intoc ab$: contradicción. Por tanto, $\frac\pi M \leq b -
a \leq \frac\pi m$; y para $q \equiv \omega^2$ ambas cotas colapsan
al espaciado exacto $\frac\pi\omega$ del oscilador armónico.

\textbf{21.} Con $u = ty$: $u'' = ty'' + 2y'$, luego $ty'' + 2y' + ty
= u'' + u = 0$: $u = A\sin t + B\cos t = R\sin(t + \varphi)$, e $y =
\frac{u}{t}$ recupera $\frac{\sin t}t$ y $\frac{\cos t}t$
(\cref{exo:b2:diffeq:7}) sin bajar el orden. Los ceros de toda
solución no nula son los de $R\sin(t + \varphi)$: espaciados
exactamente $\pi$; la filosofía de Sturm en acción, según la cual el
coeficiente $q \equiv 1$ dicta los ceros, haya fórmula o no.

\textbf{22.} Póngase $x(t) =
\frac1\omega\int_0^t\sin(\omega(t-s))F(s)\dd s$. Entonces $x(0) = 0$;
\[
x'(t) = \frac1\omega\sin(0)F(t) +
\int_0^t\cos(\omega(t-s))F(s)\dd s
= \int_0^t\cos(\omega(t-s))F(s)\dd s ,
\]
luego $x'(0) = 0$; y $x''(t) = F(t) -
\omega\int_0^t\sin(\omega(t-s))F(s)\dd s = F(t) - \omega^2x(t)$
(derivación de una \omterm{thm:b2:integration:continuity}{integral con parámetro} con límite variable, como
en el capítulo de integración). Con $F(s) = F\cos\omega s$, la
fórmula de producto a suma da
\[
\int_0^t\sin(\omega(t-s))\cos(\omega s)\dd s
= \frac12\int_0^t\bigl(\sin\omega t + \sin(\omega t -
2\omega s)\bigr)\dd s
= \frac{t}{2}\sin\omega t ,
\]
(el segundo trozo integra cero), de modo que $x =
\frac{F}{2\omega}t\sin\omega t$: de nuevo la pregunta 15, ahora desde
Duhamel.

\textbf{23.} En forma de sistema, $X' = AX + (0, F(t))^{\mathsf T}$
con $\operatorname{Sp}A$ de partes reales negativas ($\zeta > 0$): la
variación de las constantes y el~\cref{exo:b2:diffeq:8}
($\vertiii{\eu^{tA}} \leq C\eu^{-\alpha t}$) dan
\[
\norm{X(t)} \leq C\eu^{-\alpha t}\norm{X_0}
+ \int_0^t C\eu^{-\alpha(t-s)}\norm{F}_\infty\dd s
\leq C\norm{X_0} + \frac{C\norm F_\infty}{\alpha} :
\]
entrada acotada, salida acotada; y uniformemente en los datos
iniciales una vez pasado el transitorio.

\textbf{24.} Para $\zeta = 0$ y $\gamma \neq \omega$, la solución de
la pregunta 14 está acotada, y sumarle cualquier solución homogénea
(acotada: las órbitas del centro son circunferencias) la mantiene
acotada; en $\gamma = \omega$, la pregunta 15 crece linealmente. Así
pues, para el oscilador sin amortiguamiento, la acotación bajo
excitación periódica falla en exactamente una frecuencia ---la
resonancia---, mientras que la pregunta 23 muestra que cualquier
amortiguamiento positivo restaura la acotación para \emph{todas} las
entradas acotadas.

\textbf{25.} (i) El plano traza--determinante clasifica toda la
dinámica lineal plana autónoma, y el oscilador de la parte II recorre
una de sus rectas verticales; la excitación abandona el plano (deja
de ser autónoma) y toma el relevo Duhamel. (ii) $\gamma_*$ es la
frecuencia que el sistema prefiere, $A(\gamma_*)$ el precio de
excitarla y $\frac{1}{2\zeta}$ el factor de amplificación: la agudeza
de la \omterm{pb:b2:diffeq:1}{resonancia} que los ingenieros llaman factor de calidad. (iii)
Los teoremas de Sturm leen la oscilación únicamente en el signo y el
tamaño de $q$: gobiernan ecuaciones (Bessel, Schrödinger) cuyas
soluciones no tienen fórmulas elementales. (iv) Los
\omterm{def:b2:diffeq:wronskian}{wronskianos} constantes (pregunta 17, vía Liouville) y la estabilidad
con entrada acotada (pregunta 23) se reutilizan en silencio cada vez
que el libro se topa con ecuaciones de coeficientes variables o con
sistemas perturbados.
\end{solution}
