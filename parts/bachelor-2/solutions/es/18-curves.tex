\section*{Capítulo \ref{ch:b2:curves} --- Curvas}

\begin{solution}{exo:b2:curves:1}
$\gamma'(t) = (1 - \cos t,\ \sin t)$, de modo que
\[
\norm{\gamma'(t)}^2 = (1 - \cos t)^2 + \sin^2 t
= 2 - 2\cos t = 4\sin^2\tfrac t2 ,
\]
y $\norm{\gamma'(t)} = 2\sin\frac t2$ (no negativo sobre $[0,
2\pi]$). De ahí,
\[
L = \int_0^{2\pi} 2\sin\tfrac t2\,\dd t
= \Bigl[-4\cos\tfrac t2\Bigr]_0^{2\pi} = 8 :
\]
un arco de cicloide tiene \omterm{def:b2:curves:length}{longitud} $8$ (para una rueda de radio $1$);
un resultado famoso de Wren, sin ningún $\pi$ a la vista.
\end{solution}

\begin{solution}{exo:b2:curves:2}
Con $x = a\cos t$, $y = b\sin t$: $x' = -a\sin t$, $y' = b\cos t$,
$x'' = -a\cos t$, $y'' = -b\sin t$, de modo que, por la~\cref{prop:b2:curves:kappaformula},
\[
\kappa(t)
= \frac{x'y'' - y'x''}{(x'^2 + y'^2)^{3/2}}
= \frac{ab\sin^2 t + ab\cos^2 t}
       {(a^2\sin^2 t + b^2\cos^2 t)^{3/2}}
= \frac{ab}{(a^2\sin^2 t + b^2\cos^2 t)^{3/2}} .
\]
El denominador es mínimo cuando $\sin t = 0$ (valor $b^3$, puntos
$(\pm a, 0)$) y máximo cuando $\cos t = 0$ (valor $a^3$, puntos $(0,
\pm b)$), pues $a > b$. Así pues, $\kappa$ es máxima en los extremos
del eje mayor, $\kappa_{\max} = a/b^2$, y mínima en los extremos del
eje menor, $\kappa_{\min} = b/a^2$: la elipse se dobla más
bruscamente en las puntas de su eje largo.
\end{solution}

\begin{solution}{exo:b2:curves:3}
Para el grafo $\gamma(x) = (x, \cosh x)$: $\norm{\gamma'(x)} =
\sqrt{1 + \sinh^2 x} = \cosh x$, luego la \omterm{def:b2:curves:length}{longitud de arco} de $0$ a
$x$ es $\int_0^x \cosh u\,\dd u = \sinh x$. \omterm{thm:b2:curves:frenet2d}{Curvatura} de un grafo
(\cref{prop:b2:curves:kappaformula}):
\[
\kappa(x) = \frac{f''(x)}{(1 + f'(x)^2)^{3/2}}
= \frac{\cosh x}{\cosh^3 x} = \frac{1}{\cosh^2 x} ,
\]
así que $R(x) = \cosh^2 x$, como se anunciaba. Obsérvese la
coincidencia elegante $R(x) = 1 + s(x)^2$ con $s = \sinh x$ la
\omterm{def:b2:curves:length}{longitud de arco}: el \omterm{def:b2:curves:curvature}{radio de curvatura} de la catenaria crece con
el cuadrado de la \omterm{def:b2:curves:length}{longitud de arco} desde el vértice.
\end{solution}

\begin{solution}{exo:b2:curves:4}
$\gamma'(t) = e^t(\cos t - \sin t,\ \sin t + \cos t)$, luego
\[
\langle \gamma(t), \gamma'(t)\rangle = e^{2t}
\bigl(\cos t(\cos t - \sin t) + \sin t(\sin t + \cos t)\bigr)
= e^{2t},
\]
mientras que $\norm{\gamma(t)} = e^t$ y $\norm{\gamma'(t)} =
e^t\sqrt 2$. Por tanto,
\[
\cos\angle\bigl(\gamma, \gamma'\bigr)
= \frac{e^{2t}}{e^t \cdot e^t\sqrt2} = \frac{1}{\sqrt2} :
\]
la tangente forma siempre un ángulo $\pi/4$ con el radio; la
propiedad \emph{equiangular} de la espiral logarítmica. \omterm{def:b2:curves:length}{Longitud de
arco} sobre $(-\infty, 0]$:
\[
\int_{-\infty}^0 \norm{\gamma'(t)}\,\dd t
= \sqrt2\int_{-\infty}^0 e^t\,\dd t = \sqrt 2 ,
\]
finita aunque la espiral dé infinitas vueltas alrededor del origen.
\omterm{thm:b2:curves:frenet2d}{Curvatura}: con $x'y'' - y'x''$ calculado a partir de $\gamma'' =
e^t(-2\sin t,\ 2\cos t)$,
\[
x'y'' - y'x'' = e^{2t}\bigl(2\cos t(\cos t - \sin t)
+ 2\sin t(\sin t + \cos t)\bigr) = 2e^{2t},
\]
luego $\kappa(t) = \dfrac{2e^{2t}}{(e^t\sqrt2)^3} =
\dfrac{1}{e^t\sqrt2}$: la \omterm{thm:b2:curves:frenet2d}{curvatura} es $1/(\sqrt2\,
\norm{\gamma})$, y decae a medida que la espiral crece.
\end{solution}

\begin{solution}{exo:b2:curves:5}
\emph{Primer arco:} $\gamma(t) = (t^2,\ t^4 + t^5)$. Derivadas en
$0$: $\gamma'' = (2, 0) \neq 0$, luego $p = 2$. Después,
$\gamma^{(3)}(0) = (0, 0)$ y $\gamma^{(4)}(0) = (0, 24)$, no
colineal con $(2, 0)$: $q = 4$. Ambos pares: \emph{cúspide de segunda
especie}; las dos ramas salen en la dirección $+u = (1,0)$ y se
quedan al mismo lado de la tangente. (En efecto, $y = x^2 \pm
x^{5/2}$ en las dos ramas: mismo signo para $x$ pequeño.)

\emph{Segundo arco:} $\gamma(t) = (t^3, t^4)$. $\gamma'(0) =
\gamma''(0) = 0$ y $\gamma^{(3)}(0) = (6, 0)$: $p = 3$, impar.
Después, $\gamma^{(4)}(0) = (0, 24)$: $q = 4$, par. Impar-par:
\emph{punto ordinario}; pese a anularse la velocidad, la trayectoria
$y = x^{4/3}$ cruza el origen suavemente y se queda por encima de su
tangente $y = 0$.
\end{solution}

\begin{solution}{exo:b2:curves:6}
Derívese $c(s) = \gamma(s) + \dfrac{1}{\kappa(s)}N(s)$ usando las
fórmulas de Frenet en el plano (\cref{thm:b2:curves:frenet2d}):
\[
c'(s) = T(s) - \frac{\kappa'(s)}{\kappa(s)^2}N(s)
+ \frac{1}{\kappa(s)}\,\bigl(-\kappa(s)T(s)\bigr)
= -\frac{\kappa'(s)}{\kappa(s)^2}\,N(s) ,
\]
pues los términos tangenciales se cancelan exactamente. Así pues, la
velocidad de la evoluta la lleva $N(s)$, que dirige la recta normal
de $\gamma$ en $\gamma(s)$; y el punto $c(s)$ está sobre esa misma
recta normal: la evoluta es la \emph{\omterm{pb:b2:curves:1}{envolvente} de las normales}.
(Donde $\kappa' = 0$, la evoluta tiene un punto singular; esto es lo
que produce las cúspides de la evoluta de una elipse.)
\end{solution}

\begin{solution}{exo:b2:curves:7}
Sea $G(s) = F(s)F(s)^{\mathsf T}$. Entonces, usando $F' = FA$ y
$(F^{\mathsf T})' = (F')^{\mathsf T} = A^{\mathsf T}F^{\mathsf T}$,
\[
G' = F'F^{\mathsf T} + F(F^{\mathsf T})'
= FAF^{\mathsf T} + FA^{\mathsf T}F^{\mathsf T}
= F(A + A^{\mathsf T})F^{\mathsf T} = 0
\]
por la antisimetría. Así pues, $G$ es constante sobre el intervalo e
igual a $G(s_0) = F(s_0)F(s_0)^{\mathsf T} = I$: $F(s)$ es ortogonal
para todo $s$. (Alternativamente, sin calcular que $G'$ es nula:
tanto $G$ como la constante $I$ resuelven el sistema lineal $Y' = YA
+ A^{\mathsf T}Y$ con el mismo valor inicial, y la unicidad de
Cauchy--Lipschitz para sistemas lineales, \cref{ch:b2:diffeq},
fuerza $G \equiv I$.)

\emph{Relevancia:} el sistema de Frenet $(T, N, B)' = (T, N,
B)\,A(s)$ tiene la matriz de coeficientes antisimétrica
\[
A = \begin{pmatrix} 0 & -\kappa & 0\\ \kappa & 0 & -\tau\\
0 & \tau & 0\end{pmatrix}
\]
(cuyas columnas expresan $T', N', B'$). El cálculo anterior muestra
que un triedro solución que empieza siendo ortonormal \emph{sigue}
siéndolo: el paso clave del teorema fundamental que reconstruye una
curva a partir de $(\kappa, \tau)$.
\end{solution}

\begin{solution}{exo:b2:curves:8}
Por el~\cref{thm:b2:curves:fundamental} (parte de unicidad), hay una
función ángulo $\mathcal{C}^1$ $\varphi$ con $T(s) =
(\cos\varphi(s), \sin\varphi(s))$ y $\varphi' = \kappa$. De ahí,
\[
\int_0^L \kappa(s)\,\dd s = \varphi(L) - \varphi(0) .
\]
Como el arco es cerrado de periodo $L$, $T(L) = T(0)$:
$(\cos\varphi(L), \sin\varphi(L)) = (\cos\varphi(0),
\sin\varphi(0))$, luego $\varphi(L) - \varphi(0) \in 2\pi\Z$. La
\omterm{thm:b2:curves:frenet2d}{curvatura} total de una curva cerrada es, por tanto, siempre un
múltiplo entero de $2\pi$; el entero es el \emph{índice de giro} de
la tangente (el número de vueltas completas que da $T$). Para la
circunferencia de radio $R$: $\kappa = 1/R$ y $L = 2\pi R$,
\omterm{thm:b2:curves:frenet2d}{curvatura} total $2\pi$ e índice $1$; el teorema de las tangentes
giratorias afirma que ese valor vale para toda curva cerrada simple.
\end{solution}

\begin{solution}{exo:b2:curves:9}
Como $\tau \equiv 0$, la curva está en un plano
(\cref{prop:b2:curves:torsion}); trabajemos en ese plano.
Consideremos el centro candidato
\[
c(s) = \gamma(s) + \frac{1}{\kappa}N(s)
\qquad (\kappa \text{ constante}).
\]
Derivando con las fórmulas de Frenet ($N' = -\kappa T + \tau B =
-\kappa T$ aquí):
\[
c'(s) = T + \frac1\kappa(-\kappa T) = 0 ,
\]
de modo que $c$ es un punto constante $\Omega$. Entonces
$\norm{\gamma(s) - \Omega} = \norm{-\frac1\kappa N(s)} =
\frac1\kappa$ para todo $s$: la curva está sobre la circunferencia
de centro $\Omega$ y radio $1/\kappa$ (en su plano) y, al ser un arco
no constante de ella, es un arco de esa circunferencia.
\end{solution}

\begin{solution}{exo:b2:curves:10}
Para el grafo $\gamma(x) = (x, x^2/2)$, $\norm{\gamma'(x)} = \sqrt{1
+ x^2}$, luego $L = \int_0^a\sqrt{1+x^2}\,\dd x$. Sustituyendo $x =
\sinh u$ ($\dd x = \cosh u\,\dd u$, con $u$ de $0$ a $u_a = \ln(a +
\sqrt{1+a^2})$):
\[
L = \int_0^{u_a}\cosh^2 u\,\dd u
= \frac12\bigl[u + \sinh u\cosh u\bigr]_0^{u_a}
= \frac12\Bigl(\ln\bigl(a + \sqrt{1+a^2}\bigr) +
a\sqrt{1+a^2}\Bigr),
\]
usando $\cosh^2 u = \frac{1 + \cosh 2u}2$, $\sinh u_a = a$ y $\cosh
u_a = \sqrt{1 + a^2}$.
\end{solution}

\begin{solution}{exo:b2:curves:11}
Parametrícese por \omterm{def:b2:curves:length}{longitud de arco}
(\cref{thm:b2:curves:arclength}) y póngase $\lambda(s) = \langle P -
\gamma(s), T(s)\rangle$, una función $\mathcal C^1$; como $P$ está
sobre la \omterm{def:b2:curves:arc}{recta tangente} en $\gamma(s)$, el vector $P - \gamma(s)$ es
colineal con $T(s)$, luego $P = \gamma(s) + \lambda(s)T(s)$.
Derivando,
\[
0 = T(s) + \lambda'(s)T(s) + \lambda(s)T'(s)
= \bigl(1 + \lambda'(s)\bigr)T(s) + \lambda(s)T'(s),
\]
y $T'(s) \perp T(s)$ (derívese $\norm T^2 = 1$), así que ambas
componentes se anulan: $\lambda' = -1$ y $\lambda T' = 0$. Entonces
$\lambda(s) = c - s$ se anula a lo sumo una vez, luego $T' = 0$ sobre
un conjunto denso y, por \omterm{def:b2:metric:continuity}{continuidad}, en todas partes: $T$ es un
vector \omterm{def:b2:hermitian:adjoint}{unitario} constante y $\gamma(s) = \gamma(s_0) + (s - s_0)T$:
una recta (que pasa por $P$, como debe ser).
\end{solution}

\begin{solution}{exo:b2:curves:12}
(a) La matriz de Frenet
\[
A(s) = \begin{pmatrix} 0 & -\kappa & 0\\ \kappa & 0 & -\tau\\
0 & \tau & 0\end{pmatrix}
\]
tiene entradas \omterm{def:b2:metric:continuity}{continuas}, de modo que el sistema lineal $F' =
FA(s)$, $F(s_0) = F_0$ (una matriz ortonormal directa) tiene una
única solución sobre todo $J$ (\cref{thm:b2:diffeq:linear}). Por el~\cref{exo:b2:curves:7}, $F(s)$ es ortogonal para todo $s$; y $\det F$
es \omterm{def:b2:metric:continuity}{continuo} con valores en $\{\pm1\}$ y vale $1$ en $s_0$, luego
$F(s)$ es directa para todo $s$.

(b) Léanse las filas $T, N, B$ de $F$ (de modo que $T' = \kappa N$,
$N' = -\kappa T + \tau B$, $B' = -\tau N$) y póngase $\gamma(s) =
\gamma_0 + \int_{s_0}^s T(u)\,\dd u$. Entonces $\gamma' = T$ es un
vector \omterm{def:b2:hermitian:adjoint}{unitario}: rapidez unidad; $T' = \kappa N$ con $\kappa > 0$ y
$N$ \omterm{def:b2:hermitian:adjoint}{unitario} ortogonal a $T$, de modo que $\gamma$ es birregular con
\omterm{thm:b2:curves:frenet2d}{curvatura} $\norm{T'} = \kappa$ y normal principal $N$; la binormal
es $T \wedge N = B$ (triedro ortonormal directo), y $B' = -\tau N$
identifica la \omterm{thm:b2:curves:frenet3d}{torsión} como $\tau$.

(c) Sean $\gamma_1, \gamma_2$ curvas birregulares de rapidez unidad
con los mismos $(\kappa, \tau)$. Hay una única isometría directa
$\Phi = \rho + w$ ($\rho \in SO(3)$) que lleva $\gamma_1(s_0)$ a
$\gamma_2(s_0)$ y el triedro de Frenet de $\gamma_1$ en $s_0$ al de
$\gamma_2$ en $s_0$. La curva $\Phi\circ\gamma_1$ es de rapidez
unidad y tiene los mismos invariantes (su triedro es $\rho$ aplicado
al de $\gamma_1$, y $\rho$ conserva los productos vectoriales por ser
directa). Ahora bien, los triedros de $\Phi\circ\gamma_1$ y de
$\gamma_2$ resuelven ambos $F' = FA(s)$ con el mismo valor inicial,
así que coinciden por unicidad; en particular, las tangentes
coinciden y, integrando desde el punto común $s_0$:
$\Phi\circ\gamma_1 = \gamma_2$.
\end{solution}

\begin{solution}{pb:b2:curves:1}
\textbf{1.} El sistema es lineal en $(x, y)$ con \omterm{def:b2:linalg:det}{determinante}
$\Delta(t) = a b' - a'b \neq 0$: la regla de Cramer da la única
solución
\[
x = \frac{c b' - c' b}{\Delta}, \qquad
y = \frac{a c' - a' c}{\Delta}.
\]

\textbf{2.} Como $a(t)x(t) + b(t)y(t) = c(t)$ idénticamente, derivar
da $a'x + b'y + ax' + by' = c'$; la segunda ecuación característica
mata $a'x + b'y - c'$, luego $a(t)x'(t) + b(t)y'(t) = 0$: $E'(t)$ es
ortogonal a $(a, b)$ y, por tanto, paralelo a $(-b, a)$, la dirección
de $D_t$. Como $E(t) \in D_t$ (primera ecuación) y $E'(t) \neq 0$, la
recta $D_t$ es exactamente la \omterm{def:b2:curves:arc}{recta tangente} de la curva $E$ en
$E(t)$.

\textbf{3.} Aquí $(a, b, c) = (t, -1, t^2/2)$, luego $\Delta =
t\cdot0 - 1\cdot(-1) = 1$ y
\[
x = \frac{c b' - c' b}{\Delta} = \tfrac{t^2}2\cdot 0 +
t = t, \qquad
y = \frac{a c' - a' c}{\Delta} = t\cdot t - \tfrac{t^2}2 =
\tfrac{t^2}2 :
\]
la \omterm{pb:b2:curves:1}{envolvente} de las \omterm{def:b2:curves:arc}{rectas tangentes} de la parábola es la parábola,
como debía ser.

\textbf{4.} La recta normal es $\langle M, T(s)\rangle =
\langle\gamma(s), T(s)\rangle$: coeficientes $a = T_1$, $b = T_2$, $c
= \langle\gamma, T\rangle$. Derivando con Frenet ($T' = \kappa N$):
$a' = \kappa N_1$, $b' = \kappa N_2$ y $c' = \langle T, T\rangle +
\langle\gamma, \kappa N\rangle = 1 + \kappa\langle\gamma, N\rangle$.
La segunda ecuación característica $\kappa\langle M, N\rangle = 1 +
\kappa\langle\gamma, N\rangle$ se lee $\kappa\langle M - \gamma,
N\rangle = 1$. La primera dice que $M - \gamma \perp T$, así que $M -
\gamma = \mu N$ con $\mu = 1/\kappa$: el punto característico es
$\gamma + \frac1\kappa N$, el \omterm{def:b2:curves:curvature}{centro de curvatura}, y la
\omterm{pb:b2:curves:1}{envolvente} de las normales es la evoluta del~\cref{exo:b2:curves:6}. Por último, $\Delta = T_1\kappa N_2 - \kappa
N_1 T_2 = \kappa\det(T, N) = \kappa \neq 0$.

\textbf{5.} Haz: el sistema $x\cos\theta + y\sin\theta = 0$,
$-x\sin\theta + y\cos\theta = 0$ tiene \omterm{def:b2:linalg:det}{determinante} $1$ y solución
$(0,0)$ para todo $\theta$: $E \equiv (0,0)$, $E' \equiv 0$, y no hay
curva alguna, sino el mero punto común de todas las rectas. Familia
paralela: $a' = b' = 0$ da $\Delta \equiv 0$ y la segunda ecuación $0
= c'(t)$, que falla en cuanto la familia se mueve de verdad: no hay
punto característico y, en efecto, una familia de rectas paralelas no
toca ninguna curva a lo largo de todos sus miembros.

\textbf{6.} La recta que pasa por $(\cos t, 0)$ y $(0, \sin t)$ es
$\frac x{\cos t} + \frac y{\sin t} = 1$, es decir, $x\sin t + y\cos t
= \sin t\cos t$. Con $(a, b, c) = (\sin t, \cos t, \sin t\cos t)$:
$a' = \cos t$, $b' = -\sin t$, $c' = \cos 2t$, $\Delta = -\sin^2 t -
\cos^2 t = -1$. Cramer:
\begin{align*}
x &= \frac{cb' - c'b}{-1} = \sin^2 t\cos t + \cos 2t\cos t
= \cos t\,(\sin^2 t + \cos^2 t - \sin^2 t) = \cos^3 t,\\
y &= \frac{ac' - a'c}{-1} = \sin t\cos^2 t - \sin t\cos 2t
= \sin t\,(\cos^2 t - \cos^2 t + \sin^2 t) = \sin^3 t .
\end{align*}
Y $(\cos^3t)^{2/3} + (\sin^3t)^{2/3} = 1$: la \omterm{pb:b2:curves:1}{astroide}.

\textbf{7.} $E'(t) = 3(-\cos^2 t\sin t,\ \sin^2 t\cos t)$ se anula en
$t = 0$. Allí, $E''(0) = (-3, 0) \neq 0$ da $p = 2$; la componente
$x$ de $E$ es par en $t$, así que $E'''(0) = (0, 6)$, no colineal: $q
= 3$. Par-impar: cúspide de primera especie en $(1, 0)$
(\cref{prop:b2:curves:local}), con tangente a lo largo del eje $x$.
Las simetrías $x \mapsto -x$, $y \mapsto -y$ y $(x, y) \mapsto (y,
x)$ de la \omterm{pb:b2:curves:1}{astroide} transportan la cúspide a $(-1, 0)$ y a $(0,
\pm1)$.

\textbf{8.} $E'(t) = 3\sin t\cos t\,(-\cos t, \sin t)$, luego
$\norm{E'(t)} = 3\abs{\sin t\cos t} = \tfrac32\abs{\sin 2t}$. Un
cuadrante: $\int_0^{\pi/2}\tfrac32\sin 2t\,\dd t = \tfrac32$, y por
simetría la \omterm{def:b2:curves:length}{longitud} total es $4 \cdot \tfrac32 = 6$.

\textbf{9.} $E(t) - P_t = (\cos^3 t - \cos t,\ \sin^3 t) = \sin^2
t\,(-\cos t,\ \sin t) = \sin^2 t\,(Q_t - P_t)$. Así pues, el punto de
contacto es el \omterm{def:b2:affine:barycenter}{baricentro} de $(P_t, \cos^2 t)$ y $(Q_t, \sin^2 t)$:
cuando $t$ recorre de $0$ a $\pi/2$, se desliza del extremo del suelo
al extremo de la pared de la escalera.

\textbf{10.} En el primer cuadrante, la región bajo la \omterm{pb:b2:curves:1}{astroide} tiene
área $\int_0^1 y\,\dd x$, con $x = \cos^3 t$ decreciendo de $1$ a $0$
cuando $t$ va de $0$ a $\pi/2$:
\[
\int_0^1 y\,\dd x
= \int_{\pi/2}^{0}\sin^3 t\,(-3\cos^2 t\sin t)\,\dd t
= 3\int_0^{\pi/2}\sin^4 t\cos^2 t\,\dd t .
\]
Linealícese: $\sin^4 t\cos^2 t = (\sin t\cos t)^2\sin^2 t =
\tfrac18\bigl(\sin^2 2t - \sin^2 2t\cos 2t\bigr)$, y
$\int_0^{\pi/2}\sin^2 2t\,\dd t = \tfrac\pi4$, mientras que
$\int_0^{\pi/2}\sin^2 2t\cos 2t\,\dd t = \bigl[\tfrac{\sin^3
2t}6\bigr]_0^{\pi/2} = 0$. La integral vale, pues, $\tfrac\pi{32}$,
el área del cuadrante $\tfrac{3\pi}{32}$ y el área encerrada $4 \cdot
\tfrac{3\pi}{32} = \tfrac{3\pi}8$.

\textbf{11.} La tangente en $\gamma(t) = (t, t^2/2)$ tiene dirección
$(1, t)$, de modo que la recta normal es $\{(x, y) : (x - t) + t\,(y
- t^2/2) = 0\}$, es decir, $x + t\,y = t + \tfrac{t^3}2$.

\textbf{12.} $(a, b, c) = (1, t, t + t^3/2)$: $a' = 0$, $b' = 1$, $c'
= 1 + \tfrac32 t^2$, $\Delta = 1$. Entonces $y = ac' - a'c = 1 +
\tfrac32t^2$ y
\[
x = cb' - c'b = t + \tfrac{t^3}2 - t\Bigl(1 +
\tfrac32t^2\Bigr) = -t^3 .
\]
Eliminando $t$: $t^2 = \tfrac23(y - 1)$ y $x^2 = t^6 = \tfrac8{27}(y
- 1)^3$: una parábola semicúbica de vértice $(0, 1)$.

\textbf{13.} $T = (1, t)/\sqrt{1+t^2}$, $N = (-t, 1)/\sqrt{1+t^2}$ y
$\kappa = (1+t^2)^{-3/2}$, luego
\[
\gamma + \frac1\kappa N = (t, \tfrac{t^2}2) +
(1+t^2)\,(-t, 1) = \Bigl(-t^3,\ 1 + \tfrac32t^2\Bigr),
\]
la misma curva: la \omterm{pb:b2:curves:1}{envolvente} de las normales es el lugar de los
\omterm{def:b2:curves:curvature}{centros de curvatura}, como prometía la pregunta 4.

\textbf{14.} $E'(t) = (-3t^2, 3t)$ se anula en $t = 0$; $E''(0) = (0,
3) \neq 0$ da $p = 2$, y $E'''(0) = (-6, 0)$ da $q = 3$: una cúspide
de primera especie en $(0, 1)$. El vértice es donde $\kappa =
(1+t^2)^{-3/2}$ es máxima, así que $\kappa'(0) = 0$ y la velocidad de
la evoluta $-\frac{\kappa'}{\kappa^2}N$ se anula justo ahí: las
cúspides de la evoluta se sitúan en los extremos de la \omterm{thm:b2:curves:frenet2d}{curvatura}.

\textbf{15.} La normal de parámetro $t$ pasa por $(x_0, y_0)$ si y
solo si $x_0 + t\,y_0 = t + \tfrac{t^3}2$, es decir,
\[
\frac{t^3}2 + (1 - y_0)\,t - x_0 = 0 ,
\]
una cúbica en $t$: una o tres raíces reales (contadas sin
multiplicidad, para puntos genéricos). Sobre el eje $x_0 = 0$ se
factoriza como $t\bigl(\tfrac{t^2}2 + 1 - y_0\bigr) = 0$: la raíz $t
= 0$ (el eje es la normal en el vértice), más $t = \pm\sqrt{2(y_0 -
1)}$ cuando $y_0 > 1$. Así pues: una normal para $y_0 < 1$, tres para
$y_0 > 1$, y en $y_0 = 1$ la raíz triple marca la cúspide de la
evoluta. En general, una raíz doble de la cúbica significa que el
punto satisface a la vez la ecuación de la recta y su derivada
respecto de $t$: está \emph{sobre la \omterm{pb:b2:curves:1}{envolvente}}; la evoluta es
precisamente la frontera entre las regiones de una y de tres
normales.

\textbf{16.} La reflexión en el espejo invierte la componente normal
de la dirección y conserva la tangencial: escribiendo $u = \langle u,
n\rangle n + u_{\mathrm{tan}}$, la dirección reflejada es
$u_{\mathrm{tan}} - \langle u, n\rangle n = u - 2\langle u, n\rangle
n$. Aquí $\langle u, n\rangle = \cos\theta$, luego
\[
v = (1, 0) - 2\cos\theta\,(\cos\theta, \sin\theta)
= (1 - 2\cos^2\theta,\ -2\sin\theta\cos\theta)
= -(\cos2\theta,\ \sin2\theta).
\]

\textbf{17.} El rayo reflejado pasa por $P_\theta = (\cos\theta,
\sin\theta)$ con dirección $(\cos2\theta, \sin2\theta)$; un vector
normal es $(-\sin2\theta, \cos2\theta)$, de modo que la recta es
\[
-\sin2\theta\,(x - \cos\theta) + \cos2\theta\,(y -
\sin\theta) = 0,
\]
y la constante es $-\sin2\theta\cos\theta + \cos2\theta\sin\theta =
-\sin\theta$: multiplicando por $-1$, $x\sin2\theta - y\cos2\theta =
\sin\theta$.

\textbf{18.} $(a, b, c) = (\sin2\theta, -\cos2\theta, \sin\theta)$:
$a' = 2\cos2\theta$, $b' = 2\sin2\theta$, $c' = \cos\theta$, $\Delta
= 2\sin^22\theta + 2\cos^22\theta = 2$. Cramer y después las fórmulas
de producto a suma:
\begin{align*}
x &= \frac{2\sin\theta\sin2\theta +
\cos\theta\cos2\theta}{2}
= \frac{(\cos\theta - \cos3\theta) + \frac12(\cos\theta +
\cos3\theta)}{2} = \frac{3\cos\theta - \cos3\theta}4,\\
y &= \frac{\sin2\theta\cos\theta - 2\cos2\theta\sin\theta}2
= \frac{\frac12(\sin3\theta + \sin\theta) - (\sin3\theta -
\sin\theta)}2 = \frac{3\sin\theta - \sin3\theta}4 :
\end{align*}
la \omterm{pb:b2:curves:1}{nefroide}, una curva cerrada con dos cúspides.

\textbf{19.} Derivando y factorizando con $\sin3\theta - \sin\theta =
2\cos2\theta\sin\theta$ y $\cos\theta - \cos3\theta =
2\sin2\theta\sin\theta$:
\[
E'(\theta) = \tfrac34\bigl(\sin3\theta - \sin\theta,\
\cos\theta - \cos3\theta\bigr)
= \tfrac32\sin\theta\,(\cos2\theta, \sin2\theta),
\]
paralelo a la dirección reflejada de la pregunta 16: cada rayo
reflejado es tangente a la cáustica, como exige la propiedad de
\omterm{pb:b2:curves:1}{envolvente}. $E' = 0$ exactamente cuando $\sin\theta = 0$: $E(0) =
(\tfrac12, 0)$ y $E(\pi) = (-\tfrac12, 0)$, las dos cúspides.
Poniendo $y = 0$ en la ecuación de la recta: $x\sin2\theta =
\sin\theta$, luego $x = \frac1{2\cos\theta} \to \frac12$ cuando
$\theta \to 0$: los rayos paraxiales enfocan a distancia $R/2$ del
centro, la distancia focal del espejo esférico.

\textbf{20.} $\norm{E'(\theta)} = \tfrac32\abs{\sin\theta}$, así que
la \omterm{def:b2:curves:length}{longitud} es $\tfrac32\int_0^{2\pi}\abs{\sin\theta}\, \dd\theta =
\tfrac32\cdot4 = 6$. Cerca de $\theta = 0$, con $\cos k\theta = 1 -
\tfrac{k^2\theta^2}2 + O(\theta^4)$ y $\sin k\theta = k\theta -
\tfrac{k^3\theta^3}6 + O(\theta^5)$:
\[
x - \tfrac12 = \frac{3\theta^2 + O(\theta^4)}{4}
= \tfrac34\theta^2 + O(\theta^4),
\qquad
y = \frac{4\theta^3 + O(\theta^5)}{4}
= \theta^3 + O(\theta^5) :
\]
$p = 2$, $q = 3$, una cúspide de primera especie apuntando a lo largo
del eje: el punto brillante de la cáustica de la taza de café.

\textbf{21.} Parametrícense los puntos iluminados mediante
$\Phi(\theta, r) = P_\theta + r\,v_\theta$ (la posición a lo largo de
cada rayo reflejado). El \omterm{def:b2:linalg:det}{determinante} jacobiano
$\det(\partial_\theta\Phi, \partial_r\Phi) = \det(P_\theta' +
rv_\theta', v_\theta)$ es \omterm{def:b2:affine:subspace}{afín} en $r$ y se anula para exactamente un
$r = r_*(\theta)$; y $\Phi(\theta, r_*(\theta))$ es el punto
característico, pues ahí la dirección del rayo y la variación de la
familia se vuelven dependientes. Fuera de la \omterm{pb:b2:curves:1}{envolvente}, la
aplicación es un difeomorfismo local, y un punto justo dentro de la
cáustica recibe dos rayos próximos (dos soluciones $\theta$), y uno
de fuera, ninguno de esa parte de la familia; sobre la cáustica, los
dos se funden. La intensidad luminosa es inversamente proporcional al
valor absoluto del jacobiano, de modo que estalla a lo largo de la
\omterm{pb:b2:curves:1}{envolvente}: la cáustica es la curva brillante, y lo es aún más en la
cúspide, donde la degeneración es peor.

\textbf{22.} $x' = 1 - \cos t$, $y' = \sin t$, $x'' = \sin t$, $y'' =
\cos t$, luego $x'y'' - y'x'' = \cos t - 1$ y $\norm{\gamma'}^2 = 2(1
- \cos t) = 4\sin^2\tfrac t2$; por la~\cref{prop:b2:curves:kappaformula},
\[
\kappa(t) = \frac{-(1 - \cos t)}{8\sin^3\tfrac t2}
= \frac{-2\sin^2\tfrac t2}{8\sin^3\tfrac t2}
= -\frac1{4\sin\tfrac t2}
\qquad (0 < t < 2\pi).
\]
Con $T = (\sin\tfrac t2, \cos\tfrac t2)$ (divídase $\gamma'$ entre
$2\sin\tfrac t2$) y $N = (-\cos\tfrac t2, \sin\tfrac t2)$:
\[
\gamma + \frac1\kappa N
= \gamma - 4\sin\tfrac t2\,\Bigl(-\cos\tfrac t2,\
\sin\tfrac t2\Bigr)
= \bigl(t - \sin t + 2\sin t,\ 1 - \cos t - 2(1 - \cos
t)\bigr),
\]
es decir, $c(t) = (t + \sin t,\ \cos t - 1)$. Sustituyendo $t = u +
\pi$:
\[
c = \bigl(u + \pi - \sin u,\ -\cos u - 1\bigr)
= \bigl((u - \sin u) + \pi,\ (1 - \cos u) - 2\bigr) :
\]
la cicloide $\gamma(u)$ trasladada por $(\pi, -2)$. La evoluta de una
cicloide es una cicloide congruente, colgada un nivel más abajo.

\textbf{23.} $R(t) = 1/\abs{\kappa(t)} = 4\sin\tfrac t2$, así que
$R(\pi) = 4$: la mitad de la \omterm{def:b2:curves:length}{longitud} $8$ del arco calculada en el~\cref{exo:b2:curves:1}. La velocidad de la evoluta $c'(t) = (1 +
\cos t, -\sin t)$ se anula en $t = \pi$: la cúspide es $c(\pi) =
(\pi, -2)$, directamente bajo el ápice $\gamma(\pi) = (\pi, 2)$, a
distancia $4 = R(\pi)$, exactamente la \omterm{def:b2:curves:length}{longitud} del radio osculador
allí.

\textbf{24.} Por el~\cref{exo:b2:curves:6}, $c'(s) =
-\frac{\kappa'(s)}{\kappa(s)^2}N(s) = R'(s)\,N(s)$, luego
$\norm{c'(s)} = \abs{R'(s)}$ y, para $R$ monótona,
\[
\int_{s_0}^{s_1}\norm{c'(s)}\,\dd s =
\Bigl|\int_{s_0}^{s_1}R'(s)\,\dd s\Bigr| = \abs{R(s_1) -
R(s_0)} .
\]
Digamos que $R$ decrece. Un hilo tendido a lo largo de la evoluta más
allá de $c(s_0)$ y prolongado por el segmento de $c(s_0)$ a
$\gamma(s_0)$ (tangente a la evoluta, por la pregunta 4) tiene, al
despegarlo hasta $c(s)$ y tensarlo, una parte recta de \omterm{def:b2:curves:length}{longitud}
$R(s_0) - \bigl(R(s_0) - R(s)\bigr) = R(s)$ que apunta desde $c(s)$ a
lo largo de la normal, y aterriza exactamente en $\gamma(s)$: el
extremo libre traza la curva original (la ``involuta''). Huygens
colgó un péndulo entre dos mejillas cicloidales: la cuerda se enrolla
sobre la evoluta, así que la lenteja describe una cicloide, la
tautócrona, cuyo periodo de oscilación no depende de la amplitud.

\textbf{25.} Tangentes de la parábola: $\Delta = 1$ y $E' = (1, t)
\neq 0$ en todas partes; \omterm{pb:b2:curves:1}{envolvente} regular (la propia parábola).
Escalera que resbala: $\Delta = -1$, pero $E' = \tfrac32\sin 2t\,
(-\cos t, \sin t)$ se anula en los extremos de los cuadrantes: las
cuatro cúspides de la \omterm{pb:b2:curves:1}{astroide}. Normales de la parábola y de la
cicloide: $\Delta = \kappa \neq 0$, y $E' = R'N$ se anula exactamente
donde la \omterm{thm:b2:curves:frenet2d}{curvatura} es extremal: las cúspides de las evolutas en
$(0,1)$ y $(\pi, -2)$. Cáustica: $\Delta = 2$, y $E' =
\tfrac32\sin\theta\,(\cos2\theta, \sin2\theta)$ se anula en $\theta =
0, \pi$: las dos cúspides de la \omterm{pb:b2:curves:1}{nefroide}, los puntos focales del
espejo. Los extremos de la \omterm{thm:b2:curves:frenet2d}{curvatura} \emph{tienen} que producir
cúspides en la \omterm{pb:b2:curves:1}{envolvente} de las normales, pues la velocidad de la
evoluta es $R'N$: por eso la evoluta de la elipse tiene cuatro
cúspides (cuatro vértices), y las familias degeneradas (haz,
paralelas) son los casos en que la máquina devuelve un punto o
directamente nada.
\end{solution}
