\section*{الفصل \ref{ch:b2:integration} --- التكامل}

\begin{solution}{exo:b2:integration:1}
$\int_0^1 \frac{\dd t}{\sqrt{t(1-t)}}$: قرب $0$، $\sim t^{-1/2}$
(مع $\alpha = \frac12 < 1$: فيتقارب)؛ وقرب $1$، $\sim
(1-t)^{-1/2}$: فيتقارب. فهو متقارب (وقيمته $\pi$،
بالتعويض $t = \sin^2\theta$).

$\int_1^\infty \frac{\ln t}{t^2}$: $\frac{\ln t}{t^2} = o(t^{-3/2})$:
فهو متقارب (وقيمته $1$ بالتجزئة).

$\int_0^\infty \frac{\dd t}{1 + t^2\sin^2 t}$: متباعد. فقرب $t =
n\pi$، نكتب $t = n\pi + u$: $\sin^2 t = \sin^2 u \leq u^2$، ومن ثم على
$\abs u \leq \frac{1}{n}$، $1 + t^2\sin^2 t \leq 1 + (n\pi +
1)^2u^2 \leq C n^2 u^2 + 1$؛ ومنه
\[
\int_{n\pi - 1/n}^{n\pi + 1/n} \frac{\dd t}{1 + t^2\sin^2 t}
\geq \int_{-1/n}^{1/n} \frac{\dd u}{1 + Cn^2u^2}
= \frac{2\arctan\sqrt C}{\sqrt C}\cdot\frac{1}{n} ,
\]
وهو حدّ من متسلسلة متباعدة من النمط التوافقي: فبالجمع على $n$، تكون
الدالة الأصلية غير محدودة.
\end{solution}

\begin{solution}{exo:b2:integration:2}
نعوّض $u = \lambda t$:
\[
\int_0^\infty t^n \eu^{-\lambda t}\dd t
= \frac{1}{\lambda^{n+1}}\int_0^\infty u^n\eu^{-u}\dd u
= \frac{\Gamma(n+1)}{\lambda^{n+1}} = \frac{n!}{\lambda^{n+1}} .
\]
ومع $t = \eu^{-u}$ (مع $\dd t = -\eu^{-u}\dd u$):
\[
\int_0^1 (\ln t)^n \dd t = \int_0^{\infty} (-u)^n \eu^{-u}\,\dd u
= (-1)^n\, n! .
\]
\end{solution}

\begin{solution}{exo:b2:integration:3}
التقارب: الدالة المكامَلة $\leq \frac{1}{1+t^2}$ قرب $\infty$
ومحدودة قرب $0$ (فالعاملان محدودان من أسفل بعيدًا عن $0$):
فالتقارب مطلق، من أجل كل $x$. وبالتعويض $t = \frac1u$
(مع $\dd t = -\frac{\dd u}{u^2}$):
\[
I(x) = \int_0^\infty \frac{1}{\bigl(1 +
\frac1{u^2}\bigr)\bigl(1 + u^{-x}\bigr)}\cdot\frac{\dd u}{u^2}
= \int_0^\infty \frac{u^x}{(1 + u^2)(1 + u^x)}\,\dd u .
\]
وبجمع العبارتين للمقدار $I(x)$:
\[
2I(x) = \int_0^\infty \frac{1 + t^x}{(1+t^2)(1+t^x)}\dd t
= \int_0^\infty \frac{\dd t}{1 + t^2} = \frac{\pi}{2} :
\]
أي $I(x) = \frac\pi4$، وهو لا يتعلق بالوسيط $x$.
\end{solution}

\begin{solution}{exo:b2:integration:4}
قرب $0^+$، مع $u = \abs{\ln t} \to \infty$. فإذا كان $\alpha < 1$:
فالتقارب مهما كان $\beta$ (بالمقارنة مع $t^{-\alpha'}$ من أجل
$\alpha < \alpha' < 1$: فيُغلَب عامل اللوغاريتم). وإذا كان $\alpha > 1$:
فالتباعد مهما كان $\beta$ (بالمقارنة مع $t^{-\alpha''}$،
مع $1 < \alpha'' < \alpha$). وإذا كان $\alpha = 1$: فنعوّض $t =
\eu^{-u}$:
\[
\int_0^{1/2} \frac{\dd t}{t\,\abs{\ln t}^\beta}
= \int_{\ln 2}^{\infty} \frac{\dd u}{u^\beta},
\]
وهو متقارب إذا وفقط إذا $\beta > 1$. والخلاصة: التقارب إذا وفقط إذا $\alpha < 1$،
أو ($\alpha = 1$ و$\beta > 1$) --- وهي مرآة متسلسلة
برتران.
\end{solution}

\begin{solution}{exo:b2:integration:5}
الاتصال على $\intco{0}{\infty}$: بالهيمنة
$\bigl|\frac{\eu^{-xt}}{1+t^2}\bigr| \leq \frac{1}{1+t^2}$،
وهي قابلة للمكاملة ومنتظمة في $x \geq 0$:
\cref{thm:b2:integration:continuity}.

من الصنف $C^2$ على $\intoo{0}{\infty}$: على $x \geq a > 0$، يكون المشتقان
الأولان بالنسبة إلى $x$، أي $\frac{-t\,\eu^{-xt}}{1+t^2}$ و
$\frac{t^2\eu^{-xt}}{1+t^2}$، مهيمَنًا عليهما بالمقدارين $t\,\eu^{-at}$ و
$\eu^{-at}$: فبتطبيقين للنتيجة
\cref{thm:b2:integration:leibnizrule}. ثم
\[
F''(x) + F(x) = \int_0^\infty \frac{t^2 + 1}{1 + t^2}\,\eu^{-xt}\dd
t = \int_0^\infty \eu^{-xt}\dd t = \frac1x .
\]
والنهاية: $0 \leq F(x) \leq \int_0^\infty \eu^{-xt}\dd t = \frac1x \to
0$.
\end{solution}

\begin{solution}{exo:b2:integration:6}
على $\intcc{\varepsilon}{M}$، نعوّض $u = at$ و$u = bt$ في
النصفين:
\[
\int_\varepsilon^M \frac{f(at) - f(bt)}{t}\dd t
= \int_{a\varepsilon}^{aM}\frac{f(u)}{u}\dd u
- \int_{b\varepsilon}^{bM}\frac{f(u)}{u}\dd u
= \int_{a\varepsilon}^{b\varepsilon} \frac{f(u)}{u}\dd u
- \int_{aM}^{bM} \frac{f(u)}{u}\dd u .
\]
القطعة الأولى: $f(u) = f(0) + o(1)$ قرب $0$، و$\int_{a\varepsilon}
^{b\varepsilon} \frac{\dd u}{u} = \ln\frac ba$: فتؤول القطعة إلى
$f(0)\ln\frac ba$. والقطعة الثانية: $f(u) \to f(\infty)$، وبالحساب نفسه:
تؤول إلى $f(\infty)\ln\frac ba$. ومن ثم يتقارب \omterm{def:b2:integration:improper}{التكامل
المعتل} إلى $\bigl(f(0) - f(\infty)\bigr)\ln\frac ba$.

ومع $f(t) = \eu^{-t}$ (مع $f(0) = 1$، $f(\infty) = 0$)، $a = 1$، $b
= 2$:
\[
\int_0^\infty \frac{\eu^{-t} - \eu^{-2t}}{t}\dd t = \ln 2 .
\]
\end{solution}

\begin{solution}{exo:b2:integration:7}
نمدّد الدالة المكامَلة بالقيمة $0$ بعد $t = n$: $g_n(t) = (1 -
\frac tn)^n t^{x-1}\mathbf{1}_{t \leq n}$. ونقطيًا، $g_n(t) \to
\eu^{-t}t^{x-1}$ (وهي نهاية الفائدة المركّبة، مجلد السنة الأولى).
والهيمنة: يعطي $\ln(1 - u) \leq -u$ أن $(1 - \frac tn)^n \leq
\eu^{-t}$ على $\intcc{0}{n}$، ومنه $\abs{g_n(t)} \leq
\eu^{-t}t^{x-1} = \varphi(t)$، وهو قابل للمكاملة. وبالتقارب المهيمَن عليه:
\[
\int_0^n \Bigl(1 - \frac tn\Bigr)^n t^{x-1}\dd t
\xrightarrow[n\to\infty]{} \int_0^\infty \eu^{-t}t^{x-1}\dd t
= \Gamma(x) .
\]
(وحساب الطرف الأيسر بتجزئات متكررة يعطي صيغة أويلر الجدائية
$\Gamma(x) = \lim \frac{n!\,n^x}{x(x+1)\cdots(x+n)}$.)
\end{solution}

\begin{solution}{exo:b2:integration:8}
$H$ قابلة للاشتقاق في $x$ (فالدالة المكامَلة من الصنف $C^1$ في $x$، ومشتقها
$-2x(1+t^2)\cdot\frac{\eu^{-x^2(1+t^2)}}{1+t^2} =
-2x\,\eu^{-x^2}\eu^{-x^2t^2}$ \omterm{def:b2:metric:continuity}{متصل} ومحدود على متراصات
$x$، والهيمنة على $t \in \intcc{0}{1}$ بديهية):
\[
H'(x) = -2x\,\eu^{-x^2}\int_0^1 \eu^{-x^2t^2}\,\dd t
\overset{u = xt}{=} -2\,\eu^{-x^2}\int_0^x \eu^{-u^2}\,\dd u
= -G'(x),
\]
لأن $G'(x) = 2\eu^{-x^2}\int_0^x \eu^{-t^2}\dd t$ (بقاعدة سلسلة الاشتقاق على
المربّع، وبالمبرهنة الأساسية للتفاضل والتكامل). ومن ثم فإن $G + H$
ثابتة، وتساوي $G(0) + H(0) = 0 + \int_0^1 \frac{\dd t}{1+t^2}
= \frac\pi4$.

وحين $x \to \infty$: $0 \leq H(x) \leq \eu^{-x^2}\int_0^1 \dd t \to
0$، ومنه $G(x) \to \frac\pi4$:
\[
\int_0^\infty \eu^{-t^2}\dd t = \sqrt{\frac\pi4} =
\frac{\sqrt\pi}{2} .
\]
(ومن ثم $\Gamma\bigl(\frac12\bigr) = 2\int_0^\infty
\eu^{-t^2}\dd t = \sqrt\pi$، بالتعويض $t = \sqrt u$.)
\end{solution}

\begin{solution}{exo:b2:integration:9}
كوشي--شوارتز (مجلد السنة الأولى، وهي صالحة على $\intcc{\varepsilon}{M}$
ومُمرَّرة إلى النهاية) مطبَّقةً على التفكيك
$t^{\frac{x+y}{2}-1}\eu^{-t} = \bigl(t^{x-1}\eu^{-t}\bigr)^{1/2}
\bigl(t^{y-1}\eu^{-t}\bigr)^{1/2}$:
\[
\Gamma\Bigl(\frac{x+y}{2}\Bigr) \leq
\Gamma(x)^{1/2}\,\Gamma(y)^{1/2}
\quad\Longrightarrow\quad
\ln\Gamma\Bigl(\frac{x+y}{2}\Bigr) \leq \frac{\ln\Gamma(x) +
\ln\Gamma(y)}{2} :
\]
ومن ثم فإن $\ln\Gamma$ محدبة عند النقطة الوسطى؛ وبما أنها متصلة
(\cref{thm:b2:integration:gammaprops})، فهي محدبة
(\cref{exo:b2:realfun:8}). (ويثبّت التحدب اللوغاريتمي $\Gamma$
تثبيتًا وحيدًا بين استكمالات العاملي --- وهي
مبرهنة بور--مولروب، وهي لؤلؤة من لآلئ السنة الثالثة.)
\end{solution}

\begin{solution}{exo:b2:integration:10}
\begin{enumerate}
  \item على $x \geq a > 0$: المشتق بالنسبة إلى $x$ للدالة المكامَلة هو
        $-\sin t\,\eu^{-xt}$، وهو مهيمَن عليه بالمقدار $\eu^{-at}$:
        فتعطي \cref{thm:b2:integration:leibnizrule} أن $F'(x) =
        -\int_0^\infty \eu^{-xt}\sin t\,\dd t$. وبمكاملتين
        بالتجزئة (أو بالأسّي العقدي):
        \[
        \int_0^\infty \eu^{-xt}\sin t\,\dd t
        = \Im \int_0^\infty \eu^{(-x+\iu)t}\dd t
        = \Im\frac{1}{x - \iu} = \frac{1}{1 + x^2} .
        \]
  \item $\abs{F(x)} \leq \int_0^\infty \eu^{-xt}\dd t = \frac1x \to
        0$. وبمكاملة $F' = -\frac{1}{1+x^2}$ من $x$ إلى
        $\infty$: $0 - F(x) = -\bigl(\frac\pi2 - \arctan x\bigr)$،
        ومنه $F(x) = \frac\pi2 - \arctan x$.
  \item وبأخذ $x \to 0^+$ مع الاتصال المقبول: $F(0^+) =
        \frac\pi2$، و$F(0) = \int_0^\infty \frac{\sin t}{t}\dd
        t$ (وهو تكامل ديريكليه النصف متقارب،
        \cref{ex:b2:integration:sint}): فقيمته
        $\frac\pi2$.
\end{enumerate}
\end{solution}

\begin{solution}{exo:b2:integration:11}
التقارب: قرب $0$ تمتدّ الدالة المكامَلة بالاتصال بالقيمة
$1$ (لأن $\sin t \sim t$)؛ وعند اللانهاية تكون $\leq t^{-2}$:
فالتقارب مطلق. وعلى $\intcc{\varepsilon}{M}$، نكامل
بالتجزئة مع $u = \sin^2 t$، $v' = t^{-2}$:
\[
\int_\varepsilon^M \frac{\sin^2 t}{t^2}\dd t
= \Bigl[-\frac{\sin^2 t}{t}\Bigr]_\varepsilon^M
+ \int_\varepsilon^M \frac{2\sin t\cos t}{t}\dd t
= \Bigl[-\frac{\sin^2 t}{t}\Bigr]_\varepsilon^M
+ \int_{2\varepsilon}^{2M} \frac{\sin u}{u}\dd u
\]
(مع $u = 2t$ في التكامل الأخير). ويؤول القوس إلى $0$ عند
الطرفين (لأن $\sin^2\varepsilon/\varepsilon \leq \varepsilon$؛ و
$\sin^2 M/M \leq 1/M$)، ويؤول التكامل الأخير إلى
$\int_0^\infty \frac{\sin u}{u}\dd u = \frac\pi2$
(\cref{exo:b2:integration:10}). ومنه
\[
\int_0^\infty \Bigl(\frac{\sin t}{t}\Bigr)^{\!2}\dd t
= \frac\pi2 .
\]
\end{solution}

\begin{solution}{exo:b2:integration:12}
\begin{enumerate}
  \item بالتجزئة مع $u = \frac{-1}{2t}$، $v' = -2t\,\eu^{-t^2}$
        (ومن ثم $v = \eu^{-t^2}$):
        \[
        T(x) = \Bigl[\frac{-\eu^{-t^2}}{2t}\Bigr]_x^\infty
        - \int_x^\infty \frac{\eu^{-t^2}}{2t^2}\dd t
        = \frac{\eu^{-x^2}}{2x}
        - \int_x^\infty \frac{\eu^{-t^2}}{2t^2}\dd t .
        \]
        والحيلة نفسها على التكامل الجديد (مع $u = \frac{-1}{4t^3}$،
        $v' = -2t\,\eu^{-t^2}$):
        \[
        \int_x^\infty \frac{\eu^{-t^2}}{2t^2}\dd t
        = \frac{\eu^{-x^2}}{4x^3}
        - \frac34\int_x^\infty \frac{\eu^{-t^2}}{t^4}\dd t ,
        \]
        ومنه المتطابقة المعلَنة.
  \item ومكاملة أخرى بالتجزئة تحدّ الباقي:
        \[
        \int_x^\infty \frac{\eu^{-t^2}}{t^4}\dd t
        = \frac{\eu^{-x^2}}{2x^5}
        - \frac52\int_x^\infty\frac{\eu^{-t^2}}{t^6}\dd t
        \leq \frac{\eu^{-x^2}}{2x^5},
        \]
        ومنه $0 \leq \frac34\int_x^\infty t^{-4}\eu^{-t^2}\dd t
        \leq \frac{3}{8x^5}\eu^{-x^2}$. وإسقاط الباقي (الموجب)
        في متطابقة السؤال 1 يعطي الحدّ
        الأدنى؛ وإسقاط الحدّ الثاني (السالب) في التجزئة الأولى
        يعطي $T(x) \leq \frac{\eu^{-x^2}}{2x}$. وبقسمة
        الحصر على $\frac{\eu^{-x^2}}{2x}$: تُحصر النسبة
        بين $1 - \frac{1}{2x^2}$ و$1$، ومنه $T(x)
        \sim \frac{\eu^{-x^2}}{2x}$.
  \item يُنتج تكرار التجزئة المتسلسلةَ الشكلية
        \[
        T(x) \approx \frac{\eu^{-x^2}}{2x}\Bigl(1 -
        \frac{1}{2x^2} + \frac{1\cdot3}{(2x^2)^2} -
        \frac{1\cdot3\cdot5}{(2x^2)^3} + \cdots\Bigr),
        \]
        التي ينمو معاملها رقم $k$، أي $1\cdot3\cdots(2k-1) =
        \frac{(2k)!}{2^k k!}$، أسرع من أي متتالية
        هندسية: فمن أجل $x$ ثابت تؤول الحدود
        $\frac{1\cdot3\cdots(2k-1)}{(2x^2)^k}$ إلى اللانهاية
        (فنسبتها $\frac{2k+1}{2x^2} \to \infty$)، ومن ثم تتباعد
        المتسلسلة من أجل كل $x$. وهي نشر \emph{مقارب}:
        فباقتطاعها عند أي رتبة ثابتة، يكون الخطأ من رتبة الحدّ
        الأول المُسقَط حين $x \to \infty$ --- لكنها ليست متسلسلة
        متقاربة أبدًا. (وتقدير الذيل هذا هو حدّ الذيل الغاوسي
        القياسي في فصول الاحتمال.)
\end{enumerate}
\end{solution}

\begin{solution}{pb:b2:integration:1}
\textbf{1.} عند $0^+$ تكون الدالة المكامَلة $\sim t^{x-1}$: ويتقارب
سلّم الطرف المنتهي إذا وفقط إذا $1 - x < 1$، أي $x > 0$
(ومن أجل $x \leq 0$، يتباعد $t^{x-1} \geq t^{-1}$)؛ وعند
$+\infty$، يتقارب $t^{x-1}\eu^{-t} = o(t^{-2})$ من أجل كل
$x$. ثم $\Gamma(x) = \frac{\Gamma(x+1)}{x}$ و$\Gamma(x+1)
\to \Gamma(1) = 1$ حين $x \to 0^+$ (بالاتصال،
\cref{thm:b2:integration:gammaprops}): ومنه $\Gamma(x) \sim
\frac1x$.

\textbf{2.} مع $t = u^2$، $\dd t = 2u\,\dd u$:
\[
\Gamma\Bigl(\frac12\Bigr) = \int_0^\infty
t^{-1/2}\eu^{-t}\dd t
= \int_0^\infty \frac{\eu^{-u^2}}{u}\,2u\,\dd u
= 2\int_0^\infty \eu^{-u^2}\dd u = \sqrt\pi
\]
حسب \cref{exo:b2:integration:8}. ومع $u = v/\sqrt2$:
\[
\int_\R \eu^{-v^2/2}\dd v
= 2\sqrt2\int_0^\infty \eu^{-u^2}\dd u
= \sqrt2\,\sqrt\pi = \sqrt{2\pi} .
\]

\textbf{3.} صحيح من أجل $n = 0$ (فالطرفان $\sqrt\pi$). وإذا كان
$\Gamma(n + \frac12) = \frac{(2n)!}{4^n n!}\sqrt\pi$، فإن المعادلة
الوظيفية تعطي
\[
\Gamma\Bigl(n + 1 + \frac12\Bigr)
= \Bigl(n + \frac12\Bigr)\Gamma\Bigl(n + \frac12\Bigr)
= \frac{2n+1}{2}\cdot\frac{(2n)!}{4^n n!}\sqrt\pi
= \frac{(2n+2)!}{4^{n+1}(n+1)!}\sqrt\pi ,
\]
والخطوة الأخيرة لأن $\frac{(2n+2)!}{(2n)!} = (2n+2)(2n+1)$
و$\frac{2n+1}{2} = \frac{(2n+2)(2n+1)}{4(n+1)}$.

\textbf{4.} يعطي \cref{thm:b2:integration:gammaprops} أن
$\Gamma''(x) = \int_0^\infty t^{x-1}\eu^{-t}(\ln t)^2\dd t$
(بتطبيقين لقاعدة لايبنتز، وبهيمنات كما في برهان
المبرهنة)؛ والدالة المكامَلة $\geq 0$ وليست معدومة
تمامًا، ومنه $\Gamma'' > 0$: أي إن $\Gamma$ محدبة تمامًا و
$\Gamma'$ متزايدة تمامًا. وبما أن $\Gamma(1) = \Gamma(2)
= 1$، يعطي رول عددًا $x_0 \in \intoo12$ يحقق $\Gamma'(x_0) =
0$؛ وتجعل رتابة $\Gamma'$ التامة العددَ $x_0$ جذرَه
الوحيد، مع $\Gamma' < 0$ قبله و$\Gamma' > 0$ بعده:
فتتناقص $\Gamma$ على $\intoo0{x_0}$، وتتزايد على
$\intoo{x_0}\infty$، ويكون $x_0$ الحدّ الأدنى الوحيد.

\textbf{5.} ليكن $k \in \N$ وليكن $x \geq 3$؛ ونختار العدد الصحيح
$n$ بحيث $n + 1 \leq x < n + 2$ (ومن ثم $n \geq 1$). وبرتابة
السؤال 4 (وهي صالحة ابتداءً من $x_0 < 2$): $\Gamma(x)
\geq \Gamma(n + 1) = n!$، في حين أن $x^k \leq (n+2)^k$. وبما أن
$\frac{n!}{(n+2)^k} \to \infty$ (فالعامليات تغلب القوى، مجلد السنة
الأولى)، فإن $\frac{\Gamma(x)}{x^k} \geq \frac{n!}{(n+2)^k} \to
\infty$ حين $x \to \infty$: أي $x^k = o(\Gamma(x))$.

\textbf{6.} قرب $0$ تكون الدالة المكامَلة $\sim t^{x-1}$
(ويتقارب إذا وفقط إذا $x > 0$)، وقرب $1$ تكون $\sim (1-t)^{y-1}$
(إذا وفقط إذا $y > 0$)؛ والمقارنتان بين دالتين موجبتين،
ومن ثم يتقارب $B(x,y)$ من أجل $x, y > 0$ بالضبط. ويبدّل التعويض
$t \mapsto 1 - t$ العاملين: أي $B(x,y) = B(y,x)$.

\textbf{7.} $B(x,1) = \int_0^1 t^{x-1}\dd t = \frac1x$. وبالتجزئة
على $\intcc\varepsilon{1-\varepsilon}$ مع $u = (1-t)^y$، $v =
\frac{t^x}{x}$:
\[
\int t^{x-1}(1-t)^{y}\dd t
= \Bigl[\frac{t^x(1-t)^y}{x}\Bigr]
+ \frac{y}{x}\int t^{x}(1-t)^{y-1}\dd t ;
\]
وينعدم القوس عند الطرفين حين $\varepsilon \to 0$ (لأن $x >
0$ عند $0$، و$y > 0$ عند $1$)، فيبقى $B(x, y+1) = \frac
yx\,B(x+1, y)$.

\textbf{8.} بما أن $t + (1-t) = 1$:
\[
t^{x-1}(1-t)^{y-1} = t^{x}(1-t)^{y-1} + t^{x-1}(1-t)^{y},
\]
ومنه $B(x,y) = B(x+1,y) + B(x,y+1)$. ويُقرأ السؤال 7 هكذا: $B(x+1,y) =
\frac xy B(x,y+1)$؛ وبالتعويض،
\[
B(x,y) = \Bigl(\frac xy + 1\Bigr)B(x,y+1)
= \frac{x+y}{y}\,B(x,y+1),
\]
أي $B(x,y+1) = \frac{y}{x+y}B(x,y)$؛ وتنتج العلاقة التوأم
من تناظر السؤال 6.

\textbf{9.} بالتراجع على $n$ عند $m$ ثابت: $B(m,1) = \frac1m =
\frac{(m-1)!\,0!}{m!}$، وإذا صحّت الصيغة عند $n$، فإن
\[
B(m, n+1) = \frac{n}{m+n}\,B(m,n)
= \frac{n}{m+n}\cdot\frac{(m-1)!(n-1)!}{(m+n-1)!}
= \frac{(m-1)!\,n!}{(m+n)!} .
\]
وبإعادة الكتابة: $B(m,n) = \frac{(m-1)!(n-1)!}{(m+n-1)!} =
\bigl[(m+n-1)\binom{m+n-2}{m-1}\bigr]^{-1}$.

\textbf{10.} يساوي طرفا $B(x,n) =
\frac{\Gamma(x)\Gamma(n)}{\Gamma(x+n)}$ المقدارَ $\frac1x$ عند $n =
1$ (لأن $\Gamma(1) = 1$، $\Gamma(x+1) = x\Gamma(x)$). وإذا اتفقا
عند $n$، فبعلاقة النزول والمعادلة
الوظيفية:
\[
B(x, n+1) = \frac{n}{x+n}\,B(x,n), \qquad
\frac{\Gamma(x)\Gamma(n+1)}{\Gamma(x+n+1)}
= \frac{n}{x+n}\cdot
\frac{\Gamma(x)\Gamma(n)}{\Gamma(x+n)} :
\]
فالمتتاليتان تخضعان للتراجع نفسه انطلاقًا من البذرة نفسها،
ومن ثم تتفقان من أجل كل $n \in \N^*$ وكل $x > 0$.

\textbf{11.} مع $t = \sin^2\theta$ (مع $\theta \in
\intoo0{\pi/2}$، $\dd t = 2\sin\theta\cos\theta\,\dd\theta$)،
$t^{x-1} = \sin^{2x-2}\theta$ و$(1-t)^{y-1} =
\cos^{2y-2}\theta$:
\[
B(x,y) = \int_0^{\pi/2}\sin^{2x-2}\theta\,\cos^{2y-2}\theta
\cdot 2\sin\theta\cos\theta\,\dd\theta
= 2\int_0^{\pi/2}\sin^{2x-1}\theta\,\cos^{2y-1}\theta\,
\dd\theta .
\]

\textbf{12.} نأخذ $y = \frac12$ (فيموت عامل جيب التمام) و
$2x - 1 = n$: $B\bigl(\frac{n+1}2, \frac12\bigr) = 2W_n$، أي
$W_n = \frac12 B\bigl(\frac{n+1}2,\frac12\bigr)$. وتعطي علاقة النزول
في المتغير الأول أن
\[
\frac{W_n}{W_{n-2}}
= \frac{B\bigl(\frac{n-1}2 + 1, \frac12\bigr)}
{B\bigl(\frac{n-1}2, \frac12\bigr)}
= \frac{\frac{n-1}2}{\frac{n-1}2 + \frac12}
= \frac{n-1}{n} :
\]
وهو تراجع واليس، هذه المرة بلا أي مكاملة بالتجزئة
على الجيوب --- فقد أنجز الجزء الثاني العمل مرة واحدة إلى الأبد.

\textbf{13.} $B\bigl(\frac12,\frac12\bigr) = 2W_0 =
2\cdot\frac\pi2 = \pi$، في حين أن
$\Gamma\bigl(\frac12\bigr)^2/\Gamma(1) = (\sqrt\pi)^2 = \pi$:
فصيغة أويلر تصحّ عند $\bigl(\frac12,\frac12\bigr)$.

\textbf{14.} بتكرار $W_{2n} = \frac{2n-1}{2n}W_{2n-2}$ انطلاقًا من
$W_0 = \frac\pi2$:
\[
W_{2n} = \frac\pi2\prod_{k=1}^{n}\frac{2k-1}{2k}
= \frac\pi2\cdot\frac{(2n)!}{4^n(n!)^2},
\]
لأن $\prod(2k-1) = \frac{(2n)!}{2^n n!}$ و$\prod 2k = 2^n
n!$. ومنه، باستعمال السؤال 3:
\[
B\Bigl(n+\frac12, \frac12\Bigr) = 2W_{2n}
= \pi\,\frac{(2n)!}{4^n(n!)^2}
= \frac{(2n)!\sqrt\pi}{4^n n!}\cdot\frac{\sqrt\pi}{n!}
= \frac{\Gamma\bigl(n+\frac12\bigr)\Gamma\bigl(\frac12\bigr)}
{\Gamma(n+1)} .
\]
ولنثبّت الآن $x \in \frac12\N^*$. فصيغة أويلر تصحّ عند $(x,
\frac12)$: فمن أجل $x$ صحيح هذا هو السؤال 10 (مع التناظر)،
ومن أجل $x = n + \frac12$ هي الصيغة المعروضة أعلاه. ويخضع طرفا
صيغة أويلر لتراجع النزول $y \mapsto y + 1$
(بالسؤال 8 على اليسار، وبالمعادلة الوظيفية على اليمين،
كما في السؤال 10): فينشر التراجع الصيغةَ من $y =
\frac12$ ومن $y = 1$ إلى كل $y \in \frac12\N^*$. ومن ثم تصحّ صيغة
أويلر كلما كان $2x, 2y \in \N^*$.

\textbf{15.} مع $u = \frac{t}{1-t}$، أي $t =
\frac{u}{1+u}$، $1 - t = \frac{1}{1+u}$، $\dd t =
\frac{\dd u}{(1+u)^2}$:
\[
B(x,y) = \int_0^\infty
\Bigl(\frac{u}{1+u}\Bigr)^{x-1}
\Bigl(\frac{1}{1+u}\Bigr)^{y-1}
\frac{\dd u}{(1+u)^2}
= \int_0^\infty \frac{u^{x-1}}{(1+u)^{x+y}}\,\dd u .
\]
وعند $x = y = \frac12$، مع $u = v^2$:
\[
\int_0^\infty \frac{u^{-1/2}}{1+u}\dd u
= \int_0^\infty \frac{2\,\dd v}{1+v^2} = \pi
= B\Bigl(\frac12,\frac12\Bigr) . \checkmark
\]

\textbf{16.} مكاملة واحدة بالتجزئة، من أجل $1 \leq k \leq n$
و$s > 0$ (مع $u = (1 - t/n)^k$، $v = t^s/s$؛ وتنعدم الحدود الحدّية):
\[
\int_0^n \Bigl(1-\frac tn\Bigr)^{\!k} t^{s-1}\dd t
= \frac{k}{ns}\int_0^n \Bigl(1-\frac tn\Bigr)^{\!k-1}
t^{s}\dd t .
\]
وانطلاقًا من $k = n$، $s = x$ وبالتكرار $n$ مرة:
\[
\int_0^n \Bigl(1-\frac tn\Bigr)^{\!n} t^{x-1}\dd t
= \frac{n(n-1)\cdots1}{n^n\,x(x+1)\cdots(x+n-1)}
\int_0^n t^{x+n-1}\dd t
= \frac{n!}{n^n}\cdot
\frac{n^{x+n}}{x(x+1)\cdots(x+n)} ,
\]
وهو $\dfrac{n!\,n^x}{x(x+1)\cdots(x+n)}$.

\textbf{17.} حسب \cref{exo:b2:integration:7} يؤول الطرف الأيسر
إلى $\Gamma(x)$ (بالتقارب المهيمَن عليه بالدالة المهيمِنة
$t^{x-1}\eu^{-t}$)؛ والطرف الأيمن هو خارج قسمة غاوس:
\[
\Gamma(x) = \lim_{n\to\infty}
\frac{n!\,n^x}{x(x+1)\cdots(x+n)} .
\]

\textbf{18.} بأخذ اللوغاريتمات في خارج قسمة السؤال 16
$G_n(x)$ وبفصل $\ln(x+k) = \ln k + \ln(1 + x/k)$ من أجل $k
\geq 1$:
\[
\ln G_n(x) = x\ln n - \ln x - \sum_{k=1}^n
\ln\Bigl(1+\frac xk\Bigr)
= -\ln x + x(\ln n - H_n)
+ \sum_{k=1}^n\Bigl(\frac xk -
\ln\Bigl(1+\frac xk\Bigr)\Bigr).
\]
ومن أجل $u \geq 0$، $u - \frac{u^2}2 \leq \ln(1+u) \leq u$، ومن ثم يقع
الحدّ العام في $\intcc{0}{x^2/(2k^2)}$: فتتقارب المتسلسلة
(بالمقارنة مع $\sum k^{-2}$). وبما أن $\ln n - H_n
\to -\gamma$ (\cref{ex:b2:comparison:harmonic}) و$\ln G_n(x)
\to \ln\Gamma(x)$ (بالسؤال 17 وباتصال $\ln$):
\[
\ln\Gamma(x) = -\ln x - \gamma x
+ \sum_{k=1}^\infty\Bigl(\frac xk -
\ln\Bigl(1+\frac xk\Bigr)\Bigr) .
\]

\textbf{19.} عند $x = \frac12$، يكون المقام
$\prod_{k=0}^n\bigl(k+\frac12\bigr) =
\frac{(2n+1)!}{2^{2n+1}n!}$ (بضرب الأنصاف)، ومنه
\[
G_n\Bigl(\frac12\Bigr)
= \frac{n!\,\sqrt n\;2^{2n+1}n!}{(2n+1)!}
= \frac{2\sqrt n\;4^n}{(2n+1)\binom{2n}{n}} .
\]
ومع $\binom{2n}{n} \sim \frac{4^n}{\sqrt{\pi n}}$
(\cref{ex:b2:comparison:centralbinomial}):
\[
G_n\Bigl(\frac12\Bigr)
\sim \frac{2\sqrt n\,\sqrt{\pi n}}{2n+1}
\longrightarrow \sqrt\pi = \Gamma\Bigl(\frac12\Bigr) .
\]
فالمقدار $\sqrt\pi$ للمعامل الثنائي المركزي (وهو الآتي من
واليس، ومن ثم من ثابت ستيرلينغ) والمقدار $\sqrt\pi$ في التكامل
الغاوسي هما العدد نفسه.

\textbf{20.} المتطابقة جبر مباشر: نضرب
$\frac{n!\,n^x}{x(x+1)\cdots(x+n)}$ في $\frac{nx}{x+n+1}$ و
نمتصّ $x$ في الجداء، و$n$ في $n^x$. وبأخذ $n \to
\infty$: يؤول الطرف الأيسر إلى $\Gamma(x+1)$ (غاوس عند $x+1$)،
ويؤول الأيمن إلى $\Gamma(x)\cdot x\cdot 1$ لأن
$\frac{n}{x+n+1} \to 1$: أي $\Gamma(x+1) = x\Gamma(x)$ ---
مستعادًا بلا أي مكاملة بالتجزئة.

\textbf{21.} مع $u = t^a$، $t = u^{1/a}$، $\dd t = \frac1a
u^{1/a - 1}\dd u$:
\[
\int_0^\infty \eu^{-t^a}\dd t
= \frac1a\int_0^\infty u^{\frac1a - 1}\eu^{-u}\dd u
= \frac1a\,\Gamma\Bigl(\frac1a\Bigr)
= \Gamma\Bigl(1 + \frac1a\Bigr) .
\]
وحين $a \to +\infty$ (على امتداد أي متتالية): $\eu^{-t^a} \to 1$ من أجل
$0 < t < 1$، و$\to \eu^{-1}$ عند $t = 1$، و$\to 0$ من أجل $t > 1$؛
ومن أجل $a \geq 2$ نُهيمن بالمقدار $\mathbf 1_{t \leq 1} +
\eu^{-t^2}\mathbf 1_{t > 1}$ (لأن $t^a \geq t^2$ من أجل $t \geq 1$)،
وهو قابل للمكاملة. وبالتقارب المهيمَن عليه: يؤول التكامل إلى
$\int_0^1 1\,\dd t = 1$ --- كما يجب، لأن $\Gamma(1 +
\frac1a) \to \Gamma(1) = 1$ بالاتصال.

\textbf{22.} مع $u = t^n$، $\dd t = \frac1n u^{1/n - 1}\dd
u$:
\[
\int_0^1 \frac{\dd t}{\sqrt{1-t^n}}
= \frac1n\int_0^1 u^{\frac1n-1}(1-u)^{-1/2}\dd u
= \frac1n\,B\Bigl(\frac1n, \frac12\Bigr) .
\]
$n = 1$: $B\bigl(1,\frac12\bigr) = B\bigl(\frac12,1\bigr) = 2$،
وهو يطابق $\int_0^1\frac{\dd t}{\sqrt{1-t}} = 2$. و$n = 2$:
$\frac12 B\bigl(\frac12,\frac12\bigr) = \frac\pi2 = \arcsin 1$.
ومن أجل $n = 4$ تكون القيمة $\frac14 B\bigl(\frac14,\frac12\bigr)$ هي
ثابت اللمنيسكات: ولا صورة مغلقة أوّلية له.

\textbf{23.} بتكرار المعادلة الوظيفية:
\[
\frac{1}{\Gamma(x)}\int_0^\infty t^{x+k-1}\eu^{-t}\dd t
= \frac{\Gamma(x+k)}{\Gamma(x)}
= (x+k-1)(x+k-2)\cdots x ,
\]
وهو العاملي الصاعد بعدد $k$ من العوامل. وعند $x = 1$:
$\Gamma(1+k)/\Gamma(1) = k!$، وهي عزوم $\eu^{-t}$ في
\cref{exo:b2:integration:2}.

\textbf{24.} نعوّض $t = \frac{1+s}2$ (مع $s \in
\intoo{-1}1$، $\dd t = \frac{\dd s}2$، $t(1-t) =
\frac{1-s^2}4$):
\[
B(x,x) = \int_{-1}^{1}\Bigl(\frac{1-s^2}{4}\Bigr)^{x-1}
\frac{\dd s}{2}
= 4^{1-x}\int_0^1 (1-s^2)^{x-1}\dd s
\]
(فالدالة المكامَلة زوجية). ثم $s = \sqrt v$ (مع $\dd s =
\frac{\dd v}{2\sqrt v}$):
\[
B(x,x) = \frac{4^{1-x}}{2}\int_0^1
v^{-1/2}(1-v)^{x-1}\dd v
= 2^{1-2x}\,B\Bigl(\frac12, x\Bigr) .
\]
ومن أجل $2x \in \N^*$ يقع كل وسيط ظاهر في
$\frac12\N^*$، ومن ثم تنطبق صيغة أويلر (السؤال 14) على الطرفين:
\[
\frac{\Gamma(x)^2}{\Gamma(2x)}
= 2^{1-2x}\,
\frac{\Gamma\bigl(\frac12\bigr)\Gamma(x)}
{\Gamma\bigl(x+\frac12\bigr)}
\quad\Longleftrightarrow\quad
\Gamma(x)\,\Gamma\Bigl(x+\frac12\Bigr)
= 2^{1-2x}\sqrt\pi\;\Gamma(2x) .
\]
وبالتحقق المباشر عند $x = n$: الطرف الأيسر $(n-1)!\cdot
\frac{(2n)!\sqrt\pi}{4^n n!} = \frac{(2n)!\sqrt\pi}{4^n n}$،
والطرف الأيمن $2\cdot4^{-n}\sqrt\pi\,(2n-1)! =
\frac{(2n)!\sqrt\pi}{4^n n}$: فهما متساويان.

\textbf{25.} (1) أنتجت المكاملة بالتجزئة العلاقةَ $B(x,y+1) =
\frac yx B(x+1,y)$، وهي المتطابقة الواحدة التي تتدفق منها كل
علاقات النزول، والقيم الصحيحة وأنصاف الصحيحة، وتراجع
واليس. (2) وحوّل التقارب المهيمَن عليه
التكاملات الأوّلية $\int_0^n(1-t/n)^n t^{x-1}$ إلى
$\Gamma(x)$ (صيغة غاوس، السؤال 17)، وحسب
النهاية $a \to \infty$ في السؤال 21. (3) واستوردنا من فصل
المقارنة ثابت أويلر ($\ln n - H_n \to
-\gamma$، السؤال 18) ومقاربات المعامل الثنائي المركزي
(السؤال 19) --- أي صيغة ستيرلينغ بلباس. (4)
وصيغة أويلر $B(x,y) = \Gamma(x)\Gamma(y)/\Gamma(x+y)$
مبرهَنة الآن من أجل $y \in \N^*$ مع $x > 0$ كيفي (السؤال
10) ومن أجل كل الأزواج من أنصاف الأعداد الصحيحة (السؤال 14)؛ أما الحالة
العامة $x, y > 0$ فتنتظر حساب التكامل المزدوج في
فصل التكاملات المتعددة، الذي يفكك
$\Gamma(x)\Gamma(y)$ على ربع مستوٍ.
\end{solution}
