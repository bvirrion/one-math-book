\section*{الفصل \ref{ch:b2:comparison} --- مقارنة الدوال}

\begin{solution}{exo:b2:comparison:1}
$\sqrt{x^2 + x + 1} = x\sqrt{1 + \tfrac1x + \tfrac{1}{x^2}} = x +
\frac12 + \frac38\cdot\frac1x + o\bigl(\frac1x\bigr)$ (بالنشر
الثنائي: يعطي $\frac12 u - \frac18 u^2$ مع $u = \frac1x +
\frac{1}{x^2}$ أن $\frac{1}{2x} + \frac{1}{2x^2} -
\frac{1}{8x^2} = \frac{1}{2x} + \frac{3}{8x^2}$، ثم نضرب في
$x$).

$\ln(x^2 + x) - 2\ln x = \ln\bigl(1 + \tfrac1x\bigr) = \frac1x -
\frac{1}{2x^2} + o\bigl(\frac{1}{x^2}\bigr)$.

الدالة الثالثة: ننشر كل عامل،
\[
\frac{x + \sin x}{x - \ln x}
= \Bigl(1 + \frac{\sin x}{x}\Bigr)
\Bigl(1 + \frac{\ln x}{x} + \frac{(\ln x)^2}{x^2} +
O\Bigl(\frac{(\ln x)^3}{x^3}\Bigr)\Bigr).
\]
ونرتّب الإسهامات على السلّم عند $+\infty$: $\frac{\ln x}{x}
\gg \frac{1}{x} \geq \bigl|\frac{\sin x}{x}\bigr| \gg \frac{(\ln
x)^2}{x^2}$. ومن ثم فالحدّان اللذان يليان الحدّ الرئيسي $1$ هما
$\frac{\ln x}{x}$، ثم حدّ التذبذب المحدود $\frac{\sin
x}{x}$:
\[
\frac{x + \sin x}{x - \ln x}
= 1 + \frac{\ln x}{x} + \frac{\sin x}{x}
+ O\Bigl(\frac{(\ln x)^2}{x^2}\Bigr).
\]
\end{solution}

\begin{solution}{exo:b2:comparison:2}
$f(t) = t^\alpha$ ($t \geq 1$).

$\alpha > -1$: تباعد، وحسب
\cref{thm:b2:comparison:seriesintegral}~(2)، $\sum_{k\leq n}
k^\alpha = \frac{n^{\alpha+1}}{\alpha+1} + C + o(1)$ إذا كان $\alpha <
0$ (حيث تتناقص $f$)؛ ومن أجل $\alpha \geq 0$ (حيث تتزايد $f$) يعطي
الحصر نفسه بمتراجحات معكوسة أن $\sum_{k \leq n}
k^\alpha \sim \frac{n^{\alpha + 1}}{\alpha + 1}$.

$\alpha = -1$: $H_n = \ln n + \gamma + o(1)$
(\cref{ex:b2:comparison:harmonic}).

$\alpha < -1$: تقارب، بباقٍ $\sum_{k > n} k^\alpha
\sim \frac{n^{\alpha+1}}{-(\alpha+1)}$ بالحصر (1) (فحدّا
التكامل كلاهما مكافئ لتلك القيمة).
\end{solution}

\begin{solution}{exo:b2:comparison:3}
لتكن $v_n = H_n - \ln n - \gamma \to 0$. عندئذٍ
\[
v_n - v_{n+1} = \ln\frac{n+1}{n} - \frac{1}{n+1}
= \Bigl(\frac1n - \frac{1}{2n^2}\Bigr) - \Bigl(\frac1n -
\frac{1}{n^2}\Bigr) + O\Bigl(\frac{1}{n^3}\Bigr)
= \frac{1}{2n^2} + O\Bigl(\frac{1}{n^3}\Bigr),
\]
باستعمال $\frac{1}{n+1} = \frac1n - \frac{1}{n^2} + O(n^{-3})$. وبما أن
$v_n \to 0$، فبالتلسكوب على الذيل:
\[
v_n = \sum_{k \geq n} (v_k - v_{k+1})
= \sum_{k\geq n} \Bigl(\frac{1}{2k^2} + O(k^{-3})\Bigr)
= \frac{1}{2n} + O\Bigl(\frac{1}{n^2}\Bigr),
\]
حسب \cref{thm:b2:comparison:seriesintegral}~(1) مطبَّقة على $t^{-2}$
(بباقٍ $\sim \frac1n$، منصَّفًا) وعلى $t^{-3}$. ومنه $H_n =
\ln n + \gamma + \frac{1}{2n} + o(\frac1n)$.
\end{solution}

\begin{solution}{exo:b2:comparison:4}
ستيرلينغ ثلاث مرات:
\[
\frac{(3n)!}{(n!)^3}
\sim \frac{\sqrt{6\pi n}\,(3n/\eu)^{3n}}
{(2\pi n)^{3/2}\,(n/\eu)^{3n}}
= \frac{\sqrt{6}\; 27^{\,n}}{2\pi n} \cdot
\frac{1}{\sqrt{2\pi n}}\cdot\sqrt{2\pi n}\;
= \frac{\sqrt3\,27^n}{2\pi n} .
\]
(وبانتباه: $\frac{\sqrt{6\pi n}}{(2\pi n)^{3/2}} =
\frac{\sqrt6}{(2\pi n)\sqrt{2\pi n}}\sqrt{\pi n} =
\frac{\sqrt3}{2\pi n}$.)

$\dfrac{n!}{n^n} \sim \sqrt{2\pi n}\,\eu^{-n}$.

$\sqrt[n]{n!} = \exp\bigl(\frac{\ln n!}{n}\bigr)$ مع $\ln n! = n\ln
n - n + \frac12\ln(2\pi n) + o(1)$:
\[
\sqrt[n]{n!} = \exp\Bigl(\ln n - 1 + \frac{\ln(2\pi n)}{2n} +
o\Bigl(\frac{\ln n}{n}\Bigr)\Bigr)
= \frac{n}{\eu}\Bigl(1 + \frac{\ln(2\pi n)}{2n} +
o\Bigl(\frac{\ln n}{n}\Bigr)\Bigr).
\]
\end{solution}

\begin{solution}{exo:b2:comparison:5}
$g(x) = x^n + x - 1$ متزايدة تمامًا على $\intcc{0}{1}$ من $-1$
إلى $1$: فجذر وحيد $x_n$. وبما أن $x_n^n = 1 - x_n \in
\intoo{0}{1}$: فلو كان $x_n \leq c < 1$ على متتالية جزئية، لكان $x_n^n
\leq c^n \to 0$، ومنه $1 - x_n \to 0$: وهو تناقض مع $x_n \leq
c$. ومنه $x_n \to 1$.

نكتب $x_n = 1 - \varepsilon_n$، $\varepsilon_n \to 0^+$. وتُقرأ
المعادلة $(1 - \varepsilon_n)^n = \varepsilon_n$، أي
\[
n\ln(1 - \varepsilon_n) = \ln \varepsilon_n
\quad\Longrightarrow\quad
-n\varepsilon_n\bigl(1 + o(1)\bigr) = \ln\varepsilon_n .
\]
ومنه $n\varepsilon_n = -\ln\varepsilon_n\,(1 + o(1)) \to +\infty$، و
بأخذ اللوغاريتمات من جديد: $\ln n + \ln\varepsilon_n =
\ln(-\ln\varepsilon_n) + o(1)$. وبما أن $\ln(-\ln \varepsilon_n) =
o(\ln(1/\varepsilon_n))$، يعطي هذا أن $\ln\varepsilon_n \sim -\ln
n$، ومنه $\varepsilon_n = \frac{-\ln\varepsilon_n}{n}(1 + o(1))
\sim \frac{\ln n}{n}$:
\[
x_n = 1 - \frac{\ln n}{n} + o\Bigl(\frac{\ln n}{n}\Bigr) .
\]
\end{solution}

\begin{solution}{exo:b2:comparison:6}
العلاقة الدقيقة: $\cot y_n = x_n = n\pi + \frac\pi2 - y_n$، مع $y_n
\sim \frac{1}{n\pi}$ (لأن \cref{ex:b2:comparison:tan}). وننشر $\cot y
= \frac1y - \frac y3 + O(y^3)$:
\[
\frac{1}{y_n} - \frac{y_n}{3} + O(y_n^3) = n\pi + \frac\pi2 - y_n
\quad\Longrightarrow\quad
\frac{1}{y_n} = n\pi + \frac\pi2 + O\Bigl(\frac1n\Bigr),
\]
(فالحدّان $-y_n$ و$-\frac{y_n}{3}$ هما $O(\frac1n)$). ونقلب:
\[
y_n = \frac{1}{n\pi}\cdot\frac{1}{1 + \frac{1}{2n} + O(n^{-2})}
= \frac{1}{n\pi}\Bigl(1 - \frac{1}{2n} + O\Bigl(\frac{1}{n^2}\Bigr)\Bigr)
= \frac{1}{n\pi} - \frac{1}{2n^2\pi} + O\Bigl(\frac{1}{n^3}\Bigr).
\]
ومنه
\[
x_n = n\pi + \frac{\pi}{2} - y_n
= n\pi + \frac\pi2 - \frac{1}{n\pi} + \frac{1}{2n^2\pi} +
o\Bigl(\frac{1}{n^2}\Bigr).
\]
\end{solution}

\begin{solution}{exo:b2:comparison:7}
نفصل الحدّين الأكبرين:
\[
\sum_{k=0}^{n} k! = n! + (n-1)! + \sum_{k \leq n-2} k! ,
\qquad
\sum_{k\leq n-2} k! \leq (n-1)\,(n-2)! = (n-1)! .
\]
ومنه $1 \leq \frac{1}{n!}\sum k! \leq 1 + \frac{2}{n}$: فالنهاية
$1$. وبالتنقيح: $\frac{(n-1)!}{n!} = \frac1n$، ويمكن شحذ الحدّ الخام
$\sum_{k \leq n-2}k! \leq (n-1)!$ بالطريقة نفسها:
$\sum_{k\leq n-2} k! = (n-2)!\,(1 + O(\frac1n)) = O\bigl(\frac{n!}{n^2}\bigr)$.
ومنه
\[
\sum_{k=0}^{n} k! = n!\Bigl(1 + \frac1n + O\Bigl(\frac{1}{n^2}\Bigr)\Bigr).
\]
\end{solution}

\begin{solution}{exo:b2:comparison:8}
$(u_n)$ متزايدة؛ ولو كانت محدودة لتقاربت إلى $\ell$ مع
$\ell = \ell + \frac1\ell$: وهذا محال. ومنه $u_n \to \infty$.

المربّعات: $u_{n+1}^2 = u_n^2 + 2 + u_n^{-2}$، ومنه
\[
u_n^2 = u_0^2 + 2n + \sum_{k=0}^{n-1} \frac{1}{u_k^2} .
\]
والمجموع $o(n)$ (فالحدود تؤول إلى $0$، بتشيزارو)، ومنه $u_n^2 \sim 2n$
و$u_n \sim \sqrt{2n}$.

التنقيح: $\frac{1}{u_k^2} \sim \frac{1}{2k}$، ومن ثم بالمقارنة
(\cref{thm:b2:comparison:seriesintegral}، أو بمكافئات المجاميع
الجزئية للمتسلسلات الموجبة) $\sum_{k<n} u_k^{-2} \sim \frac12 \ln n$.
ومنه
\[
u_n^2 = 2n + \frac{\ln n}{2}\,(1 + o(1)) + O(1)
\quad\Longrightarrow\quad
u_n = \sqrt{2n}\sqrt{1 + \frac{\ln n}{4n} + o\Bigl(\frac{\ln
n}{n}\Bigr)}
= \sqrt{2n}\Bigl(1 + \frac{\ln n}{8n} + o\Bigl(\frac{\ln
n}{n}\Bigr)\Bigr).
\]
\end{solution}

\begin{solution}{exo:b2:comparison:9}
نُخرِج العامل $n$ ونضع $t = \ln n$:
\[
S_n = \frac1n \sum_{k=1}^{n} \frac{1}{1 + t\,\frac kn} .
\]
ومن أجل $t$ ثابت، يكون المجموع مجموعَ ريمان للدالة $u \mapsto \frac{1}{1 +
tu}$ على $\intcc{0}{1}$؛ والدالة رتيبة في $u$، ومن ثم
يُحصر مجموع ريمان بالتكامل مزاحًا بخطوة واحدة:
\[
\int_0^1 \frac{\dd u}{1 + tu} - \frac1n
\leq S_n \leq \int_0^1 \frac{\dd u}{1 + tu} + \frac1n
\]
(وهي مقارنة مجاميع ريمان لدالة رتيبة بتكاملها،
وهي صالحة من أجل كل $n$ بالعدد $t = \ln n$ الخاص به). والآن
$\int_0^1 \frac{\dd u}{1 + tu} = \frac{\ln(1 + t)}{t}$، و
$\frac1n = o\bigl(\frac{\ln t}{t}\bigr)$: ومنه
\[
S_n = \frac{\ln(1 + \ln n)}{\ln n} + O\Bigl(\frac 1n\Bigr)
\;\sim\; \frac{\ln\ln n}{\ln n} .
\]
\end{solution}

\begin{solution}{exo:b2:comparison:10}
المتطابقة: $(\ln n)^{\ln n} = \eu^{\ln n\,\ln\ln n} =
\bigl(\eu^{\ln n}\bigr)^{\ln\ln n} = n^{\ln\ln n}$. والترتيب:
نقارن اللوغاريتمات. $\ln(n^2) = 2\ln n$؛ $\ln\bigl((\ln
n)^{\ln n}\bigr) = \ln n\ln\ln n$؛ $\ln(2^n) = n\ln2$؛
$\ln(n!) = n\ln n - n + O(\ln n)$ (بستيرلينغ، أو بالحصر الأخشن
$\ln n! \sim n\ln n$)؛ $\ln(n^n) = n\ln n$. وبما أن
$2\ln n = o(\ln n\ln\ln n)$، $\ln n\ln\ln n = o(n)$، $n\ln 2 =
o(n\ln n - n)$، و$n \ln n - n \sim n\ln n$ لكن $n! / n^n \to
0$ (ففرق اللوغاريتمين $-n + O(\ln n) \to -\infty$):
\[
n^2 = o\bigl((\ln n)^{\ln n}\bigr),\quad
(\ln n)^{\ln n} = o(2^n),\quad
2^n = o(n!),\quad
n! = o(n^n).
\]
(ففي كل خطوة: يؤول فرق اللوغاريتمين إلى
$+\infty$، ومن ثم تؤول النسبة إلى $0$.)
\end{solution}

\begin{solution}{exo:b2:comparison:11}
نفكك $\frac1{k^2} = \frac1{k(k+1)} + \frac1{k^2(k+1)}$ ونجمع
من أجل $k > n$:
\[
\sum_{k>n}\frac1{k^2} = \frac1{n+1} +
\sum_{k>n}\frac{1}{k^2(k+1)} ,
\]
فالمجموع الأول تلسكوبي بالضبط ($\frac1{k(k+1)} = \frac1k -
\frac1{k+1}$). وأما الثاني: فلدينا $\frac{1}{k^2(k+1)} = \frac1{k^3}
+ O\bigl(\frac1{k^4}\bigr)$ (لأن $\frac{1}{k^2(k+1)} -
\frac1{k^3} = \frac{-1}{k^3(k+1)}$)، وبمقارنة
التكامل $\sum_{k>n}\frac1{k^3} = \frac1{2n^2} +
O\bigl(\frac1{n^3}\bigr)$، $\sum_{k>n}\frac1{k^4} =
O\bigl(\frac1{n^3}\bigr)$. ومنه
\[
\sum_{k>n}\frac1{k^2}
= \frac1{n+1} + \frac{1}{2n^2} + O\Bigl(\frac1{n^3}\Bigr)
= \frac1n - \frac1{n^2} + \frac{1}{2n^2} +
O\Bigl(\frac1{n^3}\Bigr)
= \frac1n - \frac{1}{2n^2} + O\Bigl(\frac1{n^3}\Bigr),
\]
باستعمال $\frac1{n+1} = \frac1n - \frac1{n^2} +
O\bigl(\frac1{n^3}\bigr)$.
\end{solution}

\begin{solution}{exo:b2:comparison:12}
$(u_n)$ متزايدة؛ ولو كانت محدودة لتقاربت إلى قيمة
منتهية $\ell$ مع $\ell = \ell + \eu^{-\ell}$: وهذا مستحيل. ومنه
$u_n \to \infty$. ولتكن $v_n = \eu^{u_n} \to \infty$: عندئذٍ
\[
v_{n+1} = \eu^{u_n + \eu^{-u_n}} = v_n\,\eu^{1/v_n}
= v_n\Bigl(1 + \frac1{v_n} + \frac1{2v_n^2} +
O\bigl(v_n^{-3}\bigr)\Bigr)
= v_n + 1 + \frac{1}{2v_n} + O\bigl(v_n^{-2}\bigr).
\]
ويعطي جمع $v_{k+1} - v_k = 1 + O(1)$ أولًا أن $v_n = n +
O(n)$، ومنه $v_n \geq cn$ ابتداءً من رتبة ما؛ وبإعادة الجمع مع
$\frac1{2v_k} = O(\frac1k)$ نجد $v_n = n + O(\ln n)$. وبمرور آخر:
$\frac{1}{2v_k} = \frac{1}{2k}\bigl(1 +
O\bigl(\tfrac{\ln k}k\bigr)\bigr)$، ومنه
\[
v_n = n + \sum_{k<n}\frac1{2k} + O(1) = n + \frac{\ln n}2 +
O(1).
\]
وأخيرًا $u_n = \ln v_n = \ln n + \ln\Bigl(1 + \frac{\ln n}{2n} +
O\bigl(\tfrac1n\bigr)\Bigr) = \ln n + \frac{\ln n}{2n} +
O\bigl(\tfrac1n\bigr)$.
\end{solution}

\begin{solution}{pb:b2:comparison:1}
\textbf{1.} بطرح النشرين: $\sum_i (c_i -
c_i')\varphi_i = o(\varphi_k)$. فإذا اختلف معامل ما، وليكن
$i_0$ أولها: فبالقسمة على $\varphi_{i_0}$ واستعمال
$\varphi_j = o(\varphi_{i_0})$ من أجل $j > i_0$ نجد $c_{i_0} -
c_{i_0}' = o(1)$: أي صفرًا، وهو تناقض. وأما النشر: فمع
$u = \frac{\ln x}x \to 0$،
\[
\frac{1}{x - \ln x} = \frac1x\cdot\frac{1}{1 - u}
= \frac1x\bigl(1 + u + u^2 + O(u^3)\bigr)
= \frac1x + \frac{\ln x}{x^2} + \frac{(\ln x)^2}{x^3} +
o\Bigl(\frac{(\ln x)^2}{x^3}\Bigr).
\]
ولا يظهر أي حدّ $\frac c{x^2}$ لأن النشر متسلسلة
هندسية في $u = \frac{\ln x}{x}$: فكل حدّ يحمل من قوى
$\ln x$ بقدر ما يحمل من قوى $\frac1x$ بعد الأولى؛ ودرجة
السلّم $\frac1{x^2}$ (أي معامل $(\ln x)^0$) غائبة ببساطة،
بمعامل $0$.

\textbf{2.} $f(x) = \eu^x + x$ متصلة ومتزايدة
تمامًا، ونهايتاها $-\infty$ و$+\infty$: فهي تقابل
$\R \to \R$، ومن ثم فإن $x_n = f^{-1}(n)$ موجود ووحيد، و
$x_n \to +\infty$ (لأن $f^{-1}$ متزايدة إلى $+\infty$). ومن
$\eu^{x_n} = n - x_n$: $x_n = \ln(n - x_n) \leq \ln n$، ومنه
$x_n/n \to 0$ و$x_n = \ln n + \ln(1 - x_n/n) = \ln n + o(1)
\sim \ln n$.

\textbf{3.} نكتب $u_n = x_n/n$. المرور الثاني: $u_n = \frac{\ln n
+ o(1)}{n}$، ومنه
\[
x_n = \ln n + \ln(1 - u_n) = \ln n - u_n + O(u_n^2)
= \ln n - \frac{\ln n}{n} + o\Bigl(\frac{\ln n}n\Bigr).
\]
المرور الثالث: الآن $u_n = \frac{\ln n}{n} - \frac{\ln n}{n^2} +
o\bigl(\frac{\ln n}{n^2}\bigr)$، و$\ln(1 - u_n) = -u_n -
\frac{u_n^2}2 + O(u_n^3)$:
\[
x_n = \ln n - \frac{\ln n}n + \frac{\ln n}{n^2}
- \frac{(\ln n)^2}{2n^2} + o\Bigl(\frac{(\ln
n)^2}{n^2}\Bigr)
= \ln n - \frac{\ln n}{n} - \frac{(\ln n)^2}{2n^2} +
o\Bigl(\frac{(\ln n)^2}{n^2}\Bigr),
\]
مع ابتلاع الحدّ $\frac{\ln n}{n^2}$ في
$o\bigl(\frac{(\ln n)^2}{n^2}\bigr)$.

\textbf{4.} عند $n = 1000$: $\ln 1000 \approx 6.90776$ (بخطأ
$7\cdot10^{-3}$)؛ وبحدّين: $6.90085$ (بخطأ
$2\cdot10^{-5}$)؛ وبثلاثة حدود: $6.90082$ (بخطأ دون
$10^{-5}$)، في مقابل $x_{1000} \approx 6.90083$. ويشتري كل مرور
العامل المتوقع $\frac{\ln n}{n}$ تقريبًا.

\textbf{5.} مكاملتان بالتجزئة، ابتداءً من اليمين:
مع $\frac{\dd}{\dd t}\bigl[\tfrac12t(1-t)\bigr] = \tfrac12 -
t$ و$t(1-t)$ منعدمة عند الطرفين،
\[
\frac12\int_0^1 t(1-t)g''(t)\dd t
= -\int_0^1\Bigl(\frac12 - t\Bigr)g'(t)\dd t
= -\Bigl[\Bigl(\frac12 - t\Bigr)g\Bigr]_0^1 - \int_0^1 g
= \frac{g(0) + g(1)}2 - \int_0^1 g .
\]
وبإعادة الترتيب، هذه هي المتطابقة المذكورة.

\textbf{6.} نحسب التزايد، ثم نطبّق السؤال 5 على
$g(t) = f(n + t)$:
\begin{align*}
E_{n+1} - E_n
&= f(n{+}1) - \int_n^{n+1}\!f - \frac{f(n{+}1) - f(n)}2 \\
&= \frac{f(n) + f(n{+}1)}2 - \int_n^{n+1}\!f
= \frac12\int_0^1 t(1-t)f''(n+t)\dd t .
\end{align*}
وبما أن $0 \leq t(1-t) \leq \frac14$: فإن $\abs{E_{n+1} - E_n} \leq
\frac18\int_n^{n+1}\abs{f''}$، ويتقارب مجموعه على $n$ بحكم
الفرض: ومنه تتقارب $(E_n)$ (فتزايداتها قابلة للجمع بإطلاق)
إلى $E$ ما، مع
\[
\abs{E - E_n} \leq \sum_{k\geq n}\abs{E_{k+1} - E_k} \leq
\frac18\int_n^\infty\abs{f''} .
\]

\textbf{7.} $f(t) = \frac1t$: $f''(t) = \frac2{t^3}$،
$\int_1^\infty\abs{f''} = 1 < \infty$. والسؤال 6:
\[
H_n = \ln n + \frac{1 + \frac1n}{2} + E + (E_n - E)
= \ln n + \Bigl(E + \frac12\Bigr) + \frac1{2n} +
\varepsilon_n,
\]
مع $\abs{\varepsilon_n} = \abs{E_n - E} \leq
\frac18\int_n^\infty\frac{2\dd t}{t^3} = \frac1{8n^2}$.
وبالمقارنة مع $H_n = \ln n + \gamma + o(1)$
(\cref{ex:b2:comparison:harmonic}) يتعيّن $E + \frac12 =
\gamma$.

\textbf{8.} من صيغة التزايد في السؤال 6،
\[
\varepsilon_n = E_n - E = -\sum_{k\geq n}\frac12\int_0^1
t(1-t)\,\frac{2\,\dd t}{(k+t)^3}
= -\sum_{k \geq n}\Bigl(\frac1{k^3}\int_0^1t(1-t)\dd t +
O\Bigl(\frac1{k^4}\Bigr)\Bigr),
\]
باستعمال $\frac{1}{(k+t)^3} = \frac1{k^3} +
O\bigl(\frac1{k^4}\bigr)$ بانتظام من أجل $t \in \intcc01$. ومع
$\int_0^1 t(1-t) = \frac16$ و
$\sum_{k\geq n}\frac1{k^3} \sim \frac{1}{2n^2}$
(\cref{thm:b2:comparison:seriesintegral}):
\[
\varepsilon_n = -\frac16\cdot\frac{1}{2n^2} +
o\Bigl(\frac1{n^2}\Bigr) = -\frac{1}{12n^2} +
o\Bigl(\frac{1}{n^2}\Bigr).
\]

\textbf{9.} $f = \ln$: $f''(t) = -\frac1{t^2}$، وهي قابلة
للمكاملة بإطلاق. ويعطي السؤال 6 أن
\[
\ln n! = \int_1^n\ln t\,\dd t + \frac{\ln n}2 + E + O\Bigl(
\frac1{8}\int_n^\infty\frac{\dd t}{t^2}\Bigr)
= \Bigl(n + \frac12\Bigr)\ln n - n + 1 + E +
O\Bigl(\frac1n\Bigr),
\]
ومنه $d_n = 1 + E + O\bigl(\frac1n\bigr)$: أي تقارب $(d_n)$
--- وهو الخطوة 1 في \cref{thm:b2:comparison:stirling} --- مع
المعدل $O(1/n)$. (وتعطي قيمة النهاية عند ستيرلينغ أن $E =
\ln\sqrt{2\pi} - 1$.)

\textbf{10.} $f(t) = t^{-1/2}$: $f''(t) = \frac34 t^{-5/2}$،
وهي قابلة للمكاملة بإطلاق. والسؤال 6:
\[
\sum_{k=1}^n \frac1{\sqrt k}
= 2\sqrt n - 2 + \frac{1 + \frac1{\sqrt n}}2 + E +
O\bigl(n^{-3/2}\bigr)
= 2\sqrt n + c + \frac{1}{2\sqrt n} +
O\bigl(n^{-3/2}\bigr),
\]
مع $c = E - \frac32$. وعند $n = 10^4$: $2\sqrt n = 200$، $c
\approx -1.46035$، $\frac1{2\sqrt n} = 0.005$: فالمتوقع
$198.54465$، وفعلًا $\sum_{k\leq10^4}k^{-1/2} =
198.544645\dots$ --- ثلاثة حدود، وسبعة أرقام.

\textbf{11.} $t \mapsto t\ln t$ متصلة ومتزايدة تمامًا
على $\intco1\infty$ (بمشتق $\ln t + 1 \geq 1$)،
من $0$ إلى $+\infty$: فيوجد $x_n$ وحيد، و$x_n \to
\infty$ (وإلا لبقيت $x_n\ln x_n$ محدودة). وبأخذ
اللوغاريتمات في $x_n\ln x_n = n$: $\ln x_n + \ln\ln x_n = \ln n$؛
وبما أن $\ln\ln x_n = o(\ln x_n)$، تعطي القسمة على $\ln x_n$
أن $\frac{\ln n}{\ln x_n} \to 1$: أي $\ln x_n \sim \ln n$.

\textbf{12.} من $x_n = \frac{n}{\ln x_n}$ ومن $\ln x_n \sim
\ln n$: $x_n \sim \frac{n}{\ln n}$. والمرور التالي: $\ln\ln x_n =
\ln\bigl(\ln n\,(1 + o(1))\bigr) = \ln\ln n + o(1)$، ومنه $\ln
x_n = \ln n - \ln\ln n + o(1)$ و
\[
x_n = \frac{n}{\ln n - \ln\ln n + o(1)}
= \frac{n}{\ln n}\cdot\frac{1}{1 - \frac{\ln\ln n +
o(1)}{\ln n}}
= \frac{n}{\ln n}\Bigl(1 + \frac{\ln\ln n}{\ln n} +
o\Bigl(\frac{\ln\ln n}{\ln n}\Bigr)\Bigr).
\]

\textbf{13.} عند $n = 10^6$: $\frac{n}{\ln n} \approx 72\,382$
(بخطأ $18\%$)، ويعطي الحدّان $\approx 86\,140$ (بخطأ
$1.9\%$)، في مقابل القيمة الحقيقية $x \approx 87\,848$. والمكسب في كل
مرور ليس إلا العامل $\frac{\ln\ln n}{\ln n} \approx
\frac{2.63}{13.8} \approx 0.19$: فالسلالم اللوغاريتمية تتقارب
ببطء يثير الجنون --- وهي واقعة حياة أينما كانت الأعداد الأولية
حاضرة.

\textbf{14.} توجد $n$ عددًا أوليًا بالضبط $\leq p_n$ (وهي
$p_1, \dots, p_n$): أي $\pi(p_n) = n$. وتعطي مبرهنة الأعداد الأولية
(المقبولة؛ مجلد السنة الثالثة) أن $n = \pi(p_n) \sim
\frac{p_n}{\ln p_n}$، أي $p_n \sim n\ln p_n$: وهذه هي
المعادلة $x\ln x \approx n$ مقروءةً في الاتجاه المعاكس. وبأخذ اللوغاريتمات:
$\ln p_n = \ln n + \ln\ln p_n + o(1)$، ويفرض $\ln\ln p_n =
o(\ln p_n)$ أن $\ln p_n \sim \ln n$ كما في السؤال 11.
وبالتعويض رجوعًا:
\[
p_n \sim n\ln p_n = n\,\ln n\,\frac{\ln p_n}{\ln n} \sim n\ln
n .
\]

\textbf{15.} (أ) نثبّت $\varepsilon > 0$؛ ومن أجل $k$ الكبيرة، $(1 -
\varepsilon)k\ln k \leq p_k \leq (1 + \varepsilon)k\ln k$. وبالمقارنة مع الدالة المتزايدة $t\ln t$
(بحصر من نمط \cref{thm:b2:comparison:seriesintegral})،
$\sum_{k\leq n}k\ln k = \int_1^n t\ln t\,\dd t + O(n\ln n) =
\frac{n^2\ln n}2 - \frac{n^2}4 + O(n\ln n) \sim
\frac{n^2\ln n}2$. ومنه $\sum_{k\leq n}p_k = \frac{n^2\ln
n}{2}(1 + O(\varepsilon) + o(1))$ من أجل كل $\varepsilon$:
أي $\sum_{k\leq n}p_k \sim \frac{n^2\ln n}2$. (ب) بمبرهنة الأعداد
الأولية، تكون بين الأعداد الصحيحة حتى $10^{100}$ نسبةٌ
$\sim \frac{1}{\ln 10^{100}} = \frac1{230.26\dots}$
أولية: فاحتمال أن يكون عدد صحيح عشوائي منتظم ذو $100$ رقمًا أوليًا
هو نحو $\frac1{230}$.

\textbf{16.} على $\intoo{n\pi}{n\pi + \frac\pi2}$، تكون $g(x) = \tan
x - \frac1x$ متصلة ومتزايدة تمامًا (لأن $g' = 1 +
\tan^2x + \frac1{x^2} > 0$)، بالقيمة $g \to -\frac1{n\pi} < 0$ عند
الطرف الأيسر و$g \to +\infty$ عند الأيمن: فجذر واحد بالضبط
$x_n$. وبما أن $\tan z_n = \tan x_n = \frac1{x_n} \to 0$
مع $z_n \in \intoo{0}{\frac\pi2}$: فإن $z_n \to 0^+$.

\textbf{17.} $\tan z_n \sim z_n$ و$\frac1{x_n} \sim
\frac1{n\pi}$: ومنه $z_n \sim \frac1{n\pi}$.

\textbf{18.} $z_n = \arctan\frac1{x_n}$ و$\arctan u = u +
O(u^3)$. ومع $z_n = O(\frac1n)$:
\[
\frac1{x_n} = \frac{1}{n\pi}\cdot\frac1{1 + \frac{z_n}{n\pi}}
= \frac1{n\pi} - \frac{z_n}{n^2\pi^2} +
O\Bigl(\frac1{n^4}\Bigr)
= \frac1{n\pi} + O\Bigl(\frac1{n^3}\Bigr),
\]
ومنه $z_n = \frac1{n\pi} + O\bigl(\frac1{n^3}\bigr)$: فدرجة
السلّم $\frac{c}{n^2}$ تحمل المعامل $0$، لأن
أول تصحيح للمقدار $\frac1{x_n}$ هو نفسه من الحجم
$\frac{z_n}{n^2} = O(n^{-3})$.

\textbf{19.} نُدخِل $z_n = \frac1{n\pi} + O(n^{-3})$ في
الصيغة السابقة:
\[
\frac{1}{x_n} = \frac{1}{n\pi} - \frac{1}{n^3\pi^3} +
O\Bigl(\frac1{n^5}\Bigr),
\]
ثم $z_n = \arctan\frac1{x_n} = \frac1{x_n} -
\frac{1}{3}\Bigl(\frac1{x_n}\Bigr)^3 + O\Bigl(\frac1{n^5}\Bigr)
= \frac1{n\pi} - \frac{1}{n^3\pi^3} - \frac{1}{3n^3\pi^3} +
O\Bigl(\frac1{n^5}\Bigr)$:
\[
x_n = n\pi + \frac{1}{n\pi} - \frac{4}{3\pi^3n^3} +
O\Bigl(\frac1{n^5}\Bigr).
\]

\textbf{20.} عند $n = 3$: حدٌّ واحد يعطي $9.53088$، وثلاثة حدود تعطي
$9.52929$، والجذر الحقيقي $9.52933$: فالخطآن $1.5\cdot10^{-3}$ و
$5\cdot10^{-5}$. والمقابلة: من أجل $\tan x = x$ يجب أن يجعل الجذر
$\tan$ ضخمًا، ومن ثم يلتصق بالطرف \emph{الأيمن} $n\pi +
\frac\pi2$ من النافذة، على مسافة $\sim\frac1{n\pi}$
قبل المقارب؛ ومن أجل $x\tan x = 1$ يجب أن يجعل الجذر
$\tan$ ضئيلًا، ومن ثم يقع بُعيد الطرف \emph{الأيسر} $n\pi$،
على مسافة $\sim\frac1{n\pi}$ بعد الصفر. الطريقة نفسها،
وجغرافيا معكوسة.

\textbf{21.} $\sin u < u$ على $\intoo0\pi$، و$\sin$ يرسل
$\intoo0\pi$ في $\intoc01 \subseteq \intoo0\pi$: فبعد خطوة
واحدة $u_1 \in \intoc{0}{1}$، ثم تتناقص $(u_n)$ وتُحدّ
من أسفل بالقيمة $0$: فتتقارب إلى نقطة ثابتة للتطبيق $\sin$،
أي إلى $0$. والنشر: $\sin u = u(1 - \frac{u^2}6 +
o(u^2))$، ومنه
\[
\frac{1}{u_{n+1}^2} - \frac1{u_n^2}
= \frac{1}{u_n^2}\Bigl(\bigl(1 - \tfrac{u_n^2}6 +
o(u_n^2)\bigr)^{-2} - 1\Bigr)
= \frac{1}{u_n^2}\Bigl(\frac{u_n^2}{3} + o(u_n^2)\Bigr)
\longrightarrow \frac13 .
\]

\textbf{22.} بتشيزارو (مجلد السنة الأولى)، يتقارب متوسط
التزايدات إلى النهاية نفسها:
\[
\frac{1}{n}\cdot\frac{1}{u_n^2}
= \frac1n\Bigl(\frac1{u_0^2} + \sum_{k=0}^{n-1}
\Bigl(\frac1{u_{k+1}^2} - \frac1{u_k^2}\Bigr)\Bigr)
\longrightarrow \frac13 ,
\]
ومنه $u_n^2 \sim \frac3n$، وبما أن كل الحدود موجبة، $u_n
\sim \sqrt{3/n}$.

\textbf{23.} حسب السؤال 7، $\abs{H_n - \ln n - \gamma -
\frac1{2n}} \leq \frac1{8n^2}$. وعند $n = 10^6$ يكون هذا الحدّ
$\frac{1}{8\cdot10^{12}} = 1.25\cdot10^{-13}$: فثلاثة حدود محسوبة
تعطي المجموع التوافقي ذا المليون حدّ إلى ثلاثة عشر
رقمًا، بشهادة خطأ صارمة تمامًا --- وهو كل
مغزى الصيغة المقاربة ذات الباقي الصريح.

\textbf{24.} (أ) صحيح: $\ln u_n - \ln v_n = \ln\frac{u_n}{v_n}
\to 0$ في حين أن $\ln v_n \to +\infty$، ومن ثم تؤول نسبة اللوغاريتمين
إلى $1$. (ب) خاطئ: $u_n = n + 1 \sim v_n = n$، لكن
$\eu^{u_n}/\eu^{v_n} = \eu \neq 1$. فالتكافؤ يحتمل
أخطاءً جمعية $o(1)$ في الأس، لا $O(1)$. (ج) خاطئ:
$f(x) = x + \sin(x^2) \sim g(x) = x$ عند $+\infty$، لكن $f'(x)
= 1 + 2x\cos(x^2)$ يتذبذب بلا حدّ في حين أن $g' = 1$:
فلا يلزم أن تكون مشتقات دالتين متكافئتين قابلة للمقارنة
إطلاقًا.

\textbf{25.} جرت حلقة \cref{met:b2:comparison:implicit}
على النحو نفسه ثلاث مرات: وطّن الجذر، واستخرج حدًّا
خامًا، وأعِد إدخاله من أجل الرتبة التالية --- على $\eu^x + x = n$
(الجزء الأول)، وعلى $x\ln x = n$ (الجزء الثالث)، وعلى $x\tan x = 1$ (الجزء
الرابع). ويرقّي تصحيح شبه المنحرف مقارنةَ المتسلسلة
بالتكامل من ``الفرق يتقارب'' إلى حدٍّ صريح
$\frac{f(1) + f(n)}2$ بباقٍ مصدَّق $O(\int_n^\infty
\abs{f''})$ --- أي ثوابت وأشرطة خطأ بدل
مجرد التقارب. والجسر إلى الأعداد الأولية قلبٌ محض: فمبرهنة
الأعداد الأولية تقول $\pi(x)\ln x \approx x$، ومن ثم فإن $p_n$،
المعرَّف بالعلاقة $\pi(p_n) = n$، يحلّ معادلة من نمط $x\ln x = n$ ---
ويرث مقارباتها. وقد شرّعت القاعدة (أ) في السؤال 24
كل انتقال من $u_n \sim v_n$ إلى $\ln u_n \sim
\ln v_n$ (السؤالان 11 و14)؛ وخطأُ (ب) هو السبب في أننا
لا نأخذ أسّي التكافؤات أبدًا. والقمتان: صيغة أويلر--ماكلورين
من الرتبة الأولى (السؤال 6)، والقانون المقارب
$p_n \sim n\ln n$ للعدد الأولي رقم $n$ (السؤال 14).
\end{solution}
