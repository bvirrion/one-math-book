\section*{Capítulo \ref{ch:b2:diffeq} --- Equações diferenciais}

\begin{solution}{exo:b2:diffeq:1}
Primeira matriz: \omterm{def:b2:reduction:eigen}{autovalores} $1, 2$, \omterm{def:b2:reduction:eigen}{autovetores} $(1,0)$ e $(1,1)$.
Decomponha $X_0 = (1,0) = 1\cdot(1,0) + 0\cdot(1,1)$: a solução
é
\[
X(t) = \eu^{t}\begin{pmatrix}1\\ 0\end{pmatrix} .
\]
(O vetor inicial é ele próprio um \omterm{def:b2:reduction:eigen}{autovetor}.)

Segunda: $A = 2I + N$, $N = E_{12}$, $N^2 = 0$: $\eu^{tA} =
\eu^{2t}(I + tN)$, logo
\[
X(t) = \eu^{2t}\begin{pmatrix} 1 & t\\ 0 & 1\end{pmatrix}
\begin{pmatrix}1\\ 0\end{pmatrix}
= \eu^{2t}\begin{pmatrix}1\\ 0\end{pmatrix}.
\]
\end{solution}

\begin{solution}{exo:b2:diffeq:2}
$y' = \sin(ty)$: $\abs{\sin(ty) - \sin(tz)} \leq \abs t\,\abs{y -
z}$ --- \omterm{def:b2:metric:continuity}{lipschitziana} em $y$ uniformemente em todo segmento de tempos:
soluções globais únicas em $\R$ (aplique o teorema em cada
segmento).

$y' = y^2$: apenas localmente \omterm{def:b2:metric:continuity}{lipschitziana}; sem teorema global e, de fato,
$y(0) = 1$ explode em $t = 1$.

$y' = \abs y$: $\abs\cdot$ é $1$-lipschitziana: existência e unicidade
globais. Com $y(0) = 0$: $y \equiv 0$ (unicidade!). Com $y(0)
= 1$: $y$ permanece positiva (não pode cruzar a solução nula), logo $y' =
y$: $y = \eu^t$.
\end{solution}

\begin{solution}{exo:b2:diffeq:3}
Se $y(t_1) = z(t_1)$ em algum instante, então $y$ e $z$ resolvem o mesmo
problema de Cauchy em $t_1$: a unicidade força $y = z$ em seu intervalo
comum --- soluções distintas nunca se encontram.

Para $y' = y(1-y)$: as constantes $0$ e $1$ são soluções. Uma
solução que começa em $\intoo{0}{1}$ nunca pode atingir $0$ nem $1$
(ela colidiria com uma solução constante): ela permanece em
$\intoo{0}{1}$ e portanto é global (limitada: sem explosão --- por
exemplo, pelo critério do~\cref{exo:b2:diffeq:9}, ou porque o campo de vetores
é limitado na faixa em que fica presa).
\end{solution}

\begin{solution}{exo:b2:diffeq:4}
$(A - 2I)^2 = \begin{pmatrix}1 & 1\\ -1 & -1\end{pmatrix}^2 = 0$:
Dunford com $D = 2I$, $N = A - 2I$:
\[
\eu^{tA} = \eu^{2t}\,(I + tN)
= \eu^{2t}\begin{pmatrix} 1 + t & t\\ -t & 1 - t\end{pmatrix}.
\]
Variação das constantes com $B(s) = (\eu^{2s}, 0)^{\mathsf T}$:
\[
X(t) = \int_0^t \eu^{(t-s)A}B(s)\,\dd s
= \eu^{2t}\int_0^t \begin{pmatrix} 1 + (t-s)\\ -(t-s)
\end{pmatrix}\dd s
= \eu^{2t}\begin{pmatrix} t + \frac{t^2}{2}\\[2pt] -\frac{t^2}{2}
\end{pmatrix},
\]
usando $\eu^{(t-s)A}B(s) = \eu^{2(t-s)}(I + (t-s)N)\,(\eu^{2s},
0)^{\mathsf T} = \eu^{2t}\bigl(1 + (t-s),\, -(t-s)\bigr)^{\mathsf
T}$. (Confira: $X(0) = 0$; $X' - AX = (\eu^{2t}, 0)^{\mathsf T}$
por derivação.)
\end{solution}

\begin{solution}{exo:b2:diffeq:5}
\emph{Série:} $A = \alpha I + \beta J$ com $J =
\begin{pmatrix}0 & -1\\ 1 & 0\end{pmatrix}$, $J^2 = -I$; as duas
parcelas comutam, de modo que $\eu^{tA} = \eu^{\alpha t}\,\eu^{\beta tJ}$,
e a série de $\eu^{\beta t J}$ se separa nas potências pares e ímpares
em $\cos(\beta t)I + \sin(\beta t)J$: a matriz de rotação com escala
enunciada.

\emph{Complexo:} identifique $(x, y) \in \R^2$ com $z = x + \iu y$;
o sistema $X' = AX$ lê-se $z' = (\alpha + \iu\beta)z$, cuja
solução $z(t) = \eu^{\alpha t}\eu^{\iu\beta t}z_0$ é exatamente a
espiral: módulo $\eu^{\alpha t}$, argumento avançando com velocidade
$\beta$.
\end{solution}

\begin{solution}{exo:b2:diffeq:6}
Seja $v(t) = \eu^{-k(t - t_0)}\int_{t_0}^t u$. Então
\[
v'(t) = \eu^{-k(t-t_0)}\Bigl(u(t) - k\int_{t_0}^t u\Bigr)
\leq C\,\eu^{-k(t-t_0)},
\]
pela hipótese. Integrando de $t_0$ a $t$ ($v(t_0) = 0$):
$v(t) \leq \frac{C}{k}\bigl(1 - \eu^{-k(t - t_0)}\bigr)$, isto é,
$k\int_{t_0}^t u \leq C\bigl(\eu^{k(t-t_0)} - 1\bigr)$; reinjetando
isso na hipótese: $u(t) \leq C\eu^{k(t-t_0)}$.

Unicidade/dependência: duas soluções $y, z$ da equação integral
satisfazem
\[
\norm{y(t) - z(t)} \leq \norm{y_0 - z_0} +
k\int_{t_0}^{t}\norm{y - z},
\]
e Gronwall com $C = \norm{y_0 - z_0}$ dá a cota
exponencial; $C = 0$ dá a unicidade.
\end{solution}

\begin{solution}{exo:b2:diffeq:7}
Substitua $y = y_1 z$ com $y_1 = \frac{\sin t}{t}$: a fórmula geral de
redução (\cref{prop:b2:diffeq:secondorder}) dá, para $u
= z'$,
\[
y_1 u' + \Bigl(2y_1' + \frac{2}{t}\,y_1\Bigr)u = 0
\]
(com a equação normalizada como $y'' + \frac2t y' + y = 0$). Calcule
$2y_1' + \frac2t y_1 = 2\,\frac{t\cos t - \sin t}{t^2} +
\frac{2\sin t}{t^2} = \frac{2\cos t}{t}$: assim
\[
\frac{u'}{u} = -\frac{2\cos t}{t}\cdot\frac{t}{\sin t}
= -2\cot t
\quad\Longrightarrow\quad
u = \frac{1}{\sin^2 t}
\quad (\text{a menos de constante}),
\]
e $z = -\cot t$, dando $y_2 = y_1 z = -\frac{\cos t}{t}$.
Solução geral em $\intoo{0}{\pi}$:
\[
y(t) = \alpha\,\frac{\sin t}{t} + \beta\,\frac{\cos t}{t} .
\]
(Essas são as funções esféricas de Bessel de \omterm{def:b2:structures:generated}{ordem} zero.)
\end{solution}

\begin{solution}{exo:b2:diffeq:8}
Dunford: $A = D + N$ comutando, $D$ \omterm{def:b2:reduction:diag}{diagonalizável} com os mesmos
\omterm{def:b2:reduction:eigen}{autovalores}, $N$ nilpotente, de modo que
\[
\eu^{tA} = \eu^{tD}\,\eu^{tN},
\qquad
\eu^{tN} = \sum_{k < n} \frac{t^kN^k}{k!}
\ \text{(um polinômio matricial em } t).
\]
Seja $-2\alpha = \max_i \Re\lambda_i < 0$. Numa base que diagonaliza
$D$, $\vertiii{\eu^{tD}} \leq \eu^{-2\alpha t}$ (entradas
$\eu^{t\lambda_i}$ de módulo $\eu^{t\Re\lambda_i}$); \omterm{def:b2:nvs:norm}{normas} em
bases diferentes diferem por constantes. Logo
\[
\norm{X(t)} \leq \vertiii{\eu^{tA}}\,\norm{X_0}
\leq C'\,\eu^{-2\alpha t}\,(1 + t)^{n-1}\,\norm{X_0}
\leq C\,\eu^{-\alpha t}\norm{X_0} ,
\]
absorvendo o polinômio num fator exponencial
($\eu^{-\alpha t}(1+t)^{n-1} \to 0$, logo limitada).
\end{solution}

\begin{solution}{exo:b2:diffeq:9}
Seja $y$ uma solução maximal em $\intco{0}{T}$, $T \leq \infty$,
e suponha $T < \infty$. A forma integral dá, para $t < T$,
\[
\norm{y(t)} \leq \norm{y_0} + \int_0^t \bigl(a\norm{y(s)} +
b\bigr)\dd s
\leq \bigl(\norm{y_0} + bT\bigr) + a\int_0^t\norm{y(s)}\,\dd s ,
\]
e Gronwall majora $\norm{y(t)} \leq (\norm{y_0} +
bT)\,\eu^{aT} =: M$ em $\intco{0}{T}$: a solução permanece numa
bola \omterm{def:b2:metric:compact}{compacta}. Então $y' = f(t, y)$ é limitada perto de $T$, de modo que $y$ é
\omterm{def:b2:metric:continuity}{lipschitziana} perto de $T$ e se estende \omterm{def:b2:metric:continuity}{continuamente} a $T$ (critério
de Cauchy); resolver o problema de Cauchy em $(T, y(T))$ prolonga $y$
além de $T$, contradizendo a maximalidade. Logo $T = \infty$: sem
fuga em tempo finito sob crescimento linear.
\end{solution}

\begin{solution}{exo:b2:diffeq:10}
Raízes características de $r^2 - 3r + 2$: $1$ e $2$, de modo que as
soluções homogêneas são $a\eu^t + b\eu^{2t}$. O palpite
$\alpha\eu^t$ falha porque $\eu^t$ já resolve a
equação homogênea (a raiz $r = 1$ ``entra em \omterm{pb:b2:diffeq:1}{ressonância}'' com o
membro da direita). Com $y = \alpha t\,\eu^t$: $y' = \alpha(1 +
t)\eu^t$, $y'' = \alpha(2 + t)\eu^t$, e
\[
y'' - 3y' + 2y = \alpha\eu^t\bigl(2 + t - 3 - 3t + 2t\bigr)
= -\alpha\,\eu^t :
\]
$\alpha = -1$, $y_p = -t\,\eu^t$. Solução geral: $y = a\eu^t
+ b\eu^{2t} - t\eu^t$. Dados iniciais $y(0) = y'(0) = 0$: $a + b
= 0$ e $a + 2b - 1 = 0$: $b = 1$, $a = -1$:
\[
y(t) = \eu^{2t} - (1 + t)\,\eu^{t} .
\]

\end{solution}

\begin{solution}{exo:b2:diffeq:11}
$A = \lambda I + N$ com $N = E_{12} + E_{23}$: $N^2 = E_{13}$,
$N^3 = 0$, e $\lambda I$ comuta com $N$:
\[
\eu^{tA} = \eu^{\lambda t}\Bigl(I + tN +
\frac{t^2}{2}N^2\Bigr)
= \eu^{\lambda t}\begin{pmatrix}
1 & t & \frac{t^2}{2}\\
0 & 1 & t\\
0 & 0 & 1
\end{pmatrix}.
\]
Soluções: $X(t) = \eu^{\lambda t}\bigl(X_0 + tNX_0 +
\frac{t^2}2N^2X_0\bigr)$ --- cada componente é $\eu^{\lambda
t}$ vezes um polinômio de grau $\leq 2$. A cota do grau é
o índice de nilpotência menos um: a série de $\eu^{tN}$
trunca em $N^2$.
\end{solution}

\begin{solution}{exo:b2:diffeq:12}
\begin{enumerate}
  \item Se $X(T) = X(0)$, então $Y(t) = X(t + T)$ resolve $Y' =
        AY + B(t + T) = AY + B(t)$ com $Y(0) = X(0)$:
        a unicidade (\cref{thm:b2:diffeq:linear}) dá $Y = X$,
        isto é, $X$ é $T$-periódica. A recíproca é trivial.
  \item Triangularize sobre $\C$: $A = PT'P^{-1}$ com $T'$ triangular
        superior, de diagonal $(\lambda_i)$. Toda potência de uma
        matriz triangular é triangular com diagonal
        $(\lambda_i^k)$, logo $\eu^{TA} = P\eu^{TT'}P^{-1}$ é
        triangular na mesma base com diagonal
        $(\eu^{T\lambda_i})$: esses são os \omterm{def:b2:reduction:eigen}{autovalores}. Então
        $I - \eu^{TA}$ é invertível se e somente se $\eu^{T\lambda} \neq
        1$ para todo \omterm{def:b2:reduction:eigen}{autovalor}, isto é, se e somente se $T\lambda \notin
        2\iu\pi\Z$, que é a hipótese enunciada.
  \item Variação das constantes: $X(T) = \eu^{TA}X(0) +
        \int_0^T\eu^{(T-s)A}B(s)\dd s$, de modo que $X(T) = X(0)$ se lê
        \[
        \bigl(I - \eu^{TA}\bigr)X(0) =
        \int_0^{T}\eu^{(T-s)A}B(s)\,\dd s ,
        \]
        que tem solução única $X(0)$ sob a
        hipótese de invertibilidade: exatamente uma solução
        $T$-periódica. Para o oscilador harmônico ($\lambda =
        \pm\iu\omega$), o caso excluído é $\omega T \in
        2\pi\Z$: forçamento cujo período é múltiplo do
        período natural --- \omterm{pb:b2:diffeq:1}{ressonância}, como o problema de fim de semana
        quantifica.
\end{enumerate}
\end{solution}

\begin{solution}{pb:b2:diffeq:1}
\textbf{1.} O \omterm{def:b2:reduction:charpoly}{polinômio característico} é $\lambda^2 -
\tau\lambda + \delta$, de raízes
$\frac{\tau\pm\sqrt\Delta}{2}$. Se $\delta < 0$, então $\Delta =
\tau^2 - 4\delta > 0$ e as duas raízes reais têm produto
$\delta < 0$: sinais opostos. Se $\delta > 0$ e $\Delta \geq
0$: raízes reais de produto $> 0$ e soma $\tau$: ambas do
sinal de $\tau$. Se $\Delta < 0$: par conjugado $\alpha \pm
\iu\beta$ com $\alpha = \frac\tau2$, $\beta =
\frac{\sqrt{-\Delta}}2$.

\textbf{2.} $X(t) = a\,\eu^{\mu t}v_- + b\,\eu^{\lambda t}v_+$.
As órbitas com $a = 0$ (resp.\ $b = 0$) correm ao longo da reta própria
instável (resp.\ estável); todas as demais têm $\norm X \to \infty$
nas duas direções do tempo, assintóticas a $\R v_+$ quando $t \to
+\infty$ e a $\R v_-$ quando $t \to -\infty$: o retrato da
sela. A limitação em todo o $\R$ força $b = 0$ (senão há
explosão em $+\infty$) e $a = 0$ (em $-\infty$): só a
origem.

\textbf{3.} Com $\mu < \lambda < 0$, as duas exponenciais decaem:
$X(t) \to 0$. Se $b \neq 0$, ponha $\eu^{\lambda t}$ em evidência:
\[
X(t) = \eu^{\lambda t}\bigl(b\,v_\lambda + a\,\eu^{(\mu -
\lambda)t}v_\mu\bigr),
\qquad \eu^{(\mu-\lambda)t} \to 0 :
\]
a direção de $X(t)$ tende a $\R v_\lambda$, a direção
própria \emph{lenta} --- todas as órbitas exceto as do eixo rápido
chegam tangentes a ela (o painel direito dos retratos de fase
do capítulo).

\textbf{4.} Na base em que $A = \begin{pmatrix} \alpha &
-\beta\\ \beta & \alpha\end{pmatrix}$
(\cref{exo:b2:diffeq:5}; para os sistemas do oscilador da Parte II
essa forma é atingida por uma mudança de base real explícita), a
solução é $\eu^{\alpha t}$ vezes uma rotação de ângulo $\beta
t$: espirais logarítmicas, contraindo quando $\alpha = \frac\tau2
< 0$, expandindo quando $\tau > 0$, e curvas fechadas (elipses
nas coordenadas originais) quando $\tau = 0$: o centro.

\textbf{5.} $\Delta = 0$ dá o \omterm{def:b2:reduction:eigen}{autovalor} duplo $\lambda =
\frac\tau2$; por Cayley--Hamilton
(\cref{thm:b2:reduction:cayleyhamilton}), $(A - \lambda I)^2 =
0$, de modo que $N = A - \lambda I$ é nilpotente, comuta com
$\lambda I$, e $\eu^{tA} = \eu^{\lambda t}(I + tN)$. Se $N =
0$: $A = \lambda I$, e todas as semirretas são órbitas (nó estrela). Se $N
\neq 0$: $X(t) = \eu^{\lambda t}(X_0 + tNX_0)$ e, para $NX_0
\neq 0$, a direção converge para a única direção própria
$\operatorname{im}N$: nó impróprio. Isso \omterm{def:b2:metric:complete}{completa} o
retrato traço--determinante.

\textbf{6.} $\tau = -2\zeta\omega$, $\delta = \omega^2 > 0$,
$\Delta = 4\omega^2(\zeta^2 - 1)$. Assim: $0 < \zeta < 1$ dá
$\Delta < 0$, $\tau < 0$: espiral estável; $\zeta = 1$: $\Delta =
0$: nó estável degenerado; $\zeta > 1$: $\Delta > 0$, $\tau <
0$, $\delta > 0$: nó estável; $\zeta = 0$: $\tau = 0$,
$\delta > 0$: centro. Uma viagem vertical no plano em
$\delta = \omega^2$.

\textbf{7.} Raízes $r = -\zeta\omega \pm
\omega\sqrt{\zeta^2-1}$. Para $\zeta < 1$: $r = -\zeta\omega
\pm \iu\omega_d$, $\omega_d = \omega\sqrt{1-\zeta^2}$:
\[
x(t) = \eu^{-\zeta\omega t}\bigl(a\cos\omega_dt +
b\sin\omega_dt\bigr)
= R\,\eu^{-\zeta\omega t}\cos(\omega_dt - \varphi) .
\]
Para $\zeta = 1$: $x = (a + bt)\eu^{-\omega t}$. Para $\zeta >
1$: $x = a\eu^{r_-t} + b\eu^{r_+t}$, com as duas taxas negativas.
Os máximos sucessivos de $\abs x$ no caso subamortecido ocorrem em
instantes separados pelo pseudoperíodo $\frac{2\pi}{\omega_d}$
(mesma fase do cosseno), e sua razão é
$\eu^{-\zeta\omega\cdot2\pi/\omega_d} =
\eu^{-2\pi\zeta/\sqrt{1-\zeta^2}}$: o decremento logarítmico,
um medidor de amortecimento legível num osciloscópio.

\textbf{8.} Racionalizando,
\[
\abs{\lambda_{\mathrm{slow}}} = \omega\bigl(\zeta -
\sqrt{\zeta^2-1}\bigr)
= \frac{\omega}{\zeta + \sqrt{\zeta^2 - 1}} ,
\]
cujo denominador cresce com $\zeta \geq 1$: a taxa de
decaimento é máxima em $\zeta = 1$, onde vale $\omega$. Uma
porta superamortecida fecha sem bater, mas \emph{devagar};
o amortecimento crítico é o ótimo do engenheiro.

\textbf{9.} $E' = x'x'' + \omega^2xx' =
x'\bigl(-2\zeta\omega x' - \omega^2x\bigr) + \omega^2xx' =
-2\zeta\omega\,x'^2 \leq 0$. Se $x$ fosse periódica e
não constante, $E$ seria periódica e não crescente, logo
constante, forçando $x' \equiv 0$: $x$ constante, e então
$\omega^2x = 0$: $x \equiv 0$. Assim, para $\zeta > 0$, a única
solução periódica é o repouso: o amortecimento mata todo \omterm{def:b2:structures:sn}{ciclo}.

\textbf{10.} O centro $\zeta = 0$ mora na reta $\tau =
0$ do plano traço--determinante --- um conjunto de interior
vazio: uma perturbação arbitrariamente pequena da matriz
(qualquer amortecimento físico) tira $\tau$ do zero e transforma as
órbitas fechadas em espirais. A periodicidade do oscilador não
amortecido é, portanto, um fenômeno de fio de navalha, não
robusto.

\textbf{11.} Substituindo $x_p = \Re(z\eu^{\iu\gamma t})$ na
equação:
\[
\bigl(-\gamma^2 + 2\iu\zeta\omega\gamma +
\omega^2\bigr)z = F
\quad\Longrightarrow\quad
z = \frac{F}{\omega^2 - \gamma^2 + 2\iu\zeta\omega\gamma},
\]
logo $x_p = \abs z\cos(\gamma t - \varphi)$ com $\varphi =
\arg(\omega^2 - \gamma^2 + 2\iu\zeta\omega\gamma)$, isto é,
$\tan\varphi = \frac{2\zeta\omega\gamma}{\omega^2 -
\gamma^2}$, e o $A(\gamma) = \abs z$ enunciado.

\textbf{12.} A diferença de duas soluções resolve a
equação homogênea, que para $\zeta > 0$ decai a $0$
(questão 7): toda solução é $x_p$ mais um transiente
que se anula no infinito. O regime permanente é um atrator global:
as condições iniciais são esquecidas, restam apenas $A(\gamma)$ e o
atraso de fase $\varphi$.

\textbf{13.} Minimize $g(u) = (\omega^2 - u)^2 +
4\zeta^2\omega^2u$ em $u = \gamma^2 \geq 0$: $g'(u) =
-2(\omega^2 - u) + 4\zeta^2\omega^2 = 0$ em $u = \omega^2(1 -
2\zeta^2)$, interior se e somente se $\zeta < \frac{1}{\sqrt2}$. Aí
\[
g(u_*) = 4\zeta^4\omega^4 + 4\zeta^2\omega^4(1 - 2\zeta^2)
= 4\zeta^2\omega^4(1 - \zeta^2),
\qquad
A(\gamma_*) = \frac{F}{2\zeta\omega^2\sqrt{1 - \zeta^2}} .
\]
Contra a resposta estática $A(0) = \frac{F}{\omega^2}$:
amplificação $\frac{1}{2\zeta\sqrt{1-\zeta^2}} \approx
\frac{1}{2\zeta}$ para $\zeta$ pequeno --- um
sistema pouco amortecido perto de $\gamma_* \approx \omega$ multiplica a entrada
cem vezes quando $\zeta = 0.005$.

\textbf{14.} O $x$ enunciado satisfaz $x(0) = x'(0) = 0$ e
\[
x'' + \omega^2 x = \frac{F(\omega^2 -
\gamma^2)\cos\gamma t}{\omega^2 - \gamma^2} = F\cos\gamma t
\]
(as partes $\cos\omega t$ se cancelam). A forma de produto decorre
de $\cos p - \cos q = 2\sin\frac{q+p}{2}\sin\frac{q-p}2$
com $p = \gamma t$, $q = \omega t$. Para $\gamma$ próximo de
$\omega$, o fator $\sin\frac{(\omega-\gamma)t}2$ é um envelope
lento que modula a oscilação rápida
$\sin\frac{(\omega+\gamma)t}2$: batimentos, de amplitude
$\frac{2F}{\abs{\omega^2-\gamma^2}}$ --- grande, mas limitada.

\textbf{15.} Para $x_p = \frac{F}{2\omega}t\sin\omega t$:
\[
x_p'' = \frac{F}{2\omega}\bigl(2\omega\cos\omega t -
\omega^2t\sin\omega t\bigr)
= F\cos\omega t - \omega^2x_p :
\]
é solução. E, com $t$ fixo, fazendo $\gamma \to \omega$ na
questão 14:
\[
\frac{2F\sin\frac{(\omega-\gamma)t}2
\sin\frac{(\omega+\gamma)t}{2}}
{(\omega-\gamma)(\omega+\gamma)}
\longrightarrow
\frac{2F\cdot\frac{(\omega-\gamma)t}2\big/(\omega-\gamma)
\cdot\sin\omega t}{2\omega}
= \frac{F\,t\sin\omega t}{2\omega} .
\]
A amplitude cresce linearmente sem limite: a catástrofe de
\omterm{pb:b2:diffeq:1}{ressonância} --- a razão pela qual soldados quebram o passo em pontes.

\textbf{16.} A onda quadrada carrega harmônicos em toda frequência
ímpar $n = 1, 3, 5, \dots$; por linearidade, cada harmônico
$n$ é amplificado pela resposta do oscilador em $\gamma = n$.
Para $\omega = 3$, o terceiro harmônico atinge a \omterm{pb:b2:diffeq:1}{ressonância} exatamente.
Os engenheiros temem entradas quadradas (e dente de serra) porque elas
excitam \emph{todos} os harmônicos ímpares de uma vez: qualquer que seja a frequência
natural da estrutura, algum harmônico está à espera dela.

\textbf{17.} $W' = y_1y_2'' - y_1''y_2 = -qy_1y_2 + qy_1y_2 =
0$: $W$ é constante (Liouville com uma matriz companheira de
traço nulo). $W = 0$ num ponto torna os dados iniciais de $y_2$
proporcionais aos de $y_1$, logo $y_2$ proporcional a
$y_1$ (unicidade); $W \neq 0$ se e somente se independentes. Se $y(t_0) =
y'(t_0) = 0$, então $y \equiv 0$ (unicidade): uma solução não nula
tem zeros simples, e um zero simples é isolado
($y'$ de sinal fixo por perto).

\textbf{18.} Entre zeros consecutivos $a < b$, $y_1$ mantém
um sinal, digamos $y_1 > 0$ em $\intoo ab$: então $y_1'(a) > 0$ e
$y_1'(b) < 0$ (zeros simples). Avaliando a constante $W =
y_1y_2' - y_1'y_2$ em $a$ e $b$:
\[
W = -y_1'(a)\,y_2(a) = -y_1'(b)\,y_2(b) ,
\]
de modo que $y_2(a)$ e $y_2(b)$ têm sinais opostos ($W \neq 0$
proíbe que qualquer um se anule): $y_2$ se anula em $\intoo ab$
(valores intermediários). Ela não pode se anular duas vezes ali: dois zeros
de $y_2$ ladeariam um zero de $y_1$ pelo mesmo argumento com
os papéis trocados, contradizendo o fato de serem consecutivos: exatamente
um zero --- intercalação.

\textbf{19.} Suponha que $z$ não tenha zero em $\intoo ab$; substituindo
$y, z$ por seus opostos, suponha $y > 0$ e $z > 0$ em
$\intoo ab$. Ponha $\varphi = yz' - y'z$: $\varphi' = yz'' -
y''z = (q_1 - q_2)\,yz \leq 0$ em $\intoo ab$: $\varphi$ é
não crescente. Mas $\varphi(a) = -y'(a)z(a) \leq 0$ (pois
$y'(a) > 0$, $z(a) \geq 0$) e $\varphi(b) = -y'(b)z(b) \geq
0$ (pois $y'(b) < 0$, $z(b) \geq 0$): uma função não crescente
que vai de $\leq 0$ a $\geq 0$ anula-se identicamente, logo
$(q_1 - q_2)yz \equiv 0$ em $\intoo ab$. Se $q_1 < q_2$
em algum ponto de $\intoo ab$, isso é absurdo ($y, z > 0$ ali):
$z$ tem de se anular estritamente dentro. Em geral ($q_1 \leq q_2$),
ou $z$ se anula em $\intoo ab$, ou $\varphi \equiv 0$
força $z$ proporcional a $y$, que se anula em $a$ e $b$:
em todos os casos $z$ tem zero em $\intcc ab$.

\textbf{20.} Cota superior: compare $y$ (coeficiente $q \geq
m^2$) com $u(t) = \sin(m(t - a))$ (coeficiente $m^2 \leq q$,
de modo que $y$ faz o papel de $z$ na questão 19): se $y$ não tivesse
zero em $\intoc{a}{a + \pi/m}$, os zeros $a$ e $a +
\frac\pi m$ de $u$ seriam consecutivos com $y \neq 0$
entre eles, contradizendo a questão 19: os zeros consecutivos de
$y$ estão a distância $\leq \frac\pi m$. Cota inferior: se dois
zeros consecutivos $a < b$ de $y$ tivessem $b - a < \frac\pi M$,
então $z(t) = \sin(M(t-a))$ (coeficiente $M^2 \geq q$) teria de
se anular em $\intcc ab \subset \intoo{a}{a + \pi/M}
\cup\{a\}$, onde seu único zero é o próprio $a$ --- mas a
questão 19 aplicada em $\intoo{a}{b}$ com estrita nas
extremidades dá um zero em $\intcc ab$, e $z > 0$ em
$\intoc ab$: contradição. Logo $\frac\pi M \leq b - a \leq
\frac\pi m$; para $q \equiv \omega^2$ as duas cotas colapsam no
espaçamento exato $\frac\pi\omega$ do oscilador
harmônico.

\textbf{21.} Com $u = ty$: $u'' = ty'' + 2y'$, logo $ty'' + 2y'
+ ty = u'' + u = 0$: $u = A\sin t + B\cos t = R\sin(t +
\varphi)$, e $y = \frac{u}{t}$ recupera $\frac{\sin t}t$ e
$\frac{\cos t}t$ (\cref{exo:b2:diffeq:7}) sem redução de
\omterm{def:b2:structures:generated}{ordem}. Os zeros de qualquer solução não nula são os de
$R\sin(t + \varphi)$: espaçados exatamente de $\pi$ --- a
filosofia de Sturm em ação: o coeficiente $q \equiv 1$ dita
os zeros, com fórmula ou sem.

\textbf{22.} Ponha $x(t) =
\frac1\omega\int_0^t\sin(\omega(t-s))F(s)\dd s$. Então $x(0) =
0$;
\[
x'(t) = \frac1\omega\sin(0)F(t) +
\int_0^t\cos(\omega(t-s))F(s)\dd s
= \int_0^t\cos(\omega(t-s))F(s)\dd s ,
\]
logo $x'(0) = 0$; e $x''(t) = F(t) -
\omega\int_0^t\sin(\omega(t-s))F(s)\dd s = F(t) - \omega^2x(t)$
(derivação de uma \omterm{thm:b2:integration:continuity}{integral com parâmetro} de limite variável,
como no capítulo de integração). Com $F(s) = F\cos\omega s$,
produto-soma dá
\[
\int_0^t\sin(\omega(t-s))\cos(\omega s)\dd s
= \frac12\int_0^t\bigl(\sin\omega t + \sin(\omega t -
2\omega s)\bigr)\dd s
= \frac{t}{2}\sin\omega t ,
\]
(a segunda peça integra zero), logo $x =
\frac{F}{2\omega}t\sin\omega t$: a questão 15 de novo, agora por
Duhamel.

\textbf{23.} Na forma de sistema $X' = AX + (0, F(t))^{\mathsf T}$
com $\operatorname{Sp}A$ de partes reais negativas ($\zeta >
0$): a variação das constantes e \cref{exo:b2:diffeq:8}
($\vertiii{\eu^{tA}} \leq C\eu^{-\alpha t}$) dão
\[
\norm{X(t)} \leq C\eu^{-\alpha t}\norm{X_0}
+ \int_0^t C\eu^{-\alpha(t-s)}\norm{F}_\infty\dd s
\leq C\norm{X_0} + \frac{C\norm F_\infty}{\alpha} :
\]
entrada limitada, saída limitada --- \omterm{def:b2:funcseq:def}{uniformemente} nos dados
iniciais após o transiente.

\textbf{24.} Para $\zeta = 0$ e $\gamma \neq \omega$,
a solução da questão 14 é limitada, e somar qualquer solução
homogênea (limitada: as órbitas do centro são círculos) a mantém
limitada; em $\gamma = \omega$, a questão 15 cresce linearmente. Assim,
para o oscilador não amortecido, a limitação sob forçamento
periódico falha em exatamente uma frequência --- a \omterm{pb:b2:diffeq:1}{ressonância} ---
enquanto a questão 23 mostra que qualquer amortecimento positivo restaura
a limitação para \emph{todas} as entradas limitadas.

\textbf{25.} (i) O plano traço--determinante classifica toda
dinâmica linear plana autônoma, e o oscilador da Parte II
percorre uma reta vertical dele; o forçamento deixa o plano
(não autônomo), e Duhamel assume. (ii) $\gamma_*$ é
a frequência que o sistema prefere, $A(\gamma_*)$ o preço de
excitá-la, e $\frac{1}{2\zeta}$ o fator de amplificação
--- a agudeza da \omterm{pb:b2:diffeq:1}{ressonância} que os engenheiros chamam de fator
de qualidade. (iii) Os teoremas de Sturm leem a oscilação a partir do sinal
e do tamanho de $q$ apenas: eles governam equações (Bessel,
Schrödinger) cujas soluções não têm fórmulas elementares.
(iv) Os \omterm{def:b2:diffeq:wronskian}{wronskianos} constantes (questão 17, via Liouville) e
a estabilidade com entrada limitada (questão 23) são reutilizados em silêncio
sempre que o livro encontra equações de coeficientes variáveis ou
sistemas perturbados.
\end{solution}
