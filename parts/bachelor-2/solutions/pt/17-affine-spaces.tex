\section*{Capítulo \ref{ch:b2:affine} --- Espaços afins}

\begin{solution}{exo:b2:affine:1}
$\{x + y + z = 1\}$: plano \omterm{def:b2:affine:subspace}{afim}, de direção o plano vetorial $\{x
+ y + z = 0\}$, dimensão $2$. Acrescentando $x - z = 3$: uma reta \omterm{def:b2:affine:subspace}{afim}
(duas equações independentes), direção $\{x + y + z = 0,\ x = z\}
= \operatorname{Vect}\bigl((1, -2, 1)\bigr)$, dimensão $1$. $\{x^2
+ y^2 = 1\}$: um cilindro --- não estável por \omterm{def:b2:affine:barycenter}{baricentros} (o
ponto médio de $(1,0,0)$ e $(-1,0,0)$ é a origem, fora do
cilindro): não é \omterm{def:b2:affine:subspace}{afim}. Um sistema compatível $MX = B$: subespaço
\omterm{def:b2:affine:subspace}{afim} $X_0 + \ker M$ de dimensão $\dim\ker M$, como lembrado na
\cref{def:b2:affine:subspace}.
\end{solution}

\begin{solution}{exo:b2:affine:2}
$C = \operatorname{bar}(A, 1-t; B, t)$ significa $\vect{AC} =
t\,\vect{AB}$: a existência de tal $t$ é exatamente a ligação de
$\vect{AC}$ com $\vect{AB} \neq 0$, isto é, o alinhamento. Posições: $t
= \frac12$: ponto médio; $t = 2$: além de $B$, à distância de $B$ dele
($\vect{AC} = 2\vect{AB}$); $t = -1$: o \omterm{def:b2:quadratic:adjoint}{simétrico} de $B$
por $A$.
\end{solution}

\begin{solution}{exo:b2:affine:3}
$M$ é a matriz da projeção sobre $\operatorname{Vect}(1,1)$ ao longo de
$(1,-1)$ (confira $M^2 = M$). Imagem de $f$: $\{MX + C\} = C +
\operatorname{im} M$: a reta \omterm{def:b2:affine:subspace}{afim} por $(1,0)$ dirigida por
$(1,1)$. Pontos fixos: $X = MX + C$, isto é, $(I - M)X = C$; mas $C =
(1, 0)^{\mathsf T}$ e $\operatorname{im}(I - M) =
\operatorname{Vect}(1,-1)$; será que $(1,0)$ está nela? $(1, 0) =
\alpha(1,-1)$ força $\alpha = 1$ e $0 = -1$: não. Nenhum ponto
fixo. E
\[
f(f(X)) = M(MX + C) + C = MX + MC + C = f(X) + MC,
\qquad MC = \tfrac12(1,1)^{\mathsf T} :
\]
$f\circ f$ é $f$ seguida de uma translação ao longo da reta imagem
--- $f$ é uma ``projeção deslizante'': projeção sobre a reta
composta com um deslizamento.
\end{solution}

\begin{solution}{exo:b2:affine:4}
$I = \operatorname{bar}(B, 2; C, 1)$ (pois $\vect{BI} =
\frac13\vect{BC}$ coloca $I$ mais perto de $B$: pesos $2$ em $B$,
$1$ em $C$ --- confira: $\vect{BI} = \frac{1}{3}\vect{BC}$).
Do mesmo modo, $J = \operatorname{bar}(C, 2; A, 1)$, $K =
\operatorname{bar}(A, 2; B, 1)$. Somando os três sistemas
ponderados, o \omterm{def:b2:affine:barycenter}{baricentro} de $(I, 1; J, 1; K, 1)$ (cada um de peso
total $3$, de modo que se substitui $I$ por seu sistema, etc.) é
\[
\operatorname{bar}\bigl(A, 1 + 2;\ B, 2 + 1;\ C, 1 + 2\bigr)
= \operatorname{bar}(A, 1; B, 1; C, 1) = G ,
\]
o centroide de $ABC$: o triângulo $IJK$ tem o mesmo centroide ---
previsível, porque a construção trata $A, B, C$ ciclicamente
e o centroide é o único ponto fixo da simetria \omterm{def:b2:structures:generated}{cíclica}
dos pesos.
\end{solution}

\begin{solution}{exo:b2:affine:5}
Ponha $g(u) = f(O + u) - f(O)$ (trabalhando em $\R^n$ vetorializado em
$O$), $g(0) = 0$.

\emph{Aditividade:} $\frac{(O + u) + (O + v)}{2} = O + \frac{u +
v}{2}$, de modo que a preservação de pontos médios dá $g\bigl(\frac{u+v}{2}\bigr)
= \frac{g(u) + g(v)}{2}$; com $v = 0$: $g(u/2) = g(u)/2$;
combinando, $g(u + v) = 2g\bigl(\frac{u+v}{2}\bigr) = g(u) + g(v)$.

\emph{$\Q$-homogeneidade:} a aditividade dá $g(nu) = ng(u)$ ($n \in
\N$, indução), depois $g(-u) = -g(u)$ (some), depois $g(\frac pq u) =
\frac pq g(u)$ (aplique $q$, use a injetividade do reescalamento).

\emph{$\R$-homogeneidade:} para $t \in \R$, tome racionais $t_n \to
t$: $g(t_nu) = t_ng(u)$, e a \omterm{def:b2:metric:continuity}{continuidade} de $g$ (herdada de $f$)
passa ao limite: $g(tu) = tg(u)$. Logo $g$ é linear e $f =
f(O) + g$: \omterm{def:b2:affine:subspace}{afim}.
\end{solution}

\begin{solution}{exo:b2:affine:6}
Parte linear $\varphi(x,y) = (y, x)$: a reflexão na diagonal
$y = x$ (ortogonal, \omterm{def:b2:linalg:det}{determinante} $-1$). Pontos fixos: $(x, y) = (y
+ 1, x - 1)$ reduz-se à única equação $y = x - 1$ (as duas
componentes são equivalentes): todo ponto da reta $y = x - 1$ é
fixo. Assim $f$ fixa essa reta ponto a ponto: $f$
é a \emph{reflexão} nesse eixo (uma isometria com uma reta de
pontos fixos e parte linear uma reflexão). Coerentemente, $f \circ
f(x,y) = f(y+1, x-1) = (x - 1 + 1, y + 1 - 1) = (x, y)$: uma
involução, como toda reflexão deve ser.
\end{solution}

\begin{solution}{exo:b2:affine:7}
Os $n + 1$ vetores $\vect{A_1A_i}$ ($i \geq 2$) são ligados em
dimensão $n$: existem $\mu_i$, não todos nulos, com $\sum_{i\geq2}
\mu_i\vect{A_1A_i} = 0$; ponha $\mu_1 = -\sum_{i \geq 2}\mu_i$, de modo que
$\sum_{i}\mu_i = 0$ e $\sum_i \mu_i\,\vect{OA_i} = 0$ para toda
$O$, com nem todos os $\mu_i$ nulos. Separe os índices: $P = \{i : \mu_i >
0\}$, $N = \{i : \mu_i < 0\}$, ambos não vazios (os $\mu_i$ somam
zero e não são todos nulos). Com $s = \sum_{i\in P}\mu_i =
-\sum_{i \in N}\mu_i > 0$:
\[
\operatorname{bar}\bigl(A_i, \tfrac{\mu_i}{s}\bigr)_{i \in P}
= \operatorname{bar}\bigl(A_i, \tfrac{-\mu_i}{s}\bigr)_{i \in N},
\]
(os dois lados valem o ponto $X$ com $\vect{OX} = \frac1s\sum_{i\in
P}\mu_i\vect{OA_i}$, pela relação): um ponto comum às duas
\omterm{def:b2:affine:convex}{envoltórias convexas}, com grupos de índices disjuntos.
\end{solution}

\begin{solution}{exo:b2:affine:8}
\emph{Imagem = pontos fixos:} para $Y = f(X)$, $f(Y) = f(f(X)) =
f(X) = Y$: todo ponto imagem é fixo; reciprocamente, os pontos fixos são
imagens. Assim $\mathcal{F} = \operatorname{im} f =
\operatorname{Fix}(f)$ é não vazio e é um subespaço \omterm{def:b2:affine:subspace}{afim}
(imagem de uma \omterm{def:b2:affine:subspace}{aplicação afim}), de direção $\operatorname{im}\vec f$.

\emph{Estrutura de projeção:} $\vec f$ é idempotente ($\vec{f\circ
f} = \vec f^{\,2} = \vec f$), logo $E = \operatorname{im}\vec f
\oplus \ker \vec f$ (\cref{ex:b2:reduction:projections}). Para qualquer
ponto $X$, considere o vetor $\vect{f(X)\,X}$; aplicando $\vec f$:
\[
\vec f\bigl(\vect{f(X)\,X}\bigr) = \vect{f(f(X))\,f(X)} = 0
\qquad (f \circ f = f),
\]
logo $\vect{f(X)\,X} \in \ker\vec f$. Portanto $X = f(X) +
\vect{f(X)X}$ exibe $X$ como um ponto de $\mathcal{F}$ transladado
por um vetor de $\ker\vec f$: $f$ é exatamente a projeção sobre
$\mathcal{F}$ ao longo de $\ker\vec f$. Reciprocamente, tais projeções
claramente satisfazem $f \circ f = f$.
\end{solution}

\begin{solution}{exo:b2:affine:9}
Seja $M = \operatorname{bar}(B, 2;\ C, 3)$, de peso total $5$.
A associatividade dá $G = \operatorname{bar}(A, 1;\ M, 5)$, logo
$\vect{AG} = \frac56\,\vect{AM}$: $G$ está no segmento
$\intcc AM$ a cinco sextos dele a partir de $A$. Como $A \notin
(BC)$, a reta $(AG) = (AM)$ encontra $(BC)$ no único ponto
$M$, com $\vect{BM} = \frac35\,\vect{BC}$. Do mesmo modo, com $N =
\operatorname{bar}(C, 3;\ A, 1)$ (peso total $4$, $\vect{CN}
= \frac14\,\vect{CA}$), a associatividade dá $G =
\operatorname{bar}(B, 2;\ N, 4)$: a reta $(BG)$ encontra $(CA)$
em $N$, e $\vect{BG} = \frac46\,\vect{BN} =
\frac23\,\vect{BN}$.
\end{solution}

\begin{solution}{exo:b2:affine:10}
Vetorialize numa origem $O$ e escreva os pontos como vetores:
$h_{\Omega, \lambda}(x) = \omega + \lambda(x - \omega)$ com
$\omega = \vect{O\Omega}$. A composição $g = h_{\Omega',
\mu} \circ h_{\Omega, \lambda}$ é \omterm{def:b2:affine:subspace}{afim} com parte linear
$\mu\lambda\,\mathrm{id}$. Se $\lambda\mu \neq 1$: $1 \notin
\operatorname{Sp}(\lambda\mu\,\mathrm{id})$, de modo que
a~\cref{prop:b2:affine:fixedpoint} fornece um único ponto fixo
$\Omega''$ e, vetorializada aí, $g =
\lambda\mu\,\mathrm{id}$: a homotetia $h_{\Omega'',
\lambda\mu}$. Se $\lambda\mu = 1$, a parte linear é a
identidade, logo $g$ é uma translação; desenvolvendo,
\[
g(x) = \omega' + \mu\bigl(\omega + \lambda(x - \omega) -
\omega'\bigr) = x + (1 - \mu)\,\omega' + \mu(1 -
\lambda)\,\omega .
\]
Para $\lambda = \mu = -1$ (simetrias centrais) o vetor é
$2\omega' - 2\omega = 2\,\vect{\Omega\Omega'}$: a
composição das simetrias centrais em $\Omega$ e depois
$\Omega'$ é a translação por $2\,\vect{\Omega\Omega'}$.
\end{solution}

\begin{solution}{exo:b2:affine:11}
$\vect{A'B} = \alpha\,\vect{A'C}$ diz exatamente que $1\cdot
\vect{A'B} - \alpha\,\vect{A'C} = 0$, isto é, $A' =
\operatorname{bar}(B, 1;\ C, -\alpha)$ (peso total $1 -
\alpha \neq 0$, pois $B \neq C$); do mesmo modo $B' =
\operatorname{bar}(C, 1;\ A, -\beta)$ e $C' =
\operatorname{bar}(A, 1;\ B, -\gamma)$.

\emph{O critério de alinhamento.} Dê a cada ponto $P$ sua
linha baricêntrica normalizada $p = (p_A, p_B, p_C)$, $p_A + p_B +
p_C = 1$, em relação a $(A, B, C)$. Se $\sum_i c_i p_i = 0$
com $(c_1, c_2, c_3) \neq 0$ para três pontos $P_1, P_2,
P_3$, então somar as entradas dá $\sum c_i = 0$, e
$\sum_i c_i \vect{OP_i} = \sum_j \bigl(\sum_i
c_ip_{ij}\bigr)\vect{OV_j} = 0$: os $P_i$ são afinmente
dependentes, isto é, alinhados. Reciprocamente, uma dependência \omterm{def:b2:affine:subspace}{afim}
$(t_i)$ dá $w = \sum t_ip_i$ com entradas somando $0$
e $\sum_j w_j\vect{OV_j} = 0$; desenvolvendo a partir de $A$, $w_B
\vect{AB} + w_C\vect{AC} = 0$, logo $w = 0$ pela independência
\omterm{def:b2:affine:subspace}{afim} de $(A, B, C)$: as linhas são linearmente dependentes.
Assim o alinhamento equivale à anulação de um \omterm{def:b2:linalg:det}{determinante} $3 \times 3$,
e reescalar as linhas pelos fatores não nulos $1 - \alpha$, $1 -
\beta$, $1 - \gamma$ nada muda:
\[
\det\begin{pmatrix} 0 & 1 & -\alpha\\ -\beta & 0 & 1\\ 1 &
-\gamma & 0\end{pmatrix}
= 1 - \alpha\beta\gamma .
\]
Logo $A', B', C'$ estão alinhados se e somente se $\alpha\beta\gamma = 1$:
o teorema de Menelau.
\end{solution}

\begin{solution}{exo:b2:affine:12}
Seja $\Delta = \{\lambda \in \R^{n+1} : \lambda_i \geq 0,\
\sum\lambda_i = 1\}$: fechado e limitado em $\R^{n+1}$, logo
\omterm{def:b2:metric:compact}{compacto}, e $K^{n+1}$ é \omterm{def:b2:metric:compact}{compacto} como produto finito. A aplicação
\[
\Phi \colon \Delta \times K^{n+1} \to \R^n, \qquad
\Phi(\lambda, x_0, \dots, x_n) = \sum_{i=0}^n \lambda_i x_i
\]
é \omterm{def:b2:metric:continuity}{contínua}, e o~\cref{thm:b2:affine:caratheodory} diz
precisamente que $\operatorname{conv}(K) = \Phi(\Delta \times
K^{n+1})$: imagem \omterm{def:b2:metric:continuity}{contínua} de um \omterm{def:b2:metric:compact}{compacto}
(\cref{thm:b2:metric:compactprops}), logo \omterm{def:b2:metric:compact}{compacta}.

Para um conjunto fechado: tome $S = (\R \times \{0\}) \cup \{(0,
1)\}$, fechado em $\R^2$. Uma combinação \omterm{def:b2:affine:convex}{convexa} que põe peso
$t$ em $(0,1)$ e $1 - t$ em pontos do eixo tem segunda
coordenada $t$, de modo que
\[
\operatorname{conv}(S) = \bigl(\R \times \intco01\bigr) \cup
\{(0,1)\}
\]
(para $0 \leq t < 1$, $(x, t) = t\,(0,1) + (1-t)\,\bigl(\tfrac
x{1-t}, 0\bigr)$). O ponto $(1, 1) = \lim_{t \to 1}(1, t)$
é aderente mas não está na envoltória: não é fechada.
\end{solution}

\begin{solution}{pb:b2:affine:1}
\textbf{1.} Rebase em $A_j$: para $i \neq j$, $\vect{A_jA_i} =
\vect{A_0A_i} - \vect{A_0A_j}$. Se $\sum_{i \neq j}
c_i\vect{A_jA_i} = 0$, desenvolver dá $\sum_{i \notin \{0,
j\}} c_i\,\vect{A_0A_i} - \bigl(\sum_{i\neq j}
c_i\bigr)\vect{A_0A_j} = 0$; a independência dos
$\vect{A_0A_i}$ força $c_i = 0$ para $i \notin \{0, j\}$, e depois
$c_0 = 0$: independência em $A_j$. Para a equivalência: duas
famílias de pesos $(\lambda_i)$, $(\lambda_i')$ somando $1$
com o mesmo \omterm{def:b2:affine:barycenter}{baricentro} dão, com $\nu = \lambda -
\lambda'$: $\sum\nu_i = 0$ e (origem $A_0$) $\sum_{i \geq
1}\nu_i\,\vect{A_0A_i} = 0$, logo $\nu = 0$ sob a independência.
Reciprocamente, uma relação não trivial $\sum_{i\geq1}\mu_i
\vect{A_0A_i} = 0$, completada por $\mu_0 = -\sum_{i\geq1}
\mu_i$, permite somar $t(\mu_i)$ a qualquer família de pesos sem
mover o \omterm{def:b2:affine:barycenter}{baricentro}: não unicidade.

\textbf{2.} $(\vect{AB}, \vect{AC})$ é uma base de $\R^2$:
escreva $\vect{AM} = \beta\,\vect{AB} + \gamma\,\vect{AC}$
(único) e ponha $\alpha = 1 - \beta - \gamma$; a condição de \omterm{def:b2:affine:barycenter}{baricentro}
na origem $A$ lê-se exatamente $\vect{AM} =
\beta\,\vect{AB} + \gamma\,\vect{AC}$. A unicidade é a questão
1.

\textbf{3.} De $\alpha\vect{MA} + \beta\vect{MB} +
\gamma\vect{MC} = 0$ e Chasles, $\vect{MA} =
-\beta\,\vect{AB} - \gamma\,\vect{AC}$. Com $D =
\det(\vect{AB}, \vect{AC})$:
\begin{align*}
\det(\vect{MB}, \vect{MC})
&= \det(\vect{MA} + \vect{AB},\ \vect{MA} + \vect{AC})\\
&= \det(\vect{MA}, \vect{AC}) + \det(\vect{AB}, \vect{MA}) +
D = -\beta D - \gamma D + D = \alpha D,
\end{align*}
usando a bilinearidade e $\det(\vect{MA}, \vect{AC}) = -\beta D$,
$\det(\vect{AB}, \vect{MA}) = -\gamma D$. As outras duas
fórmulas seguem do mesmo cálculo com os papéis
permutados ciclicamente.

\textbf{4.} Por definição, $\operatorname{conv}\{A, B, C\}$ é
o conjunto dos \omterm{def:b2:affine:barycenter}{baricentros} com pesos não negativos; normalizando
os pesos para somar $1$ e invocando a unicidade (questão 2),
$M \in \operatorname{conv}\{A,B,C\}$ se e somente se $\alpha, \beta,
\gamma \geq 0$. Cada coordenada é uma função \omterm{def:b2:affine:subspace}{afim} de $M$
(questão 3: um \omterm{def:b2:linalg:det}{determinante} $2\times2$ com uma coluna \omterm{def:b2:affine:subspace}{afim}
em $M$), de modo que cada condição \omterm{def:b2:metric:topology}{aberta} de sinal define um semiplano
\omterm{def:b2:metric:topology}{aberto}. O padrão $(-,-,-)$ contradiz $\alpha + \beta
+ \gamma = 1$; cada um dos sete padrões restantes é
realizado: reescale uma tripla que respeite os sinais e tenha ao menos uma
entrada $+$ de modo que a soma (positiva) valha $1$ --- por exemplo,
$(-1, 1, 1)$, $(3, -1, -1)$, $(\frac13, \frac13, \frac13)$,
e permutações.

\textbf{5.} Uma \omterm{def:b2:affine:subspace}{aplicação afim} preserva \omterm{def:b2:affine:barycenter}{baricentros}
(\cref{def:b2:affine:subspace}), logo $u(M) = \alpha u(A) +
\beta u(B) + \gamma u(C)$. Escrevendo $u(x, y) = ax + by + c$
com $(a, b) \neq (0,0)$: $\{u = c'\}$ é uma reta, e toda
reta $ax + by = c'$ é um tal conjunto de nível. Se $u(M), u(N) \geq
c$ e $t \in \intcc01$, então $u\bigl(\operatorname{bar}(M,
1-t; N, t)\bigr) = (1-t)u(M) + tu(N) \geq c$: os semiplanos são
\omterm{def:b2:affine:convex}{convexos}.

\textbf{6.} As condições $\sum\mu_i = 0$, $3\mu_2 + \mu_4 =
0$, $3\mu_3 + \mu_4 = 0$ dão (tomando $\mu_4 = 3$) a
dependência $(\mu_1, \mu_2, \mu_3, \mu_4) = (-1, -1, -1, 3)$.
Os sinais se separam como $\{A_4\} \mid \{A_1, A_2, A_3\}$ e,
normalizando cada lado por $3$:
\[
A_4 = \operatorname{bar}\bigl(A_1, \tfrac13;\ A_2, \tfrac13;\
A_3, \tfrac13\bigr) = (1,1),
\]
o centroide do triângulo: o ponto de Radon é o próprio $A_4$,
que de fato está dentro do triângulo $A_1A_2A_3$.

\textbf{7.} A aplicação linear $\Phi \colon \R^{n+2} \to \R
\times \R^n$, $\mu \mapsto (\sum\mu_i,\
\sum\mu_i\vect{OA_i})$, tem posto $\leq n + 1$, logo $\dim\ker
\Phi \geq 1$. Se existissem duas dependências independentes $\mu, \mu'$,
uma combinação adequada $\nu = \mu'_{n+2}\mu -
\mu_{n+2}\mu'$ (ou o próprio $\mu$, se os dois últimos coeficientes
se anulassem) seria uma dependência \emph{não nula} com $\nu_{n+2}
= 0$; restringindo a $A_1, \dots, A_{n+1}$ e rebaseando em
$A_1$, algum $\nu_i \neq 0$ com $i \geq 2$ (um único peso não nulo
não pode somar zero), dando uma relação não trivial
$\sum_{i\geq2}\nu_i\vect{A_1A_i} = 0$: os $n+1$ pontos seriam
afinmente dependentes, contra a posição geral. Logo $\dim\ker
\Phi = 1$. O mesmo argumento de restrição mostra que uma dependência
não nula não tem coeficiente nulo.

\textbf{8.} Seja $\mu \neq 0$ uma dependência, $P = \{i :
\mu_i > 0\}$ e $N = \{i : \mu_i < 0\}$: ambos não vazios
($\sum\mu_i = 0$, $\mu \neq 0$) e exaustivos (nenhum coeficiente
nulo). A construção de Radon (\cref{exo:b2:affine:7})
produz o ponto comum das envoltórias exatamente a partir dessa partição.
Como a dependência é única a menos de escalar não nulo
(questão 7), o par não ordenado $\{P, N\}$ --- logo a
partição de Radon --- é único.

\textbf{9.} Os blocos são não vazios, de modo que o tipo é $(1,3)$ ou
$(2,2)$. Tipo $(1,3)$, bloco $\{j\}$: o ponto de Radon está em
$\operatorname{conv}\{A_j\} = \{A_j\}$, logo $A_j \in
\operatorname{conv}$ dos outros três; ele não pode estar numa
aresta (três dos pontos seriam alinhados, contra a posição
geral), de modo que $A_j$ é interior ao triângulo. Tipo
$(2,2)$, blocos $\{i,j\} \mid \{k,l\}$: o ponto de Radon $z$
está em $\intcc{A_i}{A_j} \cap \intcc{A_k}{A_l}$, e $z$ não é
extremidade (isso alinharia três pontos): os dois
segmentos se cruzam num ponto interior. Além disso, nenhum ponto está
na envoltória dos outros: uma tal contenção $A_l =
\operatorname{bar}(A_i, \lambda_i)_{i \neq l}$ com
$\lambda_i \geq 0$ é uma dependência \omterm{def:b2:affine:subspace}{afim} de padrão de sinais
$(+,+,+,-)$, o que, pela unicidade (questão 8), tornaria a
partição $(1,3)$. Assim, no caso $(2,2)$, os quatro pontos estão
em posição \omterm{def:b2:affine:convex}{convexa} e os segmentos que se cruzam são as
diagonais.

\textbf{10.} As equações $\mu_2 + \mu_3 = 0$, $\mu_3 +
\mu_4 = 0$, $\sum\mu_i = 0$ dão a dependência $(1, -1, 1,
-1)$: partição $\{(0,0), (1,1)\} \mid \{(1,0), (0,1)\}$, e
\[
\operatorname{bar}\bigl((0,0), \tfrac12;\ (1,1),
\tfrac12\bigr) = \bigl(\tfrac12, \tfrac12\bigr) =
\operatorname{bar}\bigl((1,0), \tfrac12;\ (0,1),
\tfrac12\bigr) :
\]
o ponto de Radon é o centro do quadrado, onde as duas
diagonais se cruzam --- tipo $(2,2)$, como a figura prevê.

\textbf{11.} Radon aplicado a $x_1, \dots, x_4$ dá blocos
$I \mid J$ e um ponto $z \in \operatorname{conv}\{x_i : i
\in I\} \cap \operatorname{conv}\{x_j : j \in J\}$. Fixe $k
\in \{1, \dots, 4\}$, digamos $k \in I$. Todo $j \in J$
satisfaz $j \neq k$, logo $x_j \in C_k$ pela escolha $x_j \in
\bigcap_{l \neq j}C_l$; como $C_k$ é \omterm{def:b2:affine:convex}{convexo}, $z \in
\operatorname{conv}\{x_j : j \in J\} \subseteq C_k$. Como $k$
era arbitrário, $z \in C_1 \cap C_2 \cap C_3 \cap C_4$.

\textbf{12.} Indução sobre $m$. Para $m = 3$ a hipótese é
a conclusão; $m = 4$ é a questão 11. Seja $m \geq 4$,
suponha o enunciado para $m$ conjuntos e tome $C_1, \dots,
C_{m+1}$ com a propriedade das interseções triplas. Ponha $C_m' =
C_m \cap C_{m+1}$, \omterm{def:b2:affine:convex}{convexo}. A família $C_1, \dots, C_{m-1},
C_m'$ tem $m$ membros; uma tripla que evita $C_m'$ se intersecta por
hipótese, e uma tripla $\{C_i, C_j, C_m'\}$ tem
interseção $C_i \cap C_j \cap C_m \cap C_{m+1}$, não vazia
pela questão 11 aplicada a $C_i, C_j, C_m, C_{m+1}$ (quaisquer três
deles se encontram, por hipótese). A hipótese de indução agora
fornece um ponto comum à nova família, isto é, aos $m+1$
conjuntos.

\textbf{13.} (a) Os lados fechados $\intcc PQ$, $\intcc QR$,
$\intcc RP$ de um triângulo não degenerado: quaisquer dois compartilham um
vértice, mas um ponto comum aos três estaria em $\intcc
PQ \cap \intcc RP = \{P\}$ e em $\intcc QR$, o que exclui
$P$. (b) Quaisquer três dos conjuntos $S_i = \{x_1, \dots, x_4\}
\setminus \{x_i\}$ omitem três dos quatro pontos, deixando
exatamente um ponto comum; a interseção total omite todos os
pontos. Os $S_i$ são finitos, não \omterm{def:b2:affine:convex}{convexos}: a convexidade é
essencial. (c) Um número finito de $H_k = \intco{k}{+\infty} \times
\R$ se intersecta em $\intco{k_{\max}}{+\infty} \times \R \neq
\emptyset$, e no entanto nenhum ponto tem $x \geq k$ para todo $k \in \N$:
para famílias infinitas, a \omterm{def:b2:metric:compact}{compacidade} é essencial.

\textbf{14.} Suponha $\bigcap_{i \in I}K_i = \emptyset$ e
fixe $i_0$. Todo $x \in K_{i_0}$ escapa de algum $K_i$, logo
$K_{i_0} \subseteq \bigcup_{i \in I}(\R^2 \setminus K_i)$, uma
cobertura por \omterm{def:b2:metric:topology}{abertos} ($K_i$ é \omterm{def:b2:metric:compact}{compacto}, logo fechado). Por
Borel--Lebesgue (\cref{thm:b2:metric:borellebesgue}), um número finito
basta: $K_{i_0} \cap K_{i_1} \cap \dots \cap K_{i_N} =
\emptyset$. Mas quaisquer três membros dessa família finita de
\omterm{def:b2:affine:convex}{conjuntos convexos} se intersectam, de modo que a questão 12 torna a interseção
não vazia: contradição.

\textbf{15.} Ponha $D_p = \overline D(p, r)$ para $p \in S$:
\omterm{def:b2:affine:convex}{conjuntos convexos} \omterm{def:b2:metric:compact}{compactos}. Para $p, q, s \in S$, a hipótese
dá um disco fechado $\overline D(z, r)$ que contém $p, q, s$;
então $\norm{\vect{zp}}, \norm{\vect{zq}}, \norm{\vect{zs}}
\leq r$, isto é, $z \in D_p \cap D_q \cap D_s$. Por Helly
(questão 12; a família é finita) existe $c \in
\bigcap_{p \in S}D_p$: todo $p \in S$ satisfaz
$\norm{\vect{cp}} \leq r$, logo $S \subseteq \overline D(c,
r)$.

\textbf{16.} Seja $c = x_{\lceil n/2\rceil}$. Uma semirreta
fechada que contém $c$ é $\intoc{-\infty}{t}$ com $t \geq
c$ ou $\intco{t}{+\infty}$ com $t \leq c$. A primeira
contém $x_1, \dots, x_{\lceil n/2\rceil}$: ao menos
$\lceil n/2\rceil \geq n/2$ pontos. A segunda contém
$x_{\lceil n/2\rceil}, \dots, x_n$: exatamente $n - \lceil
n/2\rceil + 1 = \floor{n/2} + 1 > n/2$ pontos.

\textbf{17.} $\abs{A \cap B} = \abs A + \abs B - \abs{A \cup
B} \geq \abs A + \abs B - n > \tfrac{4n}3 - n = \tfrac n3$,
logo
\[
\abs{A \cap B \cap C} \geq \abs{A \cap B} + \abs C - n >
\tfrac n3 + \tfrac{2n}3 - n = 0 .
\]

\textbf{18.} Note que $m > 2n/3$. Para $T_1, T_2, T_3 \subseteq S$
de cardinalidade $m$, a questão 17 fornece um ponto $x \in T_1
\cap T_2 \cap T_3$; então $x \in \operatorname{conv}(T_i)$ para
cada $i$: quaisquer três membros de $\mathcal F$ se encontram. A família
é finita (finitos subconjuntos de $S$) e consiste em
\omterm{def:b2:affine:convex}{conjuntos convexos}, de modo que Helly (questão 12) dá $c \in
\bigcap_{\abs T = m}\operatorname{conv}(T)$.

\textbf{19.} Suponha que algum semiplano fechado $H \ni c$
contenha menos de $n/3$ pontos de $S$. Seu complementar $U$
é um semiplano \omterm{def:b2:metric:topology}{aberto}, \omterm{def:b2:affine:convex}{convexo}, com $\abs{S \cap U} > 2n/3$,
logo $\abs{S \cap U} \geq \floor{2n/3} + 1 = m$; escolha $T
\subseteq S \cap U$ com $\abs T = m$. Então
$\operatorname{conv}(T) \subseteq U$ pela convexidade de $U$, de modo que
$c \in \operatorname{conv}(T) \subseteq U$: contradição
com $c \in H$. Portanto todo semiplano fechado que contém $c$
contém ao menos $n/3$ pontos: $c$ é um ponto central.

\textbf{20.} Tome o triângulo equilátero de lado $L$ e
$\varepsilon = L/100$. Seja $c$ um ponto qualquer; exibimos um
semiplano fechado que contém $c$ e no máximo $k$ pontos.

\emph{Caso 1: $c$ está a menos de $L/10$ de um vértice, digamos $B$.}
As direções de $c$ para $A$ e para $C$ desviam das
direções $B \to A$, $B \to C$ em no máximo
$\arcsin\bigl(\tfrac{L/10}{9L/10}\bigr) = \arcsin\tfrac19$,
de modo que fazem um ângulo $\leq \tfrac\pi3 + 2\arcsin\tfrac19 <
\tfrac{2\pi}3$. \emph{Caso 2: $c$ está a distância $> L/10$
de todos os vértices.} Se $c$ está no triângulo, as três
lacunas angulares entre as direções $u_A, u_B, u_C$ de $c$
aos vértices somam $2\pi$, de modo que alguma lacuna é $\leq 2\pi/3$;
se $c$ está fora, as três direções estão num semiplano
\omterm{def:b2:metric:topology}{aberto} de direções e duas delas fazem um ângulo $<
\pi/2$. Em todos os casos duas direções, digamos rumo a $X$ e $Y$,
fazem um ângulo $\leq 2\pi/3$; seja $w$ sua bissetriz unitária,
de modo que $\langle u_X, w\rangle, \langle u_Y, w\rangle \geq
\cos\tfrac\pi3 = \tfrac12$. Para qualquer ponto $b$ do disco
em torno de $X$:
\[
\langle \vect{cb}, w\rangle \geq
\tfrac12\norm{\vect{cX}} - \varepsilon > 0,
\]
pois $\norm{\vect{cX}} \geq L/10 > 2\varepsilon$ (e
do mesmo modo para $Y$): o semiplano \omterm{def:b2:metric:topology}{aberto} $\{\langle \vect{cx},
w\rangle > 0\}$ engole os dois aglomerados. Seu complementar
fechado contém $c$ e no máximo os $k$ pontos do
terceiro aglomerado. Assim nenhum ponto do plano supera $n/3$: com
a questão 19, a constante do ponto central é exatamente $1/3$.

\textbf{21.} Ordene os ângulos; o maior, $\theta$,
satisfaz $\theta \geq \pi/3$ (os três somam $\pi$). Se
$\theta \geq \pi/2$, digamos em $R$, seja $M$ o ponto médio do
lado oposto $\intcc PQ$. A fórmula da mediana ($\vect{RM}
= \tfrac12(\vect{RP} + \vect{RQ})$, desenvolva e elimine
$\langle\vect{RP}, \vect{RQ}\rangle$ pela lei dos cossenos)
dá
\[
\norm{\vect{RM}}^2 = \tfrac12\norm{\vect{RP}}^2 +
\tfrac12\norm{\vect{RQ}}^2 - \tfrac14\norm{\vect{PQ}}^2 \leq
\tfrac12\norm{\vect{PQ}}^2 - \tfrac14\norm{\vect{PQ}}^2 =
\tfrac14\norm{\vect{PQ}}^2,
\]
usando $\norm{\vect{PQ}}^2 = \norm{\vect{RP}}^2 +
\norm{\vect{RQ}}^2 - 2\langle\vect{RP}, \vect{RQ}\rangle \geq
\norm{\vect{RP}}^2 + \norm{\vect{RQ}}^2$ (o produto interno
é $\leq 0$). Assim o disco de diâmetro $\intcc PQ$, de raio
$\leq \tfrac12 < \tfrac1{\sqrt3}$, contém os três pontos
(as triplas alinhadas degeneradas caem sob $\theta = \pi$). Se
$\theta < \pi/2$, o triângulo é acutângulo; pela lei dos senos
o raio da circunferência circunscrita é $R_c = a/(2\sin\theta)$, com $a \leq 1$
o lado oposto a $\theta$, e $\theta \in
\intco{\pi/3}{\pi/2}$ dá $\sin\theta \geq \sqrt3/2$, logo
$R_c \leq 1/\sqrt3$: o disco circunscrito resolve.

\textbf{22.} Para $p \in S$ seja $K_p = \overline D(p,
1/\sqrt3)$: \omterm{def:b2:affine:convex}{convexo} \omterm{def:b2:metric:compact}{compacto}. Quaisquer três pontos $p, q, s$ de $S$
estão dois a dois a distância $\leq 1$, de modo que a questão 21 dá um
disco de raio $1/\sqrt3$ que os contém: seu centro está em
$K_p \cap K_q \cap K_s$. Por Helly \omterm{def:b2:metric:compact}{compacto} (questão 14,
famílias arbitrárias permitidas) existe $c \in \bigcap_{p\in
S}K_p$: todo $p \in S$ está a menos de $1/\sqrt3$ de $c$, isto é, $S
\subseteq \overline D(c, 1/\sqrt3)$. Esse é o teorema de Jung
no plano.

\textbf{23.} Com $G$ o centroide, $\sum_i\vect{GA_i} = 0$,
logo
\[
\sum_i\norm{\vect{OA_i}}^2 = \sum_i\norm{\vect{OG} +
\vect{GA_i}}^2 = 3\norm{\vect{OG}}^2 + 2\Bigl\langle
\vect{OG}, \sum_i\vect{GA_i}\Bigr\rangle +
\sum_i\norm{\vect{GA_i}}^2,
\]
e o termo do meio se anula: a identidade de Leibniz. Para o
triângulo equilátero de lado $1$, $\norm{\vect{GA_i}} =
1/\sqrt3$ (dois terços da altura $\sqrt3/2$), logo
$\sum_i\norm{\vect{GA_i}}^2 = 1$. Se $\overline D(O, r)$
contém os vértices, então $3r^2 \geq
\sum_i\norm{\vect{OA_i}}^2 = 3\norm{\vect{OG}}^2 + 1 \geq 1$:
$r \geq 1/\sqrt3$, com igualdade forçando $O = G$ e as três
distâncias iguais a $r$ --- o disco circunscrito. A constante
$1/\sqrt3$ de Jung é ótima.

\textbf{24.} \emph{Helly em $\R^n$: se $C_1, \dots, C_m$
($m \geq n + 1$) são partes \omterm{def:b2:affine:convex}{convexas} de $\R^n$ e quaisquer $n + 1$
delas se intersectam, então todas se intersectam.} Caso base $m = n +
2$: escolha $x_i \in \bigcap_{j\neq i}C_j$; o lema de Radon
(\cref{exo:b2:affine:7}) separa $x_1, \dots, x_{n+2}$ em
blocos $I \mid J$ com um ponto comum das envoltórias $z$ e, para cada
$k$, o bloco que não contém $k$ consiste em pontos de
$C_k$, logo $z \in C_k$ por convexidade, exatamente como na questão
11. Passo de indução para $m \geq n + 2$: substitua $C_m,
C_{m+1}$ por $C_m \cap C_{m+1}$; uma $(n+1)$-upla da nova
família que contenha o membro intersectado equivale a $n + 2$
dos conjuntos antigos, tratados pelo caso base, e as demais
uplas estão cobertas pela hipótese. Conclua pela hipótese
de indução.

\textbf{25.} A álgebra linear entra uma única vez: $n + 2$ vetores no
espaço de dimensão $(n+1)$ dos pares (peso total,
posição ponderada) têm de ser dependentes --- essa é a dependência
\omterm{def:b2:affine:subspace}{afim}. Os sinais de seus coeficientes separam os pontos
nos dois blocos de Radon e transformam uma relação linear numa
igualdade de dois \omterm{def:b2:affine:barycenter}{baricentros} não negativos. A convexidade é usada
exatamente duas vezes: no passo de Helly (a envoltória de pontos de $C_k$
permanece em $C_k$) e nas aplicações (semiplanos e
discos são \omterm{def:b2:affine:convex}{convexos}). A dimensão do plano entrou apenas
pelo número $4 = 2 + 2$ de pontos entregues a Radon, isto é,
o ``$3 = 2 + 1$'' da hipótese de Helly; todo o resto
foi independente da dimensão, como a questão 24 confirma. Em $\R^n$ as
constantes tornam-se: número de Helly $n + 1$; constante do ponto central
$\frac1{n+1}$ (todo conjunto finito tem um ponto tal que todo
semiespaço fechado que passa por ele contém uma fração $\geq
\frac1{n+1}$ do conjunto); raio de Jung
$\sqrt{\frac{n}{2(n+1)}}$ para conjuntos de diâmetro $1$ --- igual
a $1/\sqrt3$ quando $n = 2$.
\end{solution}
