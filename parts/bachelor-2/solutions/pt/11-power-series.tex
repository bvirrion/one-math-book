\section*{Capítulo \ref{ch:b2:powerseries} --- Séries de potências}

\begin{solution}{exo:b2:powerseries:1}
$\sum \frac{n^2}{2^n}z^n$: razão $\frac{(n+1)^2}{2^{n+1}}\cdot
\frac{2^n}{n^2} \to \frac12$: $R = 2$.

$\sum \frac{z^n}{\binom{2n}{n}}$: $\binom{2n}{n} \sim
\frac{4^n}{\sqrt{\pi n}}$ (\cref{ex:b2:comparison:centralbinomial}),
logo $\abs{a_n}^{-1} \approx 4^n$ a menos de fatores polinomiais: $R = 4$
(critério da razão: $\frac{\binom{2n}{n}}{\binom{2n+2}{n+1}} =
\frac{(n+1)^2}{(2n+1)(2n+2)} \to \frac14$).

$\sum z^{n!}$: coeficientes $a_k = 1$ se $k = n!$, e $0$ caso contrário. Para
$\abs z < 1$, $\sum \abs z^{n!}$ converge (dominada por uma geométrica);
para $\abs z \geq 1$ os termos não tendem a $0$: $R = 1$.

$\sum (2 + (-1)^n)^n z^n$: coeficientes $3^n$ ($n$ par) e $1$
($n$ ímpar). A limitação de $a_nr^n$ exige $3r \leq 1$; e $r <
\frac13$ funciona: $R = \frac13$.
\end{solution}

\begin{solution}{exo:b2:powerseries:2}
As três têm raio $1$ (critério da razão). Em $z = 1$: $\sum 1$
diverge; $\sum\frac1n$ diverge; $\sum\frac{1}{n^2}$ converge. Em
$z = -1$: $\sum(-1)^n$ diverge; $\sum\frac{(-1)^n}{n}$ converge
(alternada); $\sum\frac{(-1)^n}{n^2}$ converge (\omterm{def:b2:series:def}{absolutamente}).
O comportamento na fronteira é invisível para o raio.
\end{solution}

\begin{solution}{exo:b2:powerseries:3}
De $\frac{1}{1-x} = \sum x^n$, derive e multiplique por $x$
(\cref{thm:b2:powerseries:calculus}):
\[
\sum n x^n = \frac{x}{(1-x)^2} .
\]
Derive mais uma vez e multiplique por $x$ de novo:
\[
\sum n^2 x^n = x\,\frac{\dd}{\dd x}\Bigl(\frac{x}{(1-x)^2}\Bigr)
= \frac{x(1 + x)}{(1-x)^3} .
\]
Terceira soma: é a parte ímpar de $-\ln(1 - x)$:
\[
\sum_{n\geq0} \frac{x^{2n+1}}{2n+1}
= \frac{-\ln(1-x) + \ln(1+x)}{2}
= \frac12 \ln\frac{1+x}{1-x}
= \operatorname{artanh} x .
\]
\end{solution}

\begin{solution}{exo:b2:powerseries:4}
Frações parciais: $\frac{1}{(1-x)(2-x)} = \frac{1}{1-x} -
\frac{1}{2 - x} = \sum x^n - \frac12\sum \bigl(\frac x2\bigr)^n$:
\[
\frac{1}{(1-x)(2-x)} = \sum_{n\geq0} \Bigl(1 -
\frac{1}{2^{n+1}}\Bigr)x^n,
\qquad R = 1 .
\]

$\ln(1 + x + x^2) = \ln\frac{1 - x^3}{1 - x} = \ln(1 - x^3) -
\ln(1 - x) = \sum_{n\geq1}\frac{x^n}{n} -
\sum_{m\geq1}\frac{x^{3m}}{m}$: o coeficiente de $x^n$ é
$\frac1n$ se $3 \nmid n$, e $\frac1n - \frac{3}{n} = -\frac2n$ se
$3 \mid n$. Raio $1$ (o obstáculo mais próximo: a série de
$\ln(1-x^3)$).
\end{solution}

\begin{solution}{exo:b2:powerseries:5}
\omterm{thm:b2:series:fubini}{Produto de Cauchy} de $\sum_{m \geq 0} x^m$ (coeficientes $1$) e
$\sum_{k\geq1} \frac{x^k}{k}$ (coeficientes $\frac1k$, $k \geq
1$), ambas \omterm{def:b2:series:def}{absolutamente} convergentes para $\abs x < 1$: o coeficiente
de $x^n$ no produto é $\sum_{k=1}^{n} \frac1k \cdot 1 = H_n$.
Logo
\[
\Bigl(\sum x^m\Bigr)\Bigl(\sum \frac{x^k}{k}\Bigr)
= \frac{1}{1-x}\cdot\bigl(-\ln(1-x)\bigr)
= \sum_{n\geq1} H_n x^n .
\]
\end{solution}

\begin{solution}{exo:b2:powerseries:6}
Multiplique a recorrência por $x^{n+1}$ e some ($\abs x <
\frac12$):
\[
U(x) - 1 = 2x\,U(x) + \sum_{n\geq0} n\,x^{n+1}
= 2x\,U(x) + \frac{x^2}{(1-x)^2} ,
\]
usando \cref{exo:b2:powerseries:3}. Logo
\[
U(x) = \frac{1}{1 - 2x}\Bigl(1 + \frac{x^2}{(1-x)^2}\Bigr)
= \frac{1 - 2x + 2x^2}{(1-2x)(1-x)^2} .
\]
Frações parciais (o método de encobrir em $x = \frac12$ dá o coeficiente $2$;
no polo duplo $x = 1$, o coeficiente $-1$; o coeficiente do meio
se anula avaliando em $x = 0$):
\[
U(x) = \frac{2}{1-2x} - \frac{1}{(1 - x)^2} .
\]
Desenvolvendo os dois:
\[
u_n = 2\cdot 2^n - (n + 1) = 2^{n+1} - n - 1 .
\]
(Confira: $u_0 = 1$, $u_1 = 2u_0 + 0 = 2 = 4 - 2$.)
\end{solution}

\begin{solution}{exo:b2:powerseries:7}
Indução: $f'(x) = \frac{2}{x^3}\eu^{-1/x^2}$ e, se $f^{(n)}(x)
= P_n(\tfrac1x)\eu^{-1/x^2}$, então
\[
f^{(n+1)}(x) =
\Bigl(-\frac{1}{x^2}\,P_n'\Bigl(\frac1x\Bigr) +
\frac{2}{x^3}\,P_n\Bigl(\frac1x\Bigr)\Bigr)\eu^{-1/x^2} :
\]
de novo da forma enunciada. Em $0$: os quocientes de diferenças $\frac{f^{(n)}(h)}{h} =
\frac1h P_n(\frac1h)\eu^{-1/h^2} \to 0$ quando $h \to 0$, pois
$Q(u)\,\eu^{-u^2} \to 0$ quando $u \to \pm\infty$ para qualquer polinômio
$Q$ (a exponencial vence as potências): por indução, todas as $f^{(n)}(0)$ existem
e se anulam, e cada $f^{(n)}$ é \omterm{def:b2:metric:continuity}{contínua} em $0$ pelo mesmo
limite. Logo $f \in C^\infty$ com série de Taylor nula em $0$; a
série de Taylor soma $0 \neq f$: não \omterm{def:b2:powerseries:analytic}{analítica} em $0$.
\end{solution}

\begin{solution}{exo:b2:powerseries:8}
Série binomial: $\sqrt{1-4x} = \sum_{k\geq0}
\binom{1/2}{k}(-4x)^k$. Para $k = n + 1 \geq 1$:
\begin{align*}
\binom{1/2}{n+1}(-4)^{n+1}
&= \frac{\frac12\bigl(\frac12 - 1\bigr)\cdots\bigl(\frac12 -
n\bigr)}{(n+1)!}\,(-4)^{n+1}\\
&= \frac{(-1)^n\,1\cdot3\cdots(2n-1)}{2^{n+1}(n+1)!}\,(-4)^{n+1}
= -\frac{2}{n+1}\cdot\frac{(2n)!}{n!\,n!} ,
\end{align*}
usando $1\cdot3\cdots(2n-1) = \frac{(2n)!}{2^n n!}$. Logo
\[
C(x) = \frac{1 - \sqrt{1-4x}}{2x}
= \frac{1}{2x}\sum_{n\geq0}\frac{2}{n+1}\binom{2n}{n}x^{n+1}
= \sum_{n\geq0} \frac{1}{n+1}\binom{2n}{n}\,x^n :
\]
$C_n = \frac{1}{n+1}\binom{2n}{n}$. Raio: $\frac14$ (a série
binomial em $4x$). Assintótica via
\cref{ex:b2:comparison:centralbinomial}:
\[
C_n \sim \frac{4^n}{\sqrt{\pi}\; n^{3/2}} .
\]
\end{solution}

\begin{solution}{exo:b2:powerseries:9}
Com $A_n = \sum_{k \leq n} a_k \to A$: a \omterm{thm:b2:series:abel}{transformação de Abel} dá, para
$0 \leq x < 1$,
\[
\sum_{n=0}^{\infty} a_n x^n = (1 - x)\sum_{n=0}^{\infty} A_n x^n
\]
(os dois lados convergem: $(A_n)$ é limitada; e a identidade decorre
de $a_n = A_n - A_{n-1}$ e da reindexação). Como $(1 -
x)\sum x^n = 1$:
\[
\sum_n a_nx^n - A = (1-x)\sum_{n} (A_n - A)x^n .
\]
Dado $\varepsilon$, escolha $N$ com $\abs{A_n - A} \leq
\varepsilon$ para $n > N$; então
\[
\Bigl|\sum a_nx^n - A\Bigr|
\leq (1-x)\sum_{n \leq N}\abs{A_n - A} + \varepsilon(1 -
x)\sum_{n > N}x^n
\leq (1-x)\,C_N + \varepsilon ,
\]
e fazendo $x \to 1^-$: limsup $\leq \varepsilon$ para todo
$\varepsilon$. Logo o limite radial é $A$.

Aplicações: $\sum \frac{(-1)^{n-1}}{n}$ converge (alternada),
e para $x < 1$ sua série de potências soma $\ln(1 + x)$: por Abel,
a soma é $\ln 2$. Do mesmo modo, $\sum\frac{(-1)^n}{2n+1}x^{2n+1} =
\arctan x$ dá $\frac\pi4$ em $x = 1$ --- as demonstrações por
integral do primeiro ano, agora estruturais.
\end{solution}

\begin{solution}{exo:b2:powerseries:10}
Tanto $\sum\frac{x^n}{n}$ quanto $\sum\frac{x^n}{n+1}$ têm raio
$1$, e $\frac{1}{n(n+1)} = \frac1n - \frac1{n+1}$, logo, para $0
< \abs x < 1$:
\[
\sum_{n\geq1}\frac{x^n}{n(n+1)}
= -\ln(1-x) - \frac1x\sum_{n\geq1}\frac{x^{n+1}}{n+1}
= -\ln(1-x) - \frac{-\ln(1-x) - x}{x}
= 1 + \frac{1-x}{x}\ln(1-x) .
\]
Raio $1$; e $\norm{x^n/(n(n+1))}_{\infty,\intcc{-1}1} =
\frac{1}{n(n+1)}$ é \omterm{def:b2:series:summable}{somável}: \omterm{def:b2:funcseq:series}{convergência normal} em
$\intcc{-1}{1}$, de modo que a soma é \omterm{def:b2:metric:continuity}{contínua} aí. Quando $x \to
1^-$, $(1-x)\ln(1-x) \to 0$ e a forma fechada tende a $1$ ---
de acordo com o valor telescópico $\sum\frac{1}{n(n+1)} =
\lim_N\bigl(1 - \frac{1}{N+1}\bigr) = 1$ em $x = 1$.
\end{solution}

\begin{solution}{exo:b2:powerseries:11}
Classificando as $n!$ permutações por seu conjunto de pontos fixos: escolher
os $k$ pontos fixos ($\binom nk$ maneiras) e desarranjar os outros
$n - k$ objetos dá $n! = \sum_{k=0}^n\binom nk D_{n-k}$.
Dividindo por $n!$:
\[
1 = \sum_{k=0}^{n}\frac{1}{k!}\cdot\frac{D_{n-k}}{(n-k)!} ,
\]
o que diz exatamente que o \omterm{thm:b2:series:fubini}{produto de Cauchy} de $\eu^x =
\sum\frac{x^k}{k!}$ e $D(x) = \sum D_n\frac{x^n}{n!}$ é
$\sum x^n = \frac{1}{1-x}$. Os dois fatores convergem \omterm{def:b2:series:def}{absolutamente}
para $\abs x < 1$ ($D_n \leq n!$, logo $D$ é dominada pela
série geométrica): a identidade do produto é legítima
(\cref{prop:b2:powerseries:operations}), e
\[
D(x) = \frac{\eu^{-x}}{1-x} .
\]
\omterm{thm:b2:series:fubini}{Produto de Cauchy} de $\eu^{-x} = \sum\frac{(-1)^kx^k}{k!}$ e
$\sum x^m$: o coeficiente de $x^n$ é
$\sum_{k=0}^{n}\frac{(-1)^k}{k!}$ e, pela unicidade dos coeficientes de
séries de potências (\cref{thm:b2:powerseries:calculus}):
\[
\frac{D_n}{n!} = \sum_{k=0}^{n}\frac{(-1)^k}{k!}
\xrightarrow[n\to\infty]{} \eu^{-1} :
\]
cerca de $37\%$ de todas as permutações são desarranjos, qualquer que seja
$n$.
\end{solution}

\begin{solution}{exo:b2:powerseries:12}
Série binomial com $\alpha = -\frac12$ em $-4x$:
\[
\binom{-1/2}{n}(-4)^n
= \frac{\bigl(-\frac12\bigr)\bigl(-\frac32\bigr)\cdots
\bigl(-\frac{2n-1}2\bigr)}{n!}(-4)^n
= \frac{1\cdot3\cdots(2n-1)}{2^n\,n!}\,4^n
= \frac{(2n)!}{2^n n!}\cdot\frac{2^n}{n!}
= \binom{2n}{n},
\]
usando $1\cdot3\cdots(2n-1) = \frac{(2n)!}{2^nn!}$. Logo
$(1-4x)^{-1/2} = \sum\binom{2n}nx^n$ para $\abs{4x} < 1$.
Elevando ao quadrado (\omterm{thm:b2:series:fubini}{produto de Cauchy}, legítimo pela convergência absoluta)
e comparando com $\frac{1}{1-4x} = \sum 4^nx^n$: o
coeficiente de $x^n$ no quadrado é
$\sum_{k=0}^n\binom{2k}k\binom{2n-2k}{n-k}$, e a unicidade dos
coeficientes dá
\[
\sum_{k=0}^{n}\binom{2k}{k}\binom{2n-2k}{n-k} = 4^n .
\]
\end{solution}

\begin{solution}{pb:b2:powerseries:1}
\textbf{1.} Com $a_n = r_n - r_{n+1}$, somação por partes:
\[
\sum_{n=N}^{M} a_nx^n
= r_Nx^N + \sum_{n=N+1}^{M} r_n\bigl(x^n - x^{n-1}\bigr)
- r_{M+1}x^M .
\]
Para $0 \leq x \leq 1$ os incrementos $x^{n-1} - x^n$ são
não negativos e telescopam para $x^N - x^M$; com $s = \sup_{n\geq
N}\abs{r_n}$:
\[
\Bigl|\sum_{n=N}^{M}a_nx^n\Bigr|
\leq s\bigl(x^N + (x^N - x^M) + x^M\bigr) = 2s\,x^N \leq 2s .
\]

\textbf{2.} Como $r_n \to 0$, $\sup_{n\geq N}\abs{r_n} \to 0$:
a questão 1 é exatamente o critério de Cauchy uniforme em
$\intcc{0}{1}$, de modo que $\sum a_nx^n$ converge \omterm{def:b2:funcseq:def}{uniformemente} aí e
sua soma é \omterm{def:b2:metric:continuity}{contínua} (\cref{thm:b2:funcseq:seriestransfer}).
Sendo o valor em $1$ igual a $\sum a_n$, a \omterm{def:b2:metric:continuity}{continuidade} em $1$ é o
limite radial do~\cref{exo:b2:powerseries:9}.

\textbf{3.} Para $\abs x < 1$ as três séries de potências convergem
\omterm{def:b2:series:def}{absolutamente} e $\bigl(\sum a_nx^n\bigr)\bigl(\sum b_nx^n\bigr)
= \sum c_nx^n$ (\cref{prop:b2:powerseries:operations}). Pela
questão 2, cada fator e o lado do produto são \omterm{def:b2:metric:continuity}{contínuos} em
$\intcc{0}{1}$ (suas séries de coeficientes convergem por
hipótese); fazendo $x \to 1^-$ na identidade: $AB = C$.

\textbf{4.} Aqui
\[
\abs{c_n} = \sum_{k=0}^{n}
\frac{1}{\sqrt{(k+1)(n-k+1)}}
\geq \sum_{k=0}^{n}\frac{2}{n+2}
= \frac{2(n+1)}{n+2} \geq 1,
\]
por MA--MG: $\sqrt{(k+1)(n-k+1)} \leq \frac{(k+1) +
(n-k+1)}{2} = \frac{n+2}{2}$. O termo geral de $\sum c_n$
não tende a $0$: o \omterm{thm:b2:series:fubini}{produto de Cauchy} diverge, embora
os dois fatores convirjam (séries alternadas).

\textbf{5.} \cref{exo:b2:powerseries:5} dá
$\frac{-\ln(1-x)}{1-x} = \sum H_nx^n$ ($\abs x < 1$).
Primitivas termo a termo
(\cref{thm:b2:powerseries:calculus}~(2)), anulando-se os dois lados
em $0$:
\[
\frac{\bigl(\ln(1-x)\bigr)^2}{2}
= \sum_{n\geq1}\frac{H_n}{n+1}\,x^{n+1} .
\]
Decrescimento: $(n+2)H_n \geq (n+1)H_{n+1}$ equivale a $H_n \geq
1$, verdadeiro para $n \geq 1$; e $\frac{H_n}{n+1} \sim \frac{\ln
n}{n} \to 0$: em $x = -1$ a série converge pelo
critério das alternadas. Substituindo $x \mapsto -x$ e aplicando a
questão 2 em $x = 1$:
\[
\frac{(\ln 2)^2}{2}
= \sum_{n\geq1}\frac{H_n}{n+1}(-1)^{n+1},
\]
o valor anunciado.

\textbf{6.} $f(x) = \sum(-1)^nx^n = \frac{1}{1+x} \to \frac12$
quando $x \to 1^-$: Abel-somável para $\frac12$. Mas as somas
parciais são $1, 0, 1, 0, \dots$: divergente.

\textbf{7.} Dado $\varepsilon > 0$, escolha $m$ com $\abs{u_n}
\leq \varepsilon$ para $n > m$; para $N \geq m$:
\[
\Bigl|\frac{u_1 + \dots + u_N}{N}\Bigr|
\leq \frac{\abs{u_1} + \dots + \abs{u_m}}{N} +
\varepsilon\,\frac{N - m}{N}
\leq \frac{C_m}{N} + \varepsilon,
\]
logo $\limsup \leq \varepsilon$ para todo $\varepsilon$: as
médias tendem a $0$.

\textbf{8.} Para $0 \leq x \leq 1$: $1 - x^n = (1-x)(1 + x +
\dots + x^{n-1}) \leq n(1-x)$, logo
\[
\Bigl|\sum_{n=0}^N a_n(1 - x_N^n)\Bigr|
\leq (1 - x_N)\sum_{n=1}^N n\abs{a_n}
= \frac1N\sum_{n=1}^{N}n\abs{a_n} .
\]
Para $n > N$: $\abs{a_n} = \frac{n\abs{a_n}}{n} \leq
\frac{1}{N}\sup_{m>N}m\abs{a_m}$, e $\sum_{n>N}x_N^n \leq
\frac{1}{1 - x_N} = N$:
\[
\Bigl|\sum_{n>N}a_nx_N^n\Bigr|
\leq \frac{\sup_{m>N}m\abs{a_m}}{N}\cdot N
= \sup_{m>N}\,m\abs{a_m} .
\]

\textbf{9.} Decomponha
\[
A_N - L = \sum_{n=0}^{N}a_n\bigl(1 - x_N^n\bigr)
- \sum_{n>N}a_nx_N^n + \bigl(f(x_N) - L\bigr) .
\]
O primeiro termo tende a $0$ pela questão 7 (as médias de
$n\abs{a_n} \to 0$), o segundo pela questão 8 (o sup tende a
$0$), o terceiro porque $x_N \to 1^-$ e $f(x) \to L$. Logo
$A_N \to L$: o teorema de Tauber.

\textbf{10.} Para $x \in \intco{0}{1}$ e qualquer $N$:
$\sum_{n\leq N}a_nx^n \leq f(x) \leq M$ (termos não negativos).
Faça $x \to 1^-$ na soma finita: $\sum_{n\leq N}a_n \leq M$.
As somas parciais são crescentes e limitadas: $\sum a_n$
converge, e então a questão 2 dá $\lim_{x\to1^-}f(x) =
\sum a_n$.

\textbf{11.} $\sigma_N - L$ é a média dos $N$ números
$A_n - L$ ($0 \leq n < N$), que tendem a $0$: questão 7.

\textbf{12.} $A_n = 1$ para $n$ par, $0$ para ímpar: $A_0 + \dots
+ A_{N-1} = \lceil N/2\rceil$, logo $\sigma_N =
\frac{\lceil N/2\rceil}{N} \to \frac12$, o valor de Abel da
questão 6.

\textbf{13.} Sob $\sigma_N \to L$ tem-se $S_n = O(n)$, logo
$A_n = S_n - S_{n-1} = O(n)$ e $a_n = O(n)$: todas as séries
abaixo têm raio $\geq 1$. Para $\abs x < 1$, de $a_n = A_n -
A_{n-1}$ e $A_nx^n \to 0$:
\[
(1-x)\sum_n A_nx^n = \sum_n A_nx^n - \sum_n A_nx^{n+1}
= \sum_n a_nx^n = f(x),
\]
e identicamente $(1-x)\sum S_nx^n = \sum A_nx^n$, logo $f(x) =
(1-x)^2\sum_n S_nx^n = (1-x)^2\sum_n(n+1)\sigma_{n+1}x^n$.
Por fim, $\sum(n+1)x^n = \frac{1}{(1-x)^2}$
(\cref{exo:b2:powerseries:3}), que é a segunda identidade.

\textbf{14.} Subtraindo $L$ vezes a segunda identidade da
primeira:
\[
f(x) - L = (1-x)^2\sum_{n\geq0}(n+1)
\bigl(\sigma_{n+1} - L\bigr)x^n .
\]
Dado $\varepsilon$, escolha $N$ com $\abs{\sigma_{n+1} - L}
\leq \varepsilon$ para $n \geq N$; então
\[
\abs{f(x) - L} \leq (1-x)^2 C_N +
\varepsilon(1-x)^2\sum_{n}(n+1)x^n
= (1-x)^2C_N + \varepsilon ,
\]
e fazendo $x \to 1^-$: $\limsup \leq \varepsilon$. Logo
$f(x) \to L$: o teorema de Frobenius.

\textbf{15.} $f(x) = \sum(-1)^n(n+1)x^n = \frac{1}{(1+x)^2}$
(derive a série geométrica em $-x$): valor de Abel
$\frac14$. Somas parciais: $A_{2k} = k+1$, $A_{2k+1} = -(k+1)$
(indução imediata). Então $S_{2m-1} = 0$ (pares consecutivos
se cancelam) e $S_{2m} = m + 1$, logo
\[
\sigma_{2m} = \frac{S_{2m-1}}{2m} = 0,
\qquad
\sigma_{2m+1} = \frac{m+1}{2m+1} \to \frac12 :
\]
$(\sigma_N)$ tem dois valores de acumulação distintos: não é
Cesàro-somável. Com as questões 6 e 11--14, a hierarquia
convergente $\Rightarrow$ Cesàro $\Rightarrow$ Abel é estrita
nas duas setas.

\textbf{16.} A série primitiva $F(x) =
\sum\frac{a_n}{n+1}x^{n+1}$ tem o mesmo raio e $F' = f$ em
$\intco{0}{1}$ (\cref{thm:b2:powerseries:calculus}); como
$\sum\frac{a_n}{n+1}$ converge, a Parte I (questão 2) torna $F$
\omterm{def:b2:metric:continuity}{contínua} em $\intcc{0}{1}$. Como $\int_0^x f = F(x)$ (derivadas
iguais, mesmo valor $0$ em $0$),
\[
\int_0^x f \xrightarrow[x\to1^-]{} F(1)
= \sum_{n\geq0}\frac{a_n}{n+1} :
\]
a \omterm{def:b2:integration:improper}{integral imprópria} existe com o valor enunciado.

\textbf{17.} $\frac{\ln(1+x)}{x} =
\sum_{n\geq1}\frac{(-1)^{n-1}}{n}x^{n-1}$ (raio $1$;
\omterm{def:b2:metric:continuity}{contínua} em $0$). A série de $\frac{a_m}{m+1}$ é
$\sum_{n\geq1}\frac{(-1)^{n-1}}{n^2}$, \omterm{def:b2:series:def}{absolutamente} convergente:
a questão 16 dá $\int_0^1\frac{\ln(1+x)}{x}\dd x = \eta$. Na
série \omterm{def:b2:series:def}{absolutamente} convergente $\sum\frac1{n^2}$, reagrupe pares e
ímpares:
\[
\eta = \sum_{\text{ímpar}}\frac1{n^2} -
\sum_{\text{par}}\frac1{n^2}
= \sum_{n}\frac1{n^2} - 2\sum_{k}\frac1{(2k)^2}
= \Bigl(1 - \frac12\Bigr)\sum_n\frac1{n^2}
= \frac12\sum_{n\geq1}\frac1{n^2} .
\]

\textbf{18.} Leibniz: $\frac{1}{3n+1}\downarrow0$, a série
converge. Em $\intco{0}{1}$, $\sum(-1)^nx^{3n} =
\frac{1}{1+x^3}$, e $\sum\frac{(-1)^n}{3n+1}$ converge:
a questão 16 dá $\sum\frac{(-1)^n}{3n+1} =
\int_0^1\frac{\dd x}{1+x^3}$. Frações parciais (confira:
$\frac13(x^2-x+1) + \frac{2-x}{3}(1+x) = 1$):
\[
\int_0^1\frac{\dd x}{1+x^3}
= \frac13\ln2 + \frac13\int_0^1\frac{2-x}{x^2-x+1}\dd x .
\]
Escrevendo $2 - x = -\frac12(2x-1) + \frac32$: a parte $\ln(x^2-x+1)$
se anula nas duas pontas, e
\[
\frac32\int_0^1\frac{\dd x}{(x-\frac12)^2 + \frac34}
= \frac32\cdot\frac{2}{\sqrt3}
\Bigl[\arctan\frac{2x-1}{\sqrt3}\Bigr]_0^1
= \sqrt3\cdot\frac{\pi}{3} = \frac{\pi}{\sqrt3} .
\]
Total: $\frac13\bigl(\ln2 + \frac{\pi}{\sqrt3}\bigr)$.

\textbf{19.} Substituindo $t = x^2$ na série de
\cref{exo:b2:powerseries:12} e integrando termo a
termo (a primitiva de $(1-x^2)^{-1/2}$ que se anula em $0$ é
$\arcsin$):
\[
\arcsin x = \sum_{n\geq0}
\frac{\binom{2n}n}{4^n(2n+1)}x^{2n+1}
\qquad(\abs x < 1) .
\]
Os coeficientes são $\sim \frac{1}{2\sqrt\pi\,n^{3/2}}$
(\cref{ex:b2:comparison:centralbinomial}), \omterm{def:b2:series:summable}{somáveis}: a série
converge \emph{\omterm{def:b2:funcseq:series}{normalmente}} em $\intcc{-1}{1}$, sua soma é
\omterm{def:b2:metric:continuity}{contínua} aí e coincide com a \omterm{def:b2:metric:continuity}{contínua} $\arcsin$
em $\intoo{-1}{1}$, logo também em $x = 1$:
\[
\sum_{n\geq0}\frac{\binom{2n}n}{4^n(2n+1)}
= \arcsin 1 = \frac\pi2 .
\]

\textbf{20.} $C_n4^{-n} \sim \frac{1}{\sqrt\pi\,n^{3/2}}$
(\cref{exo:b2:powerseries:8}): \omterm{def:b2:funcseq:series}{convergência normal} de $\sum
C_nx^n$ em $\intcc{0}{\frac14}$, logo sua soma é \omterm{def:b2:metric:continuity}{contínua}
aí; em $\intoo{0}{\frac14}$ ela vale
$\frac{1-\sqrt{1-4x}}{2x}$
(\cref{ex:b2:powerseries:catalan}), cujo limite em
$\frac14^-$ é $\frac{1-0}{1/2} = 2$. Portanto $\sum_{n\geq0}
C_n4^{-n} = 2$.

\textbf{21.} $f(x) = \sum_{n\geq m}a_nx^n = x^m g(x)$ com
$g(x) = \sum_{k\geq0}a_{m+k}x^k$; se $(a_nr^n)$ é limitada, então
$(a_{m+k}r^k)$ também é (divida por $r^m$): $g$ tem raio $\geq
R$. $g$ é \omterm{def:b2:metric:continuity}{contínua} com $g(0) = a_m \neq 0$, logo $g \neq 0$
em algum $\intcc{-\delta}{\delta}$, e $f(x) = x^mg(x) \neq 0$
para $0 < \abs x \leq \delta$.

\textbf{22.} $d = f - h$ é a soma de uma série de potências perto de $0$
que se anula nos pontos não nulos $x_k \to 0$. Se algum
coeficiente de $d$ fosse não nulo, a questão 21 daria uma
vizinhança perfurada de $0$ livre de zeros de $d$ ---
contradizendo $d(x_k) = 0$. Logo todos os coeficientes de $d$ se anulam:
$f$ e $h$ têm coeficientes iguais e coincidem perto de $0$.

\textbf{23.} A função $h(x) = \frac{1}{1+x^2} =
\sum(-1)^nx^{2n}$ (raio $1$) satisfaz $h(\frac1k) =
\frac{1}{1 + 1/k^2} = \frac{k^2}{k^2+1}$. Toda $f$ \omterm{def:b2:powerseries:analytic}{analítica} com
os mesmos valores coincide com $h$ nos pontos $\frac1k \to 0$:
pelo teorema da identidade (questão 22), $f = \frac{1}{1+x^2}$
perto de $0$ --- a única solução.

\textbf{24.} Seja $Z$ o conjunto dos pontos de $I$ que têm uma
vizinhança na qual $f$ se anula identicamente: \omterm{def:b2:metric:topology}{aberto} por
definição, não vazio (o subintervalo). Fechado em $I$: se $y
\in I$ é limite de pontos de $Z$, então $y$ é ponto de
acumulação de zeros de $f$; desenvolvendo $f$ em série de potências em $y$
(analiticidade) e aplicando as questões 21--22 recentradas em $y$,
todos os coeficientes em $y$ se anulam, logo $f \equiv 0$ perto de $y$: $y
\in Z$. Um intervalo é \omterm{def:b2:metric:connected}{conexo}, logo $Z = I$: $f \equiv 0$ em
$I$. Em particular, uma \omterm{def:b2:powerseries:analytic}{função analítica} em $\R$ que se anula
fora de um \omterm{def:b2:metric:compact}{compacto} se anula num intervalo, logo
em toda parte: nenhuma bolha \omterm{def:b2:powerseries:analytic}{analítica} não nula. O mundo $C^\infty$ é
diferente: colar a função achatada do
\cref{exo:b2:powerseries:7} (por exemplo,\ $x \mapsto
\eu^{-1/x^2}\mathbf 1_{x>0}$ e sua espelhada) produz bolhas
suaves de suporte \omterm{def:b2:metric:compact}{compacto}.

\textbf{25.} (i) A \omterm{def:b2:funcseq:series}{convergência normal} mora em subdiscos \omterm{def:b2:metric:compact}{compactos}
estritamente dentro do disco; o teorema de Abel estende a \omterm{def:b2:metric:continuity}{continuidade} a
um ponto da fronteira, sob a única hipótese de que a série dos
coeficientes convirja ali. (ii) A recíproca vale
sob a condição de Tauber $na_n \to 0$ (questão 9), e a
\omterm{def:b2:series:summable}{somabilidade} de Cesàro fica estritamente entre a convergência e a
\omterm{def:b2:series:summable}{somabilidade} de Abel (Frobenius, questões 14--15). (iii) Para um
amigo: $\sum\frac{(-1)^n}{3n+1} = \int_0^1\frac{\dd x}{1+x^3}$,
integrando a série geométrica até a fronteira e depois usando
frações parciais. (iv) As médias de Cesàro voltam no capítulo de Fourier
como o teorema de Fejér, em que promediar as somas parciais
conserta o fracasso da \omterm{def:b2:funcseq:def}{convergência pontual} --- mesmo remédio,
paciente novo.
\end{solution}
