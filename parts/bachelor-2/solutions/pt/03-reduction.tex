\section*{Capítulo \ref{ch:b2:reduction} --- Redução de endomorfismos}

\begin{solution}{exo:b2:reduction:1}
$A$: $\chi_A = X^2 - 2X - 3 = (X - 3)(X + 1)$. \omterm{def:b2:reduction:eigen}{Autovetores}: para
$3$: $(1,1)$; para $-1$: $(1,-1)$. Logo $P = \begin{pmatrix} 1 & 1\\ 1
& -1\end{pmatrix}$ dá $P^{-1}AP = \operatorname{diag}(3, -1)$.

$B = J - I$, em que $J$ é a matriz de uns. $J$ tem posto $1$ com
$Jv = 3v$ para $v = (1,1,1)$ e $Jw = 0$ no plano $x + y + z =
0$: o \omterm{def:b2:reduction:eigen}{espectro} de $B$ é $\{2, -1\}$ com \omterm{def:b2:reduction:eigen}{autoespaços}
$\operatorname{Vect}(1,1,1)$ (dimensão $1$) e $\{x + y + z = 0\}$
(dimensão $2$, base $(1,-1,0), (1,0,-1)$). $P$ com essas três
colunas dá $P^{-1}BP = \operatorname{diag}(2, -1, -1)$.
\end{solution}

\begin{solution}{exo:b2:reduction:2}
\emph{\omterm{def:b2:reduction:eigen}{Autoespaços}:} $\chi_C = (X-1)^2$, único \omterm{def:b2:reduction:eigen}{autovalor} $1$;
$\ker(C - I) = \ker\begin{pmatrix} 0&1\\ 0&0\end{pmatrix}$ é a
reta $\operatorname{Vect}(e_1)$: dimensão $1 < 2 = m_1$, logo não é
\omterm{def:b2:reduction:diag}{diagonalizável} (\cref{thm:b2:reduction:diagcrit}).

\emph{\omterm{def:b2:reduction:polyu}{Polinômio minimal}:} $\mu_C$ divide $(X-1)^2$ e $C \neq I$,
logo $\mu_C = (X-1)^2$: uma raiz dupla, logo não é \omterm{def:b2:reduction:diag}{diagonalizável}
(\cref{cor:b2:reduction:minpolycrit}).
\end{solution}

\begin{solution}{exo:b2:reduction:3}
$X^2 - 5X + 6 = (X-2)(X-3)$: decompõe-se com raízes simples, logo $u$ é
\omterm{def:b2:reduction:diag}{diagonalizável} (\cref{cor:b2:reduction:minpolycrit}), com
$\operatorname{Sp}(u) \subseteq \{2, 3\}$. \omterm{def:b2:reduction:eigen}{Espectros} possíveis:
$\{2\}$ ($u = 2\,\mathrm{id}$), $\{3\}$ ($u = 3\,\mathrm{id}$) ou
$\{2, 3\}$.

Potências: procure $u^k = a_k\,\mathrm{id} + b_k\,u$. Nos \omterm{def:b2:reduction:eigen}{autoespaços},
isso se lê $2^k = a_k + 2b_k$ e $3^k = a_k + 3b_k$: resolvendo, $b_k
= 3^k - 2^k$, $a_k = 3\cdot2^k - 2\cdot 3^k$:
\[
u^k = (3\cdot 2^k - 2\cdot 3^k)\,\mathrm{id} + (3^k - 2^k)\, u .
\]
(Válido para os três \omterm{def:b2:reduction:eigen}{espectros}: as identidades valem \omterm{def:b2:reduction:eigen}{autovalor} a \omterm{def:b2:reduction:eigen}{autovalor}.)
\end{solution}

\begin{solution}{exo:b2:reduction:4}
$u$ \omterm{def:b2:reduction:diag}{diagonalizável}: $P = \prod_{\lambda}(X - \lambda)$ sobre o
\omterm{def:b2:reduction:eigen}{espectro} anula $u$, decompõe-se e tem raízes simples. Então $P(u|_F) =
P(u)|_F = 0$: a restrição é anulada por um polinômio que se decompõe
com raízes simples, logo é \omterm{def:b2:reduction:diag}{diagonalizável}
(\cref{cor:b2:reduction:minpolycrit}).
\end{solution}

\begin{solution}{exo:b2:reduction:5}
$\chi_A = X^2 - X - 1$, de raízes $\varphi$ e $\psi$ (distintas):
\omterm{def:b2:reduction:diag}{diagonalizável}, com \omterm{def:b2:reduction:eigen}{autovetores} $(\varphi, 1)$ e $(\psi, 1)$.
A recorrência dá $\begin{pmatrix} F_{n+1}\\ F_n \end{pmatrix} =
A^n \begin{pmatrix}1\\ 0\end{pmatrix}$. Decomponha $(1, 0)$ nos
\omterm{def:b2:reduction:eigen}{autovetores}: $(1,0) = \frac{1}{\varphi - \psi}\bigl((\varphi, 1) -
(\psi, 1)\bigr)$ com $\varphi - \psi = \sqrt5$. Aplicar $A^n$
multiplica cada componente própria pela $n$-ésima potência de seu \omterm{def:b2:reduction:eigen}{autovalor};
lendo a segunda coordenada:
\[
F_n = \frac{\varphi^n - \psi^n}{\sqrt 5} .
\]
(Confira: $n = 1$ dá $\frac{\varphi - \psi}{\sqrt5} = 1$.)
\end{solution}

\begin{solution}{exo:b2:reduction:6}
Seja $P = \prod_i (X - \mu_i)$ um \omterm{def:b2:linalg:annihilator}{anulador} de $u^2$ que se decompõe com raízes
simples $\mu_i$ (o \omterm{def:b2:reduction:eigen}{espectro} de $u^2$). Como $u$ é invertível, $0$
não é \omterm{def:b2:reduction:eigen}{autovalor} de $u^2$ ($\det u^2 = (\det u)^2 \neq 0$), logo
todos os $\mu_i \neq 0$. Então
\[
Q(X) = \prod_i (X^2 - \mu_i) = \prod_i (X - \sqrt{\mu_i})(X +
\sqrt{\mu_i})
\]
anula $u$: $\;Q(u) = \prod_i (u^2 - \mu_i\,\mathrm{id}) =
P(u^2) = 0$. Suas raízes $\pm
\sqrt{\mu_i}$ (raízes quadradas complexas) são duas a duas distintas porque
os $\mu_i$ são distintos e não nulos ($\sqrt{\mu_i} = -\sqrt{\mu_j}$
daria $\mu_i = \mu_j$). Decompõe-se com raízes simples: $u$ é
\omterm{def:b2:reduction:diag}{diagonalizável}.

Contraexemplo sem invertibilidade: $u = \begin{pmatrix} 0 & 1\\
0 & 0\end{pmatrix}$: $u^2 = 0$ é \omterm{def:b2:reduction:diag}{diagonalizável}, $u$ não é.
\end{solution}

\begin{solution}{exo:b2:reduction:7}
$A = 2I + N$ com $N$ o deslocamento ($N e_2 = e_1$, $Ne_3 = e_2$),
$N^3 = 0$, $N^2 = E_{13}$: isso \emph{é} a \omterm{thm:b2:reduction:dunford}{decomposição de Dunford}
($2I$ diagonal, $N$ nilpotente, e comutam; a unicidade faz dela
a decomposição). Binômio com termos que comutam:
\[
A^k = 2^k I + k 2^{k-1} N + \binom k2 2^{k-2} N^2
= \begin{pmatrix}
2^k & k2^{k-1} & \binom k2 2^{k-2}\\
0 & 2^k & k2^{k-1}\\
0 & 0 & 2^k
\end{pmatrix},
\]
\[
\eu^{tA} = \eu^{2t}\Bigl(I + tN + \frac{t^2}{2}N^2\Bigr)
= \eu^{2t}\begin{pmatrix}
1 & t & t^2/2\\
0 & 1 & t\\
0 & 0 & 1
\end{pmatrix}.
\]
\end{solution}

\begin{solution}{exo:b2:reduction:8}
$X^k - 1$ anula $A$ e se decompõe sobre $\C$ com as $k$ raízes
distintas $\eu^{2\iu\pi j/k}$: $A$ é \omterm{def:b2:reduction:diag}{diagonalizável}
(\cref{cor:b2:reduction:minpolycrit}) e seus \omterm{def:b2:reduction:eigen}{autovalores}, raízes de
$X^k - 1$, são raízes $k$-ésimas da unidade.

Se além disso $A$ é semelhante a uma matriz triangular com diagonal
de uns: todos os \omterm{def:b2:reduction:eigen}{autovalores} valem $1$, e $A$, \omterm{def:b2:reduction:diag}{diagonalizável} com
único \omterm{def:b2:reduction:eigen}{autovalor} $1$, é $P\,I\,P^{-1} = I$.
\end{solution}

\begin{solution}{exo:b2:reduction:9}
Escreva $E = \bigoplus_\lambda E_\lambda(u)$
(\cref{thm:b2:reduction:diagcrit}). Cada $E_\lambda(u)$ é
estável por $v$: para $x \in E_\lambda$, $u(v(x)) = v(u(x)) = \lambda
v(x)$. A restrição de $v$ a $E_\lambda(u)$ é \omterm{def:b2:reduction:diag}{diagonalizável}
(\cref{exo:b2:reduction:4}): escolha uma base de $E_\lambda(u)$ formada
de \omterm{def:b2:reduction:eigen}{autovetores} de $v$. Concatenando essas bases sobre todos os $\lambda$
obtém-se uma base de $E$ cujos vetores são \omterm{def:b2:reduction:eigen}{autovetores} de \emph{ambos}
$u$ (por pertencerem a $E_\lambda(u)$) e $v$ (por construção).
\end{solution}

\begin{solution}{exo:b2:reduction:10}
($\Rightarrow$) Seja $u$ \omterm{def:b2:reduction:diag}{diagonalizável} e $F$ estável. Então
$u|_F$ é \omterm{def:b2:reduction:diag}{diagonalizável} (\cref{exo:b2:reduction:4}): $F$ tem uma
base de \omterm{def:b2:reduction:eigen}{autovetores}, que se estende, dentro de cada \omterm{def:b2:reduction:eigen}{autoespaço} global
$E_\lambda$, a uma base de $E_\lambda$ (teorema da base incompleta
dentro de $E_\lambda$, partindo da parte da base de $F$ que aí se encontra
--- note que $F = \bigoplus_\lambda (F \cap E_\lambda)$, pois
$u|_F$ é \omterm{def:b2:reduction:diag}{diagonalizável}). Os vetores acrescentados geram um
suplementar estável (cada um está em algum $E_\lambda$, logo o espaço que geram é
estável por $u$).

($\Leftarrow$) Seja $F = \sum_\lambda E_\lambda(u)$ (um subespaço
estável) e $G$ um suplementar estável. Se $G \neq \{0\}$: $\chi_{u|_G}$
se decompõe sobre $\C$, logo $u|_G$ tem um \omterm{def:b2:reduction:eigen}{autovetor} $x \in G$
(\cref{thm:b2:reduction:trigonalization} ou diretamente a existência
de uma raiz); mas todo \omterm{def:b2:reduction:eigen}{autovetor} de $u$ está em $F$, logo $x \in F
\cap G = \{0\}$: contradição. Portanto $G = \{0\}$ e $E = F$: os
\omterm{def:b2:reduction:eigen}{autoespaços} preenchem $E$, isto é, $u$ é \omterm{def:b2:reduction:diag}{diagonalizável}.
\end{solution}

\begin{solution}{exo:b2:reduction:11}
Triangularize: $A = PTP^{-1}$, $T = \begin{pmatrix} \lambda & c\\ 0 &
\mu\end{pmatrix}$, $\abs\lambda, \abs\mu < 1$. Então $A^k =
PT^kP^{-1}$, e basta que $T^k \to 0$.

\emph{\omterm{def:b2:reduction:eigen}{Autovalores} distintos:} a indução dá
\[
T^k = \begin{pmatrix}
\lambda^k & c\,\dfrac{\lambda^k - \mu^k}{\lambda - \mu}\\[4pt]
0 & \mu^k
\end{pmatrix},
\]
e cada entrada tende a $0$ ($\abs{\lambda}^k, \abs\mu^k \to 0$).

\emph{\omterm{def:b2:reduction:eigen}{Autovalores} iguais ($\mu = \lambda$):} $T = \lambda I + cE_{12}$
e $T^k = \lambda^k I + k\lambda^{k-1}cE_{12}$; a entrada
$k\lambda^{k-1} \to 0$, pois $\abs\lambda < 1$ (a geométrica vence a
polinomial). Nos dois casos $T^k \to 0$ entrada a entrada, logo $A^k =
PT^kP^{-1} \to 0$ (a multiplicação matricial por $P, P^{-1}$ fixas é
contínua nas entradas --- cada entrada do produto é uma combinação
linear fixa).
\end{solution}

\begin{solution}{exo:b2:reduction:12}
$\ker u$ tem dimensão $n - 1$ (teorema do núcleo e da imagem), logo $0$ é
\omterm{def:b2:reduction:eigen}{autovalor} de \omterm{def:b2:reduction:charpoly}{multiplicidade geométrica} $n - 1$, e $\chi_u$ é
divisível por $X^{n-1}$
(\cref{def:b2:reduction:charpoly}: geométrica $\leq$ algébrica).
Escreva $\chi_u = X^{n-1}(X - \alpha)$; sendo o coeficiente de
$X^{n-1}$ igual a $-\operatorname{tr} u$, obtemos $\alpha =
\operatorname{tr} u$: $\chi_u = X^{n-1}(X - \operatorname{tr}
u)$.

Se $\operatorname{tr} u \neq 0$: o \omterm{def:b2:reduction:eigen}{autovalor}
$\operatorname{tr} u$ é raiz de $\chi_u$, logo carrega um
\omterm{def:b2:reduction:eigen}{autovetor}; os \omterm{def:b2:reduction:eigen}{autoespaços} de $0$ e de $\operatorname{tr} u$
têm dimensões $n - 1$ e $\geq 1$, com soma $\geq n$: eles
preenchem $E$, e $u$ é \omterm{def:b2:reduction:diag}{diagonalizável}
(\cref{thm:b2:reduction:diagcrit}). Se $\operatorname{tr} u = 0$:
pelo~\cref{exo:b2:linalg:5}, $u^2 = (\operatorname{tr} u)u = 0$
com $u \neq 0$: $u$ é um nilpotente não nulo, e um nilpotente
\omterm{def:b2:reduction:diag}{diagonalizável} é nulo (\cref{prop:b2:reduction:nilpotent}): não é
\omterm{def:b2:reduction:diag}{diagonalizável}.
\end{solution}

\begin{solution}{pb:b2:reduction:1}
\textbf{1.} As $k - 1$ primeiras coordenadas de $Cv_n$ são $u_{n+1},
\dots, u_{n+k-1}$ (a superdiagonal desloca), e a última é
$a_0 u_n + \dots + a_{k-1}u_{n+k-1}$. Assim $v_{n+1} = Cv_n$ vale
para todo $n$ se e somente se as últimas coordenadas coincidem para todo $n$, isto é, se e somente se
$(\mathcal R)$ vale. Iterando, $v_n = C^nv_0$.

\textbf{2.} Desenvolva $D_k(X) = \det(XI_k - C)$ ao longo da primeira
coluna: as duas entradas não nulas são $X$ (posição $(1,1)$) e
$-a_0$ (posição $(k,1)$). O primeiro menor tem a forma $D_{k-1}$
para os coeficientes $a_1, \dots, a_{k-1}$; o segundo menor é
triangular superior com diagonal $-1$: \omterm{def:b2:linalg:det}{determinante} $(-1)^{k-1}$,
com sinal $(-1)^{k+1}$ vindo da posição. A indução sobre $k$
(base $k = 1$: $X - a_0$) dá
\[
D_k(X) = X\bigl(X^{k-1} - a_{k-1}X^{k-2} - \dots - a_1\bigr) -
a_0 = P(X).
\]
Para $\mu_C$: como $Q(C^{\mathsf T}) = Q(C)^{\mathsf T}$ para qualquer
polinômio, $C$ e $C^{\mathsf T}$ têm os mesmos \omterm{def:b2:linalg:annihilator}{anuladores},
logo o mesmo \omterm{def:b2:reduction:polyu}{polinômio minimal}. Para $C^{\mathsf T}$: as
colunas se leem $C^{\mathsf T}e_1 = e_2$, \dots, $C^{\mathsf
T}e_{k-1} = e_k$, de modo que $(e_1, C^{\mathsf T}e_1, \dots, (C^{\mathsf
T})^{k-1}e_1)$ é a base canônica: livre. Um polinômio $Q
\neq 0$ de grau $< k$ tem então $Q(C^{\mathsf T})e_1 \neq 0$
(é uma combinação não trivial de vetores da base): $\deg\mu \geq
k$. Como $\mu \mid \chi = P$ com $\deg P = k$: $\mu_C = P$.

\textbf{3.} Para $v = (1, \lambda, \dots, \lambda^{k-1})^{\mathsf
T}$: as linhas $1$ a $k-1$ de $Cv$ dão $\lambda, \lambda^2, \dots,
\lambda^{k-1}$, isto é, $\lambda$ vezes as primeiras $k - 1$ entradas
de $v$; a última linha dá $\sum_m a_m\lambda^m = \lambda^k -
P(\lambda) = \lambda^k = \lambda\cdot\lambda^{k-1}$. Logo $Cv =
\lambda v$. Reciprocamente, as equações $(Cx)_i = \lambda x_i$ para
$i < k$ leem-se $x_{i+1} = \lambda x_i$: todo \omterm{def:b2:reduction:eigen}{autovetor} é
proporcional a $v$ --- todo \omterm{def:b2:reduction:eigen}{autoespaço} tem dimensão exatamente
$1$. \omterm{def:b2:reduction:diag}{Diagonalizável} se e somente se as dimensões dos \omterm{def:b2:reduction:eigen}{autoespaços} somam $k$
(\cref{thm:b2:reduction:diagcrit}), se e somente se há $k$ \omterm{def:b2:reduction:eigen}{autovalores}
distintos, se e somente se $P$ tem $k$ raízes distintas (os \omterm{def:b2:reduction:eigen}{autovalores} são
as raízes de $\chi_C = P$).

\textbf{4.} Cada $(\lambda_i^n)_n$ resolve $(\mathcal R)$:
$\lambda_i^{n+k} = \lambda_i^n\,\lambda_i^k =
\lambda_i^n\sum_m a_m\lambda_i^m$. Liberdade: uma combinação
nula $\sum_i c_i\lambda_i^n = 0$ para $n = 0, \dots, k-1$
é um sistema de Vandermonde (\cref{exo:b2:linalg:11}) nos $c_i$:
todos os $c_i = 0$. O espaço das soluções tem dimensão $k$ (questão 6,
cuja demonstração é elementar e independente): $k$ soluções livres
formam uma base, e as coordenadas são únicas.

\textbf{5.} $P = X^2 - X - 6 = (X - 3)(X + 2)$: solução geral
$u_n = A\,3^n + B(-2)^n$. Condições iniciais: $A + B = 1$, $3A -
2B = 8$: $A = 2$, $B = -1$:
\[
u_n = 2\cdot 3^n - (-2)^n .
\]
(Confira: $u_2 = u_1 + 6u_0 = 14$ e $2\cdot9 - 4 = 14$.)

\textbf{6.} $P(S)\bigl((u_n)\bigr)$ é a sequência $n \mapsto
u_{n+k} - a_{k-1}u_{n+k-1} - \dots - a_0u_n$: ela se anula se e somente se
$(\mathcal R)$ vale, de modo que o conjunto solução é $\ker P(S)$, um
subespaço. A aplicação $\ker P(S) \to \C^k$, $u \mapsto (u_0, \dots,
u_{k-1})$, é linear, injetiva (a recorrência determina
$u_{k}, u_{k+1}, \dots$ a partir dos $k$ primeiros valores, por indução)
e sobrejetiva (defina $u_n$ recursivamente a partir de dados iniciais quaisquer):
dimensão $k$.

\textbf{7.} A demonstração do~\cref{thm:b2:reduction:kernels} usa
apenas: a identidade de Bézout em $\C[X]$ e o fato de que
polinômios num endomorfismo fixo comutam. Nenhum dos dois menciona
a dimensão do espaço ambiente: o lema vale palavra por palavra para
$S \in \mathcal{L}(\mathcal{S})$. Logo
\[
\ker P(S) = \bigoplus_{i=1}^{r}
\ker\,(S - \lambda_i\,\mathrm{id})^{m_i}.
\]

\textbf{8.} Para $Q \in \C[X]$: $(S -
\lambda)\bigl(Q(n)\lambda^n\bigr)_n$ tem $n$-ésimo termo
$Q(n{+}1)\lambda^{n+1} - \lambda Q(n)\lambda^n =
\lambda^{n+1}(\Delta Q)(n)$, com $\Delta Q = Q(X{+}1) - Q(X)$
de grau $\deg Q - 1$ (os termos dominantes se cancelam). Iterando,
$(S - \lambda)^m\bigl(Q(n)\lambda^n\bigr) =
\bigl(\lambda^{n+m}(\Delta^m Q)(n)\bigr)_n$, e $\Delta^m Q = 0$
quando $\deg Q \leq m - 1$: o conjunto da direita está contido no
núcleo. Ele é um subespaço de dimensão $m$: as sequências
$(n^j\lambda^n)_n$, $0 \leq j < m$, são livres, pois
$\sum_j c_j n^j\lambda^n = 0$ para todo $n$ força (dividindo por
$\lambda^n \neq 0$) o polinômio $\sum_j c_jX^j$ a se anular em
todo $n \in \N$, logo a ser nulo. Reciprocamente, $\dim\ker(S -
\lambda)^m \leq m$: desenvolvendo $(S - \lambda)^m = \sum_j
\binom mj(-\lambda)^{m-j}S^j$, a equação $(S - \lambda)^m u =
0$ é uma recorrência linear de ordem $m$ (coeficiente dominante
$1$), de modo que $u$ fica determinada por $u_0, \dots, u_{m-1}$ como na
questão 6. A igualdade das dimensões conclui.

\textbf{9.} Combine as questões 7 e 8: toda solução se decompõe
de modo único como soma de elementos dos $\ker(S -
\lambda_i)^{m_i}$, isto é, $u_n = \sum_i Q_i(n)\lambda_i^n$ com
$\deg Q_i \leq m_i - 1$; os $Q_i$ são únicos porque a
decomposição é direta e, dentro de cada parcela, os
coeficientes de $Q_i$ são coordenadas na base
$(n^j\lambda_i^n)_j$ (questão 8). Verificação de bom senso nas dimensões:
$\sum_i m_i = k$.

\textbf{10.} $P = X^2 - 4X + 4 = (X - 2)^2$: soluções $(a +
bn)2^n$. Dados iniciais: $a = 1$, $2(a + b) = 0$, logo $b = -1$:
\[
u_n = (1 - n)\,2^n .
\]
Confira: $u_2 = 4u_1 - 4u_0 = -4$ e $(1 - 2)\cdot4 = -4$.

\textbf{11.} Escreva $u_n = \lambda_1^n\bigl(c_1 + \sum_{i\geq2}
c_i(\lambda_i/\lambda_1)^n\bigr)$; cada razão tem módulo $< 1$,
logo o colchete tende a $c_1 \neq 0$: $u_n \sim c_1\lambda_1^n$.
Em particular $u_n \neq 0$ para $n$ grande, e
\[
\frac{u_{n+1}}{u_n} =
\lambda_1\,\frac{c_1 + o(1)}{c_1 + o(1)} \longrightarrow
\lambda_1 .
\]

\textbf{12.} Calcule:
\[
q(a_{n+1}, b_{n+1}) = (a_n + 2b_n)^2 - 2(a_n + b_n)^2
= -a_n^2 + 2b_n^2 = -q(a_n, b_n).
\]
Com $q(a_0, b_0) = 1 - 2 = -1$: $a_n^2 - 2b_n^2 = (-1)^{n+1}$.
Estruturalmente: $q(a, b) = (a - \sqrt2\,b)(a + \sqrt2\,b)$ e
a aplicação linear $M$ multiplica o fator $a + \sqrt2 b$ por $1 +
\sqrt2$ e o fator $a - \sqrt2 b$ por $1 - \sqrt2$ (calcule:
$a_{n+1} + \sqrt2 b_{n+1} = (1 + \sqrt2)(a_n + \sqrt2 b_n)$);
o produto é multiplicado por $(1 + \sqrt2)(1 - \sqrt2) = -1 =
\det M$ a cada passo.

\textbf{13.} Como $a_n^2 - 2b_n^2 = (a_n - \sqrt2 b_n)(a_n +
\sqrt2 b_n) = (-1)^{n+1}$,
\[
\Bigl|\frac{a_n}{b_n} - \sqrt2\Bigr|
= \frac{\abs{a_n^2 - 2b_n^2}}{b_n(a_n + \sqrt2 b_n)}
= \frac{1}{b_n(a_n + \sqrt2 b_n)} \leq \frac1{2b_n^2},
\]
usando $a_n \geq b_n \geq 1$ (indução: ambos crescem), logo $a_n +
\sqrt2 b_n \geq (1 + \sqrt2)b_n \geq 2b_n$. \omterm{def:b2:reduction:eigen}{Autovalores} de $M$:
$\chi_M = X^2 - 2X - 1$, de raízes $1 \pm \sqrt2$; como $(a_0, b_0)$
tem componente não nula no \omterm{def:b2:reduction:eigen}{autovetor} dominante (todas as entradas
positivas), $b_n \sim c(1 + \sqrt2)^n$ com $c > 0$ (questão
11). Logo o erro é $\asymp (1 + \sqrt2)^{-2n} = (3 +
2\sqrt2)^{-n}$: decaimento geométrico de razão $1/(3 + 2\sqrt2) = 3
- 2\sqrt2 \approx 0.172$.

\textbf{14.} (a) Da questão 9: $\abs{u_n} \leq \sum_i
\abs{Q_i(n)}\abs{\lambda_i}^n \leq \bigl(\sum_i
\abs{Q_i(n)}\bigr)\rho^n$, e cada $\abs{Q_i(n)} \leq C_i
n^{m_i - 1} \leq C_i n^{m-1}$ para $n \geq 1$: some as constantes.
(b) Sejam $\rho' = \max_{i \geq 2}\abs{\lambda_i} <
\abs{\lambda_1}$ e $d = \deg Q_1$, de coeficiente dominante $c
\neq 0$. Então $u_n = Q_1(n)\lambda_1^n + R_n$ com $\abs{R_n}
\leq Cn^{m-1}\rho'^n$, e
\[
\frac{R_n}{Q_1(n)\lambda_1^n} = O\Bigl(n^{m-1-d}
\bigl(\rho'/\abs{\lambda_1}\bigr)^n\Bigr) \longrightarrow 0
\]
(a geométrica vence a polinomial). Logo
\[
u_n \sim Q_1(n)\,\lambda_1^n \sim c\,n^d\lambda_1^n,
\qquad
\frac{u_{n+1}}{u_n} \longrightarrow \lambda_1
\quad\text{(pois } Q_1(n{+}1)/Q_1(n) \to 1\text{)}.
\]
Confira na questão 10: para $u_n = (1-n)2^n$ a razão é
\[
\frac{(-n)2^{n+1}}{(1-n)2^n} = 2\,\frac{-n}{1-n}
\longrightarrow 2 = \lambda_1 .
\]

\textbf{15.} Indução sobre $n$. Para $n = 1$, $A_{ij}$ conta
os passeios de comprimento $1$. Passo: um passeio de comprimento $n + 1$ de $i$ a
$j$ é um passeio de comprimento $n$ de $i$ até algum vértice $\ell$
seguido de uma aresta $\ell j$:
\[
\#\{\text{passeios}\} = \sum_{\ell} (A^n)_{i\ell}A_{\ell j} =
(A^{n+1})_{ij}.
\]

\textbf{16.} $J = 3\Pi$, em que $\Pi = J/3$ é a projeção sobre
$\operatorname{Vect}(1,1,1)$ ao longo do plano $x + y + z = 0$
($\Pi^2 = \Pi$, pois $J^2 = 3J$). Então $A = J - I = 2\Pi - (I -
\Pi)$ e, como $\Pi$ e $I - \Pi$ são projeções
complementares,
\[
A^n = 2^n\,\Pi + (-1)^n (I - \Pi),
\qquad\text{i.e.}\qquad
(A^n)_{ij} = \frac{2^n}3 + (-1)^n\Bigl(\delta_{ij} -
\frac13\Bigr),
\]
o que dá as duas fórmulas exibidas. Em $n = 2$: diagonal
$(4 + 2)/3 = 2$ (passeios $i \to \ell \to i$ para os dois vizinhos
$\ell$); fora da diagonal $(4 - 1)/3 = 1$ (o único passeio $i \to
\ell \to j$ pelo terceiro vértice).

\textbf{17.} Sejam $w_n^{(0)}, w_n^{(1)}$ o número de palavras admissíveis de
comprimento $n$ terminadas em $0$, resp.\ em $1$. Acrescentando uma letra: um $0$
pode seguir qualquer coisa, um $1$ só pode seguir um $0$:
\[
\begin{pmatrix} w_{n+1}^{(0)}\\ w_{n+1}^{(1)}\end{pmatrix}
= \begin{pmatrix} 1 & 1\\ 1 & 0\end{pmatrix}
\begin{pmatrix} w_n^{(0)}\\ w_n^{(1)}\end{pmatrix}.
\]
Somando, $w_{n+2} = w_{n+1} + w_n$ (ou: condicione na primeira
letra). Com $w_1 = 2$, $w_2 = 3$: $w_n = F_{n+2}$ por indução
($F_3 = 2$, $F_4 = 3$, mesma recorrência). Crescimento: as raízes de
$X^2 - X - 1$ são $\varphi > \abs\psi$
(\cref{exo:b2:reduction:5}), e a componente em $\varphi$ é
não nula (os $w_n$ são positivos e $\psi^n \to 0$), logo a questão
11 dá $w_{n+1}/w_n \to \varphi = \frac{1 + \sqrt5}2$.

\textbf{18.} $A = \begin{psmallmatrix} 0&1&0\\ 1&0&1\\
0&1&0\end{psmallmatrix}$. Verificação:
\[
A(1, \pm\sqrt2, 1)^{\mathsf T}
= (\pm\sqrt2, 2, \pm\sqrt2)^{\mathsf T}
= \pm\sqrt2\,(1, \pm\sqrt2, 1)^{\mathsf T},
\qquad
A(1, 0, -1)^{\mathsf T} = 0 :
\]
\omterm{def:b2:reduction:eigen}{autovalores} $\sqrt2, -\sqrt2, 0$ ($= 2\cos\frac\pi4,
2\cos\frac{3\pi}4, 2\cos\frac\pi2$). Decomponha $e_1$ na
base de \omterm{def:b2:reduction:eigen}{autovetores} e leia a terceira coordenada, ou use a simetria: com
$v_\pm = (1, \pm\sqrt2, 1)$, $v_0 = (1, 0, -1)$, verifica-se $e_1
= \frac14 v_+ + \frac14 v_- + \frac12 v_0$, de modo que, para $n \geq 1$,
\[
(A^n)_{13} = \Bigl(\tfrac14(\sqrt2)^n v_+ +
\tfrac14(-\sqrt2)^n v_- + 0\Bigr)_{\!3}
= \frac{(\sqrt2)^n + (-\sqrt2)^n}{4},
\]
zero para $n$ ímpar (grafo bipartido: as pontas estão a distância par),
e $2\cdot 2^{n/2}/4 = 2^{n/2 - 1}$ para $n$ par. Em $n = 4$:
$2^{1} = 2$, o que corresponde aos dois passeios $1\,2\,1\,2\,3$ e
$1\,2\,3\,2\,3$.

\textbf{19.} Os passeios fechados de comprimento $n$ a partir de $i$ são
$(A^n)_{ii}$; somando sobre $i$ obtém-se $\operatorname{tr}(A^n)$.
Triangularizando $A$ (sobre $\C$), $A^n$ é triangular com diagonal
$\lambda_i^n$: $\operatorname{tr}(A^n) = \sum_i\lambda_i^n$.
Triângulo: $\operatorname{tr}(A^n) = 3\,\frac{2^n + 2(-1)^n}3 =
2^n + 2(-1)^n = 2^n + (-1)^n + (-1)^n$: o \omterm{def:b2:reduction:eigen}{espectro} $\{2, -1,
-1\}$, coerente com a questão 16.

\textbf{20.} As colunas de $W^{\mathsf T}$: $W^{\mathsf T}e_i =
e_{i-1}$ para $i \geq 1$ e $W^{\mathsf T}e_0 = e_{k-1}$;
renomeando na ordem $e_0, e_1, \dots$ isso é exatamente a
matriz companheira de $X^k - 1$ ($a_0 = 1$, os demais $a_m = 0$).
Questão 2: $\chi_{W} = \chi_{W^{\mathsf T}} = X^k - 1 =
\mu_{W}$. As raízes $\omega^j$ ($j = 0, \dots, k-1$) são as $k$
raízes $k$-ésimas distintas da unidade: $W$ é \omterm{def:b2:reduction:diag}{diagonalizável} (questão
3, ou \cref{exo:b2:reduction:8}: $W^k = I$). \omterm{def:b2:reduction:eigen}{Autovetores}: $Wf_j
= \sum_m \omega^{-jm}e_{m+1} = \sum_{m'}\omega^{-j(m'-1)}e_{m'}
= \omega^j f_j$.

\textbf{21.} As circulantes são polinômios em $W$, e polinômios
numa matriz fixa comutam entre si. Cada $f_j$ é
\omterm{def:b2:reduction:eigen}{autovetor} de toda potência: $W^m f_j = \omega^{jm}f_j$, logo
\[
Cf_j = \sum_m c_m\omega^{jm} f_j = \widehat c(\omega^j)\,f_j :
\]
a base $(f_0, \dots, f_{k-1})$ (livre: Vandermonde nos
$\omega^{-j}$ distintos, \cref{exo:b2:linalg:11}) diagonaliza
todas as circulantes de uma só vez, com os \omterm{def:b2:reduction:eigen}{autovalores} enunciados.

\textbf{22.} O \omterm{def:b2:linalg:det}{determinante} é o produto dos \omterm{def:b2:reduction:eigen}{autovalores}
(diagonalize): $\det C = \prod_{j}\widehat c(\omega^j)$. Para $k
= 3$, $c_0 = a$, $c_1 = b$, $c_2 = c$ e $\omega = j =
\eu^{2\iu\pi/3}$:
\[
\det C = (a + b + c)(a + bj + cj^2)(a + bj^2 + cj^4),
\]
e $j^4 = j$: exatamente a fatoração de
\cref{exo:b2:linalg:8}.

\textbf{23.} $M = \frac12(W + W^{-1})$ é uma circulante ($W^{-1} =
W^{k-1}$), com \omterm{def:b2:reduction:eigen}{autovalores} $\frac12(\omega^j + \omega^{-j}) =
\cos\frac{2\pi j}k$ na mesma base $f_j$. Coordenadas: escreva
$x^{(0)} = \sum_j \alpha_j f_j$. As coordenadas de $f_j$ somam
$\sum_m \omega^{-jm}$, o que vale $k$ para $j = 0$ e $0$
caso contrário (soma geométrica de razão $\omega^{-j} \neq 1$).
Somando as coordenadas de $x^{(0)}$: $\sum_m x^{(0)}_m =
\alpha_0\,k$, logo $\alpha_0 = \frac1k\sum_m x^{(0)}_m$, a média.

\textbf{24.} $x^{(n)} = M^nx^{(0)} = \sum_j
\alpha_j\cos^n\bigl(\tfrac{2\pi j}k\bigr)f_j$. Para $k$ ímpar,
$\abs{\cos(2\pi j/k)} < 1$ para todo $j \neq 0$ (o ângulo nunca
vale $0$ nem $\pi$), de modo que todos os termos exceto $j = 0$ tendem a $0$:
$x^{(n)} \to \alpha_0 f_0$, o vetor constante igual à
média --- promediar num anel ímpar iguala tudo. Para $k = 4$ os
\omterm{def:b2:reduction:eigen}{autovalores} são $1, 0, -1, 0$: o termo $j = 2$
$\alpha_2(-1)^nf_2$ com $f_2 = (1, -1, 1, -1)^{\mathsf T}$
oscila para sempre. A obstrução é a média alternada:
multiplicando as coordenadas de $x^{(0)}$ por $(-1)^m$ e
somando, o mesmo cálculo de soma geométrica dá $\sum_m
(-1)^mx^{(0)}_m = 4\alpha_2$: o processo converge se e somente se
$x^{(0)}_0 - x^{(0)}_1 + x^{(0)}_2 - x^{(0)}_3 = 0$, e então
converge para a média.

\textbf{25.} A matriz companheira converte uma recorrência escalar
de ordem $k$ numa recorrência vetorial de primeira ordem, de modo que
as fórmulas fechadas viram enunciados sobre $C^n$ --- o terreno de
casa da redução (questões 1--5). O lema da decomposição em núcleos é
álgebra polinomial pura (Bézout mais comutação), logo ele reparte
$\ker P(S)$ mesmo com $\mathcal{S}$ de dimensão infinita
(questões 7--9). Os \omterm{def:b2:reduction:eigen}{autovalores} dominantes governam o crescimento porque
toda outra contribuição é geometricamente desprezível após a
normalização --- e é também por isso que o erro de Pell decai como o
quadrado da raiz dominante (questões 11--14). As potências da
matriz de adjacência contam passeios porque a multiplicação de matrizes soma
sobre os vértices intermediários, de modo que os \omterm{def:b2:reduction:eigen}{espectros} contam passeios fechados
(questões 15--19). Matrizes que comutam compartilham uma base de \omterm{def:b2:reduction:eigen}{autovetores}, e
uma única base de Fourier diagonaliza toda a \omterm{def:b2:structures:algebra}{álgebra} das circulantes
de um golpe só (questões 20--24). Cumes: o teorema
fundamental das recorrências lineares (questão 9); e, para matrizes
não negativas, a razão pela qual raízes dominantes como $\varphi$ ou $1 +
\sqrt2$ são automaticamente reais, positivas e simples é o
teorema de Perron--Frobenius, demonstrado no volume do terceiro ano de graduação.
\end{solution}
