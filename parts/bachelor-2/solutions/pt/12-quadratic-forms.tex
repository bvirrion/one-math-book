\section*{Capítulo \ref{ch:b2:quadratic} --- Formas quadráticas}

\begin{solution}{exo:b2:quadratic:1}
$q_1 = (x + 2y)^2 - 3y^2$: posto $2$, assinatura $(1, 1)$ (uma
forma do tipo hipérbole).

$q_2$: complete o quadrado em $x$: $q_2 = (x + y)^2 + y^2 + 2yz +
3z^2 = (x+y)^2 + (y + z)^2 + 2z^2$: posto $3$, assinatura $(3, 0)$
--- positiva definida.
\end{solution}

\begin{solution}{exo:b2:quadratic:2}
\omterm{def:b2:reduction:eigen}{Autovalores} $2$ (em $\operatorname{Vect}(1,1,1)$) e $-1$ (no
plano $x + y + z = 0$). Ortonormalize: $e_1 =
\frac{1}{\sqrt3}(1,1,1)$; no plano, Gram--Schmidt em $(1,-1,0),
(1,0,-1)$ dá $e_2 = \frac{1}{\sqrt2}(1,-1,0)$, $e_3 =
\frac{1}{\sqrt6}(1,1,-2)$. Então $P = (e_1\ e_2\ e_3)$ é ortogonal
com $P^{\mathsf T}AP = \operatorname{diag}(2,-1,-1)$.

A forma $q = 2xy + 2yz + 2zx$ tem matriz $A$: nas coordenadas
giradas $q = 2X^2 - Y^2 - Z^2$ --- eixos principais; assinatura
$(1,2)$, coerente com o~\cref{ex:b2:quadratic:gaussexample}
(a mesma forma!).
\end{solution}

\begin{solution}{exo:b2:quadratic:3}
$u^{**} = u$: $\langle u^{**}x, y\rangle = \langle x, u^*y\rangle =
\langle ux, y\rangle$ para todo $y$. $(uv)^* = v^*u^*$: $\langle
uvx, y\rangle = \langle vx, u^*y\rangle = \langle x,
v^*u^*y\rangle$. Núcleo: $y \in \ker u^* \iff \langle x, u^*y
\rangle = 0\ \forall x \iff \langle u(x), y\rangle = 0\ \forall x
\iff y \perp \operatorname{im} u$. Postos: $\dim\ker u^* = n -
\operatorname{rk} u$ (complemento ortogonal), logo
$\operatorname{rk} u^* = \operatorname{rk} u$ pelo teorema do núcleo e da imagem ---
o avatar euclidiano do teorema do posto da \omterm{def:b2:linalg:transpose}{transposta}.
\end{solution}

\begin{solution}{exo:b2:quadratic:4}
Espectral: $A = PDP^{\mathsf T}$, $D$ diagonal com entradas
$\lambda_i$ satisfazendo $\lambda_i^3 = \lambda_i$: $\lambda_i \in
\{-1, 0, 1\}$. Então $A^2 = PD^2P^{\mathsf T}$ com $D^2$ diagonal
de entradas $0/1$: $A^2$ é \omterm{def:b2:quadratic:adjoint}{simétrica} e idempotente
($(A^2)^2 = A^4 = A\cdot A^3 = A^2$) --- \omterm{def:b2:quadratic:adjoint}{simétrica} idempotente
$=$ projeção ortogonal (é a projeção sobre
$\ker(A^2 - I) = \ker(A-I)\oplus\ker(A+I)$ ao longo de $\ker A$, e
esses são ortogonais pelo teorema espectral).

Em geral, $P(A) = P\!\left(\text{diag}\right)$: $P(A)$ tem os
mesmos \omterm{def:b2:reduction:eigen}{autovetores} e \omterm{def:b2:reduction:eigen}{autovalores} $P(\lambda_i)$ --- a ``aplicação
espectral'' no nível \omterm{def:b2:reduction:diag}{diagonalizável}.
\end{solution}

\begin{solution}{exo:b2:quadratic:5}
Fechado: $O(n) = g^{-1}(\{I\})$ para a aplicação \omterm{def:b2:metric:continuity}{contínua} $g(P) =
P^{\mathsf T}P$ (entradas polinomiais). Limitado: cada coluna de $P
\in O(n)$ é um vetor unitário, logo todas as entradas estão em $\intcc{-1}{1}$.
Fechado e limitado em $\mathcal{M}_n(\R) \simeq \R^{n^2}$: \omterm{def:b2:metric:compact}{compacto}
(\cref{thm:b2:metric:compactprops}~(2)).

Não \omterm{def:b2:metric:connected}{conexo}: $\det$ assume os dois valores $\pm1$ em $O(n)$, e uma
sobrejeção \omterm{def:b2:metric:continuity}{contínua} sobre $\{-1, 1\}$ reparte o espaço
(o argumento do~\cref{ex:b2:metric:glnr}).
\end{solution}

\begin{solution}{exo:b2:quadratic:6}
\emph{Existência:} $A = PDP^{\mathsf T}$ com $D =
\operatorname{diag}(\lambda_i)$, $\lambda_i \geq 0$; ponha $B =
P\sqrt D P^{\mathsf T}$ com $\sqrt D =
\operatorname{diag}(\sqrt{\lambda_i})$: \omterm{def:b2:quadratic:adjoint}{simétrica}, positiva
semidefinida, $B^2 = A$.

\emph{Unicidade:} seja $B$ \omterm{def:b2:quadratic:adjoint}{simétrica} psd com $B^2 = A$. $B$
comuta com $A$; logo $B$ preserva cada \omterm{def:b2:reduction:eigen}{autoespaço}
$E_\lambda(A)$ (para $Ax = \lambda x$: $A(Bx) = BAx = \lambda Bx$).
Em $E_\lambda(A)$, a restrição de $B$ é \omterm{def:b2:quadratic:adjoint}{simétrica} psd com
quadrado $\lambda\,\mathrm{id}$; seus \omterm{def:b2:reduction:eigen}{autovalores} $\mu$ satisfazem $\mu^2
= \lambda$, $\mu \geq 0$: $\mu = \sqrt\lambda$ --- logo a
restrição, sendo \omterm{def:b2:reduction:diag}{diagonalizável} com o único \omterm{def:b2:reduction:eigen}{autovalor}
$\sqrt\lambda$, \emph{é} $\sqrt\lambda\,\mathrm{id}$. Como $E =
\bigoplus E_\lambda(A)$, $B$ fica determinada: $B = \sqrt A$.
\end{solution}

\begin{solution}{exo:b2:quadratic:7}
Numa base ortonormal de \omterm{def:b2:reduction:eigen}{autovetores}, $\norm{Ax}_2^2 = \sum \lambda_i^2
x_i^2 \leq (\max_i \lambda_i^2)\norm x_2^2$, com igualdade no
\omterm{def:b2:reduction:eigen}{autovetor} correspondente: $\vertiii A_2 = \max\abs{\lambda_i}$.
Para a matriz dada: \omterm{def:b2:reduction:eigen}{autovalores} $3, -1$
(a gêmea do~\cref{ex:b2:quadratic:spectralexample}):
$\vertiii A_2 = 3$.
\end{solution}

\begin{solution}{exo:b2:quadratic:8}
($\Rightarrow$) O bloco superior esquerdo $k \times k$ $A_k$ é a matriz
da restrição da forma (definida) ao espaço \omterm{def:b2:structures:generated}{gerado} pelos primeiros
$k$ vetores da base: positiva definida, logo seus \omterm{def:b2:reduction:eigen}{autovalores} são
positivos e $\Delta_k = \det A_k > 0$.

($\Leftarrow$) Indução sobre $n$; $n = 1$ é claro. Suponha todos os
$\Delta_k > 0$. Por indução, $A_{n-1}$ é positiva definida: a
forma $q$ restrita a $F = \operatorname{Vect}(e_1, \dots,
e_{n-1})$ é definida. Diagonalize $q|_F$ (Gauss): coordenadas
$y_1, \dots, y_{n-1}$ com $q|_F = \sum y_i^2$. No espaço todo,
completando o quadrado na última variável,
\[
q = \sum_{i=1}^{n-1} \bigl(y_i + c_i x_n\bigr)^2 + c\,x_n^2
\]
para constantes adequadas (recolha os termos cruzados nos quadrados).
A redução exibe a assinatura $(n-1 + \epsilon, \cdot)$, sendo
$\epsilon$ a contribuição de sinal de $c$; e o \omterm{def:b2:linalg:det}{determinante} guarda
o sinal do produto dos coeficientes diagonais sob
\omterm{def:b2:quadratic:def}{congruência} ($\det(P^{\mathsf T}AP) = (\det P)^2\det A$): $\Delta_n
> 0$ força $c > 0$. Logo $q$ é soma de $n$ quadrados de
formas independentes: positiva definida.
\end{solution}

\begin{solution}{exo:b2:quadratic:9}
Seja $(e_1, \dots, e_n)$ uma base ortonormal de \omterm{def:b2:reduction:eigen}{autovetores} de $\lambda_1
\geq \dots \geq \lambda_n$.

\emph{$\lambda_2 \leq$ o mín-máx:} para qualquer hiperplano $H$, o
espaço $V = \operatorname{Vect}(e_1, e_2)$ de dimensão $2$ satisfaz
$\dim(H \cap V) \geq 1$ (Grassmann): escolha um unitário $x \in H \cap V$,
$x = ae_1 + be_2$, $a^2 + b^2 = 1$:
\[
\langle u(x), x\rangle = \lambda_1 a^2 + \lambda_2 b^2 \geq
\lambda_2 :
\]
o máximo em todo hiperplano é $\geq \lambda_2$.

\emph{$\geq$:} para $H = e_1^{\perp}$, todo unitário $x = \sum_{i\geq2}
x_ie_i \in H$ tem $\langle u(x), x\rangle = \sum_{i \geq 2}
\lambda_i x_i^2 \leq \lambda_2$, atingido em $e_2$: o máximo
nesse hiperplano é exatamente $\lambda_2$. O mínimo sobre $H$ é,
portanto, $\lambda_2$.
\end{solution}

\begin{solution}{exo:b2:quadratic:10}
\emph{Caminho algébrico:} $2q = \bigl(\sum x_i\bigr)^2 - \sum
x_i^2$. No hiperplano $H : \sum x_i = 0$ (dimensão $n -
1$), $q = -\frac12\sum x_i^2$ é negativa definida; na reta
$\R(1, \dots, 1)$, $q(t, \dots, t) = \binom n2 t^2 > 0$. Um
subespaço em que $q$ é positiva definida encontra $H$ trivialmente, logo
tem dimensão $\leq 1$: por Sylvester
(\cref{thm:b2:quadratic:sylvester}), $s = 1$, e $t \geq n-1$
de $H$; o posto $\leq n$ força assinatura $(1, n-1)$, posto $n$.

\emph{Caminho espectral:} a matriz é $\frac12(J - I)$; $J$ tem
\omterm{def:b2:reduction:eigen}{autovalores} $n$ (em $(1,\dots,1)$) e $0$ (em $H$), logo
$\frac12(J-I)$ tem \omterm{def:b2:reduction:eigen}{autovalores} $\frac{n-1}{2}$ (uma vez) e
$-\frac12$ ($n-1$ vezes): um positivo, $n-1$ negativos --- mesma
assinatura, pelo~\cref{cor:b2:quadratic:principalaxes}.
\end{solution}

\begin{solution}{exo:b2:quadratic:11}
Escreva $B = C^{\mathsf T}C$ (\cref{exo:b2:quadratic:6} via $C =
\sqrt B$). Então, com $c_1, \dots, c_n$ as colunas de
$C^{\mathsf T}$:
\[
\operatorname{tr}(AB) = \operatorname{tr}(AC^{\mathsf T}C)
= \operatorname{tr}(CAC^{\mathsf T})
= \sum_{i=1}^n \langle A c_i, c_i\rangle .
\]
Pelo~\cref{cor:b2:quadratic:principalaxes}~(2), cada termo fica
entre $\lambda_{\min}(A)\norm{c_i}^2$ e
$\lambda_{\max}(A)\norm{c_i}^2$, e $\sum\norm{c_i}^2 =
\operatorname{tr}(C^{\mathsf T}C)^{\vphantom1} =
\operatorname{tr} B$: a dupla desigualdade segue. Se $A$ é
também positiva semidefinida, $\lambda_{\min}(A) \geq 0$:
$\operatorname{tr}(AB) \geq 0$.
\end{solution}

\begin{solution}{exo:b2:quadratic:12}
\begin{enumerate}
  \item $\varphi(M, N) = \operatorname{tr}(MN)$ é bilinear e
        \omterm{def:b2:quadratic:adjoint}{simétrica} ($\operatorname{tr}(MN) =
        \operatorname{tr}(NM)$), e $\varphi(M, M) = q(M)$:
        $q$ é a \omterm{def:b2:quadratic:def}{forma quadrática} de $\varphi$.
  \item Para $S$ \omterm{def:b2:quadratic:adjoint}{simétrica} e $K$ antissimétrica:
        $\operatorname{tr}(SK) =
        \operatorname{tr}\bigl((SK)^{\mathsf T}\bigr) =
        \operatorname{tr}(K^{\mathsf T}S^{\mathsf T}) =
        -\operatorname{tr}(KS) = -\operatorname{tr}(SK)$, logo
        $\varphi(S, K) = 0$: os dois subespaços são
        $\varphi$-ortogonais. Entrada a entrada, $\operatorname{tr}
        (M^2) = \sum_{i,j} m_{ij}m_{ji}$: para $M$ \omterm{def:b2:quadratic:adjoint}{simétrica}
        isso é $\sum m_{ij}^2 > 0$ ($M \neq 0$); para
        $M$ antissimétrica é $-\sum m_{ij}^2 < 0$.
  \item $\mathcal M_n(\R) = S_n \oplus A_n$ com dimensões
        $\frac{n(n+1)}2$ e $\frac{n(n-1)}2$; uma \omterm{thm:b2:quadratic:gauss}{redução de
        Gauss} adaptada a essa decomposição $\varphi$-ortogonal
        escreve $q$ como $\frac{n(n+1)}2$ quadrados positivos e
        $\frac{n(n-1)}2$ negativos: assinatura
        $\bigl(\frac{n(n+1)}2, \frac{n(n-1)}2\bigr)$
        (Sylvester), posto $n^2$: a forma é não degenerada.
\end{enumerate}
\end{solution}

\begin{solution}{pb:b2:quadratic:1}
\textbf{1.} $G$ é \omterm{def:b2:quadratic:adjoint}{simétrica} pela simetria do produto interno,
e
\[
X^{\mathsf T}GX = \sum_{i,j}x_ix_j\langle v_i, v_j\rangle
= \Bigl\|\sum_i x_iv_i\Bigr\|^2 \geq 0 ,
\]
com igualdade se e somente se $\sum x_iv_i = 0$: $G$ é definida se e somente se a
única combinação nula é a trivial, isto é, se e somente se a família é
independente.

\textbf{2.} Com $B = \sqrt A$ (\cref{exo:b2:quadratic:6}): $A
= B^2 = B^{\mathsf T}B$, a matriz de Gram das colunas de $B$;
tome $C = B$. E $X^{\mathsf T}AX = \norm{CX}^2$, logo $A$ é
definida se e somente se $CX \neq 0$ para $X \neq 0$, se e somente se $C$ é invertível.

\textbf{3.} A restrição da forma a
$\operatorname{Vect}(e_i, e_j)$ tem matriz
$\begin{pmatrix} a_{ii} & a_{ij}\\ a_{ij} & a_{jj}
\end{pmatrix}$, ainda positiva semidefinida: seu \omterm{def:b2:linalg:det}{determinante}
(produto de seus \omterm{def:b2:reduction:eigen}{autovalores} não negativos) é $\geq 0$:
$a_{ij}^2 \leq a_{ii}a_{jj}$. Essa é Cauchy--Schwarz para os
vetores $v_i, v_j$ de uma realização de Gram.

\textbf{4.} \emph{Existência:} escreva $A$ como matriz de Gram de
uma família independente $(v_1, \dots, v_n)$ (questões 1--2).
Gram--Schmidt produz uma base ortonormal $(e_1, \dots, e_n)$ com
\[
v_k = \sum_{i \leq k} t_{ik}\,e_i,
\qquad t_{kk} = \bigl\| v_k - \operatorname{proj}_{k-1}v_k
\bigr\| > 0 ,
\]
de modo que $T = (t_{ik})$ é triangular superior com diagonal positiva,
e
\[
a_{jk} = \langle v_j, v_k\rangle
= \sum_i t_{ij}t_{ik} = (T^{\mathsf T}T)_{jk} .
\]
\emph{Unicidade:} se $T_1^{\mathsf T}T_1 = T_2^{\mathsf T}T_2$,
então $R = T_1T_2^{-1}$ satisfaz $R^{\mathsf T}R = I$: $R$ é
ortogonal e também triangular superior com diagonal positiva
(produto de tais). Então $R^{-1} = R^{\mathsf T}$ é
simultaneamente triangular superior (inversa de superior) e inferior (\omterm{def:b2:linalg:transpose}{transposta} de
superior): diagonal; uma matriz diagonal ortogonal tem
entradas $\pm1$, e a positividade força $R = I$: $T_1 = T_2$.

\textbf{5.} Para $i, j \leq k$, $(T^{\mathsf T}T)_{ij} = \sum_m
t_{mi}t_{mj}$ só envolve $m \leq \min(i,j) \leq k$: o
bloco dominante $k\times k$ de $A$ é $T_k^{\mathsf T}T_k$, com
$T_k$ o bloco dominante de $T$. Logo $\Delta_k = (\det T_k)^2
= (t_{11}\cdots t_{kk})^2$. Ora, a \omterm{thm:b2:quadratic:gauss}{redução de Gauss} de uma
forma positiva definida na ordem natural nunca encontra coeficiente
quadrático nulo (os pivôs são as entradas diagonais dos
blocos positivos definidos sucessivamente reduzidos): ela produz $q
= \sum_k d_k\ell_k^2$ com $\ell_k = x_k + (\text{termos em }
x_{k+1}, \dots)$, isto é, $A = L^{\mathsf T}DL$ com $L$
triangular unipotente; então $T = \sqrt D\,L$ é um fator de
Cholesky, logo, pela unicidade, $t_{kk}^2 = d_k$ e
\[
d_k = \frac{(t_{11}\cdots t_{kk})^2}
{(t_{11}\cdots t_{k-1,k-1})^2}
= \frac{\Delta_k}{\Delta_{k-1}} .
\]
Em \cref{ex:b2:quadratic:tworoads}: $\Delta_1, \Delta_2,
\Delta_3 = 2, 3, 4$ e os pivôs eram $2, \frac32, \frac43$.

\textbf{6.} Cada $a_{ii} = e_i^{\mathsf T}Ae_i > 0$. Sejam $D =
\operatorname{diag}(a_{ii}^{-1/2})$ e $B = DAD$: positiva
definida (\omterm{def:b2:quadratic:def}{congruência}), com $b_{ii} = 1$, logo $\operatorname{tr}
B = n$. Seus \omterm{def:b2:reduction:eigen}{autovalores} $\mu_i > 0$ satisfazem, por MA--MG,
\[
\det B = \prod_i\mu_i
\leq \Bigl(\frac{\sum\mu_i}{n}\Bigr)^{\!n} = 1 ,
\]
e $\det B = (\det D)^2\det A = \dfrac{\det A}{\prod
a_{ii}}$: $\det A \leq \prod a_{ii}$.

\textbf{7.} MA--MG é igualdade se e somente se todos os $\mu_i$ são iguais
(a $1$); uma matriz \omterm{def:b2:quadratic:adjoint}{simétrica} com único \omterm{def:b2:reduction:eigen}{autovalor} $1$ é
$PIP^{\mathsf T} = I$. Logo igualdade se e somente se $B = I$, se e somente se $a_{ij} = 0$
para $i \neq j$: $A$ diagonal.

\textbf{8.} Se $M$ é singular, os dois lados valem $\geq 0 =
\abs{\det M}$. Caso contrário, $A = M^{\mathsf T}M$ é positiva
definida com $a_{ii} = \norm{c_i}^2$ e $\det A = (\det
M)^2$: a questão 6 dá $(\det M)^2 \leq \prod\norm{c_i}^2$.
Igualdade se e somente se $A = M^{\mathsf T}M$ é diagonal (questão 7),
isto é, se e somente se as colunas são duas a duas ortogonais.

\textbf{9.} $\abs{\det M}$ é o volume do paralelepípedo
\omterm{def:b2:structures:generated}{gerado} pelas colunas: o volume é no máximo o produto dos
comprimentos das arestas, com igualdade exatamente para caixas retangulares.
Se $\abs{m_{ij}} \leq 1$, então $\norm{c_i} \leq \sqrt n$, logo
$\abs{\det M} \leq n^{n/2}$. (Atingir isso força colunas ortogonais
de entradas $\pm1$: uma matriz de Hadamard.)

\textbf{10.} $X^{\mathsf T}A^{\mathsf T}AX = \norm{AX}^2 > 0$
para $X \neq 0$ ($A$ invertível): $A^{\mathsf T}A$ é positiva
definida. Sua raiz quadrada $S$ é positiva definida (\omterm{def:b2:reduction:eigen}{autovalores}
$\sqrt{\lambda_i} > 0$), logo invertível, e $Q = AS^{-1}$
satisfaz
\[
Q^{\mathsf T}Q = S^{-1}A^{\mathsf T}AS^{-1}
= S^{-1}S^2S^{-1} = I :
\]
$A = QS$ com $Q$ ortogonal e $S$ positiva definida.

\textbf{11.} Se $A = QS = Q'S'$, então $S'^{\,2} =
S'^{\mathsf T}Q'^{\mathsf T}Q'S' = A^{\mathsf T}A = S^2$; duas
matrizes positivas semidefinidas com o mesmo quadrado coincidem
(\cref{exo:b2:quadratic:6}): $S' = S$, e depois $Q' = AS^{-1} =
Q$.

\textbf{12.} $\det(A + \varepsilon I)$ é um polinômio não nulo
em $\varepsilon$: ele tem finitas raízes, logo alguma sequência
$\varepsilon_k \to 0$ as evita. Escreva $A + \varepsilon_kI =
Q_kS_k$ (questão 10). $O(n)$ é \omterm{def:b2:metric:compact}{compacto}
(\cref{exo:b2:quadratic:5}): uma subsequência dá $Q_{\varphi(k)}
\to Q \in O(n)$. Então
\[
S_{\varphi(k)} = Q_{\varphi(k)}^{\mathsf T}
\bigl(A + \varepsilon_{\varphi(k)}I\bigr)
\longrightarrow Q^{\mathsf T}A =: S,
\]
\omterm{def:b2:quadratic:adjoint}{simétrica} positiva semidefinida como limite de tais (condições
fechadas), e $A = QS$. Além disso, $S^2 = S^{\mathsf T}S =
A^{\mathsf T}QQ^{\mathsf T}A = A^{\mathsf T}A$, logo $S =
\sqrt{A^{\mathsf T}A}$ pela unicidade. Para $A$ singular, $S$ é
singular e $Q$ não é única: ela pode ser modificada arbitrariamente
em $(\operatorname{im} S)^{\perp}$ --- caso extremo $A = 0$,
em que toda $Q$ ortogonal serve.

\textbf{13.} Diagonalize $S = P\Sigma P^{\mathsf T}$ (teorema
espectral), $\Sigma = \operatorname{diag}(\sigma_i)$ com
$\sigma_i \geq 0$ os \omterm{def:b2:reduction:eigen}{autovalores} de $S =
\sqrt{A^{\mathsf T}A}$. Então
\[
A = QS = (QP)\,\Sigma\,P^{\mathsf T} = U\Sigma V^{\mathsf T},
\qquad U = QP,\ V = P \in O(n) .
\]

\textbf{14.} $\norm{Ax}^2 = x^{\mathsf T}S^2x \leq
\sigma_{\max}^2\norm x^2$ com igualdade num \omterm{def:b2:reduction:eigen}{autovetor} de topo de
$S$: $\vertiii A_2 = \sigma_{\max}$ --- para $A$ \omterm{def:b2:quadratic:adjoint}{simétrica},
$S = \sqrt{A^2}$ tem \omterm{def:b2:reduction:eigen}{autovalores} $\abs{\lambda_i}$, recuperando
o~\cref{exo:b2:quadratic:7}. \omterm{def:b2:linalg:det}{Determinante}: $\abs{\det A} =
\abs{\det U}\det\Sigma\abs{\det V} = \sigma_1\cdots\sigma_n$.
Esfera: escrevendo $x = Vy$ com $\norm y = 1$, $Ax = U\Sigma y$
tem coordenadas $z_i = \sigma_iy_i$ no referencial ortonormal das
colunas de $U$: a imagem é $\{\sum z_i^2/\sigma_i^2 =
1\}$, um elipsoide de semieixos $\sigma_i$.

\textbf{15.} Uma rotação para eixos principais
(\cref{cor:b2:quadratic:principalaxes}) transforma $f$ em
$\lambda_1X^2 + \lambda_2Y^2 + \beta_1X + \beta_2Y + c$, com
$\lambda_1\lambda_2 = \det A$. Se $\det A \neq 0$, absorva os
termos lineares pela translação $X \mapsto X -
\frac{\beta_1}{2\lambda_1}$ (e do mesmo modo $Y$): $\lambda_1X'^2
+ \lambda_2Y'^2 = c'$. Para $\det A > 0$ (sinais iguais): uma
elipse ($c'$ do sinal certo), um ponto ou o vazio. Para $\det
A < 0$: uma hipérbole ($c' \neq 0$) ou duas retas concorrentes. Se
$\det A = 0$ com posto $1$ (digamos $\lambda_2 = 0 \neq
\lambda_1$): $\lambda_1X'^2 + \beta_2Y + c''$, uma parábola quando
$\beta_2 \neq 0$; caso contrário, duas retas paralelas, uma reta ou o
vazio. Formas não degeneradas: elipse, hipérbole, parábola,
governadas pelo sinal de $\det A$.

\textbf{16.} A parte quadrática $x^2 + 4xy + y^2$ tem matriz
$\begin{pmatrix}1 & 2\\ 2 & 1\end{pmatrix}$, \omterm{def:b2:reduction:eigen}{autovalores} $3$
e $-1$ com direções ortonormais $\frac{1}{\sqrt2}(1,1)$,
$\frac{1}{\sqrt2}(1,-1)$. Nas coordenadas giradas $u =
\frac{x+y}{\sqrt2}$, $v = \frac{x-y}{\sqrt2}$: $x^2 + y^2 = u^2
+ v^2$, $2xy = u^2 - v^2$, de modo que a forma é $3u^2 - v^2$, e $2x
- 2y = 2\sqrt2\,v$. A equação torna-se
\[
3u^2 - v^2 + 2\sqrt2\,v = 4
\quad\Longleftrightarrow\quad
3u^2 - \bigl(v - \sqrt2\bigr)^2 = 2 :
\]
uma hipérbole, de centro $(u, v) = (0, \sqrt2)$, isto é, $(x, y) =
(1, -1)$, com eixos ao longo do referencial girado.

\textbf{17.} $f(x) = (x - x_0)^{\mathsf T}A(x - x_0) + f(x_0)$
sempre que $Ax_0 = -\frac b2$, isto é, $x_0 = -\frac12A^{-1}b$: o
gradiente de $f$ se anula exatamente aí ($x_0$ é o centro de
simetria). Com $M = \begin{pmatrix} I & x_0\\ 0 &
1\end{pmatrix}$:
\[
M^{\mathsf T}\widetilde QM
= \begin{pmatrix}
A & Ax_0 + \frac b2\\[2pt]
\bigl(Ax_0 + \frac b2\bigr)^{\mathsf T} &
x_0^{\mathsf T}Ax_0 + b^{\mathsf T}x_0 + c
\end{pmatrix}
= \begin{pmatrix} A & 0\\ 0 & f(x_0)\end{pmatrix},
\]
e $\det M = 1$: $\det\widetilde Q = f(x_0)\det A$. A
equação centrada lê-se $q(X) = -f(x_0)$: para $f(x_0) = 0$ ela
degenera em $q(X) = 0$ (duas retas pelo centro se a
assinatura é $(1,1)$, o único ponto $x_0$ se $q$ é
definida); para $f(x_0) \neq 0$ a cônica é uma elipse ou
hipérbole genuína.

\textbf{18.} $A^{-1} = -\frac13\begin{pmatrix} 1 & -2\\ -2 &
1\end{pmatrix}$, $\frac b2 = (1, -1)$: $x_0 =
-A^{-1}\frac b2 = (1, -1)$, como se achou na questão 16. $f(x_0) =
q(1,-1) + 2 + 2 - 4 = (1 - 4 + 1) + 0 = -2 \neq 0$, e
$\det\widetilde Q = f(x_0)\det A = (-2)(-3) = 6 \neq 0$:
não degenerada; $\det A = -3 < 0$: uma hipérbole --- e de fato
a equação centrada $3u^2 - (v - \sqrt2)^2 = -f(x_0) = 2$
coincide com a da questão 16.

\textbf{19.} A matriz é $(1-\lambda)I + \lambda J$:
\omterm{def:b2:reduction:eigen}{autovalores} $1 + 2\lambda$ (direção $(1,1,1)$) e $1 -
\lambda$ (duplo, em $x + y + z = 0$). Casos:
\begin{itemize}
  \item $-\frac12 < \lambda < 1$: todos os \omterm{def:b2:reduction:eigen}{autovalores} positivos:
        um elipsoide de revolução em torno de $(1,1,1)$ (uma esfera
        para $\lambda = 0$);
  \item $\lambda = 1$: $q = (x+y+z)^2$: a equação dá os
        dois planos paralelos $x + y + z = \pm1$;
  \item $\lambda = -\frac12$: \omterm{def:b2:reduction:eigen}{autovalores} $0, \frac32,
        \frac32$: um cilindro circular de eixo $(1,1,1)$;
  \item $\lambda > 1$: assinatura $(1, 2)$: um hiperboloide de
        duas folhas;
  \item $\lambda < -\frac12$: assinatura $(2, 1)$: um
        hiperboloide de uma folha.
\end{itemize}

\textbf{20.} A forma polar de $q$ é um produto interno
$\langle\cdot,\cdot\rangle_q$ em $E$. Para $x$ fixo, $y \mapsto
\varphi'(x, y)$ (forma polar de $q'$) é linear, logo é igual a
$\langle z_x, y\rangle_q$ para um único $z_x$; $u(x) := z_x$ é
linear (unicidade), e $\langle u(x), y\rangle_q =
\varphi'(x,y) = \varphi'(y,x) = \langle u(y), x\rangle_q$: $u$
é \omterm{def:b2:quadratic:adjoint}{simétrico} no espaço euclidiano $(E,
\langle\cdot,\cdot\rangle_q)$. O teorema espectral
(\cref{thm:b2:quadratic:spectral}) dá uma base $(\varepsilon_i)$ ortonormal para $q$
e de \omterm{def:b2:reduction:eigen}{autovetores}, $u(\varepsilon_i) =
\mu_i\varepsilon_i$: nela $q(x) = \sum x_i^2$ e $q'(x) =
\langle u(x), x\rangle_q = \sum\mu_ix_i^2$.

\textbf{21.} Seja $P$ a matriz dessa base: a \omterm{def:b2:quadratic:def}{congruência}
dá $P^{\mathsf T}AP = I$ e $P^{\mathsf T}BP =
\operatorname{diag}(\mu_i)$. Então
\[
\det(B - \mu A) = \det(P^{-\mathsf T})
\det\bigl(\operatorname{diag}(\mu_i) - \mu I\bigr)
\det(P^{-1})
= (\det P)^{-2}\prod_i(\mu_i - \mu) :
\]
os $\mu_i$ são as raízes do feixe $\det(B - \mu A)$.

\textbf{22.} $\det(B - \mu A) =
\det\begin{pmatrix} -2\mu & 1-\mu\\ 1-\mu & -\mu\end{pmatrix}
= 2\mu^2 - (1-\mu)^2 = \mu^2 + 2\mu - 1$: raízes $\mu_\pm = -1
\pm \sqrt2$. Resolvendo $(B - \mu_\pm A)v = 0$: $v_\pm = (1 -
\mu_\pm,\ 2\mu_\pm)$ (a identidade da segunda linha $(1-\mu)^2 =
2\mu^2$ nas raízes confirma). As \omterm{def:b2:nvs:norm}{normas} $A$ saem
limpas: $q_A(v_\pm) = 2(1 + \mu_\pm^2)$, e verifica-se que
$\varphi_A(v_+, v_-) = 0$ usando $\mu_+ + \mu_- = -2$,
$\mu_+\mu_- = -1$. A base $\Bigl(\frac{v_+}{\sqrt{2(1 +
\mu_+^2)}}, \frac{v_-}{\sqrt{2(1+\mu_-^2)}}\Bigr)$ é
ortonormal para $A$ e diagonaliza $B$ com entradas
$\mu_\pm$.

\textbf{23.} Se alguma $P$ invertível diagonalizasse as duas formas,
o cálculo da questão 21 daria $\det(B - \mu A) = (\det
P)^{-2}\prod(d_{2i} - \mu d_{1i})$, um polinômio real que se decompõe
em fatores lineares reais. Mas aqui
\[
\det(B - \mu A) = \det\begin{pmatrix} -\mu & 1\\ 1 &
\mu\end{pmatrix} = -\mu^2 - 1 ,
\]
de grau $2$ sem raiz real: contradição. (A assinatura
de Lorentz $A$ admite ``rotações'' $B$ sem eixos
reais.)

\textbf{24.} $A^{-1}B = A^{-1/2}\bigl(A^{-1/2}BA^{-1/2}\bigr)
A^{1/2}$ com $A^{1/2} = \sqrt A$ positiva definida
(\cref{exo:b2:quadratic:6}): $A^{-1}B$ é semelhante à
matriz \emph{\omterm{def:b2:quadratic:adjoint}{simétrica}} $A^{-1/2}BA^{-1/2}$, logo é \omterm{def:b2:reduction:diag}{diagonalizável} com
\omterm{def:b2:reduction:eigen}{autovalores} reais. E $\det(B - \mu A) = \det A\cdot
\det(A^{-1}B - \mu I)$: as raízes do feixe $\mu_i$ da questão
21 são exatamente os \omterm{def:b2:reduction:eigen}{autovalores} de $A^{-1}B$.

\textbf{25.} (i) Toda Parte se apoiou no teorema espectral:
pela raiz quadrada (Cholesky, polar), pelas cotas de
\omterm{def:b2:reduction:eigen}{autovalores} (Hadamard), pelos eixos principais (cônicas) e pela
versão adaptada a $q$ (redução simultânea). (ii) As realizações
de Gram, Hadamard e a \omterm{pb:b2:quadratic:1}{decomposição polar} sobrevivem no
mundo semidefinido; a unicidade de Cholesky, a fórmula dos
pivôs e a redução simultânea precisam da definição estrita
(as questões 12 e 23 mostram exatamente como elas falham). (iii)
Pequenas oscilações acopladas: a energia cinética (positiva definida)
e a energia potencial são duas \omterm{def:b2:quadratic:def}{formas quadráticas}; a base das
questões 20--22 é a dos \emph{modos normais} do sistema, e os
$\mu_i$ são os quadrados das frequências angulares.
\end{solution}
