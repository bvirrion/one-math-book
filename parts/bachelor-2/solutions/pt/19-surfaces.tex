\section*{Capítulo \ref{ch:b2:surfaces} --- Superfícies}

\begin{solution}{exo:b2:surfaces:1}
Com $\sigma(\theta, r) = (r\cos\theta,\ r\sin\theta,\ r)$:
\[
\sigma_\theta = (-r\sin\theta,\ r\cos\theta,\ 0),
\qquad
\sigma_r = (\cos\theta,\ \sin\theta,\ 1),
\]
independentes para $r > 0$. O \omterm{def:b2:surfaces:tangent}{plano tangente} em $M_0 =
\sigma(\theta_0, r_0)$ passa por $M_0$ com direções
$\sigma_\theta, \sigma_r$. Ora, $M_0 - O = r_0(\cos\theta_0,
\sin\theta_0, 1) = r_0\,\sigma_r(\theta_0, r_0)$ é ele próprio uma
direção tangente: o vértice $O$ está no \omterm{def:b2:surfaces:tangent}{plano tangente}. (Esse
é o comportamento geral dos cones: eles são regrados por retas passando pelo
vértice, e um \omterm{def:b2:surfaces:tangent}{plano tangente} contém a reta geratriz que passa pelo
ponto de tangência.)
\end{solution}

\begin{solution}{exo:b2:surfaces:2}
Aplique a~\cref{prop:b2:surfaces:gradient} a $f(x,y,z) =
\frac{x^2}{a^2} + \frac{y^2}{b^2} + \frac{z^2}{c^2}$: $\nabla
f(x_0,y_0,z_0) = 2\bigl(\frac{x_0}{a^2}, \frac{y_0}{b^2},
\frac{z_0}{c^2}\bigr) \neq 0$ sobre a superfície. O \omterm{def:b2:surfaces:tangent}{plano tangente} é
\[
\frac{x_0}{a^2}(x - x_0) + \frac{y_0}{b^2}(y - y_0)
+ \frac{z_0}{c^2}(z - z_0) = 0,
\qquad\text{i.e.}\qquad
\frac{x_0\,x}{a^2} + \frac{y_0\,y}{b^2} + \frac{z_0\,z}{c^2} = 1,
\]
usando que $(x_0, y_0, z_0)$ satisfaz a equação do elipsoide ---
a regra de ``desdobrar os quadrados'' que generaliza o $\langle
M_0, M \rangle = R^2$ da esfera.
\end{solution}

\begin{solution}{exo:b2:surfaces:3}
$\sigma_u = (-v\sin u,\ v\cos u,\ a)$ e $\sigma_v = (\cos u,\
\sin u,\ 0)$, logo
\[
E = v^2 + a^2, \qquad F = 0, \qquad G = 1,
\qquad
\sqrt{EG - F^2} = \sqrt{v^2 + a^2} > 0
\]
(o helicoide é regular em toda parte, inclusive sobre seu eixo $v =
0$). \omterm{def:b2:surfaces:area}{Área} do pedaço:
\[
\mathcal{A} = \int_0^{2\pi}\!\!\int_0^1
\sqrt{v^2 + a^2}\;\dd v\,\dd u
= 2\pi\cdot\frac12\Bigl[v\sqrt{v^2 + a^2}
+ a^2\ln\bigl(v + \sqrt{v^2 + a^2}\bigr)\Bigr]_0^1 ,
\]
isto é, $\mathcal{A} = \pi\Bigl(\sqrt{1 + a^2} +
a^2\ln\frac{1 + \sqrt{1 + a^2}}{a}\Bigr)$.
\end{solution}

\begin{solution}{exo:b2:surfaces:4}
Derivadas:
\[
\sigma_\theta = \bigl(-(R + r\cos\psi)\sin\theta,\
(R + r\cos\psi)\cos\theta,\ 0\bigr),
\qquad
\sigma_\psi = \bigl(-r\sin\psi\cos\theta,\
-r\sin\psi\sin\theta,\ r\cos\psi\bigr).
\]
Então
\[
E = (R + r\cos\psi)^2,
\qquad
F = r\sin\psi\cos\psi\,(R + r\cos\psi)
\bigl(\sin\theta\cos\theta - \sin\theta\cos\theta\bigr) = 0,
\qquad
G = r^2 ,
\]
logo $\sqrt{EG - F^2} = r(R + r\cos\psi) \geq r(R - r) > 0$:
regular em toda parte. \omterm{def:b2:surfaces:area}{Área}:
\[
\mathcal{A}
= \int_0^{2\pi}\!\!\int_0^{2\pi} r(R + r\cos\psi)
\,\dd\theta\,\dd\psi
= 2\pi r \Bigl(2\pi R + r\int_0^{2\pi}\cos\psi\,\dd\psi\Bigr)
= 4\pi^2 R r ,
\]
o termo em $\cos\psi$ integrando a zero --- teorema de Pappus: \omterm{def:b2:surfaces:area}{área}
$=$ (\omterm{def:b2:curves:length}{comprimento} do círculo girado) $\times$ (distância percorrida
por seu centro).
\end{solution}

\begin{solution}{exo:b2:surfaces:5}
Para $\sigma(x, y) = (x, y, f(x,y))$: $\sigma_x = (1, 0, f_x)$,
$\sigma_y = (0, 1, f_y)$, logo $E = 1 + f_x^2$, $F = f_xf_y$, $G = 1
+ f_y^2$ e
\[
EG - F^2 = (1 + f_x^2)(1 + f_y^2) - f_x^2f_y^2
= 1 + f_x^2 + f_y^2 ,
\]
o que dá a fórmula de \omterm{def:b2:surfaces:area}{área} anunciada. Para $f = \frac12(x^2 + y^2)$ sobre
o disco unitário: $1 + f_x^2 + f_y^2 = 1 + x^2 + y^2$ e, em coordenadas
polares ($x = \rho\cos\alpha$, $y = \rho\sin\alpha$, jacobiano
$\rho$, \cref{ch:b2:multint}):
\[
\mathcal{A}
= \int_0^{2\pi}\!\!\int_0^1 \sqrt{1 + \rho^2}\;\rho
\,\dd\rho\,\dd\alpha
= 2\pi\Bigl[\tfrac13(1 + \rho^2)^{3/2}\Bigr]_0^1
= \frac{2\pi}{3}\bigl(2\sqrt2 - 1\bigr).
\]
\end{solution}

\begin{solution}{exo:b2:surfaces:6}
Pelo cálculo do~\cref{ex:b2:surfaces:spherearea}, $E =
R^2\cos^2\varphi$, $F = 0$, $G = R^2$, logo, pela
\cref{prop:b2:surfaces:curvelength}
\[
L = \int_a^b R\sqrt{\cos^2\varphi(t)\,\theta'(t)^2
+ \varphi'(t)^2}\;\dd t .
\]
Sejam os extremos $(\theta_0, \varphi_1)$ e $(\theta_0,
\varphi_2)$, $\varphi_1 < \varphi_2$. Para qualquer curva que os una,
\[
L \geq \int_a^b R\,\abs{\varphi'(t)}\,\dd t
\geq R\,\Bigl|\int_a^b \varphi'(t)\,\dd t\Bigr|
= R\,(\varphi_2 - \varphi_1),
\]
descartando o termo não negativo $\cos^2\varphi\,\theta'^2$ e usando
a desigualdade triangular para integrais. O arco de meridiano
$\theta \equiv \theta_0$, com $\varphi$ crescendo de $\varphi_1$ a
$\varphi_2$, tem \omterm{def:b2:curves:length}{comprimento} exatamente $R(\varphi_2 - \varphi_1)$: ele é o
mais curto. (Meridianos são círculos máximos; esse é o primeiro
caso, elementar, do fato de que as geodésicas da esfera são círculos
máximos.)
\end{solution}

\begin{solution}{exo:b2:surfaces:7}
Fixe uma curva $\gamma$ desenhada sobre $S$ e ponha $g(t) = \norm{\gamma(t)
- \Omega}^2$. Então $g'(t) = 2\langle \gamma'(t),\ \gamma(t) -
\Omega\rangle$. A reta normal em $M = \gamma(t)$ passa por
$\Omega$ por hipótese, de modo que $\gamma(t) - \Omega$ é um vetor
\emph{normal}, ortogonal ao \omterm{def:b2:surfaces:tangent}{plano tangente}, em particular à
velocidade $\gamma'(t)$ (\cref{prop:b2:surfaces:velocity}): $g' =
0$, e $g$ é constante ao longo de toda curva desenhada.

Ora, o conjunto $S_c = \{M \in S : \norm{M - \Omega}^2 = c\}$ é
fechado em $S$; ele também é \omterm{def:b2:metric:topology}{aberto} em $S$: perto de qualquer de seus pontos, $S$
é um gráfico regular, de modo que todo ponto vizinho de $S$ liga-se a ele por
uma curva desenhada (um segmento levantado), ao longo da qual $g$ é constante. Como
$S$ é \omterm{def:b2:metric:connected}{conexa} e $S_c$ é não vazio para o $c$ adequado, $S = S_c
\subseteq$ a esfera de centro $\Omega$ e raio $\sqrt c$
(\cref{ch:b2:metric}: argumento de \omterm{def:b2:metric:connected}{conexidade}).
\end{solution}

\begin{solution}{exo:b2:surfaces:8}
Escreva $\Phi(u', v') = (u, v)$. Pela regra da cadeia,
\[
\tilde\sigma_{u'} = \frac{\partial u}{\partial u'}\sigma_u
+ \frac{\partial v}{\partial u'}\sigma_v,
\qquad
\tilde\sigma_{v'} = \frac{\partial u}{\partial v'}\sigma_u
+ \frac{\partial v}{\partial v'}\sigma_v ,
\]
com as parciais de $\sigma$ avaliadas em $\Phi(u',v')$. Desenvolvendo o
produto vetorial bilinearmente e usando $\sigma_u \wedge \sigma_u =
\sigma_v \wedge \sigma_v = 0$, $\sigma_v \wedge \sigma_u =
-\sigma_u \wedge \sigma_v$:
\[
\tilde\sigma_{u'} \wedge \tilde\sigma_{v'}
= \Bigl(\frac{\partial u}{\partial u'}
\frac{\partial v}{\partial v'}
- \frac{\partial v}{\partial u'}
\frac{\partial u}{\partial v'}\Bigr)\,
\sigma_u \wedge \sigma_v
= \det J_\Phi \cdot (\sigma_u \wedge \sigma_v)\circ\Phi .
\]
Tomando \omterm{def:b2:nvs:norm}{normas} obtém-se a identidade. Então, pela fórmula de
mudança de variáveis (\cref{ch:b2:multint}) aplicada à aplicação $\Phi$ em $K' =
\Phi^{-1}(K)$:
\[
\iint_{K'} \norm{\tilde\sigma_{u'} \wedge \tilde\sigma_{v'}}
\,\dd u'\dd v'
= \iint_{K'} \norm{\sigma_u \wedge \sigma_v}\circ\Phi\;
\abs{\det J_\Phi}\,\dd u'\dd v'
= \iint_K \norm{\sigma_u \wedge \sigma_v}\,\dd u\,\dd v :
\]
as duas parametrizações atribuem a mesma \omterm{def:b2:surfaces:area}{área} ao mesmo pedaço de
superfície.
\end{solution}

\begin{solution}{exo:b2:surfaces:9}
Na carta esférica, $z = R\sin\varphi$, e a zona
corresponde a $\varphi_1 \leq \varphi \leq \varphi_2$ com
$z_i = R\sin\varphi_i$. Com o elemento de \omterm{def:b2:surfaces:area}{área}
$R^2\cos\varphi\,\dd\theta\,\dd\varphi$
(\cref{ex:b2:surfaces:spherearea}):
\[
\mathcal A = \int_{\varphi_1}^{\varphi_2}\!\!\int_0^{2\pi}
R^2\cos\varphi\,\dd\theta\,\dd\varphi
= 2\pi R^2(\sin\varphi_2 - \sin\varphi_1)
= 2\pi R\,(z_2 - z_1) .
\]
O resultado depende apenas da altura $z_2 - z_1$: fatias de
espessura igual carregam \omterm{def:b2:surfaces:area}{áreas} iguais, sejam cortadas no equador
ou no polo --- o teorema da caixa de chapéu de Arquimedes, e a razão
pela qual a \omterm{def:b2:surfaces:area}{área} lateral do cilindro circunscrito ($2\pi R \cdot
2R = 4\pi R^2$) é igual à \omterm{def:b2:surfaces:area}{área} da esfera.
\end{solution}

\begin{solution}{exo:b2:surfaces:10}
$\sigma_\theta = (-r\sin\theta,\ r\cos\theta,\ 0)$ e
$\sigma_z = (r'\cos\theta,\ r'\sin\theta,\ 1)$, logo
\[
\sigma_\theta \wedge \sigma_z
= (r\cos\theta,\ r\sin\theta,\ -r\,r') ,
\]
um vetor normal em $M = (r\cos\theta, r\sin\theta, z)$. A
reta normal é
\[
t \mapsto \bigl(r(1 + t)\cos\theta,\ r(1 + t)\sin\theta,\
z - t\,r\,r'\bigr),
\]
que em $t = -1$ atinge $(0,\ 0,\ z + r(z)\,r'(z))$: toda
reta normal encontra o eixo, na altura $z + rr'$. (Essa é
a razão tridimensional pela qual a simetria de rotação sobrevive no
campo normal.)
\end{solution}

\begin{solution}{exo:b2:surfaces:11}
$\sigma_u = (-\sin u, \cos u, 0)$, $\sigma_v = (0, 0, 1)$: $E
= 1$, $F = 0$, $G = 1$. Pela
\cref{prop:b2:surfaces:curvelength}, o \omterm{def:b2:curves:length}{comprimento} de uma curva
desenhada é $\int\sqrt{u'^2 + v'^2}\,\dd t$ --- o \omterm{def:b2:curves:length}{comprimento} de sua
sombra de parâmetros $(u(t), v(t))$ no plano: a carta é uma
isometria local (o desenrolamento do cilindro). A hélice
$t \mapsto (\cos t, \sin t, ct)$, $t \in \intcc0{2\pi}$, tem por
sombra o segmento de $(0,0)$ a $(2\pi, 2\pi c)$, de \omterm{def:b2:curves:length}{comprimento}
$2\pi\sqrt{1 + c^2}$. Toda curva desenhada de $(1,0,0)$ a $(1,
0, 2\pi c)$ dando uma volta completa tem uma sombra \omterm{def:b2:metric:continuity}{contínua}
ligando $(0, 0)$ a $(2\pi, 2\pi c)$, de \omterm{def:b2:curves:length}{comprimento} no plano $\geq$
o do segmento reto; como os \omterm{def:b2:curves:length}{comprimentos} coincidem, a hélice é a
mais curta.
\end{solution}

\begin{solution}{exo:b2:surfaces:12}
A função $g(M) = \norm{M}^2$ é \omterm{def:b2:metric:continuity}{contínua} no \omterm{def:b2:metric:compact}{compacto}
$S$, logo atinge seu máximo em algum $M_0$. Para toda curva
$\gamma$ desenhada sobre $S$ com $\gamma(0) = M_0$, a função $t
\mapsto \norm{\gamma(t)}^2$ tem um máximo em $t = 0$, de modo que sua
derivada $2\langle\gamma'(0), M_0\rangle$ se anula: $M_0$ é
ortogonal a todo vetor tangente, isto é, normal a $S$ em
$M_0$. Como $\nabla f(M_0) \neq 0$ também dirige a reta normal
(\cref{prop:b2:surfaces:gradient}), $\nabla f(M_0)$ e
$\vect{OM_0}$ são colineares. Para o elipsoide, radialidade
significa
\[
\Bigl(\frac{x_0}{a^2}, \frac{y_0}{b^2},
\frac{z_0}{c^2}\Bigr) = \mu\,(x_0, y_0, z_0) :
\]
cada coordenada satisfaz $x_0(\frac1{a^2} - \mu) = 0$ etc.;
como $a^{-2}, b^{-2}, c^{-2}$ são distintos, no máximo uma
coordenada é não nula, e as soluções sobre a superfície são
os seis extremos dos eixos $(\pm a, 0, 0)$, $(0, \pm b, 0)$, $(0,
0, \pm c)$. Os pontos mais distantes são $(\pm a, 0, 0)$, à
distância $a = \max(a,b,c)$.
\end{solution}

\begin{solution}{pb:b2:surfaces:1}
\textbf{1.} Desenvolvendo, e usando a simetria de $A$
($t^{\mathsf T}\!APY = (At)^{\mathsf T}PY$):
\[
q(PY + t) = Y^{\mathsf T}P^{\mathsf T}\!APY +
2\,(At + b)^{\mathsf T}PY + \bigl(t^{\mathsf T}\!At +
2b^{\mathsf T}t + c\bigr),
\]
que é a tripla exibida. $A' = P^{\mathsf T}\!AP =
P^{-1}AP$ é semelhante a $A$: mesmo \omterm{def:b2:reduction:charpoly}{polinômio característico},
mesmo \omterm{def:b2:reduction:eigen}{espectro}, mesmo posto, mesma assinatura. Por fim, $\{q = 0\} = \{sq = 0\}$
para $s \neq 0$, de modo que só a equação \emph{a menos de um escalar} está
ligada ao conjunto; multiplicar por $s$ multiplica todos os \omterm{def:b2:reduction:eigen}{autovalores}
por $s$.

\textbf{2.} O teorema espectral fornece $P \in O(3)$ com
$P^{\mathsf T}\!AP = \operatorname{diag}(\lambda_1, \lambda_2,
\lambda_3)$; a questão 1 com $t = 0$ transforma a equação em
$\sum\lambda_iy_i^2 + 2\sum\beta_iy_i + c = 0$, em que $\beta =
P^{\mathsf T}b$.

\textbf{3.} Para $\lambda_i \neq 0$: $\lambda_iy_i^2 +
2\beta_iy_i = \lambda_i\bigl(y_i +
\frac{\beta_i}{\lambda_i}\bigr)^2 -
\frac{\beta_i^2}{\lambda_i}$; a translação $z_i = y_i +
\beta_i/\lambda_i$ (e $z_i = y_i$ para $i > r$) dá
\[
\sum_{i \leq r}\lambda_iz_i^2 + 2\sum_{i > r}\beta_iz_i +
c'' = 0, \qquad
c'' = c - \sum_{i\leq r}\frac{\beta_i^2}{\lambda_i}.
\]

\textbf{4.} $q(2\Omega - X) = (2\Omega - X)^{\mathsf T}
A(2\Omega - X) + 2b^{\mathsf T}(2\Omega - X) + c$; desenvolvendo
e subtraindo $q(X)$, os termos quadráticos se cancelam e
\[
q(2\Omega - X) - q(X) = -4\,\langle A\Omega + b,\ X\rangle +
4\,\langle A\Omega + b,\ \Omega\rangle .
\]
Se $A\Omega + b = 0$, isso se anula identicamente: a simetria
\omterm{def:b2:funcseq:def}{pontual} preserva $q$, logo a \omterm{pb:b2:surfaces:1}{quádrica}. Reciprocamente,
``$q(2\Omega - X) = q(X)$ para todo $X$'' diz que a função \omterm{def:b2:affine:subspace}{afim}
acima se anula em todo $\R^3$, o que força sua
parte linear $A\Omega + b$ a ser nula. Assim, centros $=$ soluções de
$A\Omega = -b$: um conjunto não vazio se e somente se $b \in \operatorname{im}A$
(um subespaço \omterm{def:b2:affine:subspace}{afim} dirigido por $\ker A$), e um único ponto
se e somente se $A$ é invertível.

\textbf{5.} Assinatura $(3,0)$ (todos os $\lambda_i > 0$): $\delta >
0$ dá $\frac{x^2}{a^2} + \frac{y^2}{b^2} + \frac{z^2}{c^2}
= 1$ com $a = \sqrt{\delta/\lambda_1}$ etc.\ --- um
elipsoide; $\delta = 0$: o único ponto $O$; $\delta < 0$:
vazio. Assinatura $(2,1)$ ($\lambda_1, \lambda_2 > 0 >
\lambda_3$): $\delta > 0$: $\frac{x^2}{a^2} + \frac{y^2}{b^2}
- \frac{z^2}{c^2} = 1$, o hiperboloide de uma folha; $\delta =
0$: o cone $\frac{x^2}{a^2} + \frac{y^2}{b^2} =
\frac{z^2}{c^2}$; $\delta < 0$: $\frac{z^2}{c^2} -
\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1$, o hiperboloide de
duas folhas ($\abs z \geq c$: duas componentes).

\textbf{6.} A parte quadrática tem matriz $A$ com diagonal
$1$ e fora da diagonal $2$: $A = 2J - I$. Como $J$ tem \omterm{def:b2:reduction:eigen}{espectro}
$\{3, 0, 0\}$ (\omterm{def:b2:reduction:eigen}{autovetor} $(1,1,1)$ para $3$), $A$ tem
\omterm{def:b2:reduction:eigen}{espectro} $\{5, -1, -1\}$, sendo o \omterm{def:b2:reduction:eigen}{autovalor} $5$ carregado por
$\R(1,1,1)$. Nas coordenadas rodadas: $5u^2 - v^2 - w^2 =
1$, isto é, $\frac{u^2}{1/5} - v^2 - w^2 = 1$: um hiperboloide de duas
folhas, de revolução (\omterm{def:b2:reduction:eigen}{autovalores} iguais $-1$) em torno do
eixo $\R(1,1,1)$.

\textbf{7.} A superfície é $\bigl(\frac xa - \frac
zc\bigr)\bigl(\frac xa + \frac zc\bigr) = \bigl(1 - \frac
yb\bigr)\bigl(1 + \frac yb\bigr)$. Para $(\lambda : \mu) \neq
(0:0)$ defina a reta
\[
D_{\lambda:\mu} :\quad
\lambda\Bigl(\frac xa - \frac zc\Bigr) = \mu\Bigl(1 - \frac
yb\Bigr), \qquad
\mu\Bigl(\frac xa + \frac zc\Bigr) = \lambda\Bigl(1 + \frac
yb\Bigr)
\]
(duas equações \omterm{def:b2:affine:subspace}{afins} independentes: uma reta). Multiplicando as
duas equações vê-se que todo ponto de $D_{\lambda:\mu}$ está
sobre a superfície quando $\lambda\mu \neq 0$; os casos $\lambda = 0$
ou $\mu = 0$ verificam-se diretamente (por exemplo $\lambda = 0$: $y =
b$, $\frac xa = -\frac zc$, o que satisfaz a equação). A
segunda família $D'_{\lambda:\mu}$ troca os dois fatores do
membro da direita.

\textbf{8.} Fixe $M$ sobre a superfície. As condições para $M \in
D_{\lambda:\mu}$ formam um sistema linear homogêneo $2\times2$
em $(\lambda, \mu)$ cujo \omterm{def:b2:linalg:det}{determinante} vale
\[
\Bigl(\frac{x^2}{a^2} - \frac{z^2}{c^2}\Bigr) - \Bigl(1 -
\frac{y^2}{b^2}\Bigr) = 0
\]
precisamente porque $M$ está sobre a \omterm{pb:b2:surfaces:1}{quádrica}: existe uma solução
não trivial $(\lambda : \mu)$. A matriz de coeficientes nunca é
nula (isso forçaria $1 - \frac yb = 1 + \frac yb =
0$), de modo que seu posto é $1$ e a solução é única a menos de
escala: exatamente uma reta da família passa por $M$.
O mesmo vale para a segunda família, e as duas retas são
distintas (em $(a, 0, 0)$ elas são $\{x = a,\ \frac zc = \frac
yb\}$ e $\{x = a,\ \frac zc = -\frac yb\}$): o hiperboloide
de uma folha é duplamente regrado.

\textbf{9.} Seja $M = (x, y, z)$ sobre o hiperboloide, $\rho^2 =
\frac{x^2}{a^2} + \frac{y^2}{b^2} = 1 + \frac{z^2}{c^2} \geq
1$. O ponto $N = (x,\ y,\ \varepsilon c\rho)$ com
$\varepsilon$ o sinal de $z$ satisfaz $\frac{x^2}{a^2} +
\frac{y^2}{b^2} - \frac{(c\rho)^2}{c^2} = 0$: $N \in C$, e
\[
d(M, C) \leq \abs{z - \varepsilon c\rho}
= c\bigl(\rho - \sqrt{\rho^2 - 1}\bigr)
= \frac{c}{\rho + \sqrt{\rho^2 - 1}} .
\]
Se $\norm M \to \infty$, então $\rho \to \infty$ (as três
coordenadas são limitadas por múltiplos de $\rho$), logo $d(M, C)
\to 0$. A seção $x = a$: $\frac{y^2}{b^2} -
\frac{z^2}{c^2} = 0$, o par de retas concorrentes da questão
8 --- e do mesmo modo em $x = -a$.

\textbf{10.} Com $\lambda_3 = 0$, a questão 3 deixa
$\lambda_1z_1^2 + \lambda_2z_2^2 + 2\beta_3z_3 + c'' = 0$. Na
base de \omterm{def:b2:reduction:eigen}{autovetores}, $\operatorname{im}A = \operatorname{Vect}(e_1,
e_2)$, de modo que, pela questão 4, existem centros se e somente se $\beta_3 = 0$. Se
$\beta_3 \neq 0$: a translação $z_3 \mapsto z_3 -
c''/(2\beta_3)$ elimina a constante, deixando $\lambda_1z_1^2
+ \lambda_2z_2^2 + 2\beta_3z_3 = 0$, isto é, $z_3 = px^2 +
qy^2$ após renomear: um \emph{paraboloide elíptico} se
$\lambda_1\lambda_2 > 0$, um \emph{paraboloide hiperbólico} se
$\lambda_1\lambda_2 < 0$ --- e, de fato, nenhum centro. Se
$\beta_3 = 0$: a equação $\lambda_1z_1^2 + \lambda_2z_2^2 +
c'' = 0$ não envolve $z_3$: a \omterm{pb:b2:surfaces:1}{quádrica} é um cilindro
sobre a cônica plana correspondente --- cilindro elíptico,
reta ou conjunto vazio quando $\lambda_1\lambda_2 > 0$; cilindro
hiperbólico ou par de planos concorrentes quando
$\lambda_1\lambda_2 < 0$ --- com toda uma reta de centros
$\{(z_1^*, z_2^*)\} \times \R$.

\textbf{11.} $xy - z = 0$. Substituindo $x = \frac{u +
v}{\sqrt2}$, $y = \frac{u - v}{\sqrt2}$ (rotação de $\pi/4$):
$xy = \frac{u^2 - v^2}2$, de modo que a equação se torna $z =
\frac12(u^2 - v^2)$: um paraboloide hiperbólico --- a sela
da figura, a menos do fator $\frac12$.

\textbf{12.} A reta $\{x = x_0,\ z = x_0y\}$ (parametrizada
por $y$) está claramente sobre $z = xy$, assim como $\{y = y_0,\ z =
xy_0\}$; por $(x_0, y_0, x_0y_0)$ passam as duas.
Unicidade: se $t \mapsto (x_0 + tv_1, y_0 + tv_2, z_0 +
tv_3)$ permanece sobre a superfície, o coeficiente de $t^2$ em $(x_0
+ tv_1)(y_0 + tv_2) - z_0 - tv_3$ dá $v_1v_2 = 0$, logo
$v_1 = 0$ ou $v_2 = 0$, caindo em uma das duas famílias:
uma reta de cada uma por cada ponto --- a segunda \omterm{pb:b2:surfaces:1}{quádrica}
duplamente regrada.

\textbf{13.} Completando os quadrados: $(x - 1)^2 + (y + 2)^2 = 2$,
sem condição alguma sobre $z$: um cilindro circular reto de raio
$\sqrt2$ e eixo a reta vertical $\{(1, -2, z) : z \in
\R\}$ --- uma reta de centros, como a questão 10 prevê.

\textbf{14.} Com $\lambda_2 = \lambda_3 = 0$, a equação reduzida
é $\lambda_1z_1^2 + 2\beta_2z_2 + 2\beta_3z_3 + c''
= 0$. Uma rotação do plano do núcleo $(z_2, z_3)$ alinha
a forma linear: $2\beta_2z_2 + 2\beta_3z_3 = 2\beta w$ com
$\beta = \sqrt{\beta_2^2 + \beta_3^2}$. Se $\beta \neq 0$,
transladar $w$ absorve $c''$: $\lambda_1z_1^2 + 2\beta w =
0$, um \emph{cilindro parabólico}; se $\beta = 0$:
$\lambda_1z_1^2 = -c''$ dá dois planos paralelos ($c''
\lambda_1 < 0$), um plano duplo ($c'' = 0$) ou o conjunto
vazio. Para $(x + y)^2 = z$: com $u = \frac{x + y}{\sqrt2}$ a
equação se lê $z = 2u^2$: um cilindro parabólico, invariante
por translações ao longo de $(1, -1, 0)$.

\textbf{15.} Toda \omterm{pb:b2:surfaces:1}{quádrica} é levada por uma rotação mais
translações sobre uma de: (posto 3) elipsoide, ponto, conjunto
vazio, hiperboloide de uma folha, cone, hiperboloide de duas folhas;
(posto 2) paraboloide elíptico, paraboloide hiperbólico, cilindro
elíptico, reta, conjunto vazio, cilindro hiperbólico, par de
planos concorrentes; (posto 1) cilindro parabólico, par de
planos paralelos, plano duplo, conjunto vazio. Identificando as
três variantes vazias como \emph{tipos \omterm{def:b2:affine:subspace}{afins}} distintos de
equações, a contagem é dezessete; entre eles nove são superfícies
de verdade: elipsoide, os dois hiperboloides, o cone, os dois
paraboloides e os três cilindros.

\textbf{16.} Algoritmo. (i) Leia $(A, b, c)$; calcule o
\omterm{def:b2:reduction:charpoly}{polinômio característico} de $A$, seus \omterm{def:b2:reduction:eigen}{autovalores} (reais, pelo
teorema espectral) e $r = \operatorname{rank}A$. (ii)
Resolva $A\Omega = -b$ (eliminação gaussiana): solúvel ou não
--- há centros ou não. (iii) Se for solúvel, transladar para um centro:
a equação se torna $\sum\lambda_iz_i^2 + c'' = 0$ com $c''
= c + \langle b, \Omega\rangle$; classifique por $r$, pela assinatura
e pelo sinal de $c''$, usando as questões 5, 10 e 14. (iv) Se não
for solúvel ($r \leq 2$), rode e reduza como nas questões 10
e 14: paraboloide ($r = 2$) ou cilindro parabólico ($r = 1$),
elíptico/hiperbólico conforme o sinal de
$\lambda_1\lambda_2$. Cada passo é um cálculo finito:
raízes de uma cúbica de raízes reais, postos, sistemas lineares.

\textbf{17.} $A$ tem diagonal $1$ e fora da diagonal $-1$: $A = 2I
- J$, \omterm{def:b2:reduction:eigen}{espectro} $\{2 - 3,\ 2,\ 2\} = \{-1, 2, 2\}$ com $-1$
em $\R(1,1,1)$. Assinatura $(2,1)$, $b = 0$, membro da direita
$\delta = 1 > 0$: $2u^2 + 2v^2 - w^2 = 1$, um hiperboloide de uma
folha de revolução em torno do eixo $\R(1,1,1)$.

\textbf{18.} Complete os quadrados: $(x-1)^2 + (y+2)^2 - (z-1)^2 +
(-1 - 4 + 1 + 4) = 0$, isto é,
\[
(x-1)^2 + (y+2)^2 = (z-1)^2 :
\]
a constante desapareceu --- um \emph{cone} circular reto de
vértice (e centro único) $(1, -2, 1)$ e eixo paralelo a
$Oz$.

\textbf{19.} A parte quadrática $x^2 + 4xy + y^2$ tem matriz
$\begin{psmallmatrix}1 & 2\\ 2 & 1\end{psmallmatrix}$ (no plano
$xy$), \omterm{def:b2:reduction:eigen}{autovalores} $3$ (em $(1,1)$) e $-1$ (em
$(1,-1)$): com $u = \frac{x+y}{\sqrt2}$, $v =
\frac{x-y}{\sqrt2}$ ela vale $3u^2 - v^2$, e a \omterm{pb:b2:surfaces:1}{quádrica}
é
\[
z = \tfrac32u^2 - \tfrac12v^2 :
\]
um paraboloide hiperbólico ($A$ tem posto $2$ e $b$ tem uma
componente ao longo de $\ker A = \R e_z$: sem centro).

\textbf{20.} Um movimento rígido transforma os dados da equação por
$A \mapsto P^{\mathsf T}\!AP$ (mesmos \omterm{def:b2:reduction:eigen}{autovalores}) e as
formas normais não têm liberdade residual além de permutar
coordenadas e multiplicar a equação inteira por um escalar
($> 0$ para preservar a escrita): duas formas normais centrais
coincidem a menos de isometria se e somente se as listas de coeficientes coincidem a menos de
permutação e de um fator positivo comum --- para o elipsoide,
se e somente se os semieixos $(a, b, c)$ coincidem. Afimmente, pode-se também
escalonar cada coordenada separadamente ($z_i \mapsto z_i\sqrt{\abs
{\lambda_i}}$), o que apaga os \omterm{def:b2:reduction:eigen}{autovalores} e deixa apenas
seus sinais: todo elipsoide se torna $u^2 + v^2 + w^2 = 1$,
a esfera. O teorema da inércia de Sylvester
(\cref{thm:b2:quadratic:sylvester}) garante que a assinatura
sobrevive a qualquer mudança linear invertível: os tipos \omterm{def:b2:affine:subspace}{afins} da
questão 15 são genuinamente distintos.

\textbf{21.} Parametrize o plano afimmente: $M = P + su +
tv$. Então $q(P + su + tv)$ é um polinômio de grau $\leq 2$
em $(s, t)$ (desenvolva a \omterm{def:b2:quadratic:def}{forma quadrática} bilinearmente), de modo que a
seção $\{q = 0\}$ é uma cônica do plano, possivelmente
degenerada. Para $z = xy$ e o plano $z = c$: $xy = c$, uma
hipérbole para $c \neq 0$ e, para $c = 0$, as duas retas
coordenadas --- o par de geratrizes pela origem.

\textbf{22.} \emph{Elipsoide}: limitado, não contém reta alguma.
\emph{Hiperboloide de duas folhas} $\frac{z^2}{c^2} -
\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1$: restrinja a $p + tv$;
o coeficiente de $t^2$, $\frac{v_3^2}{c^2} - \frac{v_1^2}{a^2}
- \frac{v_2^2}{b^2}$, tem de se anular, logo $v_3 \neq 0$ (senão $v =
0$); o coeficiente de $t$ dá $\frac{p_3v_3}{c^2} =
\frac{p_1v_1}{a^2} + \frac{p_2v_2}{b^2}$ e Cauchy--Schwarz
fornece
\[
\frac{p_3^2}{c^2}
= \frac{c^2}{v_3^2}\Bigl(\frac{p_1v_1}{a^2} +
\frac{p_2v_2}{b^2}\Bigr)^2
\leq \frac{c^2}{v_3^2}\Bigl(\frac{p_1^2}{a^2} +
\frac{p_2^2}{b^2}\Bigr)\frac{v_3^2}{c^2}
= \frac{p_1^2}{a^2} + \frac{p_2^2}{b^2},
\]
de modo que o termo constante é $\leq 0 \neq 1$: nenhuma reta.
\emph{Paraboloide elíptico} $z = \frac{x^2}{a^2} +
\frac{y^2}{b^2}$: o coeficiente de $t^2$ força $v_1 = v_2 =
0$ e então a equação é linear não constante em $t$: nenhuma reta.
\emph{Cone} $x^2 + y^2 = z^2$: uma reta sobre ele satisfaz $q(p) =
q(v) = B(p, v) = 0$ para a forma de Lorentz; a igualdade na
desigualdade de Cauchy--Schwarz plana $\abs{p_1v_1 + p_2v_2} = \abs{p_3v_3} =
\sqrt{(p_1^2 + p_2^2)(v_1^2 + v_2^2)}$ força $(p_1, p_2)
\parallel (v_1, v_2)$ e então $p = kv$: todas as retas passam
pelo vértice --- uma só família. \emph{Cilindros}: para os
cilindros elíptico e parabólico o coeficiente de $t^2$ força
$v_1 = v_2 = 0$ (só as geratrizes); para o cilindro hiperbólico
$\frac{x^2}{a^2} - \frac{y^2}{b^2} = 1$, $\frac{v_1}a = \pm
\frac{v_2}b$ com $v_2 \neq 0$ leva, pelo coeficiente de $t$,
a $\frac{p_1^2}{a^2} = \frac{p_2^2}{b^2}$, contradizendo o
termo constante $1$: de novo só as geratrizes verticais. Assim, as
superfícies \omterm{pb:b2:surfaces:1}{quádricas} duplamente regradas são exatamente o hiperboloide
de uma folha e o paraboloide hiperbólico.

\textbf{23.} Gauss: $x^2 - 2xy - 2zx = (x - y - z)^2 - y^2 -
z^2 - 2yz$, logo
\[
q = (x - y - z)^2 - 4yz = (x - y - z)^2 + (y - z)^2 - (y +
z)^2 :
\]
três quadrados independentes com sinais $(+, +, -)$ --- assinatura
$(2, 1)$, coincidindo com a questão 17, sem cálculo algum de
\omterm{def:b2:reduction:eigen}{autovalores}. Gauss perde os dados métricos: as novas coordenadas
não são ortonormais, de modo que os \omterm{def:b2:reduction:eigen}{autovalores} (a forma do
hiperboloide, seus eixos e seus \omterm{def:b2:curves:length}{comprimentos}) se foram; só resta o
tipo \omterm{def:b2:affine:subspace}{afim}.

\textbf{24.} Sejam $p$, $n$, $z$ os números de \omterm{def:b2:reduction:eigen}{autovalores} positivos,
negativos e nulos, $p + n + z = 3$. A regra de Descartes
limita $p$ pelo número $V$ de mudanças de sinal de
$\chi_A$, e $n$ pelo número $V'$ de mudanças de sinal de
$\chi_A(-\lambda)$; além disso, cada par de coeficientes não nulos
consecutivos produz uma mudança em exatamente um dos dois
polinômios, de modo que $V + V' \leq 3 - z$ (as raízes nulas são visíveis
como coeficientes finais que se anulam). Então $p + n = 3 - z \geq
V + V' \geq p + n$: igualdade, logo $p = V$ exatamente --- os
sinais dos \omterm{def:b2:reduction:eigen}{autovalores} podem ser lidos diretamente. Para $\chi_A(\lambda)
= \lambda^3 - 3\lambda^2 - 9\lambda - 5$: os sinais $+,-,-,-$
dão $V = 1$, e $\chi_A(-\lambda) = -\lambda^3 - 3\lambda^2
+ 9\lambda - 5$ tem sinais $-,-,+,-$: $V' = 2$. Assinatura $(1,
2)$ --- coerente com a fatoração exata $\chi_A =
(\lambda - 5)(\lambda + 1)^2$ da questão 6.

\textbf{25.} Algoritmo: diagonalizar ortonormalmente a parte
quadrática (teorema espectral: esse é o único passo
analiticamente profundo, e é o que torna a
classificação \emph{euclidiana}); resolver $A\Omega = -b$ para
decidir entre tipos centrais e parabólicos e para eliminar por translação
a parte linear quando possível (geometria \omterm{def:b2:affine:subspace}{afim}); ler o
tipo a partir do posto, da assinatura e da constante (Sylvester
garante que esses são invariantes); quando só importa o tipo
\omterm{def:b2:affine:subspace}{afim}, a \omterm{thm:b2:quadratic:gauss}{redução de Gauss} substitui o teorema espectral ao
custo da informação métrica. Em $\R^2$ a mesma
máquina classifica as cônicas: elipse, hipérbole, parábola, mais
pares de retas, uma reta, um ponto e conjuntos vazios. Em $\R^n$
nada muda a não ser a contabilidade: os tipos são indexados pela
assinatura de $A$, pela posição de $b$ em relação a
$\operatorname{im}A$ e por uma constante --- com a matriz orlada de
dimensão $(n{+}2)$ de $(A, b, c)$
fornecendo um invariante \omterm{def:b2:metric:compact}{compacto}.
\end{solution}
