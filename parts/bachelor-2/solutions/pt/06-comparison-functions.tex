\section*{Capítulo \ref{ch:b2:comparison} --- Comparação de funções}

\begin{solution}{exo:b2:comparison:1}
$\sqrt{x^2 + x + 1} = x\sqrt{1 + \tfrac1x + \tfrac{1}{x^2}} = x +
\frac12 + \frac38\cdot\frac1x + o\bigl(\frac1x\bigr)$ (binomial
expansão: $\frac12 u - \frac18 u^2$ com $ u = \frac1x +
\frac{1}{x^2}$ dá $\frac{1}{2x} + \frac{1}{2x^2} -
\frac{1}{8x^2} = \frac{1}{2x} + \frac{3}{8x^2}$, então multiplique por
$ x $).

$\ln(x^2 + x) - 2\ln x = \ln\bigl(1 + \tfrac1x\bigr) = \frac1x -
\frac{1}{2x^2} + o\bigl(\frac{1}{x^2}\bigr)$.

Terceira função: expandir cada fator,
\[
\frac{x + \sin x}{x - \ln x}
= \Bigl(1 + \frac{\sin x}{x}\Bigr)
\Bigl(1 + \frac{\ln x}{x} + \frac{(\ln x)^2}{x^2} +
O\Bigl(\frac{(\ln x)^3}{x^3}\Bigr)\Bigr).
\]
Encomende as contribuições na escala em $+\infty $: $\frac{\ln x}{x}
\gg \frac{1}{x} \geq \bigl|\frac{\sin x}{x}\bigr| \gg \frac{(\ln
x)^2}{x^2}$. Os dois termos após o líder $1$ são portanto
$\frac{\ln x}{x}$, então o termo de oscilação limitada $\frac{\sin
x}{x}$:
\[
\frac{x + \sin x}{x - \ln x}
= 1 + \frac{\ln x}{x} + \frac{\sin x}{x}
+ O\Bigl(\frac{(\ln x)^2}{x^2}\Bigr).
\]
\end{solution}

\begin{solution}{exo:b2:comparison:2}
$ f(t) = t^\alpha $ ($ t \geq 1$).

$\alpha > -1$: divergência, e por
\cref{thm:b2:comparison:seriesintegral}~(2), $\sum_{k\leq n}
k^\alpha = \frac{n^{\alpha+1}}{\alpha+1} + C + o(1)$ se $\alpha <
0$ (onde $ f $ diminui); para $\alpha \geq 0$ ($ f $ aumentando) o
mesmo colchetes com desigualdades invertidas dá $\sum_{k \leq n}
k^\alpha \sim \frac{n^{\alpha + 1}}{\alpha + 1}$.

$\alpha = -1$: $ H_n = \ln n + \gamma + o(1)$
(\cref{ex:b2:comparison:harmonic}).

$\alpha < -1$: convergência, com resto $\sum_{k > n} k^\alpha
\sim \frac{n^{\alpha+1}}{-(\alpha+1)}$ pelo colchete (1) (ambos
limites integrais são equivalentes a esse valor).
\end{solution}

\begin{solution}{exo:b2:comparison:3}
Deixar $ v_n = H_n - \ln n - \gamma \to 0$. Então
\[
v_n - v_{n+1} = \ln\frac{n+1}{n} - \frac{1}{n+1}
= \Bigl(\frac1n - \frac{1}{2n^2}\Bigr) - \Bigl(\frac1n -
\frac{1}{n^2}\Bigr) + O\Bigl(\frac{1}{n^3}\Bigr)
= \frac{1}{2n^2} + O\Bigl(\frac{1}{n^3}\Bigr),
\]
usando $\frac{1}{n+1} = \frac1n - \frac{1}{n^2} + O(n^{-3})$. Desde
$ v_n \to 0$, telescópicamente a cauda:
\[
v_n = \sum_{k \geq n} (v_k - v_{k+1})
= \sum_{k\geq n} \Bigl(\frac{1}{2k^2} + O(k^{-3})\Bigr)
= \frac{1}{2n} + O\Bigl(\frac{1}{n^2}\Bigr),
\]
por \cref{thm:b2:comparison:seriesintegral}~(1) aplicado a $ t^{-2}$
(restante $\sim \frac1n $, dividido pela metade) e para $ t^{-3}$. Por isso $ H_n =
\ln n + \gamma + \frac{1}{2n} + o(\frac1n)$.
\end{solution}

\begin{solution}{exo:b2:comparison:4}
Stirling três vezes:
\[
\frac{(3n)!}{(n!)^3}
\sim \frac{\sqrt{6\pi n}\,(3n/\eu)^{3n}}
{(2\pi n)^{3/2}\,(n/\eu)^{3n}}
= \frac{\sqrt{6}\; 27^{\,n}}{2\pi n} \cdot
\frac{1}{\sqrt{2\pi n}}\cdot\sqrt{2\pi n}\;
= \frac{\sqrt3\,27^n}{2\pi n} .
\]
(Com cuidado: $\frac{\sqrt{6\pi n}}{(2\pi n)^{3/2}} =
\frac{\sqrt6}{(2\pi n)\sqrt{2\pi n}}\sqrt{\pi n} =
\frac{\sqrt3}{2\pi n}$.)

$\dfrac{n!}{n^n} \sim \sqrt{2\pi n}\,\eu^{-n}$.

$\sqrt[n]{n!} = \exp\bigl(\frac{\ln n!}{n}\bigr)$ com $\ln n! = n\ln
n - n + \frac12\ln(2\pi n) + o(1)$:
\[
\sqrt[n]{n!} = \exp\Bigl(\ln n - 1 + \frac{\ln(2\pi n)}{2n} +
o\Bigl(\frac{\ln n}{n}\Bigr)\Bigr)
= \frac{n}{\eu}\Bigl(1 + \frac{\ln(2\pi n)}{2n} +
o\Bigl(\frac{\ln n}{n}\Bigr)\Bigr).
\]
\end{solution}

\begin{solution}{exo:b2:comparison:5}
$ g(x) = x^n + x - 1$ aumenta estritamente em $\intcc{0}{1}$ de $-1$ a $1$: raiz única $ x_n $. Desde $ x_n^n = 1 - x_n \in
\intoo{0}{1}$: se $ x_n \leq c < 1$ ao longo de uma subsequência, então $ x_n^n
\leq c^n \to 0$, então $1 - x_n \to 0$: contradição com $ x_n \leq
c $. Por isso $ x_n \to 1$.

Escrever $ x_n = 1 - \varepsilon_n $, $\varepsilon_n \to 0^+$. O
leituras de equações $(1 - \varepsilon_n)^n = \varepsilon_n $, ou seja
\[
n\ln(1 - \varepsilon_n) = \ln \varepsilon_n
\quad\Longrightarrow\quad
-n\varepsilon_n\bigl(1 + o(1)\bigr) = \ln\varepsilon_n .
\]
Então $ n\varepsilon_n = -\ln\varepsilon_n\,(1 + o(1)) \to +\infty $ e
tomando logaritmos novamente: $\ln n + \ln\varepsilon_n =
\ln(-\ln\varepsilon_n) + o(1)$. Desde $\ln(-\ln \varepsilon_n) =
o(\ln(1/\varepsilon_n))$, isso dá $\ln\varepsilon_n \sim -\ln
n $, de onde $\varepsilon_n = \frac{-\ln\varepsilon_n}{n}(1 + o(1))
\sim \frac{\ln n}{n}$:
\[
x_n = 1 - \frac{\ln n}{n} + o\Bigl(\frac{\ln n}{n}\Bigr) .
\]
\end{solution}

\begin{solution}{exo:b2:comparison:6}
Relação exata: $\cot y_n = x_n = n\pi + \frac\pi2 - y_n $, com $ y_n
\sim \frac{1}{n\pi}$ (\cref{ex:b2:comparison:tan}). Expandir $\cot y
= \frac1y - \frac y3 + O(y^3)$:
\[
\frac{1}{y_n} - \frac{y_n}{3} + O(y_n^3) = n\pi + \frac\pi2 - y_n
\quad\Longrightarrow\quad
\frac{1}{y_n} = n\pi + \frac\pi2 + O\Bigl(\frac1n\Bigr),
\]
(os termos $-y_n $ e $-\frac{y_n}{3}$ são $ O(\frac1n)$). Invertido:
\[
y_n = \frac{1}{n\pi}\cdot\frac{1}{1 + \frac{1}{2n} + O(n^{-2})}
= \frac{1}{n\pi}\Bigl(1 - \frac{1}{2n} + O\Bigl(\frac{1}{n^2}\Bigr)\Bigr)
= \frac{1}{n\pi} - \frac{1}{2n^2\pi} + O\Bigl(\frac{1}{n^3}\Bigr).
\]
Por isso
\[
x_n = n\pi + \frac{\pi}{2} - y_n
= n\pi + \frac\pi2 - \frac{1}{n\pi} + \frac{1}{2n^2\pi} +
o\Bigl(\frac{1}{n^2}\Bigr).
\]
\end{solution}

\begin{solution}{exo:b2:comparison:7}
Divida os dois maiores termos:
\[
\sum_{k=0}^{n} k! = n! + (n-1)! + \sum_{k \leq n-2} k! ,
\qquad
\sum_{k\leq n-2} k! \leq (n-1)\,(n-2)! = (n-1)! .
\]
Então $1 \leq \frac{1}{n!}\sum k! \leq 1 + \frac{2}{n}$: o limite é
$1$. Refino: $\frac{(n-1)!}{n!} = \frac1n $ e o limite bruto
$\sum_{k \leq n-2}k! \leq (n-1)!$ pode ser afiado da mesma maneira:
$\sum_{k\leq n-2} k! = (n-2)!\,(1 + O(\frac1n)) = O\bigl(\frac{n!}{n^2}\bigr)$.
Por isso
\[
\sum_{k=0}^{n} k! = n!\Bigl(1 + \frac1n + O\Bigl(\frac{1}{n^2}\Bigr)\Bigr).
\]
\end{solution}

\begin{solution}{exo:b2:comparison:8}
$(u_n)$ aumenta; se limitado, convergiria para $\ell $ com
$\ell = \ell + \frac1\ell $: absurdo. Então $ u_n \to \infty $.

Quadrados: $ u_{n+1}^2 = u_n^2 + 2 + u_n^{-2}$, então
\[
u_n^2 = u_0^2 + 2n + \sum_{k=0}^{n-1} \frac{1}{u_k^2} .
\]
A soma é $ o(n)$ (os termos tendem a $0$, Cesàro), então $ u_n^2 \sim 2n $
e $ u_n \sim \sqrt{2n}$.

Refinamento: $\frac{1}{u_k^2} \sim \frac{1}{2k}$, então por comparação
(\cref{thm:b2:comparison:seriesintegral}, ou equivalentes de parcial
somas de séries positivas) $\sum_{k<n} u_k^{-2} \sim \frac12 \ln n $.
Por isso
\[
u_n^2 = 2n + \frac{\ln n}{2}\,(1 + o(1)) + O(1)
\quad\Longrightarrow\quad
u_n = \sqrt{2n}\sqrt{1 + \frac{\ln n}{4n} + o\Bigl(\frac{\ln
n}{n}\Bigr)}
= \sqrt{2n}\Bigl(1 + \frac{\ln n}{8n} + o\Bigl(\frac{\ln
n}{n}\Bigr)\Bigr).
\]
\end{solution}

\begin{solution}{exo:b2:comparison:9}
Fator $ n $ e definir $ t = \ln n $:
\[
S_n = \frac1n \sum_{k=1}^{n} \frac{1}{1 + t\,\frac kn} .
\]
Para fixo $ t $, a soma é uma soma de Riemann de $ u \mapsto \frac{1}{1 +
tu}$ sobre $\intcc{0}{1}$; a função é monótona em $ u $, então o
A soma de Riemann está entre colchetes pela integral deslocada por uma malha:
\[
\int_0^1 \frac{\dd u}{1 + tu} - \frac1n
\leq S_n \leq \int_0^1 \frac{\dd u}{1 + tu} + \frac1n
\]
(comparação das somas de Riemann de uma função monótona com seu
integral, válido para cada $ n $ com o seu próprio $ t = \ln n $). Agora
$\int_0^1 \frac{\dd u}{1 + tu} = \frac{\ln(1 + t)}{t}$, e
$\frac1n = o\bigl(\frac{\ln t}{t}\bigr)$: por isso
\[
S_n = \frac{\ln(1 + \ln n)}{\ln n} + O\Bigl(\frac 1n\Bigr)
\;\sim\; \frac{\ln\ln n}{\ln n} .
\]
\end{solution}

\begin{solution}{exo:b2:comparison:10}
Identidade: $(\ln n)^{\ln n} = \eu^{\ln n\,\ln\ln n} =
\bigl(\eu^{\ln n}\bigr)^{\ln\ln n} = n^{\ln\ln n}$. Classificação:
comparar logaritmos. $\ln(n^2) = 2\ln n $; $\ln\bigl((\ln
n)^{\ln n}\bigr) = \ln n\ln\ln n $; $\ln(2^n) = n\ln2$;
$\ln(n!) = n\ln n - n + O(\ln n)$ (Stirling, ou o mais bruto
colchetes $\ln n! \sim n\ln n $); $\ln(n^n) = n\ln n $. Desde
$2\ln n = o(\ln n\ln\ln n)$, $\ln n\ln\ln n = o(n)$, $ n\ln 2 =
o(n\ln n - n)$, e $ n \ln n - n \sim n\ln n $ mas $ n! / n^n \to
0$ (a diferença de logs é $-n + O(\ln n) \to -\infty $):
\[
n^2 = o\bigl((\ln n)^{\ln n}\bigr),\quad
(\ln n)^{\ln n} = o(2^n),\quad
2^n = o(n!),\quad
n! = o(n^n).
\]
(Para cada etapa: a diferença dos logaritmos tende a
$+\infty $, então a proporção tende a $0$.)
\end{solution}

\begin{solution}{exo:b2:comparison:11}
Decompor $\frac1{k^2} = \frac1{k(k+1)} + \frac1{k^2(k+1)}$ e
soma para $ k > n $:
\[
\sum_{k>n}\frac1{k^2} = \frac1{n+1} +
\sum_{k>n}\frac{1}{k^2(k+1)} ,
\]
a primeira soma telescópica exatamente ($\frac1{k(k+1)} = \frac1k -
\frac1{k+1}$). Para o segundo: $\frac{1}{k^2(k+1)} = \frac1{k^3}
+ O\bigl(\frac1{k^4}\bigr)$ (desde $\frac{1}{k^2(k+1)} -
\frac1{k^3} = \frac{-1}{k^3(k+1)}$) e pelo
comparação integral $\sum_{k>n}\frac1{k^3} = \frac1{2n^2} +
O\bigl(\frac1{n^3}\bigr)$, $\sum_{k>n}\frac1{k^4} =
O\bigl(\frac1{n^3}\bigr)$. Por isso
\[
\sum_{k>n}\frac1{k^2}
= \frac1{n+1} + \frac{1}{2n^2} + O\Bigl(\frac1{n^3}\Bigr)
= \frac1n - \frac1{n^2} + \frac{1}{2n^2} +
O\Bigl(\frac1{n^3}\Bigr)
= \frac1n - \frac{1}{2n^2} + O\Bigl(\frac1{n^3}\Bigr),
\]
usando $\frac1{n+1} = \frac1n - \frac1{n^2} +
O\bigl(\frac1{n^3}\bigr)$.
\end{solution}

\begin{solution}{exo:b2:comparison:12}
$(u_n)$ aumenta; se fosse limitado, convergiria para um
finito $\ell $ com $\ell = \ell + \eu^{-\ell}$: impossível. Então
$ u_n \to \infty $. Deixar $ v_n = \eu^{u_n} \to \infty $: então
\[
v_{n+1} = \eu^{u_n + \eu^{-u_n}} = v_n\,\eu^{1/v_n}
= v_n\Bigl(1 + \frac1{v_n} + \frac1{2v_n^2} +
O\bigl(v_n^{-3}\bigr)\Bigr)
= v_n + 1 + \frac{1}{2v_n} + O\bigl(v_n^{-2}\bigr).
\]
Resumindo $ v_{k+1} - v_k = 1 + O(1)$ primeiro dá $ v_n = n +
O(n)$, por isso $ v_n \geq cn $ eventualmente; resumindo com
$\frac1{2v_k} = O(\frac1k)$ dá $ v_n = n + O(\ln n)$. Mais um
passar: $\frac{1}{2v_k} = \frac{1}{2k}\bigl(1 +
O\bigl(\tfrac{\ln k}k\bigr)\bigr)$, então
\[
v_n = n + \sum_{k<n}\frac1{2k} + O(1) = n + \frac{\ln n}2 +
O(1).
\]
Finalmente $ u_n = \ln v_n = \ln n + \ln\Bigl(1 + \frac{\ln n}{2n} +
O\bigl(\tfrac1n\bigr)\Bigr) = \ln n + \frac{\ln n}{2n} +
O\bigl(\tfrac1n\bigr)$.
\end{solution}

\begin{solution}{pb:b2:comparison:1}
\textbf{1.} Subtraindo as duas expansões: $\sum_i (c_i -
c_i')\varphi_i = o(\varphi_k)$. Se algum coeficiente for diferente, deixe
$ i_0$ seja o primeiro: dividindo por $\varphi_{i_0}$ e usando
$\varphi_j = o(\varphi_{i_0})$ para $ j > i_0$ dá $ c_{i_0} -
c_{i_0}' = o(1)$: zero, contradição. Para a expansão: com
$ u = \frac{\ln x}x \to 0$,
\[
\frac{1}{x - \ln x} = \frac1x\cdot\frac{1}{1 - u}
= \frac1x\bigl(1 + u + u^2 + O(u^3)\bigr)
= \frac1x + \frac{\ln x}{x^2} + \frac{(\ln x)^2}{x^3} +
o\Bigl(\frac{(\ln x)^2}{x^3}\Bigr).
\]
Não $\frac c{x^2}$ termo aparece porque a expansão é uma
série geométrica em $ u = \frac{\ln x}{x}$: cada termo carrega como
muitos poderes de $\ln x $ a partir de $\frac1x $ além do primeiro; o
degrau de escala $\frac1{x^2}$ (coeficiente de $(\ln x)^0$) é simplesmente
ausente, com coeficiente $0$.

\textbf{2.} $ f(x) = \eu^x + x $ é \omterm{def:b2:metric:continuity}{contínuo}, estritamente
aumentando, com limites $-\infty $ e $+\infty $: uma bijeção
$\R \to \R $, então $ x_n = f^{-1}(n)$ existe e é único, e
$ x_n \to +\infty $ ($ f^{-1}$ aumenta para $+\infty $). De
$\eu^{x_n} = n - x_n $: $ x_n = \ln(n - x_n) \leq \ln n $, então
$ x_n/n \to 0$ e $ x_n = \ln n + \ln(1 - x_n/n) = \ln n + o(1)
\sim \ln n $.

\textbf{3.} Escrever $ u_n = x_n/n $. Segunda passagem: $ u_n = \frac{\ln n
+ o(1)}{n}$, então
\[
x_n = \ln n + \ln(1 - u_n) = \ln n - u_n + O(u_n^2)
= \ln n - \frac{\ln n}{n} + o\Bigl(\frac{\ln n}n\Bigr).
\]
Terceira passagem: agora $ u_n = \frac{\ln n}{n} - \frac{\ln n}{n^2} +
o\bigl(\frac{\ln n}{n^2}\bigr)$, e $\ln(1 - u_n) = -u_n -
\frac{u_n^2}2 + O(u_n^3)$:
\[
x_n = \ln n - \frac{\ln n}n + \frac{\ln n}{n^2}
- \frac{(\ln n)^2}{2n^2} + o\Bigl(\frac{(\ln
n)^2}{n^2}\Bigr)
= \ln n - \frac{\ln n}{n} - \frac{(\ln n)^2}{2n^2} +
o\Bigl(\frac{(\ln n)^2}{n^2}\Bigr),
\]
O termo $\frac{\ln n}{n^2}$ sendo absorvido em
$ o\bigl(\frac{(\ln n)^2}{n^2}\bigr)$.

\textbf{4.} No $ n = 1000$: $\ln 1000 \approx 6.90776$ (erro
$7\cdot10^{-3}$); dois termos: $6.90085$ (erro
$2\cdot10^{-5}$); três termos: $6.90082$ (erro abaixo
$10^{-5}$), contra $ x_{1000} \approx 6.90083$. Cada passe compra
aproximadamente o fator previsto $\frac{\ln n}{n}$.

\textbf{5.} Duas integrações por partes, começando pela direita:
com $\frac{\dd}{\dd t}\bigl[\tfrac12t(1-t)\bigr] = \tfrac12 -
t $ e $ t(1-t)$ desaparecendo em ambas as extremidades,
\[
\frac12\int_0^1 t(1-t)g''(t)\dd t
= -\int_0^1\Bigl(\frac12 - t\Bigr)g'(t)\dd t
= -\Bigl[\Bigl(\frac12 - t\Bigr)g\Bigr]_0^1 - \int_0^1 g
= \frac{g(0) + g(1)}2 - \int_0^1 g .
\]
Reorganizada, esta é a identidade declarada.

\textbf{6.} Calcule o incremento e aplique a questão 5 a
$ g(t) = f(n + t)$:
\begin{align*}
E_{n+1} - E_n
&= f(n{+}1) - \int_n^{n+1}\!f - \frac{f(n{+}1) - f(n)}2 \\
&= \frac{f(n) + f(n{+}1)}2 - \int_n^{n+1}\!f
= \frac12\int_0^1 t(1-t)f''(n+t)\dd t .
\end{align*}
Desde $0 \leq t(1-t) \leq \frac14$: $\abs{E_{n+1} - E_n} \leq
\frac18\int_n^{n+1}\abs{f''}$, cuja soma excede $ n $ converge por
hipótese: $(E_n)$ converge (incrementos absolutamente somáveis)
para alguns $ E $, com
\[
\abs{E - E_n} \leq \sum_{k\geq n}\abs{E_{k+1} - E_k} \leq
\frac18\int_n^\infty\abs{f''} .
\]

\textbf{7.} $ f(t) = \frac1t $: $ f''(t) = \frac2{t^3}$,
$\int_1^\infty\abs{f''} = 1 < \infty $. Pergunta 6:
\[
H_n = \ln n + \frac{1 + \frac1n}{2} + E + (E_n - E)
= \ln n + \Bigl(E + \frac12\Bigr) + \frac1{2n} +
\varepsilon_n,
\]
com $\abs{\varepsilon_n} = \abs{E_n - E} \leq
\frac18\int_n^\infty\frac{2\dd t}{t^3} = \frac1{8n^2}$.
Comparando com $ H_n = \ln n + \gamma + o(1)$
(\cref{ex:b2:comparison:harmonic}) identifica $ E + \frac12 =
\gamma $.

\textbf{8.} Da fórmula de incremento da pergunta 6,
\[
\varepsilon_n = E_n - E = -\sum_{k\geq n}\frac12\int_0^1
t(1-t)\,\frac{2\,\dd t}{(k+t)^3}
= -\sum_{k \geq n}\Bigl(\frac1{k^3}\int_0^1t(1-t)\dd t +
O\Bigl(\frac1{k^4}\Bigr)\Bigr),
\]
usando $\frac{1}{(k+t)^3} = \frac1{k^3} +
O\bigl(\frac1{k^4}\bigr)$ uniformemente para $ t \in \intcc01$. Com
$\int_0^1 t(1-t) = \frac16$ e
$\sum_{k\geq n}\frac1{k^3} \sim \frac{1}{2n^2}$
(\cref{thm:b2:comparison:seriesintegral}):
\[
\varepsilon_n = -\frac16\cdot\frac{1}{2n^2} +
o\Bigl(\frac1{n^2}\Bigr) = -\frac{1}{12n^2} +
o\Bigl(\frac{1}{n^2}\Bigr).
\]

\textbf{9.} $ f = \ln $: $ f''(t) = -\frac1{t^2}$, absolutamente
integrável. A questão 6 dá
\[
\ln n! = \int_1^n\ln t\,\dd t + \frac{\ln n}2 + E + O\Bigl(
\frac1{8}\int_n^\infty\frac{\dd t}{t^2}\Bigr)
= \Bigl(n + \frac12\Bigr)\ln n - n + 1 + E +
O\Bigl(\frac1n\Bigr),
\]
então $ d_n = 1 + E + O\bigl(\frac1n\bigr)$: convergência de $(d_n)$
--- Etapa 1 de \cref{thm:b2:comparison:stirling} --- mais o
taxa $ O(1/n)$. (O valor do limite de Stirling dá $ E =
\ln\sqrt{2\pi} - 1$.)

\textbf{10.} $ f(t) = t^{-1/2}$: $ f''(t) = \frac34 t^{-5/2}$,
absolutamente integrável. Pergunta 6:
\[
\sum_{k=1}^n \frac1{\sqrt k}
= 2\sqrt n - 2 + \frac{1 + \frac1{\sqrt n}}2 + E +
O\bigl(n^{-3/2}\bigr)
= 2\sqrt n + c + \frac{1}{2\sqrt n} +
O\bigl(n^{-3/2}\bigr),
\]
com $ c = E - \frac32$. No $ n = 10^4$: $2\sqrt n = 200$, $ c
\approx -1.46035$, $\frac1{2\sqrt n} = 0.005$: previsto
$198.54465$, e de fato $\sum_{k\leq10^4}k^{-1/2} =
198.544645\dots $ --- três termos, sete dígitos.

\textbf{11.} $ t \mapsto t\ln t $ é \omterm{def:b2:metric:continuity}{contínuo} e estritamente
aumentando em $\intco1\infty $ (derivado $\ln t + 1 \geq 1$),
de $0$ para $+\infty $: um único $ x_n $ existe, e $ x_n \to
\infty $ (outro $ x_n\ln x_n $ permaneceria limitado). Tomando
logaritmos em $ x_n\ln x_n = n $: $\ln x_n + \ln\ln x_n = \ln n $;
desde $\ln\ln x_n = o(\ln x_n)$, dividindo por $\ln x_n $ dá
$\frac{\ln n}{\ln x_n} \to 1$: $\ln x_n \sim \ln n $.

\textbf{12.} De $ x_n = \frac{n}{\ln x_n}$ e $\ln x_n \sim
\ln n $: $ x_n \sim \frac{n}{\ln n}$. Próxima passagem: $\ln\ln x_n =
\ln\bigl(\ln n\,(1 + o(1))\bigr) = \ln\ln n + o(1)$, então $\ln
x_n = \ln n - \ln\ln n + o(1)$ e
\[
x_n = \frac{n}{\ln n - \ln\ln n + o(1)}
= \frac{n}{\ln n}\cdot\frac{1}{1 - \frac{\ln\ln n +
o(1)}{\ln n}}
= \frac{n}{\ln n}\Bigl(1 + \frac{\ln\ln n}{\ln n} +
o\Bigl(\frac{\ln\ln n}{\ln n}\Bigr)\Bigr).
\]

\textbf{13.} No $ n = 10^6$: $\frac{n}{\ln n} \approx 72\,382$
(desligado por $18\%$), dois termos dão $\approx 86\,140$ (desligado por
$1.9\%$), contra a verdade $ x \approx 87\,848$. O ganho por
passar é apenas o fator $\frac{\ln\ln n}{\ln n} \approx
\frac{2.63}{13.8} \approx 0.19$: escalas logarítmicas convergem
com uma lentidão enlouquecedora --- um fato da vida onde quer que haja números primos
envolvido.

\textbf{14.} Existem exatamente $ n $ primos $\leq p_n $ (nomeadamente
$ p_1, \dots, p_n $): $\pi(p_n) = n $. O teorema dos números primos
(admitido; volume do ano 3) dá $ n = \pi(p_n) \sim
\frac{p_n}{\ln p_n}$, ou seja\ $ p_n \sim n\ln p_n $: este é o
equação $ x\ln x \approx n $ leia de trás para frente. Tomando logaritmos:
$\ln p_n = \ln n + \ln\ln p_n + o(1)$, e $\ln\ln p_n =
o(\ln p_n)$ forças $\ln p_n \sim \ln n $ como na questão 11.
Substituindo de volta:
\[
p_n \sim n\ln p_n = n\,\ln n\,\frac{\ln p_n}{\ln n} \sim n\ln
n .
\]

\textbf{15.} (a) Correção $\varepsilon > 0$; para $ k $ grande, $(1 -
\varepsilon)k\ln k \leq p_k \leq (1 + \varepsilon)k\ln k $. Por
comparação com o aumento $ t\ln t $
(\cref{thm:b2:comparison:seriesintegral}-tipo bracketing),
$\sum_{k\leq n}k\ln k = \int_1^n t\ln t\,\dd t + O(n\ln n) =
\frac{n^2\ln n}2 - \frac{n^2}4 + O(n\ln n) \sim
\frac{n^2\ln n}2$. Por isso $\sum_{k\leq n}p_k = \frac{n^2\ln
n}{2}(1 + O(\varepsilon) + o(1))$ para cada $\varepsilon $:
$\sum_{k\leq n}p_k \sim \frac{n^2\ln n}2$. (b) Pelo primo
teorema dos números, entre os inteiros até $10^{100}$ um
proporção $\sim \frac{1}{\ln 10^{100}} = \frac1{230.26\dots}$
são primos: um uniformemente aleatório $100$-dígito inteiro é primo
com probabilidade sobre $\frac1{230}$.

\textbf{16.} Sobre $\intoo{n\pi}{n\pi + \frac\pi2}$, $ g(x) = \tan
x - \frac1x $ é \omterm{def:b2:metric:continuity}{contínuo} e aumentando estritamente ($ g' = 1 +
\tan^2x + \frac1{x^2} > 0$), com $ g \to -\frac1{n\pi} < 0$ em
a extremidade esquerda e $ g \to +\infty $ à direita: exatamente um
raiz $ x_n $. Desde $\tan z_n = \tan x_n = \frac1{x_n} \to 0$
com $ z_n \in \intoo{0}{\frac\pi2}$: $ z_n \to 0^+$.

\textbf{17.} $\tan z_n \sim z_n $ e $\frac1{x_n} \sim
\frac1{n\pi}$: $ z_n \sim \frac1{n\pi}$.

\textbf{18.} $ z_n = \arctan\frac1{x_n}$ e $\arctan u = u +
O(u^3)$. Com $ z_n = O(\frac1n)$:
\[
\frac1{x_n} = \frac{1}{n\pi}\cdot\frac1{1 + \frac{z_n}{n\pi}}
= \frac1{n\pi} - \frac{z_n}{n^2\pi^2} +
O\Bigl(\frac1{n^4}\Bigr)
= \frac1{n\pi} + O\Bigl(\frac1{n^3}\Bigr),
\]
então $ z_n = \frac1{n\pi} + O\bigl(\frac1{n^3}\bigr)$: o
$\frac{c}{n^2}$ degrau carrega coeficiente $0$, porque o
primeira correção para $\frac1{x_n}$ é em si de tamanho
$\frac{z_n}{n^2} = O(n^{-3})$.

\textbf{19.} Inserir $ z_n = \frac1{n\pi} + O(n^{-3})$ para dentro do
exibição anterior:
\[
\frac{1}{x_n} = \frac{1}{n\pi} - \frac{1}{n^3\pi^3} +
O\Bigl(\frac1{n^5}\Bigr),
\]
então $ z_n = \arctan\frac1{x_n} = \frac1{x_n} -
\frac{1}{3}\Bigl(\frac1{x_n}\Bigr)^3 + O\Bigl(\frac1{n^5}\Bigr)
= \frac1{n\pi} - \frac{1}{n^3\pi^3} - \frac{1}{3n^3\pi^3} +
O\Bigl(\frac1{n^5}\Bigr)$:
\[
x_n = n\pi + \frac{1}{n\pi} - \frac{4}{3\pi^3n^3} +
O\Bigl(\frac1{n^5}\Bigr).
\]

\textbf{20.} No $ n = 3$: um termo $9.53088$, três termos
$9.52929$, raiz verdadeira $9.52933$: erros $1.5\cdot10^{-3}$ e
$5\cdot10^{-5}$. Contraste: para $\tan x = x $ a raiz deve fazer
$\tan $ enorme, então ele abraça o \emph{certo} fim $ n\pi +
\frac\pi2$ da janela, à distância $\sim\frac1{n\pi}$
antes da assíntota; para $ x\tan x = 1$ a raiz deve fazer
$\tan $ minúsculo, então fica logo depois do \emph{esquerda} fim $ n\pi $,
à distância $\sim\frac1{n\pi}$ depois do zero. Mesmo método,
geografia do espelho.

\textbf{21.} $\sin u < u $ sobre $\intoo0\pi $ e $\sin $ aplicações
$\intoo0\pi $ em $\intoc01 \subseteq \intoo0\pi $: depois de um
passo $ u_1 \in \intoc{0}{1}$, então $(u_n)$ diminui e é
limitado abaixo por $0$: converge para um ponto fixo de $\sin $,
ou seja,\ para $0$. Expansão: $\sin u = u(1 - \frac{u^2}6 +
o(u^2))$, então
\[
\frac{1}{u_{n+1}^2} - \frac1{u_n^2}
= \frac{1}{u_n^2}\Bigl(\bigl(1 - \tfrac{u_n^2}6 +
o(u_n^2)\bigr)^{-2} - 1\Bigr)
= \frac{1}{u_n^2}\Bigl(\frac{u_n^2}{3} + o(u_n^2)\Bigr)
\longrightarrow \frac13 .
\]

\textbf{22.} Por Cesàro (volume do ano 1), a média do
incrementos convergem para o mesmo limite:
\[
\frac{1}{n}\cdot\frac{1}{u_n^2}
= \frac1n\Bigl(\frac1{u_0^2} + \sum_{k=0}^{n-1}
\Bigl(\frac1{u_{k+1}^2} - \frac1{u_k^2}\Bigr)\Bigr)
\longrightarrow \frac13 ,
\]
então $ u_n^2 \sim \frac3n $ e, sendo todos os termos positivos, $ u_n
\sim \sqrt{3/n}$.

\textbf{23.} Pela pergunta 7, $\abs{H_n - \ln n - \gamma -
\frac1{2n}} \leq \frac1{8n^2}$. No $ n = 10^6$ esse limite é
$\frac{1}{8\cdot10^{12}} = 1.25\cdot10^{-13}$: três computados
termos entregam a soma harmônica de um milhão de termos para treze
dígitos, com um certificado de erro totalmente rigoroso --- o todo
ponto de uma fórmula assintótica com resto explícito.

\textbf{24.} (a) Verdadeiro: $\ln u_n - \ln v_n = \ln\frac{u_n}{v_n}
\to 0$ enquanto $\ln v_n \to +\infty $, então a proporção de logaritmos
tende a $1$. (b) Falso: $ u_n = n + 1 \sim v_n = n $, mas
$\eu^{u_n}/\eu^{v_n} = \eu \neq 1$. Equivalência tolera
erros aditivos $ o(1)$ no expoente, não $ O(1)$. (c) Falso:
$ f(x) = x + \sin(x^2) \sim g(x) = x $ no $+\infty $, mas $ f'(x)
= 1 + 2x\cos(x^2)$ oscila ilimitadamente enquanto $ g' = 1$:
derivadas de funções equivalentes não precisam ser comparáveis em
tudo.

\textbf{25.} O \omterm{def:b2:structures:sn}{ciclo} de \cref{met:b2:comparison:implicit} correu
de forma idêntica três vezes: localizar a raiz, extrair um bruto
prazo, envie-o de volta para o próximo pedido --- em $\eu^x + x = n $
(Parte I), em $ x\ln x = n $ (Parte III), em $ x\tan x = 1$ (Parte
IV). A correção trapezoidal atualiza a série - integral
comparação de “a diferença converge” para uma expressão explícita
$\frac{f(1) + f(n)}2$ prazo com um certificado $ O(\int_n^\infty
\abs{f''})$ restante --- constantes e barras de erro em vez de
mera convergência. A ponte para os números primos é pura inversão: o
teorema dos números primos diz $\pi(x)\ln x \approx x $, então $ p_n $,
definido por $\pi(p_n) = n $, resolve um $ x\ln x = n $ equação ---
e herda seus assintóticos. Regra (a) da questão 24
legitimou cada passagem de $ u_n \sim v_n $ a $\ln u_n \sim
\ln v_n $ (questões 11, 14); a falsidade de (b) é a razão pela qual
nunca exponenciar equivalências. Cúpulas: o Euler-Maclaurin
fórmula de primeira ordem (questão 6), e a lei assintótica
$ p_n \sim n\ln n $ do $ n$-ª linha (questão 14).
