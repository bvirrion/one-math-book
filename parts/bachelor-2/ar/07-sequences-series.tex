\chapter{المتتاليات والمتسلسلات}\label{ch:b2:series}

تنضج نظرية المتسلسلات العددية (مجلد السنة الأولى) هنا في ثلاثة
اتجاهات: المتسلسلات ذوات القيم في فضاءات باناخ، حيث ينهض
التقارب المطلق بالعمل؛ والمحكّات الأدقّ للمتسلسلات الحقيقية (\omterm{thm:b2:series:abel}{جمع
آبل})؛ و\emph{العائلات \omterm{def:b2:series:summable}{القابلة للجمع}} --- أي الجمع محرَّرًا من
ترتيب الحدود --- مع مبرهنة فوبيني للمجاميع المزدوجة \omterm{thm:b2:series:fubini}{وجداء كوشي}.
وتحمل هذه الأدوات كل فصول متسلسلات الدوال الآتية.

\section{المتسلسلات في الفضاءات المعيارية}

\begin{definition}\label{def:b2:series:def}
من أجل متتالية $(u_n)$ في فضاء معياري $E$، تتقارب المتسلسلة $\sum u_n$
حين تتقارب مجاميعها الجزئية؛ وتتقارب
\emph{بإطلاق}\index{تقارب مطلق!في فضاء باناخ} حين يكون
$\sum \norm{u_n} < \infty$. وفي فضاء \emph{باناخ}، يستلزم التقارب
المطلق التقاربَ (\cref{thm:b2:nvs:absoluteconvergence})؛
وقد يفشل ذلك في فضاء غير تام
(\cref{exo:b2:series:9}).
\end{definition}

\begin{example}\label{ex:b2:series:neumann}
في $\mathcal{M}_n(K)$ (أو في $\mathcal{L}_c(E)$، حيث $E$ \omterm{def:b2:nvs:banach}{فضاء باناخ}): من أجل
$\vertiii A < 1$، تتقارب \emph{متسلسلة نويمان} $\sum A^k$
\omterm{def:b2:series:def}{بإطلاق} إلى $(I - A)^{-1}$ (وقد بُرهن على ذلك في \cref{exo:b2:nvs:5})؛
وتتقارب $\sum \frac{A^k}{k!}$ \omterm{def:b2:series:def}{بإطلاق} إلى $\eu^A$ من أجل كل $A$
(\cref{ex:b2:nvs:matrixexp}). فالمتسلسلتان الهندسية والأسّية
ذواتا القيم المؤثِّرة تسلكان مثل نموذجيهما العدديين --- وهو كل
مغزى إطار باناخ.
\end{example}

\section{جمع آبل}

\begin{theorem}[جمع آبل ومحكّه]\label{thm:b2:series:abel}
(الجمع بالتجزئة) من أجل أعداد $a_n$ ومتجهات $b_n$، مع $B_n
= \sum_{k=0}^{n} b_k$:\index{جمع آبل}
\[
\sum_{n=0}^{N} a_n b_n
= a_N B_N - \sum_{n=0}^{N-1} (a_{n+1} - a_n) B_n .
\]
(محك آبل) إذا كانت $(a_n)$ متتالية حقيقية متناقصة إلى $0$، وكانت
المجاميع الجزئية $B_n$ \emph{محدودة} (في \omterm{def:b2:nvs:banach}{فضاء باناخ})، فإن
$\sum a_n b_n$ تتقارب.
\end{theorem}

\begin{proof}
المتطابقة، خطوةً خطوة: مع $B_{-1} = 0$، نكتب $b_n = B_n -
B_{n-1}$ ونفصل،
\[
\sum_{n=0}^{N} a_nb_n
= \sum_{n=0}^{N} a_nB_n - \sum_{n=0}^{N} a_nB_{n-1}
= \sum_{n=0}^{N} a_nB_n - \sum_{n=0}^{N-1} a_{n+1}B_{n} ,
\]
ونعيد ترقيم المجموع الثاني بالدليل $n \mapsto n + 1$ (فالحدّ $B_{-1}$
ينعدم)؛ وبجمع المجال المشترك $0 \leq n \leq N-1$
يبقى $a_NB_N$ مضافًا إليه $\sum_{n\leq N-1}(a_n - a_{n+1})B_n$: وهي
الصيغة المذكورة. وهي المكاملة بالتجزئة المتقطعة، مع
$(B_n)$ بوصفه دالة أصلية للمقدار $(b_n)$، والفرق $a_{n+1}
- a_n$ بوصفه مشتقًا للمقدار $(a_n)$. ومن أجل المحك، مع $\norm{B_n}
\leq M$: يكون
الحدّ الحدّي $a_N B_N \to 0$؛ وتتقارب المتسلسلة $\sum (a_n -
a_{n+1})B_n$ \omterm{def:b2:series:def}{بإطلاق}، لأن
\[
\sum_n \norm{(a_{n+1} - a_n)B_n} \leq M \sum_n (a_n - a_{n+1})
= M a_0 < \infty
\]
(بالتلسكوب، مع $a_n \downarrow 0$). فيتقارب جزءا المتطابقة كلاهما،
ومن ثم تتقارب $\sum a_n b_n$.
\end{proof}

\begin{example}\label{ex:b2:series:sinn}
$\sum \frac{\sin n}{n}$ تتقارب: فلدينا $a_n = \frac1n \downarrow 0$ و
$B_n = \sum_{k=1}^{n} \sin k$ محدودة --- بل إنها $B_n =
\Im\sum_{k \leq n} \eu^{\iu k} = \Im\,\frac{\eu^{\iu}(\eu^{\iu n} -
1)}{\eu^{\iu} - 1}$، وطويلتها $\leq \frac{2}{\abs{\eu^{\iu} - 1}}$.
و\emph{لا} تتقارب \omterm{def:b2:series:def}{بإطلاق} (إذ $\abs{\sin n} \geq \sin^2 n
= \frac{1 - \cos 2n}{2}$، و$\sum \frac{1 - \cos 2n}{2n}$ متباعدة
لأن $\sum \frac{\cos 2n}{n}$ متقاربة بمحك آبل نفسه في حين أن
$\sum \frac{1}{2n}$ متباعدة). ومحك المتسلسلات المتناوبة هو
الحالة الخاصة $b_n = (-1)^n$.
\end{example}

\begin{example}[آبل على دائرة التقارب]\label{ex:b2:series:abelboundary}
من أجل أي أعداد عقدية $z$ بشرط $\abs z = 1$ تتقارب $\sum_{n \geq 1}
\frac{z^n}{n}$؟ عند $z = 1$ تكون المتسلسلة التوافقية:
فتتباعد. ومن أجل $z \neq 1$ على الدائرة، ينطبق محك آبل
مع $a_n = \frac1n \downarrow 0$ ومع $b_n = z^n$، التي تكون مجاميعها
الجزئية محدودة بغض النظر عن $N$:
\[
\Bigl|\sum_{n=1}^{N} z^n\Bigr|
= \Bigl|\frac{z(z^N - 1)}{z - 1}\Bigr|
\leq \frac{2}{\abs{z - 1}} .
\]
فهي متقاربة --- وإن لم تكن \omterm{def:b2:series:def}{بإطلاق} أبدًا ($\sum\frac1n$). متسلسلة
واحدة ودائرة من السلوكات: تباعد عند نقطة واحدة،
وتقارب نصفي في كل مكان آخر. وهذا هو السلوك الحدّي القياسي
لمتسلسلات القوى (\cref{ch:b2:powerseries})، ملقىً هنا
باليدين المجرّدتين؛ وعند $z = -1$ نستعيد المتسلسلة
التوافقية المتناوبة، وعند $z = \eu^{\iu\theta}$ يكون جزآها الحقيقي
والتخيلي هما المتسلسلتين $\sum\frac{\cos
n\theta}{n}$ و$\sum\frac{\sin n\theta}{n}$ في
\cref{exo:b2:series:4}.
\end{example}

\begin{example}[متسلسلة متناوبة ملغومة]\label{ex:b2:series:trap}
هل تتقارب $\sum_{n\geq2} \dfrac{(-1)^n}{\sqrt n + (-1)^n}$؟
الإشارات تتناوب والحدود تؤول إلى $0$ --- ومع ذلك \emph{لا}
ينطبق محك المتناوبات: فالطوائل
$\frac{1}{\sqrt n + (-1)^n}$ ليست متناقصة (إذ تقفز إلى أعلى عند
كل $n$ فردي). فننشر بدل ذلك:
\[
\frac{(-1)^n}{\sqrt n + (-1)^n}
= \frac{(-1)^n}{\sqrt n}\cdot
\frac{1}{1 + \frac{(-1)^n}{\sqrt n}}
= \frac{(-1)^n}{\sqrt n} - \frac{1}{n}
+ O\Bigl(\frac{1}{n^{3/2}}\Bigr).
\]
الجزء الأول يتقارب (بمحك المتناوبات، مطبَّقًا بأمانة على
$\frac1{\sqrt n}\downarrow0$)، والثالث يتقارب \omterm{def:b2:series:def}{بإطلاق}
--- لكن الجزء الأوسط هو المتسلسلة التوافقية المتباعدة: ومن ثم
\emph{يتباعد} المجموع إلى $-\infty$. والفكرة الختامية: حين تفشل
الرتابة، انشر إلى أن يصير كل جزء إما
متقاربًا \omterm{def:b2:series:def}{بإطلاق} وإما حالةَ محك نظيفة؛ فالمقدار $-\frac1n$
المختبئ غير مرئي لعدّ الإشارات.
\end{example}

\begin{remark}[مزالق شائعة]\label{rem:b2:series:pitfalls}
(1) قولُ ``الحدود تؤول إلى $0$'' لا يبرهن على شيء: فالمتسلسلة التوافقية
تتباعد. (2) يقتضي محك المتناوبات طوائل \emph{متناقصة}
--- والمتسلسلة \cref{ex:b2:series:trap} هي المثال المضاد
القانوني، والمتسلسلة الثالثة في \cref{exo:b2:series:1} هي
التمرين. (3) لا يجوز إعادة ترتيب المتسلسلات المتقاربة شرطيًا
(\cref{ex:b2:series:rearrange})، وقد تتباعد جداءاتها
الكوشية: فمن أجل مربّع $\sum\frac{(-1)^n}{\sqrt{n+1}}$،
تحقق الحدود القطرية
\[
\abs{c_k} = \sum_{m=0}^{k}
\frac{1}{\sqrt{(m+1)(k-m+1)}}
\geq (k+1)\cdot\frac{2}{k+2} \longrightarrow 2 \neq 0
\]
(فكل عامل لا يتجاوز $\frac{k+2}2$ بمتراجحة المتوسطين)، ومن ثم تتباعد $\sum c_k$
--- فالتقارب المطلق لأحد العاملين على الأقل
(\cref{exo:b2:series:8}) ليس ترفًا. (4) القابلية للجمع
تتعلق بحدود \emph{مطلقة} بحكم التعريف: فلا وجود
\omterm{def:b2:series:summable}{لعائلة قابلة للجمع} شرطيًا.
\end{remark}

\section{العائلات القابلة للجمع}

\begin{definition}\label{def:b2:series:summable}
لتكن $I$ مجموعة أدلة \omterm{def:b2:structures:countable}{قابلة للعد}. تكون عائلة $(u_i)_{i \in I}$ من
\emph{الأعداد الحقيقية غير السالبة} \emph{قابلة للجمع}\index{عائلة قابلة للجمع}
إذا كانت مجاميعها الجزئية المنتهية محدودة؛ ويكون مجموعها
\[
\sum_{i \in I} u_i = \sup_{F \subseteq I \text{ منته}} \sum_{i
\in F} u_i \in \intcc{0}{+\infty} .
\]
وتكون عائلة من الأعداد الحقيقية أو العقدية (أو من متجهات باناخ) قابلة للجمع إذا كانت
$(\norm{u_i})$ كذلك؛ ويُعرَّف مجموعها عندئذٍ بالفصل إلى
الجزأين الموجب والسالب (أو الحقيقي والتخيلي) --- وبكيفية مكافئة، بوصفه
القيمة المشتركة للمقدار $\sum_{n} u_{\sigma(n)}$ على كل ترقيمات
$\sigma$ للمجموعة $I$ (انظر أدناه).
\end{definition}

\begin{method}[كيف نختار محكًّا]\label{met:b2:series:choosing}
أمام $\sum u_n$، وبهذا الترتيب: (1) إذا كان $u_n \not\to 0$، فتباعد،
وتوقّف. (2) إذا كانت الحدود ثابتة الإشارة، فقارِن: جد
مكافئًا (\cref{ch:b2:comparison}) وضعه على
خريطة ريمان--برتران. (3) إذا تناوبت الإشارات بطوائل
\emph{متناقصة}، فمحك المتناوبات؛ وإذا لم تكن الطوائل
رتيبة، فانشر الحدّ إلى أن يصير كل جزء متقاربًا \omterm{def:b2:series:def}{بإطلاق}
أو حالةَ محك نظيفة (\cref{ex:b2:series:trap}). (4)
إذا كان نمط الإشارات متذبذبًا لكنه منظَّم ($\sin
n\theta$، $\eu^{\iu n\theta}$، أو قوى مصفوفات)، فمحك آبل بمجاميع
جزئية محدودة. (5) والتقارب المطلق جدير بالفحص أولًا
دائمًا: فهو أقوى، ولا يتأثر بالترتيب، ويفتح جداءات كوشي
وفوبيني.
\end{method}

\begin{example}[القابلية للجمع بعدّ الأقطار]\label{ex:b2:series:diagonalcount}
من أجل أي $s > 0$ تكون العائلة
$\bigl((m + n)^{-s}\bigr)_{m, n \geq 1}$ \omterm{def:b2:series:summable}{قابلة للجمع}؟ نجمّع
المجاميع الجزئية المنتهية حسب الأقطار $m + n = k$: فالقطر $k$
يحمل $k - 1$ زوجًا، ويسهم كلٌّ منها بالمقدار $k^{-s}$، ومن ثم تُحدّ
المجاميع المنتهية بالضبط (وتستنفد)
\[
\sum_{k \geq 2} \frac{k - 1}{k^{s}} ,
\]
وهي متسلسلة ذات حدود موجبة مكافئة للمقدار $k^{1-s}$: فهي \omterm{def:b2:series:summable}{قابلة للجمع}
إذا وفقط إذا $s - 1 > 1$، أي $s > 2$. فالدليل ذو البعدين يلتهم
قوة كاملة: أي إن مستويًا من الحدود ``أكثر تباعدًا ببعد
واحد'' من مستقيم --- فهندسة عدّ مجموعة الأدلة، لا حجم
الحدود المفردة، هي التي تقرّر القابلية للجمع.
(ويبيّن الإحصاء نفسه أن $\bigl((m^2 + n^2)^{-1}\bigr)$ ليست
\omterm{def:b2:series:summable}{قابلة للجمع}: فعلى القطر $m + n = k$، كل حدّ لا يقلّ عن
$k^{-2}$، و$(k-1)\cdot k^{-2}$ يُجمع مثل المتسلسلة
التوافقية.)
\end{example}

\begin{theorem}[القابلية للجمع والترتيب]\label{thm:b2:series:rearrangement}
\begin{enumerate}
  \item من أجل العائلات غير السالبة، لا يتغير المجموع تحت أي
        ترقيم: $\sum_{i} u_i = \sum_{n=0}^{\infty}
        u_{\sigma(n)}$ من أجل كل تقابل $\sigma \colon \N \to I$.
  \item تكون متسلسلة حقيقية أو عقدية $\sum u_n$ \emph{متقاربة
        تبديليًا} (أي إن كل إعادة ترتيب تتقارب، بالمجموع نفسه)
        إذا وفقط إذا كانت متقاربة \omterm{def:b2:series:def}{بإطلاق}.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) كل مجموع جزئي $\sum_{n \leq N} u_{\sigma(n)}$ هو مجموع جزئي
منته للعائلة (ومن ثم فهو $\leq$ النهاية العليا)؛ وكل مجموعة منتهية $F$
محتواة في $\{\sigma(0), \dots, \sigma(N)\}$ ما (ومن ثم فالنهاية العليا
$\leq$ نهاية المتسلسلة). فيتطابق الحدّان.

(2) إذا كان $\sum\abs{u_n} < \infty$: فمن أجل أي إعادة ترتيب $\sigma$ وأي
$\varepsilon > 0$، نختار $N$ بحيث $\sum_{n > N}\abs{u_n} \leq
\varepsilon$؛ وبعد الرتبة التي يكون فيها $\sigma$ قد استنفد
$\intint{0}{N}$، تختلف المجاميع الجزئية المعاد ترتيبها عن
النهاية الأصلية بمقدار لا يتجاوز $\varepsilon$: أي المجموع نفسه. وإذا كان $\sum
\abs{u_n} = \infty$ مع تقارب $\sum u_n$ (في الحالة الحقيقية؛ والعقدية
تتبع إحداثيةً إحداثية): فإن الجزأين الموجب والسالب يتباعدان
معًا، ويمكن إعادة الترتيب لبلوغ أي نهاية مفروضة
--- وهي مبرهنة ريمان، المنفَّذة في \cref{exo:b2:series:5} ---
ومن ثم يفشل التقارب التبديلي.
\end{proof}

\begin{example}[إعادة ترتيب متلبّسة بالجرم]\label{ex:b2:series:rearrange}
مجموع المتسلسلة التوافقية المتناوبة هو
$\sum_{n\geq1}\frac{(-1)^{n-1}}{n} = \ln 2$ (مجلد السنة
الأولى). ولنعد ترتيبها على صورة ``موجب واحد، ثم سالبان'':
\[
1 - \frac12 - \frac14 + \frac13 - \frac16 - \frac18 + \frac15 -
\cdots
\]
وبتجميع كل كتلة من ثلاثة،
\[
\frac{1}{2k-1} - \frac{1}{4k-2} - \frac1{4k}
= \frac{1}{4k-2} - \frac{1}{4k}
= \frac12\Bigl(\frac{1}{2k-1} - \frac1{2k}\Bigr),
\]
ومن ثم تتقارب المتسلسلة المعاد ترتيبها إلى $\frac12\ln 2$ --- أي نصف
المجموع الأصلي، بالحدود نفسها بالضبط. فالمتسلسلات المتقاربة
لا \omterm{def:b2:series:def}{بإطلاق} تتذكّر ترتيب حدودها؛ والعائلات \omterm{def:b2:series:summable}{القابلة للجمع} هي
بالضبط تلك التي لا تتذكّره.
\end{example}

\begin{example}[التجميع آمن، وفكّ التجميع ليس كذلك]\label{ex:b2:series:grouping}
لا يغيّر تجميع الحدود المتتالية لمتسلسلة \emph{متقاربة}
المجموعَ أبدًا: فالمجاميع الجزئية المجمَّعة متتالية جزئية من
الأصلية. أما العملية العكسية فممنوعة:
\[
(1 - 1) + (1 - 1) + (1 - 1) + \cdots = 0 + 0 + \cdots = 0,
\]
ومع ذلك تتباعد $1 - 1 + 1 - 1 + \cdots$ بعد فكّ التجميع (فمجاميعها
الجزئية تتذبذب بين $1$ و$0$). ولا يكون فكّ التجميع مشروعًا
إلا بفرضية معوِّضة --- مثل أن تؤول الحدود إلى $0$
مع أطوال كتل محدودة: فبين مجموعين جزئيين مجمَّعين تنجرف
المجاميع الأصلية بمقدار لا يتجاوز مجموع عدد محدود من الحدود
$o(1)$، فينتقل التقارب رجوعًا.
وهذا بالضبط هو الشرط الذي يجعل حساب الكتل في
\cref{ex:b2:series:rearrange} برهانًا لا خفّة يد.
\end{example}

\begin{theorem}[فوبيني للعائلات؛ جداءات كوشي]\label{thm:b2:series:fubini}
لتكن $(u_{m,n})_{(m,n) \in \N^2}$ عائلة مزدوجة \omterm{def:b2:series:summable}{قابلة للجمع} (أي
$\sup_F \sum_F \abs{u_{m,n}} < \infty$). عندئذٍ\index{مبرهنة فوبيني
للمتسلسلات}
\[
\sum_{(m,n)} u_{m,n}
= \sum_{m=0}^{\infty}\Bigl(\sum_{n=0}^{\infty} u_{m,n}\Bigr)
= \sum_{n=0}^{\infty}\Bigl(\sum_{m=0}^{\infty} u_{m,n}\Bigr)
= \sum_{k=0}^{\infty}\Bigl(\sum_{m+n=k} u_{m,n}\Bigr),
\]
مع تقارب كل المتسلسلات الداخلية (\omterm{def:b2:series:def}{بإطلاق}). وعلى الخصوص، إذا تقاربت $\sum
a_m$ و$\sum b_n$ \omterm{def:b2:series:def}{بإطلاق}، فإن \emph{جداءهما
الكوشي}\index{جداء كوشي} يتقارب \omterm{def:b2:series:def}{بإطلاق} مع
\[
\Bigl(\sum_m a_m\Bigr)\Bigl(\sum_n b_n\Bigr)
= \sum_{k=0}^{\infty} c_k,
\qquad
c_k = \sum_{m=0}^{k} a_m b_{k-m} .
\]
\end{theorem}

\begin{proof}
\emph{الحالة غير السالبة.} يحسب كل تجميع (بالأسطر أو الأعمدة أو
الأقطار) النهاية العليا نفسها: فأي مجموعة منتهية من الأزواج
محتواة في كتلة منتهية من الأسطر (فيُحدّ كل مجموع مجمَّع من أسفل
بالمجاميع الجزئية المنتهية ومن أعلى بالمجموع الكلي)، ويقوم التقارب
الرتيب للمجاميع الجزئية بالباقي --- وبالتحديد، من أجل الأسطر:
يعطي $\sum_{m \leq M}\sum_{n \leq N} u_{m,n} \leq S$، بأخذ $N
\to \infty$ ثم $M \to \infty$، أن $\sum_m \sum_n u_{m,n} \leq S$؛
وعكسيًا تقع كل مجموعة منتهية $F$ في مستطيل كهذا، ومنه $S \leq
\sum_m\sum_n u_{m,n}$. وأما الأقطار: فبالحدّين نفسيهما مع مثلثات
بدل المستطيلات.

\emph{الحالة العامة.} نفصل إلى الجزأين الموجب والسالب (أو الحقيقي
والتخيلي)، وكلٌّ منهما عائلة غير سالبة \omterm{def:b2:series:summable}{قابلة للجمع}؛ وتتفق التجميعات
الأربعة على كل جزء، ومن ثم على الفرق؛ ويأتي التقارب المطلق
للمتسلسلات الداخلية من الحالة غير السالبة
مطبَّقةً على $\abs{u_{m,n}}$.

\emph{\omterm{thm:b2:series:fubini}{جداء كوشي}.} العائلة $u_{m,n} = a_m b_n$ \omterm{def:b2:series:summable}{قابلة للجمع}: فالمجاميع
الجزئية المنتهية للمقدار $\abs{a_mb_n}$ محدودة بالمقدار
$\bigl(\sum\abs{a_m}\bigr)\bigl(\sum\abs{b_n}\bigr)$. وتعطي الأسطر
$\bigl(\sum a_m\bigr)\bigl(\sum b_n\bigr)$؛ وتعطي الأقطار $\sum_k
c_k$.
\end{proof}

\begin{example}[متطابقة الأسّي، بأمانة]\label{ex:b2:series:exp}
من أجل $a, b \in \C$ (أو مصفوفتين متبادلتين):
\[
\Bigl(\sum_m \frac{a^m}{m!}\Bigr)\Bigl(\sum_n
\frac{b^n}{n!}\Bigr)
= \sum_k \sum_{m+n=k} \frac{a^m b^n}{m!\,n!}
= \sum_k \frac{(a + b)^k}{k!},
\]
بمبرهنة ثنائي الحدّ على كل قطر: $\eu^a \eu^b = \eu^{a+b}$
--- أي المعادلة الوظيفية للأسّي $\exp$ مستنبَطةً من المتسلسلة وحدها.
(ويُستعمل التبادل في خطوة ثنائي الحدّ؛ ومن أجل المصفوفات غير
المتبادلة تفشل المتطابقة فعلًا،
\cref{ch:b2:diffeq}.)
\end{example}

\begin{example}[جداءات كوشي بوصفها أداة حساب]\label{ex:b2:series:squaresum}
من المتسلسلة الهندسية ومن المقدار $\sum_{n
\geq 1} nz^n = \frac{z}{(1-z)^2}$ في \cref{exo:b2:series:2} ($\abs z < 1$)، يُنهي \omterm{thm:b2:series:fubini}{جداء
كوشي} إضافي العزمَ الثاني. نضرب $\sum_m
mz^m$ في $\sum_n z^n$: فالمعامل القطري هو $\sum_{m=0}^k
m = \frac{k(k+1)}2$، ومنه
\[
\frac{z}{(1-z)^3} = \sum_{k\geq0}\frac{k(k+1)}{2}\,z^k ,
\]
وتُركّب المتطابقة $n^2 = 2\cdot\frac{n(n+1)}2 - n$ المقدارَ
\[
\sum_{n\geq1} n^2z^n = \frac{2z}{(1-z)^3} - \frac{z}{(1-z)^2}
= \frac{z(1+z)}{(1-z)^3} .
\]
وعند $z = \frac12$: $\sum_{n\geq1}\frac{n^2}{2^n} =
\frac{\frac12\cdot\frac32}{\frac18} = 6$ --- أي قيمة مغلقة بلا
أي اشتقاق، بمجرد ضرب متسلسلات متقاربة \omterm{def:b2:series:def}{بإطلاق}
كما تُضرب كثيرات الحدود. ويحسب تلسكوب المتطابقات نفسه
كل مقدار $\sum n^dz^n$، وسيتعرّف أهل الاحتمال على
العزم العاملي الثاني للقانون الهندسي
(\cref{ch:b2:genfun}).
\end{example}

\begin{example}[تقييم مجموع مزدوج]\label{ex:b2:series:doublesum}
من أجل $s > 1$ حقيقي، لتكن $\zeta(s) = \sum_{n\geq1} n^{-s}$. وبعدّ
القواسم بالجمع المزدوج --- فالعائلة $(m^{-s}n^{-s})$ على
$(m,n) \in (\N^*)^2$ \omterm{def:b2:series:summable}{قابلة للجمع} (وهي جداء متسلسلتين موجبتين متقاربتين)
--- وبالتجميع حسب الجداء $q = mn$:
\[
\zeta(s)^2 = \sum_{m,n} \frac{1}{(mn)^s}
= \sum_{q=1}^{\infty} \frac{d(q)}{q^s},
\]
حيث $d(q)$ عدد قواسم $q$. فالعائلات \omterm{def:b2:series:summable}{القابلة للجمع}
تحوّل التوافقيات إلى تحليل.
\end{example}

\begin{example}[تقييم بفوبيني: $\sum_n (\zeta(n) - 1) = 1$]\label{ex:b2:series:zetasum}
من أجل $n \geq 2$ صحيح، $\zeta(n) - 1 = \sum_{k \geq 2} k^{-n}$.
والعائلة المزدوجة $(k^{-n})_{k, n \geq 2}$ \omterm{def:b2:series:summable}{قابلة للجمع}: فبجمع
الأعمدة الهندسية أولًا،
\[
\sum_{k\geq2}\sum_{n\geq2} \frac{1}{k^n}
= \sum_{k\geq2} \frac{1/k^2}{1 - 1/k}
= \sum_{k\geq2} \frac{1}{k(k-1)} = 1
\]
(بالتلسكوب)، وكل الحدود موجبة، ومن ثم تُرخّص
\cref{thm:b2:series:fubini} بالجمع حسب الأسطر بدلًا من ذلك:
\[
\sum_{n\geq2}\bigl(\zeta(n) - 1\bigr) = 1 .
\]
فقيم $\zeta$ غير المنتهية العدد، وكلٌّ منها يبدو متساميًا،
لها ذيول تُجمع لتعطي $1$ بالضبط. والفكرة الختامية: حين تكون
حدود مجموع مزدوج موجبة، فاحسبه بالترتيب الذي
ينهار فيه --- وهنا الأعمدة هندسية والأسطر غامضة،
وينقل فوبيني الانهيار.
\end{example}

\begin{example}[المتسلسلة الهندسية تحلّ معادلة]\label{ex:b2:series:neumannsolve}
في \omterm{def:b2:nvs:banach}{فضاء باناخ} $\bigl(C(\intcc01),
\norm\cdot_\infty\bigr)$، حُلّ $x - K(x) = y$ حيث $K(f)$ هي
الدالة الثابتة $\frac12\int_0^1 f$. \omterm{thm:b2:nvs:continuouslinear}{معيار المؤثر} هو
$\vertiii K \leq \frac12 < 1$، ومن ثم تنطبق متسلسلة نويمان
(\cref{ex:b2:series:neumann}): $x = \sum_{n\geq0} K^n(y)$.
ونحسب التكرارات: $K(y) = \frac12\int_0^1 y$ (وهي ثابتة)،
وتطبيق $K$ على ثابت $c$ يعطي $\frac c2$، ومنه
$K^n(y) = \frac{1}{2^{n-1}}\cdot\frac12\int_0^1 y$ من أجل $n \geq
1$. وبجمع الثوابت الهندسية:
\[
x = y + \Bigl(\int_0^1 y\Bigr)
\sum_{n\geq1}\frac{1}{2^n} = y + \int_0^1 y .
\]
وللتحقق: $x - K(x) = y + \int y - \frac12\bigl(\int y + \int
y\bigr) = y$. متسلسلة غير منتهية، وجواب منته،
وتحقق في سطر واحد --- فالمتسلسلة الهندسية خوارزمية قلب،
لا مجرّد قول في التقارب.
\end{example}

\begin{example}[التلسكوب بالكسور الجزئية]\label{ex:b2:series:telescope}
الجمع الدقيق نادر؛ والتلسكوب مورده الرئيسي.
نفكك
\[
\frac{1}{n(n+1)(n+2)}
= \frac{1}{2}\Bigl(\frac{1}{n(n+1)} -
\frac{1}{(n+1)(n+2)}\Bigr),
\]
(بالتحقق بالتوحيد على مقام مشترك)، ومن ثم تنهار المجاميع
الجزئية:
\[
\sum_{n=1}^{N}\frac{1}{n(n+1)(n+2)}
= \frac12\Bigl(\frac{1}{1\cdot2} -
\frac{1}{(N+1)(N+2)}\Bigr)
\longrightarrow \frac14 .
\]
والنمط نفسه --- أن نكتب الحدّ على صورة $c(u_n - u_{n+1})$ من أجل
$(u_n)$ صريح --- حلّ \cref{exo:b2:series:10}
(بالظلال العكسية) ويحسب كل مقدار $\sum\frac{1}{n(n+1)\cdots(n +
k)} = \frac{1}{k\cdot k!}$. وحين يوجد مجموع دقيق عند هذا
المستوى، يكون التلسكوب مختبئًا في الحدّ عادةً.
\end{example}

\begin{remark}[آفاق داخل هذا المجلد]\label{rem:b2:series:perspectives}
ثلاثة فصول آتية زبائن مباشرون. فمن أجل
\cref{ch:b2:funcseq}: التقارب الناظمي للمقدار $\sum f_n$ هو
تقارب مطلق للمقدار $\sum\norm{f_n}_\infty$ في \omterm{def:b2:nvs:banach}{فضاء باناخ}
$\bigl(C, \norm\cdot_\infty\bigr)$ --- أي مبرهنة
\cref{thm:b2:nvs:absoluteconvergence} في هذا الفصل بلباس آخر. ومن أجل
\cref{ch:b2:powerseries}: يكون كل شيء داخل قرص التقارب
مطلقًا وقابلًا للجمع، ومن ثم تجري جداءات كوشي وإعادات الترتيب
بحرية (ولهذا تُضرب متسلسلات القوى مثل
كثيرات الحدود)؛ وأما على الحدّ فيتولى محك آبل
(\cref{ex:b2:series:abelboundary}). ومن أجل \cref{ch:b2:genfun}:
تكون الدوال المولّدة الاحتمالية متسلسلات قوى تعالجها
--- بالجداءات من أجل مجاميع المتغيرات المستقلة، وبالمجاميع
المزدوجة من أجل القوانين المركّبة --- كلها بترخيص من
\cref{thm:b2:series:fubini}. فالعائلات \omterm{def:b2:series:summable}{القابلة للجمع} هي
الدائرة القانونية للتحليل الآتي.
\end{remark}

\begin{remark}[أين يُستعمَل هذا الفصل]
كل ما فيه مجموع غير منته يمرّ من هنا: فمتسلسلات القوى
(\cref{ch:b2:powerseries}) عائلات \omterm{def:b2:series:summable}{قابلة للجمع} متخفية،
وتُضرب معاملات فورييه بجداءات كوشي ويُعاد ترتيبها
ببارسفال (\cref{ch:b2:fourier})، والدوال المولّدة
الاحتمالية (\cref{ch:b2:genfun}) هي مبرهنة فوبيني
مطبَّقةً على الآمال. ويستوعب مجلد السنة الثالثة العائلات
\omterm{def:b2:series:summable}{القابلة للجمع} في تكامل لوبيغ على قياس العدّ ---
حيث تصير \cref{thm:b2:series:fubini} حالةً خاصة من
مبرهنة فوبيني--تونيللي.
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:b2:series:7}
برهن على أن العائلة $\bigl(\frac{1}{m^2 n^2}\bigr)_{m,n \geq 1}$
\omterm{def:b2:series:summable}{قابلة للجمع}، وأن التجميع حسب $\gcd$ يعطي: مع $q = \gcd(m,n)$،
\[
\zeta(2)^2 = \sum_{q\geq1} \frac{1}{q^4}
\sum_{\substack{a,b \geq 1\\ \gcd(a,b)=1}} \frac{1}{a^2b^2}
= \zeta(4) \cdot S,
\]
حيث $S = \sum_{\gcd(a,b)=1} \frac{1}{a^2b^2}$: واستنتج $S =
\zeta(2)^2/\zeta(4)$. \emph{(فكل زوج $(m,n)$ يُكتب بكيفية وحيدة
$(qa, qb)$ مع $\gcd(a,b) = 1$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:8}
(مبرهنة آبل على الجداءات، نسخة خفيفة) لنفترض أن $\sum a_n$

\omterm{thm:b2:series:fubini}{جداءهما الكوشي} $\sum c_n$ يتقارب، مع $\sum c_n = (\sum
a_n)(\sum b_n)$. \emph{(اكتب $C_N = \sum_{k\leq N} c_k = \sum_n
a_n B_{N-n}$ حيث $B$ المجاميع الجزئية للمتسلسلة $b$؛ وافصل حسب

$(a_n)$ المطلق.)}
\end{exercise}

\section{مسألة: مقدار أويلر $\zeta(2) = \pi^2/6$، بمجموع ظل التمام عند كوشي}

تقبل أشهر متطابقات أويلر، $1 + \frac14 + \frac19 + \cdots =
\frac{\pi^2}6$، برهانًا أوّليًا تمامًا يعود
إلى كوشي: فصيغة دي موافر تُنتج كثير حدود جذوره
هي الأعداد $\cot^2\frac{k\pi}{2n+1}$، وتجمع صيغ فييت تلك الجذور
بالضبط، ويسحق الحصر $\cot^2\theta < \frac{1}{\theta^2} <
1 + \cot^2\theta$ المجاميعَ الجزئية للمقدار $\sum\frac1{k^2}$
بين حدّين ناطقين صريحين. ونشغّل البرهان كاملًا،
ونستخرج $\zeta(4) = \frac{\pi^4}{90}$ بالطريقة نفسها، ثم
نرسم خريطة الحدود كلها بين التقارب والتباعد بمتسلسلات
برتران --- ونبرهن على أن هذه الحدود لا تحمل أي متسلسلة
متقاربة أبطأ من كل ما سواها.

\begin{problem}[{مسألة نهاية الأسبوع --- $\zeta(2) = \pi^2/6$ وبانوراما
برتران}]
\label{pb:b2:series:1}
في كل ما يلي، $n \geq 1$ و$\theta_k = \dfrac{k\pi}{2n+1}$ من أجل $k
= 1, \dots, n$؛ ولاحظ أن $0 < \theta_k < \frac\pi2$.

\medskip
\noindent\textbf{الجزء الأول --- متطابقة ظل التمام.}
\begin{enumerate}
  \item برهن على صيغة دي موافر $(\cos\theta +
        \iu\sin\theta)^m = \cos m\theta + \iu\sin m\theta$ (مع $m
        \in \N$)، واستنتج، من أجل $m = 2n + 1$،
        \[
        \sin\bigl((2n{+}1)\theta\bigr) = \sum_{j=0}^{n}
        (-1)^j\binom{2n+1}{2j+1}\cos^{2(n-j)}\theta\,
        \sin^{2j+1}\theta .
        \]
  \item استنتج أنه من أجل $\theta \in \intoo{0}{\frac\pi2}$،
        \[
        \sin\bigl((2n{+}1)\theta\bigr) =
        \sin^{2n+1}\theta\; P_n(\cot^2\theta),
        \qquad
        P_n(x) = \sum_{j=0}^{n}
        (-1)^j\binom{2n+1}{2j+1}x^{\,n-j},
        \]
        وهو كثير حدود درجته $n$ ومعامله الرئيسي $2n
        + 1$.
  \item بيّن أن الأعداد $x_k = \cot^2\theta_k$، مع $k = 1, \dots, n$، هي
        $n$ جذرًا \emph{مختلفًا} لكثير الحدود $P_n$ --- ومن ثم فهي كل
        جذوره.
  \item بصيغ فييت، برهن على المتطابقة الدقيقة
        \[
        \sum_{k=1}^{n} \cot^2\frac{k\pi}{2n+1}
        = \frac{n(2n-1)}{3}.
        \]
  \item استنتج أيضًا $\displaystyle\sum_{k=1}^{n}
        \frac{1}{\sin^2\theta_k} = \frac{2n(n+1)}{3}$.
  \item برهن على الحصر: $\cot^2\theta < \dfrac1{\theta^2} <
        \dfrac{1}{\sin^2\theta}$ من أجل $\theta \in
        \intoo{0}{\frac\pi2}$ \emph{(من $\sin\theta < \theta <
        \tan\theta$)}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الحصر ينغلق: مبرهنة
أويلر.}
\begin{enumerate}[resume]
  \item بجمع السؤال 6 على $k = 1, \dots, n$ مع $\theta =
        \theta_k$، أثبت
        \[
        \frac{n(2n-1)}{3} \;<\;
        \frac{(2n+1)^2}{\pi^2}\sum_{k=1}^{n}\frac1{k^2}
        \;<\; \frac{2n(n+1)}{3}.
        \]
  \item اختم (\emph{مبرهنة أويلر، ببرهان كوشي}):
        \[
        \zeta(2) = \sum_{k=1}^{\infty}\frac{1}{k^2} =
        \frac{\pi^2}{6}.
        \]
  \item استخرج معدلًا من الحصر: بيّن أن
        \[
        \Bigl|\sum_{k=1}^{n}\frac1{k^2} - \frac{\pi^2}6\Bigr|
        = O\Bigl(\frac1n\Bigr),
        \]
        وهو متوافق مع الذيل الدقيق $\sum_{k>n}k^{-2} =
        \frac1n - \frac{1}{2n^2} + O(n^{-3})$ في
        \cref{exo:b2:comparison:11}.
  \item شغّل الآلة طابقًا أعلى: باستعمال دالة فييت المتناظرة الثانية
        للأعداد $P_n$، بيّن أن
        \[
        \sum_{k=1}^{n}\cot^4\theta_k =
        \Bigl(\frac{n(2n-1)}3\Bigr)^{\!2} -
        \frac{2n(2n-1)(2n-2)(2n-3)}{60}
        \;\sim\; \frac{8n^4}{45},
        \]
        واحصر بالمقدار $\cot^4 < \theta^{-4} < (1 +
        \cot^2)^2$ للحصول على $\zeta(4) = \dfrac{\pi^4}{90}$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الأرباح.}
\begin{enumerate}[resume]
  \item استنتج من $\zeta(2) = \frac{\pi^2}6$:
        \[
        \sum_{k\geq0}\frac{1}{(2k+1)^2} = \frac{\pi^2}{8},
        \qquad
        \sum_{k\geq1}\frac{(-1)^{k-1}}{k^2} = \frac{\pi^2}{12}.
        \]
  \item اجمع ذلك مع \cref{exo:b2:series:7}: احسب $S =
        \sum_{\gcd(a,b)=1}\frac{1}{a^2b^2} =
        \frac{\zeta(2)^2}{\zeta(4)} = \frac52$، وفسّر
        $\frac{1}{\zeta(2)} = \frac{6}{\pi^2} \approx 0.608$ بوصفه
        كثافة الأزواج الأولية فيما بينها (وصُغ الحدس
        بأمانة: فالعدّ الصارم مسألةٌ في مجلد
        السنة الثالثة).
  \item (تسريع مصدَّق) تعطي صيغة الذيل في السؤال 9
        أن $\sum_{k\leq n}k^{-2} + \frac1n - \frac1{2n^2} =
        \frac{\pi^2}6 + O(n^{-3})$. قارِن الجهد اللازم من أجل
        ستة أرقام من $\zeta(2)$: الجمع المباشر في مقابل
        المجموع المصحَّح عند $n = 100$ (حيث الخطأ
        $1.7\cdot10^{-7}$).
  \item تحقق من السؤال 4 يدويًا عند $n = 1$ و$n = 2$ (والقيمتان
        $\cot^2\frac\pi3 = \frac13$ و
        $\cot^2\frac\pi5 + \cot^2\frac{2\pi}5 = 2$)، مستعملًا
        $\cos\frac\pi5 = \frac{1+\sqrt5}4$ أو تقييمًا عدديًا.
  \item برهن على المتطابقة المرافقة
        \[
        \sum_{k=1}^{n}\tan^2\frac{k\pi}{2n+1} = n(2n+1)
        \]
        \emph{(فالأعداد $\tan^2\theta_k$ هي جذور
        كثير الحدود المعكوس $x^nP_n(1/x)$)}، وتحقق منها عند $n
        = 1$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- بانوراما برتران.} من أجل $\alpha,
\beta \in \R$، لننظر في \emph{متسلسلة برتران}
\[
\sum_{n \geq 3} \frac{1}{n^{\alpha}(\ln n)^{\beta}} .
\]
\begin{enumerate}[resume]
  \item بيّن أنه من أجل $\alpha > 1$ تتقارب المتسلسلة،
        مهما كان $\beta$ \emph{(بالمقارنة مع
        $n^{-(1+\alpha)/2}$)}.
  \item بيّن أنه من أجل $\alpha < 1$ تتباعد، مهما كان
        $\beta$.
  \item ومن أجل $\alpha = 1$: باستعمال مقارنة المتسلسلة بالتكامل
        (\cref{thm:b2:comparison:seriesintegral}) مع $f(t) =
        \frac{1}{t(\ln t)^\beta}$، برهن على أنها تتقارب إذا وفقط إذا $\beta
        > 1$.
  \item كرّر الحدود: بيّن أن $\sum\frac{1}{n\ln n\,\ln\ln
        n}$ تتباعد في حين أن $\sum\frac{1}{n\ln n\,(\ln\ln n)^2}$
        تتقارب.
  \item فخّان: عيّن طبيعة
        \[
        \sum_n \frac{1}{n^{1 + 1/\ln n}}
        \qquad\text{و}\qquad
        \sum_n \frac{1}{n^{1 + 1/\ln\ln n}}
        \]
        \emph{(احسب $n^{1/\ln n}$ بالضبط؛ وقارِن
        $n^{1/\ln\ln n}$ بكل قوة للمقدار $\ln n$)}.
  \item (لا توجد متسلسلة متقاربة أبطأ من الجميع) لتكن $\sum a_n$ أي
        متسلسلة متقاربة مع $a_n > 0$، ولتكن $R_n = \sum_{k
        \geq n}a_k$ ذيولها. برهن على أن $\sum
        \frac{a_n}{\sqrt{R_n}}$ تبقى \emph{متقاربة}
        \emph{(بالمقارنة مع التلسكوب $2(\sqrt{R_n} -
        \sqrt{R_{n+1}})$)}، رغم أن $\frac{a_n/\sqrt{R_n}}{a_n}
        \to \infty$: أي إن كل متسلسلة متقاربة مهيمَن عليها تمامًا
        بمتسلسلة متقاربة أخرى. فحدود التقارب
        ليست منحنى بل ضباب.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- تحققات متقاطعة وتركيب.}
\begin{enumerate}[resume]
  \item (تكثيف كوشي) برهن على أنه من أجل $(a_n)$ موجبة
        متناقصة، تتقارب $\sum a_n$ إذا وفقط إذا تقاربت $\sum 2^k a_{2^k}$. وأعد استنباط حدود السؤال 18 منها.
  \item (ثمن البطء) من أجل $\sum\frac1{n(\ln n)^2}$،
        حُدّ الذيل بتكامل وبيّن أن الجمع إلى غاية
        $N = 10^6$ يترك خطأً أكبر من
        $0.07$: فالتقارب المصدَّق نظريًا قد يكون عديم الفائدة
        عدديًا --- وقارِن بالسؤال 13.
  \item صنّف (بتبريرات من سطر واحد):
        $\sum\frac1{n\ln n}$، $\sum\frac1{n^{1.01}}$،
        $\sum\frac{(\ln n)^{100}}{n^{1.001}}$،
        $\sum\frac1{n(\ln n)(\ln\ln n)^{3}}$.
  \item (تركيب) بجملة واحدة لكل بند: كيف حوّل دي موافر
        متطابقةً مثلثية إلى كثير حدود بمجاميع جذور قابلة
        للحساب؛ وأين احتاج الحصر إلى متطابقات
        \emph{دقيقة} عند الطرفين لا إلى مكافئات؛ وأي أداة
        من \cref{ch:b2:comparison} شغّلت الجزء الرابع؛ وماذا
        يقول السؤال 21 عن حلم ``محك المقارنة
        الشامل''. وسمِّ القمتين: مقدار أويلر
        $\zeta(2) = \frac{\pi^2}6$ (وطابقه الأعلى،
        $\zeta(4) = \frac{\pi^4}{90}$)، وتصنيف برتران.
        ولاحظ أين سيُبرهن على $\zeta(2)$ من جديد:
        ببارسفال في \cref{ch:b2:fourier} --- مبرهنة
        واحدة، وحضارتان.
\end{enumerate}
\end{problem}
