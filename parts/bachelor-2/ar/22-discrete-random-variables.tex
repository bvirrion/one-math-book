\chapter{المتغيرات العشوائية المتقطعة}\label{ch:b2:randomvar}

تنظم \omterm{def:b2:randomvar:law}{المتغيرات العشوائية} الحسابات الاحتمالية حول الدوال بدل \omterm{def:b2:proba:space}{الأحداث}.
وعلى الفضاءات \omterm{def:b2:structures:countable}{القابلة للعد}، تشغّل النظريةَ العائلاتُ \omterm{def:b2:series:summable}{القابلة للجمع} في
\cref{ch:b2:series}: \omterm{def:b2:randomvar:expectation}{فالأمل الرياضي} مجموع عائلة مفهرسة \omterm{def:b2:proba:space}{بالفضاء العيني}، وجميع
خواصه --- الخطية، والنقل، وصيغة الجداء من أجل المتغيرات المستقلة ---
مبرهنات عن العائلات \omterm{def:b2:series:summable}{القابلة للجمع}. ويبرهن الفصل على المتراجحات
المفتاحية لماركوف وتشيبيشيف وكوشي--شوارتز وجنسن، وينتهي بالقوانين
الكلاسيكية وبالقانون الضعيف للأعداد الكبيرة، الذي يصير برهانه سطرين
بمجرد توفر متراجحة تشيبيشيف.

\section{المتغيرات العشوائية وقوانينها}

\begin{definition}[المتغير العشوائي المتقطع؛ القانون]\label{def:b2:randomvar:law}
ليكن $(\Omega, \P)$ فضاء احتماليا قابلا للعد. \emph{المتغير العشوائي}\index{متغير عشوائي}
هو تطبيق $X \colon \Omega \to E$ (حيث $E$ مجموعة كيفما كانت؛ ونقول متغيرا عشوائيا
\emph{حقيقيا} عندما $E = \R$). و\emph{قانونه}\index{قانون!متغير عشوائي}
(أو \emph{توزيعه}\index{توزيع}) هو \omterm{def:b2:proba:space}{القياس الاحتمالي} $\P_X$ على
\omterm{def:b2:structures:countable}{المجموعة القابلة للعد} $X(\Omega)$ المعرَّف بالعلاقة
\[
\P_X(\{x\}) = \P(X = x)
= \P\bigl(\{\omega : X(\omega) = x\}\bigr) .
\]
\end{definition}

\begin{example}[القوانين الكلاسيكية]\label{ex:b2:randomvar:laws}
\leavevmode
\begin{itemize}
\item \emph{برنولي} $\mathcal{B}(p)$: $X \in \{0, 1\}$، $\P(X =
1) = p$. وهو الدالة المميزة \omterm{def:b2:proba:space}{لحدث}.
\item \emph{ثنائي الحد} $\mathcal{B}(n, p)$: $\P(X = k) =
\binom nk p^k(1-p)^{n-k}$، $0 \leq k \leq n$: أي عدد النجاحات في
$n$ تجربة برنولي مستقلة (مجلد الثانوية؛ ويُعاد البرهان عليه أدناه
عبر مجاميع المتغيرات المستقلة).
\item \emph{الهندسي} $\mathcal{G}(p)$: $\P(X = k) =
(1-p)^{k-1}p$، $k \in \N^*$: أي رتبة أول نجاح
(\cref{ex:b2:proba:geometric}).
\item \emph{بواسون} $\mathcal{P}(\lambda)$: $\P(X = k) =
e^{-\lambda}\frac{\lambda^k}{k!}$، $k \in \N$ --- وهو \omterm{def:b2:proba:space}{قياس احتمالي}
بحكم المتسلسلة الأسية. وهو قانون \omterm{def:b2:proba:space}{الأحداث} النادرة (\cref{ch:b2:genfun}).
\end{itemize}
\end{example}

\begin{remark}[أي قانون ينمذج ماذا]
تجيب القوانين الأربعة عن أربعة أسئلة أولية: برنولي، ``هل وقع؟''؛
وثنائي الحد، ``كم مرة في $n$ محاولة؟''؛ والهندسي، ``كم يطول
الانتظار حتى المرة الأولى؟''؛ وبواسون، ``كم حدثا عند معدل معطى،
عندما تكون المحاولات كثيرة وكل منها غير مرجح؟''. والتعرف على السؤال
تسعة أعشار النمذجة: فمجاميع الدوال المميزة تشير إلى ثنائي الحد،
وأزمنة الانتظار إلى الهندسي، وعدود \omterm{def:b2:proba:space}{الأحداث} النادرة إلى بواسون ---
مع جعل المرور من ثنائي الحد إلى بواسون دقيقا بقانون \omterm{def:b2:proba:space}{الأحداث} النادرة
في \cref{ch:b2:genfun}.
\end{remark}

\begin{proposition}[انعدام ذاكرة القانون الهندسي]\label{prop:b2:randomvar:memoryless}
إذا كان $X \sim \mathcal{G}(p)$، فإن من أجل كل $m, n \in \N$:
\[
\P(X > m + n \mid X > m) = \P(X > n) ,
\]
والقوانين الهندسية هي القوانين الوحيدة على $\N^*$ التي لها هذه
الخاصية.
\end{proposition}

\begin{proof}
بجمع الأوزان الهندسية، $\P(X > n) = (1-p)^n$. ومنه
\[
\P(X > m + n \mid X > m)
= \frac{\P(X > m + n)}{\P(X > m)}
= \frac{(1-p)^{m+n}}{(1-p)^m} = (1-p)^n = \P(X > n).
\]
وبالعكس، إذا حقق $G(n) = \P(X > n)$ العلاقة $G(m + n) =
G(m)G(n)$ مع $G(0) = 1$، فإن $G(n) = G(1)^n$ بالتراجع؛
و$q =
G(1) \in \intco{0}{1}$، ويكون $q = 0$ أو يكون القانون $\mathcal{G}(1 - q)$: $\P(X = k) = G(k-1) - G(k) =
q^{k-1}(1 - q)$.
\end{proof}

\begin{example}[لا رقم ``مستحق'' أبدا]
نرمي نردا انتظارا لستة: فزمن الانتظار $X \sim
\mathcal G(1/6)$. ويقول انعدام الذاكرة
إنه بعد $10$ رمية بلا جدوى، يكون الانتظار \emph{المتبقي} $X - 10$،
علما \omterm{def:b2:proba:space}{بالحدث} $X > 10$، هو $\mathcal G(1/6)$ من جديد: فالانتظار المتوقع شرطيا لا
يزال $6$ رمية، تماما كما في البداية. فالنرد لا يتذكر، ولا ``تستحق''
أي ستة الظهور أبدا --- ومغالطة المقامر هي الاعتقاد بأن القانون الشرطي
كان ينبغي أن ينزاح. وبالعكس، يقول شق الوحدانية من القضية إن هذه
اللامبالاة \emph{تميّز} أزمنة الانتظار الهندسية: فكل زمن انتظار لا
يتحدث تنبؤه أبدا يكون هندسيا. أما الطوابير والأعمار الحقيقية فعادة
ما تتحدث، وهذا بالضبط كيف نكتشف أنها ليست هندسية.
\end{example}

\section{الأمل الرياضي}

\begin{definition}[الأمل الرياضي]\label{def:b2:randomvar:expectation}
نقول إن متغيرا عشوائيا حقيقيا $X$ على $(\Omega, \P)$ \emph{يقبل أملا
رياضيا} إذا كانت العائلة $\bigl(X(\omega)\,\P(\{\omega\})\bigr)_{\omega \in \Omega}$ \omterm{def:b2:series:summable}{قابلة للجمع} (\cref{ch:b2:series})؛ وعندئذ يكون
\emph{أمله الرياضي}\index{أمل رياضي}
\[
\E(X) = \sum_{\omega \in \Omega} X(\omega)\,\P(\{\omega\}) .
\]
\end{definition}

\begin{theorem}[مبرهنة النقل]\label{thm:b2:randomvar:transfer}
يقبل $X$ أملا رياضيا إذا وفقط إذا كانت العائلة $\bigl(x\,\P(X =
x)\bigr)_{x \in X(\Omega)}$ \omterm{def:b2:series:summable}{قابلة للجمع}،
وعندئذ
\[
\E(X) = \sum_{x \in X(\Omega)} x\,\P(X = x) .
\]
وبوجه أعم، من أجل $f \colon X(\Omega) \to \R$، يقبل المتغير $f(X)$ أملا رياضيا إذا وفقط
إذا كان $\sum_x \abs{f(x)}\,\P(X = x) <
\infty$، وعندئذ $\E(f(X)) = \sum_x f(x)\,\P(X = x)$.
\end{theorem}

\begin{proof}
نجزّئ $\Omega$ إلى مجموعات المستوى $\Omega_x = \{X = x\}$، $x
\in X(\Omega)$. وبمبرهنة الجمع
بالرزم من أجل العائلات \omterm{def:b2:series:summable}{القابلة للجمع} (\cref{ch:b2:series})، تكون العائلة $(X(\omega)\P(\{\omega\}))_\omega$
\omterm{def:b2:series:summable}{قابلة للجمع} إذا وفقط إذا كانت كل رزمة كذلك (وهو تلقائي: $\sum_{\omega \in \Omega_x}\abs{x}\P(\{\omega\}) =
\abs x\,\P(X = x)$)
\emph{وكانت} عائلة مجاميع الرزم $\bigl(x\,\P(X = x)\bigr)_x$ \omterm{def:b2:series:summable}{قابلة للجمع} --- وعندئذ يتطابق
المجموعان الكليان. ومن أجل $f(X)$: نطبق النص المبرهن على المتغير
$Y =
f \circ X$، الذي مجموعات مستواه $\{Y = y\} =
\bigsqcup_{x : f(x) = y}\{X = x\}$؛ ويحوّل جمع بالرزم ثان المقدار
$\sum_y y\,\P(Y = y)$ إلى $\sum_x
f(x)\,\P(X = x)$، إذ ترزم الرزم الآن القيم $x$ بحسب صورتها $f(x)$،
مع تكافؤ القابلية للجمع المطلقة لإحدى العائلتين مع القابلية للجمع
المطلقة للأخرى.
\end{proof}

\begin{theorem}[خواص الأمل الرياضي]\label{thm:b2:randomvar:linearity}
على مجموعة \omterm{def:b2:randomvar:law}{المتغيرات العشوائية} التي تقبل أملا رياضيا:
\begin{enumerate}
\item (الخطية) $\E(aX + bY) = a\,\E(X) + b\,\E(Y)$.
\item (الموجبية والرتابة) $X \geq 0 \Rightarrow \E(X)
\geq 0$؛ $X \leq Y \Rightarrow \E(X) \leq \E(Y)$؛ و$\abs{\E(X)} \leq \E(\abs X)$.
\item (الهيمنة) إذا كان $\abs X \leq Z$ وكان $Z$ يقبل أملا رياضيا، فكذلك
$X$.
\end{enumerate}
\end{theorem}

\begin{proof}
كلها خواص لمجاميع العائلات \omterm{def:b2:series:summable}{القابلة للجمع} (\cref{ch:b2:series}): خطية المجموع،
والموجبية حدا حدا، ومحك الهيمنة من أجل القابلية للجمع. (ولاحظ أن
الخطية فورية على \emph{التعريف} على $\Omega$، بينما كانت ستكون محرجة
على صيغة النقل --- وهي إحدى فوائد تعريف $\E$ في الأعلى.)
\end{proof}

\begin{example}\label{ex:b2:randomvar:expectations}
$X \sim \mathcal{B}(n, p)$: بكتابة $X = X_1 + \dots + X_n$ مجموعا لدوال برنولي المميزة وباستعمال الخطية،
$\E(X) = np$ --- دون حاجة إلى أي معاملات ثنائية الحد. $X \sim \mathcal{G}(p)$: $\E(X) =
\sum_{k\geq1}k(1-p)^{k-1}p = p\cdot\frac{1}{(1 - (1-p))^2} =
\frac1p$،
بمفاضلة المتسلسلة الهندسية داخل قرصها (\cref{ch:b2:powerseries}). $X \sim \mathcal{P}(\lambda)$: $\E(X)
= \sum_{k\geq1}k e^{-\lambda}\frac{\lambda^k}{k!} = \lambda
e^{-\lambda}\sum_{j\geq0}\frac{\lambda^j}{j!} = \lambda$.
\end{example}

\begin{example}[النقل في العمل]\label{ex:b2:randomvar:transferex}
من أجل $X \sim \mathcal P(\lambda)$، احسب $\E\bigl(\frac1{1+X}\bigr)$ --- فقانون $\frac1{1+X}$ نفسه محرج، لكن مبرهنة
النقل لا تطلبه أبدا:
\[
\E\Bigl(\frac1{1+X}\Bigr)
= \sum_{k\geq0}\frac{1}{k+1}\,\eu^{-\lambda}
\frac{\lambda^k}{k!}
= \frac{\eu^{-\lambda}}{\lambda}\sum_{k\geq0}
\frac{\lambda^{k+1}}{(k+1)!}
= \frac{\eu^{-\lambda}}{\lambda}\bigl(\eu^\lambda - 1\bigr)
= \frac{1 - \eu^{-\lambda}}{\lambda} .
\]
ودرسان. حسابيا: التعرف على متسلسلة أسية مزاحة هو العمل كله ---
فمبرهنة النقل تختزل آمال $f(X)$ إلى معالجة متسلسلات. وبنيويا: القيمة
الساذجة بالتعويض ستكون $\frac1{1 + \E X} =
\frac1{1 + \lambda}$، والجواب الحقيقي أكبر،
\[
\frac{1 - \eu^{-\lambda}}{\lambda} \geq
\frac{1}{1 + \lambda},
\]
تماما كما تقتضي متراجحة جنسن من أجل الدالة المحدبة $t
\mapsto \frac1{1+t}$. فآمال
الصور المحدبة تقع فوق القيمة الساذجة بالتعويض، وتجعل مبرهنة النقل مع
تحقق من متسلسلة المتراجحة المجردة ملموسة.
\end{example}

\begin{theorem}[الاستقلال والجداءات]\label{thm:b2:randomvar:product}
يكون المتغيران العشوائيان $X, Y$ \emph{مستقلين} إذا كان $\P(X = x, Y =
y) = \P(X = x)\P(Y = y)$ من
أجل كل $x, y$ --- وبصورة مكافئة، إذا كان الحدثان $\{X \in A\}$ و$\{Y \in B\}$
مستقلين من أجل كل $A,
B$. وإذا كان $X$ و$Y$ متغيرين حقيقيين
مستقلين يقبلان أملا رياضيا، فإن $XY$ يقبل أملا رياضيا و
\[
\E(XY) = \E(X)\,\E(Y) .
\]
\end{theorem}

\begin{proof}
ينتج تكافؤ الصياغتين بالجمع النقطي للمتطابقة على $(x, y) \in A \times B$ (بالجمعية
$\sigma$ مرتين). أما الجداء: فالعائلة المزدوجة $\bigl(xy\,\P(X = x)\P(Y = y)\bigr)_{(x,y)}$ \omterm{def:b2:series:summable}{قابلة للجمع}،
لأنه بحسب مبرهنة فوبيني من أجل العائلات (\cref{ch:b2:series})
\[
\sum_{x, y}\abs x \abs y\,\P(X{=}x)\P(Y{=}y)
= \Bigl(\sum_x \abs x \P(X{=}x)\Bigr)
\Bigl(\sum_y \abs y \P(Y{=}y)\Bigr) < \infty ;
\]
وبالاستقلال تكون هذه العائلة بالضبط $\bigl(xy\,\P(X = x, Y =
y)\bigr)$، ومجموعها $\E(XY)$ بمبرهنة
النقل مطبقة على المتغير $(X, Y) \mapsto xy$؛ وتقوّم مبرهنة فوبيني من جديد المجموع
غير الموقّع بالجداء $\E(X)\E(Y)$.
\end{proof}

\begin{example}[الجداءات، مع الاستقلال وبدونه]\label{ex:b2:randomvar:productcontrast}
نرمي نردين متزنين. فإذا كان $Y$ هو النرد الثاني (المستقل عن
الأول)، فإن $\E(XY) = \E(X)\E(Y) = 3.5^2 = 12.25$. أما إذا كان $Y = X$ (أي ``جداء'' نرد بنفسه)،
\[
\E(X^2) = \frac{1 + 4 + 9 + 16 + 25 + 36}{6} = \frac{91}{6}
\approx 15.17 \neq 12.25 :
\]
فالقانونان الحديان أنفسهما في السيناريوهين، لكن القانونين المشتركين
مختلفان، وآملا الجداء مختلفان. والعبرة، وهي جديرة بأن تُنقش: أن
$\E(XY)$ دالة تخص \emph{الزوج}، لا القانونين الحديين --- والفرق $\E(X^2) - \E(X)^2 \approx 2.92$
هو بالضبط، بحسب كونيغ--هويغنز، \omterm{def:b2:randomvar:variance}{تباين} $\frac{35}{12}$ النرد.
\end{example}

\section{التباين والتغاير والمتراجحات الكلاسيكية}

\begin{definition}[العزوم والتباين]\label{def:b2:randomvar:variance}
نقول إن $X$ يقبل \emph{عزما من الرتبة 2} إذا كان $X^2$ يقبل
أملا رياضيا (وعندئذ يقبله $X$ أيضا، بالهيمنة: $\abs X \leq \frac{1 +
X^2}{2}$). وعندئذ
يكون \emph{تباينه}\index{تباين} و\emph{انحرافه المعياري}\index{انحراف معياري}
\[
V(X) = \E\bigl((X - \E(X))^2\bigr)
= \E(X^2) - \E(X)^2 ,
\qquad
\sigma(X) = \sqrt{V(X)} ,
\]
(والصيغة الثانية --- وهي صيغة \emph{كونيغ--هويغنز} --- بنشر المربع
واستعمال الخطية:
\[
\E\bigl((X - \E X)^2\bigr)
= \E\bigl(X^2 - 2X\,\E X + \E(X)^2\bigr)
= \E(X^2) - 2\,\E(X)^2 + \E(X)^2 ,
\]
مع كون الحد الأوسط ناتجا عن كون $\E X$ ثابتا). ومن أجل $X, Y$ ذوي
عزوم من الرتبة الثانية، يكون \emph{التغاير}\index{تغاير}
\[
\operatorname{Cov}(X, Y)
= \E\bigl((X - \E X)(Y - \E Y)\bigr)
= \E(XY) - \E(X)\E(Y) .
\]
\end{definition}

\begin{theorem}[عدة التباين]\label{thm:b2:randomvar:variancerules}
من أجل المتغيرات ذات العزوم من الرتبة الثانية:
\begin{enumerate}
\item $V(aX + b) = a^2\,V(X)$؛
\item $V(X + Y) = V(X) + V(Y) + 2\operatorname{Cov}(X, Y)$، وبوجه أعم
\[
V\Bigl(\sum_{i=1}^n X_i\Bigr)
= \sum_{i=1}^n V(X_i)
+ 2\sum_{i < j}\operatorname{Cov}(X_i, X_j) ;
\]
\item إذا كان $X, Y$ مستقلين، فإن $\operatorname{Cov}(X, Y) = 0$ (والعكس غير صحيح)، ومنه
فإن تباينات المتغيرات المستقلة تُجمع.
\end{enumerate}
\end{theorem}

\begin{proof}
البندان \emph{1} و\emph{2} نشران لمربعات مع الخطية؛ وللجداءات $X_iX_j$
آمال رياضية بحكم متراجحة كوشي--شوارتز أدناه (أو بحكم $\abs{X_iX_j} \leq \frac{X_i^2 + X_j^2}{2}$). والبند
\emph{3} هو \cref{thm:b2:randomvar:product} مطبقا على المتغيرين المركزيين. ومثال مضاد قياسي
للعكس: $X$ منتظم على $\{-1, 0, 1\}$ و$Y = X^2$ غير مترابطين ($\E(XY) = \E(X^3) = 0 = \E X \cdot \E Y$) لكنهما
متعلقان بوضوح.
\end{proof}

\begin{theorem}[متراجحتا ماركوف وتشيبيشيف]\label{thm:b2:randomvar:markov}
\leavevmode
\begin{enumerate}
\item (ماركوف) إذا كان $X \geq 0$ يقبل أملا رياضيا، فإن من أجل كل $a
> 0$:
\[
\P(X \geq a) \leq \frac{\E(X)}{a} .
\]
\item (تشيبيشيف) إذا كان $X$ يقبل عزما من الرتبة الثانية، فإن من
أجل كل $\varepsilon > 0$:
\[
\P\bigl(\abs{X - \E(X)} \geq \varepsilon\bigr)
\leq \frac{V(X)}{\varepsilon^2} .
\]
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} نقطيا، $a\,\mathbf{1}_{X \geq a} \leq X$ (فعلى \omterm{def:b2:proba:space}{الحدث} يكون الطرف الأيسر $a \leq X$؛ وخارجه
$0 \leq X$). وبأخذ الآمال: $a\,\P(X \geq a) \leq \E(X)$ بالرتابة و$\E(\mathbf{1}_A) = \P(A)$.
\emph{2.} نطبق متراجحة ماركوف على المتغير الموجب $(X - \E
X)^2$ عند المستوى
$a = \varepsilon^2$: \omterm{def:b2:proba:space}{فالحدث} $\{(X - \E X)^2
\geq \varepsilon^2\}$ هو بالضبط $\{\abs{X - \E X} \geq
\varepsilon\}$.
\end{proof}

\begin{example}[غير مترابطين لكن ملتصقين]\label{ex:b2:randomvar:sumdiff}
نرمي نردين متزنين، $X$ و$Y$ مستقلين، ونضع $S = X +
Y$، $D = X - Y$.
وبثنائية خطية \omterm{def:b2:randomvar:variance}{التغاير}،
\[
\operatorname{Cov}(S, D) = V(X) - V(Y) +
\operatorname{Cov}(Y, X) - \operatorname{Cov}(X, Y) = V(X) -
V(Y) = 0 :
\]
فالمجموع والفرق غير مترابطين. فهل هما مستقلان؟ قطعا لا: إذ يفرض
$S = 12$ أن $D = 0$، بينما $\P(D = 0) =
\frac16$ دون شرط. فالترابط لا يختبر إلا
الجزء \emph{الخطي} من التعلق؛ وهنا يحمل التعلقَ القيدُ القائل إن
$S$ و$D$ لهما الزوجية نفسها، وهو غير مرئي \omterm{def:b2:randomvar:variance}{للتغاير}. (ومن أجل
هذا الزوج، احتاج انعدام \omterm{def:b2:randomvar:variance}{التغاير} إلى $V(X) = V(Y)$: فالتوزيعان المتطابقان،
لا الاستقلال، هما اللذان أنجزا العمل.)
\end{example}

\begin{example}[متى تكون متراجحة ماركوف مساواة]\label{ex:b2:randomvar:markovsharp}
تكون متراجحة ماركوف مساواة بالضبط عندما لا يُهدر شيء في الحصر $a\,\mathbf 1_{X\geq a} \leq X$:
أي عندما لا يأخذ المتغير إلا القيمتين $0$ و$a$. وبصورة ملموسة،
إذا كان $\P(X = a) = \pi$ و$\P(X = 0) = 1 - \pi$، فإن $\E(X)
= a\pi$ و
\[
\P(X \geq a) = \pi = \frac{\E(X)}{a} .
\]
وقراءة واقعية: ففي مجتمع متوسط الثروة فيه $100$ والثروة إما $0$
وإما $10^6$، تكون نسبة أصحاب الملايين هي $10^{-4}$ بالضبط --- أي حصر
ماركوف، مبلوغا بالضبط بمتراجحة قصوى. وكلما انتشر $X$ على قيم
وسيطة كان الحصر تاما، وغالبا بفارق هائل؛ لكن كما تبين الحالة الحدية،
لا يمكن استخراج متراجحة أفضل من المتوسط وحده.
\end{example}

\begin{example}[متراجحة تشيبيشيف أمثلية --- دون فرضيات إضافية]\label{ex:b2:randomvar:chebsharp}
نثبّت $\varepsilon > 0$، $q \in \intoc01$، ونجعل $X$ يأخذ القيمتين $\pm\varepsilon$ باحتمال
$\frac q2$ لكل منهما و$0$ باحتمال $1 - q$. عندئذ $\E(X) = 0$، $V(X) =
q\varepsilon^2$، و
\[
\P\bigl(\abs{X - \E X} \geq \varepsilon\bigr) = q
= \frac{V(X)}{\varepsilon^2} :
\]
أي التساوي في متراجحة تشيبيشيف. ومنه فلا يمكن تحسين المتراجحة
باستعمال \omterm{def:b2:randomvar:variance}{التباين} وحده --- فالتلاشي $1/\varepsilon^2$ هو الثمن المضبوط لمعلومة
العزم من الرتبة الثانية. أما التلاشي الأسرع فيتطلب فرضيات أقوى:
فحصر المتغير يشتري تركيزا \emph{أسيا}، كما يستعرض \cref{exo:b2:randomvar:7} وكما تطوره
مسألة نهاية الأسبوع لهذا الفصل منهجيا.
\end{example}

\begin{theorem}[كوشي--شوارتز وجنسن]\label{thm:b2:randomvar:jensen}
\leavevmode
\begin{enumerate}
\item (كوشي--شوارتز) إذا كان للمتغيرين $X, Y$ عزمان من الرتبة الثانية، فإن
$XY$ يقبل أملا رياضيا و$\E(XY)^2 \leq \E(X^2)\,\E(Y^2)$؛ ومنه $\operatorname{Cov}(X,Y)^2 \leq V(X)V(Y)$.
\item (جنسن) إذا كانت $\varphi \colon I \to \R$ محدبة على فترة تحوي $X(\Omega)$، وكان
$X$ و$\varphi(X)$ يقبلان أملين رياضيين، فإن
\[
\varphi\bigl(\E(X)\bigr) \leq \E\bigl(\varphi(X)\bigr) .
\]
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} قابلية جمع $XY$: $\abs{XY} \leq \frac{X^2 + Y^2}2$. والتطبيق $(X, Y) \mapsto \E(XY)$ صورة ثنائية
الخطية متناظرة موجبة على فضاء المتغيرات ذات العزوم من الرتبة الثانية،
ومنه تنطبق متراجحة كوشي--شوارتز المجردة في \cref{ch:b2:quadratic} (فالصورة \emph{شبه} المعرَّفة
الموجبة تكفي من أجل المتراجحة). وتطبيقها على المتغيرين المركزيين
يعطي حصر \omterm{def:b2:randomvar:variance}{التغاير}.

\emph{2.} أولا، يقع $m = \E(X)$ في $I$: فالفترة $I$ تحوي جميع قيم
$X$، \omterm{def:b2:randomvar:expectation}{والأمل الرياضي} رتيب، ومنه فإن $m$ بين $\inf X(\Omega)$ و$\sup X(\Omega)$.
وبمبرهنة المستقيم المساند من أجل الدوال المحدبة (\cref{ch:b2:realfun})، يوجد $\alpha, \beta$
يحقق $\varphi(t) \geq \alpha t + \beta$ من أجل كل $t \in I$ و$\varphi(m) = \alpha m + \beta$. وعندئذ، نقطيا على $\Omega$،
$\varphi(X) \geq \alpha X + \beta$؛ وبأخذ الآمال،
\[
\E\bigl(\varphi(X)\bigr) \geq \alpha\,\E(X) + \beta
= \varphi\bigl(\E(X)\bigr). \qedhere
\]
\end{proof}

\begin{example}
تعطي متراجحة جنسن مع $\varphi(t) = t^2$ المقدار $\E(X)^2 \leq \E(X^2)$ --- أي موجبية \omterm{def:b2:randomvar:variance}{التباين}؛
ومع $\varphi(t) = 1/t$ على $\intoo{0}{\infty}$: $\frac{1}{\E X} \leq \E\bigl(\frac1X\bigr)$ --- أي إن المتوسط التوافقي دون
المتوسط الحسابي، في صيغة عشوائية الآن.
\end{example}

\begin{remark}[مزالق شائعة]
(1) $\E(XY) = \E(X)\E(Y)$ \emph{يقتضي} الاستقلال (أو على الأقل انعدام \omterm{def:b2:randomvar:variance}{التغاير}):
فأخذ $Y = X$ يعطي $\E(X^2) \neq
\E(X)^2$ كلما كان $V(X) > 0$. (2) وبالمثل $V(X + X) =
4V(X)$، لا
$2V(X)$: فالتباينات لا تُجمع إلا على الحدود المستقلة (أو غير
المترابطة). (3) وليس $\E(f(X))$ هو $f(\E(X))$؛ ومن أجل $f$ محدبة تخبرك
متراجحة جنسن حتى باتجاه الخطأ، كما في \cref{ex:b2:randomvar:transferex}. (4) والوجود فرضية
حقيقية: فمن أجل متغير سان بطرسبورغ $X = 2^K$ مع $\P(K = k) = 2^{-k}$ ($k \geq 1$)،
\[
\sum_{k\geq1}2^k\cdot2^{-k} = \sum_{k\geq1}1 = \infty :
\]
يكون $X$ منتهيا بشكل شبه أكيد ومع ذلك لا يقبل أملا رياضيا، ولا
يوجد ثمن دخول عادل للعبة. فالقابلية للجمع في تعريف $\E$ ليست
تنطعا في مسك الحسابات --- بل هي حيث تُكتشف الذيول الثقيلة. (5)
وأخيرا، تحتاج مبرهنة النقل إلى القابلية للجمع \emph{المطلقة} قبل أن
تكون أي إعادة ترتيب للمجموع على القيم مشروعة (\cref{ch:b2:series}).
\end{remark}


\begin{example}[تشيبيشيف على مائة رمية]\label{ex:b2:randomvar:hundredtosses}
من أجل $X \sim \mathcal B(100, \frac12)$: $\E X = 50$، $V(X) =
25$. وتشيبيشيف مع $\varepsilon = 6$:
\[
\P(45 \leq X \leq 55) = \P(\abs{X - 50} < 6)
\geq 1 - \frac{25}{36} \approx 0.31 ,
\]
بينما يعطي المجموع الثنائي المضبوط $\approx 0.73$. فالمقدار المضمون $31\%$
بعيد عن الحقيقة، لكنه لم يتطلب \emph{إلا} المتوسط \omterm{def:b2:randomvar:variance}{والتباين} ---
فالشهادة نفسها تنطبق حرفيا على أي متغير له $\E = 50$ و$V = 25$، مهما
يكن غريبا، ويبين \cref{ex:b2:randomvar:chebsharp} أن متغيرا من هذا النوع يشبعها. فللشمولية
ثمن؛ وعندما يكون \omterm{def:b2:randomvar:law}{التوزيع} ثنائي الحد فعلا، تغلق الأدوات الأسية في
مسألة نهاية الأسبوع معظم الفجوة.
\end{example}

\begin{example}[ترابط الجزء مع الكل]\label{ex:b2:randomvar:partwhole}
من أجل متغيرات $X, Y$ مستقلة ومتماثلة \omterm{def:b2:randomvar:law}{التوزيع} تباينها $\sigma^2 > 0$، ما
مقدار ترابط أحد الحدود مع المجموع $S = X + Y$؟ نحسب
\[
\operatorname{Cov}(X, S) = \operatorname{Cov}(X, X) +
\operatorname{Cov}(X, Y) = \sigma^2 + 0 = \sigma^2,
\qquad V(S) = 2\sigma^2,
\]
ومنه فإن معامل الترابط هو
\[
\rho(X, S) = \frac{\operatorname{Cov}(X,
S)}{\sigma(X)\,\sigma(S)}
= \frac{\sigma^2}{\sigma\cdot\sigma\sqrt2}
= \frac{1}{\sqrt2} \approx 0.707 ,
\]
مهما يكن القانون المشترك --- نردا أو قطعة نقود أو عدود بواسون. ومع
$n$ حدا يعطي الحساب نفسه $\rho(X_1, S_n) =
1/\sqrt n$: فتأثير كل حد مفرد على المجموع
يتخفف كجذر تربيعي، وهو الظل الترابطي للسلّم $\sqrt n$ للتقلبات. وتضمن
متراجحة كوشي--شوارتز $\abs\rho \leq 1$ دائما؛ وهنا يُبلغ الحصر بالضبط في الحالة
المنحلة $n = 1$ ويتلاشى على نحو متوقع بعدها.
\end{example}

\begin{example}[متراجحة المتوسطين الموزونة من جنسن]\label{ex:b2:randomvar:amgm}
ليكن $Y$ يأخذ القيم الموجبة $a_1, \dots, a_k$ بالاحتمالات $\lambda_1, \dots, \lambda_k$. والدالة
$-\ln$ محدبة على $\intoo0\infty$، ومنه تعطي متراجحة جنسن $-\ln\E(Y) \leq \E(-\ln Y)$، أي
\[
a_1^{\lambda_1}a_2^{\lambda_2}\cdots a_k^{\lambda_k}
\;\leq\; \lambda_1a_1 + \lambda_2a_2 + \dots + \lambda_ka_k :
\]
وهي متراجحة المتوسطين الحسابي والهندسي الموزونة، مع التساوي إذا وفقط
إذا كان $Y$ ثابتا. والأوزان المتساوية $\lambda_i = \frac1k$ تستعيد متراجحة
المتوسطين الكلاسيكية. وهكذا برهنت الاحتمالات بصمت على مبرهنة جبرية
محضة: فاختيار قانون احتمال ليس إلا أداة لمسك حسابات التوليفات
المحدبة --- أي وجهة نظر مراكز الثقل في \cref{ch:b2:affine} مرة أخرى، مع جنسن
محركا هذه المرة.
\end{example}

\section{القانون الضعيف للأعداد الكبيرة}

\begin{theorem}[القانون الضعيف للأعداد الكبيرة]\label{thm:b2:randomvar:lln}
لتكن $(X_k)_{k \geq 1}$ متغيرات عشوائية مستقلة مثنى مثنى لها القانون نفسه وتقبل
عزما من الرتبة الثانية؛ ونكتب $m = \E(X_1)$ و$S_n = X_1 + \dots + X_n$. عندئذ، من أجل كل
$\varepsilon > 0$:
\[
\P\Bigl(\,\Bigl|\frac{S_n}{n} - m\Bigr| \geq \varepsilon\Bigr)
\;\leq\; \frac{V(X_1)}{n\,\varepsilon^2}
\xrightarrow[n \to \infty]{} 0 .
\]
\end{theorem}

\begin{proof}
بالخطية $\E(S_n/n) = m$؛ وبحسب \cref{thm:b2:randomvar:variancerules} (فالاستقلال مثنى مثنى يقتل التغايرات)
$V(S_n) = n\,V(X_1)$، ومنه $V(S_n/n) =
V(X_1)/n$. وتعطي متراجحة تشيبيشيف مطبقة على $S_n/n$
الحصر.
\end{proof}

\begin{remark}
هذه هي المبرهنة التي تربط الاحتمال بالتواتر: فإذا كان $X_k$ الدالة
المميزة \omterm{def:b2:proba:space}{لحدث} $A$ في تكرارات مستقلة، كان $S_n/n$ هو التواتر
الملاحظ \omterm{def:b2:proba:space}{للحدث} $A$، ويقول قانون الأعداد الكبيرة إنه يتركز حول
$\P(A)$ بمعدل $\frac{p(1-p)}{n\varepsilon^2}$. أما القانون \emph{القوي} ($S_n/n \to
m$ بشكل شبه
أكيد) فمبرهنة من السنة الثالثة --- غير أن برهانه من أجل العزوم من
الرتبة الرابعة في المتناول: انظر \cref{exo:b2:randomvar:9}، الذي يشغّل بوريل--كانتيلي
على حصر من نوع تشيبيشيف. وقد شغّل التقدير التشيبيشيفي نفسه برهان
\omterm{thm:b2:funcseq:weierstrass}{كثيرات حدود برنشتاين} لمبرهنة فايرشتراس في التقريب في \cref{ch:b2:funcseq} ---
فمبرهنة العدّ هناك \emph{كانت} القانون الضعيف للأعداد الكبيرة
متنكرا.
\end{remark}

\begin{example}[جمع خمسين قسيمة]
\label{ex:b2:randomvar:couponnumeric}
جامع القسائم في \cref{exo:b2:randomvar:3} مع $n =
50$ لعبة متمايزة: فالعدد الكلي المتوقع
هو
\[
\E(T_{50}) = 50\,H_{50} = 50\sum_{k=1}^{50}\frac1k
\approx 50 \times 4.499 \approx 225
\]
علبة --- أي أربع مرات ونصف التخمين الساذج $50$. والنمو التوافقي هو
القصة كلها: فاللعب $25$ الأولى تصل في نحو $50\ln2 \approx 35$ علبة، بينما
تكلف اللعبة \emph{الأخيرة} وحدها $50$ علبة وسطيا (وهو انتظار
هندسي بالوسيط $\frac1{50}$). فمسائل الإكمال يهيمن عليها شوطها الأخير،
ولهذا يجد \cref{exo:b2:randomvar:12} تقلبات من رتبة $n$ --- أي حجم ذلك الانتظار
الهندسي الأخير --- حول المتوسط $n\ln n$.
\end{example}

\begin{example}[كم يجب أن يكون $n$ كبيرا؟]\label{ex:b2:randomvar:llnnumeric}
لتثبيت التواتر الملاحظ ضمن $\varepsilon = 0.01$ من $\P(A)$ بثقة $95\%$، يقتضي حصر
تشيبيشيف
\[
\frac{p(1-p)}{n\varepsilon^2} \leq \frac{1}{4n\varepsilon^2}
\leq 0.05,
\qquad\text{أي}\qquad
n \geq \frac{1}{4\cdot0.05\cdot(0.01)^2} = 50\,000 .
\]
والتعلق وحشي بالنسبة إلى $\varepsilon$ (تربيعي)، ولطيف بالنسبة إلى الثقة
(خطي بدلالة $1/\alpha$). وكلتا الخاصيتين خاصية \emph{للحصر} لا للحقيقة:
فمتراجحات مسألة نهاية الأسبوع الأسية تخفض ثمن الثقة من $1/\alpha$ إلى
$\ln(1/\alpha)$ --- فالمواصفة نفسها ستكلف نحو $18\,500$ عينة هناك --- بينما
السلّم $1/\varepsilon^2$ حقيقي ولا يمكن تحسينه. ومعرفة أي جزء من الحصر مرتخ
لا تقل فائدة عن الحصر نفسه.
\end{example}

إلى الأمام، يغذي كل ما هنا \cref{ch:b2:genfun}: \omterm{def:b2:randomvar:expectation}{فالأمل الرياضي} $\E(t^X)$ لدالة
ماكرة واحدة للمتغير $X$ يحزم القانون كله في متسلسلة قوى، وتصير
العزوم مشتقات عند $1$، وتحمل متطابقات من نوع فالد من أجل المجاميع
العشوائية نظرية مسارات التفرع؛ وتصير مبرهنة الجداء من أجل المتغيرات

مرتقية بتغيير متغير واحد جيد، تمتد على ثلاثة فصول.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:1}
احسب $\E(X)$ و$V(X)$ من أجل $X \sim \mathcal{B}(n, p)$ (عبر الدوال المميزة)، ومن أجل
$X \sim \mathcal{P}(\lambda)$ (بيّن $V(X) =
\lambda$)، ومن أجل $X \sim \mathcal{G}(p)$ (بيّن $V(X) =
\frac{1-p}{p^2}$؛ واستعمل $\E(X(X-1))$
والمشتقة الثانية للمتسلسلة الهندسية).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:2}
ليكن $X \sim \mathcal{P}(\lambda)$ و$Y \sim \mathcal{P}(\mu)$ مستقلين. بيّن أن $X + Y \sim \mathcal{P}(\lambda + \mu)$ (وهو التفاف الأوزان؛
بمبرهنة ذات الحدين)، وأن القانون الشرطي للمتغير $X$ علما \omterm{def:b2:proba:space}{بالحدث}
$X + Y = n$ هو القانون الثنائي $\mathcal{B}\bigl(n, \frac{\lambda}{\lambda + \mu}\bigr)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:3}
(جامع القسائم، \omterm{def:b2:randomvar:expectation}{الأمل الرياضي}) تخفي علامة حبوب واحدة من $n$ لعبة
متمايزة، بانتظام، في كل علبة. وليكن $T_n$ عدد العلب اللازمة لجمع
اللعب $n$ كلها. بكتابة $T_n$ مجموعا لمتغيرات هندسية مستقلة (وهي
زمن رؤية لعبة \emph{جديدة} عندما لا تزال $k$ لعبة ناقصة)، بيّن
\[
\E(T_n) = n\sum_{k=1}^{n}\frac{1}{k} \sim n\ln n
\]
(وهو مكافئ بمقارنة المتسلسلة بالتكامل في \cref{ch:b2:comparison}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:4}
ليكن $X \geq 0$ ذا قيم صحيحة. برهن على \emph{صيغة الذيل}
\[
\E(X) = \sum_{n=1}^{\infty} \P(X \geq n)
\]
(عندما يكون أحد الطرفين منتهيا)، بكتابة $X =
\sum_{n\geq1}\mathbf{1}_{X \geq n}$ وتبديل ترتيب الجمع
(مبرهنة فوبيني من أجل العائلات الموجبة). واستعد $\E(X) = \frac1p$ من أجل
القانون الهندسي.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:5}
(المعاينة دون إرجاع أشد تركيزا) تحوي جرة $N$ كرة، منها $M$
بيضاء. نسحب $n \leq N$ دون إرجاع ونجعل $X$ يعد البيضاوات (وهو
القانون \emph{فوق الهندسي}). باستعمال الدوال المميزة $X = \sum_{i=1}^n Y_i$ مع
$Y_i$ السحبة رقم $i$: بيّن أن كل $Y_i$ برنولي بالوسيط $p = M/N$
(بالتناظر!)، واستنتج $\E(X) = np$ تماما كما في حالة الإرجاع، وبيّن $\operatorname{Cov}(Y_i, Y_j) = -\frac{p(1-p)}{N-1} < 0$
من أجل $i
\neq j$، ومنه $V(X) = np(1-p)\frac{N - n}{N - 1} \leq np(1-p)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:6}
ليكن $X$ ذا عزم من الرتبة الثانية. بيّن أن $c \mapsto \E\bigl((X -
c)^2\bigr)$ أصغري بالضبط
عند $c = \E(X)$، وأن قيمته الصغرى $V(X)$. ثم بيّن أن $\P(X = \E(X)) = 1$ إذا وفقط
إذا كان $V(X) =
0$. \emph{(من أجل النقطة الثانية: إذا كان $V(X) = 0$،
فاستعمل متراجحة تشيبيشيف مع $\varepsilon = 1/n$ والاتصال الرتيب، \cref{thm:b2:proba:continuity}.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:7}
(التركيز يتفوق على ماركوف) ليكن $S_n \sim \mathcal{B}(n,
\frac12)$ (أي عدد الصور في $n$
رمية متزنة). قارن بين الحصور التي تعطيها متراجحة ماركوف ($\P(S_n \geq \frac{3n}{4})$)،
ومتراجحة تشيبيشيف، والطريقة الأسية (تشيرنوف):
\[
\P\Bigl(S_n \geq \frac{3n}4\Bigr)
\leq \E\bigl(e^{tS_n}\bigr)e^{-3nt/4}
= \Bigl(\frac{1 + e^t}{2}\Bigr)^n e^{-3nt/4}
\quad (t > 0),
\]
وأمثل $t$ للحصول على حصر صغير أسيا. \emph{(عند $t = \ln 3$: احصر
$\bigl(2\cdot 3^{-3/4}\bigr)^n \approx
(0.877)^n$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:8}
(فايرشتراس من جديد، احتماليا) لتكن $f \colon [0,1] \to \R$ متصلة وليكن $S_n \sim \mathcal{B}(n, x)$.
بيّن أن كثير حدود برنشتاين $B_nf(x) = \sum_{k=0}^n f\bigl(\frac
kn\bigr)\binom nk x^k(1-x)^{n-k}$ يساوي $\E\bigl[f\bigl(\frac{S_n}{n}\bigr)\bigr]$، وأعد اشتقاق التقدير
$\abs{B_nf(x) - f(x)} \leq \omega_f(\delta) +
\frac{2\norm f_\infty}{4n\delta^2}$ في \cref{ch:b2:funcseq} بهذه اللغة الاحتمالية (بالتقسيم على $\bigl|\frac{S_n}{n} -
x\bigr| \geq \delta$
واستعمال متراجحة تشيبيشيف).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:9}
(القانون القوي تحت عزوم من الرتبة الرابعة) لتكن $(X_k)$ مستقلة
ومتماثلة \omterm{def:b2:randomvar:law}{التوزيع} ومركزية ($\E X_1 = 0$)، مع $\E(X_1^4) < \infty$. بنشر $\E(S_n^4)$ وعدّ
الحدود الباقية (فقط الحدود من نوع $\E(X_i^4)$ و$\E(X_i^2X_j^2)$، $i
\neq j$)، بيّن
$\E(S_n^4) \leq C n^2$ من أجل ثابت $C$. واستنتج
$\sum_n \P\bigl(\abs{S_n/n} \geq \varepsilon\bigr) < \infty$
من أجل كل $\varepsilon > 0$ (بمتراجحة ماركوف من الرتبة 4) واختم ببوريل--كانتيلي
(\cref{thm:b2:proba:borelcantelli}) أن $S_n/n \to 0$ بشكل شبه أكيد وفق صياغة مناسبة: أي إن احتمال
\omterm{def:b2:proba:space}{الحدث} $\bigcap_{j}\bigcup_N\bigcap_{n \geq N}\{\abs{S_n/n} <
\frac1j\}$ هو $1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:10}
يُرمى نردان متزنان؛ وليكن $M$ الأكبر من النتيجتين. باستعمال صيغة
الذيل في \cref{exo:b2:randomvar:4} (بصيغتها المنتهية)، بيّن
\[
\E(M) = \sum_{k=1}^{6}\P(M \geq k)
= 6 - \sum_{j=0}^5\Bigl(\frac j6\Bigr)^2 = \frac{161}{36}
\approx 4.47 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:11}
ليكن $F_n$ عدد النقط الثابتة لتبديلة منتظمة عشوائية على $\{1, \dots, n\}$
($n \geq 2$). بكتابة $F_n =
\sum_i\mathbf 1_{\sigma(i) = i}$، احسب $\E(F_n) = 1$ و$\operatorname{Cov}(\mathbf 1_{\sigma(i)=i}, \mathbf
1_{\sigma(j)=j}) = \frac1{n^2(n-1)}$ من أجل $i \neq j$،
واستنتج $V(F_n) = 1$: أي إن رسالة واحدة وسطيا تبقى في مكانها، \omterm{def:b2:randomvar:variance}{بتباين}
يساوي $1$ بالضبط، مهما يكن $n$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:12}
(جامع القسائم، التركيز) في إطار \cref{exo:b2:randomvar:3}، بيّن
\[
V(T_n) = \sum_{k=1}^n\frac{1 - k/n}{(k/n)^2}
\leq n^2\sum_{k=1}^n\frac{1}{k^2} \leq \frac{\pi^2}{6}n^2,
\]
باستعمال استقلال المراحل الهندسية و$V(\mathcal
G(p)) = \frac{1-p}{p^2}$ (\cref{exo:b2:randomvar:1}؛ والقيمة $\pi^2/6$
هي \cref{ex:b2:fourier:basel}). واستنتج بمتراجحة تشيبيشيف أن $\dfrac{T_n}{n\ln n} \to 1$ \emph{بالاحتمال}:
أي إن الزمن الكلي للجامع هو $n\ln n$ إلى غاية تقلبات من رتبة $n$.
\end{exercise}

\section{مسألة: عدة التركيز، من ماركوف إلى هوفدينغ}

\begin{problem}[{مسألة نهاية الأسبوع --- التركيز الأسي باليد، وكم
شخصا يجب أن يسأل استطلاع الرأي}]
\label{pb:b2:randomvar:1}
تكلف متراجحة ماركوف عزما واحدا وتشتري تلاشيا $1/a$؛ وتكلف متراجحة
تشيبيشيف عزمين وتشتري $1/\varepsilon^2$ --- ويبين \cref{ex:b2:randomvar:chebsharp} أن هذا كل ما يمكن
لهذين العزمين أن يشترياه. وتصعد هذه المسألة بقية السلّم: الطريقة
الأسية (تشيرنوف) بمعدلها \emph{المضبوط} من أجل رمي قطع النقود،
ومتراجحة هوفدينغ من أجل جميع المتغيرات المحدودة، والعائد --- أي
أحجام عينات صريحة وصادقة من أجل استطلاعات الرأي، وإعلان نتائج
الانتخابات، واختبار قطع النقود.
\index{متراجحات التركيز}\index{متراجحة هوفدينغ}\index{حصر تشيرنوف}
وفي كل ما يلي، $S_n \sim
\mathcal B(n, p)$ مجموع $n$ متغير برنولي مستقل و$\widehat p_n = S_n/n$ هو
التواتر التجريبي.

\medskip
\noindent\textbf{الجزء الأول --- المعايرة على قطعة نقود متزنة.}
هنا $p = \frac12$ و$a \in \intoo{\frac12}{1}$.
\begin{enumerate}
  \item متراجحة ماركوف عند المستوى $an$: بيّن $\P(S_n \geq an) \leq
        \frac1{2a}$، وهو حصر لا
        يؤول حتى إلى $0$. فأين تخسر متراجحة ماركوف كل هذا؟
  \item متراجحة تشيبيشيف: باستعمال تناظر القانون الثنائي المتزن حول
        $n/2$، بيّن
        \[
        \P(S_n \geq an) = \tfrac12\,
        \P\bigl(\abs{S_n - \tfrac n2} \geq n(a -
        \tfrac12)\bigr)
        \leq \frac{1}{8n(a - 1/2)^2},
        \]
        أي $\frac2n$ عند $a = \frac34$: فتلاش كثير الحدود أخيرا.
  \item (تشيرنوف، عند مستوى عام) احسب $\E(\eu^{tS_n}) = \bigl(\frac{1 + \eu^t}2\bigr)^n$ وأمثل $\P(S_n \geq an) \leq
        \E(\eu^{tS_n})\eu^{-tan}$ على
        $t > 0$: بيّن أن القيمة المثلى $t$ هي $\ln\frac{a}{1-a}$ وأن
        \[
        \P(S_n \geq an) \leq \eu^{-n\,I(a)},
        \qquad
        I(a) = \ln 2 + a\ln a + (1-a)\ln(1-a) > 0 .
        \]
        وتحقق من أن $a = \frac34$ يستعيد الحصر $\bigl(2\cdot3^{-3/4}\bigr)^n$ في \cref{exo:b2:randomvar:7}.
  \item (الأسّ مضبوط) ليكن $k = an$ عددا صحيحا. انطلاقا من كون $\binom nk a^k(1-a)^{n-k}$
        أكبر حدود \omterm{def:b2:randomvar:law}{التوزيع} الاحتمالي وعددها $n + 1$، برهن على $\binom nk \geq
        \frac{\eu^{nH(a)}}{n+1}$
        مع $H(a) = -a\ln a -
        (1-a)\ln(1-a)$، واستنتج الحصر السفلي المطابق
        \[
        \P(S_n \geq an) \geq \binom{n}{an}2^{-n}
        \geq \frac{\eu^{-n\,I(a)}}{n + 1} .
        \]
  \item جدول الحصور الثلاثة عند $n = 100$، $a =
        \frac34$: ماركوف $\frac23$،
        وتشيبيشيف $0.02$، وتشيرنوف $\approx 2.1\cdot10^{-6}$ (والقيمة الحقيقية $\approx 2.8\cdot10^{-7}$).
        والعبرة، في جملة واحدة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- متراجحة هوفدينغ.}
\begin{enumerate}[resume]
  \item (حالة رادماخر) من أجل $\varepsilon = \pm1$ باحتمال $\frac12$ لكل منهما،
        برهن على
        \[
        \E(\eu^{t\varepsilon}) = \cosh t \leq \eu^{t^2/2}
        \qquad (t \in \R)
        \]
        بمقارنة المتسلسلتين حدا حدا ($(2k)! \geq
        2^kk!$).
  \item استنتج، من أجل متغيرات رادماخر مستقلة $\varepsilon_1, \dots, \varepsilon_n$ ومن أجل كل
        $s >
        0$:
        \[
        \P\Bigl(\sum_{i=1}^n\varepsilon_i \geq s\Bigr)
        \leq \eu^{-s^2/(2n)} .
        \]
  \item ترجم ذلك إلى قطع النقود المتزنة ($X_i =
        \frac{1+\varepsilon_i}2$): $\P\bigl(\widehat p_n -
        \tfrac12 \geq \delta\bigr) \leq \eu^{-2n\delta^2}$، ومع
        الصيغة ذات الطرفين بعامل $2$.
  \item (مبرهنة هوفدينغ المساعدة) ليكن $X \in \intcc01$ مع $\E X =
        p$، و$\psi(t) = \ln\E(\eu^{tX})$.
        برر أن $\psi$ قابل \omterm{def:b2:diffcalc:differential}{للتفاضل} مرتين مع
        \[
        \psi''(t) = \E_t(X^2) - \E_t(X)^2, \qquad
        \E_t(Y) := \frac{\E(Y\eu^{tX})}{\E(\eu^{tX})},
        \]
        وهو \emph{\omterm{def:b2:randomvar:variance}{تباين}} لمتغير معاد ترجيحه لا يزال يأخذ قيمه في
        $\intcc01$؛ واحصره بالمقدار $\frac14$ (بحجة الأصغرية في \cref{exo:b2:randomvar:6})
        واختم بمبرهنة تايلور:
        \[
        \E\bigl(\eu^{t(X - p)}\bigr) \leq \eu^{t^2/8} .
        \]
  \item (متراجحة هوفدينغ) من أجل متغيرات مستقلة $X_i \in
        \intcc01$ لها المتوسط
        المشترك $p$، استنتج
        \[
        \P\bigl(\abs{\widehat p_n - p} \geq \delta\bigr)
        \leq 2\,\eu^{-2n\delta^2}
        \qquad (\delta > 0).
        \]
  \item قارن معدل تشيبيشيف $\frac{p(1-p)}{n\delta^2}$ بمعدل هوفدينغ $2\eu^{-2n\delta^2}$: أي فرضية
        يقتضي كل منهما، وانطلاقا من أي $n$ (تقريبا) يفوز الحصر
        الأسي عند $\delta = 0.03$، $p =
        \frac12$؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- كم شخصا يجب أن يسأل استطلاع
الرأي؟} يسأل استطلاع $n$ ناخبا مستقلا مختارا بانتظام؛ ويجيب كل
منهم بصدق؛ و$p$ هي النتيجة الحقيقية و$\widehat p_n$ هي نتيجة الاستطلاع.
\begin{enumerate}[resume]
  \item بيّن أن الاستطلاع دقيق في حدود $\pm\delta$ بثقة $1 - \alpha$ (أي
        $\P(\abs{\widehat p_n - p} \geq \delta) \leq \alpha$) بمجرد أن يكون
        \[
        n \;\geq\; \frac{\ln(2/\alpha)}{2\,\delta^2} .
        \]
  \item احسب $n$ اللازم من أجل المواصفة القياسية ``ثلاث نقاط،
        خمسة وتسعون في المائة'' ($\delta
        = 0.03$، $\alpha = 0.05$): $n \geq 2050$؛ ومن أجل
        نقطة واحدة: $n \geq 18\,445$. ولاحظ --- وفسّر --- الحقيقة اللافتة
        القائلة إن الجواب لا يتضمن حجم المجتمع.
  \item أعد السؤال 13 بمتراجحة تشيبيشيف ($V(X_1) = p(1-p)
        \leq \frac14$): $n \geq \frac1{4\alpha\delta^2} =
        5556$ عند ثلاث
        نقاط. ولاحظ أن المعاينة \emph{دون} إرجاع لا تزيد الأمر إلا
        حسنا (\cref{exo:b2:randomvar:5}: \omterm{def:b2:randomvar:variance}{فالتباين} يتقلص بالعامل $\frac{N-n}{N-1}$).
  \item (إعلان نتيجة انتخابات) النتيجة الحقيقية لمرشح هي $p
        = 0.52$. كم
        ناخبا يجب استطلاعه ليكون $\P(\widehat p_n \leq \tfrac12) \leq 0.01$؟ بيّن $n
        \geq \frac{\ln 100}{2\cdot(0.02)^2} \approx 5757$ --- فإعلان
        سباق متقارب يكلف أكثر بكثير من تقدير نتيجة.
  \item ما \emph{لا} تغطيه الرياضيات: اسرد فرضيات النمذجة المستعملة
        (المعاينة المنتظمة المستقلة، وصدق الإجابات، وثبات $p$)،
        واشرح في فقرة قصيرة لماذا يهيمن \emph{التحيز} على أخطاء
        الاستطلاعات الحقيقية (المعاينة غير المنتظمة، وعدم الاستجابة)،
        وهو ما لا تخفضه أي زيادة في $n$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- أدق وأرخص.}
\begin{enumerate}[resume]
  \item (وسيط المتوسطات: تلاش أسي انطلاقا من عزمين) نقسم ميزانية
        من $km$ عينة إلى $k$ مجموعة مستقلة من $m$؛ وليكن
        $\widehat p^{(1)}, \dots, \widehat
        p^{(k)}$ متوسطات المجموعات و$M$ وسيطها. اختر $m$ بحيث
        تحقق كل مجموعة $\P(\abs{\widehat p^{(i)} - p} \geq \delta) \leq
        \frac18$ (تشيبيشيف: يكفي $m \geq \frac2{\delta^2}$). بيّن أنه إذا
        كان $\abs{M - p} \geq \delta$ فإن ما لا يقل عن $k/2$ مجموعة تخطئ، واستنتج
        \[
        \P(\abs{M - p} \geq \delta)
        \leq \binom{k}{\lceil k/2\rceil}\Bigl(\frac18
        \Bigr)^{k/2}
        \leq 2^k\cdot 8^{-k/2} = 2^{-k/2} :
        \]
        أي تركيزا أسيا دون استعمال أي شيء يتجاوز التباينات.
  \item (بالي--زيغموند) من أجل $X \geq 0$ ذي عزم من الرتبة الثانية،
        برهن على $\P(X > 0) \geq \dfrac{\E(X)^2}{\E(X^2)}$ \emph{(بمتراجحة كوشي--شوارتز على $X\mathbf 1_{X>0}$)}:
        وهي الأداة في الاتجاه المعاكس --- فالعزوم قد تفرض وقوع
        \omterm{def:b2:proba:space}{الأحداث} أيضا.
  \item (بينسكر مخففة) بيّن $I(a) \geq 2\bigl(a -
        \tfrac12\bigr)^2$ على $\intoo{\frac12}1$ \emph{(فالفرق ينعدم
        حتى الرتبة الثانية عند $\frac12$ ومشتقته الثانية $\frac1{a(1-a)} - 4 \geq
        0$)}: أي
        إن أسّ تشيرنوف المضبوط يتفوق دائما على أسّ هوفدينغ التربيعي.
  \item انشر $I\bigl(\tfrac12 + \delta\bigr) = 2\delta^2 +
        O(\delta^4)$ واجمعه مع السؤال 4: ففي حالة الانحرافات
        الصغيرة يكون أسّ هوفدينغ $2n\delta^2$ \emph{مضبوطا} مقاربا ---
        فلا طريقة تتفوق عليه بأكثر من عوامل كثيرة الحدود.
  \item ارسم جدول العدة: من أجل ماركوف، وتشيبيشيف، وحصر العزم من
        الرتبة الرابعة في \cref{exo:b2:randomvar:9}، وهوفدينغ، وتشيرنوف بالأسّ $I$،
        اذكر في سطر واحد لكل منها: الفرضية المطلوبة، والتلاشي
        المحصل، والسؤال في هذه المسألة الذي كان فيه أدق ما يكون.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- العوائد.}
\begin{enumerate}[resume]
  \item (اختبار قطعة نقود) قطعة نقود إما متزنة وإما منحازة بالوسيط
        $p = 0.55$. ترميها $n$ مرة وتعلن ``منحازة'' عندما $\widehat p_n > 0.525$.
        بيّن أن احتمالي الخطأ كليهما لا يتجاوز $\eu^{-2n(0.025)^2}$، وأن $n \geq 3685$
        رمية تضمن أن يكونا كلاهما دون $1\%$.
  \item (\omterm{def:b2:proba:space}{الأحداث} النادرة تحتاج إلى حصر يعي \omterm{def:b2:randomvar:variance}{التباين}) ليكن $p =
        0.01$
        وخذ المواصفة النسبية $\delta =
        p/2 = 0.005$، $\alpha = 0.05$. قارن أحجام العينات
        التي تقتضيها متراجحة هوفدينغ ($n \approx 74\,000$) وتلك التي تقتضيها
        متراجحة تشيبيشيف \omterm{def:b2:randomvar:variance}{بالتباين} الحقيقي $p(1-p)$ ($n
        \approx 7920$): فالحصر
        الأسي الأعمى عن \omterm{def:b2:randomvar:variance}{التباين} يخسر أمام العزم الثاني المتواضع.
        اذكر العبرة، ومن أين ستأتي الأداة الناقصة (حصر أسي يعي
        \omterm{def:b2:randomvar:variance}{التباين}؛ وتقريب بواسون في \cref{ch:b2:genfun}).
  \item (القانون القوي من أجل قطع النقود) انطلاقا من $\sum_n
        2\eu^{-2n\delta^2} < \infty$
        وبوريل--كانتيلي (\cref{thm:b2:proba:borelcantelli})، برهن على أن $\widehat p_n \to p$ بشكل شبه أكيد
        من أجل رميات مستقلة لقطعة نقود: صُغ \omterm{def:b2:proba:space}{الحدث} شبه الأكيد على
        الصورة $\bigcap_j\bigcup_N\bigcap_{n\geq N}
        \{\abs{\widehat p_n - p} < \tfrac1j\}$ كما في \cref{exo:b2:randomvar:9}، واختم. (فالحصر يحل محل العزم
        من الرتبة الرابعة المستعمل هناك.)
  \item تركيب. في خمس جمل: ماذا تكلف وماذا تشتري كل درجة من درجات
        السلّم (العزوم الأول والثاني والرابع؛ والأسي المحدود؛ والأسّ
        المضبوط)؛ ولماذا يكفي استطلاع $2050$ شخص من أجل بلد أيا يكن
        حجمه؛ وأي هذه الحصور سيشحذه مجلد السنة الثالثة ليصير الثوابت
        المضبوطة في مبرهنة النهاية المركزية.
\end{enumerate}
\end{problem}
