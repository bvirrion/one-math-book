\chapter{متسلسلات القوى}\label{ch:b2:powerseries}

متسلسلات القوى أحسنُ متسلسلات الدوال سلوكًا في
الرياضيات: فهي تتقارب داخل قرص تقاربها
\omterm{def:b2:funcseq:series}{ناظميًا} على المتراصات، ويمكن اشتقاقها ومكاملتها حدًّا
حدًّا بلا تردد، ومجاميعها --- أي \emph{الدوال
\omterm{def:b2:powerseries:analytic}{التحليلية}} --- تتحدد بمعاملاتها. ويبرهن هذا الفصل
على تلك الحزمة كاملةً ويستعيد، بأمانة، كل متسلسلة
تايلور في مجلد السنة الأولى؛ وتختمه الدوال المولّدة
بأرباح جبرية.

\section{نصف قطر التقارب}

\begin{lemma}[آبل]\label{lem:b2:powerseries:abel}
إذا كانت المتتالية $(a_n z_0^n)$ محدودة من أجل $z_0 \neq 0$ ما،
فإن $\sum a_n z^n$ تتقارب \omterm{def:b2:series:def}{بإطلاق} من أجل كل $\abs z < \abs
{z_0}$، \omterm{def:b2:funcseq:series}{وناظميًا} على كل قرص $\abs z \leq r < \abs{z_0}$.
\end{lemma}

\begin{proof}
مع $\abs{a_n z_0^n} \leq M$ و$\abs z \leq r$:
\[
\abs{a_n z^n} = \abs{a_n z_0^n}\,\Bigl|\frac{z}{z_0}\Bigr|^n
\leq M\Bigl(\frac{r}{\abs{z_0}}\Bigr)^{\!n},
\]
وهو حدّ هندسي متقارب، منتظم على القرص.
\end{proof}

\begin{definition}[نصف قطر التقارب]\label{def:b2:powerseries:radius}
\emph{نصف قطر التقارب}\index{نصف قطر التقارب} للمتسلسلة
$\sum a_n z^n$ هو
\[
R = \sup\{r \geq 0 : (a_n r^n) \text{ محدودة}\} \in
\intcc{0}{+\infty} .
\]
وحسب \cref{lem:b2:powerseries:abel}: التقارب المطلق من أجل $\abs
z < R$ (وناظمي على الأقراص الجزئية المتراصة)، والتباعد --- بل عدم
تحدّد الحدود --- من أجل $\abs z > R$. وعلى الدائرة الحدّية، يمكن أن يحدث
أي شيء (\cref{exo:b2:powerseries:2}). وعمليًا يُحسب $R$
بمحك النسبة عند دالمبير على $\abs{a_n}\abs z^n$ أو بالمقارنة.
\end{definition}

\begin{example}[نصف قطر بلا محك نسبة]\label{ex:b2:powerseries:sinn}
ما نصف قطر $\sum \sin(n)\,z^n$؟ ليس للنسبة
$\abs{\sin(n+1)/\sin n}$ نهاية، لكن التعريف يعمل
مباشرةً. \emph{$R \geq 1$:} $\abs{\sin n} \leq 1$، ومن ثم تكون $(\sin
n\cdot r^n)$ محدودة من أجل كل $r < 1$ --- بل من أجل $r =
1$. \emph{$R \leq 1$:} يكفي أن يكون $\sin n \not\to 0$.
ولنفترض $\sin n \to 0$؛ فصيغة الجمع
\[
\sin(n+1) = \sin n\cos 1 + \cos n\sin 1
\]
كانت ستفرض $\cos n \to 0$ (بالحلّ من أجل $\cos n$: $\sin 1 \neq
0$)، وهذا يناقض $\sin^2 n + \cos^2 n = 1$. ومن ثم فالحدود
$\sin(n)\,1^n$ لا تؤول إلى $0$: فتتباعد المتسلسلة عند $z =
1$، ويكون $R \leq 1$. والخلاصة: $R = 1$. والفكرة الختامية:
نصف القطر قولٌ في كون $\abs{a_n}r^n$ محدودة ---
ولا يُطلب أبدًا وجود نهاية للنسب، وتحسم حجج
الحدّية حالات لا يستطيع محك النسبة أن يلمسها (وقارِن
بالمعاملات المتذبذبة في \cref{exo:b2:powerseries:1}).
\end{example}

\begin{proposition}[العمليات]\label{prop:b2:powerseries:operations}
لتكن $\sum a_nz^n$ و$\sum b_nz^n$ بنصفَي قطر $R_a, R_b$. عندئذٍ، من أجل
$\abs z < \min(R_a, R_b)$:
\[
\sum (a_n + b_n)z^n = \sum a_nz^n + \sum b_nz^n,
\qquad
\Bigl(\sum a_nz^n\Bigr)\Bigl(\sum b_nz^n\Bigr) = \sum c_n z^n,
\quad c_n = \sum_{k=0}^{n} a_kb_{n-k},
\]
وللمتسلسلتين نصف القطر $\geq \min(R_a, R_b)$. (والجداء هو
\omterm{thm:b2:series:fubini}{جداء كوشي}، وهو مشروع بالتقارب المطلق وبالمبرهنة
\cref{thm:b2:series:fubini}.)
\end{proposition}

\begin{proof}
صيغة المجموع هي خطية المتسلسلات المتقاربة، و
$(a_n + b_n)r^n$ محدودة كلما كانت $a_nr^n$ و$b_nr^n$
محدودتين: فنصف القطر $\geq \min(R_a, R_b)$. وأما الجداء: فنثبّت $\abs z
< \min(R_a, R_b)$: فتتقارب المتسلسلتان كلتاهما \emph{\omterm{def:b2:series:def}{بإطلاق}}
هناك (\cref{lem:b2:powerseries:abel})، ومن ثم تكون العائلة ذات الدليلين
$(a_kz^k\,b_lz^l)_{k,l}$ \omterm{def:b2:series:summable}{قابلة للجمع}، وتسمح
\cref{thm:b2:series:fubini} بأي تجميع. وبالتجميع حسب $k
+ l = n$:
\[
\Bigl(\sum_k a_kz^k\Bigr)\Bigl(\sum_l b_lz^l\Bigr)
= \sum_{n\geq0}\Bigl(\sum_{k+l=n}a_kb_l\Bigr)z^n
= \sum_{n\geq0}c_nz^n ,
\]
وهي متقاربة \omterm{def:b2:series:def}{بإطلاق} من أجل كل $z$ كهذا: ومن ثم فلمتسلسلة الجداء
نصف القطر $\geq \min(R_a, R_b)$ أيضًا.
\end{proof}

\begin{example}[مربّع كوشي، متحقَّق منه متقاطعًا]\label{ex:b2:powerseries:cauchysquare}
لنربّع المتسلسلة الهندسية: من أجل $\abs x < 1$، معامل
$x^n$ في $\bigl(\sum x^k\bigr)^2$ هو $c_n = \sum_{k+l=n}
1\cdot1 = n + 1$، ومنه
\[
\frac{1}{(1-x)^2} = \sum_{n\geq0}(n+1)\,x^n .
\]
وللتحقق المتقاطع بالاشتقاق حدًّا حدًّا
(\cref{thm:b2:powerseries:calculus} أدناه): يعطي اشتقاق
$\frac{1}{1-x} = \sum x^n$ أن $\frac{1}{(1-x)^2} = \sum
nx^{n-1} = \sum(n+1)x^n$ --- أي المتسلسلة نفسها بآليتين
غير مترابطتين. والفكرة الختامية: حين تبدو متطابقة معاملات
غامضة، يُنتجها عادةً أحد هذين المحرّكين (الالتفاف أو
الاشتقاق) في سطر واحد؛ ويشغّل سؤال مسألة نهاية الأسبوع عن
$\sum\binom{2k}k\binom{2n-2k}{n-k} = 4^n$ محرّك الالتفاف بكامل طاقته.
\end{example}

\begin{example}[الضرب في $\frac{1}{1-x}$ يجمع المعاملات]\label{ex:b2:powerseries:partialsums}
\omterm{thm:b2:series:fubini}{لجداء كوشي} في مقابل المتسلسلة الهندسية معنى جدير
بالتذكّر: فمن أجل أي $\sum a_nx^n$ نصف قطرها $R > 0$ ومن أجل $\abs x <
\min(R, 1)$،
\[
\frac{1}{1-x}\sum_{n\geq0}a_nx^n
= \sum_{n\geq0}\Bigl(\sum_{k=0}^{n}a_k\Bigr)x^n :
\]
فالضرب في $\frac{1}{1-x}$ يعوّض المعاملات
بمجاميعها الجزئية (أي الالتفاف مع متتالية الواحدات).
ومثال: $\dfrac{\eu^x}{1-x} = \sum_n s_n x^n$ مع $s_n =
\sum_{k\leq n}\frac{1}{k!}$، وهي المجاميع الجزئية للمقدار $\eu$ ---
وقارِن بالمثال \cref{exo:b2:powerseries:11}، حيث يرمّز الجداء نفسه
مع $\eu^{-x}$ أعدادَ الاختلالات. والفكرة الختامية: العمليات على متسلسلات
القوى عملياتٌ على متتاليات المعاملات متخفية (فالضرب في $\frac1{1-x}$:
جمع؛ والضرب في $x$: إزاحة؛ والاشتقاق: ضرب في $n$
وإزاحة) --- وهو معجم سيقرؤه فصل الدوال المولّدة
بطلاقة.
\end{example}

\section{انتظام المجموع}

\begin{theorem}[الحساب حدًّا حدًّا]\label{thm:b2:powerseries:calculus}
لتكن $f(x) = \sum_{n\geq0} a_n x^n$ بنصف قطر $R > 0$ (بمتغيّر
حقيقي $x \in \intoo{-R}{R}$).
\begin{enumerate}
  \item للمتسلسلة المشتقة $\sum n\,a_n x^{n-1}$ نصف القطر $R$ \emph{نفسه}، وتكون $f$ من الصنف $C^1$ مع $f'(x) = \sum_{n \geq 1}
        n a_n x^{n-1}$. وبالتكرار، تكون $f$ من الصنف $C^\infty$ و
        \[
        a_n = \frac{f^{(n)}(0)}{n!} :
        \]
        ومعاملات متسلسلة القوى وحيدة (فمتسلسلتان
        لهما المجموع نفسه قرب $0$ لهما المعاملات نفسها).
  \item الدالة الأصلية حدًّا حدًّا: للمتسلسلة $\sum \frac{a_n}{n+1}x^{n+1}$
        نصف القطر $R$ والمشتق $f$.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{نصف القطر نفسه:} إذا كانت $(a_nr^n)$ محدودة وكان $r' < r$، فإن
$n\abs{a_n} r'^{\,n-1} = \frac{n}{r'}\abs{a_nr^n}
\bigl(\frac{r'}{r}\bigr)^n$ محدودة (بل $\to 0$: فالهندسي
يغلب $n$)، ومنه $R' \geq R$؛ وعكسيًا يعطي $\abs{a_n x^n} \leq \abs x
\cdot n\abs{a_n}\abs x^{n-1}$ أن $R \geq R'$.

\emph{الاشتقاق:} على $\intcc{-r}{r}$، مع $r < R$، تتقارب المتسلسلة
المشتقة \omterm{def:b2:funcseq:series}{ناظميًا} (لأن $n\abs{a_n}r^{n-1}$ \omterm{def:b2:series:summable}{قابلة للجمع} بحساب
نصف القطر)؛ وتتقارب الأصلية عند $x = 0$: فتنطبق مبرهنة
الاشتقاق للمتسلسلات
(\cref{thm:b2:funcseq:seriestransfer}) على كل قطعة كهذه،
ومن ثم على $\intoo{-R}{R}$. وبالتكرار $k$ مرة و
التقييم عند $0$: وبالتحديد، تكون المتسلسلة المشتقة رقم $k$ هي
\[
f^{(k)}(x) = \sum_{n\geq k} n(n-1)\cdots(n-k+1)\,a_n\,x^{n-k},
\]
وعند $x = 0$ ينعدم كل حدّ ذي $n > k$، فلا يبقى إلا
الحدّ الثابت $k(k-1)\cdots1\cdot a_k$: أي $f^{(k)}(0) =
k!\,a_k$. وتنتج وحدانية المعاملات: فلمتسلسلتَي قوى
لهما المجموع نفسه قرب $0$ المشتقاتُ نفسها عند $0$،
ومن ثم المعاملات $a_k$ نفسها. وأما الدوال الأصلية: فنصف القطر نفسه بالحساب
نفسه، ثم نشتق حدًّا حدًّا رجوعًا.
\end{proof}

\begin{example}[تقييم متسلسلة عند نقطة]\label{ex:b2:powerseries:evaluate}
ما قيمة $\sum_{n\geq1}\dfrac{n^2}{2^n}$؟ إنها مجموع
$\sum n^2x^n$ للمقدار \cref{exo:b2:powerseries:3} مقيَّمًا
\emph{داخل} القرص، عند $x = \frac12 < 1 = R$، حيث كانت كل
معالجة استُعملت لاستنباط الصورة المغلقة مشروعة:
\[
\sum_{n\geq1} n^2x^n = \frac{x(1+x)}{(1-x)^3}
\quad\Longrightarrow\quad
\sum_{n\geq1}\frac{n^2}{2^n}
= \frac{\frac12\cdot\frac32}{(\frac12)^3}
= \frac{3/4}{1/8} = 6 .
\]
والمحرّك نفسه بأقراص أخرى: يعطي $x = \frac13$ أن
$\sum\frac{n^2}{3^n} = \frac{\frac13\cdot\frac43}{(2/3)^3} =
\frac32$. والفكرة الختامية: متطابقة متسلسلة قوى
آلةٌ لا صيغة واحدة --- فاستنباط واحد يسعّر كل
متسلسلة عددية $\sum n^2q^n$ دفعةً واحدة، من أجل كل $\abs q < 1$؛
وهكذا سيحسب فصل الدوال المولّدة
الآمال والتباينات بالجملة.
\end{example}

\begin{example}[الكلاسيكيات، هذه المرة بأمانة]\label{ex:b2:powerseries:classics}
$\displaystyle\frac{1}{1 - x} = \sum x^n$ (مع $R = 1$)؛ وبالمكاملة
حدًّا حدًّا (البند (2) في \cref{thm:b2:powerseries:calculus}):
\[
-\ln(1 - x) = \sum_{n\geq1} \frac{x^n}{n},
\qquad
\arctan x = \sum_{n \geq 0} \frac{(-1)^n x^{2n+1}}{2n+1}
\quad (\abs x < 1),
\]
والثانية في خطوتين: نعوّض $-x^2$ في المتسلسلة
الهندسية للحصول على $\frac{1}{1+x^2} = \sum(-1)^nx^{2n}$ (بنصف قطر
$1$، لأن $\abs{x^2} < 1 \iff \abs x < 1$)، ثم نأخذ
الدالة الأصلية حدًّا حدًّا المنعدمة عند $0$؛ فالطرفان
دالتان أصليتان للدالة نفسها بالقيمة نفسها عند $0$،
ومن ثم فهما متساويان على $\intoo{-1}{1}$. وأما $\exp$:
فالمتسلسلة $E(x) = \sum \frac{x^n}{n!}$ (مع $R = \infty$) تحقق
$E' = E$، $E(0) = 1$ بالاشتقاق حدًّا حدًّا، ومنه $E = \exp$ بوحدانية
السنة الأولى. فكل ``نشر قياسي'' في مجلد السنة
الأولى صار الآن مبرهنة عن متسلسلة قواه الكاملة.
\end{example}

\begin{example}[لوغاريتم محسوب من داخل القرص]\label{ex:b2:powerseries:lntwo}
بتقييم $-\ln(1-x) = \sum\frac{x^n}{n}$ عند النقطة
الداخلية $x = \frac12$:
\[
\sum_{n\geq1}\frac{1}{n\,2^n} = \ln 2 ,
\]
وهو تمثيل سريع التقارب للعدد $\ln 2$ (فعشرة حدود تعطي بالفعل
$0.69306\ldots$ في مقابل $\ln 2 = 0.69314\ldots$)، وهو أفضل بكثير
من المتسلسلة المتناوبة $1 - \frac12 + \frac13 -
\dots$ المتاحة عند الحدّ فقط. والفكرة الختامية:
كلما أمكن بلوغ ثابت عند الحافة وداخل القرص تمامًا معًا،
فاز الداخل عدديًا --- تناقص هندسي في مقابل
تناقص توافقي.
\end{example}

\begin{example}[الاشتقاق يحفظ نصف القطر لا الحدّ]\label{ex:b2:powerseries:boundaryloss}
للمتسلسلة $\sum_{n\geq1}\frac{x^n}{n^2}$ نصف القطر $1$ وهي
تتقارب عند \emph{كلا} الطرفين ($\sum\frac1{n^2}$ وتوأمها
المتناوب). ولمتسلسلتها المشتقة،
\[
\sum_{n\geq1}\frac{x^{n-1}}{n} ,
\]
نصف القطر نفسه $1$ --- كما تضمن
\cref{thm:b2:powerseries:calculus} --- لكنها الآن
تتباعد عند $x = 1$ (المتسلسلة التوافقية) وما تزال تتقارب
عند $x = -1$ (متناوبة). واشتقاق آخر يعطي
$\sum_{n\geq2}\frac{n-1}{n}x^{n-2}$، وهي متباعدة عند الطرفين
(فالحدود لا تؤول إلى $0$). والفكرة الختامية: يضرب كل
اشتقاق المعاملات في $n$، وهذا لا يحرّك نصف القطر أبدًا
(فالهندسي يغلب كثير الحدود) لكنه يأكل
رتبة واحدة من التناقص عند الحدّ؛ فالحساب حدًّا حدًّا
رياضة داخلية، وكل ما يحدث على الحافة يجب
إعادة فحصه --- ونظرية آبل--تاوبر في مسألة نهاية الأسبوع هي
بالضبط إعادة الفحص تلك.
\end{example}

\begin{example}[فصل متسلسلة حسب البواقي --- منفَّذًا إلى النهاية]\label{ex:b2:powerseries:quarter}
لنحسب $f(x) = \sum_{n\geq0} \dfrac{x^{4n}}{(4n)!}$ بصورة
مغلقة. لكلٍّ من $\cosh x = \sum \frac{x^{2m}}{(2m)!}$ و$\cos x =
\sum \frac{(-1)^m x^{2m}}{(2m)!}$ نصف القطر $\infty$، ومن ثم يمكن حساب
متوسطهما حدًّا حدًّا:
\[
\frac{\cosh x + \cos x}{2}
= \sum_{m\geq0}\frac{1 + (-1)^m}{2}\,\frac{x^{2m}}{(2m)!}
= \sum_{m \text{ زوجي}}\frac{x^{2m}}{(2m)!}
= \sum_{n\geq0}\frac{x^{4n}}{(4n)!} = f(x) .
\]
ويُبقي المرشّح $\frac{1+(-1)^m}{2}$ القيم الزوجية للدليل $m$ بالضبط: وهذه
هي الصورة الحقيقية لمرشّح جذور الوحدة (والنسخة العقدية،
مع $\iu^n$، تستخرج البواقي بترديد $4$ دفعةً
واحدة). وللتحقق الختامي: $f$ تحلّ $f'''' = f$ مع $f(0) = 1$،
$f'(0) = f''(0) = f'''(0) = 0$ --- فاشتق المتسلسلة أربع مرات
(\cref{thm:b2:powerseries:calculus}) وراقبها وهي
تعيد إنتاج نفسها؛ و$\frac{\cosh + \cos}{2}$ تحقق المعطيات
نفسها.
\end{example}

\begin{definition}[الدوال التحليلية]\label{def:b2:powerseries:analytic}
تكون $f$ \emph{تحليلية}\index{دالة تحليلية} عند $x_0$ إذا كانت
مجموع متسلسلة قوى في $(x - x_0)$ على جوار؛ وعلى
فترة، إذا كانت كذلك عند كل نقطة. ومجاميع متسلسلات القوى تحليلية
داخل قرصها (بإعادة ترتيب النشر --- ونقبل ذلك دون برهان في هذا
المستوى من أجل إعادة التمركز، والحالة $x_0 = 0$ هي
\cref{thm:b2:powerseries:calculus}). والتحليلية تستلزم الصنف $C^\infty$؛
والعكس \emph{يفشل}: كالدالة المسطحة $\eu^{-1/x^2}$
(\cref{exo:b2:powerseries:7}).
\end{definition}

\begin{example}[إعادة التمركز، ونصف القطر بوصفه مسافة]\label{ex:b2:powerseries:recenter}
لننشر $f(x) = \frac{1}{1-x}$ حول $x_0 = \frac12$: فبكتابة
$x = \frac12 + h$،
\[
\frac{1}{1 - x} = \frac{1}{\frac12 - h}
= \frac{2}{1 - 2h}
= \sum_{n\geq0} 2^{n+1}\,h^n
= \sum_{n\geq0} 2^{n+1}\Bigl(x - \frac12\Bigr)^{\!n},
\]
وهو صالح من أجل $\abs{2h} < 1$، أي $\abs{x - \frac12} < \frac12$.
ونصف القطر الجديد هو بالضبط المسافة من المركز الجديد إلى
الشذوذ $x = 1$: فإعادة التمركز تقلّص (أو توسّع)
القرص ليلائم أقرب عائق. والفكرة الختامية: هذه هي
الصورة وراء تعريف \omterm{def:b2:powerseries:analytic}{التحليلية} --- دالة
واحدة، ومتسلسلات قوى محلية كثيرة، تعيش كلٌّ منها على أكبر
قرص يتجنب المشكلة؛ ويحوّل مجلد السنة الثالثة
الحدسَ ``نصف القطر $=$ المسافة إلى أقرب شذوذ
عقدي'' إلى مبرهنة.
\end{example}

\begin{remark}[مزالق شائعة]
\emph{(1) محك النسبة كافٍ لا لازم:} فحين لا تكون
للنسبة $\abs{a_{n+1}/a_n}$ نهاية
(\cref{ex:b2:powerseries:sinn}، \cref{exo:b2:powerseries:1})،
عُد إلى التعريف: $R = \sup\{r : (a_nr^n)$
محدودة$\}$. \emph{(2) لا شيء يعبر الحدّ مجانًا:}
فالاشتقاق والمكاملة حدًّا حدًّا مبرهنتان
\emph{داخل} القرص المفتوح؛ وعند $\abs x = R$ يجب إعادة فحص
كل متسلسلة (وهذا هو موضوع مسألة نهاية الأسبوع
كله). \emph{(3) نصف قطر المجموع:} المقدار $\min(R_a, R_b)$
حدٌّ أدنى فقط --- فقد تكبّره التلاشيات ($a_n = 1,
b_n = -1$: فالمجموع معدوم تمامًا، ونصف القطر $0$ يساوي $\infty$). \emph{(4)
الصنف $C^\infty$ ليس \omterm{def:b2:powerseries:analytic}{تحليلية}:} فقد تتقارب متسلسلة تايلور المتقاربة
إلى الدالة \emph{الخطأ}
(\cref{exo:b2:powerseries:7})؛ فقبل كتابة $f(x) = \sum
\frac{f^{(n)}(0)}{n!}x^n$، برهن على ذلك --- عبر معادلة تفاضلية
(\cref{met:b2:powerseries:ode})، أو تقدير للباقي، أو
صيغة تكاملية.
\end{remark}

\begin{remark}[أين يُستعمَل هذا]
متسلسلات القوى حصان عمل ثلاثة فصول لاحقة: فيحلّ
فصل المعادلات التفاضلية المعادلاتِ الخطية بحقن
$\sum a_nx^n$ (وهو صندوق الطريقة أدناه، مصنَّعًا)؛ ويحوّل
فصل الدوال المولّدة المتطابقاتِ عن الاحتمالات إلى متطابقات
عن متسلسلات القوى وبالعكس؛ ويجعل مجلد السنة الثالثة
المتغيّر العقدي رسميًا، حيث تصير \omterm{def:b2:powerseries:analytic}{التحليلية} مكافئة لقابلية
الاشتقاق العقدية وتنال ``إعادة التمركز المقبولة'' أعلاه
برهانها الأمين. وتستكشف مسألة نهاية الأسبوع الموضع الوحيد
الذي تصمت فيه مبرهنات هذا الفصل: الحدّ $\abs x = R$ نفسه.
\end{remark}

\begin{method}[النشر عبر معادلة تفاضلية]\label{met:b2:powerseries:ode}
لنشر دالة $f$ في متسلسلة قوى: جد معادلة تفاضلية خطية بمعاملات
كثيرات حدود تحققها $f$؛ واحقن $\sum a_nx^n$؛
وعيّن المعاملات للحصول على تراجع للمقادير $(a_n)$؛ ثم حُلّ، و
تحقق من نصف القطر ومن الشروط الابتدائية. ومثال --- المتسلسلة
الثنائية: تحقق $f(x) = (1+x)^\alpha$ المعادلةَ $(1+x)f' = \alpha f$، $f(0)
= 1$؛ ويعطي الحقن $(n+1)a_{n+1} = (\alpha - n)a_n$، ومنه $a_n =
\binom{\alpha}{n}$، ونصف القطر $1$ (بمحك النسبة)، ويساوي المجموع، الذي يحقق
المعادلة نفسها بالقيمة الابتدائية نفسها، المقدارَ $(1 + x)^\alpha$
بمبرهنة الوحدانية للمعادلات التفاضلية الخطية (مجلد السنة الأولى).
\end{method}

\begin{example}[الطريقة على معادلة مقسورة]\label{ex:b2:powerseries:odeforced}
حُلّ $y' = y + x$، $y(0) = 0$، بمتسلسلات القوى. وبحقن $y =
\sum a_nx^n$:
\[
\sum_{n\geq0}(n+1)a_{n+1}x^n
= \sum_{n\geq0}a_nx^n + x ,
\]
وبتعيين المعاملات: $a_1 = a_0 = 0$، $2a_2 = a_1 + 1
= 1$، و$(n+1)a_{n+1} = a_n$ من أجل $n \geq 2$. ومنه $a_2 =
\frac{1}{2!}$، وبالتراجع، $a_n = \frac{1}{n!}$ من أجل كل
$n \geq 2$: فنصف القطر $\infty$، و
\[
y(x) = \sum_{n\geq2}\frac{x^n}{n!} = \eu^x - 1 - x .
\]
وللتحقق: $y' = \eu^x - 1 = y + x$ و$y(0) = 0$. والفكرة
الختامية: التراجع \emph{هو} المعادلة، معاملًا بمعامل؛
ولا يضطرب حدّ القسر إلا عددًا منتهيًا من المعاملات
الأولى، وبعده يتولى النمط المتجانس ---
وهو ظلّ متقطع لقولنا ``حلّ خاص زائد
حلّ متجانس''.
\end{example}

\section{الدوال المولّدة}

\begin{example}[فيبوناتشي]\label{ex:b2:powerseries:fibonacci}
لتكن $F(x) = \sum_{n\geq0} F_n x^n$ (أعداد فيبوناتشي، مع $F_0 = 0$،
$F_1 = 1$). ويُترجَم التراجع $F_{n+2} = F_{n+1} + F_n$، بالضرب
في $x^{n+2}$ والجمع، إلى
\[
F(x) - x = x\,F(x) + x^2 F(x)
\quad\Longrightarrow\quad
F(x) = \frac{x}{1 - x - x^2} ,
\]
وهو صالح حيث تتقارب المتسلسلة. ونصف القطر هو
$\frac{1}{\varphi}$: فمن $F_n \sim
\frac{\varphi^n}{\sqrt5}$ (بينيه، وهو المثال التالي --- أو بالتراجع
الخام $F_n \leq 2^n$ مع التراجع)، يعطي محك النسبة أن
\[
\frac{F_{n+1}\abs x^{n+1}}{F_n\abs x^n}
\longrightarrow \varphi\abs x ,
\qquad\text{التقارب إذا وفقط إذا } \abs x < \frac1\varphi
\approx 0.618 .
\] والكسور الجزئية على
$\frac{x}{1 - x - x^2}$ مع المتسلسلة الهندسية تعيدان استنباط صيغة
بينيه --- فالدوال المولّدة تصنّع التراجعات
الخطية.\index{دالة مولّدة}
\end{example}

\begin{example}[صيغة بينيه، منفَّذةً]\label{ex:b2:powerseries:binet}
ليكن $\varphi = \frac{1+\sqrt5}{2}$ و$\psi =
\frac{1-\sqrt5}{2}$، وهما جذرا $X^2 = X + 1$؛ وبما أن $\varphi
+ \psi = 1$ و$\varphi\psi = -1$،
\[
1 - x - x^2 = (1 - \varphi x)(1 - \psi x) .
\]
والكسور الجزئية: بالبحث عن $\frac{x}{(1-\varphi x)(1-\psi x)} =
\frac{A}{1 - \varphi x} + \frac{B}{1 - \psi x}$، يعطي الحدّ
الثابت $A + B = 0$ ويعطي معامل $x$ أن $-A\psi - B\varphi
= 1$، ومنه $A(\varphi - \psi) = 1$: أي $A = \frac{1}{\sqrt5} = -B$.
وبعد متسلسلتين هندسيتين،
\[
F(x) = \frac{1}{\sqrt5}\sum_{n\geq0}
\bigl(\varphi^n - \psi^n\bigr)x^n
\quad\Longrightarrow\quad
F_n = \frac{\varphi^n - \psi^n}{\sqrt5}
\]
بوحدانية المعاملات
(\cref{thm:b2:powerseries:calculus}). وبما أن $\abs\psi < 1$، فإن
الحدّ $\frac{\psi^n}{\sqrt5}$ قيمته المطلقة $< \frac12$:
أي إن $F_n$ هو \emph{أقرب عدد صحيح} إلى
$\frac{\varphi^n}{\sqrt5}$. والفكرة الختامية: نصف القطر
$\frac1\varphi$ للدالة $F$ هو مقلوب الجذر المهيمن
--- فنموّ المعاملات \omterm{def:b2:powerseries:radius}{ونصف قطر التقارب} هما المعلومة
نفسها مقروءةً في اتجاهين متعاكسين.
\end{example}

\begin{example}[أعداد كاتالان]\label{ex:b2:powerseries:catalan}
تحقق \omterm{ex:b2:powerseries:catalan}{أعداد كاتالان} $C_n$ (وهي عدد المثلثات، والتقويسات،
ومسارات دايك، \dots) العلاقتين $C_0 = 1$ و$C_{n+1} = \sum_{k=0}^n
C_kC_{n-k}$. وعندئذٍ تحقق \omterm{ex:b2:powerseries:fibonacci}{الدالة المولّدة} $C(x) = \sum C_nx^n$
(\omterm{thm:b2:series:fubini}{بجداء كوشي}!)
\[
C(x) = 1 + x\,C(x)^2
\quad\Longrightarrow\quad
C(x) = \frac{1 - \sqrt{1 - 4x}}{2x} ,
\]
وباختيار الجذر الذي $C(0) = 1$: يعطي حلّ المعادلة من الدرجة الثانية
$xC^2 - C + 1 = 0$ المرشحين $\frac{1 \pm
\sqrt{1-4x}}{2x}$، وحين $x \to 0$ ينفجر الجذر ذو ``$+$''
مثل $\frac1x$ في حين أن الجذر ذا ``$-$'' يؤول إلى $1$ (بنشر
$\sqrt{1-4x} = 1 - 2x + O(x^2)$) --- فالإشارة السالبة وحدها يمكن أن
تحمل متسلسلة قوى مع $C_0 = 1$. وبنشر $\sqrt{1 - 4x}$
بالمتسلسلة الثنائية نحصل على الصورة المغلقة
\[
C_n = \frac{1}{n+1}\binom{2n}{n} ,
\]
المنفَّذة في \cref{exo:b2:powerseries:8}.\index{أعداد كاتالان}
\end{example}

\begin{remark}[المتسلسلات الشكلية في مقابل المتقاربة]
كل حساب بالدوال المولّدة أعلاه يُختم باستدعاء
وحدانية المعاملات، وتلك المبرهنة تعيش
\emph{داخل} قرص ذي نصف قطر موجب: فقبل ``قراءة''
$F_n$ أو $C_n$، يجب أن نعرف أن $R > 0$. ويكفي حدّ قبلي خام
--- فالحدّ $F_n \leq 2^n$ (بتراجع مباشر) يعطي $R \geq
\frac12$ من أجل فيبوناتشي؛ و$C_n \leq 4^n$ (فكل عدد كاتالان
يعدّ أجزاءً من المسارات) يعطي $R \geq \frac14$. واحذر
الطرف المنحلّ من السلّم: فللمتسلسلة $\sum n!\,x^n$ نصف القطر $0$، و
معالجتها بوصفها دالة بلا معنى --- والمتطابقات
التي تشمل متسلسلات كهذه تنتمي إلى الحساب \emph{الشكلي}
للمعاملات، وهي لعبة جبرية محضة بقواعدها (المختلفة).
وفي هذا المستوى: أمّن دائمًا نصف قطر موجب أولًا،
ثم احسب بحرية داخله.
\end{remark}

\begin{remark}[آفاق داخل هذا المجلد]
متسلسلات القوى إحدى آلتَي النشر العظيمتين في هذا
الكتاب؛ والأخرى هي متسلسلات فورييه في الفصول
التوافقية، والمقارنة بينهما مفيدة. فمتسلسلة القوى
صلبة: معاملاتها مفروضة ($a_n = f^{(n)}(0)/n!$)،
وتقاربها لا يرحم (ناظمي في الداخل، ميؤوس منه في الخارج)،
ومجموعها تحليلي --- أي صلب بلا نهاية
(\cref{def:b2:powerseries:analytic}). أما متسلسلة فورييه فليّنة:
فهي تمثّل مجرد إشارات ملساء على قطع، بثمن
أسئلة تقارب دقيقة عند حدود الملاسة. وتلتقي النظريتان في
مسألة نهاية الأسبوع في هذا الفصل: فجمع تشيزارو وآبل، المطوَّران هنا من أجل
الدائرة الحدّية، يعودان في فصل فورييه بوصفهما نواتَي
فييير وبواسون. وفي الأثناء يستهلك فصل المعادلات
التفاضلية متسلسلات القوى مباشرةً ($\eu^{tA}$، والحلول
بالمتسلسلات)، ويحوّل فصل الدوال المولّدة
حيلةَ \cref{ex:b2:powerseries:fibonacci} إلى حساب منهجي
للاحتمالات.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:powerseries:1}
أنصاف أقطار التقارب:
$\sum \dfrac{n^2}{2^n}z^n$؛ و$\;\sum \dfrac{z^n}{\binom{2n}{n}}$؛
و$\;\sum z^{n!}$؛ و$\;\sum \bigl(2 + (-1)^n\bigr)^n z^n$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:powerseries:2}
بيّن أن $\sum z^n$ و$\sum \frac{z^n}{n}$ و$\sum \frac{z^n}{n^2}$
لها كلها نصف القطر $1$ لكنها تسلك سلوكًا مختلفًا عند $z = 1$ و$z = -1$:
تباعد/تباعد، وتباعد/تقارب،
وتقارب/تقارب.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:powerseries:3}
احسب المجاميع، من أجل $\abs x < 1$:
\[
\sum_{n\geq0} n x^n,
\qquad
\sum_{n\geq0} n^2 x^n,
\qquad
\sum_{n\geq0} \frac{x^{2n+1}}{2n+1} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:powerseries:4}
انشر في متسلسلة قوى عند $0$، مع نصف القطر: $\dfrac{1}{(1-x)(2-x)}$
(بالكسور الجزئية)؛ و$\;\ln(1 + x + x^2)$ \emph{(اكتب $1 + x + x^2
= \frac{1 - x^3}{1 - x}$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:powerseries:5}
برهن على أن $f(x) = \sum_{n\geq1} H_n x^n = -\dfrac{\ln(1 -
x)}{1 - x}$ من أجل $\abs x < 1$، حيث $H_n$ هو العدد التوافقي
\emph{(\omterm{thm:b2:series:fubini}{جداء كوشي} للمتسلسلتين $\sum x^n$ و$\sum \frac{x^n}{n}$)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:powerseries:6}
حُلّ \omterm{ex:b2:powerseries:fibonacci}{بالدالة المولّدة} التراجعَ $u_0 = 1$، $u_{n+1} =
2u_n + n$: احسب $U(x) = \sum u_nx^n$ بصورة مغلقة، وفكك،
واقرأ $u_n = 2^{n+1} - n - 1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:powerseries:7}
لتكن $f(x) = \eu^{-1/x^2}$ من أجل $x \neq 0$، و$f(0) = 0$. برهن على أن
$f$ من الصنف $C^\infty$ على $\R$ مع $f^{(n)}(0) = 0$ من أجل كل $n$
\emph{(بيّن بالتراجع أن $f^{(n)}(x) = P_n\bigl(\frac1x\bigr)
\eu^{-1/x^2}$ من أجل كثيرات حدود $P_n$، واستعمل مقارنة
النموّ)}. واختم بأن $f$ ليست \omterm{def:b2:powerseries:analytic}{تحليلية} عند $0$: فمتسلسلة تايلور
الخاصة بها عند $0$ تتقارب --- إلى الدالة الخطأ.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:powerseries:8}
أكمل \cref{ex:b2:powerseries:catalan}: انشر $\sqrt{1 - 4x}$
بالمتسلسلة الثنائية، مبيّنًا أن
\[
\binom{1/2}{n+1}(-4)^{n+1} = -\frac{2}{n+1}\binom{2n}{n},
\]
واستنتج $C_n = \frac{1}{n+1}\binom{2n}{n}$؛ وعيّن نصف قطر
تقارب $C(x)$ ومقاربات $C_n$ عبر
ستيرلينغ.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:powerseries:9}
(مبرهنة النهاية القطرية عند آبل، حالة خاصة) لنفترض أن $\sum a_n$
تتقارب. برهن على أن $\lim_{x \to 1^-} \sum_{n} a_n x^n = \sum_n
a_n$. \emph{(\omterm{thm:b2:series:abel}{جمع آبل}: مع $A_n$ المجاميع الجزئية و$A =
\lim A_n$، اكتب $\sum a_nx^n = (1 - x)\sum A_n x^n$؛ ثم $\sum
a_nx^n - A = (1-x)\sum (A_n - A)x^n$، وافصل المجموع عند
$N$ كبير.)} وللتطبيق: $\sum \frac{(-1)^{n-1}}{n} = \ln 2$ و$\sum
\frac{(-1)^n}{2n+1} = \frac\pi4$، مبرهَنتين من جديد انطلاقًا من متسلسلة القوى.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:powerseries:10}
بيّن أن $\displaystyle\sum_{n\geq1}\frac{x^n}{n(n+1)} = 1 +
\frac{1-x}{x}\,\ln(1-x)$ من أجل $0 < \abs x < 1$، وعيّن
نصف القطر، وتحقق من أن \omterm{def:b2:funcseq:series}{التقارب ناظمي} على
$\intcc{-1}{1}$؛ وتحقق من أن القيمة عند $x = 1$ المتنبَّأ بها
بالاتصال توافق المجموع التلسكوبي $\sum
\frac{1}{n(n+1)} = 1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:powerseries:11}
(الاختلالات) ليكن $D_n$ عدد تبديلات $n$
عنصرًا بلا نقطة ثابتة (مع $D_0 = 1$). ويعطي فرز
تبديلات $\{1, \dots, n\}$ حسب مجموعة نقاطها الثابتة أن
$n! = \sum_{k=0}^{n}\binom nk D_{n-k}$. اضرب في
$\frac{x^n}{n!}$، واجمع، وتعرّف على \omterm{thm:b2:series:fubini}{جداء كوشي} للحصول على
\omterm{ex:b2:powerseries:fibonacci}{الدالة المولّدة} الأسّية
\[
\sum_{n\geq0} D_n\,\frac{x^n}{n!} = \frac{\eu^{-x}}{1-x}
\qquad (\abs x < 1),
\]
ثم اقرأ الصورة المغلقة $\dfrac{D_n}{n!} =
\sum_{k=0}^{n}\dfrac{(-1)^k}{k!}$ والنهاية $\dfrac{D_n}{n!}
\to \eu^{-1}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:powerseries:12}
برهن، بالمتسلسلة الثنائية في
\cref{met:b2:powerseries:ode}، على أن
\[
\frac{1}{\sqrt{1 - 4x}} = \sum_{n\geq0}\binom{2n}{n}x^n
\qquad \Bigl(\abs x < \frac14\Bigr),
\]
واستنتج، بالتربيع (\omterm{thm:b2:series:fubini}{بجداء كوشي} في مقابل
$\frac{1}{1-4x} = \sum 4^nx^n$)، متطابقةَ الالتفاف
\[
\sum_{k=0}^{n}\binom{2k}{k}\binom{2n-2k}{n-k} = 4^n .
\]
\end{exercise}

\section{مسألة: آبل وتاوبر وحدّ التقارب}

\begin{problem}\label{pb:b2:powerseries:1}
كل شيء سهل داخل قرص التقارب؛ وكل
دراما متسلسلات القوى تقع \emph{على} الحدّ. وتبني هذه
المسألة نظرية الحدّ في المتغيّر الحقيقي: مبرهنة آبل
بصورتها المنتظمة، وعكسها تحت شرط تاوبر،
وسلّم طرق الجمع عند تشيزارو وآبل
(مع مبرهنة فروبينيوس)، والمكاملة حدًّا حدًّا حتى
الحدّ مع ثوابت كلاسيكية بوصفها أرباحًا، وأخيرًا
صلابة الدوال \omterm{def:b2:powerseries:analytic}{التحليلية} --- أي مبرهنة
التطابق.\index{مبرهنة آبل}\index{مبرهنة تاوبر}
وفي كل ما يلي، $(a_n)$ متتالية حقيقية، و$f(x) = \sum_{n\geq0}
a_nx^n$، و$A_n = a_0 + \dots + a_n$.

\medskip
\noindent\textbf{الجزء الأول --- مبرهنة آبل، بانتظام.} نفترض
في هذا الجزء أن $\sum a_n$ تتقارب، ونضع $r_n =
\sum_{k\geq n} a_k$ (ومنه $r_n \to 0$ و$a_n = r_n -
r_{n+1}$).
\begin{enumerate}
  \item برهن، بالجمع بالتجزئة، على أنه من أجل كل $0 \leq x
        \leq 1$ و$N \leq M$:
        \[
        \Bigl|\sum_{n=N}^{M} a_n x^n\Bigr|
        \leq 2\sup_{n \geq N}\,\abs{r_n} .
        \]
  \item استنتج أن $\sum a_nx^n$ تتقارب \emph{بانتظام} على
        $\intcc{0}{1}$، وأن مجموعها \omterm{def:b2:metric:continuity}{متصل} هناك، و
        استعد النهاية القطرية في
        \cref{exo:b2:powerseries:9}: أي $f(x) \to \sum a_n$ حين $x
        \to 1^-$.
  \item (مبرهنة آبل لجداءات كوشي) لتكن $\sum a_n = A$،
        و$\sum b_n = B$ ولنفترض أن \omterm{thm:b2:series:fubini}{جداء كوشي} $\sum
        c_n$، مع $c_n = \sum_{k} a_kb_{n-k}$، \emph{يتقارب}،
        بالمجموع $C$. برهن على أن $C = AB$ \emph{(داخل القرص تصحّ
        متطابقة الجداء حسب
        \cref{prop:b2:powerseries:operations}؛ ثم خذ $x \to
        1^-$)}.
  \item بيّن أن الفرضية مهمة: فمن أجل $a_n = b_n =
        \frac{(-1)^n}{\sqrt{n+1}}$، تتقارب المتسلسلتان، ومع ذلك
        $\abs{c_n} \geq \frac{2(n+1)}{n+2} \geq 1$ \emph{(حُدّ
        كل عامل $\sqrt{(k+1)(n-k+1)}$ بمتراجحة الحسابي والهندسي)}: فقد
        يتباعد \omterm{thm:b2:series:fubini}{جداء كوشي} لمتسلسلتين متقاربتين.
  \item (ربح من \cref{exo:b2:powerseries:5}) بيّن أن
        $\bigl(\ln(1-x)\bigr)^2 = 2\sum_{n\geq1}
        \frac{H_n}{n+1}x^{n+1}$ على $\intoo{-1}{1}$، وتحقق من أن
        $\bigl(\frac{H_n}{n+1}\bigr)_{n\geq1}$ تتناقص إلى
        $0$، واختم بآبل:
        \[
        \sum_{n\geq1} (-1)^{n+1}\,\frac{H_n}{n+1}
        = \frac{(\ln 2)^2}{2} .
        \]
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- عكس تاوبر.} نقول إن $\sum
a_n$ \emph{\omterm{def:b2:series:summable}{قابلة للجمع} بمعنى آبل} إلى $L$ إذا كان $f(x) \to L$ حين $x \to
1^-$.
\begin{enumerate}[resume]
  \item بيّن أن $\sum (-1)^n$ \omterm{def:b2:series:summable}{قابلة للجمع} بمعنى آبل إلى $\frac12$
        ومع ذلك متباعدة: فليس لمبرهنة آبل أي عكس
        غير مشروط.
  \item (مبرهنة تشيزارو المساعدة) إذا كان $u_n \to 0$ فإن $\frac{u_1 + \dots
        + u_N}{N} \to 0$ \emph{(افصل المجموع عند $m$ ثابت)}.
  \item لنفترض الآن $n\,a_n \to 0$ و$f(x) \to L$. ومع $x_N =
        1 - \frac1N$، برهن على التقديرين
        \[
        \Bigl|\sum_{n=0}^{N} a_n\bigl(1 - x_N^n\bigr)\Bigr|
        \leq \frac{1}{N}\sum_{n=1}^{N} n\,\abs{a_n},
        \qquad
        \Bigl|\sum_{n>N} a_n x_N^n\Bigr|
        \leq \sup_{n>N}\bigl(n\abs{a_n}\bigr)
        \]
        \emph{(من أجل الأول، $1 - x^n \leq n(1-x)$؛ ومن أجل
        الثاني، $\abs{a_n} \leq \frac{1}{N}\sup_{m>N}
        m\abs{a_m}$ و$\sum x_N^n \leq N$)}.
  \item اختم \emph{مبرهنة تاوبر}: إذا كان $n\,a_n \to 0$ وكانت
        $\sum a_n$ \omterm{def:b2:series:summable}{قابلة للجمع} بمعنى آبل إلى $L$، فإن $\sum a_n$
        تتقارب إلى $L$.
  \item (المبرهنة التاوبرية السهلة من أجل المعاملات الموجبة) إذا كان $a_n
        \geq 0$ وكانت $f$ محدودة على $\intco{0}{1}$، فبيّن أن
        $\sum a_n$ تتقارب وأن $\sum a_n = \lim_{x\to1^-}
        f(x)$ \emph{(حُدّ $\sum_{n\leq N}a_nx^n \leq f(x)$ و
        خذ $x \to 1^-$، ثم استعمل آبل)}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- متوسطات تشيزارو ومبرهنة
فروبينيوس.} نقول إن $\sum a_n$ \emph{\omterm{def:b2:series:summable}{قابلة للجمع} بمعنى تشيزارو} إلى $L$ إذا كان
$\sigma_N = \frac{A_0 + \dots + A_{N-1}}{N} \to L$.
\begin{enumerate}[resume]
  \item بيّن أن المتسلسلة المتقاربة \omterm{def:b2:series:summable}{قابلة للجمع} بمعنى تشيزارو إلى
        مجموعها \emph{(السؤال 7 مطبَّقًا على $A_n - L$)}.
  \item احسب قيمة تشيزارو للمتسلسلة $\sum(-1)^n$ وتحقق من أنها
        توافق قيمة آبل $\frac12$ في السؤال 6.
  \item مع $S_n = A_0 + \dots + A_n = (n+1)\,\sigma_{n+1}$،
        برهن على المتطابقتين، من أجل $0 \leq x < 1$:
        \[
        f(x) = (1-x)^2\sum_{n\geq0}(n+1)\,\sigma_{n+1}x^n,
        \qquad
        (1-x)^2\sum_{n\geq0}(n+1)x^n = 1 .
        \]
  \item (فروبينيوس) استنتج: إذا كان $\sigma_N \to L$ فإن $f(x) \to
        L$ حين $x \to 1^-$ --- أي إن القابلية للجمع بمعنى تشيزارو تستلزم
        القابلية للجمع بمعنى آبل، بالقيمة نفسها \emph{(اطرح
        المتطابقتين وافصل المجموع عند $N$ كبير، كما في
        \cref{exo:b2:powerseries:9})}.
  \item بيّن أن السلّم
        \[
        \text{متقاربة} \;\Longrightarrow\;
        \text{قابلة للجمع بمعنى تشيزارو} \;\Longrightarrow\;
        \text{قابلة للجمع بمعنى آبل}
        \]
        تامّ عند السهمين معًا: بالسؤال 6 من أجل الأول؛
        ومن أجل الثاني، بيّن أن $\sum(-1)^n(n+1)$
        \omterm{def:b2:series:summable}{قابلة للجمع} بمعنى آبل إلى $\frac14$ (احسب $f$) وليست
        \omterm{def:b2:series:summable}{قابلة للجمع} بمعنى تشيزارو (احسب $\sigma_N$ على حدة من أجل
        $N$ الزوجية والفردية).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- المكاملة حتى الحدّ.}
\begin{enumerate}[resume]
  \item لنفترض أن $\sum a_nx^n$ تتقارب على $\intco{0}{1}$ وأن
        $\sum \frac{a_n}{n+1}$ تتقارب. برهن على أن
        \omterm{def:b2:integration:improper}{التكامل المعتل} $\int_0^1 f$ موجود وأن
        \[
        \int_0^1 \Bigl(\sum_{n\geq0} a_nx^n\Bigr)\dd x
        = \sum_{n\geq0}\frac{a_n}{n+1}
        \]
        \emph{(فالدالة الأصلية $F(x) = \sum\frac{a_n}{n+1}x^{n+1}$
        متصلة عند $1$ حسب الجزء الأول)}.
  \item لتكن $\eta = \sum_{n\geq1}\frac{(-1)^{n-1}}{n^2}$. بيّن أن
        $\int_0^1 \frac{\ln(1+x)}{x}\dd x = \eta$، وبفصل الأدلة الزوجية والفردية في
        المتسلسلة المتقاربة \omterm{def:b2:series:def}{بإطلاق} $\sum \frac1{n^2}$، أن $\eta =
        \frac12\sum_{n\geq1}\frac{1}{n^2}$. (وتقيّم مسألة نهاية الأسبوع
        في فصل فورييه المقدارَ $\sum\frac1{n^2} = \frac{\pi^2}{6}$.)
  \item برهن على أن
        \[
        \sum_{n\geq0}\frac{(-1)^n}{3n+1}
        = \int_0^1\frac{\dd x}{1+x^3}
        = \frac13\Bigl(\ln 2 + \frac{\pi}{\sqrt3}\Bigr)
        \]
        \emph{(تتقارب المتسلسلة بمحك لايبنتز؛ وكامِل
        المتسلسلة الهندسية $\sum(-1)^nx^{3n}$ بالسؤال 16؛
        ثم بالكسور الجزئية: $\frac{1}{1+x^3} =
        \frac{1/3}{1+x} + \frac{(2-x)/3}{x^2-x+1}$)}.
  \item انطلاقًا من المتسلسلة الثنائية للمقدار $(1-t)^{-1/2}$
        (\cref{exo:b2:powerseries:12}) استنبط
        \[
        \arcsin x = \sum_{n\geq0}
        \frac{\binom{2n}{n}}{4^n(2n+1)}\,x^{2n+1}
        \quad(\abs x < 1),
        \qquad\text{ثم}\qquad
        \sum_{n\geq0}\frac{\binom{2n}{n}}{4^n(2n+1)}
        = \frac\pi2 ,
        \]
        مبرّرًا القيمة الحدّية بالتقارب \emph{الناظمي}
        على $\intcc{-1}{1}$ (استعمل
        $\binom{2n}n4^{-n} \sim \frac{1}{\sqrt{\pi n}}$،
        \cref{ex:b2:comparison:centralbinomial}) --- فهنا لا حاجة
        حتى إلى آبل.
  \item (كاتالان عند الحدّ) بيّن أن $\sum C_n 4^{-n} =
        2$: فمتسلسلة كاتالان في
        \cref{ex:b2:powerseries:catalan} تتقارب \emph{عند}
        نصف قطرها $\frac14$ (بمقاربات
        \cref{exo:b2:powerseries:8})، ومجموعها \omterm{def:b2:metric:continuity}{متصل} على
        $\intcc{0}{\frac14}$، وللصورة المغلقة النهاية $2$
        هناك.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- الصلابة: مبرهنة التطابق.}
\begin{enumerate}[resume]
  \item (الأصفار المعزولة) لتكن $f = \sum a_nx^n$ بنصف قطر $R >
        0$ وليست كل $a_n = 0$ معدومة؛ وليكن $m$ أصغر دليل
        يحقق $a_m \neq 0$. بيّن أن $f(x) = x^m g(x)$ مع $g$
        متسلسلة قوى نصف قطرها $R$، مع $g(0) = a_m \neq 0$، و
        استنتج أنه ليس للدالة $f$ أي جذر في جوار مثقوب
        ما للنقطة $0$.
  \item (مبرهنة التطابق) لتكن $f, h$ مجموعَي متسلسلتَي قوى
        قرب $0$ ولتكن $(x_k)$ متتالية من النقاط \emph{غير المعدومة}
        مع $x_k \to 0$ و$f(x_k) = h(x_k)$. برهن
        على أن للدالتين $f$ و$h$ المعاملات نفسها، ومن ثم فهما
        تتطابقان قرب $0$.
  \item جد \emph{كل} الدوال $f$ \omterm{def:b2:powerseries:analytic}{التحليلية} قرب $0$ التي تحقق
        \[
        f\Bigl(\frac1k\Bigr) = \frac{k^2}{k^2+1}
        \qquad\text{من أجل كل عدد صحيح كبير } k .
        \]
  \item بيّن أن \omterm{def:b2:powerseries:analytic}{الدالة التحليلية} على فترة \omterm{def:b2:metric:topology}{مفتوحة} $I$
        والمنعدمة على فترة جزئية تنعدم تمامًا على
        $I$ \emph{(فمجموعة النقاط التي تنعدم حولها $f$
        تمامًا \omterm{def:b2:metric:topology}{مفتوحة}، وهي مغلقة في $I$ بمبرهنة التطابق
        مطبَّقةً عند نقاط التراكم)}.
        واختم بأنه لا توجد \omterm{def:b2:powerseries:analytic}{دالة تحليلية} غير معدومة على $\R$ ذات
        حامل متراص --- في حين توجد دوال نتوء من الصنف $C^\infty$
        (وتوفّر \cref{exo:b2:powerseries:7} لبنة
        البناء): \omterm{def:b2:powerseries:analytic}{فالتحليلية} صلبة، والملاسة
        رخوة.
  \item تركيب. بجملة واحدة لكل بند: (1) ماذا تضيف مبرهنة
        آبل إلى حزمة \omterm{def:b2:funcseq:series}{التقارب الناظمي} في
        \cref{lem:b2:powerseries:abel}؛ (2) وما الفرضيات
        الدقيقة التي يصحّ تحتها العكس (تاوبر) و
        ما الدرجة الوسيطة (فروبينيوس)؛ (3) وأي ثابت حدّي
        من الجزء الرابع تستطيع الآن أن تستنبطه لصديق
        في سطرين؛ (4) وأين ستعود متوسطات تشيزارو في هذا الكتاب،
        من أجل متسلسلات من نوع مختلف تمامًا.
\end{enumerate}
\end{problem}
