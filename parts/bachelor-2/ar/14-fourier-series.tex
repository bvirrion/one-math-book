\chapter{متسلسلات فورييه}\label{ch:b2:fourier}

هل يمكن إعادة بناء كل إشارة دورية من جيوب وجيوب تمام محضة؟
خلق جوابُ فورييه الجريء ``نعم'' قرنًا من التحليل. ويبرهن هذا
الفصل على الركنين اللذين في المتناول في هذا المستوى:
\emph{مبرهنة ديريكليه} (إعادة البناء النقطية للدوال المتصلة على قطع من الصنف $C^1$،
عبر \omterm{lem:b2:fourier:kernel}{نواة ديريكليه}) و\emph{متطابقة بارسفال} (طاقة
الإشارة هي مجموع طاقات توافقياتها)، ويحصد
المتسلسلات العددية الكلاسيكية --- ومقدارُ بازل $\sum 1/n^2 =
\pi^2/6$ أولها.

وفي كل ما يلي، الدوال دورية بالدور $2\pi$، ومتصلة على قطع،
وذات قيم عقدية؛ ويرمز $\mathcal{C}$ إلى المتصلة منها.

\section{معاملات فورييه}

\begin{definition}\label{def:b2:fourier:coefficients}
\emph{معاملات فورييه}\index{معاملات فورييه} للدالة $f$
هي
\[
c_n(f) = \frac{1}{2\pi}\int_{-\pi}^{\pi} f(t)\,\eu^{-\iu n t}\,\dd
t \qquad (n \in \Z),
\]
والمعاملات في الصورة الحقيقية $a_n = c_n + c_{-n}$، $b_n = \iu(c_n
- c_{-n})$، بحيث تكون \emph{المجاميع الجزئية لفورييه}\index{مجاميع فورييه
الجزئية}
\[
S_N(f)(t) = \sum_{n=-N}^{N} c_n(f)\,\eu^{\iu nt}
= \frac{a_0}{2} + \sum_{n=1}^{N}\bigl(a_n\cos nt + b_n \sin
nt\bigr).
\]
وعلى $\mathcal{C}$، نعرّف \omterm{def:b2:hermitian:def}{الجداء السلّمي الهرميتي} $\langle f,
g\rangle = \frac{1}{2\pi}\int_{-\pi}^{\pi}\conj f\,g$: فالأسّيات
$e_n(t) = \eu^{\iu nt}$ \emph{متعامدة ممنظمة}
(لأن $\langle e_m, e_n\rangle = \delta_{mn}$، بالحساب المباشر)، و
$c_n(f) = \langle e_n, f\rangle$: فتحليل فورييه هندسةٌ هرميتية
(\cref{ch:b2:hermitian}) في البعد غير المنتهي.
\end{definition}

\begin{proposition}[متراجحة بيسل]\label{prop:b2:fourier:bessel}
$S_N(f)$ هو الإسقاط المتعامد للدالة $f$ على فضاء
$\mathcal{T}_N$ كثيرات الحدود المثلثية من الدرجة $\leq N$،
ويكون\index{متراجحة بيسل}
\[
\sum_{n=-N}^{N} \abs{c_n(f)}^2 \leq \norm f_2^2 =
\frac{1}{2\pi}\int_{-\pi}^{\pi} \abs f^2 :
\]
فتتقارب المتسلسلة $\sum \abs{c_n}^2$، ويكون $c_n(f) \to 0$ حين
$\abs n \to \infty$ (وهو ريمان--لوبيغ من أجل المعاملات).
\end{proposition}

\begin{proof}
$f - S_N(f)$ متعامدة مع كل $e_k$، مع $\abs k \leq N$ (لأن $\langle
e_k, f - S_N f\rangle = c_k - c_k = 0$): ومن ثم فإن $S_Nf$ هو الإسقاط
المتعامد على $\mathcal{T}_N = \operatorname{Vect}(e_{-N}, \dots,
e_N)$ (وهي مبرهنة الإسقاط في مجلد السنة الأولى، حرفيًا في
الإطار الهرميتي). وبمبرهنة فيثاغورس: $\norm f_2^2 = \norm{S_Nf}_2^2 +
\norm{f - S_Nf}_2^2 \geq \norm{S_Nf}_2^2 = \sum_{\abs n \leq N}
\abs{c_n}^2$؛ ثم نأخذ $N \to \infty$.
\end{proof}

\begin{example}[أفضل تقريب، مقيسًا]\label{ex:b2:fourier:bestapprox}
إلى أي حدّ تقرّب كثيرات الحدود المثلثية ذوات الدرجة الصغيرة
السنَّ المنشارية $f(t) = t$ (على $\intoo{-\pi}{\pi}$) في
المتوسط التربيعي؟ حسب \cref{prop:b2:fourier:bessel} يكون أفضل
تقريب من الدرجة $N$ \emph{هو} $S_N(f)$، بخطأ مربّع
\[
\norm{f - S_Nf}_2^2 = \norm f_2^2 -
\sum_{\abs n\leq N}\abs{c_n}^2 .
\]
وهنا $\norm f_2^2 = \frac{1}{2\pi}\int_{-\pi}^\pi t^2\dd t =
\frac{\pi^2}{3}$، ومن $b_n = \frac{2(-1)^{n+1}}{n}$
(\cref{ex:b2:fourier:basel}): $\abs{c_n}^2 + \abs{c_{-n}}^2 =
\frac{b_n^2}{2} = \frac{2}{n^2}$. ومن ثم
\[
\norm{f - S_Nf}_2^2
= \frac{\pi^2}{3} - \sum_{n=1}^{N}\frac{2}{n^2}
\qquad\text{: وعدديًا } 1.29,\ 0.79,\ 0.57,\ 0.44
\]
من أجل $N = 1, 2, 3, 4$ --- وهو متناقص، لكن ببطء: فالذيل
$\sum_{n>N}\frac2{n^2} \sim \frac2N$ يحكمه التناقص البطيء
$\frac1n$ للمعاملات، وهو نفسه بصمة
القفزة (\cref{exo:b2:fourier:6} مقروءةً في الاتجاه المعاكس). والفكرة
الختامية: يحوّل بارسفال جودة التقريب إلى ذيل
متسلسلة عددية --- ويتنبأ، قبل أي رسم، بأن
القفزات تجعل متسلسلات فورييه تتقارب على مضض.
\end{example}

\begin{method}[كيف نحسب معاملات فورييه بكفاءة]\label{met:b2:fourier:coefficients}
قبل مكاملة أي شيء:
\begin{enumerate}
  \item \emph{التماثل:} إذا كانت $f$ زوجية فإن $b_n = 0$، وإذا كانت $f$ فردية فإن $a_n
        = 0$، وتؤول التكاملات الباقية إلى $\frac2\pi
        \int_0^\pi$ --- أي نصف العمل وضعف الموثوقية.
  \item \emph{كثيرات الحدود المثلثية منجزة أصلًا:}
        خطّط الجداءات ($\cos^3$، $\sin^2\cos$، \dots) و
        اقرأ المعاملات مباشرةً
        (\cref{exo:b2:fourier:9})؛ فالتعامد والتنظيم يجعلان أي
        مكاملة إضافية زائدة.
  \item \emph{الأسّيات العقدية من أجل الأسّيات:} من أجل
        العوامل $\eu^{at}$ أو التذبذبات المخمَّدة، احسب
        $c_n$ مباشرةً --- فتكامل واحد للمقدار $\eu^{(a - \iu
        n)t}$ يغلب مكاملتين بالتجزئة
        (\cref{exo:b2:fourier:10}).
  \item \emph{اشتق نشرًا معلومًا:} إذا كانت معاملات $f'$
        معلومة وكانت $f$ متصلة، فإن $c_n(f) =
        \frac{c_n(f')}{\iu n}$ (مع $n \neq 0$) تستعيد كل شيء إلا
        $c_0$، وهو المتوسط --- وكثيرًا ما يكون هذا أسرع طريق،
        وهو مشروع بالضبط تحت فرضيات
        البند (1) في \cref{thm:b2:fourier:parseval}.
\end{enumerate}
\end{method}

\section{مبرهنة ديريكليه}

\begin{lemma}[نواة ديريكليه]\label{lem:b2:fourier:kernel}
$S_N(f)(x) = \frac{1}{2\pi}\int_{-\pi}^{\pi} f(x + u)\,D_N(u)\,\dd
u$، حيث\index{نواة ديريكليه}
\[
D_N(u) = \sum_{n=-N}^{N} \eu^{\iu nu}
= \frac{\sin\bigl((N + \frac12)u\bigr)}{\sin\frac u2}
\quad (u \notin 2\pi\Z),
\qquad
\frac{1}{2\pi}\int_{-\pi}^{\pi} D_N = 1 .
\]
\end{lemma}

\begin{proof}
نُدخِل تعريف $c_n$ في $S_N$ ونبدّل المجموع والتكامل
(وهذا مشروع: فالمجموع منته):
\[
S_N(f)(x)
= \sum_{n=-N}^{N}\Bigl(\frac{1}{2\pi}\int_{-\pi}^{\pi}
f(t)\,\eu^{-\iu nt}\dd t\Bigr)\eu^{\iu nx}
= \frac{1}{2\pi}\int_{-\pi}^{\pi} f(t)
\sum_{n=-N}^{N}\eu^{\iu n(x-t)}\,\dd t ;
\]
ونعوّض $u = t - x$ ونُزحلق قطعة المكاملة رجوعًا
إلى $\intcc{-\pi}{\pi}$ بدورية الدالة المكامَلة بالدور $2\pi$؛
ويجعل مجال الدليل المتناظر $\sum_n\eu^{-\iu nu} = D_N(u)$.
والصورة المغلقة: مجموع هندسي بنسبة $\eu^{\iu u}$،
\[
D_N(u) = \eu^{-\iu Nu}\,\frac{\eu^{\iu(2N+1)u} - 1}{\eu^{\iu u} -
1}
= \frac{\eu^{\iu(N + \frac12)u} - \eu^{-\iu(N+\frac12)u}}
{\eu^{\iu u/2} - \eu^{-\iu u/2}} ,
\]
وهو خارج قسمة الجيوب. ومتوسطها $1$: فلا يسهم إلا $n = 0$.
\end{proof}

\begin{theorem}[مبرهنة ريمان--لوبيغ المساعدة]\label{thm:b2:fourier:riemannlebesgue}
من أجل $g$ متصلة على قطع في قطعة،
$\int_a^b g(t)\sin(\lambda t + \varphi)\,\dd t \to 0$ حين $\lambda
\to +\infty$.\index{مبرهنة ريمان--لوبيغ المساعدة}
\end{theorem}

\begin{proof}
نقرّب $g$ بانتظام بدوال درجية
(ولا حاجة إلى \cref{thm:b2:funcseq:weierstrass} --- فالتقريب الأوّلي
بالدوال الدرجية للدوال المتصلة على قطع
يكفي) ونكامل كل درجة صراحةً: فتسهم كل قطعة
بالمقدار $O\bigl(\frac1\lambda\bigr)$، ويسهم خطأ التقريب
بالمقدار $\varepsilon(b - a)$. وقد نُفّذت هذه الحجة كاملةً
في التمرين الأخير من فصل التكامل في مجلد
السنة الأولى؛ ومن أجل القطع من الصنف $C^1$ يمكن بدلًا من ذلك المكاملة بالتجزئة
والحدّ بالمقدار $\frac C\lambda$.
\end{proof}

\begin{example}[كم تسرع المعاملات إلى الموت؟]\label{ex:b2:fourier:decaytable}
يقول ريمان--لوبيغ إن المعاملات تؤول إلى $0$؛ و\emph{معدلها}
مقياسُ ملاسة. وثلاث عينات من هذا الفصل وتمارينه:
\[
\text{موجة مربّعة: } b_n = \frac{4}{\pi n}\ (n\ \text{فردي}),
\qquad
\abs t : a_n = \frac{-4}{\pi n^2}\ (n\ \text{فردي}),
\qquad
\abs{\sin t} : a_{2k} = \frac{-4}{\pi(4k^2-1)} .
\]
فالقفزة في $f$ (الموجة المربّعة، والسنّ المنشارية) تترك معاملات من
الرتبة $\frac1n$: فلا \omterm{def:b2:funcseq:series}{تقارب ناظمي}، وتجاوز غيبس عند
القفزات. أما الاتصال مع زاوية --- أي قفزة في $f'$ فقط ---
فيحسّن الرتبة إلى $\frac{1}{n^2}$: أي \omterm{def:b2:funcseq:series}{تقارب ناظمي} و
إعادة بناء منتظمة. وعمومًا تشتري $k$ مشتقات المعدلَ $c_n =
O(n^{-k})$ (\cref{exo:b2:fourier:6})، وعكسيًا يفرض طيفٌ
يتناقص أسرع من كل قوة أن تكون $f$ من الصنف $C^\infty$
(بالاشتقاق حدًّا حدًّا، وهو مشروع الآن). والفكرة
الختامية: انتظام الإشارة وتناقص \omterm{def:b2:reduction:eigen}{الطيف}
هما المعلومة نفسها --- فيقرأ المهندس إحداهما من
ميل الأخرى بلا رسم الدالة أبدًا.
\end{example}

\begin{theorem}[ديريكليه]\label{thm:b2:fourier:dirichlet}
لتكن $f$ دورية بالدور $2\pi$ ومتصلة على قطع من الصنف $C^1$. عندئذٍ من أجل كل $x$،
\[
S_N(f)(x) \xrightarrow[N \to \infty]{}
\frac{f(x^+) + f(x^-)}{2}
\]
(أي متوسط النهايتين من جهة واحدة) --- وعلى الخصوص $S_N(f)(x) \to
f(x)$ عند كل نقطة اتصال.\index{مبرهنة ديريكليه}
\end{theorem}


\begin{proof}
بمبرهنة النواة المساعدة وبكون متوسطها واحدًا، وبفصل التكامل إلى
نصفين $u > 0$ و$u < 0$ (متوسط كلٍّ منهما $\frac12$):
\begin{align*}
S_N(f)(x) - \frac{f(x^+) + f(x^-)}{2}
&= \frac{1}{2\pi}\int_{0}^{\pi} \bigl(f(x+u) -
f(x^+)\bigr)D_N(u)\,\dd u\\
&\quad+ \frac{1}{2\pi}\int_{-\pi}^{0}\bigl(f(x+u) -
f(x^-)\bigr)D_N(u)\,\dd u .
\end{align*}
ونعالج الأول (والثاني متناظر). ونكتب
\[
\bigl(f(x + u) - f(x^+)\bigr)\,D_N(u)
= \underbrace{\frac{f(x+u) - f(x^+)}{\sin\frac u2}}_{g(u)}\,
\sin\Bigl(\Bigl(N + \frac12\Bigr)u\Bigr) .
\]
والدالة $g$ متصلة على قطع في $\intoc{0}{\pi}$ و
لها \emph{نهاية منتهية عند $0^+$}: فبكتابة
\[
g(u) = \frac{f(x+u) - f(x^+)}{u}\cdot\frac{u}{\sin\frac u2} ,
\]
يؤول العامل الأول إلى $f'(x^+)$ (بقابلية الاشتقاق من
جهة واحدة، الآتية من كون الدالة من الصنف $C^1$ على قطع) ويؤول الثاني إلى $2$
(بالنهاية القياسية $\frac{\sin v}{v} \to 1$ عند $v = \frac
u2$): ومن ثم توجد $g(0^+) = 2f'(x^+)$. فتمتدّ $g$ بالاتصال
على قطع إلى $\intcc{0}{\pi}$، وترسل مبرهنة ريمان--لوبيغ المساعدة
(\cref{thm:b2:fourier:riemannlebesgue}) التكاملَ إلى
$0$. وهذا هو كل مغزى الفرضية: فبلا
مشتقات من جهة واحدة، ينفجر العامل $\frac{1}{\sin(u/2)}$
عند $0$ أسرع مما يستطيع ريمان--لوبيغ تعويضه، وقد
يفشل \omterm{def:b2:funcseq:def}{التقارب النقطي} فعلًا من أجل $f$ المتصلة فحسب
--- وهي الفجوة التي تسدّها مبرهنة فييير (مسألة نهاية الأسبوع)
بأخذ المتوسطات.
\end{proof}

\begin{example}[ديريكليه عند قفزة]\label{ex:b2:fourier:jumpvalue}
من أجل السنّ المنشارية $f(t) = t$ على $\intoo{-\pi}{\pi}$
(\cref{ex:b2:fourier:basel} أدناه)، يقفز التمديد الدوري
عند $t = \pi$ من $f(\pi^-) = \pi$ إلى $f(\pi^+) = -\pi$.
وتَعِد مبرهنة ديريكليه بالقيمة $\frac{\pi + (-\pi)}{2} = 0$
هناك، وفعلًا ينعدم كل حدّ من $\sum
\frac{2(-1)^{n+1}}{n}\sin nt$ عند $t = \pi$: فتتقارب
المتسلسلة بأدب إلى النقطة الوسطى، متجاهلةً القيمتين
من جهة واحدة. وبنقل نقطة التقييم إلى $t =
\frac\pi2$ بدلًا من ذلك (وهي نقطة اتصال) تصير المتسلسلة نفسها
متسلسلةَ لايبنتز $\frac\pi4$. متسلسلة واحدة، وسلوكان ---
وهما بالضبط بندا المبرهنة.
\end{example}

\begin{theorem}[التقارب الناظمي من أجل $C^1$؛ بارسفال]\label{thm:b2:fourier:parseval}
\begin{enumerate}
  \item إذا كانت $f$ متصلة ودورية بالدور $2\pi$ ومن الصنف $C^1$ على قطع،
        فإن $c_n(f') = \iu n\,c_n(f)$، وتتقارب متسلسلة فورييه للدالة $f$
        \emph{\omterm{def:b2:funcseq:series}{ناظميًا}} على $\R$، ومجموعها $f$.
  \item (بارسفال) من أجل كل دالة $f$ متصلة على قطع ودورية بالدور $2\pi$:\index{متطابقة بارسفال}
        \[
        \frac{1}{2\pi}\int_{-\pi}^{\pi}\abs{f}^2
        = \sum_{n=-\infty}^{\infty} \abs{c_n(f)}^2
        = \frac{\abs{a_0}^2}{4} + \frac12\sum_{n\geq1}
        \bigl(\abs{a_n}^2 + \abs{b_n}^2\bigr).
        \]
        \emph{(وهي مبرهَنة هنا من أجل $f$ المتصلة من الصنف $C^1$ على قطع؛
        ومقبولة في الحالة العامة.)}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) بالمكاملة بالتجزئة على كل قطعة من الصنف $C^1$ (فتتلاشى الحدود الحدّية
بالاتصال والدورية): $c_n(f') = \iu n c_n(f)$. ثم، بمتراجحة
كوشي--شوارتز على العائلتين القابلتين للجمع تربيعيًا
(\cref{prop:b2:fourier:bessel} من أجل $f'$):
\[
\sum_{n \neq 0} \abs{c_n(f)} = \sum_{n\neq0}
\frac{\abs{c_n(f')}}{\abs n}
\leq \Bigl(\sum \abs{c_n(f')}^2\Bigr)^{1/2}
\Bigl(\sum_{n\neq0}\frac{1}{n^2}\Bigr)^{1/2} < \infty :
\]
أي \omterm{def:b2:funcseq:series}{التقارب الناظمي} لمتسلسلة فورييه. ومجموعها \omterm{def:b2:metric:continuity}{متصل} و
يتطابق مع $f$ عند كل نقطة بمبرهنة ديريكليه
(\cref{thm:b2:fourier:dirichlet}: فالدالة $f$ متصلة): ومن ثم تتقارب المتسلسلة
إلى $f$ بانتظام.

(2) من أجل $f$ كهذه: $S_N f \to f$ بانتظام، ومنه $\norm{f - S_Nf}_2
\leq \norm{f - S_Nf}_\infty \to 0$، وتمرّ مبرهنة فيثاغورس
($\norm f_2^2 = \sum_{\abs n \leq N}\abs{c_n}^2 + \norm{f -
S_Nf}_2^2$) إلى النهاية. والصورة الحقيقية مسك دفاتر مع
$a_n, b_n$.
\end{proof}

\begin{example}[حدّ الذيل $C^1$، مصوغًا كمّيًا]\label{ex:b2:fourier:tailbound}
يخفي برهان البند (1) في \cref{thm:b2:fourier:parseval} تقديرًا
قابلًا للاستعمال. فمن أجل $f$ المتصلة من الصنف $C^1$ على قطع، تعطي متراجحة
كوشي--شوارتز نفسها مطبَّقةً على الذيل وحده
\[
\sum_{\abs n > N}\abs{c_n(f)}
= \sum_{\abs n > N}\frac{\abs{c_n(f')}}{\abs n}
\leq \Bigl(\sum_{\abs n > N}\abs{c_n(f')}^2\Bigr)^{\!1/2}
\Bigl(\sum_{\abs n>N}\frac{1}{n^2}\Bigr)^{\!1/2}
\leq \norm{f'}_2\,\sqrt{\frac{2}{N}} ,
\]
باستعمال بيسل من أجل $f'$ و$\sum_{n>N}n^{-2} \leq \frac1N$. ومن ثم
يخضع الخطأ المنتظم للمجاميع الجزئية للعلاقة
\[
\norm{f - S_Nf}_\infty
\leq \sum_{\abs n>N}\abs{c_n(f)}
\leq \norm{f'}_2\,\sqrt{\frac2N} .
\]
ومن أجل $f(t) = \abs t$: $\norm{f'}_2 = 1$ (فالمشتق
$\pm1$)، ومن ثم فإن عشرة حدود تعيد بناء $\abs t$ بانتظام
ضمن $\sqrt{0.2} \approx 0.45$، و$N = 10^4$ ضمن
$0.015$. والفكرة الختامية: يشتري مشتق واحد المعدلَ
المنتظم $\frac{1}{\sqrt N}$؛ وبالمقارنة مع عالم المعاملات
$\frac1n$ في الموجة المربّعة (حيث لا \omterm{def:b2:funcseq:def}{تقارب منتظم} إطلاقًا)،
يكتسب معجم \cref{ex:b2:fourier:decaytable} أعدادًا.
\end{example}

\begin{example}[بازل وأصدقاؤه]\label{ex:b2:fourier:basel}
لتكن $f(t) = t$ على $\intoo{-\pi}{\pi}$، ممدَّدةً بالدور $2\pi$
(وهي سنّ منشارية، من الصنف $C^1$ على قطع). وبالحساب، $a_n = 0$ (بالفردية) و
\[
b_n = \frac{1}{\pi}\int_{-\pi}^{\pi} t\sin nt\,\dd t
= \frac{2(-1)^{n+1}}{n} .
\]
ويستعيد ديريكليه عند $t = \frac\pi2$ متسلسلةَ لايبنتز $\frac\pi4 = 1 -
\frac13 + \frac15 - \dots$؛ ويعطي بارسفال
\[
\frac{1}{2\pi}\int_{-\pi}^{\pi} t^2\,\dd t = \frac{\pi^2}{3}
= \frac12\sum_{n\geq1}\frac{4}{n^2}
\quad\Longrightarrow\quad
\boxed{\;\sum_{n\geq1}\frac{1}{n^2} = \frac{\pi^2}{6}\;}
\]
--- أي مجموع بازل عند أويلر، في سطرين. وتعطي الدالة $f(t) = t^2$
كذلك $\sum \frac1{n^4} = \frac{\pi^4}{90}$
(\cref{exo:b2:fourier:3}).
\end{example}

\begin{example}[نشر كامل بتحقق مدمج: $\abs{\sin t}$]\label{ex:b2:fourier:abssin}
الدالة $f(t) = \abs{\sin t}$ متصلة وزوجية، ودورية بالدور
$\pi$ (ومن ثم بالدور $2\pi$)، ومن الصنف $C^1$ على قطع.
وتقتل الزوجية المعاملات $b_n$؛ و$a_0 = \frac1\pi\int_{-\pi}^{\pi}
\abs{\sin t}\dd t = \frac4\pi$؛ ومن أجل $n \geq 1$،
يعطي تحويل الجداء إلى مجموع
\[
a_n = \frac2\pi\int_0^\pi \sin t\cos nt\,\dd t
= \frac{1}{\pi}\int_0^\pi\bigl(\sin(1+n)t +
\sin(1-n)t\bigr)\dd t
= \frac2\pi\cdot\frac{1 + \cos n\pi}{1 - n^2}
\]
من أجل $n \neq 1$ (و$a_1 = 0$ مباشرةً): فهو معدوم من أجل $n$ الفردية، و
$a_{2k} = \frac{-4}{\pi(4k^2-1)}$. وحسب
البند (1) في \cref{thm:b2:fourier:parseval} يكون التقارب \omterm{def:b2:funcseq:series}{ناظميًا}،
و
\[
\abs{\sin t} = \frac{2}{\pi} - \frac{4}{\pi}
\sum_{k\geq1}\frac{\cos(2kt)}{4k^2 - 1}
\qquad (t \in \R) .
\]
وللتحقق المدمج عند $t = 0$: تقتضي المتطابقة
$\sum_{k\geq1}\frac{1}{4k^2-1} = \frac12$، وهو ما يؤكده
التلسكوب:
\[
\sum_{k\geq1}\frac{1}{4k^2-1}
= \frac12\sum_{k\geq1}\Bigl(\frac{1}{2k-1} -
\frac{1}{2k+1}\Bigr) = \frac12 . \checkmark
\]
والفكرة الختامية: يعيش \omterm{def:b2:reduction:eigen}{طيف} $\abs{\sin}$ على التواترات
\emph{الزوجية} وحدها --- فتقويم الجيب يضاعف محتواه
التواتري، ولهذا تطنّ المقوّمات كاملة الموجة عند
$100$ أو $120$ هرتز، أي ضعف تواتر الشبكة.
\end{example}

\begin{example}[بارسفال أداةَ حساب]\label{ex:b2:fourier:zetafourodd}
يحوّل بارسفال النشور إلى متسلسلات عددية بالجملة.
ولنطبّقه على $f(t) = \abs t$ (\cref{exo:b2:fourier:2}: $a_0 =
\pi$، $a_n = \frac{-4}{\pi n^2}$ من أجل $n$ الفردية، والباقي معدوم):
\[
\frac{1}{2\pi}\int_{-\pi}^{\pi}t^2\,\dd t = \frac{\pi^2}{3}
= \frac{a_0^2}{4} + \frac12\sum_{n \text{ فردي}} a_n^2
= \frac{\pi^2}{4} +
\frac{8}{\pi^2}\sum_{n\text{ فردي}}\frac{1}{n^4} ,
\]
ومنه
\[
\sum_{n\text{ فردي}}\frac{1}{n^4}
= \frac{\pi^2}{8}\Bigl(\frac{\pi^2}{3} -
\frac{\pi^2}{4}\Bigr) = \frac{\pi^4}{96} .
\]
وللتحقق المتقاطع في مقابل \cref{exo:b2:fourier:3}: يعطي فصل
$\sum\frac1{n^4}$ إلى جزأين فردي وزوجي مسكَ دفاتر من نمط $\zeta$، أي $S = S_{\mathrm{odd}} + \frac{S}{16}$، ومنه $S =
\frac{16}{15}\cdot\frac{\pi^4}{96} = \frac{\pi^4}{90}$ ---
وهي بالضبط القيمة الموجودة هناك بدالة مختلفة. نشران
وعدد واحد: والتوافق هو تقايس بارسفال
في العمل. والفكرة الختامية: كل نشر فورييه جديد
آلةٌ مولّدة لمتطابقات المتسلسلات؛ ويشرح الجزء الثاني
من مسألة نهاية الأسبوع لماذا لا تستطيع الآلة أن
تناقض نفسها أبدًا.
\end{example}

\begin{example}[الانسحاب والتعديل]\label{ex:b2:fourier:shiftrules}
تولّد قاعدتان من سطر واحد نشورًا كثيرة من نشر واحد. فمن أجل $a
\in \R$، وبتعويض $s = t - a$:
\[
c_n\bigl(f(\cdot - a)\bigr)
= \frac{1}{2\pi}\int_{-\pi}^{\pi}f(t - a)\eu^{-\iu nt}\dd t
= \eu^{-\iu na}\,c_n(f)
\qquad\text{(فالانسحاب يعدّل الطيف)},
\]
ومباشرةً من التعريف،
\[
c_n\bigl(\eu^{\iu kt}f\bigr) = c_{n-k}(f)
\qquad\text{(والتعديل يُسحب الطيف)}.
\]
ومثال منفَّذ: للسنّ المنشارية المزاحة بالمقدار $\pi$، أي $g(t) = f(t -
\pi)$ مع $f(t) = t$، معاملاتٌ من نمط $b_n$
$(-1)^n\cdot\frac{2(-1)^{n+1}}{n} = -\frac2n$: فهو نشر
$g \sim -2\sum\frac{\sin nt}{n}$ للسنّ المنشارية التي تقفز عند
$0$ بدلًا من $\pi$ --- بلا إعادة حساب أي تكامل. والفكرة
الختامية: لا تدير الإزاحات الزمنية إلا الأطوار، ولا تغيّر السعات
أبدًا (فالمقدار $\abs{c_n}$ لا يتغير بالإزاحة)، ولهذا تكون الطاقة
(بارسفال) وصنف التقارب خاصيتين \emph{لشكل}
الإشارة، لا لموضع بدء الساعة.
\end{example}

\begin{remark}[مزالق شائعة]
\emph{(1) ثلاثة تقاربات، وثلاث عملات:} النقطي
(ديريكليه: يحتاج الصنف $C^1$ على قطع، ويدفع \emph{النقطة الوسطى} عند
كل قفزة --- لا القيمة من جهة واحدة أبدًا)، والمنتظم (يحتاج نهاية
متصلة؛ وهو مستحيل عبر قفزة، وغيبس هو العَرَض
المرئي)، وفي المتوسط التربيعي (بارسفال: وهو الأمتن،
والأعمى عن النقاط المفردة). فسمِّ دائمًا أيّها
تدّعي. \emph{(2) لا اشتقاق حدًّا حدًّا
افتراضيًا:} فاشتقاق متسلسلة السنّ المنشارية في
\cref{ex:b2:fourier:basel} حدًّا حدًّا يعطي $\sum
2(-1)^{n+1}\cos nt$، وحدودها لا تؤول حتى إلى $0$ ---
فمبرهنات النقل في فصل متتاليات الدوال تحتاج
تقاربًا منتظمًا للمتسلسلة \emph{المشتقة}، وهو ما تهدمه
القفزة. فالملاسة أولًا، والاشتقاق ثانيًا
(\cref{exo:b2:fourier:6} هو المعجم). \emph{(3)
المجاميع الجزئية المتناظرة:} تتعلق مبرهنة ديريكليه بالمقدار $S_N =
\sum_{-N}^{N}$؛ وقد تحوّل إعادةُ الترتيب أو جمعُ جهة أولًا
التباعدَ إلى تقارب وبالعكس. \emph{(4)
انحراف التنظيم:} تختلف الاصطلاحات بين الكتب
($\frac{1}{2\pi}$ أو $\frac1\pi$ في المقدمة، والدور $2\pi$ أو
$1$)؛ واللامتغيّرات الموثوقة هي علاقات التعامد والتنظيم
--- فأعد حساب $\langle e_m, e_n\rangle$ في الاصطلاح
المستعمل قبل الوثوق بأي صيغة.
\end{remark}

\begin{remark}[أين يُستعمَل هذا]
بارسفال هو بذرة نظرية $L^2$ لمتسلسلات فورييه: ويُكمل
مجلد السنة الثالثة الصورة (فالأسّيات أساس
هيلبرتي للفضاء $L^2$، والتطبيق $f \mapsto (c_n)$
تقايس تقابلي). وداخل هذا المجلد، تبرهن مسألة نهاية الأسبوع
على \emph{مبرهنة فييير} --- أي إن متوسطات تشيزارو
لمتسلسلة فورييه تتقارب بانتظام من أجل كل دالة $f$ دورية
متصلة --- وهذا يرقّي بارسفالَ العام المقبول إلى
مبرهنة، ويعطي مبرهنة فايرشتراس المثلثية، ويدفع
ربحين مذهلين: مبرهنة فايل في التوزع المتساوي و
متراجحة القياس المتساوي. وتقرأ الرياضيات التطبيقية هذا
الفصل يوميًا: أطياف الإشارات، وتوافقيات الجمل
المهتزة، وتحويل فورييه السريع (وقد كان نظيره المنتهي
\cref{exo:b2:hermitian:10}).
\end{remark}

\begin{remark}[آفاق داخل هذا المجلد]
ثلاثة فصول تحاور هذا الفصل. رجوعًا: قدّم الفصل
الهرميتي الهندسةَ (العائلات المتعامدة الممنظمة، و
الإسقاطات، وبيسل)، وقدّم فصل متتاليات الدوال
التحليلَ (\omterm{def:b2:funcseq:def}{التقارب المنتظم}، ومبرهنات النقل، والمتطابقات
التقريبية --- فنواة فييير لمتسلسلات فورييه هي ما كانت
\omterm{thm:b2:funcseq:weierstrass}{كثيرات حدود برنشتاين} لفايرشتراس). وجانبيًا: تعود
نظرية الحدّ في فصل متسلسلات القوى عبر \omterm{thm:b2:series:abel}{جمع
آبل}، مُحقَّقًا هنا بنواة بواسون
(\cref{exo:b2:fourier:12}) --- ويلعب نصف قطر القرص $r$
دور وسيط الجمع. وأمامًا: يفكك فصل
المعادلات التفاضلية القسرَ الدوري
إلى توافقيات ويُطعم كلًا منها لاستجابة المتذبذب
التواترية؛ ويقع الرنين حين يوافق نمطُ فورييه للدخل
تواترًا طبيعيًا، ولهذا تكون مسألة نهاية الأسبوع هناك
ومسألة هذا الفصل نصفَي حكاية واحدة.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:fourier:1}
احسب \omterm{def:b2:fourier:coefficients}{معاملات فورييه} للموجة المربّعة (وهي $f = -1$ على
$\intoo{-\pi}{0}$، و$+1$ على $\intoo{0}{\pi}$)، وصُغ استنتاج مبرهنة
ديريكليه عند $t = \frac\pi2$ وعند القفزة $t = 0$، واستعد
متسلسلة لايبنتز.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:fourier:2}
انشر $f(t) = \abs t$ (مع $\abs t \leq \pi$، ودورية بالدور $2\pi$) في
متسلسلة فورييه؛ وبرّر التقارب \emph{الناظمي}؛ وقيّم عند $t =
0$ للحصول على $\sum_{k\geq0}\frac{1}{(2k+1)^2} = \frac{\pi^2}{8}$،
وأعد استنباط بازل منها.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:fourier:3}
انشر $f(t) = t^2$ (مع $\abs t \leq \pi$) واستنتج
\[
\sum_{n\geq1}\frac{(-1)^{n+1}}{n^2} = \frac{\pi^2}{12},
\qquad
\sum_{n\geq1}\frac{1}{n^4} = \frac{\pi^4}{90}
\quad\text{(بارسفال)}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:4}
لتكن $f$ متصلة ودورية بالدور $2\pi$ مع $c_n(f) = 0$ من أجل كل
$n$. برهن على أن $f = 0$ \emph{(بارسفال --- ومن أجل أي صنف بُرهن عليه
هنا؟ وبرّر أنه يمكن إسقاط الاتصال مع الصنف $C^1$ على قطع
بقبول بارسفال العام، أو أعطِ حجة الكثافة في
الخطوط العريضة)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:5}
من أجل $\alpha \notin \Z$، انشر $f(t) = \cos(\alpha t)$ (مع $\abs t
\leq \pi$) واستنتج نشر ظل التمام إلى كسور جزئية:
\[
\pi\cot(\pi\alpha) = \frac{1}{\alpha} + \sum_{n\geq1}
\frac{2\alpha}{\alpha^2 - n^2} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:6}
برهن على أنه إذا كانت $f$ دورية بالدور $2\pi$ ومن الصنف $C^k$ مع $f^{(k)}$
متصلة على قطع، فإن $c_n(f) = O\bigl(\abs n^{-k}\bigr)$:
أي إن ملاسة الإشارة $=$ تناقصُ طيفها.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:fourier:7}
(متراجحة فيرتينغر) لتكن $f$ من الصنف $C^1$، ودورية بالدور $2\pi$، مع
$\int_{-\pi}^{\pi} f = 0$. برهن على أن
\[
\int_{-\pi}^{\pi} \abs{f}^2 \leq \int_{-\pi}^{\pi} \abs{f'}^2 ,
\]
مع المساواة إذا وفقط إذا $f(t) = a\cos t + b\sin t$. \emph{(بارسفال على
الطرفين؛ وقارِن $\abs{c_n}^2$ و$n^2\abs{c_n}^2$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:fourier:8}
(ثابت غيبس) من أجل الموجة المربّعة في
\cref{exo:b2:fourier:1}، قيّم المجموع الجزئي عند $x_N =
\frac{\pi}{2N}$: فبكتابة $u_k = \frac{(2k+1)\pi}{2N}$ و$\Delta u
= \frac{\pi}{N}$، بيّن أن
\[
S_{2N-1}\Bigl(\frac{\pi}{2N}\Bigr)
= \frac{4}{\pi}\sum_{k=0}^{N-1} \frac{\sin u_k}{2k+1}
= \frac{2}{\pi}\sum_{k=0}^{N-1} \frac{\sin u_k}{u_k}\,\Delta u
\xrightarrow[N\to\infty]{}
\frac{2}{\pi}\int_0^{\pi}\frac{\sin u}{u}\,\dd u \approx 1.179 :
\]
وهو مجموع ريمان للدالة $\frac{2}{\pi}\cdot\frac{\sin u}{u}$ على
$\intcc{0}{\pi}$ عند النقاط الوسطى. واختم بأن التجاوز فوق
قيمة القفزة $1$ لا يتلاشى حين $N \to \infty$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:fourier:9}
انشر $\cos^3 t$ و$\sin^2 t\,\cos t$ في متسلسلتَي فورييه
\emph{(بالتخطيط؛ فكثير الحدود المثلثي هو متسلسلة فورييه
الخاصة به، بوحدانية المعاملات)}. وما قيم $c_n$ و
$a_n$ و$b_n$ من أجل كل منهما؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:10}
ليكن $a > 0$ ولتكن $f(t) = \eu^{at}$ على $\intoc{-\pi}{\pi}$،
ممدَّدةً بالدور $2\pi$. احسب
\[
c_n(f) = \frac{(-1)^n\sinh(a\pi)}{\pi(a - \iu n)} ,
\]
وطبّق مبرهنة ديريكليه عند القفزة $t = \pi$، واستنتج
نشر ظل التمام الزائدي إلى كسور جزئية:
\[
\coth(\pi a) = \frac{1}{\pi a} + \sum_{n\geq1}
\frac{2a}{\pi(a^2 + n^2)} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:11}
(الالتفاف) من أجل $f, g$ متصلتين ودوريتين بالدور $2\pi$، نعرّف
\[
(f * g)(x) = \frac{1}{2\pi}\int_{-\pi}^{\pi} f(x - t)\,g(t)\,
\dd t .
\]
بيّن أن $f * g = g * f$، وأن $c_n(f*g) = c_n(f)\,c_n(g)$
\emph{(ولتبديل تكاملَي دالة مكامَلة متصلة،
قارِن بين دالتَي الحدّ الأعلى: فكلتاهما تنعدم عند
الطرف الأيسر ولهما المشتق نفسه، بالاتصال و
الاشتقاق تحت علامة التكامل)}، وأن $S_N(f) =
f * D_N$ من أجل \omterm{lem:b2:fourier:kernel}{نواة ديريكليه}. (ومتوسطات فييير في مسألة نهاية
الأسبوع التفافاتٌ كذلك، $\sigma_N(f) = f *
F_N$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:fourier:12}
(نواة بواسون: متوسطات آبل لمتسلسلات فورييه) من أجل $0 \leq r <
1$ نضع $P_r(t) = \sum_{n\in\Z} r^{\abs n}\eu^{\iu nt}$.
\begin{enumerate}
  \item اجمع المتسلسلتين الهندسيتين وبيّن أن
        \[
        P_r(t) = \frac{1 - r^2}{1 - 2r\cos t + r^2} > 0,
        \qquad
        \frac{1}{2\pi}\int_{-\pi}^{\pi}P_r = 1 .
        \]
  \item بيّن أنه من أجل $\delta \leq \abs t \leq \pi$: $P_r(t)
        \leq \frac{1 - r^2}{1 - 2r\cos\delta + r^2} \to 0$ حين
        $r \to 1^-$، بانتظام.
  \item استنتج أنه من أجل كل دالة $f$ متصلة ودورية بالدور $2\pi$،
        تتقارب \emph{متوسطات آبل} $(f * P_r)(x) = \sum_n
        r^{\abs n}c_n(f)\,\eu^{\iu nx}$ إلى $f$
        بانتظام حين $r \to 1^-$ --- وهي شقيقة مبرهنة فييير
        ذات الوسيط \omterm{def:b2:metric:continuity}{المتصل}، وتجسيدُ \omterm{thm:b2:series:abel}{جمع آبل}
        من فصل متسلسلات القوى في لغة فورييه.
\end{enumerate}
\end{exercise}

\section{مسألة: مبرهنة فييير وأرباحها}

\begin{problem}\label{pb:b2:fourier:1}
تحتاج مبرهنة ديريكليه أن تكون $f$ من الصنف $C^1$ على قطع؛ ومن أجل
$f$ المتصلة فحسب قد تسيء المجاميع الجزئية $S_N(f)$ السلوك. واكتشاف
فييير: أن \emph{متوسطاتها التشيزارية} لا تسيء أبدًا. والمحرّك
إيجابية نواة فييير، والحصاد
هائل: التقريب المثلثي المنتظم (فايرشتراس)،
ووحدانية \omterm{def:b2:fourier:coefficients}{معاملات فورييه}، وبارسفال من أجل كل دالة
في هذا الفصل (فيُزال ``المقبول'' في
\cref{thm:b2:fourier:parseval})، ومبرهنة فايل في التوزع
المتساوي، و--- تتويجًا لقرن من الهندسة ---
متراجحة القياس المتساوي.\index{مبرهنة فييير}\index{توزع متساوٍ}\index{متراجحة القياس المتساوي}
وفي كل ما يلي، $f$ دورية بالدور $2\pi$ ومتصلة على قطع،
و
\[
\sigma_N(f) = \frac{S_0(f) + S_1(f) + \dots +
S_{N-1}(f)}{N} .
\]

\medskip
\noindent\textbf{الجزء الأول --- نواة فييير.}
\begin{enumerate}
  \item بيّن أن $\sigma_N(f)(x) =
        \frac{1}{2\pi}\int_{-\pi}^{\pi}f(x+u)\,F_N(u)\,\dd u$
        مع $F_N = \frac{D_0 + \dots + D_{N-1}}{N}$، وأن
        $\frac{1}{2\pi}\int_{-\pi}^{\pi}F_N = 1$.
  \item برهن على الصورة المغلقة، من أجل $u \notin 2\pi\Z$:
        \[
        F_N(u) = \frac{1}{N}\,
        \frac{\sin^2\bigl(\frac{Nu}{2}\bigr)}
        {\sin^2\bigl(\frac u2\bigr)} \;\geq\; 0
        \]
        \emph{(اجمع $\sin\bigl((n+\frac12)u\bigr)$ بوصفه
        الجزء التخيلي لمتسلسلة هندسية)}.
  \item بيّن تقدير التركّز: من أجل $0 < \delta \leq
        \abs u \leq \pi$،
        \[
        F_N(u) \leq \frac{1}{N\sin^2\frac\delta2}
        \xrightarrow[N\to\infty]{} 0
        \quad\text{بانتظام} :
        \]
        أي إن $(F_N)$ متطابقة تقريبية موجبة.
  \item (مبرهنة فييير) برهن على أنه إذا كانت $f$ متصلة و
        دورية بالدور $2\pi$، فإن $\sigma_N(f) \to f$
        \emph{بانتظام} على $\R$ \emph{(افصل تكامل
        $\bigl(f(x+u) - f(x)\bigr)F_N(u)$ عند $\abs u =
        \delta$؛ واستعمل هاينه والأسئلة 1--3)}.
  \item من أجل $f$ المتصلة على قطع، بيّن النسخة
        النقطية $\sigma_N(f)(x) \to \frac{f(x^+) +
        f(x^-)}{2}$ عند كل $x$، والحدّ المنتظم
        $\norm{\sigma_N(f)}_\infty \leq \norm f_\infty$
        \emph{(بالإيجابية!)}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- فايرشتراس والوحدانية
وبارسفال.}
\begin{enumerate}[resume]
  \item (فايرشتراس المثلثية) استنتج: كل دالة متصلة
        دورية بالدور $2\pi$ هي نهاية منتظمة
        لكثيرات حدود مثلثية.
  \item (الوحدانية) استنتج: الدالة المتصلة $f$ التي $c_n(f) = 0$
        من أجل كل $n$ معدومة تمامًا --- فدالتان دوريتان
        متصلتان لهما \omterm{def:b2:fourier:coefficients}{معاملات فورييه} نفسها
        تتطابقان (\cref{exo:b2:fourier:4}، الآن بلا أي
        قبول).
  \item (بارسفال، الحالة المتصلة) باستعمال خاصية الإسقاط
        للمقدار $S_N$ (\cref{prop:b2:fourier:bessel}) و
        $\sigma_Nf \in \mathcal T_{N-1} \subseteq \mathcal
        T_N$، برهن على أن
        \[
        \norm{f - S_Nf}_2 \leq \norm{f - \sigma_Nf}_2
        \leq \norm{f - \sigma_Nf}_\infty
        \xrightarrow[N\to\infty]{} 0 ,
        \]
        واختم متطابقة بارسفال من أجل كل دالة $f$ دورية بالدور $2\pi$
        \emph{متصلة}.
  \item (بارسفال، حالة الاتصال على قطع) من أجل $f$
        متصلة على قطع و$\varepsilon > 0$ معطى، أنشئ
        دالة دورية متصلة $g$ تحقق $\norm{f - g}_2 \leq
        \varepsilon$ \emph{(عوّض $f$ باستكمال
        أفيني على فترات صغيرة جدًا حول القفزات)}،
        واستنتج $\norm{f - S_Nf}_2 \to 0$
        \emph{(استعمل بيسل: $\norm{S_Nh}_2 \leq \norm h_2$)}:
        فيصحّ بارسفال بالعمومية الكاملة المذكورة في
        \cref{thm:b2:fourier:parseval} --- وقد زال ``المقبول''.
  \item (لا غيبس عند فييير) قارِن بالنتيجة \cref{exo:b2:fourier:8}: بيّن أنه من أجل الموجة المربّعة
        $f$، يكون $\abs{\sigma_N(f)} \leq 1$ في كل مكان، من أجل كل
        $N$ --- فمعدّل تشيزارو يمحو التجاوز الذي
        يطارد $S_N$. واشرح بجملة واحدة أي خاصية
        للنواة $F_N$ مسؤولة عن ذلك.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- المعدلات.}
\begin{enumerate}[resume]
  \item برهن على حدّي النواة، من أجل $0 < \abs u \leq
        \pi$:
        \[
        F_N(u) \leq N,
        \qquad
        F_N(u) \leq \frac{\pi^2}{N u^2}
        \]
        \emph{(من أجل الأول، $\abs{\sin N\theta} \leq
        N\abs{\sin\theta}$ بالتراجع؛ ومن أجل الثاني،
        $\sin\frac u2 \geq \frac{u}{\pi}$ على
        $\intcc{0}{\pi}$)}.
  \item استنتج تقدير العزم الأول
        \[
        \frac{1}{2\pi}\int_{-\pi}^{\pi}\abs u\,F_N(u)\,\dd u
        \;\leq\; \frac{C\,\ln N}{N}
        \qquad (N \geq 2)
        \]
        من أجل ثابت صريح \emph{(افصل عند $\abs u =
        \frac1N$)}.
  \item اختم: إذا كانت $f$ ليبشيتزية بالثابت $L$ ودورية بالدور $2\pi$،
        فإن
        \[
        \norm{\sigma_N f - f}_\infty \leq \frac{C\,L\ln N}{N} .
        \]
  \item (التشبّع) احسب $\sigma_N(e_1)$ من أجل $e_1(t) =
        \eu^{\iu t}$ وبيّن أن $\norm{\sigma_Ne_1 - e_1}_\infty
        = \frac1N$: أي إن فييير لا يتقارب أسرع من
        $\frac1N$ حتى من أجل أملس الدوال --- وهو النظير
        الدقيق لتشبّع برنشتاين في مسألة نهاية الأسبوع
        في فصل متتاليات الدوال.
  \item (التوطين) بيّن أنه إذا انعدمت $f$ (المتصلة على قطع)
        على $\intoo{x - \delta}{x + \delta}$، فإن
        $\sigma_N(f)(x) \to 0$، مهما كانت $f$ جامحة في ما عدا ذلك
        --- فتقارب المتوسطات عند $x$ لا يرى إلا $f$ قرب
        $x$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- مبرهنة فايل في التوزع المتساوي.}
تكون المتتالية $(x_n)_{n\geq1}$ في $\intco{0}{1}$
\emph{متوزعة بالتساوي} إذا كان، من أجل كل فترة
$\intcc{a}{b} \subseteq \intcc{0}{1}$،
\[
\frac{\#\{n \leq N : x_n \in \intcc ab\}}{N}
\xrightarrow[N\to\infty]{} b - a .
\]
\begin{enumerate}[resume]
  \item بيّن أن $(x_n)$ متوزعة بالتساوي بمجرد أن يكون
        $\frac1N\sum_{n\leq N}f(x_n) \to \int_0^1 f$ من أجل كل دالة $f$ \emph{متصلة}
        دورية بالدور $1$ \emph{(احصر
        الدالة المميّزة للفترة $\intcc ab$ بين دالتين متصلتين
        أفينيتين على قطع يختلف تكاملاهما بمقدار
        $\varepsilon$)}.
  \item (محك فايل، الكفاية) لنفترض
        \[
        \frac{1}{N}\sum_{n=1}^{N}\eu^{2\iu\pi kx_n}
        \xrightarrow[N\to\infty]{} 0
        \qquad\text{من أجل كل } k \in \Z\setminus\{0\} .
        \]
        بيّن أن $\frac1N\sum f(x_n) \to \int_0^1f$ أولًا من أجل
        كثيرات الحدود المثلثية، ثم من أجل كل دالة $f$ متصلة دورية
        بالدور $1$ بالسؤال 6 (منقولًا إلى الدور
        $1$): ومع السؤال 16، تكون $(x_n)$ متوزعة بالتساوي.
  \item ليكن $\alpha$ عددًا أصمّ ولتكن $x_n = \{n\alpha\}$
        (وهي الجزء الكسري). حُدّ المجموع الهندسي
        \[
        \Bigl|\sum_{n=1}^{N}\eu^{2\iu\pi kn\alpha}\Bigr|
        \leq \frac{2}{\abs{1 - \eu^{2\iu\pi k\alpha}}}
        \qquad (k \neq 0),
        \]
        واختم \emph{مبرهنة فايل}: أي إن $(\{n\alpha\})$
        متوزعة بالتساوي في $\intco{0}{1}$.
  \item استنتج أن $(\{n\alpha\})$ كثيفة في
        $\intcc{0}{1}$ من أجل $\alpha$ الأصمّ، واشرح بجملة
        واحدة لماذا يكون التوزع المتساوي أقوى تمامًا
        من الكثافة.
  \item (الأرقام الرائدة) برهن على أن نسبة الأعداد الصحيحة
        $n \leq N$ التي يكون فيها الرقم العشري الرائد للمقدار $2^n$
        هو $1$ تؤول إلى $\log_{10}2 \approx 0.301$ \emph{(الرقم
        الرائد $1$ يعني $\{n\log_{10}2\} \in
        \intco{0}{\log_{10}2}$؛ وبيّن أن $\log_{10}2$
        أصمّ)}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- متراجحة القياس المتساوي.} ليكن
$\Gamma$ منحنيًا بسيطًا مغلقًا من الصنف $C^1$ وطوله $L$ ويحصر
مساحة موجَّهة $A$، ممثَّلًا وسيطيًا بطول القوس المُحجَّم: أي $z(t) =
x(t) + \iu y(t)$، دوريًا بالدور $2\pi$، مع $\abs{z'(t)} =
\frac{L}{2\pi}$ ثابتة؛ والمساحة المحصورة
\[
A = \frac12\int_0^{2\pi}\bigl(x\,y' - y\,x'\bigr)\dd t
= \frac{1}{2}\,\Im\int_0^{2\pi}\conj{z}\,z'\,\dd t
\]
(ونأخذها هنا تعريفًا للمساحة الموجَّهة؛ ويبرهن فصل
التكاملات المتعددة على أنها توافق المساحة الحدسية،
عبر صيغة غرين).
\begin{enumerate}[resume]
  \item انشر $z(t) = \sum_{n\in\Z}c_n\eu^{\iu nt}$ (وهي
        متسلسلة دالة من الصنف $C^1$، متقاربة \omterm{def:b2:funcseq:series}{ناظميًا}) و
        برهن، ببارسفال مطبَّقًا على $z'$:
        \[
        \frac{L^2}{2\pi} = \int_0^{2\pi}\abs{z'}^2\dd t
        = 2\pi\sum_{n\in\Z}n^2\abs{c_n}^2 .
        \]
  \item برهن كذلك على أن $A = \pi\sum_{n\in\Z}
        n\,\abs{c_n}^2$ \emph{(بارسفال في صورته المستقطبة:
        $\frac{1}{2\pi}\int\conj f g = \sum
        \conj{c_n(f)}c_n(g)$، مطبَّقةً على $f = z$، $g = z'$)}.
  \item (هورفيتس) اختم:
        \[
        L^2 - 4\pi A = 4\pi^2\sum_{n\in\Z}
        (n^2 - n)\abs{c_n}^2 \;\geq\; 0 ,
        \]
        مع المساواة إذا وفقط إذا $z(t) = c_0 +
        c_1\eu^{\iu t}$ --- أي دائرة. وهي متراجحة القياس
        المتساوي: فمن بين المنحنيات المغلقة ذوات الطول $L$، لا تحصر
        الدائرةُ وحدها مساحة $\frac{L^2}{4\pi}$.
  \item تحققات: تحقق من المساواة من أجل الدائرة ذات
        نصف القطر $R$ ومن المتراجحة التامة من أجل المربّع ذي
        الضلع $a$؛ واشرح لماذا $n^2 - n \geq 0$ من أجل كل
        عدد صحيح $n$، بما فيها السالبة، وأين استُعملت
        سرعة التمثيل الوسيطي الثابتة.
  \item تركيب. بجملة واحدة لكل بند: (1) الخاصية الوحيدة
        للنواة $F_N$ التي تتدفق منها الأجزاء الأول--الثالث، و
        التي تفتقر إليها $D_N$؛ (2) وكيف يرتبط جمع تشيزارو هنا
        بمسألة نهاية الأسبوع في فصل متسلسلات القوى
        (فروبينيوس)؛ (3) وأي ربح استعمل
        فايرشتراس وحده (السؤال 6) وأيها احتاج بارسفال
        كاملًا؛ (4) وجملة واحدة عمّا يضيفه مجلد السنة الثالثة
        (تمام $L^2$: أي متسلسلات فورييه بوصفها أساسًا
        هيلبرتيًا).
\end{enumerate}
\end{problem}
