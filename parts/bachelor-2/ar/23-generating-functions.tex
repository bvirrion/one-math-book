\chapter{الدوال المولّدة للاحتمالات}\label{ch:b2:genfun}

تعود متسلسلات القوى في \cref{ch:b2:powerseries} بمهمة احتمالية: فنربط \omterm{def:b2:randomvar:law}{بمتغير عشوائي} ذي
قيم في $\N$ متسلسلةَ القوى ذات المعاملات $\P(X = n)$. وتحوّل هذه
\emph{\omterm{ex:b2:powerseries:fibonacci}{الدالة المولّدة}} مجاميع المتغيرات المستقلة إلى جداءات، والعزوم
إلى مشتقات عند $1$، والمتطابقات التوفيقية الصعبة إلى ضرب من سطر
واحد. ويختم الفصل الكتاب بقطعتين معروضتين: تقريب بواسون للأحداث
النادرة، ومحك الانقراض من أجل مسارات التفرع --- وهو حساب احتمالي
لانهائي حقا يُحل كله بهندسة منحن محدب.

\section{التعريف والخواص الأساسية}

\begin{definition}[الدالة المولّدة للاحتمالات]\label{def:b2:genfun:pgf}
ليكن $X$ متغيرا عشوائيا ذا قيم في $\N$، $p_n = \P(X = n)$. \emph{الدالة
المولّدة للاحتمالات}\index{دالة مولدة للاحتمالات} للمتغير $X$ هي
مجموع متسلسلة القوى
\[
G_X(t) = \E\bigl(t^X\bigr) = \sum_{n=0}^{\infty} p_n\,t^n .
\]
\end{definition}

\begin{example}[ردود الفعل الأولى]\label{ex:b2:genfun:firstreflexes}
للمتغير الثابت $X = c$ الدالة $G_X(t) = t^c$؛ والإزاحة تحقق $G_{X+c}(t) = t^c\,G_X(t)$؛ والتقويم
عند نقط خاصة يقرأ معلومات دون أي نشر: $G_X(0) = \P(X
= 0)$، $G_X(1) = 1$، و$G_X(-1) = \P(X\text{ زوجي}) -
\P(X\text{
فردي})$، وهو ميزان الزوجية المستغل في \cref{exo:b2:genfun:10}. وتُستعمل هذه الأسطر
المفردة بصمت في كل ما يلي --- والتقويم $G_X(0)$ هو بالضبط كيف ستُستخرج
احتمالات الانقراض من الدوال المولّدة المكررة في نهاية الفصل.
\end{example}

\begin{proposition}[نصف القطر والخواص الأولى]\label{prop:b2:genfun:radius}
للمتسلسلة التي تعرّف $G_X$ نصف قطر تقارب $\geq 1$؛ والدالة $G_X$
معرَّفة ومتصلة على $\intcc{-1}{1}$، ومن الصنف $\mathcal{C}^\infty$ على $\intoo{-1}{1}$، مع $G_X(1) = 1$
و$\abs{G_X(t)} \leq 1$ هناك. وعلاوة على ذلك تحدد $G_X$ قانون $X$:
\[
p_n = \frac{G_X^{(n)}(0)}{n!} .
\]
\end{proposition}

\begin{proof}
بما أن $\sum p_n = 1$ متقاربة، فإن الحدود $p_n\,1^n$ محدودة، ومنه فنصف القطر
$\geq 1$ (بمبرهنة آبل المساعدة، \cref{ch:b2:powerseries})؛ وعند $t = \pm1$ تتقارب المتسلسلة
تقاربا مطلقا (فالمقدار $\sum p_n = 1$ يهيمن)؛ بل أفضل من ذلك، على الفترة
$\intcc{-1}1$ كلها،
\[
\sup_{\abs t\leq1}\,\abs{p_nt^n} = p_n
\quad\text{مع}\quad \sum_np_n < \infty :
\]
فالمتسلسلة تتقارب \emph{تقاربا ناظميا} على $\intcc{-1}1$، ومنه فمجموعها
\omterm{def:b2:metric:continuity}{متصل} هناك (\cref{thm:b2:funcseq:weierstrass,thm:b2:funcseq:continuity}). أما الملاسة في الداخل وصيغة المعاملات فهما
النظرية العامة لمتسلسلات القوى؛ وبما أن المعاملات قابلة للاستعادة،
فإن لمتغيرين لهما \omterm{ex:b2:powerseries:fibonacci}{الدالة المولّدة} نفسها القانون نفسه.
\end{proof}

\begin{example}[القوانين الكلاسيكية]\label{ex:b2:genfun:classical}
\leavevmode
\begin{itemize}
\item برنولي $\mathcal{B}(p)$: $G(t) = 1 - p + pt$.
\item ثنائي الحد $\mathcal{B}(n, p)$: $G(t) = \sum_k \binom nk
(pt)^k(1-p)^{n-k} = (1 - p + pt)^n$ (بمبرهنة ذات الحدين).
\item الهندسي $\mathcal{G}(p)$: $G(t) =
\sum_{k\geq1}(1-p)^{k-1}p\,t^k = \dfrac{pt}{1 - (1-p)t}$ (بنصف قطر $\frac{1}{1-p} > 1$).
\item بواسون $\mathcal{P}(\lambda)$: $G(t) =
\sum_k e^{-\lambda}\frac{(\lambda t)^k}{k!} = e^{\lambda(t - 1)}$ (بنصف قطر $\infty$).
\end{itemize}
\end{example}

\begin{example}[مكاملة الدالة المولّدة]\label{ex:b2:genfun:integraltrick}
تعطي مشتقات $G_X$ عند $1$ العزوم الموجبة؛ أما \emph{التكامل}
فيعطي عزما سالبا. انطلاقا من $\int_0^1t^k\dd t
= \frac1{k+1}$ وبالمكاملة حدا حدا (\omterm{def:b2:funcseq:series}{بالتقارب
الناظمي} على $\intcc01$):
\[
\int_0^1G_X(t)\,\dd t = \sum_{k\geq0}\frac{\P(X =
k)}{k+1} = \E\Bigl(\frac1{1+X}\Bigr).
\]
ومن أجل $X \sim \mathcal P(\lambda)$:
\[
\E\Bigl(\frac1{1+X}\Bigr) =
\int_0^1\eu^{\lambda(t-1)}\,\dd t = \frac{1 -
\eu^{-\lambda}}{\lambda},
\]
فنستعيد في سطر واحد حساب المتسلسلة في \cref{ex:b2:randomvar:transferex}. \omterm{ex:b2:powerseries:fibonacci}{فالدالة المولّدة} أداة
ذات اتجاهين: نفاضل عند $1$ من أجل العزمين $\E(X)$ و$\E(X(X-1))$، ونكامل
على $\intcc01$ من أجل $\E\bigl(\frac1{1+X}\bigr)$ --- كائن تحليلي واحد، يُستجوب في أي اتجاه
تحتاج إليه المسألة.
\end{example}

\begin{example}[قانون نصف قطره يساوي واحدا بالضبط]\label{ex:b2:genfun:heavytail}
لنضع $\P(X = k) = \dfrac{6}{\pi^2k^2}$ من أجل $k \geq 1$ --- وهو قانون احتمال بحكم متطابقة بازل
(\cref{ex:b2:fourier:basel}). ولدالته المولّدة $G(t) = \frac6{\pi^2}\sum_{k\geq1}\frac{t^k}{k^2}$ نصف قطر تقارب يساوي $1$ بالضبط:
فالحصر العام ``نصف القطر $\geq
1$'' في \cref{prop:b2:genfun:radius} لا يمكن تحسينه. والمتوسط
هو
\[
\sum_{k\geq1}k\,\P(X = k) =
\frac6{\pi^2}\sum_{k\geq1}\frac1k = \infty :
\]
فالدالة $G$ متصلة على $\intcc{-1}1$، وملساء في الداخل، لكن مشتقتها
تنفجر عند $1^-$ --- فالتمثيل البياني يصل إلى النقطة $(1, 1)$ \omterm{def:b2:curves:arc}{بمماس}
شاقولي. فالذيول الثقيلة مرئية \emph{هندسيا} على \omterm{ex:b2:powerseries:fibonacci}{الدالة المولّدة}، عند
النقطة الوحيدة $t = 1$؛ وتجعل مبرهنة العزوم أدناه هذا التقابل مضبوطا.
\end{example}

\begin{theorem}[العزوم من الدالة المولّدة]\label{thm:b2:genfun:moments}
يقبل $X$ أملا رياضيا إذا وفقط إذا كانت $G_X$ \omterm{def:b2:diffcalc:differential}{قابلة للتفاضل} عند
$1^-$ (بالمشتقة اليسرى، وهي منتهية)، وعندئذ $\E(X) = G_X'(1)$.
وبالمثل يقبل $X$ عزما من الرتبة الثانية إذا وفقط إذا كانت $G_X$
\omterm{def:b2:diffcalc:differential}{قابلة للتفاضل} مرتين عند $1^-$، وعندئذ
\[
\E\bigl(X(X - 1)\bigr) = G_X''(1),
\qquad
V(X) = G_X''(1) + G_X'(1) - G_X'(1)^2 .
\]
\end{theorem}

\begin{proof}
من أجل $t \in \intoo{0}{1}$، تعطي المفاضلة حدا حدا داخل القرص $G_X'(t) = \sum_{n\geq1} np_n t^{n-1}$، وهي متسلسلة
ذات معاملات موجبة: ومنه فإن $t \mapsto G_X'(t)$ غير متناقصة على $\intoo{0}{1}$، وبالتقارب
الرتيب للمجاميع الجزئية (أو بمبرهنة آبل من أجل المعاملات الموجبة،
\cref{ch:b2:powerseries})،
\[
\lim_{t \to 1^-} G_X'(t)
= \sum_{n\geq1} n\,p_n \in \intcc{0}{+\infty} ,
\]
ويكون كل طرف منتهيا بالضبط عندما يكون الآخر كذلك. وعندما يكون منتهيا،
تحصر مبرهنة القيمة المتوسطة نسب الفروق $\frac{G_X(1) -
G_X(t)}{1 - t}$ بين قيم $G_X'$، ومنه
فإن $G_X$ \omterm{def:b2:diffcalc:differential}{قابلة للتفاضل} عند $1^-$ مع $G_X'(1) = \sum np_n = \E(X)$ (بمبرهنة النقل). أما
النص من الرتبة الثانية فيكرر الحجة درجة أعلى: فالمقدار $G''_X(t) = \sum_{n\geq2}n(n-1)p_nt^{n-2}$ غير
متناقص على $\intoo01$ ونهايته الرتيبة $\sum_nn(n-1)p_n = \E(X(X-1))$، وهو منته بالضبط عندما
يقبل $X$ عزما من الرتبة الثانية. وتنتج صيغة \omterm{def:b2:randomvar:variance}{التباين} عندئذ من صيغة
كونيغ--هويغنز:
\[
V(X) = \E(X^2) - \E(X)^2 = \E\bigl(X(X-1)\bigr) + \E(X) -
\E(X)^2 = G''_X(1) + G'_X(1) - G'_X(1)^2 .
\]
\end{proof}

\begin{example}
بواسون: $G'(t) = \lambda e^{\lambda(t-1)}$، ومنه $\E(X) = \lambda$؛ و$G''(1) = \lambda^2$، ومنه $V(X) = \lambda^2 + \lambda - \lambda^2 =
\lambda$ --- أي حسابا
\cref{ch:b2:randomvar} في سطر واحد لكل منهما.
\end{example}

\begin{example}[منوال قانون بواسون]
\label{ex:b2:genfun:poissonmode}
أين يكون $\P(X = k)$ أكبر ما يكون من أجل $X \sim \mathcal
P(\lambda)$؟ تُقارن الأوزان
المتتالية عبر النسبة
\[
\frac{\P(X = k+1)}{\P(X = k)} = \frac{\lambda}{k + 1} ,
\]
التي تتجاوز $1$ ما دام $k < \lambda - 1$ وتهبط دون $1$ بمجرد أن $k > \lambda - 1$:
فالأوزان تصعد ثم تهبط، والمنوال $\floor\lambda$ (مع تساو بين $\lambda - 1$ و$\lambda$
عندما يكون $\lambda$ عددا صحيحا: فمن أجل $\lambda = 3$، $\P(X = 2) = \P(X = 3) = \frac92\eu^{-3} \approx 0.224$).
واختبارات النسب على المعاملات كثيرا ما تكون أسرع طريق إلى حقائق
نوعية عن قانون متقطع --- دون حاجة إلى \omterm{ex:b2:powerseries:fibonacci}{دالة مولّدة}، لكن المعاملات
\emph{هي} \omterm{ex:b2:powerseries:fibonacci}{الدالة المولّدة}، مقروءة حدا حدا.
\end{example}

\section{مجاميع المتغيرات المستقلة}

\begin{theorem}[الضربية]\label{thm:b2:genfun:product}
إذا كان $X$ و$Y$ متغيرين عشوائيين مستقلين ذوي قيم في $\N$،
فإن
\[
G_{X + Y}(t) = G_X(t)\,G_Y(t)
\qquad (\abs t \leq 1),
\]
وبالتراجع $G_{X_1 + \dots + X_n} = \prod_i G_{X_i}$ من أجل متغيرات مستقلة $X_1, \dots, X_n$.
\end{theorem}

\begin{proof}
برهانان، وكلاهما مفيد. \emph{بالآمال الرياضية:} المتغيران $t^X$
و$t^Y$ مستقلان ومحدودان، ومنه (\cref{thm:b2:randomvar:product})
\[
G_{X+Y}(t) = \E\bigl(t^{X+Y}\bigr)
= \E\bigl(t^X t^Y\bigr)
= \E\bigl(t^X\bigr)\E\bigl(t^Y\bigr) .
\]
\emph{بجداءات كوشي:} قانون $X + Y$ هو الالتفاف $\P(X + Y = n) = \sum_{k=0}^n \P(X = k)\P(Y = n - k)$، ويضرب \omterm{thm:b2:series:fubini}{جداء
كوشي} للمتسلسلات المتقاربة مطلقا (\cref{ch:b2:series}) متسلسلتي القوى بالضبط وفق
هذا الالتفاف.
\end{proof}

\begin{example}[استقرار القوانين الكلاسيكية]\label{ex:b2:genfun:stability}
تُجمع المتغيرات الثنائية المستقلة ذات $p$ نفسه: $(1 - p + pt)^m(1 -
p + pt)^n = (1 - p + pt)^{m+n}$، ومنه
$\mathcal{B}(m, p) +
\mathcal{B}(n, p) = \mathcal{B}(m + n, p)$ --- وبوجه خاص فإن مجموع $n$ متغير برنولي مستقل ثنائي
الحد، وهو ما يعيد البرهان على قانون عدد النجاحات. وتُجمع متغيرات
بواسون المستقلة: $e^{\lambda(t-1)}e^{\mu(t-1)} = e^{(\lambda + \mu)(t-1)}$، ومنه $\mathcal{P}(\lambda) + \mathcal{P}(\mu) = \mathcal{P}(\lambda +
\mu)$ --- أي حساب الالتفاف في \cref{exo:b2:randomvar:2}،
دون حساب الآن.
\end{example}

\begin{example}[نردان، وكثير حدود واحد مربّع]\label{ex:b2:genfun:twodice}
من أجل نرد متزن واحد، $G(t) = \frac{t + t^2 + \dots + t^6}{6}$؛ ومن أجل مجموع نردين،
\[
G(t)^2 = \frac{1}{36}\bigl(t^2 + 2t^3 + 3t^4 + 4t^5 + 5t^6 +
6t^7 + 5t^8 + 4t^9 + 3t^{10} + 2t^{11} + t^{12}\bigr) :
\]
أي القانون المثلثي لمجاميع النرد (فالمنوال $7$ باحتمال $\frac6{36} = \frac16$)،
مقروءا من مربع كثير حدود يُنشر مرة واحدة في العمر. وكانت صيغة
الالتفاف ستتطلب إحدى عشرة حجة عد منفصلة؛ أما \omterm{ex:b2:powerseries:fibonacci}{الدالة المولّدة} فتنجزها
كلها في آن واحد، لأن ضرب كثيرات الحدود \emph{هو} التفاف المعاملات.
وهذه الترجمة الآلية --- من القوانين إلى المعاملات، ومن المجاميع إلى
الجداءات --- هي نموذج عمل الفصل كله، ويدفعها \cref{exo:b2:genfun:11} إلى نردي سيشرمان
المدهشين.
\end{example}

\begin{example}[ثلاثة نرد واستخراج معامل]\label{ex:b2:genfun:threedice}
من أجل مجموع $S$ لثلاثة نرد متزنة، يكون $\P(S = 10)$ معامل $t^{10}$ في
$\bigl(\frac{t + \dots +
t^6}6\bigr)^3$. ونفكك وننشر بمتسلسلتي ذات الحدين والهندسية:
\[
\Bigl(\frac{t(1 - t^6)}{6(1 - t)}\Bigr)^{\!3}
= \frac{t^3}{216}\,\bigl(1 - 3t^6 + 3t^{12} -
t^{18}\bigr)\sum_{j\geq0}\binom{j+2}2t^j .
\]
ويقتضي معامل $t^{10}$ أخذ $t^7$ من الجداء: $j = 7$ مع الحد $1$،
و$j = 1$ مع الحد $-3t^6$:
\[
\P(S = 10) = \frac{1}{216}\Bigl(\binom92 -
3\binom32\Bigr) = \frac{36 - 9}{216} = \frac{27}{216} =
\frac18 .
\]
والإحصاء المباشر للثلاثيات وعددها $27$ كثير الأخطاء؛ أما \omterm{def:b2:structures:algebra}{الجبر}
فآلي ويتحاكى مع أي عدد من النرد --- فالاحتواء والاستبعاد الظاهر في
$(1 - t^6)^3$ يقوم بمناقشة الحالات تلقائيا.
\end{example}

\begin{example}[قراءة قانون من دالته المولّدة]\label{ex:b2:genfun:identify}
أي قانون دالته المولّدة $G(t) = \dfrac1{2 - t}$؟ ننشره في متسلسلة قوى:
\[
\frac{1}{2 - t} = \frac12\cdot\frac1{1 - t/2}
= \sum_{k\geq0}\frac{t^k}{2^{k+1}} :
\]
فالمعاملات موجبة ومجموعها $G(1) = 1$، ومنه فهذا قانون حقيقي، $\P(X = k) = 2^{-(k+1)}$ على
$\N$ --- أي قانون هندسي يبدأ عند $0$. وبالوحدانية (\cref{prop:b2:genfun:radius})،
لا يشترك أي قانون آخر في هذه الدالة $G$. والتعرف على القوانين من
دوالها المولّدة مهارة تستحق التمرين: فهي كيف يُكشف المكرَّر التفرعي
الحرج $G_n(t) = \frac{n - (n-1)t}{n+1 - nt}$ في مسألة نهاية الأسبوع على أنه قانون هندسي مشروط
بالبقاء.
\end{example}

\begin{remark}
يسير الاستقرار في اتجاه واحد فقط: فمجاميع متغيرات بواسون المستقلة
بواسونية، أما الفروق فلا --- إذ يأخذ $X - Y$ قيما سالبة، ومنه فليس
له أي \omterm{ex:b2:powerseries:fibonacci}{دالة مولّدة}، ويقع \omterm{def:b2:randomvar:law}{قانونه} (\omterm{def:b2:randomvar:law}{توزيع} سكيلام) خارج عدة هذا الفصل.
وبالمثل فإن $\mathcal B(m, p) + \mathcal B(n, p')$ مع $p \neq p'$ \emph{ليس} ثنائي الحد: فللجداء $(1 - p +
pt)^m(1 - p' + p't)^n$
موضعا جذر متمايزان، بينما لكل \omterm{ex:b2:powerseries:fibonacci}{دالة مولّدة} ثنائية الحد جذر مضاعف
وحيد. وقراءة الاستقرار من أنماط الجذور استعراض صغير لمقدار البنية
التي يرمّزها كثير الحدود.
\end{remark}

\begin{remark}[مرشح جذور الوحدة]
التقويم عند $-1$ يفصل الزوجي عن الفردي؛ والتقويم عند جميع جذور
الوحدة من الرتبة $m$ يفصل كل صف بواق: فمع $\omega = \eu^{2\iu\pi/m}$،
\[
\P(X \equiv r \bmod m)
= \frac1m\sum_{j=0}^{m-1}\omega^{-jr}\,G_X(\omega^j),
\]
لأن متوسط $\omega^{j(k-r)}$ على $j$ يعطي $1$ إذا كان $k
\equiv r$ و$0$
فيما عدا ذلك. ومثال على العائد: من أجل مجموع $S$ لنردين متزنين،
يكون كل $G(\omega^j) = \frac16\sum_{k=1}^6
\omega^{jk} = -\frac16$ من أجل $j \neq 0$ (فمجموع جذور الوحدة السبعة من الرتبة
السابعة معدوم)، ومنه
\[
\P(7 \mid S) = \frac17\Bigl(1 +
6\cdot\frac1{36}\Bigr) = \frac16 ,
\]
وهو ما يؤكد العد في \cref{ex:b2:genfun:twodice} --- والطريقة تتحاكى مع أسئلة لا يتحاكى
معها العد المباشر.
\end{remark}

\begin{theorem}[المجاميع العشوائية: متطابقة فالد من أجل الدوال
المولّدة]\label{thm:b2:genfun:compound}
لتكن $(X_k)_{k\geq1}$ متغيرات مستقلة ذات قيم في $\N$ لها القانون نفسه
\omterm{ex:b2:powerseries:fibonacci}{والدالة المولّدة} $G_X$، وليكن $N$ متغيرا ذا قيم في $\N$
مستقلا عن المتغيرات $X_k$، ودالته المولّدة $G_N$. عندئذ يكون
للمجموع العشوائي $S = X_1 + \dots + X_N$ (مع $S = 0$ عندما $N = 0$) \omterm{ex:b2:powerseries:fibonacci}{الدالة المولّدة}
\[
G_S = G_N \circ G_X .
\]
وبوجه خاص، إذا قبل $N$ و$X_1$ أملين رياضيين، فإن $\E(S) =
\E(N)\,\E(X_1)$.
\end{theorem}

\begin{proof}
نشترط على $N$ (بالاحتمالات الكلية، \cref{thm:b2:proba:bayes}): فمن أجل $\abs t \leq 1$،
\[
G_S(t) = \sum_{n=0}^\infty \P(N = n)\,
\E\bigl(t^{X_1 + \dots + X_n}\bigr)
= \sum_{n=0}^\infty \P(N = n)\,G_X(t)^n
= G_N\bigl(G_X(t)\bigr),
\]
باستعمال الضربية من أجل كل $n$ ثابتة وقابلية جمع العائلة المزدوجة
كلها ($\abs{G_X(t)} \leq 1$). وتبديل ترتيب الجمع هو مبرهنة فوبيني من أجل العائلات
\omterm{def:b2:series:summable}{القابلة للجمع} (\cref{ch:b2:series}). وبالمفاضلة عند $1^-$ بقاعدة السلسلة
و\cref{thm:b2:genfun:moments}: $\E(S) = G_N'(G_X(1))\,G_X'(1)
= G_N'(1)G_X'(1) = \E(N)\E(X_1)$.
\end{proof}

\begin{example}[بواسون المركب: خسائر التأمين السنوية]\label{ex:b2:genfun:compoundpoisson}
تتلقى شركة تأمين $N \sim \mathcal P(\lambda)$ مطالبة في السنة، وتكلف كل مطالبة $X_k$
(بوحدات صحيحة، مستقلة ومتماثلة \omterm{def:b2:randomvar:law}{التوزيع}، دالتها المولّدة $G_X$،
ومتوسطها $\mu$، ومستقلة عن $N$). وبحسب \cref{thm:b2:genfun:compound}، يكون للخسارة
الكلية $S$
\[
G_S(t) = \eu^{\lambda(G_X(t) - 1)},
\qquad
\E(S) = \lambda\mu ,
\]
وبالمفاضلة مرتين عند $1^-$:
\[
V(S) = \lambda\,G_X''(1) + \lambda^2\mu^2 + \lambda\mu -
(\lambda\mu)^2 = \lambda\,\E(X^2) .
\]
ويتضمن \omterm{def:b2:randomvar:variance}{التباين} العزم \emph{الثاني} لمطالبة واحدة، لا تباينها: فمجموع
بواسون المركب يشعر بالمطالبة الكبيرة العارضة مرتين --- مرة عبر كم
عددها، ومرة عبر كم حجمها. ومن أجل $\lambda = 10$ مطالبة ذات قانون هندسي
متوسطه $2$ ($\E X^2 = 6$): $\E S = 20$، $V(S) = 60$، وتعطي متراجحة تشيبيشيف
(\cref{ch:b2:randomvar}) هوامش ملاءة قابلة للاستعمال أصلا. وهذا النمط من ``المجموع
الموقوف عشوائيا'' هو النمط نفسه الذي سيقود التراجع التفرعي في
\cref{prop:b2:genfun:branching}: فتركيب الدوال المولّدة هو جبر المجتمعات العشوائية.
\end{example}

\begin{remark}
استقلال $N$ عن الحدود ليس زينة. خذ $X_k \in \{0, 2\}$ بالاحتمالات نفسها
واجعل $N =
X_1$ (وهو متعلق على نحو صارخ): عندئذ يكون $S = X_1 + \dots + X_N$ هو $0$
عندما $X_1 = 0$، و$2 + X_2$ عندما $X_1 = 2$، ومنه $\E(S) =
\frac12(2 + 1) = \frac32$، بينما $\E(N)\E(X_1) = 1\cdot1 =
1$:
فمتطابقة فالد تسقط. وعندما يُسمح لعدد الحدود \emph{بأن يتفاعل} مع
الحدود نفسها، تنهار البنية الجدائية النظيفة --- والنظرية الكاملة
لقواعد ``التوقف'' هذه هي فصل المارتينغال في مجلد السنة الثالثة.
\end{remark}

\section{تقريب بواسون}

\begin{theorem}[قانون الأحداث النادرة]\label{thm:b2:genfun:rareevents}
ليكن $X_n \sim \mathcal{B}(n, p_n)$ مع $n\,p_n \to \lambda > 0$. عندئذ، من أجل كل $k \in \N$:
\[
\P(X_n = k)
\xrightarrow[n\to\infty]{}
e^{-\lambda}\frac{\lambda^k}{k!} :
\]
أي إن القانون الثنائي \omterm{def:b2:proba:independence}{لأحداث مستقلة} نادرة كثيرة يتقارب نحو قانون
بواسون ذي الوسيط $\lambda$.
\end{theorem}

\begin{proof}
بحساب مباشر مع $p_n = \frac{\lambda_n}{n}$، $\lambda_n
\to \lambda$:
\[
\P(X_n = k)
= \binom nk p_n^k(1 - p_n)^{n-k}
= \frac{n(n-1)\cdots(n-k+1)}{n^k}\cdot
\frac{\lambda_n^k}{k!}\,
\bigl(1 - \tfrac{\lambda_n}{n}\bigr)^{n-k} .
\]
وعندما $n \to \infty$ مع $k$ ثابتة: يؤول العامل الأول إلى $1$ (فهو
جداء $k$ عاملا $\to 1$)؛ و$\lambda_n^k \to \lambda^k$؛ و$\bigl(1 - \frac{\lambda_n}{n}\bigr)^{n-k} =
\exp\bigl((n-k)\ln(1 - \frac{\lambda_n}{n})\bigr) \to
e^{-\lambda}$ لأن $(n - k)\ln\bigl(1 - \frac{\lambda_n}{n}\bigr)
\sim -\lambda_n \to -\lambda$ (\cref{ch:b2:comparison}).
وبديلا عن ذلك، على مستوى الدوال المولّدة: $G_{X_n}(t) = \bigl(1 +
\frac{\lambda_n(t-1)}{n}\bigr)^n \to e^{\lambda(t - 1)} =
G_{\mathcal{P}(\lambda)}(t)$ من أجل كل $t \in [0, 1]$
ثابتة --- أي تقارب الدوال المولّدة، وهو مكافئ (من أجل المتغيرات ذات
القيم في $\N$) لتقارب كل $\P(X_n = k)$؛ انظر \cref{exo:b2:genfun:9}.
\end{proof}

\begin{remark}
ولهذا تنمذج قوانين بواسون عدود \omterm{def:b2:proba:space}{الأحداث} النادرة --- الأخطاء المطبعية
في الصفحة، والتفككات الإشعاعية في الثانية، والحوادث في اليوم عند
تقاطع: فكل فرصة تكاد تكون مهملة، والفرص كثيرة، ولا يبقى في النهاية
إلا المعدل المتوسط $\lambda$.
\end{remark}

\begin{example}[مشاهدة نهاية بواسون وهي تتقارب]\label{ex:b2:genfun:poissonnumerics}
نثبّت $\lambda = 2$ ونجعل $X_n \sim \mathcal B(n, 2/n)$. واحتمال عدم وقوع أي \omterm{def:b2:proba:space}{حدث} هو $\P(X_n = 0) = (1 - 2/n)^n$
بالضبط:
\[
n = 10:\ 0.107, \qquad
n = 20:\ 0.122, \qquad
n = 50:\ 0.130, \qquad
n = 100:\ 0.133,
\]
مقابل النهاية $\eu^{-2} \approx 0.135$. والتقارب رتيب وسرعته $O(1/n)$ --- فبالنشر،
$(1 -
2/n)^n = \eu^{-2}\bigl(1 - \tfrac2n + O(n^{-2})\bigr)$ --- ومنه فمن أجل $n$ بالمئات يكون نموذج بواسون دقيقا
أصلا إلى الرقم الثالث. وهذا هو المضمون العملي لقانون \omterm{def:b2:proba:space}{الأحداث} النادرة:
فالنمذج لا يعرف أبدا $n$ و$p$ على حدة (فكم فرصة مجهرية لخطأ
مطبعي تحملها صفحة؟)، بل يعرف جداءهما $\lambda$ وحده، والقانون النهائي
لا يتعلق برحمة بأي شيء آخر.
\end{example}

\section{مسارات التفرع}

لننظر في مجتمع ينطلق من سلف واحد؛ ولكل فرد، بصورة مستقلة، عدد عشوائي
من الأبناء \omterm{def:b2:randomvar:law}{قانونه} $(p_k)_{k
\in \N}$ ودالته المولّدة $G$ (وهو \emph{قانون
النسل}). وليكن $Z_n$ حجم الجيل $n$ ($Z_0 =
1$)، وليكن $m = G'(1) = \E(Z_1)$
متوسط عدد الأبناء.

\begin{proposition}\label{prop:b2:genfun:branching}
\omterm{ex:b2:powerseries:fibonacci}{الدالة المولّدة} للمتغير $Z_n$ هي المكرَّر رقم $n$ $G_{Z_n} = G \circ G \circ \dots \circ G$
($n$ مرة)، وتحقق احتمالات الانقراض $q_n = \P(Z_n = 0)$
\[
q_0 = 0, \qquad q_{n+1} = G(q_n),
\]
وتتزايد نحو احتمال الانقراض النهائي $q$، وهو نقطة ثابتة للدالة
$G$.
\end{proposition}

\begin{proof}
الجيل $n + 1$ هو المجموع العشوائي لنسل أفراد الجيل $n$ وعددهم
$Z_n$، والعدود مستقلة بعضها عن بعض وعن $Z_n$: ومنه يعطي \cref{thm:b2:genfun:compound}
المقدار $G_{Z_{n+1}} =
G_{Z_n} \circ G$، ويعطي التراجع انطلاقا من $G_{Z_0}(t) = t$ المكرَّر $n$
مرة --- وهو ما يمكن قراءته بالقدر نفسه، بحكم تجميعية التركيب، على
الصورة $G_{Z_{n+1}} = G \circ G_{Z_n}$. وبتقويم هذه الصيغة الثانية عند $0$: $q_{n+1} = G_{Z_{n+1}}(0) =
G\bigl(G_{Z_n}(0)\bigr) = G(q_n)$.
\omterm{def:b2:proba:space}{والأحداث} $\{Z_n = 0\}$ متزايدة (فالمجتمعات المنقرضة تبقى منقرضة)، ومنه
$q_n \uparrow q = \P\bigl(\bigcup_n\{Z_n = 0\}\bigr)$ بالاتصال الرتيب (\cref{thm:b2:proba:continuity})، ويحول اتصال $G$ على $[0, 1]$
المقدار $q_{n+1} = G(q_n)$ إلى $q =
G(q)$ عند النهاية.
\end{proof}

\begin{example}[مشاهدة الانقراض وهو يتقارب]
\label{ex:b2:genfun:cobwebnumerics}
من أجل قانون النسل $(p_0, p_1, p_2) = (\tfrac14,
\tfrac14, \tfrac12)$ في \cref{ex:b2:genfun:branchingexample}، لدينا $G(t) = \tfrac14 + \tfrac14t + \tfrac12t^2$ ويعطي التكرار
$q_{n+1} = G(q_n)$
\[
q_1 = 0.25, \quad q_2 = 0.34375, \quad q_3 \approx 0.39502,
\quad q_4 \approx 0.42678, \quad q_5 \approx 0.44776,
\]
صاعدا نحو احتمال الانقراض $q = \tfrac12$. والفروق $q - q_n$ هي $0.25$، $0.156$،
$0.105$، $0.073$، $0.052$: وكل منها نحو $\tfrac34$ من سابقه، وفعلا تعطي
مبرهنة القيمة المتوسطة $q - q_{n+1} = G'(c_n)(q
- q_n)$ مع $G'(q) = \tfrac14 + q = \tfrac34$. وعبرتان: أن لخط عائلة لا
يزال حيا عند الجيل $n$، وفق الحساب نفسه، احتمال $q - q_n$ في أن
يكون محكوما عليه بالفناء لاحقا؛ وأن سرعة تقارب الدرج في الشكل أدناه
هي المشتقة عند النقطة الثابتة --- وتحول مسألة نهاية الأسبوع الملاحظتين
كلتيهما إلى مبرهنتين.
\end{example}

\begin{theorem}[محك الانقراض]\label{thm:b2:genfun:extinction}
لنفترض $p_1 \neq 1$. احتمال الانقراض $q$ هو النقطة الثابتة
\emph{الصغرى} للدالة $G$ في $\intcc{0}{1}$، ولدينا:
\begin{itemize}
\item إذا كان $m \leq 1$ (تحت حرج أو حرج)، فإن $q = 1$: أي إن الانقراض
مؤكد؛
\item إذا كان $m > 1$ (فوق حرج)، فإن $q < 1$: أي إن المجتمع يبقى إلى
الأبد باحتمال موجب $1 - q$.
\end{itemize}
\end{theorem}

\begin{proof}
$G$ محدبة على $\intcc{0}{1}$ (فهي متسلسلة قوى ذات معاملات موجبة: $G'' \geq 0$)،
وغير متناقصة، مع $G(1) = 1$.

\emph{النقطة الثابتة الصغرى:} لتكن $r \in \intcc{0}{1}$ نقطة ثابتة كيفما كانت.
عندئذ $q_0 = 0 \leq r$، وبالتراجع $q_{n+1} = G(q_n)
\leq G(r) = r$ (بالرتابة): ومنه $q = \lim q_n \leq r$.

\emph{الحالة $m \leq 1$:} لنفترض أن $r < 1$ نقطة ثابتة. فبمبرهنة القيمة
المتوسطة على $[r, 1]$، يوجد $c \in \intoo{r}{1}$ يحقق $G'(c) = \frac{G(1) - G(r)}{1 - r} = \frac{1 - r}{1 - r} = 1$. لكن $G'$ غير
متناقصة (بالتحدب) و$\lim_{t\to1^-}G'(t) = m
\leq 1$، ومنه $G' \leq 1$ على $\intoo{0}{1}$؛ وتفرض المساواة
$G'(c) =
1$ عندئذ أن تكون $G'$ ثابتة تساوي $1$ على $\intco{c}{1}$، ومنه
$G'' = \sum n(n-1)p_nt^{n-2} \equiv 0$ هناك. ومتسلسلة قوى ذات معاملات موجبة تنعدم على فترة تكون هذه
المعاملات كلها معدومة: $p_n = 0$ من أجل $n \geq 2$، ومنه $G(t) = p_0
+ p_1t$ و$1 = G'(c) = p_1$
--- وهو ما يناقض الفرضية $p_1 \neq 1$. ومنه فإن $1$ هي النقطة الثابتة
الوحيدة: $q = 1$.

\emph{الحالة $m > 1$:} بجوار $1$، لمشتقة $G(t) - t$ القيمة $G'(t) -
1 \to m - 1 > 0$
عندما $t \to 1^-$، ومنه $G(t) - t < G(1) - 1 = 0$ على فترة ما $\intoo{1 - \delta}{1}$: فالدالة المتصلة
$G(t) - t$ تساوي $\geq 0$ عند $t = 0$ ($G(0) = p_0 \geq 0$) وتكون $< 0$ تحت $1$
مباشرة، ومنه فهي تنعدم عند $r < 1$ ما (بمبرهنة القيم الوسطى).
والنقطة الثابتة الصغرى عندئذ $q \leq r < 1$.
\end{proof}

\begin{remark}[كيف تقرأ مخطط بيت العنكبوت]
في الشكل، تطبّق حركة شاقولية الدالة $G$ (من $(q_n, q_n)$ صعودا إلى
$(q_n, G(q_n))$)، وتحوّل حركة أفقية نحو القطر الخرج إلى دخل: فالدرج \emph{هو}
التراجع $q_{n+1} = G(q_n)$. ولا يترك تحدب $G$ و$G(1) = 1$ إلا هندستين. فإما أن
يبقى المنحنى فوق القطر على $\intco01$ (بمتوسط $m \leq 1$): فلا يجد الدرج
موضعا يتوقف عنده قبل $1$. وإما أن يعبر المنحنى عند $q <
1$ ما
($m > 1$): فينحصر الدرج تحت نقطة العبور ويتقارب نحوها، بالمعدل
الهندسي $G'(q) < 1$ المقيس في \cref{ex:b2:genfun:cobwebnumerics}. وكل تحليل مبرهنة الانقراض مرئي
في هذه الصورة الواحدة --- ولهذا تستحق أن تُرسم قبل الحساب.
\end{remark}

\begin{example}\label{ex:b2:genfun:branchingexample}
قانون النسل: لا ابن، أو ابن واحد، أو ابنان بالاحتمالات $\frac14, \frac14, \frac12$.
عندئذ $m = \frac14 + 1 =
\frac54 > 1$ و$G(t) = \frac14 + \frac14 t + \frac12 t^2$. والنقط الثابتة: $\frac12 t^2 - \frac34 t + \frac14 = 0$، أي $2t^2 - 3t
+ 1 = (2t - 1)(t - 1) = 0$: $q = \frac12$.
فينقرض خط العائلة باحتمال $\frac12$ --- وباحتمال $\frac12$ يعيش إلى الأبد.
\end{example}

\begin{remark}[آفاق داخل هذا المجلد]
هذا الفصل مفترق طرق الكتاب، وقد وصل كل مكوّن من مكان مسمى: فجبر
المتسلسلات من \cref{ch:b2:series} و\cref{ch:b2:powerseries}، والاحتمالات من \cref{ch:b2:proba} (فالاتصال
الرتيب يبرهن على $q_n \uparrow q$) ومن \cref{ch:b2:randomvar} (فالمقدار $G_X =
\E(t^X)$ \omterm{def:b2:randomvar:expectation}{أمل رياضي}،
والضربية هي مبرهنة الجداء)، والتحدب من \cref{ch:b2:realfun} عبر \cref{ch:b2:affine}. بل حتى
أمراض الذيول الثقيلة متصلة: فلمتغير سان بطرسبورغ في الفصل السابق
الدالة $G(t) = \sum_k2^{-k}t^{2^k}$، وهي متسلسلة متقاربة تماما على $\intcc01$ لكن مشتقتها
عند $1^-$ متباعدة --- أي متوسط لانهائي، مرئي في لمحة. كائن واحد،
وكل أدوات السنة: فصل أخير لائق.
\end{remark}

\begin{remark}[مزالق شائعة]
(1) لا تنطبق الدوال المولّدة إلا على المتغيرات ذات القيم في $\N$:
ففي حالة المتغيرات ذات الإشارة أو غير الصحيحة يفقد الكائن $\E(t^X)$
بنيته بوصفه متسلسلة قوى (وتستبدل به السنة الثالثة تحويلات ملائمة
للفضاء $\R$). (2) التحقق الأول من أي $G$ محسوبة هو $G(1) = 1$؛
والثاني هو أن تكون المعاملات موجبة --- فالمعامل السالب يعني زلة جبرية
لا قانونا جديدا. (3) وفي المجاميع العشوائية يهم ترتيب التركيب:
$G_S = G_N \circ G_X$، فالدالة \emph{الخارجية} هي التي تعد الحدود؛ والتركيب في
الاتجاه الآخر بلا معنى (فالمقدار $G_X \circ G_N$ سيعد عناصر العناصر). (4)
وتحتاج الضربية إلى الاستقلال وإلى مصادر عشوائية متمايزة: $G_{2X}(t) = G_X(t^2)$، لا
$G_X(t)^2$. (5) والمفاضلة عند $1$ عملية حدية: فعندما يكون نصف القطر
$1$ بالضبط، كما في \cref{ex:b2:genfun:heavytail}، قد يكون $G'(1^-)$ لانهائيا، ولا تكون
صياغة مبرهنة العزوم بالنهاية الرتيبة تنطعا بل هي النص الصادق.
\end{remark}

\section*{ختام المجلد}

\omterm{ex:b2:powerseries:fibonacci}{الدالة المولّدة} كائن ختامي لائق بهذا الكتاب: فهي في آن واحد متسلسلة
قوى (\cref{ch:b2:powerseries})، وأداة من أدوات العائلات \omterm{def:b2:series:summable}{القابلة للجمع} (\cref{ch:b2:series})، \omterm{def:b2:randomvar:expectation}{وأمل
رياضي} (\cref{ch:b2:randomvar})، ودالة محدبة تقرر هندستها الانقراض (\cref{ch:b2:realfun})، وتكرار
نقطة ثابتة (\cref{ch:b2:metric}). فرياضيات السنة الثانية موضوع واحد. وسيفتح
مجلد السنة الثالثة الأبواب المتروكة مغلقة عمدا هنا: تكامل لوبيغ

القياس على الفضاءات غير \omterm{def:b2:structures:countable}{القابلة للعد}، والبرهان الكامل لمبرهنة الدالة
العكسية (\cref{ch:b2:diffcalc}) في إطار الهندسة التفاضلية.

\begin{exercise}[$\star\star$]\label{exo:b2:genfun:5}

هندسية مستقلة عددها $r$، واستنتج

G_{T_r}(t) = \Bigl(\frac{pt}{1 - (1-p)t}\Bigr)^{r},
\qquad
\E(T_r) = \frac rp,
\qquad
V(T_r) = \frac{r(1-p)}{p^2},
\]
وانشر $G_{T_r}$ لإيجاد $\P(T_r = n) = \binom{n-1}{r-1}
p^r(1-p)^{n-r}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:8}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:9}

\section{مسألة: مسار غالتون--واطسون، محلولا}

\begin{problem}[{مسألة نهاية الأسبوع --- معدلات النمو، والحلول
المضبوطة، والنسل الكلي، وتقدير كولموغوروف الحرج}]\label{pb:b2:genfun:1}
يقسم محك الانقراض (\cref{thm:b2:genfun:extinction}) مسارات التفرع\index{مسار تفرع} إلى تحت
حرجة وحرجة وفوق حرجة --- لكنه لا يقول شيئا عن \emph{المعدلات}: كم
بسرعة يموت خط محكوم عليه، وكم يكبر خط باق. وتحسب هذه المسألة ذلك.
ونحتفظ برموز الفصل: قانون النسل $(p_k)$ ودالته المولّدة $G$،
والمتوسط $m = G'(1)$، وأحجام الأجيال $Z_n$ ($Z_0 = 1$)، والمكرَّرات
$G_n = G_{Z_n}$، واحتمالات الانقراض $q_n = \P(Z_n = 0)
\uparrow q$؛ ونفترض دائما $p_1 \neq 1$، ونفترض
$G''(1) < \infty$ حيث تظهر عزوم من الرتبة الثانية، ونكتب $\sigma^2 =
V(Z_1)$.

\medskip
\noindent\textbf{الجزء الأول --- عزوم الأجيال.}
\begin{enumerate}
  \item بيّن $\E(Z_n) = m^n$ \emph{(بقاعدة السلسلة على $G_n = G
        \circ G_{n-1}$ عند $1^-$،
        باستعمال $G_{n-1}(1) = 1$ و\cref{thm:b2:genfun:moments})}.
  \item أنشئ التراجع $G_n''(1) =
        G''(1)\,m^{2(n-1)} + m\,G_{n-1}''(1)$ وحله:
        $G_n''(1) = G''(1)\,m^{n-1}\dfrac{m^n - 1}{m - 1}$
        من أجل $m \neq 1$، و$G_n''(1) = n\,G''(1)$ من أجل $m =
        1$.
  \item استنتج
        \[
        V(Z_n) = \sigma^2m^{n-1}\,\frac{m^n - 1}{m - 1}
        \quad (m \neq 1),
        \qquad
        V(Z_n) = n\,\sigma^2 \quad (m = 1).
        \]
  \item (المعدل تحت الحرج، الحصر العلوي) من أجل $m < 1$، بيّن $\P(Z_n > 0) \leq m^n$
        \emph{(بمتراجحة ماركوف على المتغير الصحيح $Z_n$)}: أي إن
        الانقراض مؤكد بمعدل هندسي --- وهو تدقيق كمي لمحك الفصل.
  \item (المعدل تحت الحرج، الحصر السفلي) باستعمال متراجحة
        كوشي--شوارتز على $Z_n\mathbf 1_{Z_n > 0}$، بيّن
        \[
        \P(Z_n > 0) \geq \frac{\E(Z_n)^2}{\E(Z_n^2)}
        \geq c\,m^{n}
        \quad\text{مع}\quad
        c = \Bigl(\frac{\sigma^2}{m(1-m)} + 1\Bigr)^{-1} :
        \]
        أي إن المعدل الهندسي $m^n$ مضبوط إلى غاية ثوابت.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- العائلة الهندسية، محلولة
بالضبط.} ليكن قانون النسل هندسيا على $\N$: $p_k =
qp^k$ ($k \geq 0$)، مع
$0 < p < 1$، $q = 1 - p$.
\begin{enumerate}[resume]
  \item احسب $G(t) = \dfrac{q}{1 - pt}$ و$m = \dfrac
        pq$؛ وحدد الأنظمة الثلاثة بدلالة $p$.
  \item حل $G(t) = t$: بيّن أن النقطتين الثابتتين هما $1$ و$q/p = 1/m$،
        واستعد احتمال الانقراض $q_{\mathrm{ext}} = \min(1, 1/m)$.
  \item برهن بالتراجع على الصيغتين المغلقتين
        \[
        q_n = \frac{m^n - 1}{m^{n+1} - 1} \quad (m \neq 1),
        \qquad
        q_n = \frac{n}{n+1} \quad (m = 1).
        \]
  \item استنتج المعدلات المضبوطة: $1 - q_n \sim (1 - m)\,m^n$ في الحالة تحت الحرجة،
        و$q_{\mathrm{ext}} - q_n
        \sim \dfrac{m - 1}{m^{2}}\cdot m^{-n}$ في الحالة فوق الحرجة؛ وتحقق من أن نسبة التقلص فوق
        الحرجة هي $G'(q_{\mathrm{ext}}) = 1/m$.
  \item الحالة الحرجة ($p = \tfrac12$): احسب $\sigma^2 =
        2$ ولاحظ $1 - q_n = \frac1{n+1}$: فاحتمال
        البقاء يتلاشى مثل $\frac1n$ --- فلا هو هندسي ولا قابل للجمع.
  \item وما زلنا في الحالة الحرجة: برهن بالتراجع على المكرَّر الكامل
        \[
        G_n(t) = \frac{n - (n-1)t}{n + 1 - nt},
        \]
        واستنتج أنه بشرط البقاء يكون $Z_n$ هندسيا على $\N^*$
        بالوسيط $\frac1{n+1}$:
        \[
        \P(Z_n = k \mid Z_n > 0) = \frac1{n+1}
        \Bigl(\frac{n}{n+1}\Bigr)^{k-1},
        \qquad
        \E(Z_n \mid Z_n > 0) = n + 1 .
        \]
        فالخط المتوسط يموت، لكن الخطوط الباقية حجمها من رتبة $n$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- النسل الكلي.} ليكن $Y =
\sum_{n\geq0}Z_n \in \N^* \cup \{\infty\}$ العدد
الكلي للأفراد المولودين على الإطلاق، وليكن $H(t) =
\sum_{k\geq1}\P(Y = k)t^k$.
\begin{enumerate}[resume]
  \item برر $\P(Y < \infty) = q_{\mathrm{ext}}$، وذكّر من \cref{exo:b2:genfun:7} بالمعادلة الدالية $H(t) = t\,G(H(t))$ (التي
        لم يستعمل اشتقاقها $m < 1$).
  \item (التفرع الثنائي) من أجل $p_0 = p_2 = \frac12$ (الحرج)، حل المعادلة الدالية:
        \[
        H(t) = \frac{1 - \sqrt{1 - t^2}}{t},
        \]
        وانشرها بالمقدار \cref{ex:b2:powerseries:catalan} للحصول على
        \[
        \P(Y = 2k + 1) = \frac{C_k}{2^{2k+1}},
        \qquad C_k = \frac1{k+1}\binom{2k}k ;
        \]
        وتحقق من القيمتين $\P(Y = 1) = \frac12$ و$\P(Y = 3)
        = \frac18$ بالعد المباشر.
  \item بمفاضلة المعادلة الدالية عند $1^-$، بيّن أن $\E(Y) = \frac{1}{1-m}$ من أجل
        $m < 1$، بينما تفرض الحرجية $\E(Y) = \infty$: أي إن النسل الكلي الحرج
        منته بشكل شبه أكيد بمتوسط لانهائي.
  \item مع السلوك المقارب للمعامل الثنائي المركزي (\cref{ex:b2:comparison:centralbinomial})، بيّن
        \[
        \P(Y = 2k+1) \sim \frac{1}{2\sqrt\pi\,k^{3/2}},
        \]
        أي ذيلا ثقيلا $k^{-3/2}$، واستنتج $\P(Y > n)
        \asymp n^{-1/2}$ (ويكفي حصران علوي
        وسفلي من هذه الرتبة).
  \item قارن مع \omterm{pb:b2:proba:1}{السير العشوائي} المتزن (مسألة نهاية الأسبوع في
        \cref{ch:b2:proba}): فهناك أزمنة عودة مؤكدة بمتوسط لانهائي، وهنا نسل
        كلي مؤكد بمتوسط لانهائي، وكلاهما بقوانين موضعية $n^{-3/2}$.
        وفقرة واحدة عن سبب إنتاج الحرجية لهذه البصمة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- تقدير كولموغوروف عند الحرجية.}
لنفترض $m = 1$، $0 < \sigma^2 = G''(1) <
\infty$.
\begin{enumerate}[resume]
  \item بيّن أن $G''$ يمتد بالاتصال إلى $\intcc01$ \emph{(فهو موجب
        متزايد ذو نهاية منتهية)} واستنتج نشر تايلور عند $1$:
        \[
        G(t) = t + b\,(1-t)^2 + o\bigl((1-t)^2\bigr),
        \qquad b = \frac{G''(1)}2 = \frac{\sigma^2}2 .
        \]
  \item من أجل $t \in \intco01$، ضع $h(t) = \dfrac1{1 - G(t)} -
        \dfrac1{1 - t}$. بيّن
        \[
        h(t) = \frac{G(t) - t}{(1 - G(t))(1 - t)}
        \xrightarrow[t\to1^-]{} b .
        \]
  \item تلسكب على طول التكرار $q_{j+1} = G(q_j)$:
        \[
        \frac1{1 - q_n} = 1 + \sum_{j=0}^{n-1}h(q_j),
        \]
        واختم بحجة تشيزارو أن
        \[
        \P(Z_n > 0) = 1 - q_n \sim \frac{2}{\sigma^2\,n}
        \]
        --- وهو \emph{تقدير كولموغوروف}: فكل \omterm{pb:b2:genfun:1}{مسار تفرع} حرج يموت
        بالمعدل الشامل $1/n$، ولا يتذكر قانون النسل إلا الثابت.
  \item تحقق من التقدير بمقابلته بالحالة الهندسية الحرجة في السؤال
        10.
  \item استنتج $\E(Z_n \mid Z_n > 0) = \dfrac{1}{1 - q_n}
        \sim \dfrac{\sigma^2 n}{2}$ \emph{(لاحظ $\E(Z_n
        \mathbf 1_{Z_n>0}) = \E(Z_n) = 1$)}، وتحقق منه بمقابلته
        بالسؤال 11: فبشرط البقاء ينمو المجتمع \emph{خطيا} --- وهو
        الحبل المشدود الحرج بين الموت والانفجار.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- تطبيقات وتركيب.}
\begin{enumerate}[resume]
  \item (الأوبئة والتفاعلات المتسلسلة) من أجل قانون نسل بواسوني
        $\mathcal P(\lambda)$ --- إذ تعدي كل حالة $\mathcal P(\lambda)$ حالة جديدة --- اكتب
        معادلة الانقراض $q = \eu^{\lambda(q-1)}$ وحلها عدديا من أجل $\lambda = 1.5$ ($q \approx
        0.417$)
        ومن أجل $\lambda = 2$ ($q \approx 0.203$): فانطلاقا من حالة واحدة، لا يكون
        تفشٍّ كبير \emph{مؤكدا} حتى عندما $\lambda > 1$. واشرح لماذا
        يتقارب التكرار $q_{n+1} = \eu^{\lambda(q_n - 1)}$ انطلاقا من $q_0 = 0$ نحو الجذر الصحيح.
  \item انطلاقا من $k$ سلفا بدل واحد، بيّن أن احتمال الانقراض هو
        $q^k$. وتطبيقا: مع $\lambda = 1.5$، كم حالة ابتدائية تجعل احتمال
        التفشي لا يقل عن $99\%$؟
  \item (شرط مسار فوق حرج على الانقراض) من أجل $m > 1$ باحتمال
        انقراض $q \in
        \intoo01$: برهن أولا بالتحدب على أن $G'(q) < 1$ عند النقطة
        الثابتة الصغرى، واستنتج
        $q_{\mathrm{ext}} - q_n = O\bigl(G'(q)^n\bigr)$
        (أي تقاربا هندسيا، كما يبين السؤال 9). ثم بيّن أن $\widehat G(t) = G(qt)/q$
        \omterm{ex:b2:powerseries:fibonacci}{دالة مولّدة} لقانون نسل حقيقي، متوسطه $\widehat m =
        G'(q) < 1$: أي مسار رفيق
        تحت حرج. وتحقق من ذلك على العائلة الهندسية: فشرط المسار فوق
        الحرج ذي الوسيط $(p, q)$ على الانقراض يبادل بين $p$
        و$q$. (والنص الكامل --- وهو أن المسار المشروط \emph{هو}
        المسار الرفيق --- مبرهن في مجلد السنة الثالثة؛ وقد تحققت
        هنا من ظله على مستوى الدوال المولّدة.)
  \item تركيب: ارسم جدول الثلاثية --- من أجل $m <
        1$ و$m = 1$
        و$m > 1$: قيمة $q$؛ ومعدل $\P(Z_n >
        0)$ أو $q - q_n$؛ و$\E(Y)$؛
        وحجم جيل باق. واذكر في جملة واحدة لكل أداة كيف حمل تركيب
        الدوال المولّدة والتحدب وتايلور عند $1^-$ ومتوسطات تشيزارو
        المسألة كلها، وماذا يضيف مجلد السنة الثالثة (المارتينغال
        $Z_n/m^n$ وقانون ياغلوم النهائي الأسي).
\end{enumerate}
\end{problem}
