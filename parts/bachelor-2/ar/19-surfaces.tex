\chapter{السطوح}\label{ch:b2:surfaces}

بعد المنحنيات، السطوح: وهي كائنات ذات وسيطين في $\R^3$. ويوفر حساب
\omterm{def:b2:diffcalc:differential}{التفاضل} في \cref{ch:b2:diffcalc} كل ما نحتاج إليه --- فالمشتقات الجزئية تعطي المتجهات
المماسة، والجداء الشعاعي يعطي الناظم، والمحددات تعطي المساحات. ونعرّف
\omterm{def:b2:surfaces:param}{السطوح الموسّمة} النظامية ومستوياتها المماسة و\emph{\omterm{def:b2:surfaces:fff}{الصورة الأساسية
الأولى}}، التي ترمّز جميع قياسات الأطوال والمساحات على السطح. وتنشأ
السطوح أيضا مجموعاتِ مستوى $f(x, y, z) = c$؛ ويوجه التدرج عندئذ الناظم.

\section{السطوح الموسّمة}

\begin{definition}[السطح الموسّم النظامي]\label{def:b2:surfaces:param}
ليكن $U \subseteq \R^2$ مفتوحا. \emph{السطح الموسّم}\index{سطح موسم} من الصنف
$\mathcal{C}^k$ ($k \geq 1$) هو تطبيق $\sigma \colon U \to
\R^3$، $(u, v) \mapsto \sigma(u, v)$، من الصنف $\mathcal{C}^k$. وتكون النقطة
\emph{نظامية} إذا كانت متجهتا المشتقتين الجزئيتين
\[
\sigma_u(u, v) = \frac{\partial\sigma}{\partial u}(u,v),
\qquad
\sigma_v(u, v) = \frac{\partial\sigma}{\partial v}(u,v)
\]
مستقلتين خطيا، أي $\sigma_u \wedge \sigma_v \neq 0$؛ ويكون السطح \emph{نظاميا} إذا كانت كل نقطة
كذلك.
\end{definition}

\begin{example}[الأوصاف القياسية الثلاثة]\label{ex:b2:surfaces:standard}
\leavevmode
\begin{enumerate}
\item \emph{التمثيل البياني:} $\sigma(u, v) = (u,\ v,\ f(u, v))$ من أجل $f \in
\mathcal{C}^1(U)$. وهو نظامي دائما:
فالمتجهتان $\sigma_u = (1, 0, f_u)$ و$\sigma_v = (0, 1, f_v)$ مستقلتان.
\item \emph{الكرة (بالإحداثيات الكروية):} من أجل الكرة ذات نصف
القطر $R$،
\[
\sigma(\theta, \varphi)
= (R\cos\theta\cos\varphi,\ R\sin\theta\cos\varphi,\
R\sin\varphi),
\qquad
(\theta, \varphi) \in \R \times
\bigl(-\tfrac\pi2, \tfrac\pi2\bigr),
\]
حيث $\theta$ هو الطول و$\varphi$ هو العرض. ونتحقق من أن $\norm{\sigma_\theta \wedge \sigma_\varphi} = R^2\cos\varphi > 0$: فهو نظامي
بعيدا عن القطبين (اللذين تهملهما هذه الخريطة).
\item \emph{مجموعة المستوى:} $S = \{(x,y,z) : f(x, y, z) = c\}$ حيث $f$ من الصنف $\mathcal{C}^1$
و$\nabla f \neq 0$ على $S$. فبجوار كل نقطة، يمكن التعبير عن إحدى الإحداثيات
بدلالة الأخريين بمبرهنة الدالة الضمنية (\cref{ch:b2:diffcalc})، ومنه فإن $S$
تمثيل بياني محليا.
\end{enumerate}
\end{example}

\begin{example}[من مجموعة المستوى إلى التمثيل البياني]
\label{ex:b2:surfaces:levelgraph}
تستحق مبرهنة الدالة الضمنية في البند 3 تشغيلا صريحا واحدا. خذ الكرة
$x^2 + y^2 + z^2 = R^2$ بجوار قطبها الشمالي $(0, 0, R)$: فهناك $\frac{\partial f}{\partial z} = 2z =
2R \neq 0$، والحل بدلالة $z$
يعطي خريطة التمثيل البياني
\[
z = \sqrt{R^2 - x^2 - y^2},
\qquad x^2 + y^2 < R^2 ,
\]
وهي نظامية في كل نقطة من مجالها (المفتوح) --- بما فيها القطب الذي
فوّتته الخريطة الكروية. وبجوار نقطة من خط الاستواء مثل $(R, 0, 0)$، تحل
المبرهنة نفسها بدلالة $x$ بدلا من ذلك ($\frac{\partial f}{\partial x} = 2R \neq 0$). والقاعدة العملية:
سطح المستوى تمثيل بياني فوق المستوي الإحداثي \emph{المتعامد مع أكبر
مركبات التدرج}، وبتغطية الكرة بست خرائط تمثيل بياني من هذا النوع
نتحقق من ملاستها في كل نقطة دون أي حساب مثلثي البتة.
\end{example}

\section{المستوي المماس والناظم}

\begin{definition}[المستوي المماس]\label{def:b2:surfaces:tangent}
ليكن $\sigma$ نظاميا في $(u_0, v_0)$، $M_0 = \sigma(u_0, v_0)$. \emph{المستوي المماس}\index{مستو مماس}
$T_{M_0}S$ هو المستوي المار بالنقطة $M_0$ والموجه بالمتجهتين $\operatorname{Vect}(\sigma_u, \sigma_v)$
(المشتقتان الجزئيتان في $(u_0, v_0)$). و\emph{الناظم الواحدي}\index{ناظم واحدي}
هو
\[
n(u_0, v_0)
= \frac{\sigma_u \wedge \sigma_v}
       {\norm{\sigma_u \wedge \sigma_v}} .
\]
\end{definition}

\begin{proposition}[المتجهات المماسة متجهات سرعة]\label{prop:b2:surfaces:velocity}
اتجاه $T_{M_0}S$ هو بالضبط مجموعة المتجهات $\gamma'(0)$، حيث يجري $\gamma = \sigma \circ c$ على
المنحنيات من الصنف $\mathcal{C}^1$ المرسومة على السطح والمارة بالنقطة $M_0$
(أي إن $c \colon (-\varepsilon, \varepsilon) \to U$ يساوي $\mathcal{C}^1$ مع $c(0) = (u_0, v_0)$).
\end{proposition}

\begin{proof}
إذا كان $c(t) = (u(t), v(t))$، فإن قاعدة السلسلة (\cref{ch:b2:diffcalc}) تعطي
\[
\gamma'(0) = u'(0)\,\sigma_u + v'(0)\,\sigma_v
\in \operatorname{Vect}(\sigma_u, \sigma_v).
\]
وبالعكس، تُبلَغ المتجهة $a\sigma_u + b\sigma_v$ بالمنحنى $c(t) = (u_0 + at,\ v_0 + bt)$، الذي يبقى في المفتوح
$U$ من أجل $\abs t$ صغيرة.
\end{proof}

\begin{example}[المستوي المماس للسطح اللولبي]
\label{ex:b2:surfaces:helicoidtangent}
من أجل السطح اللولبي $\sigma(u, v) = (v\cos u,\ v\sin u,\ au)$، في النقطة $\sigma(0, 1) = (1, 0, 0)$:
\[
\sigma_u = (0,\ 1,\ a), \qquad \sigma_v = (1,\ 0,\ 0),
\qquad
\sigma_u \wedge \sigma_v = (0,\ a,\ -1),
\]
\omterm{def:b2:surfaces:tangent}{فالمستوي المماس} هو $a\,y = z$. وهو يحوي المولّد المستقيمي الأفقي كله
$\{(t, 0, 0)\}$ (باتجاه $\sigma_v$): وكما في مخروط \cref{exo:b2:surfaces:1}، فإن كل سطح مسطّر
بمستقيمات يقع كل مولّد مستقيمي فيه داخل \omterm{def:b2:surfaces:tangent}{المستوي المماس} على طوله.
والاتجاه \omterm{def:b2:curves:arc}{المماس} الآخر $\sigma_u$ هو سرعة اللولب $u \mapsto \sigma(u, 1)$: فخريطة واحدة،
ومنحنيان مرسومان، ويتولد \omterm{def:b2:surfaces:tangent}{المستوي المماس} كله --- أي \cref{prop:b2:surfaces:velocity} في العمل.
\end{example}

\begin{proposition}[ناظم سطح مستوى]\label{prop:b2:surfaces:gradient}
ليكن $S = \{f = c\}$ مع $f$ من الصنف $\mathcal{C}^1$ و$\nabla
f(M_0) \neq 0$. عندئذ يكون \omterm{def:b2:surfaces:tangent}{المستوي
المماس} للسطح $S$ في $M_0$ هو المستوي المار بالنقطة $M_0$
والمتعامد مع $\nabla f(M_0)$:
\[
T_{M_0}S :\quad
\langle \nabla f(M_0),\ M - M_0\rangle = 0 .
\]
\end{proposition}

\begin{proof}
من أجل أي منحن $\gamma$ مرسوم على $S$ ومار بالنقطة $M_0$، يكون
$f(\gamma(t)) =
c$ تطابقا، ومنه تعطي قاعدة السلسلة $\langle
\nabla f(M_0), \gamma'(0)\rangle = 0$: أي إن جميع متجهات السرعة
متعامدة مع التدرج، ومنه فإن الاتجاه \omterm{def:b2:curves:arc}{المماس} محتوى في المستوي $\nabla f(M_0)^\perp$.
وكلاهما فضاء جزئي بعده $2$ --- الاتجاه \omterm{def:b2:curves:arc}{المماس} لأن $S$ تمثيل
بياني نظامي محليا (\cref{ex:b2:surfaces:standard})، والمتمم المتعامد لأن $\nabla f(M_0) \neq 0$ --- ومنه فهما
متساويان.
\end{proof}

\begin{example}\label{ex:b2:surfaces:spheretangent}
من أجل الكرة $x^2 + y^2 + z^2 = R^2$: $\nabla f = 2(x, y, z)$، ومنه فإن \omterm{def:b2:surfaces:tangent}{المستوي المماس} في $M_0$ متعامد
مع نصف القطر $\vect{OM_0}$ --- وهي الحقيقة الكلاسيكية القائلة إن نصف القطر
\omterm{def:b2:surfaces:tangent}{والمستوي المماس} متعامدان، بالمعادلة $\langle M_0, M\rangle = R^2$.
\end{example}

\begin{example}[المستوي المماس لتمثيل بياني]\label{ex:b2:surfaces:graphtangent}
من أجل $z = f(x, y)$ في $(x_0, y_0)$: بتطبيق \cref{prop:b2:surfaces:gradient} على $F(x,y,z) = f(x,y) - z$،
\[
z = f(x_0, y_0)
+ f_x(x_0, y_0)(x - x_0)
+ f_y(x_0, y_0)(y - y_0) ,
\]
أي الجزء الأفيني من نشر تايلور من الرتبة الأولى --- \omterm{def:b2:surfaces:tangent}{فالمستوي المماس}
هو التمثيل البياني \omterm{def:b2:diffcalc:differential}{للتفاضل}، كما يجب.
\end{example}

\begin{example}[أقرب نقطة من سطح]\label{ex:b2:surfaces:closest}
أي نقط المجسم المكافئ $z = x^2 + y^2$ أقرب إلى $P = (0, 0, 1)$؟ نصغّر مربع المسافة
على طول السطح: فمع $\rho^2 = x^2 + y^2$،
\[
g(\rho^2) = \rho^2 + (\rho^2 - 1)^2,
\qquad
g'(\rho^2) = 1 + 2(\rho^2 - 1) = 0
\iff \rho^2 = \tfrac12 ,
\]
وهو ما يعطي دائرة النقط الواقعة على الارتفاع $z = \frac12$ وعلى المسافة
$\sqrt{\tfrac12 + \tfrac14} = \frac{\sqrt3}2$. والبصمة الهندسية للأصغرية: في نقطة كهذه $M$، يجب أن تكون
المتجهة $\vect{MP}$ \emph{ناظمية} للسطح --- وإلا لأدى الانزلاق على طول
منحن مرسوم سرعته لها مركبة نحو $P$ إلى إنقاص المسافة. وللتحقق:
$\nabla(z - x^2 - y^2) = (-2x, -2y, 1)$ في $M = (x, y,
\tfrac12)$، بينما $\vect{MP} = (-x, -y, \tfrac12) =
\tfrac12(-2x, -2y, 1)$: فهما متوازيان، كما تُنبئ. وشرط
الرتبة الأولى ``قدم العمود'' هو الشرط نفسه الذي سيقود القيم القصوى
على مجموعات المستوى في \cref{exo:b2:surfaces:12}.
\end{example}

\begin{example}[المستويات المماسة للسطوح من الدرجة الثانية: قاعدة
الاستقطاب]\label{ex:b2:surfaces:polarization}
ليكن $S : xy + yz + zx = 1$ و$M_0 = (1, 1, 0) \in S$. هنا $\nabla f = (y + z,\ x + z,\ x + y)$، ومنه $\nabla f(M_0) = (1,
1, 2)$ ويكون \omterm{def:b2:surfaces:tangent}{المستوي المماس}
\[
(x - 1) + (y - 1) + 2z = 0,
\qquad\text{أي}\qquad x + y + 2z = 2 .
\]
والجواب نفسه يأتي من \emph{قاعدة الاستقطاب} التي تعمم \cref{ex:b2:surfaces:spheretangent}
و\cref{exo:b2:surfaces:2}: ففي معادلة \omterm{pb:b2:surfaces:1}{السطح من الدرجة الثانية}، نستبدل $x^2$
بالمقدار $x_0x$، وكل جداء $xy$ بالمقدار $\frac{x_0y +
y_0x}2$ (وهكذا دواليك):
\[
\frac{x_0y + y_0x}2 + \frac{y_0z + z_0y}2 + \frac{z_0x +
x_0z}2 = 1
\;\xrightarrow{\ M_0 = (1,1,0)\ }\;
\frac{x + y}2 + \frac z2 + \frac z2 = 1 ,
\]
وهو $x + y + 2z = 2$ من جديد. وتنجح القاعدة لأن $\nabla$ لصورة تربيعية هو
الصورة الثنائية الخطية المرافقة مقوَّمة عند نقطة الأساس --- فالتماس
مع \omterm{pb:b2:surfaces:1}{سطح من الدرجة الثانية} استقطاب، وهو وجه آخر من وجوه \cref{ch:b2:quadratic}.
\end{example}

\section{الصورة الأساسية الأولى}

\begin{definition}[الصورة الأساسية الأولى]\label{def:b2:surfaces:fff}
ليكن $\sigma \colon U \to \R^3$ سطحا نظاميا من الصنف $\mathcal{C}^1$. \emph{صورته الأساسية
الأولى}\index{صورة أساسية أولى} في $(u,v)$ هي الصورة التربيعية
المعرفة الموجبة على $\R^2$
\[
I(h, k) = \norm{h\,\sigma_u + k\,\sigma_v}^2
= E\,h^2 + 2F\,hk + G\,k^2,
\]
حيث
\[
E = \norm{\sigma_u}^2,
\qquad
F = \langle \sigma_u, \sigma_v\rangle,
\qquad
G = \norm{\sigma_v}^2 .
\]
\end{definition}

\begin{remark}
$I$ هو قصر الجداء السلمي الإقليدي المحيط على \omterm{def:b2:surfaces:tangent}{المستوي المماس}،
مقروءا في الأساس $(\sigma_u, \sigma_v)$: وهو معرَّف موجب بالضبط لأن $\sigma_u, \sigma_v$ مستقلان
(\cref{ch:b2:quadratic}). وتُحسب كل كمية مترية على السطح --- أطوال المنحنيات
المرسومة والزوايا بينها والمساحات --- انطلاقا من $E, F, G$ وحده.
ويكون سطحان لهما $E,
F, G$ نفسه في وسيطين مناسبين \emph{متقايسين} حتى
لو استقرا في الفضاء استقرارا مختلفا: وهذه هي نقطة انطلاق الهندسة
الجوهرية.
\end{remark}

\begin{example}[الزوايا بين المنحنيات الإحداثية]\label{ex:b2:surfaces:coordangle}
تقيس \omterm{def:b2:surfaces:fff}{الصورة الأساسية الأولى} الزوايا أيضا: فالمنحنيان الإحداثيان
$u \mapsto \sigma(u, v_0)$ و$v \mapsto
\sigma(u_0, v)$ يتلاقيان بالزاوية $\theta$ حيث
\[
\cos\theta = \frac{\langle\sigma_u,
\sigma_v\rangle}{\norm{\sigma_u}\,\norm{\sigma_v}}
= \frac{F}{\sqrt{EG}} :
\]
فالمعامل الوحيد $F$ هو الذي يقرر تعامد شبكة الوسيطين. وفي حالة
خريطة الكرة والسطح اللولبي، $F =
0$: فخطوط الزوال تقطع دوائر العرض،
واللوالب تقطع المولّدات المستقيمية الأفقية، بزوايا قائمة --- ولهذا
انهارت مكاملات مساحتيهما إلى $\sqrt{EG}$. وفي حالة خريطة تمثيل بياني، لا
ينعدم $F = f_xf_y$ إلا حيث تنعدم مشتقة جزئية: فالشبكة الإحداثية لتمثيل
بياني مائل \emph{ليست} متعامدة، مع أن الشبكة $(x, y)$ في الأسفل
متعامدة. وعندما تبدو الحسابات على سطح ثقيلة، تكون الخطوة الأولى هي
البحث عن خريطة يكون فيها $F = 0$.
\end{example}

\begin{example}[خريطة السرج]\label{ex:b2:surfaces:saddlechart}
من أجل السرج $z = xy$ بالخريطة $\sigma(u, v) = (u, v,
uv)$:
\[
\sigma_u = (1, 0, v), \qquad \sigma_v = (0, 1, u),
\qquad
E = 1 + v^2, \quad F = uv, \quad G = 1 + u^2 ,
\]
و$EG - F^2 = 1 + u^2 + v^2 > 0$: فهو نظامي في كل نقطة. والمنحنيان الإحداثيان المارّان بنقطة
\emph{مستقيمان} في $\R^3$ (بتثبيت $u$ أو بتثبيت $v$: وهما
المولّدان المستقيميان للسرج مزدوج التسطير)، ومع ذلك فإن $F \neq 0$ خارج
المحورين: فالمولّدان المارّان بنقطة عامة غير متعامدين. ويقع
المولّدان كلاهما في \omterm{def:b2:surfaces:tangent}{المستوي المماس}، ويولّدانه --- ومنه فإن \omterm{def:b2:surfaces:tangent}{المستوي
المماس} يقطع السطح على طول مستقيمين كاملين، وهو النقيض الأقصى للكرة،
التي تلامس مستوياتها المماسة في نقطة واحدة فقط. أما إشارة ``التماس
من الرتبة الثانية'' بين سطح ومستوياته المماسة فقصة انحناء، يتناولها
مجلد السنة الثالثة.
\end{example}

\begin{proposition}[طول منحن مرسوم على سطح]\label{prop:b2:surfaces:curvelength}
إذا كان $\gamma(t) = \sigma(u(t), v(t))$، $t \in [a, b]$، من الصنف $\mathcal{C}^1$، فإن
\[
L(\gamma) = \int_a^b
\sqrt{E\,u'^2 + 2F\,u'v' + G\,v'^2}\;\dd t ,
\]
مع تقويم $E, F, G$ في $(u(t), v(t))$.
\end{proposition}

\begin{proof}
$\gamma' = u'\sigma_u + v'\sigma_v$ بقاعدة السلسلة، ومنه $\norm{\gamma'}^2 = I(u', v') = Eu'^2 + 2Fu'v' + Gv'^2$؛ ونكامل $\norm{\gamma'}$ (\cref{def:b2:curves:length}).
\end{proof}

\begin{example}[لماذا تطير الطائرات فوق القطب]\label{ex:b2:surfaces:polarroute}
يقع مطاران على خط العرض $\varphi_0$ وعلى خطي طول متقابلين: $A = \sigma(0, \varphi_0)$ و$B = \sigma(\pi,
\varphi_0)$
على الكرة ذات نصف القطر $R$. فعلى طول دائرة العرض ($\varphi \equiv \varphi_0$)،
يكون الطول $\int_0^\pi R\cos\varphi_0\,\dd\theta = \pi R\cos\varphi_0$. وعلى طول المسار فوق القطب (صعودا على خط الزوال
$\theta = 0$، ونزولا على خط الزوال $\theta = \pi$)، يكون الطول $2R(\frac\pi2 -
\varphi_0)$. وعند خط
العرض $\varphi_0 = \frac\pi3$ (ستون درجة): مسار دائرة العرض $\pi R/2 \approx 1.571\,R$، والمسار القطبي
$\pi R/3 \approx 1.047\,R$ --- أي أقصر بالثلث. وفي الواقع $\pi\cos\varphi_0 \geq \pi - 2\varphi_0$ على $\intcc0{\pi/2}$ (فالدالة
$\pi\cos\varphi - \pi +
2\varphi$ تنعدم عند الطرفين ومشتقتها $2 -
\pi\sin\varphi$ تغير الإشارة مرة واحدة،
ومنه فهي متزايدة أولا ثم متناقصة، ومنه فهي موجبة): فالمسار القطبي لا
يخسر أبدا. وهكذا حولت \omterm{def:b2:surfaces:fff}{الصورة الأساسية الأولى} سؤال ملاحة إلى تكاملين
من سطر واحد؛ ويدفع \cref{exo:b2:surfaces:6} الفكرة إلى برهان أصغرية حقيقي من أجل خطوط
الزوال.
\end{example}

\begin{lemma}[متطابقة لاغرانج]\label{lem:b2:surfaces:lagrange}
من أجل كل $a, b \in \R^3$: $\norm{a \wedge b}^2 = \norm a^2 \norm b^2 - \langle a, b\rangle^2$.
وبوجه خاص
\[
\norm{\sigma_u \wedge \sigma_v} = \sqrt{EG - F^2}.
\]
\end{lemma}

\begin{proof}
لا يتغير الطرفان إذا استبدلنا $b$ بمركبته $b_\perp
= b - \frac{\langle a, b\rangle}{\norm a^2}a$ المتعامدة مع
$a$ (من أجل $a \neq 0$؛ والحالة $a = 0$ تافهة): فالطرف الأيسر لأن
$a \wedge a = 0$، والطرف الأيمن بنشر $\norm
{b_\perp}^2 = \norm b^2 - \frac{\langle a,b\rangle^2}{\norm a^2}$ و$\langle a, b_\perp\rangle = 0$. ومنه يكفي أن نبرهن على
المتطابقة من أجل $a, b$ متعامدين، حيث تُكتب $\norm{a \wedge
b} = \norm a \norm b$: وهو صحيح، لأن
\omterm{def:b2:nvs:norm}{معيار} الجداء الشعاعي لمتجهتين متعامدتين هو $\norm a\norm b\,\abs{\sin\frac\pi2}$. والصيغة المكتوبة
هي الحالة $a = \sigma_u$، $b = \sigma_v$.
\end{proof}

\begin{remark}
تقول متطابقة لاغرانج إن $EG - F^2 = \det\begin{psmallmatrix}
E & F\\ F & G\end{psmallmatrix}$ هو \emph{\omterm{def:b2:linalg:det}{محدد} غرام} للمتجهتين $(\sigma_u, \sigma_v)$:
أي مربع \omterm{def:b2:surfaces:area}{مساحة} متوازي الأضلاع الذي يولّدانه. والنظامية، ومعرَّفية
\omterm{def:b2:surfaces:fff}{الصورة الأساسية الأولى} الموجبة، وموجبية \omterm{def:b2:linalg:det}{محدد} غرام، ثلاث صياغات لشرط
واحد --- ولهذا لا تنعدم أبدا دالة \omterm{def:b2:surfaces:area}{المساحة} تحت التكامل أدناه على
خريطة نظامية.
\end{remark}

\begin{definition}[المساحة]\label{def:b2:surfaces:area}
ليكن $\sigma \colon U \to \R^3$ سطحا نظاميا متباينا من الصنف $\mathcal{C}^1$ ولتكن $K \subseteq U$
منطقة متراصة تكون التكاملات المزدوجة عليها ذات معنى (\cref{ch:b2:multint}).
\emph{مساحة}\index{مساحة!سطح} القطعة $\sigma(K)$ هي
\[
\mathcal{A}
= \iint_K \norm{\sigma_u \wedge \sigma_v}\,\dd u\,\dd v
= \iint_K \sqrt{EG - F^2}\;\dd u\,\dd v .
\]
\end{definition}

\begin{remark}[لماذا هذه الصيغة]
يُرسل المستطيل $[u, u + \dd u] \times [v, v + \dd v]$، من الرتبة الأولى، على متوازي الأضلاع المولَّد
بالمتجهتين $\sigma_u\,\dd u$ و$\sigma_v\,\dd v$، ومساحته $\norm{\sigma_u \wedge
\sigma_v}\,\dd u\,\dd v$: فالتعريف يكامل عامل تشوه
\omterm{def:b2:surfaces:area}{المساحة} المحلي، تماما كما يكامل \omterm{def:b2:curves:length}{طول القوس} السرعة المحلية. أما
الانسجام مع تغيير الوسيطين فهو \cref{exo:b2:surfaces:8}؛ والانسجام مع صيغة تغيير
المتغيرات في التكاملات المزدوجة يناقَش في \cref{ch:b2:multint}.
\end{remark}

\begin{example}[تمثيلان بيانيان مختلفان، ومساحة واحدة]\label{ex:b2:surfaces:samearea}
فوق قرص الوحدة، قارن الطاسة $z = \frac12(x^2 + y^2)$ بالسرج $z = xy$. فدالتا \omterm{def:b2:surfaces:area}{المساحة}
تحت التكامل (\cref{exo:b2:surfaces:5}) هما
\[
\sqrt{1 + x^2 + y^2}
\qquad\text{و}\qquad
\sqrt{1 + y^2 + x^2} :
\]
وهما متطابقتان. فللسطحين --- أحدهما ينحني في الاتجاهات كلها بالطريقة
نفسها والآخر على شكل سرج --- مساحتان متساويتان تماما فوق أي منطقة،
وهي $\frac{2\pi}3(2\sqrt2 - 1)$ فوق قرص الوحدة. فعنصر \omterm{def:b2:surfaces:area}{المساحة} لا يرى إلا \emph{طول}
التدرج، لا ترتيب الانثناء؛ وتمييز الطاسة من السرج يتطلب معطيات من
الرتبة الثانية (بنية الإشارات المعروضة في \cref{fig:b2:surfaces:saddle})، لا يكشفها أي قدر
من قياس المساحات. \omterm{def:b2:surfaces:fff}{فالصورة الأساسية الأولى} متري أعمى عن الشكل؛ أما
الصورة الثانية المبصرة للشكل فتنتمي إلى السنة الثالثة.
\end{example}

\begin{example}[مساحة الكرة]\label{ex:b2:surfaces:spherearea}
من أجل الخريطة الكروية في \cref{ex:b2:surfaces:standard}:
\[
\sigma_\theta = R(-\sin\theta\cos\varphi,\
\cos\theta\cos\varphi,\ 0),
\qquad
\sigma_\varphi = R(-\cos\theta\sin\varphi,\
-\sin\theta\sin\varphi,\ \cos\varphi),
\]
ومنه $E = R^2\cos^2\varphi$، $F = 0$، $G = R^2$، و$\sqrt{EG - F^2}
= R^2\cos\varphi$. ومنه
\[
\mathcal{A}
= \int_{-\pi/2}^{\pi/2}\!\!\int_0^{2\pi}
R^2\cos\varphi\;\dd\theta\,\dd\varphi
= 2\pi R^2\,\bigl[\sin\varphi\bigr]_{-\pi/2}^{\pi/2}
= \boxed{4\pi R^2} .
\]
\end{example}

\begin{example}[المخروط، متحققا منه بصيغة المدرسة]\label{ex:b2:surfaces:conearea}
من أجل المخروط $z = \sqrt{x^2 + y^2}$ فوق التاج $a \leq
\rho \leq b$، تعطي صيغة التمثيل البياني
في \cref{exo:b2:surfaces:5} المقدار $1 + f_x^2 + f_y^2 = 2$ (بحساب $f_x = x/\rho$ و$f_y = y/\rho$)، ومنه
\[
\mathcal A = \sqrt2\,\pi\,(b^2 - a^2) .
\]
وللتحقق مع صيغة طول الراسم $\pi\rho\ell$ في \cref{ex:b2:surfaces:revolution}: للمخروطين الكاملين ذوي
نصفي قطر القاعدة $b$ و$a$ مساحتان جانبيتان $\pi b\cdot b\sqrt2$ و$\pi a\cdot a\sqrt2$،
وفرقهما هو بالضبط $\sqrt2\pi(b^2 - a^2)$. فخريطتان، وصيغتان، \omterm{def:b2:surfaces:area}{ومساحة} واحدة --- وهو
الثبات المبرهن في \cref{exo:b2:surfaces:8}، مشاهَدا في الطبيعة.
\end{example}

\begin{remark}[تحققات سريعة من المساحات]
ثلاثة تحققات فورية تمسك معظم الأخطاء في حساب المساحات.
\emph{التحاكي}: تكبير سطح بالمعامل $\lambda$ يضرب $E, F, G$ في $\lambda^2$
ويضرب \omterm{def:b2:surfaces:area}{المساحة} في $\lambda^2$ --- فالجواب الذي لا يتعلق بالمقدار $R$
تعلقا تربيعيا (مثل $4\pi R^2$) خاطئ. \emph{موجبية العنصر}: يجب أن يكون
$\sqrt{EG - F^2}$ موجبا تماما داخل الخريطة؛ وانعدام القيمة يشير إلى انحلال في
الخريطة، ينبغي استئصاله كما في قطبي الكرة. \emph{التناظر}: يجب أن
ينسجم حساب على قطعة متناظرة مع جمع أجزائها المتطابقة --- فمن الأفضل
أن يعطي نصف الكرة القيمة $2\pi R^2$.
\end{remark}

\begin{example}[سطح دوران]\label{ex:b2:surfaces:revolution}
لندر المنحنى $z \mapsto (r(z), 0, z)$، $r > 0$، من الصنف $\mathcal{C}^1$، حول المحور $z$:
\[
\sigma(\theta, z) = (r(z)\cos\theta,\ r(z)\sin\theta,\ z).
\]
عندئذ $E = r(z)^2$، $F = 0$، $G = 1 + r'(z)^2$، ومنه
\[
\mathcal{A} = \int_a^b\!\!\int_0^{2\pi}
r(z)\sqrt{1 + r'(z)^2}\;\dd\theta\,\dd z
= 2\pi\int_a^b r(z)\sqrt{1 + r'(z)^2}\,\dd z ,
\]
وهي الصيغة الكلاسيكية (المحيط $2\pi r$ مضروبا في عنصر طول الراسم).
ومن أجل المخروط $r(z) = kz$، $z \in [0, h]$: $\mathcal
A = 2\pi k\sqrt{1 + k^2}\,\frac{h^2}{2} = \pi \rho \ell$ مع $\rho = kh$ نصف قطر القاعدة
و$\ell = h\sqrt{1 + k^2}$ طول الراسم --- وهي صيغة المدرسة، وقد استُنتجت الآن بدل أن
تُسلَّم.
\end{example}

\begin{example}[السطح السلسلي]\label{ex:b2:surfaces:catenoid}
لندر السلسلية $r(z) = \cosh z$، $z \in \intcc{-1}{1}$، حول محورها: فيكون \emph{للسطح
السلسلي} الناتج، بصيغة الدوران و$1 + \sinh^2 z = \cosh^2 z$،
\[
\mathcal A = 2\pi\int_{-1}^1\cosh z\,\sqrt{1 + \sinh^2z}\,
\dd z = 2\pi\int_{-1}^1\cosh^2z\,\dd z
= \pi\bigl[z + \sinh z\cosh z\bigr]_{-1}^{1},
\]
أي
\[
\mathcal A = 2\pi + \pi\sinh 2 \approx 17.68 .
\]
فقد اندمج عامل الراسم $\sqrt{1 + r'^2}$ مع نصف القطر في مربع تام --- وهي
المتطابقة نفسها التي جعلت طول قوس السلسلية أوليا في فصل المنحنيات.
وليست هذه مصادفة جبرية: فمن بين جميع سطوح الدوران الممتدة بين
الدائرتين الحديتين، يصغّر السطح السلسلي \omterm{def:b2:surfaces:area}{المساحة} (فهو شكل غشاء صابوني
بين حلقتين)، وهذه الخاصية التغايرية هي بالضبط ما يميز $\cosh$؛
ويبرهن مجلد السنة الثالثة على ذلك بحساب التغايرات.
\end{example}

\begin{remark}[مزالق شائعة]
(1) \emph{شذوذ الخريطة ليس شذوذا في السطح}: فالخريطة الكروية تنحل
عند القطبين ($\cos\varphi = 0$)، لكن الكرة ملساء تماما هناك --- وخريطة أخرى
(بتبادل أدوار المحاور) نظامية عند القطبين. فقبل إعلان نقطة شاذة،
جرّب توسيما ثانيا. (2) \emph{نظامية $\sigma$ تخص التوسيم لا الصورة}:
فالسطح $\sigma(u, v) = (u^3, v,
0)$ غير نظامي على طول $u = 0$ مع أن صورته مستو. (3) تتطلب
صيغة \omterm{def:b2:surfaces:area}{المساحة} أن يكون $\sigma$ \emph{متباينا} على $K$: فخريطة تغطي
قطعة مرتين تحسبها مرتين (فجريان $\theta$ على $\intcc0{4\pi}$ يضاعف \omterm{def:b2:surfaces:area}{مساحة}
الكرة). (4) \omterm{def:b2:surfaces:tangent}{الناظم الواحدي} معرَّف إلى غاية إشارة بحكم السطح لكن
\emph{تختاره} الخريطة (بترتيب $u, v$)؛ فالنصوص التي تتضمن التوجيه
يجب أن تثبّت ذلك الاختيار. (5) وأخيرا، ليس $EG - F^2 > 0$ فرضية إضافية: بل
هو النظامية بالضبط، بحكم متطابقة لاغرانج --- فإن انعدم في موضع ما،
كانت المشكلة في الخريطة، ولا تنطبق هناك أي صيغة \omterm{def:b2:surfaces:area}{مساحة} أو \omterm{def:b2:surfaces:tangent}{مستو مماس}.
\end{remark}

\begin{remark}[آفاق داخل هذا المجلد]
الروابط المتقدمة انطلاقا من هنا. عنصر \omterm{def:b2:surfaces:area}{المساحة} $\sqrt{EG -
F^2}\,\dd u\,\dd v$ جاكوبي ذو
بعدين متنكر، ويجعل \cref{ch:b2:multint} المماثلة مضبوطة مع مبرهنة تغيير
المتغيرات --- فالتكاملات السطحية هناك هي مساحات هذا الفصل مع دالة
تحت التكامل. \omterm{def:b2:surfaces:fff}{والصورة الأساسية الأولى} حقل من \omterm{def:b2:quadratic:def}{الصور التربيعية} الموجبة،
تعالَج نقطة نقطة بأدوات \cref{ch:b2:quadratic}، ومبرهنتها الطيفية تشغّل أيضا تصنيف
السطوح من الدرجة الثانية في مسألة نهاية الأسبوع لهذا الفصل. ويقود
الناظم مسائل القيم القصوى على مجموعات القيود (\cref{ex:b2:surfaces:closest})، وهو الجرثومة
الهندسية لطريقة مضاعف لاغرانج المرسومة مع \cref{thm:b2:diffcalc:extrema}.
\end{remark}

\begin{remark}[أين يُستعمل هذا]
عنصر \omterm{def:b2:surfaces:area}{المساحة} $\norm{\sigma_u\wedge\sigma_v}\,\dd u\,\dd v$ هو قياس التكامل السطحي في \cref{ch:b2:multint}، حيث يلتقي
بصيغة غرين؛ \omterm{def:b2:surfaces:fff}{والصورة الأساسية الأولى} هي النموذج الأولي لحقل من \omterm{def:b2:quadratic:def}{الصور
التربيعية}، يُدرس نقطة نقطة بأدوات \cref{ch:b2:quadratic}؛ وتصنّف مسألة نهاية
الأسبوع لهذا الفصل جميع السطوح من الدرجة الثانية بالمبرهنة الطيفية.
ويعود مجلد السنة الثالثة إلى السطوح بالصور التفاضلية ومبرهنة
التباعد، \omterm{thm:b2:curves:frenet2d}{والانحناء} الجوهري --- أي ما تعرفه $E, F, G$ عن الانثناء ---
هو البوابة إلى الهندسة التفاضلية بمعناها الدقيق.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b2:surfaces:3}

\begin{exercise}[$\star\star$]\label{exo:b2:surfaces:4}
(الطارة) وسّم الطارة الناتجة عن دوران الدائرة ذات المركز $(R, 0, 0)$
ونصف القطر $r < R$ في المستوي $xz$ حول المحور $z$:

\sigma(\theta, \psi) = \bigl((R + r\cos\psi)\cos\theta,\
(R + r\cos\psi)\sin\theta,\ r\sin\psi\bigr).
\]
احسب $E, F, G$، وتحقق من النظامية، وبيّن أن \omterm{def:b2:surfaces:area}{المساحة} $4\pi^2 R r$ (بابوس:
المحيط المتوسط $2\pi R$ مضروبا في طول الدائرة $2\pi r$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:surfaces:5}
بيّن أن \omterm{def:b2:surfaces:area}{مساحة} التمثيل البياني للدالة $f \in \mathcal{C}^1(K)$ هي $\iint_K \sqrt{1 + f_x^2 + f_y^2}\,\dd x\,\dd y$، واحسبها من
أجل قطعة المجسم المكافئ $z = \frac12(x^2 + y^2)$ فوق القرص $x^2 + y^2 \leq 1$ (بالإحداثيات
القطبية، \cref{ch:b2:multint}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:surfaces:6}
لمنحن مرسوم $\gamma(t) = \sigma(u(t), v(t))$ على الكرة ذات نصف القطر $R$ (بالخريطة
الكروية) $u = \theta(t)$، $v =
\varphi(t)$. اكتب طوله على صورة تكامل بدلالة $\theta, \varphi$
وبرهن على أن قوس خط الزوال هو الأقصر بين المنحنيات الواصلة بين نقطتين
من خط الزوال نفسه $\theta = \theta_0$. \emph{(احصر دالة التكامل من الأسفل
بالمقدار $R\abs{\varphi'}$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:surfaces:7}
(نواظم الكرة) ليكن $S$ سطح مستوى نظاميا $\{f = c\}$، مترابطا، تمر
جميع نواظمه بنقطة ثابتة $\Omega$. بيّن أن $S$ محتوى في كرة مركزها
$\Omega$. \emph{(بيّن أن مشتقة $\norm{M - \Omega}^2$ معدومة على طول كل منحن مرسوم
على $S$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:surfaces:8}
(\omterm{def:b2:surfaces:area}{المساحة} هندسية) ليكن $\Phi \colon U' \to U$ تقابلا تفاضليا من الصنف $\mathcal{C}^1$ بين
مفتوحين من $\R^2$ و$\tilde\sigma =
\sigma \circ \Phi$. بيّن أن
\[
\norm{\tilde\sigma_{u'} \wedge \tilde\sigma_{v'}}
= \abs{\det J_\Phi}\,
\norm{(\sigma_u \wedge \sigma_v)\circ\Phi} ,
\]
واستنتج، باستعمال صيغة تغيير المتغيرات في \cref{ch:b2:multint}، أن \omterm{def:b2:surfaces:area}{مساحة} \cref{def:b2:surfaces:area}
لا تتعلق بالتوسيم النظامي المختار.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:surfaces:9}
(مبرهنة أرخميدس في علبة القبعة) على الكرة ذات نصف القطر $R$،
يكون \emph{للنطاق} الواقع بين خطي العرض اللذين فيهما $z_1 \leq z \leq z_2$
($-R \leq z_1 < z_2 \leq R$) \omterm{def:b2:surfaces:area}{مساحة} $2\pi R\,(z_2 - z_1)$: برهن على ذلك بالخريطة الكروية، واستنتج أن
\omterm{def:b2:surfaces:area}{مساحة} نطاق لا تتعلق إلا بارتفاعه --- ففي تقطيع برتقالة إلى شرائح
متساوية السماكة نحصل على كميات متساوية من القشر.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:surfaces:10}
بيّن أن كل ناظم لسطح دوران $\sigma(\theta, z) = (r(z)\cos\theta,\ r(z)\sin\theta,\ z)$ ($r > 0$ من الصنف $\mathcal C^1$) يلاقي
محور الدوران، وحدد نقطة التقاطع.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:surfaces:11}
(فرد الأسطوانة) للخريطة $\sigma(u, v) = (\cos u,\
\sin u,\ v)$ لأسطوانة الوحدة $E = G = 1$، $F = 0$:
تحقق من ذلك، واشرح لماذا يكون لكل منحن مرسوم $t \mapsto
\sigma(u(t), v(t))$ الطول نفسه الذي
للمنحنى المستوي $t
\mapsto (u(t), v(t))$. واستنتج أن اللولب من $(1, 0, 0)$ إلى $(1, 0, 2\pi c)$
الذي يصنع دورة واحدة طوله $2\pi\sqrt{1 + c^2}$، وأنه لا يوجد منحن مرسوم بالطرفين
نفسيهما وبدورة كاملة واحدة يكون أقصر منه.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:surfaces:12}
ليكن $S = \{f = c\}$ سطح مستوى نظاميا متراصا ولتكن $M_0 \in S$ نقطة على مسافة
أعظمية من المبدأ. بيّن أن $\nabla f(M_0)$ خطي مع $\vect{OM_0}$ --- فالناظم عند
النقطة الأبعد قطري. وطبق ذلك على المجسم الإهليلجي $\frac{x^2}{a^2} + \frac{y^2}{b^2} + \frac{z^2}{c^2} = 1$ ($a >
b > c > 0$):
جد جميع النقط التي يكون فيها الناظم قطريا، وحدد أبعدها.
\end{exercise}

\section{مسألة: تصنيف السطوح من الدرجة الثانية في $\R^3$}

\begin{problem}[{مسألة نهاية الأسبوع --- كل سطح من الدرجة الثانية،
مرتبا بالمبرهنة الطيفية}]\label{pb:b2:surfaces:1}
\emph{السطح من الدرجة الثانية}\index{سطح من الدرجة الثانية} هو
مجموعة أصفار كثير حدود من الدرجة الثانية في $\R^3$
\[
q(X) = X^{\mathsf T}\!AX + 2\,\langle b, X\rangle + c,
\qquad A \in \mathcal S_3(\R),\ A \neq 0,\ b \in \R^3,\
c \in \R .
\]
والسطوح الواردة في أشكال هذا الفصل --- الكرات والمجسمات الإهليلجية
والسروج والمخاريط والأسطوانات --- كلها سطوح من الدرجة الثانية. وتصنف
هذه المسألة هذه السطوح \emph{كلها}: فالمبرهنة الطيفية (\cref{thm:b2:quadratic:spectral})
تقوّم الجزء التربيعي، والانسحابات الأفينية (\cref{ch:b2:affine}) تمتص الجزء
الخطي، وما يبقى قائمة قصيرة كاملة من الصيغ الموحدة.

\medskip
\noindent\textbf{الجزء الأول --- آلة الاختزال.}
\begin{enumerate}
  \item ليكن $X = PY + t$ مع $P \in O(3)$ و$t \in \R^3$ (وهو تغيير إحداثيات صلب).
        بيّن أن $q(PY + t) =
        Y^{\mathsf T}\!A'Y + 2\langle b', Y\rangle + c'$ مع
        \[
        A' = P^{\mathsf T}\!AP, \qquad
        b' = P^{\mathsf T}(At + b), \qquad
        c' = q(t) .
        \]
        واستنتج أن \omterm{def:b2:reduction:eigen}{طيف} $A$ (ومنه رتبته وبصمته) ثابت صلب للمعادلة،
        واشرح لماذا لا تتحدد معادلة سطح معطى من الدرجة الثانية إلا
        إلى غاية عامل عددي غير معدوم.
  \item باستعمال المبرهنة الطيفية، بيّن أن المعادلة تصير بعد دوران
        $\sum_i \lambda_i y_i^2 +
        2\sum_i\beta_iy_i + c = 0$ حيث $\lambda_1, \lambda_2,
        \lambda_3$ هي \omterm{def:b2:reduction:eigen}{القيم الذاتية} للمصفوفة $A$.
  \item من أجل كل $i$ يحقق $\lambda_i \neq 0$، امتص $\beta_iy_i$ بانسحاب ($y_i \mapsto y_i -
        \beta_i/\lambda_i$).
        اكتب المعادلة المختزلة عندما $\operatorname{rank} A = r$: $\sum_{i\leq r}\lambda_i
        z_i^2 + 2\sum_{i > r}\beta_iz_i + c'' = 0$.
  \item \emph{مركز} السطح ذي المعادلة $q = 0$ هو نقطة $\Omega$ تحقق
        $q(2\Omega - X) = q(X)$ من أجل كل $X$: فالتناظر المركزي حول $\Omega$ يحافظ
        على المعادلة، ومنه على السطح. بيّن أن $q(2\Omega - X)
        - q(X) = -4\langle A\Omega + b, X\rangle + 4\langle
        A\Omega + b, \Omega\rangle$، واستنتج: أن
        المراكز هي بالضبط حلول $A\Omega = -b$؛ وأنها توجد إذا وفقط إذا كان
        $b \in \operatorname{im}A$، وأن المركز وحيد إذا وفقط إذا كانت $A$ قابلة
        للقلب.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- السطوح المركزية من الدرجة الثانية
($\operatorname{rank}A = 3$).} هنا تكون المعادلة المختزلة $\lambda_1z_1^2 + \lambda_2z_2^2 + \lambda_3z_3^2 = \delta$.
\begin{enumerate}[resume]
  \item بالضرب في $-1$ عند الحاجة، افترض أن اثنين على الأقل من
        المقادير $\lambda_i > 0$. أحص الإمكانات: البصمة $(3, 0)$ مع $\delta > 0$
        و$= 0$ و$< 0$، والبصمة $(2, 1)$ مع $\delta > 0$ و$= 0$ و$<
        0$؛
        وسمّ المجموعات الست الناتجة (المجسم الإهليلجي، النقطة،
        المجموعة الخالية، المجسم الزائدي ذو الطية الواحدة، المخروط،
        المجسم الزائدي ذو الطيتين) وضع كلا منها في صيغته الموحدة
        الإقليدية ($\frac{x^2}{a^2} + \frac{y^2}{b^2} +
        \frac{z^2}{c^2} = 1$، وهكذا).
  \item صنّف $x^2 + y^2 + z^2 + 4xy + 4yz + 4zx = 1$: بيّن $A = 2J - I$ حيث $J$ هي المصفوفة التي جميع
        معاملاتها واحد، واحسب \omterm{def:b2:reduction:eigen}{الطيف} $\{5, -1, -1\}$، وحدد مجسما زائديا ذا
        طيتين ودورانيا حول المحور $\R(1,1,1)$.
  \item (المولّدات المستقيمية) من أجل المجسم الزائدي ذي الطية
        الواحدة $\frac{x^2}{a^2} + \frac{y^2}{b^2} - \frac{z^2}{c^2}
        = 1$، فكّك
        \[
        \Bigl(\frac xa - \frac zc\Bigr)
        \Bigl(\frac xa + \frac zc\Bigr)
        = \Bigl(1 - \frac yb\Bigr)\Bigl(1 + \frac yb\Bigr)
        \]
        وأنتج عائلتين ذواتي وسيط واحد من المستقيمات الواقعة على
        السطح.
  \item بيّن أن بكل نقطة من المجسم الزائدي ذي الطية الواحدة يمر
        مستقيم واحد بالضبط من كل عائلة: فالسطح \emph{مزدوج التسطير}.
  \item \emph{المخروط المقارب} للمجسم الزائدي ذي الطية الواحدة هو
        $C : \frac{x^2}{a^2} + \frac{y^2}{b^2}
        - \frac{z^2}{c^2} = 0$. مع $\rho^2 = \frac{x^2}{a^2}
        + \frac{y^2}{b^2}$، بيّن أن أي نقطة من المجسم الزائدي تقع
        على مسافة لا تتجاوز $c\bigl(\rho -
        \sqrt{\rho^2 - 1}\bigr) = \frac{c}{\rho +
        \sqrt{\rho^2 - 1}}$ من $C$، ومنه فإن السطح يعانق
        مخروطه عند اللانهاية. وما مقاطع المجسم الزائدي بالمستويات
        $x = \pm a$؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الرتبة $2$ والرتبة $1$:
المجسمات المكافئة والأسطوانات والمستويات.}
\begin{enumerate}[resume]
  \item لنفترض $\operatorname{rank}A = 2$، ولتكن $\lambda_1,
        \lambda_2 \neq 0 = \lambda_3$. انطلاقا من السؤال 3، انقسم
        إلى حالتين بحسب $\beta_3 \neq 0$ (فلا مركز، بحسب السؤال 4) أو $\beta_3 = 0$
        (فمستقيم من المراكز)، واختزل إلى
        \[
        \lambda_1z_1^2 + \lambda_2z_2^2 + 2\beta_3z_3 = 0
        \qquad\text{أو}\qquad
        \lambda_1z_1^2 + \lambda_2z_2^2 + c'' = 0 :
        \]
        أي مجسمات مكافئة إهليلجية أو زائدية في الحالة الأولى،
        وأسطوانات فوق مقاطع مخروطية مركزية (أو زوج مستويين
        متقاطعين، أو مستقيم، أو المجموعة الخالية) في الحالة الثانية.
  \item بيّن أن السرج $z = xy$ مجسم مكافئ زائدي: أدر بالزاوية $\pi/4$
        في المستوي $xy$ لتبلغ $z = \tfrac12(u^2 - v^2)$، وهو سطح \cref{fig:b2:surfaces:saddle} إلى غاية
        تحاك.
  \item بيّن أن السرج $z = xy$ يحمل عائلتي المستقيمات $\{x = x_0,\ z = x_0y\}$ و$\{y = y_0,\ z
        = xy_0\}$،
        مع مرور مستقيم واحد بالضبط من كل عائلة بكل نقطة: وهو ثاني
        \omterm{pb:b2:surfaces:1}{سطح من الدرجة الثانية} مزدوج التسطير.
  \item صنّف $x^2 + y^2 - 2x + 4y + 3 = 0$ في $\R^3$ (بإتمام المربعات؛ وحدد أسطوانة
        دائرية قائمة، وأعط محورها ونصف قطرها).
  \item ولنأخذ الآن $\operatorname{rank}A = 1$، ولتكن $\lambda_1
        \neq 0 = \lambda_2 = \lambda_3$. بالدوران داخل مستوي
        النواة وبالانسحاب، اختزل إلى
        \[
        \lambda_1z_1^2 + 2\beta z_2 = 0 \quad (\beta \neq 0)
        \qquad\text{أو}\qquad
        \lambda_1z_1^2 + c'' = 0 :
        \]
        أي أسطوانة مكافئة، أو زوج مستويين متوازيين، أو مستو مضاعف،
        أو المجموعة الخالية. وصنّف $(x + y)^2 =
        z$ تصنيفا كاملا (الصيغة
        الموحدة، ومحور الثبات بالانسحاب).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- مبرهنة التصنيف.}
\begin{enumerate}[resume]
  \item اجمع الأجزاء الأول والثاني والثالث في مبرهنة: \emph{كل \omterm{pb:b2:surfaces:1}{سطح
        من الدرجة الثانية} في $\R^3$ تُرسله حركة صلبة على صيغة موحدة
        واحدة بالضبط}. اسرد الأنماط الأفينية السبعة عشر (بعدّ
        المتغيرات الخالية والمجموعات المنحلة)، وميّز \emph{السطوح} التسعة
        من الدرجة الثانية: المجسم الإهليلجي،
        والمجسمان الزائديان ذو الطية الواحدة وذو الطيتين، والمخروط،
        والمجسمان المكافئان الإهليلجي والزائدي، والأسطوانات
        الإهليلجية والزائدية والمكافئة.
  \item اكتب \emph{خوارزمية} التصنيف: بمعطى $(A,
        b, c)$، أي المقادير
        تحسبها، وبأي ترتيب، وأي تفرع يقرر أي نمط؟ وبرر أن كل خطوة
        فعالة (\omterm{def:b2:reduction:eigen}{القيم الذاتية} لمصفوفة متناظرة من الحجم $3\times3$،
        والرتبة، وقابلية حل $A\Omega = -b$).
  \item شغّل الخوارزمية على $x^2 + y^2 + z^2 - 2xy - 2yz -
        2zx = 1$: بيّن أن \omterm{def:b2:reduction:eigen}{طيف} $A = 2I - J$ هو $\{2, 2,
        -1\}$
        واستنتج: مجسم زائدي ذو طية واحدة ودوراني حول $\R(1,1,1)$.
  \item شغّلها على $x^2 + y^2 - z^2 - 2x + 4y + 2z + 4 = 0$: جد المركز وحدد \omterm{pb:b2:surfaces:1}{السطح من الدرجة الثانية}.
  \item شغّلها على $x^2 + 4xy + y^2 = 2z$: قطّر الكتلة $xy$ ($u = \frac{x+y}{\sqrt2}$، $v =
        \frac{x-y}{\sqrt2}$)
        وحدد \omterm{pb:b2:surfaces:1}{السطح من الدرجة الثانية}.
  \item الإقليدي مقابل الأفيني. بيّن أن سطحين مركزيين من الدرجة
        الثانية في صيغة موحدة يكونان متكافئين \emph{صلبا} إذا وفقط
        إذا كانت لهما قائمتا معاملات متساويتان (إلى غاية تبديل وعدد
        موجب مشترك على المعادلة)، بينما لا تبقى \emph{\omterm{def:b2:affine:subspace}{أفينيا}} إلا
        معطيات البصمة: فكل مجسم إهليلجي صورة أفينية للكرة المدورة.
        وأي مبرهنة تضمن أن البصمة لا يمكن أن تتغير في الطريق
        (\cref{thm:b2:quadratic:sylvester})؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- العوائد.}
\begin{enumerate}[resume]
  \item بيّن أن كل مقطع \omterm{pb:b2:surfaces:1}{لسطح من الدرجة الثانية} بمستو أفيني هو مقطع
        مخروطي (وقد يكون منحلا) من ذلك المستوي. وحدد مقاطع السرج
        $z = xy$ بالمستويات $z = c$ ($c \neq 0$ و$c = 0$).
  \item أي السطوح من الدرجة الثانية تحوي مستقيمات؟ بيّن أن المجسم
        الإهليلجي والمجسم الزائدي ذا الطيتين والمجسم المكافئ
        الإهليلجي لا يحوي أي منها مستقيما \emph{(اقصر $q$ على
        مستقيم واستعمل متراجحة كوشي--شوارتز في حالة الطيتين)}؛ وأن
        المخروط والأسطوانات مسطّرة بعائلة واحدة؛ واستنتج أن السطوح
        من الدرجة الثانية مزدوجة التسطير هي بالضبط المجسم الزائدي ذو
        الطية الواحدة والمجسم المكافئ الزائدي.
  \item عندما لا يُطلب إلا النمط \emph{الأفيني}، يكون \omterm{thm:b2:quadratic:gauss}{اختزال غاوس}
        (\cref{thm:b2:quadratic:gauss}) أرخص من التقطير. أعد السؤال 17 بخوارزمية غاوس
        وتحقق من البصمة $(2, 1)$؛ وأي معلومات إقليدية يفقدها غاوس؟
  \item جميع \omterm{def:b2:reduction:eigen}{القيم الذاتية} لمصفوفة متناظرة حقيقية؛ بيّن أنه يمكن
        بالتالي قراءة \emph{إشارات} \omterm{def:b2:reduction:eigen}{القيم الذاتية} للمصفوفة $A$
        من \omterm{def:b2:reduction:charpoly}{كثير الحدود المميز} بقاعدة ديكارت في الإشارات، وتحقق من
        ذلك في السؤال 6: فالمقدار $\chi_A(\lambda) = \lambda^3 - 3\lambda^2
        - 9\lambda - 5$ فيه تغيير إشارة واحد
        بالضبط، ومنه فالبصمة $(1, 2)$.
  \item تركيب. لخّص الخوارزمية في أسطر قليلة؛ واذكر الدور المضبوط
        الذي تؤديه (1) المبرهنة الطيفية، (2) معادلة المركز $A\Omega = -b$،
        (3) مبرهنة سيلفستر في القصور الذاتي، (4) \omterm{thm:b2:quadratic:gauss}{اختزال غاوس}. وماذا
        تعطي الآلة نفسها في $\R^2$، وما الذي يتغير في $\R^n$؟
\end{enumerate}
\end{problem}
