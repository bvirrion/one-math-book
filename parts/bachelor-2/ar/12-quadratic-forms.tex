\chapter{الصيغ التربيعية}\label{ch:b2:quadratic}

\omterm{def:b2:quadratic:def}{الصيغة التربيعية} هي الظلّ الجبري لهندسة ما: فالأجزاء ذوات
التوقيع المعدوم تُسطّح، والموجبة تُقوّس في اتجاه، والسالبة في
الاتجاه الآخر. ويُرجع هذا الفصل كل \omterm{def:b2:quadratic:def}{صيغة تربيعية} حقيقية إلى مجموع
$\pm$ مربّعًا (غاوس)، ويبرهن على أن عدّ الإشارات ذاتي
(سيلفستر)، ثم يتوّج الهندسة الإقليدية
\emph{بالمبرهنة الطيفية}: فالتشاكلات الذاتية المتماثلة تتقطّر في
أسس متعامدة ممنظمة --- وهي المبرهنة الأكثر استعمالًا في \omterm{def:b2:structures:algebra}{الجبر} الخطي
التطبيقي.

\section{الصيغ ثنائية الخطية والتربيعية}

\begin{definition}\label{def:b2:quadratic:def}
\emph{الصيغة ثنائية الخطية المتماثلة} على فضاء متجهي حقيقي $E$ هي
صيغة ثنائية الخطية $\varphi \colon E \times E \to \R$ تحقق $\varphi(x, y) =
\varphi(y, x)$؛ و\emph{الصيغة
التربيعية}\index{صيغة تربيعية} المرافقة هي $q(x) = \varphi(x, x)$. وتُستعاد الصيغة
$\varphi$ من $q$ \emph{بالاستقطاب}\index{متطابقة الاستقطاب}:
\[
\varphi(x, y) = \tfrac12\bigl(q(x + y) - q(x) - q(y)\bigr).
\]
وفي أساس $(e_i)$، تكون \emph{مصفوفة} $\varphi$ هي المصفوفة المتماثلة
$B = (\varphi(e_i, e_j))$، مع $q(x) = X^{\mathsf T} B X$؛ ويعوّض
تغيير الأساس بمصفوفة $P$ المصفوفةَ $B$ بالمصفوفة $P^{\mathsf T} B P$
(\emph{التوافق}\index{مصفوفتان متوافقتان} --- لا التشابه!).
و\emph{رتبة} $q$ هي $\operatorname{rk} B$ (وهي لا متغيّرة:
لأن التوافق يضرب في مصفوفات قابلة للقلب).
\end{definition}

\begin{example}
على $\R^2$: للصيغة $q(x, y) = x^2 + 4xy + y^2$ المصفوفةُ $\begin{pmatrix}
1 & 2\\ 2 & 1\end{pmatrix}$. والجداء السلّمي هو بالضبط
صيغة ثنائية الخطية متماثلة صيغتها التربيعية معرَّفة موجبة؛
ويدرس هذا الفصل الحالة العامة غير محدَّدة الإشارة.
\end{example}

\begin{example}[التوافق في العمل]\label{ex:b2:quadratic:congruence}
لنأخذ $q(x, y) = x^2 + 4xy + y^2$ (بالمصفوفة $B = \begin{pmatrix} 1
& 2\\ 2 & 1\end{pmatrix}$) والأساس الجديد $e_1' = (1, 1)$،
$e_2' = (1, -1)$، أي $P = \begin{pmatrix} 1 & 1\\ 1 &
-1\end{pmatrix}$. عندئذٍ
\[
P^{\mathsf T}BP
= \begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
\begin{pmatrix} 1 & 2\\ 2 & 1\end{pmatrix}
\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
= \begin{pmatrix} 6 & 0\\ 0 & -2\end{pmatrix} :
\]
وفي الإحداثيات $(u, v)$ على امتداد الأساس الجديد، $q = 6u^2 -
2v^2$ --- وللتحقق: يعطي $x = u + v$، $y = u - v$ أن $x^2 + 4xy +
y^2 = 6u^2 - 2v^2$ مباشرةً. ولاحظ أن العناصر القطرية الجديدة $6,
-2$ \emph{ليست} \omterm{def:b2:reduction:eigen}{القيم الذاتية} $3, -1$ للمصفوفة $B$: فالتوافق
يعيد تحجيم السلّم، ولا يحفظ الأطياف إلا التشابه --- لكن الإشارات
تتفق، كما تقتضي مبرهنة سيلفستر. (والأساس هنا
متعامد لكنه غير ممنظم؛ ونظمُه بالمقدار
$\frac{1}{\sqrt2}$ يقسم القطر على $2$ و
يستعيد \omterm{def:b2:reduction:eigen}{القيم الذاتية}.)
\end{example}

\begin{example}[محددات غرام تقيس المساحة]\label{ex:b2:quadratic:gramarea}
من أجل $v_1, v_2$ في فضاء إقليدي، تحزم مصفوفة غرام $G =
\bigl(\langle v_i, v_j\rangle\bigr)$ الأطوالَ
والزاوية؛ ويحزم محددها \emph{المساحة}:
\[
\det G = \norm{v_1}^2\norm{v_2}^2 - \langle v_1, v_2\rangle^2
= \norm{v_1}^2\norm{v_2}^2\bigl(1 - \cos^2\theta\bigr)
= \bigl(\norm{v_1}\,\norm{v_2}\sin\theta\bigr)^2 ,
\]
وهو مربّع مساحة متوازي الأضلاع المبني على $v_1, v_2$ ---
ومتراجحة كوشي--شوارتز هي بالضبط القول $\det G \geq 0$.
ومثال منفَّذ: $v_1 = (1, 2, 2)$، $v_2 = (2, 1, -2)$ في
$\R^3$:
\[
G = \begin{pmatrix} 9 & 0\\ 0 & 9 \end{pmatrix},
\qquad
\det G = 81 :
\]
فالمتجهتان متعامدتان وطولهما $3$، وتولّدان
متوازي أضلاع (وهو هنا مربّع) مساحته $\sqrt{81} = 9$.
والفكرة الختامية: لم يُستعمل أي جداء شعاعي ولا أي سحر خاص بالبعد
$3$ --- فالمقدار $\sqrt{\det G}$ يقيس الحجم ذا البعد $k$
في \emph{أي} بعد، وهو منطلق الجزء الأول من مسألة نهاية
الأسبوع وتكاملات مساحة السطوح لاحقًا في
هذا المجلد.
\end{example}

\section{اختزال غاوس وقصور سيلفستر الذاتي}

\begin{theorem}[اختزال غاوس]\label{thm:b2:quadratic:gauss}
كل \omterm{def:b2:quadratic:def}{صيغة تربيعية} $q$ على فضاء حقيقي منتهي البعد يمكن
كتابتها\index{اختزال غاوس}
\[
q = \sum_{i=1}^{s} \ell_i^2 - \sum_{j=1}^{t} m_j^2 ,
\]
حيث $\ell_1, \dots, \ell_s, m_1, \dots, m_t$ صيغ خطية مستقلة
خطيًا؛ وبكيفية مكافئة، يجعل أساسٌ ما مصفوفةَ
$q$ قطرية بعناصر $+1$ ($s$ مرة)، و$-1$ ($t$
مرة)، و$0$.
\end{theorem}

\begin{proof}
بالتراجع على عدد المتغيرات، بالإحداثيات: $q(x_1,
\dots, x_n)$.

\emph{الحالة 1: يظهر مربّع ما}، وليكن معامل $a$ للمتغير
$x_1^2$ غير معدوم. نجمّع كل الحدود التي فيها $x_1$ ونُتمّم
المربّع:
\[
q = a\Bigl(x_1 + \frac{1}{a}\,\lambda(x_2, \dots,
x_n)\Bigr)^{\!2} + q_1(x_2, \dots, x_n),
\]
حيث $\lambda$ خطية و$q_1$ تربيعية في المتغيرات
الباقية: فقد انفصلت صيغة مستقلة واحدة (فهي تشمل $x_1$، وأما
الأخريات فلا)، وينطبق التراجع على $q_1$، وتأتي إشارات $\pm$
من إشارة $a$ بعد تغيير السلّم بالمعامل $\sqrt{\abs a}$.

\emph{الحالة 2: لا مربّع، لكن ثمة حدّ متقاطع}، وليكن $b\,x_1x_2$ مع
$b \neq 0$. نستعمل المتطابقة
\[
x_1x_2 = \tfrac14\bigl((x_1 + x_2)^2 - (x_1 - x_2)^2\bigr)
\]
بعد التجميع: فبكتابة $q = b\,x_1x_2 + x_1\alpha + x_2\beta +
q_2$ (مع $\alpha, \beta, q_2$ في المتغيرات الأخرى)، نتحقق من أن
\[
q = \frac{b}{4}\Bigl[\Bigl(x_1 + x_2 + \frac{\alpha +
\beta}{b}\Bigr)^{2} - \Bigl(x_1 - x_2 + \frac{\beta -
\alpha}{b}\Bigr)^{2}\Bigr] + \widetilde q ,
\]
مع $\widetilde q$ خالية من $x_1, x_2$: فتنفصل صيغتان مستقلتان،
ويُنهي التراجع العمل.

استقلال الصيغ المجموعة: نرتّب الدفعات كما أُنتجت.
فصيغ الدفعة الأولى تحتوي $x_1$ (في الحالة 1)
أو $x_1, x_2$ (في الحالة 2)؛ وكل الصيغ اللاحقة خالية من تلك
المتغيرات. ولنفترض انعدام تركيب خطي لكل الصيغ
المجموعة. فبقراءة معامل $x_1$ (و$x_2$): لا تسهم إلا
الدفعة الأولى، وداخل تلك الدفعة تكون الصيغة أو الصيغتان
مستقلتين بوضوح ($\ell$ وحدها؛ أو $\ell \pm
m$ مع $\ell, m$ مستقلتين): فتنعدم معاملات الدفعة الأولى.
ثم نجرّد الدفعة ونكرّر: فبالتراجع على امتداد
الدفعات، تنعدم كل المعاملات --- فالعائلة كلها حرة،
وقد صار التثليث صريحًا.
\end{proof}

\begin{theorem}[قانون سيلفستر للقصور الذاتي]\label{thm:b2:quadratic:sylvester}
لا يتعلق الزوج $(s, t)$ في \cref{thm:b2:quadratic:gauss} إلا
بالصيغة $q$، لا بالاختزال: وهو
\emph{توقيع}\index{توقيع صيغة تربيعية} الصيغة $q$.
زيادةً على ذلك،
\[
s = \max\{\dim F : q|_F \text{ معرَّفة موجبة}\},
\]
وبالتناظر من أجل $t$.
\end{theorem}

\begin{proof}
ليكن $q = \sum_{i \leq s}\ell_i^2 - \sum_{j\leq t} m_j^2$ وليكن
$F_+$ الفضاء المولَّد بالمتجهات (القبلية) التي تقتصر عليها $(\ell_i)$
لتصير إحداثيات --- وبالتحديد: نُتمّ العائلة
المستقلة $(\ell_1, \dots, \ell_s, m_1, \dots, m_t)$
إلى \omterm{def:b2:linalg:dual}{أساس للثنوي} $E^*$، ولتكن $(u_1, \dots, u_n)$
الأساس للفضاء $E$ الذي تكون هذه صيغَ إحداثياته (وهو الأساس
القبلي: $\ell_i(u_k) = \delta_{ik}$ من أجل $k \leq s$،
وتنعدم الصيغ اللاحقة على المتجهات السابقة). ونضع $F_+ =
\operatorname{Vect}(u_1, \dots, u_s)$: فمن أجل $x = \sum_{i\leq
s}x_iu_i \in F_+$،
\[
\ell_i(x) = x_i,
\qquad m_j(x) = 0,
\qquad\text{ومنه}\qquad
q(x) = \sum_{i\leq s}x_i^2 > 0 \quad (x \neq 0) :
\]
فالصيغة $q|_{F_+}$ معرَّفة موجبة وتكون الأعظمية في الصيغة المعروضة
$\geq s$. وعكسيًا، ليكن $F$ أي فضاء جزئي تكون عليه $q|_F$
معرَّفة موجبة، وليكن $G = \{x : \ell_1(x) = \dots = \ell_s(x) =
0\}$، وبعده \omterm{def:b2:quadratic:adjoint}{المرافق} $\leq s$؛ وعلى $G$، $q(x) = -\sum m_j^2 \leq 0$.
عندئذٍ $F \cap G = \{0\}$ (فمتجهة غير معدومة هناك ستحقق $q > 0$
و$q \leq 0$)، ومنه $\dim F \leq \dim E - \dim G \leq s$. ومن ثم تساوي
الأعظمية $s$ من أجل \emph{كل} اختزال: أي إن $s$ ذاتي، وكذلك $t
= \operatorname{rk} q - s$.
\end{proof}

\begin{example}\label{ex:b2:quadratic:gaussexample}
$q(x, y, z) = xy + yz + zx$ (لا مربّعات). ومع $x_1 = x$، $x_2 =
y$: $q = xy + z(x + y)$، وتعطي متطابقة المربّعين
\[
q = \tfrac14(x + y + 2z)^2 - \tfrac14(x - y)^2 - z^2 ,
\]
(انشر للتحقق). ثلاث صيغ مستقلة: فالتوقيع $(1, 2)$،
والرتبة $3$. اتجاه موجب واحد، واتجاهان سالبان: أي هندسة
``مخروط الضوء'' لهذه الصيغة.
\end{example}

\begin{example}[صيغة منحلّة، مختزَلة بالكامل]\label{ex:b2:quadratic:degenerate}
$q(x, y, z) = xy + yz$ على $\R^3$: لا مربّعات، فالحالة 2 مع
التجميع $q = y(x + z)$. ومتطابقة المربّعين على جداء
الصيغتين المستقلتين $y$ و$x + z$:
\[
q = \frac14\bigl(y + x + z\bigr)^2 -
\frac14\bigl(y - x - z\bigr)^2 .
\]
والصيغتان الخطيتان $y + x + z$ و$y - x - z$
مستقلتان (ففرقهما $2(x+z)$، ومجموعهما $2y$)، ومن ثم
يقرأ سيلفستر: التوقيع $(1, 1)$، والرتبة $2$ ---
فهي \emph{منحلّة}. وتُوجَد نواة الصيغة القطبية
بحلّ $\varphi(v, \cdot) = 0$: فبالمصفوفة
$\frac12\begin{pmatrix} 0&1&0\\ 1&0&1\\ 0&1&0\end{pmatrix}$،
تكون النواة $\{y = 0,\ x + z = 0\} = \R\,(1, 0, -1)$، وهو
الاتجاه الذي لا ترى $q$ عليه شيئًا. والفكرة الختامية: يظهر
نقص الرتبة في غاوس بوصفه ``نفاد المتغيرات''
--- فقد أنتج الاختزال مربّعين فقط من ثلاثة
أبعاد، والبعد الناقص هو النواة بالضبط.
\end{example}

\begin{example}[صيغة واحدة، وطريقان إلى التوقيع]\label{ex:b2:quadratic:tworoads}
$q(x, y, z) = 2x^2 + 2y^2 + 2z^2 + 2xy + 2yz$، بالمصفوفة
$\begin{pmatrix} 2 & 1 & 0\\ 1 & 2 & 1\\ 0 & 1 & 2
\end{pmatrix}$. \emph{الطريق 1، غاوس:} نُتمّم المربّعات
بالترتيب،
\[
q = 2\Bigl(x + \frac y2\Bigr)^{\!2} + \frac32 y^2 + 2yz + 2z^2
= 2\Bigl(x + \frac y2\Bigr)^{\!2}
+ \frac32\Bigl(y + \frac{2z}{3}\Bigr)^{\!2} + \frac43 z^2 :
\]
فثلاثة مربّعات موجبة على صيغ مستقلة، والتوقيع $(3,
0)$: أي معرَّفة موجبة. \emph{الطريق 2، \omterm{def:b2:reduction:eigen}{القيم الذاتية}:} المصفوفة
هي المصفوفة ثلاثية الأقطار $2I + N$ مع $N$ مصفوفة الجوار؛ وقيمها
الذاتية $2 + \sqrt2$ و$2$ و$2 - \sqrt2$ (تحقق من
\omterm{def:b2:reduction:eigen}{المتجهات الذاتية} $(1, \pm\sqrt2, 1)$ و$(1, 0, -1)$)، وكلها
موجبة: أي الحكم نفسه، حسب
\cref{cor:b2:quadratic:principalaxes}. وغاوس أسرع؛
\omterm{def:b2:reduction:eigen}{والقيم الذاتية} تقول أكثر (فهي تعطي المحاور الرئيسية والقيم
الحدّية للصيغة $q$ على الكرة). والفكرة الختامية: المحاور
الموجبة $2, \frac32, \frac43$ عند غاوس هي بالضبط
نسب $\frac{\Delta_k}{\Delta_{k-1}}$ الأقصرات الرئيسية
الرائدة ($\Delta_1 = 2$، $\Delta_2 = 3$، $\Delta_3 = 4$) --- وتبرهن
مسألة نهاية الأسبوع على ذلك عمومًا.
\end{example}

\begin{method}[كيف نحسب توقيعًا: ثلاثة طرق]\label{met:b2:quadratic:signature}
\begin{enumerate}
  \item \emph{غاوس} (يعمل دائمًا، وهو الأسرع يدويًا): أتمِم
        المربّعات بالترتيب، وانتقل إلى الحالة 2 حين لا يتوفر مربّع؛
        ثم عُدّ الإشارات. وتحقق من أن الصيغ الخطية المجموعة
        مستقلة --- فعدد صيغ أقلّ من عدد المتغيرات يعني وجود
        نواة (\cref{ex:b2:quadratic:degenerate}).
  \item \emph{الأقصرات الرائدة} (من أجل اختبارات التحديد): كل
        $\Delta_k > 0$ إذا وفقط إذا كانت معرَّفة موجبة
        (\cref{exo:b2:quadratic:8})؛ بل إن المحاور
        $\Delta_k/\Delta_{k-1}$ تعطي معاملات
        غاوس (مسألة نهاية الأسبوع). ويفشل هذا صامتًا إذا كان
        $\Delta_k = 0$ ما معدومًا: فارجع إلى الطريق 1.
  \item \emph{\omterm{def:b2:reduction:eigen}{القيم الذاتية}} (الأكثر إفادةً والأكثر كلفة):
        إشارات \omterm{def:b2:reduction:eigen}{الطيف}
        (\cref{cor:b2:quadratic:principalaxes})؛ وهي تعطي أيضًا
        المحاور الرئيسية والقيم الحدّية للصيغة $q$ على
        كرة الوحدة. وفضّلها حين تكون البنية الذاتية
        مطلوبة أصلًا.
\end{enumerate}
\end{method}

\section{المبرهنة الطيفية}

وليكن الآن $E$ فضاءً إقليديًا (بجداء سلّمي $\langle\cdot,\cdot\rangle$،
مجلد السنة الأولى).

\begin{definition}[المرافق؛ التشاكلات الذاتية المتماثلة]\label{def:b2:quadratic:adjoint}
من أجل $u \in \mathcal{L}(E)$، يكون \emph{المرافق}\index{مرافق}
$u^*$ هو التشاكل الذاتي الوحيد الذي يحقق
\[
\langle u(x), y\rangle = \langle x, u^*(y)\rangle
\qquad (x, y \in E);
\]
وفي أساس متعامد ممنظم، $\operatorname{Mat}(u^*) =
\operatorname{Mat}(u)^{\mathsf T}$. ويكون $u$
\emph{متماثلًا}\index{تشاكل ذاتي متماثل} (ذاتي المرافق) إذا كان
$u^* = u$ --- وبكيفية مكافئة، إذا كانت مصفوفته في أساس متعامد ممنظم
متماثلة.
\end{definition}

\begin{proof}[وجود المرافق ووحدانيته]
من أجل $y$ ثابتة، تكون الصيغة $x \mapsto \langle u(x), y\rangle$
خطية، ومن ثم (في البعد المنتهي) من الشكل $\langle x,
z_y\rangle$ من أجل $z_y$ وحيدة --- ويكون التطبيق $y \mapsto z_y =:
u^*(y)$ خطيًا بالوحدانية. وأما المماهاة المصفوفية:
فالعلاقة $\langle u(e_i), e_j\rangle$ مقروءةً في الاتجاهين.
\end{proof}

\begin{example}[المرافق يتعلق بالجداء السلّمي]\label{ex:b2:quadratic:weightedadjoint}
على $\R^2$ نأخذ الجداء السلّمي \emph{الموزون} $\langle x,
y\rangle_D = x_1y_1 + 2x_2y_2$ (بالمصفوفة $D =
\operatorname{diag}(1,2)$) ونأخذ $u$ بالمصفوفة $A =
\begin{pmatrix} 0 & 1\\ 0 & 0\end{pmatrix}$ في الأساس
القانوني. ومن $\langle u(x), y\rangle_D = (Ax)^{\mathsf T}Dy =
x^{\mathsf T}(A^{\mathsf T}D)y$ ومن $\langle x, u^*(y)\rangle_D
= x^{\mathsf T}(DA^*)y$، تكون مصفوفة \omterm{def:b2:quadratic:adjoint}{المرافق}
\[
A^* = D^{-1}A^{\mathsf T}D
= \begin{pmatrix} 1 & 0\\ 0 & \tfrac12\end{pmatrix}
\begin{pmatrix} 0 & 0\\ 1 & 0\end{pmatrix}
\begin{pmatrix} 1 & 0\\ 0 & 2\end{pmatrix}
= \begin{pmatrix} 0 & 0\\ \tfrac12 & 0\end{pmatrix}
\neq A^{\mathsf T} .
\]
وللتحقق على $x = (1,0)$، $y = (0,1)$:
\[
\langle u(x), y\rangle_D = \langle (0,0), y\rangle_D = 0,
\quad
\langle x, u^*(y)\rangle_D
= \langle(1,0), (0,\tfrac12)\rangle_D = 0 ;
\]
وعلى $x = (0,1)$، $y = (1,0)$:
\[
\langle u(x), y\rangle_D = \langle(1,0),(1,0)\rangle_D = 1,
\quad
\langle x, u^*(y)\rangle_D =
\langle(0,1),(0,\tfrac12)\rangle_D = 1 .
\] والفكرة الختامية: قولُ ``$\operatorname{Mat}(u^*)
= \operatorname{Mat}(u)^{\mathsf T}$'' قولٌ في الأسس
\emph{المتعامدة الممنظمة} فقط؛ وعمومًا يتدخل المتري $D$،
تمامًا كما في الاختزال المتزامن في
مسألة نهاية الأسبوع.
\end{example}

\begin{theorem}[المبرهنة الطيفية]\label{thm:b2:quadratic:spectral}
ليكن $u$ تشاكلًا ذاتيًا \omterm{def:b2:quadratic:adjoint}{متماثلًا} لفضاء إقليدي $E$. عندئذٍ
يملك $E$ \emph{أساسًا متعامدًا ممنظمًا من \omterm{def:b2:reduction:eigen}{المتجهات الذاتية}} للتشاكل $u$؛ وكل
\omterm{def:b2:reduction:eigen}{القيم الذاتية} حقيقية، \omterm{def:b2:reduction:eigen}{والفضاءات الذاتية} الموافقة لقيم ذاتية مختلفة
متعامدة. والصيغة المصفوفية: كل مصفوفة حقيقية متماثلة $A$ تُكتب
\[
A = P\,D\,P^{\mathsf T},
\qquad P \text{ متعامدة } (P^{\mathsf T} P = I),\ D
\text{ قطرية}.\index{مبرهنة طيفية}
\]
\end{theorem}

\begin{proof}
\emph{وجود \omterm{def:b2:reduction:eigen}{متجهة ذاتية}.} الدالة $x \mapsto \langle u(x),
x\rangle$ متصلة على كرة الوحدة $S$ من $E$، وهي
متراصة (فالبعد منته، \cref{thm:b2:nvs:finitedim}): فتبلغ
أعظميتها $\lambda$ عند $a \in S$ ما. وندّعي أن $u(a) = \lambda
a$. فمن أجل أي $y \perp a$ تحقق $\norm y = 1$ و$t \in \R$، تقع
المتجهة $x_t = \frac{a + ty}{\sqrt{1 + t^2}}$ على $S$
(لأن $\norm{a + ty}^2 = 1 + t^2$ بمبرهنة فيثاغورس)؛ وبنشر
الدالة المعظَّمة،
\[
g(t) = \langle u(x_t), x_t\rangle
= \frac{\langle u(a), a\rangle + 2t\langle u(a), y\rangle +
t^2\langle u(y), y\rangle}{1 + t^2}
\]
(فقد دمج تماثل $u$ الحدّين المتقاطعين: $\langle u(a),
y\rangle = \langle a, u(y)\rangle = \langle u(y), a\rangle$).
و$g$ دالة قابلة للاشتقاق في $t$ لها أعظمية عند $t =
0$؛ وتعطي قاعدة خارج القسمة عند $0$ أن
\[
g'(0) = \frac{2\langle u(a), y\rangle\cdot 1 - \langle u(a),
a\rangle\cdot 0}{1} = 2\langle u(a), y\rangle = 0 .
\]
ومن ثم فإن $u(a)$ متعامدة مع الفضاء الفوقي $a^\perp$ كله: أي $u(a)
\in (a^{\perp})^{\perp} = \R a$، أي $u(a) = \mu a$؛ و$\mu =
\langle u(a), a\rangle = \lambda$.

\emph{التراجع.} المتممة المتعامدة $F = a^\perp$
مستقرة تحت $u$: فمن أجل $x \perp a$، $\langle u(x), a\rangle = \langle x,
u(a)\rangle = \lambda\langle x, a\rangle = 0$. والمقصور
$u|_F$ متماثل من أجل الجداء السلّمي المستحثّ؛ وبالتراجع على
البعد، يملك $F$ أساسًا ذاتيًا متعامدًا ممنظمًا؛ ونضيف $a$ في المقدمة.

\emph{التتمات.} \omterm{def:b2:reduction:eigen}{القيم الذاتية} هي الأعداد الحقيقية $\langle u(e),
e\rangle$ على الأساس الذاتي. وتعامد \omterm{def:b2:reduction:eigen}{الفضاءات الذاتية}: يعطي $u(x) =
\lambda x$، $u(y) = \mu y$ أن $\lambda\langle x, y\rangle =
\langle u(x), y\rangle = \langle x, u(y)\rangle = \mu\langle x,
y\rangle$، ومنه $\langle x, y\rangle = 0$ حين $\lambda \neq \mu$.
والصيغة المصفوفية: أعمدة $P$ هي الأساس الذاتي المتعامد الممنظم.
\end{proof}

\begin{example}[تشغيلة طيفية كاملة]\label{ex:b2:quadratic:fullrun}
لنقطّر تعامديًا $A = \begin{pmatrix} 1 & 2\\ 2 &
1\end{pmatrix}$. \omterm{def:b2:reduction:charpoly}{كثير الحدود المميز} $(1 - \lambda)^2 -
4$: فالقيمتان الذاتيتان $3$ و$-1$. والمتجهتان الذاتيتان: يعطي $(A - 3I)v = 0$
أن $v_1 = \frac{1}{\sqrt2}(1,1)$؛ ويعطي $(A + I)v = 0$ أن $v_2
= \frac{1}{\sqrt2}(1,-1)$ --- وهما متعامدتان، كما
تضمن \cref{thm:b2:quadratic:spectral} بلا
حساب. ومع $P = (v_1\ v_2)$ (وهي دوران بزاوية
$\frac\pi4$):
\[
P^{\mathsf T}AP = \begin{pmatrix} 3 & 0\\ 0 & -1
\end{pmatrix},
\qquad
x^2 + 4xy + y^2 = 3u^2 - v^2
\quad\text{في المعلم المُدار} .
\]
ومن ثم فصيغة \cref{exo:b2:quadratic:1} من نمط
القطع الزائد: بتوقيع $(1,1)$، وهو متوافق مع \omterm{thm:b2:quadratic:gauss}{اختزال غاوس} لها
$(x + 2y)^2 - 3y^2$ --- مربّعات مختلفة، وتوقيع واحد، كما
تقتضي مبرهنة سيلفستر. والفكرة الختامية: أعطى غاوس الجواب
أسرع، لكن الطريق الطيفي يفيد أيضًا بأن $q$ تجري على دائرة
الوحدة على $\intcc{-1}{3}$ بالضبط، وتُبلغ على امتداد
$v_2$ و$v_1$: فالعمل الإضافي يشتري هندسة.
\end{example}

\begin{corollary}[المحاور الرئيسية؛ اختبارات الإيجابية]\label{cor:b2:quadratic:principalaxes}
\begin{enumerate}
  \item كل \omterm{def:b2:quadratic:def}{صيغة تربيعية} $q$ على فضاء إقليدي تتقطّر
        في أساس \emph{متعامد ممنظم} ما: $q(x) = \sum_i \lambda_i
        x_i^2$ مع $\lambda_i$ \omterm{def:b2:reduction:eigen}{القيم الذاتية} للمصفوفة المتماثلة
        للصيغة $q$؛ ويعدّ التوقيع \omterm{def:b2:reduction:eigen}{القيم الذاتية} الموجبة والسالبة.
  \item تكون المصفوفة المتماثلة شبه معرَّفة موجبة (أو
        معرَّفة موجبة) إذا وفقط إذا كانت كل قيمها الذاتية $\geq 0$ (أو $>
        0$)؛ وعندئذٍ تكون القيم الحدّية لخارج قسمة رايلي
        هي
        \[
        \min_{\norm x = 1} \langle Ax, x\rangle = \lambda_{\min},
        \qquad
        \max_{\norm x = 1} \langle Ax, x\rangle = \lambda_{\max} .
        \]
\end{enumerate}
\end{corollary}

\begin{proof}
(1) نكتب $q(x) = \langle A x, x\rangle$ مع $A$ متماثلة (وهي
مصفوفة $q$ في أساس متعامد ممنظم)؛ ونقطّر $A$
بالمبرهنة الطيفية: فمن أجل $x = \sum x_ie_i$ في الأساس الذاتي
المتعامد الممنظم،
\[
q(x) = \Bigl\langle \sum_i \lambda_ix_ie_i,\ \sum_j
x_je_j\Bigr\rangle = \sum_i \lambda_i x_i^2
\]
(فالتعامد والتنظيم يقتلان الحدود المتقاطعة). وتعدّ إشارات $\pm$ للقيم
$\lambda_i$ التوقيعَ بمبرهنة سيلفستر: فتغيير سلّم كل
إحداثية بالمعامل $\sqrt{\abs{\lambda_i}}$ يُبرز اختزالَ
غاوس بصيغ مستقلة.

(2) في الأساس الذاتي، $\langle Ax, x\rangle = \sum \lambda_i
x_i^2$، وهو محصور بين $\lambda_{\min}\norm x^2$ و
$\lambda_{\max}\norm x^2$، مع المساواة عند \omterm{def:b2:reduction:eigen}{المتجهات الذاتية}
الموافقة؛ ومن ثم فإيجابية كل \omterm{def:b2:reduction:eigen}{القيم الذاتية} مكافئة
لإيجابية الصيغة.
\end{proof}

\begin{example}\label{ex:b2:quadratic:spectralexample}
$A = \begin{pmatrix} 2 & 1\\ 1 & 2 \end{pmatrix}$: القيمتان الذاتيتان $3$
(\omterm{def:b2:reduction:eigen}{بمتجهة ذاتية} $\frac{1}{\sqrt2}(1,1)$) و$1$
(بالمتجهة $\frac{1}{\sqrt2}(1,-1)$). وتصير \omterm{def:b2:quadratic:def}{الصيغة التربيعية} $2x^2 + 2xy + 2y^2$
هي $3X^2 + Y^2$ في المعلم المتعامد الممنظم المُدار: أي المحاور
الرئيسية لقطع ناقص، محسوبةً. ويبلغ \omterm{thm:b2:quadratic:gauss}{اختزال غاوس} صيغة قطرية
أيضًا، لكن المبرهنة الطيفية وحدها تبلغها \emph{دون
تشويه الأطوال}.
\end{example}

\begin{example}[قطع ناقص معيَّن بالكامل]\label{ex:b2:quadratic:ellipserun}
أي منحنٍ هو $5x^2 + 4xy + 2y^2 = 6$؟ لمصفوفته
$\begin{pmatrix} 5 & 2\\ 2 & 2\end{pmatrix}$ كثيرُ الحدود
المميز $\lambda^2 - 7\lambda + 6 = (\lambda - 1)(\lambda -
6)$: فالقيمتان الذاتيتان $1$ و$6$، وكلتاهما موجبة --- أي إنه قطع ناقص.
والمتجهتان الذاتيتان المتعامدتان الممنظمتان: من أجل $\lambda = 1$، نحلّ
$\begin{pmatrix} 4 & 2\\ 2 & 1\end{pmatrix}v = 0$: فنجد $v_1 =
\frac{1}{\sqrt5}(1, -2)$؛ ومن أجل $\lambda = 6$: $v_2 =
\frac{1}{\sqrt5}(2, 1)$. وفي الإحداثيات المُدارة $(X, Y)$
على امتداد $(v_2, v_1)$ تصير المعادلة
\[
6X^2 + Y^2 = 6,
\qquad\text{أي}\qquad
X^2 + \frac{Y^2}{6} = 1 :
\]
فأنصاف المحاور $1$ (على امتداد $v_2$) و$\sqrt6$ (على امتداد $v_1$).
والفكرة الختامية: كان الشكل التقريبي مجانيًا --- فالشرط $\det
= 6 > 0$ والأثر الموجب يعلنان قطعًا ناقصًا قبل حساب أي
\omterm{def:b2:reduction:eigen}{متجهة ذاتية} --- لكن المبرهنة الطيفية وحدها
تعطي اتجاهات المحاور وأطوالها، أي الهندسة
الفعلية.
\end{example}

\begin{example}[الحدود القصوى على الكرة، مقروءةً من الطيف]\label{ex:b2:quadratic:sphereextremes}
ما القيم الحدّية للصيغة $q(x,y,z) = 2xy + 2yz + 2zx$ على
كرة الوحدة؟ لمصفوفتها (وهي التي كل عناصرها خارج القطر واحد في
\cref{exo:b2:quadratic:2}) القيمتان الذاتيتان $2$ و$-1$
(مضاعفة)، ومن ثم يعطي البند (2) في \cref{cor:b2:quadratic:principalaxes}:
\[
\max_{\norm v = 1} q(v) = 2
\ \text{ عند } v = \tfrac{1}{\sqrt3}(1,1,1),
\qquad
\min_{\norm v = 1} q(v) = -1
\ \text{ على الدائرة } x + y + z = 0 .
\]
فلا تفاضل ولا مضاعفات لاغرانج: فالمبرهنة الطيفية
تحلّ هذا التحسين المقيَّد فورًا --- وتُبرز
المعظِّم. والفكرة الختامية: قارِن بطريقة المضاعفات
في فصل حساب التفاضل، التي تجد النقاط الحرجة
نفسها بجهد أكبر؛ ومن أجل الدوال الهدف \emph{التربيعية}
على الكرات، تكون الأطياف الطريق الملكي (وتبني مسألة
نهاية الأسبوع في الفصل الهرميتي نظريةَ
كوران--فيشر كلها على هذه الملاحظة).
\end{example}

\begin{remark}[مزالق شائعة]
\emph{(1) التوافق ليس تشابهًا:} فتغيير الأساس من أجل
صيغة يؤثّر بالقانون $P^{\mathsf T}BP$ لا $P^{-1}BP$؛ \omterm{def:b2:reduction:eigen}{والقيم الذاتية}
\emph{ليست} لا متغيّرات \omterm{def:b2:quadratic:def}{لصيغة تربيعية} (فالمصفوفتان $I$ و$4I$
متوافقتان عبر $P = 2I$) --- ولا تكون كذلك إلا إشاراتها
(سيلفستر). فلا تتحدث عن \omterm{def:b2:reduction:eigen}{القيم الذاتية} لصيغة إلا بعد تثبيت
جداء سلّمي. \emph{(2) العناصر الموجبة لا تبرهن على
شيء:} فللمصفوفة $\begin{pmatrix} 1 & 2\\ 2 & 1\end{pmatrix}$ كل
العناصر موجبة ومع ذلك توقيعها $(1,1)$ (لأن $\det = -3$)؛
وعكسيًا قد يكون للمصفوفة المعرَّفة الموجبة عناصر سالبة
خارج القطر (فالمصفوفة \cref{ex:b2:quadratic:tworoads} مزاحةً:
$2I - N$ تفي كذلك). فاستعمل
\cref{met:b2:quadratic:signature}. \emph{(3) المربّعات
المرتبطة:} فكتابة $q = \ell_1^2 - \ell_2^2$ لا تقول شيئًا إذا كانت
$\ell_1, \ell_2$ متناسبة --- فرتبة $x^2 + 2xy + y^2 =
(x+y)^2$ هي $1$ لا $2$؛ فتحقق دائمًا من الاستقلال
قبل قراءة التوقيع. \emph{(4) الحدود القصوى على الكرة
دون تراص:} فحدود رايلي في
\cref{cor:b2:quadratic:principalaxes} مبلوغة لأن
الكرة متراصة؛ وعلى الكرة \omterm{def:b2:metric:topology}{المفتوحة} أو على الفضاء كله، لا
تملك الصيغة غير محدَّدة الإشارة أعظمية ولا أصغرية.
\end{remark}

\begin{remark}[أين يُستعمَل هذا]
المبرهنة الطيفية أكثر نتائج هذا الكتاب تصديرًا:
فالإحصاء يقطّر بها مصفوفات التباين المشترك
(تحليل المركّبات الرئيسية)، والميكانيك يستخرج بها الأنماط
الطبيعية للتذبذب (وهي الاختزال المتزامن في
مسألة نهاية الأسبوع)، والتحليل العددي يبني عليها \omterm{pb:b2:quadratic:1}{تفكيك
تشوليسكي} وتفكيك القيم المنفردة (المسألة نفسها)، وينقلها الفصل
التالي إلى الفضاءات الهرميتية العقدية. ويبرهن مجلد السنة
الثالثة على نظيرها غير المنتهي البعد من أجل المؤثرات المتراصة
ذاتية \omterm{def:b2:quadratic:adjoint}{المرافق}، حيث تصير حجة التراص في البرهان المنتهي البعد
الحكايةَ كلها.
\end{remark}

\begin{remark}[آفاق داخل هذا المجلد]
تخترق الصيغ التربيعية بقية الكتاب الرابع بثلاثة
أزياء. بوصفها \emph{مصفوفات هيسي}: فيصنّف فصل حساب
التفاضل النقاطَ الحرجة بتوقيع الصيغة من
الرتبة الثانية، ومن ثم فلا تغيّر سيلفستر هو ما يجعل
كلمة ``سرج'' معرَّفة جيدًا. وبوصفها \emph{طاقات}: فمتذبذبات
فصل المعادلات التفاضلية تحمل الطاقة
التربيعية $\frac12x'^2 + \frac12\omega^2x^2$، والاختزال المتزامن
في مسألة نهاية الأسبوع في هذا الفصل هو
بالضبط استخراج الأنماط الطبيعية. وبوصفها \emph{هندسة}:
فمخروطيات هذا الفصل تكبر لتصير السطوح التربيعية في
فصول الهندسة، حيث تكون للصيغة الأساسية الثانية
لسطح --- وهي \omterm{def:b2:quadratic:def}{صيغة تربيعية} على كل مستوٍ مماسّ ---
توقيعٌ يقرّر ما إذا كان السطح ينحني كوعاء أم
كسرج. ويعيد الفصل الهرميتي التالي عزف المقطوعة كلها
على $\C$.
\end{remark}

(محك سيلفستر) لتكن $A$ حقيقية متماثلة بأقصرات
رئيسية رائدة $\Delta_1, \dots, \Delta_n$ (وهي محددات
الكتل العليا اليسرى). برهن على أن $A$ معرَّفة موجبة إذا وفقط
إذا كانت كل $\Delta_k > 0$. \emph{(من أجل $\Rightarrow$: مقصورات الصيغة
المعرَّفة معرَّفة، \omterm{def:b2:linalg:det}{ومحدد المصفوفة} المعرَّفة
الموجبة --- وهو جداء قيمها الذاتية --- موجب.
ومن أجل $\Leftarrow$: استعمل التراجع على $n$؛ فالكتلة العليا اليسرى ذات القياس $(n-1)$
معرَّفة موجبة، فقطّر الصيغة على ذلك الفضاء الجزئي و
أتمِم المربّع في المتغير الأخير؛ وتحكم $\det A = \Delta_n > 0$ إشارةَ العنصر
القطري الأخير.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:quadratic:9}
(كوران--فيشر، \omterm{def:b2:reduction:eigen}{القيمة الذاتية} الثانية) لتكن $u$ متماثلة بقيم
ذاتية $\lambda_1 \geq \lambda_2 \geq \dots \geq \lambda_n$.
برهن على أن
\[
\lambda_2 = \min_{\substack{H \text{ فضاء فوقي}}}\;
\max_{\substack{x \in H,\ \norm x = 1}} \langle u(x), x\rangle .
\]
\emph{(من أجل $\leq$: يلتقي أي فضاء فوقي المستوي $2$ المولَّد
بالمتجهتين الذاتيتين العلييين. ومن أجل $\geq$: اختر $H =
(e_1)^{\perp}$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:10}
عيّن رتبة وتوقيع الصيغة $q(x_1, \dots, x_n) = \sum_{i <
j} x_ix_j$ على $\R^n$ (مع $n \geq 2$)، بطريقتين: بالمتطابقة الجبرية
$2q = \bigl(\sum x_i\bigr)^2 - \sum x_i^2$ مع مقصور $q$ على الفضاء الفوقي $\sum x_i = 0$؛
وبحساب \omterm{def:b2:reduction:eigen}{القيم الذاتية} لمصفوفتها $\frac12(J - I)$،
حيث $J$ المصفوفة التي كل عناصرها واحد.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:11}
لتكن $A$ و$B$ حقيقيتين متماثلتين مع $B$ شبه معرَّفة موجبة.
برهن على أن
\[
\lambda_{\min}(A)\operatorname{tr} B
\;\leq\;
\operatorname{tr}(AB)
\;\leq\;
\lambda_{\max}(A)\operatorname{tr} B .
\]
\emph{(اكتب $B = C^{\mathsf T}C$ و$\operatorname{tr}(AB) =
\sum_i \langle A c_i, c_i\rangle$ على أعمدة $c_i$ للمصفوفة
$C^{\mathsf T}$.)} وعلى الخصوص $\operatorname{tr}(AB) \geq 0$
حين تكون كلتاهما شبه معرَّفة موجبة.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:quadratic:12}
على $E = \mathcal{M}_n(\R)$، لننظر في $q(M) =
\operatorname{tr}(M^2)$.
\begin{enumerate}
  \item بيّن أن $q$ \omterm{def:b2:quadratic:def}{صيغة تربيعية} صيغتها القطبية
        $\varphi(M, N) = \operatorname{tr}(MN)$.
  \item بيّن أن المصفوفات المتماثلة وضدّ المتماثلة تشكّل
        فضاءين جزئيين متعامدين من أجل $\varphi$ تكون $q$ عليهما
        معرَّفة موجبة ومعرَّفة سالبة على التوالي
        \emph{(احسب $\operatorname{tr}(M^2)$ عنصرًا عنصرًا في
        كل حالة)}.
  \item اختم: توقيع $q$ هو $\bigl(\frac{n(n+1)}{2},
        \frac{n(n-1)}{2}\bigr)$ ورتبتها $n^2$.
\end{enumerate}
\end{exercise}

\section{مسألة: تشوليسكي وهادامار والتفكيك القطبي}

\begin{problem}\label{pb:b2:quadratic:1}
المبرهنة الطيفية مجهر؛ وتستعملها هذه المسألة مصنعًا.
فمن مصفوفات غرام نصنع \emph{تفكيك
تشوليسكي}\index{تفكيك تشوليسكي} (ونعيّن محاور
غاوس بوصفها نسب أقصرات)، ثم نبرهن على \emph{متراجحة
هادامار}\index{متراجحة هادامار} للمحددات، ونبني
\emph{التفكيك القطبي}\index{تفكيك قطبي} $A =
QS$ و\emph{تفكيك القيم المنفردة}، ونصنّف مخروطيات
المستوي، ونُختم بالاختزال المتزامن لصيغتين ---
وهي المبرهنة وراء الأنماط الطبيعية للتذبذب. وفي كل ما يلي، $E =
\R^n$ بجدائه السلّمي القياسي.

\medskip
\noindent\textbf{الجزء الأول --- مصفوفات غرام وتشوليسكي.} من أجل
متجهات $v_1, \dots, v_n \in E$، تكون \emph{مصفوفة غرام} الخاصة بها هي
$G = \bigl(\langle v_i, v_j\rangle\bigr)_{i,j}$.
\begin{enumerate}
  \item بيّن أن $G$ متماثلة شبه معرَّفة موجبة، و
        معرَّفة موجبة إذا وفقط إذا كانت $(v_1, \dots, v_n)$
        مستقلة خطيًا \emph{(احسب $X^{\mathsf T}GX$)}.
  \item وعكسيًا، بيّن أن كل مصفوفة متماثلة شبه معرَّفة
        موجبة $A$ مصفوفةُ غرام: أي $A = C^{\mathsf T}C$
        من أجل $C$ ما (استعمل الجذر التربيعي للمصفوفة
        \cref{exo:b2:quadratic:6})، مع $C$ قابلة للقلب إذا وفقط إذا كانت $A$
        معرَّفة.
  \item استنتج أن المصفوفة شبه المعرَّفة الموجبة $A$ تحقق
        $\abs{a_{ij}} \leq \sqrt{a_{ii}\,a_{jj}}$ من أجل كل $i,
        j$ \emph{(اقصر على إحداثيتين)} --- أي
        متراجحة كوشي--شوارتز، مقروءةً مصفوفيًا.
  \item (تشوليسكي) لتكن $A$ معرَّفة موجبة. برهن على أنه
        توجد مصفوفة مثلثية عليا \emph{وحيدة} $T$ بعناصر
        قطرية موجبة تحقق
        \[
        A = T^{\mathsf T}\,T
        \]
        \emph{(الوجود: طبّق غرام--شميت على متجهات
        تحقق $A$ بوصفها مصفوفة غرام؛ والوحدانية: إذا كان
        $T_1^{\mathsf T}T_1 = T_2^{\mathsf T}T_2$، فبيّن أن
        $T_1T_2^{-1}$ متعامدة ومثلثية بقطر
        موجب، ومن ثم $I$)}.
  \item بيّن أن الأقصرات الرئيسية الرائدة تحقق
        $\Delta_k = (t_{11}\cdots t_{kk})^2$، واستنتج أن
        المحاور التي يُنتجها \omterm{thm:b2:quadratic:gauss}{اختزال غاوس} لصيغة
        معرَّفة موجبة، مأخوذةً بترتيب المتغيرات الطبيعي،
        هي
        \[
        d_k = \frac{\Delta_k}{\Delta_{k-1}}
        \qquad (\Delta_0 = 1) :
        \]
        فأقصرات محك سيلفستر
        (\cref{exo:b2:quadratic:8}) ومحاور غاوس هما
        المعطى نفسه. وتحقق على
        \cref{ex:b2:quadratic:tworoads}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- متراجحة هادامار.}
\begin{enumerate}[resume]
  \item لتكن $A$ معرَّفة موجبة. برهن على أن
        \[
        \det A \leq a_{11}\,a_{22}\cdots a_{nn}
        \]
        \emph{(نظّم: فالمصفوفة $B = DAD$ مع $D =
        \operatorname{diag}(a_{ii}^{-1/2})$ لها قطر واحدي؛
        وحُدّ $\det B = \prod \mu_i$ بمتراجحة الحسابي والهندسي في مقابل
        $\operatorname{tr} B = n$)}.
  \item بيّن أن المساواة تتحقق إذا وفقط إذا كانت $A$ قطرية.
  \item استنتج \emph{متراجحة هادامار}: من أجل كل مصفوفة
        مربّعة حقيقية $M$ بأعمدة $c_1, \dots, c_n$،
        \[
        \abs{\det M} \leq \prod_{i=1}^{n}\norm{c_i}_2 ,
        \]
        مع المساواة (من أجل $M$ القابلة للقلب) إذا وفقط إذا كانت الأعمدة
        متعامدة مثنى مثنى \emph{(طبّق السؤالين 6--7 على
        $M^{\mathsf T}M$)}.
  \item أرباح هندسية وتوافقية: فسّر
        السؤال 8 هكذا: ``حجم متوازي السطوح لا يتجاوز
        جداء أطوال أحرفه''؛ وبيّن أن
        المصفوفة التي كل عناصرها في $\intcc{-1}{1}$ تحقق
        $\abs{\det M} \leq n^{n/2}$. (والمصفوفات التي تبلغ هذا
        الحدّ --- أي مصفوفات هادامار --- موجودة من أجل $n = 1, 2$
        وكثير من مضاعفات $4$؛ وأما وجودها من أجل \emph{كل}
        مضاعفات $4$ فمسألة \omterm{def:b2:metric:topology}{مفتوحة} شهيرة.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- \omterm{pb:b2:quadratic:1}{التفكيك القطبي} والقيم
المنفردة.}
\begin{enumerate}[resume]
  \item لتكن $A$ قابلة للقلب. بيّن أن $A^{\mathsf T}A$
        معرَّفة موجبة، وأن
        \[
        S = \sqrt{A^{\mathsf T}A}
        \quad\text{(الجذر التربيعي للمصفوفة
        \cref{exo:b2:quadratic:6})},
        \qquad Q = AS^{-1}
        \]
        يعطيان تفكيكًا $A = QS$ مع $Q$ متعامدة و
        $S$ معرَّفة موجبة.
  \item برهن على أن هذا التفكيك لمصفوفة $A$ قابلة للقلب
        وحيد.
  \item مدّد الوجود إلى $A$ كيفية: اختر
        $\varepsilon_k \to 0$ بحيث تكون $A + \varepsilon_k I$
        قابلة للقلب، واكتب $A + \varepsilon_kI = Q_kS_k$، و
        استعمل تراص $O(n)$
        (\cref{exo:b2:quadratic:5}) لاستخراج $Q_k \to Q$؛
        وبيّن أن $S_k = Q_k^{\mathsf T}(A + \varepsilon_kI)$
        تتقارب إلى مصفوفة شبه معرَّفة موجبة $S$ تحقق $A =
        QS$ و$S = \sqrt{A^{\mathsf T}A}$. وأين تفشل
        الوحدانية من أجل $A$ الشاذة؟
  \item (تفكيك القيم المنفردة) استنتج أن كل مصفوفة مربّعة
        حقيقية $A$ تُكتب
        \[
        A = U\,\Sigma\,V^{\mathsf T},
        \qquad U, V \in O(n),\quad
        \Sigma = \operatorname{diag}(\sigma_1, \dots,
        \sigma_n),\ \sigma_i \geq 0 ,
        \]
        حيث الأعداد $\sigma_i$ (وهي \emph{القيم المنفردة}) هي
        \omterm{def:b2:reduction:eigen}{القيم الذاتية} للمصفوفة $\sqrt{A^{\mathsf T}A}$.
  \item ثلاث نتائج: $\vertiii{A}_2 = \sigma_{\max}$ من أجل
        \emph{كل} مصفوفة حقيقية $A$ (وهذا يعمّم
        \cref{exo:b2:quadratic:7})؛ و$\abs{\det A} =
        \sigma_1\cdots\sigma_n$؛ وصورة كرة الوحدة
        بالتطبيق $A$ القابل للقلب هي مجسّم ناقص بأنصاف
        محاور $\sigma_1, \dots, \sigma_n$ على امتداد أعمدة
        $U$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- المخروطيات، بالمبرهنة الطيفية.}
المخروطية المستوية هي مجموعة أصفار $f(x) = q(x) + \langle b,
x\rangle + c$، مع $q \neq 0$ \omterm{def:b2:quadratic:def}{صيغة تربيعية} مصفوفتها $A$،
و$b \in \R^2$، و$c \in \R$.
\begin{enumerate}[resume]
  \item اختزل $f$ بدوران (المحاور الرئيسية،
        \cref{cor:b2:quadratic:principalaxes}) متبوعًا
        بانسحاب، وصنّف الأشكال الممكنة غير الخالية وغير
        المنحلّة بتوقيع $q$: قطع ناقص
        ($\det A > 0$)، وقطع زائد ($\det A < 0$)، وقطع مكافئ
        ($\det A = 0$، بالرتبة $1$، مع عدم امتصاص الحدّ الخطي).
  \item شغّل الاختزال كاملًا من أجل
        \[
        x^2 + 4xy + y^2 + 2x - 2y = 4 :
        \]
        الإحداثيات المُدارة، والمعادلة المختزَلة، وطبيعة
        المخروطية ومركزها.
  \item (المخروطيات المركزية) لنفترض $\det A \neq 0$. بيّن أن
        \emph{المركز} هو $x_0 = -\frac12 A^{-1}b$، وأن
        التوافق بالمصفوفة $\begin{pmatrix} I & x_0\\ 0 &
        1\end{pmatrix}$ للمصفوفة $\widetilde Q
        = \begin{pmatrix} A & b/2 \\ b^{\mathsf T}/2 &
        c\end{pmatrix}$ ذات القياس $3\times3$ يعطي
        \[
        \det\widetilde Q = f(x_0)\,\det A :
        \]
        فتكون المخروطية المركزية منحلّة (نقطة أو مستقيمين)
        بالضبط حين $\det\widetilde Q = 0$.
  \item تحقق من السؤال 17 على مثال السؤال 16:
        احسب $x_0$ و$f(x_0)$ و$\det\widetilde Q$، و
        اختم من جديد بأن المخروطية قطع زائد غير
        منحلّ.
  \item (حزمة تربيعية) من أجل $\lambda \in \R$، صنّف
        السطح
        \[
        x^2 + y^2 + z^2 + 2\lambda(xy + yz + zx) = 1
        \]
        \omterm{def:b2:reduction:eigen}{بالقيم الذاتية} لمصفوفته \emph{(ببنية الواحدات:
        القيمتان الذاتيتان $1 + 2\lambda$ و$1 -
        \lambda$ مضاعفة)}: كرة أو مجسّم ناقص، أو أسطوانة، أو زوج
        مستويات، أو مجسّم زائد ذو طبقة واحدة أو طبقتين، تبعًا
        للمقدار $\lambda$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- صيغتان معًا: الاختزال
المتزامن.}
\begin{enumerate}[resume]
  \item لتكن $q$ معرَّفة موجبة ولتكن $q'$ \omterm{def:b2:quadratic:def}{صيغة تربيعية}
        كيفية على $E$. برهن على أنه يوجد أساس للفضاء
        $E$ متعامد ممنظم من أجل $q$ ومتعامد من أجل
        $q'$: وفيه، $q = \sum x_i^2$ و$q' = \sum \mu_i
        x_i^2$ \emph{(استعمل $q$ جداءً سلّميًا وطبّق
        المبرهنة الطيفية على التشاكل الذاتي الممثِّل
        للصيغة $q'$)}.
  \item الصيغة المصفوفية: من أجل $A$ معرَّفة موجبة و$B$
        متماثلة، توجد مصفوفة $P$ قابلة للقلب تحقق
        $P^{\mathsf T}AP = I$ و$P^{\mathsf T}BP =
        \operatorname{diag}(\mu_1, \dots, \mu_n)$، حيث
        الأعداد $\mu_i$ هي جذور $\det(B - \mu A) = 0$.
  \item شغّله كاملًا من أجل
        \[
        A = \begin{pmatrix} 2 & 1\\ 1 & 1 \end{pmatrix},
        \qquad
        B = \begin{pmatrix} 0 & 1\\ 1 & 0 \end{pmatrix} :
        \]
        أي \omterm{def:b2:reduction:eigen}{القيم الذاتية} المعمَّمة $\mu_\pm$، والمتجهات
        التي تقطّر الصيغتين معًا.
  \item بيّن أنه لا يمكن إسقاط شرط التعريف الموجب: فمن أجل
        \[
        A = \begin{pmatrix} 1 & 0\\ 0 & -1 \end{pmatrix},
        \qquad
        B = \begin{pmatrix} 0 & 1\\ 1 & 0 \end{pmatrix},
        \]
        لا يقطّر أي أساس الصيغتين معًا \emph{(فلو قطّرت $P$
        كلتيهما، لانحلّ $\det(B - \mu A)$ بجذور
        حقيقية؛ فاحسبه)}.
  \item بيّن أن الأعداد $\mu_i$ في السؤال 21 هي
        \omterm{def:b2:reduction:eigen}{القيم الذاتية} للمصفوفة $A^{-1}B$، وأن $A^{-1}B$، وإن لم تكن
        متماثلة عمومًا، تكون دائمًا قابلة للتقطير بقيم ذاتية
        حقيقية \emph{(قارِن بالمصفوفة $\sqrt A$)}.
  \item تركيب. بجملة واحدة لكل بند: (1) المبرهنة الواحدة
        التي اتّكأ عليها كل جزء؛ (2) وأي نتائج الأجزاء
        الأول--الثالث تصمد من أجل المصفوفات \emph{شبه} المعرَّفة
        الموجبة، وأيها يحتاج التعريف الموجب؛ (3) والجملة
        الفيزيائية التي تقطّر الأسئلة 20--22 تذبذباتها
        الصغيرة (بالطاقتين الحركية والكامنة بوصفهما
        الصيغتين)، وماذا تعني الأعداد $\mu_i$ هناك.
\end{enumerate}
\end{problem}
