\chapter{الصيغ الهرميتية}\label{ch:b2:hermitian}

للفضاءات المتجهية العقدية هندستها الخاصة بالجداء السلّمي، بلمسة
واحدة: خطية في متغيّر، و\emph{مرافقة} الخطية في
الآخر. والثمن الذي نجنيه بقبول اللمسة نظريةٌ طيفية
أنظف من الحقيقية --- فللتشاكلات الذاتية الهرميتية قيم ذاتية
حقيقية، وللوحدوية قيم ذاتية طويلتها واحد، وكلاهما
يتقطّر في أسس متعامدة ممنظمة. ويشغّل هذا الفصل القصير
برنامجَ \cref{ch:b2:quadratic} الإقليدي على $\C$.

\section{الجداءات السلّمية الهرميتية}

\begin{definition}\label{def:b2:hermitian:def}
\emph{الجداء السلّمي الهرميتي}\index{جداء سلّمي هرميتي} على
فضاء متجهي عقدي $E$ هو تطبيق $\langle\cdot,\cdot\rangle
\colon E \times E \to \C$ خطي في المتغيّر الثاني،
ومتماثل بالمرافقة ($\langle y, x\rangle = \conj{\langle x,
y\rangle}$ --- ومن ثم مرافق الخطية في المتغيّر الأول)، و
معرَّف موجب ($\langle x, x\rangle > 0$ من أجل $x \neq 0$). والمثال
القياسي على $\C^n$:
\[
\langle x, y \rangle = \sum_{i=1}^{n} \conj{x_i}\, y_i ;
\]
وعلى الدوال المتصلة، $\langle f, g\rangle = \int_a^b \conj
f\,g$. \omterm{def:b2:nvs:norm}{والمعيار}: $\norm x = \sqrt{\langle x,x\rangle}$؛ ويُسمى الفضاء
العقدي المنتهي البعد المزوَّد بذلك \emph{فضاءً
هرميتيًا}\index{فضاء هرميتي}.
\end{definition}

\begin{example}[حسابات أولى]\label{ex:b2:hermitian:firstcomp}
في $\C^2$، نأخذ $x = (1+\iu,\ 2-\iu)$ و$y = (\iu,\ 1)$.
عندئذٍ
\[
\norm x^2 = \abs{1+\iu}^2 + \abs{2-\iu}^2 = 2 + 5 = 7,
\qquad
\norm y^2 = 1 + 1 = 2,
\]
وبمرافقة الوسيط الأول،
\[
\langle x, y\rangle
= \conj{(1+\iu)}\,\iu + \conj{(2-\iu)}\cdot1
= (1-\iu)\iu + (2+\iu)
= (\iu + 1) + (2 + \iu) = 3 + 2\iu .
\]
وتتحقق متراجحة كوشي--شوارتز: $\abs{\langle x, y\rangle}^2 = 9 +
4 = 13 \leq 14 = \norm x^2\norm y^2$ --- وهي قريبة من المساواة،
لأن $x$ قريبة من مضاعف للمتجهة $y$. ولاحظ أيضًا
$\langle y, x\rangle = \conj{3 + 2\iu} = 3 - 2\iu$:
أي التماثل بالمرافقة في العمل، والسبب في أن $\langle x,
x\rangle$ حقيقي دائمًا.
\end{example}

\begin{theorem}[كوشي--شوارتز، العقدية]\label{thm:b2:hermitian:cs}
$\abs{\langle x, y\rangle} \leq \norm x \norm y$، مع المساواة إذا وفقط إذا كانت
$x, y$ مرتبطتين خطيًا؛ و$\norm\cdot$ \omterm{def:b2:nvs:norm}{معيار}. زيادةً على ذلك،
توجد أسس متعامدة ممنظمة (فغرام--شميت يجري حرفيًا)، مع
\[
x = \sum_i \langle e_i, x\rangle\, e_i,
\qquad
\norm x^2 = \sum_i \abs{\langle e_i, x\rangle}^2 .
\]
\end{theorem}

\begin{proof}
من أجل $y \neq 0$ و$t \in \C$: $0 \leq \norm{x - ty}^2 = \norm x^2
- 2\Re\bigl(\conj t\langle y, x\rangle\bigr) + \abs t^2\norm y^2$.
ونختار $t = \frac{\langle y, x\rangle}{\norm y^2}$:
\[
0 \leq \norm x^2 - \frac{\abs{\langle y, x\rangle}^2}{\norm y^2},
\]
وهذه هي المتراجحة؛ وتفرض المساواة $x = ty$. ومتراجحة
المثلث، كاملةً:
\[
\norm{x + y}^2 = \norm x^2 + 2\,\Re\langle x, y\rangle
+ \norm y^2
\leq \norm x^2 + 2\,\abs{\langle x, y\rangle} + \norm y^2
\leq \bigl(\norm x + \norm y\bigr)^2 ,
\]
باستعمال $\Re z \leq \abs z$ ثم كوشي--شوارتز؛ والتجانس
والفصل مباشران، ومن ثم فإن $\norm\cdot$ \omterm{def:b2:nvs:norm}{معيار}.
وغرام--شميت: كما في الحالة الحقيقية، مع وضع المرافقات
حسب التعريف (ولاحظ الاصطلاح: جداءاتنا
مرافقة الخطية في الخانة \emph{الأولى}، ومن ثم فالإحداثيات
$\langle e_i, x\rangle$؛ ويشغّل المثال التالي الخوارزمية
كاملةً مرة واحدة).
\end{proof}

\begin{example}[غرام--شميت العقدي، مشغَّلًا كاملًا]\label{ex:b2:hermitian:gramschmidt}
لنُنظّم تعامديًا الأساس $v_1 = (1, \iu)$، $v_2 = (0, 1)$ من
$\C^2$. المتجهة الأولى: $\norm{v_1}^2 = \abs1^2 + \abs\iu^2 = 2$،
ومنه $e_1 = \frac{1}{\sqrt2}(1, \iu)$. ونُسقط $v_2$ --- مع
\omterm{def:b2:hermitian:adjoint}{المرافق} في الخانة الأولى:
\[
\langle e_1, v_2\rangle
= \frac{1}{\sqrt2}\bigl(\conj{1}\cdot0 +
\conj{\iu}\cdot1\bigr)
= \frac{-\iu}{\sqrt2} ,
\qquad
v_2 - \langle e_1, v_2\rangle e_1
= (0,1) + \frac{\iu}{2}\,(1, \iu)
= \Bigl(\frac\iu2,\ \frac12\Bigr) .
\]
ومعيارها $\sqrt{\frac14 + \frac14} = \frac{1}{\sqrt2}$:
$e_2 = \frac{1}{\sqrt2}(\iu, 1)$. وللتحقق: $\langle e_1,
e_2\rangle = \frac12(\conj1\cdot\iu + \conj\iu\cdot1) =
\frac12(\iu - \iu) = 0$. وإحداثيات $v_2$ في الأساس الجديد:
$v_2 = \langle e_1, v_2\rangle e_1 + \langle e_2, v_2\rangle
e_2$ مع $\langle e_2, v_2\rangle = \frac{1}{\sqrt2}$ ---
وانتبه للترتيب: فالكتابة $\langle v_2, e_1\rangle$ كانت ستعطي المعامل
\emph{\omterm{def:b2:hermitian:adjoint}{المرافق}}. والفكرة الختامية: الخوارزمية هي الإقليدية
حرفيًا؛ والفخّ الوحيد هو موضع
المرافقة، ويستعمل حساب $\norm{v_2 -
\text{الإسقاط}}^2 > 0$ الإيجابيةَ صامتًا --- وهي البديهية
التي تجعل الهندسة كلها تعمل.
\end{example}

\section{المرافق والتشاكلات الذاتية الهرميتية والوحدوية}

\begin{definition}\label{def:b2:hermitian:adjoint}
يُعرَّف \emph{المرافق} $u^*$ للتشاكل $u \in \mathcal{L}(E)$ بالعلاقة
$\langle u^*(x), y\rangle = \langle x, u(y)\rangle$؛ وفي أساس
متعامد ممنظم، $\operatorname{Mat}(u^*) = \conj{A}^{\mathsf T} =:
A^{\dagger}$ (وهو \omterm{def:b2:linalg:transpose}{المنقول} المرافق) --- فعلًا، إذا كانت $B =
(b_{ij})$ مصفوفة $u^*$ في الأساس المتعامد الممنظم
$(e_i)$، فإن $b_{ij} = \langle e_i, u^*(e_j)\rangle$، وتعطي
المتطابقة المعرِّفة
\[
\conj{b_{ij}}
= \langle u^*(e_j), e_i\rangle
= \langle e_j, u(e_i)\rangle = a_{ji} ,
\qquad\text{ومنه}\qquad
B = \conj{A}^{\mathsf T} .
\] ويكون $u$
\emph{هرميتيًا}\index{تشاكل ذاتي هرميتي} إذا كان $u^* = u$
($A^\dagger = A$)، و\emph{وحدويًا}\index{زمرة وحدوية} إذا كان $u^*u =
\mathrm{id}$ ($A^\dagger A = I$: أي الزمرة $U(n)$)، و\emph{\omterm{def:b2:funcseq:series}{ناظميًا}}
إذا كان $u^*u = uu^*$.
\end{definition}

\begin{proposition}\label{prop:b2:hermitian:eigenvalues}
\omterm{def:b2:reduction:eigen}{القيم الذاتية} \omterm{def:b2:hermitian:adjoint}{للتشاكل الذاتي الهرميتي} \emph{حقيقية}؛
\omterm{def:b2:reduction:eigen}{والقيم الذاتية} للتشاكل الوحدوي طويلتها \emph{تساوي $1$}؛ وفي
الحالتين تكون \omterm{def:b2:reduction:eigen}{الفضاءات الذاتية} الموافقة لقيم ذاتية مختلفة متعامدة.
\end{proposition}

\begin{proof}
هرميتي، $u(x) = \lambda x$، $x \neq 0$:
\[
\lambda \norm x^2 = \langle x, u(x)\rangle = \langle u(x), x\rangle
= \conj{\langle x, u(x)\rangle} = \conj\lambda\,\norm x^2 ,
\]
ومنه $\lambda \in \R$. ووحدوي: $\norm{u(x)} = \norm x$ (من $u^*u =
\mathrm{id}$)، ومنه $\abs\lambda\norm x = \norm x$. والتعامد
(في الحالة الهرميتية): $\lambda\langle x, y\rangle = \langle u(x),
y\rangle = \langle x, u(y)\rangle = \mu\langle x, y\rangle$ من أجل
متجهتين ذاتيتين بقيمتين حقيقيتين $\lambda \neq \mu$. والحالة الوحدوية،
كاملةً: من أجل $u(x) = \lambda x$، $u(y) = \mu y$ مع $\lambda \neq
\mu$ (وكلتاهما طويلتها واحد)،
\[
\langle x, y\rangle = \langle u(x), u(y)\rangle
= \conj\lambda\mu\,\langle x, y\rangle ,
\]
و$\conj\lambda\mu = \frac{\mu}{\lambda} \neq 1$: فالعامل
لا يساوي $1$، ومنه $\langle x, y\rangle = 0$.
\end{proof}

\begin{example}[مصفوفة ضدّ هرميتية، مقطَّرة]\label{ex:b2:hermitian:skewexample}
$A = \begin{pmatrix} 0 & -2\\ 2 & 0\end{pmatrix}$ تحقق
$A^\dagger = A^{\mathsf T} = -A$: فهي ضدّ هرميتية (وهي أيضًا حقيقية
ضدّ متماثلة --- وعلى $\R$ ليست لها أي \omterm{def:b2:reduction:eigen}{قيمة ذاتية} إطلاقًا).
\omterm{def:b2:reduction:charpoly}{وكثير الحدود المميز} $X^2 + 4$: فالقيمتان الذاتيتان $\pm2\iu$،
وهما تخيليتان محضًا، كما تتنبأ \cref{exo:b2:hermitian:9}
عمومًا. والمتجهتان الذاتيتان: يعطي $(A - 2\iu I)v = 0$ أن $v_1 =
\frac{1}{\sqrt2}(1, \iu)$، و$v_2 = \frac{1}{\sqrt2}(1,
-\iu)$ من أجل $-2\iu$؛ وهما متعامدتان:
\[
\langle v_1, v_2\rangle
= \tfrac12\bigl(\conj{1}\cdot1 +
\conj{\iu}\cdot(-\iu)\bigr)
= \tfrac12(1 - 1) = 0 .
\]
ومنه $A = U\operatorname{diag}(2\iu, -2\iu)\,U^\dagger$ مع $U =
(v_1\ v_2)$ وحدوية. والفكرة الختامية: $H = -\iu A =
\begin{pmatrix} 0 & 2\iu\\ -2\iu & 0\end{pmatrix}$ هرميتية
\omterm{def:b2:reduction:eigen}{بالطيف} الحقيقي $\{\pm2\}$ وبالمتجهات الذاتية
\emph{نفسها} --- أي التقابل $u \mapsto \iu u$ بين
التشاكلات الهرميتية وضدّ الهرميتية
(\cref{exo:b2:hermitian:9})، مرئيًا مصفوفةً مصفوفة؛ وعلى $\R$
تكون $A$ نفسها دورانًا مع تغيير سلّم بلا أي \omterm{def:b2:reduction:eigen}{متجهة ذاتية}
إطلاقًا، ولا يكشف صورتها الطبيعية إلا الانتقال إلى $\C$.
\end{example}

\begin{theorem}[المبرهنة الطيفية الهرميتية]\label{thm:b2:hermitian:spectral}
لكل \omterm{def:b2:hermitian:adjoint}{تشاكل ذاتي هرميتي} \omterm{def:b2:hermitian:def}{لفضاء هرميتي} أساسٌ متعامد ممنظم
من \omterm{def:b2:reduction:eigen}{المتجهات الذاتية} (بقيم ذاتية حقيقية): أي إن $A^\dagger = A$
يستلزم $A = U D U^{\dagger}$ مع $U \in U(n)$ و$D$ قطرية
حقيقية.\index{مبرهنة طيفية!هرميتية}
\end{theorem}

\begin{proof}
على $\C$، ينحلّ \omterm{def:b2:reduction:charpoly}{كثير الحدود المميز}: فتوجد \omterm{def:b2:reduction:eigen}{متجهة ذاتية}
$e_1$ (\cref{ch:b2:reduction}) --- بلا أي حجة تراص،
وهي إحدى فوائد $\C$. ونُنظّمها. ومتممتها
المتعامدة $F = e_1^\perp$ مستقرة: فمن أجل $x \perp e_1$،
\[
\langle e_1, u(x)\rangle = \langle u(e_1), x\rangle
= \lambda_1\langle e_1, x\rangle = 0
\]
(لأن $\lambda_1$ حقيقي). والمقصور هرميتي؛ فنستعمل التراجع على
البعد ونتلاصق. وبالتفصيل: المقصور $u|_F$
تشاكل ذاتي \omterm{def:b2:hermitian:def}{للفضاء الهرميتي} $F$ (وبعده $n - 1$)
مع $\langle u|_F(x), y\rangle = \langle x, u|_F(y)\rangle$
موروث من $u$؛ وتعطي فرضية التراجع أساسًا
متعامدًا ممنظمًا $(e_2, \dots, e_n)$ للفضاء $F$ من \omterm{def:b2:reduction:eigen}{المتجهات الذاتية}،
وتكون $(e_1, e_2, \dots, e_n)$ متعامدة ممنظمة في $E$ (لأن $e_1 \perp
F$) ومكوَّنة من متجهات ذاتية للتشاكل $u$. والترجمة المصفوفية:
أعمدة $U$ هي المتجهات $e_i$، ويعبّر $U^\dagger U = I$ عن
تعامدها وتنظيمها، ويجمع $AU = UD$ معادلات \omterm{def:b2:reduction:eigen}{القيم
الذاتية}، ومنه $A = UDU^\dagger$ مع $D$ قطرية حقيقية
(\cref{prop:b2:hermitian:eigenvalues}).
\end{proof}

\begin{example}\label{ex:b2:hermitian:pauli}
$A = \begin{pmatrix} 0 & -\iu\\ \iu & 0\end{pmatrix}$ هرميتية
(لأن $A^\dagger = A$): \omterm{def:b2:reduction:eigen}{والقيم الذاتية} من $\chi_A = X^2 - 1$: $\pm 1$
(حقيقية، كما وُعد)، بمتجهتين ذاتيتين متعامدتين ممنظمتين
$\frac{1}{\sqrt2}(1, \iu)^{\mathsf T}$ و$\frac{1}{\sqrt2}(1,
-\iu)^{\mathsf T}$. (ويعرف الفيزيائيون $A$ بمصفوفة باولي؛ و
حقيقيةُ الأطياف الهرميتية هي السبب في نمذجة المراقَبات الكمومية
بمؤثرات هرميتية.)
\end{example}

\begin{example}[مصفوفة هرميتية معرَّفة موجبة، منفَّذة]\label{ex:b2:hermitian:pdrun}
$A = \begin{pmatrix} 2 & 1-\iu\\ 1+\iu & 3\end{pmatrix}$:
هرميتية، لأن القطر حقيقي والعناصر خارج القطر
مترافقة. \omterm{def:b2:reduction:charpoly}{وكثير الحدود المميز}:
\[
(2-\lambda)(3-\lambda) - \abs{1-\iu}^2
= \lambda^2 - 5\lambda + 4
= (\lambda - 1)(\lambda - 4) :
\]
\omterm{def:b2:reduction:eigen}{والطيف} $\{1, 4\}$، وهو حقيقي وموجب --- ومن ثم فإن $A$ معرَّفة
موجبة. \omterm{def:b2:reduction:eigen}{والمتجهات الذاتية}: من أجل $\lambda = 4$، تعطي الجملة $(A -
4I)v = 0$ أن $v_4 = (1 - \iu,\ 2)$ (وللتحقق من السطر الثاني:
$(1+\iu)(1-\iu) - 2 = 0$)؛ ومن أجل $\lambda = 1$، $v_1 = (1 - \iu,\
-1)$. والتعامد، مع \omterm{def:b2:hermitian:adjoint}{المرافق} في الخانة الأولى:
\[
\langle v_4, v_1\rangle
= \conj{(1-\iu)}\,(1-\iu) + \conj{2}\,(-1)
= 2 - 2 = 0 . \checkmark
\]
والتنظيم ($\norm{v_4}^2 = 2 + 4 = 6$، $\norm{v_1}^2 = 2 + 1
= 3$) يعطي المصفوفة الوحدوية $U = \bigl(\frac{v_4}{\sqrt6}\
\frac{v_1}{\sqrt3}\bigr)$ مع $A =
U\operatorname{diag}(4,1)U^\dagger$. والفكرة الختامية:
قراءة رايلي مباشرة --- فعلى كرة الوحدة في
$\C^2$، تجري $\langle x, Ax\rangle$ على $\intcc{1}{4}$،
وتُبلغ عند المتجهتين الذاتيتين؛ وهذه هي بذرة $n = 2$
لنظرية كوران--فيشر المبنية في مسألة نهاية الأسبوع.
وللتحقق من أن الصيغة حقيقية خارج \omterm{def:b2:reduction:eigen}{المتجهات الذاتية} أيضًا: عند
$x = (1, \iu)$،
\[
Ax = \bigl(2 + (1-\iu)\iu,\ (1+\iu) + 3\iu\bigr)
= (3 + \iu,\ 1 + 4\iu),
\]
\[
\langle x, Ax\rangle = \conj{1}\,(3+\iu) +
\conj{\iu}\,(1+4\iu)
= (3 + \iu) + (-\iu)(1 + 4\iu) = 3 + \iu - \iu + 4 = 7 \in
\R ,
\]
كما تضمن آلية برهان \cref{prop:b2:hermitian:eigenvalues}
(التماثل بالمرافقة في مقابل $A^\dagger = A$) من أجل
كل $x$.
\end{example}

\begin{example}[مصفوفة وحدوية مقطَّرة]\label{ex:b2:hermitian:unitary}
$U = \frac{1}{\sqrt2}\begin{pmatrix} 1 & \iu\\ \iu & 1
\end{pmatrix}$ (وهي وحدوية لأن \cref{exo:b2:hermitian:2}). و
كثير حدودها المميز $\bigl(X - \frac{1}{\sqrt2}\bigr)^2
+ \frac12$، بجذرين
\[
\lambda_\pm = \frac{1 \pm \iu}{\sqrt2} = \eu^{\pm\iu\pi/4},
\]
طويلتهما واحد كما وعدت \cref{prop:b2:hermitian:eigenvalues}، و
بمتجهتين ذاتيتين متعامدتين ممنظمتين $\frac{1}{\sqrt2}(1, \pm1)$. ومن ثم $U =
V\operatorname{diag}(\eu^{\iu\pi/4},
\eu^{-\iu\pi/4})V^\dagger$: ففي الأساس المناسب، تكون $U$ زوجًا من
الدورانات المستوية بزاوية $\pm\frac\pi4$ --- فليس لمصفوفة الدوران الحقيقية
أي \omterm{def:b2:reduction:eigen}{متجهة ذاتية} حقيقية، لكنها على $\C$ تنقسم إلى
عددين طويلتهما واحد. والفكرة الختامية: تعيش الأطياف الهرميتية على
المستقيم الحقيقي، والأطياف الوحدوية على دائرة الوحدة؛ وكلاهما
ظلٌّ للناظمية نفسها، ويرسل تحويل كايلي في
\cref{exo:b2:hermitian:6} إحدى الصورتين إلى الأخرى.
\end{example}

\begin{example}[تحويل كايلي، محسوبًا]\label{ex:b2:hermitian:cayleyrun}
لنشغّل \cref{exo:b2:hermitian:6} على $H = \begin{pmatrix} 0 & 1\\
1 & 0\end{pmatrix}$ (وهي هرميتية، \omterm{def:b2:reduction:eigen}{بطيف} $\{1, -1\}$،
وبمتجهتين ذاتيتين متعامدتين ممنظمتين $\frac{1}{\sqrt2}(1, \pm1)$). وفي
الأساس الذاتي يصير كل شيء عدديًا: فيرسل التحويل $\lambda
\mapsto \frac{\lambda - \iu}{\lambda + \iu}$
\[
1 \longmapsto \frac{1 - \iu}{1 + \iu} = -\iu,
\qquad
-1 \longmapsto \frac{-1 - \iu}{-1 + \iu} = \iu ,
\]
(بالضرب في مرافق المقام)، ومن ثم فإن $U = (H -
\iu I)(H + \iu I)^{-1}$ هي المصفوفة الوحدوية ذات القيمتين الذاتيتين
$\mp\iu$ على المتجهتين الذاتيتين نفسيهما:
\[
U = \frac{1}{2}\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
\begin{pmatrix} -\iu & 0\\ 0 & \iu\end{pmatrix}
\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
= \begin{pmatrix} 0 & -\iu\\ -\iu & 0\end{pmatrix} .
\]
وللتحقق: $U^\dagger U = I$، و$1 \notin \operatorname{Sp}U =
\{\pm\iu\}$، كما تَعِد النظرية. والفكرة الختامية: يُرسَل المستقيم
الحقيقي على دائرة الوحدة ناقص النقطة $1$ ---
قيمةً ذاتية قيمةً ذاتية، فتحويل كايلي هو
تطبيق موبيوس $\frac{\lambda-\iu}{\lambda+\iu}$، والمصفوفات
تتبع أطيافها فحسب.
\end{example}

\begin{remark}[مزالق شائعة]
\emph{(1) أين تقع المرافقة:} يرافق هذا الكتاب الخانة
\emph{الأولى}، ومن ثم فالإحداثيات $\langle e_i, x\rangle$
و$\langle\lambda x, y\rangle = \conj\lambda\langle x,
y\rangle$؛ وترافق نصوص كثيرة الخانة الثانية بدلًا من ذلك ---
فترجم قبل مقارنة الصيغ، وإلا انقلبت إشارات $\iu$
صامتةً. \emph{(2) الاستقطاب العقدي أقوى:}
فعلى $\C$، إذا كان $\langle x, u(x)\rangle = 0$ من أجل \emph{كل} $x$
فإن $u = 0$ (انشر $x + y$ و$x + \iu y$: فينعدم الجزآن الحقيقي
والتخيلي للمقدار $\langle x, u(y)\rangle$)؛ وعلى $\R$ يفشل هذا --- فالدوران بزاوية $\frac\pi2$ يحقق
$\langle x, u(x)\rangle = 0$ في كل مكان. ومن ثم، على $\C$ فقط،
يفرض قولُ ``$\langle x, u(x)\rangle \in \R$ من أجل كل $x$''
أن يكون $u$ \omterm{def:b2:hermitian:adjoint}{هرميتيًا}. \emph{(3) الناظمي الحقيقي ليس
\omterm{def:b2:reduction:diag}{قابلًا للتقطير}:} فمصفوفة
\cref{ex:b2:hermitian:skewexample} ناظمية لكن ليست لها أي \omterm{def:b2:reduction:eigen}{قيمة
ذاتية} حقيقية؛ والتقطير الوحدوي مبرهنة \emph{على}
$\C$، وعلى $\R$ لا نحصل إلا على اختزالات كتلية.
\emph{(4) التحقق من الوحدوية:} يعني $U^\dagger U = I$ أن تكون
\emph{الأعمدة} متعامدة ممنظمة من أجل الجداء الهرميتي ---
واختبار $UU^{\mathsf T}$، أو نسيان المرافقة في
جداءات الأعمدة، طريقان كلاسيكيان للتصديق على مصفوفة
خاطئة.
\end{remark}

\begin{example}[التقايسات هي الوحدويات بالضبط]\label{ex:b2:hermitian:isometry}
يبدو حفظ \omterm{def:b2:nvs:norm}{المعيار} أضعف من الوحدوية، لكنه على $\C$
ليس كذلك: فإذا كان $\norm{u(x)} = \norm x$ من أجل كل $x$، فإن $u^*u =
\mathrm{id}$. وفعلًا، $v = u^*u - \mathrm{id}$ هرميتي و
يحقق $\langle x, v(x)\rangle = \norm{u(x)}^2 - \norm x^2 =
0$ من أجل كل $x$؛ \omterm{def:b2:quadratic:def}{وبالاستقطاب} العقدي (المزلق (2)
أعلاه)، يكون التطبيق الذي ينعدم ``قطره'' تمامًا معدومًا:
أي $v = 0$. وبالتحديد، يجري الاستقطاب هكذا
\[
0 = \langle x + y, v(x+y)\rangle
= \langle x, v(y)\rangle + \langle y, v(x)\rangle,
\qquad
0 = \langle x + \iu y, v(x + \iu y)\rangle
= \iu\langle x, v(y)\rangle - \iu\langle y, v(x)\rangle ,
\]
ويفرض السطران معًا أن $\langle x, v(y)\rangle = 0$
من أجل كل $x, y$. والفكرة الختامية: ولهذا يمكن التحقق من
``الوحدوية'' بقياس الأطوال وحدها --- وهي صلابة سيستغلها
فصل فورييه، حيث يكون حفظ الطاقة
$\norm f_2$ (بارسفال) هو نفسه حفظ كل الجداءات
السلّمية للمعاملات.
\end{example}

\begin{remark}[آفاق داخل هذا المجلد]
تُستهلك الآلة الهرميتية المبنية هنا فورًا
تقريبًا. ففصل فورييه \emph{هو} الهندسة الهرميتية
في البعد غير المنتهي: فالأسّيات $(e_n)$ عائلة
متعامدة ممنظمة من أجل $\langle f, g\rangle =
\frac{1}{2\pi}\int\conj fg$، ومتراجحة بيسل هي
تقدير الإسقاط في \cref{thm:b2:hermitian:cs} في هذا الفصل،
وبارسفال هي مساواتها الحدّية. ويعود تحويل فورييه المنتهي
(\cref{exo:b2:hermitian:10}) كلما لزم تقطير
الالتفاف. وتوفّر مسألة نهاية الأسبوع في هذا الفصل ---
كوران--فيشر وفايل والتداخل --- استقرارَ
\omterm{def:b2:reduction:eigen}{القيم الذاتية} الذي يستدعيه فصل المعادلات التفاضلية
حين يؤكد أن الاضطرابات الصغيرة في جملة ما
تحرّك تواتراتها قليلًا فقط. ورجوعًا، كل ما هنا
هو المرآة العقدية لفصل الصيغ التربيعية: فأبقِ
المعجمين جنبًا إلى جنب ($A^{\mathsf T} \leftrightarrow
A^\dagger$، ومتعامدة $\leftrightarrow$ وحدوية، ورايلي
حقيقي في الحالتين).
\end{remark}

\begin{remark}[التشاكلات الذاتية الناظمية]
على $\C$ يكون القول النهائي: $u$ \emph{\omterm{def:b2:reduction:diag}{قابل للتقطير}
\omterm{def:b2:hermitian:adjoint}{وحدويًا} إذا وفقط إذا كان \omterm{def:b2:funcseq:series}{ناظميًا}} ($u^*u = uu^*$) ---
وهذا يغطي التشاكلات الهرميتية والوحدوية وضدّ الهرميتية دفعةً واحدة. و
البرهان تقوية لطيفة للحجة أعلاه
(\cref{exo:b2:hermitian:8}). وفي المقابل، لا تشتري الناظمية على $\R$
إلا التقطير الكتلي (بكتل دوران): فالهندسة العقدية
أبسط فعلًا.
\end{remark}

\begin{remark}[أين يُستعمَل هذا]
النظرية الطيفية الهرميتية هي رياضيات الميكانيك
الكمومي: فالمراقَبات تُنمذَج بمؤثرات هرميتية
(الأطياف الحقيقية = القيم القابلة للقياس)، ويُنمذَج التطور الزمني بمؤثرات
وحدوية (حفظ \omterm{def:b2:nvs:norm}{المعيار} = حفظ الاحتمال). وداخل
هذا الكتاب، يقوم فصل فورييه على تعامد وتنظيم
الأسّيات --- وهو قول في \omterm{def:b2:hermitian:def}{الجداء السلّمي الهرميتي} --- ويكون
تقطير المصفوفات الدائرية
(\cref{exo:b2:hermitian:10}) هو تحويل فورييه المنتهي.
وتطوّر مسألة نهاية الأسبوع حسابَ \omterm{def:b2:reduction:eigen}{القيم الذاتية}
التغايري (كوران--فيشر، وفايل، والتداخل)، وهو الخبز
اليومي للتحليل العددي والفيزياء الرياضية؛ ويمدّده مجلد السنة
الثالثة إلى المؤثرات المتراصة ذاتية \omterm{def:b2:hermitian:adjoint}{المرافق} على
فضاءات هيلبرت.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:hermitian:1}
على $\C^2$: احسب $\langle x, y\rangle$ و$\norm x$ و$\norm y$ من أجل
$x = (1, \iu)$، $y = (\iu, 1)$؛ وهل هما متعامدتان؟ وأعطِ
أساسًا متعامدًا ممنظمًا يحتوي $\frac{x}{\norm x}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:hermitian:2}
أيها هرميتية؟ وأيها وحدوية؟ وأيها ناظمية؟
\[
\begin{pmatrix} 1 & \iu\\ -\iu & 2 \end{pmatrix},
\qquad
\frac{1}{\sqrt2}\begin{pmatrix} 1 & \iu\\ \iu & 1\end{pmatrix},
\qquad
\begin{pmatrix} 0 & 1\\ 0 & 0 \end{pmatrix}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:hermitian:3}
برهن على أن المصفوفة $A \in \mathcal{M}_n(\C)$ تُكتب بكيفية وحيدة $A =
H + \iu K$ مع $H, K$ هرميتيتين \emph{(أي ``الجزآن الحقيقي
والتخيلي'' $H = \frac{A + A^\dagger}{2}$، $K = \frac{A -
A^\dagger}{2\iu}$)}، وأن $A$ ناظمية إذا وفقط إذا تبادلت $H$ و$K$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:4}
قطّر في أساس متعامد ممنظم: $A = \begin{pmatrix} 2 & \iu\\
-\iu & 2\end{pmatrix}$، واحسب $A^k$ من أجل $k \in \N$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:5}
برهن على أن $U(n)$ متراصة، وأن تطبيق \omterm{def:b2:reduction:eigen}{القيم الذاتية}
شامل: أي إن كل $\lambda$ طويلته واحد ينشأ من مصفوفة وحدوية ما.
وبرهن على أن $\det U \in \mathbb{U}$ (وهي دائرة الوحدة) من أجل $U \in
U(n)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:6}
(تحويل كايلي) لتكن $H$ هرميتية. برهن على أن $H + \iu I$
قابلة للقلب وأن $U = (H - \iu I)(H + \iu I)^{-1}$ وحدوية،
مع $1 \notin \operatorname{Sp}(U)$. \emph{(اعمل طيفيًا: فعلى
أساس ذاتي للمصفوفة $H$، يصير كل شيء عدديًا.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:7}
من أجل $A$ هرميتية معرَّفة موجبة ($\langle x, Ax\rangle > 0$ من أجل
$x \neq 0$)، برهن على أن $\operatorname{Sp}(A) \subseteq
\intoo{0}{\infty}$، وعلى أن $A = B^2$ من أجل مصفوفة هرميتية معرَّفة
موجبة $B$، وعلى أن $\det A > 0$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:hermitian:8}
(المبرهنة الطيفية للتشاكلات الذاتية الناظمية) ليكن $u$ \omterm{def:b2:funcseq:series}{ناظميًا} على
\omterm{def:b2:hermitian:def}{فضاء هرميتي}.
\begin{enumerate}
  \item برهن على أن $\norm{u(x)} = \norm{u^*(x)}$ من أجل كل $x$، واستنتج
        $\ker(u - \lambda) = \ker(u^* - \conj\lambda)$.
  \item برهن على أن \omterm{def:b2:reduction:eigen}{الفضاءات الذاتية} للتشاكل $u$ الموافقة لقيم ذاتية مختلفة
        متعامدة، وأن المتممة المتعامدة لفضاء ذاتي
        مستقرة تحت $u$.
  \item اختم بالتراجع أن $u$ \omterm{def:b2:reduction:diag}{قابل للتقطير} \omterm{def:b2:hermitian:adjoint}{وحدويًا}؛
        والعكس بالعكس.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:hermitian:9}
يكون التشاكل الذاتي \emph{ضدّ هرميتي} إذا كان $u^* = -u$. برهن
على أن قيمه الذاتية تخيلية محضًا، وعلى أن $u \mapsto \iu
u$ تقابل من التشاكلات الهرميتية إلى التشاكلات ضدّ
الهرميتية، وعلى أن التشاكلات ضدّ الهرميتية
قابلة للتقطير \omterm{def:b2:hermitian:adjoint}{وحدويًا} (\cref{exo:b2:hermitian:8}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:10}
(تحويل فورييه المنتهي) لتكن $S$ الإزاحة الدورية في
$\C^n$: $S(x_0, x_1, \dots, x_{n-1}) = (x_{n-1}, x_0, \dots,
x_{n-2})$، ولتكن $\omega = \eu^{2\iu\pi/n}$.
\begin{enumerate}
  \item بيّن أن $S$ وحدوية، وأن المتجهات $f_k =
        \frac{1}{\sqrt n}\bigl(1, \omega^k, \omega^{2k}, \dots,
        \omega^{(n-1)k}\bigr)$، مع $0 \leq k < n$، تشكّل
        أساسًا متعامدًا ممنظمًا من \omterm{def:b2:reduction:eigen}{المتجهات الذاتية}: $Sf_k = \omega^{-k}
        f_k$.
  \item استنتج أن كل مصفوفة \emph{دائرية} $C =
        \sum_{j=0}^{n-1} c_jS^j$ ناظمية، ومقطَّرة بالأساس
        نفسه، بقيم ذاتية $\widehat c(k) = \sum_j
        c_j\,\omega^{-jk}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:11}
ليكن $P$ تشاكلًا ذاتيًا متساوي القوى ($P^2 = P$) \omterm{def:b2:hermitian:def}{لفضاء هرميتي}.
برهن على أن $P$ هو الإسقاط \emph{المتعامد} على
$\operatorname{im} P$ إذا وفقط إذا كان $P^* = P$. وأعطِ مصفوفة
الإسقاط المتعامد على $\C v$ (مع $\norm v = 1$)، و
على فضاء جزئي له أساس متعامد ممنظم $(v_1, \dots, v_k)$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:hermitian:12}
(الإسقاطات الطيفية بالاستكمال) لتكن $A$ هرميتية بقيم ذاتية
\emph{مختلفة} $\lambda_1, \dots, \lambda_p$ وبتفكيك \omterm{def:b2:reduction:eigen}{الفضاءات
الذاتية} $E = \bigoplus_i E_i$. ونعرّف كثيرات حدود
لاغرانج $L_i(X) = \prod_{j\neq i}\frac{X -
\lambda_j}{\lambda_i - \lambda_j}$. برهن على أن $P_i = L_i(A)$
هو الإسقاط المتعامد على $E_i$، وأن $P_iP_j = 0$ من أجل
$i \neq j$، وأن $\sum_i P_i = I$، وأن $A = \sum_i \lambda_iP_i$
(وهو \emph{التفكيك الطيفي})؛ وعبّر عن $f(A)$ من أجل أي
كثير حدود $f$ بدلالة الإسقاطات $P_i$.
\end{exercise}

\section{مسألة: كوران--فيشر وفايل وحساب القيم الذاتية}

\begin{problem}\label{pb:b2:hermitian:1}
\omterm{def:b2:reduction:eigen}{القيم الذاتية} لمصفوفة هرميتية ليست مجرد جذور
كثير حدود: بل هي حلول \emph{مسائل تحسين}.
وتلك النظرة التغايرية --- خوارج قسمة رايلي و
\emph{مبرهنة الأصغرية العظمى عند كوران--فيشر}\index{مبرهنة
كوران--فيشر} --- تجعل \omterm{def:b2:reduction:eigen}{القيم الذاتية} قابلة للمقارنة ومستقرة و
قابلة للحساب، وتحصد هذه المسألة محاصيلها الكلاسيكية:
\emph{متراجحات فايل في الاضطراب}\index{متراجحة
فايل}، و\emph{تداخل كوشي}، ومتراجحتَي شور وكاي فان
للأثر، ورتابة الجذر التربيعي المصفوفي،
\omterm{def:b2:reduction:eigen}{وطيف} لابلاسي المتقطع. وفي كل ما يلي، $A, B,
E$ هرميتية على $E = \C^n$ بقيم ذاتية مسرودة بترتيب
متناقص $\lambda_1(A) \geq \dots \geq \lambda_n(A)$،
ويكون $R_A(x) = \frac{\langle x, Ax\rangle}{\langle x, x\rangle}$
من أجل $x \neq 0$ هو \emph{خارج قسمة رايلي}\index{خارج قسمة
رايلي}.

\medskip
\noindent\textbf{الجزء الأول --- خوارج قسمة رايلي والأصغرية العظمى.}
نثبّت أساسًا ذاتيًا متعامدًا ممنظمًا $(e_1, \dots, e_n)$، $Ae_i =
\lambda_ie_i$.
\begin{enumerate}
  \item بيّن أن $R_A(x)$ حقيقي، وأن
        \[
        \lambda_n \leq R_A(x) \leq \lambda_1
        \qquad (x \neq 0),
        \]
        وأن الحدّين مبلوغان: $\lambda_1 = \max R_A$،
        $\lambda_n = \min R_A$.
  \item بيّن أن النقاط الحرجة للمقدار $R_A$ هي بالضبط
        \omterm{def:b2:reduction:eigen}{المتجهات الذاتية} للمصفوفة $A$ \emph{(انشر $t \mapsto R_A(x +
        tv)$ عند $t = 0$ من أجل $v$ كيفية، ثم عوّض $v$ بالمقدار
        $\iu v$)}.
  \item ليكن $V_k = \operatorname{Vect}(e_1, \dots, e_k)$ و
        $W_k = \operatorname{Vect}(e_k, \dots, e_n)$. بيّن أن
        \[
        \min_{x \in V_k\setminus\{0\}} R_A(x) = \lambda_k
        = \max_{x \in W_k\setminus\{0\}} R_A(x) .
        \]
  \item برهن على \emph{مبرهنة كوران--فيشر}: من أجل $1 \leq
        k \leq n$،
        \[
        \lambda_k = \max_{\dim V = k}\;\min_{x \in
        V\setminus\{0\}} R_A(x)
        = \min_{\dim W = n-k+1}\;\max_{x \in
        W\setminus\{0\}} R_A(x)
        \]
        \emph{(من أجل أي $V$ بعده $k$: $V \cap W_k \neq
        \{0\}$ بصيغة غراسمان، ومنه $\min_V R_A \leq \lambda_k$؛
        ويبيّن السؤال 3 أن الحدّ مبلوغ)}.
  \item (الرتابة) نكتب $A \leq B$ حين تكون $B - A$
        شبه معرَّفة موجبة. استنتج من السؤال 4: أن $A \leq
        B$ يستلزم $\lambda_k(A) \leq \lambda_k(B)$ من أجل كل
        $k$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- متراجحات فايل.}
\begin{enumerate}[resume]
  \item بيّن أن الفضاءين الجزئيين $V, W \subseteq \C^n$ اللذين $\dim V
        + \dim W > n$ يتقاطعان تقاطعًا غير محايد، وعمّم:
        $\dim(V_1 \cap V_2 \cap V_3) \geq \dim V_1 + \dim V_2 +
        \dim V_3 - 2n$.
  \item برهن على \emph{متراجحة فايل}: من أجل $i + j - 1 \leq n$،
        \[
        \lambda_{i+j-1}(A + B) \leq \lambda_i(A) +
        \lambda_j(B)
        \]
        \emph{(قاطِع الفضاءات الجزئية $W_i(A)$ و$W_j(B)$ و
        $V_{i+j-1}(A+B)$ في السؤال 3 وعُدّ
        الأبعاد)}.
  \item عرّف $\vertiii{E}_2 = \max_{\norm x = 1}\norm{Ex}$
        وبيّن أن $\vertiii E_2 = \max_k\abs{\lambda_k(E)}$ من أجل
        $E$ الهرميتية. واستنتج \emph{مبرهنة فايل في
        الاضطراب}:
        \[
        \bigl|\lambda_k(A + E) - \lambda_k(A)\bigr| \leq
        \vertiii{E}_2
        \qquad (1 \leq k \leq n) :
        \]
        فكل \omterm{def:b2:reduction:eigen}{قيمة ذاتية} دالة ليبشيتزية بالثابت $1$ في
        المصفوفة.
  \item (الاضطرابات ذوات الرتبة واحد) لتكن $P$ هرميتية شبه
        معرَّفة موجبة رتبتها $1$. بيّن أن
        \[
        \lambda_k(A) \leq \lambda_k(A + P) \leq
        \lambda_{k-1}(A) \qquad (2 \leq k \leq n),
        \]
        مع $\lambda_1(A) \leq \lambda_1(A+P)$: أي إن
        \omterm{def:b2:reduction:eigen}{القيم الذاتية} الجديدة \emph{تتداخل} مع القديمة.
  \item تحقق من السؤال 8 عدديًا: $A = \begin{pmatrix} 2 &
        \iu\\ -\iu & 2\end{pmatrix}$
        (\cref{exo:b2:hermitian:4}: \omterm{def:b2:reduction:eigen}{بطيف} $\{3, 1\}$) و
        $E = \begin{pmatrix} 0 & 1\\ 1 & 0\end{pmatrix}$
        (\omterm{def:b2:reduction:eigen}{بطيف} $\{1, -1\}$): احسب \omterm{def:b2:reduction:eigen}{طيف} $A +
        E$ وطرفَي المتراجحة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- التداخل ومتراجحات
الأثر.}
\begin{enumerate}[resume]
  \item (تداخل كوشي) لتكن $B$ المصفوفة الجزئية الرئيسية الرائدة ذات القياس $(n-1)
        \times(n-1)$ من $A$. برهن على أن
        \[
        \lambda_{k+1}(A) \leq \lambda_k(B) \leq \lambda_k(A)
        \qquad (1 \leq k \leq n-1)
        \]
        \emph{(انظر إلى $\C^{n-1} \subseteq \C^n$؛ وعليه تكون $R_B$
        مقصورَ $R_A$؛ وطبّق كوران--فيشر على
        المستويين)}.
  \item كرّر: من أجل مصفوفة جزئية رئيسية $B$ بالقياس $n - m$،
        $\lambda_{k+m}(A) \leq \lambda_k(B) \leq
        \lambda_k(A)$.
  \item (شور) لتكن $d_1 \geq d_2 \geq \dots \geq d_n$ العناصر
        القطرية للمصفوفة $A$، مرتَّبة. برهن، من أجل كل $k$:
        \[
        \sum_{i=1}^{k} d_i \leq \sum_{i=1}^{k}\lambda_i(A),
        \]
        مع المساواة عند $k = n$ (وهو الأثر)
        \emph{(فالعناصر القطرية $k$ المختارة تشكّل
        مصفوفة جزئية رئيسية ذات القياس $k\times k$؛ فحُدّ أثرها
        بالسؤال 12)}.
  \item (كاي فان) برهن على أن
        \[
        \sum_{i=1}^{k}\lambda_i(A)
        = \max\Bigl\{\sum_{i=1}^{k}\langle x_i, Ax_i\rangle
        : (x_1, \dots, x_k) \text{ متعامدة ممنظمة}\Bigr\} .
        \]
  \item تحقق من السؤالين 11 و13 على
        \[
        A = \begin{pmatrix} 2 & 1 & 0\\ 1 & 2 & 1\\
        0 & 1 & 2\end{pmatrix}
        \qquad
        \bigl(\operatorname{Sp} = \{2 - \sqrt2,\ 2,\
        2+\sqrt2\}\bigr)
        \]
        في مقابل كتلتها الرائدة ذات القياس $2\times2$ (\omterm{def:b2:reduction:eigen}{بطيف} $\{1,
        3\}$) وقطرها.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- ترتيب لوفنر.} ما يزال $A \leq B$
يعني أن $B - A$ شبه معرَّفة موجبة؛ وكل المصفوفات في هذا
الجزء هرميتية.
\begin{enumerate}[resume]
  \item بيّن أن: $A \leq B$ يستلزم $a_{ii} \leq b_{ii}$ من أجل كل
        $i$، و$\operatorname{tr} A \leq \operatorname{tr} B$،
        و$C^\dagger AC \leq C^\dagger BC$ من أجل \emph{كل}
        مصفوفة عقدية $C$.
  \item بيّن أن التربيع \emph{ليس} رتيبًا: فمن أجل
        \[
        A = \begin{pmatrix} 1 & 0\\ 0 & 0\end{pmatrix},
        \qquad
        B = \begin{pmatrix} 2 & 1\\ 1 & 1\end{pmatrix},
        \]
        تحقق من أن $0 \leq A \leq B$ لكن $A^2 \not\leq B^2$.
  \item برهن على أن الجذر التربيعي \emph{رتيب}: أي إن $0 \leq
        A \leq B$ يستلزم $\sqrt A \leq \sqrt B$ \emph{(ليكن
        $\mu$ \omterm{def:b2:reduction:eigen}{قيمة ذاتية} للمصفوفة $\sqrt B - \sqrt A$ \omterm{def:b2:reduction:eigen}{بمتجهة ذاتية} واحدية
        $v$؛ واحسب $\langle v, (B - A)v\rangle =
        \mu\bigl(\langle v, \sqrt B\,v\rangle + \langle v,
        \sqrt A\,v\rangle\bigr)$ وناقش)}.
  \item برهن على أن القلب ضدّ رتيب على المصفوفات المعرَّفة
        الموجبة: أي إن $0 < A \leq B$ يستلزم $B^{-1} \leq A^{-1}$
        \emph{(وافِق بالمصفوفة $A^{-1/2}$ لتُرجع المسألة إلى $I \leq M
        \Rightarrow M^{-1} \leq I$، وهو عددي في
        أساس طيفي)}.
  \item لتكن $A, B$ معرَّفتين موجبتين. بيّن أن
        قيم $AB$ الذاتية (وهي ليست هرميتية عمومًا!)
        حقيقية وموجبة، وأن
        \[
        \lambda_{\max}(AB) \leq
        \lambda_{\max}(A)\,\lambda_{\max}(B)
        \]
        \emph{(قارِن بالمصفوفة $\sqrt A$: $AB \sim \sqrt
        A\,B\sqrt A$)}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- لابلاسي المتقطع، منفَّذًا.}
لتكن $T_n$ المصفوفة ثلاثية الأقطار ذات القياس $n \times n$ بالعدد $2$ على
القطر و$-1$ على القطرين المجاورين.
\begin{enumerate}[resume]
  \item مع $\theta_k = \frac{k\pi}{n+1}$، تحقق من أن
        المتجهات $v_k = \bigl(\sin(j\theta_k)\bigr)_{1\leq j\leq
        n}$ تحقق $T_nv_k = (2 - 2\cos\theta_k)\,v_k$
        \emph{(بمتطابقة تحويل الجداء إلى مجموع؛ وتحقق من سطرَي الحدّ
        $j = 1, n$)}. واختم:
        \[
        \operatorname{Sp}(T_n) = \Bigl\{4\sin^2
        \frac{k\pi}{2(n+1)} : 1 \leq k \leq n\Bigr\},
        \]
        وكلها بسيطة، وكلها موجبة: أي إن $T_n$ معرَّفة موجبة.
  \item (كمون) من أجل مصفوفة قطرية حقيقية $D =
        \operatorname{diag}(d_1, \dots, d_n)$، احصر
        \omterm{def:b2:reduction:eigen}{الطيف}: من أجل كل $k$،
        \[
        \lambda_k(T_n) + \min_i d_i \;\leq\;
        \lambda_k(T_n + D) \;\leq\; \lambda_k(T_n) + \max_i
        d_i .
        \]
  \item تحقق من تداخل كوشي بين $T_3$ و$T_2$
        صراحةً (بالطيفين $\{2 \pm \sqrt2, 2\}$ و$\{1,
        3\}$)، وفسّر: $T_2$ هي $T_3$ مع حذف طرف
        واحد من المسار.
  \item بيّن أن القيمتين الذاتيتين الحدّيتين تحققان، حين $n \to
        \infty$:
        \[
        \lambda_{\min}(T_n) = 4\sin^2\frac{\pi}{2(n+1)}
        \sim \frac{\pi^2}{(n+1)^2},
        \qquad
        \lambda_{\max}(T_n) \to 4 ,
        \]
        ومن ثم ينمو عدد الشرط $\kappa_n =
        \lambda_{\max}/\lambda_{\min}$ مثل
        $\frac{4(n+1)^2}{\pi^2}$: أي إن تقطيع مشتق ثانٍ على شبكة أدقّ
        فأدقّ سيئ الشرط
        بطبيعته.
  \item تركيب. بجملة واحدة لكل بند: (1) لماذا يكون التوصيف
        التغايري، لا \omterm{def:b2:reduction:charpoly}{كثير الحدود المميز}، هو ما يجعل \omterm{def:b2:reduction:eigen}{القيم
        الذاتية} مستقرة (السؤالان 8--9)؛ (2) وأي الأسئلة استعملت
        $\lambda_1 = \max R_A$ وحده وأيها احتاج الأصغرية العظمى
        كاملةً؛ (3) وماذا يضيف ترتيب لوفنر إلى
        الحكاية؛ (4) وأين تعود هذه الأدوات (في التحليل
        العددي لمصفوفات الصلابة في السؤال 24؛ وفي نظرية
        الاضطراب الكمومية؛ وفي مجلد السنة الثالثة، في مبدأ
        الأصغرية العظمى للمؤثرات المتراصة ذاتية
        \omterm{def:b2:hermitian:adjoint}{المرافق}).
\end{enumerate}
\end{problem}
