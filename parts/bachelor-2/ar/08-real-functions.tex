\chapter{دوال المتغير الحقيقي}\label{ch:b2:realfun}

قبل أن ينتقل التحليل إلى دوال الدوال (\cref{ch:b2:funcseq})
يجدر أن نعرف منظر المتغير الواحد بتفصيل أدقّ مما
اقتضته السنة الأولى: إلى أي حدّ يمكن أن تكون دالة رتيبة غير متصلة، وإلى
أي حدّ يجب أن تكون دالة محدبة منتظمة، وما الخواص الخاصة التي
تتمتع بها المشتقات (داربو). وهذه النتائج البنيوية قصيرة
وحادّة، ومحبّبة إلى الممتحِنين.

\section{الدوال الرتيبة}

\begin{theorem}[انتظام الدوال الرتيبة]\label{thm:b2:realfun:monotone}
لتكن $f \colon I \to \R$ متزايدة على فترة.
\begin{enumerate}
  \item عند كل نقطة داخلية $a$، توجد النهايتان من جهة واحدة:
        \[
        f(a^-) = \sup_{x < a} f(x) \;\leq\; f(a) \;\leq\;
        f(a^+) = \inf_{x > a} f(x) ;
        \]
        فكل عدم اتصال \emph{قفزة}.
  \item مجموعة نقاط عدم اتصال $f$ \omterm{def:b2:structures:countable}{قابلة للعد} على
        الأكثر.\index{دالة رتيبة!نقاط عدم الاتصال}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) المجموعة $\{f(x) : x < a\}$ غير خالية ومحدودة من أعلى بالمقدار $f(a)$:
وتحقق نهايتها العليا $s$ أن $f(x) \to s$ حين $x \to a^-$ (فمن أجل
$\varepsilon$، يوجد $f(x_0) > s - \varepsilon$، وتحصر الرتابة
$f(x) \in \intoc{s - \varepsilon}{s}$ من أجل $x \in
\intoo{x_0}{a}$). وبالتناظر على اليمين.

(2) نُرفق بكل نقطة عدم اتصال $a$ الفترةَ \omterm{def:b2:metric:topology}{المفتوحة} غير الخالية
$J_a = \intoo{f(a^-)}{f(a^+)}$ (وهو قفزة حقيقية). ومن أجل نقطتَي
عدم اتصال $a < b$، يكون $J_a$ و$J_b$ منفصلين: إذ $f(a^+) \leq f(c)
\leq f(b^-)$ من أجل أي $c$ بينهما. ويحتوي كل $J_a$ عددًا ناطقًا؛
وتحصل نقاط عدم الاتصال المختلفة على أعداد ناطقة مختلفة: أي غرسة
لمجموعة نقاط عدم الاتصال في $\Q$، وهي \omterm{def:b2:structures:countable}{قابلة للعد}
(\cref{prop:b2:structures:countablestable}).
\end{proof}

\begin{example}
الحدّ أمثل: نثبّت ترقيمًا $(r_n)$ للمجموعة $\Q \cap
\intoo{0}{1}$ ونضع $f(x) = \sum_{n : r_n \leq x} 2^{-n}$
(وهو تعريف \omterm{def:b2:series:summable}{بعائلة قابلة للجمع}،
\cref{def:b2:series:summable}). عندئذٍ تكون $f$ متزايدة على
$\intcc{0}{1}$ وغير متصلة بالضبط عند كل عدد ناطق من
$\intoo{0}{1}$ (بقفزة $2^{-n}$ عند $r_n$): فالدالة الرتيبة
\emph{يمكن} أن تكون غير متصلة على مجموعة كثيفة \omterm{def:b2:structures:countable}{قابلة للعد}.
\end{example}

\begin{example}[القفزات لا يمكن أن تفوق الصعود]\label{ex:b2:realfun:jumpbudget}
من أجل $f$ متزايدة على $\intcc{a}{b}$، للقفزات ميزانية:
فإذا كانت $a < c_1 < \dots < c_m < b$ نقاط عدم اتصال بقفزات
$s_i = f(c_i^+) - f(c_i^-) > 0$، فباختيار
نقاط متداخلة $a < c_1 < t_1 < c_2 < \dots$ واستعمال الرتابة على
كل قطعة،
\[
\sum_{i=1}^{m} s_i \;\leq\; f(b) - f(a) :
\]
يحدّ الصعود الكلي القفزَ الكلي. ومن ثم: من أجل
كل $k$، لا تتجاوز نقاط عدم الاتصال ذوات القفزة $\geq \frac1k$ العددَ $k\,\bigl(f(b) - f(a)\bigr)$ --- وهو تنقيح كمّي
للبند (2) في \cref{thm:b2:realfun:monotone}، لأن مجموعة نقاط عدم الاتصال
هي الاتحاد القابل للعد على $k$ لهذه المجموعات المنتهية. وفي
دالة القفزات الناطقة أعلاه، تُنفَق الميزانية بالضبط:
فمجموع القفزات $2^{-n}$ هو $1 = f(1^+) - f(0^-)$ بالمعنى الموسَّع
الواضح. فالدوال الرتيبة قد تقفز بكثافة، لكن ضمن
مخصَّص صارم.
\end{example}

\section{الدوال المحدبة}

\begin{lemma}[متراجحة الميول]\label{lem:b2:realfun:slopes}
لتكن $f$ محدبة على $I$ ولتكن $x < y < z$ في $I$. عندئذٍ
\[
\frac{f(y) - f(x)}{y - x}
\;\leq\; \frac{f(z) - f(x)}{z - x}
\;\leq\; \frac{f(z) - f(y)}{z - y} :
\]
فميول الأوتار تتزايد في الطرفين معًا.\index{دالة
محدبة!متراجحة الميول}
\end{lemma}

\begin{proof}
نكتب $y = \frac{z - y}{z - x}\,x + \frac{y - x}{z - x}\,z$: وهو
تركيب محدب، لأن المعاملين موجبان ومجموعهما
$1$. ويعطي التحدب أن
\[
f(y) \;\leq\; \frac{z-y}{z-x}\,f(x) + \frac{y-x}{z-x}\,f(z).
\]
ومن أجل المتراجحة اليسرى، نطرح $f(x)$ من الطرفين، مستعملين
$\frac{z-y}{z-x} - 1 = -\frac{y-x}{z-x}$:
\[
f(y) - f(x) \leq \frac{y - x}{z - x}\bigl(f(z) - f(x)\bigr),
\]
ثم نقسم على $y - x > 0$. ومن أجل المتراجحة اليمنى، نطرح
بدل ذلك من $f(z)$:
\[
f(z) - f(y) \geq f(z) - \frac{z-y}{z-x}f(x) -
\frac{y-x}{z-x}f(z)
= \frac{z - y}{z - x}\bigl(f(z) - f(x)\bigr),
\]
ثم نقسم على $z - y > 0$. والخطوتان المعروضتان هما المتطابقة
المركزية نفسها مقروءةً في مقابل طرف مختلف.
\end{proof}

\begin{theorem}[انتظام الدوال المحدبة]\label{thm:b2:realfun:convexreg}
لتكن $f$ محدبة على فترة $I$.
\begin{enumerate}
  \item عند كل نقطة داخلية، يكون للدالة $f$ مشتقان منتهيان من جهة
        واحدة $f'_g \leq f'_d$؛ وكلاهما دالة متزايدة
        في النقطة؛ وعلى الخصوص تكون $f$ متصلة على
        داخل $I$ (وقد لا تكون كذلك عند الطرفين).
  \item تقع $f$ فوق كل \emph{مستقيم مساند}: فمن أجل $a$ داخلية
        وأي $m \in \intcc{f'_g(a)}{f'_d(a)}$،
        \[
        f(x) \geq f(a) + m(x - a) \qquad (x \in I).
        \]
  \item (جنسن، الموزون) من أجل $x_i \in I$ وأوزان $\lambda_i
        \geq 0$، $\sum\lambda_i = 1$:
        \[
        f\Bigl(\sum_i \lambda_i x_i\Bigr) \leq \sum_i \lambda_i
        f(x_i) .\index{متراجحة جنسن}
        \]
\end{enumerate}
\end{theorem}

\begin{proof}
(1) نثبّت $a$ داخلية. وحسب \cref{lem:b2:realfun:slopes}، يكون الميل
$\tau(h) = \frac{f(a + h) - f(a)}{h}$ دالة متزايدة في
$h$ (على الجهتين، مع $\tau(h_-) \leq \tau(h_+)$ من أجل $h_- < 0 <
h_+$). ومن ثم فللمقدار $\tau$ نهاية منتهية حين $h \to 0^-$ (فهو متزايد
ومحدود من أعلى بأي ميل يميني) --- وهي $f'_g(a)$ --- وحين
$h \to 0^+$ ($f'_d(a)$)، مع $f'_g(a) \leq f'_d(a)$. ووجود
مشتقين منتهيين من جهة واحدة يفرض الاتصال عند $a$. وأما الرتابة في
النقطة: فمن أجل $a < b$ داخليتين، $f'_d(a) \leq \frac{f(b) - f(a)}{b -
a} \leq f'_g(b)$، وذلك بمتراجحة الميول من جديد.

(2) من أجل $x > a$: $\frac{f(x) - f(a)}{x - a} \geq f'_d(a) \geq m$؛
ومن أجل $x < a$: $\frac{f(a) - f(x)}{a - x} \leq f'_g(a) \leq m$. وكلاهما
يُعاد ترتيبه ليعطي المطلوب.

(3) بالتراجع على عدد النقاط تمامًا كما في مجلد السنة
الأولى (فحالة النقطتين هي التعريف) --- أو دفعةً واحدة:
نطبّق (2) عند $a = \sum\lambda_i x_i$ ونأخذ متوسط متراجحات
المستقيم المساند عند النقاط $x_i$ بالأوزان $\lambda_i$: $\sum_i
\lambda_i f(x_i) \geq f(a) + m\sum_i\lambda_i(x_i - a) = f(a)$.
\end{proof}

\begin{example}[عدم اتصال عند الطرف]
على $\intcc{0}{1}$، الدالة $f(0) = 1$، $f(x) = 0$ من أجل $x > 0$
محدبة لكنها غير متصلة عند الطرف $0$: فالبند (1)
أمثل.
\end{example}

\begin{example}[الزوايا وحزمة المستقيمات المساندة]\label{ex:b2:realfun:corner}
من أجل $f(x) = \abs x$ عند $a = 0$: يكون المشتقان من جهة واحدة
$f'_g(0) = -1$ و$f'_d(0) = +1$، ويوزّع
البند (2) في \cref{thm:b2:realfun:convexreg} مستقيمًا مساندًا من أجل
\emph{كل} ميل $m \in \intcc{-1}{1}$:
\[
\abs x \geq m\,x \qquad (x \in \R,\ -1 \leq m \leq 1),
\]
وكلٌّ منها مساواة بالضبط على نصف مستقيم أو عند $0$. وتكون الدالة
المحدبة قابلة للاشتقاق عند $a$ بالضبط حين تنهار الحزمة
إلى مستقيم واحد ($f'_g(a) = f'_d(a)$)؛ أما الزوايا فتحمل
فترةً من المماسات. وهذه الحزمة هي البذرة المنتهية البعد
\emph{للمشتق التحتي} في التحسين المحدب ---
والسبب في أن الدوال المحدبة بالغة المتانة: فحتى حيث يفشل
المشتق، تنجو الهندسة المساندة، وهي كل
ما استعمله برهان جنسن.
\end{example}

\begin{example}[متراجحة المتوسطات القوية]\label{ex:b2:realfun:powermeans}
من أجل $0 < p < q$ وأعداد موجبة $x_i$ بأوزان $\lambda_i$
مجموعها $1$، وبتطبيق جنسن على الدالة المحدبة $t \mapsto t^{q/p}$
عند النقاط $x_i^p$:
\[
\Bigl(\sum \lambda_i x_i^{p}\Bigr)^{1/p}
\leq \Bigl(\sum \lambda_i x_i^{q}\Bigr)^{1/q} :
\]
فالمتوسطات القوية تتزايد مع الأس --- وهذا يتضمن متراجحة الحسابي
والتربيعي، ويعطي، في النهاية $p \to 0$ (\cref{exo:b2:realfun:6})، متراجحةَ
الحسابي والهندسي مرة أخرى.
\end{example}

\begin{example}[الإنتروبيا العظمى]\label{ex:b2:realfun:entropy}
من أجل متجهة احتمال $(p_1, \dots, p_n)$ (موجبة،
مجموعها $1$)، تحقق الإنتروبيا $H(p) = -\sum_i p_i\ln p_i$
العلاقةَ
\[
H(p) \leq \ln n ,
\qquad\text{مع المساواة إذا وفقط إذا } p_i = \frac1n \text{ من أجل كل }
i .
\]
والبرهان بجنسن (البند (3) في \cref{thm:b2:realfun:convexreg}) مطبَّقًا على
الدالة \emph{المقعرة} $\ln$ بأوزان $p_i$ عند النقاط
$\frac{1}{p_i}$:
\[
H(p) = \sum_i p_i\ln\frac{1}{p_i}
\leq \ln\Bigl(\sum_i p_i\,\frac1{p_i}\Bigr) = \ln n ,
\]
وتفرض المساواة تساوي كل النقاط $\frac1{p_i}$ (بالتقعر
التام)، أي أن تكون $p$ منتظمة. وبكيفية مكافئة، هذا هو
\cref{exo:b2:realfun:7} مع $q$ منتظمًا. فيبلغ الارتياب أعظمه
بجهلٍ موزَّع بانتظام --- وهو المبدأ التغايري
وراء الترميز والميكانيك الإحصائي وظهورات الإنتروبيا
في \cref{ch:b2:randomvar}.
\end{example}

\begin{method}[كيف نجد الدالة المحدبة وراء متراجحة]\label{met:b2:realfun:spotting}
معظم المتراجحات الكلاسيكية جنسن بلباس؛ ولخلع
اللباس: (1) نظّم بحيث يظهر \emph{متوسط موزون}
(بأوزان موجبة مجموعها $1$ --- بالقسمة على كتلة كلية عند
الحاجة)؛ (2) انظر إلى الدالة المطبَّقة داخل المتوسط في مقابل
المطبَّقة خارجه: فالادعاء ``$f(\text{متوسط}) \leq$
متوسط $f$'' يسمّي الدالة المحدبة $f$؛ (3) صدّق التحدب
بالمشتق الثاني، وعالج المساواة بالتمامية؛ (4)
وإذا لم يظهر أي متوسط، فخذ اللوغاريتمات أولًا --- فتصير الجداءات
والقوى متوسطات، ويحمل تقعّر $\ln$
متراجحاتِ الحسابي والهندسي ويونغ وأقاربها (وتشغّل مسألة نهاية
الأسبوع في هذا الفصل الخطوات 1--4 على كل منها). وإذا لم تكشف
اللوغاريتمات نفسها أي متوسط، فحاول قراءة المتراجحة بوصفها
رتابة ميول (\cref{lem:b2:realfun:slopes}) ---
فأقوال فوق الجمعية مثل \cref{exo:b2:realfun:9} تسكن
هناك.
\end{method}

\begin{remark}[مزالق شائعة]\label{rem:b2:realfun:pitfalls}
(1) لا يُحفظ التحدب بالجداء: فالدالتان $x$ و$(x - 1)^2$
محدبتان على $\intcc{0}{2}$، لكن جداءهما $x(x-1)^2$ مشتقه
الثاني $6x - 4$، وهو سالب على
$\intco{0}{\frac23}$ --- فليست محدبة؛ ولا يُحفظ التحدب
بالتركيب دون رتابة
(\cref{exo:b2:realfun:10}). (2) ينقلب جنسن من أجل الدوال
المقعرة: فنصف المتراجحات الكلاسيكية هو النسخة
المقعرة $\ln$؛ وتطبيق الصيغة المحدبة على $\ln$ أسرع
طريق لبرهان متراجحة الحسابي والهندسي \emph{بالمقلوب}. (3) التحدب
عند النقطة الوسطى وحده لا يستلزم التحدب --- بل يلزم الاتصال (أو
مجرد كون الدالة محدودة) (\cref{exo:b2:realfun:8})؛ أما
الأمثلة المضادة المرضية فتعيش وراء بديهيات هذا الكتاب. (4)
الدالة المحدبة على فترة \emph{\omterm{def:b2:metric:topology}{مفتوحة}} متصلة،
بل ليبشيتزية محليًا (\cref{exo:b2:realfun:12})؛ وعند
الطرفين، لا شيء مجاني. (5) تخضع المشتقات لداربو لكن
لا يلزم أن تكون متصلة (\cref{ex:b2:realfun:oscillation}):
فقول ``ليس للمقدار $f'$ قفزات'' لا يعني أبدًا ``$f'$ متصلة''.
\end{remark}

\section{خاصية داربو}

\begin{theorem}[داربو]\label{thm:b2:realfun:darboux}
لتكن $f$ قابلة للاشتقاق على فترة $I$. عندئذٍ تأخذ $f'$ كل
قيمة بين أي قيمتين من قيمها --- وإن لم يلزم أن تكون $f'$
متصلة.\index{مبرهنة داربو}
\end{theorem}

\begin{proof}
ليكن $a < b$ في $I$ وليكن $v$ واقعًا تمامًا بين $f'(a)$ و$f'(b)$،
ولنفترض $f'(a) < v < f'(b)$. الدالة $g(x) = f(x) - vx$
قابلة للاشتقاق مع $g'(a) < 0 < g'(b)$: وحدّها الأدنى على
$\intcc{a}{b}$ (وهو مبلوغ: بالاتصال على متراص) ليس عند $a$
(فبُعيد $a$ مباشرةً تتناقص $g$ تحت $g(a)$) ولا عند $b$ (فقُبيل
$b$ مباشرةً تكون $g$ تحت $g(b)$): فهو داخلي، وهناك $g'(c)
= 0$، أي $f'(c) = v$. (وقد كان هذا تمرينًا بنجمة في السنة
الأولى؛ وموضعه في النظرية هنا.)
\end{proof}

\begin{example}[أي الدوال مشتقات؟]\label{ex:b2:realfun:whichderivatives}
مبرهنة داربو آلة عدم وجود. فدالة الجزء الصحيح
$\lfloor x\rfloor$ ليست مشتقًا لأي دالة على
$\R$: فهي تأخذ القيمتين $0$ و$1$ لكنها تتخطى $\frac12$ على
$\intcc{0}{1}$، وهذا ما تمنعه \cref{thm:b2:realfun:darboux} من أجل
المشتقات. والحكم نفسه يصيب كل دالة ذات قفزة
--- كدالة الإشارة وهيفيسايد وكل الدوال الدرجية --- مهما بدت
بريئة؛ و``دوالها الأصلية'' ($\abs x$ من أجل الإشارة، وهكذا)
لا توجد إلا بعيدًا عن القفزة وتنعقد هناك بزاوية.
وفي المقابل: فإن الدالة $f'$ الشديدة عدم الاتصال في
\cref{ex:b2:realfun:oscillation} \emph{هي} مشتق --- إذ إن
عدم اتصالها تذبذب، وداربو يحتمله. والحدّ
بين السلوكين هو بالضبط النتيجة أدناه في انعدام القفزات.
\end{example}

\begin{corollary}\label{cor:b2:realfun:nojumps}
ليس للمشتق أي عدم اتصال من نوع القفزة: فإذا وُجدت $f'(a^-)$ و
$f'(a^+)$، فإنهما تساويان $f'(a)$. ونقاط عدم اتصال
المشتق تكون دائمًا من نوع التذبذب (مشتق $x^2\sin\frac1x$
عند $0$، مجلد السنة الأولى).
\end{corollary}

\begin{proof}
إذا وُجدت $f'(a^+) = \lim_{x\to a^+} f'(x)$ واختلفت عن
$f'(a)$، لتخطّت $f'$ القيم الواقعة تمامًا بينهما على
جوار يميني --- وهذا يناقض داربو على الفترات
$\intcc{a}{a + h}$. (أو بديلًا: تفرض مبرهنة التزايدات المنتهية
أن $f'(a) = \lim_{h\to0^+} \frac{f(a+h)-f(a)}{h} = f'(a^+)$،
لأن نسبة الفرق قيمةٌ للمقدار $f'$ عند نقطة وسيطة.)
والأمر نفسه على اليسار.
\end{proof}

\begin{example}[المشتق المتذبذب القانوني]\label{ex:b2:realfun:oscillation}
لتكن $f(x) = x^2\sin\frac1x$ من أجل $x \neq 0$ و$f(0) = 0$. وعند
$0$: $\bigl|\frac{f(h) - f(0)}{h}\bigr| = \abs{h\sin\frac1h}
\leq \abs h \to 0$، ومن ثم فإن $f'(0) = 0$ موجود. وبعيدًا عن $0$،
\[
f'(x) = 2x\sin\frac1x - \cos\frac1x ,
\]
ويؤول حدّها الأول إلى $0$ في حين أن $\cos\frac1x$ يتذبذب
عبر $\intcc{-1}{1}$ على كل فترة $\intoo{0}{\delta}$: فالنهاية
$f'(0^+)$ غير موجودة. ومن ثم فإن $f'$ معرَّفة في كل مكان لكنها
غير متصلة عند $0$ --- وكما تتنبأ
\cref{cor:b2:realfun:nojumps} بالضبط، فإن عدم الاتصال
تذبذب لا قفزة: إذ على كل $\intoo{0}{\delta}$، ما تزال $f'$
تمسح فترةً كاملةً حول $0$. فقد تكون المشتقات جامحة، لكن
بالطريقة المتوافقة مع داربو فقط.
\end{example}

\begin{remark}[أين يُستعمَل هذا الفصل]
التحدب محرّك صناعة المتراجحات: فمسألة نهاية الأسبوع في هذا
الفصل تصنع منه متراجحات يونغ وهولدر ومينكوفسكي
وسلسلة المتوسطات القوية، وتستهلكها نظريةُ المعايير في \cref{ch:b2:nvs}
وتقديراتُ التكاملات في \cref{ch:b2:integration}؛
ويعود جنسن في الاحتمال بوصفه متراجحات العزوم في
\cref{ch:b2:randomvar}. ويعود انتظام الرتيبات في
\cref{ch:b2:integration} (فالدوال الرتيبة قابلة للمكاملة)، وفي
مجلد السنة الثالثة بوصفه قابلية اشتقاق الدوال الرتيبة في
كل مكان تقريبًا --- حيث يصير ``عدد قابل للعد من القفزات''
الخطوةَ الأولى في نظرية لوبيغ.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:realfun:1}
عيّن مجموعات نقاط عدم الاتصال وأحجام القفزات: $\lfloor x
\rfloor$؛ و$\;x - \lfloor x\rfloor$؛ و$\;\lfloor x \rfloor + \sqrt{x
- \lfloor x\rfloor}$؛ ودالة المثال الذي يلي
\cref{thm:b2:realfun:monotone} مقصورةً على الأعداد الناطقة الثنائية
$r_n$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:realfun:2}
برهن على أن الدالة المتزايدة $f \colon I \to \R$ ذات خاصية
القيم الوسطى (أي إن صورة أي فترة جزئية بها فترة) تكون
متصلة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:realfun:3}
أي مما يلي محدب على مجال تعريفه؟ $x \mapsto
x\ln x$ (مع $x > 0$)؛ $\;x \mapsto \ln(1 + \eu^x)$؛ $\;x \mapsto
\sqrt{1 + x^2}$؛ $\;x \mapsto x^3$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:4}
لتكن $f$ محدبة على $\R$ ومحدودة من أعلى. برهن على أن $f$
ثابتة. \emph{(إذا كان $f(a) \neq f(b)$، فإن متراجحة الميول
تنشر ميل الوتر غير المعدوم: فبعد النقطة ذات القيمة الأكبر،
تنمو $f$ نموًا خطيًا على الأقل --- وهذا يناقض كونها محدودة.
وعالج إشارتَي الميل معًا.)} استنتج أن الدالة المحدبة على
$\R$ ولها مقارب عند الطرفين تكون أفينية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:5}
لتكن $f$ قابلة للاشتقاق على $I$ مع $f'$ رتيبة. برهن على أن
$f'$ متصلة \emph{(اجمع بين
\cref{thm:b2:realfun:monotone} و
\cref{cor:b2:realfun:nojumps})}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:6}
(المتوسط الهندسي بوصفه نهاية) من أجل أعداد موجبة $x_i$ وأوزان
$\lambda_i$ مجموعها $1$، برهن على أن
\[
\lim_{p \to 0^+} \Bigl(\sum_i \lambda_i x_i^p\Bigr)^{1/p}
= \prod_i x_i^{\lambda_i} ,
\]
عبر $x_i^p = \eu^{p\ln x_i} = 1 + p\ln x_i + O(p^2)$، واستنتج
متراجحة الحسابي والهندسي الموزونة من
\cref{ex:b2:realfun:powermeans}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:7}
(متراجحة الإنتروبيا) باستعمال التحدب التام للدالة $t \mapsto t\ln t$،
برهن على أنه من أجل أعداد موجبة $p_i, q_i$ مع $\sum p_i = \sum q_i = 1$:
\[
\sum_i p_i \ln\frac{p_i}{q_i} \geq 0 ,
\]
مع المساواة إذا وفقط إذا $p = q$. \emph{(اكتب الطرف الأيسر على صورة $\sum q_i\,
\varphi\bigl(\frac{p_i}{q_i}\bigr)$ مع $\varphi(t) = t\ln t$ ثم
طبّق جنسن بالأوزان $q_i$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:realfun:8}
(التحدب عند النقطة الوسطى) تكون $f \colon I \to \R$ \emph{محدبة عند النقطة الوسطى}
إذا كان $f\bigl(\frac{x+y}{2}\bigr) \leq \frac{f(x) + f(y)}{2}$
دائمًا. برهن على أن الدالة \emph{المتصلة} المحدبة عند النقطة الوسطى تكون
محدبة. \emph{(أثبت متراجحة التحدب من أجل الأوزان
الثنائية $\frac{k}{2^m}$ بالتراجع على $m$، ثم انتقل إلى النهاية
مستعملًا الكثافة والاتصال.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:realfun:9}
لتكن $f$ محدبة على $\intco{0}{+\infty}$ مع $f(0) \leq 0$. برهن على
أن $x \mapsto \frac{f(x)}{x}$ متزايدة على
$\intoo{0}{+\infty}$، واستنتج أنه من أجل $f$ محدبة مع $f(0) =
0$: $f(x + y) \geq f(x) + f(y)$ من أجل $x, y \geq 0$
(فوق الجمعية).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:realfun:10}
لتكن $f$ محدبة على $I$ ولتكن $g$ محدبة \emph{متزايدة} على
فترة تحتوي $f(I)$. برهن على أن $g \circ f$ محدبة، و
بيّن بمثال مضاد أنه لا يمكن إسقاط رتابة $g$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:11}
(هيرميت--هادامار) لتكن $f$ محدبة ومتصلة على
$\intcc{a}{b}$. برهن على أن
\[
f\Bigl(\frac{a+b}{2}\Bigr) \;\leq\; \frac{1}{b -
a}\int_a^b f(t)\,\dd t \;\leq\; \frac{f(a) + f(b)}{2} .
\]
\emph{(اليسار: كامِل مستقيمًا مساندًا عند النقطة الوسطى. واليمين:
حُدّ $f$ بالوتر.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:realfun:12}
برهن على أن الدالة المحدبة على فترة \emph{\omterm{def:b2:metric:topology}{مفتوحة}} $I$ تكون
ليبشيتزية محليًا: فمن أجل كل قطعة $\intcc{a}{b} \subseteq I$
وهامش $\delta > 0$ مع $\intcc{a - \delta}{b + \delta}
\subseteq I$، يكون مقصور $f$ على $\intcc{a}{b}$
\omterm{def:b2:metric:continuity}{ليبشيتزيًا} بالثابت
$\max\Bigl(\bigl|\frac{f(a) - f(a - \delta)}{\delta}\bigr|,
\bigl|\frac{f(b + \delta) - f(b)}{\delta}\bigr|\Bigr)$
\emph{(احصر كل ميل وتر بين هذين بمتراجحة الميول)}.
\end{exercise}

\section{مسألة: صندوق أدوات التحدب}

تعريف واحد --- الوتر فوق المنحني --- يولّد صندوق
أدوات المتراجحات الكلاسيكية كله. وتبني مسألة نهاية الأسبوع هذه
الصندوقَ بترتيب منطقي: محكّات التحدب وجنسن التام،
ثم يونغ وهولدر ومينكوفسكي (وهي شهادات ميلاد
المعايير $p$)، وسلسلة المتوسطات القوية كاملةً من الأصغر إلى
الأكبر، وربحان تاجيان --- متراجحة كارلمان، وهولدر
مقروءًا بوصفه ثنوية. وكل شيء مبرهَن عليه؛ ولا شيء
مستورد.

\begin{problem}[{مسألة نهاية الأسبوع --- يونغ وهولدر ومينكوفسكي،
وسلسلة المتوسطات القوية}]
\label{pb:b2:realfun:1}
في كل ما يلي، $p, q > 1$ \emph{أسّان مترافقان}: $\frac1p +
\frac1q = 1$؛ والمتجهات $a = (a_1, \dots, a_n) \in \R^n$؛
والأوزان $\lambda_i > 0$ تحقق $\sum_i\lambda_i = 1$.

\medskip
\noindent\textbf{الجزء الأول --- المحكّات وجنسن التام.}
\begin{enumerate}
  \item لتكن $f$ قابلة للاشتقاق على فترة $I$. برهن على أن
        $f$ محدبة إذا وفقط إذا كانت $f'$ متزايدة
        \emph{(اتجاه بالانتقال إلى النهاية في
        متراجحة الميول \cref{lem:b2:realfun:slopes}؛ والاتجاه
        الآخر بمبرهنة التزايدات المنتهية)}. واستنتج محك
        $C^2$: $f'' \geq 0$.
  \item لنفترض $f''> 0$ على $I$. برهن على أن $f$
        \emph{محدبة تمامًا} (بمتراجحة تامة من أجل $x \neq y$
        و$\lambda \in \intoo01$)، وأن الدالة المحدبة تمامًا
        تحقق متراجحة جنسن
        (البند (3) في \cref{thm:b2:realfun:convexreg}) بالمساواة
        \emph{فقط} حين تتطابق كل القيم $x_i$.
  \item صدّق المواد الخام لصندوق الأدوات: $-\ln$ محدبة تمامًا
        على $\intoo{0}{+\infty}$؛ و$t \mapsto t^r$
        محدبة تمامًا هناك من أجل $r > 1$ ومقعرة تمامًا
        من أجل $0 < r < 1$؛ و$\exp$ محدبة تمامًا على $\R$.
  \item (متراجحة يونغ) من أجل $a, b \geq 0$، برهن على أن
        \[
        ab \;\leq\; \frac{a^p}{p} + \frac{b^q}{q},
        \]
        مع المساواة إذا وفقط إذا $a^p = b^q$ \emph{(طبّق
        تقعّر $\ln$ على النقطتين $a^p, b^q$
        بالأوزان $\frac1p, \frac1q$)}.
  \item أعد استنباط متراجحة الحسابي والهندسي الموزونة في سطر واحد من
        تقعّر $\ln$:
        \[
        \prod_i x_i^{\lambda_i} \leq \sum_i\lambda_ix_i \qquad
        (x_i > 0),
        \]
        مع حالة المساواة؛ وقارِن بطريق النهاية في
        \cref{exo:b2:realfun:6}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- هولدر ومينكوفسكي.} نكتب
$\norm{a}_p = \bigl(\sum_i \abs{a_i}^p\bigr)^{1/p}$ و
$\norm{a}_\infty = \max_i\abs{a_i}$.
\begin{enumerate}[resume]
  \item (هولدر) برهن على أن
        \[
        \sum_{i=1}^{n}\abs{a_ib_i} \;\leq\;
        \norm a_p\,\norm b_q ,
        \]
        مع المساواة إذا وفقط إذا تناسبت المتجهتان $(\abs{a_i}^p)$ و
        $(\abs{b_i}^q)$ \emph{(نظّم
        $\norm a_p = \norm b_q = 1$ وطبّق يونغ حدًّا حدًّا)}.
  \item عيّن الحالات الخاصة: $p = q = 2$
        (كوشي--شوارتز)، والزوج الحدّي $(p, q) = (1,
        \infty)$: صُغ وبرهن على $\sum\abs{a_ib_i} \leq
        \norm a_1\norm b_\infty$.
  \item (مينكوفسكي) من أجل $p \geq 1$، برهن على أن
        \[
        \norm{a + b}_p \leq \norm a_p + \norm b_p
        \]
        \emph{(اكتب $\abs{a_i + b_i}^p \leq \abs{a_i +
        b_i}^{p-1}(\abs{a_i} + \abs{b_i})$ وطبّق هولدر
        على كل جداء)}. واختم: $\norm\cdot_p$ \omterm{def:b2:nvs:norm}{معيار}
        على $\R^n$ من أجل كل $p \in \intco{1}{+\infty}$،
        فيكتمل مشهد \cref{ch:b2:nvs}.
  \item النسخ التكاملية: من أجل $f, g$ متصلتين على
        $\intcc{a}{b}$، صُغ وبرهن على هولدر ومينكوفسكي
        من أجل $\norm f_p = \bigl(\int_a^b\abs
        f^p\bigr)^{1/p}$ \emph{(بالبراهين نفسها، مع الإيجابية
        التامة للتكامل من أجل مناقشة
        المساواة)}.
  \item برهن على الرتابة $\norm a_q \leq \norm a_p$ من أجل $1
        \leq p \leq q$، وعلى النهاية $\norm a_p \to
        \norm a_\infty$ حين $p \to \infty$، وعلى المقارنة
        العكسية بالثابت الأمثل:
        \[
        \norm a_p \leq n^{\frac1p - \frac1q}\,\norm a_q
        \]
        \emph{(هولدر في مقابل المتجهة الثابتة)}. وعيّن
        المتجهات المحققة لكل مساواة.
  \item (الاستكمال) من أجل $1 \leq p < r < q$ و$\theta \in
        \intoo01$ مع $\frac1r = \frac\theta p +
        \frac{1-\theta}q$، برهن على أن
        \[
        \norm a_r \leq \norm a_p^{\theta}\,
        \norm a_q^{1-\theta}
        \]
        \emph{(طبّق هولدر بالأسّين $\frac{p}{\theta
        r}$ و$\frac{q}{(1-\theta)r}$ على $\abs{a_i}^{\theta
        r}\abs{a_i}^{(1-\theta)r}$)}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- سلسلة المتوسطات القوية، كاملةً.}
من أجل $p \neq 0$ نضع $M_p = \bigl(\sum_i\lambda_i
x_i^p\bigr)^{1/p}$ (مع $x_i > 0$)، و$M_0 = \prod_i
x_i^{\lambda_i}$.
\begin{enumerate}[resume]
  \item برهن على أن $p \mapsto M_p$ متزايد على $\R^*$ كله:
        وعالج $p < q < 0$ بمتطابقة المقلوب
        $M_{-p}(x) = M_p(1/x)^{-1}$، وأنشئ الجسر عبر $0$ بأن
        تبيّن $M_p \leq M_0 \leq M_q$ من أجل $p < 0 < q$
        \emph{(طبّق تقعّر $\ln$ على $x_i^q$، والمتراجحة
        المعكوسة من أجل الأسس السالبة)}.
  \item برهن على النهايتين $M_p \to \max_i x_i$ حين $p \to
        +\infty$ و$M_p \to \min_i x_i$ حين $p \to -\infty$.
  \item اكتب السلسلة $\min \leq \mathrm{HM} \leq
        \mathrm{GM} \leq \mathrm{AM} \leq \mathrm{QM} \leq
        \max$ من أجل أوزان متساوية، وبرهن على النتيجة
        الكلاسيكية: من أجل أعداد موجبة $a_1, \dots, a_n$،
        \[
        \Bigl(\sum_i a_i\Bigr)\Bigl(\sum_i\frac1{a_i}\Bigr)
        \geq n^2 .
        \]
  \item اربط المتوسطات بالمعايير: من أجل أوزان متساوية $\lambda_i =
        \frac1n$، $M_p(x) = n^{-1/p}\norm x_p$. ووفّق بين
        الرتابتين --- فالمتوسطات \emph{تتزايد} مع $p$
        في حين أن المعايير \emph{تتناقص} (السؤال 10) --- بجملة
        واحدة عن العامل $n^{-1/p}$.
  \item عيّن حالات المساواة على امتداد سلسلة
        السؤال 14 كلها (بأوزان موجبة): فالمساواة في أي موضع
        تفرض تساوي كل القيم $x_i$ --- وهنا يؤتي التحدب التام ثماره.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الأرباح.}
\begin{enumerate}[resume]
  \item (يونغ بمقبض) من أجل $a, b \geq 0$ و$\varepsilon >
        0$، برهن على أن
        \[
        ab \leq \varepsilon\,\frac{a^p}{p} +
        \varepsilon^{-q/p}\,\frac{b^q}{q},
        \]
        وعلى الحالة العملية $ab \leq \varepsilon a^2 +
        \frac{b^2}{4\varepsilon}$: وهي حيلة الامتصاص المستعملة
        في التحليل كله.
  \item (نحو كارلمان) لتكن $c_k = \frac{(k+1)^k}{k^{k-1}}$.
        برهن على المتطابقة التلسكوبية $\prod_{k=1}^{n}c_k =
        (n+1)^n$، واستنتج، بتطبيق متراجحة الحسابي والهندسي على الأعداد
        $c_ka_k$،
        \[
        (a_1a_2\cdots a_n)^{1/n} \leq
        \frac{1}{n(n+1)}\sum_{k=1}^{n} c_k a_k
        \qquad (a_k > 0).
        \]
  \item (متراجحة كارلمان) اجمع على $n$، وبدّل
        ترتيب الجمع (فهي عائلات موجبة \omterm{def:b2:series:summable}{قابلة للجمع}،
        \cref{thm:b2:series:fubini})، واستعمل $\sum_{n \geq
        k}\frac{1}{n(n+1)} = \frac1k$ و$c_k/k = \bigl(1 +
        \frac1k\bigr)^k < \eu$ لتختم: من أجل كل
        متسلسلة متقاربة $\sum a_k$ ذات حدود موجبة،
        \[
        \sum_{n=1}^{\infty}(a_1a_2\cdots a_n)^{1/n}
        \;\leq\; \eu\sum_{k=1}^{\infty}a_k .
        \]
  \item من أجل $f$ متصلة وموجبة على $\intcc{0}{1}$،
        برهن على أن
        \[
        \Bigl(\int_0^1 f\Bigr)\Bigl(\int_0^1\frac1f\Bigr)
        \geq 1,
        \]
        مع المساواة إذا وفقط إذا كانت $f$ ثابتة
        \emph{(كوشي--شوارتز على $\sqrt f\cdot\frac1{\sqrt
        f}$)}.
  \item (هندسة الكرات) باستعمال حالة المساواة في
        مينكوفسكي، بيّن أنه من أجل $1 < p < \infty$ لا تحتوي
        الكرة الواحدية في $\norm\cdot_p$ أي قطعة (\omterm{def:b2:nvs:norm}{فالمعيار}
        محدب تمامًا بمعنى
        \cref{pb:b2:nvs:1})، في حين أنها تحتويها من أجل $p = 1$ و$p =
        \infty$: أبرِز القطع المستقيمة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- الثنوية والتركيب.}
\begin{enumerate}[resume]
  \item (هولدر بوصفه ثنوية) برهن على أنه من أجل كل $a \in
        \R^n$،
        \[
        \norm a_p = \max_{\norm b_q \leq 1}\ \sum_i a_ib_i ,
        \]
        مع إبراز $b$ معظِّمة صراحةً. (\omterm{def:b2:nvs:norm}{فالمعيار} $p$
        هو ثنوي \omterm{def:b2:nvs:norm}{المعيار} $q$ --- وهو البذرة المنتهية البعد
        لثنوية $L^p$.)
  \item (العزوم) ليكن $X$ متغيرًا عشوائيًا يأخذ عددًا منتهيًا
        من القيم الموجبة $x_i$ باحتمالات
        $\lambda_i$. أعد صياغة السؤال 12 هكذا: $r \mapsto
        \E[X^r]^{1/r}$ متزايد --- وهي متراجحة العزوم
        (ليابونوف)، التي سيُعاد استعمالها في
        \cref{ch:b2:randomvar}.
  \item حُلّ بأدوات مسمّاة، في سطرين لكل بند: (1) من أجل
        أعداد موجبة $a, b, c$: $a^3 + b^3 + c^3 \geq
        \frac{(a+b+c)^3}{9}$؛ (2) من أجل أعداد موجبة $x_1, \dots,
        x_n$: $\bigl(\sum_i\sqrt{x_i}\bigr)^2 \leq
        n\sum_i x_i$.
  \item (تركيب) ارسم شجرة النسب في خمس جمل: من تعريف الوتر
        إلى مبرهنة الميول المساعدة؛ ومن الميول إلى المستقيمات المساندة إلى
        جنسن؛ ومن تقعّر $\ln$ إلى يونغ إلى هولدر إلى
        مينكوفسكي إلى المعايير $p$؛ ومن جنسن إلى سلسلة
        المتوسطات القوية إلى العزوم؛ ومن متراجحة الحسابي والهندسي إلى كارلمان. وسمِّ القمتين
        (هولدر--مينكوفسكي؛ وكارلمان)، وقل إلى أين يتجه
        صندوق الأدوات: إلى فضاءات $L^p$ في مجلد السنة
        الثالثة، التي بديهياتها هي بالضبط السؤالان 6 و8.
\end{enumerate}
\end{problem}
