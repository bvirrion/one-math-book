\chapter{مقارنة الدوال}\label{ch:b2:comparison}

بدأ التحليل المقارب --- وهو فنّ تعويض مقدار معقّد
بمقدار بسيط مع خطأ محكوم --- في مجلد السنة الأولى
مع نشور تايلور. ويجعله هذا الفصل تخصّصًا قائمًا
بذاته: النشور على سلالم عامة، ومقارنة المتسلسلة بالتكامل
بكل قوتها المقاربة، وصيغة ستيرلينغ (مبرهَنةً
كاملةً)، والدراسة المنهجية للمتتاليات المعرَّفة ضمنيًا. وهذه
التقنيات هي الخبز اليومي للتحليل المقارب، وكل فصل لاحق يقدّر
أي شيء --- متسلسلات أو تكاملات أو احتمالات --- يأكل من
هذه المائدة.

\section{علاقات المقارنة والسلالم}

\begin{definition}\label{def:b2:comparison:landau}
قرب نقطة $a$ (مع $a \in \R$ أو $\pm\infty$)، ومن أجل الدوال (أو
المتتاليات، مع $n \to \infty$): تُعرَّف $f = o(g)$ و$f = O(g)$ و$f \sim g$
كما في مجلد السنة الأولى. و\emph{سلّم المقارنة}\index{سلّم مقارنة}
عند $a$ عائلة من الدوال الموجبة، قابلة للمقارنة مثنى مثنى،
ومرتَّبة ترتيبًا كليًا بالعلاقة $o(\cdot)$ ---
والسلّم القياسي عند $+\infty$ هو
\[
x^{\alpha} (\ln x)^{\beta}
\qquad (\alpha, \beta \in \R),
\]
مرتَّبًا معجميًا في $(\alpha, \beta)$، ويُنقَّح عند الحاجة
بالأسّيات $\eu^{\gamma x}$.
\end{definition}

\begin{definition}[النشر المقارب]\label{def:b2:comparison:expansion}
تقبل $f$ \emph{النشر المقارب}\index{نشر مقارب}
\[
f = c_1 \varphi_1 + c_2\varphi_2 + \dots + c_k \varphi_k +
o(\varphi_k)
\qquad (\varphi_{i+1} = o(\varphi_i) \text{ في السلّم})
\]
إذا حققت البواقي المتتالية التقديرات المعروضة. وتكون
المعاملات عندئذٍ وحيدة: $c_1 = \lim f/\varphi_1$، ثم
بالتراجع $c_{i+1} = \lim\,(f - \sum_{j \leq i}
c_j\varphi_j)/\varphi_{i+1}$.
\end{definition}

\begin{example}
نشور تايلور نشورٌ مقاربة على السلّم $(x -
a)^k$ عند $a$. لكن المفهوم أوسع تمامًا: فعند $+\infty$،
\[
\frac{1}{x - \ln x}
= \frac1x \cdot \frac{1}{1 - \frac{\ln x}{x}}
= \frac1x + \frac{\ln x}{x^2} + o\Bigl(\frac{\ln x}{x^2}\Bigr),
\]
وهو نشر على سلّم مختلط --- ولا تنطبق أي مبرهنة تايلور،
بل المتسلسلة الهندسية وحساب المقادير $o$ فقط.
\end{example}

\begin{example}[السلّم القياسي مرتَّب فعلًا]\label{ex:b2:comparison:scaleorder}
يحتاج الادعاء المعجمي في \cref{def:b2:comparison:landau}
سطرَ برهان لكل حالة. لنقارن
$x^{\alpha}(\ln x)^{\beta}$ و$x^{\alpha'}(\ln x)^{\beta'}$
عند $+\infty$. إذا كان $\alpha < \alpha'$: فالنسبة
$x^{\alpha - \alpha'}(\ln x)^{\beta - \beta'} \to 0$، لأن
قوة سالبة للمقدار $x$ تسحق أي قوة للمقدار $\ln x$ (ضع $x =
\eu^t$: $\eu^{(\alpha - \alpha')t}\,t^{\beta - \beta'} \to 0$
بنهاية ``الأسّي يغلب كثير الحدود'' في مجلد السنة
الأولى). وإذا كان $\alpha = \alpha'$ و$\beta < \beta'$: فالنسبة
$(\ln x)^{\beta - \beta'} \to 0$ مباشرةً. ومن ثم فإن الأزواج
$(\alpha, \beta)$، مرتَّبةً معجميًا، ترتّب السلّم
بالعلاقة $o(\cdot)$ --- والتعويض $x = \eu^t$ هو
الحيلة الصالحة لكل المقارنات المختلطة بين القوى واللوغاريتمات.
\end{example}

\begin{example}[ترتيب حديقة حيوان]\label{ex:b2:comparison:ranking}
يجب أن تكون السلالم \emph{مرتَّبة}؛ وهذا هو التمرين القياسي. عند
$+\infty$، نقارن $n^{10}$ و$\eu^{\sqrt{\ln n}\,\cdot\,\sqrt n}$
و$2^n$ و$n^{\ln n}$ بأخذ اللوغاريتمات:
\[
10\ln n
\;\ll\; (\ln n)^2
\;\ll\; \sqrt{n\ln n}
\;\ll\; n\ln 2 ,
\]
حيث تعني $a_n \ll b_n$ أن $a_n = o(b_n)$؛ والمدخل الثاني
هو $\ln(n^{\ln n})$. وتحافظ الأسّيات على هذه الفجوات التامة (فإذا كان
$\ln u_n - \ln v_n \to -\infty$ فإن $u_n/v_n \to 0$)، ومنه
\[
n^{10} = o\bigl(n^{\ln n}\bigr),
\qquad
n^{\ln n} = o\bigl(\eu^{\sqrt{n\ln n}}\bigr),
\qquad
\eu^{\sqrt{n\ln n}} = o(2^n) .
\]
والعبرة مزدوجة: قارِن دائمًا عبر اللوغاريتمات
(بفروق اللوغاريتمات، لا بنسبها)، ولا تستنتج أبدًا
$u_n \sim v_n$ من $\ln u_n \sim \ln v_n$ --- فالزوج $n^{10}$
و$n^{\ln n}$ نسبة $\ln$ فيه تؤول إلى $\infty$، لكن $2^n$
و$4^n$ نسبة $\ln$ فيهما تساوي $2$ بالضبط وهما غير
متكافئين إطلاقًا.
\end{example}

\section{مقارنة المتسلسلة بالتكامل، مقاربيًا}

\begin{theorem}\label{thm:b2:comparison:seriesintegral}
لتكن $f$ متصلة وموجبة و\emph{متناقصة} على
$\intco{1}{+\infty}$.
\begin{enumerate}
  \item إذا تقاربت $\int_1^{\infty} f$، فإن البواقي تحقق
        \[
        \int_{n+1}^{\infty} f \;\leq\; \sum_{k > n} f(k) \;\leq\;
        \int_{n}^{\infty} f .
        \]
  \item وإذا تباعدت $\int_1^\infty f$، فإن المجاميع الجزئية تحقق
        $\sum_{k=1}^{n} f(k) = \int_1^n f + C + o(1)$ من أجل ثابت
        $C$ ما: أي إن الفرق $\sum_{k \leq n} f(k) -
        \int_1^n f$ \emph{يتقارب}.
\end{enumerate}
\end{theorem}

\begin{proof}
كان الحصر $f(k+1) \leq \int_k^{k+1} f \leq f(k)$ (بالتناقص) هو
حيلة السنة الأولى؛ ويعطي جمعه على $k \geq n+1$ أو على $k \geq n$
البندَ (1). ومن أجل (2)، نضع $u_k = f(k) - \int_k^{k+1} f$: فبالحصر،
$0 \leq u_k \leq f(k) - f(k+1)$، ومن ثم فإن المجاميع الجزئية
للمتسلسلة $\sum u_k$ محدودة بالمجموع التلسكوبي $f(1) - f(n+1) \leq
f(1)$: فتتقارب المتسلسلة. زيادةً على ذلك، تكون المتتالية $\bigl(\int_n^{n+1}
f\bigr)_n$ غير متزايدة (لأن $f$ متناقصة) وغير سالبة، ومن ثم فهي
متقاربة. وبكتابة
\[
\sum_{k=1}^{n} f(k) - \int_1^n f
= \sum_{k=1}^{n} u_k + \int_n^{n+1} f ,
\]
يتقارب الطرف الأيمن حين $n \to \infty$: أي إن الفرق يتقارب
إلى ثابت $C$، وهو البند (2).
\end{proof}

\begin{example}[نشر المتسلسلة التوافقية]\label{ex:b2:comparison:harmonic}
من أجل $f(t) = \frac1t$: $H_n = \ln n + \gamma + o(1)$، فنستعيد
ثابت أويلر (مجلد السنة الأولى) ببرهان أنظف. وبدفعها رتبة
أخرى (\cref{exo:b2:comparison:3}):
\[
H_n = \ln n + \gamma + \frac{1}{2n} + o\Bigl(\frac1n\Bigr).
\]
وتُظهر الأعداد المكسب عند $n = 10$: فلدينا $H_{10} =
2.928968\dots$ و$\ln 10 = 2.302585\dots$، ومن ثم فالتقدير الخام
للمقدار $\gamma$ هو $H_{10} - \ln 10 = 0.626383$، بخطأ $0.049$؛
وطرح التصحيح $\frac1{20}$ يعطي $0.576383$،
وهو يبعد عن $\gamma = 0.577216$ بمقدار $8.3\cdot10^{-4}$ فقط ---
وهو نفسه الحدّ التالي $\frac{1}{12\cdot100}$ في
النشر، كما تبرهن مسألة نهاية الأسبوع (السؤال 8).
\end{example}

\begin{example}[تقدير خام للمقدار $\ln(n!)$ بلا ستيرلينغ]\label{ex:b2:comparison:lnfactorial}
تحدّد حيلة الحصر وحدها موقع $\ln(n!)$ بالفعل. فبما أن
$\ln$ متزايدة،
\[
\int_{k-1}^{k}\ln t\,\dd t \;\leq\; \ln k \;\leq\;
\int_{k}^{k+1}\ln t\,\dd t ,
\]
وبالجمع على $k = 2, \dots, n$ (مع $\int_1^n\ln = n\ln n
- n + 1$):
\[
n\ln n - n + 1 \;\leq\; \ln(n!) \;\leq\; (n+1)\ln(n+1) - n .
\]
والسياجان كلاهما $n\ln n - n + O(\ln n)$: ومنه $\ln(n!) = n\ln
n - n + O(\ln n)$، وعلى الخصوص $\ln(n!) \sim n\ln n$.
وما تضيفه صيغة ستيرلينغ هو الدرجتان التاليتان --- المقدار $\frac12\ln n$
والثابت $\ln\sqrt{2\pi}$ --- وهما يكلّفان الحصر التلسكوبي الأدقّ
في \cref{thm:b2:comparison:stirling}. ومعرفةُ
أي دقة يشتريها كل أداة نصفُ صنعة التحليل المقارب.
\end{example}

\begin{example}[التلاشي يقتضي النشور]\label{ex:b2:comparison:cancellation}
لنحسب نهاية $\sqrt{n^2 + n} - n$. فالحدّان كلاهما
$\sim n$، والكتابة ``$\sim n - n$'' لا معنى لها: إذ لا يمكن
طرح المكافئات. فننشر بدل ذلك:
\[
\sqrt{n^2 + n} - n
= n\Bigl(\sqrt{1 + \tfrac1n} - 1\Bigr)
= n\Bigl(\frac{1}{2n} - \frac{1}{8n^2} +
O\Bigl(\frac1{n^3}\Bigr)\Bigr)
= \frac12 - \frac{1}{8n} + O\Bigl(\frac1{n^2}\Bigr) :
\]
فالنهاية $\frac12$، مع سرعة الاقتراب $\frac1{8n}$
مكافأةً. وتستحق الآلية اسمًا: فالفرق بين مقدارين كبيرين
متكافئين يعيش كله في حدّيهما \emph{التاليين}،
ومن ثم يجب النشر إلى أول رتبة يختلف عندها الطرفان --- مع
حمل الباقي للتصديق على أن لا شيء آخر ينجو عند تلك
الرتبة.
\end{example}

\begin{example}[مقارنة متباعدة، منفَّذة]\label{ex:b2:comparison:loglog}
من أجل $f(t) = \frac{1}{t\ln t}$ على $\intco{2}{+\infty}$
(وهي متصلة وموجبة ومتناقصة): $\int_2^x f = \ln\ln x -
\ln\ln 2 \to \infty$، ومن ثم حسب
\cref{thm:b2:comparison:seriesintegral}~(2)،
\[
\sum_{k=2}^{n}\frac{1}{k\ln k} = \ln\ln n + C + o(1)
\]
من أجل ثابت $C$ ما. ودرسان. أولًا، التباعد
حقيقي لكنه جليدي: فالمجموع الجزئي يتجاوز $4$ أول مرة قرب $n
\approx \eu^{\eu^{4 - C}}$، وهو عدد فلكي. ثانيًا، أعطت الدالة الأصلية
\emph{الشكل} $\ln\ln n$، ولم يُخمَّن: فمن أجل الحدود الرتيبة،
يكون التكامل أداة الجمع القانونية، والثابت $C$ --- مثل ثابت أويلر $\gamma$
--- هو ذاكرة الحدود الأولى.
\end{example}

\section{صيغة ستيرلينغ}

\begin{lemma}[تكاملات واليس، من جديد]\label{lem:b2:comparison:wallis}
لتكن $W_n = \int_0^{\pi/2} \sin^n t\,\dd t$. عندئذٍ $nW_nW_{n-1} =
\frac\pi2$ من أجل $n \geq 1$، و$(W_n)$ متناقصة، و$W_n \sim
\sqrt{\dfrac{\pi}{2n}}$.
\end{lemma}

\begin{proof}
تعطي المكاملة بالتجزئة $nW_n = (n-1)W_{n-2}$ (مع $n \geq 2$)، ومن ثم فإن
$nW_nW_{n-1}$ ثابت في $n$، ويساوي $1 \cdot W_1 W_0 =
\frac\pi2$. التناقص: $\sin^{n+1} \leq \sin^n$ على
$\intcc{0}{\frac\pi2}$. والحصر، بالتفصيل: تعطي الرتابة
$W_{n+1} \leq W_n \leq W_{n-1}$، وبالقسمة على $W_{n-1}
> 0$،
\[
\frac{n}{n+1} = \frac{W_{n+1}}{W_{n-1}} \leq
\frac{W_n}{W_{n-1}} \leq 1 ,
\]
والمتطابقة اليسرى من التراجع عند الدليل $n + 1$. ويؤول
الحدّان إلى $1$: ومنه $W_n \sim W_{n-1}$، ومن ثم
\[
nW_n^2 \sim nW_nW_{n-1} = \frac\pi2
\qquad\Longrightarrow\qquad
W_n \sim \sqrt{\frac{\pi}{2n}} .
\]
\end{proof}

\begin{example}[تكاملات واليس الأولى]\label{ex:b2:comparison:wallisvalues}
من $W_0 = \frac\pi2$ و$W_1 = 1$ والتراجع $nW_n =
(n-1)W_{n-2}$:
\[
W_2 = \frac\pi4, \qquad
W_3 = \frac23, \qquad
W_4 = \frac{3\pi}{16}, \qquad
W_5 = \frac{8}{15}, \qquad
W_6 = \frac{5\pi}{32}.
\]
تحمل الأدلة الزوجية عاملًا $\pi$، والفردية ناطقة --- وهما الجداءان
المتداخلان للصيغ المغلقة. وعدديًا $W_6
\approx 0.4909$ في مقابل المقارب
$\sqrt{\pi/12} \approx 0.5116$: فعند $n = 6$ يكون المكافئ
ضمن $5\%$ بالفعل، والمتطابقة الجدائية دقيقة عند كل
$n$: $6\,W_6W_5 = 6\cdot\frac{5\pi}{32}\cdot\frac8{15} =
\frac\pi2$. وجداول صغيرة كهذه أرخص طريقة
لاصطياد زلة جبرية قبل أن تعدي حجة مقاربة.
\end{example}

\begin{theorem}[ستيرلينغ]\label{thm:b2:comparison:stirling}
\[
n! \;\sim\; \sqrt{2\pi n}\, \Bigl(\frac{n}{\eu}\Bigr)^{\!n}
.\index{صيغة ستيرلنغ}
\]
\end{theorem}

\begin{proof}
\emph{الخطوة 1: $n! \sim C \sqrt n\, (n/\eu)^n$ من أجل ثابت $C >
0$ ما.} نضع
\[
d_n = \ln(n!) - \Bigl(n + \frac12\Bigr)\ln n + n .
\]
عندئذٍ
\[
d_n - d_{n+1}
= \Bigl(n + \frac12\Bigr) \ln\frac{n+1}{n} - 1
= \Bigl(n + \frac12\Bigr)\Bigl(\frac1n - \frac{1}{2n^2} +
\frac{1}{3n^3} + o\bigl(n^{-3}\bigr)\Bigr) - 1
= \frac{1}{12n^2} + o\Bigl(\frac{1}{n^2}\Bigr),
\]
بنشر تايلور للدالة $\ln(1 + \frac1n)$. ومن ثم تتقارب المتسلسلة $\sum (d_n
- d_{n+1})$ بإطلاق (بالمقارنة مع $\sum
n^{-2}$)، فتتقارب $(d_n)$، ولتكن نهايتها $d$؛ وبأخذ الأسّي، $n! \sim
C\sqrt n\,(n/\eu)^n$ مع $C = \eu^{d}$.

\emph{الخطوة 2: $C = \sqrt{2\pi}$ عبر واليس.} تتضافر الصيغة المغلقة $W_{2p}
= \frac{(2p)!}{4^p (p!)^2}\cdot\frac\pi2$ (الآتية من التراجع،
وهي حساب السنة الأولى معادًا في إطار
\cref{lem:b2:comparison:wallis}) مع الخطوة 1:
\[
W_{2p} \sim \frac{C\sqrt{2p}\,(2p/\eu)^{2p}}
{4^p\,\bigl(C\sqrt p\,(p/\eu)^p\bigr)^2}\cdot\frac{\pi}{2}
= \frac{\sqrt{2p}}{C\,p}\cdot\frac{\pi}{2}
= \frac{\pi}{C}\cdot\frac{1}{\sqrt{2p}} .
\]
وبالمقارنة مع $W_{2p} \sim \sqrt{\frac{\pi}{4p}}$
(\cref{lem:b2:comparison:wallis}): يفرض $\frac{\pi}{C\sqrt{2p}} =
\sqrt{\frac{\pi}{4p}}\,(1 + o(1))$ أن $C = \pi
\sqrt{\frac{4p}{2p\,\pi}} = \sqrt{2\pi}$.
\end{proof}

\begin{example}[المعامل الثنائي المركزي]\label{ex:b2:comparison:centralbinomial}
\[
\binom{2n}{n} = \frac{(2n)!}{(n!)^2}
\sim \frac{\sqrt{4\pi n}\,(2n/\eu)^{2n}}{2\pi n\,(n/\eu)^{2n}}
= \frac{4^n}{\sqrt{\pi n}} :
\]
واحتمال أن يعود مسير عشوائي متناظر إلى $0$ عند الزمن
$2n$ هو $\sim \frac{1}{\sqrt{\pi n}}$ --- وهو إعلان عن
\cref{ch:b2:randomvar}.
\end{example}

\begin{remark}[آفاق داخل هذا المجلد]\label{rem:b2:comparison:perspectives}
يتكلم كل فصل كمّي آتٍ لغةَ هذا الفصل.
فالفصل \cref{ch:b2:series} يصنّف المتسلسلات بمقارنة
حدودها بالسلّم $n^{-\alpha}(\ln n)^{-\beta}$ --- وترسم
مسألة نهاية الأسبوع فيه تلك الحدود كاملةً.
ويفعل \cref{ch:b2:integration} الشيء نفسه من أجل التكاملات المعتلة،
بالسلّم نفسه في المتغيّر \omterm{def:b2:metric:continuity}{المتصل}.
ويحسب \cref{ch:b2:powerseries} أنصاف أقطار التقارب من
$\limsup\abs{a_n}^{1/n}$، وهو تمرين في مكافئ الجذور من الرتبة $n$
يكون فيه ستيرلينغ المفتاحَ القياسي ($\sqrt[n]{n!}
\sim \frac n\eu$، \cref{exo:b2:comparison:4}). وتصرف فصول
الاحتمال ستيرلينغ مباشرةً: فالتقديرات المحلية
في \cref{ch:b2:randomvar} للمعاملات الثنائية
هي \cref{ex:b2:comparison:centralbinomial} و
\cref{ex:b2:comparison:lopsided} حرفيًا. فالتحليل المقارب ليس
فصلًا هنا؛ بل هو لكنة المجلد.
\end{remark}

\begin{method}[قائمة تدقيق الإقلاع المتدرّج]\label{met:b2:comparison:checklist}
قبل الوثوق بنشر متدرّج، دقّق أربع نقاط.
(1) \emph{الوجود أولًا:} يجب تثبيت الجذر أو المتتالية
(بالرتابة والقيم الوسطى) قبل أي نشر
--- فالرموز بلا مرجع تنتشر بجمال ولا تعني
شيئًا. (2) \emph{رتبة واحدة في كل مرور:} لا يُوثق بكل تعويض
إلا إلى رتبة التقدير المُدخَل؛ واستخراج
حدّين جديدين من مرور واحد هو المصدر الكلاسيكي للمعاملات
الخاطئة. (3) \emph{البواقي ترافقك:} احمل
المقدار $o(\cdot)$ عبر كل خطوة جبرية ودَع الابتلاع
(ابتلاع البواقي الأكبر للحدود الأصغر) يقع في
النهاية، صراحةً. (4) \emph{التدقيق العددي:} قيّم عند قيمة
أمينة واحدة للمقدار $n$؛ فخطأ المعامل ينجو من إعادة الاستنباط
الجبري كثيرًا على نحو مفاجئ، ولا ينجو من الحساب أبدًا تقريبًا.
\end{method}

\begin{remark}[مزالق شائعة]\label{rem:b2:comparison:pitfalls}
(1) \emph{تُجمَع} المكافئات جمعًا سيئًا: فمن $u_n \sim n + \ln n$ و
$v_n \sim -n$ \emph{لا} يجوز استنتاج $u_n + v_n \sim \ln
n$؛ فالتلاشيات تقتضي نشورًا ببواقٍ صريحة،
لا مكافئات مجرّدة. (2) لا تأخذ أسّي تكافؤ أبدًا:
فلدينا $n + 1 \sim n$ لكن $\eu^{n+1} \not\sim \eu^n$؛ والاتجاه
الآمن هو أخذ لوغاريتمات المكافئات المؤولة إلى
$+\infty$ (مسألة نهاية الأسبوع في هذا الفصل، السؤال 24). (3)
\omterm{def:b2:comparison:expansion}{النشر المقارب} مرتبط \emph{بسلّم}: فكتابة
$f = \frac1x + o\bigl(\frac1{x^2}\bigr)$ تدّعي أكثر مما تدّعيه $f =
\frac1x + o\bigl(\frac1x\bigr)$، وخلط الاثنين يُبطل
\omterm{def:b2:structures:algebra}{الجبر} اللاحق. (4) في الإقلاع المتدرّج، عوّض
النشر الجاري \emph{كله}، بما فيه الباقي --- فإسقاط
مقدار $o(\cdot)$ في منتصف المرور يُنتج معاملات معقولة لكنها
خاطئة. (5) تحتاج مقارنة المتسلسلة بالتكامل إلى
الرتابة: فهي تفشل تمامًا من أجل الحدود المتذبذبة (قارِن
$\sum\frac{\sin k}k$، \cref{ch:b2:series}).
\end{remark}

\begin{example}[ستيرلينغ بالأعداد]\label{ex:b2:comparison:stirlingnum}
عند $n = 10$: تعطي الصيغة $\sqrt{20\pi}\,(10/\eu)^{10}
\approx 3\,598\,696$ في مقابل $10! = 3\,628\,800$: أي خطأ نسبي
$8.3\cdot10^{-3}$، وهو مذهل من أجل قول ``مقارب'' عند
$n = 10$. وللخطأ بنية --- وهي التنقيح الدقيق $n!
= \sqrt{2\pi n}\,(n/\eu)^n\bigl(1 + \frac1{12n} +
O(n^{-2})\bigr)$ --- ويشرح تصحيحه الأول $\frac1{120} \approx
8.3\cdot10^{-3}$ الفجوةَ المرصودة بدقة تقريبًا. وآلة
أويلر--ماكلورين في مسألة نهاية الأسبوع هي بالضبط المصدر
المنهجي لحدود التصحيح هذه.
\end{example}

\begin{remark}[أين يُستعمَل هذا الفصل]
المقارنة المقاربة هي نحو كل ما هو كمّي
فيما بعد: محكّات التقارب وبانوراما برتران في
\cref{ch:b2:series}، ومحكّات قابلية المكاملة في
\cref{ch:b2:integration}، وحسابات نصف قطر التقارب
في \cref{ch:b2:powerseries}، ومبرهنات النهاية في
\cref{ch:b2:randomvar} (حيث يُشغّل ستيرلينغ تقديرات
دي موافر--لابلاس). ويُصنّع مجلد السنة الثالثة الفكرةَ الواحدة
التي نبرهن عليها هنا يدويًا --- استخرج الحدّ الرئيسي وحُدّ
الباقي --- فتصير طريقة لابلاس والتقارب المهيمَن عليه.
\end{remark}

\begin{example}[تكامل مقارَن بنفسه: $\int_2^x \frac{\dd t}{\ln t}$]\label{ex:b2:comparison:liworked}
تعمل صندوق أدوات المقارنة على التكاملات أيضًا. لتكن $F(x) =
\int_2^x\frac{\dd t}{\ln t}$ (والدالة المكامَلة متصلة على
$\intco2\infty$). نكامل بالتجزئة:
\[
F(x) = \Bigl[\frac{t}{\ln t}\Bigr]_2^x +
\int_2^x\frac{\dd t}{(\ln t)^2}
= \frac{x}{\ln x} + O\Bigl(\int_2^x\frac{\dd t}{(\ln
t)^2}\Bigr) + O(1),
\]
والتكامل الباقي هو $o\bigl(\frac{x}{\ln x}\bigr)$:
فنقسمه عند $\sqrt x$، ونحدّه بالمقدارين
\[
\int_2^{\sqrt x}\frac{\dd t}{(\ln t)^2} \leq \sqrt x
\qquad\text{و}\qquad
\int_{\sqrt x}^{x}\frac{\dd t}{(\ln t)^2} \leq
\frac{x}{(\ln\sqrt x)^2} = \frac{4x}{(\ln x)^2} .
\]
ومنه $F(x) \sim \frac{x}{\ln x}$. وسيتعرّف القراء الذين لقوا
مبرهنة الأعداد الأولية في مسألة نهاية الأسبوع في هذا الفصل على
$F$: فهو التكامل اللوغاريتمي، وهو المقدِّر الأفضل
للمقدار $\pi(x)$، ويبيّن الحساب أنه يتفق مع
$\frac{x}{\ln x}$ إلى الرتبة الأولى.
\end{example}

\begin{example}[ستيرلينغ على معامل ثنائي غير متوازن]\label{ex:b2:comparison:lopsided}
يعطي روتين العواملات الثلاثة نفسه المستعمل من أجل
\cref{ex:b2:comparison:centralbinomial}، ومن أجل
$\binom{3n}{n} = \frac{(3n)!}{n!\,(2n)!}$:
\[
\binom{3n}{n} \sim
\frac{\sqrt{6\pi n}\,(3n/\eu)^{3n}}
{\sqrt{2\pi n}\,(n/\eu)^{n}\cdot\sqrt{4\pi n}\,(2n/\eu)^{2n}}
= \sqrt{\frac{3}{4\pi n}}\,
\Bigl(\frac{27}{4}\Bigr)^{\!n} .
\]
والمعدل الأسّي $\frac{27}4 = \frac{3^3}{2^2}$ هو
$\eu^{3n\,H(1/3)}$ بترميز الإنتروبيا في نظرية
المعلومات: فالمعاملات الثنائية غير المتوازنة تنمو أبطأ تمامًا من
المعامل المركزي $4^n$ لكل خطوتين --- وهنا $(27/4)^{1/3} \approx 1.89 < 2$ لكل
خطوة. وكل مقارب ثنائي في التوافقيات والاحتمال
(\cref{ch:b2:randomvar}) هو هذا الحساب الواحد بأوزان مختلفة.
\end{example}

\section{المتتاليات المعرَّفة ضمنيًا}

\begin{method}\label{met:b2:comparison:implicit}
لإيجاد مقاربات حلول $x_n$ لمعادلة $F(x, n) =
0$:\index{متتالية ضمنية}
\begin{enumerate}
  \item \emph{وطّن}: برهن على وجود $x_n$ ووحدانيته في
        فترة محدَّدة (بالرتابة ومبرهنة القيم
        الوسطى)، وجد سلوكه الخام (النهاية، ورتبة
        النموّ).
  \item \emph{أقلِع تدريجيًا}: عوّض الصورة الخامة $x_n = (\text{الحدّ
        الرئيسي})(1 + \varepsilon_n)$ في المعادلة وحُلّ من أجل
        الرتبة التالية للمقدار $\varepsilon_n$؛ وكرّر، فينقّح كل مرور
        رتبةً واحدة.
\end{enumerate}
\end{method}

\begin{example}\label{ex:b2:comparison:tan}
من أجل $n \geq 1$، للمعادلة $\tan x = x$ حلٌّ واحد بالضبط
$x_n$ في $\intoo{n\pi - \frac\pi2}{n\pi + \frac\pi2}$ (فالدالة
$\tan x - x$ متزايدة من $-\infty$ إلى $+\infty$ هناك، ومشتقها
$\tan^2 x \geq 0$). \emph{الخام:} $x_n = n\pi +
\frac\pi2 - y_n$ مع $y_n \in \intoo{0}{\pi}$؛ وبما أن $x_n \to
\infty$ و$\tan x_n = x_n \to +\infty$، فإن $x_n$ تقترب من
المقارب من اليسار: $y_n \to 0$. \emph{الإقلاع المتدرّج:} $\tan x_n =
\cot y_n = \frac{1}{\tan y_n} \sim \frac{1}{y_n}$، وتعطي المعادلة
$\cot y_n = x_n \sim n\pi$ أن $y_n \sim \frac{1}{n\pi}$. ومنه
\[
x_n = n\pi + \frac\pi2 - \frac{1}{n\pi} +
o\Bigl(\frac1n\Bigr),
\]
وتستمر العملية إلى أي رتبة
(\cref{exo:b2:comparison:6}).
\end{example}

\begin{example}[تشغيلة ثانية للطريقة]\label{ex:b2:comparison:xplusln}
حُلّ $x + \ln x = n$ مقاربيًا. \emph{التوطين:} $x
\mapsto x + \ln x$ متزايدة من $-\infty$ إلى $+\infty$ على
$\intoo{0}{+\infty}$: فجذر وحيد $x_n$، و$x_n \to \infty$.
\emph{الخام:} يعطي $\ln x_n = o(x_n)$ أن $x_n \sim n$.
\emph{الإقلاع المتدرّج:} من $x_n = n - \ln x_n$ ومن $\ln x_n = \ln n
+ o(1)$ (لوغاريتمات مكافئات، وكلا الطرفين $\to \infty$):
\[
x_n = n - \ln n + o(1) ;
\]
وبمرور آخر، مع $\ln x_n = \ln\bigl(n - \ln n + o(1)\bigr) =
\ln n - \frac{\ln n}{n} + o\bigl(\frac{\ln n}n\bigr)$:
\[
x_n = n - \ln n + \frac{\ln n}{n} +
o\Bigl(\frac{\ln n}{n}\Bigr).
\]
(وللتحقق عند $n = 100$: الجذر هو $x \approx 95.4415$؛ وتعطي
صيغة الحدود الثلاثة $100 - 4.6052 + 0.0461 = 95.4409$، وصيغة
الحدّين $95.3948$ --- فيكسب كل مرور الرتبة المتوقعة.)
الحلقة نفسها، ومنظر ثالث: فطريقة
\cref{met:b2:comparison:implicit} لا تعبأ بشكل
المعادلة، بل بأن يعزل كل مرور المجهولَ المهيمن فحسب.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:comparison:1}
انشر عند $+\infty$، حدّين بعد الحدّ الرئيسي:
\[
\sqrt{x^2 + x + 1} ,
\qquad
\ln(x^2 + x) - 2\ln x,
\qquad
\frac{x + \sin x}{x - \ln x} .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:comparison:2}
أعطِ الطبيعة (تقارب أو تباعد) ومقاربات الحدّ الرئيسي عند
التباعد للمقدار $\sum_{k \leq n} k^\alpha$ من أجل $\alpha >
-1$ و$\alpha = -1$ و$\alpha < -1$، عبر
\cref{thm:b2:comparison:seriesintegral}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:comparison:3}
برهن على أن $H_n = \ln n + \gamma + \frac{1}{2n} + o\bigl(\frac1n\bigr)$.
\emph{(ادرس $v_n = H_n - \ln n - \gamma$: بيّن أن $v_n - v_{n+1} =
\frac{1}{2n^2} + O(n^{-3})$ واجمع الذيل، بالمقارنة مع
$\sum_{k \geq n} \frac{1}{2k^2} \sim \frac{1}{2n}$ ---
\cref{thm:b2:comparison:seriesintegral}~(1).)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:comparison:4}
باستعمال ستيرلينغ، جد مكافئات المقادير: $\dfrac{(3n)!}{(n!)^3}$؛
و$\;\dfrac{n!}{n^n}$؛ و$\;\sqrt[n]{n!}$ (حين $\frac n\eu(1 + o(1))$،
مصوغًا بدقة إلى حدّين).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:comparison:5}
من أجل $n \geq 2$، برهن على أن للمعادلة $x^n + x = 1$ حلًا وحيدًا $x_n
\in \intoo{0}{1}$، وأن $x_n \to 1$، وأثبت
\[
x_n = 1 - \frac{\ln n}{n} + o\Bigl(\frac{\ln n}{n}\Bigr).
\]
\emph{(من $x_n^n = 1 - x_n$: خذ اللوغاريتمات وأقلِع تدريجيًا مع
$x_n = 1 - \varepsilon_n$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:comparison:6}
ادفع \cref{ex:b2:comparison:tan} رتبة أخرى:
\[
x_n = n\pi + \frac\pi2 - \frac{1}{n\pi} + \frac{1}{2n^2\pi} +
o\Bigl(\frac{1}{n^2}\Bigr).
\]
\emph{(اكتب $\cot y_n = x_n$ بالضبط، وانشر $\cot y = \frac1y -
\frac y3 + o(y)$ و$x_n = n\pi(1 + \frac{1}{2n} - \dots)$، ثم
عيّن المعاملات.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:comparison:7}
عيّن $\lim_{n\to\infty} \dfrac{1}{n!}\sum_{k=0}^{n} k!$
\emph{(حُدّ مجموع كل الحدود عدا الأخيرين)}، واستنتج
\omterm{def:b2:comparison:expansion}{النشر المقارب} $\sum_{k \leq n} k! = n!\bigl(1 + \frac1n +
O(n^{-2})\bigr)$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:comparison:8}
لتكن $u_0 > 0$ و$u_{n+1} = u_n + \dfrac{1}{u_n}$. برهن على أن $u_n
\to \infty$، ثم على أن $u_n \sim \sqrt{2n}$ \emph{(ادرس $u_n^2$:
فتزايداتها $2 + u_n^{-2}$؛ ثم اجمع)}، ونقّح:
\[
u_n = \sqrt{2n}\Bigl(1 + \frac{\ln n}{8n} +
o\Bigl(\frac{\ln n}{n}\Bigr)\Bigr).
\]
\emph{(من $u_n^2 = 2n + \sum_{k<n} u_k^{-2} + u_0^2$ ومن $u_k^2
\sim 2k$: يكون المجموع $\sim \frac12\ln n$ حسب
\cref{thm:b2:comparison:seriesintegral}.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:comparison:9}
(مجموع ريمان بلمسة) عيّن السلوك المقارب للمقدار
\[
S_n = \sum_{k=1}^{n} \frac{1}{n + k\ln n} .
\]
\emph{(أخرِج العامل $n$: $S_n = \frac1n\sum_k \bigl(1 +
\frac{k\ln n}{n}\bigr)^{-1}$؛ وتعرّف على مجموع من نمط ريمان بوسيط
بطيء التغيّر $t = \ln n$، واحسب $\int_0^1
\frac{\dd u}{1 + tu} = \frac{\ln(1+t)}{t}$، واختم بأن $S_n \sim
\frac{\ln\ln n}{\ln n}$.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:comparison:10}
برهن على المتطابقة $(\ln n)^{\ln n} = n^{\ln\ln n}$، ثم رتّب
ما يلي ترتيبًا تصاعديًا بالعلاقة $o(\cdot)$ عند اللانهاية، مع
البراهين: $n^2$، $(\ln n)^{\ln n}$، $2^n$، $n!$، $n^n$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:comparison:11}
(ذيل المقدار $\sum 1/k^2$، بحدّين) باستعمال التلسكوب الدقيق
$\sum_{k > n} \frac{1}{k(k+1)} = \frac{1}{n+1}$ والتفكيك
$\frac1{k^2} = \frac{1}{k(k+1)} +
\frac{1}{k^2(k+1)}$، برهن على أن
\[
\sum_{k > n} \frac{1}{k^2}
= \frac1n - \frac{1}{2n^2} + O\Bigl(\frac{1}{n^3}\Bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:comparison:12}
لتكن $u_0 = \frac12$ و$u_{n+1} = u_n + \eu^{-u_n}$. برهن على أن
$u_n \to \infty$، ثم --- بوضع $v_n = \eu^{u_n}$ وتبيان أن
$v_{n+1} = v_n + 1 + \frac{1}{2v_n} + O\bigl(v_n^{-2}\bigr)$ ---
أثبت
\[
u_n = \ln n + \frac{\ln n}{2n} + O\Bigl(\frac1n\Bigr).
\]
\end{exercise}

\section{مسألة: الإقلاع المتدرّج، من أويلر--ماكلورين إلى
الأعداد الأولية}

نادرًا ما يسلّم مقدار ضمني أو متراكم مقارباته
دفعةً واحدة؛ بل تُستخرج على مرورات، يُغذّي كل مرور فيها
العلاقةَ المعرِّفة بالتقدير السابق.
وتدرّب مسألة نهاية الأسبوع هذه تلك الحلقة على معادلات جديدة، وتبرهن
على \emph{صيغة أويلر--ماكلورين} من الرتبة الأولى (وهي ترقية شبه المنحرف
لمقارنة المتسلسلة بالتكامل، بأشرطة خطأ صارمة)،
وتقلب $x\ln x = n$، وتصرف أشهر شيك للطريقة:
فمن مبرهنة الأعداد الأولية المقبولة، نستخرج القانون
المقارب $p_n \sim n\ln n$ للعدد الأولي رقم $n$.

\begin{problem}[{مسألة نهاية الأسبوع --- تصحيح أويلر--ماكلورين
ومقاربات العدد الأولي رقم $n$}]
\label{pb:b2:comparison:1}

\medskip
\noindent\textbf{الجزء الأول --- حلقة الإقلاع المتدرّج على معادلة
جديدة.}
\begin{enumerate}
  \item برهن على قول الوحدانية في
        \cref{def:b2:comparison:expansion}: إذا كان $f = \sum_{i\leq
        k} c_i\varphi_i + o(\varphi_k) = \sum_{i \leq k}
        c_i'\varphi_i + o(\varphi_k)$ على السلّم نفسه، فإن
        $c_i = c_i'$ من أجل كل $i$. ثم ادفع مثال الدرس
        المختلط رتبة أخرى:
        \[
        \frac{1}{x - \ln x} = \frac1x + \frac{\ln x}{x^2} +
        \frac{(\ln x)^2}{x^3} + o\Bigl(\frac{(\ln
        x)^2}{x^3}\Bigr) \qquad (x \to +\infty),
        \]
        واشرح لماذا لا يظهر أي حدّ $\frac{c}{x^2}$.
  \item بيّن أنه من أجل كل $n \geq 1$ يوجد للمعادلة $\eu^x + x =
        n$ حلٌّ حقيقي واحد بالضبط $x_n$، وأن $x_n
        \to +\infty$ مع $x_n \sim \ln n$.
  \item أقلِع تدريجيًا مرتين:
        \[
        x_n = \ln n - \frac{\ln n}{n} - \frac{(\ln n)^2}{2n^2} +
        o\Bigl(\frac{(\ln n)^2}{n^2}\Bigr).
        \]
  \item تحقق عدديًا عند $n = 1000$: قارِن $x_{1000}
        \approx 6.90083$ بالقيم ذات الحدّ الواحد والحدّين والثلاثة
        في السؤال 3، إلى خمسة أرقام عشرية.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- أويلر--ماكلورين، من الرتبة الأولى.}
\begin{enumerate}[resume]
  \item برهن على متطابقة نواة شبه المنحرف: من أجل $g$ من الصنف
        $C^2$ على $\intcc{0}{1}$،
        \[
        \int_0^1 g(t)\,\dd t = \frac{g(0) + g(1)}{2}
        - \frac12\int_0^1 t(1 - t)\,g''(t)\,\dd t
        \]
        \emph{(كامِل $\frac12 t(1-t)g''$ بالتجزئة مرتين)}.
  \item لتكن $f$ من الصنف $C^2$ على $\intco{1}{+\infty}$ مع
        $\int_1^\infty \abs{f''} < \infty$. بيّن أن
        \[
        E_n = \sum_{k=1}^{n} f(k) - \int_1^n f -
        \frac{f(1) + f(n)}{2}
        \]
        تتقارب إلى ثابت $E$، مع حدّ الذيل
        $\abs{E - E_n} \leq \frac18\int_n^\infty\abs{f''}$: وهي
        \emph{صيغة أويلر--ماكلورين} من الرتبة الأولى.
  \item طبّق هذا على $f(t) = \frac1t$: برهن على أن
        \[
        H_n = \ln n + \gamma + \frac{1}{2n} + \varepsilon_n,
        \qquad \abs{\varepsilon_n} \leq \frac{1}{8n^2},
        \]
        فتقوّي \cref{exo:b2:comparison:3} (وعيّن
        الثابت بالمقدار $\gamma$ بالمقارنة مع
        \cref{ex:b2:comparison:harmonic}).
  \item استخرج المعامل التالي: بيّن أن $\varepsilon_n =
        -\frac{1}{12n^2} + o\bigl(\frac1{n^2}\bigr)$
        \emph{(فتزايدات $E_n$ هي
        $\frac12\int_0^1t(1-t)f''(n+t)\dd t =
        \frac1{12}f''(n) + o(f''(n))$؛ واجمع الذيل مع
        \cref{thm:b2:comparison:seriesintegral})}.
  \item طبّق السؤال 6 على $f = \ln$: أعد استنباط تقارب
        $d_n = \ln n! - (n +
        \frac12)\ln n + n$ في ثلاثة أسطر (وهو الخطوة 1 في
        \cref{thm:b2:comparison:stirling})، مع مكافأة معدل
        الخطأ $d_n = d + O\bigl(\frac1n\bigr)$.
  \item طبّق السؤال 6 على $f(t) = \frac{1}{\sqrt t}$: بيّن أن
        \[
        \sum_{k=1}^{n}\frac1{\sqrt k} = 2\sqrt n + c +
        \frac{1}{2\sqrt n} + O\Bigl(\frac{1}{n^{3/2}}\Bigr)
        \]
        من أجل ثابت $c$ ما، وقيّم كل الحدود عند $n =
        10^4$ (والثابت هو $c \approx -1.4604$).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- القلب: المعادلة $x\ln x =
n$.}
\begin{enumerate}[resume]
  \item بيّن أن للمعادلة $x\ln x = n$ حلًا واحدًا بالضبط $x_n \in
        \intco{1}{+\infty}$ من أجل $n \geq 1$، وأن $x_n \to
        \infty$، وأن $\ln x_n \sim \ln n$.
  \item استنتج القلب ذا الحدّ الواحد $x_n \sim
        \dfrac{n}{\ln n}$، ثم أقلِع تدريجيًا مرة أخرى:
        \[
        \ln x_n = \ln n - \ln\ln n + o(1),
        \qquad
        x_n = \frac{n}{\ln n}\Bigl(1 + \frac{\ln\ln n}{\ln n}
        + o\Bigl(\frac{\ln\ln n}{\ln n}\Bigr)\Bigr).
        \]
  \item اختبر عند $n = 10^6$: الجذر الحقيقي هو $x \approx
        87\,848$؛ قارِن بالقيمة ذات الحدّ الواحد ($\approx 72\,382$)
        وبالقيمة ذات الحدّين ($\approx 86\,140$)، واشرح
        بطء المكسب (فوسيط النشر هو $\frac{\ln\ln
        n}{\ln n}$، ولا يساوي إلا $\approx 0.19$ عند $n = 10^6$).
  \item \emph{نقبل} الآن مبرهنة الأعداد الأولية: يحقق عدد
        $\pi(x)$ الأعداد الأولية التي لا تتجاوز $\leq x$ العلاقةَ $\pi(x) \sim
        \frac{x}{\ln x}$ حين $x \to \infty$ (وهي مبرهَنة بأمانة في
        مجلد السنة الثالثة). وبكتابة $p_n$ للعدد الأولي رقم $n$،
        برّر $\pi(p_n) = n$، وشغّل قلب
        السؤالين 11--12 لتبرهن على أن
        \[
        p_n \sim n \ln n .
        \]
  \item الأرباح: (أ) بيّن أن $\sum_{k \leq n} p_k \sim
        \frac{n^2\ln n}{2}$ \emph{(قارِن $\sum k\ln k$ بالمقدار
        $\int t\ln t\,\dd t$)}؛ (ب) احسب الفرصة التقريبية
        لأن يكون عدد صحيح عشوائي منتظم ذو $100$
        رقمًا أوليًا ($\ln 10^{100} \approx 230.26$: نحو
        واحد من كل $230$).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- تصدير الطريقة: $x\tan x =
1$.}
\begin{enumerate}[resume]
  \item بيّن أنه من أجل كل $n \geq 1$ يوجد للمعادلة $\tan x =
        \frac1x$ حلٌّ واحد بالضبط $x_n$ في
        $\intoo{n\pi}{\,n\pi + \frac\pi2}$، وأن $z_n = x_n
        - n\pi \to 0^+$.
  \item حدٌّ واحد: $z_n \sim \dfrac{1}{n\pi}$.
  \item بيّن أن نشر $z_n$ \emph{لا} يحتوي أي حدّ
        $\frac{c}{n^2}$: أي $z_n = \frac1{n\pi} +
        O\bigl(\frac{1}{n^3}\bigr)$.
  \item ثلاثة حدود: باستعمال $\arctan u = u - \frac{u^3}3 +
        O(u^5)$ و$\frac1{x_n} = \frac{1}{n\pi} -
        \frac{z_n}{(n\pi)^2} + O(n^{-3}\cdot z_n^2)$، برهن على أن
        \[
        x_n = n\pi + \frac{1}{n\pi} -
        \frac{4}{3\pi^3 n^3} + o\Bigl(\frac{1}{n^3}\Bigr).
        \]
  \item تحقق عند $n = 3$: الجذر الحقيقي $x_3 \approx 9.5293344$؛
        قارِن القيمتين ذات الحدّ الواحد وذات الحدود الثلاثة، وقابِل
        بجملة واحدة بينها وبين المتتالية $\tan x = x$ في الدرس
        (\cref{ex:b2:comparison:tan}): أين تقع كل متتالية
        في نافذتها، ولماذا.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- إقلاع متدرّج حركي، وقواعد
اللعبة، والتركيب.}
\begin{enumerate}[resume]
  \item لتكن $u_0 \in \intoo{0}{\pi}$ و$u_{n+1} = \sin u_n$.
        بيّن أن $u_n \to 0$ تناقصيًا، واحسب نهاية
        $\dfrac{1}{u_{n+1}^2} - \dfrac{1}{u_n^2}$ \emph{(انشر
        $\sin^{-2}$ عبر $\sin u = u - \frac{u^3}6 + o(u^3)$)}.
  \item استنتج، عبر متوسطات تشيزارو (مجلد السنة الأولى)، الكلاسيكيةَ
        \[
        u_n \sim \sqrt{\frac{3}{n}} .
        \]
  \item (حسابات مصدَّقة) باستعمال الحدّ الصارم في السؤال 7،
        بيّن أن تقييم $\ln n + \gamma + \frac1{2n}$ عند $n
        = 10^6$ يعطي $H_{10^6}$ بخطأ لا يتجاوز
        $1.25\cdot10^{-13}$ --- أي مجموع بمليون حدّ محسوبًا إلى
        ثلاثة عشر رقمًا بثلاثة حدود.
  \item (قواعد اللعبة) برهن أو ادحض، بالبراهين أو
        بالأمثلة المضادة: (أ) إذا كان $u_n \sim v_n \to +\infty$ فإن
        $\ln u_n \sim \ln v_n$؛ (ب) إذا كان $u_n \sim v_n$ فإن
        $\eu^{u_n} \sim \eu^{v_n}$؛ (ج) إذا كان $f \sim g$ عند
        $+\infty$ (مع $f, g$ قابلة للاشتقاق) فإن $f' \sim g'$.
  \item (تركيب) بجملة واحدة لكل بند: حلقة الإقلاع المتدرّج في
        \cref{met:b2:comparison:implicit} كما استُعملت في الأجزاء الأول
        والثالث والرابع؛ وماذا يضيف تصحيح شبه المنحرف إلى
        \cref{thm:b2:comparison:seriesintegral}؛ ولماذا يكون قلب
        $x\ln x$ هو بالضبط الجسر من $\pi(x)$ إلى
        $p_n$؛ وأي قاعدة من قواعد السؤال 24 حرست أي
        خطوة. وسمِّ القمتين: صيغة أويلر--ماكلورين
        (من الرتبة الأولى)، والقانون المقارب للعدد الأولي رقم $n$.
\end{enumerate}
\end{problem}
