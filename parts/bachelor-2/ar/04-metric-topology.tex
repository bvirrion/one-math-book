\chapter{طوبولوجيا الفضاءات المترية}\label{ch:b2:metric}

تتعمّم طوبولوجيا المستقيم الحقيقي (مجلد السنة الأولى)، دون تغيير
كلمة تقريبًا، إلى أي مجموعة مزوَّدة بمسافة. والمكسب هائل:
فمتتاليات الدوال والمصفوفات والمنحنيات --- كلها تصير نقاطًا في
فضاءات مترية، وتنطبق عليها بانتظام الأركان الثلاثة المبرهَن عليها
هنا --- التمام مع مبرهنة النقطة الثابتة عند باناخ، والتراص،
والترابط. وهذا الفصل هو العمود الفقري لنصف الكتاب التحليلي كله.

\section{الفضاءات المترية}

\begin{definition}\label{def:b2:metric:def}
\emph{الفضاء المتري}\index{فضاء متري} مجموعة $X$ مزوَّدة بتطبيق $d
\colon X \times X \to \R_+$ يحقق، من أجل كل $x, y, z$:
\[
d(x,y) = 0 \iff x = y,
\qquad
d(x,y) = d(y,x),
\qquad
d(x,z) \leq d(x,y) + d(y,z).
\]
والكرات: $B(a, r) = \{x : d(a,x) < r\}$ (\omterm{def:b2:metric:topology}{مفتوحة})، $\overline B(a,r) =
\{x : d(a,x) \leq r\}$ (مغلقة). ويصير الجزء $A \subseteq X$
فضاءً متريًا بالمسافة المستحثّة.
\end{definition}

\begin{example}\label{ex:b2:metric:examples}
$\R$ مع $\abs{x - y}$؛ و$\R^n$ مع أي من المسافات
\[
d_1(x,y) = \sum_i \abs{x_i - y_i},
\quad
d_2(x,y) = \Bigl(\sum_i (x_i - y_i)^2\Bigr)^{1/2},
\quad
d_\infty(x,y) = \max_i \abs{x_i - y_i};
\]
ومجموعة الدوال المتصلة $C(\intcc{a}{b})$ مع
\emph{مسافة النهاية العليا} $d_\infty(f, g) = \sup_{\intcc{a}{b}} \abs{f -
g}$ (وهي منتهية لأن $f - g$ محدودة)؛ وأي مجموعة مع المسافة
\emph{المتقطعة} ($d(x,y) = 1$ من أجل $x \neq y$). أما المسافات الآتية من
المعايير فهي موضوع \cref{ch:b2:nvs}.
\end{example}

\begin{definition}[طوبولوجيا الفضاء المتري]\label{def:b2:metric:topology}
تكون $U \subseteq X$ \emph{مفتوحة}\index{مجموعة مفتوحة!في فضاء متري}
إذا كانت كل نقطة من $U$ مركزَ كرة محتواة في $U$؛
وتكون $F$ \emph{مغلقة} إذا كانت متممتها مفتوحة. وتُعرَّف الجوارات
والداخل والمنغلق والكثافة والحافة تمامًا كما على المستقيم
الحقيقي (مجلد السنة الأولى)، مع كرات بدل الفترات، وتنتقل الأقوال
المبرهَن عليها هناك --- اتحادات المفتوحات وتقاطعاتها،
وتوصيفا الداخل والمنغلق، والمنغلق بوصفه أصغر مغلقة تحتوي المجموعة ---
بالبراهين نفسها. والكرات المفتوحة مفتوحة، والكرات المغلقة مغلقة
(بمتراجحة المثلث).
\end{definition}

\begin{example}[الداخل والمنغلق والحافة على مجموعة واحدة]\label{ex:b2:metric:intclosure}
في $\R$، لتكن $A = \intoc{0}{1} \cup \{2\}$. الداخل:
$\intoo{0}{1}$ --- إذ تبقى كرة صغيرة حول أي $x \in \intoo01$
داخل $A$؛ أما حول $1$، فكل كرة $\intoo{1-r}{1+r}$ تتسرّب
خارج $A$ من اليمين، ومن ثم فإن $1$ ليست داخلية؛ والنقطة
المعزولة $2$ ليست داخلية أيضًا. المنغلق:
$\intcc{0}{1} \cup \{2\}$ (فالنقطة $0$ نهاية للمجموعة $A$،
ولا يُضاف شيء آخر). الحافة (المنغلق ناقص الداخل):
$\{0, 1, 2\}$. ولاحظ التباينات الجديرة بالتذكّر: قد تنتمي نقطة
طرفية إلى مجموعة دون أن تكون داخلية ($1$)، وقد تكون
ملتصقة دون أن تنتمي ($0$)، والنقطة المعزولة
هي حافة نفسها ($2$). ويجري مسك الدفاتر نفسه حرفيًا في
أي \omterm{def:b2:metric:def}{فضاء متري}، بكرات بدل الفترات.
\end{example}

\begin{definition}[النهايات والاتصال]\label{def:b2:metric:continuity}
نكتب $x_n \to x$ في $X$ إذا كان $d(x_n, x) \to 0$. ويكون التطبيق $f \colon X \to Y$
بين فضاءين مترين \emph{متصلًا}\index{متصل} عند $a$ إذا كان
\[
\forall \varepsilon > 0,\ \exists\delta > 0,\quad
d_X(x, a) \leq \delta \implies d_Y\bigl(f(x), f(a)\bigr) \leq
\varepsilon ;
\]
وبكيفية مكافئة (بالبرهان نفسه المستعمل على $\R$)، $f(x_n) \to f(a)$ من أجل كل
متتالية $x_n \to a$. ويكون $f$ \emph{ليبشيتزيًا}\index{ليبشيتزي} بالثابت $k$ إذا كان
$d_Y(f(x), f(y)) \leq k\, d_X(x, y)$ دائمًا --- ويكون عندئذٍ متصلًا
بانتظام، ومن ثم متصلًا.
\end{definition}

\begin{theorem}[التوصيف الشمولي للاتصال]\label{thm:b2:metric:globalcontinuity}
يكون $f \colon X \to Y$ \omterm{def:b2:metric:continuity}{متصلًا} (عند كل نقطة) إذا وفقط إذا كانت
الصورة العكسية لكل \omterm{def:b2:metric:topology}{مفتوحة} مفتوحةً --- إذا وفقط إذا كانت الصورة العكسية لكل
مغلقة مغلقةً.
\end{theorem}

\begin{proof}
($\Rightarrow$) لتكن $V \subseteq Y$ \omterm{def:b2:metric:topology}{مفتوحة} وليكن $a \in f^{-1}(V)$:
فتوجد كرة $B(f(a), \varepsilon) \subseteq V$؛ ويعطي الاتصال عند $a$
عددًا $\delta$ يحقق $f(B(a, \delta)) \subseteq B(f(a),
\varepsilon)$، ومنه $B(a, \delta) \subseteq f^{-1}(V)$.

($\Leftarrow$) لتكن $a$ ولتكن $\varepsilon$: فالمجموعة $f^{-1}\bigl(B(f(a),
\varepsilon)\bigr)$ \omterm{def:b2:metric:topology}{مفتوحة} وتحتوي $a$، ومن ثم تحتوي كرة
$B(a, \delta)$: وهذا هو تعريف الاتصال عند $a$. وأما المغلقات:
فبالمتممات (\cref{prop:b2:structures:images}).
\end{proof}

\section{الفضاءات التامة}

\begin{definition}\label{def:b2:metric:complete}
تكون المتتالية $(x_n)$ \emph{متتالية كوشي} إذا كان $\sup_{p, q \geq N} d(x_p,
x_q) \to 0$ حين $N \to \infty$. ويكون \omterm{def:b2:metric:def}{الفضاء المتري}
\emph{تامًا}\index{فضاء متري تام} إذا تقاربت فيه كل متتالية
كوشي. والمتقاربة $\Rightarrow$ كوشي دائمًا؛ والأجزاء المغلقة
من الفضاءات التامة تامة، والأجزاء التامة من أي فضاء
مغلقة (بالبراهين نفسها المستعملة على $\R$: مجلد السنة الأولى).
\end{definition}

\begin{example}[متتالية كوشي بلا نهاية]\label{ex:b2:metric:cauchynolimit}
في $X = \Q$ مع المسافة المعتادة، تحقق الاقتطاعات العشرية للعدد
$\sqrt2$،
\[
x_0 = 1,\quad x_1 = 1.4,\quad x_2 = 1.41,\quad x_3 = 1.414,
\quad\dots
\]
العلاقةَ $\abs{x_p - x_q} \leq 10^{-\min(p,q)}$: فهي متتالية كوشي في $\Q$.
وأي نهاية في $\Q$ تكون نهاية في $\R$ أيضًا، أي
$\sqrt2 \notin \Q$: فلا نهاية لها في $X$. وعدم التمام هو
وجود مثل هذه ``النهايات الشبحية''؛ وقد صُمّم تمام $\R$
في مجلد السنة الأولى بالضبط ليعطي كل متتالية كوشي بيتًا.
\end{example}

\begin{theorem}\label{thm:b2:metric:rncomplete}
$\R^n$ (بأي من المسافات الثلاث في
\cref{ex:b2:metric:examples}) و$\bigl(C(\intcc{a}{b}),
d_\infty\bigr)$ فضاءان تامّان.
\end{theorem}

\begin{proof}
$\R^n$: متتالية كوشي هي كوشي في كل إحداثية (فكل
$\abs{x_i - y_i} \leq d(x,y)$ من أجل المسافات الثلاث)، ومن ثم تتقارب كل
إحداثية (بتمام $\R$، مجلد السنة الأولى)، ويستلزم
التقارب الإحداثي التقارب من أجل $d_\infty$
(فالإحداثيات منتهية العدد)، ومن ثم من أجل الثلاث (لأن المسافات
الثلاث يهيمن بعضها على بعض بمعاملات ثابتة: $d_\infty
\leq d_2 \leq d_1 \leq n\,d_\infty$).

$C(\intcc{a}{b})$: لتكن $(f_n)$ متتالية كوشي من أجل $d_\infty$. من أجل كل $x$،
تكون $(f_n(x))$ متتالية كوشي في $\R$ ($\abs{f_p(x) - f_q(x)} \leq
d_\infty(f_p, f_q)$): فتتقارب إلى $f(x)$ ما. وبالانتقال إلى
النهاية في $\abs{f_p(x) - f_q(x)} \leq \varepsilon$ (وهي صالحة من أجل $p, q
\geq N_\varepsilon$ وكل $x$) حين $q \to \infty$: نجد $\abs{f_p(x) -
f(x)} \leq \varepsilon$ من أجل كل $x$، أي $d_\infty(f_p, f) \leq
\varepsilon$: أي التقارب المنتظم. والنهاية متصلة: فليكن
$\varepsilon$، ونختار $p$ بحيث $\sup\abs{f_p - f} \leq \varepsilon$،
ثم نستعمل اتصال $f_p$ عند $a$ والتقسيم الثلاثي
\[
\abs{f(x) - f(a)} \leq \abs{f(x) - f_p(x)} + \abs{f_p(x) - f_p(a)}
+ \abs{f_p(a) - f(a)} \leq 3\varepsilon
\]
من أجل $x$ قريبة من $a$. (وتعود ``حجة $3\varepsilon$'' هذه بوصفها
مبرهنةَ النهاية المنتظمة في \cref{ch:b2:funcseq}.)
\end{proof}

\begin{example}[المفتوحات والمغلقات تُعرَف بالاتصال]\label{ex:b2:metric:recognize}
التوصيف الشمولي
(\cref{thm:b2:metric:globalcontinuity}) هو الأداة اليومية لمسك
الدفاتر الطوبولوجي. ففي $\R^2$: المجموعة $\{(x, y) : x^2 + y^2
< 1,\ y > x^3\}$ \omterm{def:b2:metric:topology}{مفتوحة} --- إذ هي $g^{-1}(\intoo{-\infty}{1})
\cap h^{-1}(\intoo{0}{+\infty})$ من أجل الدالتين المتصلتين $g(x,y) =
x^2 + y^2$ و$h(x, y) = y - x^3$، أي تقاطع صورتين عكسيتين
مفتوحتين. وفي $\bigl(C(\intcc01), d_\infty\bigr)$: مجموعة
الدوال التي $f(0) = f(1)$ و$\int_0^1 f = 0$ مغلقة ---
إذ هي الصورة العكسية للمجموعة $\{(0,0)\}$ بالتطبيق \omterm{def:b2:metric:continuity}{المتصل} $f \mapsto
\bigl(f(0) - f(1),\ \int_0^1 f\bigr)$ نحو $\R^2$ (وكل
إحداثية ليبشيتزية بالثابت $1$، كما في \cref{exo:b2:metric:3}). ولا
ترسم هذه الطريقة أي شكل: أبرِز تطبيقًا \omterm{def:b2:metric:continuity}{متصلًا}، واقرأ
المجموعة بوصفها صورة عكسية، ثم استشهد بالمبرهنة.
\end{example}

\begin{example}[مجموعة مغلقة معرَّفة بعدد غير منته من الشروط]\label{ex:b2:metric:lipclosed}
في $\bigl(C(\intcc01), d_\infty\bigr)$، تكون المجموعة
\[
L = \{f : \abs{f(x) - f(y)} \leq \abs{x - y}
\ \text{من أجل كل } x, y\}
\]
المكوَّنة من الدوال الليبشيتزية بالثابت $1$ مغلقةً، رغم أنها مقتطَعة
بشروط غير \omterm{def:b2:structures:countable}{قابلة للعد}: فمن أجل كل زوج ثابت $(x, y)$، يكون
التطبيق $f \mapsto \abs{f(x) - f(y)} - \abs{x - y}$ \omterm{def:b2:metric:continuity}{متصلًا}
(فالتقييمات ليبشيتزية بالثابت $1$)، ومن ثم يعرّف كل شرط منفرد
مجموعة مغلقة، وتكون $L$ \emph{تقاطع}
هذه العائلة --- وأي تقاطع لمغلقات مغلق. ويصدّق القالب
نفسه انغلاقَ مجموعات الدوال الرتيبة والدوال المحدبة والدوال
المحدودة بعدد ثابت $g$: فالنهايات المنتظمة ترث كل خاصية
يمكن التعبير عنها بعائلة من القيود النقطية المغلقة. أما ما
\emph{لا} ترثه النهايات المنتظمة تلقائيًا --- كقابلية
الاشتقاق --- فهو بالضبط ما يجب أن يكدح \cref{ch:b2:funcseq} من أجله.
\end{example}

\begin{theorem}[مبرهنة النقطة الثابتة عند باناخ]\label{thm:b2:metric:banach}
ليكن $X$ فضاءً متريًا \omterm{def:b2:metric:complete}{تامًا} غير خالٍ وليكن $f \colon X \to X$
\emph{تطبيقًا متقلّصًا}: أي \omterm{def:b2:metric:continuity}{ليبشيتزيًا} بثابت $k < 1$. عندئذٍ يملك $f$
نقطة ثابتة وحيدة $\ell$، ويتقارب كل مدار $x_{n+1} = f(x_n)$
إلى $\ell$، مع\index{مبرهنة النقطة الثابتة عند باناخ}
\[
d(x_n, \ell) \leq \frac{k^n}{1 - k}\, d(x_1, x_0) .
\]
\end{theorem}

\begin{proof}
الوحدانية: المسافة بين نقطتين ثابتتين تساوي $\leq k$ مضروبًا في نفسها.
الوجود: $d(x_{n+1}, x_n) \leq k^n d(x_1, x_0)$ بالتراجع، ومن ثم
من أجل $q > p$،
\[
d(x_q, x_p) \leq \sum_{j=p}^{q-1} d(x_{j+1}, x_j)
\leq d(x_1, x_0) \sum_{j \geq p} k^j
= \frac{k^p}{1-k}\, d(x_1, x_0) \xrightarrow[p\to\infty]{} 0 :
\]
فهي متتالية كوشي، ومن ثم متقاربة إلى $\ell$ ما؛ ويُمرّر اتصال $f$
العلاقةَ $x_{n+1} = f(x_n)$ إلى النهاية: $\ell = f(\ell)$. وأما حدّ الخطأ
فهو التقدير المعروض مع $q \to \infty$.
\end{proof}

\begin{example}[معادلة تكاملية]\label{ex:b2:metric:integralequation}
على الفضاء التام $X = C(\intcc{0}{1})$ (مع
\cref{thm:b2:metric:rncomplete})، لننظر في $T(f)(x) = 1 + \frac12
\int_0^x f(t)\,\dd t$. من أجل $f, g \in X$:
\[
\abs{T(f)(x) - T(g)(x)} \leq \frac12 \int_0^x \abs{f - g} \leq
\frac12\, d_\infty(f, g),
\]
ومن ثم فإن $T$ متقلّص بالثابت $\frac12$: فله نقطة ثابتة متصلة
وحيدة --- وهي حلّ المعادلة $f' = \frac f2$، $f(0) = 1$، أي
$\eu^{x/2}$. وهذه الخطة، بعد تصنيعها، تصير
مبرهنةَ كوشي--ليبشيتز في \cref{ch:b2:diffeq}.
\end{example}

\begin{example}[نقطة ثابتة عددية: $x = \cos x$]\label{ex:b2:metric:cosfix}
على الفترة التامة $X = \intcc{0}{1}$، يرسل التطبيق $f = \cos$ المجموعةَ $X$
داخل $\intcc{\cos 1}{1} \subseteq X$ وهو متقلّص: فبمتراجحة
التزايدات المنتهية،
\[
\abs{\cos x - \cos y} \leq \bigl(\sup_{\intcc01}\abs{\sin}\bigr)
\abs{x - y} = (\sin 1)\abs{x - y},
\qquad \sin 1 \approx 0.841 < 1 .
\]
وبمبرهنة باناخ: حلّ وحيد للمعادلة $x = \cos x$ في $\intcc01$ (ومن ثم في
$\R$: فأي نقطة ثابتة حقيقية تقع في $\intcc{-1}{1}$، ثم في
$\intcc{\cos 1}{1}$ بعد تطبيق واحد)، ويتقارب التكرار
$x_{n+1} = \cos x_n$ إليه انطلاقًا من أي بداية: $x_\infty
\approx 0.739085$، وهو العدد الشهير الذي نحصل عليه بالنقر
المتواصل على زرّ جيب التمام في الآلة الحاسبة. ويتنبأ حدّ الخطأ
بتناقص من رتبة $(\sin1)^n/(1 - \sin1)$ --- أي نحو رقم واحد لكل $13$
نقرة؛ ويصدّق الحدّ البعدي في مسألة نهاية الأسبوع في هذا الفصل
(السؤال 14) كل خطوة أثناء الحساب.
\end{example}

\section{التراص}

\begin{definition}\label{def:b2:metric:compact}
يكون \omterm{def:b2:metric:def}{الفضاء المتري} $X$ \emph{متراصًا}\index{فضاء متري متراص}
إذا كان لكل متتالية في $X$ متتالية جزئية متقاربة \emph{في}
$X$ (وهي خاصية بولزانو--فايرشتراس). ويكون الجزء متراصًا إذا كان
كذلك بالمسافة المستحثّة.
\end{definition}

\begin{theorem}[الخواص الأولى]\label{thm:b2:metric:compactprops}
\begin{enumerate}
  \item الجزء المتراص مغلق ومحدود؛ والجزء المغلق من فضاء
        متراص متراص.
  \item في $\R^n$، يصحّ العكس: متراص $\iff$ مغلق
        ومحدود.
  \item الصورة المتصلة لفضاء متراص متراصة؛ والدالة الحقيقية
        المتصلة على فضاء متراص غير خالٍ محدودة وتبلغ
        حدّيها.
  \item (هاينه) التطبيق \omterm{def:b2:metric:continuity}{المتصل} على فضاء متراص \omterm{def:b2:metric:continuity}{متصل} بانتظام.
  \item الجداءات: إذا كان $X, Y$ متراصين، فكذلك $X \times Y$ (مع
        $d\bigl((x,y),(x',y')\bigr) = d(x,x') + d(y,y')$).
\end{enumerate}
\end{theorem}

\begin{proof}
(1) الحجج نفسها المستعملة على المستقيم (مجلد السنة الأولى): فالمتتالية
الهاربة إلى % أو المتقاربة إلى الخارج ليس لها متتالية جزئية
متقاربة في الداخل؛ وأما القول الثاني: فنستخرج في المتراص
المحيط ونستعمل الانغلاق.

(2) للمتتاليات المحدودة في $\R^n$ متتاليات جزئية متقاربة
مركّبةً مركّبة: فنستخرج على الإحداثية الأولى
(بولزانو--فايرشتراس على $\R$)، ثم من تلك المتتالية الجزئية على
الثانية، وهكذا ($n$ استخراجًا متتاليًا)؛ ويُبقي الانغلاق
النهايةَ في الداخل.

(3) من أجل $(f(x_n))$، نستخرج $x_{\varphi(n)} \to x \in X$؛
ويعطي الاتصال $f(x_{\varphi(n)}) \to f(x) \in f(X)$. وأما الحالة الحقيقية:
فالتراص يجعل $f(X) \subseteq \R$ مغلقة ومحدودة، ويكون
$\sup f(X) \in f(X)$ (فالنهاية العليا لمجموعة ملتصقة بها، و
$f(X)$ مغلقة).

(4) ينتقل برهان السنة الأولى حرفيًا؛ وها هو بلباس متري.
لنفترض أن $f \colon X \to Y$ \omterm{def:b2:metric:continuity}{متصل} على المتراص $X$
وغير \omterm{def:b2:metric:continuity}{متصل} بانتظام: فيوجد $\varepsilon > 0$ يقبل، من أجل
كل $n$، نقطتين تحققان
\[
d_X(x_n, y_n) \leq \frac{1}{n+1}
\qquad\text{و}\qquad
d_Y\bigl(f(x_n), f(y_n)\bigr) > \varepsilon .
\]
نستخرج $x_{\varphi(n)} \to a \in X$؛ عندئذٍ $y_{\varphi(n)} \to
a$ أيضًا (لأن المسافة بينهما تؤول إلى $0$). ويرسل الاتصال عند $a$
متتاليتَي الصور كلتيهما إلى $f(a)$، ومنه
$d_Y\bigl(f(x_{\varphi(n)}), f(y_{\varphi(n)})\bigr) \to 0$ ---
وهذا يناقض الفجوة المنتظمة $> \varepsilon$. ولم يقدّم التراص
إلا شيئًا واحدًا: نقطة التراكم $a$ التي نطبّق عندها
الاتصال العادي.

(5) نستخرج على الإحداثيات في $X$، ثم من جديد على
الإحداثيات في $Y$.
\end{proof}

\begin{example}[مبرهنة هاينه، مع التراص وبدونه]\label{ex:b2:metric:heinework}
على $\intcc{0}{1}$، تكون الدالة $x \mapsto x^2$ متصلة بانتظام ---
وتقول مبرهنة هاينه ذلك بلا حساب، لكن التقدير المباشر
مفيد:
\[
\abs{x^2 - y^2} = \abs{x + y}\,\abs{x - y} \leq 2\abs{x - y},
\]
ومن ثم يفي $\delta = \varepsilon/2$ \emph{بكل النقاط
دفعةً واحدة}. أما على $\R$ فالدالة نفسها ليست متصلة بانتظام:
فمع $x_n = n$ و$y_n = n + \frac1n$، تكون الفجوة $\abs{x_n - y_n}
= \frac1n \to 0$ في حين أن $\abs{x_n^2 - y_n^2} = 2 + \frac1{n^2}
\geq 2$: فلا يوجد $\delta$ واحد يفي بالاختيار $\varepsilon = 1$. والآلية
ظاهرة: فالثابت \omterm{def:b2:metric:continuity}{الليبشيتزي} المحلي $\abs{x + y}$
محدود على متراص وغير محدود على $\R$ --- ومبرهنة
هاينه هي بالضبط القول بأن التراص يضع سقفًا منتظمًا
لمثل هذه الثوابت المحلية.
\end{example}

\begin{method}[كيف نبرهن على أن مجموعة متراصة]\label{met:b2:metric:compactness}
ثلاثة طرق، مرتبةً بحسب التواتر. (1) \emph{التعرّف في الفضاء
المحيط:} في $\R^n$ (أو في أي فضاء معياري منتهي البعد،
\cref{ch:b2:nvs})، تحقّق من الانغلاق --- عادةً بوصفها
صورة عكسية، \cref{ex:b2:metric:recognize} --- ومن كونها محدودة. (2)
\emph{الوراثة:} الجزء المغلق من متراص معلوم
متراص؛ واتحاد منته أو جداء لمتراصات متراص؛ والصورة
المتصلة لمتراص متراصة. (3) \emph{باليدين
المجرّدتين:} استخرج متتالية جزئية متقاربة من متتالية
كيفية --- عادةً باستخراجات متتالية إحداثيةً إحداثية.
وللبرهان على \emph{عدم} التراص يكفي شاهد واحد:
متتالية بلا متتالية جزئية متقاربة، وغالبًا
نقاط المسافة بينها $\geq \varepsilon$.
\end{method}

\begin{example}[المسافات بين المجموعات: التراص يستحق أجره]\label{ex:b2:metric:setdistance}
لتكن $K$ متراصة و$F$ مغلقة، مع $K \cap F = \emptyset$، في \omterm{def:b2:metric:def}{فضاء
متري}. عندئذٍ
\[
d(K, F) = \inf\,\{d(x, y) : x \in K,\ y \in F\} > 0 :
\]
فالدالة $x \mapsto d(x, F)$ متصلة
(\cref{exo:b2:metric:11}) وموجبة على $K$ (إذ إن $d(x, F) = 0$
يضع $x \in \overline F = F$)، ومن ثم تبلغ حدًّا أدنى موجبًا
على المتراص $K$
(\cref{thm:b2:metric:compactprops}~(3)). وليس التراص
زينة: فمن أجل مجموعتين \emph{مغلقتين} قد ينعدم الحدّ الأدنى
دون أن يُبلغ --- ففي $\R^2$، القطعُ الزائد $F_1 = \{xy
= 1\}$ والمحور $F_2 = \{y = 0\}$ مجموعتان مغلقتان منفصلتان
مع $d(F_1, F_2) = 0$ (فالنقاط $(n, \frac1n)$ تقترب من
المحور). والهرب إلى اللانهاية هو بالضبط ما يمنعه التراص.
\end{example}

\begin{theorem}[بوريل--لوبيغ]\label{thm:b2:metric:borellebesgue}
يكون \omterm{def:b2:metric:def}{الفضاء المتري} $X$ \omterm{def:b2:metric:compact}{متراصًا} إذا وفقط إذا كان لكل تغطية للفضاء $X$
بمفتوحات تغطيةٌ جزئية \emph{منتهية}.\index{مبرهنة
بوريل--لوبيغ}
\end{theorem}

\begin{proof}
($\Leftarrow$) لنفترض أنه ليس للمتتالية $(x_n)$ متتالية جزئية متقاربة. وندّعي
أن لكل $x \in X$ كرةً $B(x, r_x)$ تحتوي $x_n$
من أجل عدد منته فقط من الأدلة $n$: وإلا لاحتوت كل كرة
$B(x, \frac1{k+1})$ عددًا غير منته من الحدود، ولبنى اختيار
الأدلة
\[
\varphi(0) < \varphi(1) < \varphi(2) < \cdots
\quad\text{مع}\quad
x_{\varphi(k)} \in B\Bigl(x, \frac{1}{k+1}\Bigr)
\]
(وهو ممكن عند كل خطوة بالضبط لأن عددًا غير منته من المرشحين
يبقى) متتاليةً جزئية متقاربة إلى $x$.
وتغطي الكرات $B(x, r_x)$ الفضاءَ $X$؛ فلو غطّى عدد منته منها
$X$، لكانت مجموعة الأدلة $\N$ اتحادًا منتهيًا
لمجموعات منتهية: وهذا محال.

($\Rightarrow$) خطوتان. \emph{عدد لوبيغ:} من أجل تغطية \omterm{def:b2:metric:topology}{مفتوحة}
$(U_i)$ لمتراص $X$، يوجد $\rho > 0$ بحيث تقع كل كرة
نصف قطرها $\rho$ داخل $U_i$ ما. وإلا، فنختار من أجل كل $n$
عنصرًا $x_n$ بحيث لا تقع $B(x_n, \frac{1}{n+1})$ في أي $U_i$؛ ونستخرج
$x_{\varphi(n)} \to x \in U_{i_0} \supseteq B(x, r)$؛ ومن أجل $n$ الكبيرة،
$B(x_{\varphi(n)}, \frac{1}{\varphi(n)+1}) \subseteq B(x, r)
\subseteq U_{i_0}$: وهو تناقض. \emph{والحدّ الكلي:} من أجل
كل $\rho > 0$، تغطي كراتٌ منتهية العدد نصف قطرها $\rho$ الفضاءَ $X$.
وإلا فنختار بالتراجع $x_{n+1}$ خارج $B(x_0, \rho) \cup
\dots \cup B(x_n, \rho)$: فتكون المسافات المتبادلة في المتتالية $\geq
\rho$، ومن ثم فلا متتالية جزئية كوشي --- ولا متقاربة إذن:
وهو تناقض. وبالجمع: نغطي $X$ بعدد منته من الكرات نصف قطرها
$\rho$ (وهو عدد لوبيغ)، وكلٌّ منها داخل $U_i$ ما: فنحصل على تغطية
جزئية منتهية.
\end{proof}

\begin{example}[شبكة $\varepsilon$، معدودة]\label{ex:b2:metric:epsnet}
الحدّ الكلي (من برهان
\cref{thm:b2:metric:borellebesgue}) ملموس جدًا على
$\intcc{0}{1}$: فمن أجل $\varepsilon > 0$، تغطيه الكرات $\lceil
\frac{1}{2\varepsilon}\rceil$ ذوات المراكز $\varepsilon,
3\varepsilon, 5\varepsilon, \dots$ ونصف القطر $\varepsilon$
--- أي نحو $\frac1{2\varepsilon}$ كرة، ولا تستطيع أي تغطية
أن تكتفي بأقلّ من $\frac{1}{2\varepsilon}$ منها (فكل
كرة تغطي طولًا لا يتجاوز $2\varepsilon$). وفي $\intcc01^2$
يتربّع العدد فيصير من رتبة $\varepsilon^{-2}$: فأعداد التغطية
تنمو مثل $\varepsilon^{-d}$ في البعد $d$ --- وهو وجه كمّي
للتراص، والسبب في أن كرات الوحدة غير المنتهية البعد في
\cref{ch:b2:nvs} (حيث لا توجد أي شبكة
$\frac13$ منتهية) لا يمكن أن تكون متراصة.
\end{example}

\begin{example}[قراءة التراص على التغطيات]\label{ex:b2:metric:covers}
الفترة نصف \omterm{def:b2:metric:topology}{المفتوحة} $\intoc{0}{1}$ مغطّاة بالمفتوحات
$U_n = \intoo{\frac1n}{2}$، مع $n \geq 1$؛ ولأي عائلة جزئية منتهية
دليل أعظم $N$ فتفوتها $\intoc{0}{\frac1N}$: فلا تغطية
جزئية منتهية، ومن ثم فإن $\intoc{0}{1}$ غير متراص --- وهو ما يراه
التعريف المتتالي عبر $x_n = \frac1n$، التي تهرب نهايتها
$0$. وفي المقابل، تُصلح إضافة النقطة الواحدة $0$
التشخيصين معًا: فعلى $\intcc{0}{1}$ يجب أن تحتوي كل تغطية كهذه
مجموعةً تحتوي $0$، وهذه تبتلع قطعة ابتدائية كاملة،
ويُنهي عدد منته من المجموعات الباقي. ولغتا
\cref{thm:b2:metric:borellebesgue} تفشلان أو تنجحان معًا
دائمًا --- فالتغطيات تكشف الهرب حيث تكشفه المتتاليات بالضبط.
\end{example}

\begin{remark}[آفاق داخل هذا المجلد]\label{rem:b2:metric:perspectives}
هذا الفصل هو الجدار الحامل في المجلد؛ فراقب أين يحمل
كل ركن ثقلًا. \emph{التمام}: يصير محك كوشي محكَّ تقارب
المتسلسلات في فضاءات باناخ (\cref{ch:b2:series})، والتقارب
المنتظم في \cref{ch:b2:funcseq} هو بالضبط التقارب في الفضاء
التام $\bigl(C, d_\infty\bigr)$، ومبرهنة كوشي--ليبشيتز
(\cref{ch:b2:diffeq}) هي مبرهنة النقطة الثابتة عند باناخ
بلباس معادلة تكاملية. \emph{التراص}: يبرهن على
تكافؤ المعايير (\cref{ch:b2:nvs})، وعلى بلوغ
الحدود القصوى في تحسين \cref{ch:b2:diffcalc}، وعلى
وجود أفضل التقريبات (مسألة نهاية الأسبوع في \cref{ch:b2:nvs}).
\emph{الترابط}: يعمّم الأقوال المحلية
--- وحدانية حلول المعادلات التفاضلية، ومبرهنة القيم
الوسطى على المنحنيات (\cref{ch:b2:curves})، والمركّبتان
في $GL_n(\R)$ اللتان ستُبقيهما نظرية التوجيه
(\cref{ch:b2:multint}) منفصلتين.
\end{remark}

\begin{remark}[مزالق شائعة]\label{rem:b2:metric:pitfalls}
(1) قولُ ``مغلقة ومحدودة يستلزم متراصة'' مبرهنة عن
$\R^n$، لا عن \omterm{def:b2:metric:def}{الفضاءات المترية}: فالمجموعة غير المنتهية بالمسافة
المتقطعة مغلقة ومحدودة في نفسها ومع ذلك ليست متراصة
(\cref{exo:b2:metric:4})، وكذلك تفشل كرة الوحدة المغلقة في
$C(\intcc01)$ (\cref{ch:b2:nvs}). (2) التمام خاصية
\emph{المسافة} لا الطوبولوجيا: فالمجموعة $\R$
مع $d(x,y) = \abs{\arctan x - \arctan y}$ لها المتتاليات المتقاربة
المعتادة لكنها غير تامة
(\cref{exo:b2:metric:1}). (3) لا يلزم أن يكون التقابل \omterm{def:b2:metric:continuity}{المتصل}
تماثلًا طوبولوجيًا --- كتمثيل الدائرة في
\cref{exo:b2:metric:7}؛ ويُصلح تراصُ المنطلق ذلك.
(4) تحتاج مبرهنة باناخ إلى $k < 1$ \emph{بانتظام}: فالشرط
$d(f(x), f(y)) < d(x,y)$ وحده لا يضمن شيئًا على فضاء
غير متراص (\cref{exo:b2:metric:5}). (5) المترابط لا
يستلزم المترابط بالأقواس عمومًا --- لكن من أجل الأجزاء
\omterm{def:b2:metric:topology}{المفتوحة} من الفضاءات المعيارية التي نلقاها في هذا الكتاب
يتطابق المفهومان (\cref{ch:b2:nvs}).
\end{remark}

\begin{remark}[أين يُستعمَل هذا الفصل]
في كل مكان من النصف التحليلي. فتمام
$C(\intcc{a}{b})$ يُشغّل مبرهنات التقارب في
\cref{ch:b2:funcseq} ونظريةَ كوشي--ليبشيتز في
\cref{ch:b2:diffeq} (وتبرهن مسألة نهاية الأسبوع في هذا الفصل على
مبرهنة بيكار--ليندلوف المحلية بالفعل)؛ ويعطي التراص
تكافؤ المعايير في البعد المنتهي (\cref{ch:b2:nvs}) و
وجود الحدود القصوى في \cref{ch:b2:diffcalc}؛ ويقوم الترابط
تحت حجج القيم الوسطى في \cref{ch:b2:realfun}
وتحت تعميم الوحدانية للمعادلات التفاضلية. وفي
مجلد السنة الثالثة، يصير التراص في فضاءات الدوال
(مبرهنة أرزيلا--أسكولي) ومبرهنة بير في التصنيف
(\cref{exo:b2:metric:12} هنا) أدواتٍ للاستعمال اليومي.
\end{remark}

\section{الترابط}

\begin{definition}\label{def:b2:metric:connected}
يكون $X$ \emph{مترابطًا}\index{فضاء مترابط} إذا لم يقبل أي
تقسيم إلى جزأين مفتوحين غير خاليين --- وبكيفية مكافئة، إذا كانت
أجزاؤه الوحيدة \omterm{def:b2:metric:topology}{المفتوحة} والمغلقة معًا هي $\emptyset$ و$X$. ويكون $X$
\emph{مترابطًا بالأقواس} إذا كان كل نقطتين فيه موصولتين بتطبيق \omterm{def:b2:metric:continuity}{متصل}
$\gamma \colon \intcc{0}{1} \to X$.
\end{definition}

\begin{theorem}\label{thm:b2:metric:connectedness}
\begin{enumerate}
  \item الأجزاء المترابطة من $\R$ هي بالضبط الفترات.
  \item الصورة المتصلة \omterm{def:b2:metric:connected}{لفضاء مترابط} مترابطة ---
        ومنها مبرهنة القيم الوسطى العامة: تأخذ الدالة الحقيقية
        المتصلة على \omterm{def:b2:metric:connected}{فضاء مترابط} كل قيمة بين
        أي قيمتين من قيمها.
  \item المترابط بالأقواس $\Rightarrow$ مترابط. (والعكس خاطئ
        عمومًا؛ لكنه يصحّ من أجل الأجزاء \omterm{def:b2:metric:topology}{المفتوحة} من الفضاءات المعيارية،
        \cref{ch:b2:nvs}.)
\end{enumerate}
\end{theorem}

\begin{proof}
(1) المجموعة $A$ التي ليست فترةً تفوتها نقطة $z$ بين نقطتين منها:
فالتقسيم
$A = (A \cap \intoo{-\infty}{z}) \cup (A \cap \intoo{z}{+\infty})$
يقسمها إلى جزأين غير خاليين مفتوحين (في $A$). وعكسيًا، لتكن
$I$ فترةً وليكن $I = U \cup V$ تقسيمًا إلى مجموعتين غير خاليتين
مفتوحتين نسبيًا؛ ونختار $a \in U$، $b \in V$، ولنفترض $a < b$،
ونضع $s = \sup\,(U \cap \intcc{a}{b})$، وهي نقطة من $\intcc{a}{b}
\subseteq I$. فإذا كان $s \in U$: فإن $s \neq b$، ويضع الانفتاح النسبي
للمجموعة $U$ فترةً كاملةً حول $s$ (متقاطعةً مع $I$)
داخل $U$ --- ومن ثم توجد في $U \cap \intcc{a}{b}$ نقاط تتجاوز $s$،
وهذا يناقض النهاية العليا. وإذا كان $s \in V$: فإن الانفتاح النسبي للمجموعة $V$
يضع فترةً $\intoo{s - r}{s + r} \cap I$ داخل $V$؛ لكن النهاية
العليا ملتصقة بالمجموعة $U \cap \intcc{a}{b}$، التي يجب أن تلتقي تلك
الفترة --- وهذا يناقض $U \cap V = \emptyset$. (وهذه هي
حجة السنة الأولى في المفتوح المغلق من أجل $\R$، مطبَّقة داخل $I$.)

(2) إذا انقسمت $f(X) = U' \cup V'$ إلى مجموعتين غير خاليتين مفتوحتين
نسبيًا، فإن $X = f^{-1}(U') \cup f^{-1}(V')$ تقسم $X$
(\cref{thm:b2:metric:globalcontinuity}). وأما مبرهنة القيم الوسطى: فالمجموعة $f(X) \subseteq \R$
مترابطة، ومن ثم فهي فترة حسب (1).

(3) لنفترض $X = U \cup V$، وكلتاهما غير خالية \omterm{def:b2:metric:topology}{ومفتوحة}، ولنصل $a \in U$
بالنقطة $b \in V$ بقوس $\gamma$: عندئذٍ تقسم $\gamma^{-1}(U),
\gamma^{-1}(V)$ المجموعةَ $\intcc{0}{1}$، وهذا يناقض (1).
\end{proof}

\begin{example}\label{ex:b2:metric:glnr}
$GL_n(\R)$ غير مترابطة: \omterm{def:b2:linalg:det}{فالمحدد} $\det$ \omterm{def:b2:metric:continuity}{متصل} (وهو كثير حدود في
العناصر) على $\R^*$، وهذه غير مترابطة؛ وتقسم الصورتان العكسيتان
للمجموعتين $\R_+^*$ و$\R_-^*$ المجموعةَ $GL_n(\R)$. (وكل قطعة مترابطة بالأقواس
في الواقع --- وهو تمرين \omterm{def:b2:reduction:eigen}{لطيف} يتجاوز حاجتنا.) وفي
المقابل، $GL_n(\C)$ \emph{مترابطة} بالأقواس:
\cref{exo:b2:metric:10}.
\end{example}

\begin{example}[نقطة ثابتة من الترابط وحده]\label{ex:b2:metric:ivtfixed}
لكل تطبيق \omterm{def:b2:metric:continuity}{متصل} $f \colon \intcc01 \to \intcc01$ نقطة
ثابتة --- بلا فرضية تقلّص وبلا تكرار. لننظر في
$g(x) = f(x) - x$، وهي متصلة على المترابط $\intcc01$:
\[
g(0) = f(0) \geq 0,
\qquad
g(1) = f(1) - 1 \leq 0 ,
\]
وتعطي مبرهنة القيم الوسطى
(\cref{thm:b2:metric:connectedness}~(2)) جذرًا للدالة
$g$، أي نقطة ثابتة للتطبيق $f$. وقارِن بمبرهنة باناخ
(\cref{thm:b2:metric:banach}): فالوجود هنا طوبولوجي
ومجاني، لكن الوحدانية والخوارزمية ضائعتان --- فللتطبيق $f =
\mathrm{id}$ كل نقطة ثابتة، وقد يدور تكرار تطبيق $f$
غير متقلّص إلى الأبد. ومبرهنتا النقطة الثابتة في هذا الفصل
تجيبان عن سؤالين مختلفين بعملتين
مختلفتين.
\end{example}

\begin{example}[$\R$ و$\R^2$ غير متماثلين طوبولوجيًا]\label{ex:b2:metric:rnotr2}
الترابط بصمة طوبولوجية. لنفترض أن $h \colon
\R^2 \to \R$ تماثل طوبولوجي (أي تقابل \omterm{def:b2:metric:continuity}{متصل} ذو مقلوب
\omterm{def:b2:metric:continuity}{متصل}). ونحذف نقطة واحدة $a \in \R^2$: فيبقى
المقصور $h \colon \R^2\setminus\{a\} \to
\R\setminus\{h(a)\}$ تماثلًا طوبولوجيًا. لكن $\R^2$ ناقصَ
نقطة مترابط بالأقواس --- فنصل أي نقطتين بقطعة،
مع الالتفاف على قطعة ثانية عبر نقطة مساعدة إذا
اعترضت $a$ الطريقَ المباشر --- ومن ثم فهو مترابط
(\cref{thm:b2:metric:connectedness}~(3))؛ في حين أن $\R$ ناقصَ
نقطة ينقسم إلى نصفَي مستقيم مفتوحين غير خاليين: فهو غير مترابط.
والترابط محفوظ بالتطبيقات المتصلة: وهو تناقض.
فالمستوي والمستقيم مختلفان فعلًا \emph{بوصفهما فضاءين
طوبولوجيين} --- وهي واقعة لا تراها القوة وحدها
(فالتقابلات من نمط \cref{exo:b2:structures:3} موجودة فعلًا!) لشدة
خشونتها.
\end{example}

\begin{exercise}[$\star\star$]\label{exo:b2:metric:11}
من أجل $\emptyset \neq A \subseteq X$، نضع $d(x, A) = \inf_{a \in A}

منفصلتين.
\end{exercise}

\end{exercise}

\section{مسألة: تكرار بيكار}

التمام مع التقلّص آلةُ حلّ: أطعِمها معادلة مكتوبة
على صورة مسألة نقطة ثابتة، فتردّ عليك
بالوجود والوحدانية وخوارزمية وأشرطة خطأ. وتُشغّل مسألة نهاية
الأسبوع هذه الآلة بكامل طاقتها على المعادلة $y' = f(t,
y)$: فنبرهن على مبرهنة \emph{بيكار--ليندلوف} المحلية (وهي

التعلّق \omterm{def:b2:metric:continuity}{المتصل} بالمعطيات، ومعادلة كبلر،
وتشابه مجموعة كانتور مع نفسها.

\begin{problem}[{مسألة نهاية الأسبوع --- مبرهنة
بيكار--ليندلوف}]
\label{pb:b2:metric:1}
في كل ما يلي، $t_0 \in \R$، $y_0 \in \R$، $a, b > 0$، و$f$
دالة متصلة على المستطيل $R = \intcc{t_0 - a}{t_0 +
a} \times \intcc{y_0 - b}{y_0 + b}$، محدودة بالعدد $M =
\sup_R\,\abs f$، و\emph{ليبشيتزية بالثابت $L$ في متغيّرها الثاني}:
$\abs{f(t, y) - f(t, z)} \leq L\abs{y - z}$ كلما وقعت النقطتان
في $R$. ونضع
\[
h = \min\Bigl(a, \frac bM\Bigr) \quad (\text{مع } h = a
\text{ إذا كان } M = 0), \qquad I = \intcc{t_0 - h}{t_0 + h}.
\]

\medskip
\noindent\textbf{الجزء الأول --- المسرح التام.}
\begin{enumerate}
  \item برهن على القولين المذكورين في
        \cref{def:b2:metric:complete}: الجزء المغلق من \omterm{def:b2:metric:complete}{فضاء
        متري تام} تام، والجزء التام من أي \omterm{def:b2:metric:def}{فضاء متري}
        مغلق. استنتج أن كل جزء مغلق
        من $\bigl(C(I), d_\infty\bigr)$ \omterm{def:b2:metric:complete}{فضاء متري
        تام}.
  \item بيّن أن التطبيق $C(I) \to C(I)$، $y \mapsto \bigl(t
        \mapsto \int_{t_0}^{t}y(s)\,\dd s\bigr)$،
        \omterm{def:b2:metric:continuity}{ليبشيتزي} بالثابت $h$ من أجل $d_\infty$.
  \item (النقاط الثابتة تتحرك أقلّ من التطبيقات) ليكن $g \colon X
        \to X$ تطبيقًا متقلّصًا بالثابت $k$ \omterm{def:b2:metric:def}{لفضاء متري} ونقطته
        الثابتة $\ell_g$، وليكن $\widetilde g \colon X \to X$
        تطبيقًا \emph{كيفيًا} له نقطة ثابتة $\ell_{\widetilde g}$.
        برهن على أن
        \[
        d(\ell_g, \ell_{\widetilde g}) \leq
        \frac{d\bigl(g(\ell_{\widetilde g}),
        \widetilde g(\ell_{\widetilde g})\bigr)}{1 - k}
        \leq \frac{\sup_{x \in X} d\bigl(g(x), \widetilde
        g(x)\bigr)}{1 - k}.
        \]
  \item (حيلة التكرار) ليكن $X$ \omterm{def:b2:metric:complete}{تامًا} غير خالٍ وليكن $g
        \colon X \to X$ تطبيقًا --- لا يُفترض \omterm{def:b2:metric:continuity}{متصلًا} ---
        بحيث يكون تكرارٌ ما $g^m$ متقلّصًا بالثابت $k$.
        برهن على أن للتطبيق $g$ نقطة ثابتة وحيدة $\ell$ وأن
        \emph{كل} مدار $x_{n+1} = g(x_n)$ يتقارب إلى
        $\ell$. \emph{(النقاط الثابتة للتطبيق $g$ نقاط ثابتة للتطبيق
        $g^m$؛ وعكسيًا $g(\ell)$ نقطة ثابتة للتطبيق $g^m$؛
        وقسّم المدار حسب البواقي بترديد $m$.)}
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- مبرهنة بيكار--ليندلوف.}
\begin{enumerate}[resume]
  \item بيّن أن الدالة $y \colon I \to \intcc{y_0 - b}{y_0
        + b}$ تكون $C^1$ مع $y(t_0) = y_0$ و$y' = f(t, y)$ على
        $I$ إذا وفقط إذا كانت متصلة وحققت
        المعادلة التكاملية
        \[
        y(t) = y_0 + \int_{t_0}^{t} f\bigl(s, y(s)\bigr)\dd s
        \qquad (t \in I).
        \]
  \item ليكن $X_h = \{y \in C(I) : \abs{y(t) - y_0} \leq b
        \text{ على } I\}$ وليكن $T$ معرَّفًا بالعلاقة $T(y)(t) =
        y_0 + \int_{t_0}^{t}f(s, y(s))\dd s$. بيّن أن $X_h$
        جزء مغلق غير خالٍ من $C(I)$، ومن ثم فهو تام،
        وأن $T$ يرسل $X_h$ داخل $X_h$ --- وهنا يعمل $h \leq
        b/M$.
  \item بيّن أن $d_\infty\bigl(T(y), T(z)\bigr) \leq
        Lh\,d_\infty(y, z)$ على $X_h$: فإذا كان $Lh < 1$، فإن
        مبرهنة باناخ تختم فورًا. وسنزيل شرط الصغر هذا
        فيما يلي.
  \item برهن بالتراجع على $n$ أن:
        \[
        \abs{T^n(y)(t) - T^n(z)(t)} \leq
        \frac{\bigl(L\abs{t - t_0}\bigr)^n}{n!}\,
        d_\infty(y, z) \qquad (y, z \in X_h,\ t \in I),
        \]
        ومن ثم فإن تكرارًا ما للتطبيق $T$ متقلّص. واختم بالسؤال
        4 (\emph{مبرهنة بيكار--ليندلوف}): لمسألة
        كوشي $y' = f(t,y)$، $y(t_0) = y_0$ حلٌّ واحد بالضبط
        على $I = \intcc{t_0 - h}{t_0 + h}$ بقيم في
        $\intcc{y_0 - b}{y_0 + b}$.
  \item بيّن أن القيد ``بقيم في $\intcc{y_0 -
        b}{y_0 + b}$'' تلقائي: فكل حلّ لمسألة
        كوشي معرَّف على $I$ ويبدأ منحنيه في $R$ يبقى
        في $\intcc{y_0 - b}{y_0 + b}$ \emph{(انظر في أول زمن خروج
        وحُدّ $\abs{y(t) - y_0}$ بالمقدار $M\abs{t -
        t_0}$)}. ومنه فالوحدانية صحيحة بين كل الحلول على
        $I$.
  \item شغّل الآلة على $y' = y$، $y(0) = 1$، انطلاقًا من
        الدالة الثابتة $y^{(0)} \equiv 1$: احسب تكرارات
        بيكار $y^{(n)}$، وعيّنها، وصِف
        التقارب.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- كل فرضية تستحق أجرها.}
\begin{enumerate}[resume]
  \item (تفشل الليبشيتزية، فتفشل الوحدانية) من أجل $y' =
        2\sqrt{\abs y}$، $y(0) = 0$: تحقق من أن $y \equiv 0$
        وأن الدوال، من أجل كل $c \geq 0$، المعرَّفة بالعلاقة $y_c(t) = 0$ من أجل
        $t \leq c$، وبالعلاقة $y_c(t) = (t - c)^2$ من أجل $t > c$، كلها
        حلول من الصنف $C^1$ على $\R$. وأين بالضبط تفشل ليبشيتزية $y \mapsto
        2\sqrt{\abs y}$؟
  \item (المحلية حقيقية) من أجل $y' = y^2$، $y(0) = 1$: حُلّ
        صراحةً، وأعطِ الفترة الأعظمية للوجود،
        واحسب أفضل $h$ تستطيع المبرهنة تصديقه على كل
        اختيارات المستطيل (مع $a$ كبيرة و$b$ حرّة): بيّن أن
        $h_{\max} = \sup_{b>0} \frac{b}{(1+b)^2} = \frac14$،
        في حين أن الحلّ الحقيقي يعيش على $\intoo{-\infty}{1}$.
  \item (التمام ليس زينة) على $X = \Q \cap
        \intcc{1}{2}$ مع المسافة المعتادة، ليكن $g(x) =
        \frac x2 + \frac1x$. بيّن أن $g(X) \subseteq X$، وأن $g$
        متقلّص بالثابت $\frac12$ \emph{(بمتراجحة التزايدات
        المنتهية)}، وأنه ليس للتطبيق $g$ نقطة ثابتة في $X$.
        وأي فرضيات مبرهنة باناخ تفشل، وما النقطة الثابتة
        في التمام؟
  \item (أشرطة الخطأ) من أجل تطبيق متقلّص $g$ بالثابت $k$ على فضاء
        تام، برهن على التقدير \emph{البعدي} $d(x_n,
        \ell) \leq \frac{k}{1-k}\,d(x_n, x_{n-1})$. ومن أجل تطبيق هيرون
        $g(x) = \frac x2 + \frac 1x$ على $\intcc{1}{2}$
        (بنقطة ثابتة $\sqrt2$)، انطلاقًا من $x_0 = \frac32$:
        كم خطوة يطلب الحدّ \emph{القبلي}
        $\frac{k^n}{1-k}d(x_1, x_0)$ من أجل دقة
        $10^{-6}$، وكم خطوة تكفي في الواقع؟
        (احسب $x_1, x_2, x_3$ وأخطاءها؛ فحدّ التقلّص
        أمين لكنه متشائم --- إذ يتقارب هيرون
        تربيعيًا.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- التعلّق \omterm{def:b2:metric:continuity}{المتصل}.} في هذا
الجزء $Lh < 1$، ومن ثم فإن $T$ نفسه متقلّص على $X_h$ (السؤال
7)؛ ونكتب $y[\,y_0\,]$ للحلّ ذي القيمة الابتدائية
$y_0$.
\begin{enumerate}[resume]
  \item (التعلّق بالقيمة الابتدائية) لتكن $z_0$ قيمة
        ابتدائية أخرى مع $\abs{z_0 - y_0}$ صغيرة بما يكفي
        لتقع المسألتان في المستطيل. وباستعمال السؤال 3،
        برهن على أن
        \[
        d_\infty\bigl(y[y_0], y[z_0]\bigr) \leq
        \frac{\abs{y_0 - z_0}}{1 - Lh}.
        \]
  \item (الفترات الطويلة بالتسلسل) لنفترض أن الحلول موجودة
        على قطعة طويلة مقسَّمة إلى $m$ قطع متتالية ينطبق
        على كل منها الحدّ السابق بالعامل $Lh \leq
        \frac12$. بيّن أن الانحراف ينمو بعامل لا يتجاوز
        $2$ لكل قطعة، ومن ثم $d_\infty \leq 2^m\abs{y_0 - z_0}$
        إجمالًا --- وهو حدّ أسّي في الطول، وهو الظلّ
        المتقطع لعامل $\eu^{L\abs{t - t_0}}$ في مبرهنة غرونوال المساعدة
        (\cref{ch:b2:diffeq}).
  \item (التعلّق بالحقل) ليكن $g$ حقلًا آخر على
        $R$، \omterm{def:b2:metric:continuity}{ليبشيتزيًا} بالثابت $L$ في $y$ أيضًا، مع $\sup_R \abs{f - g}
        \leq \varepsilon$. برهن على أن الحلّين
        الموافقين يحققان $d_\infty \leq
        \frac{\varepsilon h}{1 - Lh}$: أي إن خطأ النمذجة
        ينتشر خطيًا.
  \item (الوسائط) إذا كانت عائلة $f_\lambda$ من الحقول
        ليبشيتزية بالثابت $L$ في $y$ بانتظام و$\sup_R\abs{f_\lambda
        - f_\mu} \leq C\abs{\lambda - \mu}$، فاستنتج أن
        $\lambda \mapsto y_\lambda$ \omterm{def:b2:metric:continuity}{ليبشيتزي} من فضاء الوسائط
        إلى $\bigl(C(I), d_\infty\bigr)$.
  \item (الجمل لا تكلّف شيئًا) اشرح لماذا تصحّ الأجزاء الأول والثاني والرابع
        حرفيًا من أجل $y$ بقيم في $\R^n$ (بمسافات
        عليا مبنية على أي من مسافات
        \cref{ex:b2:metric:examples})، ثم احسب كل تكرارات بيكار
        للجملة $y' = Ay$، $y(0) = (c_1,
        c_2)$، $A = \begin{psmallmatrix} 0 & 1\\ 0 &
        0\end{psmallmatrix}$: بيّن أن التكرار يصير
        ساكنًا عند الحلّ الدقيق بعد خطوة واحدة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- الأرباح المترية والتركيب.}
\begin{enumerate}[resume]
  \item (اضطراب المطابق) ليكن $X$ مسرحًا \omterm{def:b2:metric:complete}{تامًا}
        شبيهًا بفضاء معياري: خذ $X = C(I)$ أو $\R^n$. فإذا كان
        $\eta \colon X \to X$ \omterm{def:b2:metric:continuity}{ليبشيتزيًا} بالثابت $k$ مع $k < 1$،
        فبرهن على أن $x \mapsto x + \eta(x)$ تقابل للفضاء $X$
        مقلوبه \omterm{def:b2:metric:continuity}{ليبشيتزي} بالثابت $\frac1{1-k}$ \emph{(من أجل كل
        $y$، طبّق باناخ على $x \mapsto y - \eta(x)$)}. وهذا
        هو القلب المتري لمبرهنة الدالة العكسية
        (\cref{ch:b2:diffcalc}).
  \item (معادلة كبلر) من أجل $0 \leq e < 1$ و$m \in \R$،
        برهن على أن للمعادلة $x = m + e\sin x$ حلًا واحدًا بالضبط،
        وأن التكرار $x_{n+1} = m + e\sin x_n$ يتقارب
        إليه انطلاقًا من أي بداية، وقدّر: من أجل $e = \frac12$،
        $m = 1$، كم تكرارًا يضمن خطأً $\leq
        10^{-3}$ بالحدّ القبلي؟ (الحلّ هو $x
        \approx 1.4987$.)
  \item (مجموعة كانتور نقطة ثابتة) ليكن $S_1(x) = \frac
        x3$ و$S_2(x) = \frac x3 + \frac23$ على $\R$، ولتكن
        $C$ مجموعة كانتور في \cref{exo:b2:metric:8}. برهن على أن
        $C = S_1(C) \cup S_2(C)$، واشرح بجملة واحدة
        لماذا لا تحقق هذه المعادلة أي مجموعة متراصة غير خالية
        \emph{أخرى} (فالتطبيق $A \mapsto S_1(A) \cup S_2(A)$
        متقلّص من أجل مسافة بين المجموعات المتراصة --- وهي
        مسافة هاوسدورف، التي تُصاغ بأمانة في مجلد السنة الثالثة).
  \item (الترابط يعمّم الوحدانية) ليكن $f$ \omterm{def:b2:metric:continuity}{ليبشيتزيًا} محليًا
        في $y$ على مجموعة \omterm{def:b2:metric:topology}{مفتوحة}، ولتكن $y, z$ حلّين
        للمعادلة $y' = f(t, y)$ على فترة مشتركة $J$
        مع $y(t_0) = z(t_0)$. برهن على أن $y = z$ على $J$: بيّن أن
        $\{t \in J : y(t) = z(t)\}$ غير خالية ومغلقة في $J$،
        \omterm{def:b2:metric:topology}{ومفتوحة} في $J$ (بالوحدانية المحلية)، واستعمل
        ترابط الفترات
        (\cref{thm:b2:metric:connectedness}).
  \item (لا حاجة إلى الصغر من أجل المعادلات الخطية) من أجل $y' =
        \alpha(t)y + \beta(t)$ مع $\alpha, \beta$ متصلتين
        على قطعة $\intcc{A}{B}$، كيّف السؤال 8 لتبيّن أن
        الحدّ العاملي يصحّ على القطعة \emph{كلها}،
        ومن ثم فالوجود والوحدانية شموليان هناك --- وهي
        الحالة العددية لمبرهنة كوشي--ليبشيتز في \cref{ch:b2:diffeq}،
        بلا أي قيد على الطول $B - A$.
  \item (تركيب) بجملة واحدة لكل بند: ماذا أسهم به التمام؛
        وماذا أسهم به التقلّص؛ وماذا اشترت حيلة
        التكرار مقارنةً بمبرهنة باناخ العادية؛ وأين دخل
        الترابط؛ وأي مثال مضاد في الجزء
        الثالث يحرس أي فرضية. وسمِّ المبرهنة القمة،
        وقل ما الذي يحلّ محلّ التقلّص حين تكون $f$ مجرّد
        دالة متصلة (مبرهنة بيانو، عبر التراص في فضاءات
        الدوال --- أي أرزيلا--أسكولي في مجلد السنة الثالثة).
\end{enumerate}
\end{problem}
