\chapter{المعادلات التفاضلية}\label{ch:b2:diffeq}

حلّت السنة الأولى المعادلات الخطية التي تقبل صيغا صريحة. ويوفر هذا
الفصل ما تعجز عنه الصيغ: \emph{مبرهنة كوشي--ليبشيتز} --- الوجود
والوحدانية من أجل $y' = f(t, y)$ --- مبرهنة بمبرهنة نقطة بناخ الثابتة، تماما
كما وُعد في \cref{ch:b2:metric}؛ ثم النظرية الكاملة \emph{للجمل الخطية} $X' = A(t)X + B(t)$، مع
الأسي المصفوفي \omterm{def:b2:diffeq:wronskian}{والفرونسكيان} محركين للحساب.

\section{مبرهنة كوشي--ليبشيتز}

\begin{theorem}[كوشي--ليبشيتز، الصيغة الليبشيتزية الشاملة]\label{thm:b2:diffeq:cauchylipschitz}
لتكن $I$ قطعة، ولتكن $f \colon I \times \R^n \to \R^n$ متصلة و\emph{ليبشيتزية بدلالة المتغير
الثاني}، بانتظام بالنسبة إلى الأول: $\norm{f(t, y) - f(t, z)} \leq k\,\norm{y - z}$ من أجل كل $t
\in I$. عندئذ، من
أجل كل $(t_0, y_0) \in I \times \R^n$، تقبل مسألة كوشي\index{مبرهنة كوشي--ليبشيتز}
\[
y' = f(t, y), \qquad y(t_0) = y_0
\]
حلا وحيدا $y \colon I \to \R^n$ من الصنف $C^1$.
\end{theorem}

\begin{proof}
\emph{إعادة الصياغة.} تكون الدالة المتصلة $y$ حلا للمسألة إذا
وفقط إذا حققت المعادلة التكاملية
\[
y(t) = y_0 + \int_{t_0}^{t} f\bigl(s, y(s)\bigr)\,\dd s
=: T(y)(t)
\]
(بالمبرهنة الأساسية للتحليل في الاتجاهين؛ وكل حل \omterm{def:b2:metric:continuity}{متصل} للمعادلة
التكاملية يكون تلقائيا من الصنف $C^1$).

\emph{تطبيق تقلصي، بعد تغيير \omterm{def:b2:nvs:norm}{المعيار}.} على فضاء بناخ $E =
C(I, \R^n)$ مزودا
\emph{\omterm{def:b2:nvs:norm}{بالمعيار} الموزون}
\[
N(y) = \sup_{t \in I}\; \eu^{-2k\abs{t - t_0}}\,\norm{y(t)} ,
\]
(وهو مكافئ \omterm{def:b2:nvs:norm}{لمعيار} الحد الأعلى: فالوزن محدود من الأعلى ومن الأسفل على
القطعة $I$، ومنه يبقى $E$ تاما)، نقدر من أجل $y, z \in
E$ ومن أجل
$t \geq t_0$ مثلا:
\[
\norm{T(y)(t) - T(z)(t)}
\leq \int_{t_0}^{t} k\,\norm{y(s) - z(s)}\,\dd s
\leq k\,N(y - z)\int_{t_0}^{t} \eu^{2k(s - t_0)}\dd s
\leq \frac{N(y-z)}{2}\,\eu^{2k(t - t_0)} .
\]
وبالضرب في $\eu^{-2k(t - t_0)}$ وأخذ الحد الأعلى (والحالة $t
< t_0$ متناظرة): $N\bigl(T(y) - T(z)\bigr) \leq \frac12 N(y -
z)$:
أي إن $T$ تطبيق تقلصي بالنسبة $\frac12$ على الفضاء التام $(E, N)$. وتعطي
مبرهنة نقطة بناخ الثابتة (\cref{thm:b2:metric:banach}) نقطة ثابتة وحيدة: وهي الحل الوحيد.
\end{proof}

\begin{remark}
إذا كانت $f$ من الصنف $C^1$ فحسب (أي ليبشيتزية محليا)، فإن
المبرهنة تبقى صحيحة \emph{محليا}، مع حل أعظمي على فترة \omterm{def:b2:metric:topology}{مفتوحة} أعظمية؛
وقد تنفجر الحلول في زمن منته ($y' = y^2$، $y(0) = 1$: $y(t)
= \frac{1}{1-t}$، ويختفي عند
$t = 1$). أما الفرضية الليبشيتزية الشاملة فهي التي تشتري القطعة كلها.
ونتيجتان جديرتان بأن تُنقشا: منحنيات حلول معادلة تفاضلية ذات حقل
\omterm{def:b2:metric:continuity}{ليبشيتزي} \emph{لا تتقاطع أبدا}؛ والدالة المعدومة هي الحل الوحيد
لمعادلة خطية متجانسة ينعدم في نقطة ما.
\end{remark}

\begin{example}[الوحدانية مبرهنة: حقل يتسرب]\label{ex:b2:diffeq:leaky}
لننظر في $y' = 2\sqrt{\abs y}$ مع $y(0) = 0$. الدالة المعدومة حل لها؛ وكذلك
\[
y(t) = \begin{cases} 0 & t \leq 0,\\ t^2 & t \geq 0,
\end{cases}
\]
وهي من الصنف $C^1$ (فمشتقة كل من القطعتين تساوي $0$ في نقطة
اللصق) وتحقق $y'(t) = 2t = 2\sqrt{t^2}$ من أجل $t > 0$ --- بل إن تأخير الإقلاع يعطي حلا من
أجل \emph{كل} لحظة انطلاق $c \geq 0$: فهناك عدد لانهائي من الحلول
بالمعطيات الابتدائية نفسها. ولا تناقض مع
\cref{thm:b2:diffeq:cauchylipschitz}: فبجوار $y = 0$،
\[
\frac{\abs{2\sqrt y - 2\sqrt z}}{\abs{y - z}}
= \frac{2}{\sqrt y + \sqrt z} \longrightarrow +\infty ,
\]
لا يكون الحقل ليبشيتزيا بدلالة $y$، والمبرهنة صامتة.
والخلاصة: القراءة الفيزيائية هي دلو يفرغ تحت تأثير الجاذبية مع عكس
اتجاه الزمن --- فانطلاقا من الحالة الفارغة لا يمكن معرفة متى بدأ
الامتلاء؛ فحتمية المعادلات التفاضلية هي بالضبط شرط ليبشيتز، لا قانون
من قوانين الطبيعة.
\end{example}

\section{الجمل الخطية}

\begin{theorem}[بنية الجمل الخطية]\label{thm:b2:diffeq:linear}
لتكن $A \colon I \to \mathcal{M}_n(\R)$ و$B \colon I \to \R^n$ متصلتين على فترة $I$. من أجل كل $(t_0, X_0)$ تقبل
المسألة
\[
X' = A(t)X + B(t), \qquad X(t_0) = X_0
\]
حلا وحيدا على $I$ \emph{كله}. وحلول الجملة المتجانسة ($B = 0$) تشكل
فضاء متجهيا $\mathcal{S}_H$ بعده $n$ بالضبط، والتقويم $X \mapsto X(t_0)$ تشاكل تقابلي
$\mathcal{S}_H \to \R^n$؛ والحل العام يساوي $=$ حلا خاصا $+$ حلا متجانسا.
\end{theorem}

\begin{proof}
على كل قطعة $J \subseteq I$ تحوي $t_0$: تكون $f(t, X) =
A(t)X + B(t)$ متصلة وليبشيتزية بدلالة
$X$ بالثابت $k
= \sup_J \vertiii{A(t)}$ (وهو منته: فالدالة متصلة على قطعة): ومنه تنطبق
\cref{thm:b2:diffeq:cauchylipschitz} على $J$؛ وبجعل $J$ يستنفد $I$، تلصق الوحدانية الحلول في
حل واحد على $I$.
أما خطية مجموعة الحلول وخطية تطبيق التقويم فواضحتان؛ والتقويم تقابل
بالوجود (فهو غامر) وبالوحدانية (فهو متباين): ومنه $\dim \mathcal{S}_H = n$. أما البنية
الأفينية فهي حجة السنة الأولى حرفيا.
\end{proof}

\begin{example}[تشاكل التقويم، بصورة ملموسة]\label{ex:b2:diffeq:evaliso}
من أجل $y'' + y = 0$، منظورا إليها بوصفها الجملة $X' =
\begin{pmatrix} 0 & 1\\ -1 & 0\end{pmatrix}X$ مع $X = (y,
y')$: تقول
المبرهنة إن فضاء الحلول مستو، وإن $X \mapsto X(0) = (y(0), y'(0))$ تشاكل تقابلي على $\R^2$.
والحلان $\cos$ و$\sin$ يقوَّمان إلى $(1, 0)$ و$(0, 1)$ --- وهما الأساس
القانوني للفضاء $\R^2$ --- ومنه فهما يشكلان أساسا لفضاء الحلول،
و\emph{كل} حل يكتب
\[
y(t) = y(0)\cos t + y'(0)\sin t ,
\]
مع قراءة المعاملين مباشرة من المعطيات الابتدائية، دون أي جملة خطية
تُحل. والخلاصة: اختيار \omterm{def:b2:diffeq:wronskian}{الجملة الأساسية} التي قيمها الابتدائية هي
الأساس القانوني (هنا $\cos, \sin$) هو بالضبط اختيار أعمدة المصفوفة $\eu^{tA}$؛
وتشاكل التقويم هو السبب في أن الشروط الابتدائية توسّم المسارات ---
وهو المضمون الهندسي ``للديناميكا الحتمية'' في حالة المعادلات الخطية.
\end{example}

\begin{definition}[الفرونسكيان]\label{def:b2:diffeq:wronskian}
من أجل حلول $X_1, \dots, X_n$ للجملة المتجانسة، يكون \emph{الفرونسكيان}\index{الفرونسكيان}
هو $W(t) = \det\bigl(X_1(t),
\dots, X_n(t)\bigr)$. وبحسب التشاكل السابق، إما أن ينعدم $W$ تطابقا (فالعائلة
مرتبطة) وإما ألا ينعدم أبدا (\emph{جملة أساسية}\index{جملة أساسية})؛
وكمّيا، لدينا $W' = \operatorname{tr}\bigl(A(t)\bigr)
W$، ومنه
\[
W(t) = W(t_0)\,\exp\Bigl(\int_{t_0}^{t}
\operatorname{tr} A(s)\,\dd s\Bigr)
\quad \text{(صيغة ليوفيل)}.
\]
\end{definition}

\begin{example}[التحقق من ليوفيل على معادلة أويلر]\label{ex:b2:diffeq:eulerwronskian}
على $\intoo{0}{\infty}$، تقبل المعادلة $t^2y'' + ty' - y = 0$ الحلين $y_1(t) = t$ و$y_2(t) = \frac1t$ (بالتعويض).
وفرونسكيانهما:
\[
W(t) = \det\begin{pmatrix} t & \tfrac1t\\[2pt]
1 & -\tfrac{1}{t^2}\end{pmatrix}
= -\frac1t - \frac1t = -\frac2t ,
\]
لا ينعدم أبدا: فهما \omterm{def:b2:diffeq:wronskian}{جملة أساسية}. ولنتحقق الآن من ليوفيل: في الصيغة
الموحدة $y'' + \frac1t\,y' - \frac{1}{t^2}\,y = 0$، يكون أثر المصفوفة المرافقة $A(t) = \begin{pmatrix} 0 & 1\\ \frac{1}{t^2}
& -\frac1t\end{pmatrix}$ هو $-\frac1t$، ومنه
\[
W(t) = W(1)\exp\Bigl(-\int_1^t\frac{\dd s}{s}\Bigr)
= -2\,\eu^{-\ln t} = -\frac2t . \checkmark
\]
والخلاصة: يتنبأ ليوفيل \emph{بشكل} \omterm{def:b2:diffeq:wronskian}{الفرونسكيان} قبل معرفة أي حل ---
وهنا، بأن $W$ يجب أن يكون $\frac{c}{t}$؛ وهذا هو ما يشغّل طريقة خفض
الرتبة (\cref{prop:b2:diffeq:secondorder})، حيث تحدد معرفة $y_1$ وشكل \omterm{def:b2:diffeq:wronskian}{الفرونسكيان} الحل $y_2$
بمكاملة واحدة.
\end{example}

\begin{proof}[برهان صيغة ليوفيل]
$W(t) = \det M(t)$ مع $M' = AM$. وبمفاضلة \omterm{def:b2:linalg:det}{المحدد} بوصفه دالة متعددة الخطية بدلالة
الأعمدة،
\[
W'(t) = \sum_j \det(X_1, \dots, X_j', \dots, X_n)
= \sum_j \det(X_1, \dots, AX_j, \dots, X_n) .
\]
والتطبيق $(C_1, \dots, C_n) \mapsto \sum_j \det(C_1, \dots,
AC_j, \dots, C_n)$ متعدد الخطية $n$ مرات ومتناوب (فإذا تساوى عمودان
$C_i = C_k$، انعدمت الحدود $j \notin \{i, k\}$ رأسا، وتلاشى الحدان $j = i$ و$j = k$
مثنى مثنى بعد تبديل عمودين): ومنه فهو $c \cdot \det$ بحسب مبرهنة الوحدانية
(\cref{thm:b2:linalg:detspace})، مع قراءة $c$ على الأعمدة القانونية: $c = \sum_j \det(e_1, \dots, Ae_j, \dots,
e_n) = \sum_j a_{jj} = \operatorname{tr} A$. ومنه
$W' =
\operatorname{tr}\bigl(A(t)\bigr)W$: وهي معادلة تفاضلية خطية سلمية، تُحل بصيغة السنة الأولى.
\end{proof}

\section{المعاملات الثابتة: الأسي المصفوفي}

\begin{theorem}\label{thm:b2:diffeq:matrixexp}
من أجل $A \in \mathcal{M}_n(\R)$ (أو $\C$)، يحقق الأسي $\eu^{tA}
= \sum_k \frac{(tA)^k}{k!}$ (\cref{ex:b2:nvs:matrixexp}) ما يلي:
$t \mapsto \eu^{tA}$ من الصنف $C^1$ (بل $C^\infty$) مع\index{الأسي المصفوفي}
\[
\frac{\dd}{\dd t}\,\eu^{tA} = A\,\eu^{tA} = \eu^{tA}A ,
\qquad
\eu^{(s+t)A} = \eu^{sA}\,\eu^{tA},
\qquad
(\eu^{A})^{-1} = \eu^{-A} ;
\]
ولدينا $\eu^{A + B} = \eu^A\eu^B$ \emph{عندما} $AB = BA$. ومسألة كوشي
$X' = AX$، $X(0) = X_0$ لها الحل الوحيد $X(t) =
\eu^{tA}X_0$؛ ومع طرف ثان، تتحقق صيغة
\emph{تغيير الثوابت}:
\[
X(t) = \eu^{(t - t_0)A}X_0 + \int_{t_0}^{t} \eu^{(t-s)A}B(s)\,\dd
s .
\]
\end{theorem}

\begin{proof}
\emph{قابلية \omterm{def:b2:diffcalc:differential}{التفاضل}:} تتقارب المتسلسلة $\sum \frac{t^kA^k}{k!}$ ومتسلسلتها المشتقة
حدا حدا $\sum \frac{t^{k-1}A^k}{(k-1)!} = A\sum
\frac{(tA)^{k-1}}{(k-1)!}$ تقاربا ناظميا على كل قطعة (بالمعايير $\leq \frac{(\abs t\,\vertiii A)^k}{k!}$): فنفاضل
حدا حدا (\cref{thm:b2:funcseq:seriestransfer}، بقيم متجهية).
والترتيبان $A\eu^{tA}$ و$\eu^{tA}A$ متطابقان لأن كل مجموع جزئي يتبادل مع
$A$.

\emph{قانون الزمرة:} من أجل $A, B$ متبادلتين، يعاد تنظيم \omterm{thm:b2:series:fubini}{جداء كوشي}
للمتسلسلتين الأسيتين بمبرهنة ذات الحدين تماما كما في \cref{ex:b2:series:exp} (والتقارب
المطلق في \omterm{def:b2:structures:algebra}{الجبر} البناخي يبرر ذلك): $\eu^{A+B} = \eu^A\eu^B$؛ وبأخذ $B = sA$ نحصل على
قانون الزمرة ذات الوسيط الواحد، ويعطي $B = -A$ المقلوب.

\emph{مسألة كوشي:} المقدار $X(t) = \eu^{tA}X_0$ يحلها (بالمفاضلة)؛ والوحدانية
بحسب \cref{thm:b2:diffeq:linear}. أما تغيير الثوابت: فنضع $Y(t) = \eu^{-tA}X(t)$؛ وبالمفاضلة، $Y' = \eu^{-tA}(X' -
AX) = \eu^{-tA}B(t)$؛ ثم
نكامل من $t_0$ إلى $t$ ونضرب من جديد في $\eu^{tA}$.
\end{proof}

\begin{method}[حساب $\eu^{tA}$]\label{met:b2:diffeq:computeexp}
نختزل $A$ (\cref{ch:b2:reduction}): فإذا كانت $A = PDP^{-1}$ قطرية، فإن
$\eu^{tA} = P\,\eu^{tD}P^{-1}$ مع $\eu^{tD}$ قطرية معاملاتها $\eu^{t\lambda_i}$؛ وفي الحالة العامة نستعمل
تفكيك دونفور $A = D + N$ (بمركبتين متبادلتين): $\eu^{tA} = \eu^{tD}\,\eu^{tN}$ حيث $\eu^{tN}$ \emph{كثير
حدود} بدلالة $t$ (فانعدام القوى يبتر المتسلسلة). وتتزاوج \omterm{def:b2:reduction:eigen}{القيم
الذاتية} المركبة في كتل من نوع دوران مضروب في أسي
(\cref{exo:b2:diffeq:5}).
\end{method}

\begin{remark}[مزالق شائعة]
\emph{(1) كتابة $\eu^{A+B} \neq \eu^A\eu^B$ دون تبادل:}
خذ $A = \begin{pmatrix} 0 & 1\\ 0 & 0\end{pmatrix}$، $B =
\begin{pmatrix} 0 & 0\\ 1 & 0\end{pmatrix}$. عندئذ $\eu^A = I +
A$، $\eu^B = I + B$ (لانعدام القوى)، ومنه
\[
\eu^A\eu^B = \begin{pmatrix} 2 & 1\\ 1 & 1\end{pmatrix},
\qquad\text{بينما}\qquad
\eu^{A+B} = \cosh(1)\,I + \sinh(1)\,(A + B)
= \begin{pmatrix} \cosh 1 & \sinh 1\\ \sinh 1 & \cosh 1
\end{pmatrix},
\]
باستعمال $(A+B)^2 = I$؛ ثم $\cosh 1 \approx 1.54 \neq 2$. فقانون الزمرة في \cref{thm:b2:diffeq:matrixexp} يحمل فرضية
حقيقية. \emph{(2) الحدس اللاخطي على أرض خطية:} تعيش حلول جملة
\emph{خطية} ذات معاملات متصلة على الفترة كلها
(\cref{thm:b2:diffeq:linear}) --- فإذا انفجر حل مرشح داخل $I$، فإما أن المعادلة لم
تكن خطية وإما أن الحساب خاطئ؛ وبالعكس، لا تعد أبدا بالطابع الشامل
في المعادلات اللاخطية دون حجة ($y' = y^2$).
\emph{(3) القسمة على المجهول:} فصل المتغيرات في
$y' = y(1-y)$ يهمل بصمت الحلين الثابتين $0$ و$1$ --- وهما بالضبط
الحلان اللذان ينظمان الخط الطوري
(\cref{exo:b2:diffeq:3})؛ فاسرد الحلول الثابتة أولا.
\emph{(4) المعطيات الابتدائية تثبت متجهات لا أعدادا:} تحتاج المعادلة
السلمية من الرتبة $n$ إلى $n$ شرطا ($y, y', \dots$ في
$t_0$)؛ والاكتفاء بمطابقة $y(t_0)$ يترك عائلة ذات $(n-1)$ وسيطا، وهو
مصدر كلاسيكي للثوابت ``الضائعة''.
\end{remark}

\begin{example}[أسي من الحجم $3\times3$ بتفكيك دونفور]\label{ex:b2:diffeq:dunford3}
لنحل $X' = AX$ من أجل $A = \begin{pmatrix} 2 & 1 & 0\\ 0 & 2 &
0\\ 0 & 0 & 3\end{pmatrix}$. تفكيك دونفور بالكتل: $A = D + N$ مع $D = \operatorname{diag}(2, 2, 3)$
و$N = E_{12}$، وهما متبادلتان (فالمصفوفة $N$ تعيش داخل الكتلة الموافقة
\omterm{def:b2:reduction:eigen}{للقيمة الذاتية} $2$)، و$N^2 =
0$:
\[
\eu^{tA} = \eu^{tD}\,\eu^{tN}
= \begin{pmatrix}
\eu^{2t} & t\,\eu^{2t} & 0\\
0 & \eu^{2t} & 0\\
0 & 0 & \eu^{3t}
\end{pmatrix} .
\]
ويُقرأ الحل العام عمودا عمودا: $X(t) =
\bigl(\eu^{2t}(x_0 + ty_0),\ \eu^{2t}y_0,\
\eu^{3t}z_0\bigr)$. وللتحقق: في $t = 0$ تساوي
المصفوفة $I$؛ ومحددها $\eu^{7t} =
\eu^{t\operatorname{tr}A}$، كما يقتضي ليوفيل؛ ويظهر العامل
$t$ بالضبط حيث تكون \omterm{def:b2:reduction:eigen}{القيمة الذاتية} $2$ ناقصة. والخلاصة: كثيرات
الحدود مضروبة في الأسيات ليست تخمينا يُحفظ --- بل هي المتسلسلة
المبتورة $\eu^{tN}$، ودرجتها محصورة بدليل انعدام القوى، لا أكثر أبدا.
\end{example}

\begin{method}[حل $X' = AX + B(t)$، من البداية إلى النهاية]\label{met:b2:diffeq:solvelinear}
\begin{enumerate}
  \item \omterm{def:b2:reduction:eigen}{طيف} $A$؛ ثم $\eu^{tA}$ بحسب
        \cref{met:b2:diffeq:computeexp} (بالتقطير؛ أو بتفكيك دونفور كما في
        \cref{ex:b2:diffeq:dunford3}؛ أو بحيلة كثير حدود مثل $A^2 = -I$).
  \item حل خاص: تغيير الثوابت $\int_{t_0}^t\eu^{(t-s)A}B(s)\dd s$ يفي بالغرض دائما؛ أما إذا كان
        الطرف الثاني $B$ من نوع أسي مضروب في كثير حدود، فإن
        فرضية حل من الشكل نفسه (مع رفع الدرجة عند \omterm{pb:b2:diffeq:1}{الرنين}،
        \cref{exo:b2:diffeq:10}) أسرع.
  \item الحل العام يساوي $= \eu^{(t-t_0)A}X_0 +$ حلا خاصا؛ وطابق المعطيات الابتدائية
        \emph{في النهاية}، على الصيغة الكاملة.
  \item للتحقق: صحة $X(t_0)$؛ وتوافق نمو الجزء المتجانس مع الأجزاء
        الحقيقية للقيم الذاتية (\cref{exo:b2:diffeq:8})؛ وخضوع $\det$ لمصفوفة أساسية
        لصيغة ليوفيل.
\end{enumerate}
\end{method}

\begin{example}[مخطط طوري]\label{ex:b2:diffeq:phaseportrait}
$X' = AX$ مع $A = \begin{pmatrix} 0 & 1\\ -1 & 0\end{pmatrix}$:
لدينا $A^2 = -I$، ومنه تنقسم المتسلسلة إلى
\[
\eu^{tA} = (\cos t)\,I + (\sin t)\,A
= \begin{pmatrix} \cos t & \sin t\\ -\sin t & \cos t
\end{pmatrix} :
\]
فالمسارات دوائر تُقطع في اتجاه عقارب الساعة --- وهو المتذبذب التوافقي
$x'' + x = 0$ في ثوب الرتبة الأولى. والقيمتان الذاتيتان $\pm\iu$ على المحور
التخيلي: \emph{مركز}. وبوجه أعم تحدد الأجزاء الحقيقية للقيم الذاتية
للمصفوفة $A$ نمو $\norm{X(t)}$ أو تلاشيه
(\cref{exo:b2:diffeq:8}).
\end{example}

\section{الرتبة الثانية بمعاملات متغيرة}

\begin{proposition}\label{prop:b2:diffeq:secondorder}
المعادلة $y'' + a(t)y' + b(t)y = c(t)$ (حيث $a, b, c$ متصلة على $I$) هي الجملة $X' = A(t)X + B(t)$ من
أجل $X = (y, y')$: فالحلول موجودة ووحيدة على $I$ كله من أجل أي معطيات
ابتدائية $(y(t_0),
y'(t_0))$؛ والحلول المتجانسة تشكل مستويا. وإذا عُرف حل متجانس
غير معدوم $y_1$، وُجد حل ثان مستقل \emph{بخفض الرتبة}: فوضع $y = y_1 z$
يحول المعادلة المتجانسة إلى معادلة من الرتبة الأولى بدلالة $z'$،
تُحل بالمكاملات.
\end{proposition}

\begin{proof}
تعطي الصيغة الجملية والمبرهنة \cref{thm:b2:diffeq:linear} كل ما هو بنيوي. أما خفض الرتبة:
فبالتعويض $y = y_1z$،
\[
y_1 z'' + (2y_1' + a y_1)z' + \underbrace{(y_1'' + ay_1' +
by_1)}_{=\,0}\,z = 0 :
\]
وهي معادلة خطية من الرتبة الأولى بدلالة $u = z'$، تُحل بصيغة السنة
الأولى؛ وبمكاملة $u$ نجد $z$، ومنه $y_2 = y_1 z$، وهو مستقل عن $y_1$
كلما كانت $z$ غير ثابتة.
\end{proof}

\begin{example}\label{ex:b2:diffeq:lowering}
$t^2y'' - 2y = 0$ على $\intoo{0}{\infty}$: الدالة $y_1 = t^2$ حل. نعوض $y = t^2z$: انطلاقا من $y' = t^2z' + 2tz$
و$y''
= t^2z'' + 4tz' + 2z$،
\[
t^2y'' - 2y = t^4 z'' + 4t^3z' = 0,
\qquad\text{أي}\qquad \frac{z''}{z'} = -\frac4t :
\]
ومنه $z' = t^{-4}$ (إلى غاية ثابت)، و$z = -\frac{1}{3t^3}$، و$y_2
= t^2z = -\frac{1}{3t}$. والحل العام: $y = \alpha t^2 +
\frac{\beta}{t}$.
\end{example}

\begin{exercise}[$\star$]\label{exo:b2:diffeq:1}
حل $X' = AX$، $X(0) = (1, 0)^{\mathsf T}$، من أجل $A =
\begin{pmatrix} 1 & 1\\ 0 & 2\end{pmatrix}$ (بالتقطير) ومن أجل $A =

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:diffeq:2}
أي مسائل كوشي التالية لها حلول شاملة وحيدة على $\R$ بحسب \cref{thm:b2:diffeq:cauchylipschitz}؟
$y' = \sin(ty)$؛ $\;y' =
y^2$؛ $\;y' = \abs y$. وفي الأخيرة، حل صراحة من أجل $y(0) =
0$ ومن
أجل $y(0) = 1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:diffeq:3}
برهن على أن حلين أعظميين متمايزين للمعادلة $y' = f(t,y)$ (حيث $f$
ليبشيتزية بدلالة $y$) لا يأخذان القيمة نفسها في اللحظة نفسها
أبدا، واستنتج أن حلول $y' = y(1 - y)$ المنطلقة من $\intoo{0}{1}$ تبقى في $\intoo{0}{1}$ إلى
الأبد.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:4}
احسب $\eu^{tA}$ من أجل $A = \begin{pmatrix} 3 & 1\\ -1 &
1\end{pmatrix}$ \emph{(بتفكيك دونفور: $(A - 2I)^2 = 0$)}، ثم حل
$X' =
AX + \begin{pmatrix} \eu^{2t}\\ 0\end{pmatrix}$، $X(0) = 0$، بتغيير الثوابت.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:5}
من أجل $A = \begin{pmatrix} \alpha & -\beta\\ \beta &
\alpha\end{pmatrix}$، برهن على $\eu^{tA} =
\eu^{\alpha t}\begin{pmatrix} \cos\beta t & -\sin\beta t\\
\sin\beta t & \cos\beta t\end{pmatrix}$ --- أي إن المسارات حلزونية ---
بطريقتين: بالمتسلسلة (اكتب $A = \alpha I + \beta J$، $J^2 =
-I$)، وبالمطابقة المركبة
$z' = (\alpha +
\iu\beta)z$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:6}
(مبرهنة غرونوال) لتكن $u$ متصلة موجبة تحقق $u(t)
\leq C + k\int_{t_0}^{t} u(s)\,\dd s$ على $\intco{t_0}{T}$.
برهن على $u(t) \leq C\,\eu^{k(t - t_0)}$ \emph{(فاضل $v(t) =
\eu^{-kt}\int_{t_0}^t u$)}. واستنتج من جديد الوحدانية في
مبرهنة كوشي--ليبشيتز، والتعلق \omterm{def:b2:metric:continuity}{المتصل} $\norm{y(t) -
z(t)} \leq \norm{y_0 - z_0}\,\eu^{k\abs{t - t_0}}$ من أجل حلين بمعطيات
ابتدائية مختلفة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:7}
علما أن $y_1(t) = \frac{\sin t}{t}$ حل للمعادلة $ty'' + 2y' + ty =
0$ على $\intoo{0}{\pi}$، جد حلا ثانيا مستقلا
بخفض الرتبة، وأعط الحل العام.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffeq:8}
لتكن $A \in \mathcal{M}_n(\C)$ جميع قيمها الذاتية ذات جزء حقيقي سالب (تماما). برهن على
أن كل حل للمعادلة $X' = AX$ يؤول إلى $0$ عندما $t \to +\infty$، وبسرعة
أسية: $\norm{X(t)}
\leq C\,\eu^{-\alpha t}$ من أجل $\alpha > 0$ ما.
\emph{(ثلّث المصفوفة؛ وعالج الجملة المثلثية انطلاقا من السطر الأخير
صعودا، أو استعمل تفكيك دونفور: $\eu^{tA} = \eu^{tD}\eu^{tN}$ مع $\norm{\eu^{tD}} \leq \eu^{-\alpha' t}$ و$\eu^{tN}$ كثير حدود
بدلالة $t$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffeq:9}
(لا هروب في زمن منته عند النمو الخطي) لنفترض أن $f$ متصلة تحقق
$\norm{f(t, y)} \leq a\norm y + b$ على $\intco{0}{\infty} \times \R^n$، وليبشيتزية محليا بدلالة $y$. باستعمال مبرهنة
غرونوال (\cref{exo:b2:diffeq:6}) على الصيغة التكاملية، برهن على أن الحلول الأعظمية
شاملة (أي معرفة على $\intco{0}{\infty}$ كله).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:diffeq:10}
حل $y'' - 3y' + 2y = \eu^{t}$: الحلول المتجانسة، ثم حل خاص من الشكل $\alpha t\,\eu^{t}$
\emph{(لماذا يفشل التخمين الساذج $\alpha\eu^t$؟)}؛ ثم الحل العام والحل
الذي يحقق $y(0) = y'(0) = 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:11}
احسب $\eu^{tA}$ من أجل كتلة جوردان
\[
A = \begin{pmatrix} \lambda & 1 & 0\\ 0 & \lambda & 1\\
0 & 0 & \lambda\end{pmatrix},
\]
وصف جميع حلول $X' = AX$: أسيات مضروبة في متجهات كثيرات الحدود، بدرجات
تصل إلى $2$. ومن أين تأتي درجة كثير الحدود؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffeq:12}
(إثارة دورية، استجابة دورية) لتكن $A \in
\mathcal{M}_n(\R)$ ولتكن $B \colon \R \to \R^n$ متصلة ودورية
بالدور $T$.
\begin{enumerate}
  \item بيّن أن حلا للمعادلة $X' = AX + B(t)$ يكون دوريا بالدور $T$ إذا
        وفقط إذا كان $X(T) = X(0)$
        \emph{(قارن $X(\cdot + T)$ و$X$)}.
  \item بيّن أن \omterm{def:b2:reduction:eigen}{القيم الذاتية} للمصفوفة $\eu^{TA}$ هي المقادير
        $\eu^{T\lambda}$، $\lambda \in \operatorname{Sp}A$
        \emph{(ثلّث على $\C$)}. واستنتج: إذا لم تقع أي \omterm{def:b2:reduction:eigen}{قيمة ذاتية}
        للمصفوفة $A$ في $\frac{2\iu\pi}{T}\Z$، فإن
        $I - \eu^{TA}$ قابلة للقلب.
  \item تحت هذه الفرضية، برهن على أن للجملة حلا دوريا وحيدا بالدور
        $T$، وأن
        \[
        X(0) = \bigl(I - \eu^{TA}\bigr)^{-1}
        \int_0^{T}\eu^{(T-s)A}B(s)\,\dd s .
        \]
        وماذا توافق الحالة المستثناة في حالة المتذبذب التوافقي؟
        (تجيب مسألة نهاية الأسبوع: \omterm{pb:b2:diffeq:1}{الرنين}.)
\end{enumerate}
\end{exercise}

\section{مسألة: التذبذبات والرنين ومبرهنتا شتورم في المقارنة}

\begin{problem}\label{pb:b2:diffeq:1}
معادلة واحدة تحكم العالمين الميكانيكي والكهربائي:
\[
x'' + 2\zeta\omega\,x' + \omega^2 x = F(t),
\qquad \omega > 0,\ \zeta \geq 0 .
\]
وتدرسها هذه المسألة دراسة كاملة --- عبر تصنيف الجمل الخطية المستوية
بمستوي الأثر--المحدد، وأنظمة التخميد الثلاثة، والاستجابة الدائمة
للإثارة الدورية بقمة \emph{الرنين}\index{الرنين} فيها وكارثة \omterm{pb:b2:diffeq:1}{الرنين}
--- ثم تترك المعاملات الثابتة إلى \emph{مبرهنتي شتورم في الفصل
والمقارنة}\index{مبرهنة شتورم في المقارنة}، اللتين تتحكمان في أصفار
حلول $y'' + q(t)y = 0$ دون أي صيغة صريحة على الإطلاق.

\medskip
\noindent\textbf{الجزء الأول --- مستوي الأثر--المحدد.} لتكن
$A \in \mathcal M_2(\R)$، $\tau = \operatorname{tr}A$، $\delta
= \det A$، $\Delta = \tau^2 - 4\delta$.
\begin{enumerate}
  \item بيّن أن القيمتين الذاتيتين للمصفوفة $A$ هما $\frac{\tau\pm\sqrt\Delta}{2}$، وصنّف:
        قيمتان ذاتيتان حقيقيتان من إشارتين متعاكستين إذا وفقط إذا
        كان $\delta < 0$؛ وقيمتان ذاتيتان حقيقيتان من الإشارة نفسها إذا
        وفقط إذا كان $\delta > 0$ و$\Delta
        \geq 0$ (بإشارة $\tau$)؛ وزوج مرافق غير
        حقيقي إذا وفقط إذا كان $\Delta < 0$ (بجزء حقيقي $\frac\tau2$).
  \item (السرج، $\delta < 0$) بقيمتين ذاتيتين $\mu < 0 <
        \lambda$ ومتجهتين ذاتيتين
        $v_\pm$، اكتب الحل العام وصف المسارات: نصفا مستقيم مستقران
        ونصفا مستقيم غير مستقرين، وجميع المدارات الأخرى مقاربة
        للاثنين. ولماذا لا يمكن لأي حل غير $0$ أن يبقى محدودا على
        $\R$ كله؟
  \item (العقد، $\delta > 0$، $\Delta > 0$) من أجل $\mu < \lambda
        < 0$: بيّن أن كل حل غير
        معدوم يؤول إلى $0$، وأن جميع المدارات عدا الواقعة على
        المحور السريع تصل \emph{ماسّة للاتجاه الذاتي البطيء}
        \emph{(قارن $\eu^{\mu t}$ و$\eu^{\lambda t}$)}.
  \item (الحلزونات والمراكز، $\Delta < 0$) بكتابة القيمتين الذاتيتين
        $\alpha \pm \iu\beta$، استعمل \cref{exo:b2:diffeq:5} (بعد تغيير أساس حقيقي، مقبولا في تلك
        العمومية أو مبرهنا من أجل جمل الجزء الثاني، وهي المستعملة
        فيما يلي) لوصف المدارات: حلزونات متقاربة من أجل $\alpha =
        \frac\tau2 < 0$،
        ومتباعدة من أجل $\tau > 0$، ومنحنيات مغلقة (مركز) من أجل $\tau = 0$.
  \item (الحالات الحدية) في حالة \omterm{def:b2:reduction:eigen}{قيمة ذاتية} مضاعفة ($\Delta =
        0$): بيّن
        أن $\eu^{tA} = \eu^{\lambda t}(I + tN)$ مع $N
        = A - \lambda I$ معدومة القوى، وميّز بين النجمة ($N = 0$)
        والعقدة غير السوية ($N \neq 0$).
        ولخّص الجزء الأول في صورة مستوي الأثر--المحدد الواردة في شكل
        هذا الفصل.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- المتذبذب المخمد.} نأخذ الآن
$F =
0$: $x'' + 2\zeta\omega x' + \omega^2x = 0$، أي $X' = AX$ مع $A = \begin{pmatrix} 0 & 1\\ -\omega^2 &
-2\zeta\omega\end{pmatrix}$.
\begin{enumerate}[resume]
  \item احسب $\tau, \delta, \Delta$ وضع أنظمة التخميد الثلاثة في مستوي
        الأثر--المحدد: \emph{ضعيف التخميد} $0 < \zeta < 1$ (حلزون مستقر)،
        \emph{حرج التخميد} $\zeta = 1$ (\omterm{def:b2:reduction:eigen}{قيمة ذاتية} مضاعفة)،
        \emph{قوي التخميد} $\zeta > 1$ (عقدة مستقرة)؛ والحالة $\zeta = 0$ هي
        المركز.
  \item حل الأنظمة الثلاثة صراحة:
        \[
        \zeta < 1:\ \eu^{-\zeta\omega t}\bigl(a\cos\omega_d t
        + b\sin\omega_dt\bigr),\ \omega_d =
        \omega\sqrt{1-\zeta^2};
        \qquad
        \zeta = 1:\ (a + bt)\,\eu^{-\omega t};
        \]
        $\zeta > 1$: أسيان حقيقيان. عرّف شبه الدور $\frac{2\pi}{\omega_d}$ وبيّن أن نسبة
        القيم العظمى المتتالية للدالة $\abs x$ هي الثابت $\eu^{-2\pi\zeta/\sqrt{1-\zeta^2}}$ (وهو
        التناقص اللوغاريتمي).
  \item (مبدأ مغلاق الباب) من أجل $\zeta \geq 1$، تحكم \omterm{def:b2:reduction:eigen}{القيمة الذاتية}
        الأبطأ $\lambda_{\mathrm{slow}} = -\omega\bigl(\zeta -
        \sqrt{\zeta^2-1}\bigr)$ سرعة التلاشي. بيّن أن $\abs{\lambda_{\mathrm{slow}}} =
        \frac{\omega}{\zeta + \sqrt{\zeta^2 - 1}}$ دالة
        \emph{متناقصة} بدلالة $\zeta \geq 1$: فالتخميد الحرج $\zeta = 1$ يعطي
        أسرع عودة إلى السكون دون تذبذب.
  \item (الطاقة) لنضع $E(t) = \frac12x'^2 +
        \frac12\omega^2x^2$. برهن على $E' = -2\zeta\omega\,x'^2
        \leq 0$، واستنتج أنه من أجل
        $\zeta > 0$ لا تقبل المعادلة أي حل دوري غير معدوم \emph{(فوجود
        دور يفرض أن يكون $E$ ثابتا، ومنه $x' \equiv 0$)}.
  \item فسّر في جملتين لماذا يكون المركز $\zeta = 0$ \emph{هشا بنيويا}:
        فأي $\zeta > 0$، مهما صغر، يدمر الدورية --- وأين يظهر ذلك في
        مستوي الأثر--المحدد (فخط المراكز داخله خال).
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- التذبذبات القسرية \omterm{pb:b2:diffeq:1}{والرنين}.}
نأخذ الآن $F(t) = F\cos(\gamma t)$ مع $F, \gamma > 0$.
\begin{enumerate}[resume]
  \item ($\zeta > 0$: النظام الدائم) ابحث عن $x_p =
        \Re\bigl(z\,\eu^{\iu\gamma t}\bigr)$: بيّن أن
        \[
        z = \frac{F}{\omega^2 - \gamma^2 +
        2\iu\zeta\omega\gamma},
        \qquad
        A(\gamma) := \abs z = \frac{F}{\sqrt{(\omega^2 -
        \gamma^2)^2 + 4\zeta^2\omega^2\gamma^2}} ,
        \]
        واكتب $x_p = A(\gamma)\cos(\gamma t - \varphi)$ مع $\tan\varphi =
        \frac{2\zeta\omega\gamma}{\omega^2-\gamma^2}$.
  \item بيّن أن \emph{كل} حل هو $x_p$ مضافا إليه نظام عابر من
        الجزء الثاني يؤول إلى $0$: فمهما تكن المعطيات الابتدائية،
        تنغلق الجملة على النظام الدائم --- بسعة $A(\gamma)$ وتأخر طوري
        $\varphi$.
  \item (منحنى \omterm{pb:b2:diffeq:1}{الرنين}) عظّم $A$: بيّن أن $A(\gamma)$ يقبل قيمة عظمى
        داخلية إذا وفقط إذا كان $\zeta <
        \frac{1}{\sqrt2}$، وذلك في
        \[
        \gamma_* = \omega\sqrt{1 - 2\zeta^2},
        \qquad
        A(\gamma_*) =
        \frac{F}{2\zeta\omega^2\sqrt{1-\zeta^2}} ,
        \]
        وأن القمة تضخّم، من أجل $\zeta$ صغيرة، الاستجابة الساكنة
        $A(0) = \frac F{\omega^2}$ بالعامل $\approx \frac{1}{2\zeta}$.
  \item ($\zeta = 0$، بعيدا عن \omterm{pb:b2:diffeq:1}{الرنين}) من أجل $\gamma \neq \omega$، بيّن أن الحل الذي
        يحقق $x(0) = x'(0) = 0$ هو
        \[
        x(t) = \frac{F}{\omega^2 -
        \gamma^2}\bigl(\cos\gamma t - \cos\omega t\bigr)
        = \frac{2F}{\omega^2-\gamma^2}
        \sin\frac{(\omega-\gamma)t}{2}
        \sin\frac{(\omega+\gamma)t}{2} :
        \]
        وهو محدود، مع \emph{ضربات} --- أي تذبذب سريع تحت غلاف بطيء
        --- عندما يكون $\gamma$ قريبا من $\omega$.
  \item ($\zeta = 0$، عند \omterm{pb:b2:diffeq:1}{الرنين}) من أجل $\gamma = \omega$، بيّن أن $x_p(t) = \frac{F}{2\omega}\,t\sin(\omega t)$ حل،
        واستعده بوصفه نهاية السؤال 14 عندما $\gamma \to \omega$: فالسعة تنمو
        خطيا إلى الأبد --- وهي كارثة \omterm{pb:b2:diffeq:1}{الرنين}.
  \item (الصلة بفورييه) تتفكك إثارة دورية عامة إلى توافقيات (فصل
        فورييه)؛ وبالخطية يكون النظام الدائم مجموع الاستجابات
        التوافقية. ومن أجل متذبذب غير مخمد تواتره $\omega = 3$ يُثار
        بالموجة المربعة من نوع \cref{exo:b2:fourier:1} (وتوافقياتها عند جميع الأعداد
        الفردية)، أي توافقية ترنّ؟ وجملة واحدة عن سبب خوف المهندسين
        من الموجات المربعة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- مبرهنتا شتورم.} لننظر في $y''
+ q(t)\,y = 0$
على فترة $I$، حيث $q$ متصلة. (وكل معادلة $y'' + ay' + by = 0$ تختزل إلى
هذه الصيغة الموحدة بالتعويض $y = u\exp\bigl(-\frac12\int a\bigr)$؛ ويبين السؤال 21 صورة من هذه
الحيلة في العمل.)
\begin{enumerate}[resume]
  \item من أجل حلين $y_1, y_2$، بيّن أن \omterm{def:b2:diffeq:wronskian}{الفرونسكيان} $W = y_1y_2' - y_1'y_2$ \emph{ثابت}،
        وأنه معدوم إذا وفقط إذا كان الحلان متناسبين؛ وأن أصفار حل
        غير معدوم \emph{بسيطة ومعزولة}.
  \item (فصل شتورم) ليكن $y_1, y_2$ حلين مستقلين وليكن $a < b$ صفرين
        متتاليين للحل $y_1$. برهن على أن $y_2$ ينعدم مرة واحدة
        بالضبط في $\intoo{a}{b}$ \emph{(قوّم الثابت $W$ في $a$
        و$b$: فلدينا $W = y_1'y_2$ هناك، والمقداران $y_1'(a)$ و$y_1'(b)$ من
        إشارتين متعاكستين)}: فأصفار الحلول المستقلة تتناوب.
  \item (مقارنة شتورم) ليكن $q_1 \leq q_2$ على $I$، ولتكن $y \neq
        0$ تحقق
        $y'' + q_1y = 0$، ولتكن $z \neq 0$ تحقق $z'' + q_2z
        = 0$، وليكن $a < b$ صفرين
        متتاليين للحل $y$. بيّن أن $z$ ينعدم في $\intcc{a}{b}$ ---
        وينعدم في الداخل تماما إذا كان $q_1 < q_2$ في نقطة ما من $\intoo ab$
        \emph{(إذا كان $z \neq
        0$ على $\intoo ab$، فادرس $(yz' - y'z)' = (q_1 -
        q_2)yz$ بإشارتين
        ثابتتين للمقدارين $y, z$ وقارن القيم الحدية)}.
  \item استنتج \emph{حصور التباعد}: إذا كان $0 < m^2 \leq
        q(t) \leq M^2$ على $I$، فإن
        أي صفرين متتاليين $a < b$ لحل غير معدوم للمعادلة $y'' + qy = 0$
        يحققان
        \[
        \frac{\pi}{M} \;\leq\; b - a \;\leq\; \frac{\pi}{m}
        \]
        \emph{(قارن مع $u'' + M^2u = 0$ و$u'' + m^2u
        = 0$، اللذين تتباعد أصفارهما
        بالمقدارين $\frac\pi M$ و$\frac\pi m$)}. وتحقق من ذلك على المتذبذب
        التوافقي.
  \item حوّل المعادلة $ty'' + 2y' + ty = 0$ (\cref{exo:b2:diffeq:7}) بالتعويض $u = ty$ إلى $u'' + u =
        0$،
        واستعد حلّيها $\frac{\sin t}t$ و$\frac{\cos t}{t}$ على الفور، واستنتج أن أصفار كل
        حل غير معدوم تتباعد بالمقدار $\pi$ بالضبط: وهي رؤية شتورم
        للعالم --- فالأصفار يتحكم فيها المعامل $q$، بصيغة أو
        بغير صيغة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- دوهاميل وحدود الحصر.}
\begin{enumerate}[resume]
  \item (دوهاميل من أجل المتذبذب) بيّن أنه من أجل $F$ متصلة،
        يكون حل المعادلة $x'' + \omega^2x =
        F(t)$ الذي يحقق $x(0) = x'(0) = 0$ هو
        \[
        x(t) = \frac1\omega\int_0^t\sin\bigl(\omega(t -
        s)\bigr)F(s)\,\dd s ,
        \]
        واستعد منه الحل الرنيني للسؤال 15 بأخذ $F(s) = F\cos(\omega s)$
        \emph{(بتحويل الجداء إلى مجموع)}.
  \item ($\zeta > 0$: دخل محدود، خرج محدود) بيّن أنه من أجل $\zeta > 0$ ومن
        أجل \emph{أي} دالة متصلة محدودة $F$، يكون كل حل للمعادلة
        المخمدة محدودا على $\intco{0}{\infty}$ \emph{(بتغيير الثوابت مضافا إليه
        التلاشي الأسي $\vertiii{\eu^{tA}} \leq
        C\eu^{-\alpha t}$ للمقدار \cref{exo:b2:diffeq:8})}.
  \item ($\zeta = 0$) بيّن أنه دون تخميد، تُبقي الإثارة الدورية المحدودة
        جميع الحلول محدودة \emph{إلا} عند \omterm{pb:b2:diffeq:1}{الرنين} بالضبط ($\gamma =
        \omega$،
        السؤال 15 مقابل السؤال 14): فالتخميد هو ما يحول حدود الحصر
        إلى استقرار منتظم.
  \item تركيب. في جملة واحدة عن كل بند: (1) كيف ينظم مستوي
        الأثر--المحدد الجزأين الأول والثاني، وأين تخرج الإثارة
        (الجزء الثالث) عنه؛ (2) المعنى الفيزيائي للمقدارين $\gamma_*$
        و$A(\gamma_*)$ وللعامل $\frac1{2\zeta}$؛ (3) ماذا تقول مبرهنتا شتورم مما
        تعجز عنه الصيغ الصريحة؛ (4) أي نتيجتين من هذه المسألة
        سيعيد بقية الكتاب استعمالهما بصمت (ثبات \omterm{def:b2:diffeq:wronskian}{الفرونسكيان} بحسب
        ليوفيل؛ واستقرار الدخل المحدود).
\end{enumerate}
\end{problem}
