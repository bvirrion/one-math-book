\chapter{الفضاءات الأفينية}\label{ch:b2:affine}

للفضاءات المتجهية نقطة متميزة --- المبدأ --- لا تريدها الهندسة.
و\emph{\omterm{def:b2:affine:def}{الفضاء الأفيني}} فضاء متجهي نسي مبدأه: فتصير النقط والمتجهات
نوعين مختلفين، يربط بينهما الانسحاب. ويبني هذا الفصل القصير المعجم
(النقط ومراكز الثقل \omterm{def:b2:affine:def}{والفضاءات الأفينية} الجزئية والتطبيقات الأفينية)،
والنظرة الأفينية إلى التحدب، وأدوات التصنيف المستعملة في فصول
الهندسة الآتية.

\section{النقط والمتجهات}

\begin{definition}\label{def:b2:affine:def}
\emph{الفضاء الأفيني}\index{فضاء أفيني} الموجه بفضاء متجهي حقيقي
$E$ هو مجموعة غير خالية $\mathcal{E}$ مزودة بتطبيق $(A, B) \mapsto \vect{AB} \in E$ يحقق
\[
\vect{AB} + \vect{BC} = \vect{AC}
\quad \text{(علاقة شال)},
\qquad
\text{ومن أجل كل } A,\ B \mapsto \vect{AB} \text{ يكون تقابلا }
\mathcal{E} \to E .
\]
ونكتب $B = A + u$ للدلالة على النقطة الوحيدة التي تحقق $\vect{AB} = u$.
و\emph{بُعد} $\mathcal{E}$ هو $\dim E$. وكل فضاء متجهي فضاء أفيني على نفسه
($\vect{AB} = B - A$)؛ وكل اختيار لمبدأ $O \in \mathcal{E}$ يطابق $\mathcal{E}$ مع $E$ عبر $M \mapsto \vect{OM}$.
\end{definition}

\begin{example}[فضاء أفيني بلا مبدأ طبيعي]\label{ex:b2:affine:planeexample}
مستوي الحلول $\mathcal E = \{(x, y, z) \in \R^3 : x + y
+ z = 1\}$ ليس فضاء متجهيا جزئيا ($0 \notin \mathcal E$)، لكنه \omterm{def:b2:affine:def}{فضاء أفيني}
موجه بالفضاء $E = \{x + y + z =
0\}$: فمن أجل $A, B \in \mathcal E$ يقع الفرق $\vect{AB} = B
- A$ في $E$
(فالمجموعان يتلاشيان)، وعلاقة شال موروثة من $\R^3$، والتطبيق $B \mapsto \vect{AB}$
تقابلي على $E$.
ولا تتميز أي نقطة من $\mathcal E$ --- فأي اختيار ``لمبدأ'' $O \in \mathcal E$ صالح
بالقدر نفسه، وجميع المطابقات $M \mapsto \vect{OM}$ تختلف بانسحابات.
وهذا هو الوضع النمطي: فمجموعات حلول المسائل الخطية غير المتجانسة
(الجمل الخطية، والمعادلات التفاضلية الخطية في \cref{ch:b2:diffeq}) أفينية لا خطية
أبدا، والشعار ``حل خاص زائد النواة'' هو بالضبط النص $\mathcal F = A + F$ في التعريف
التالي.
\end{example}

\begin{definition}[مركز الثقل]\label{def:b2:affine:barycenter}
لتكن $(A_i, \lambda_i)_{i \leq k}$ نقطا موزونة تحقق $\sum\lambda_i \neq 0$. \emph{مركز الثقل}\index{مركز الثقل}
$G
= \operatorname{bar}\bigl((A_i, \lambda_i)\bigr)$ هو النقطة الوحيدة التي تحقق
\[
\sum_i \lambda_i\, \vect{GA_i} = 0
\qquad\text{وبصورة مكافئة}\qquad
\vect{OG} = \frac{1}{\sum\lambda_i}\sum_i \lambda_i\,\vect{OA_i}
\quad (\text{من أجل أي } O).
\]
ومراكز الثقل \emph{تجميعية} (إذ يمكن استبدال أي مجموعة جزئية من
النقط بمركز ثقلها الجزئي مع مجموع أوزانها)، وثابتة عند ضرب جميع
الأوزان في عدد.
\end{definition}

\begin{proof}[برهان الوجود والصيغ]
نثبت $O$ ونكتب $s = \sum_i\lambda_i \neq 0$. وبعلاقة شال،
\[
\sum_i\lambda_i\,\vect{GA_i} = 0
\iff \sum_i\lambda_i\bigl(\vect{GO} + \vect{OA_i}\bigr) = 0
\iff s\,\vect{OG} = \sum_i\lambda_i\,\vect{OA_i},
\]
وهو ما يحدد $G = O + \frac1s\sum_i\lambda_i\vect{OA_i}$ تحديدا وحيدا. \emph{الاستقلال عن $O$:} من أجل
مبدأ آخر $O'$،
\[
\frac1s\sum_i\lambda_i\,\vect{O'A_i}
= \frac1s\sum_i\lambda_i\bigl(\vect{O'O} + \vect{OA_i}\bigr)
= \vect{O'O} + \frac1s\sum_i\lambda_i\,\vect{OA_i}
= \vect{O'G} :
\]
أي النقطة $G$ نفسها. \emph{التجميعية:} نقسم مجموعة الأدلة على
الصورة $I \sqcup J$ مع $s_I = \sum_{i\in I}\lambda_i \neq 0$، ونسمي $G_I$ مركز ثقل $(A_i, \lambda_i)_{i\in
I}$، بحيث يكون
$\sum_{i\in I}\lambda_i\vect{OA_i} =
s_I\,\vect{OG_I}$. عندئذ
\[
s\,\vect{OG}
= \sum_{i\in I}\lambda_i\vect{OA_i} +
\sum_{j\in J}\lambda_j\vect{OA_j}
= s_I\,\vect{OG_I} + \sum_{j\in J}\lambda_j\vect{OA_j} :
\]
فالنقطة $G$ هي مركز ثقل $(G_I, s_I)$ مع $(A_j,
\lambda_j)_{j\in J}$، كما ادُّعي. \emph{ضرب
الأوزان:} استبدال كل $\lambda_i$ بالمقدار $t\lambda_i$ ($t \neq 0$) يضرب $s$
والمجموع الموزون في $t$، فيبقى $\vect{OG}$ دون تغيير.
\end{proof}

\begin{remark}[مزالق شائعة]
يحيط بالتعريف فخّان. أولا، إذا كان مجموع الأوزان \emph{معدوما}، فلا
وجود \omterm{def:b2:affine:barycenter}{لمركز الثقل}: فالتطبيق $O \mapsto
\sum\lambda_i\vect{OA_i}$ يصير عندئذ مستقلا عن $O$ ويعرّف
\emph{متجهة} لا نقطة --- فالمقدار $(A, -1;\ B, 1)$ مثلا يرمز إلى $\vect{AB}$. وتتبع
أي من الكائنين ينتجه حساب ما هو نصف نظافة حساب مراكز الثقل. وثانيا،
لا معنى للأوزان إلا إلى غاية عامل مشترك غير معدوم؛ فصيغ من نوع
``إحداثيات $G$ هي $\lambda_1, \dots, \lambda_k$'' تفترض توحيدا مسبقا (وهو عادة $\sum\lambda_i = 1$)،
ونسيان التوحيد هو المصدر المعتاد للنسب الخاطئة على الشكل.
\end{remark}

\begin{definition}[الفضاءات الأفينية الجزئية؛ التطبيقات الأفينية]\label{def:b2:affine:subspace}
\emph{\omterm{def:b2:affine:def}{الفضاء الأفيني} الجزئي} هو مجموعة $\mathcal{F} = A + F = \{A + u :
u \in F\}$ حيث $F$ فضاء متجهي
جزئي (وهو \emph{اتجاهه})؛ وبصورة مكافئة، مجموعة غير خالية مستقرة
بمراكز الثقل. \omterm{def:b2:affine:def}{والفضاءات الأفينية} الجزئية للفضاء $\R^n$ هي بالضبط
مجموعات حلول الجمل الخطية $MX = B$ (السنة الأولى: حل خاص زائد النواة).
ويكون التطبيق $f \colon
\mathcal{E} \to \mathcal{E}'$ \emph{أفينيا}\index{تطبيق أفيني} إذا حافظ على
مراكز الثقل --- وبصورة مكافئة إذا كان
\[
f(A + u) = f(A) + \varphi(u)
\]
من أجل تطبيق خطي (وحيد) $\varphi = \vec f$، وهو \emph{الجزء الخطي}. والتطبيقات
الأفينية للفضاء $\R^n$: $X \mapsto MX + C$. والمركبات أفينية وأجزاؤها الخطية
مركبة؛ ويكون $f$ تقابليا إذا وفقط إذا كان $\vec f$ كذلك.
\end{definition}

\begin{proof}[برهان التكافؤ في حالة التطبيقات]
إذا كان $f(A + u) = f(A) + \varphi(u)$: فمن أجل مركز ثقل $G$ للنقط $(A_i,
\lambda_i)$، وبنشر كل نقطة
انطلاقا من $A$، لدينا $f(G) = f(A) +
\varphi(\vect{AG})$ و$\varphi(\vect{AG}) =
\frac{\sum\lambda_i\varphi(\vect{AA_i})}{\sum\lambda_i}$: ومنه فإن $f(G)$ مركز ثقل
الصور. وبالعكس، نثبت $A$ ونضع $\varphi(u) = \vect{f(A)\,f(A + u)}$. \emph{التجانس:} لدينا $A + tu =
\operatorname{bar}\bigl(A, 1-t;\ A + u, t\bigr)$
من أجل كل عدد حقيقي $t$، ومنه فإن الحفاظ على مراكز الثقل (بأوزان
حقيقية كيفما كانت، كما هو مفترض) يعطي $\varphi(tu) = t\,\varphi(u)$ مباشرة.
\emph{الجمعية:} $A + u + v = \operatorname{bar}\bigl(A + 2u,
\tfrac12;\ A + 2v, \tfrac12\bigr)$، ومنه $\varphi(u + v) =
\tfrac12\varphi(2u) + \tfrac12\varphi(2v) = \varphi(u) +
\varphi(v)$، باستعمال التجانس. ومنه فإن $\varphi$
خطي.
\end{proof}

\begin{remark}
استعمل البرهان مراكز ثقل بأوزان \emph{حقيقية كيفما كانت}: فخطوة
التجانس تخرج بالمقدار $t$ خارج $\intcc01$.
فإذا اكتفينا بافتراض أن التطبيق يحافظ على مراكز الثقل ذات الأوزان
الموجبة --- وبصورة مكافئة على المنتصفات والقطع --- فلا تأتي خطية
التطبيق المتجهي مجانا: إذ لا نحصل إلا على الخطية بالنسبة إلى $\Q$،
ونحتاج إلى فرضية اتصال كي نستنتج، تماما كما في
\cref{exo:b2:affine:5}. والتمييز بين ``يحافظ على جميع مراكز الثقل'' و``يحافظ على
التوليفات المحدبة'' دقة صغيرة لكنها حقيقية في المعجم الأفيني.
\end{remark}

\begin{example}[هندسة مراكز الثقل الكلاسيكية]\label{ex:b2:affine:median}
مركز ثقل المثلث $ABC$ هو \omterm{def:b2:affine:barycenter}{مركز الثقل} $G =
\operatorname{bar}(A,1; B,1; C,1)$. والتجميعية مع منتصف
$A' = \operatorname{bar}(B, 1; C, 1)$ تبين أن
\[
G = \operatorname{bar}(A, 1;\ A', 2) :
\]
فالنقطة $G$ تقع على المتوسط $AA'$ عند ثلثيه --- والأمر نفسه
للمتوسطين الآخرين: فالمتوسطات الثلاثة متلاقية، في سطر واحد من حساب
مراكز الثقل.
\end{example}

\begin{example}[متوسطا الرباعي الثنائيان]\label{ex:b2:affine:bimedians}
ليكن $ABCD$ رباعيا كيفما كان (مستويا أو غير مستوي!) ولننظر في
\emph{متوسطيه الثنائيين}: أي القطعتين الواصلتين بين منتصفي الضلعين
المتقابلين، $M_{AB}M_{CD}$ و$M_{BC}M_{DA}$. نُدخل \omterm{def:b2:affine:barycenter}{مركز الثقل} $G$ للنقط $(A,1; B,1; C,1; D,1)$
ونجمّع الأوزان بطريقتين:
\[
G = \operatorname{bar}\bigl(M_{AB}, 2;\ M_{CD}, 2\bigr)
= \operatorname{bar}\bigl(M_{BC}, 2;\ M_{DA}, 2\bigr) :
\]
فالنقطة $G$ هي منتصف \emph{كلا} المتوسطين الثنائيين --- ومنه فإن
المتوسطين الثنائيين ينصف كل منهما الآخر دائما، وإن الرباعي المكوّن
من المنتصفات الأربعة متوازي أضلاع (وقطراه هما المتوسطان الثنائيان).
ولا مناقشة حالات ولا إحداثيات، وتصمد الحجة دون تغيير من أجل رباعي
غير مستوي في $\R^3$، حيث يكون البرهان القائم على الشكل دقيقا أصلا:
فالتجميعية لا تأبه بالبعد.
\end{example}

\begin{example}[تصنيف تطبيق أفيني، من البداية إلى النهاية]\label{ex:b2:affine:dilation}
لنأخذ $f(x, y) = (2x - 1,\ 3y - 4)$ على $\R^2$. جزؤه الخطي هو $\varphi = \operatorname{diag}(2, 3)$، وطيفه $\{2, 3\}$ يتجنب
$1$: فبمحك النقطة الثابتة المبرهن أدناه (\cref{prop:b2:affine:fixedpoint})، يقبل $f$
نقطة ثابتة وحيدة، تُوجد بحل
\[
x = 2x - 1, \qquad y = 3y - 4
\qquad\Longrightarrow\qquad \Omega = (1, 2).
\]
وبإعادة التمركز في $\Omega$ (بوضع $x = 1 + u$، $y = 2 + v$):
\[
f(1 + u,\ 2 + v) = (1 + 2u,\ 2 + 3v) :
\]
ففي المعلم ذي المبدأ $\Omega$، \emph{يكون} $f$ جزأه الخطي نفسه،
وهو تمدد غير موحد الاتجاه يمدد بالمعامل $2$ أفقيا وبالمعامل
$3$ شاقوليا انطلاقا من المركز $(1, 2)$. والدرس العام: \omterm{def:b2:affine:subspace}{التطبيق
الأفيني} هو ``تطبيق خطي زائد معطى موضعي''، وينهار المعطى الموضعي إلى
مبدأ واحد حسن الاختيار كلما لم يكن $1$ \omterm{def:b2:reduction:eigen}{قيمة ذاتية}. وبالعكس، فإن
نقل المبدأ نقلا سيئا \emph{يخلق} الحدود الثابتة: فالهندسة الأفينية
هي فن اختيار موضع $0$.
\end{example}

\begin{remark}[طريقة: التلاقي والاستقامة بمراكز الثقل]
المثال \cref{ex:b2:affine:median} حالة خاصة من وصفة عامة. فلإثبات تلاقي ثلاثة خطوط
سيفيانية في مثلث، أظهر جملة موزونة \emph{واحدة} $(A,
\alpha; B, \beta; C, \gamma)$ واستعمل
التجميعية بثلاث طرائق: فتجميع $(B, C)$ يبين أن \omterm{def:b2:affine:barycenter}{مركز الثقل} يقع على
السيفيان الخارج من $A$، وتجميع $(C, A)$ يبين وقوعه على السيفيان
الخارج من $B$، وتجميع $(A, B)$ على الثالث. وفي حالة المتوسطات،
تقوم الجملة $(A, 1; B, 1; C, 1)$ بالعمل كله؛ وفي حالة الخطوط السيفيانية التي تقطع
الأضلاع بنسب مفروضة، تُقرأ الأوزان من النسب. ولإثبات \emph{استقامة}
ثلاث نقط، اكتب إحداها مركز ثقل للنقطتين الأخريين (\cref{exo:b2:affine:2})، أو
استعمل محك \omterm{def:b2:linalg:det}{المحدد} في \cref{exo:b2:affine:11}. وتستبدل الوصفتان كلتاهما بالبراعة
الهندسية مسك حسابات الأوزان --- وهذا بالضبط ما وُجد له حساب مراكز
الثقل.
\end{remark}

\section{التحدب من زاوية أفينية}

\begin{definition}\label{def:b2:affine:convex}
تكون مجموعة جزئية $C$ من \omterm{def:b2:affine:def}{فضاء أفيني} \emph{محدبة}\index{مجموعة محدبة}
إذا احتوت كل مركز ثقل بأوزان \emph{موجبة} لنقطها --- وبصورة مكافئة،
كل قطعة $\intcc{A}{B} = \{\operatorname{bar}(A,
1-t; B, t) : t \in \intcc{0}{1}\}$ بين نقطتين منها. و\emph{الغلاف المحدب}\index{الغلاف المحدب}
$\operatorname{conv}(S)$ هو مجموعة جميع مراكز الثقل ذات الأوزان الموجبة لنقط $S$
--- أي أصغر مجموعة محدبة تحوي $S$.
\end{definition}

\begin{example}[المناطق الواقعة فوق المنحنيات مجموعات محدبة]\label{ex:b2:affine:epigraph}
المنطقة $C = \{(x, y) : y \geq x^2\}$ الواقعة فوق القطع المكافئ محدبة: فمن أجل $(x_1, y_1), (x_2, y_2) \in C$ ومن
أجل $t \in
\intcc01$، تعطي متراجحة تحدب دالة المربع
\[
\bigl((1-t)x_1 + tx_2\bigr)^2 \leq (1-t)x_1^2 + tx_2^2
\leq (1-t)y_1 + ty_2 ,
\]
ومنه يبقى \omterm{def:b2:affine:barycenter}{مركز الثقل} فوق القطع المكافئ. والحساب عام: فالمجموعة
$\{y \geq f(x)\}$ محدبة إذا وفقط إذا كانت $f$ دالة محدبة --- \emph{\omterm{def:b2:affine:convex}{فالمجموعات المحدبة}}
و\emph{الدوال المحدبة} (\cref{ch:b2:realfun}) وجهان لمفهوم واحد، والمناطق
الواقعة فوق المنحنيات هي المعجم بينهما. وهذا هو السبب الهندسي في
وجود مستقيمات مساندة للدوال المحدبة، وهي الحقيقة التي ستبرهن على
متراجحة جنسن في \cref{ch:b2:randomvar}.
\end{example}

\begin{example}[مولّدات زائدة لغلاف محدب]\label{ex:b2:affine:squarehull}
لنأخذ $S = \{(0,0), (2,0), (2,2), (0,2), (1,1)\}$. النقطة الخامسة هي \omterm{def:b2:affine:barycenter}{مركز الثقل}
\[
(1,1) = \operatorname{bar}\bigl((0,0), \tfrac12;\ (2,2),
\tfrac12\bigr),
\]
ومنه فهي تقع أصلا في غلاف النقط الأربع الأخرى: فالمجموعة $\operatorname{conv}(S)$ هي
المربع ذو الرؤوس الأربعة. وبوجه عام، يمكن حذف أي نقطة من $S$ تكون
مركز ثقل بأوزان موجبة للنقط \emph{الأخرى} من $S$ دون تغيير
الغلاف؛ والنقط التي لا يمكن حذفها أبدا (وهي هنا الرؤوس الأربعة) هي
\emph{النقط المتطرفة} للغلاف. وتحديدها حساب مراكز ثقل خالص: فالنقطة
$(2,0)$ مثلا لا يمكن كتابتها $\operatorname{bar}$ بالنقط الباقية بأوزان موجبة، لأن
الإحداثية الأولى ستفرض تركيز كل الوزن على النقط ذات $x = 2$، ثم تسقط
الإحداثية الثانية. فأسئلة التحدب تختزل، مرة بعد مرة، إلى حل جمل
موزونة صغيرة.
\end{example}

\begin{theorem}[كاراتيودوري]\label{thm:b2:affine:caratheodory}
في \omterm{def:b2:affine:def}{فضاء أفيني} بعده $n$، تكون كل نقطة من $\operatorname{conv}(S)$ مركز ثقل لعدد من
نقط $S$ لا يتجاوز $n + 1$.\index{مبرهنة كاراتيودوري}
\end{theorem}

\begin{proof}
لتكن $G = \operatorname{bar}(A_0, \lambda_0; \dots; A_k, \lambda_k)$ مع $\lambda_i > 0$، $\sum\lambda_i = 1$، و$k + 1 > n + 1$ نقطة. المتجهات $\vect{A_0A_i}$ وعددها
$k$ ($i \geq 1$) مرتبطة ($k > n$): ومنه $\sum_{i\geq1}\mu_i \vect{A_0A_i} = 0$ بصورة غير تافهة؛
وبوضع $\mu_0 = -\sum_{i\geq1}\mu_i$، نحصل على أوزان $(\mu_i)$ مع $\sum\mu_i = 0$، $\sum \mu_i\,\vect{OA_i} = 0$ (من أجل أي
$O$)، وهي ليست معدومة كلها. عندئذ، من أجل كل عدد حقيقي $t$،
تبقى الأوزان $\lambda_i - t\mu_i$ مجموعها $1$، وبما أن $\sum_i\mu_i\vect{OA_i} = 0$، فإن
\[
\sum_i(\lambda_i - t\mu_i)\,\vect{OA_i}
= \sum_i\lambda_i\,\vect{OA_i} :
\]
فهي تنتج النقطة $G$ \emph{نفسها}. ولننزلق الآن بالمقدار $t$
انطلاقا من $0$: بعض المقادير $\mu_i$ موجب (فمجموعها معدوم وهي
ليست معدومة كلها)، ومنه فإن
\[
t^* = \min\Bigl\{\frac{\lambda_i}{\mu_i} : \mu_i > 0\Bigr\}
\]
معرَّف وموجب. وعند $t = t^*$: من أجل الأدلة التي فيها $\mu_i > 0$، لدينا
$\lambda_i - t^*\mu_i \geq 0$ بحكم الأصغرية، مع تساو عند دليل يحقق الأصغرية؛ ومن أجل الأدلة
التي فيها $\mu_i
\leq 0$، لدينا $\lambda_i - t^*\mu_i \geq \lambda_i > 0$. فتبقى جميع الأوزان موجبة وقد مات
واحد منها على الأقل: ومنه تُعاد كتابة $G$ مركز ثقل لعدد أقل من
النقط. ونكرر ما دام العدد المتبقي من النقط يتجاوز $n + 1$.
\end{proof}

\begin{example}
في المستوي ($n = 2$): تقع كل نقطة من \omterm{def:b2:affine:convex}{الغلاف المحدب} لمجموعة منتهية في
مثلث رؤوسه من المجموعة --- وهو المضمون الهندسي لمبرهنة كاراتيودوري،
المستعملة في الأمثلة وفي الاحتمالات (الخلائط) على حد سواء.
\end{example}

\begin{example}[تشغيل خوارزمية كاراتيودوري]\label{ex:b2:affine:caratheodoryrun}
لنكتب مركز مربع المثال \cref{ex:b2:affine:squarehull} بدلالة رؤوسه الأربعة $A_1 =
(0,0)$ و$A_2 = (2,0)$
و$A_3 = (2,2)$ و$A_4 = (0,2)$:
\[
(1,1) = \operatorname{bar}\bigl(A_1, \tfrac14;\ A_2,
\tfrac14;\ A_3, \tfrac14;\ A_4, \tfrac14\bigr),
\]
أي أربع نقط في البعد $2$ --- واحدة أكثر من اللازم. وتطلب وصفة
البرهان أوزانا $(\mu_i)$ تحقق $\sum\mu_i = 0$ و$\sum\mu_i\vect{OA_i} = 0$: وهنا يفي $\mu = (1, -1, 1, -1)$ بالغرض
(فالقطران يتقاسمان منتصفهما). والانزلاق بالمقدار $\lambda_i
\mapsto \lambda_i - t\mu_i$ يُبقي \omterm{def:b2:affine:barycenter}{مركز
الثقل} ثابتا من أجل كل $t$؛ والقيمة الحدية المقبولة $t = \frac14$ تجعل
الأوزان $(0, \tfrac12, 0, \tfrac12)$، فتقتل $A_1$ و$A_3$ في آن واحد:
\[
(1,1) = \operatorname{bar}\bigl(A_2, \tfrac12;\ A_4,
\tfrac12\bigr),
\]
أي تمثيل بنقطتين --- وهو أفضل حتى من النقط الثلاث التي تضمنها
المبرهنة، لأن المركز يقع مصادفةً على قطعة بين مولّدين. والخوارزمية
آلية تماما: جد تعلقا، وانزلق حتى يموت وزن، ثم كرّر.
\end{example}

\section{أدوات التصنيف الأفيني}

\begin{proposition}[النقط الثابتة للتطبيقات الأفينية]\label{prop:b2:affine:fixedpoint}
ليكن $f$ تشاكلا \omterm{def:b2:affine:subspace}{أفينيا} داخليا \omterm{def:b2:affine:def}{لفضاء أفيني} منتهي البعد جزؤه الخطي
$\varphi$. إذا كان $1 \notin
\operatorname{Sp}(\varphi)$، فإن $f$ يقبل نقطة ثابتة وحيدة $\Omega$،
وباتخاذ $\Omega$ مبدأ \emph{يكون} $f$ جزأه الخطي نفسه. (والانسحابات،
التي فيها $\varphi = \mathrm{id}$ ولا نقطة ثابتة، هي العائق الأساسي.)
\end{proposition}

\begin{proof}
نثبت $O$ ونكتب $f(O + x) = f(O) + \varphi(x)$. تكون النقطة $O + x$ ثابتة إذا وفقط إذا كان
$O + x = f(O) + \varphi(x)$، أي
\[
(\mathrm{id} - \varphi)(x) = \vect{O f(O)} .
\]
وفي البعد المنتهي، يكون $\mathrm{id} - \varphi$ قابلا للقلب إذا وفقط إذا لم يكن
$0$ \omterm{def:b2:reduction:eigen}{قيمة ذاتية} للتطبيق $\mathrm{id} - \varphi$، أي إذا وفقط إذا كان $1 \notin \operatorname{Sp}\varphi$ ---
وعندئذ تقبل المعادلة المكتوبة أعلاه حلا وحيدا $x^*$، فيعطي النقطة
الثابتة الوحيدة $\Omega = O + x^*$. وإعادة التمركز: من أجل أي متجهة $u$،
\[
f(\Omega + u) = f(\Omega) + \varphi(u) = \Omega + \varphi(u),
\]
ومنه يُكتب التطبيق في المعلم ذي المبدأ $\Omega$ على الصورة $u \mapsto
\varphi(u)$: أي
خطيا محضا. أما إذا كان $1 \in
\operatorname{Sp}\varphi$، فإما ألا توجد أي نقطة ثابتة (فقد
تكون المعادلة المكتوبة أعلاه غير قابلة للحل، كما في حالة انسحاب)
وإما أن يوجد \omterm{def:b2:affine:def}{فضاء أفيني} جزئي كامل منها (بإضافة أي \omterm{def:b2:reduction:eigen}{متجهة ذاتية} موافقة
\omterm{def:b2:reduction:eigen}{للقيمة الذاتية} $1$ إلى حل): فالوحدانية هي بالضبط الشرط الطيفي.
\end{proof}

\begin{example}[تقايسات المستوي، مكتملة]\label{ex:b2:affine:isometries}
للتقايس الأفيني للمستوي الإقليدي جزء خطي في $O(2)$: إما دوران $R_\theta$
وإما انعكاس (مجلد السنة الأولى). فإذا كان $\theta \neq 0$: فإن $1 \notin \operatorname{Sp} R_\theta$، ومنه
يكون التطبيق \emph{دورانا حول مركز وحيد} (\cref{prop:b2:affine:fixedpoint}). وإذا كان الجزء
الخطي انعكاسا: فإما انعكاس حول محور (فتوجد نقط ثابتة) وإما
\emph{انعكاس منزلق} (أي انعكاس مركب مع انسحاب موازٍ للمحور، بلا نقطة
ثابتة). ومع الانسحابات، هذا هو التصنيف الكامل لتقايسات المستوي.
\end{example}

\begin{remark}[تقايسات المستوي في نظرة واحدة]
بجمع الحالات: التطبيق المطابق؛ والانسحابات ($\vec f =
\mathrm{id}$، بلا نقطة ثابتة
إلا في الحالة التافهة)؛ والدورانات (جزؤها الخطي $R_\theta$، $\theta \neq 0$: بمركز
واحد)؛ والانعكاسات (جزؤها الخطي انعكاس، ولها مستقيم من النقط
الثابتة)؛ والانعكاسات المنزلقة (بالجزء الخطي نفسه، بلا نقطة ثابتة).
أربع عائلات زائد التطبيق المطابق، وكل منها يُعرف بمعطيين اثنين فقط:
الجزء الخطي ومجموعة النقط الثابتة --- وهو نمط \cref{prop:b2:affine:fixedpoint} بعد جعله
شاملا.
\end{remark}

\begin{example}[انعكاس منزلق، متلبسا بالجرم]\label{ex:b2:affine:glide}
لنأخذ $f(x, y) = (y + 1,\ x + 1)$. الجزء الخطي $(x, y)
\mapsto (y, x)$ هو الانعكاس حول القطر $y = x$،
ومنه $1 \in \operatorname{Sp}\vec f$ ويكون \cref{prop:b2:affine:fixedpoint} صامتا. أما النقط الثابتة فتستلزم $x = y + 1$
و$y = x + 1$ في آن واحد: وهو مستحيل --- فلا وجود لها، ومنه فإن $f$
ليس انعكاسا. ويحسم التربيع التصنيف:
\[
f\bigl(f(x, y)\bigr) = f(y + 1,\ x + 1) = (x + 2,\ y + 2),
\]
أي الانسحاب بالمتجهة $(2, 2)$: ومنه فإن $f$ هو \emph{الانعكاس
المنزلق} الذي محوره المستقيم $y = x$ (منقولا بما يناسب: فمنتصف $M$
و$f(M)$ يقع دائما على $y = x + {}$ثابت، وهو هنا $y = x$، كما يُتحقق على
$M = (0,
0) \mapsto (1,1)$) ومتجهة انزلاقه $(1, 1)$، وهو نصف $f
\circ f$. وقارن ذلك بالمثال
\cref{exo:b2:affine:6}، حيث أعطى الجزء الخطي نفسه مع ثابت مختلف انعكاسا صريحا: فمع
حضور \omterm{def:b2:reduction:eigen}{القيمة الذاتية} $1$، يقرر الحد الثابت كل شيء.
\end{example}

\begin{example}[المتتاليات التراجعية الأفينية ديناميكا أفينية]\label{ex:b2:affine:recursion}
المتتالية التراجعية الكلاسيكية $u_{n+1} = au_n + b$ ($a \neq 1$) تكرّر \omterm{def:b2:affine:subspace}{التطبيق
الأفيني} $f(x) = ax + b$ للمستقيم، وجزؤه الخطي $a$ يتجنب \omterm{def:b2:reduction:eigen}{القيمة الذاتية}
$1$: فتوجد نقطة ثابتة وحيدة $\omega = \frac{b}{1-a}$، وإعادة التمركز فيها (وهي
الحالة ذات البعد الواحد من القضية أعلاه) تحول $f$ إلى ضرب في
$a$:
\[
u_{n+1} - \omega = a\,(u_n - \omega)
\qquad\Longrightarrow\qquad
u_n = \omega + a^n(u_0 - \omega) .
\]
ومن أجل $u_{n+1} = \frac{u_n}2 + 3$: $\omega = 6$ و$u_n = 6 +
(u_0 - 6)2^{-n} \to 6$. والوصفة التي تُعلَّم من أجل مثل هذه
المتتاليات في \cref{ch:b2:series} --- ``اطرح النقطة الثابتة'' --- هي بالضبط
إعادة تمركز \omterm{def:b2:affine:subspace}{تطبيق أفيني} عند نقطته الثابتة؛ والتقارب من أجل $\abs a < 1$ هو ظاهرة
التقلص التي حولها \cref{ch:b2:metric} إلى مبرهنة نقطة بناخ الثابتة. فكرة واحدة،
وثلاثة فصول.
\end{example}

\begin{example}[إيجاد مركز دوران]\label{ex:b2:affine:rotationcenter}
لنأخذ $f(x, y) = (-y + 2,\ x)$. الجزء الخطي هو $\varphi(x, y) = (-y, x)$: أي الدوران بالزاوية $\frac\pi2$،
وطيفه $\{\iu, -\iu\}$ يتجنب $1$. وبحسب \cref{prop:b2:affine:fixedpoint} توجد نقطة ثابتة وحيدة
بالضبط: فالمعادلتان $x = -y + 2$ و$y = x$ تعطيان $x = 1$، $y = 1$، ومنه
$\Omega = (1, 1)$، و\emph{يكون} $f$ الدوران ذا المركز $(1, 1)$ والزاوية
$\frac\pi2$. والدرس العام: عندما يكون $1
\notin \operatorname{Sp}\vec f$، يكلف تصنيف $f$ جملة
خطية واحدة --- فالهندسة كلها في الجزء الخطي، والحساب كله في تحديد
موضع المركز.
\end{example}

\begin{remark}[أين تُستعمل اللغة الأفينية فيما يلي]
مراكز الثقل والتطبيقات الأفينية هي قواعد نحو فصول الهندسة الآتية:
فالمماسات والمستويات المماسة كائنات أفينية (\cref{ch:b2:curves,ch:b2:surfaces})، وتغيير المتغيرات
الأفيني يضرب المساحات والحجوم في $\abs{\det \vec f}$ (\cref{ch:b2:multint})، والأمل الرياضي
مركز ثقل بأوزان يعطيها قانون احتمال، وهذا هو سبب تحكم التحدب في
متراجحة جنسن (\cref{ch:b2:randomvar}). وفي مجلد السنة الثالثة يحمل معجم التحدب نفسه
دراسة معايير $L^p$ ومتراجحات التكامل.
\end{remark}

\begin{remark}[آفاق داخل هذا المجلد]
يخرج من هذا الفصل خيطان. الخيط \emph{الأفيني}: فالمماسات (\cref{ch:b2:curves})
والمستويات المماسة (\cref{ch:b2:surfaces}) فضاءات أفينية جزئية مرتبطة بكائنات
لاخطية، ويجري تصنيف السطوح من الدرجة الثانية في فصل السطوح على
معادلة المركز $A\Omega = -b$ الواردة في هذا الفصل. أما الخيط \emph{المحدب}
فأطول: فتحدب أنصاف المستويات والأقراص يشغّل نظرية هيلي في مسألة
نهاية الأسبوع؛ وتحدب الدوال يعطي متراجحة جنسن (\cref{ch:b2:randomvar})؛ ومبرهنة
الكتاب الأخيرة --- محك الانقراض في مسارات التفرع (\cref{ch:b2:genfun}) ---
يقررها موضع منحن محدب بالنسبة إلى القطر، وهي صورة تنتمي إلى هذا
الفصل بقدر انتمائها إلى الاحتمالات. وتعود مراكز الثقل هناك أيضا:
فالأمل الرياضي مركز ثقل بأوزان احتمالية.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b2:affine:1}
في $\R^3$، هل ما يلي فضاءات أفينية جزئية؟ أعط الاتجاهات والأبعاد.
$\{x + y + z = 1\}$؛ $\;\{x + y + z = 1,\ x - z =
3\}$؛ $\;\{x^2 + y^2 = 1\}$؛ ومجموعة حلول $MX = B$ من أجل جملة متوافقة
معطاة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:affine:2}
برهن على أن ثلاث نقط متمايزة $A, B, C$ من \omterm{def:b2:affine:def}{فضاء أفيني} تكون على استقامة
واحدة إذا وفقط إذا كانت $C$ مركز ثقل للنقطتين $A$ و$B$، وإذا
وفقط إذا كانت المتجهتان $\vect{AB}, \vect{AC}$ مرتبطتين. واستنتج مسك أوزان على طريقة
منيلاوس: إذا كان $C = \operatorname{bar}(A, 1 - t; B, t)$، فحدد موضع $C$ من أجل $t = \frac12$ و$t = 2$
و$t = -1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:affine:3}
ليكن $f$ \omterm{def:b2:affine:subspace}{التطبيق الأفيني} للفضاء $\R^2$ المعطى بالعلاقة $f(X) = MX + C$
مع $M = \frac12\begin{pmatrix} 1 & 1\\ 1 & 1\end{pmatrix}$ و$C =
(1, 0)^{\mathsf T}$. عيّن صورة $f$ ونقطه الثابتة (إن وُجدت)
و$f \circ f$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:4}
(التجميعية في العمل) في مثلث $ABC$، لتقسم النقط $I, J, K$ القطع
$BC$ و$CA$ و$AB$ بالنسب $\vect{BI} = \frac13\vect{BC}$ و$\vect{CJ} = \frac13\vect{CA}$ و$\vect{AK} = \frac13\vect{AB}$.
عبّر عن $I, J, K$ بمراكز ثقل واحسب مركز ثقل $(I,1;J,1;K,1)$: ماذا تجد، ولماذا
كان ذلك متوقعا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:5}
برهن على أن كل تطبيق $f \colon \R^n \to \R^n$ يحافظ على \emph{المنتصفات} ($f\bigl(\frac{A+B}{2}\bigr) = \frac{f(A) + f(B)}{2}$)
ويكون \emph{متصلا} يكون \omterm{def:b2:affine:subspace}{أفينيا}. \emph{(بيّن أن التطبيق المتجهي
$u \mapsto
f(O + u) - f(O)$ جمعي بفضل المنتصفات، ثم متجانس بالنسبة إلى $\Q$، ثم متجانس
بالنسبة إلى $\R$ بالاتصال --- وهي استراتيجية الكثافة نفسها
المستعملة من أجل معادلة كوشي الدالية في مجلد السنة الأولى؛ فأعد
اشتقاق الخطوات اللازمة هنا.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:6}
صنّف \omterm{def:b2:affine:subspace}{التطبيق الأفيني} $f(x, y) = (y + 1,\; x - 1)$ للمستوي الإقليدي: الجزء الخطي، والنقط
الثابتة، والطبيعة الهندسية (انعكاس؟ انعكاس منزلق؟). واحسب $f \circ f$
واستنتج.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:affine:7}
(رادون) لتكن $A_1, \dots, A_{n+2}$ نقطا من \omterm{def:b2:affine:def}{فضاء أفيني} بعده $n$. برهن على أنه
يمكن تقسيمها إلى مجموعتين منفصلتين يتقاطع غلافاهما المحدبان.
\emph{(كما في برهان كاراتيودوري، جد أوزانا $\mu_i$ ليست معدومة كلها
تحقق $\sum\mu_i = 0$ و$\sum\mu_i\vect{OA_i} = 0$؛ ثم افصل الأوزان الموجبة عن السالبة ووحّد
الطرفين.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:affine:8}
ليكن $f$ تشاكلا \omterm{def:b2:affine:subspace}{أفينيا} داخليا للفضاء $\R^n$ يحقق $f \circ f = f$. برهن
على أن $f$ هو الإسقاط الأفيني على \omterm{def:b2:affine:def}{الفضاء الأفيني} الجزئي $\operatorname{Fix}(f) = \operatorname{im} f$
وفق الاتجاه $\ker\vec f$، وبالعكس أن كل إسقاط من هذا النوع متساوي القوى.
\emph{(بيّن أولا أن $\operatorname{im} f$ يتكون من نقط ثابتة.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:affine:9}
لتكن $G = \operatorname{bar}(A, 1;\ B, 2;\ C, 3)$ في مثلث $ABC$. باستعمال التجميعية، بيّن أن المستقيم
$AG$ يقطع $BC$ في $M = \operatorname{bar}(B, 2;\ C, 3)$، وحدد موضع $G$ على القطعة $\intcc AM$؛
وحدد بالمثل موضع تقاطع $BG$ مع $CA$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:10}
من أجل $\lambda \neq 0$، \emph{التحاكي} $h_{\Omega,
\lambda}$ هو \omterm{def:b2:affine:subspace}{التطبيق الأفيني} الذي يثبّت
$\Omega$ وجزؤه الخطي $\lambda\,\mathrm{id}$. برهن على أن المركب $h_{\Omega', \mu} \circ h_{\Omega, \lambda}$ تحاك نسبته
$\lambda\mu$ عندما $\lambda\mu \neq 1$، وانسحاب عندما $\lambda\mu = 1$؛ وفي الحالة $\lambda = \mu = -1$
(تناظران مركزيان)، احسب متجهة الانسحاب.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:11}
(منيلاوس) في مثلث $ABC$، لتكن $A' \in (BC)$ و$B' \in
(CA)$ و$C' \in (AB)$، وكلها
متمايزة عن الرؤوس، ولنعرّف $\alpha, \beta, \gamma$ بالعلاقات $\vect{A'B} =
\alpha\,\vect{A'C}$ و$\vect{B'C} = \beta\,\vect{B'A}$
و$\vect{C'A} = \gamma\,\vect{C'B}$. برهن على أن $A', B', C'$ تقع على استقامة واحدة إذا وفقط إذا كان
$\alpha\beta\gamma = 1$. \emph{(اكتب كل نقطة مركز ثقل لرأسين؛ وبيّن أن ثلاث نقط تقع
على استقامة واحدة إذا وفقط إذا شكلت أسطر إحداثياتها الثقلية بالنسبة
إلى $(A, B, C)$ مصفوفة شاذة من الحجم $3 \times 3$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:affine:12}
برهن على أن \omterm{def:b2:affine:convex}{الغلاف المحدب} لمجموعة جزئية متراصة $K$ من $\R^n$
متراص. \emph{(بحسب \cref{thm:b2:affine:caratheodory}، تكون $\operatorname{conv}(K)$ صورة مجموعة متراصة بتطبيق
\omterm{def:b2:metric:continuity}{متصل}.)} وبيّن بمثال في $\R^2$ أن \omterm{def:b2:affine:convex}{الغلاف المحدب} لمجموعة \emph{مغلقة}
ليس بالضرورة مغلقا.
\end{exercise}

\begin{problem}[{مسألة نهاية الأسبوع --- مبرهنة هيلي واثنان من
عوائدها}]\label{pb:b2:affine:1}
تقول مبرهنة رادون (\cref{exo:b2:affine:7}) إن $n + 2$ نقطة من \omterm{def:b2:affine:def}{فضاء أفيني} بعده
$n$ تنقسم دائما إلى مجموعتين يتقاطع غلافاهما المحدبان. وتحول هذه
المسألة تلك الحقيقة الواحدة من \omterm{def:b2:structures:algebra}{الجبر} الخطي إلى سلسلة من مبرهنات
الهندسة التوفيقية: مبرهنة هيلي في التقاطع، ومبرهنة نقطة المركز (وهي
وسيط ذو بعدين)، ومبرهنة يونغ في التغطية. وفي كل ما يلي، المستوي هو
$\R^2$ ببنيته الإقليدية المعتادة، و$\det$ هو \omterm{def:b2:linalg:det}{المحدد} في الأساس
القانوني.\index{مبرهنة هيلي}\index{مبرهنة رادون}\index{مبرهنة يونغ}\index{نقطة المركز}

\medskip
\noindent\textbf{الجزء الأول --- الإحداثيات الثقلية.} تكون النقط
$A_0, \dots, A_k$ \emph{مستقلة \omterm{def:b2:affine:subspace}{أفينيا}} إذا كانت المتجهات $\vect{A_0A_1}, \dots, \vect{A_0A_k}$ مستقلة خطيا.
\begin{enumerate}
  \item بيّن أن الاستقلال الأفيني لا يتعلق باختيار نقطة الأساس
        $A_0$، وأنه يكافئ ما يلي: كلما عرّفت عائلتان من الأوزان،
        مجموع كل منهما $1$، \omterm{def:b2:affine:barycenter}{مركز الثقل} نفسه للنقط $(A_0,
        \dots, A_k)$، تطابقت
        الأوزان.
  \item لتكن $(A, B, C)$ مستقلة \omterm{def:b2:affine:subspace}{أفينيا} في المستوي. بيّن أن كل نقطة $M$
        تقبل ثلاثية وحيدة $(\alpha, \beta, \gamma)$ تحقق $\alpha + \beta +
        \gamma = 1$ و$M = \operatorname{bar}(A, \alpha; B,
        \beta; C, \gamma)$ --- وهي
        \emph{إحداثياتها الثقلية}.
  \item برهن على صيغ المحددات
        \[
        \alpha = \frac{\det(\vect{MB},
        \vect{MC})}{\det(\vect{AB}, \vect{AC})},
        \qquad
        \beta = \frac{\det(\vect{MC},
        \vect{MA})}{\det(\vect{AB}, \vect{AC})},
        \qquad
        \gamma = \frac{\det(\vect{MA},
        \vect{MB})}{\det(\vect{AB}, \vect{AC})} :
        \]
        فالإحداثيات الثقلية نسب مساحات جبرية.
  \item المستقيمات $BC$ و$CA$ و$AB$ هي مستقيمات الإحداثيات
        $\{\alpha = 0\}$ و$\{\beta = 0\}$ و$\{\gamma = 0\}$.
        بيّن أن $M$ تقع في المثلث المغلق $\operatorname{conv}\{A, B, C\}$ إذا وفقط إذا كان
        $\alpha, \beta,
        \gamma \geq 0$، وأن المستقيمات الثلاثة تقطع المستوي إلى سبع مناطق
        بالضبط، تصنّفها إشارات $(\alpha, \beta, \gamma)$ (مع استحالة نمط الإشارات
        $(-, -,
        -)$).
  \item ليكن $u \colon \R^2 \to \R$ تطبيقا \omterm{def:b2:affine:subspace}{أفينيا} (\emph{صورة أفينية}). بيّن أن
        $u(M) = \alpha\,u(A) +
        \beta\,u(B) + \gamma\,u(C)$، وأن مجموعات مستوى صورة أفينية غير ثابتة مستقيمات،
        وأن كل مستقيم ينشأ هكذا، وأن نصفي المستوي المغلقين $\{u
        \geq c\}$
        محدبان.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- تقسيمات رادون، بتدقيق.} تكون
عائلة من $n + 2$ نقطة من $\R^n$ في \emph{وضع عام} إذا كانت كل
$n + 1$ منها مستقلة \omterm{def:b2:affine:subspace}{أفينيا}. و\emph{التعلق الأفيني} للنقط $(A_1, \dots, A_{n+2})$ هو
عائلة $(\mu_i)$ تحقق $\sum_i \mu_i = 0$ و$\sum_i
\mu_i\,\vect{OA_i} = 0$ من أجل مبدأ $O$ واحد (ومنه
من أجل كل مبدأ).
\begin{enumerate}[resume]
  \item احسب تعلقا \omterm{def:b2:affine:subspace}{أفينيا} غير معدوم للنقط الأربع $A_1 = (0,0)$ و$A_2 = (3,0)$
        و$A_3 = (0,3)$ و$A_4 = (1,1)$؛ وأعط تقسيم رادون ونقطة رادون.
  \item بيّن أنه في حالة نقط في وضع عام، يكون بعد فضاء التعلقات
        الأفينية المتجهي $1$ بالضبط، وأن كل تعلق غير معدوم
        \emph{لا} يحوي أي معامل معدوم.
  \item استنتج أن تقسيم رادون للنقط $n + 2$ في وضع عام وحيد (إلى
        غاية تبادل الكتلتين)، وأن كل كتلة هي مجموعة الأدلة التي
        يأخذ فيها $\mu_i$ إشارة ثابتة واحدة.
  \item من أجل أربع نقط من المستوي في وضع عام، بيّن الثنائية: إما
        أن يكون التقسيم من النمط $(1, 3)$ --- أي نقطة داخل مثلث النقط
        الثلاث الأخرى --- وإما من النمط $(2, 2)$: فتكون النقط الأربع
        في وضع محدب وتتقاطع القطعتان الواصلتان بين الزوجين (أي
        القطران) في نقطة رادون.
  \item نفّذ السؤال 6 من أجل المربع الواحدي $(0,0)$ و$(1,0)$ و$(1,1)$
        و$(0,1)$: التعلق والتقسيم ونقطة رادون.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- مبرهنة هيلي في المستوي.}
\begin{enumerate}[resume]
  \item لتكن $C_1, C_2, C_3, C_4$ مجموعات جزئية محدبة من $\R^2$، لكل ثلاث منها
        نقطة مشتركة. خذ $x_i \in
        \bigcap_{j \neq i} C_j$ وطبق مبرهنة رادون على $x_1, \dots, x_4$: بيّن
        أن نقطة رادون تنتمي إلى المجموعات الأربع. \emph{(من أجل كل
        $k$، تتكون الكتلة التي لا تحوي $x_k$ من نقط من
        $C_k$.)}
  \item (هيلي) لتكن $C_1, \dots, C_m$ ($m \geq 3$) مجموعات جزئية محدبة من $\R^2$،
        تتقاطع كل ثلاث منها. برهن على $\bigcap_{i=1}^m C_i \neq \emptyset$، بالتراجع على $m$:
        استبدل $C_{m-1}$ و$C_m$ بالمجموعة $C_{m-1} \cap C_m$ وتحقق من الفرضية من
        أجل العائلة الجديدة باستعمال السؤال 11.
  \item ثلاثة أمثلة مضادة، واحد لكل فرضية: (1) أضلاع مثلث المغلقة
        الثلاثة تتقاطع مثنى مثنى لكن لا نقطة مشتركة بينها (فلا
        يمكن خفض $3$ إلى $2$)؛ (2) المجموعات الأربع $S_i = \{x_1, \dots, x_4\}
        \setminus \{x_i\}$،
        من أجل أربع نقط في وضع عام، تحقق فرضية التقاطع الثلاثي دون
        النتيجة (فالتحدب مهم)؛ (3) أنصاف المستويات المغلقة $H_k = \intco{k}{+\infty} \times
        \R$
        و$k \in \N$ تتقاطع مثنى مثنى وثلاثا ثلاثا لكن $\bigcap_k H_k = \emptyset$ (فالعائلات
        اللانهائية تحتاج إلى التراص).
  \item (هيلي المتراصة) لتكن $(K_i)_{i \in I}$ عائلة كيفما كانت من المجموعات
        الجزئية المحدبة المتراصة من $\R^2$، تتقاطع كل ثلاث منها.
        باستعمال السؤال 12 وخاصية بوريل--لوبيغ (\cref{thm:b2:metric:borellebesgue})، بيّن
        $\bigcap_{i \in I} K_i \neq \emptyset$.
  \item (العائد الأول) لتكن $S$ مجموعة منتهية من نقط المستوي
        ولتكن $r > 0$. بيّن: إذا كانت كل ثلاث نقط من $S$ تقع في
        قرص مغلق نصف قطره $r$، فإن $S$ تقع في قرص مغلق واحد
        نصف قطره $r$. \emph{(طبق مبرهنة هيلي على الأقراص $\overline D(p, r)$،
        $p \in S$.)}
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- مبرهنة نقطة المركز.}
\emph{نقطة المركز} لمجموعة منتهية $S$ من $n$ نقطة من المستوي
هي نقطة $c$ (لا تنتمي بالضرورة إلى $S$) بحيث يحوي كل نصف
مستو مغلق يحوي $c$ ما لا يقل عن $n/3$ نقطة من $S$.
\begin{enumerate}[resume]
  \item (البعد $1$) من أجل أعداد حقيقية $x_1 \leq \dots \leq x_n$، بيّن أن الوسيط
        $c = x_{\lceil n/2 \rceil}$ يحقق: كل نصف مستقيم مغلق يحوي $c$ يحوي ما لا يقل
        عن $n/2$ من المقادير $x_i$.
  \item (مبرهنة العد) إذا كانت $A, B, C$ مجموعات جزئية من $S$ تحقق
        $\abs A, \abs B, \abs C > \tfrac{2n}3$، فبيّن $A \cap
        B \cap C \neq \emptyset$.
  \item لتكن $m = \floor{2n/3} + 1$ ولتكن $\mathcal F$ العائلة (المنتهية) للأغلفة المحدبة
        $\operatorname{conv}(T)$ و$T \subseteq S$ و$\abs T =
        m$. بيّن أن لكل ثلاثة عناصر من $\mathcal F$
        نقطة مشتركة، واستنتج من مبرهنة هيلي وجود نقطة $c$ مشتركة
        بينها جميعا.
  \item برهن على أن هذه النقطة $c$ نقطة مركز للمجموعة $S$:
        وهي \emph{مبرهنة نقطة المركز}. \emph{(لو حوى نصف مستو مغلق
        يمر بالنقطة $c$ أقل من $n/3$ نقطة، لاحتوى متممه المفتوح
        مجموعة $T$ من $m$ نقطة، ولتجنبت $\operatorname{conv}(T)$ النقطة
        $c$.)}
  \item الأمثلية: خذ $n = 3k$ وضع $k$ نقطة في كل قرص من ثلاثة
        أقراص نصف قطرها صغير $\varepsilon$ ومراكزها رؤوس مثلث كبير. بيّن
        أنه من أجل \emph{كل} نقطة $c$ من المستوي يوجد نصف مستو
        مغلق يحوي $c$ ولا يحوي أكثر من $n/3$ نقطة من $S$،
        ومنه فإن الثابت $1/3$ لا يمكن تحسينه. \emph{(من بين
        الاتجاهات الثلاثة من $c$ نحو مراكز الأقراص، اثنان يصنعان
        زاوية لا تتجاوز $2\pi/3$.)}
\end{enumerate}

\medskip
\noindent\textbf{الجزء الخامس --- مبرهنة يونغ وتركيب.}
\begin{enumerate}[resume]
  \item (مبرهنة المثلث) لتكن $P, Q, R$ ثلاث نقط المسافات بينها مثنى
        مثنى $\leq 1$. بيّن أنها تقع في قرص مغلق نصف قطره $1/\sqrt3$.
        \emph{(إذا كانت إحدى الزوايا $\geq \pi/2$، فخذ القرص الذي قطره
        الضلع الأطول، مستعملا صيغة المتوسط $\norm{RM}^2 = \tfrac12\norm{RP}^2 +
        \tfrac12\norm{RQ}^2 - \tfrac14\norm{PQ}^2$؛ وإذا كان المثلث
        حاد الزوايا، فاحصر نصف قطر الدائرة المحيطة $a/(2\sin
        \widehat A)$ باستعمال
        زاويته الكبرى، الواقعة في $\intco{\pi/3}{\pi/2}$.)}
  \item (يونغ) استنتج: كل مجموعة جزئية متراصة من المستوي قطرها
        $\leq 1$ محتواة في قرص مغلق نصف قطره $1/\sqrt3$.
  \item الأمثلية: من أجل المثلث المتساوي الأضلاع $A_1A_2A_3$ ذي الضلع
        $1$ ومركز ثقله $G$، برهن على متطابقة ليبنتز $\sum_i \norm{\vect{OA_i}}^2 =
        3\norm{\vect{OG}}^2 + \sum_i \norm{\vect{GA_i}}^2$
        من أجل كل نقطة $O$، واستنتج أن نصف قطر أي قرص يحوي الرؤوس
        الثلاثة يبلغ $\geq
        1/\sqrt3$، مع تساو من أجل القرص المحيط وحده.
  \item (هيلي في $\R^n$) اذكر مبرهنة هيلي في $\R^n$ وبرهن عليها:
        إذا كانت مجموعات محدبة منتهية العدد بحيث تتقاطع كل $n + 1$
        منها، فإنها تتقاطع كلها. \emph{(تعالج مبرهنة رادون \cref{exo:b2:affine:7}
        حالة $n + 2$ مجموعة؛ ثم تراجع كما في السؤال 12.)}
  \item تركيب. اجمع السلسلة
        \[
        \text{التعلق الأفيني} \Rightarrow \text{رادون}
        \Rightarrow \text{هيلي} \Rightarrow
        \text{نقطة المركز ويونغ},
        \]
        مبيّنا في جملة واحدة عن كل بند: أين يدخل \omterm{def:b2:structures:algebra}{الجبر} الخطي، وأين
        تدخل إشارات الأوزان، وأين يدخل التحدب، وأي خطوة وحيدة
        استعملت بعد المستوي. وإلى ماذا تصير الثوابت $3$ (في
        مبرهنة هيلي) و$1/3$ (في نقطة المركز) و$1/\sqrt3$ (في مبرهنة
        يونغ) في $\R^n$؟ (اذكر دون برهان.)
\end{enumerate}
\end{problem}
