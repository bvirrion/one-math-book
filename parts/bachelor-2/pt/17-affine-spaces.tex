\chapter{Espaços afins}\label{ch:b2:affine}

Os espaços vetoriais têm um ponto privilegiado --- a origem --- que
a geometria não quer. Um \emph{\omterm{def:b2:affine:def}{espaço afim}} é um espaço vetorial
que esqueceu sua origem: pontos e vetores tornam-se espécies
diferentes, relacionadas por translação. Este capítulo curto constrói o
dicionário (pontos, \omterm{def:b2:affine:barycenter}{baricentros}, subespaços e \omterm{def:b2:affine:subspace}{aplicações afins}), a
visão \omterm{def:b2:affine:subspace}{afim} da convexidade e as ferramentas de classificação usadas nos
capítulos de geometria adiante.

\section{Pontos e vetores}

\begin{definition}\label{def:b2:affine:def}
Um \emph{espaço afim}\index{espaço afim} dirigido por um espaço
vetorial real $E$ é um conjunto não vazio $\mathcal{E}$ munido de uma aplicação
$(A, B) \mapsto \vect{AB} \in E$ satisfazendo
\[
\vect{AB} + \vect{BC} = \vect{AC}
\quad \text{(Chasles)},
\qquad
\text{para cada } A,\ B \mapsto \vect{AB} \text{ é uma bijeção }
\mathcal{E} \to E .
\]
Escreve-se $B = A + u$ para o único ponto com $\vect{AB} = u$.
A \emph{dimensão} de $\mathcal{E}$ é $\dim E$. Todo espaço
vetorial é um espaço \omterm{def:b2:affine:subspace}{afim} sobre si mesmo ($\vect{AB} = B - A$); toda
escolha de origem $O \in \mathcal{E}$ identifica $\mathcal{E}$ com
$E$ via $M \mapsto \vect{OM}$.
\end{definition}

\begin{example}[Um espaço afim sem origem natural]\label{ex:b2:affine:planeexample}
O plano solução $\mathcal E = \{(x, y, z) \in \R^3 : x + y
+ z = 1\}$ não é um subespaço vetorial ($0 \notin \mathcal E$),
mas é um \omterm{def:b2:affine:def}{espaço afim} dirigido por $E = \{x + y + z =
0\}$: para $A, B \in \mathcal E$ a diferença $\vect{AB} = B
- A$ cai em $E$ (as somas se cancelam), Chasles é herdado
de $\R^3$, e $B \mapsto \vect{AB}$ é bijetiva sobre $E$.
Nenhum ponto de $\mathcal E$ é distinguido --- qualquer escolha de
``origem'' $O \in \mathcal E$ funciona igualmente bem, e todas as
identificações $M \mapsto \vect{OM}$ diferem por translações.
Essa é a situação típica: os conjuntos solução de problemas
lineares não homogêneos (sistemas lineares, equações diferenciais
lineares no~\cref{ch:b2:diffeq}) são \omterm{def:b2:affine:subspace}{afins}, nunca lineares,
e o lema ``solução particular mais núcleo'' é exatamente
o enunciado $\mathcal F = A + F$ da definição seguinte.
\end{example}

\begin{definition}[Baricentro]\label{def:b2:affine:barycenter}
Sejam $(A_i, \lambda_i)_{i \leq k}$ pontos ponderados com
$\sum\lambda_i \neq 0$. O \emph{baricentro}\index{baricentro} $G
= \operatorname{bar}\bigl((A_i, \lambda_i)\bigr)$ é o único
ponto com
\[
\sum_i \lambda_i\, \vect{GA_i} = 0
\qquad\text{equivalentemente}\qquad
\vect{OG} = \frac{1}{\sum\lambda_i}\sum_i \lambda_i\,\vect{OA_i}
\quad (\text{quaisquer } O).
\]
Os baricentros são \emph{associativos} (subgrupos de pontos podem ser
substituídos por seu baricentro parcial com o peso somado) e
invariantes por reescalamento de todos os pesos.
\end{definition}

\begin{proof}[Demonstração da existência e das fórmulas]
Fixe $O$ e escreva $s = \sum_i\lambda_i \neq 0$. Por Chasles,
\[
\sum_i\lambda_i\,\vect{GA_i} = 0
\iff \sum_i\lambda_i\bigl(\vect{GO} + \vect{OA_i}\bigr) = 0
\iff s\,\vect{OG} = \sum_i\lambda_i\,\vect{OA_i},
\]
o que determina $G = O + \frac1s\sum_i\lambda_i\vect{OA_i}$
univocamente. \emph{Independência de $O$:} para outra origem
$O'$,
\[
\frac1s\sum_i\lambda_i\,\vect{O'A_i}
= \frac1s\sum_i\lambda_i\bigl(\vect{O'O} + \vect{OA_i}\bigr)
= \vect{O'O} + \frac1s\sum_i\lambda_i\,\vect{OA_i}
= \vect{O'G} :
\]
o mesmo ponto $G$. \emph{Associatividade:} separe o conjunto de índices
como $I \sqcup J$ com $s_I = \sum_{i\in I}\lambda_i \neq 0$,
e seja $G_I$ o \omterm{def:b2:affine:barycenter}{baricentro} de $(A_i, \lambda_i)_{i\in
I}$, de modo que $\sum_{i\in I}\lambda_i\vect{OA_i} =
s_I\,\vect{OG_I}$. Então
\[
s\,\vect{OG}
= \sum_{i\in I}\lambda_i\vect{OA_i} +
\sum_{j\in J}\lambda_j\vect{OA_j}
= s_I\,\vect{OG_I} + \sum_{j\in J}\lambda_j\vect{OA_j} :
\]
$G$ é o \omterm{def:b2:affine:barycenter}{baricentro} de $(G_I, s_I)$ junto com $(A_j,
\lambda_j)_{j\in J}$, como afirmado. \emph{Reescalamento:} substituir
cada $\lambda_i$ por $t\lambda_i$ ($t \neq 0$) multiplica $s$
e a soma ponderada por $t$, deixando $\vect{OG}$ inalterado.
\end{proof}

\begin{remark}[Armadilhas comuns]
Duas armadilhas cercam a definição. Primeiro, se os pesos somam
\emph{zero}, não há \omterm{def:b2:affine:barycenter}{baricentro}: a aplicação $O \mapsto
\sum\lambda_i\vect{OA_i}$ é então independente de $O$ e
define um \emph{vetor}, não um ponto --- por exemplo,
$(A, -1;\ B, 1)$ codifica $\vect{AB}$. Acompanhar qual
dos dois objetos um cálculo produz é metade da
higiene baricêntrica. Segundo, os pesos só fazem sentido a menos
de um fator comum não nulo; fórmulas como ``as coordenadas
de $G$ são $\lambda_1, \dots, \lambda_k$'' pressupõem uma
normalização (em geral $\sum\lambda_i = 1$), e esquecer
de normalizar é a fonte padrão de razões erradas numa
figura.
\end{remark}

\begin{definition}[Subespaços afins; aplicações afins]\label{def:b2:affine:subspace}
Um \emph{subespaço afim} é um conjunto $\mathcal{F} = A + F = \{A + u :
u \in F\}$ com $F$ um subespaço vetorial (sua \emph{direção});
equivalentemente, um conjunto não vazio estável por \omterm{def:b2:affine:barycenter}{baricentros}. Os subespaços
afins de $\R^n$ são exatamente os conjuntos solução de sistemas lineares
$MX = B$ (primeiro ano: solução particular mais núcleo). Uma aplicação $f \colon
\mathcal{E} \to \mathcal{E}'$ é \emph{afim}\index{aplicação afim}
quando preserva \omterm{def:b2:affine:barycenter}{baricentros} --- equivalentemente, quando
\[
f(A + u) = f(A) + \varphi(u)
\]
para uma (única) aplicação linear $\varphi = \vec f$, a \emph{parte
linear}. Aplicações afins de $\R^n$: $X \mapsto MX + C$. As composições são
afins com partes lineares compostas; $f$ é bijetiva se e somente se $\vec f$
o é.
\end{definition}

\begin{proof}[Demonstração da equivalência para aplicações]
Se $f(A + u) = f(A) + \varphi(u)$: para um \omterm{def:b2:affine:barycenter}{baricentro} $G$ de $(A_i,
\lambda_i)$, desenvolvendo todos os pontos a partir de $A$, $f(G) = f(A) +
\varphi(\vect{AG})$ e $\varphi(\vect{AG}) =
\frac{\sum\lambda_i\varphi(\vect{AA_i})}{\sum\lambda_i}$: $f(G)$ é
o \omterm{def:b2:affine:barycenter}{baricentro} das imagens. Reciprocamente, fixe $A$ e defina
$\varphi(u) = \vect{f(A)\,f(A + u)}$. \emph{Homogeneidade:} $A + tu =
\operatorname{bar}\bigl(A, 1-t;\ A + u, t\bigr)$ para todo real
$t$, de modo que a preservação de \omterm{def:b2:affine:barycenter}{baricentros} (com pesos reais arbitrários,
como hipotetizado) dá $\varphi(tu) = t\,\varphi(u)$ diretamente.
\emph{Aditividade:} $A + u + v = \operatorname{bar}\bigl(A + 2u,
\tfrac12;\ A + 2v, \tfrac12\bigr)$, logo $\varphi(u + v) =
\tfrac12\varphi(2u) + \tfrac12\varphi(2v) = \varphi(u) +
\varphi(v)$, usando a homogeneidade. Portanto $\varphi$ é linear.
\end{proof}

\begin{remark}
A demonstração usou \omterm{def:b2:affine:barycenter}{baricentros} com pesos \emph{reais arbitrários}:
o passo de homogeneidade leva $t$ para fora de $\intcc01$.
Se apenas se supõe que uma aplicação preserva \omterm{def:b2:affine:barycenter}{baricentros} com
pesos não negativos --- equivalentemente, pontos médios e segmentos
--- a linearidade da aplicação vetorial deixa de ser gratuita:
obtém-se apenas $\Q$-linearidade, e é preciso uma hipótese de \omterm{def:b2:metric:continuity}{continuidade}
para concluir, exatamente como no
\cref{exo:b2:affine:5}. Distinguir ``preserva todos os
\omterm{def:b2:affine:barycenter}{baricentros}'' de ``preserva combinações \omterm{def:b2:affine:convex}{convexas}'' é uma
sutileza pequena mas real do vocabulário \omterm{def:b2:affine:subspace}{afim}.
\end{remark}

\begin{example}[Geometria baricêntrica clássica]\label{ex:b2:affine:median}
O centroide de um triângulo $ABC$ é o \omterm{def:b2:affine:barycenter}{baricentro} $G =
\operatorname{bar}(A,1; B,1; C,1)$. A associatividade com o
ponto médio $A' = \operatorname{bar}(B, 1; C, 1)$ mostra que
\[
G = \operatorname{bar}(A, 1;\ A', 2) :
\]
$G$ está na mediana $AA'$ a dois terços dela --- e o mesmo
para as outras duas medianas: as três medianas são concorrentes, em
uma linha de cálculo baricêntrico.
\end{example}

\begin{example}[As bimedianas de um quadrilátero]\label{ex:b2:affine:bimedians}
Seja $ABCD$ um quadrilátero qualquer (plano ou não!) e considere
suas \emph{bimedianas}: os segmentos que unem os pontos médios de
lados opostos, $M_{AB}M_{CD}$ e $M_{BC}M_{DA}$. Introduza
o \omterm{def:b2:affine:barycenter}{baricentro} $G$ de $(A,1; B,1; C,1; D,1)$ e agrupe os
pesos de duas maneiras:
\[
G = \operatorname{bar}\bigl(M_{AB}, 2;\ M_{CD}, 2\bigr)
= \operatorname{bar}\bigl(M_{BC}, 2;\ M_{DA}, 2\bigr) :
\]
$G$ é o ponto médio de \emph{ambas} as bimedianas --- de modo que as duas
bimedianas sempre se bissectam, e o quadrilátero dos
quatro pontos médios é um paralelogramo (suas diagonais são as
bimedianas). Sem análise de casos, sem coordenadas, e o
argumento sobrevive intacto para um quadrilátero reverso em
$\R^3$, em que uma demonstração baseada em figura já seria
delicada: a associatividade não se importa com a dimensão.
\end{example}

\begin{example}[Classificar uma aplicação afim, do começo ao fim]\label{ex:b2:affine:dilation}
Seja $f(x, y) = (2x - 1,\ 3y - 4)$ em $\R^2$. Sua parte linear
é $\varphi = \operatorname{diag}(2, 3)$, cujo \omterm{def:b2:reduction:eigen}{espectro}
$\{2, 3\}$ evita $1$: pelo critério de ponto fixo demonstrado
adiante (\cref{prop:b2:affine:fixedpoint}), $f$ tem exatamente um
ponto fixo, achado resolvendo
\[
x = 2x - 1, \qquad y = 3y - 4
\qquad\Longrightarrow\qquad \Omega = (1, 2).
\]
Recentrando em $\Omega$ (ponha $x = 1 + u$, $y = 2 + v$):
\[
f(1 + u,\ 2 + v) = (1 + 2u,\ 2 + 3v) :
\]
no referencial em $\Omega$, $f$ \emph{é} sua parte linear, uma
dilatação anisotrópica que estica por $2$ na horizontal e por $3$
na vertical a partir do centro $(1, 2)$. A lição geral: uma
\omterm{def:b2:affine:subspace}{aplicação afim} é ``aplicação linear mais dado de posição'', e o
dado de posição colapsa numa única origem bem escolhida sempre que
$1$ não é \omterm{def:b2:reduction:eigen}{autovalor}. Reciprocamente, transladar mal a origem
\emph{cria} os termos constantes: a geometria \omterm{def:b2:affine:subspace}{afim} é
a arte de escolher onde pôr $0$.
\end{example}

\begin{remark}[Método: concorrência e alinhamento por baricentros]
O~\cref{ex:b2:affine:median} é um caso de uma
receita geral. Para provar que três cevianas de um triângulo são
concorrentes, exiba um \emph{único} sistema ponderado $(A,
\alpha; B, \beta; C, \gamma)$ e use a associatividade de três
maneiras: agrupar $(B, C)$ mostra que o \omterm{def:b2:affine:barycenter}{baricentro} está na
ceviana que sai de $A$, agrupar $(C, A)$ que está na ceviana que sai de $B$,
agrupar $(A, B)$ que está na terceira. Para as medianas, o sistema
$(A, 1; B, 1; C, 1)$ faz todo o trabalho; para cevianas que cortam
os lados em razões prescritas, os pesos são lidos nas
razões. Para provar que três pontos são \emph{alinhados}, escreva um como
\omterm{def:b2:affine:barycenter}{baricentro} dos outros dois (\cref{exo:b2:affine:2}), ou use
o critério do \omterm{def:b2:linalg:det}{determinante} do~\cref{exo:b2:affine:11}. As duas
receitas substituem a engenhosidade geométrica pela contabilidade de pesos ---
é precisamente para isso que serve o cálculo baricêntrico.
\end{remark}

\section{Convexidade, afinamente}

\begin{definition}\label{def:b2:affine:convex}
Uma parte $C$ de um \omterm{def:b2:affine:def}{espaço afim} é \emph{convexa}\index{conjunto convexo} quando contém
todo \omterm{def:b2:affine:barycenter}{baricentro} com pesos \emph{não negativos} de seus pontos ---
equivalentemente, todo segmento $\intcc{A}{B} = \{\operatorname{bar}(A,
1-t; B, t) : t \in \intcc{0}{1}\}$ entre seus pontos. A
\emph{envoltória convexa}\index{envoltória convexa} $\operatorname{conv}(S)$ é o conjunto de todos os
\omterm{def:b2:affine:barycenter}{baricentros} com pesos não negativos de pontos de $S$ --- o menor
convexo que contém $S$.
\end{definition}

\begin{example}[Os epígrafos são conjuntos convexos]\label{ex:b2:affine:epigraph}
A região $C = \{(x, y) : y \geq x^2\}$ acima da parábola
é \omterm{def:b2:affine:convex}{convexa}: para $(x_1, y_1), (x_2, y_2) \in C$ e $t \in
\intcc01$, a desigualdade de convexidade da função quadrado
dá
\[
\bigl((1-t)x_1 + tx_2\bigr)^2 \leq (1-t)x_1^2 + tx_2^2
\leq (1-t)y_1 + ty_2 ,
\]
de modo que o \omterm{def:b2:affine:barycenter}{baricentro} permanece acima da parábola. O cálculo
é geral: $\{y \geq f(x)\}$ é \omterm{def:b2:affine:convex}{convexo} exatamente quando $f$ é uma
função \omterm{def:b2:affine:convex}{convexa} --- os \emph{conjuntos} \omterm{def:b2:affine:convex}{convexos} e as \emph{funções}
\omterm{def:b2:affine:convex}{convexas} (\cref{ch:b2:realfun}) são duas faces de uma só
noção, sendo os epígrafos o dicionário. Essa é a razão geométrica
pela qual existem retas de suporte para funções \omterm{def:b2:affine:convex}{convexas}, fato
que demonstrará a desigualdade de Jensen no
\cref{ch:b2:randomvar}.
\end{example}

\begin{example}[Geradores redundantes de uma envoltória convexa]\label{ex:b2:affine:squarehull}
Seja $S = \{(0,0), (2,0), (2,2), (0,2), (1,1)\}$. O quinto
ponto é o \omterm{def:b2:affine:barycenter}{baricentro}
\[
(1,1) = \operatorname{bar}\bigl((0,0), \tfrac12;\ (2,2),
\tfrac12\bigr),
\]
de modo que ele já está na envoltória dos outros quatro:
$\operatorname{conv}(S)$ é o quadrado com os quatro cantos
como vértices. Em geral, um ponto de $S$ que é um
\omterm{def:b2:affine:barycenter}{baricentro} de pesos não negativos dos \emph{outros} pontos de
$S$ pode ser apagado sem mudar a envoltória; os pontos que
nunca podem ser apagados (aqui os quatro cantos) são os
\emph{pontos extremos} da envoltória. Determiná-los é um puro
cálculo de \omterm{def:b2:affine:barycenter}{baricentros}: $(2,0)$, digamos, não pode ser escrito como
$\operatorname{bar}$ dos pontos restantes com pesos não
negativos, porque a primeira coordenada forçaria todo o peso
sobre pontos com $x = 2$, e a segunda coordenada então
falha. As questões de convexidade se reduzem, vez após vez, a
resolver pequenos sistemas ponderados.
\end{example}

\begin{theorem}[Carathéodory]\label{thm:b2:affine:caratheodory}
Num \omterm{def:b2:affine:def}{espaço afim} de dimensão $n$, todo ponto de
$\operatorname{conv}(S)$ é \omterm{def:b2:affine:barycenter}{baricentro} de no máximo $n + 1$ pontos
de $S$.\index{teorema de Carathéodory}
\end{theorem}

\begin{proof}
Seja $G = \operatorname{bar}(A_0, \lambda_0; \dots; A_k, \lambda_k)$
com $\lambda_i > 0$, $\sum\lambda_i = 1$ e $k + 1 > n + 1$
pontos. Os $k$ vetores $\vect{A_0A_i}$ ($i \geq 1$) são ligados
($k > n$): $\sum_{i\geq1}\mu_i \vect{A_0A_i} = 0$ não trivialmente;
pondo $\mu_0 = -\sum_{i\geq1}\mu_i$, obtemos pesos $(\mu_i)$
com $\sum\mu_i = 0$, $\sum \mu_i\,\vect{OA_i} = 0$ (qualquer $O$), não
todos nulos. Então, para todo real $t$, os pesos $\lambda_i - t\mu_i$
ainda somam $1$ e, como $\sum_i\mu_i\vect{OA_i} = 0$,
\[
\sum_i(\lambda_i - t\mu_i)\,\vect{OA_i}
= \sum_i\lambda_i\,\vect{OA_i} :
\]
eles produzem o \emph{mesmo} ponto $G$. Deslize agora $t$ a partir de
$0$: algum $\mu_i$ é positivo (eles somam zero e não são
todos nulos), de modo que
\[
t^* = \min\Bigl\{\frac{\lambda_i}{\mu_i} : \mu_i > 0\Bigr\}
\]
está bem definido e é positivo. Em $t = t^*$: para os índices com
$\mu_i > 0$, $\lambda_i - t^*\mu_i \geq 0$ pela minimalidade,
com igualdade num índice minimizante; para os índices com $\mu_i
\leq 0$, $\lambda_i - t^*\mu_i \geq \lambda_i > 0$. Todos os
pesos permanecem não negativos e ao menos um morreu: $G$ é
reescrito como \omterm{def:b2:affine:barycenter}{baricentro} de menos pontos. Itere enquanto restarem mais
de $n + 1$ pontos.
\end{proof}

\begin{example}
No plano ($n = 2$): todo ponto da \omterm{def:b2:affine:convex}{envoltória convexa} de um conjunto
finito está num triângulo com vértices no conjunto --- o conteúdo
geométrico de Carathéodory, usado tanto em otimização quanto em probabilidade
(misturas).
\end{example}

\begin{example}[Rodando o algoritmo de Carathéodory]\label{ex:b2:affine:caratheodoryrun}
Escreva o centro do quadrado do
\cref{ex:b2:affine:squarehull} com seus quatro cantos $A_1 =
(0,0)$, $A_2 = (2,0)$, $A_3 = (2,2)$, $A_4 = (0,2)$:
\[
(1,1) = \operatorname{bar}\bigl(A_1, \tfrac14;\ A_2,
\tfrac14;\ A_3, \tfrac14;\ A_4, \tfrac14\bigr),
\]
quatro pontos em dimensão $2$ --- um a mais do que o necessário. A
receita da demonstração pede pesos $(\mu_i)$ com $\sum\mu_i = 0$ e
$\sum\mu_i\vect{OA_i} = 0$: aqui $\mu = (1, -1, 1, -1)$ funciona
(as duas diagonais compartilham o ponto médio). Deslizar $\lambda_i
\mapsto \lambda_i - t\mu_i$ mantém o \omterm{def:b2:affine:barycenter}{baricentro} fixo para
todo $t$; o valor admissível extremo $t = \frac14$ torna
os pesos $(0, \tfrac12, 0, \tfrac12)$, matando $A_1$ e
$A_3$ simultaneamente:
\[
(1,1) = \operatorname{bar}\bigl(A_2, \tfrac12;\ A_4,
\tfrac12\bigr),
\]
uma representação por dois pontos --- melhor ainda que os três
garantidos pelo teorema, porque o centro por acaso
está num segmento entre geradores. O algoritmo é
inteiramente mecânico: ache uma dependência, deslize até um peso
morrer, repita.
\end{example}

\section{Ferramentas de classificação afim}

\begin{proposition}[Pontos fixos de aplicações afins]\label{prop:b2:affine:fixedpoint}
Seja $f$ um endomorfismo \omterm{def:b2:affine:subspace}{afim} de um \omterm{def:b2:affine:def}{espaço afim} de dimensão
finita, com parte linear $\varphi$. Se $1 \notin
\operatorname{Sp}(\varphi)$, então $f$ tem exatamente um ponto fixo
$\Omega$ e, na vetorialização em $\Omega$, $f$ \emph{é}
sua parte linear. (As translações, com $\varphi = \mathrm{id}$ e sem
ponto fixo, são a obstrução básica.)
\end{proposition}

\begin{proof}
Fixe $O$ e escreva $f(O + x) = f(O) + \varphi(x)$. O ponto
$O + x$ é fixo se e somente se $O + x = f(O) + \varphi(x)$, isto é,
\[
(\mathrm{id} - \varphi)(x) = \vect{O f(O)} .
\]
Em dimensão finita, $\mathrm{id} - \varphi$ é invertível
se e somente se $0$ não é \omterm{def:b2:reduction:eigen}{autovalor} de $\mathrm{id} - \varphi$, se e somente se
$1 \notin \operatorname{Sp}\varphi$ --- e, nesse caso, a
equação exibida tem exatamente uma solução $x^*$, dando o
ponto fixo único $\Omega = O + x^*$. Recentramento: para qualquer
vetor $u$,
\[
f(\Omega + u) = f(\Omega) + \varphi(u) = \Omega + \varphi(u),
\]
de modo que, no referencial de origem $\Omega$, a aplicação se lê $u \mapsto
\varphi(u)$: puramente linear. Quando $1 \in
\operatorname{Sp}\varphi$, ou não existe ponto fixo (a
equação exibida pode ser insolúvel, como para uma translação),
ou existe todo um subespaço \omterm{def:b2:affine:subspace}{afim} deles (some qualquer \omterm{def:b2:reduction:eigen}{autovetor}
de \omterm{def:b2:reduction:eigen}{autovalor} $1$ a uma solução): a unicidade é exatamente a
condição espectral.
\end{proof}

\begin{example}[Isometrias do plano, completadas]\label{ex:b2:affine:isometries}
Uma isometria \omterm{def:b2:affine:subspace}{afim} do plano euclidiano tem parte linear em
$O(2)$: uma rotação $R_\theta$ ou uma reflexão (volume do primeiro ano de graduação). Se
$\theta \neq 0$: $1 \notin \operatorname{Sp} R_\theta$, de modo que a aplicação
é uma \emph{rotação em torno de um centro único}
(\cref{prop:b2:affine:fixedpoint}). Se a parte linear é uma
reflexão: ou uma reflexão num eixo (há pontos fixos) ou
uma \emph{reflexão deslizante} (reflexão composta com uma translação
ao longo do eixo, sem ponto fixo). Com as translações, essa é a
classificação \omterm{def:b2:metric:complete}{completa} das isometrias do plano.
\end{example}

\begin{remark}[As isometrias do plano, num relance]
Reunindo os casos: identidade; translações ($\vec f =
\mathrm{id}$, sem ponto fixo salvo a trivial); rotações
(parte linear $R_\theta$, $\theta \neq 0$: um centro);
reflexões (parte linear uma reflexão, uma reta de pontos
fixos); reflexões deslizantes (mesma parte linear, sem ponto
fixo). Quatro famílias mais a identidade, cada uma reconhecida por
apenas dois dados: a parte linear e o conjunto de pontos fixos ---
o padrão da~\cref{prop:b2:affine:fixedpoint} tornado
exaustivo.
\end{remark}

\begin{example}[Uma reflexão deslizante, pega em flagrante]\label{ex:b2:affine:glide}
Seja $f(x, y) = (y + 1,\ x + 1)$. A parte linear $(x, y)
\mapsto (y, x)$ é a reflexão na diagonal $y = x$, logo
$1 \in \operatorname{Sp}\vec f$ e a
\cref{prop:b2:affine:fixedpoint} se cala. Pontos fixos
exigiriam $x = y + 1$ e $y = x + 1$ simultaneamente:
impossível --- não existem, de modo que $f$ não é uma reflexão. Elevar ao quadrado
resolve a classificação:
\[
f\bigl(f(x, y)\bigr) = f(y + 1,\ x + 1) = (x + 2,\ y + 2),
\]
a translação por $(2, 2)$: $f$ é a \emph{reflexão
deslizante} de eixo a reta $y = x$ (deslocada
adequadamente: o ponto médio de $M$ e $f(M)$ está sempre em
$y = x + {}$constante, aqui $y = x$, como se verifica em $M = (0,
0) \mapsto (1,1)$) e vetor de deslizamento $(1, 1)$, metade de $f
\circ f$. Compare com o~\cref{exo:b2:affine:6}, em que a mesma
parte linear mas uma constante diferente produziu uma reflexão
honesta: com o \omterm{def:b2:reduction:eigen}{autovalor} $1$ presente, o termo constante
decide tudo.
\end{example}

\begin{example}[Recorrências afins são dinâmica afim]\label{ex:b2:affine:recursion}
A recorrência clássica $u_{n+1} = au_n + b$ ($a \neq 1$)
itera a \omterm{def:b2:affine:subspace}{aplicação afim} $f(x) = ax + b$ da reta, cuja
parte linear $a$ evita o \omterm{def:b2:reduction:eigen}{autovalor} $1$: existe um único
ponto fixo $\omega = \frac{b}{1-a}$ e, recentrando aí
(o caso unidimensional da proposição acima), $f$ vira multiplicação por $a$:
\[
u_{n+1} - \omega = a\,(u_n - \omega)
\qquad\Longrightarrow\qquad
u_n = \omega + a^n(u_0 - \omega) .
\]
Para $u_{n+1} = \frac{u_n}2 + 3$: $\omega = 6$ e $u_n = 6 +
(u_0 - 6)2^{-n} \to 6$. A receita ensinada para tais
recorrências no~\cref{ch:b2:series} --- ``subtraia o ponto
fixo'' --- é exatamente a vetorialização de uma \omterm{def:b2:affine:subspace}{aplicação afim}
em seu ponto fixo; a convergência para $\abs a < 1$ é o fenômeno de
contração que o~\cref{ch:b2:metric} transformou no
teorema do ponto fixo de Banach. Uma ideia, três capítulos.
\end{example}

\begin{example}[Achar o centro de uma rotação]\label{ex:b2:affine:rotationcenter}
Seja $f(x, y) = (-y + 2,\ x)$. A parte linear é
$\varphi(x, y) = (-y, x)$: a rotação de ângulo $\frac\pi2$,
cujo \omterm{def:b2:reduction:eigen}{espectro} $\{\iu, -\iu\}$ evita $1$. Pela
\cref{prop:b2:affine:fixedpoint} existe exatamente um ponto
fixo: $x = -y + 2$ e $y = x$ dão $x = 1$, $y = 1$, logo
$\Omega = (1, 1)$, e $f$ \emph{é} a rotação de centro
$(1, 1)$ e ângulo $\frac\pi2$. A lição geral: quando $1
\notin \operatorname{Sp}\vec f$, classificar $f$ custa um
sistema linear --- a geometria está inteiramente na parte linear,
a aritmética inteiramente em localizar o centro.
\end{example}

\begin{remark}[Onde a linguagem afim é usada em seguida]
Os \omterm{def:b2:affine:barycenter}{baricentros} e as \omterm{def:b2:affine:subspace}{aplicações afins} são a gramática dos capítulos
de geometria adiante: retas e planos tangentes são objetos \omterm{def:b2:affine:subspace}{afins}
(\cref{ch:b2:curves,ch:b2:surfaces}), uma mudança \omterm{def:b2:affine:subspace}{afim} de
variáveis multiplica áreas e volumes por
$\abs{\det \vec f}$ (\cref{ch:b2:multint}), e a esperança
é um \omterm{def:b2:affine:barycenter}{baricentro} com pesos dados por uma lei de probabilidade, e é por isso
que a convexidade governa a desigualdade de Jensen
(\cref{ch:b2:randomvar}). No volume do terceiro ano de graduação, o mesmo
vocabulário de convexidade sustenta o estudo das \omterm{def:b2:nvs:norm}{normas} $L^p$ e das
desigualdades integrais.
\end{remark}

\begin{remark}[Perspectivas dentro deste volume]
Dois fios saem deste capítulo. O fio \emph{\omterm{def:b2:affine:subspace}{afim}}: as retas
tangentes (\cref{ch:b2:curves}) e os planos tangentes
(\cref{ch:b2:surfaces}) são subespaços \omterm{def:b2:affine:subspace}{afins} ligados a
objetos não lineares, e a classificação das quádricas no
capítulo de superfícies roda sobre a equação do centro
$A\Omega = -b$ deste capítulo. O fio \emph{\omterm{def:b2:affine:convex}{convexo}} é mais longo: a convexidade
dos semiplanos e dos discos move a teoria de Helly do
problema de fim de semana; a convexidade das funções dá a desigualdade
de Jensen (\cref{ch:b2:randomvar}); e o teorema final
do livro --- o critério de extinção para processos de ramificação
(\cref{ch:b2:genfun}) --- é decidido pela posição de uma
curva \omterm{def:b2:affine:convex}{convexa} em relação à diagonal, um retrato que pertence
a este capítulo tanto quanto à probabilidade. Os \omterm{def:b2:affine:barycenter}{baricentros}
voltam lá também: uma esperança é um \omterm{def:b2:affine:barycenter}{baricentro} com
pesos de probabilidade.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:affine:1}
Em $\R^3$, os seguintes são subespaços \omterm{def:b2:affine:subspace}{afins}? Dê direções e
dimensões. $\{x + y + z = 1\}$; $\;\{x + y + z = 1,\ x - z =
3\}$; $\;\{x^2 + y^2 = 1\}$; o conjunto solução de $MX = B$ para um
sistema compatível dado.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:affine:2}
Prove que três pontos distintos $A, B, C$ de um \omterm{def:b2:affine:def}{espaço afim} são
alinhados se e somente se $C$ é \omterm{def:b2:affine:barycenter}{baricentro} de $A$ e $B$, se e somente se os vetores
$\vect{AB}, \vect{AC}$ são ligados. Deduza a contabilidade de pesos ao estilo
Menelau: se $C = \operatorname{bar}(A, 1 - t; B, t)$, localize
$C$ para $t = \frac12$, $t = 2$, $t = -1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:affine:3}
Seja $f$ a \omterm{def:b2:affine:subspace}{aplicação afim} de $\R^2$ dada por $f(X) = MX + C$ com
$M = \frac12\begin{pmatrix} 1 & 1\\ 1 & 1\end{pmatrix}$ e $C =
(1, 0)^{\mathsf T}$. Determine a imagem de $f$, seus pontos fixos
(se houver) e $f \circ f$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:4}
(Associatividade em ação) Num triângulo $ABC$, sejam $I, J, K$ os pontos que dividem
$BC$, $CA$, $AB$ nas razões $\vect{BI} = \frac13\vect{BC}$,
$\vect{CJ} = \frac13\vect{CA}$, $\vect{AK} = \frac13\vect{AB}$.
Exprima $I, J, K$ como \omterm{def:b2:affine:barycenter}{baricentros} e calcule o \omterm{def:b2:affine:barycenter}{baricentro} de
$(I,1;J,1;K,1)$: o que você encontra, e por que era previsível?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:5}
Prove que uma aplicação $f \colon \R^n \to \R^n$ que preserva \emph{pontos médios}
($f\bigl(\frac{A+B}{2}\bigr) = \frac{f(A) + f(B)}{2}$) e é
\emph{\omterm{def:b2:metric:continuity}{contínua}} é \omterm{def:b2:affine:subspace}{afim}. \emph{(Mostre que a aplicação vetorial $u \mapsto
f(O + u) - f(O)$ é aditiva via pontos médios, depois
$\Q$-homogênea, depois $\R$-homogênea por \omterm{def:b2:metric:continuity}{continuidade} ---
a mesma estratégia de densidade usada para a equação funcional de Cauchy no
volume do primeiro ano de graduação; refaça aqui os passos necessários.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:6}
Classifique a \omterm{def:b2:affine:subspace}{aplicação afim} $f(x, y) = (y + 1,\; x - 1)$ do
plano euclidiano: parte linear, pontos fixos, natureza geométrica
(reflexão? deslizante?). Calcule $f \circ f$ e conclua.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:affine:7}
(Radon) Sejam $A_1, \dots, A_{n+2}$ pontos de um \omterm{def:b2:affine:def}{espaço afim} de
dimensão $n$. Prove que eles podem ser separados em dois grupos
disjuntos cujas \omterm{def:b2:affine:convex}{envoltórias convexas} se intersectam. \emph{(Como na
demonstração de Carathéodory, ache pesos $\mu_i$, não todos nulos, com $\sum\mu_i = 0$ e
$\sum\mu_i\vect{OA_i} = 0$; separe os pesos positivos dos negativos
e normalize os dois lados.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:affine:8}
Seja $f$ um endomorfismo \omterm{def:b2:affine:subspace}{afim} de $\R^n$ com $f \circ f = f$.
Prove que $f$ é a projeção \omterm{def:b2:affine:subspace}{afim} sobre o subespaço \omterm{def:b2:affine:subspace}{afim}
$\operatorname{Fix}(f) = \operatorname{im} f$ na direção
$\ker\vec f$, e que, reciprocamente, todas essas projeções são
idempotentes. \emph{(Mostre primeiro que $\operatorname{im} f$ consiste
em pontos fixos.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:affine:9}
Sejam $G = \operatorname{bar}(A, 1;\ B, 2;\ C, 3)$ num triângulo
$ABC$. Usando a associatividade, mostre que a reta $AG$ encontra $BC$
em $M = \operatorname{bar}(B, 2;\ C, 3)$ e localize $G$ no
segmento $\intcc AM$; localize do mesmo modo a interseção de $BG$
com $CA$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:10}
Para $\lambda \neq 0$, a \emph{homotetia} $h_{\Omega,
\lambda}$ é a \omterm{def:b2:affine:subspace}{aplicação afim} que fixa $\Omega$ com parte linear
$\lambda\,\mathrm{id}$. Prove que a composição
$h_{\Omega', \mu} \circ h_{\Omega, \lambda}$ é uma homotetia de
razão $\lambda\mu$ quando $\lambda\mu \neq 1$, e uma translação
quando $\lambda\mu = 1$; no caso $\lambda = \mu = -1$ (duas
simetrias centrais), calcule o vetor de translação.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:11}
(Menelau) Num triângulo $ABC$, sejam $A' \in (BC)$, $B' \in
(CA)$, $C' \in (AB)$, todos distintos dos vértices, e
defina $\alpha, \beta, \gamma$ por $\vect{A'B} =
\alpha\,\vect{A'C}$, $\vect{B'C} = \beta\,\vect{B'A}$,
$\vect{C'A} = \gamma\,\vect{C'B}$. Prove que $A', B', C'$ estão
alinhados se e somente se $\alpha\beta\gamma = 1$. \emph{(Escreva
cada ponto como \omterm{def:b2:affine:barycenter}{baricentro} de dois vértices; mostre que três
pontos estão alinhados se e somente se suas linhas de coordenadas baricêntricas
em relação a $(A, B, C)$ formam uma matriz $3 \times 3$ singular.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:affine:12}
Prove que a \omterm{def:b2:affine:convex}{envoltória convexa} de uma parte \omterm{def:b2:metric:compact}{compacta} $K$ de $\R^n$
é \omterm{def:b2:metric:compact}{compacta}. \emph{(Pelo~\cref{thm:b2:affine:caratheodory},
$\operatorname{conv}(K)$ é a imagem de um \omterm{def:b2:metric:compact}{compacto} por uma aplicação
\omterm{def:b2:metric:continuity}{contínua}.)} Mostre, por um exemplo em $\R^2$, que a \omterm{def:b2:affine:convex}{envoltória
convexa} de um conjunto \emph{fechado} não precisa ser fechada.
\end{exercise}

\begin{problem}[{Problema de fim de semana --- o teorema de Helly e dois de
seus dividendos}]\label{pb:b2:affine:1}
O lema de Radon (\cref{exo:b2:affine:7}) diz que $n + 2$
pontos de um \omterm{def:b2:affine:def}{espaço afim} de dimensão $n$ sempre se separam em
dois grupos com \omterm{def:b2:affine:convex}{envoltórias convexas} que se intersectam. Este problema transforma
esse único fato de álgebra linear numa cadeia de teoremas de
geometria combinatória: o teorema de interseção de Helly, o
teorema do ponto central (uma mediana bidimensional) e o teorema
de recobrimento de Jung. Em todo o problema, o plano é $\R^2$ com sua
estrutura euclidiana usual, e $\det$ é o \omterm{def:b2:linalg:det}{determinante} na
base canônica.\index{teorema de Helly}\index{lema de
Radon}\index{teorema de Jung}\index{ponto central}

\medskip
\noindent\textbf{Parte I --- Coordenadas baricêntricas.} Pontos
$A_0, \dots, A_k$ são \emph{afinmente independentes} quando os
vetores $\vect{A_0A_1}, \dots, \vect{A_0A_k}$ são linearmente
independentes.
\begin{enumerate}
  \item Mostre que a independência \omterm{def:b2:affine:subspace}{afim} não depende da
        escolha do ponto-base $A_0$ e que ela equivale a: sempre
        que duas famílias de pesos, cada uma somando $1$, definem
        o mesmo \omterm{def:b2:affine:barycenter}{baricentro} de $(A_0,
        \dots, A_k)$, os pesos coincidem.
  \item Sejam $(A, B, C)$ afinmente independentes no plano.
        Mostre que todo ponto $M$ admite uma única tripla
        $(\alpha, \beta, \gamma)$ com $\alpha + \beta +
        \gamma = 1$ e $M = \operatorname{bar}(A, \alpha; B,
        \beta; C, \gamma)$ --- suas \emph{coordenadas
        baricêntricas}.
  \item Demonstre as fórmulas com \omterm{def:b2:linalg:det}{determinantes}
        \[
        \alpha = \frac{\det(\vect{MB},
        \vect{MC})}{\det(\vect{AB}, \vect{AC})},
        \qquad
        \beta = \frac{\det(\vect{MC},
        \vect{MA})}{\det(\vect{AB}, \vect{AC})},
        \qquad
        \gamma = \frac{\det(\vect{MA},
        \vect{MB})}{\det(\vect{AB}, \vect{AC})} :
        \]
        as coordenadas baricêntricas são razões de áreas com sinal.
  \item As retas $BC$, $CA$, $AB$ são as retas de coordenada
        $\{\alpha = 0\}$, $\{\beta = 0\}$, $\{\gamma = 0\}$.
        Mostre que $M$ está no triângulo fechado
        $\operatorname{conv}\{A, B, C\}$ se e somente se $\alpha, \beta,
        \gamma \geq 0$, e que as três retas cortam o plano
        em exatamente sete regiões, classificadas pelos sinais de
        $(\alpha, \beta, \gamma)$ (sendo impossível o padrão de sinais $(-, -,
        -)$).
  \item Seja $u \colon \R^2 \to \R$ uma \omterm{def:b2:affine:subspace}{aplicação afim} (uma
        \emph{forma \omterm{def:b2:affine:subspace}{afim}}). Mostre que $u(M) = \alpha\,u(A) +
        \beta\,u(B) + \gamma\,u(C)$, que os conjuntos de nível de uma
        forma \omterm{def:b2:affine:subspace}{afim} não constante são retas, que toda reta
        surge assim, e que os semiplanos fechados $\{u
        \geq c\}$ são \omterm{def:b2:affine:convex}{convexos}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Partições de Radon, refinadas.} Uma
família de $n + 2$ pontos de $\R^n$ está em \emph{posição
geral} quando quaisquer $n + 1$ deles são afinmente independentes.
Uma \emph{dependência \omterm{def:b2:affine:subspace}{afim}} de $(A_1, \dots, A_{n+2})$ é uma
família $(\mu_i)$ com $\sum_i \mu_i = 0$ e $\sum_i
\mu_i\,\vect{OA_i} = 0$ para uma (logo toda) origem $O$.
\begin{enumerate}[resume]
  \item Calcule uma dependência \omterm{def:b2:affine:subspace}{afim} não nula dos quatro
        pontos $A_1 = (0,0)$, $A_2 = (3,0)$, $A_3 = (0,3)$,
        $A_4 = (1,1)$; dê a partição de Radon e o ponto de
        Radon.
  \item Mostre que, para pontos em posição geral, o espaço
        vetorial das dependências \omterm{def:b2:affine:subspace}{afins} tem dimensão exatamente
        $1$, e que uma dependência não nula \emph{não} tem
        coeficiente nulo.
  \item Deduza que a partição de Radon de $n + 2$ pontos em
        posição geral é única (a menos de trocar os dois
        blocos), sendo cada bloco o conjunto dos índices em que
        $\mu_i$ tem um sinal fixo.
  \item Para quatro pontos do plano em posição geral,
        mostre a dicotomia: ou a partição é do tipo
        $(1, 3)$ --- um ponto interior ao triângulo dos
        outros três --- ou do tipo $(2, 2)$: os quatro pontos
        estão em posição \omterm{def:b2:affine:convex}{convexa} e os segmentos que unem os
        dois pares (as diagonais) se intersectam, no ponto de
        Radon.
  \item Execute a questão 6 para o quadrado unitário $(0,0)$,
        $(1,0)$, $(1,1)$, $(0,1)$: dependência, partição,
        ponto de Radon.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O teorema de Helly no plano.}
\begin{enumerate}[resume]
  \item Sejam $C_1, C_2, C_3, C_4$ partes \omterm{def:b2:affine:convex}{convexas} de $\R^2$,
        quaisquer três das quais têm ponto comum. Escolha $x_i \in
        \bigcap_{j \neq i} C_j$ e aplique o lema de Radon a
        $x_1, \dots, x_4$: mostre que o ponto de Radon pertence
        aos quatro conjuntos. \emph{(Para cada $k$, o bloco que não
        contém $x_k$ é formado por pontos de $C_k$.)}
  \item (Helly) Sejam $C_1, \dots, C_m$ ($m \geq 3$) partes \omterm{def:b2:affine:convex}{convexas}
        de $\R^2$, quaisquer três das quais se intersectam.
        Prove que $\bigcap_{i=1}^m C_i \neq \emptyset$, por
        indução sobre $m$: substitua $C_{m-1}$ e $C_m$ por
        $C_{m-1} \cap C_m$ e verifique a hipótese para a
        nova família usando a questão 11.
  \item Três contraexemplos, um por hipótese: (a) os
        três lados fechados de um triângulo se intersectam dois a dois
        mas não têm ponto comum ($3$ não pode baixar para
        $2$); (b) os quatro conjuntos $S_i = \{x_1, \dots, x_4\}
        \setminus \{x_i\}$, para quatro pontos em posição geral,
        satisfazem a hipótese das interseções triplas
        mas não a conclusão (a convexidade importa); (c) os
        semiplanos fechados $H_k = \intco{k}{+\infty} \times
        \R$, $k \in \N$, se intersectam dois a dois e três a três
        mas $\bigcap_k H_k = \emptyset$ (famílias infinitas
        precisam de \omterm{def:b2:metric:compact}{compacidade}).
  \item (Helly \omterm{def:b2:metric:compact}{compacto}) Seja $(K_i)_{i \in I}$ uma família arbitrária
        de partes \omterm{def:b2:affine:convex}{convexas} \omterm{def:b2:metric:compact}{compactas} de $\R^2$, quaisquer três
        das quais se intersectam. Usando a questão 12 e a
        propriedade de Borel--Lebesgue
        (\cref{thm:b2:metric:borellebesgue}), mostre que
        $\bigcap_{i \in I} K_i \neq \emptyset$.
  \item (Primeiro dividendo) Sejam $S$ um conjunto finito de pontos do
        plano e $r > 0$. Mostre: se quaisquer três pontos de
        $S$ estão em algum disco fechado de raio $r$, então $S$
        está num único disco fechado de raio $r$. \emph{(Aplique
        Helly aos discos $\overline D(p, r)$, $p \in S$.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O teorema do ponto central.} Um
\emph{ponto central} de um conjunto finito $S$ de $n$ pontos do
plano é um ponto $c$ (não necessariamente em $S$) tal que todo
semiplano fechado que contém $c$ contém ao menos $n/3$
pontos de $S$.
\begin{enumerate}[resume]
  \item (Dimensão $1$) Para reais $x_1 \leq \dots \leq x_n$,
        mostre que a mediana $c = x_{\lceil n/2 \rceil}$
        satisfaz: toda semirreta fechada que contém $c$
        contém ao menos $n/2$ dos $x_i$.
  \item (Lema de contagem) Se $A, B, C$ são partes de $S$ com
        $\abs A, \abs B, \abs C > \tfrac{2n}3$, mostre que $A \cap
        B \cap C \neq \emptyset$.
  \item Sejam $m = \floor{2n/3} + 1$ e seja $\mathcal F$
        a família (finita) das \omterm{def:b2:affine:convex}{envoltórias convexas}
        $\operatorname{conv}(T)$, $T \subseteq S$, $\abs T =
        m$. Mostre que quaisquer três membros de $\mathcal F$ têm
        ponto comum e deduza de Helly um ponto $c$
        comum a todos eles.
  \item Prove que esse $c$ é um ponto central de $S$: o
        \emph{teorema do ponto central}. \emph{(Se um
        semiplano fechado por $c$ contivesse menos de $n/3$
        pontos, seu complementar \omterm{def:b2:metric:topology}{aberto} conteria um conjunto $T$
        de $m$ pontos, e $\operatorname{conv}(T)$ evitaria
        $c$.)}
  \item Otimalidade: seja $n = 3k$ e coloque $k$ pontos em cada
        um de três discos de raio pequeno $\varepsilon$ centrados
        nos vértices de um triângulo grande. Mostre que, para
        \emph{todo} ponto $c$ do plano, algum semiplano
        fechado contendo $c$ contém no máximo $n/3$
        pontos de $S$, de modo que a constante $1/3$ não pode ser
        melhorada. \emph{(Entre as três direções de $c$
        aos centros dos discos, duas formam ângulo no máximo
        $2\pi/3$.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- O teorema de Jung e síntese.}
\begin{enumerate}[resume]
  \item (Lema do triângulo) Sejam $P, Q, R$ três pontos com
        distâncias mútuas $\leq 1$. Mostre que eles estão num
        disco fechado de raio $1/\sqrt3$. \emph{(Se algum
        ângulo é $\geq \pi/2$, tome o disco de diâmetro o maior
        lado, usando a fórmula da mediana
        $\norm{RM}^2 = \tfrac12\norm{RP}^2 +
        \tfrac12\norm{RQ}^2 - \tfrac14\norm{PQ}^2$; se o
        triângulo é acutângulo, majore o raio da circunferência circunscrita $a/(2\sin
        \widehat A)$ usando seu maior ângulo, que está em
        $\intco{\pi/3}{\pi/2}$.)}
  \item (Jung) Deduza: toda parte \omterm{def:b2:metric:compact}{compacta} do plano de
        diâmetro $\leq 1$ está contida num disco fechado de
        raio $1/\sqrt3$.
  \item Otimalidade: para o triângulo equilátero $A_1A_2A_3$
        de lado $1$ e centroide $G$, demonstre a identidade
        de Leibniz $\sum_i \norm{\vect{OA_i}}^2 =
        3\norm{\vect{OG}}^2 + \sum_i \norm{\vect{GA_i}}^2$
        para todo ponto $O$ e conclua que todo disco
        que contém os três vértices tem raio $\geq
        1/\sqrt3$, com igualdade apenas para o disco circunscrito.
  \item (Helly em $\R^n$) Enuncie e demonstre o teorema de Helly em
        $\R^n$: se um número finito de \omterm{def:b2:affine:convex}{conjuntos convexos} é tal que
        quaisquer $n + 1$ deles se intersectam, então todos eles
        se intersectam. \emph{(O lema de Radon
        \cref{exo:b2:affine:7} trata de $n + 2$ conjuntos; depois
        faça indução como na questão 12.)}
  \item Síntese. Monte a cadeia
        \[
        \text{dependência afim} \Rightarrow \text{Radon}
        \Rightarrow \text{Helly} \Rightarrow
        \text{ponto central e Jung},
        \]
        indicando em uma frase cada: onde entra a álgebra linear,
        onde entram os sinais dos pesos, onde entra a
        convexidade e qual único passo usou a
        dimensão do plano. O que se tornam as constantes $3$ (em
        Helly), $1/3$ (ponto central) e $1/\sqrt3$ (Jung)
        em $\R^n$? (Enuncie sem demonstrar.)
\end{enumerate}
\end{problem}
