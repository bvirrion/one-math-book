\chapter{Integração}\label{ch:b2:integration}

O volume do primeiro ano de graduação construiu a integral num segmento. Este capítulo
a estende a intervalos arbitrários (\omterm{def:b2:integration:improper}{integrais impróprias}, com toda a
caixa de ferramentas de comparação) e depois estuda integrais \emph{que dependem de um
parâmetro} --- \omterm{def:b2:metric:continuity}{continuidade} e derivação sob o sinal de
integral --- movidas pelo teorema da convergência dominada, o único
resultado deste capítulo aceito por confiança. A função $\Gamma$ serve
de exemplo condutor e de portal para metade das funções especiais
da matemática.

\section{Integrais num intervalo arbitrário}

\begin{definition}\label{def:b2:integration:improper}
Seja $f$ \omterm{def:b2:metric:continuity}{contínua} por partes em $\intco{a}{b}$ ($b \in \R$ ou
$+\infty$), com \omterm{def:b2:metric:continuity}{continuidade} por partes. A integral \emph{converge}\index{integral imprópria}
quando $\lim_{x \to b^-} \int_a^x f$ existe; escreve-se então
$\int_a^b f$ para o limite. (Analogamente em $\intoc{a}{b}$, e em
$\intoo{a}{b}$ separando num ponto interior --- a escolha não
importa, por Chasles.) A integral \emph{converge \omterm{def:b2:series:def}{absolutamente}}
quando $\int_a^b \abs f$ converge; a convergência absoluta implica a
convergência, pelo critério de Cauchy:
\[
\Bigl| \int_x^{y} f \Bigr| \leq \int_x^{y} \abs f
\]
e pela \omterm{def:b2:metric:complete}{completude} de $\R$ (a primitiva tem a propriedade de Cauchy).
Em detalhe: sejam $F(x) = \int_a^x f$ e $G(x) = \int_a^x \abs
f$. Se $\int^b\abs f$ converge, $G$ tem limite em $b^-$, de modo que,
para todo $\varepsilon > 0$, existe $c < b$ com $G(y) - G(x)
\leq \varepsilon$ sempre que $c \leq x \leq y < b$; a fórmula
exibida transfere essa propriedade de Cauchy a $F$. Para qualquer sequência $x_n
\to b^-$, os valores $F(x_n)$ formam então uma sequência de Cauchy de
reais, convergente pela \omterm{def:b2:metric:complete}{completude}, e intercalar duas dessas
sequências mostra que o limite é o mesmo para todas: $F$ tem
limite em $b^-$.
\end{definition}

\begin{theorem}[Caixa de ferramentas de comparação positiva]\label{thm:b2:integration:comparison}
Para $f, g \geq 0$ \omterm{def:b2:metric:continuity}{contínua} por partes em $\intco{a}{b}$, com \omterm{def:b2:metric:continuity}{continuidade} por partes:
\begin{enumerate}
  \item $\int_a^b f$ \omterm{def:b2:integration:improper}{converge} se e somente se a primitiva $x \mapsto \int_a^x
        f$ é limitada;
  \item $f \leq g$: a convergência de $\int g$ força a de $\int f$;
        a divergência se transfere no outro sentido;
  \item $f \sim g$ em $b$: as duas integrais têm a mesma natureza;
  \item as escalas de referência: em $+\infty$, $\int^{\infty}
        \frac{\dd t}{t^\alpha}$ \omterm{def:b2:integration:improper}{converge} se e somente se $\alpha > 1$, e
        $\int^\infty \frac{\dd t}{t(\ln t)^\beta}$ se e somente se $\beta > 1$;
        numa extremidade finita $b$, $\int^b \frac{\dd t}{(b -
        t)^\alpha}$ \omterm{def:b2:integration:improper}{converge} se e somente se $\alpha < 1$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) A primitiva $F(x) = \int_a^x f$ é não decrescente ($f \geq
0$). Se ela é limitada, $\ell = \sup_{x < b}F$ é finito e
$F(x) \to \ell$: dado $\varepsilon > 0$, algum $F(x_0) > \ell -
\varepsilon$, e a monotonicidade prende $F(x) \in
\intoc{\ell - \varepsilon}{\ell}$ para $x_0 \leq x < b$. Se ela é
ilimitada, $F \to +\infty$: divergência.

(2) De $f \leq g$: $\int_a^x f \leq \int_a^x g$ para todo $x$;
se $\int^b g$ \omterm{def:b2:integration:improper}{converge}, o membro da direita é limitado, logo o da esquerda
também, e (1) conclui. A contraposição transfere a divergência
no outro sentido.

(3) $f \sim g$ em $b$ fornece $c < b$ com
\[
\tfrac12\,g(t) \;\leq\; f(t) \;\leq\; 2\,g(t)
\qquad (c \leq t < b) :
\]
por (2) aplicado nos dois sentidos em $\intco{c}{b}$, as duas integrais
têm a mesma natureza; a peça inicial $\intcc{a}{c}$ é uma
integral própria e nada muda.

(4) Primitivas explícitas: para $\alpha \neq 1$ e $\beta \neq 1$,
\[
\int_c^x \frac{\dd t}{t^\alpha}
= \frac{x^{1-\alpha} - c^{1-\alpha}}{1 - \alpha},
\qquad
\int_c^x \frac{\dd t}{t(\ln t)^\beta}
= \frac{(\ln x)^{1-\beta} - (\ln c)^{1-\beta}}{1 - \beta},
\]
com logaritmos nos casos excluídos: limitadas quando $x \to
+\infty$ exatamente se $\alpha > 1$, resp.\ $\beta > 1$. Numa
extremidade finita, a substituição $u = b - t$ reduz à
escala $\int_0 u^{-\alpha}\,\dd u$, limitada se e somente se $\alpha < 1$.
Aplique (1) em cada caso.
\end{proof}

\begin{example}[Dois aquecimentos, levados até o fim]\label{ex:b2:integration:warmups}
\emph{(a)} $\displaystyle\int_0^1 \ln t\,\dd t$: o integrando
explode em $0^+$, mas $\abs{\ln t} = o\bigl(t^{-1/2}\bigr)$
aí (os logaritmos perdem para as potências), e $\int_0 t^{-1/2}$
\omterm{def:b2:integration:improper}{converge}: convergência absoluta. O valor, por partes em
$\intcc{\varepsilon}{1}$:
\[
\int_\varepsilon^1 \ln t\,\dd t
= \bigl[t\ln t - t\bigr]_\varepsilon^1
= -1 - \varepsilon\ln\varepsilon + \varepsilon
\xrightarrow[\varepsilon\to0^+]{} -1 .
\]
\emph{(b)} $\displaystyle\int_0^\infty \frac{\ln t}{1 +
t^2}\,\dd t$: problema nas duas pontas, logo separe em $1$. Perto de $0$:
$\abs{\ln t}$, integrável como em (a); perto de $\infty$: $\frac{\ln
t}{1+t^2} = o(t^{-3/2})$: \omterm{def:b2:series:def}{absolutamente} convergente. A
substituição $t = \frac1u$ leva $\intoo{0}{1}$ em
$\intoo{1}{\infty}$ e
\[
\int_0^1 \frac{\ln t}{1+t^2}\,\dd t
= \int_1^{\infty} \frac{-\ln u}{1 + u^{-2}}\cdot
\frac{\dd u}{u^2}
= -\int_1^\infty \frac{\ln u}{1+u^2}\,\dd u :
\]
as duas metades se cancelam, e a integral vale $0$. Lição
final: a simetria sob $t \mapsto \frac1t$ vale uma página de
cálculo --- o mesmo truque já movia o
\cref{exo:b2:integration:3}.
\end{example}

\begin{example}[Um valor, três integrais]\label{ex:b2:integration:onecos}
Estude $I = \displaystyle\int_0^{\infty} \frac{1 - \cos
t}{t^2}\,\dd t$. Em $0$: $1 - \cos t \sim \frac{t^2}2$, de modo que o
integrando se estende \omterm{def:b2:metric:continuity}{continuamente} pelo valor $\frac12$ --- não há
singularidade alguma. Em $\infty$: $0 \leq \frac{1 - \cos t}{t^2}
\leq \frac{2}{t^2}$: convergência absoluta
(\cref{thm:b2:integration:comparison}). Valor: integre por
partes em $\intcc{\varepsilon}{M}$ com $u = 1 - \cos t$, $v' =
t^{-2}$:
\[
\int_\varepsilon^M \frac{1 - \cos t}{t^2}\,\dd t
= \Bigl[-\frac{1 - \cos t}{t}\Bigr]_\varepsilon^M
+ \int_\varepsilon^M \frac{\sin t}{t}\,\dd t .
\]
O colchete se anula nas duas pontas ($\frac{1 -
\cos\varepsilon}{\varepsilon} \sim \frac\varepsilon2$; numerador
limitado em $M$), e a integral tende ao valor de Dirichlet
$\frac\pi2$ (\cref{exo:b2:integration:10}): $I = \frac\pi2$.
Lição final: com $1 - \cos t = 2\sin^2\frac t2$ e $u =
\frac t2$,
\[
I = \int_0^\infty \frac{2\sin^2 u}{(2u)^2}\;2\,\dd u
= \int_0^\infty \Bigl(\frac{\sin u}{u}\Bigr)^{\!2}\dd u :
\]
os três clássicos $\int_0^\infty\frac{\sin t}{t}\dd t$,
$\int_0^\infty\bigl(\frac{\sin t}{t}\bigr)^2\dd t$
(\cref{exo:b2:integration:11}) e $I$ compartilham o valor
$\frac\pi2$, passado adiante por partes e substituição --- e
só o primeiro é semiconvergente: a integração por partes trocou
a convergência absoluta por um integrando mais simples.
\end{example}

\begin{example}[Uma integral semiconvergente]\label{ex:b2:integration:sint}
$\displaystyle\int_1^{\infty} \frac{\sin t}{t}\,\dd t$ \omterm{def:b2:integration:improper}{converge}:
integre por partes,
\[
\int_1^x \frac{\sin t}{t}\dd t
= \Bigl[\frac{-\cos t}{t}\Bigr]_1^x - \int_1^x \frac{\cos
t}{t^2}\dd t ,
\]
em que o colchete tem limite e a última integral converge
\omterm{def:b2:series:def}{absolutamente} ($\abs{\cos t}/t^2 \leq t^{-2}$). Mas não \omterm{def:b2:series:def}{absolutamente}:
de $\abs{\sin t} \geq \sin^2 t$,
\[
\int_1^x \frac{\abs{\sin t}}{t}\,\dd t
\;\geq\; \int_1^x \frac{\sin^2t}{t}\,\dd t
= \underbrace{\int_1^x \frac{\dd t}{2t}}_{=\ \frac12\ln x
\ \to\ \infty}
\;-\; \underbrace{\int_1^x \frac{\cos 2t}{2t}\,\dd
t}_{\text{convergente}} ,
\]
em que a última integral \omterm{def:b2:integration:improper}{converge} pela \emph{mesma}
integração por partes acima (com $\sin 2t$ no colchete):
uma peça divergente menos uma convergente diverge. Assim
$\int_1^\infty\frac{\sin t}{t}\dd t$ \omterm{def:b2:integration:improper}{converge} sem convergir
\omterm{def:b2:series:def}{absolutamente} --- o análogo integral das
séries alternadas, com a integração por partes fazendo o papel
do critério das alternadas.
\end{example}

\section{O teorema de convergência}

\begin{theorem}[Convergência dominada]\label{thm:b2:integration:dominated}
Sejam $(f_n)$ \omterm{def:b2:metric:continuity}{contínuas} por partes num intervalo $I$, com \omterm{def:b2:metric:continuity}{continuidade} por partes, convergindo
pontualmente para uma $f$ \omterm{def:b2:metric:continuity}{contínua} por partes, com \omterm{def:b2:metric:continuity}{continuidade} por partes, e suponha que exista uma
função integrável fixa $\varphi \geq 0$ ($\int_I \varphi < \infty$)
com\index{teorema da convergência dominada}
\[
\abs{f_n(t)} \leq \varphi(t) \qquad (t \in I,\ n \in \N).
\]
Então todas as $\int_I f_n$ e $\int_I f$ convergem \omterm{def:b2:series:def}{absolutamente}, e
\[
\int_I f_n \xrightarrow[n \to \infty]{} \int_I f .
\]
\end{theorem}
\admitted

\begin{remark}
A demonstração honesta pertence à teoria da integração de Lebesgue do terceiro
ano; o enunciado, porém, é usado constantemente daqui em diante. A hipótese de
\emph{dominação} é o ponto todo: a convergência pontual
sozinha não basta ($f_n = n\,\mathbf{1}_{\intoo{0}{1/n}}$,
corcovas deslizantes: $\int f_n = 1 \not\to 0 = \int f$). O teorema
também vale para um parâmetro \omterm{def:b2:metric:continuity}{contínuo} ($f_\lambda$, $\lambda \to
\lambda_0$), pela caracterização sequencial dos limites.
\end{remark}

\begin{example}[Um limite gaussiano, por dominação]\label{ex:b2:integration:wallislimit}
Calcule $\displaystyle\lim_{n\to\infty} I_n$, em que $I_n =
\int_0^\infty \Bigl(1 + \frac{t^2}{n}\Bigr)^{\!-n}\dd t$.
Pontualmente, $(1 + t^2/n)^n \to \eu^{t^2}$ (o limite dos juros
compostos), de modo que os integrandos tendem a $\eu^{-t^2}$. Dominação: a
sequência $n \mapsto (1 + u/n)^n$ é não decrescente para $u \geq 0$
(MA--MG nos $n + 1$ fatores $1, 1 + \frac un, \dots, 1 +
\frac un$ dá $(1 + \frac u{n+1})^{n+1} \geq (1 + \frac
un)^n$), logo, para $n \geq 2$:
\[
\Bigl(1 + \frac{t^2}{n}\Bigr)^{\!-n}
\leq \Bigl(1 + \frac{t^2}{2}\Bigr)^{\!-2},
\]
um dominante integrável ($\sim 4t^{-4}$ no infinito). Convergência
dominada:
\[
I_n \xrightarrow[n\to\infty]{} \int_0^\infty \eu^{-t^2}\dd t
= \frac{\sqrt\pi}{2}
\]
(a integral gaussiana de \cref{exo:b2:integration:8}). Verificação
final: a substituição $t = \sqrt n\tan\theta$ calcula $I_n$
exatamente, $I_n = \sqrt n\int_0^{\pi/2}\cos^{2n-2}\theta\,\dd\theta
= \sqrt n\,W_{2n-2}$, e a assintótica de Wallis $W_m \sim
\sqrt{\pi/(2m)}$ (\cref{lem:b2:comparison:wallis}) dá $\sqrt
n\,W_{2n-2} \to \frac{\sqrt\pi}2$ de novo: os dois pilares deste
capítulo e do anterior concordam.
\end{example}

\begin{example}[Convergência dominada, parâmetro contínuo]
\label{ex:b2:integration:arctanlimit}
Calcule
\[
\lim_{x\to+\infty}\int_0^\infty
\frac{\arctan(xt)}{1+t^2}\,\dd t .
\]
Para cada $t > 0$,
$\arctan(xt) \to \frac\pi2$ quando $x \to \infty$; e a
dominação
\[
\Bigl|\frac{\arctan(xt)}{1+t^2}\Bigr|
\leq \frac{\pi/2}{1+t^2},
\qquad\text{integrável, independente de } x,
\]
vale para todo $x$. Pela forma com parâmetro \omterm{def:b2:metric:continuity}{contínuo} do
\cref{thm:b2:integration:dominated} (caracterização
sequencial: teste ao longo de toda $x_n \to \infty$),
\[
\int_0^\infty\frac{\arctan(xt)}{1+t^2}\,\dd t
\xrightarrow[x\to+\infty]{}
\frac\pi2\int_0^\infty\frac{\dd t}{1+t^2}
= \frac{\pi^2}{4} .
\]
Lição final: o único ponto $t = 0$, onde o limite pontual
vale $0$ em vez de $\frac\pi2$, nada muda --- a
função limite só entra por sua integral, uma das
misericórdias silenciosas do teorema.
\end{example}

\section{Integrais com parâmetro}

\begin{theorem}[Continuidade sob o sinal de integral]\label{thm:b2:integration:continuity}
Seja $f \colon A \times I \to \R$ ($A$ um \omterm{def:b2:metric:def}{espaço métrico}, $I$ um
intervalo) com: $t \mapsto f(x, t)$ \omterm{def:b2:metric:continuity}{contínua} por partes para cada
$x$, com \omterm{def:b2:metric:continuity}{continuidade} por partes; $x \mapsto f(x, t)$ \omterm{def:b2:metric:continuity}{contínua} para cada $t$; e uma
\emph{dominação} $\abs{f(x,t)} \leq \varphi(t)$ ($\varphi$
integrável em $I$, independente de $x$). Então
\[
F(x) = \int_I f(x, t)\,\dd t
\]
está definida e é \omterm{def:b2:metric:continuity}{contínua} em $A$.\index{integral com parâmetro}
\end{theorem}

\begin{proof}
Boa definição: a dominação dá a convergência absoluta. \omterm{def:b2:metric:continuity}{Continuidade} em
$x_0$: para toda sequência $x_n \to x_0$, as funções $g_n(t) =
f(x_n, t)$ convergem pontualmente para $f(x_0, t)$ (\omterm{def:b2:metric:continuity}{continuidade} em $x$)
sob a dominação fixa $\varphi$: a convergência dominada dá
$F(x_n) \to F(x_0)$; conclua pela caracterização sequencial da
\omterm{def:b2:metric:continuity}{continuidade} (\cref{def:b2:metric:continuity}).
\end{proof}

\begin{theorem}[Derivação sob o sinal de integral]\label{thm:b2:integration:leibnizrule}
Seja $f \colon J \times I \to \R$ ($J$ um intervalo de parâmetros)
com: $t \mapsto f(x,t)$ integrável em $I$ para cada $x$; $x \mapsto
f(x,t)$ de classe $C^1$ para cada $t$, sendo a derivada parcial
$\frac{\partial f}{\partial x}$ \omterm{def:b2:metric:continuity}{contínua} por partes em $t$, com \omterm{def:b2:metric:continuity}{continuidade} por partes,
e dominada: $\bigl|\frac{\partial f}{\partial x}(x,t)\bigr| \leq
\psi(t)$ com $\psi$ integrável. Então $F(x) = \int_I f(x,t)\dd t$
é $C^1$ em $J$ e
\[
F'(x) = \int_I \frac{\partial f}{\partial x}(x, t)\,\dd t .
\]
\end{theorem}

\begin{proof}
Fixe $x$ e $h_n \to 0$. Os quocientes de diferenças
\[
\frac{F(x + h_n) - F(x)}{h_n}
= \int_I \frac{f(x + h_n, t) - f(x, t)}{h_n}\,\dd t
\]
têm integrandos convergindo pontualmente para $\frac{\partial f}{\partial
x}(x, t)$ e dominados por $\psi(t)$: pela desigualdade do valor
médio aplicada em $x$ com $t$ fixo,
\[
\Bigl|\frac{f(x + h_n, t) - f(x,t)}{h_n}\Bigr|
\leq \sup_{\xi} \Bigl|\frac{\partial f}{\partial x}(\xi, t)\Bigr|
\leq \psi(t) .
\]
A convergência dominada dá o limite $\int_I \frac{\partial
f}{\partial x}(x,t)\dd t$ dos quocientes: $F$ é derivável
com a derivada anunciada, que é \omterm{def:b2:metric:continuity}{contínua} pelo
\cref{thm:b2:integration:continuity} aplicado a $\frac{\partial
f}{\partial x}$.
\end{proof}

\begin{example}[Uma integral com parâmetro conferida contra uma fórmula]\label{ex:b2:integration:arctanparam}
Seja $F(x) = \displaystyle\int_0^\infty \frac{\dd t}{t^2 + x}$
para $x > 0$. Em todo $\intcc{a}{b} \subset \intoo{0}{\infty}$,
o integrando é dominado por $\frac{1}{t^2 + a}$, integrável
e independente de $x$: $F$ é \omterm{def:b2:metric:continuity}{contínua}
(\cref{thm:b2:integration:continuity}). Aqui o teorema pode ser
conferido contra um valor explícito:
\[
F(x) = \Bigl[\frac{1}{\sqrt x}\arctan\frac{t}{\sqrt
x}\Bigr]_0^\infty = \frac{\pi}{2\sqrt x} ,
\]
visivelmente \omterm{def:b2:metric:continuity}{contínua}. Derive agora sob a integral: a
derivada em $x$, $-\frac{1}{(t^2+x)^2}$, é dominada em
$\intcc ab$ por $\frac{1}{(t^2+a)^2}$, integrável:
o~\cref{thm:b2:integration:leibnizrule} dá
\[
F'(x) = -\int_0^\infty \frac{\dd t}{(t^2 + x)^2}
\qquad\text{enquanto}\qquad
F'(x) = -\frac{\pi}{4}\,x^{-3/2} ,
\]
de modo que \emph{calculamos} uma integral nova de graça:
$\int_0^\infty\frac{\dd t}{(t^2+x)^2} = \frac{\pi}{4x^{3/2}}$.
Lição final: derivar uma \omterm{thm:b2:integration:continuity}{integral com parâmetro} conhecida é uma
fábrica de fórmulas novas --- iterando obtém-se
$\int_0^\infty\frac{\dd t}{(t^2+1)^n}$ para todo $n$, sem nenhuma
substituição trigonométrica.
\end{example}

\begin{method}[Estudar uma integral imprópria]\label{met:b2:integration:study}
Dada $\int_a^b f$:
\begin{enumerate}
  \item Localize o problema: liste as extremidades (ou pontos
        interiores) em que $f$ é ilimitada ou o intervalo é
        infinito, e separe de modo que cada peça tenha exatamente uma
        ponta problemática.
  \item Se $f$ tem sinal constante perto dessa ponta, ache um
        equivalente e compare com as escalas de referência do
        \cref{thm:b2:integration:comparison}.
  \item Se $f$ oscila, teste $\abs f$ primeiro (convergência
        absoluta). Se $\int\abs f$ diverge, tente a integração
        por partes para trocar a oscilação por decaimento, como no
        \cref{ex:b2:integration:sint}; minorações do tipo
        $\abs{\sin t} \geq \sin^2t$ detectam a
        semiconvergência genuína.
  \item Para um valor, e não apenas a natureza: partes, substituição
        ou um parâmetro (derive uma integral mais simples, como no
        \cref{ex:b2:integration:arctanparam} e no
        \cref{ex:b2:integration:laplace}).
  \item Verificações de bom senso em qualquer valor calculado: sinal e ordem de
        grandeza contra uma cota bruta ($\int_0^\infty
        \eu^{-t^2}\dd t \in \intoo{0}{1 + \int_1^\infty
        \eu^{-t}}$, logo $\frac{\sqrt\pi}{2} \approx 0.886$ é
        plausível); e coerência dimensional por reescalamento
        ($t \mapsto \lambda t$ deve reescalar os dois lados do mesmo
        jeito --- o detector mais rápido de um fator perdido).
\end{enumerate}
\end{method}

\begin{remark}[Armadilhas comuns]
Três erros recorrentes. \emph{(i) Dominantes que dependem do
parâmetro:} a dominação $\abs{f(x,t)} \leq \varphi(t)$ tem de
ser uniforme em $x$ no conjunto considerado; ela costuma valer em
segmentos $\intcc ab$ mas não globalmente --- para
$\int_0^\infty\eu^{-xt}\dd t$ não há dominante integrável
válido para todo $x > 0$, e no entanto dominar em $x \geq a > 0$ basta
para trabalhar em toda a semirreta \omterm{def:b2:metric:topology}{aberta}, pois \omterm{def:b2:metric:continuity}{continuidade}
e derivadas são noções locais. \emph{(ii) Comparar integrandos
com sinal:} a caixa de ferramentas de comparação é para funções
\emph{não negativas}; de $\abs f \leq g$ com $\int g$ divergente nada se pode
concluir --- $\int_1^\infty\frac{\sin t}t\,\dd t$
\omterm{def:b2:integration:improper}{converge} embora toda comparação com $\frac1t$ falhe.
\emph{(iii) Esquecer metade do problema:} em
$\intoo{0}{\infty}$ estude sempre as duas pontas separadamente;
$\int_0^\infty\frac{\dd t}{t}$ diverge em \emph{ambas}, e uma separação
de aparência convergente pode cancelar silenciosamente dois infinitos.
O reflexo seguro é a lista de verificação do
\cref{met:b2:integration:study}.
\end{remark}

\begin{example}[Um caso de fronteira de Bertrand, até o algarismo]\label{ex:b2:integration:bertrandboundary}
A escala $\int^\infty\frac{\dd t}{t(\ln t)^\beta}$ do
\cref{thm:b2:integration:comparison} fica exatamente na borda
das escalas de potências; seus casos de fronteira merecem um
cálculo \omterm{def:b2:metric:complete}{completo}. Para $\beta = 2$:
\[
\int_\eu^{\infty}\frac{\dd t}{t(\ln t)^2}
= \Bigl[-\frac{1}{\ln t}\Bigr]_\eu^{\infty}
= 0 - (-1) = 1 ,
\]
uma integral convergente de valor agradavelmente exato; ao passo que, para
$\beta = 1$,
\[
\int_\eu^{x}\frac{\dd t}{t\ln t}
= \bigl[\ln\ln t\bigr]_\eu^{x} = \ln\ln x
\longrightarrow \infty ,
\]
divergente --- mas tão devagar que atingir $\ln\ln x = 10$
exige $x = \eu^{\eu^{10}} \approx 10^{9566}$. Lição
final: entre ``toda potência $t^{-1-\varepsilon}$
\omterm{def:b2:integration:improper}{converge}'' e ``$t^{-1}$ diverge'' mora uma escada infinita
de escalas logarítmicas, cada uma refinando a anterior; a substituição
$u = \ln t$ colapsa cada degrau sobre o anterior, e é por isso
que os critérios de Bertrand ecoam os de Riemann um nível
acima.
\end{example}

\begin{remark}[Perspectivas dentro deste volume]
As ferramentas deste capítulo estão prestes a aparecer em toda parte. A convergência
dominada é o motor por trás das identidades aproximadas do
capítulo seguinte (núcleos deslizantes, tanto Bernstein quanto Fejér);
a \omterm{def:b2:metric:continuity}{continuidade} e a derivação sob o sinal de integral
produzem o cálculo dos coeficientes de Fourier no capítulo de
Fourier, onde todo $c_n(f)$ é uma \omterm{thm:b2:integration:continuity}{integral com parâmetro}
disfarçada. A função $\Gamma$ volta duas vezes: no capítulo
sobre integrais múltiplas, em que uma integral dupla finalmente demonstra
a fórmula Beta--Gama de Euler por inteiro, e nos capítulos de probabilidade,
em que integrais do tipo $\Gamma$ normalizam as densidades
padrão e calculam seus momentos. E a semiconvergente
$\int\frac{\sin t}{t}$ ressurge como a constante de Gibbs do
capítulo de Fourier --- a mesma integral, medindo a ultrapassagem
das somas parciais num salto.
\end{remark}

\begin{definition}[A função $\Gamma$]\label{def:b2:integration:gamma}
Para $x > 0$:\index{função Gama}
\[
\Gamma(x) = \int_0^{\infty} t^{x-1}\,\eu^{-t}\,\dd t ,
\]
convergente nas duas pontas ($t^{x-1}$ integrável em $0^+$ para $x > 0$;
decaimento exponencial em $\infty$).
\end{definition}

\begin{theorem}\label{thm:b2:integration:gammaprops}
$\Gamma$ é \omterm{def:b2:metric:continuity}{contínua} em $\intoo{0}{+\infty}$, satisfaz a
equação funcional
\[
\Gamma(x + 1) = x\,\Gamma(x),
\qquad \Gamma(1) = 1,
\qquad\text{logo}\qquad \Gamma(n + 1) = n! ,
\]
e é de classe $C^1$ (na verdade $C^\infty$) com $\Gamma'(x) =
\int_0^\infty t^{x-1}\eu^{-t}\ln t\,\dd t$.
\end{theorem}

\begin{proof}
Equação funcional: integre por partes em $\intcc{\varepsilon}{M}$
e faça as pontas irem ao limite: $\int t^{x}\eu^{-t} = [-t^x\eu^{-t}] + x\int
t^{x-1}\eu^{-t}$, anulando-se os termos de bordo --- de fato,
$\varepsilon^x\eu^{-\varepsilon} \to 0$ quando $\varepsilon \to
0^+$, pois $x > 0$, e $M^x\eu^{-M} \to 0$ quando $M \to \infty$,
pois a exponencial vence toda potência; as duas integrais truncadas
convergem para seus valores impróprios pela convergência
estabelecida na~\cref{def:b2:integration:gamma}. $\Gamma(1) = \int
\eu^{-t} = 1$; a indução dá o fatorial.

\omterm{def:b2:metric:continuity}{Continuidade} em $\intcc{a}{b} \subset \intoo{0}{\infty}$: domine
$t^{x-1}\eu^{-t}$ por $\varphi(t) = (t^{a-1} + t^{b-1})\eu^{-t}$,
integrável e independente de $x \in \intcc{a}{b}$:
o~\cref{thm:b2:integration:continuity} se aplica em todo segmento desse tipo,
logo em toda a semirreta. Derivabilidade: a derivada em $x$,
$t^{x-1}\eu^{-t}\ln t$, é dominada em $\intcc{a}{b}$ por $(t^{a-1}
+ t^{b-1})\eu^{-t}\,\abs{\ln t}$, ainda integrável:
\cref{thm:b2:integration:leibnizrule}; iterando obtêm-se todas as
derivadas (cada uma acrescenta uma potência de $\ln t$, inofensiva).
\end{proof}

\begin{example}[Fatoriais semi-inteiros]\label{ex:b2:integration:halffactorial}
A equação funcional e $\Gamma\bigl(\frac12\bigr) =
\sqrt\pi$ (a uma substituição de
\cref{exo:b2:integration:8}: ponha $t = u^2$ na integral que
define) geram todos os valores semi-inteiros:
\[
\Gamma\Bigl(\frac32\Bigr) = \frac12\,\Gamma\Bigl(\frac12\Bigr)
= \frac{\sqrt\pi}{2},
\qquad
\Gamma\Bigl(\frac52\Bigr) = \frac32\cdot\frac{\sqrt\pi}{2}
= \frac{3\sqrt\pi}{4},
\qquad
\Gamma\Bigl(\frac72\Bigr) = \frac{15\sqrt\pi}{8} .
\]
Como $\Gamma(n+1) = n!$, é justo dizer ``$\frac12! =
\frac{\sqrt\pi}{2} \approx 0.886$'': o fatorial foi
interpolado, e a curva interpoladora mergulha abaixo de $1$
entre $0! = 1$ e $1! = 1$ (seu mínimo $\approx 0.8856$ em
$x \approx 1.4616$ combina com o retrato de convexidade da
Parte I do problema de fim de semana). Lição final: nada na
integral $\int_0^\infty t^{x-1}\eu^{-t}\dd t$ privilegia os
inteiros --- a discretude do fatorial era um acidente da
contagem, e $\sqrt\pi$ é o que vive entre $1$ e $1$.
\end{example}

\begin{remark}[Para onde vai $\Gamma$ daqui]
O problema de fim de semana deste capítulo constrói todo o cálculo
de Euler em torno de $\Gamma$: a \omterm{pb:b2:integration:1}{função Beta}, suas recursões por
integração por partes, as integrais de Wallis como valores de Beta e a
fórmula-limite de Gauss. O capítulo sobre integrais múltiplas demonstra a
fórmula Beta--Gama de Euler para todos os argumentos por uma integral dupla; os
capítulos de probabilidade reencontram $\Gamma$ na normalização das
densidades mais comuns e nos momentos dos tempos de espera.
O volume do terceiro ano de graduação reconstrói $\Gamma$ sobre fundamentos de Lebesgue,
demonstra o teorema de unicidade de Bohr--Mollerup e estende
a fórmula de Stirling dos inteiros à semirreta real por
convergência dominada.
\end{remark}

\begin{example}[Um cálculo clássico por derivação]\label{ex:b2:integration:laplace}
Para $x \in \R$, seja $F(x) = \int_0^{\infty} \eu^{-t^2}\cos(xt)\,\dd
t$ (\omterm{def:b2:series:def}{absolutamente} convergente, dominada por $\eu^{-t^2}$). Pelo
\cref{thm:b2:integration:leibnizrule} (dominação da
derivada em $x$ por $t\,\eu^{-t^2}$, integrável):
\[
F'(x) = -\int_0^\infty t\,\eu^{-t^2}\sin(xt)\,\dd t
= \Bigl[\frac{\eu^{-t^2}}{2}\sin(xt)\Bigr]_0^\infty
- \frac x2\int_0^\infty \eu^{-t^2}\cos(xt)\,\dd t
= -\frac x2\,F(x),
\]
(partes com $u' = t\eu^{-t^2}$). A equação diferencial $F' =
-\frac x2 F$ integra-se em $F(x) = F(0)\,\eu^{-x^2/4}$: a
integral do tipo gaussiano se reproduz. A constante $F(0) =
\int_0^\infty \eu^{-t^2}\dd t = \frac{\sqrt\pi}{2}$ é calculada no
\cref{exo:b2:integration:8} --- e de novo, por integração dupla,
no~\cref{ch:b2:multint}.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:integration:1}
Natureza de:
$\displaystyle\int_0^1 \frac{\dd t}{\sqrt{t(1-t)}}$;
$\;\displaystyle\int_1^\infty \frac{\ln t}{t^2}\dd t$;
$\;\displaystyle\int_0^\infty \frac{\dd t}{1 + t^2\sin^2 t}$
\emph{(compare com o comportamento divergente do tipo harmônico perto de $t =
n\pi$)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:integration:2}
Calcule $\displaystyle\int_0^\infty t^n \eu^{-\lambda t}\,\dd t$
($\lambda > 0$) via $\Gamma$, e
$\displaystyle\int_0^1 (\ln t)^n \dd t$ pela substituição $t =
\eu^{-u}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:integration:3}
Prove que $\displaystyle\int_0^{\infty} \frac{\dd t}{(1 +
t^2)(1 + t^x)}$ está bem definida para todo $x \in \R$ e é
independente de $x$. \emph{(Substitua $t \mapsto \frac1t$ e
faça a média das duas expressões.)} Qual é seu valor?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:4}
(Integrais de Bertrand numa extremidade finita) Para quais $(\alpha,
\beta)$ a integral $\displaystyle\int_0^{1/2}
\frac{\dd t}{t^\alpha\,\abs{\ln t}^\beta}$ \omterm{def:b2:integration:improper}{converge}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:5}
Seja $F(x) = \displaystyle\int_0^{\infty} \frac{\eu^{-xt}}{1 +
t^2}\,\dd t$ para $x \geq 0$. Prove que $F$ é \omterm{def:b2:metric:continuity}{contínua} em
$\intco{0}{\infty}$, de classe $C^2$ em $\intoo{0}{\infty}$, satisfaz $F'' +
F = \frac1x$ aí, e que $F(x) \to 0$ quando $x \to +\infty$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:6}
(Frullani) Seja $f$ \omterm{def:b2:metric:continuity}{contínua} em $\intco{0}{+\infty}$ com
limite finito $f(\infty)$ em $+\infty$. Prove que, para $a, b > 0$:
\[
\int_0^{\infty} \frac{f(at) - f(bt)}{t}\,\dd t
= \bigl(f(0) - f(\infty)\bigr)\,\ln\frac ba .
\]
\emph{(Em $\intcc{\varepsilon}{M}$, substitua em cada peça e
reagrupe em $\int_{a\varepsilon}^{b\varepsilon} -
\int_{aM}^{bM}$ de $\frac{f(u)}u\,\dd u$; confronte usando a
\omterm{def:b2:metric:continuity}{continuidade} em $0$ e o limite em $\infty$.)} Calcule
$\int_0^\infty \frac{\eu^{-t} - \eu^{-2t}}{t}\dd t$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:7}
Justifique e calcule $\lim_{n\to\infty} \displaystyle\int_0^n
\Bigl(1 - \frac tn\Bigr)^{\!n} t^{x-1}\,\dd t$ para $x > 0$
\emph{(convergência dominada com $\varphi(t) = \eu^{-t}t^{x-1}$,
usando $(1 - t/n)^n \leq \eu^{-t}$; o limite é $\Gamma(x)$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:8}
(A integral gaussiana por um truque de parâmetro) Para $x \geq 0$ ponha
\[
G(x) = \Bigl(\int_0^x \eu^{-t^2}\dd t\Bigr)^{\!2},
\qquad
H(x) = \int_0^1 \frac{\eu^{-x^2(1+t^2)}}{1 + t^2}\,\dd t .
\]
Prove que $G' + H' = 0$ (derive $H$ sob a integral e
substitua $u = xt$ na integral resultante), deduza $G(x) + H(x)
= \frac\pi4$ para todo $x$ e conclua
\[
\int_0^{\infty} \eu^{-t^2}\,\dd t = \frac{\sqrt\pi}{2} .
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:9}
Prove que $\Gamma$ é \emph{log-convexa}: $\ln\Gamma$ é convexa em
$\intoo{0}{\infty}$. \emph{(A desigualdade de Cauchy--Schwarz para integrais aplicada
a $t^{(x+y)/2 - 1}\eu^{-t} = \bigl(t^{x-1}\eu^{-t}\bigr)^{1/2}
\bigl(t^{y-1}\eu^{-t}\bigr)^{1/2}$ dá
$\Gamma\bigl(\frac{x+y}{2}\bigr)^2 \leq \Gamma(x)\Gamma(y)$;
combine com a \omterm{def:b2:metric:continuity}{continuidade} e \cref{exo:b2:realfun:8}.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:10}
(Integral de Dirichlet) Ponha $F(x) = \displaystyle\int_0^{\infty}
\frac{\sin t}{t}\,\eu^{-xt}\,\dd t$ para $x > 0$.
\begin{enumerate}
  \item Justifique $F'(x) = -\frac{1}{1 + x^2}$ (derive sob
        a integral; calcule $\int_0^\infty \eu^{-xt}\sin t\,\dd
        t$ por duas integrações por partes).
  \item Prove que $F(x) \to 0$ quando $x \to +\infty$ e deduza $F(x) =
        \frac\pi2 - \arctan x$.
  \item Admitindo a \omterm{def:b2:metric:continuity}{continuidade} de $F$ em $0^+$ (um teorema
        do tipo Abel), conclua o valor da integral
        semiconvergente:
        \[
        \int_0^{\infty} \frac{\sin t}{t}\,\dd t = \frac{\pi}{2}.
        \]
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:11}
Justifique a convergência de $\displaystyle\int_0^\infty
\Bigl(\frac{\sin t}{t}\Bigr)^{\!2}\dd t$ e depois calcule-a por uma
integração por partes e \cref{exo:b2:integration:10}:
\[
\int_0^\infty \Bigl(\frac{\sin t}{t}\Bigr)^{\!2}\dd t
= \frac{\pi}{2} .
\]
(O mesmo valor do $\int_0^\infty \frac{\sin t}{t}\dd t$ --- mas
desta vez a convergência é absoluta.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:12}
(A cauda gaussiana) Para $x > 0$ ponha $T(x) = \displaystyle
\int_x^\infty \eu^{-t^2}\dd t$.
\begin{enumerate}
  \item Escrevendo $\eu^{-t^2} = \frac{1}{-2t}\cdot(-2t\,\eu^{-t^2})$,
        integre por partes duas vezes para obter
        \[
        T(x) = \eu^{-x^2}\Bigl(\frac{1}{2x} -
        \frac{1}{4x^3}\Bigr) +
        \frac34\int_x^\infty \frac{\eu^{-t^2}}{t^4}\,\dd t .
        \]
  \item Majore o resto: $0 \leq \frac34\int_x^\infty
        t^{-4}\eu^{-t^2}\dd t \leq \frac{3}{8x^5}\,\eu^{-x^2}$,
        e deduza o enquadramento
        \[
        \eu^{-x^2}\Bigl(\frac{1}{2x} - \frac{1}{4x^3}\Bigr)
        \leq T(x) \leq \frac{\eu^{-x^2}}{2x},
        \qquad\text{logo}\qquad
        T(x) \sim \frac{\eu^{-x^2}}{2x} \quad (x \to +\infty).
        \]
  \item Por que a série alternada \omterm{def:b2:metric:complete}{completa} obtida iterando
        as partes nunca pode convergir para $x$ fixo? \emph{(Compare o
        crescimento dos coeficientes $1\cdot3\cdots(2k-1)$ com as
        potências $(2x^2)^k$.)}
\end{enumerate}
\end{exercise}

\section{Problema: as integrais de Euler --- Beta, Gama e a fórmula-limite
de Gauss}

\begin{problem}\label{pb:b2:integration:1}
A função $\Gamma$ da~\cref{def:b2:integration:gamma} é uma
metade do cálculo de integrais de Euler; a outra metade é a
\emph{função Beta}\index{função Beta}
\[
B(x, y) = \int_0^1 t^{x-1}(1 - t)^{y-1}\,\dd t .
\]
Este problema desenvolve o par $(\Gamma, B)$ apenas com as ferramentas
deste capítulo --- integração por partes, substituição,
convergência dominada --- e culmina na \emph{fórmula
Beta--Gama de Euler} $B(x,y) = \frac{\Gamma(x)\Gamma(y)}
{\Gamma(x+y)}$ nos semi-inteiros e na \emph{fórmula-limite
de Gauss}\index{fórmula-limite de Gauss} para $\Gamma$. Pelo caminho,
as integrais de Wallis do~\cref{lem:b2:comparison:wallis} reaparecem
como valores de Beta, e a fórmula de duplicação de Legendre cai no colo.

\medskip
\noindent\textbf{Parte I --- Estrutura fina de $\Gamma$.}
\begin{enumerate}
  \item Relembre por que $\Gamma(x) = \int_0^\infty t^{x-1}\eu^{-t}\dd
        t$ \omterm{def:b2:integration:improper}{converge} exatamente para $x > 0$ e mostre que
        \[
        \Gamma(x) \sim \frac1x \qquad (x \to 0^+)
        \]
        \emph{(equação funcional mais \omterm{def:b2:metric:continuity}{continuidade} de $\Gamma$ em
        $1$)}.
  \item Prove $\Gamma\bigl(\tfrac12\bigr) = \sqrt\pi$
        \emph{(substitua $t = u^2$ e invoque
        o~\cref{exo:b2:integration:8})} e deduza
        $\int_\R \eu^{-u^2/2}\dd u = \sqrt{2\pi}$.
  \item Mostre por indução, para $n \in \N$:
        \[
        \Gamma\Bigl(n + \frac12\Bigr)
        = \frac{(2n)!}{4^n\,n!}\,\sqrt\pi .
        \]
  \item Justifique $\Gamma''(x) = \int_0^\infty
        t^{x-1}\eu^{-t}(\ln t)^2\dd t > 0$ e deduza que
        $\Gamma$ é estritamente convexa, atinge um mínimo único em
        algum $x_0 \in \intoo{1}{2}$ \emph{($\Gamma(1) = \Gamma(2)
        = 1$ e Rolle)}, decresce em $\intoo{0}{x_0}$ e
        cresce em $\intoo{x_0}{\infty}$.
  \item Mostre que $\Gamma$ vence toda potência: para cada $k \in
        \N$, $x^k = o\bigl(\Gamma(x)\bigr)$ quando $x \to +\infty$
        \emph{(confronte $x$ entre inteiros e use $\Gamma(n+1)
        = n!$ com a monotonicidade da questão 4)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A \omterm{pb:b2:integration:1}{função Beta}, por partes.}
\begin{enumerate}[resume]
  \item Mostre que $B(x,y)$ \omterm{def:b2:integration:improper}{converge} exatamente para $x > 0$ e $y >
        0$, e que $B(x,y) = B(y,x)$.
  \item Calcule $B(x, 1) = \frac1x$ e prove, por integração por
        partes, para $x, y > 0$:
        \[
        B(x, y+1) = \frac{y}{x}\,B(x+1, y) .
        \]
  \item Da decomposição $t^{x-1}(1-t)^{y-1} =
        t^{x}(1-t)^{y-1} + t^{x-1}(1-t)^{y}$ deduza $B(x,y) =
        B(x+1,y) + B(x,y+1)$ e combine com a questão 7 nas
        \emph{relações de descida}
        \[
        B(x, y+1) = \frac{y}{x+y}\,B(x,y),
        \qquad
        B(x+1, y) = \frac{x}{x+y}\,B(x,y) .
        \]
  \item Deduza, para $m, n \geq 1$ inteiros:
        \[
        B(m, n) = \frac{(m-1)!\,(n-1)!}{(m+n-1)!}
        = \frac{1}{(m+n-1)\binom{m+n-2}{m-1}} .
        \]
  \item Demonstre a \emph{fórmula de Euler com um argumento inteiro}:
        para todo $x > 0$ e $n \in \N^*$,
        \[
        B(x, n) = \frac{\Gamma(x)\,\Gamma(n)}{\Gamma(x + n)}
        \]
        \emph{(indução sobre $n$: os dois lados valem $\frac1x$ em $n
        = 1$ e obedecem à mesma relação de descida)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- As integrais de Wallis como valores de Beta.}
\begin{enumerate}[resume]
  \item Substitua $t = \sin^2\theta$ para obter a forma
        trigonométrica
        \[
        B(x, y) = 2\int_0^{\pi/2}
        \sin^{2x-1}\theta\,\cos^{2y-1}\theta\,\dd\theta .
        \]
  \item Deduza $W_n = \frac12\,B\bigl(\frac{n+1}2, \frac12\bigr)$
        para a integral de Wallis $W_n = \int_0^{\pi/2}\sin^n
        \theta\,\dd\theta$ e recupere a recorrência $W_n =
        \frac{n-1}{n}W_{n-2}$ do~\cref{lem:b2:comparison:wallis}
        só a partir das relações de descida da questão 8.
  \item Calcule $B\bigl(\frac12, \frac12\bigr) = 2W_0 = \pi$ e
        confira contra $\Gamma\bigl(\frac12\bigr)^2/\Gamma(1)$:
        a fórmula de Euler vale em $\bigl(\frac12,
        \frac12\bigr)$.
  \item Deduza a forma fechada $W_{2n} = \frac\pi2\,
        \frac{(2n)!}{4^n(n!)^2}$ da recorrência e verifique
        \[
        B\Bigl(n + \frac12, \frac12\Bigr)
        = \frac{\Gamma\bigl(n + \frac12\bigr)\Gamma\bigl(
        \frac12\bigr)}{\Gamma(n+1)} .
        \]
        Conclua, por indução com as relações de descida, que
        a fórmula de Euler $B(x,y) =
        \frac{\Gamma(x)\Gamma(y)}{\Gamma(x+y)}$ vale sempre que
        $2x$ e $2y$ são inteiros positivos.
  \item Substitua $u = \frac{t}{1-t}$ para obter a terceira
        forma clássica
        \[
        B(x,y) = \int_0^\infty
        \frac{u^{x-1}}{(1+u)^{x+y}}\,\dd u ,
        \]
        e confira o caso $x = y = \frac12$ diretamente ($u = v^2$
        reduz a $\int_0^\infty\frac{2\,\dd v}{1+v^2}$).
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A fórmula-limite de Gauss.}
\begin{enumerate}[resume]
  \item Para $x > 0$ e $n \in \N^*$, prove por $n$ integrações
        por partes sucessivas:
        \[
        \int_0^n \Bigl(1 - \frac tn\Bigr)^{\!n} t^{x-1}\,\dd t
        = \frac{n!\;n^x}{x(x+1)\cdots(x+n)} .
        \]
  \item Conclua com \cref{exo:b2:integration:7} (convergência
        dominada) a \emph{fórmula-limite de Gauss}:
        \[
        \Gamma(x) = \lim_{n\to\infty}
        \frac{n!\;n^x}{x(x+1)\cdots(x+n)}
        \qquad (x > 0).
        \]
  \item Tomando logaritmos, mostre que, para $x > 0$:
        \[
        \ln\Gamma(x) = -\ln x - \gamma x +
        \sum_{k=1}^{\infty}\Bigl(\frac xk -
        \ln\Bigl(1 + \frac xk\Bigr)\Bigr),
        \]
        em que $\gamma$ é a constante de Euler
        (\cref{ex:b2:comparison:harmonic}); justifique a
        convergência da série \emph{(o termo geral é
        $\sim \frac{x^2}{2k^2}$)}.
  \item Use a fórmula de Gauss em $x = \frac12$ e a assintótica do
        binomial central $\binom{2n}{n} \sim
        \frac{4^n}{\sqrt{\pi n}}$
        (\cref{ex:b2:comparison:centralbinomial}) para recalcular
        $\Gamma\bigl(\frac12\bigr) = \sqrt\pi$: a constante de Stirling
        e a integral gaussiana são o mesmo número
        em dois disfarces.
  \item Verifique que a fórmula de Gauss redemonstra a equação
        funcional: da identidade exata
        \[
        \frac{n!\,n^{x+1}}{(x+1)\cdots(x+n+1)}
        = \frac{n!\,n^{x}}{x(x+1)\cdots(x+n)}\cdot
        \frac{n\,x}{x+n+1},
        \]
        conclua $\Gamma(x+1) = x\,\Gamma(x)$ mais uma vez. (A fórmula de
        Gauss determina $\Gamma$ de imediato; o volume do terceiro ano de graduação
        demonstra o teorema mais fino de Bohr--Mollerup: a
        equação funcional mais a log-convexidade já fixam
        $\Gamma$.)
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dividendos.}
\begin{enumerate}[resume]
  \item Para $a > 0$ mostre que $\int_0^\infty \eu^{-t^a}\dd t =
        \Gamma\bigl(1 + \frac1a\bigr)$ e calcule o limite quando
        $a \to +\infty$ por convergência dominada \emph{(limite
        pontual $\mathbf 1_{t < 1}$; domine por $1$ em
        $\intoc{0}{1}$ e por $\eu^{-t^2}$ além, para $a \geq
        2$)}. Confira a resposta contra a \omterm{def:b2:metric:continuity}{continuidade} de
        $\Gamma$.
  \item Para $n \geq 1$ mostre que
        \[
        \int_0^1 \frac{\dd t}{\sqrt{1 - t^n}}
        = \frac1n\,B\Bigl(\frac1n, \frac12\Bigr),
        \]
        e recupere os valores $2$ ($n = 1$) e $\frac\pi2$ ($n
        = 2$). (Para $n = 4$ essa é a \emph{constante da
        lemniscata}, que não tem forma fechada elementar; sua
        história pertence à teoria das integrais elípticas.)
  \item (Momentos) Para $x > 0$ e $k \in \N$, mostre que
        \[
        \frac{1}{\Gamma(x)}\int_0^\infty
        t^{k}\,t^{x-1}\eu^{-t}\,\dd t
        = \frac{\Gamma(x+k)}{\Gamma(x)}
        = x(x+1)\cdots(x+k-1),
        \]
        o fatorial ascendente; confira que $x = 1$ dá $k!$. (Nos
        capítulos de probabilidade esse é o $k$-ésimo momento de uma
        densidade padrão de tempo de espera.)
  \item Demonstre a identidade Beta, válida para \emph{todos} $x > 0$:
        \[
        B(x, x) = 2^{1-2x}\,B\Bigl(x, \frac12\Bigr)
        \]
        \emph{(substitua $t = \frac{1+s}2$, explore a
        simetria em $s$ e depois ponha $s = \sqrt v$)}. Deduza, para
        $2x \in \N^*$, a \emph{fórmula de duplicação de Legendre}
        \[
        \Gamma(x)\,\Gamma\Bigl(x + \frac12\Bigr)
        = 2^{1-2x}\,\sqrt\pi\;\Gamma(2x),
        \]
        e verifique-a diretamente em $x = n$ pela questão 3. (Para
        $x$ geral ela decorre da mesma identidade uma vez
        conhecida a fórmula de Euler para todos os argumentos --- a
        demonstração por integral dupla no capítulo sobre integrais
        múltiplas.)
  \item Síntese. Uma frase para cada: (i) onde a integração
        por partes carregou toda a Parte II; (ii) onde a
        convergência dominada entrou nas Partes IV e V; (iii) quais
        insumos assintóticos foram importados do capítulo de
        comparação; (iv) o que está agora demonstrado da fórmula de Euler
        $B(x,y) = \Gamma(x)\Gamma(y)/\Gamma(x+y)$, e o que
        resta para a integral dupla resolver.
\end{enumerate}
\end{problem}
