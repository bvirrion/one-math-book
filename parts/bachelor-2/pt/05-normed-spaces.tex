\chapter{Espaços vetoriais normados}\label{ch:b2:nvs}

Quando o \omterm{def:b2:metric:def}{espaço métrico} é um espaço vetorial e a distância provém de
uma \omterm{def:b2:nvs:norm}{norma}, topologia e álgebra linear começam a interagir: as aplicações lineares
são \omterm{def:b2:metric:continuity}{contínuas} exatamente quando são limitadas na bola unitária,
a dimensão finita obriga todas as \omterm{def:b2:nvs:norm}{normas} a coincidirem, e a \omterm{def:b2:metric:complete}{completude} transforma
séries absolutamente convergentes em séries convergentes. A
divisão entre dimensão finita e infinita --- cristalizada no teorema de Riesz
--- é a lição mais profunda do capítulo.

Em todo o capítulo, $E, F$ são espaços vetoriais sobre $K = \R$ ou $\C$.

\section{Normas}

\begin{definition}\label{def:b2:nvs:norm}
Uma \emph{norma}\index{norma} em $E$ é uma aplicação $\norm{\,\cdot\,} \colon E
\to \R_+$ tal que, para todos $x, y \in E$, $\lambda \in K$:
\[
\norm x = 0 \iff x = 0,
\qquad
\norm{\lambda x} = \abs\lambda\,\norm x,
\qquad
\norm{x + y} \leq \norm x + \norm y .
\]
Então $d(x, y) = \norm{x - y}$ é uma distância, e tudo o que está no
\cref{ch:b2:metric} se aplica. A desigualdade triangular invertida
$\bigl|\norm x - \norm y\bigr| \leq \norm{x - y}$ torna a própria norma
$1$-lipschitziana; a adição e a multiplicação por escalar são
\omterm{def:b2:metric:continuity}{contínuas} (estimativas $\norm{(x + y) - (x' + y')} \leq \norm{x -
x'} + \norm{y - y'}$, etc.).
\end{definition}

\begin{example}\label{ex:b2:nvs:examples}
Em $K^n$:
\[
\norm{x}_1 = \sum_i \abs{x_i},
\qquad
\norm{x}_2 = \Bigl(\sum_i \abs{x_i}^2\Bigr)^{1/2},
\qquad
\norm{x}_\infty = \max_i \abs{x_i}
\]
($\norm\cdot_2$ é uma \omterm{def:b2:nvs:norm}{norma} por Cauchy--Schwarz, volume do primeiro ano de graduação). Em
$C(\intcc{a}{b})$:
\[
\norm f_\infty = \sup \abs f,
\qquad
\norm f_1 = \int_a^b \abs f,
\qquad
\norm f_2 = \Bigl(\int_a^b \abs f^2\Bigr)^{1/2},
\]
sendo as duas últimas \omterm{def:b2:nvs:norm}{normas} graças à positividade estrita da
integral e à desigualdade de Cauchy--Schwarz integral (volume do primeiro ano de graduação). Sobre matrizes:
qualquer \omterm{def:b2:nvs:norm}{norma} em $\mathcal M_n(K) \simeq K^{n^2}$; as \omterm{thm:b2:nvs:continuouslinear}{normas de operador}
adiante são as estruturalmente importantes.
\end{example}

\begin{definition}[Normas equivalentes]\label{def:b2:nvs:equivalent}
Duas \omterm{def:b2:nvs:norm}{normas} $N_1, N_2$ em $E$ são \emph{equivalentes}\index{normas
equivalentes} quando existem constantes $c, C > 0$ com
\[
c\,N_1 \leq N_2 \leq C\, N_1 .
\]
\omterm{def:b2:nvs:norm}{Normas} equivalentes têm os mesmos \omterm{def:b2:metric:topology}{abertos}, as mesmas sequências convergentes e
de Cauchy, as mesmas partes \omterm{def:b2:metric:compact}{compactas} e \omterm{def:b2:metric:complete}{completas}: a mesma
análise.
\end{definition}

\begin{example}[Não equivalência em dimensão infinita]\label{ex:b2:nvs:nonequivalent}
Em $C(\intcc{0}{1})$: $\norm f_1 \leq \norm f_\infty$ sempre, mas
nenhuma cota recíproca vale: $f_n(x) = x^n$ tem $\norm{f_n}_\infty = 1$
e $\norm{f_n}_1 = \frac{1}{n+1} \to 0$. Logo $f_n \to 0$ para
$\norm\cdot_1$ mas não para $\norm\cdot_\infty$: as duas \omterm{def:b2:nvs:norm}{normas}
discordam sobre a própria convergência.
\end{example}

\begin{example}[Constantes explícitas em dimensão $n$]\label{ex:b2:nvs:constants}
Em $K^n$ as três \omterm{def:b2:nvs:norm}{normas} clássicas são \omterm{def:b2:nvs:equivalent}{equivalentes} com constantes
ótimas:
\[
\norm x_\infty \leq \norm x_2 \leq \norm x_1
\leq \sqrt n\,\norm x_2 \leq n\,\norm x_\infty ,
\]
vindo a cota do meio $\norm x_1 \leq \sqrt n\norm x_2$ de
Cauchy--Schwarz contra o vetor de uns. Vetores extremais:
$e_1$ torna as duas primeiras desigualdades igualdades, e $(1, 1,
\dots, 1)$ as duas últimas. A dimensão $n$ aparece visivelmente nas
constantes --- a semente quantitativa do fracasso em dimensão
infinita: quando $n \to \infty$ nenhuma constante uniforme sobrevive,
que é exatamente o que o~\cref{ex:b2:nvs:nonequivalent} exibe em
espaços de funções.
\end{example}

\section{Aplicações lineares contínuas}

\begin{theorem}[Caracterização]\label{thm:b2:nvs:continuouslinear}
Para uma aplicação linear $u \colon E \to F$ entre espaços normados, as
afirmações seguintes são \omterm{def:b2:nvs:equivalent}{equivalentes}:
\begin{enumerate}
  \item $u$ é \omterm{def:b2:metric:continuity}{contínua};
  \item $u$ é \omterm{def:b2:metric:continuity}{contínua} em $0$;
  \item $u$ é limitada na bola unitária fechada: $\sup_{\norm x \leq
        1} \norm{u(x)} < \infty$;
  \item existe $C \geq 0$ com $\norm{u(x)} \leq C \norm x$ para
        todo $x$;
  \item $u$ é \omterm{def:b2:metric:continuity}{lipschitziana}.
\end{enumerate}
O menor tal $C$ é a \emph{norma de operador}\index{norma de operador}
$\vertiii{u} = \sup_{\norm x \leq 1}\norm{u(x)} = \sup_{x \neq 0}
\frac{\norm{u(x)}}{\norm x}$; ela faz do espaço $\mathcal{L}_c(E,
F)$ das aplicações lineares \omterm{def:b2:metric:continuity}{contínuas} um espaço normado, com
\[
\vertiii{v \circ u} \leq \vertiii v\, \vertiii u .
\]
\end{theorem}

\begin{proof}
(1 $\Rightarrow$ 2) trivial. (2 $\Rightarrow$ 3): a \omterm{def:b2:metric:continuity}{continuidade} em $0$
com $\varepsilon = 1$ dá $\delta$ com $\norm x \leq \delta
\Rightarrow \norm{u(x)} \leq 1$; a homogeneidade então reduz qualquer $x$
com $\norm x \leq 1$ para dentro dessa bola e de volta:
\[
\norm{u(x)} = \frac1\delta\,\norm{u(\delta x)} \leq
\frac1\delta ,
\]
pois $\norm{\delta x} \leq \delta$. (3 $\Rightarrow$ 4):
para $x \neq 0$, aplique a cota a $\frac{x}{\norm x}$. (4
$\Rightarrow$ 5): $\norm{u(x) - u(y)} = \norm{u(x - y)} \leq
C\norm{x - y}$. (5 $\Rightarrow$ 1) conhecido.

Axiomas de \omterm{def:b2:nvs:norm}{norma} para $\vertiii\cdot$: a homogeneidade e a separação são
claras ($\vertiii u = 0$ força $u = 0$ na bola, logo
em toda parte); a desigualdade triangular vem de $\norm{(u + v)(x)} \leq
\norm{u(x)} + \norm{v(x)}$. Submultiplicatividade: $\norm{v(u(x))}
\leq \vertiii v\,\norm{u(x)} \leq \vertiii v \vertiii u \norm x$.
\end{proof}

\begin{example}\label{ex:b2:nvs:operatornorms}
Em $\bigl(C(\intcc{0}{1}), \norm\cdot_\infty\bigr)$: a avaliação
$f \mapsto f(0)$ tem \omterm{thm:b2:nvs:continuouslinear}{norma de operador} $1$; a integração $f \mapsto
\int_0^1 f$ tem \omterm{def:b2:nvs:norm}{norma} $1$; a aplicação $f \mapsto \int_0^1 t f(t)\dd t$
tem \omterm{def:b2:nvs:norm}{norma} $\int_0^1 t\,\dd t = \frac12$ (cota superior pela desigualdade
triangular para integrais; atingida em $f \equiv 1$). Mas
a derivação, de $(C^1, \norm\cdot_\infty)$ em $(C^0,
\norm\cdot_\infty)$, \emph{não} é \omterm{def:b2:metric:continuity}{contínua}: $\norm{\sin(nx)}_\infty
= 1$ enquanto a derivada tem \omterm{def:b2:nvs:norm}{norma} do sup $n$. Linear não implica
\omterm{def:b2:metric:continuity}{contínua} em dimensão infinita.
\end{example}

\begin{example}[Duas normas, dois veredictos sobre uma sequência]\label{ex:b2:nvs:sqrtnxn}
Em $C(\intcc01)$, seja $g_n(x) = \sqrt{n}\,x^n$. Então
\[
\norm{g_n}_1 = \frac{\sqrt n}{n + 1} \longrightarrow 0,
\qquad
\norm{g_n}_2^2 = \frac{n}{2n + 1} \longrightarrow \frac12,
\qquad
\norm{g_n}_\infty = \sqrt n \longrightarrow \infty :
\]
uma sequência, três \omterm{def:b2:nvs:norm}{normas}, três comportamentos --- convergência para
zero, ausência de convergência (as \omterm{def:b2:nvs:norm}{normas} se estabilizam em $\frac1{\sqrt2}$ mas
o limite pontual é $0$) e explosão. A massa que se concentra
perto de $x = 1$ é invisível para $\norm\cdot_1$, meio visível para
$\norm\cdot_2$ e dominante para $\norm\cdot_\infty$. Em dimensão
infinita, ``isso converge?'' não é uma pergunta sobre uma
sequência: é uma pergunta sobre uma sequência \emph{e} uma \omterm{def:b2:nvs:norm}{norma}.
\end{example}

\begin{method}[Calcular uma norma de operador]\label{met:b2:nvs:opnorm}
Sempre em dois movimentos. \emph{Cota superior:} estime
$\norm{u(x)}$ por $C\norm x$ usando desigualdades triangulares,
Cauchy--Schwarz ou cotas integrais --- isso demonstra $\vertiii u
\leq C$. \emph{Testemunha:} exiba ou um $x_0 \neq 0$ específico
com $\norm{u(x_0)} = C\norm{x_0}$ (a cota é atingida), ou uma
sequência de vetores unitários $x_n$ com $\norm{u(x_n)} \to C$ (a
cota é aproximada). Os dois movimentos são obrigatórios: só a cota superior
dá apenas $\vertiii u \leq C$, e só a testemunha dá apenas
$\vertiii u \geq C$. Em dimensão infinita a testemunha pode ter
de ser uma sequência --- o supremo não precisa ser atingido
(\cref{exo:b2:nvs:8}).
\end{method}

\begin{example}[Os operadores diagonais veem todas as normas do mesmo jeito]\label{ex:b2:nvs:diagonalnorm}
Para $D = \operatorname{diag}(d_1, \dots, d_n)$ em $K^n$ com
\emph{qualquer} uma das \omterm{def:b2:nvs:norm}{normas} $\norm\cdot_1, \norm\cdot_2,
\norm\cdot_\infty$: de $\abs{d_ix_i} \leq
\bigl(\max_j\abs{d_j}\bigr)\abs{x_i}$ coordenada a coordenada,
$\norm{Dx} \leq \max_j\abs{d_j}\,\norm x$; e $x = e_{j_0}$
(um índice que maximiza) a atinge. Logo $\vertiii D =
\max_j\abs{d_j}$ nos três casos: para aplicações diagonais, todas as
\omterm{def:b2:nvs:norm}{normas} razoáveis contam a mesma história, o maior fator de
alongamento. Tudo o que é difícil nas \omterm{thm:b2:nvs:continuouslinear}{normas de operador} diz respeito ao
comportamento \emph{não} diagonal --- e é por isso que as \omterm{def:b2:nvs:norm}{normas} adaptadas
do problema de fim de semana deste capítulo (questão 22) funcionam
forçando primeiro a matriz a ficar diagonal.
\end{example}

\begin{example}[Somas por colunas: o gêmeo em $1$ do~\cref{exo:b2:nvs:4}]\label{ex:b2:nvs:columnsums}
Em $(\R^n, \norm\cdot_1)$, a \omterm{thm:b2:nvs:continuouslinear}{norma de operador} de uma matriz $A$ é
a maior soma de valores absolutos por \emph{coluna}. Rode o método: para
$\norm x_1 \leq 1$,
\[
\norm{Ax}_1 = \sum_i\Bigl|\sum_j a_{ij}x_j\Bigr|
\leq \sum_j \abs{x_j}\sum_i\abs{a_{ij}}
\leq \Bigl(\max_j\sum_i\abs{a_{ij}}\Bigr)\norm x_1 ,
\]
e a cota é atingida em $x = e_{j_0}$ para uma coluna $j_0$ que
maximize --- a testemunha mais arrumada que se pode imaginar. Assim, para $A =
\begin{psmallmatrix}1 & -2\\ 3 & 1\end{psmallmatrix}$:
$\vertiii A_1 = \max(1 + 3,\ 2 + 1) = 4$, ao passo que
$\vertiii A_\infty = 4$ também (linhas) --- uma coincidência aqui, não
uma lei: transponha as entradas da matriz de forma assimétrica e as duas
\omterm{def:b2:nvs:norm}{normas} se separam. Linhas para $\norm\cdot_\infty$, colunas para
$\norm\cdot_1$: a mnemônica é que os vetores unitários de cada \omterm{def:b2:nvs:norm}{norma}
(padrões de sinais, resp.\ vetores da base) selecionam as somas
correspondentes.
\end{example}

\begin{proposition}[Aplicações bilineares]\label{prop:b2:nvs:bilinear}
Uma aplicação bilinear $b \colon E \times F \to G$ é \omterm{def:b2:metric:continuity}{contínua} se e somente se
$\norm{b(x,y)} \leq C\norm x\,\norm y$ para algum $C$; e então ela é
\omterm{def:b2:metric:continuity}{lipschitziana} em conjuntos limitados. (Mesmo padrão de demonstração; o produto
$(u, v) \mapsto v \circ u$ e a multiplicação de matrizes são os exemplos
principais.)
\end{proposition}

\begin{proof}
Se a cota vale:
\[
b(x,y) - b(x_0,y_0) = b(x - x_0,\, y) + b(x_0,\, y - y_0),
\]
logo $\norm{b(x,y) - b(x_0,y_0)} \leq C\norm{x - x_0}\norm y +
C\norm{x_0}\norm{y - y_0}$: \omterm{def:b2:metric:continuity}{continuidade} em $(x_0, y_0)$, e uma
cota \omterm{def:b2:metric:continuity}{lipschitziana} onde $\norm x, \norm y \leq R$. Reciprocamente, a
\omterm{def:b2:metric:continuity}{continuidade} em $(0,0)$ dá $\delta$ com $\norm{b(x,y)} \leq 1$ em
$\norm x, \norm y \leq \delta$; reescale as duas variáveis.
\end{proof}

\section{Dimensão finita}

\begin{theorem}[Equivalência das normas em dimensão finita]\label{thm:b2:nvs:finitedim}
Num espaço de dimensão finita, \emph{todas as \omterm{def:b2:nvs:norm}{normas} são
\omterm{def:b2:nvs:equivalent}{equivalentes}}.\index{equivalência das normas} Por consequência, em dimensão
finita: convergência, abertura, \omterm{def:b2:metric:compact}{compacidade} e \omterm{def:b2:metric:complete}{completude} são
noções independentes da \omterm{def:b2:nvs:norm}{norma}; \omterm{def:b2:metric:compact}{compacto} $=$ fechado e limitado;
o espaço é \omterm{def:b2:metric:complete}{completo}; e toda aplicação linear (ou multilinear)
\emph{a partir de} um espaço de dimensão finita é \omterm{def:b2:metric:continuity}{contínua}.
\end{theorem}

\begin{proof}
Fixe uma base e identifique $E \simeq K^n$; basta comparar uma
\omterm{def:b2:nvs:norm}{norma} qualquer $N$ com $\norm\cdot_\infty$.

\emph{Uma direção é álgebra:} $N(x) = N(\sum x_i e_i) \leq
\sum \abs{x_i} N(e_i) \leq C \norm x_\infty$ com $C = \sum N(e_i)$.
Isso também mostra que $N$ é \omterm{def:b2:metric:continuity}{contínua} em $(K^n, \norm\cdot_\infty)$
(ela é $C$-lipschitziana: $\abs{N(x) - N(y)} \leq N(x - y)$).

\emph{A outra é topologia:} a esfera unitária $S = \{x :
\norm{x}_\infty = 1\}$ é fechada e limitada em $(K^n,
\norm\cdot_\infty)$, logo \omterm{def:b2:metric:compact}{compacta}
(\cref{thm:b2:metric:compactprops}~(2), válido para $\C^n \simeq
\R^{2n}$). A função \omterm{def:b2:metric:continuity}{contínua} $N$ atinge seu mínimo $c$ em
$S$; $c > 0$, pois $N$ só se anula em $0 \notin S$. A homogeneidade
espalha a cota: $N(x) \geq c \norm{x}_\infty$ para todo $x$.

Consequências: todos os enunciados se reduzem a $(K^n,
\norm\cdot_\infty)$, onde são conhecidos
(\cref{thm:b2:metric:rncomplete},
\cref{thm:b2:metric:compactprops}); uma aplicação linear $u$ a partir de
$E$ de dimensão finita satisfaz $\norm{u(x)} \leq \sum\abs{x_i}
\norm{u(e_i)} \leq C'\norm{x}_\infty$: cota (4) do
\cref{thm:b2:nvs:continuouslinear}.
\end{proof}

\begin{corollary}\label{cor:b2:nvs:closedsubspace}
Todo subespaço de dimensão finita de um espaço normado é fechado.
\end{corollary}

\begin{proof}
Ele é \omterm{def:b2:metric:complete}{completo} para a \omterm{def:b2:nvs:norm}{norma} induzida
(\cref{thm:b2:nvs:finitedim}), e as partes \omterm{def:b2:metric:complete}{completas} são fechadas
(\cref{def:b2:metric:complete}).
\end{proof}

\begin{example}[Uma melhor aproximação calculada por simetria]\label{ex:b2:nvs:bestaffine}
Em $\bigl(C(\intcc{-1}{1}), \norm\cdot_\infty\bigr)$, a que distância está
$f(x) = \abs x$ do subespaço (fechado, de dimensão dois) das
funções afins $a + bx$? Por simetria, substituir $a + bx$ por
$a - bx$ deixa $\norm{f - (a \pm bx)}_\infty$ inalterada, e
o ponto médio $a$ vai pelo menos tão bem (desigualdade triangular na
média): basta considerar as constantes. Para uma constante
$a$:
\[
\norm{\abs x - a}_\infty = \max\,(1 - a,\ a)
\geq \frac12 ,
\]
minimizada em $a = \frac12$: a distância é $\frac12$, atingida
pela constante $\frac12$. Note a curva de erro $\abs x -
\frac12$: ela atinge $\pm\frac12$ alternadamente em $x = -1, 0,
1$ --- três extremos de sinais alternados para uma melhor
aproximação por uma família de dois parâmetros. Esse padrão de
\emph{equioscilação} não é acaso; é a
assinatura da otimalidade que o problema de fim de semana deste capítulo
transforma no teorema de Chebyshev.
\end{example}

\begin{example}[Subespaços fechados contra subespaços densos]\label{ex:b2:nvs:closeddense}
Em $E = \bigl(C(\intcc{0}{1}), \norm\cdot_\infty\bigr)$: cada
$\R_n[X]$ (polinômios de grau $\leq n$, restritos a
$\intcc01$) é um subespaço de dimensão finita, logo \emph{fechado}
--- um limite uniforme de polinômios de grau $\leq n$ ainda é um deles. Mas
a união $\R[X]$ de todos eles é \emph{densa} em $E$ (o
teorema de aproximação de Weierstrass, demonstrado no
\cref{ch:b2:funcseq}), e subespaços próprios densos são o mais
não fechado que se pode imaginar. A moral: o fechamento dos subespaços é um
privilégio da dimensão finita; empilhar andares fechados pode construir um
arranha-céu denso.
\end{example}

\begin{theorem}[Riesz]\label{thm:b2:nvs:riesz}
A bola unitária fechada de um espaço normado $E$ é \omterm{def:b2:metric:compact}{compacta} \emph{se e
somente se} $\dim E < \infty$.\index{teorema de Riesz}
\end{theorem}

\begin{proof}
Dimensão finita: fechada e limitada basta
(\cref{thm:b2:nvs:finitedim}).

Reciprocamente, suponha $\dim E = \infty$. \emph{Lema de Riesz:} para
todo subespaço fechado próprio $F \subsetneq E$ e todo $\varepsilon \in
\intoo{0}{1}$, existe um vetor unitário $x$ com $d(x, F) \geq 1 -
\varepsilon$. Demonstração: escolha $y \notin F$, ponha $\delta = d(y, F) > 0$
($F$ fechado), escolha $f \in F$ com $\norm{y - f} \leq
\frac{\delta}{1 - \varepsilon}$ e ponha $x = \frac{y - f}{\norm{y -
f}}$: para todo $g \in F$,
\[
\norm{x - g} = \frac{\norm{y - (f + \norm{y-f}\,g)}}{\norm{y - f}}
\geq \frac{\delta}{\norm{y-f}} \geq 1 - \varepsilon ,
\]
sendo o numerador uma distância de $y$ a um ponto de $F$.

Construa agora vetores unitários $x_1, x_2, \dots$ por indução: $F_k =
\operatorname{Vect}(x_1, \dots, x_k)$ tem dimensão finita, logo é
fechado (\cref{cor:b2:nvs:closedsubspace}) e próprio; o lema de Riesz
com $\varepsilon = \frac12$ fornece um unitário $x_{k+1}$ com $d(x_{k+1},
F_k) \geq \frac12$. A sequência satisfaz $\norm{x_p - x_q} \geq
\frac12$ para $p \neq q$: nenhuma subsequência convergente --- a bola unitária
não é \omterm{def:b2:metric:compact}{compacta}.
\end{proof}

\begin{example}[Riesz como detector de dimensão]\label{ex:b2:nvs:rieszdetector}
$C(\intcc01)$ tem dimensão finita? Riesz responde sem
exibir nenhuma família livre infinita explícita: a sequência
$f_n(x) = x^n$ está na bola unitária fechada e satisfaz, para
$m > n$, $\norm{f_n - f_m}_\infty \geq f_n(x_0) - f_m(x_0) > 0$
em pontos adequados --- quantificado com limpeza no problema de fim de semana
deste capítulo (questão 16), onde uma subsequência permanece a
distância mútua $\geq \frac14$. Nenhuma subsequência convergente, logo
a bola não é \omterm{def:b2:metric:compact}{compacta}, logo $\dim C(\intcc01) = \infty$ pelo
\cref{thm:b2:nvs:riesz}. A \omterm{def:b2:metric:compact}{compacidade} da bola unitária é uma
dicotomia perfeita: vale em dimensão finita, falha em dimensão
infinita, sem meio-termo --- só a geometria já
lê o tipo de dimensão.
\end{example}

\section{Espaços de Banach}

\begin{definition}\label{def:b2:nvs:banach}
Um \emph{espaço de Banach}\index{espaço de Banach} é um espaço normado
\omterm{def:b2:metric:complete}{completo}. Exemplos: todo espaço normado de dimensão finita
(\cref{thm:b2:nvs:finitedim}); $\bigl(C(\intcc{a}{b}),
\norm\cdot_\infty\bigr)$ (\cref{thm:b2:metric:rncomplete});
$\mathcal{L}_c(E, F)$ com $F$ de Banach (mesmo padrão de demonstração que para as
funções \omterm{def:b2:metric:continuity}{contínuas}). Contraexemplo: $\bigl(C(\intcc{0}{1}),
\norm\cdot_1\bigr)$ (\cref{exo:b2:nvs:7}).
\end{definition}

\begin{example}[A norma de operador da integração]\label{ex:b2:nvs:integrationnorm}
Em $\bigl(C(\intcc01), \norm\cdot_\infty\bigr)$, seja $T(f)(x) =
\int_0^x f(t)\,\dd t$ (um endomorfismo: $T(f)$ é \omterm{def:b2:metric:continuity}{contínua}).
Rode o~\cref{met:b2:nvs:opnorm}. Cota superior:
\[
\abs{T(f)(x)} \leq \int_0^x\abs f \leq x\,\norm f_\infty \leq
\norm f_\infty ,
\]
logo $\vertiii T \leq 1$. Testemunha: $f \equiv 1$ dá $T(f)(x) =
x$ e $\norm{T(f)}_\infty = 1 = \norm f_\infty$: atingida,
$\vertiii T = 1$. Mas note que $\vertiii{T^2} = \frac12 \neq
\vertiii T^2$: de fato $T^2(f)(x) = \int_0^x(x - t)f(t)\dd t$ tem
$\abs{T^2(f)(x)} \leq \frac{x^2}2\norm f_\infty$, atingida de novo
em $f \equiv 1$; e, em geral, $\vertiii{T^n} = \frac1{n!}$ ---
a cota submultiplicativa $\vertiii T^n = 1$ erra por um
fatorial. Esse é exatamente o fenômeno que o truque do iterado do
problema de fim de semana do~\cref{ch:b2:metric} converte em
resolubilidade global de equações diferenciais lineares.
\end{example}

\begin{theorem}[Convergência absoluta em espaços de Banach]\label{thm:b2:nvs:absoluteconvergence}
Num \omterm{def:b2:nvs:banach}{espaço de Banach}, se $\sum \norm{u_n} < \infty$ então $\sum u_n$
converge, e $\norm{\sum u_n} \leq \sum\norm{u_n}$. (A teoria completa
das séries em espaços normados é o~\cref{ch:b2:series}.)
\end{theorem}

\begin{proof}
Somas parciais $S_N$: para $q > p$, $\norm{S_q - S_p} \leq
\sum_{n=p+1}^{q}\norm{u_n}$, que tende a $0$ (critério de Cauchy
para a série real das \omterm{def:b2:nvs:norm}{normas}): $(S_N)$ é de Cauchy, logo convergente.
A desigualdade passa ao limite a partir da desigualdade triangular
finita.
\end{proof}

\begin{example}[Exponencial de matriz, primeiro contato]\label{ex:b2:nvs:matrixexp}
A \emph{exponencial de matriz}\index{exponencial de matriz}:
$\mathcal{M}_n(K)$ com qualquer \omterm{def:b2:nvs:norm}{norma} submultiplicativa ($\vertiii{AB}
\leq \vertiii A \vertiii B$) é de Banach (dimensão finita). Então, para
toda $A$,
\[
\eu^A = \sum_{k=0}^{\infty} \frac{A^k}{k!}
\]
converge absolutamente ($\vertiii{A^k/k!} \leq \vertiii A^k /k!$,
somável): está bem definida. O~\cref{ch:b2:diffeq} a explora
sistematicamente.
\end{example}

\begin{example}[Uma série de Neumann que termina]\label{ex:b2:nvs:neumannfinite}
Para $A = \begin{psmallmatrix}0 & \frac12\\ 0 &
0\end{psmallmatrix}$: $\vertiii A < 1$ em qualquer \omterm{thm:b2:nvs:continuouslinear}{norma de operador}
construída sobre as \omterm{def:b2:nvs:norm}{normas} do~\cref{ex:b2:nvs:examples}, e $A^2 = 0$, de modo que
a série geométrica desaba:
\[
(I - A)^{-1} = \sum_{k \geq 0} A^k = I + A =
\begin{pmatrix}1 & \tfrac12\\ 0 & 1\end{pmatrix},
\]
verificado por $(I - A)(I + A) = I - A^2 = I$. A nilpotência
trunca a série exatamente como truncou a exponencial no
\cref{ch:b2:reduction}; e o exemplo calibra as
expectativas: a inversa de Neumann é uma série infinita em
geral, e um polinômio precisamente quando a perturbação é
nilpotente, sendo o erro após $N$ termos sempre majorado pela
cauda geométrica $\vertiii A^{N+1}/(1 - \vertiii A)$.
\end{example}

\begin{example}[A exponencial de um gerador de rotações]\label{ex:b2:nvs:rotationexp}
Tome $A = \begin{psmallmatrix}0 & -\theta\\ \theta &
0\end{psmallmatrix}$. Então $A^2 = -\theta^2 I$, de modo que as potências
ciclam com período quatro, e a série se separa em partes
pares e ímpares:
\[
\eu^{A} = \sum_{k}\frac{A^k}{k!}
= \Bigl(\sum_{j}\frac{(-1)^j\theta^{2j}}{(2j)!}\Bigr) I
+ \Bigl(\sum_{j}\frac{(-1)^j\theta^{2j+1}}{(2j+1)!}\Bigr)
\frac{A}{\theta}
= \begin{pmatrix}
\cos\theta & -\sin\theta\\
\sin\theta & \cos\theta
\end{pmatrix},
\]
sendo todos os rearranjos autorizados pela convergência absoluta. A
exponencial de um gerador antissimétrico é uma rotação ---
calculada aqui apenas a partir da série, três capítulos antes de a
equação diferencial $x' = Ax$ (\cref{ch:b2:diffeq}) explicar
\emph{por quê}: $\eu^{tA}$ é movimento circular uniforme. A lição
final: as identidades entre séries de matrizes demonstram-se exatamente como as
escalares, uma vez que uma \omterm{def:b2:nvs:norm}{norma} submultiplicativa certifique a convergência
absoluta.
\end{example}

\begin{example}[Norma do sup significa uniforme: o dicionário]\label{ex:b2:nvs:uniformdictionary}
A afirmação $\norm{f_n - f}_\infty \to 0$ \emph{é} a convergência
uniforme: um único número, $\sup_x\abs{f_n(x) - f(x)}$, majora
o erro em todos os pontos simultaneamente. O dicionário em
ação para $f_n(x) = x^n$ em $\intcc{0}{1}$: pontualmente, $f_n
\to 0$ em $\intco{0}{1}$ e $f_n(1) = 1$; na \omterm{def:b2:nvs:norm}{norma},
$\norm{f_n - 0}_\infty = 1 \not\to 0$ e, de fato, o limite pontual
é descontínuo, logo fora do alcance de um
limite em $\norm\cdot_\infty$ dentro de $C(\intcc01)$ (que é fechado
para limites uniformes, \cref{thm:b2:metric:rncomplete}). Em
$\intcc{0}{a}$, $a < 1$: $\norm{f_n}_\infty = a^n \to 0$ ---
a convergência uniforme é restaurada ao encolher o domínio. Todo
enunciado de convergência do~\cref{ch:b2:funcseq} é um enunciado
sobre essa única \omterm{def:b2:nvs:norm}{norma}; ter o dicionário em mente reduz aquele capítulo à
metade.
\end{example}

\begin{remark}[Armadilhas comuns]\label{rem:b2:nvs:pitfalls}
(i) Uma \omterm{thm:b2:nvs:continuouslinear}{norma de operador} depende \emph{das duas} \omterm{def:b2:nvs:norm}{normas} escolhidas: a
mesma matriz tem $\vertiii\cdot_\infty$ dada pelas somas por linha
(\cref{exo:b2:nvs:4}) e uma $\vertiii\cdot_1$ diferente (somas por
coluna); citar ``a'' \omterm{def:b2:nvs:norm}{norma} de uma matriz sem nomear as
\omterm{def:b2:nvs:norm}{normas} subjacentes não significa nada. (ii) $\vertiii{AB} \leq
\vertiii A\,\vertiii B$ é uma desigualdade, em geral estrita ---
as potências podem encolher muito mais depressa do que a cota $\vertiii A^k$
sugere, e é justamente esse o propósito das \omterm{def:b2:nvs:norm}{normas} adaptadas (problema
de fim de semana deste capítulo, questão 22). (iii) ``Linear implica
\omterm{def:b2:metric:continuity}{contínua}'' é um privilégio da dimensão finita: a derivação
sobre polinômios é linear e ilimitada
(\cref{ex:b2:nvs:operatornorms}). (iv) A convergência absoluta de
$\sum u_n$ só ajuda quando o espaço é \omterm{def:b2:metric:complete}{completo}
(o~\cref{exo:b2:series:9} constrói o contraexemplo). (v) Em
dimensão infinita, um supremo sobre a bola unitária é um supremo
genuíno: não suponha que ele seja atingido
(\cref{exo:b2:nvs:8}).
\end{remark}

\begin{remark}[Perspectivas dentro deste volume]\label{rem:b2:nvs:perspectives}
Três encontros ficam marcados. Com o~\cref{ch:b2:series}: num
\omterm{def:b2:nvs:banach}{espaço de Banach}, as séries absolutamente convergentes convergem, de modo que as
séries geométrica e exponencial de operadores tornam-se ferramentas
cotidianas --- inverter $I - A$, definir $\eu^{A}$
(\cref{ex:b2:series:neumann}). Com os
\cref{ch:b2:funcseq} e \cref{ch:b2:powerseries}: a convergência
de sequências de funções e de séries de potências é a convergência em
$\bigl(C, \norm\cdot_\infty\bigr)$
(\cref{ex:b2:nvs:uniformdictionary}), e o raio de
convergência é um enunciado sobre quais séries geométricas
dominam. Com o~\cref{ch:b2:fourier}: as \omterm{def:b2:nvs:norm}{normas}
$\norm\cdot_2$ e $\norm\cdot_\infty$ discordam de verdade em
$C(\intcc{0}{1})$ (\cref{ex:b2:nvs:sqrtnxn}), e é exatamente
por isso que a convergência em média quadrática das séries de Fourier e a convergência
uniforme são dois teoremas distintos com dois preços distintos.
\end{remark}

\begin{remark}[Onde este capítulo é usado]
As \omterm{thm:b2:nvs:continuouslinear}{normas de operador} e a série geométrica movem os argumentos de perturbação
do~\cref{ch:b2:diffcalc} (teorema da função inversa) e a
\omterm{ex:b2:nvs:matrixexp}{exponencial de matriz} do~\cref{ch:b2:diffeq}; a \omterm{thm:b2:nvs:finitedim}{equivalência das
normas} autoriza silenciosamente todo argumento do tipo ``escolha sua \omterm{def:b2:nvs:norm}{norma} preferida''
no~\cref{ch:b2:funcseq} e adiante; e a divisão finito/infinito
do teorema de Riesz --- tornada quantitativa no
problema de fim de semana deste capítulo --- é a razão pela qual o volume do terceiro
ano de graduação precisa de ferramentas novas (convergência fraca, Arzelà--Ascoli,
projeções em espaços de Hilbert) onde este volume ainda podia extrair
subsequências convergentes.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:nvs:1}
Em $\R^2$, desenhe as bolas unitárias de $\norm\cdot_1$, $\norm\cdot_2$,
$\norm\cdot_\infty$ e demonstre as desigualdades $\norm x_\infty
\leq \norm x_2 \leq \norm x_1 \leq 2\norm x_\infty$ com as melhores
constantes em dimensão $2$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:nvs:2}
$N(f) = \abs{f(0)} + \norm{f'}_\infty$ é uma \omterm{def:b2:nvs:norm}{norma} em
$C^1(\intcc{0}{1})$? Compare-a com $\norm{f}_\infty$: uma
desigualdade vale, a outra falha (exiba).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:nvs:3}
Calcule a \omterm{thm:b2:nvs:continuouslinear}{norma de operador} de $u(f) = \int_0^1 f(t)\,\eu^t\,\dd t$
em $\bigl(C(\intcc{0}{1}), \norm\cdot_\infty\bigr) \to \R$, e a do
deslocamento $S(x_1, x_2, \dots, x_n) = (x_2, \dots, x_n, 0)$ em
$(K^n, \norm\cdot_\infty)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:nvs:4}
Em $(\R^n, \norm\cdot_\infty)$, prove que a \omterm{thm:b2:nvs:continuouslinear}{norma de operador} de uma
matriz $A$ é $\vertiii A_\infty = \max_i \sum_j \abs{a_{ij}}$ (a
maior soma de valores absolutos por linha). Calcule-a para $\begin{pmatrix} 1 & -2\\
3 & 1\end{pmatrix}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:nvs:5}
Prove que $GL_n(K)$ é \omterm{def:b2:metric:topology}{aberto} em $\mathcal{M}_n(K)$ e que $A
\mapsto A^{-1}$ é \omterm{def:b2:metric:continuity}{contínua} nele. \emph{Sugestão: para a abertura, se
$\vertiii H < \frac{1}{\vertiii{A^{-1}}}$ então $A + H = A(I +
A^{-1}H)$ com $\vertiii{A^{-1}H} < 1$, e $I + B$ é invertível
para $\vertiii B < 1$ pela série geométrica
(\cref{thm:b2:nvs:absoluteconvergence}); para a \omterm{def:b2:metric:continuity}{continuidade}, majore
$(A+H)^{-1} - A^{-1}$ usando a mesma série.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:nvs:6}
Seja $\varphi$ uma forma linear num espaço normado $E$. Prove que
$\varphi$ é \omterm{def:b2:metric:continuity}{contínua} se e somente se $\ker\varphi$ é fechado.
\emph{(Se $\ker\varphi$ é fechado e $\varphi \neq 0$, escolha $a$
com $\varphi(a) = 1$ e $r > 0$ com $B(a, r) \cap \ker\varphi =
\emptyset$; deduza $\abs{\varphi(h)} \leq \frac{1}{r}\norm h$ por um
argumento de reescalamento em $a - \frac{h}{\varphi(h)}$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:nvs:7}
Prove que $\bigl(C(\intcc{0}{1}), \norm\cdot_1\bigr)$ não é
\omterm{def:b2:metric:complete}{completo}: mostre que as funções $f_n$, rampas afins de $0$ a
$1$ em $\bigl[\frac12 - \frac1n, \frac12\bigr]$ (valendo $0$
antes e $1$ depois), formam uma sequência de Cauchy sem limite
\omterm{def:b2:metric:continuity}{contínuo} em $\norm\cdot_1$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:nvs:8}
Em $E = C(\intcc{0}{1})$ com $\norm\cdot_\infty$, considere
\[
\varphi(f) = \sum_{n \geq 1} (-1)^n\, 2^{-n} f\bigl(\tfrac1n\bigr).
\]
Prove que $\varphi$ é uma forma linear \omterm{def:b2:metric:continuity}{contínua} bem definida com
$\vertiii\varphi = 1$, mas que o supremo que define
$\vertiii\varphi$ \emph{não é atingido} na bola unitária fechada.
\emph{(Cota superior: desigualdade triangular. \omterm{def:b2:nvs:norm}{Norma} $= 1$: construa
$f_K$ \omterm{def:b2:metric:continuity}{contínuas} com $\norm{f_K}_\infty \leq 1$ e $f_K(\frac1n)
= (-1)^n$ para $n \leq K$ --- os pontos $\frac1n$ estão isolados uns
dos outros. Não atingimento: a igualdade forçaria $f(\frac1n) =
(-1)^n$ para todo $n$, incompatível com a \omterm{def:b2:metric:continuity}{continuidade} de $f$ em $0$,
pois $\frac1n \to 0$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:nvs:9}
Seja $E$ um espaço normado no qual a bola unitária fechada é \omterm{def:b2:metric:compact}{compacta}.
Redemonstre, sem citar o~\cref{thm:b2:nvs:riesz}, que toda
sequência limitada tem subsequência convergente, e prove que toda
forma linear em $E$ é \omterm{def:b2:metric:continuity}{contínua} se e somente se $\dim E < \infty$.
\emph{(Para a dimensão infinita, construa uma forma descontínua
definindo-a livremente numa sequência normalizada linearmente independente
e estendendo --- admitindo a existência de um complemento
algébrico.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:nvs:10}
Em $C(\intcc{0}{1})$, prove $\norm f_1 \leq \norm f_2 \leq \norm
f_\infty$ \emph{(Cauchy--Schwarz para a primeira)} e mostre, com
a família $f_n(x) = x^n$, que nenhuma das desigualdades pode ser invertida
a menos de constante: as três \omterm{def:b2:nvs:norm}{normas} são duas a duas não \omterm{def:b2:nvs:equivalent}{equivalentes}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:nvs:11}
(Distância a um hiperplano) Seja $\varphi$ uma forma linear \omterm{def:b2:metric:continuity}{contínua}
não nula num espaço normado $E$. Prove que
\[
d\bigl(x, \ker\varphi\bigr) =
\frac{\abs{\varphi(x)}}{\vertiii\varphi}
\qquad (x \in E),
\]
e verifique em \cref{exo:b2:nvs:8} que o ínfimo não precisa ser
atingido por nenhum ponto do hiperplano.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:nvs:12}
Em $E = \R[X]$ (todos os polinômios), sejam $N_1(P) =
\sup_{\intcc{0}{1}}\abs P$ e $N_2(P) =
\sup_{\intcc{0}{2}}\abs P$. Mostre que $N_1 \leq N_2$ mas que
$N_1$ e $N_2$ \emph{não} são \omterm{def:b2:nvs:equivalent}{equivalentes}; deduza que a
identidade $(E, N_2) \to (E, N_1)$ é uma bijeção linear \omterm{def:b2:metric:continuity}{contínua}
cuja inversa é descontínua. Mostre por fim que $(E, N_1)$ não é
\omterm{def:b2:metric:complete}{completo} \emph{(somas parciais de Taylor de $\eu^x$)}. Os três
fenômenos são impossíveis em dimensão finita --- diga por quê.
\end{exercise}

\section{Problema: melhor aproximação e o teorema de Chebyshev}

Quão bem uma função pode ser aproximada por polinômios de um dado
grau, e qual polinômio faz isso melhor? Do lado da existência,
a resposta pertence a este capítulo: a \omterm{def:b2:metric:compact}{compacidade} em dimensão
finita faz as melhores aproximações existirem. Do lado explícito,
um caso não trivial pode ser resolvido inteiramente na unha ---
entre todos os polinômios \emph{mônicos} de grau $n$, o de
menor \omterm{def:b2:nvs:norm}{norma} do sup em $\intcc{-1}{1}$ é o polinômio de Chebyshev
(normalizado), de \omterm{def:b2:nvs:norm}{norma} $2^{1-n}$: o \emph{teorema extremal de
Chebyshev}. O problema demonstra os dois lados e depois mede quão mal
a \omterm{def:b2:metric:compact}{compacidade} falha em dimensão infinita: a bola unitária de
$C(\intcc{0}{1})$ contém constelações infinitas de pontos a
distância mútua $1$.

\begin{problem}[{Problema de fim de semana --- o teorema extremal de Chebyshev
e a geometria da bola unitária}]
\label{pb:b2:nvs:1}
As \omterm{def:b2:nvs:norm}{normas} sem índice são \omterm{def:b2:nvs:norm}{normas} do sup no segmento indicado.

\medskip
\noindent\textbf{Parte I --- Melhor aproximação em espaços
normados.}
\begin{enumerate}
  \item Seja $F$ um subespaço de dimensão finita de um espaço
        normado $E$ e $x \in E$. Prove que a distância $d(x,
        F) = \inf_{f \in F}\norm{x - f}$ é \emph{atingida}
        \emph{(reduza a uma parte fechada e limitada de $F$ e use
        o~\cref{thm:b2:nvs:finitedim})}.
  \item Uma \omterm{def:b2:nvs:norm}{norma} é \emph{estritamente convexa} quando $\norm u = \norm v
        = 1$ e $u \neq v$ implicam $\bigl\Vert\frac{u +
        v}2\bigr\Vert < 1$. Mostre que $\norm\cdot_2$ em $\R^n$
        é estritamente convexa \emph{(identidade do paralelogramo)} e
        que $\norm\cdot_1$ e $\norm\cdot_\infty$ não o são para
        $n \geq 2$.
  \item Prove que, para uma \omterm{def:b2:nvs:norm}{norma} estritamente convexa, a melhor
        aproximação da questão 1 é \emph{única}.
  \item Em $(\R^2, \norm\cdot_\infty)$, calcule todas as melhores
        aproximações de $x = (0, 1)$ pela reta $F =
        \operatorname{Vect}\bigl((1,0)\bigr)$: um intervalo de
        minimizantes.
  \item Em $\bigl(C(\intcc{a}{b}), \norm\cdot_\infty\bigr)$,
        mostre que a melhor aproximação de $f$ por
        \emph{constantes} é única, igual a $c^* = \frac{\max f
        + \min f}{2}$, com distância $\frac{\max f - \min
        f}{2}$; calcule as duas para $f(x) = x^2$ em
        $\intcc{0}{1}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Polinômios de Chebyshev.}
\begin{enumerate}[resume]
  \item Mostre que existe exatamente um polinômio $T_n$ com
        $T_n(\cos\theta) = \cos n\theta$ para todo $\theta$
        \emph{(recorrência $T_{n+1} = 2XT_n - T_{n-1}$ a partir da
        fórmula de adição do cosseno)}, que $\deg T_n = n$, e que
        seu coeficiente dominante é $2^{n-1}$ para $n \geq 1$.
  \item Mostre que $\abs{T_n} \leq 1$ em $\intcc{-1}{1}$, com
        $T_n(\eta_k) = (-1)^k$ nos $n + 1$ pontos $\eta_k =
        \cos\frac{k\pi}{n}$ ($k = 0, \dots, n$), e que as
        raízes de $T_n$ são os $n$ pontos
        $\cos\frac{(2k-1)\pi}{2n}$, que se intercalam com os $\eta_k$.
  \item Calcule $T_2, T_3, T_4$ e verifique a alternância de
        $T_3$ em $\eta_0, \dots, \eta_3 = 1, \frac12, -\frac12,
        -1$ por avaliação direta.
  \item Para $\abs x \geq 1$, prove
        \[
        T_n(x) = \frac{\bigl(x + \sqrt{x^2 - 1}\bigr)^n +
        \bigl(x - \sqrt{x^2 - 1}\bigr)^n}{2},
        \]
        e deduza $T_n(x) \sim \frac12\bigl(x + \sqrt{x^2 -
        1}\bigr)^n \to \infty$ geometricamente para $x > 1$ fixo.
  \item Prove a lei de composição $T_m \circ T_n = T_{mn}$
        \emph{(verifique em $\intcc{-1}{1}$ e invoque a rigidez
        dos polinômios)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O teorema extremal de Chebyshev.}
Escreva $Q_n = 2^{1-n}T_n$ (mônico, pela questão 6).
\begin{enumerate}[resume]
  \item Seja $P$ mônico de grau $n \geq 1$ com
        $\sup_{\intcc{-1}{1}}\abs P < 2^{1-n}$. Avaliando $D
        = Q_n - P$ nos pontos $\eta_k$ e contando as trocas
        de sinal, obtenha uma contradição. Conclua:
        \[
        \sup_{\intcc{-1}{1}}\abs P \;\geq\; 2^{1-n}
        \qquad\text{para todo } P \text{ mônico de grau } n.
        \]
  \item (Caso de igualdade) Suponha $\sup_{\intcc{-1}{1}}\abs P =
        2^{1-n}$ com $P$ mônico de grau $n$, e ponha $D = Q_n
        - P \neq 0$. Mostre que $(-1)^kD(\eta_k) \geq 0$ para todo $k$;
        mostre que cada um dos $n$ intervalos
        $\intcc{\eta_{k}}{\eta_{k-1}}$ contém um zero de $D$,
        e que um zero comum a dois intervalos consecutivos é
        um ponto \emph{interior} $\eta_k$ no qual também $D' = 0$.
        Conclua que $D$ tem $n$ zeros contados com
        multiplicidade, logo $D = 0$: o minimizante é exatamente
        $Q_n$ --- o \emph{teorema extremal de Chebyshev}.
  \item Reformule o teorema como uma distância: em $\intcc{-1}{1}$,
        \[
        d_\infty\bigl(X^n,\ \R_{n-1}[X]\bigr) = 2^{1-n},
        \]
        com melhor aproximação única $X^n - Q_n$; e mostre, pela
        substituição afim $x = \frac{1+t}2$, que em
        $\intcc{0}{1}$ a distância se torna $2^{1-2n}$.
  \item (Nós de interpolação ótimos) Para $n$ nós $x_1, \dots,
        x_n \in \intcc{-1}{1}$, o polinômio nodal
        $\omega(x) = \prod_i(x - x_i)$ é mônico de grau $n$.
        Deduza da questão 12 qual escolha de nós minimiza
        $\sup_{\intcc{-1}{1}}\abs\omega$, o fator dependente dos nós
        na cota clássica do erro de interpolação, e
        dê o valor mínimo.
  \item Verifique à mão o caso $n = 2$ do teorema (ache
        $\inf_c \sup_{\intcc{-1}{1}}\abs{x^2 - c}$ diretamente)
        e calcule numericamente a distância da questão 13 em
        $\intcc{0}{1}$ para $n = 10$. O que o tamanho dela diz
        sobre o gráfico de $x^{10}$?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A bola unitária de
$C(\intcc{0}{1})$.}
\begin{enumerate}[resume]
  \item Seja $g_k(x) = x^{2^k}$. Mostre que $\norm{g_k}_\infty = 1$ e
        $\norm{g_k - g_j}_\infty \geq \frac14$ para $j > k$
        \emph{(avalie no ponto em que $x^{2^k} =
        \frac12$)}: uma sequência limitada explícita sem
        subsequência convergente --- a bola unitária fechada não é
        \omterm{def:b2:metric:compact}{compacta}, na unha.
  \item (Lema de Riesz, refinado) Seja $F$ um subespaço próprio
        \emph{de dimensão finita} de um espaço normado
        $E$. Usando a questão 1, produza um vetor unitário $x$
        com $d(x, F) = 1$ exatamente --- e não apenas $\geq 1 -
        \varepsilon$ como no lema do~\cref{thm:b2:nvs:riesz}.
  \item Deduza: em todo espaço normado de dimensão infinita existe
        uma sequência de vetores unitários com distâncias mútuas
        $\geq 1$, e redemonstre com isso o teorema de Riesz.
  \item Em $C(\intcc{0}{1})$, exiba explicitamente uma tal constelação:
        as funções tenda $h_n$ com suporte em
        $\bigl[\frac1{n+1}, \frac1n\bigr]$ e valor de pico $1$.
        Verifique $\norm{h_n} = 1$, $\norm{h_n - h_m} = 1$ para $n
        \neq m$, e note que $h_n \to 0$ pontualmente mas não
        uniformemente.
  \item (A precompacidade falha) Mostre que a bola unitária fechada
        de $C(\intcc{0}{1})$ não pode ser coberta por um número finito de
        bolas de raio $\frac13$ \emph{(cada uma dessas bolas contém
        no máximo um $h_n$)} --- em contraste com a
        etapa de precompacidade na demonstração do
        \cref{thm:b2:metric:borellebesgue}.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- \omterm{def:b2:nvs:norm}{Normas} em ação sobre matrizes, e
síntese.}
\begin{enumerate}[resume]
  \item Prove que todo \omterm{def:b2:reduction:eigen}{autovalor} $\lambda$ de $A \in
        \mathcal{M}_n(\C)$ satisfaz $\abs\lambda \leq
        \vertiii A$ para toda \omterm{thm:b2:nvs:continuouslinear}{norma de operador}; aplique
        o~\cref{exo:b2:nvs:4} para majorar os \omterm{def:b2:reduction:eigen}{autovalores} de
        $\begin{psmallmatrix}1 & -2\\ 3 & 1\end{psmallmatrix}$
        e compare com o módulo verdadeiro deles.
  \item (\omterm{def:b2:nvs:norm}{Normas} adaptadas) Seja $A$ \omterm{def:b2:reduction:diag}{diagonalizável}, $A =
        P\,\mathrm{diag}(\lambda_1, \dots, \lambda_n)\,P^{-1}$.
        Mostre que $N_P(x) = \norm{P^{-1}x}_\infty$ é uma \omterm{def:b2:nvs:norm}{norma} cuja
        \omterm{thm:b2:nvs:continuouslinear}{norma de operador} satisfaz $\vertiii A_{N_P} =
        \max_i\abs{\lambda_i}$.
  \item Deduza: para $A$ \omterm{def:b2:reduction:diag}{diagonalizável}, $A^k \to 0$ se e somente
        se todos os \omterm{def:b2:reduction:eigen}{autovalores} satisfazem $\abs{\lambda_i} < 1$ ---
        a \omterm{thm:b2:nvs:finitedim}{equivalência das normas} torna a conclusão
        independente da \omterm{def:b2:nvs:norm}{norma}. Verifique em $A =
        \frac14\begin{psmallmatrix}1 & 2\\ 2 &
        1\end{psmallmatrix}$.
  \item (As constantes de equivalência explodem) Em $\R_n[X]$, compare
        $N_c(P) = \max_k \abs{a_k}$ (coeficientes) e
        $\norm{P}_{\intcc{0}{1}}$: ambas são \omterm{def:b2:nvs:norm}{normas}, logo são
        \omterm{def:b2:nvs:equivalent}{equivalentes} para cada $n$ fixo; mas mostre, usando o
        minimizante mônico da questão 13 em $\intcc{0}{1}$, que a
        melhor constante $C_n$ em $N_c \leq
        C_n\norm\cdot_{\intcc{0}{1}}$ satisfaz $C_n \geq
        2^{2n-1}$. Conclua em uma frase por que ``todas as \omterm{def:b2:nvs:norm}{normas} são
        \omterm{def:b2:nvs:equivalent}{equivalentes}'' morre em dimensão infinita.
  \item (Síntese) Uma frase para cada: onde a \omterm{def:b2:metric:compact}{compacidade} das
        bolas de dimensão finita trabalhou (questões 1 e 12); o que
        a convexidade estrita governa; o que a constelação das
        questões 18--19 destrói; e como a questão 24
        quantifica o fracasso. Nomeie o cume (o teorema
        extremal de Chebyshev) e diga onde a melhor aproximação
        encontra sua casa moderna (o teorema da projeção em espaços de
        Hilbert, volume do terceiro ano de graduação, onde a \omterm{def:b2:metric:complete}{completude} substitui
        a \omterm{def:b2:metric:compact}{compacidade}).
\end{enumerate}
\end{problem}
