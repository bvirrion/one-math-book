\chapter{Sequências e séries de funções}\label{ch:b2:funcseq}

Quando funções convergem para uma função, quais propriedades sobrevivem à
passagem ao limite? A \omterm{def:b2:funcseq:def}{convergência pontual} preserva quase
nada; a convergência \emph{uniforme} --- convergência na \omterm{def:b2:nvs:norm}{norma} do sup
--- preserva a \omterm{def:b2:metric:continuity}{continuidade}, as integrais em segmentos e, com uma reviravolta,
as derivadas. Este capítulo demonstra os três teoremas de transferência,
suas versões para séries, e os coroa com o teorema de aproximação
de Weierstrass, demonstrado pelos belíssimos polinômios probabilísticos
de Bernstein.

\section{Convergência pontual e convergência uniforme}

\begin{definition}\label{def:b2:funcseq:def}
Sejam $f_n, f \colon X \to \R$ (ou $\C$, ou um espaço normado) e $X$ um
conjunto qualquer. $(f_n)$ converge para $f$
\emph{pontualmente}\index{convergência pontual} quando $f_n(x) \to
f(x)$ para todo $x$; \emph{uniformemente}\index{convergência uniforme}
quando
\[
\norm{f_n - f}_\infty = \sup_{x \in X}\, \abs{f_n(x) - f(x)}
\xrightarrow[n \to \infty]{} 0 .
\]
Uniforme implica pontual; em $C(\intcc{a}{b})$, a convergência uniforme
é exatamente a convergência no \omterm{def:b2:nvs:banach}{espaço de Banach}
$\bigl(C(\intcc{a}{b}), \norm\cdot_\infty\bigr)$ de
\cref{ch:b2:nvs}.
\end{definition}

\begin{example}\label{ex:b2:funcseq:xn}
Em $\intcc{0}{1}$, $f_n(x) = x^n$ converge \omterm{def:b2:funcseq:def}{pontualmente} para o limite
\emph{descontínuo} $f = \mathbf{1}_{\{1\}}$; a convergência
não é uniforme: $\norm{f_n - f}_\infty \geq f_n\bigl(1 -
\tfrac1n\bigr) = (1 - \tfrac1n)^n \to \eu^{-1} \neq 0$. Em
$\intcc{0}{a}$ com $a < 1$ ela \emph{é} uniforme ($\sup = a^n \to
0$): a uniformidade é uma propriedade do domínio tanto quanto da
sequência.
\end{example}

\begin{example}[Dois limites que se recusam a comutar]\label{ex:b2:funcseq:doublelimit}
O capítulo inteiro trata de trocar limites, então eis o
menor fracasso possível. Seja $a_{n,m} = \dfrac{n}{n+m}$ para
$n, m \geq 1$. Então
\[
\lim_{m\to\infty}\Bigl(\lim_{n\to\infty}a_{n,m}\Bigr)
= \lim_{m\to\infty} 1 = 1,
\qquad
\lim_{n\to\infty}\Bigl(\lim_{m\to\infty}a_{n,m}\Bigr)
= \lim_{n\to\infty} 0 = 0 :
\]
os dois limites iterados existem e diferem. Todo teorema de
transferência deste capítulo é uma licença para comutar dois limites ---
$\lim_n$ com $\lim_{x\to a}$ (\omterm{def:b2:metric:continuity}{continuidade}), com $\int$
(integração), com $\frac{\dd}{\dd x}$ (derivação) ---
e a \omterm{def:b2:funcseq:def}{convergência uniforme} é precisamente a taxa que torna a
comutação legal. Lição final: sempre que uma ``demonstração''
troca silenciosamente duas operações de limite, este quadro de duas linhas é
o contraexemplo a levantar contra ela; as corcovas deslizantes do
\cref{exo:b2:funcseq:2} são o mesmo fenômeno vestido de
sinal de integral.
\end{example}

\section{Os três teoremas de transferência}

\begin{theorem}[Continuidade]\label{thm:b2:funcseq:continuity}
Se cada $f_n$ é \omterm{def:b2:metric:continuity}{contínua} em $a$ e $f_n \to f$ \omterm{def:b2:funcseq:def}{uniformemente} numa
vizinhança de $a$, então $f$ é \omterm{def:b2:metric:continuity}{contínua} em $a$. Um limite uniforme
de funções \omterm{def:b2:metric:continuity}{contínuas} é \omterm{def:b2:metric:continuity}{contínuo}.
\end{theorem}

\begin{proof}
O argumento dos $3\varepsilon$ já usado no
\cref{thm:b2:metric:rncomplete}: escolha $n$ com $\norm{f_n -
f}_\infty \leq \varepsilon$ e depois $\delta$ pela \omterm{def:b2:metric:continuity}{continuidade} de
$f_n$ em $a$; para $\abs{x - a} \leq \delta$,
\[
\abs{f(x) - f(a)} \leq \abs{f(x) - f_n(x)} + \abs{f_n(x) - f_n(a)}
+ \abs{f_n(a) - f(a)} \leq 3\varepsilon . \qedhere
\]
\end{proof}

\begin{example}[A uniformidade falha exatamente onde o limite quebra]\label{ex:b2:funcseq:threshold}
Em $\intcc{0}{2}$, seja $f_n(x) = \dfrac{x^n}{1 + x^n}$. O
limite \omterm{def:b2:funcseq:def}{pontual} é uma função de três peças:
\[
f(x) = \begin{cases} 0 & 0 \leq x < 1,\\[2pt]
\tfrac12 & x = 1,\\[2pt]
1 & 1 < x \leq 2, \end{cases}
\]
descontínua em $1$, de modo que, pelo~\cref{thm:b2:funcseq:continuity},
a convergência não pode ser uniforme em $\intcc{0}{2}$. Nas
peças fechadas que evitam o limiar ela é: para $0 \leq x \leq
a < 1$,
\[
\sup_{\intcc{0}{a}}\abs{f_n - 0}
= \frac{a^n}{1 + a^n} \leq a^n \to 0 ,
\]
e para $1 < b \leq x \leq 2$,
\[
\sup_{\intcc{b}{2}}\abs{f_n - 1}
= \frac{1}{1 + b^n} \leq b^{-n} \to 0 ,
\]
os dois supremos calculados pela monotonicidade de $u \mapsto
\frac{u}{1+u}$ e de $x \mapsto x^n$. Lição final: o
fracasso da uniformidade está localizado na descontinuidade do
limite --- a mesma geometria do~\cref{ex:b2:funcseq:xn}, e a
razão pela qual a disciplina do ``uniforme em todo segmento interior''
volta o capítulo inteiro.
\end{example}

\begin{theorem}[Integração num segmento]\label{thm:b2:funcseq:integration}
Se $f_n \to f$ uniformemente em $\intcc{a}{b}$, com $f_n$ \omterm{def:b2:metric:continuity}{contínuas} por
partes com \omterm{def:b2:metric:continuity}{continuidade} por partes ($f$ idem), então
\[
\int_a^b f_n \longrightarrow \int_a^b f .
\]
\end{theorem}

\begin{proof}
A linearidade e a desigualdade triangular para integrais dão
\[
\Bigl|\int_a^b f_n - \int_a^b f\Bigr|
= \Bigl|\int_a^b (f_n - f)\Bigr|
\leq \int_a^b\abs{f_n - f}
\leq (b - a)\,\norm{f_n - f}_\infty \longrightarrow 0 .
\]
O fator de comprimento $(b - a)$ é onde entra a \omterm{def:b2:metric:compact}{compacidade} do segmento:
em intervalos não \omterm{def:b2:metric:compact}{compactos} a mesma estimativa produz a
cota inútil $\infty\cdot0$, e a conclusão realmente
falha sem dominação --- as corcovas achatadas $f_n =
\frac1n\mathbf 1_{\intcc{0}{n}}$ convergem \omterm{def:b2:funcseq:def}{uniformemente} para $0$ em
$\intco{0}{\infty}$ e no entanto mantêm $\int f_n = 1$ (observação sobre armadilhas
adiante), e as corcovas deslizantes da observação do
\cref{ch:b2:integration} fazem o mesmo com a \omterm{def:b2:funcseq:def}{convergência pontual}; a \omterm{def:b2:funcseq:def}{convergência
uniforme} controla alturas, nunca larguras.
\end{proof}

\begin{theorem}[Derivação]\label{thm:b2:funcseq:differentiation}
Sejam $f_n$ de classe $C^1$ num intervalo $I$, com: $(f_n')$ convergindo
\emph{\omterm{def:b2:funcseq:def}{uniformemente}} em $I$ (ou em todo segmento de $I$) para alguma $g$,
e $(f_n(x_0))$ convergindo em um ponto $x_0$. Então $(f_n)$
converge (uniformemente em segmentos) para uma função $f$ de classe $C^1$, e $f'
= g$: pode-se derivar o limite.
\end{theorem}

\begin{proof}
Defina $f(x) = \lim f_n(x_0) + \int_{x_0}^x g$: legítimo, sendo $g$
\omterm{def:b2:metric:continuity}{contínua} --- de fato $g$ é o limite \emph{uniforme} em
segmentos das \omterm{def:b2:metric:continuity}{contínuas} $f_n'$, de modo que
o~\cref{thm:b2:funcseq:continuity} se aplica, e a integral de
uma função \omterm{def:b2:metric:continuity}{contínua} está bem definida com, pelo teorema
fundamental do cálculo,
\[
f'(x) = g(x) \qquad (x \in I) :
\]
o candidato a limite é $C^1$ com a derivada certa
\emph{por construção}, antes de qualquer convergência ser demonstrada. Pelo
teorema fundamental de novo, $f_n(x) = f_n(x_0) + \int_{x_0}^x
f_n'$; subtraindo,
\[
\abs{f_n(x) - f(x)} \leq \abs{f_n(x_0) - \lim f_n(x_0)}
+ \abs{x - x_0}\,\norm{f_n' - g}_{\infty} ,
\]
que tende a $0$ uniformemente em todo segmento. E $f$ é $C^1$
com $f' = g$ por construção.
\end{proof}

\begin{example}[Por que a hipótese recai sobre as derivadas]\label{ex:b2:funcseq:sqrtabs}
Seja $F_n(x) = \sqrt{x^2 + \frac1n}$ em $\R$. Cada $F_n$ é
$C^1$ (aliás $C^\infty$), e a convergência para $\abs x$ é
uniforme em todo o $\R$:
\[
0 \leq F_n(x) - \abs x
= \frac{(x^2 + \frac1n) - x^2}{\sqrt{x^2+\frac1n} + \abs x}
= \frac{1/n}{\sqrt{x^2 + \frac1n} + \abs x}
\leq \frac{1/n}{1/\sqrt n} = \frac{1}{\sqrt n} .
\]
E no entanto o limite $\abs x$ não é derivável em $0$: a \omterm{def:b2:funcseq:def}{convergência
uniforme} das \emph{funções}, por mais rápida que seja, não transfere
derivabilidade alguma. O fracasso é visível nas derivadas:
\[
F_n'(x) = \frac{x}{\sqrt{x^2 + \frac1n}}
\longrightarrow \begin{cases} 1 & x > 0,\\ 0 & x = 0,\\
-1 & x < 0, \end{cases}
\]
um limite \omterm{def:b2:funcseq:def}{pontual} descontínuo, de modo que $(F_n')$ não pode convergir
\omterm{def:b2:funcseq:def}{uniformemente} perto de $0$ (\cref{thm:b2:funcseq:continuity} de novo).
Lição final:
o~\cref{thm:b2:funcseq:differentiation} supõe deliberadamente
a \omterm{def:b2:funcseq:def}{convergência uniforme} das $f_n'$, e não das $f_n$ ---
este exemplo é a razão.
\end{example}

\section{Séries de funções}

\begin{definition}\label{def:b2:funcseq:series}
Uma série de funções $\sum u_n$ converge \omterm{def:b2:funcseq:def}{pontualmente}/\omterm{def:b2:funcseq:def}{uniformemente} quando
suas somas parciais convergem. Ela converge
\emph{normalmente}\index{convergência normal} (em $X$) quando $\sum
\norm{u_n}_\infty < \infty$. A convergência normal implica a \omterm{def:b2:funcseq:def}{convergência
uniforme} (no \omterm{def:b2:nvs:banach}{espaço de Banach} das funções limitadas:
\cref{thm:b2:nvs:absoluteconvergence}), que implica a \omterm{def:b2:funcseq:def}{pontual};
as duas implicações são estritas.
\end{definition}

\begin{example}[Uma série, três veredictos]\label{ex:b2:funcseq:threemodes}
Tome $u_n(x) = \frac{x^n}{n}$ em $\intco{0}{1}$. \emph{\omterm{def:b2:funcseq:def}{Pontualmente}:}
converge para todo $x \in \intco01$ (comparação com a
série geométrica). \emph{\omterm{def:b2:funcseq:series}{Normalmente} em $\intcc{0}{a}$, $a < 1$:}
$\norm{u_n}_{\infty,\intcc0a} = \frac{a^n}{n}$, \omterm{def:b2:series:summable}{somável}.
\emph{Não \omterm{def:b2:funcseq:series}{normalmente} em $\intco{0}{1}$:}
$\norm{u_n}_{\infty,\intco01} = \frac1n$, e $\sum\frac1n$
diverge. \emph{Nem sequer uniformemente em $\intco{0}{1}$:} o
resto resiste perto de $1$,
\[
R_N(x) = \sum_{n>N}\frac{x^n}{n}
\geq \sum_{n=N+1}^{2N}\frac{x^n}{n}
\geq \frac{N\,x^{2N}}{2N} = \frac{x^{2N}}{2}
\xrightarrow[x\to1^-]{} \frac12 ,
\]
logo $\sup_{\intco01}\abs{R_N} \geq \frac12$ para todo $N$.
Lição final: os quatro veredictos convivem em paz --- a
soma $-\ln(1-x)$ é \omterm{def:b2:metric:continuity}{contínua} em $\intco{0}{1}$ porque a
\omterm{def:b2:metric:continuity}{continuidade} só precisa de uniformidade \emph{perto de cada ponto}, isto é,
nos segmentos $\intcc0a$; explodir na borda é direito da
soma.
\end{example}

\begin{theorem}[Transferência para séries]\label{thm:b2:funcseq:seriestransfer}
Se $\sum u_n$ converge \omterm{def:b2:funcseq:def}{uniformemente} (por exemplo, \omterm{def:b2:funcseq:series}{normalmente}) no
conjunto relevante: a \omterm{def:b2:metric:continuity}{continuidade} de todas as $u_n$ em $a$ passa à soma; a integração
num segmento pode ser feita termo a termo; e se $\sum u_n(x_0)$
converge enquanto $\sum u_n'$ converge uniformemente em segmentos, a
soma é $C^1$ com derivada $\sum u_n'$.
\end{theorem}

\begin{proof}
Tudo é o teorema correspondente aplicado às somas
parciais $S_N = \sum_{n\leq N}u_n$, que são somas finitas de
funções com a regularidade relevante. \omterm{def:b2:metric:continuity}{Continuidade}: cada $S_N$
é \omterm{def:b2:metric:continuity}{contínua} em $a$ e $S_N \to \sum u_n$ \omterm{def:b2:funcseq:def}{uniformemente}:
\cref{thm:b2:funcseq:continuity}. Integração: no segmento,
\[
\int_a^b \sum_{n\geq0} u_n
= \lim_N \int_a^b S_N
= \lim_N \sum_{n=0}^{N}\int_a^b u_n
= \sum_{n\geq0}\int_a^b u_n
\]
pelo~\cref{thm:b2:funcseq:integration} (primeira igualdade) e pela
linearidade da integral (segunda). Derivação: as
$S_N$ são $C^1$, $S_N(x_0)$ converge e $S_N' =
\sum_{n\leq N}u_n'$ converge uniformemente em segmentos:
o~\cref{thm:b2:funcseq:differentiation} dá que a soma é
$C^1$ com derivada $\lim S_N' = \sum u_n'$.
\end{proof}

\begin{remark}[Armadilhas comuns]
Quatro armadilhas, todas vistas em provas. \emph{(i)
Supremos verificados pela metade:} avaliar $f_n$ ao longo de uma sequência
$x_n$ bem escolhida só minora $\norm{f_n - f}_\infty$ por
\emph{baixo} --- suficiente para refutar a uniformidade (como no
\cref{ex:b2:funcseq:xn}), nunca para demonstrá-la; para demonstrá-la, majore
o sup por um cálculo válido para \emph{todo} $x$. \emph{(ii)
Uniformidade no conjunto errado:} a \omterm{def:b2:funcseq:series}{convergência normal} ou uniforme
costuma valer em todo $\intcc{-a}{a}$ ou $\intco\delta\infty$
mas falhar na união \omterm{def:b2:metric:topology}{aberta}; isso não é obstáculo --- a \omterm{def:b2:metric:continuity}{continuidade}
e a derivabilidade são locais, de modo que a disciplina segmento a segmento
do~\cref{ex:b2:funcseq:zeta} as fornece em todo o
\omterm{def:b2:metric:topology}{aberto}. \emph{(iii) Integrar fora de segmentos:} o
\cref{thm:b2:funcseq:integration} é um enunciado sobre
segmentos; em $\intco0\infty$, a \omterm{def:b2:funcseq:def}{convergência uniforme} não
impede a massa de escapar para o infinito ($f_n =
\frac1n\mathbf 1_{\intcc{0}{n}}$ converge \omterm{def:b2:funcseq:def}{uniformemente} para $0$,
com $\int f_n = 1$) --- use a convergência dominada ali.
\emph{(iv) Derivar o limite:}
\cref{ex:b2:funcseq:sqrtabs}; a hipótese sobre derivadas recai sobre
$(f_n')$, e nenhuma velocidade de convergência de $(f_n)$ pode
substituí-la.
\end{remark}

\begin{example}[A função $\zeta$ de Riemann]\label{ex:b2:funcseq:zeta}
$\zeta(s) = \sum_{n\geq1} n^{-s}$ converge \omterm{def:b2:funcseq:series}{normalmente} em toda
semirreta $\intco{a}{+\infty}$, $a > 1$ ($\norm{n^{-s}}_\infty =
n^{-a}$, \omterm{def:b2:series:summable}{somável}): $\zeta$ é \omterm{def:b2:metric:continuity}{contínua} em $\intoo{1}{+\infty}$;
derivando termo a termo (a série derivada $\sum -\ln n\; n^{-s}$
também converge \omterm{def:b2:funcseq:series}{normalmente} em $\intco{a}{\infty}$), $\zeta$ é $C^1$
--- e, iterando, $C^\infty$ --- com $\zeta'(s) = -\sum \frac{\ln
n}{n^s}$. Note a disciplina: a \omterm{def:b2:funcseq:series}{convergência normal} é verificada em
\emph{sub}-semirretas, nunca no \omterm{def:b2:metric:topology}{aberto} $\intoo{1}{\infty}$
propriamente dito, onde ela falha.
\end{example}

\begin{example}[Uma série logarítmica, trabalhada até o fim]\label{ex:b2:funcseq:logseries}
Seja $F(x) = \sum_{n\geq1} \frac{\eu^{-nx}}{n}$ em
$\intoo{0}{\infty}$. Cada termo é majorado em $\intco{\delta}
\infty$ por $\frac{\eu^{-n\delta}}{n} \leq \eu^{-n\delta}$, uma
série geométrica convergente: \omterm{def:b2:funcseq:series}{convergência normal} em todo
$\intco\delta\infty$, logo $F$ é \omterm{def:b2:metric:continuity}{contínua} em
$\intoo{0}{\infty}$. A série derivada $\sum -\eu^{-nx}$ é
igualmente \omterm{def:b2:funcseq:series}{normalmente} convergente em $\intco\delta\infty$
($\norm{\eu^{-nx}}_{\infty,\intco\delta\infty} =
\eu^{-n\delta}$), de modo que $F$ é $C^1$ com derivada geométrica:
\[
F'(x) = -\sum_{n\geq1}\eu^{-nx}
= \frac{-\eu^{-x}}{1 - \eu^{-x}}
= \frac{-1}{\eu^{x} - 1} .
\]
Iterando, $F$ é $C^\infty$. Integrando $F'$ (tanto $F$ quanto $x
\mapsto -\ln(1 - \eu^{-x})$ se anulam em $+\infty$ e têm a
mesma derivada em $\intoo0\infty$):
\[
F(x) = -\ln\bigl(1 - \eu^{-x}\bigr),
\]
a série logarítmica em $t = \eu^{-x}$. Lição final: quando $x
\to 0^+$, $F(x) = -\ln(x + O(x^2)) = \ln\frac1x + O(x)$ --- a
série diverge logaritmicamente na fronteira, exatamente como a
série harmônica em que ela se transforma em $x = 0$; a \omterm{def:b2:funcseq:series}{convergência normal} em
$\intco\delta\infty$ mas não em $\intoo0\infty$ é o
sintoma.
\end{example}

\begin{method}[Demonstrar ou refutar a convergência uniforme]\label{met:b2:funcseq:uniform}
Para $f_n \to f$ \omterm{def:b2:funcseq:def}{pontualmente} em $X$:
\begin{enumerate}
  \item Calcule ou majore $\norm{f_n - f}_\infty$: estude a
        função $x \mapsto \abs{f_n(x) - f(x)}$ (derivada,
        monotonicidade) para localizar seu máximo; uma cota válida para
        todo $x$ que tenda a $0$ demonstra a uniformidade.
  \item Para \emph{refutar}: exiba pontos $x_n$ com
        $\abs{f_n(x_n) - f(x_n)} \not\to 0$ (muitas vezes $x_n$ acompanha
        a corcova móvel, como no~\cref{exo:b2:funcseq:1}); ou
        invoque um teorema de transferência na contrapositiva --- um
        limite descontínuo de funções \omterm{def:b2:metric:continuity}{contínuas}
        (\cref{ex:b2:funcseq:threshold}), ou $\int f_n
        \not\to \int f$ num segmento.
  \item Para séries, tente primeiro a \omterm{def:b2:funcseq:series}{convergência normal}
        ($\sum\sup\abs{u_n} < \infty$); se ela falhar globalmente,
        teste-a nos subsegmentos que importam
        (\cref{ex:b2:funcseq:threemodes}); se falhar
        em toda parte, a \omterm{def:b2:funcseq:def}{convergência uniforme} ainda pode valer pela
        cota do resto alternado
        (\cref{exo:b2:funcseq:4}) ou por somação por partes.
\end{enumerate}
\end{method}

\section{O teorema de aproximação de Weierstrass}

\begin{theorem}[Weierstrass, via Bernstein]\label{thm:b2:funcseq:weierstrass}
Toda $f \colon \intcc{0}{1} \to \R$ \omterm{def:b2:metric:continuity}{contínua} é limite uniforme
de polinômios --- explicitamente, de seus \emph{polinômios de
Bernstein}\index{polinômios de Bernstein}\index{teorema de aproximação
de Weierstrass}
\[
B_n(f)(x) = \sum_{k=0}^{n} f\Bigl(\frac kn\Bigr)\binom nk x^k
(1-x)^{n-k} .
\]
\end{theorem}

\begin{proof}
Fixe $x \in \intcc{0}{1}$ e ponha $p_k(x) = \binom nk x^k(1 -
x)^{n-k}$. Três identidades binomiais, obtidas avaliando
$(x + y)^n$ e suas duas derivadas em $x$ em $y = 1 - x$:
\[
\sum_k p_k = 1,
\qquad
\sum_k k\,p_k = nx,
\qquad
\sum_k k(k-1) p_k = n(n-1)x^2 .
\]
Em detalhe: $(x+y)^n = \sum_k\binom nk x^ky^{n-k}$ em $y = 1-x$
é a primeira; derivando em $x$,
\[
n(x+y)^{n-1} = \sum_k k\binom nk x^{k-1}y^{n-k} ,
\]
depois multiplicando por $x$ e pondo $y = 1 - x$ obtém-se a
segunda; derivando duas vezes e multiplicando por $x^2$ obtém-se
a terceira. Desenvolvendo $(k - nx)^2 = k(k-1) + k(1 - 2nx) +
n^2x^2$ e combinando as três:
\[
\sum_k (k - nx)^2 p_k
= n(n-1)x^2 + nx(1 - 2nx) + n^2x^2
= nx(1 - x) \leq \frac n4 ,
\]
a \emph{identidade da variância}.

Agora estime, usando $\sum p_k = 1$:
\[
\abs{B_n(f)(x) - f(x)}
\leq \sum_{k} \Bigl| f\Bigl(\frac kn\Bigr) - f(x)\Bigr|\, p_k(x)
= \Sigma_{\text{perto}} + \Sigma_{\text{longe}} ,
\]
separando conforme $\abs{\frac kn - x} \leq \delta$ ou não.
Dado $\varepsilon > 0$, a \omterm{def:b2:metric:continuity}{continuidade} uniforme de $f$ (Heine) fornece
$\delta$ com $\Sigma_{\text{perto}} \leq \varepsilon$. Para a soma
distante, com $M = \norm f_\infty$: pela identidade da variância e pelo
truque de contagem de Chebyshev,
\[
\Sigma_{\text{longe}} \leq 2M \sum_{\abs{k - nx} > n\delta} p_k
\leq 2M\,\frac{\sum_k (k - nx)^2 p_k}{n^2\delta^2}
\leq \frac{2M}{4 n \delta^2} \cdot 1
= \frac{M}{2n\delta^2}
\xrightarrow[n\to\infty]{} 0 ,
\]
uniformemente em $x$. Logo $\norm{B_n(f) - f}_\infty \leq \varepsilon +
\frac{M}{2n\delta^2} \leq 2\varepsilon$ para $n$ grande.
\end{proof}

\begin{remark}
Por substituição afim o teorema vale em qualquer segmento
$\intcc{a}{b}$. Ele falha em $\R$ (um limite uniforme de polinômios em
$\R$ é um polinômio: \cref{exo:b2:funcseq:8}). A leitura
probabilística --- $B_n(f)(x)$ é o valor esperado de $f$ numa
média binomial, e a cota da variância é a desigualdade de Chebyshev --- é
tornada honesta no~\cref{ch:b2:genfun}.
\end{remark}

\begin{remark}[Onde isso é usado]
A aproximação de Weierstrass é o teorema de densidade da análise
clássica: ele torna $C(\intcc ab)$ separável, permite verificar
identidades integrais apenas em polinômios (problemas de momentos) e
sustenta a versão trigonométrica demonstrada no capítulo de Fourier
pelo núcleo de Fejér. O problema de fim de semana deste
capítulo extrai o conteúdo quantitativo da demonstração de Bernstein
--- taxas de convergência governadas pelo módulo de \omterm{def:b2:metric:continuity}{continuidade} ---
e depois isola o que realmente a fez funcionar, no teorema de
Korovkin: positividade mais três funções de teste. O volume do terceiro ano de
graduação generaliza o enunciado de densidade a subálgebras
arbitrárias (Stone--Weierstrass) e a espaços \omterm{def:b2:metric:compact}{compactos}.
\end{remark}

\begin{example}[Aproximação poligonal, com uma taxa]\label{ex:b2:funcseq:polygonal}
Para $f$ $L$-lipschitziana em $\intcc{0}{1}$, seja $I_nf$ a
interpolante afim por partes nos nós $\frac kn$. Numa
célula $\intcc{\frac kn}{\frac{k+1}n}$, tanto $f(x)$ quanto
$I_nf(x)$ ficam entre os valores extremos que uma função
$L$-lipschitziana pode assumir dados os dois valores nodais, de modo que, para $x$ na
célula, escrevendo $x_k = \frac kn$:
\[
\abs{I_nf(x) - f(x)}
\leq \abs{I_nf(x) - f(x_k)} + \abs{f(x_k) - f(x)}
\leq L\,\abs{x - x_k} + L\,\abs{x - x_k}
\leq \frac{2L}{n}
\]
(a interpolante é ela própria $L$-lipschitziana na célula: sua
inclinação é um quociente de diferenças de $f$). Logo $\norm{I_nf -
f}_\infty \leq \frac{2L}{n}$: a aproximação poligonal de
funções \omterm{def:b2:metric:continuity}{lipschitzianas} converge com velocidade $\frac1n$ ---
\emph{mais rápido} que o $\frac{1}{\sqrt n}$ de Bernstein para a
mesma classe (problema de fim de semana, Parte II). Lição final: o
polígono interpola mas não é suave, Bernstein é suave
mas lento; não há almoço grátis entre a regularidade do
aproximante e a velocidade --- um compromisso precisado pelos
resultados de saturação do problema de fim de semana.
\end{example}

\begin{remark}[Perspectivas dentro deste volume]
A \omterm{def:b2:funcseq:def}{convergência uniforme} é o cavalo de batalha deste livro daqui em diante. O
capítulo de séries de potências roda inteiramente sobre a \omterm{def:b2:funcseq:series}{convergência normal} em
subdiscos \omterm{def:b2:metric:compact}{compactos} --- todo teorema termo a termo lá é um
caso particular dos teoremas de transferência deste capítulo. O capítulo de Fourier
mora um andar acima: suas somas parciais $S_N$ falham
exatamente onde este capítulo avisa que poderiam (\omterm{def:b2:funcseq:def}{pontualmente} mas não
\omterm{def:b2:funcseq:def}{uniformemente} nos saltos), e suas médias de Fejér têm êxito pela mesma
mecânica dos $3\varepsilon$ que demonstrou o
\cref{thm:b2:funcseq:continuity}. O capítulo de equações
diferenciais define $\eu^{tA}$ por uma série \omterm{def:b2:funcseq:series}{normalmente} convergente e
a deriva termo a termo --- literalmente
o~\cref{thm:b2:funcseq:seriestransfer} aplicado às entradas
das matrizes. Quando, mais adiante no livro, restar dúvida sobre ``por que
podemos fazer isso'', a resposta costuma ser um teorema deste capítulo.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:funcseq:1}
Estude a \omterm{def:b2:funcseq:def}{convergência pontual} e a \omterm{def:b2:funcseq:def}{convergência uniforme} em $\intcc{0}{1}$, e depois
em $\intcc{0}{a}$ ($a < 1$) ou $\intco{\delta}{1}$ conforme o caso, de:
\[
f_n(x) = \frac{x}{1 + nx},
\qquad
g_n(x) = n x\,\eu^{-n x^2},
\qquad
h_n(x) = x^n(1 - x^n).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:funcseq:2}
Prove que $\displaystyle\int_0^1 g_n \not\to \int_0^1 \lim g_n$
para as $g_n$ do~\cref{exo:b2:funcseq:1}, e concilie com o
\cref{thm:b2:funcseq:integration}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:funcseq:3}
Prove que $S(x) = \sum_{n\geq1} \dfrac{x^n}{n^2}$ é \omterm{def:b2:metric:continuity}{contínua} em
$\intcc{-1}{1}$ e que $S$ é $C^1$ em $\intoo{-1}{1}$ com
$S'(x) = -\frac{\ln(1-x)}{x}$ para $0 < \abs x < 1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:funcseq:4}
Seja $F(x) = \sum_{n \geq 0} \dfrac{(-1)^n}{n + x}$ em
$\intoo{0}{+\infty}$. Prove a \omterm{def:b2:funcseq:def}{convergência uniforme} (não normal) em
$\intco{\delta}{\infty}$ pela cota do resto de série alternada,
a \omterm{def:b2:metric:continuity}{continuidade} e a equação funcional $F(x) + F(x + 1) =
\frac1x$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:funcseq:5}
(Dini) Sejam $f_n \colon K \to \R$ \omterm{def:b2:metric:continuity}{contínuas} num \emph{\omterm{def:b2:metric:compact}{espaço métrico
compacto}}, com $f_n \to f$ \emph{\omterm{def:b2:funcseq:def}{pontualmente}}, $f$ \omterm{def:b2:metric:continuity}{contínua}
e $(f_n(x))$ \emph{decrescente} em $n$ para cada $x$. Prove que
a convergência é uniforme. \emph{(Dado $\varepsilon$, os \omterm{def:b2:metric:topology}{abertos}
$U_n = \{x : f_n(x) - f(x) < \varepsilon\}$ crescem e cobrem
$K$; extraia uma subcobertura finita ---
\cref{thm:b2:metric:borellebesgue}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:funcseq:6}
Prove que $\displaystyle\lim_{n\to\infty} \int_0^1
\frac{n\,f(x)}{1 + n^2x^2}\,\dd x = \frac{\pi}{2} f(0)$ para toda
$f$ \omterm{def:b2:metric:continuity}{contínua} em $\intcc{0}{1}$. \emph{(Substitua $u = nx$; separe
$f(0)$; domine.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:funcseq:7}
Calcule explicitamente os \omterm{thm:b2:funcseq:weierstrass}{polinômios de Bernstein} de $f(x) = x^2$ e
verifique o erro uniforme $\norm{B_n f - f}_\infty =
O\bigl(\frac1n\bigr)$ previsto pela demonstração do
\cref{thm:b2:funcseq:weierstrass} --- aqui exatamente $\frac{x(1 -
x)}{n}$ em cada ponto.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:funcseq:8}
Prove que, se polinômios $P_n$ convergem \omterm{def:b2:funcseq:def}{uniformemente} \emph{em todo
o $\R$} para $f$, então $f$ é um polinômio. \emph{(Para $m, n$ grande,
$P_n - P_m$ é um polinômio limitado em $\R$, logo constante; assim a
sequência se estabiliza a menos de constantes.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:funcseq:9}
(Uma função \omterm{def:b2:metric:continuity}{contínua} em nenhum ponto derivável --- guiado) Seja
$\varphi$ a distância ao inteiro mais próximo ($1$-periódica,
$\norm{\varphi}_\infty = \frac12$, $1$-lipschitziana) e
\[
W(x) = \sum_{n=0}^{\infty} \Bigl(\frac{3}{4}\Bigr)^{\!n}
\varphi(4^n x) .
\]
Prove: (a) $W$ é \omterm{def:b2:metric:continuity}{contínua} em $\R$ (\omterm{def:b2:funcseq:series}{convergência normal}); (b) para
todo $x$ e todo $m$, escolhendo $h_m = \pm\frac12\cdot 4^{-m}$
com o sinal que torna $\varphi$ afim no segmento de $4^m x$
a $4^m(x + h_m)$, o quociente de diferenças satisfaz
\[
\Bigl|\frac{W(x + h_m) - W(x)}{h_m}\Bigr| \geq 3^m -
\sum_{n<m} 3^n \geq \frac{3^m + 1}{2} \xrightarrow[m\to\infty]{}
\infty
\]
(os termos $n > m$ se anulam por periodicidade; o termo $n = m$ contribui
exatamente com $3^m$; os termos $n < m$ são majorados pela propriedade \omterm{def:b2:metric:continuity}{lipschitziana}).
Conclua que $W$ não é derivável em ponto algum.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:funcseq:10}
Seja $u_n(x) = (-1)^n x^n(1 - x)$ em $\intcc{0}{1}$. Mostre que
$\sum u_n$ converge \omterm{def:b2:funcseq:def}{pontualmente} em $\intcc{0}{1}$ e calcule sua
soma; mostre que a convergência é uniforme em $\intcc{0}{1}$
\emph{(majore o resto $R_N(x) = \sum_{n > N} u_n(x)$, uma
cauda geométrica, por seu primeiro termo e maximize $x^{N+1}(1-x)$)}
mas \emph{não} normal \emph{(calcule $\norm{u_n}_\infty$)}:
a \omterm{def:b2:funcseq:def}{convergência uniforme} é estritamente mais fraca que a \omterm{def:b2:funcseq:series}{convergência normal}.
Contraste com $\sum x^n(1-x)$, cuja soma é descontínua em
$1$: ali até a uniformidade falha.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:funcseq:11}
Seja $f_n \to f$ \omterm{def:b2:funcseq:def}{uniformemente} num \omterm{def:b2:metric:def}{espaço métrico} $X$, cada $f_n$
\omterm{def:b2:metric:continuity}{contínua}, e seja $x_n \to x$ em $X$. Prove que $f_n(x_n) \to
f(x)$. Mostre, por um exemplo em $X = \intcc{0}{1}$, que a \omterm{def:b2:funcseq:def}{convergência
pontual} não basta, mesmo com $f$ \omterm{def:b2:metric:continuity}{contínua}
\emph{(use as corcovas $g_n$ do~\cref{exo:b2:funcseq:1} e $x_n =
\frac{1}{\sqrt{2n}}$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:funcseq:12}
(Uma equação integral de Volterra por séries) Para $f \in
C(\intcc{0}{1})$ defina $Tf(x) = \int_0^x f(t)\,\dd t$.
\begin{enumerate}
  \item Mostre por indução que, para $n \geq 1$:
        \[
        T^n f(x) = \int_0^x \frac{(x -
        t)^{n-1}}{(n-1)!}\,f(t)\,\dd t,
        \qquad
        \norm{T^n f}_\infty \leq \frac{\norm f_\infty}{n!} .
        \]
  \item Deduza que $S = \sum_{n\geq0} T^n f$ converge \omterm{def:b2:funcseq:series}{normalmente}
        em $\intcc{0}{1}$ e resolve a equação integral $S = f
        + TS$.
  \item Verifique que $S(x) = f(x) + \int_0^x \eu^{x-t}f(t)\,\dd
        t$ resolve a mesma equação e demonstre a unicidade das
        soluções \omterm{def:b2:metric:continuity}{contínuas} \emph{(se $S = TS$ então
        $\norm{S}_\infty \leq \norm{T^nS}_\infty \to 0$)}:
        conclua a forma fechada da soma.
\end{enumerate}
\end{exercise}

\section{Problema: taxas de aproximação e o teorema de Korovkin}

\begin{problem}\label{pb:b2:funcseq:1}
A demonstração de Bernstein do~\cref{thm:b2:funcseq:weierstrass} esconde
dois tesouros. Primeiro, ela é \emph{quantitativa}: quão rápido $B_nf
\to f$ é governado pelo módulo de \omterm{def:b2:metric:continuity}{continuidade} de $f$, com a
taxa ótima atingida por $\abs{x - \frac12}$. Segundo, ela é
\emph{estrutural}: tudo o que importou foi que $B_n$ é um
operador linear positivo que se comporta bem em $1$, $x$, $x^2$ ---
essa observação, isolada, é o \emph{teorema de
Korovkin}\index{teorema de Korovkin}. Este problema demonstra os dois
e fecha com a assintótica exata de Voronovskaya. Em todo o problema,
$f \in C(\intcc{0}{1})$, $M = \norm f_\infty$, $p_k(x) = \binom
nk x^k(1-x)^{n-k}$ e $e_j$ denota $x \mapsto x^j$.

\medskip
\noindent\textbf{Parte I --- O operador de Bernstein.}
\begin{enumerate}
  \item Mostre que $B_n$ é linear, \emph{positivo} ($f \geq 0
        \Rightarrow B_nf \geq 0$), logo monótono ($f \leq g
        \Rightarrow B_nf \leq B_ng$), com
        $\norm{B_nf}_\infty \leq \norm f_\infty$, e que
        $B_nf$ interpola $f$ nas duas extremidades.
  \item Redemonstre as identidades $B_n e_0 = e_0$, $B_n e_1 =
        e_1$ e $B_n e_2 = e_2 + \frac{e_1 - e_2}{n}$
        \emph{(derive $(x + y)^n$ duas vezes e ponha $y = 1 -
        x$)}.
  \item Deduza a identidade da variância $\sum_k \bigl(\frac kn -
        x\bigr)^2 p_k(x) = \frac{x(1-x)}{n}$ e, por
        Cauchy--Schwarz, a cota do primeiro momento
        \[
        \sum_{k=0}^{n}\Bigl|\frac kn - x\Bigr|\,p_k(x)
        \leq \sqrt{\frac{x(1-x)}{n}} \leq \frac{1}{2\sqrt n} .
        \]
  \item Mostre que, se $f$ é convexa, então $B_nf \geq f$ em
        $\intcc{0}{1}$ \emph{(desigualdade de Jensen finita para os
        pesos $p_k(x)$)}.
  \item (A cota de contagem de Chebyshev, reformulada) Para $\delta > 0$
        mostre que
        \[
        \sum_{\abs{k/n - x} > \delta} p_k(x)
        \leq \frac{x(1-x)}{n\delta^2}
        \leq \frac{1}{4n\delta^2} ,
        \]
        e dê a leitura probabilística: $B_nf(x)$ promedia
        $f$ sobre uma média amostral binomial que se concentra em $x$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Taxas: o módulo de \omterm{def:b2:metric:continuity}{continuidade}.}
Para $\delta > 0$ ponha $\omega(\delta) = \sup\{\abs{f(s) - f(t)}
: s, t \in \intcc{0}{1},\ \abs{s - t} \leq \delta\}$.
\begin{enumerate}[resume]
  \item Mostre: $\omega$ é finito, não decrescente,
        $\omega(\delta) \to 0$ quando $\delta \to 0^+$ (Heine),
        subaditivo ($\omega(\delta_1 + \delta_2) \leq
        \omega(\delta_1) + \omega(\delta_2)$) e
        $\omega(\lambda\delta) \leq (1 +
        \lambda)\,\omega(\delta)$ para todos $\lambda > 0$.
  \item Demonstre a estimativa mestra, para todo $\delta > 0$:
        \[
        \abs{B_nf(x) - f(x)}
        \leq \sum_k \omega\Bigl(\Bigl|\frac kn -
        x\Bigr|\Bigr)p_k(x)
        \leq \Bigl(1 + \frac1\delta\sum_k\Bigl|\frac kn -
        x\Bigr|p_k(x)\Bigr)\,\omega(\delta) .
        \]
  \item Escolha $\delta = n^{-1/2}$ e conclua o
        \emph{teorema de Weierstrass quantitativo}:
        \[
        \norm{B_nf - f}_\infty \leq
        \frac32\,\omega\Bigl(\frac{1}{\sqrt n}\Bigr)
        \xrightarrow[n\to\infty]{} 0 .
        \]
  \item Deduza as taxas: $\norm{B_nf - f}_\infty \leq
        \frac{3L}{2\sqrt n}$ para $f$ $L$-lipschitziana, e $\leq
        \frac32 C n^{-\alpha/2}$ para $f$ $\alpha$-hölderiana
        ($\abs{f(s) - f(t)} \leq C\abs{s-t}^\alpha$).
  \item (O exemplo ótimo --- uma identidade binomial) Para $m
        \geq 1$ prove
        \[
        \sum_{k=m+1}^{2m} (k - m)\binom{2m}{k}
        = \frac{m}{2}\binom{2m}{m},
        \qquad\text{logo}\qquad
        \sum_{k=0}^{2m}\abs{k - m}\binom{2m}{k}
        = m\binom{2m}{m}
        \]
        \emph{(use $k\binom{2m}k = 2m\binom{2m-1}{k-1}$ e a
        simetria da linha binomial, que dá
        $\sum_{j=m}^{2m-1}\binom{2m-1}{j} = 2^{2m-2}$)}.
  \item Para $f(t) = \abs{t - \frac12}$ deduza o valor exato
        e sua assintótica (binomial central,
        \cref{ex:b2:comparison:centralbinomial}):
        \[
        B_{2m}f\Bigl(\frac12\Bigr) - f\Bigl(\frac12\Bigr)
        = \frac{\binom{2m}{m}}{2\cdot4^{m}}
        \;\sim\; \frac{1}{2\sqrt{\pi m}} :
        \]
        a taxa $\omega(n^{-1/2})$ da questão 8 é atingida
        (a menos de uma constante) --- para $f$ apenas \omterm{def:b2:metric:continuity}{contínua},
        o $n^{-1/2}$ de Bernstein é honesto.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O teorema de Korovkin.} Seja $(L_n)$
uma sequência de operadores \emph{lineares positivos} de
$C(\intcc{0}{1})$ em si mesmo tal que $L_ne_j \to e_j$
\omterm{def:b2:funcseq:def}{uniformemente} para $j = 0, 1, 2$.
\begin{enumerate}[resume]
  \item Mostre que um $L$ linear positivo é monótono e
        satisfaz $\abs{Lf} \leq L\abs f$ \omterm{def:b2:funcseq:def}{pontualmente}.
  \item Mostre: para todo $\varepsilon > 0$ existe $\delta > 0$
        tal que, para \emph{todos} $s, x \in \intcc{0}{1}$:
        \[
        \abs{f(s) - f(x)} \leq \varepsilon +
        \frac{2M}{\delta^2}(s - x)^2
        \]
        \emph{(trate $\abs{s - x} \leq \delta$ por Heine e
        $\abs{s-x} > \delta$ pela cota bruta $2M$)}.
  \item Fixe $x$, aplique $L_n$ à desigualdade da questão 13
        na variável $s$ e deduza
        \[
        \abs{L_nf(x) - f(x)\,L_ne_0(x)}
        \leq \varepsilon\,L_ne_0(x) + \frac{2M}{\delta^2}
        \bigl(L_ne_2(x) - 2x\,L_ne_1(x) + x^2 L_ne_0(x)\bigr).
        \]
  \item Mostre que $\sup_x \bigl(L_ne_2(x) - 2x\,L_ne_1(x) + x^2
        L_ne_0(x)\bigr) \to 0$ e depois monte
        o \emph{teorema de Korovkin}: $L_nf \to f$ \omterm{def:b2:funcseq:def}{uniformemente} para
        \emph{toda} $f \in C(\intcc{0}{1})$.
  \item Verifique que $(B_n)$ satisfaz as hipóteses de Korovkin:
        Weierstrass pela terceira vez, a partir de três monômios.
  \item Seja $I_n$ o operador de interpolação afim por partes
        nos nós $\frac kn$. Mostre que $I_n$ é
        linear positivo, $I_ne_0 = e_0$, $I_ne_1 = e_1$, e
        $\norm{I_ne_2 - e_2}_\infty = \frac{1}{4n^2}$
        \emph{(em cada célula o erro da interpolação afim
        de $t^2$ é $(t - a)(b - t)$)}. Conclua por Korovkin:
        as interpolantes poligonais convergem \omterm{def:b2:funcseq:def}{uniformemente} para toda
        $f$ \omterm{def:b2:metric:continuity}{contínua}.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Dividendos: densidade, momentos,
derivadas.}
\begin{enumerate}[resume]
  \item Mostre que os polinômios com coeficientes \emph{racionais}
        são densos em $\bigl(C(\intcc{0}{1}),
        \norm\cdot_\infty\bigr)$: esse \omterm{def:b2:nvs:banach}{espaço de Banach} é
        separável.
  \item (Os momentos determinam a função) Seja $f \in
        C(\intcc{0}{1})$ com $\int_0^1 f(t)\,t^n \dd t = 0$
        para todo $n \in \N$. Mostre que $\int_0^1 f P = 0$ para todo
        polinômio, depois que $\int_0^1 f^2 = 0$ e por fim que $f = 0$.
  \item Demonstre a identidade da derivada
        \[
        (B_nf)'(x) = n\sum_{k=0}^{n-1}\Bigl(
        f\Bigl(\frac{k+1}{n}\Bigr) -
        f\Bigl(\frac kn\Bigr)\Bigr)\,
        \binom{n-1}{k}x^k(1-x)^{n-1-k}
        \]
        \emph{(derive $p_k$ e reindexe --- uma \omterm{thm:b2:series:abel}{transformação
        de Abel})}.
  \item Suponha $f$ de classe $C^1$. Usando o teorema do valor médio em
        cada incremento e comparando com $B_{n-1}(f')$, mostre que
        $(B_nf)' \to f'$ uniformemente em $\intcc{0}{1}$. Deduza:
        para $f \in C^1$ existem polinômios que convergem para $f$
        \emph{junto com} suas derivadas.
  \item Suponha $f$ de classe $C^2$. Por Taylor--Lagrange em $x$ mostre que
        \[
        \abs{B_nf(x) - f(x)} \leq
        \frac{\norm{f''}_\infty}{2}\cdot\frac{x(1-x)}{n}
        \leq \frac{\norm{f''}_\infty}{8n} :
        \]
        a suavidade eleva a taxa de $n^{-1/2}$ para
        $n^{-1}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Saturação: o teorema de
Voronovskaya.}
\begin{enumerate}[resume]
  \item Demonstre a identidade do quarto momento
        \[
        \sum_k (k - nx)^4 p_k(x)
        = nx(1-x)\bigl(1 + 3(n-2)x(1-x)\bigr) \leq n^2
        \quad (n \geq 1)
        \]
        \emph{(desenvolva $k^4$ em fatoriais descendentes $k(k-1)\cdots$
        e use o truque de derivação da questão 2 mais duas
        vezes)}.
  \item (Voronovskaya) Seja $f$ de classe $C^2$ e $x \in
        \intcc{0}{1}$. Escrevendo $f(t) = f(x) + f'(x)(t-x) +
        \frac{f''(x)}2(t-x)^2 + \eta(t)(t-x)^2$ com $\eta$
        limitada e $\eta(t) \to 0$ quando $t \to x$, prove que
        \[
        n\bigl(B_nf(x) - f(x)\bigr)
        \xrightarrow[n\to\infty]{}
        \frac{x(1-x)}{2}\,f''(x)
        \]
        \emph{(separe a soma em $\eta$ em $\abs{t - x} \leq
        \delta$; controle a parte distante com a questão 23)}. Assim
        o erro da questão 22 é exato em ordem \emph{e} em
        constante: $B_n$ \emph{satura} em $\frac1n$, por mais
        suave que $f$ seja --- compare com o
        \cref{exo:b2:funcseq:7}.
  \item Síntese. Uma frase para cada: (i) o que só a positividade
        comprou (Partes I e III); (ii) onde a \omterm{def:b2:metric:compact}{compacidade}
        de $\intcc{0}{1}$ entrou em todas as partes; (iii) por que três
        funções de teste bastam no teorema de Korovkin; (iv) o
        compromisso que Bernstein faz ($n^{-1/2}$ robusto para $f$
        rugosa, mas um teto de $\frac1n$ para $f$ suave), e qual
        capítulo deste livro jogará o mesmo jogo com
        polinômios trigonométricos.
\end{enumerate}
\end{problem}
