\chapter{Sequências e séries}\label{ch:b2:series}

A teoria das séries numéricas (volume do primeiro ano de graduação) amadurece aqui em três
direções: séries com valores em \omterm{def:b2:nvs:banach}{espaços de Banach}, onde a convergência
absoluta faz o trabalho; os critérios mais finos para séries reais (\omterm{thm:b2:series:abel}{transformação
de Abel}); e as \emph{\omterm{def:b2:series:summable}{famílias somáveis}} --- a somação libertada
da ordem dos termos --- com o teorema de Fubini para somas duplas
e o \omterm{thm:b2:series:fubini}{produto de Cauchy}. Essas ferramentas sustentam todos os capítulos de séries de
funções que virão.

\section{Séries em espaços normados}

\begin{definition}\label{def:b2:series:def}
Para uma sequência $(u_n)$ num espaço normado $E$, a série $\sum u_n$
converge quando suas somas parciais convergem; ela converge
\emph{absolutamente}\index{convergência absoluta!num espaço de Banach} quando
$\sum \norm{u_n} < \infty$. Num espaço de \emph{Banach}, a convergência
absoluta implica a convergência (\cref{thm:b2:nvs:absoluteconvergence});
num espaço não \omterm{def:b2:metric:complete}{completo} isso pode falhar
(\cref{exo:b2:series:9}).
\end{definition}

\begin{example}\label{ex:b2:series:neumann}
Em $\mathcal{M}_n(K)$ (ou $\mathcal{L}_c(E)$, com $E$ de Banach): para
$\vertiii A < 1$, a \emph{série de Neumann} $\sum A^k$ converge
\omterm{def:b2:series:def}{absolutamente} para $(I - A)^{-1}$ (demonstrado no~\cref{exo:b2:nvs:5});
$\sum \frac{A^k}{k!}$ converge \omterm{def:b2:series:def}{absolutamente} para $\eu^A$ para toda $A$
(\cref{ex:b2:nvs:matrixexp}). As séries geométrica e exponencial com valores em operadores
comportam-se como seus modelos escalares --- exatamente o
propósito do quadro de Banach.
\end{example}

\section{Transformação de Abel}

\begin{theorem}[Transformação de Abel e critério de Abel]\label{thm:b2:series:abel}
(Somação por partes) Para escalares $a_n$ e vetores $b_n$, com $B_n
= \sum_{k=0}^{n} b_k$:\index{transformação de Abel}
\[
\sum_{n=0}^{N} a_n b_n
= a_N B_N - \sum_{n=0}^{N-1} (a_{n+1} - a_n) B_n .
\]
(Critério de Abel) Se $(a_n)$ é uma sequência real decrescente para $0$ e
as somas parciais $B_n$ são \emph{limitadas} (num \omterm{def:b2:nvs:banach}{espaço de Banach}), então
$\sum a_n b_n$ converge.
\end{theorem}

\begin{proof}
A identidade, passo a passo: com $B_{-1} = 0$, escreva $b_n = B_n -
B_{n-1}$ e separe,
\[
\sum_{n=0}^{N} a_nb_n
= \sum_{n=0}^{N} a_nB_n - \sum_{n=0}^{N} a_nB_{n-1}
= \sum_{n=0}^{N} a_nB_n - \sum_{n=0}^{N-1} a_{n+1}B_{n} ,
\]
reindexando a segunda soma por $n \mapsto n + 1$ (o termo $B_{-1}$
se anula); reunindo o intervalo comum $0 \leq n \leq N-1$
sobram $a_NB_N$ mais $\sum_{n\leq N-1}(a_n - a_{n+1})B_n$: a
fórmula enunciada. É a integração por partes discreta, com
$(B_n)$ como primitiva de $(b_n)$ e a diferença $a_{n+1}
- a_n$ como derivada de $(a_n)$. Para o critério, com $\norm{B_n}
\leq M$: o
termo de bordo $a_N B_N \to 0$; a série $\sum (a_n -
a_{n+1})B_n$ converge \omterm{def:b2:series:def}{absolutamente}, pois
\[
\sum_n \norm{(a_{n+1} - a_n)B_n} \leq M \sum_n (a_n - a_{n+1})
= M a_0 < \infty
\]
(telescopagem, $a_n \downarrow 0$). As duas peças da identidade
convergem, logo $\sum a_n b_n$ também converge.
\end{proof}

\begin{example}\label{ex:b2:series:sinn}
$\sum \frac{\sin n}{n}$ converge: $a_n = \frac1n \downarrow 0$ e
$B_n = \sum_{k=1}^{n} \sin k$ é limitada --- de fato $B_n =
\Im\sum_{k \leq n} \eu^{\iu k} = \Im\,\frac{\eu^{\iu}(\eu^{\iu n} -
1)}{\eu^{\iu} - 1}$, de módulo $\leq \frac{2}{\abs{\eu^{\iu} - 1}}$.
Ela \emph{não} converge \omterm{def:b2:series:def}{absolutamente} ($\abs{\sin n} \geq \sin^2 n
= \frac{1 - \cos 2n}{2}$, e $\sum \frac{1 - \cos 2n}{2n}$ diverge,
pois $\sum \frac{\cos 2n}{n}$ converge pelo mesmo critério de Abel enquanto
$\sum \frac{1}{2n}$ diverge). O critério das séries alternadas é o
caso particular $b_n = (-1)^n$.
\end{example}

\begin{example}[Abel sobre o círculo de convergência]\label{ex:b2:series:abelboundary}
Para quais complexos $z$ com $\abs z = 1$ a série $\sum_{n \geq 1}
\frac{z^n}{n}$ converge? Em $z = 1$ ela é a série harmônica:
divergente. Para $z \neq 1$ no círculo, o critério de Abel se aplica
com $a_n = \frac1n \downarrow 0$ e $b_n = z^n$, cujas somas
parciais são limitadas independentemente de $N$:
\[
\Bigl|\sum_{n=1}^{N} z^n\Bigr|
= \Bigl|\frac{z(z^N - 1)}{z - 1}\Bigr|
\leq \frac{2}{\abs{z - 1}} .
\]
Convergente --- embora nunca \omterm{def:b2:series:def}{absolutamente} ($\sum\frac1n$). Uma
série, um círculo de comportamentos: divergência num único ponto,
semiconvergência em todos os demais. Esse é o comportamento padrão na fronteira
das séries de potências (\cref{ch:b2:powerseries}), aqui encontrado
na unha; em $z = -1$ ele recupera a série harmônica
alternada, e em $z = \eu^{\iu\theta}$ suas partes real e
imaginária são as séries $\sum\frac{\cos
n\theta}{n}$ e $\sum\frac{\sin n\theta}{n}$ do
\cref{exo:b2:series:4}.
\end{example}

\begin{example}[Uma série alternada armadilhada]\label{ex:b2:series:trap}
A série $\sum_{n\geq2} \dfrac{(-1)^n}{\sqrt n + (-1)^n}$ converge?
Os sinais alternam e os termos tendem a $0$ --- e no entanto o
critério das alternadas \emph{não} se aplica: os módulos
$\frac{1}{\sqrt n + (-1)^n}$ não são decrescentes (eles pulam para cima a
cada $n$ ímpar). Desenvolva, em vez disso:
\[
\frac{(-1)^n}{\sqrt n + (-1)^n}
= \frac{(-1)^n}{\sqrt n}\cdot
\frac{1}{1 + \frac{(-1)^n}{\sqrt n}}
= \frac{(-1)^n}{\sqrt n} - \frac{1}{n}
+ O\Bigl(\frac{1}{n^{3/2}}\Bigr).
\]
A primeira peça converge (critério das alternadas, honestamente aplicado a
$\frac1{\sqrt n}\downarrow0$), a terceira converge \omterm{def:b2:series:def}{absolutamente}
--- mas a peça do meio é a série harmônica divergente: a
soma \emph{diverge} para $-\infty$. A lição final: quando a
monotonicidade falha, desenvolva até que cada peça seja
\omterm{def:b2:series:def}{absolutamente} convergente ou um caso limpo de aplicação de critério; o
$-\frac1n$ escondido é invisível para a contagem de sinais.
\end{example}

\begin{remark}[Armadilhas comuns]\label{rem:b2:series:pitfalls}
(i) ``Os termos tendem a $0$'' nada prova: a série harmônica
diverge. (ii) O critério das alternadas exige módulos
\emph{decrescentes} --- \cref{ex:b2:series:trap} é o
contraexemplo canônico, e a terceira série do~\cref{exo:b2:series:1}, o
treino. (iii) Séries condicionalmente convergentes não podem ser
reordenadas (\cref{ex:b2:series:rearrange}), e seus \omterm{thm:b2:series:fubini}{produtos
de Cauchy} podem divergir: para $\sum\frac{(-1)^n}{\sqrt{n+1}}$
ao quadrado, os termos da diagonal satisfazem
\[
\abs{c_k} = \sum_{m=0}^{k}
\frac{1}{\sqrt{(m+1)(k-m+1)}}
\geq (k+1)\cdot\frac{2}{k+2} \longrightarrow 2 \neq 0
\]
(cada fator é no máximo $\frac{k+2}2$ pela desigualdade das médias), de modo que $\sum c_k$
diverge --- a convergência absoluta de ao menos um fator
(\cref{exo:b2:series:8}) não é luxo. (iv) A \omterm{def:b2:series:summable}{somabilidade} é,
por definição, uma questão de cotas \emph{absolutas}: não existe
família condicionalmente \omterm{def:b2:series:summable}{somável}.
\end{remark}

\section{Famílias somáveis}

\begin{definition}\label{def:b2:series:summable}
Seja $I$ um conjunto de índices \omterm{def:b2:structures:countable}{enumerável}. Uma família $(u_i)_{i \in I}$ de
\emph{reais não negativos} é \emph{somável}\index{família somável}
quando as somas parciais finitas são limitadas; sua soma é
\[
\sum_{i \in I} u_i = \sup_{F \subseteq I \text{ finito}} \sum_{i
\in F} u_i \in \intcc{0}{+\infty} .
\]
Uma família de reais ou complexos (ou de vetores de Banach) é somável quando
$(\norm{u_i})$ o é; sua soma é então definida separando as partes
positiva e negativa (ou real e imaginária) --- equivalentemente, como o
valor comum de $\sum_{n} u_{\sigma(n)}$ para todas as enumerações
$\sigma$ de $I$ (veja adiante).
\end{definition}

\begin{method}[Escolher um critério]\label{met:b2:series:choosing}
Diante de $\sum u_n$, nesta ordem: (1) se $u_n \not\to 0$, divergência,
pare. (2) Se os termos têm sinal constante, compare: ache um
equivalente (\cref{ch:b2:comparison}) e situe-o no
mapa de Riemann--Bertrand. (3) Se os sinais alternam com
módulos \emph{decrescentes}, o critério das alternadas; se os módulos
não são monótonos, desenvolva o termo até que cada peça seja \omterm{def:b2:series:def}{absolutamente}
convergente ou um caso limpo (\cref{ex:b2:series:trap}). (4)
Se o padrão de sinais é oscilatório mas estruturado ($\sin
n\theta$, $\eu^{\iu n\theta}$, potências de matrizes), o critério de Abel com
somas parciais limitadas. (5) A convergência absoluta vale sempre
a pena ser verificada primeiro: é mais forte, imune à \omterm{def:b2:structures:generated}{ordem} e destranca \omterm{thm:b2:series:fubini}{produtos
de Cauchy} e Fubini.
\end{method}

\begin{example}[Somabilidade por contagem em diagonais]\label{ex:b2:series:diagonalcount}
Para quais $s > 0$ a família
$\bigl((m + n)^{-s}\bigr)_{m, n \geq 1}$ é \omterm{def:b2:series:summable}{somável}? Agrupe as
somas parciais finitas por diagonais $m + n = k$: a diagonal $k$
carrega $k - 1$ pares, cada um contribuindo com $k^{-s}$, de modo que as
somas finitas são exatamente majoradas por (e esgotam)
\[
\sum_{k \geq 2} \frac{k - 1}{k^{s}} ,
\]
uma série de termos positivos equivalente a $k^{1-s}$: \omterm{def:b2:series:summable}{somável}
se e somente se $s - 1 > 1$, isto é, $s > 2$. O índice bidimensional come
uma potência inteira: um plano de termos é ``uma dimensão mais
divergente'' do que uma reta --- é a geometria de contagem do conjunto
de índices, e não o tamanho dos termos individuais, que decide a \omterm{def:b2:series:summable}{somabilidade}.
(O mesmo recenseamento mostra que $\bigl((m^2 + n^2)^{-1}\bigr)$ não é
\omterm{def:b2:series:summable}{somável}: na diagonal $m + n = k$, cada termo vale ao menos
$k^{-2}$, e $(k-1)\cdot k^{-2}$ soma como a série
harmônica.)
\end{example}

\begin{theorem}[Somabilidade e ordem]\label{thm:b2:series:rearrangement}
\begin{enumerate}
  \item Para famílias não negativas, a soma é invariante por qualquer
        enumeração: $\sum_{i} u_i = \sum_{n=0}^{\infty}
        u_{\sigma(n)}$ para toda bijeção $\sigma \colon \N \to I$.
  \item Uma série real ou complexa $\sum u_n$ é \emph{comutativamente
        convergente} (toda reordenação converge, com a mesma
        soma) se e somente se ela é \omterm{def:b2:series:def}{absolutamente} convergente.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Toda soma parcial $\sum_{n \leq N} u_{\sigma(n)}$ é uma soma parcial
finita da família (logo $\leq$ o sup); todo $F$ finito está
contido em algum $\{\sigma(0), \dots, \sigma(N)\}$ (logo o sup é
$\leq$ o limite da série). As duas cotas se encontram.

(2) Se $\sum\abs{u_n} < \infty$: para qualquer reordenação $\sigma$ e
$\varepsilon > 0$, escolha $N$ com $\sum_{n > N}\abs{u_n} \leq
\varepsilon$; a partir do posto em que $\sigma$ tenha esgotado
$\intint{0}{N}$, as somas parciais reordenadas diferem do
limite original por no máximo $\varepsilon$: mesma soma. Se $\sum
\abs{u_n} = \infty$ mas $\sum u_n$ converge (caso real; o complexo
segue coordenada a coordenada): as partes positiva e negativa ambas
divergem, e é possível reordenar para atingir qualquer limite prescrito
--- o teorema de Riemann, executado no~\cref{exo:b2:series:5} ---
de modo que a convergência comutativa falha.
\end{proof}

\begin{example}[Uma reordenação pega em flagrante]\label{ex:b2:series:rearrange}
A série harmônica alternada soma
$\sum_{n\geq1}\frac{(-1)^{n-1}}{n} = \ln 2$ (volume do primeiro ano de
graduação). Reordene-a como ``um positivo, dois negativos'':
\[
1 - \frac12 - \frac14 + \frac13 - \frac16 - \frac18 + \frac15 -
\cdots
\]
Agrupando cada bloco de três,
\[
\frac{1}{2k-1} - \frac{1}{4k-2} - \frac1{4k}
= \frac{1}{4k-2} - \frac{1}{4k}
= \frac12\Bigl(\frac{1}{2k-1} - \frac1{2k}\Bigr),
\]
de modo que a série reordenada converge para $\frac12\ln 2$ --- metade da
soma original, com exatamente os mesmos termos. As séries
não \omterm{def:b2:series:def}{absolutamente} convergentes lembram a ordem de seus termos; as \omterm{def:b2:series:summable}{famílias
somáveis} são precisamente as que não lembram.
\end{example}

\begin{example}[Agrupar é seguro, desagrupar não é]\label{ex:b2:series:grouping}
Agrupar termos consecutivos de uma série \emph{convergente} nunca
muda a soma: as somas parciais agrupadas formam uma subsequência das
originais. A operação inversa é proibida:
\[
(1 - 1) + (1 - 1) + (1 - 1) + \cdots = 0 + 0 + \cdots = 0,
\]
e no entanto a série desagrupada $1 - 1 + 1 - 1 + \cdots$ diverge (as somas
parciais oscilam entre $1$ e $0$). Desagrupar só é legítimo
com uma hipótese compensatória --- por exemplo, termos
tendendo a $0$ com comprimentos de bloco limitados: então, entre duas
somas parciais agrupadas, as originais derivam no máximo por uma soma de
$o(1)$ termos em número limitado, e a convergência volta a se transferir.
Essa é exatamente a cláusula sob a qual o cálculo por blocos do
\cref{ex:b2:series:rearrange} é uma demonstração e não um truque
de mágica.
\end{example}

\begin{theorem}[Fubini para famílias; produtos de Cauchy]\label{thm:b2:series:fubini}
Seja $(u_{m,n})_{(m,n) \in \N^2}$ uma família dupla \omterm{def:b2:series:summable}{somável} (isto é,
$\sup_F \sum_F \abs{u_{m,n}} < \infty$). Então\index{teorema de Fubini
para séries}
\[
\sum_{(m,n)} u_{m,n}
= \sum_{m=0}^{\infty}\Bigl(\sum_{n=0}^{\infty} u_{m,n}\Bigr)
= \sum_{n=0}^{\infty}\Bigl(\sum_{m=0}^{\infty} u_{m,n}\Bigr)
= \sum_{k=0}^{\infty}\Bigl(\sum_{m+n=k} u_{m,n}\Bigr),
\]
convergindo (\omterm{def:b2:series:def}{absolutamente}) todas as séries internas. Em particular, se $\sum
a_m$ e $\sum b_n$ convergem \omterm{def:b2:series:def}{absolutamente}, seu \emph{produto
de Cauchy}\index{produto de Cauchy} converge \omterm{def:b2:series:def}{absolutamente} com
\[
\Bigl(\sum_m a_m\Bigr)\Bigl(\sum_n b_n\Bigr)
= \sum_{k=0}^{\infty} c_k,
\qquad
c_k = \sum_{m=0}^{k} a_m b_{k-m} .
\]
\end{theorem}

\begin{proof}
\emph{Caso não negativo.} Cada agrupamento (por linhas, colunas ou
diagonais) calcula o mesmo supremo: todo conjunto finito de pares está
contido num bloco finito de linhas (o que minora cada soma agrupada
por somas parciais finitas e a majora pelo total), e a convergência
monótona das somas parciais faz o resto --- concretamente, para linhas:
$\sum_{m \leq M}\sum_{n \leq N} u_{m,n} \leq S$ dá, fazendo $N
\to \infty$ e depois $M \to \infty$, $\sum_m \sum_n u_{m,n} \leq S$;
reciprocamente, todo $F$ finito cabe em tal retângulo, logo $S \leq
\sum_m\sum_n u_{m,n}$. Diagonais: as mesmas duas cotas com triângulos
em vez de retângulos.

\emph{Caso geral.} Separe em partes positiva e negativa (real e
imaginária), cada uma uma família não negativa \omterm{def:b2:series:summable}{somável}; os quatro
agrupamentos coincidem em cada parte, logo na diferença; a convergência
absoluta das séries internas vem do caso não negativo
aplicado a $\abs{u_{m,n}}$.

\emph{\omterm{thm:b2:series:fubini}{Produto de Cauchy}.} A família $u_{m,n} = a_m b_n$ é \omterm{def:b2:series:summable}{somável}:
as somas parciais finitas de $\abs{a_mb_n}$ são majoradas por
$\bigl(\sum\abs{a_m}\bigr)\bigl(\sum\abs{b_n}\bigr)$. As linhas dão
$\bigl(\sum a_m\bigr)\bigl(\sum b_n\bigr)$; as diagonais dão $\sum_k
c_k$.
\end{proof}

\begin{example}[A identidade exponencial, honestamente]\label{ex:b2:series:exp}
Para $a, b \in \C$ (ou matrizes que comutam):
\[
\Bigl(\sum_m \frac{a^m}{m!}\Bigr)\Bigl(\sum_n
\frac{b^n}{n!}\Bigr)
= \sum_k \sum_{m+n=k} \frac{a^m b^n}{m!\,n!}
= \sum_k \frac{(a + b)^k}{k!},
\]
pelo binômio de Newton em cada diagonal: $\eu^a \eu^b = \eu^{a+b}$
--- a equação funcional de $\exp$ deduzida só da série.
(A comutatividade é usada no passo do binômio; para matrizes que não comutam
a identidade falha de verdade,
\cref{ch:b2:diffeq}.)
\end{example}

\begin{example}[Produtos de Cauchy como dispositivo de cálculo]\label{ex:b2:series:squaresum}
A partir da série geométrica e do $\sum_{n
\geq 1} nz^n = \frac{z}{(1-z)^2}$ do~\cref{exo:b2:series:2} ($\abs z < 1$), mais um
\omterm{thm:b2:series:fubini}{produto de Cauchy} conclui o segundo momento. Multiplique $\sum_m
mz^m$ por $\sum_n z^n$: o coeficiente diagonal é $\sum_{m=0}^k
m = \frac{k(k+1)}2$, logo
\[
\frac{z}{(1-z)^3} = \sum_{k\geq0}\frac{k(k+1)}{2}\,z^k ,
\]
e a identidade $n^2 = 2\cdot\frac{n(n+1)}2 - n$ monta
\[
\sum_{n\geq1} n^2z^n = \frac{2z}{(1-z)^3} - \frac{z}{(1-z)^2}
= \frac{z(1+z)}{(1-z)^3} .
\]
Em $z = \frac12$: $\sum_{n\geq1}\frac{n^2}{2^n} =
\frac{\frac12\cdot\frac32}{\frac18} = 6$ --- um valor fechado sem
nenhuma derivação, apenas séries \omterm{def:b2:series:def}{absolutamente} convergentes
multiplicadas como polinômios. A mesma telescopagem de identidades
calcula todo $\sum n^dz^n$, e os probabilistas reconhecerão
o segundo momento fatorial da lei geométrica
(\cref{ch:b2:genfun}).
\end{example}

\begin{example}[Uma avaliação de soma dupla]\label{ex:b2:series:doublesum}
Para $s > 1$ real, seja $\zeta(s) = \sum_{n\geq1} n^{-s}$. Contando os
divisores por soma dupla --- a família $(m^{-s}n^{-s})$ sobre
$(m,n) \in (\N^*)^2$ é \omterm{def:b2:series:summable}{somável} (produto de séries positivas
convergentes) --- e agrupando pelo produto $q = mn$:
\[
\zeta(s)^2 = \sum_{m,n} \frac{1}{(mn)^s}
= \sum_{q=1}^{\infty} \frac{d(q)}{q^s},
\]
em que $d(q)$ é o número de divisores de $q$. As \omterm{def:b2:series:summable}{famílias somáveis}
transformam combinatória em análise.
\end{example}

\begin{example}[Uma avaliação por Fubini: $\sum_n (\zeta(n) - 1) = 1$]\label{ex:b2:series:zetasum}
Para $n \geq 2$ inteiro, $\zeta(n) - 1 = \sum_{k \geq 2} k^{-n}$.
A família dupla $(k^{-n})_{k, n \geq 2}$ é \omterm{def:b2:series:summable}{somável}: somando
primeiro as colunas geométricas,
\[
\sum_{k\geq2}\sum_{n\geq2} \frac{1}{k^n}
= \sum_{k\geq2} \frac{1/k^2}{1 - 1/k}
= \sum_{k\geq2} \frac{1}{k(k-1)} = 1
\]
(telescopagem), e todos os termos são positivos, de modo que o
\cref{thm:b2:series:fubini} autoriza somar por linhas:
\[
\sum_{n\geq2}\bigl(\zeta(n) - 1\bigr) = 1 .
\]
Os infinitos valores de $\zeta$, cada um de aparência transcendente,
têm caudas que somam exatamente $1$. Lição final: quando uma
soma dupla tem termos positivos, calcule-a na ordem que
colapsar --- aqui as colunas são geométricas, as linhas são misteriosas,
e Fubini transfere o colapso.
\end{example}

\begin{example}[A série geométrica resolve uma equação]\label{ex:b2:series:neumannsolve}
No \omterm{def:b2:nvs:banach}{espaço de Banach} $\bigl(C(\intcc01),
\norm\cdot_\infty\bigr)$, resolva $x - K(x) = y$, em que $K(f)$ é
a função constante $\frac12\int_0^1 f$. A \omterm{thm:b2:nvs:continuouslinear}{norma de operador} é
$\vertiii K \leq \frac12 < 1$, de modo que a série de Neumann se aplica
(\cref{ex:b2:series:neumann}): $x = \sum_{n\geq0} K^n(y)$.
Calcule os iterados: $K(y) = \frac12\int_0^1 y$ (uma constante) e,
aplicando $K$ a uma constante $c$, obtém-se $\frac c2$, logo
$K^n(y) = \frac{1}{2^{n-1}}\cdot\frac12\int_0^1 y$ para $n \geq
1$. Somando as constantes geométricas:
\[
x = y + \Bigl(\int_0^1 y\Bigr)
\sum_{n\geq1}\frac{1}{2^n} = y + \int_0^1 y .
\]
Verificação: $x - K(x) = y + \int y - \frac12\bigl(\int y + \int
y\bigr) = y$. Uma série infinita, uma resposta finita e uma
verificação de uma linha --- a série geométrica é um algoritmo de inversão,
e não apenas um enunciado de convergência.
\end{example}

\begin{example}[Telescopagem por frações parciais]\label{ex:b2:series:telescope}
A somação exata é rara; a telescopagem é seu principal fornecedor.
Decomponha
\[
\frac{1}{n(n+1)(n+2)}
= \frac{1}{2}\Bigl(\frac{1}{n(n+1)} -
\frac{1}{(n+1)(n+2)}\Bigr),
\]
(verifique reduzindo ao denominador comum), de modo que as somas
parciais colapsam:
\[
\sum_{n=1}^{N}\frac{1}{n(n+1)(n+2)}
= \frac12\Bigl(\frac{1}{1\cdot2} -
\frac{1}{(N+1)(N+2)}\Bigr)
\longrightarrow \frac14 .
\]
O mesmo padrão --- escrever o termo como $c(u_n - u_{n+1})$ para
um $(u_n)$ explícito --- resolveu o~\cref{exo:b2:series:10}
(arco-tangentes) e calcula todo $\sum\frac{1}{n(n+1)\cdots(n +
k)} = \frac{1}{k\cdot k!}$. Quando existe uma soma exata neste
nível, costuma haver um telescópio escondido no termo.
\end{example}

\begin{remark}[Perspectivas dentro deste volume]\label{rem:b2:series:perspectives}
Três capítulos adiante são clientes diretos. Para o
\cref{ch:b2:funcseq}: a convergência normal de $\sum f_n$ é a
convergência absoluta de $\sum\norm{f_n}_\infty$ no \omterm{def:b2:nvs:banach}{espaço de
Banach} $\bigl(C, \norm\cdot_\infty\bigr)$ --- o
\cref{thm:b2:nvs:absoluteconvergence} deste capítulo, a rigor, de fantasia. Para o
\cref{ch:b2:powerseries}: dentro do disco de convergência
tudo é absoluto e \omterm{def:b2:series:summable}{somável}, de modo que \omterm{thm:b2:series:fubini}{produtos de Cauchy} e
reordenações correm soltos (é por isso que as séries de potências se multiplicam como
polinômios); na fronteira, o critério de Abel assume
(\cref{ex:b2:series:abelboundary}). Para o~\cref{ch:b2:genfun}:
as funções geradoras de probabilidade são séries de potências cujas
manipulações --- produtos para somas de variáveis independentes,
somas duplas para leis compostas --- são todas autorizadas pelo
\cref{thm:b2:series:fubini}. As \omterm{def:b2:series:summable}{famílias somáveis} são o departamento
jurídico da análise que vem a seguir.
\end{remark}

\begin{remark}[Onde este capítulo é usado]
Tudo o que tem soma infinita passa por aqui: as séries de potências
(\cref{ch:b2:powerseries}) são \omterm{def:b2:series:summable}{famílias somáveis} disfarçadas,
os coeficientes de Fourier são multiplicados por \omterm{thm:b2:series:fubini}{produtos de Cauchy} e
reordenados por Parseval (\cref{ch:b2:fourier}), e as funções
geradoras de probabilidade (\cref{ch:b2:genfun}) são o teorema de Fubini
aplicado a esperanças. O volume do terceiro ano de graduação absorve as \omterm{def:b2:series:summable}{famílias somáveis}
na integração de Lebesgue com a medida de contagem ---
onde o~\cref{thm:b2:series:fubini} se torna um caso particular do
teorema de Fubini--Tonelli.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:series:1}
Natureza de: $\sum \dfrac{\cos n}{n}$; $\;\sum \dfrac{(-1)^n}{\ln
n}$; $\;\sum \dfrac{(-1)^n}{n^{3/4} + \cos n}$ \emph{(desenvolva como na
armadilha do primeiro ano: o critério das alternadas precisa de monotonicidade)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:series:2}
Prove que, para $\abs z < 1$: $\sum_{n\geq1} n z^{n} =
\dfrac{z}{(1-z)^2}$, via o \omterm{thm:b2:series:fubini}{produto de Cauchy} de $\sum z^n$ consigo
mesma.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:3}
(Lema do tipo Kronecker) Seja $\sum b_n$ uma série real convergente.
Prove, pela \omterm{thm:b2:series:abel}{transformação de Abel}, que $\dfrac{1}{n}\sum_{k=1}^{n} k\,b_k
\to 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:4}
Estude a convergência de $\sum \dfrac{\sin(n\theta)}{n^\alpha}$
($\theta \in \R$, $\alpha > 0$) --- para quais $(\theta, \alpha)$ ela é
\omterm{def:b2:series:def}{absolutamente} convergente, semiconvergente, divergente?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:5}
(Reordenação de Riemann) Seja $\sum u_n$ uma série real convergente mas não
\omterm{def:b2:series:def}{absolutamente} convergente, e seja $\ell \in \R$. Prove que
alguma reordenação de $\sum u_n$ converge para $\ell$.
\emph{(Mostre que as duas subséries de termos positivos e negativos divergem;
depois alterne gulosamente: tome termos positivos até ultrapassar
$\ell$, depois negativos até cair abaixo, e assim por diante; os termos
tendem a $0$, forçando a convergência para $\ell$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:6}
Prove que a família
$\Bigl(\dfrac{x^{m+n}}{m!\,n!}\Bigr)_{(m,n)\in\N^2}$ é \omterm{def:b2:series:summable}{somável}
para todo $x \in \R$, e redemonstre a identidade $(\eu^x)^2 =
\eu^{2x}$ agrupando a soma dupla ao longo das diagonais $m + n =
k$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:7}
Prove que a família $\bigl(\frac{1}{m^2 n^2}\bigr)_{m,n \geq 1}$
é \omterm{def:b2:series:summable}{somável} e que o agrupamento por $\gcd$ dá: com $q = \gcd(m,n)$,
\[
\zeta(2)^2 = \sum_{q\geq1} \frac{1}{q^4}
\sum_{\substack{a,b \geq 1\\ \gcd(a,b)=1}} \frac{1}{a^2b^2}
= \zeta(4) \cdot S,
\]
em que $S = \sum_{\gcd(a,b)=1} \frac{1}{a^2b^2}$: deduza $S =
\zeta(2)^2/\zeta(4)$. \emph{(Todo par $(m,n)$ se escreve de modo único
$(qa, qb)$ com $\gcd(a,b) = 1$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:8}
(Teorema de Abel sobre produtos, versão leve) Suponha que $\sum a_n$
convirja \omterm{def:b2:series:def}{absolutamente} e que $\sum b_n$ convirja. Prove que seu
\omterm{thm:b2:series:fubini}{produto de Cauchy} $\sum c_n$ converge, com $\sum c_n = (\sum
a_n)(\sum b_n)$. \emph{(Escreva $C_N = \sum_{k\leq N} c_k = \sum_n
a_n B_{N-n}$ com $B$ as somas parciais de $b$; separe conforme
$n \leq N/2$ ou não, usando a limitação de $(B_m)$ e a
cauda absoluta de $(a_n)$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:9}
No espaço (não \omterm{def:b2:metric:complete}{completo}) $E$ das sequências reais nulas a partir de certa ordem,
com a \omterm{def:b2:nvs:norm}{norma} do sup, exiba uma série \omterm{def:b2:series:def}{absolutamente} convergente que não
converge em $E$. \emph{(Tente $u_n = 2^{-n} e_n$ com $(e_n)$ as
sequências canônicas.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:10}
Verifique a identidade $\arctan(n+1) - \arctan(n) =
\arctan\dfrac{1}{n^2 + n + 1}$ e deduza o valor exato de
\[
\sum_{n=1}^{\infty} \arctan\frac{1}{n^2 + n + 1} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:11}
Determine a natureza (com equivalentes) de
\[
\sum_n \bigl(\sqrt{n+1} - \sqrt n\bigr)^{\alpha}
\ (\alpha > 0), \qquad
\sum_n \Bigl(1 - \cos\frac1n\Bigr), \qquad
\sum_n \Bigl(\eu - \Bigl(1 + \frac1n\Bigr)^{\!n}\Bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:12}
Seja $(a_n)$ positiva e \emph{decrescente} com $\sum a_n$
convergente. Prove que $n\,a_n \to 0$ \emph{(majore $n a_{2n}$ por uma
cauda)}. Mostre que a recíproca é falsa e que a monotonicidade é
essencial, com contraexemplos explícitos.
\end{exercise}

\section{Problema: o $\zeta(2) = \pi^2/6$ de Euler, pela soma de
cotangentes de Cauchy}

A identidade mais famosa de Euler, $1 + \frac14 + \frac19 + \cdots =
\frac{\pi^2}6$, admite uma demonstração inteiramente elementar, devida a
Cauchy: a fórmula de de Moivre produz um polinômio cujas raízes são
os números $\cot^2\frac{k\pi}{2n+1}$, as relações de Girard somam essas raízes
exatamente, e o confronto $\cot^2\theta < \frac{1}{\theta^2} <
1 + \cot^2\theta$ esmaga as somas parciais de $\sum\frac1{k^2}$
entre duas cotas racionais explícitas. Rodamos a demonstração inteira,
extraímos $\zeta(4) = \frac{\pi^4}{90}$ pelo mesmo método e depois
mapeamos toda a fronteira entre convergência e divergência com
as séries de Bertrand --- e provamos que essa fronteira não carrega
nenhuma série convergente mais lenta que todas.

\begin{problem}[{Problema de fim de semana --- $\zeta(2) = \pi^2/6$ e o
panorama de Bertrand}]
\label{pb:b2:series:1}
Em todo o problema, $n \geq 1$ e $\theta_k = \dfrac{k\pi}{2n+1}$ para $k
= 1, \dots, n$; note que $0 < \theta_k < \frac\pi2$.

\medskip
\noindent\textbf{Parte I --- A identidade das cotangentes.}
\begin{enumerate}
  \item Prove a fórmula de de Moivre $(\cos\theta +
        \iu\sin\theta)^m = \cos m\theta + \iu\sin m\theta$ ($m
        \in \N$) e deduza, para $m = 2n + 1$,
        \[
        \sin\bigl((2n{+}1)\theta\bigr) = \sum_{j=0}^{n}
        (-1)^j\binom{2n+1}{2j+1}\cos^{2(n-j)}\theta\,
        \sin^{2j+1}\theta .
        \]
  \item Deduza que, para $\theta \in \intoo{0}{\frac\pi2}$,
        \[
        \sin\bigl((2n{+}1)\theta\bigr) =
        \sin^{2n+1}\theta\; P_n(\cot^2\theta),
        \qquad
        P_n(x) = \sum_{j=0}^{n}
        (-1)^j\binom{2n+1}{2j+1}x^{\,n-j},
        \]
        um polinômio de grau $n$ com coeficiente dominante $2n
        + 1$.
  \item Mostre que $x_k = \cot^2\theta_k$, $k = 1, \dots, n$, são
        $n$ raízes \emph{distintas} de $P_n$ --- logo, todas
        elas.
  \item Pelas relações de Girard, prove a identidade exata
        \[
        \sum_{k=1}^{n} \cot^2\frac{k\pi}{2n+1}
        = \frac{n(2n-1)}{3}.
        \]
  \item Deduza também $\displaystyle\sum_{k=1}^{n}
        \frac{1}{\sin^2\theta_k} = \frac{2n(n+1)}{3}$.
  \item Prove o confronto: $\cot^2\theta < \dfrac1{\theta^2} <
        \dfrac{1}{\sin^2\theta}$ para $\theta \in
        \intoo{0}{\frac\pi2}$ \emph{(a partir de $\sin\theta < \theta <
        \tan\theta$)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O confronto se fecha: o teorema de
Euler.}
\begin{enumerate}[resume]
  \item Somando a questão 6 sobre $k = 1, \dots, n$ com $\theta =
        \theta_k$, estabeleça
        \[
        \frac{n(2n-1)}{3} \;<\;
        \frac{(2n+1)^2}{\pi^2}\sum_{k=1}^{n}\frac1{k^2}
        \;<\; \frac{2n(n+1)}{3}.
        \]
  \item Conclua (\emph{o teorema de Euler, pela demonstração de Cauchy}):
        \[
        \zeta(2) = \sum_{k=1}^{\infty}\frac{1}{k^2} =
        \frac{\pi^2}{6}.
        \]
  \item Extraia uma taxa do sanduíche: mostre que
        \[
        \Bigl|\sum_{k=1}^{n}\frac1{k^2} - \frac{\pi^2}6\Bigr|
        = O\Bigl(\frac1n\Bigr),
        \]
        coerente com a cauda exata $\sum_{k>n}k^{-2} =
        \frac1n - \frac{1}{2n^2} + O(n^{-3})$ do
        \cref{exo:b2:comparison:11}.
  \item Rode a máquina um andar acima: usando a segunda função de Girard
        dos $P_n$, mostre que
        \[
        \sum_{k=1}^{n}\cot^4\theta_k =
        \Bigl(\frac{n(2n-1)}3\Bigr)^{\!2} -
        \frac{2n(2n-1)(2n-2)(2n-3)}{60}
        \;\sim\; \frac{8n^4}{45},
        \]
        e confronte com $\cot^4 < \theta^{-4} < (1 +
        \cot^2)^2$ para obter $\zeta(4) = \dfrac{\pi^4}{90}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Dividendos.}
\begin{enumerate}[resume]
  \item Deduza do $\zeta(2) = \frac{\pi^2}6$:
        \[
        \sum_{k\geq0}\frac{1}{(2k+1)^2} = \frac{\pi^2}{8},
        \qquad
        \sum_{k\geq1}\frac{(-1)^{k-1}}{k^2} = \frac{\pi^2}{12}.
        \]
  \item Combine com \cref{exo:b2:series:7}: calcule $S =
        \sum_{\gcd(a,b)=1}\frac{1}{a^2b^2} =
        \frac{\zeta(2)^2}{\zeta(4)} = \frac52$ e interprete
        $\frac{1}{\zeta(2)} = \frac{6}{\pi^2} \approx 0.608$ como
        a densidade de pares coprimos (enuncie a heurística
        honestamente: a contagem rigorosa é assunto do volume do
        terceiro ano de graduação).
  \item (Aceleração certificada) A fórmula da cauda da questão 9
        dá $\sum_{k\leq n}k^{-2} + \frac1n - \frac1{2n^2} =
        \frac{\pi^2}6 + O(n^{-3})$. Compare o trabalho necessário para
        seis algarismos de $\zeta(2)$: somação direta contra a
        soma corrigida em $n = 100$ (onde o erro é
        $1.7\cdot10^{-7}$).
  \item Verifique a questão 4 à mão em $n = 1$ e $n = 2$ (os
        valores $\cot^2\frac\pi3 = \frac13$ e
        $\cot^2\frac\pi5 + \cot^2\frac{2\pi}5 = 2$), usando
        $\cos\frac\pi5 = \frac{1+\sqrt5}4$ ou uma avaliação
        numérica.
  \item Demonstre a identidade companheira
        \[
        \sum_{k=1}^{n}\tan^2\frac{k\pi}{2n+1} = n(2n+1)
        \]
        \emph{(os números $\tan^2\theta_k$ são as raízes do
        polinômio invertido $x^nP_n(1/x)$)} e verifique-a em $n
        = 1$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O panorama de Bertrand.} Para $\alpha,
\beta \in \R$, considere a \emph{série de Bertrand}
\[
\sum_{n \geq 3} \frac{1}{n^{\alpha}(\ln n)^{\beta}} .
\]
\begin{enumerate}[resume]
  \item Mostre que, para $\alpha > 1$, a série converge,
        qualquer que seja $\beta$ \emph{(compare com
        $n^{-(1+\alpha)/2}$)}.
  \item Mostre que, para $\alpha < 1$, ela diverge, qualquer que seja
        $\beta$.
  \item Para $\alpha = 1$: usando a comparação série--integral
        (\cref{thm:b2:comparison:seriesintegral}) com $f(t) =
        \frac{1}{t(\ln t)^\beta}$, prove a convergência se e somente se $\beta
        > 1$.
  \item Itere a fronteira: mostre que $\sum\frac{1}{n\ln n\,\ln\ln
        n}$ diverge enquanto $\sum\frac{1}{n\ln n\,(\ln\ln n)^2}$
        converge.
  \item Duas armadilhas: determine a natureza de
        \[
        \sum_n \frac{1}{n^{1 + 1/\ln n}}
        \qquad\text{e}\qquad
        \sum_n \frac{1}{n^{1 + 1/\ln\ln n}}
        \]
        \emph{(calcule $n^{1/\ln n}$ exatamente; compare
        $n^{1/\ln\ln n}$ com toda potência de $\ln n$)}.
  \item (Não existe série convergente mais lenta) Seja $\sum a_n$ uma
        série convergente qualquer com $a_n > 0$, e sejam $R_n = \sum_{k
        \geq n}a_k$ suas caudas. Prove que $\sum
        \frac{a_n}{\sqrt{R_n}}$ ainda \emph{converge}
        \emph{(compare com a telescopagem $2(\sqrt{R_n} -
        \sqrt{R_{n+1}})$)}, embora $\frac{a_n/\sqrt{R_n}}{a_n}
        \to \infty$: toda série convergente é estritamente
        dominada por outra série convergente. A fronteira da
        convergência não é uma curva, mas uma névoa.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Verificações cruzadas e síntese.}
\begin{enumerate}[resume]
  \item (Condensação de Cauchy) Prove: para $(a_n)$ positiva e
        decrescente, $\sum a_n$ converge se e somente se $\sum 2^k a_{2^k}$
        converge. Redemonstre com isso a fronteira da questão 18.
  \item (O custo da lentidão) Para $\sum\frac1{n(\ln n)^2}$,
        majore a cauda por uma integral e mostre que somar até
        $N = 10^6$ ainda deixa um erro maior que
        $0.07$: uma convergência certificada pela teoria pode ser inútil
        para a numérica --- contraste com a questão 13.
  \item Classifique (com justificativas de uma linha):
        $\sum\frac1{n\ln n}$, $\sum\frac1{n^{1.01}}$,
        $\sum\frac{(\ln n)^{100}}{n^{1.001}}$,
        $\sum\frac1{n(\ln n)(\ln\ln n)^{3}}$.
  \item (Síntese) Uma frase para cada: como de Moivre transformou uma
        identidade trigonométrica num polinômio com somas de raízes
        calculáveis; onde o confronto precisou de identidades \emph{exatas}
        nas extremidades, e não de equivalentes; qual ferramenta
        do~\cref{ch:b2:comparison} moveu a Parte IV; e o que a
        questão 21 diz sobre o sonho de um ``critério de comparação
        universal''. Nomeie os dois cumes: o $\zeta(2) = \frac{\pi^2}6$
        de Euler (e o andar acima, $\zeta(4) = \frac{\pi^4}{90}$) e a
        classificação de Bertrand. Note onde $\zeta(2)$ será demonstrado
        de novo: por Parseval no~\cref{ch:b2:fourier} --- um
        teorema, duas civilizações.
\end{enumerate}
\end{problem}
