\chapter{Séries de Fourier}\label{ch:b2:fourier}

Todo sinal periódico pode ser reconstruído a partir de senos e cossenos puros?
O audacioso ``sim'' de Fourier criou um século de análise. Este
capítulo demonstra os dois pilares ao alcance deste nível:
o \emph{teorema de Dirichlet} (reconstrução \omterm{def:b2:funcseq:def}{pontual} para funções $C^1$ por partes,
via o \omterm{lem:b2:fourier:kernel}{núcleo de Dirichlet}) e a \emph{identidade de Parseval} (a energia
de um sinal é a soma das energias de seus harmônicos), e
colhe as séries numéricas clássicas --- o $\sum 1/n^2 =
\pi^2/6$ de Basileia à frente de todas.

Em todo o capítulo, as funções são $2\pi$-periódicas, \omterm{def:b2:metric:continuity}{contínuas} por partes
e com valores complexos; $\mathcal{C}$ designa as \omterm{def:b2:metric:continuity}{contínuas}.

\section{Coeficientes de Fourier}

\begin{definition}\label{def:b2:fourier:coefficients}
Os \emph{coeficientes de Fourier}\index{coeficientes de Fourier} de $f$
são
\[
c_n(f) = \frac{1}{2\pi}\int_{-\pi}^{\pi} f(t)\,\eu^{-\iu n t}\,\dd
t \qquad (n \in \Z),
\]
e os coeficientes na forma real $a_n = c_n + c_{-n}$, $b_n = \iu(c_n
- c_{-n})$, de modo que as \emph{somas parciais de Fourier}\index{somas parciais
de Fourier} são
\[
S_N(f)(t) = \sum_{n=-N}^{N} c_n(f)\,\eu^{\iu nt}
= \frac{a_0}{2} + \sum_{n=1}^{N}\bigl(a_n\cos nt + b_n \sin
nt\bigr).
\]
Em $\mathcal{C}$, defina o \omterm{def:b2:hermitian:def}{produto interno hermitiano} $\langle f,
g\rangle = \frac{1}{2\pi}\int_{-\pi}^{\pi}\conj f\,g$: as
exponenciais $e_n(t) = \eu^{\iu nt}$ são \emph{ortonormais}
($\langle e_m, e_n\rangle = \delta_{mn}$, cálculo direto), e
$c_n(f) = \langle e_n, f\rangle$: a análise de Fourier é geometria
\omterm{def:b2:hermitian:adjoint}{hermitiana} (\cref{ch:b2:hermitian}) em dimensão infinita.
\end{definition}

\begin{proposition}[Desigualdade de Bessel]\label{prop:b2:fourier:bessel}
$S_N(f)$ é a projeção ortogonal de $f$ sobre o espaço
$\mathcal{T}_N$ dos polinômios trigonométricos de grau $\leq N$,
e\index{desigualdade de Bessel}
\[
\sum_{n=-N}^{N} \abs{c_n(f)}^2 \leq \norm f_2^2 =
\frac{1}{2\pi}\int_{-\pi}^{\pi} \abs f^2 :
\]
a série $\sum \abs{c_n}^2$ converge, e $c_n(f) \to 0$ quando
$\abs n \to \infty$ (Riemann--Lebesgue para coeficientes).
\end{proposition}

\begin{proof}
$f - S_N(f)$ é ortogonal a cada $e_k$, $\abs k \leq N$ ($\langle
e_k, f - S_N f\rangle = c_k - c_k = 0$): $S_Nf$ é a projeção
ortogonal sobre $\mathcal{T}_N = \operatorname{Vect}(e_{-N}, \dots,
e_N)$ (o teorema da projeção do volume do primeiro ano de graduação, palavra por palavra no
contexto \omterm{def:b2:hermitian:adjoint}{hermitiano}). Pitágoras: $\norm f_2^2 = \norm{S_Nf}_2^2 +
\norm{f - S_Nf}_2^2 \geq \norm{S_Nf}_2^2 = \sum_{\abs n \leq N}
\abs{c_n}^2$; faça $N \to \infty$.
\end{proof}

\begin{example}[Melhor aproximação, medida]\label{ex:b2:fourier:bestapprox}
Quão bem os polinômios trigonométricos de grau baixo aproximam
a dente de serra $f(t) = t$ (em $\intoo{-\pi}{\pi}$) na
média quadrática? Pela~\cref{prop:b2:fourier:bessel}, a melhor
aproximação de grau $N$ \emph{é} $S_N(f)$, com erro quadrático
\[
\norm{f - S_Nf}_2^2 = \norm f_2^2 -
\sum_{\abs n\leq N}\abs{c_n}^2 .
\]
Aqui $\norm f_2^2 = \frac{1}{2\pi}\int_{-\pi}^\pi t^2\dd t =
\frac{\pi^2}{3}$ e, de $b_n = \frac{2(-1)^{n+1}}{n}$
(\cref{ex:b2:fourier:basel}): $\abs{c_n}^2 + \abs{c_{-n}}^2 =
\frac{b_n^2}{2} = \frac{2}{n^2}$. Logo
\[
\norm{f - S_Nf}_2^2
= \frac{\pi^2}{3} - \sum_{n=1}^{N}\frac{2}{n^2}
\qquad\text{: numericamente } 1.29,\ 0.79,\ 0.57,\ 0.44
\]
para $N = 1, 2, 3, 4$ --- decrescente, mas devagar: a cauda
$\sum_{n>N}\frac2{n^2} \sim \frac2N$ é governada pelo decaimento lento
$\frac1n$ dos coeficientes, que é ele próprio a assinatura
do salto (o~\cref{exo:b2:fourier:6} lido ao contrário). Lição
final: Parseval transforma a qualidade da aproximação numa cauda de
série numérica --- e prevê, antes de qualquer figura, que
saltos fazem as séries de Fourier convergirem a contragosto.
\end{example}

\begin{method}[Calcular coeficientes de Fourier com eficiência]\label{met:b2:fourier:coefficients}
Antes de integrar seja o que for:
\begin{enumerate}
  \item \emph{Paridade:} $f$ par tem $b_n = 0$, $f$ ímpar tem $a_n
        = 0$, e as integrais sobreviventes se reduzem a $\frac2\pi
        \int_0^\pi$ --- metade do trabalho, o dobro de confiabilidade.
  \item \emph{Os polinômios trigonométricos já estão prontos:}
        linearize produtos ($\cos^3$, $\sin^2\cos$, \dots) e
        leia os coeficientes
        (\cref{exo:b2:fourier:9}); a ortonormalidade torna qualquer
        integração adicional redundante.
  \item \emph{Exponenciais complexas para exponenciais:} para
        fatores $\eu^{at}$ ou oscilações amortecidas, calcule
        $c_n$ diretamente --- uma integral de $\eu^{(a - \iu
        n)t}$ vence duas integrações por partes
        (\cref{exo:b2:fourier:10}).
  \item \emph{Derive um desenvolvimento conhecido:} se $f' $ tem
        coeficientes conhecidos e $f$ é \omterm{def:b2:metric:continuity}{contínua}, $c_n(f) =
        \frac{c_n(f')}{\iu n}$ ($n \neq 0$) recupera tudo exceto
        $c_0$, que é a média --- muitas vezes a rota mais rápida,
        e legítima exatamente sob as hipóteses do
        \cref{thm:b2:fourier:parseval}~(1).
\end{enumerate}
\end{method}

\section{O teorema de Dirichlet}

\begin{lemma}[Núcleo de Dirichlet]\label{lem:b2:fourier:kernel}
$S_N(f)(x) = \frac{1}{2\pi}\int_{-\pi}^{\pi} f(x + u)\,D_N(u)\,\dd
u$, em que\index{núcleo de Dirichlet}
\[
D_N(u) = \sum_{n=-N}^{N} \eu^{\iu nu}
= \frac{\sin\bigl((N + \frac12)u\bigr)}{\sin\frac u2}
\quad (u \notin 2\pi\Z),
\qquad
\frac{1}{2\pi}\int_{-\pi}^{\pi} D_N = 1 .
\]
\end{lemma}

\begin{proof}
Insira a definição de $c_n$ em $S_N$ e troque a soma com a integral
(legítimo: a soma é finita):
\[
S_N(f)(x)
= \sum_{n=-N}^{N}\Bigl(\frac{1}{2\pi}\int_{-\pi}^{\pi}
f(t)\,\eu^{-\iu nt}\dd t\Bigr)\eu^{\iu nx}
= \frac{1}{2\pi}\int_{-\pi}^{\pi} f(t)
\sum_{n=-N}^{N}\eu^{\iu n(x-t)}\,\dd t ;
\]
substitua $u = t - x$ e desloque o segmento de integração de volta
a $\intcc{-\pi}{\pi}$ pela $2\pi$-periodicidade do integrando;
o intervalo \omterm{def:b2:quadratic:adjoint}{simétrico} de índices faz $\sum_n\eu^{-\iu nu} = D_N(u)$.
A forma fechada: soma geométrica de razão $\eu^{\iu u}$,
\[
D_N(u) = \eu^{-\iu Nu}\,\frac{\eu^{\iu(2N+1)u} - 1}{\eu^{\iu u} -
1}
= \frac{\eu^{\iu(N + \frac12)u} - \eu^{-\iu(N+\frac12)u}}
{\eu^{\iu u/2} - \eu^{-\iu u/2}} ,
\]
que é o quociente de senos. Sua média vale $1$: só $n = 0$
contribui.
\end{proof}

\begin{theorem}[Lema de Riemann--Lebesgue]\label{thm:b2:fourier:riemannlebesgue}
Para $g$ \omterm{def:b2:metric:continuity}{contínua} por partes num segmento,
$\int_a^b g(t)\sin(\lambda t + \varphi)\,\dd t \to 0$ quando $\lambda
\to +\infty$.\index{lema de Riemann--Lebesgue}
\end{theorem}

\begin{proof}
Aproxime $g$ \omterm{def:b2:funcseq:def}{uniformemente} por funções escada
(o~\cref{thm:b2:funcseq:weierstrass} não é necessário --- a aproximação
elementar por funções escada de funções \omterm{def:b2:metric:continuity}{contínuas} por partes
basta) e integre cada degrau explicitamente: cada peça
contribui com $O\bigl(\frac1\lambda\bigr)$, e o erro de aproximação
contribui com $\varepsilon(b - a)$. Esse argumento foi executado
por inteiro como exercício final do capítulo de integração do
volume do primeiro ano de graduação; para peças $C^1$ pode-se, em vez disso, integrar por
partes e majorar por $\frac C\lambda$.
\end{proof}

\begin{example}[Com que rapidez morrem os coeficientes?]\label{ex:b2:fourier:decaytable}
Riemann--Lebesgue diz que os coeficientes tendem a $0$; sua
\emph{velocidade} é um medidor de suavidade. Três espécimes deste
capítulo e de seus exercícios:
\[
\text{onda quadrada: } b_n = \frac{4}{\pi n}\ (n\ \text{ímpar}),
\qquad
\abs t : a_n = \frac{-4}{\pi n^2}\ (n\ \text{ímpar}),
\qquad
\abs{\sin t} : a_{2k} = \frac{-4}{\pi(4k^2-1)} .
\]
Um salto em $f$ (onda quadrada, dente de serra) deixa coeficientes de
ordem $\frac1n$: sem \omterm{def:b2:funcseq:series}{convergência normal}, com ultrapassagem de Gibbs nos
saltos. A \omterm{def:b2:metric:continuity}{continuidade} com um bico --- um salto apenas em $f'$ ---
melhora a ordem para $\frac{1}{n^2}$: \omterm{def:b2:funcseq:series}{convergência normal},
reconstrução uniforme. Em geral $k$ derivadas compram $c_n =
O(n^{-k})$ (\cref{exo:b2:fourier:6}) e, reciprocamente, um \omterm{def:b2:reduction:eigen}{espectro}
que decai mais depressa que toda potência força $f$ a ser $C^\infty$
(derive termo a termo, agora legitimamente). Lição
final: a regularidade do sinal e o decaimento do \omterm{def:b2:reduction:eigen}{espectro}
são a mesma informação --- um engenheiro lê uma na
inclinação da outra sem nunca traçar a função.
\end{example}

\begin{theorem}[Dirichlet]\label{thm:b2:fourier:dirichlet}
Seja $f$ $2\pi$-periódica e $C^1$ por partes. Então, para todo $x$,
\[
S_N(f)(x) \xrightarrow[N \to \infty]{}
\frac{f(x^+) + f(x^-)}{2}
\]
(a média dos limites laterais) --- em particular, $S_N(f)(x) \to
f(x)$ em todo ponto de \omterm{def:b2:metric:continuity}{continuidade}.\index{teorema de Dirichlet}
\end{theorem}


\begin{proof}
Pelo lema do núcleo e por sua média unitária, separando a integral nas
metades $u > 0$ e $u < 0$ (cada uma de média $\frac12$):
\begin{align*}
S_N(f)(x) - \frac{f(x^+) + f(x^-)}{2}
&= \frac{1}{2\pi}\int_{0}^{\pi} \bigl(f(x+u) -
f(x^+)\bigr)D_N(u)\,\dd u\\
&\quad+ \frac{1}{2\pi}\int_{-\pi}^{0}\bigl(f(x+u) -
f(x^-)\bigr)D_N(u)\,\dd u .
\end{align*}
Trate a primeira (a segunda é \omterm{def:b2:quadratic:adjoint}{simétrica}). Escreva
\[
\bigl(f(x + u) - f(x^+)\bigr)\,D_N(u)
= \underbrace{\frac{f(x+u) - f(x^+)}{\sin\frac u2}}_{g(u)}\,
\sin\Bigl(\Bigl(N + \frac12\Bigr)u\Bigr) .
\]
A função $g$ é \omterm{def:b2:metric:continuity}{contínua} por partes em $\intoc{0}{\pi}$ e
tem \emph{limite finito em $0^+$}: escrevendo
\[
g(u) = \frac{f(x+u) - f(x^+)}{u}\cdot\frac{u}{\sin\frac u2} ,
\]
o primeiro fator tende a $f'(x^+)$ (derivabilidade
lateral, vinda de ser $C^1$ por partes) e o segundo a $2$
(o limite padrão $\frac{\sin v}{v} \to 1$ em $v = \frac
u2$): $g(0^+) = 2f'(x^+)$ existe. Assim $g$ estende-se
\omterm{def:b2:metric:continuity}{continuamente} por partes a $\intcc{0}{\pi}$, e Riemann--Lebesgue
(\cref{thm:b2:fourier:riemannlebesgue}) manda a integral para
$0$. Esse é o ponto todo da hipótese: sem
derivadas laterais, o fator $\frac{1}{\sin(u/2)}$ explode
em $0$ mais depressa do que Riemann--Lebesgue consegue compensar, e a
\omterm{def:b2:funcseq:def}{convergência pontual} pode falhar de verdade para $f$ meramente
\omterm{def:b2:metric:continuity}{contínua} --- a lacuna que o teorema de Fejér (problema de fim de semana)
fecha promediando.
\end{proof}

\begin{example}[Dirichlet num salto]\label{ex:b2:fourier:jumpvalue}
Para a dente de serra $f(t) = t$ em $\intoo{-\pi}{\pi}$
(\cref{ex:b2:fourier:basel} adiante), a extensão periódica
salta em $t = \pi$ de $f(\pi^-) = \pi$ a $f(\pi^+) = -\pi$.
Dirichlet promete o valor $\frac{\pi + (-\pi)}{2} = 0$
aí, e de fato todo termo de $\sum
\frac{2(-1)^{n+1}}{n}\sin nt$ se anula em $t = \pi$: a
série converge educadamente para o ponto médio, ignorando os dois
valores laterais. Mover o ponto de avaliação para $t =
\frac\pi2$ (um ponto de \omterm{def:b2:metric:continuity}{continuidade}) transforma a mesma série
na $\frac\pi4$ de Leibniz. Uma série, dois comportamentos ---
exatamente as duas cláusulas do teorema.
\end{example}

\begin{theorem}[Convergência normal para $C^1$; Parseval]\label{thm:b2:fourier:parseval}
\begin{enumerate}
  \item Se $f$ é \omterm{def:b2:metric:continuity}{contínua}, $2\pi$-periódica e $C^1$ por partes,
        então $c_n(f') = \iu n\,c_n(f)$, a série de Fourier de $f$
        converge \emph{\omterm{def:b2:funcseq:series}{normalmente}} em $\R$, e sua soma é $f$.
  \item (Parseval) Para toda $f$ $2\pi$-periódica \omterm{def:b2:metric:continuity}{contínua} por
        partes:\index{identidade de Parseval}
        \[
        \frac{1}{2\pi}\int_{-\pi}^{\pi}\abs{f}^2
        = \sum_{n=-\infty}^{\infty} \abs{c_n(f)}^2
        = \frac{\abs{a_0}^2}{4} + \frac12\sum_{n\geq1}
        \bigl(\abs{a_n}^2 + \abs{b_n}^2\bigr).
        \]
        \emph{(Demonstrado aqui para $f$ \omterm{def:b2:metric:continuity}{contínua} e $C^1$ por partes;
        admitido em geral.)}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Integração por partes em cada peça $C^1$ (os termos de bordo se cancelam
pela \omterm{def:b2:metric:continuity}{continuidade} e pela periodicidade): $c_n(f') = \iu n c_n(f)$. Então, por
Cauchy--Schwarz nas duas famílias quadraticamente \omterm{def:b2:series:summable}{somáveis}
(\cref{prop:b2:fourier:bessel} para $f'$):
\[
\sum_{n \neq 0} \abs{c_n(f)} = \sum_{n\neq0}
\frac{\abs{c_n(f')}}{\abs n}
\leq \Bigl(\sum \abs{c_n(f')}^2\Bigr)^{1/2}
\Bigl(\sum_{n\neq0}\frac{1}{n^2}\Bigr)^{1/2} < \infty :
\]
\omterm{def:b2:funcseq:series}{convergência normal} da série de Fourier. Sua soma é \omterm{def:b2:metric:continuity}{contínua} e
coincide com $f$ em todo ponto por Dirichlet
(\cref{thm:b2:fourier:dirichlet}: $f$ é \omterm{def:b2:metric:continuity}{contínua}): a série
converge para $f$, \omterm{def:b2:funcseq:def}{uniformemente}.

(2) Para tal $f$: $S_N f \to f$ \omterm{def:b2:funcseq:def}{uniformemente}, logo $\norm{f - S_Nf}_2
\leq \norm{f - S_Nf}_\infty \to 0$, e Pitágoras
($\norm f_2^2 = \sum_{\abs n \leq N}\abs{c_n}^2 + \norm{f -
S_Nf}_2^2$) passa ao limite. A forma real é contabilidade com
$a_n, b_n$.
\end{proof}

\begin{example}[A cota da cauda em $C^1$, tornada quantitativa]\label{ex:b2:fourier:tailbound}
A demonstração do~\cref{thm:b2:fourier:parseval}~(1) esconde uma
estimativa utilizável. Para $f$ \omterm{def:b2:metric:continuity}{contínua} e $C^1$ por partes, a mesma
Cauchy--Schwarz aplicada apenas à cauda dá
\[
\sum_{\abs n > N}\abs{c_n(f)}
= \sum_{\abs n > N}\frac{\abs{c_n(f')}}{\abs n}
\leq \Bigl(\sum_{\abs n > N}\abs{c_n(f')}^2\Bigr)^{\!1/2}
\Bigl(\sum_{\abs n>N}\frac{1}{n^2}\Bigr)^{\!1/2}
\leq \norm{f'}_2\,\sqrt{\frac{2}{N}} ,
\]
usando Bessel para $f'$ e $\sum_{n>N}n^{-2} \leq \frac1N$. Assim
o erro uniforme das somas parciais obedece a
\[
\norm{f - S_Nf}_\infty
\leq \sum_{\abs n>N}\abs{c_n(f)}
\leq \norm{f'}_2\,\sqrt{\frac2N} .
\]
Para $f(t) = \abs t$: $\norm{f'}_2 = 1$ (a derivada é
$\pm1$), de modo que dez termos já reconstroem $\abs t$ \omterm{def:b2:funcseq:def}{uniformemente}
a menos de $\sqrt{0.2} \approx 0.45$, e $N = 10^4$ a menos de
$0.015$. Lição final: uma derivada compra a taxa
uniforme $\frac{1}{\sqrt N}$; comparando com o mundo de coeficientes
$\frac1n$ da onda quadrada (sem \omterm{def:b2:funcseq:def}{convergência
uniforme} alguma), o dicionário do
\cref{ex:b2:fourier:decaytable} ganha números.
\end{example}

\begin{example}[Basileia e companhia]\label{ex:b2:fourier:basel}
Seja $f(t) = t$ em $\intoo{-\pi}{\pi}$, estendida
$2\pi$-periodicamente (uma dente de serra, $C^1$ por partes). Calculando, $a_n = 0$ (imparidade) e
\[
b_n = \frac{1}{\pi}\int_{-\pi}^{\pi} t\sin nt\,\dd t
= \frac{2(-1)^{n+1}}{n} .
\]
Dirichlet em $t = \frac\pi2$ recupera a série $\frac\pi4 = 1 -
\frac13 + \frac15 - \dots$ de Leibniz; Parseval dá
\[
\frac{1}{2\pi}\int_{-\pi}^{\pi} t^2\,\dd t = \frac{\pi^2}{3}
= \frac12\sum_{n\geq1}\frac{4}{n^2}
\quad\Longrightarrow\quad
\boxed{\;\sum_{n\geq1}\frac{1}{n^2} = \frac{\pi^2}{6}\;}
\]
--- a soma de Basileia de Euler, em duas linhas. A função $f(t) = t^2$
fornece igualmente $\sum \frac1{n^4} = \frac{\pi^4}{90}$
(\cref{exo:b2:fourier:3}).
\end{example}

\begin{example}[Um desenvolvimento completo com verificação embutida: $\abs{\sin t}$]\label{ex:b2:fourier:abssin}
A função $f(t) = \abs{\sin t}$ é \omterm{def:b2:metric:continuity}{contínua}, par,
$\pi$-periódica (logo $2\pi$-periódica) e $C^1$ por partes.
A paridade mata os $b_n$; $a_0 = \frac1\pi\int_{-\pi}^{\pi}
\abs{\sin t}\dd t = \frac4\pi$; e, para $n \geq 1$,
a fórmula produto-soma dá
\[
a_n = \frac2\pi\int_0^\pi \sin t\cos nt\,\dd t
= \frac{1}{\pi}\int_0^\pi\bigl(\sin(1+n)t +
\sin(1-n)t\bigr)\dd t
= \frac2\pi\cdot\frac{1 + \cos n\pi}{1 - n^2}
\]
para $n \neq 1$ (e $a_1 = 0$ diretamente): zero para $n$ ímpar, e
$a_{2k} = \frac{-4}{\pi(4k^2-1)}$. Pelo
\cref{thm:b2:fourier:parseval}~(1) a convergência é normal,
e
\[
\abs{\sin t} = \frac{2}{\pi} - \frac{4}{\pi}
\sum_{k\geq1}\frac{\cos(2kt)}{4k^2 - 1}
\qquad (t \in \R) .
\]
Verificação embutida em $t = 0$: a identidade exige que
$\sum_{k\geq1}\frac{1}{4k^2-1} = \frac12$, o que a telescopagem
confirma:
\[
\sum_{k\geq1}\frac{1}{4k^2-1}
= \frac12\sum_{k\geq1}\Bigl(\frac{1}{2k-1} -
\frac{1}{2k+1}\Bigr) = \frac12 . \checkmark
\]
Lição final: o \omterm{def:b2:reduction:eigen}{espectro} de $\abs{\sin}$ vive apenas nas
frequências \emph{pares} --- retificar um seno dobra seu
conteúdo de frequência, e é por isso que os retificadores de onda completa zumbem a
$100$ ou $120$ hertz, o dobro da frequência da rede.
\end{example}

\begin{example}[Parseval como dispositivo de cálculo]\label{ex:b2:fourier:zetafourodd}
Parseval transforma desenvolvimentos em séries numéricas no atacado.
Aplique-a a $f(t) = \abs t$ (\cref{exo:b2:fourier:2}: $a_0 =
\pi$, $a_n = \frac{-4}{\pi n^2}$ para $n$ ímpar, o resto nulo):
\[
\frac{1}{2\pi}\int_{-\pi}^{\pi}t^2\,\dd t = \frac{\pi^2}{3}
= \frac{a_0^2}{4} + \frac12\sum_{n \text{ ímpar}} a_n^2
= \frac{\pi^2}{4} +
\frac{8}{\pi^2}\sum_{n\text{ ímpar}}\frac{1}{n^4} ,
\]
donde
\[
\sum_{n\text{ ímpar}}\frac{1}{n^4}
= \frac{\pi^2}{8}\Bigl(\frac{\pi^2}{3} -
\frac{\pi^2}{4}\Bigr) = \frac{\pi^4}{96} .
\]
Confira contra o~\cref{exo:b2:fourier:3}: separando
$\sum\frac1{n^4}$ em partes ímpar e par obtém-se a contabilidade ao estilo $\zeta$
$S = S_{\mathrm{odd}} + \frac{S}{16}$, logo $S =
\frac{16}{15}\cdot\frac{\pi^4}{96} = \frac{\pi^4}{90}$ ---
exatamente o valor achado lá por uma função diferente. Dois
desenvolvimentos, um número: a coerência é a isometria de Parseval
em ação. Lição final: cada novo desenvolvimento de Fourier é uma
máquina geradora de identidades entre séries; a Parte II do problema
de fim de semana explica por que a máquina nunca pode
contradizer a si mesma.
\end{example}

\begin{example}[Translação e modulação]\label{ex:b2:fourier:shiftrules}
Duas regras de uma linha geram muitos desenvolvimentos a partir de um. Para $a
\in \R$, substituindo $s = t - a$:
\[
c_n\bigl(f(\cdot - a)\bigr)
= \frac{1}{2\pi}\int_{-\pi}^{\pi}f(t - a)\eu^{-\iu nt}\dd t
= \eu^{-\iu na}\,c_n(f)
\qquad\text{(a translação modula o espectro)},
\]
e, diretamente da definição,
\[
c_n\bigl(\eu^{\iu kt}f\bigr) = c_{n-k}(f)
\qquad\text{(a modulação translada o espectro)}.
\]
Instância trabalhada: a dente de serra deslocada de $\pi$, $g(t) = f(t -
\pi)$ com $f(t) = t$, tem coeficientes
$b_n$ iguais a $(-1)^n\cdot\frac{2(-1)^{n+1}}{n} = -\frac2n$: o desenvolvimento
$g \sim -2\sum\frac{\sin nt}{n}$ da dente de serra que salta em
$0$ em vez de em $\pi$ --- sem recalcular integral alguma. Lição
final: deslocamentos no tempo só giram fases, nunca amplitudes
($\abs{c_n}$ é invariante por deslocamento), e é por isso que a energia
(Parseval) e a classe de convergência são propriedades da \emph{forma}
do sinal, e não de onde o relógio começa.
\end{example}

\begin{remark}[Armadilhas comuns]
\emph{(i) Três convergências, três moedas:} a \omterm{def:b2:funcseq:def}{pontual}
(Dirichlet: precisa de $C^1$ por partes, paga o \emph{ponto médio} em
cada salto --- nunca o valor lateral), a uniforme (precisa de um
limite \omterm{def:b2:metric:continuity}{contínuo}; impossível através de um salto, sendo Gibbs o
sintoma visível) e a em média quadrática (Parseval: a mais
robusta, cega a pontos individuais). Diga sempre qual delas você
está afirmando. \emph{(ii) Nada de derivar termo a termo por
padrão:} derivar a série da dente de serra do
\cref{ex:b2:fourier:basel} termo a termo produz $\sum
2(-1)^{n+1}\cos nt$, cujos termos nem sequer tendem a $0$ ---
os teoremas de transferência do capítulo de sequências de funções precisam da
\omterm{def:b2:funcseq:def}{convergência uniforme} da série \emph{derivada}, que o salto
destrói. Suavidade primeiro, derivação depois
(o~\cref{exo:b2:fourier:6} é o dicionário). \emph{(iii)
Somas parciais \omterm{def:b2:quadratic:adjoint}{simétricas}:} o teorema de Dirichlet trata de $S_N =
\sum_{-N}^{N}$; reordenar ou somar primeiro um dos lados pode
transformar divergência em convergência e vice-versa. \emph{(iv)
Deriva de normalização:} as convenções diferem entre livros
($\frac{1}{2\pi}$ ou $\frac1\pi$ na frente, período $2\pi$ ou
$1$); os invariantes confiáveis são as relações de ortonormalidade
--- recalcule $\langle e_m, e_n\rangle$ na convenção em
uso antes de confiar em qualquer fórmula.
\end{remark}

\begin{remark}[Onde isso é usado]
Parseval é o germe da teoria $L^2$ das séries de Fourier: o
volume do terceiro ano de graduação \omterm{def:b2:metric:complete}{completa} o quadro (as exponenciais são uma
base hilbertiana de $L^2$, e a aplicação $f \mapsto (c_n)$ é uma
isometria bijetiva). Dentro deste volume, o problema de fim de semana
demonstra o \emph{teorema de Fejér} --- as médias de Cesàro da
série de Fourier convergem \omterm{def:b2:funcseq:def}{uniformemente} para toda $f$ periódica
\omterm{def:b2:metric:continuity}{contínua} --- o que eleva a Parseval geral admitida a
teorema, fornece o teorema de Weierstrass trigonométrico e paga
dois dividendos espetaculares: o teorema de equidistribuição de Weyl e
a desigualdade isoperimétrica. A matemática aplicada lê este
capítulo diariamente: \omterm{def:b2:reduction:eigen}{espectros} de sinais, harmônicos de sistemas
vibrantes e a transformada rápida de Fourier (o avatar finito foi o
\cref{exo:b2:hermitian:10}).
\end{remark}

\begin{remark}[Perspectivas dentro deste volume]
Três capítulos conversam com este. Para trás: o capítulo \omterm{def:b2:hermitian:adjoint}{hermitiano}
forneceu a geometria (famílias ortonormais,
projeções, Bessel), e o capítulo de sequências de funções, a
análise (\omterm{def:b2:funcseq:def}{convergência uniforme}, teoremas de transferência, identidades
aproximadas --- o núcleo de Fejér está para as séries de Fourier assim como
os \omterm{thm:b2:funcseq:weierstrass}{polinômios de Bernstein} estavam para Weierstrass). De lado: a
teoria de fronteira do capítulo de séries de potências volta pela \omterm{thm:b2:series:abel}{transformação
de Abel}, aqui realizada pelo núcleo de Poisson
(\cref{exo:b2:fourier:12}) --- com o raio $r$ do disco fazendo
o papel do parâmetro de somação. Para a frente: o
capítulo de equações diferenciais decompõe o forçamento periódico
em harmônicos e alimenta cada um à resposta em frequência do
oscilador; a ressonância acontece quando um modo de Fourier da entrada
casa com uma frequência natural, e é por isso que o problema de fim de semana
daquele capítulo e o deste são duas metades de uma só história.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:fourier:1}
Calcule os \omterm{def:b2:fourier:coefficients}{coeficientes de Fourier} da onda quadrada ($f = -1$ em
$\intoo{-\pi}{0}$, $+1$ em $\intoo{0}{\pi}$), enuncie a conclusão de
Dirichlet em $t = \frac\pi2$ e no salto $t = 0$, e recupere a
série de Leibniz.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:fourier:2}
Desenvolva $f(t) = \abs t$ ($\abs t \leq \pi$, $2\pi$-periódica) em
série de Fourier; justifique a convergência \emph{normal}; avalie em $t =
0$ para obter $\sum_{k\geq0}\frac{1}{(2k+1)^2} = \frac{\pi^2}{8}$
e redemonstre Basileia a partir disso.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:fourier:3}
Desenvolva $f(t) = t^2$ ($\abs t \leq \pi$) e deduza
\[
\sum_{n\geq1}\frac{(-1)^{n+1}}{n^2} = \frac{\pi^2}{12},
\qquad
\sum_{n\geq1}\frac{1}{n^4} = \frac{\pi^4}{90}
\quad\text{(Parseval)}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:4}
Seja $f$ \omterm{def:b2:metric:continuity}{contínua} $2\pi$-periódica com $c_n(f) = 0$ para todo
$n$. Prove que $f = 0$ \emph{(Parseval --- para qual classe ela está demonstrada
aqui? justifique que a \omterm{def:b2:metric:continuity}{continuidade} mais $C^1$ por partes podem ser dispensadas
admitindo a Parseval geral, ou dê o argumento de densidade em
esboço)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:5}
Para $\alpha \notin \Z$, desenvolva $f(t) = \cos(\alpha t)$ ($\abs t
\leq \pi$) e deduza a decomposição em frações parciais da
cotangente:
\[
\pi\cot(\pi\alpha) = \frac{1}{\alpha} + \sum_{n\geq1}
\frac{2\alpha}{\alpha^2 - n^2} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:6}
Prove que, se $f$ é $2\pi$-periódica e $C^k$ com $f^{(k)}$
\omterm{def:b2:metric:continuity}{contínua} por partes, então $c_n(f) = O\bigl(\abs n^{-k}\bigr)$:
a suavidade do sinal $=$ o decaimento de seu \omterm{def:b2:reduction:eigen}{espectro}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:fourier:7}
(Desigualdade de Wirtinger) Seja $f$ de classe $C^1$, $2\pi$-periódica, com
$\int_{-\pi}^{\pi} f = 0$. Prove que
\[
\int_{-\pi}^{\pi} \abs{f}^2 \leq \int_{-\pi}^{\pi} \abs{f'}^2 ,
\]
com igualdade se e somente se $f(t) = a\cos t + b\sin t$. \emph{(Parseval nos
dois lados; compare $\abs{c_n}^2$ e $n^2\abs{c_n}^2$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:fourier:8}
(A constante de Gibbs) Para a onda quadrada do
\cref{exo:b2:fourier:1}, avalie a soma parcial em $x_N =
\frac{\pi}{2N}$: escrevendo $u_k = \frac{(2k+1)\pi}{2N}$ e $\Delta u
= \frac{\pi}{N}$, mostre que
\[
S_{2N-1}\Bigl(\frac{\pi}{2N}\Bigr)
= \frac{4}{\pi}\sum_{k=0}^{N-1} \frac{\sin u_k}{2k+1}
= \frac{2}{\pi}\sum_{k=0}^{N-1} \frac{\sin u_k}{u_k}\,\Delta u
\xrightarrow[N\to\infty]{}
\frac{2}{\pi}\int_0^{\pi}\frac{\sin u}{u}\,\dd u \approx 1.179 :
\]
uma soma de Riemann de $\frac{2}{\pi}\cdot\frac{\sin u}{u}$ em
$\intcc{0}{\pi}$ nos pontos médios. Conclua que a ultrapassagem além
do valor do salto $1$ não desaparece quando $N \to \infty$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:fourier:9}
Desenvolva $\cos^3 t$ e $\sin^2 t\,\cos t$ em série de Fourier
\emph{(linearize; um polinômio trigonométrico é sua própria série de
Fourier, pela unicidade dos coeficientes)}. Quais são $c_n$,
$a_n$, $b_n$ para cada uma?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:10}
Sejam $a > 0$ e $f(t) = \eu^{at}$ em $\intoc{-\pi}{\pi}$,
estendida $2\pi$-periodicamente. Calcule
\[
c_n(f) = \frac{(-1)^n\sinh(a\pi)}{\pi(a - \iu n)} ,
\]
aplique o teorema de Dirichlet no salto $t = \pi$ e deduza a
decomposição em frações parciais da cotangente hiperbólica:
\[
\coth(\pi a) = \frac{1}{\pi a} + \sum_{n\geq1}
\frac{2a}{\pi(a^2 + n^2)} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:11}
(Convolução) Para $f, g$ \omterm{def:b2:metric:continuity}{contínuas} e $2\pi$-periódicas, defina
\[
(f * g)(x) = \frac{1}{2\pi}\int_{-\pi}^{\pi} f(x - t)\,g(t)\,
\dd t .
\]
Mostre que $f * g = g * f$, que $c_n(f*g) = c_n(f)\,c_n(g)$
\emph{(para trocar as duas integrais de um integrando \omterm{def:b2:metric:continuity}{contínuo},
compare as duas funções do limite superior: ambas se anulam na
extremidade esquerda e têm a mesma derivada, pela \omterm{def:b2:metric:continuity}{continuidade} e pela
derivação sob o sinal de integral)}, e que $S_N(f) =
f * D_N$ para o \omterm{lem:b2:fourier:kernel}{núcleo de Dirichlet}. (As médias de
Fejér do problema de fim de semana também são convoluções, $\sigma_N(f) = f *
F_N$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:fourier:12}
(Núcleo de Poisson: médias de Abel das séries de Fourier) Para $0 \leq r <
1$ ponha $P_r(t) = \sum_{n\in\Z} r^{\abs n}\eu^{\iu nt}$.
\begin{enumerate}
  \item Some as duas séries geométricas e mostre que
        \[
        P_r(t) = \frac{1 - r^2}{1 - 2r\cos t + r^2} > 0,
        \qquad
        \frac{1}{2\pi}\int_{-\pi}^{\pi}P_r = 1 .
        \]
  \item Mostre que, para $\delta \leq \abs t \leq \pi$: $P_r(t)
        \leq \frac{1 - r^2}{1 - 2r\cos\delta + r^2} \to 0$ quando
        $r \to 1^-$, \omterm{def:b2:funcseq:def}{uniformemente}.
  \item Deduza que, para toda $f$ \omterm{def:b2:metric:continuity}{contínua} $2\pi$-periódica,
        as \emph{médias de Abel} $(f * P_r)(x) = \sum_n
        r^{\abs n}c_n(f)\,\eu^{\iu nx}$ convergem para $f$
        \omterm{def:b2:funcseq:def}{uniformemente} quando $r \to 1^-$ --- a irmã com parâmetro \omterm{def:b2:metric:continuity}{contínuo}
        do teorema de Fejér, e a encarnação de Fourier
        da \omterm{thm:b2:series:abel}{transformação de Abel} do capítulo de séries de
        potências.
\end{enumerate}
\end{exercise}

\section{Problema: o teorema de Fejér e seus dividendos}

\begin{problem}\label{pb:b2:fourier:1}
O teorema de Dirichlet precisa de $f$ $C^1$ por partes; para $f$ meramente
\omterm{def:b2:metric:continuity}{contínua}, as somas parciais $S_N(f)$ podem se comportar mal. A descoberta de
Fejér: suas \emph{médias de Cesàro} nunca se comportam. O motor é
a positividade do núcleo de Fejér, e a colheita é
imensa: aproximação trigonométrica uniforme (Weierstrass),
unicidade dos \omterm{def:b2:fourier:coefficients}{coeficientes de Fourier}, Parseval para toda função
deste capítulo (removendo o ``admitido'' do
\cref{thm:b2:fourier:parseval}), o teorema de equidistribuição de Weyl
e --- coroando um século de geometria --- a
desigualdade isoperimétrica.\index{teorema de Fejer@teorema de Fejér}\index{equidistribuição}\index{desigualdade isoperimétrica}
Em todo o problema, $f$ é $2\pi$-periódica e \omterm{def:b2:metric:continuity}{contínua} por partes,
e
\[
\sigma_N(f) = \frac{S_0(f) + S_1(f) + \dots +
S_{N-1}(f)}{N} .
\]

\medskip
\noindent\textbf{Parte I --- O núcleo de Fejér.}
\begin{enumerate}
  \item Mostre que $\sigma_N(f)(x) =
        \frac{1}{2\pi}\int_{-\pi}^{\pi}f(x+u)\,F_N(u)\,\dd u$
        com $F_N = \frac{D_0 + \dots + D_{N-1}}{N}$, e
        que $\frac{1}{2\pi}\int_{-\pi}^{\pi}F_N = 1$.
  \item Demonstre a forma fechada, para $u \notin 2\pi\Z$:
        \[
        F_N(u) = \frac{1}{N}\,
        \frac{\sin^2\bigl(\frac{Nu}{2}\bigr)}
        {\sin^2\bigl(\frac u2\bigr)} \;\geq\; 0
        \]
        \emph{(some $\sin\bigl((n+\frac12)u\bigr)$ como a
        parte imaginária de uma série geométrica)}.
  \item Mostre a estimativa de concentração: para $0 < \delta \leq
        \abs u \leq \pi$,
        \[
        F_N(u) \leq \frac{1}{N\sin^2\frac\delta2}
        \xrightarrow[N\to\infty]{} 0
        \quad\text{uniformemente} :
        \]
        $(F_N)$ é uma identidade aproximada positiva.
  \item (Teorema de Fejér) Prove: se $f$ é \omterm{def:b2:metric:continuity}{contínua} e
        $2\pi$-periódica, então $\sigma_N(f) \to f$
        \emph{\omterm{def:b2:funcseq:def}{uniformemente}} em $\R$ \emph{(separe a integral de
        $\bigl(f(x+u) - f(x)\bigr)F_N(u)$ em $\abs u =
        \delta$; use Heine e as questões 1--3)}.
  \item Para $f$ \omterm{def:b2:metric:continuity}{contínua} por partes, mostre a versão \omterm{def:b2:funcseq:def}{pontual}
        $\sigma_N(f)(x) \to \frac{f(x^+) +
        f(x^-)}{2}$ em todo $x$, e a cota uniforme
        $\norm{\sigma_N(f)}_\infty \leq \norm f_\infty$
        \emph{(positividade!)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Weierstrass, unicidade,
Parseval.}
\begin{enumerate}[resume]
  \item (Weierstrass trigonométrico) Deduza: toda função \omterm{def:b2:metric:continuity}{contínua}
        $2\pi$-periódica é limite uniforme de
        polinômios trigonométricos.
  \item (Unicidade) Deduza: uma $f$ \omterm{def:b2:metric:continuity}{contínua} com $c_n(f) = 0$
        para todo $n$ é identicamente nula --- duas funções
        periódicas \omterm{def:b2:metric:continuity}{contínuas} com os mesmos \omterm{def:b2:fourier:coefficients}{coeficientes de Fourier}
        coincidem (\cref{exo:b2:fourier:4}, agora sem
        admissão alguma).
  \item (Parseval, caso \omterm{def:b2:metric:continuity}{contínuo}) Usando a propriedade de projeção
        de $S_N$ (\cref{prop:b2:fourier:bessel}) e
        $\sigma_Nf \in \mathcal T_{N-1} \subseteq \mathcal
        T_N$, prove que
        \[
        \norm{f - S_Nf}_2 \leq \norm{f - \sigma_Nf}_2
        \leq \norm{f - \sigma_Nf}_\infty
        \xrightarrow[N\to\infty]{} 0 ,
        \]
        e conclua a identidade de Parseval para toda $f$
        \emph{\omterm{def:b2:metric:continuity}{contínua}} $2\pi$-periódica.
  \item (Parseval, caso \omterm{def:b2:metric:continuity}{contínuo} por partes) Dada $f$
        \omterm{def:b2:metric:continuity}{contínua} por partes e $\varepsilon > 0$, construa uma
        $g$ periódica \omterm{def:b2:metric:continuity}{contínua} com $\norm{f - g}_2 \leq
        \varepsilon$ \emph{(substitua $f$ por uma
        interpolação afim em intervalos minúsculos em torno dos saltos)}
        e deduza $\norm{f - S_Nf}_2 \to 0$
        \emph{(use Bessel: $\norm{S_Nh}_2 \leq \norm h_2$)}:
        Parseval vale na generalidade plena enunciada no
        \cref{thm:b2:fourier:parseval} --- o ``admitido''
        acabou.
  \item (Sem Gibbs para Fejér) Contraste com o
        \cref{exo:b2:fourier:8}: mostre que, para a onda quadrada
        $f$, $\abs{\sigma_N(f)} \leq 1$ em toda parte, para todo
        $N$ --- a promediação de Cesàro apaga a ultrapassagem que
        assombra $S_N$. Explique em uma frase qual propriedade
        de $F_N$ é a responsável.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Taxas.}
\begin{enumerate}[resume]
  \item Demonstre as duas cotas do núcleo, para $0 < \abs u \leq
        \pi$:
        \[
        F_N(u) \leq N,
        \qquad
        F_N(u) \leq \frac{\pi^2}{N u^2}
        \]
        \emph{(para a primeira, $\abs{\sin N\theta} \leq
        N\abs{\sin\theta}$ por indução; para a segunda,
        $\sin\frac u2 \geq \frac{u}{\pi}$ em
        $\intcc{0}{\pi}$)}.
  \item Deduza a estimativa do primeiro momento
        \[
        \frac{1}{2\pi}\int_{-\pi}^{\pi}\abs u\,F_N(u)\,\dd u
        \;\leq\; \frac{C\,\ln N}{N}
        \qquad (N \geq 2)
        \]
        para uma constante explícita \emph{(separe em $\abs u =
        \frac1N$)}.
  \item Conclua: se $f$ é $L$-lipschitziana e $2\pi$-periódica,
        então
        \[
        \norm{\sigma_N f - f}_\infty \leq \frac{C\,L\ln N}{N} .
        \]
  \item (Saturação) Calcule $\sigma_N(e_1)$ para $e_1(t) =
        \eu^{\iu t}$ e mostre que $\norm{\sigma_Ne_1 - e_1}_\infty
        = \frac1N$: mesmo para as funções mais suaves, Fejér
        não converge mais depressa que $\frac1N$ --- o análogo exato
        da saturação de Bernstein no problema de fim de semana do
        capítulo de sequências de funções.
  \item (Localização) Mostre: se $f$ (\omterm{def:b2:metric:continuity}{contínua} por partes)
        se anula em $\intoo{x - \delta}{x + \delta}$, então
        $\sigma_N(f)(x) \to 0$, por mais selvagem que $f$ seja fora dali
        --- a convergência das médias em $x$ só enxerga $f$ perto de
        $x$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O teorema de equidistribuição de Weyl.}
Uma sequência $(x_n)_{n\geq1}$ em $\intco{0}{1}$ é
\emph{equidistribuída} quando, para todo intervalo
$\intcc{a}{b} \subseteq \intcc{0}{1}$,
\[
\frac{\#\{n \leq N : x_n \in \intcc ab\}}{N}
\xrightarrow[N\to\infty]{} b - a .
\]
\begin{enumerate}[resume]
  \item Mostre que $(x_n)$ é equidistribuída assim que
        $\frac1N\sum_{n\leq N}f(x_n) \to \int_0^1 f$ para toda $f$
        \emph{\omterm{def:b2:metric:continuity}{contínua}} $1$-periódica \emph{(comprima a
        indicadora de $\intcc ab$ entre duas funções afins por partes
        \omterm{def:b2:metric:continuity}{contínuas} cujas integrais diferem de
        $\varepsilon$)}.
  \item (Critério de Weyl, suficiência) Suponha
        \[
        \frac{1}{N}\sum_{n=1}^{N}\eu^{2\iu\pi kx_n}
        \xrightarrow[N\to\infty]{} 0
        \qquad\text{para todo } k \in \Z\setminus\{0\} .
        \]
        Mostre $\frac1N\sum f(x_n) \to \int_0^1f$ primeiro para
        polinômios trigonométricos e depois para toda $f$ \omterm{def:b2:metric:continuity}{contínua}
        $1$-periódica pela questão 6 (transportada para o período
        $1$): com a questão 16, $(x_n)$ é equidistribuída.
  \item Seja $\alpha$ irracional e $x_n = \{n\alpha\}$
        (parte fracionária). Majore a soma geométrica
        \[
        \Bigl|\sum_{n=1}^{N}\eu^{2\iu\pi kn\alpha}\Bigr|
        \leq \frac{2}{\abs{1 - \eu^{2\iu\pi k\alpha}}}
        \qquad (k \neq 0),
        \]
        e conclua o \emph{teorema de Weyl}: $(\{n\alpha\})$ é
        equidistribuída em $\intco{0}{1}$.
  \item Deduza que $(\{n\alpha\})$ é denso em
        $\intcc{0}{1}$ para $\alpha$ irracional, e explique em
        uma frase por que a equidistribuição é estritamente mais forte
        que a densidade.
  \item (Primeiros algarismos) Prove que a proporção de inteiros
        $n \leq N$ tais que $2^n$ tem primeiro algarismo (decimal)
        $1$ tende a $\log_{10}2 \approx 0.301$ \emph{(primeiro
        algarismo $1$ significa $\{n\log_{10}2\} \in
        \intco{0}{\log_{10}2}$; mostre que $\log_{10}2$ é
        irracional)}.
\end{enumerate}

\noindent\textbf{Parte V --- A desigualdade isoperimétrica.} Seja
$\Gamma$ uma curva fechada simples $C^1$ de comprimento $L$ que encerra
uma área com sinal $A$, parametrizada por comprimento de arco reescalado: $z(t) =
x(t) + \iu y(t)$, $2\pi$-periódica, com $\abs{z'(t)} =
\frac{L}{2\pi}$ constante; a área encerrada é
\[
A = \frac12\int_0^{2\pi}\bigl(x\,y' - y\,x'\bigr)\dd t
= \frac{1}{2}\,\Im\int_0^{2\pi}\conj{z}\,z'\,\dd t
\]
(tomada aqui como definição da área com sinal; o capítulo
sobre integrais múltiplas demonstra que ela coincide com a intuitiva,
via a fórmula de Green).
\begin{enumerate}[resume]
  \item Desenvolva $z(t) = \sum_{n\in\Z}c_n\eu^{\iu nt}$ (a
        série de uma função $C^1$, \omterm{def:b2:funcseq:series}{normalmente} convergente) e
        prove, por Parseval aplicada a $z'$:
        \[
        \frac{L^2}{2\pi} = \int_0^{2\pi}\abs{z'}^2\dd t
        = 2\pi\sum_{n\in\Z}n^2\abs{c_n}^2 .
        \]
  \item Prove igualmente que $A = \pi\sum_{n\in\Z}
        n\,\abs{c_n}^2$ \emph{(Parseval em sua forma polarizada:
        $\frac{1}{2\pi}\int\conj f g = \sum
        \conj{c_n(f)}c_n(g)$, aplicada a $f = z$, $g = z'$)}.
  \item (Hurwitz) Conclua:
        \[
        L^2 - 4\pi A = 4\pi^2\sum_{n\in\Z}
        (n^2 - n)\abs{c_n}^2 \;\geq\; 0 ,
        \]
        com igualdade se e somente se $z(t) = c_0 +
        c_1\eu^{\iu t}$ --- um círculo. A desigualdade
        isoperimétrica: entre as curvas fechadas de comprimento $L$, só
        o círculo encerra área $\frac{L^2}{4\pi}$.
  \item Verificações de bom senso: confira a igualdade para o círculo de
        raio $R$ e a desigualdade estrita para o quadrado de
        lado $a$; explique por que $n^2 - n \geq 0$ para todo
        inteiro $n$, inclusive os negativos, e onde a
        velocidade constante da parametrização foi usada.
  \item Síntese. Uma frase para cada: (i) a única
        propriedade de $F_N$ da qual fluem as Partes I--III e
        que falta a $D_N$; (ii) como a somação de Cesàro aqui
        se relaciona com o problema de fim de semana do capítulo de séries de potências
        (Frobenius); (iii) qual dividendo usou apenas
        Weierstrass (questão 6) e qual precisou da Parseval
        \omterm{def:b2:metric:complete}{completa}; (iv) uma frase sobre o que o volume do terceiro ano de graduação
        acrescenta (a \omterm{def:b2:metric:complete}{completude} de $L^2$: as séries de Fourier como base
        hilbertiana).
\end{enumerate}
\end{problem}
