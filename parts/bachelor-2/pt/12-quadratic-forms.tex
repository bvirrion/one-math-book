\chapter{Formas quadráticas}\label{ch:b2:quadratic}

Uma \omterm{def:b2:quadratic:def}{forma quadrática} é a sombra algébrica de uma geometria: as partes de
assinatura zero achatam, as positivas curvam para um lado, as negativas para o
outro. Este capítulo reduz toda \omterm{def:b2:quadratic:def}{forma quadrática} real a uma soma de
$\pm$ quadrados (Gauss), demonstra que as contagens de sinais são intrínsecas
(Sylvester) e depois coroa a geometria euclidiana com o
\emph{teorema espectral}: os \omterm{def:b2:quadratic:adjoint}{endomorfismos simétricos} diagonalizam em
bases ortonormais --- o teorema mais usado de toda a álgebra
linear aplicada.

\section{Formas bilineares e quadráticas}

\begin{definition}\label{def:b2:quadratic:def}
Uma \emph{forma bilinear \omterm{def:b2:quadratic:adjoint}{simétrica}} num espaço vetorial real $E$ é uma
aplicação bilinear $\varphi \colon E \times E \to \R$ com $\varphi(x, y) =
\varphi(y, x)$; a \emph{forma
quadrática}\index{forma quadrática} associada é $q(x) = \varphi(x, x)$. A forma
$\varphi$ é recuperada a partir de $q$ por
\emph{polarização}\index{identidade de polarização}:
\[
\varphi(x, y) = \tfrac12\bigl(q(x + y) - q(x) - q(y)\bigr).
\]
Numa base $(e_i)$, a \emph{matriz} de $\varphi$ é a matriz \omterm{def:b2:quadratic:adjoint}{simétrica}
$B = (\varphi(e_i, e_j))$, com $q(x) = X^{\mathsf T} B X$; uma mudança
de base de matriz $P$ substitui $B$ por $P^{\mathsf T} B P$
(\emph{congruência}\index{matrizes congruentes} --- não semelhança!).
O \emph{posto} de $q$ é $\operatorname{rk} B$ (invariante:
a congruência multiplica por matrizes invertíveis).
\end{definition}

\begin{example}
Em $\R^2$: $q(x, y) = x^2 + 4xy + y^2$ tem matriz $\begin{pmatrix}
1 & 2\\ 2 & 1\end{pmatrix}$. Um produto interno é exatamente uma
\omterm{def:b2:quadratic:def}{forma bilinear simétrica} cuja \omterm{def:b2:quadratic:def}{forma quadrática} é positiva definida;
este capítulo estuda o caso geral, de sinal indefinido.
\end{example}

\begin{example}[Congruência em ação]\label{ex:b2:quadratic:congruence}
Tome $q(x, y) = x^2 + 4xy + y^2$ (matriz $B = \begin{pmatrix} 1
& 2\\ 2 & 1\end{pmatrix}$) e a nova base $e_1' = (1, 1)$,
$e_2' = (1, -1)$, isto é, $P = \begin{pmatrix} 1 & 1\\ 1 &
-1\end{pmatrix}$. Então
\[
P^{\mathsf T}BP
= \begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
\begin{pmatrix} 1 & 2\\ 2 & 1\end{pmatrix}
\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
= \begin{pmatrix} 6 & 0\\ 0 & -2\end{pmatrix} :
\]
nas coordenadas $(u, v)$ ao longo da nova base, $q = 6u^2 -
2v^2$ --- confira: $x = u + v$, $y = u - v$ dá $x^2 + 4xy +
y^2 = 6u^2 - 2v^2$ diretamente. Note que as novas entradas diagonais $6,
-2$ \emph{não} são os \omterm{def:b2:reduction:eigen}{autovalores} $3, -1$ de $B$: a \omterm{def:b2:quadratic:def}{congruência}
reescala, só a semelhança preserva \omterm{def:b2:reduction:eigen}{espectros} --- mas os sinais
coincidem, como exige o teorema de Sylvester. (A base aqui é
ortogonal mas não ortonormal; normalizá-la por
$\frac{1}{\sqrt2}$ dividiria a diagonal por $2$ e
recuperaria os \omterm{def:b2:reduction:eigen}{autovalores}.)
\end{example}

\begin{example}[Determinantes de Gram medem área]\label{ex:b2:quadratic:gramarea}
Para $v_1, v_2$ num espaço euclidiano, a matriz de Gram $G =
\bigl(\langle v_i, v_j\rangle\bigr)$ empacota comprimentos e
ângulo; seu \omterm{def:b2:linalg:det}{determinante} empacota a \emph{área}:
\[
\det G = \norm{v_1}^2\norm{v_2}^2 - \langle v_1, v_2\rangle^2
= \norm{v_1}^2\norm{v_2}^2\bigl(1 - \cos^2\theta\bigr)
= \bigl(\norm{v_1}\,\norm{v_2}\sin\theta\bigr)^2 ,
\]
o quadrado da área do paralelogramo sobre $v_1, v_2$ ---
e Cauchy--Schwarz é exatamente a afirmação $\det G \geq 0$.
Instância trabalhada: $v_1 = (1, 2, 2)$, $v_2 = (2, 1, -2)$ em
$\R^3$:
\[
G = \begin{pmatrix} 9 & 0\\ 0 & 9 \end{pmatrix},
\qquad
\det G = 81 :
\]
os vetores são ortogonais de comprimento $3$, gerando um
paralelogramo (aqui, um quadrado) de área $\sqrt{81} = 9$.
Lição final: nenhum produto vetorial e nenhuma mágica da dimensão $3$
foram usados --- $\sqrt{\det G}$ mede volume $k$-dimensional
em \emph{qualquer} dimensão, o ponto de partida da Parte I do problema
de fim de semana e das integrais de área de superfície mais adiante neste
volume.
\end{example}

\section{Redução de Gauss e a inércia de Sylvester}

\begin{theorem}[Redução de Gauss]\label{thm:b2:quadratic:gauss}
Toda \omterm{def:b2:quadratic:def}{forma quadrática} $q$ num espaço real de dimensão finita pode ser
escrita\index{redução de Gauss}
\[
q = \sum_{i=1}^{s} \ell_i^2 - \sum_{j=1}^{t} m_j^2 ,
\]
em que $\ell_1, \dots, \ell_s, m_1, \dots, m_t$ são formas lineares
linearmente independentes; equivalentemente, alguma base torna a
matriz de $q$ diagonal com entradas $+1$ ($s$ vezes), $-1$ ($t$
vezes), $0$.
\end{theorem}

\begin{proof}
Indução no número de variáveis, em coordenadas: $q(x_1,
\dots, x_n)$.

\emph{Caso 1: aparece algum quadrado}, digamos que o coeficiente $a$ de
$x_1^2$ seja não nulo. Agrupe todos os termos em $x_1$ e complete o
quadrado:
\[
q = a\Bigl(x_1 + \frac{1}{a}\,\lambda(x_2, \dots,
x_n)\Bigr)^{\!2} + q_1(x_2, \dots, x_n),
\]
em que $\lambda$ é linear e $q_1$ é quadrática nas variáveis restantes:
uma forma independente foi destacada (ela envolve $x_1$, as
outras não), a indução se aplica a $q_1$, e os sinais $\pm$ vêm
do sinal de $a$ após reescalar por $\sqrt{\abs a}$.

\emph{Caso 2: nenhum quadrado, mas algum termo cruzado}, digamos $b\,x_1x_2$ com
$b \neq 0$. Use a identidade
\[
x_1x_2 = \tfrac14\bigl((x_1 + x_2)^2 - (x_1 - x_2)^2\bigr)
\]
após agrupar: escrevendo $q = b\,x_1x_2 + x_1\alpha + x_2\beta +
q_2$ (com $\alpha, \beta, q_2$ nas outras variáveis), verifica-se que
\[
q = \frac{b}{4}\Bigl[\Bigl(x_1 + x_2 + \frac{\alpha +
\beta}{b}\Bigr)^{2} - \Bigl(x_1 - x_2 + \frac{\beta -
\alpha}{b}\Bigr)^{2}\Bigr] + \widetilde q ,
\]
com $\widetilde q$ livre de $x_1, x_2$: duas formas independentes são
destacadas, e a indução conclui.

Independência das formas colhidas: ordene os lotes como
produzidos. As formas do primeiro lote contêm $x_1$ (Caso 1)
ou $x_1, x_2$ (Caso 2); todas as formas posteriores são livres dessas
variáveis. Suponha que uma combinação linear de todas as formas colhidas
se anule. Lendo o coeficiente de $x_1$ (e de $x_2$): só
o primeiro lote contribui e, dentro desse lote, a uma ou
duas formas são visivelmente independentes ($\ell$ sozinha; ou $\ell \pm
m$ com $\ell, m$ independentes): os coeficientes do primeiro lote
se anulam. Retire o lote e repita: por indução ao longo dos
lotes, todos os coeficientes se anulam --- a família inteira é livre,
a triangularidade tornada explícita.
\end{proof}

\begin{theorem}[Lei de inércia de Sylvester]\label{thm:b2:quadratic:sylvester}
O par $(s, t)$ em \cref{thm:b2:quadratic:gauss} depende apenas de
$q$, e não da redução: é a
\emph{assinatura}\index{assinatura de uma forma quadrática} de $q$.
Além disso,
\[
s = \max\{\dim F : q|_F \text{ positiva definida}\},
\]
e simetricamente para $t$.
\end{theorem}

\begin{proof}
Seja $q = \sum_{i \leq s}\ell_i^2 - \sum_{j\leq t} m_j^2$ e seja
$F_+$ o espaço \omterm{def:b2:structures:generated}{gerado} pelos vetores (pré-)\omterm{def:b2:linalg:dual}{duais} sobre os quais $(\ell_i)$
se restringem a coordenadas --- concretamente: complete a
família independente $(\ell_1, \dots, \ell_s, m_1, \dots, m_t)$
numa base do \omterm{def:b2:linalg:dual}{dual} $E^*$ e seja $(u_1, \dots, u_n)$
a base de $E$ cujas formas coordenadas são essas (a base
\omterm{def:b2:linalg:dual}{pré-dual}: $\ell_i(u_k) = \delta_{ik}$ para $k \leq s$,
e as formas posteriores se anulam nos vetores anteriores). Ponha $F_+ =
\operatorname{Vect}(u_1, \dots, u_s)$: para $x = \sum_{i\leq
s}x_iu_i \in F_+$,
\[
\ell_i(x) = x_i,
\qquad m_j(x) = 0,
\qquad\text{logo}\qquad
q(x) = \sum_{i\leq s}x_i^2 > 0 \quad (x \neq 0) :
\]
$q|_{F_+}$ é positiva definida e o máximo na fórmula exibida
vale $\geq s$. Reciprocamente, seja $F$ um subespaço qualquer com $q|_F$
positiva definida, e $G = \{x : \ell_1(x) = \dots = \ell_s(x) =
0\}$, de codimensão $\leq s$; em $G$, $q(x) = -\sum m_j^2 \leq 0$.
Então $F \cap G = \{0\}$ (um vetor não nulo aí teria $q > 0$
e $q \leq 0$), logo $\dim F \leq \dim E - \dim G \leq s$. Portanto o
máximo vale $s$ para \emph{toda} redução: $s$ é intrínseco, e $t
= \operatorname{rk} q - s$ do mesmo modo.
\end{proof}

\begin{example}\label{ex:b2:quadratic:gaussexample}
$q(x, y, z) = xy + yz + zx$ (sem quadrados). Com $x_1 = x$, $x_2 =
y$: $q = xy + z(x + y)$, e a identidade dos dois quadrados dá
\[
q = \tfrac14(x + y + 2z)^2 - \tfrac14(x - y)^2 - z^2 ,
\]
(desenvolva para conferir). Três formas independentes: assinatura $(1, 2)$,
posto $3$. Uma direção positiva, duas negativas: a geometria do ``cone
de luz'' desta forma.
\end{example}

\begin{example}[Uma forma degenerada, reduzida por inteiro]\label{ex:b2:quadratic:degenerate}
$q(x, y, z) = xy + yz$ em $\R^3$: sem quadrados, logo Caso 2 com o
agrupamento $q = y(x + z)$. A identidade dos dois quadrados no produto
das formas independentes $y$ e $x + z$:
\[
q = \frac14\bigl(y + x + z\bigr)^2 -
\frac14\bigl(y - x - z\bigr)^2 .
\]
As duas formas lineares $y + x + z$ e $y - x - z$ são
independentes (sua diferença é $2(x+z)$, sua soma $2y$), de modo que
Sylvester informa: assinatura $(1, 1)$, posto $2$ ---
\emph{degenerada}. O núcleo da forma polar acha-se
resolvendo $\varphi(v, \cdot) = 0$: com a matriz
$\frac12\begin{pmatrix} 0&1&0\\ 1&0&1\\ 0&1&0\end{pmatrix}$,
o núcleo é $\{y = 0,\ x + z = 0\} = \R\,(1, 0, -1)$, a
direção ao longo da qual $q$ nada enxerga. Lição final: a
deficiência de posto aparece em Gauss como ``ficar sem variáveis''
--- a redução produziu apenas dois quadrados em três
dimensões, e a dimensão que falta é exatamente o núcleo.
\end{example}

\begin{example}[Uma forma, dois caminhos até a assinatura]\label{ex:b2:quadratic:tworoads}
$q(x, y, z) = 2x^2 + 2y^2 + 2z^2 + 2xy + 2yz$, matriz
$\begin{pmatrix} 2 & 1 & 0\\ 1 & 2 & 1\\ 0 & 1 & 2
\end{pmatrix}$. \emph{Caminho 1, Gauss:} complete quadrados em
ordem,
\[
q = 2\Bigl(x + \frac y2\Bigr)^{\!2} + \frac32 y^2 + 2yz + 2z^2
= 2\Bigl(x + \frac y2\Bigr)^{\!2}
+ \frac32\Bigl(y + \frac{2z}{3}\Bigr)^{\!2} + \frac43 z^2 :
\]
três quadrados positivos sobre formas independentes, assinatura $(3,
0)$: positiva definida. \emph{Caminho 2, \omterm{def:b2:reduction:eigen}{autovalores}:} a matriz
é a tridiagonal $2I + N$, com $N$ a matriz de vizinhança; seus
\omterm{def:b2:reduction:eigen}{autovalores} são $2 + \sqrt2$, $2$, $2 - \sqrt2$ (confira os
\omterm{def:b2:reduction:eigen}{autovetores} $(1, \pm\sqrt2, 1)$ e $(1, 0, -1)$), todos
positivos: mesmo veredicto, pelo
\cref{cor:b2:quadratic:principalaxes}. Gauss é mais rápido;
os \omterm{def:b2:reduction:eigen}{autovalores} dizem mais (dão os eixos principais e os
valores extremos de $q$ na esfera). Lição final: os
pivôs positivos $2, \frac32, \frac43$ de Gauss são exatamente as
razões $\frac{\Delta_k}{\Delta_{k-1}}$ de menores principais
dominantes ($\Delta_1 = 2$, $\Delta_2 = 3$, $\Delta_3 = 4$) --- o
problema de fim de semana demonstra isso em geral.
\end{example}

\begin{method}[Calcular uma assinatura: três caminhos]\label{met:b2:quadratic:signature}
\begin{enumerate}
  \item \emph{Gauss} (sempre funciona, o mais rápido à mão): complete
        quadrados em ordem, Caso 2 quando não houver quadrado disponível;
        conte os sinais. Verifique que as formas lineares colhidas
        são independentes --- menos formas que variáveis significa
        núcleo (\cref{ex:b2:quadratic:degenerate}).
  \item \emph{Menores dominantes} (para testes de definição): todos os
        $\Delta_k > 0$ se e somente se positiva definida
        (\cref{exo:b2:quadratic:8}); os pivôs
        $\Delta_k/\Delta_{k-1}$ dão até os coeficientes
        de Gauss (problema de fim de semana). Falha em silêncio se algum
        $\Delta_k = 0$: recorra ao caminho 1.
  \item \emph{\omterm{def:b2:reduction:eigen}{Autovalores}} (o mais informativo, o mais caro):
        sinais do \omterm{def:b2:reduction:eigen}{espectro}
        (\cref{cor:b2:quadratic:principalaxes}); também fornece
        os eixos principais e os valores extremos de $q$ na
        esfera unitária. Prefira-o quando a estrutura própria for
        necessária de qualquer modo.
\end{enumerate}
\end{method}

\section{O teorema espectral}

Seja agora $E$ euclidiano (produto interno $\langle\cdot,\cdot\rangle$,
volume do primeiro ano de graduação).

\begin{definition}[Adjunto; endomorfismos simétricos]\label{def:b2:quadratic:adjoint}
Para $u \in \mathcal{L}(E)$, o \emph{adjunto}\index{adjunto}
$u^*$ é o único endomorfismo com
\[
\langle u(x), y\rangle = \langle x, u^*(y)\rangle
\qquad (x, y \in E);
\]
numa base ortonormal, $\operatorname{Mat}(u^*) =
\operatorname{Mat}(u)^{\mathsf T}$. $u$ é
\emph{simétrico}\index{endomorfismo simétrico} (autoadjunto) quando
$u^* = u$ --- equivalentemente, sua matriz numa base ortonormal é
simétrica.
\end{definition}

\begin{proof}[Existência e unicidade do adjunto]
Para $y$ fixo, a forma $x \mapsto \langle u(x), y\rangle$ é
linear, logo (dimensão finita) da forma $\langle x,
z_y\rangle$ para um único $z_y$ --- sendo a aplicação $y \mapsto z_y =:
u^*(y)$ linear pela unicidade. A identificação matricial:
$\langle u(e_i), e_j\rangle$ lido nos dois sentidos.
\end{proof}

\begin{example}[O adjunto depende do produto interno]\label{ex:b2:quadratic:weightedadjoint}
Em $\R^2$ tome o produto interno \emph{ponderado} $\langle x,
y\rangle_D = x_1y_1 + 2x_2y_2$ (matriz $D =
\operatorname{diag}(1,2)$) e $u$ de matriz $A =
\begin{pmatrix} 0 & 1\\ 0 & 0\end{pmatrix}$ na base
canônica. De $\langle u(x), y\rangle_D = (Ax)^{\mathsf T}Dy =
x^{\mathsf T}(A^{\mathsf T}D)y$ e $\langle x, u^*(y)\rangle_D
= x^{\mathsf T}(DA^*)y$, a matriz do adjunto é
\[
A^* = D^{-1}A^{\mathsf T}D
= \begin{pmatrix} 1 & 0\\ 0 & \tfrac12\end{pmatrix}
\begin{pmatrix} 0 & 0\\ 1 & 0\end{pmatrix}
\begin{pmatrix} 1 & 0\\ 0 & 2\end{pmatrix}
= \begin{pmatrix} 0 & 0\\ \tfrac12 & 0\end{pmatrix}
\neq A^{\mathsf T} .
\]
Verificação de bom senso em $x = (1,0)$, $y = (0,1)$:
\[
\langle u(x), y\rangle_D = \langle (0,0), y\rangle_D = 0,
\quad
\langle x, u^*(y)\rangle_D
= \langle(1,0), (0,\tfrac12)\rangle_D = 0 ;
\]
em $x = (0,1)$, $y = (1,0)$:
\[
\langle u(x), y\rangle_D = \langle(1,0),(1,0)\rangle_D = 1,
\quad
\langle x, u^*(y)\rangle_D =
\langle(0,1),(0,\tfrac12)\rangle_D = 1 .
\] Lição final: ``$\operatorname{Mat}(u^*)
= \operatorname{Mat}(u)^{\mathsf T}$'' é um enunciado sobre
bases \emph{ortonormais} apenas; em geral a métrica $D$
intervém, exatamente como na redução simultânea do
problema de fim de semana.
\end{example}

\begin{theorem}[Teorema espectral]\label{thm:b2:quadratic:spectral}
Seja $u$ um \omterm{def:b2:quadratic:adjoint}{endomorfismo simétrico} de um espaço euclidiano $E$. Então
$E$ tem uma \emph{base ortonormal de \omterm{def:b2:reduction:eigen}{autovetores}} de $u$; todos os
\omterm{def:b2:reduction:eigen}{autovalores} são reais, e os \omterm{def:b2:reduction:eigen}{autoespaços} de \omterm{def:b2:reduction:eigen}{autovalores} distintos são
ortogonais. Forma matricial: toda matriz real \omterm{def:b2:quadratic:adjoint}{simétrica} $A$ escreve-se
\[
A = P\,D\,P^{\mathsf T},
\qquad P \text{ ortogonal } (P^{\mathsf T} P = I),\ D
\text{ diagonal}.\index{teorema espectral}
\]
\end{theorem}

\begin{proof}
\emph{Existe um \omterm{def:b2:reduction:eigen}{autovetor}.} A função $x \mapsto \langle u(x),
x\rangle$ é \omterm{def:b2:metric:continuity}{contínua} na esfera unitária $S$ de $E$, que é
\omterm{def:b2:metric:compact}{compacta} (dimensão finita, \cref{thm:b2:nvs:finitedim}): ela atinge
seu máximo $\lambda$ em algum $a \in S$. Afirmamos que $u(a) = \lambda
a$. Para todo $y \perp a$ com $\norm y = 1$ e $t \in \R$, o
vetor $x_t = \frac{a + ty}{\sqrt{1 + t^2}}$ está em $S$
($\norm{a + ty}^2 = 1 + t^2$ por Pitágoras); desenvolvendo
a função maximizada,
\[
g(t) = \langle u(x_t), x_t\rangle
= \frac{\langle u(a), a\rangle + 2t\langle u(a), y\rangle +
t^2\langle u(y), y\rangle}{1 + t^2}
\]
(a simetria de $u$ fundiu os dois termos cruzados: $\langle u(a),
y\rangle = \langle a, u(y)\rangle = \langle u(y), a\rangle$).
$g$ é uma função derivável de $t$ com máximo em $t =
0$; a regra do quociente em $0$ dá
\[
g'(0) = \frac{2\langle u(a), y\rangle\cdot 1 - \langle u(a),
a\rangle\cdot 0}{1} = 2\langle u(a), y\rangle = 0 .
\]
Logo $u(a)$ é ortogonal a todo o hiperplano $a^\perp$: $u(a)
\in (a^{\perp})^{\perp} = \R a$, isto é, $u(a) = \mu a$; e $\mu =
\langle u(a), a\rangle = \lambda$.

\emph{Indução.} O complemento ortogonal $F = a^\perp$ é
estável por $u$: para $x \perp a$, $\langle u(x), a\rangle = \langle x,
u(a)\rangle = \lambda\langle x, a\rangle = 0$. A restrição
$u|_F$ é \omterm{def:b2:quadratic:adjoint}{simétrica} para o produto interno induzido; por indução sobre
a dimensão, $F$ tem base ortonormal de \omterm{def:b2:reduction:eigen}{autovetores}; acrescente $a$ na frente.

\emph{Complementos.} Os \omterm{def:b2:reduction:eigen}{autovalores} são os números reais $\langle u(e),
e\rangle$ na base de \omterm{def:b2:reduction:eigen}{autovetores}. Ortogonalidade dos \omterm{def:b2:reduction:eigen}{autoespaços}: $u(x) =
\lambda x$, $u(y) = \mu y$ dão $\lambda\langle x, y\rangle =
\langle u(x), y\rangle = \langle x, u(y)\rangle = \mu\langle x,
y\rangle$, logo $\langle x, y\rangle = 0$ quando $\lambda \neq \mu$.
Forma matricial: as colunas de $P$ são a base ortonormal de \omterm{def:b2:reduction:eigen}{autovetores}.
\end{proof}

\begin{example}[Uma execução espectral completa]\label{ex:b2:quadratic:fullrun}
Diagonalize ortogonalmente $A = \begin{pmatrix} 1 & 2\\ 2 &
1\end{pmatrix}$. \omterm{def:b2:reduction:charpoly}{Polinômio característico} $(1 - \lambda)^2 -
4$: \omterm{def:b2:reduction:eigen}{autovalores} $3$ e $-1$. \omterm{def:b2:reduction:eigen}{Autovetores}: $(A - 3I)v = 0$
dá $v_1 = \frac{1}{\sqrt2}(1,1)$; $(A + I)v = 0$ dá $v_2
= \frac{1}{\sqrt2}(1,-1)$ --- ortogonais, como o
\cref{thm:b2:quadratic:spectral} garante sem
cálculo. Com $P = (v_1\ v_2)$ (uma rotação de
$\frac\pi4$):
\[
P^{\mathsf T}AP = \begin{pmatrix} 3 & 0\\ 0 & -1
\end{pmatrix},
\qquad
x^2 + 4xy + y^2 = 3u^2 - v^2
\quad\text{no referencial girado} .
\]
Assim a forma de \cref{exo:b2:quadratic:1} é uma forma do tipo
hipérbole: assinatura $(1,1)$, coerente com sua \omterm{thm:b2:quadratic:gauss}{redução de Gauss}
$(x + 2y)^2 - 3y^2$ --- quadrados diferentes, mesma assinatura, como
exige Sylvester. Lição final: Gauss deu a resposta
mais depressa, mas o caminho espectral também informa que, no círculo
unitário, $q$ percorre exatamente $\intcc{-1}{3}$, atingido ao longo de
$v_2$ e $v_1$: o trabalho extra compra geometria.
\end{example}

\begin{corollary}[Eixos principais; testes de positividade]\label{cor:b2:quadratic:principalaxes}
\begin{enumerate}
  \item Toda \omterm{def:b2:quadratic:def}{forma quadrática} $q$ num espaço euclidiano diagonaliza
        em alguma base \emph{ortonormal}: $q(x) = \sum_i \lambda_i
        x_i^2$ com $\lambda_i$ os \omterm{def:b2:reduction:eigen}{autovalores} da matriz \omterm{def:b2:quadratic:adjoint}{simétrica}
        de $q$; a assinatura conta os \omterm{def:b2:reduction:eigen}{autovalores} positivos e negativos.
  \item Uma matriz \omterm{def:b2:quadratic:adjoint}{simétrica} é positiva semidefinida (resp.\
        definida) se e somente se todos os seus \omterm{def:b2:reduction:eigen}{autovalores} são $\geq 0$ (resp.\ $>
        0$); e então os valores extremos do quociente de Rayleigh
        são
        \[
        \min_{\norm x = 1} \langle Ax, x\rangle = \lambda_{\min},
        \qquad
        \max_{\norm x = 1} \langle Ax, x\rangle = \lambda_{\max} .
        \]
\end{enumerate}
\end{corollary}

\begin{proof}
(1) Escreva $q(x) = \langle A x, x\rangle$ com $A$ \omterm{def:b2:quadratic:adjoint}{simétrica} (a
matriz de $q$ numa base ortonormal); diagonalize $A$ pelo
teorema espectral: para $x = \sum x_ie_i$ na base ortonormal
de \omterm{def:b2:reduction:eigen}{autovetores},
\[
q(x) = \Bigl\langle \sum_i \lambda_ix_ie_i,\ \sum_j
x_je_j\Bigr\rangle = \sum_i \lambda_i x_i^2
\]
(a ortonormalidade mata os termos cruzados). Os sinais $\pm$ dos
$\lambda_i$ contam a assinatura por Sylvester: reescalar cada
coordenada por $\sqrt{\abs{\lambda_i}}$ exibe uma \omterm{thm:b2:quadratic:gauss}{redução de
Gauss} com formas independentes.

(2) Na base de \omterm{def:b2:reduction:eigen}{autovetores}, $\langle Ax, x\rangle = \sum \lambda_i
x_i^2$, preso entre $\lambda_{\min}\norm x^2$ e
$\lambda_{\max}\norm x^2$, com igualdade nos \omterm{def:b2:reduction:eigen}{autovetores}
correspondentes; a positividade de todos os \omterm{def:b2:reduction:eigen}{autovalores} é assim equivalente
à positividade da forma.
\end{proof}

\begin{example}\label{ex:b2:quadratic:spectralexample}
$A = \begin{pmatrix} 2 & 1\\ 1 & 2 \end{pmatrix}$: \omterm{def:b2:reduction:eigen}{autovalores} $3$
(\omterm{def:b2:reduction:eigen}{autovetor} $\frac{1}{\sqrt2}(1,1)$) e $1$
($\frac{1}{\sqrt2}(1,-1)$). A \omterm{def:b2:quadratic:def}{forma quadrática} $2x^2 + 2xy + 2y^2$
torna-se $3X^2 + Y^2$ no referencial ortonormal girado: os eixos
principais de uma elipse, calculados. A \omterm{thm:b2:quadratic:gauss}{redução de Gauss} também chega a uma forma
diagonal, mas só o teorema espectral chega a ela \emph{sem
distorcer comprimentos}.
\end{example}

\begin{example}[Uma elipse plenamente identificada]\label{ex:b2:quadratic:ellipserun}
Que curva é $5x^2 + 4xy + 2y^2 = 6$? A matriz
$\begin{pmatrix} 5 & 2\\ 2 & 2\end{pmatrix}$ tem \omterm{def:b2:reduction:charpoly}{polinômio
característico} $\lambda^2 - 7\lambda + 6 = (\lambda - 1)(\lambda -
6)$: \omterm{def:b2:reduction:eigen}{autovalores} $1$ e $6$, ambos positivos --- uma elipse.
\omterm{def:b2:reduction:eigen}{Autovetores} ortonormais: para $\lambda = 1$, resolva
$\begin{pmatrix} 4 & 2\\ 2 & 1\end{pmatrix}v = 0$: $v_1 =
\frac{1}{\sqrt5}(1, -2)$; para $\lambda = 6$: $v_2 =
\frac{1}{\sqrt5}(2, 1)$. Nas coordenadas giradas $(X, Y)$
ao longo de $(v_2, v_1)$ a equação torna-se
\[
6X^2 + Y^2 = 6,
\qquad\text{i.e.}\qquad
X^2 + \frac{Y^2}{6} = 1 :
\]
semieixos $1$ (ao longo de $v_2$) e $\sqrt6$ (ao longo de $v_1$).
Lição final: a forma grosseira saiu de graça --- $\det
= 6 > 0$ e traço positivo anunciam uma elipse antes de qualquer
\omterm{def:b2:reduction:eigen}{autovetor} ser calculado --- mas só o teorema espectral
entrega as direções e os comprimentos dos eixos, isto é, a
geometria de fato.
\end{example}

\begin{example}[Extremos na esfera, lidos no espectro]\label{ex:b2:quadratic:sphereextremes}
Quais são os valores extremos de $q(x,y,z) = 2xy + 2yz + 2zx$ na
esfera unitária? Sua matriz (a de uns fora da diagonal de
\cref{exo:b2:quadratic:2}) tem \omterm{def:b2:reduction:eigen}{autovalores} $2$ e $-1$
(duplo), de modo que, pelo~\cref{cor:b2:quadratic:principalaxes}~(2):
\[
\max_{\norm v = 1} q(v) = 2
\ \text{ em } v = \tfrac{1}{\sqrt3}(1,1,1),
\qquad
\min_{\norm v = 1} q(v) = -1
\ \text{ no círculo } x + y + z = 0 .
\]
Sem cálculo diferencial, sem multiplicadores de Lagrange: o teorema espectral
resolve essa otimização com restrição de imediato --- e exibe
o maximizante. Lição final: compare com o método dos multiplicadores
do capítulo de cálculo diferencial, que acha os
mesmos pontos críticos com mais trabalho; para objetivos
\emph{quadráticos} em esferas, os \omterm{def:b2:reduction:eigen}{espectros} são a via real (o problema de fim
de semana do capítulo hermitiano constrói toda a
teoria de Courant--Fischer sobre essa observação).
\end{example}

\begin{remark}[Armadilhas comuns]
\emph{(i) \omterm{def:b2:quadratic:def}{Congruência} não é semelhança:} uma mudança de base para
uma forma age por $P^{\mathsf T}BP$, e não por $P^{-1}BP$; os \omterm{def:b2:reduction:eigen}{autovalores}
\emph{não} são invariantes de uma \omterm{def:b2:quadratic:def}{forma quadrática} ($I$ e $4I$
são congruentes via $P = 2I$) --- só seus sinais o são
(Sylvester). Fale dos \omterm{def:b2:reduction:eigen}{autovalores} de uma forma somente depois de fixado um
produto interno. \emph{(ii) Entradas positivas nada provam:}
$\begin{pmatrix} 1 & 2\\ 2 & 1\end{pmatrix}$ tem todas as
entradas positivas e no entanto assinatura $(1,1)$ ($\det = -3$);
reciprocamente, uma matriz positiva definida pode ter entradas
negativas fora da diagonal (\cref{ex:b2:quadratic:tworoads} deslocada:
$2I - N$ funciona igualmente). Use o
\cref{met:b2:quadratic:signature}. \emph{(iii) Quadrados
dependentes:} escrever $q = \ell_1^2 - \ell_2^2$ nada diz se
$\ell_1, \ell_2$ são proporcionais --- $x^2 + 2xy + y^2 =
(x+y)^2$ tem posto $1$, e não $2$; verifique sempre a independência
antes de ler a assinatura. \emph{(iv) Extremos na esfera
sem \omterm{def:b2:metric:compact}{compacidade}:} as cotas de Rayleigh do
\cref{cor:b2:quadratic:principalaxes} são atingidas porque a
esfera é \omterm{def:b2:metric:compact}{compacta}; na bola \omterm{def:b2:metric:topology}{aberta} ou no espaço todo, uma
forma indefinida não tem máximo nem mínimo.
\end{remark}

\begin{remark}[Onde isso é usado]
O teorema espectral é o resultado mais exportado deste
livro: a estatística diagonaliza matrizes de covariância com ele
(análise de componentes principais), a mecânica extrai dele os modos normais
de oscilação (a redução simultânea do
problema de fim de semana), a análise numérica constrói sobre ele as decomposições
de Cholesky e em valores singulares (mesmo problema), e o próximo
capítulo o transporta para espaços hermitianos complexos. O volume do terceiro ano de
graduação demonstra seu avatar em dimensão infinita para operadores
autoadjuntos \omterm{def:b2:metric:compact}{compactos}, em que o argumento de \omterm{def:b2:metric:compact}{compacidade}
da demonstração em dimensão finita se torna a história inteira.
\end{remark}

\begin{remark}[Perspectivas dentro deste volume]
As \omterm{def:b2:quadratic:def}{formas quadráticas} atravessam o resto do Livro 4 sob três
disfarces. Como \emph{hessianas}: o capítulo de cálculo diferencial
classifica pontos críticos pela assinatura da forma de
segunda ordem, de modo que a invariância de Sylvester é o que torna
``sela'' uma palavra bem definida. Como \emph{energias}: os
osciladores do capítulo de equações diferenciais carregam a
energia quadrática $\frac12x'^2 + \frac12\omega^2x^2$, e a
redução simultânea do problema de fim de semana deste capítulo é
exatamente a extração dos modos normais. Como \emph{geometria}:
as cônicas deste capítulo crescem nas superfícies quádricas dos
capítulos de geometria, em que a segunda forma fundamental de uma
superfície --- uma \omterm{def:b2:quadratic:def}{forma quadrática} em cada plano tangente --- tem
assinatura que decide se a superfície curva como uma tigela ou
como uma sela. O capítulo hermitiano, a seguir, reexecuta a partitura
inteira sobre $\C$.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:quadratic:1}
Reduza por Gauss e dê o posto e a assinatura:
\[
q_1(x,y) = x^2 + 4xy + y^2,
\qquad
q_2(x,y,z) = x^2 + 2y^2 + 3z^2 + 2xy + 2yz .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:quadratic:2}
Diagonalize ortogonalmente $A = \begin{pmatrix} 0 & 1 & 1\\ 1 & 0 &
1\\ 1 & 1 & 0\end{pmatrix}$ (\omterm{def:b2:reduction:eigen}{autovalores} do
cálculo do~\cref{ch:b2:reduction}; agora torne a base
ortonormal) e reduza a forma $q(x,y,z) = 2xy + 2yz + 2zx$ a
eixos principais.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:quadratic:3}
Prove que $u^{**} = u$, $(u \circ v)^* = v^* \circ u^*$ e que
$\ker u^* = (\operatorname{im} u)^{\perp}$. Deduza
$\operatorname{rk} u^* = \operatorname{rk} u$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:4}
Seja $A$ real \omterm{def:b2:quadratic:adjoint}{simétrica} com $A^3 = A$. Prove que $A^2$ é a
matriz de uma projeção ortogonal. Mais geralmente, relacione as
decomposições espectrais de $A$ e de $P(A)$ para um polinômio $P$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:5}
Prove que $O(n) = \{P : P^{\mathsf T}P = I\}$ é uma parte \omterm{def:b2:metric:compact}{compacta}
de $\mathcal{M}_n(\R)$ \emph{(fechada: imagem inversa de $\{I\}$ por uma
aplicação \omterm{def:b2:metric:continuity}{contínua}; limitada: as colunas são vetores unitários)}. Ela é
\omterm{def:b2:metric:connected}{conexa}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:6}
(Raiz quadrada) Seja $A$ \omterm{def:b2:quadratic:adjoint}{simétrica} positiva semidefinida. Construa
uma $B$ \omterm{def:b2:quadratic:adjoint}{simétrica} positiva semidefinida com $B^2 = A$ e prove
que ela é \emph{única} \emph{(existência: tome raízes quadradas dos
\omterm{def:b2:reduction:eigen}{autovalores} numa base espectral; unicidade: uma candidata $B$
comuta com $A = B^2$, logo preserva seus \omterm{def:b2:reduction:eigen}{autoespaços} --- reduza ao
caso escalar em cada um)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:7}
Para $A$ real \omterm{def:b2:quadratic:adjoint}{simétrica}, prove que $\vertiii{A}_2 :=
\sup_{\norm x_2 = 1}\norm{Ax}_2 = \max_i \abs{\lambda_i}$
(raio espectral) e calcule $\vertiii{A}_2$ para $A =
\begin{pmatrix} 1 & 2\\ 2 & 1\end{pmatrix}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:quadratic:8}
(Critério de Sylvester) Seja $A$ real \omterm{def:b2:quadratic:adjoint}{simétrica} com menores
principais dominantes $\Delta_1, \dots, \Delta_n$ (\omterm{def:b2:linalg:det}{determinantes} dos
blocos superiores esquerdos). Prove que $A$ é positiva definida se e somente
se todos os $\Delta_k > 0$. \emph{(Para $\Rightarrow$: restrições de uma
forma definida são definidas, e o \omterm{def:b2:linalg:det}{determinante de uma matriz} positiva
definida --- o produto de seus \omterm{def:b2:reduction:eigen}{autovalores} --- é positivo.
Para $\Leftarrow$: faça indução sobre $n$; o bloco superior esquerdo $(n-1)$ é
positivo definido, diagonalize a forma nesse subespaço e
complete o quadrado na última variável; o sinal da última
entrada diagonal é governado por $\det A = \Delta_n > 0$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:quadratic:9}
(Courant--Fischer, segundo \omterm{def:b2:reduction:eigen}{autovalor}) Seja $u$ \omterm{def:b2:quadratic:adjoint}{simétrica} com
\omterm{def:b2:reduction:eigen}{autovalores} $\lambda_1 \geq \lambda_2 \geq \dots \geq \lambda_n$.
Prove
\[
\lambda_2 = \min_{\substack{H \text{ hiperplano}}}\;
\max_{\substack{x \in H,\ \norm x = 1}} \langle u(x), x\rangle .
\]
\emph{(Para $\leq$: todo hiperplano encontra o plano $2$ \omterm{def:b2:structures:generated}{gerado} pelos
dois \omterm{def:b2:reduction:eigen}{autovetores} de cima. Para $\geq$: escolha $H =
(e_1)^{\perp}$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:10}
Determine o posto e a assinatura de $q(x_1, \dots, x_n) = \sum_{i <
j} x_ix_j$ em $\R^n$ ($n \geq 2$), de duas maneiras: pela
identidade algébrica $2q = \bigl(\sum x_i\bigr)^2 - \sum x_i^2$ junto
com a restrição de $q$ ao hiperplano $\sum x_i = 0$;
e calculando os \omterm{def:b2:reduction:eigen}{autovalores} de sua matriz $\frac12(J - I)$,
em que $J$ é a matriz de uns.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:11}
Sejam $A$ e $B$ reais \omterm{def:b2:quadratic:adjoint}{simétricas} com $B$ positiva semidefinida.
Prove que
\[
\lambda_{\min}(A)\operatorname{tr} B
\;\leq\;
\operatorname{tr}(AB)
\;\leq\;
\lambda_{\max}(A)\operatorname{tr} B .
\]
\emph{(Escreva $B = C^{\mathsf T}C$ e $\operatorname{tr}(AB) =
\sum_i \langle A c_i, c_i\rangle$ sobre as colunas $c_i$ de
$C^{\mathsf T}$.)} Em particular, $\operatorname{tr}(AB) \geq 0$
quando ambas são positivas semidefinidas.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:quadratic:12}
Em $E = \mathcal{M}_n(\R)$, considere $q(M) =
\operatorname{tr}(M^2)$.
\begin{enumerate}
  \item Mostre que $q$ é uma \omterm{def:b2:quadratic:def}{forma quadrática} de forma polar
        $\varphi(M, N) = \operatorname{tr}(MN)$.
  \item Mostre que as matrizes \omterm{def:b2:quadratic:adjoint}{simétricas} e as antissimétricas formam
        subespaços $\varphi$-ortogonais nos quais $q$ é
        respectivamente positiva definida e negativa definida
        \emph{(calcule $\operatorname{tr}(M^2)$ entrada a entrada em
        cada caso)}.
  \item Conclua: $q$ tem assinatura $\bigl(\frac{n(n+1)}{2},
        \frac{n(n-1)}{2}\bigr)$ e posto $n^2$.
\end{enumerate}
\end{exercise}

\section{Problema: Cholesky, Hadamard e a decomposição polar}

\begin{problem}\label{pb:b2:quadratic:1}
O teorema espectral é um microscópio; este problema o usa como
fábrica. A partir de matrizes de Gram fabricamos a \emph{fatoração
de Cholesky}\index{fatoração de Cholesky} (e identificamos os
pivôs de Gauss como razões de menores), depois demonstramos a \emph{desigualdade
de Hadamard}\index{desigualdade de Hadamard} sobre \omterm{def:b2:linalg:det}{determinantes}, construímos
a \emph{decomposição polar}\index{decomposição polar} $A =
QS$ e a \emph{decomposição em valores singulares}, classificamos as cônicas
do plano e terminamos com a redução simultânea de duas formas ---
o teorema por trás dos modos normais de oscilação. Em todo o problema, $E =
\R^n$ com seu produto interno padrão.

\medskip
\noindent\textbf{Parte I --- Matrizes de Gram e Cholesky.} Para
vetores $v_1, \dots, v_n \in E$, sua \emph{matriz de Gram} é
$G = \bigl(\langle v_i, v_j\rangle\bigr)_{i,j}$.
\begin{enumerate}
  \item Mostre que $G$ é \omterm{def:b2:quadratic:adjoint}{simétrica} positiva semidefinida, e
        positiva definida se e somente se $(v_1, \dots, v_n)$ é
        linearmente independente \emph{(calcule $X^{\mathsf T}GX$)}.
  \item Reciprocamente, mostre que toda $A$ \omterm{def:b2:quadratic:adjoint}{simétrica} positiva
        semidefinida é uma matriz de Gram: $A = C^{\mathsf T}C$
        para alguma $C$ (use a raiz quadrada de
        \cref{exo:b2:quadratic:6}), com $C$ invertível se e somente se $A$
        é definida.
  \item Deduza que uma $A$ positiva semidefinida satisfaz
        $\abs{a_{ij}} \leq \sqrt{a_{ii}\,a_{jj}}$ para todos $i,
        j$ \emph{(restrinja a duas coordenadas)} --- a
        desigualdade de Cauchy--Schwarz, relida matricialmente.
  \item (Cholesky) Seja $A$ positiva definida. Prove que
        existe uma \emph{única} $T$ triangular superior com
        entradas diagonais positivas tal que
        \[
        A = T^{\mathsf T}\,T
        \]
        \emph{(existência: aplique Gram--Schmidt a vetores
        que realizem $A$ como matriz de Gram; unicidade: se
        $T_1^{\mathsf T}T_1 = T_2^{\mathsf T}T_2$, mostre que
        $T_1T_2^{-1}$ é ortogonal e triangular com
        diagonal positiva, logo $I$)}.
  \item Mostre que os menores principais dominantes satisfazem
        $\Delta_k = (t_{11}\cdots t_{kk})^2$ e deduza que
        os pivôs produzidos pela \omterm{thm:b2:quadratic:gauss}{redução de Gauss} de uma
        forma positiva definida, tomados na ordem natural das
        variáveis, são
        \[
        d_k = \frac{\Delta_k}{\Delta_{k-1}}
        \qquad (\Delta_0 = 1) :
        \]
        os menores do critério de Sylvester
        (\cref{exo:b2:quadratic:8}) e os pivôs de Gauss são
        os mesmos dados. Verifique em
        \cref{ex:b2:quadratic:tworoads}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A desigualdade de Hadamard.}
\begin{enumerate}[resume]
  \item Seja $A$ positiva definida. Prove que
        \[
        \det A \leq a_{11}\,a_{22}\cdots a_{nn}
        \]
        \emph{(normalize: $B = DAD$ com $D =
        \operatorname{diag}(a_{ii}^{-1/2})$ tem diagonal de uns;
        majore $\det B = \prod \mu_i$ por MA--MG contra
        $\operatorname{tr} B = n$)}.
  \item Mostre que vale a igualdade se e somente se $A$ é diagonal.
  \item Deduza a \emph{desigualdade de Hadamard}: para toda matriz
        quadrada real $M$ de colunas $c_1, \dots, c_n$,
        \[
        \abs{\det M} \leq \prod_{i=1}^{n}\norm{c_i}_2 ,
        \]
        com igualdade (para $M$ invertível) se e somente se as colunas são
        duas a duas ortogonais \emph{(aplique as questões 6--7 a
        $M^{\mathsf T}M$)}.
  \item Dividendos geométricos e combinatórios: interprete
        a questão 8 como ``o volume de um paralelepípedo é no
        máximo o produto dos comprimentos de suas arestas''; e mostre que uma
        matriz com todas as entradas em $\intcc{-1}{1}$ tem
        $\abs{\det M} \leq n^{n/2}$. (As matrizes que atingem essa
        cota --- as matrizes de Hadamard --- existem para $n = 1, 2$
        e muitos múltiplos de $4$; se existem para \emph{todos}
        os múltiplos de $4$ é um famoso problema em aberto.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- \omterm{pb:b2:quadratic:1}{Decomposição polar} e valores
singulares.}
\begin{enumerate}[resume]
  \item Seja $A$ invertível. Mostre que $A^{\mathsf T}A$ é
        positiva definida e que
        \[
        S = \sqrt{A^{\mathsf T}A}
        \quad\text{(a raiz quadrada do~\cref{exo:b2:quadratic:6})},
        \qquad Q = AS^{-1}
        \]
        dão uma fatoração $A = QS$ com $Q$ ortogonal e
        $S$ positiva definida.
  \item Prove que essa fatoração de uma $A$ invertível é
        única.
  \item Estenda a existência a $A$ arbitrária: escolha
        $\varepsilon_k \to 0$ com $A + \varepsilon_k I$
        invertível, escreva $A + \varepsilon_kI = Q_kS_k$ e
        use a \omterm{def:b2:metric:compact}{compacidade} de $O(n)$
        (\cref{exo:b2:quadratic:5}) para extrair $Q_k \to Q$;
        mostre que $S_k = Q_k^{\mathsf T}(A + \varepsilon_kI)$
        converge para alguma $S$ positiva semidefinida com $A =
        QS$ e $S = \sqrt{A^{\mathsf T}A}$. Onde a
        unicidade falha para $A$ singular?
  \item (Decomposição em valores singulares) Deduza que toda
        $A$ quadrada real escreve-se
        \[
        A = U\,\Sigma\,V^{\mathsf T},
        \qquad U, V \in O(n),\quad
        \Sigma = \operatorname{diag}(\sigma_1, \dots,
        \sigma_n),\ \sigma_i \geq 0 ,
        \]
        em que os $\sigma_i$ (os \emph{valores singulares}) são
        os \omterm{def:b2:reduction:eigen}{autovalores} de $\sqrt{A^{\mathsf T}A}$.
  \item Três consequências: $\vertiii{A}_2 = \sigma_{\max}$ para
        \emph{toda} $A$ real (generalizando
        o~\cref{exo:b2:quadratic:7}); $\abs{\det A} =
        \sigma_1\cdots\sigma_n$; e a imagem da esfera
        unitária por uma $A$ invertível é um elipsoide de
        semieixos $\sigma_1, \dots, \sigma_n$ ao longo das colunas
        de $U$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Cônicas, pelo teorema espectral.}
Uma cônica plana é o conjunto de zeros de $f(x) = q(x) + \langle b,
x\rangle + c$, com $q \neq 0$ uma \omterm{def:b2:quadratic:def}{forma quadrática} de matriz $A$,
$b \in \R^2$, $c \in \R$.
\begin{enumerate}[resume]
  \item Reduza $f$ por uma rotação (eixos principais,
        \cref{cor:b2:quadratic:principalaxes}) seguida de uma
        translação, e classifique as formas não vazias e
        não degeneradas possíveis pela assinatura de $q$: elipse
        ($\det A > 0$), hipérbole ($\det A < 0$), parábola
        ($\det A = 0$, posto $1$, com o termo linear não
        absorvido).
  \item Execute a redução por inteiro para
        \[
        x^2 + 4xy + y^2 + 2x - 2y = 4 :
        \]
        coordenadas giradas, equação reduzida, natureza e
        centro da cônica.
  \item (Cônicas centrais) Suponha $\det A \neq 0$. Mostre que o
        \emph{centro} é $x_0 = -\frac12 A^{-1}b$ e que a
        \omterm{def:b2:quadratic:def}{congruência} por $\begin{pmatrix} I & x_0\\ 0 &
        1\end{pmatrix}$ da matriz $3\times3$ $\widetilde Q
        = \begin{pmatrix} A & b/2 \\ b^{\mathsf T}/2 &
        c\end{pmatrix}$ dá
        \[
        \det\widetilde Q = f(x_0)\,\det A :
        \]
        a cônica central é degenerada (um ponto ou duas retas)
        exatamente quando $\det\widetilde Q = 0$.
  \item Verifique a questão 17 no exemplo da questão 16:
        calcule $x_0$, $f(x_0)$ e $\det\widetilde Q$, e
        conclua de novo que a cônica é uma hipérbole não
        degenerada.
  \item (Um feixe de quádricas) Para $\lambda \in \R$, classifique a
        superfície
        \[
        x^2 + y^2 + z^2 + 2\lambda(xy + yz + zx) = 1
        \]
        pelos \omterm{def:b2:reduction:eigen}{autovalores} de sua matriz \emph{(estrutura de
        uns: \omterm{def:b2:reduction:eigen}{autovalores} $1 + 2\lambda$ e $1 -
        \lambda$ duplo)}: esfera/elipsoide, cilindro, par de
        planos, hiperboloides de uma e de duas folhas, conforme
        $\lambda$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Duas formas de uma vez: redução
simultânea.}
\begin{enumerate}[resume]
  \item Seja $q$ positiva definida e $q'$ uma
        \omterm{def:b2:quadratic:def}{forma quadrática} arbitrária em $E$. Prove que existe uma base de
        $E$ que é ortonormal para $q$ e ortogonal para
        $q'$: nela, $q = \sum x_i^2$ e $q' = \sum \mu_i
        x_i^2$ \emph{(use $q$ como produto interno e aplique
        o teorema espectral ao endomorfismo que representa
        $q'$)}.
  \item Forma matricial: para $A$ positiva definida e $B$
        \omterm{def:b2:quadratic:adjoint}{simétrica}, existe $P$ invertível com
        $P^{\mathsf T}AP = I$ e $P^{\mathsf T}BP =
        \operatorname{diag}(\mu_1, \dots, \mu_n)$, em que os
        $\mu_i$ são as raízes de $\det(B - \mu A) = 0$.
  \item Execute-a por inteiro para
        \[
        A = \begin{pmatrix} 2 & 1\\ 1 & 1 \end{pmatrix},
        \qquad
        B = \begin{pmatrix} 0 & 1\\ 1 & 0 \end{pmatrix} :
        \]
        os \omterm{def:b2:reduction:eigen}{autovalores} generalizados $\mu_\pm$ e os vetores
        que diagonalizam as duas formas.
  \item Mostre que a definição positiva não pode ser dispensada: para
        \[
        A = \begin{pmatrix} 1 & 0\\ 0 & -1 \end{pmatrix},
        \qquad
        B = \begin{pmatrix} 0 & 1\\ 1 & 0 \end{pmatrix},
        \]
        nenhuma base diagonaliza as duas formas \emph{(se $P$
        diagonalizasse ambas, $\det(B - \mu A)$ se decomporia com
        raízes reais; calcule-o)}.
  \item Mostre que os $\mu_i$ da questão 21 são os
        \omterm{def:b2:reduction:eigen}{autovalores} de $A^{-1}B$ e que $A^{-1}B$, embora não
        \omterm{def:b2:quadratic:adjoint}{simétrica} em geral, é sempre \omterm{def:b2:reduction:diag}{diagonalizável} com
        \omterm{def:b2:reduction:eigen}{autovalores} reais \emph{(conjugue por $\sqrt A$)}.
  \item Síntese. Uma frase para cada: (i) o único teorema
        em que toda Parte se apoiou; (ii) quais resultados das Partes
        I--III sobrevivem para matrizes positivas \emph{semi}definidas
        e quais precisam de definição estrita; (iii) o
        sistema físico cujas pequenas oscilações as questões
        20--22 diagonalizam (energias cinética e potencial como
        as duas formas), e o que os $\mu_i$ significam ali.
\end{enumerate}
\end{problem}
