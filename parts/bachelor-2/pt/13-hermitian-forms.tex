\chapter{Formas hermitianas}\label{ch:b2:hermitian}

Os espaços vetoriais complexos têm sua própria geometria de produto interno, com
uma reviravolta: linearidade em uma variável, linearidade \emph{conjugada} na
outra. A recompensa por aceitar a reviravolta é uma teoria espectral
ainda mais limpa que a real --- os \omterm{def:b2:hermitian:adjoint}{endomorfismos hermitianos} têm
\omterm{def:b2:reduction:eigen}{autovalores} reais, os \omterm{def:b2:hermitian:adjoint}{unitários} têm \omterm{def:b2:reduction:eigen}{autovalores} de módulo um, e os dois
diagonalizam em bases ortonormais. Este capítulo curto roda o
programa euclidiano do~\cref{ch:b2:quadratic} sobre $\C$.

\section{Produtos internos hermitianos}

\begin{definition}\label{def:b2:hermitian:def}
Um \emph{produto interno hermitiano}\index{produto interno hermitiano} num
espaço vetorial complexo $E$ é uma aplicação $\langle\cdot,\cdot\rangle
\colon E \times E \to \C$ linear na segunda variável,
\omterm{def:b2:quadratic:adjoint}{conjugada-simétrica} ($\langle y, x\rangle = \conj{\langle x,
y\rangle}$ --- logo conjugada-linear na primeira variável) e
positiva definida ($\langle x, x\rangle > 0$ para $x \neq 0$). O
exemplo padrão em $\C^n$:
\[
\langle x, y \rangle = \sum_{i=1}^{n} \conj{x_i}\, y_i ;
\]
sobre funções \omterm{def:b2:metric:continuity}{contínuas}, $\langle f, g\rangle = \int_a^b \conj
f\,g$. \omterm{def:b2:nvs:norm}{Norma}: $\norm x = \sqrt{\langle x,x\rangle}$; um
espaço complexo de dimensão finita assim munido é um \emph{espaço
hermitiano}\index{espaço hermitiano}.
\end{definition}

\begin{example}[Primeiros cálculos]\label{ex:b2:hermitian:firstcomp}
Em $\C^2$, tome $x = (1+\iu,\ 2-\iu)$ e $y = (\iu,\ 1)$.
Então
\[
\norm x^2 = \abs{1+\iu}^2 + \abs{2-\iu}^2 = 2 + 5 = 7,
\qquad
\norm y^2 = 1 + 1 = 2,
\]
e, conjugando o primeiro argumento,
\[
\langle x, y\rangle
= \conj{(1+\iu)}\,\iu + \conj{(2-\iu)}\cdot1
= (1-\iu)\iu + (2+\iu)
= (\iu + 1) + (2 + \iu) = 3 + 2\iu .
\]
Cauchy--Schwarz confere: $\abs{\langle x, y\rangle}^2 = 9 +
4 = 13 \leq 14 = \norm x^2\norm y^2$ --- perto da igualdade,
porque $x$ está perto de ser um múltiplo de $y$. Note também que
$\langle y, x\rangle = \conj{3 + 2\iu} = 3 - 2\iu$:
a simetria conjugada em ação, e a razão pela qual $\langle x,
x\rangle$ é sempre real.
\end{example}

\begin{theorem}[Cauchy--Schwarz, complexa]\label{thm:b2:hermitian:cs}
$\abs{\langle x, y\rangle} \leq \norm x \norm y$, com igualdade se e somente se
$x, y$ são linearmente dependentes; $\norm\cdot$ é uma \omterm{def:b2:nvs:norm}{norma}. Além disso,
existem bases ortonormais (Gram--Schmidt roda palavra por palavra), com
\[
x = \sum_i \langle e_i, x\rangle\, e_i,
\qquad
\norm x^2 = \sum_i \abs{\langle e_i, x\rangle}^2 .
\]
\end{theorem}

\begin{proof}
Para $y \neq 0$ e $t \in \C$: $0 \leq \norm{x - ty}^2 = \norm x^2
- 2\Re\bigl(\conj t\langle y, x\rangle\bigr) + \abs t^2\norm y^2$.
Escolha $t = \frac{\langle y, x\rangle}{\norm y^2}$:
\[
0 \leq \norm x^2 - \frac{\abs{\langle y, x\rangle}^2}{\norm y^2},
\]
que é a desigualdade; a igualdade força $x = ty$. Desigualdade
triangular, por inteiro:
\[
\norm{x + y}^2 = \norm x^2 + 2\,\Re\langle x, y\rangle
+ \norm y^2
\leq \norm x^2 + 2\,\abs{\langle x, y\rangle} + \norm y^2
\leq \bigl(\norm x + \norm y\bigr)^2 ,
\]
usando $\Re z \leq \abs z$ e depois Cauchy--Schwarz; a homogeneidade
e a separação são imediatas, logo $\norm\cdot$ é uma \omterm{def:b2:nvs:norm}{norma}.
Gram--Schmidt: como no caso real, com os conjugados
colocados pela definição (note a convenção: nossos produtos são
conjugados-lineares na \emph{primeira} casa, de modo que as coordenadas são
$\langle e_i, x\rangle$; o exemplo seguinte roda o algoritmo
uma vez por inteiro).
\end{proof}

\begin{example}[Gram--Schmidt complexo, executado por inteiro]\label{ex:b2:hermitian:gramschmidt}
Ortonormalize a base $v_1 = (1, \iu)$, $v_2 = (0, 1)$ de
$\C^2$. Primeiro vetor: $\norm{v_1}^2 = \abs1^2 + \abs\iu^2 = 2$,
logo $e_1 = \frac{1}{\sqrt2}(1, \iu)$. Projete $v_2$ --- com o
conjugado na primeira casa:
\[
\langle e_1, v_2\rangle
= \frac{1}{\sqrt2}\bigl(\conj{1}\cdot0 +
\conj{\iu}\cdot1\bigr)
= \frac{-\iu}{\sqrt2} ,
\qquad
v_2 - \langle e_1, v_2\rangle e_1
= (0,1) + \frac{\iu}{2}\,(1, \iu)
= \Bigl(\frac\iu2,\ \frac12\Bigr) .
\]
Sua \omterm{def:b2:nvs:norm}{norma} é $\sqrt{\frac14 + \frac14} = \frac{1}{\sqrt2}$:
$e_2 = \frac{1}{\sqrt2}(\iu, 1)$. Verificação: $\langle e_1,
e_2\rangle = \frac12(\conj1\cdot\iu + \conj\iu\cdot1) =
\frac12(\iu - \iu) = 0$. Coordenadas de $v_2$ na nova base:
$v_2 = \langle e_1, v_2\rangle e_1 + \langle e_2, v_2\rangle
e_2$ com $\langle e_2, v_2\rangle = \frac{1}{\sqrt2}$ ---
atenção à \omterm{def:b2:structures:generated}{ordem}: $\langle v_2, e_1\rangle$ daria o
coeficiente \emph{conjugado}. Lição final: o algoritmo é o
euclidiano palavra por palavra; a única armadilha é onde cai a
conjugação, e calcular $\norm{v_2 -
\text{proj}}^2 > 0$ usa silenciosamente a positividade --- o axioma
que faz toda a geometria funcionar.
\end{example}

\section{Adjunto, endomorfismos hermitianos e unitários}

\begin{definition}\label{def:b2:hermitian:adjoint}
O \emph{adjunto} $u^*$ de $u \in \mathcal{L}(E)$ define-se por
$\langle u^*(x), y\rangle = \langle x, u(y)\rangle$; numa
base ortonormal, $\operatorname{Mat}(u^*) = \conj{A}^{\mathsf T} =:
A^{\dagger}$ (\omterm{def:b2:linalg:transpose}{transposta} conjugada) --- de fato, se $B =
(b_{ij})$ é a matriz de $u^*$ na base ortonormal
$(e_i)$, então $b_{ij} = \langle e_i, u^*(e_j)\rangle$, e a
identidade de definição dá
\[
\conj{b_{ij}}
= \langle u^*(e_j), e_i\rangle
= \langle e_j, u(e_i)\rangle = a_{ji} ,
\qquad\text{logo}\qquad
B = \conj{A}^{\mathsf T} .
\] $u$ é
\emph{hermitiano}\index{endomorfismo hermitiano} quando $u^* = u$
($A^\dagger = A$), \emph{unitário}\index{grupo unitário} quando $u^*u =
\mathrm{id}$ ($A^\dagger A = I$: o grupo $U(n)$), \emph{normal}
quando $u^*u = uu^*$.
\end{definition}

\begin{proposition}\label{prop:b2:hermitian:eigenvalues}
Os \omterm{def:b2:reduction:eigen}{autovalores} de um \omterm{def:b2:hermitian:adjoint}{endomorfismo hermitiano} são \emph{reais};
os \omterm{def:b2:reduction:eigen}{autovalores} de um endomorfismo \omterm{def:b2:hermitian:adjoint}{unitário} têm \emph{módulo $1$}; nos
dois casos, os \omterm{def:b2:reduction:eigen}{autoespaços} de \omterm{def:b2:reduction:eigen}{autovalores} distintos são ortogonais.
\end{proposition}

\begin{proof}
\omterm{def:b2:hermitian:adjoint}{Hermitiano}, $u(x) = \lambda x$, $x \neq 0$:
\[
\lambda \norm x^2 = \langle x, u(x)\rangle = \langle u(x), x\rangle
= \conj{\langle x, u(x)\rangle} = \conj\lambda\,\norm x^2 ,
\]
logo $\lambda \in \R$. \omterm{def:b2:hermitian:adjoint}{Unitário}: $\norm{u(x)} = \norm x$ (de $u^*u =
\mathrm{id}$), logo $\abs\lambda\norm x = \norm x$. Ortogonalidade
(caso \omterm{def:b2:hermitian:adjoint}{hermitiano}): $\lambda\langle x, y\rangle = \langle u(x),
y\rangle = \langle x, u(y)\rangle = \mu\langle x, y\rangle$ para
\omterm{def:b2:reduction:eigen}{autovetores} com $\lambda \neq \mu$ reais. Caso \omterm{def:b2:hermitian:adjoint}{unitário}, por
inteiro: para $u(x) = \lambda x$, $u(y) = \mu y$ com $\lambda \neq
\mu$ (ambos de módulo um),
\[
\langle x, y\rangle = \langle u(x), u(y)\rangle
= \conj\lambda\mu\,\langle x, y\rangle ,
\]
e $\conj\lambda\mu = \frac{\mu}{\lambda} \neq 1$: o fator
não é $1$, logo $\langle x, y\rangle = 0$.
\end{proof}

\begin{example}[Uma matriz anti-hermitiana, diagonalizada]\label{ex:b2:hermitian:skewexample}
$A = \begin{pmatrix} 0 & -2\\ 2 & 0\end{pmatrix}$ satisfaz
$A^\dagger = A^{\mathsf T} = -A$: \omterm{def:b2:hermitian:adjoint}{anti-hermitiana} (e também real
antissimétrica --- sobre $\R$ ela não tem \omterm{def:b2:reduction:eigen}{autovalor} algum).
\omterm{def:b2:reduction:charpoly}{Polinômio característico} $X^2 + 4$: \omterm{def:b2:reduction:eigen}{autovalores} $\pm2\iu$,
puramente imaginários, como o~\cref{exo:b2:hermitian:9} prevê em
geral. \omterm{def:b2:reduction:eigen}{Autovetores}: $(A - 2\iu I)v = 0$ dá $v_1 =
\frac{1}{\sqrt2}(1, \iu)$, e $v_2 = \frac{1}{\sqrt2}(1,
-\iu)$ para $-2\iu$; eles são ortogonais:
\[
\langle v_1, v_2\rangle
= \tfrac12\bigl(\conj{1}\cdot1 +
\conj{\iu}\cdot(-\iu)\bigr)
= \tfrac12(1 - 1) = 0 .
\]
Assim $A = U\operatorname{diag}(2\iu, -2\iu)\,U^\dagger$ com $U =
(v_1\ v_2)$ \omterm{def:b2:hermitian:adjoint}{unitária}. Lição final: $H = -\iu A =
\begin{pmatrix} 0 & 2\iu\\ -2\iu & 0\end{pmatrix}$ é \omterm{def:b2:hermitian:adjoint}{hermitiana}
com o \omterm{def:b2:reduction:eigen}{espectro} real $\{\pm2\}$ e os \emph{mesmos}
\omterm{def:b2:reduction:eigen}{autovetores} --- a bijeção $u \mapsto \iu u$ entre
\omterm{def:b2:hermitian:adjoint}{endomorfismos hermitianos} e \omterm{def:b2:hermitian:adjoint}{anti-hermitianos}
(\cref{exo:b2:hermitian:9}), vista matriz a matriz; sobre $\R$
a mesma $A$ é uma rotação com escala sem \omterm{def:b2:reduction:eigen}{autovetor} algum,
e só a passagem a $\C$ revela sua forma normal.
\end{example}

\begin{theorem}[Teorema espectral hermitiano]\label{thm:b2:hermitian:spectral}
Todo \omterm{def:b2:hermitian:adjoint}{endomorfismo hermitiano} de um \omterm{def:b2:hermitian:def}{espaço hermitiano} tem base
ortonormal de \omterm{def:b2:reduction:eigen}{autovetores} (com \omterm{def:b2:reduction:eigen}{autovalores} reais): $A^\dagger = A$
implica $A = U D U^{\dagger}$ com $U \in U(n)$ e $D$ diagonal
real.\index{teorema espectral!hermitiano}
\end{theorem}

\begin{proof}
Sobre $\C$, o \omterm{def:b2:reduction:charpoly}{polinômio característico} se decompõe: existe um \omterm{def:b2:reduction:eigen}{autovetor}
$e_1$ (\cref{ch:b2:reduction}) --- nenhum argumento de \omterm{def:b2:metric:compact}{compacidade}
necessário, vantagem de $\C$. Normalize-o. Seu complemento
ortogonal $F = e_1^\perp$ é estável: para $x \perp e_1$,
\[
\langle e_1, u(x)\rangle = \langle u(e_1), x\rangle
= \lambda_1\langle e_1, x\rangle = 0
\]
($\lambda_1$ real). A restrição é \omterm{def:b2:hermitian:adjoint}{hermitiana}; faça indução sobre a
dimensão e concatene. Em detalhe: a restrição $u|_F$ é
um endomorfismo do \omterm{def:b2:hermitian:def}{espaço hermitiano} $F$ (dimensão $n - 1$)
com $\langle u|_F(x), y\rangle = \langle x, u|_F(y)\rangle$
herdado de $u$; a hipótese de indução fornece uma
base ortonormal $(e_2, \dots, e_n)$ de $F$ de \omterm{def:b2:reduction:eigen}{autovetores},
e $(e_1, e_2, \dots, e_n)$ é ortonormal em $E$ ($e_1 \perp
F$) e formada de \omterm{def:b2:reduction:eigen}{autovetores} de $u$. Tradução matricial: as
colunas de $U$ são os $e_i$, $U^\dagger U = I$ exprime sua
ortonormalidade, e $AU = UD$ reúne as equações de
\omterm{def:b2:reduction:eigen}{autovalor}, donde $A = UDU^\dagger$ com $D$ diagonal real
(\cref{prop:b2:hermitian:eigenvalues}).
\end{proof}

\begin{example}\label{ex:b2:hermitian:pauli}
$A = \begin{pmatrix} 0 & -\iu\\ \iu & 0\end{pmatrix}$ é \omterm{def:b2:hermitian:adjoint}{hermitiana}
($A^\dagger = A$): \omterm{def:b2:reduction:eigen}{autovalores} a partir de $\chi_A = X^2 - 1$: $\pm 1$
(reais, como prometido), com \omterm{def:b2:reduction:eigen}{autovetores} ortonormais
$\frac{1}{\sqrt2}(1, \iu)^{\mathsf T}$ e $\frac{1}{\sqrt2}(1,
-\iu)^{\mathsf T}$. (Os físicos conhecem $A$ como matriz de Pauli; a
realidade dos \omterm{def:b2:reduction:eigen}{espectros} \omterm{def:b2:hermitian:adjoint}{hermitianos} é o motivo pelo qual os observáveis quânticos são modelados
por operadores \omterm{def:b2:hermitian:adjoint}{hermitianos}.)
\end{example}

\begin{example}[Uma matriz hermitiana positiva definida, trabalhada]\label{ex:b2:hermitian:pdrun}
$A = \begin{pmatrix} 2 & 1-\iu\\ 1+\iu & 3\end{pmatrix}$:
\omterm{def:b2:hermitian:adjoint}{hermitiana}, pois a diagonal é real e as entradas fora da
diagonal são conjugadas. \omterm{def:b2:reduction:charpoly}{Polinômio característico}:
\[
(2-\lambda)(3-\lambda) - \abs{1-\iu}^2
= \lambda^2 - 5\lambda + 4
= (\lambda - 1)(\lambda - 4) :
\]
\omterm{def:b2:reduction:eigen}{espectro} $\{1, 4\}$, real e positivo --- $A$ é positiva
definida. \omterm{def:b2:reduction:eigen}{Autovetores}: para $\lambda = 4$, o sistema $(A -
4I)v = 0$ dá $v_4 = (1 - \iu,\ 2)$ (confira a segunda linha:
$(1+\iu)(1-\iu) - 2 = 0$); para $\lambda = 1$, $v_1 = (1 - \iu,\
-1)$. Ortogonalidade, com o conjugado na primeira casa:
\[
\langle v_4, v_1\rangle
= \conj{(1-\iu)}\,(1-\iu) + \conj{2}\,(-1)
= 2 - 2 = 0 . \checkmark
\]
Normalizando ($\norm{v_4}^2 = 2 + 4 = 6$, $\norm{v_1}^2 = 2 + 1
= 3$) obtém-se a \omterm{def:b2:hermitian:adjoint}{unitária} $U = \bigl(\frac{v_4}{\sqrt6}\
\frac{v_1}{\sqrt3}\bigr)$ com $A =
U\operatorname{diag}(4,1)U^\dagger$. Lição final: a
leitura de Rayleigh é imediata --- na esfera unitária de
$\C^2$, $\langle x, Ax\rangle$ percorre $\intcc{1}{4}$,
atingido nos dois \omterm{def:b2:reduction:eigen}{autovetores}; esse é o germe em $n = 2$ da
teoria de Courant--Fischer construída no problema de fim de semana.
Confira por amostragem que a forma é real fora dos \omterm{def:b2:reduction:eigen}{autovetores} também: em
$x = (1, \iu)$,
\[
Ax = \bigl(2 + (1-\iu)\iu,\ (1+\iu) + 3\iu\bigr)
= (3 + \iu,\ 1 + 4\iu),
\]
\[
\langle x, Ax\rangle = \conj{1}\,(3+\iu) +
\conj{\iu}\,(1+4\iu)
= (3 + \iu) + (-\iu)(1 + 4\iu) = 3 + \iu - \iu + 4 = 7 \in
\R ,
\]
como o mecanismo de demonstração da~\cref{prop:b2:hermitian:eigenvalues}
(simetria conjugada contra $A^\dagger = A$) garante para
todo $x$.
\end{example}

\begin{example}[Uma matriz unitária diagonalizada]\label{ex:b2:hermitian:unitary}
$U = \frac{1}{\sqrt2}\begin{pmatrix} 1 & \iu\\ \iu & 1
\end{pmatrix}$ (\omterm{def:b2:hermitian:adjoint}{unitária} por \cref{exo:b2:hermitian:2}). Seu
\omterm{def:b2:reduction:charpoly}{polinômio característico} é $\bigl(X - \frac{1}{\sqrt2}\bigr)^2
+ \frac12$, com raízes
\[
\lambda_\pm = \frac{1 \pm \iu}{\sqrt2} = \eu^{\pm\iu\pi/4},
\]
de módulo um, como a~\cref{prop:b2:hermitian:eigenvalues} prometeu, e
\omterm{def:b2:reduction:eigen}{autovetores} ortonormais $\frac{1}{\sqrt2}(1, \pm1)$. Assim $U =
V\operatorname{diag}(\eu^{\iu\pi/4},
\eu^{-\iu\pi/4})V^\dagger$: na base certa, $U$ é um par de
rotações planas de $\pm\frac\pi4$ --- uma matriz de rotação real não tem
\omterm{def:b2:reduction:eigen}{autovetor} real algum, mas sobre $\C$ ela se separa em dois
escalares de módulo um. Lição final: os \omterm{def:b2:reduction:eigen}{espectros} \omterm{def:b2:hermitian:adjoint}{hermitianos} vivem na
reta real, os \omterm{def:b2:hermitian:adjoint}{unitários} no círculo unitário; ambos são
sombras da mesma normalidade, e a transformada de Cayley do
\cref{exo:b2:hermitian:6} leva um retrato no outro.
\end{example}

\begin{example}[A transformada de Cayley, calculada]\label{ex:b2:hermitian:cayleyrun}
Rode o~\cref{exo:b2:hermitian:6} em $H = \begin{pmatrix} 0 & 1\\
1 & 0\end{pmatrix}$ (\omterm{def:b2:hermitian:adjoint}{hermitiana}, \omterm{def:b2:reduction:eigen}{espectro} $\{1, -1\}$,
\omterm{def:b2:reduction:eigen}{autovetores} ortonormais $\frac{1}{\sqrt2}(1, \pm1)$). Na
base de \omterm{def:b2:reduction:eigen}{autovetores} tudo é escalar: a transformada $\lambda
\mapsto \frac{\lambda - \iu}{\lambda + \iu}$ leva
\[
1 \longmapsto \frac{1 - \iu}{1 + \iu} = -\iu,
\qquad
-1 \longmapsto \frac{-1 - \iu}{-1 + \iu} = \iu ,
\]
(multiplique pelo conjugado do denominador), de modo que $U = (H -
\iu I)(H + \iu I)^{-1}$ é a \omterm{def:b2:hermitian:adjoint}{unitária} de \omterm{def:b2:reduction:eigen}{autovalores}
$\mp\iu$ nesses mesmos \omterm{def:b2:reduction:eigen}{autovetores}:
\[
U = \frac{1}{2}\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
\begin{pmatrix} -\iu & 0\\ 0 & \iu\end{pmatrix}
\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
= \begin{pmatrix} 0 & -\iu\\ -\iu & 0\end{pmatrix} .
\]
Verificação: $U^\dagger U = I$ e $1 \notin \operatorname{Sp}U =
\{\pm\iu\}$, como a teoria promete. Lição final: a reta
real é levada sobre o círculo unitário menos o ponto $1$ ---
\omterm{def:b2:reduction:eigen}{autovalor} por \omterm{def:b2:reduction:eigen}{autovalor}, a transformada de Cayley é a
aplicação de Möbius $\frac{\lambda-\iu}{\lambda+\iu}$, e as matrizes
apenas seguem seus \omterm{def:b2:reduction:eigen}{espectros}.
\end{example}

\begin{remark}[Armadilhas comuns]
\emph{(i) Onde cai a barra:} este livro conjuga a
\emph{primeira} casa, de modo que as coordenadas são $\langle e_i, x\rangle$
e $\langle\lambda x, y\rangle = \conj\lambda\langle x,
y\rangle$; muitos textos conjugam a segunda casa ---
traduza antes de comparar fórmulas, ou os sinais de $\iu$ ficam
errados em silêncio. \emph{(ii) A \omterm{def:b2:quadratic:def}{polarização} complexa é mais forte:}
sobre $\C$, se $\langle x, u(x)\rangle = 0$ para \emph{todo} $x$,
então $u = 0$ (desenvolva $x + y$ e $x + \iu y$: tanto a parte
real quanto a imaginária de $\langle x, u(y)\rangle$ se anulam); sobre
$\R$ isso falha --- a rotação de $\frac\pi2$ satisfaz
$\langle x, u(x)\rangle = 0$ em toda parte. Por consequência, apenas sobre
$\C$, ``$\langle x, u(x)\rangle \in \R$ para todo $x$''
já força $u$ \omterm{def:b2:hermitian:adjoint}{hermitiano}. \emph{(iii) Normal real não é
\omterm{def:b2:reduction:diag}{diagonalizável}:} a matriz de
\cref{ex:b2:hermitian:skewexample} é normal mas não tem \omterm{def:b2:reduction:eigen}{autovalor}
real; a diagonalização \omterm{def:b2:hermitian:adjoint}{unitária} é um teorema \emph{sobre}
$\C$, e sobre $\R$ só se obtêm reduções por blocos.
\emph{(iv) Verificar a unitariedade:} $U^\dagger U = I$ significa que as
\emph{colunas} são ortonormais para o produto \omterm{def:b2:hermitian:adjoint}{hermitiano} ---
testar $UU^{\mathsf T}$, ou esquecer a conjugação nos
produtos de colunas, são os dois modos clássicos de certificar uma matriz
errada.
\end{remark}

\begin{example}[As isometrias são exatamente as unitárias]\label{ex:b2:hermitian:isometry}
Preservar a \omterm{def:b2:nvs:norm}{norma} parece mais fraco que ser \omterm{def:b2:hermitian:adjoint}{unitário}, mas sobre $\C$ não
é: se $\norm{u(x)} = \norm x$ para todo $x$, então $u^*u =
\mathrm{id}$. De fato, $v = u^*u - \mathrm{id}$ é \omterm{def:b2:hermitian:adjoint}{hermitiano} e
satisfaz $\langle x, v(x)\rangle = \norm{u(x)}^2 - \norm x^2 =
0$ para todo $x$; pela \omterm{def:b2:quadratic:def}{polarização} complexa (armadilha (ii)
acima), uma aplicação com ``diagonal'' identicamente nula é nula:
$v = 0$. Concretamente, a \omterm{def:b2:quadratic:def}{polarização} roda
\[
0 = \langle x + y, v(x+y)\rangle
= \langle x, v(y)\rangle + \langle y, v(x)\rangle,
\qquad
0 = \langle x + \iu y, v(x + \iu y)\rangle
= \iu\langle x, v(y)\rangle - \iu\langle y, v(x)\rangle ,
\]
e as duas linhas juntas forçam $\langle x, v(y)\rangle = 0$
para todos $x, y$. Lição final: é por isso que ``\omterm{def:b2:hermitian:adjoint}{unitário}'' pode
ser verificado só medindo comprimentos --- uma rigidez que o
capítulo de Fourier vai explorar, em que preservar a energia
$\norm f_2$ (Parseval) é o mesmo que preservar todos os produtos
internos de coeficientes.
\end{example}

\begin{remark}[Perspectivas dentro deste volume]
A maquinaria \omterm{def:b2:hermitian:adjoint}{hermitiana} aqui construída é consumida quase
imediatamente. O capítulo de Fourier \emph{é} geometria \omterm{def:b2:hermitian:adjoint}{hermitiana}
em dimensão infinita: as exponenciais $(e_n)$ são uma
família ortonormal para $\langle f, g\rangle =
\frac{1}{2\pi}\int\conj fg$, a desigualdade de Bessel é a
estimativa de projeção do~\cref{thm:b2:hermitian:cs} deste capítulo,
e Parseval é sua igualdade limite. A transformada de Fourier finita
(\cref{exo:b2:hermitian:10}) reaparece sempre que a convolução
tem de ser diagonalizada. E o problema de fim de semana deste capítulo ---
Courant--Fischer, Weyl, entrelaçamento --- fornece a
estabilidade dos \omterm{def:b2:reduction:eigen}{autovalores} que o capítulo de equações diferenciais
invoca ao afirmar que pequenas perturbações de um sistema
movem suas frequências apenas ligeiramente. Para trás, tudo aqui
é o espelho complexo do capítulo de \omterm{def:b2:quadratic:def}{formas quadráticas}: mantenha os
dois dicionários lado a lado ($A^{\mathsf T} \leftrightarrow
A^\dagger$, ortogonal $\leftrightarrow$ \omterm{def:b2:hermitian:adjoint}{unitário}, Rayleigh
real nos dois).
\end{remark}

\begin{remark}[Endomorfismos normais]
Sobre $\C$ o enunciado definitivo é: $u$ é \emph{unitariamente
\omterm{def:b2:reduction:diag}{diagonalizável} se e somente se é normal} ($u^*u = uu^*$) ---
cobrindo de uma vez as aplicações \omterm{def:b2:hermitian:adjoint}{hermitianas}, \omterm{def:b2:hermitian:adjoint}{unitárias} e \omterm{def:b2:hermitian:adjoint}{anti-hermitianas}. A
demonstração é um reforço agradável do argumento acima
(\cref{exo:b2:hermitian:8}). Sobre $\R$, em contraste, a normalidade só
compra a diagonalização por blocos (blocos de rotação): a geometria complexa é
genuinamente mais simples.
\end{remark}

\begin{remark}[Onde isso é usado]
A teoria espectral \omterm{def:b2:hermitian:adjoint}{hermitiana} é a matemática da mecânica
quântica: os observáveis são modelados por operadores \omterm{def:b2:hermitian:adjoint}{hermitianos}
(\omterm{def:b2:reduction:eigen}{espectros} reais = valores mensuráveis), a evolução temporal por operadores \omterm{def:b2:hermitian:adjoint}{unitários}
(preservação da \omterm{def:b2:nvs:norm}{norma} = conservação da probabilidade). Dentro
deste livro, o capítulo de Fourier repousa sobre a ortonormalidade das
exponenciais --- um enunciado sobre \omterm{def:b2:hermitian:def}{produto interno hermitiano} --- e
a diagonalização das matrizes circulantes
(\cref{exo:b2:hermitian:10}) é a transformada de Fourier finita.
O problema de fim de semana desenvolve o cálculo variacional dos
\omterm{def:b2:reduction:eigen}{autovalores} (Courant--Fischer, Weyl, entrelaçamento), o pão de cada
dia da análise numérica e da física matemática; o volume do terceiro
ano de graduação o estende a operadores \omterm{def:b2:hermitian:adjoint}{autoadjuntos} \omterm{def:b2:metric:compact}{compactos} em
espaços de Hilbert.
\end{remark}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:hermitian:1}
Em $\C^2$: calcule $\langle x, y\rangle$, $\norm x$, $\norm y$ para
$x = (1, \iu)$, $y = (\iu, 1)$; eles são ortogonais? Dê uma
base ortonormal que contenha $\frac{x}{\norm x}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:hermitian:2}
Quais são \omterm{def:b2:hermitian:adjoint}{hermitianas}? \omterm{def:b2:hermitian:adjoint}{unitárias}? normais?
\[
\begin{pmatrix} 1 & \iu\\ -\iu & 2 \end{pmatrix},
\qquad
\frac{1}{\sqrt2}\begin{pmatrix} 1 & \iu\\ \iu & 1\end{pmatrix},
\qquad
\begin{pmatrix} 0 & 1\\ 0 & 0 \end{pmatrix}.
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:hermitian:3}
Prove que uma matriz $A \in \mathcal{M}_n(\C)$ se escreve de modo único $A =
H + \iu K$ com $H, K$ \omterm{def:b2:hermitian:adjoint}{hermitianas} \emph{(as ``partes real e
imaginária'' $H = \frac{A + A^\dagger}{2}$, $K = \frac{A -
A^\dagger}{2\iu}$)}, e que $A$ é normal se e somente se $H$ e $K$
comutam.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:4}
Diagonalize numa base ortonormal: $A = \begin{pmatrix} 2 & \iu\\
-\iu & 2\end{pmatrix}$, e calcule $A^k$ para $k \in \N$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:5}
Prove que $U(n)$ é \omterm{def:b2:metric:compact}{compacto} e que a aplicação \omterm{def:b2:reduction:eigen}{autovalor} é
sobrejetiva: todo $\lambda$ de módulo um aparece para alguma matriz unitária.
Prove que $\det U \in \mathbb{U}$ (o círculo unitário) para $U \in
U(n)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:6}
(Transformada de Cayley) Seja $H$ \omterm{def:b2:hermitian:adjoint}{hermitiana}. Prove que $H + \iu I$ é
invertível e que $U = (H - \iu I)(H + \iu I)^{-1}$ é \omterm{def:b2:hermitian:adjoint}{unitária},
com $1 \notin \operatorname{Sp}(U)$. \emph{(Trabalhe espectralmente: numa
base de \omterm{def:b2:reduction:eigen}{autovetores} de $H$, tudo é escalar.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:7}
Para $A$ \omterm{def:b2:hermitian:adjoint}{hermitiana} positiva definida ($\langle x, Ax\rangle > 0$ para
$x \neq 0$), prove que $\operatorname{Sp}(A) \subseteq
\intoo{0}{\infty}$, que $A = B^2$ para uma $B$ \omterm{def:b2:hermitian:adjoint}{hermitiana} positiva
definida e que $\det A > 0$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:hermitian:8}
(Teorema espectral para endomorfismos normais) Seja $u$ normal num
\omterm{def:b2:hermitian:def}{espaço hermitiano}.
\begin{enumerate}
  \item Prove que $\norm{u(x)} = \norm{u^*(x)}$ para todo $x$ e deduza que
        $\ker(u - \lambda) = \ker(u^* - \conj\lambda)$.
  \item Prove que os \omterm{def:b2:reduction:eigen}{autoespaços} de $u$ de \omterm{def:b2:reduction:eigen}{autovalores} distintos são
        ortogonais e que o complemento ortogonal de um
        \omterm{def:b2:reduction:eigen}{autoespaço} é estável por $u$.
  \item Conclua por indução que $u$ é unitariamente \omterm{def:b2:reduction:diag}{diagonalizável};
        e reciprocamente.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:hermitian:9}
Um endomorfismo é \emph{\omterm{def:b2:hermitian:adjoint}{anti-hermitiano}} quando $u^* = -u$. Prove
que seus \omterm{def:b2:reduction:eigen}{autovalores} são puramente imaginários, que $u \mapsto \iu
u$ é uma bijeção dos \omterm{def:b2:hermitian:adjoint}{endomorfismos hermitianos} nos
\omterm{def:b2:hermitian:adjoint}{anti-hermitianos} e que os endomorfismos \omterm{def:b2:hermitian:adjoint}{anti-hermitianos} são
unitariamente \omterm{def:b2:reduction:diag}{diagonalizáveis} (\cref{exo:b2:hermitian:8}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:10}
(A transformada de Fourier finita) Seja $S$ o deslocamento \omterm{def:b2:structures:generated}{cíclico} de
$\C^n$: $S(x_0, x_1, \dots, x_{n-1}) = (x_{n-1}, x_0, \dots,
x_{n-2})$, e $\omega = \eu^{2\iu\pi/n}$.
\begin{enumerate}
  \item Mostre que $S$ é \omterm{def:b2:hermitian:adjoint}{unitário} e que os vetores $f_k =
        \frac{1}{\sqrt n}\bigl(1, \omega^k, \omega^{2k}, \dots,
        \omega^{(n-1)k}\bigr)$, $0 \leq k < n$, formam uma
        base ortonormal de \omterm{def:b2:reduction:eigen}{autovetores}: $Sf_k = \omega^{-k}
        f_k$.
  \item Deduza que toda matriz \emph{circulante} $C =
        \sum_{j=0}^{n-1} c_jS^j$ é normal, diagonalizada pela
        mesma base, com \omterm{def:b2:reduction:eigen}{autovalores} $\widehat c(k) = \sum_j
        c_j\,\omega^{-jk}$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:11}
Seja $P$ um endomorfismo idempotente ($P^2 = P$) de um \omterm{def:b2:hermitian:def}{espaço
hermitiano}. Prove que $P$ é a projeção \emph{ortogonal} sobre
$\operatorname{im} P$ se e somente se $P^* = P$. Dê a matriz
da projeção ortogonal sobre $\C v$ ($\norm v = 1$) e
sobre um subespaço de base ortonormal $(v_1, \dots, v_k)$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:hermitian:12}
(Projetores espectrais por interpolação) Seja $A$ \omterm{def:b2:hermitian:adjoint}{hermitiana} com
\omterm{def:b2:reduction:eigen}{autovalores} \emph{distintos} $\lambda_1, \dots, \lambda_p$ e
decomposição em \omterm{def:b2:reduction:eigen}{autoespaços} $E = \bigoplus_i E_i$. Defina os
polinômios de Lagrange $L_i(X) = \prod_{j\neq i}\frac{X -
\lambda_j}{\lambda_i - \lambda_j}$. Prove que $P_i = L_i(A)$
é a projeção ortogonal sobre $E_i$, que $P_iP_j = 0$ para
$i \neq j$, $\sum_i P_i = I$ e $A = \sum_i \lambda_iP_i$
(a \emph{decomposição espectral}); exprima $f(A)$ para um polinômio
$f$ qualquer em termos dos $P_i$.
\end{exercise}

\section{Problema: Courant--Fischer, Weyl e o cálculo dos
autovalores}

\begin{problem}\label{pb:b2:hermitian:1}
Os \omterm{def:b2:reduction:eigen}{autovalores} de uma matriz \omterm{def:b2:hermitian:adjoint}{hermitiana} não são apenas raízes de um
polinômio: são soluções de \emph{problemas de otimização}.
Esse ponto de vista variacional --- os \omterm{pb:b2:hermitian:1}{quocientes de Rayleigh} e o
\emph{teorema mín-máx de Courant--Fischer}\index{teorema de Courant--Fischer}
--- torna os \omterm{def:b2:reduction:eigen}{autovalores} comparáveis, estáveis e
calculáveis, e este problema colhe suas safras clássicas:
as \emph{desigualdades de perturbação de Weyl}\index{desigualdade de
Weyl}, o \emph{entrelaçamento de Cauchy}, as desigualdades de traço de Schur e de Ky Fan,
a monotonicidade da raiz quadrada de matrizes
e o \omterm{def:b2:reduction:eigen}{espectro} do laplaciano discreto. Em todo o problema, $A, B,
E$ são \omterm{def:b2:hermitian:adjoint}{hermitianas} em $E = \C^n$ com \omterm{def:b2:reduction:eigen}{autovalores} listados em
ordem decrescente $\lambda_1(A) \geq \dots \geq \lambda_n(A)$,
e $R_A(x) = \frac{\langle x, Ax\rangle}{\langle x, x\rangle}$,
para $x \neq 0$, é o \emph{quociente de Rayleigh}\index{quociente de
Rayleigh}.

\medskip
\noindent\textbf{Parte I --- \omterm{pb:b2:hermitian:1}{Quocientes de Rayleigh} e mín-máx.}
Fixe uma base ortonormal de \omterm{def:b2:reduction:eigen}{autovetores} $(e_1, \dots, e_n)$, $Ae_i =
\lambda_ie_i$.
\begin{enumerate}
  \item Mostre que $R_A(x)$ é real e que
        \[
        \lambda_n \leq R_A(x) \leq \lambda_1
        \qquad (x \neq 0),
        \]
        com as duas cotas atingidas: $\lambda_1 = \max R_A$,
        $\lambda_n = \min R_A$.
  \item Mostre que os pontos críticos de $R_A$ são exatamente os
        \omterm{def:b2:reduction:eigen}{autovetores} de $A$ \emph{(desenvolva $t \mapsto R_A(x +
        tv)$ em $t = 0$ para $v$ arbitrário e depois substitua $v$ por
        $\iu v$)}.
  \item Sejam $V_k = \operatorname{Vect}(e_1, \dots, e_k)$ e
        $W_k = \operatorname{Vect}(e_k, \dots, e_n)$. Mostre que
        \[
        \min_{x \in V_k\setminus\{0\}} R_A(x) = \lambda_k
        = \max_{x \in W_k\setminus\{0\}} R_A(x) .
        \]
  \item Demonstre o \emph{teorema de Courant--Fischer}: para $1 \leq
        k \leq n$,
        \[
        \lambda_k = \max_{\dim V = k}\;\min_{x \in
        V\setminus\{0\}} R_A(x)
        = \min_{\dim W = n-k+1}\;\max_{x \in
        W\setminus\{0\}} R_A(x)
        \]
        \emph{(para qualquer $V$ de dimensão $k$: $V \cap W_k \neq
        \{0\}$ por Grassmann, logo $\min_V R_A \leq \lambda_k$;
        a questão 3 mostra que a cota é atingida)}.
  \item (Monotonicidade) Escreva $A \leq B$ quando $B - A$ é
        positiva semidefinida. Deduza da questão 4: $A \leq
        B$ implica $\lambda_k(A) \leq \lambda_k(B)$ para todo
        $k$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- As desigualdades de Weyl.}
\begin{enumerate}[resume]
  \item Mostre que subespaços $V, W \subseteq \C^n$ com $\dim V
        + \dim W > n$ se intersectam não trivialmente, e generalize:
        $\dim(V_1 \cap V_2 \cap V_3) \geq \dim V_1 + \dim V_2 +
        \dim V_3 - 2n$.
  \item Demonstre a \emph{desigualdade de Weyl}: para $i + j - 1 \leq n$,
        \[
        \lambda_{i+j-1}(A + B) \leq \lambda_i(A) +
        \lambda_j(B)
        \]
        \emph{(intersecte os subespaços $W_i(A)$, $W_j(B)$ e
        $V_{i+j-1}(A+B)$ da questão 3 e conte as
        dimensões)}.
  \item Defina $\vertiii{E}_2 = \max_{\norm x = 1}\norm{Ex}$
        e mostre que $\vertiii E_2 = \max_k\abs{\lambda_k(E)}$ para
        $E$ \omterm{def:b2:hermitian:adjoint}{hermitianas}. Deduza o \emph{teorema de perturbação
        de Weyl}:
        \[
        \bigl|\lambda_k(A + E) - \lambda_k(A)\bigr| \leq
        \vertiii{E}_2
        \qquad (1 \leq k \leq n) :
        \]
        cada \omterm{def:b2:reduction:eigen}{autovalor} é função $1$-lipschitziana da
        matriz.
  \item (Perturbações de posto um) Seja $P$ \omterm{def:b2:hermitian:adjoint}{hermitiana} positiva
        semidefinida de posto $1$. Mostre que
        \[
        \lambda_k(A) \leq \lambda_k(A + P) \leq
        \lambda_{k-1}(A) \qquad (2 \leq k \leq n),
        \]
        junto com $\lambda_1(A) \leq \lambda_1(A+P)$: os
        novos \omterm{def:b2:reduction:eigen}{autovalores} \emph{entrelaçam} os antigos.
  \item Confira a questão 8 numericamente: $A = \begin{pmatrix} 2 &
        \iu\\ -\iu & 2\end{pmatrix}$
        (\cref{exo:b2:hermitian:4}: \omterm{def:b2:reduction:eigen}{espectro} $\{3, 1\}$) e
        $E = \begin{pmatrix} 0 & 1\\ 1 & 0\end{pmatrix}$
        (\omterm{def:b2:reduction:eigen}{espectro} $\{1, -1\}$): calcule o \omterm{def:b2:reduction:eigen}{espectro} de $A +
        E$ e os dois lados da desigualdade.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Entrelaçamento e desigualdades de
traço.}
\begin{enumerate}[resume]
  \item (Entrelaçamento de Cauchy) Seja $B$ a submatriz principal dominante $(n-1)
        \times(n-1)$ de $A$. Prove que
        \[
        \lambda_{k+1}(A) \leq \lambda_k(B) \leq \lambda_k(A)
        \qquad (1 \leq k \leq n-1)
        \]
        \emph{(veja $\C^{n-1} \subseteq \C^n$; nele, $R_B$ é
        a restrição de $R_A$; aplique Courant--Fischer nos
        dois níveis)}.
  \item Itere: para uma submatriz principal $B$ de tamanho $n - m$,
        $\lambda_{k+m}(A) \leq \lambda_k(B) \leq
        \lambda_k(A)$.
  \item (Schur) Sejam $d_1 \geq d_2 \geq \dots \geq d_n$ as
        entradas diagonais de $A$, ordenadas. Prove, para todo $k$:
        \[
        \sum_{i=1}^{k} d_i \leq \sum_{i=1}^{k}\lambda_i(A),
        \]
        com igualdade em $k = n$ (o traço)
        \emph{(as $k$ entradas diagonais escolhidas formam uma
        submatriz principal $k\times k$; majore seu traço pela
        questão 12)}.
  \item (Ky Fan) Prove que
        \[
        \sum_{i=1}^{k}\lambda_i(A)
        = \max\Bigl\{\sum_{i=1}^{k}\langle x_i, Ax_i\rangle
        : (x_1, \dots, x_k) \text{ ortonormal}\Bigr\} .
        \]
  \item Verifique as questões 11 e 13 em
        \[
        A = \begin{pmatrix} 2 & 1 & 0\\ 1 & 2 & 1\\
        0 & 1 & 2\end{pmatrix}
        \qquad
        \bigl(\operatorname{Sp} = \{2 - \sqrt2,\ 2,\
        2+\sqrt2\}\bigr)
        \]
        contra seu bloco dominante $2\times2$ (\omterm{def:b2:reduction:eigen}{espectro} $\{1,
        3\}$) e contra sua diagonal.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A \omterm{def:b2:structures:generated}{ordem} de Loewner.} $A \leq B$
continua significando $B - A$ positiva semidefinida; todas as matrizes desta
parte são \omterm{def:b2:hermitian:adjoint}{hermitianas}.
\begin{enumerate}[resume]
  \item Mostre: $A \leq B$ implica $a_{ii} \leq b_{ii}$ para todos
        $i$, $\operatorname{tr} A \leq \operatorname{tr} B$,
        e $C^\dagger AC \leq C^\dagger BC$ para \emph{toda}
        matriz complexa $C$.
  \item Mostre que elevar ao quadrado \emph{não} é monótono: para
        \[
        A = \begin{pmatrix} 1 & 0\\ 0 & 0\end{pmatrix},
        \qquad
        B = \begin{pmatrix} 2 & 1\\ 1 & 1\end{pmatrix},
        \]
        verifique que $0 \leq A \leq B$ mas $A^2 \not\leq B^2$.
  \item Prove que a raiz quadrada \emph{é} monótona: $0 \leq
        A \leq B$ implica $\sqrt A \leq \sqrt B$ \emph{(seja
        $\mu$ um \omterm{def:b2:reduction:eigen}{autovalor} de $\sqrt B - \sqrt A$ com autovetor
        unitário $v$; calcule $\langle v, (B - A)v\rangle =
        \mu\bigl(\langle v, \sqrt B\,v\rangle + \langle v,
        \sqrt A\,v\rangle\bigr)$ e discuta)}.
  \item Prove que a inversão é antítona nas matrizes positivas
        definidas: $0 < A \leq B$ implica $B^{-1} \leq A^{-1}$
        \emph{(faça a \omterm{def:b2:quadratic:def}{congruência} por $A^{-1/2}$ para reduzir a $I \leq M
        \Rightarrow M^{-1} \leq I$, que é escalar numa
        base espectral)}.
  \item Sejam $A, B$ positivas definidas. Mostre que os
        \omterm{def:b2:reduction:eigen}{autovalores} de $AB$ (não \omterm{def:b2:hermitian:adjoint}{hermitiana} em geral!) são
        reais e positivos, e que
        \[
        \lambda_{\max}(AB) \leq
        \lambda_{\max}(A)\,\lambda_{\max}(B)
        \]
        \emph{(conjugue por $\sqrt A$: $AB \sim \sqrt
        A\,B\sqrt A$)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- O laplaciano discreto, trabalhado.}
Seja $T_n$ a matriz tridiagonal $n \times n$ com $2$ na
diagonal e $-1$ nas duas diagonais adjacentes.
\begin{enumerate}[resume]
  \item Com $\theta_k = \frac{k\pi}{n+1}$, verifique que os
        vetores $v_k = \bigl(\sin(j\theta_k)\bigr)_{1\leq j\leq
        n}$ satisfazem $T_nv_k = (2 - 2\cos\theta_k)\,v_k$
        \emph{(identidade produto-soma; confira as linhas de bordo
        $j = 1, n$)}. Conclua:
        \[
        \operatorname{Sp}(T_n) = \Bigl\{4\sin^2
        \frac{k\pi}{2(n+1)} : 1 \leq k \leq n\Bigr\},
        \]
        todos simples, todos positivos: $T_n$ é positiva definida.
  \item (Um potencial) Para uma diagonal real $D =
        \operatorname{diag}(d_1, \dots, d_n)$, enquadre o
        \omterm{def:b2:reduction:eigen}{espectro}: para todo $k$,
        \[
        \lambda_k(T_n) + \min_i d_i \;\leq\;
        \lambda_k(T_n + D) \;\leq\; \lambda_k(T_n) + \max_i
        d_i .
        \]
  \item Verifique explicitamente o entrelaçamento de Cauchy entre $T_3$ e $T_2$
        (\omterm{def:b2:reduction:eigen}{espectros} $\{2 \pm \sqrt2, 2\}$ e $\{1,
        3\}$) e interprete: $T_2$ é $T_3$ com uma extremidade
        do caminho removida.
  \item Mostre que os \omterm{def:b2:reduction:eigen}{autovalores} extremos satisfazem, quando $n \to
        \infty$:
        \[
        \lambda_{\min}(T_n) = 4\sin^2\frac{\pi}{2(n+1)}
        \sim \frac{\pi^2}{(n+1)^2},
        \qquad
        \lambda_{\max}(T_n) \to 4 ,
        \]
        de modo que o número de condicionamento $\kappa_n =
        \lambda_{\max}/\lambda_{\min}$ cresce como
        $\frac{4(n+1)^2}{\pi^2}$: discretizar uma segunda
        derivada numa malha cada vez mais fina é
        intrinsecamente mal condicionado.
  \item Síntese. Uma frase para cada: (i) por que a
        caracterização variacional, e não o \omterm{def:b2:reduction:charpoly}{polinômio
        característico}, é o que torna os \omterm{def:b2:reduction:eigen}{autovalores} estáveis
        (questões 8--9); (ii) quais questões usaram apenas
        o $\lambda_1 = \max R_A$ e quais precisaram do
        mín-máx \omterm{def:b2:metric:complete}{completo}; (iii) o que a ordem de Loewner acrescenta à
        história; (iv) onde essas ferramentas reaparecem (análise
        numérica das matrizes de rigidez da questão 24; teoria
        quântica de perturbações; e, no volume do terceiro ano de graduação, o
        princípio mín-máx para operadores \omterm{def:b2:hermitian:adjoint}{autoadjuntos}
        \omterm{def:b2:metric:compact}{compactos}).
\end{enumerate}
\end{problem}
