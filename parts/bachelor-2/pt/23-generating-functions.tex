\chapter{Funções geradoras de probabilidade}\label{ch:b2:genfun}

As séries de potências do~\cref{ch:b2:powerseries} voltam com uma
missão probabilística: a uma \omterm{def:b2:randomvar:law}{variável aleatória} com valores em $\N$ associamos
a série de potências de coeficientes $\P(X = n)$. Essa
\emph{\omterm{ex:b2:powerseries:fibonacci}{função geradora}} converte somas de variáveis \omterm{def:b2:proba:independence}{independentes}
em produtos, momentos em derivadas em $1$ e identidades
combinatórias difíceis em multiplicações de uma linha. O
capítulo fecha o livro com duas peças de exibição: a aproximação
de Poisson de \omterm{def:b2:proba:space}{eventos} raros e o critério de extinção para
\omterm{pb:b2:genfun:1}{processos de ramificação} --- um cálculo probabilístico genuinamente
infinito resolvido inteiramente pela geometria de uma curva \omterm{def:b2:affine:convex}{convexa}.

\section{Definição e propriedades básicas}

\begin{definition}[Função geradora de probabilidade]\label{def:b2:genfun:pgf}
Seja $X$ uma \omterm{def:b2:randomvar:law}{variável aleatória} com valores em $\N$, $p_n = \P(X = n)$. A
\emph{função geradora de probabilidade}\index{função geradora de
probabilidade} de $X$ é a soma da
série de potências
\[
G_X(t) = \E\bigl(t^X\bigr) = \sum_{n=0}^{\infty} p_n\,t^n .
\]
\end{definition}

\begin{example}[Primeiros reflexos]\label{ex:b2:genfun:firstreflexes}
Uma variável constante $X = c$ tem $G_X(t) = t^c$; um deslocamento obedece a
$G_{X+c}(t) = t^c\,G_X(t)$; e avaliar em pontos especiais
extrai informação sem desenvolvimento algum: $G_X(0) = \P(X
= 0)$, $G_X(1) = 1$ e $G_X(-1) = \P(X\text{ par}) -
\P(X\text{ ímpar})$, o balanço de paridades explorado no
\cref{exo:b2:genfun:10}. Essas frases de uma linha são usadas silenciosamente
em toda parte abaixo --- e a avaliação $G_X(0)$ é exatamente
como as probabilidades de extinção serão extraídas de \omterm{ex:b2:powerseries:fibonacci}{funções
geradoras} iteradas no fim do capítulo.
\end{example}

\begin{proposition}[Raio e primeiras propriedades]\label{prop:b2:genfun:radius}
A série que define $G_X$ tem \omterm{def:b2:powerseries:radius}{raio de convergência} $\geq 1$;
$G_X$ está definida e é \omterm{def:b2:metric:continuity}{contínua} em $\intcc{-1}{1}$,
$\mathcal{C}^\infty$ em $\intoo{-1}{1}$, com $G_X(1) = 1$ e
$\abs{G_X(t)} \leq 1$ ali. Além disso, $G_X$ determina a \omterm{def:b2:randomvar:law}{lei} de
$X$:
\[
p_n = \frac{G_X^{(n)}(0)}{n!} .
\]
\end{proposition}

\begin{proof}
Como $\sum p_n = 1$ converge, os termos $p_n\,1^n$ são limitados,
de modo que o raio é $\geq 1$ (lema de Abel,
\cref{ch:b2:powerseries}); em $t = \pm1$ a série converge
\omterm{def:b2:series:def}{absolutamente} ($\sum p_n = 1$ a domina); melhor, em todo o
intervalo $\intcc{-1}1$,
\[
\sup_{\abs t\leq1}\,\abs{p_nt^n} = p_n
\quad\text{com}\quad \sum_np_n < \infty :
\]
a série converge \emph{\omterm{def:b2:funcseq:series}{normalmente}} em $\intcc{-1}1$, de modo que
sua soma é \omterm{def:b2:metric:continuity}{contínua} ali
(\cref{thm:b2:funcseq:weierstrass,thm:b2:funcseq:continuity}). A suavidade no interior e a fórmula dos
coeficientes são a teoria geral das séries de potências; sendo os coeficientes
recuperáveis, duas variáveis com a mesma \omterm{ex:b2:powerseries:fibonacci}{função geradora}
têm a mesma \omterm{def:b2:randomvar:law}{lei}.
\end{proof}

\begin{example}[As leis clássicas]\label{ex:b2:genfun:classical}
\leavevmode
\begin{itemize}
\item Bernoulli $\mathcal{B}(p)$: $G(t) = 1 - p + pt$.
\item Binomial $\mathcal{B}(n, p)$: $G(t) = \sum_k \binom nk
(pt)^k(1-p)^{n-k} = (1 - p + pt)^n$ (teorema binomial).
\item Geométrica $\mathcal{G}(p)$: $G(t) =
\sum_{k\geq1}(1-p)^{k-1}p\,t^k = \dfrac{pt}{1 - (1-p)t}$ (raio
$\frac{1}{1-p} > 1$).
\item Poisson $\mathcal{P}(\lambda)$: $G(t) =
\sum_k e^{-\lambda}\frac{(\lambda t)^k}{k!} = e^{\lambda(t - 1)}$
(raio $\infty$).
\end{itemize}
\end{example}

\begin{example}[Integrando a função
geradora]\label{ex:b2:genfun:integraltrick}
As derivadas de $G_X$ em $1$ dão momentos positivos; a
\emph{integral} dá um momento negativo. De $\int_0^1t^k\dd t
= \frac1{k+1}$ e da integração termo a termo (\omterm{def:b2:funcseq:series}{convergência
normal} em $\intcc01$):
\[
\int_0^1G_X(t)\,\dd t = \sum_{k\geq0}\frac{\P(X =
k)}{k+1} = \E\Bigl(\frac1{1+X}\Bigr).
\]
Para $X \sim \mathcal P(\lambda)$:
\[
\E\Bigl(\frac1{1+X}\Bigr) =
\int_0^1\eu^{\lambda(t-1)}\,\dd t = \frac{1 -
\eu^{-\lambda}}{\lambda},
\]
recuperando numa linha o cálculo em séries do
\cref{ex:b2:randomvar:transferex}. A \omterm{ex:b2:powerseries:fibonacci}{função geradora}
é um instrumento de mão dupla: derive em $1$ para os
momentos $\E(X)$, $\E(X(X-1))$, integre em $\intcc01$
para $\E\bigl(\frac1{1+X}\bigr)$ --- um só objeto \omterm{def:b2:powerseries:analytic}{analítico},
consultado na direção de que o problema precisa.
\end{example}

\begin{example}[Uma lei de raio exatamente
um]\label{ex:b2:genfun:heavytail}
Seja $\P(X = k) = \dfrac{6}{\pi^2k^2}$ para $k \geq 1$ --- uma
\omterm{def:b2:randomvar:law}{lei} de probabilidade pela identidade de Basileia
(\cref{ex:b2:fourier:basel}). Sua \omterm{ex:b2:powerseries:fibonacci}{função geradora}
$G(t) = \frac6{\pi^2}\sum_{k\geq1}\frac{t^k}{k^2}$ tem \omterm{def:b2:powerseries:radius}{raio
de convergência} exatamente $1$: a cota geral ``raio $\geq
1$'' da~\cref{prop:b2:genfun:radius} não pode ser melhorada. E
a média vale
\[
\sum_{k\geq1}k\,\P(X = k) =
\frac6{\pi^2}\sum_{k\geq1}\frac1k = \infty :
\]
$G$ é \omterm{def:b2:metric:continuity}{contínua} em $\intcc{-1}1$, suave no interior, mas sua
derivada explode em $1^-$ --- o gráfico chega ao
ponto $(1, 1)$ com tangente vertical. As caudas pesadas são
visíveis \emph{geometricamente} na \omterm{ex:b2:powerseries:fibonacci}{função geradora}, no
único ponto $t = 1$; o teorema dos momentos abaixo torna
essa correspondência exata.
\end{example}

\begin{theorem}[Momentos a partir da função
geradora]\label{thm:b2:genfun:moments}
$X$ tem \omterm{def:b2:randomvar:expectation}{esperança} se e somente se $G_X$ é \omterm{def:b2:diffcalc:differential}{diferenciável} em
$1^-$ (derivada à esquerda, finita), e então $\E(X) = G_X'(1)$.
Do mesmo modo, $X$ tem momento de segunda ordem se e somente se $G_X$ é duas vezes
\omterm{def:b2:diffcalc:differential}{diferenciável} em $1^-$, e então
\[
\E\bigl(X(X - 1)\bigr) = G_X''(1),
\qquad
V(X) = G_X''(1) + G_X'(1) - G_X'(1)^2 .
\]
\end{theorem}

\begin{proof}
Para $t \in \intoo{0}{1}$, a derivação termo a termo dentro do
disco dá $G_X'(t) = \sum_{n\geq1} np_n t^{n-1}$, uma série com
coeficientes não negativos: $t \mapsto G_X'(t)$ é não decrescente em
$\intoo{0}{1}$ e, por convergência monótona das somas parciais (ou pelo
teorema de Abel para coeficientes não negativos,
\cref{ch:b2:powerseries}),
\[
\lim_{t \to 1^-} G_X'(t)
= \sum_{n\geq1} n\,p_n \in \intcc{0}{+\infty} ,
\]
cada membro finito exatamente quando o outro o é. Quando finito, o teorema do
valor médio espreme os quocientes de diferenças $\frac{G_X(1) -
G_X(t)}{1 - t}$ entre valores de $G_X'$, de modo que $G_X$ é
\omterm{def:b2:diffcalc:differential}{diferenciável} em $1^-$ com $G_X'(1) = \sum np_n = \E(X)$ (por
transferência). O enunciado de segunda ordem repete o argumento um nível
acima: $G''_X(t) = \sum_{n\geq2}n(n-1)p_nt^{n-2}$ é
não decrescente em $\intoo01$ com limite monótono
$\sum_nn(n-1)p_n = \E(X(X-1))$, finito exatamente quando $X$ tem
momento de segunda ordem. A fórmula da \omterm{def:b2:randomvar:variance}{variância} segue então de
König--Huygens:
\[
V(X) = \E(X^2) - \E(X)^2 = \E\bigl(X(X-1)\bigr) + \E(X) -
\E(X)^2 = G''_X(1) + G'_X(1) - G'_X(1)^2 .
\]
\end{proof}

\begin{example}
Poisson: $G'(t) = \lambda e^{\lambda(t-1)}$, logo $\E(X) = \lambda$;
$G''(1) = \lambda^2$, logo $V(X) = \lambda^2 + \lambda - \lambda^2 =
\lambda$ --- os cálculos do~\cref{ch:b2:randomvar} em uma
linha cada.
\end{example}

\begin{example}[A moda de uma lei de Poisson]
\label{ex:b2:genfun:poissonmode}
Onde $\P(X = k)$ é maior para $X \sim \mathcal
P(\lambda)$? Pesos consecutivos se comparam pela razão
\[
\frac{\P(X = k+1)}{\P(X = k)} = \frac{\lambda}{k + 1} ,
\]
que excede $1$ enquanto $k < \lambda - 1$ e cai abaixo de
$1$ assim que $k > \lambda - 1$: os pesos sobem e depois descem, com
moda $\floor\lambda$ (e um empate entre $\lambda - 1$ e
$\lambda$ quando $\lambda$ é inteiro: para $\lambda = 3$,
$\P(X = 2) = \P(X = 3) = \frac92\eu^{-3} \approx 0.224$).
Testes de razão sobre os coeficientes são muitas vezes a via mais rápida
para fatos qualitativos sobre uma \omterm{def:b2:randomvar:law}{lei} discreta --- sem precisar de \omterm{ex:b2:powerseries:fibonacci}{função
geradora}, mas os coeficientes \emph{são} a
\omterm{ex:b2:powerseries:fibonacci}{função geradora}, lida termo a termo.
\end{example}

\section{Somas de variáveis independentes}

\begin{theorem}[Multiplicatividade]\label{thm:b2:genfun:product}
Se $X$ e $Y$ são \omterm{def:b2:randomvar:law}{variáveis aleatórias} \omterm{def:b2:proba:independence}{independentes} com valores em $\N$, então
\[
G_{X + Y}(t) = G_X(t)\,G_Y(t)
\qquad (\abs t \leq 1),
\]
e, por indução, $G_{X_1 + \dots + X_n} = \prod_i G_{X_i}$ para
$X_1, \dots, X_n$ \omterm{def:b2:proba:independence}{independentes}.
\end{theorem}

\begin{proof}
Duas demonstrações, ambas instrutivas. \emph{Via \omterm{def:b2:randomvar:expectation}{esperanças}:} $t^X$ e
$t^Y$ são variáveis limitadas \omterm{def:b2:proba:independence}{independentes}, de modo que
(\cref{thm:b2:randomvar:product})
\[
G_{X+Y}(t) = \E\bigl(t^{X+Y}\bigr)
= \E\bigl(t^X t^Y\bigr)
= \E\bigl(t^X\bigr)\E\bigl(t^Y\bigr) .
\]
\emph{Via \omterm{thm:b2:series:fubini}{produtos de Cauchy}:} a \omterm{def:b2:randomvar:law}{lei} de $X + Y$ é a convolução
$\P(X + Y = n) = \sum_{k=0}^n \P(X = k)\P(Y = n - k)$, e o teorema
do \omterm{thm:b2:series:fubini}{produto de Cauchy} para séries \omterm{def:b2:series:def}{absolutamente} convergentes
(\cref{ch:b2:series}) multiplica as duas séries de potências exatamente
ao longo dessa convolução.
\end{proof}

\begin{example}[Estabilidade das leis clássicas]\label{ex:b2:genfun:stability}
Binomiais \omterm{def:b2:proba:independence}{independentes} com o mesmo $p$ se somam: $(1 - p + pt)^m(1 -
p + pt)^n = (1 - p + pt)^{m+n}$, logo $\mathcal{B}(m, p) +
\mathcal{B}(n, p) = \mathcal{B}(m + n, p)$ --- em particular, uma soma
de $n$ variáveis de Bernoulli \omterm{def:b2:proba:independence}{independentes} é binomial, redemonstrando a
\omterm{def:b2:randomvar:law}{lei} do número de sucessos. Poissons \omterm{def:b2:proba:independence}{independentes} se somam:
$e^{\lambda(t-1)}e^{\mu(t-1)} = e^{(\lambda + \mu)(t-1)}$, logo
$\mathcal{P}(\lambda) + \mathcal{P}(\mu) = \mathcal{P}(\lambda +
\mu)$ --- o cálculo de convolução do
\cref{exo:b2:randomvar:2}, agora sem cálculo algum.
\end{example}

\begin{example}[Dois dados, um polinômio ao
quadrado]\label{ex:b2:genfun:twodice}
Para um dado honesto, $G(t) = \frac{t + t^2 + \dots + t^6}{6}$;
para a soma de dois,
\[
G(t)^2 = \frac{1}{36}\bigl(t^2 + 2t^3 + 3t^4 + 4t^5 + 5t^6 +
6t^7 + 5t^8 + 4t^9 + 3t^{10} + 2t^{11} + t^{12}\bigr) :
\]
a \omterm{def:b2:randomvar:law}{lei} triangular das somas de dados ($7$ é a moda, com
probabilidade $\frac6{36} = \frac16$), lida num quadrado de
polinômio que se multiplica uma vez na vida. A
fórmula de convolução teria exigido onze argumentos de
contagem separados; a \omterm{ex:b2:powerseries:fibonacci}{função geradora} os faz todos
simultaneamente, porque multiplicar polinômios \emph{é}
convoluir coeficientes. Essa tradução mecânica ---
\omterm{def:b2:randomvar:law}{leis} em coeficientes, somas em produtos --- é todo o
modelo de negócios do capítulo, e o
\cref{exo:b2:genfun:11} o leva até os surpreendentes
dados de Sicherman.
\end{example}

\begin{example}[Três dados e uma extração de
coeficiente]\label{ex:b2:genfun:threedice}
Para a soma $S$ de três dados honestos, $\P(S = 10)$ é o
coeficiente de $t^{10}$ em $\bigl(\frac{t + \dots +
t^6}6\bigr)^3$. Fatore e desenvolva com as séries binomial e
geométrica:
\[
\Bigl(\frac{t(1 - t^6)}{6(1 - t)}\Bigr)^{\!3}
= \frac{t^3}{216}\,\bigl(1 - 3t^6 + 3t^{12} -
t^{18}\bigr)\sum_{j\geq0}\binom{j+2}2t^j .
\]
O coeficiente de $t^{10}$ exige $t^7$ do produto:
$j = 7$ com o termo $1$, e $j = 1$ com o termo
$-3t^6$:
\[
\P(S = 10) = \frac{1}{216}\Bigl(\binom92 -
3\binom32\Bigr) = \frac{36 - 9}{216} = \frac{27}{216} =
\frac18 .
\]
A enumeração direta das $27$ triplas é propensa a erros; a
álgebra é mecânica e escala para qualquer número de dados ---
a inclusão--exclusão visível em $(1 - t^6)^3$ faz
a análise de casos automaticamente.
\end{example}

\begin{example}[Lendo uma lei em sua função
geradora]\label{ex:b2:genfun:identify}
Que \omterm{def:b2:randomvar:law}{lei} tem $G(t) = \dfrac1{2 - t}$? Desenvolva em série
de potências:
\[
\frac{1}{2 - t} = \frac12\cdot\frac1{1 - t/2}
= \sum_{k\geq0}\frac{t^k}{2^{k+1}} :
\]
coeficientes não negativos de soma $G(1) = 1$, de modo que essa é uma
\omterm{def:b2:randomvar:law}{lei} genuína, $\P(X = k) = 2^{-(k+1)}$ em $\N$ --- uma
\omterm{def:b2:randomvar:law}{lei} geométrica começando em $0$. Por unicidade
(\cref{prop:b2:genfun:radius}), nenhuma outra \omterm{def:b2:randomvar:law}{lei} compartilha essa
$G$. Reconhecer \omterm{def:b2:randomvar:law}{leis} a partir de suas \omterm{ex:b2:powerseries:fibonacci}{funções geradoras} é uma
habilidade que vale treinar: é assim que o iterado crítico de
ramificação $G_n(t) = \frac{n - (n-1)t}{n+1 - nt}$ do
problema de fim de semana é desmascarado como uma \omterm{def:b2:randomvar:law}{lei} geométrica condicionada
à sobrevivência.
\end{example}

\begin{remark}
A estabilidade vale num sentido só: somas de Poissons \omterm{def:b2:proba:independence}{independentes} são
Poisson, mas diferenças não são --- $X - Y$ assume valores
negativos, de modo que não tem \omterm{ex:b2:powerseries:fibonacci}{função geradora} alguma, e sua \omterm{def:b2:randomvar:law}{lei}
(a \omterm{def:b2:randomvar:law}{distribuição} de Skellam) fica fora do instrumental deste
capítulo. Do mesmo modo, $\mathcal B(m, p) + \mathcal B(n, p')$ com
$p \neq p'$ \emph{não} é binomial: o produto $(1 - p +
pt)^m(1 - p' + p't)^n$ tem duas localizações de raiz distintas, ao passo que
toda \omterm{ex:b2:powerseries:fibonacci}{função geradora} binomial tem uma única raiz repetida. Ler a
estabilidade nos padrões de raízes é uma pequena antevisão de quanta
estrutura o polinômio codifica.
\end{remark}

\begin{remark}[O filtro das raízes da unidade]
Avaliar em $-1$ separa pares de ímpares; avaliar em todas as
raízes $m$-ésimas da unidade separa cada classe de resto: com
$\omega = \eu^{2\iu\pi/m}$,
\[
\P(X \equiv r \bmod m)
= \frac1m\sum_{j=0}^{m-1}\omega^{-jr}\,G_X(\omega^j),
\]
pois a média de $\omega^{j(k-r)}$ sobre $j$ dá $1$ se $k
\equiv r$ e $0$ caso contrário. Dividendo de amostra: para a soma $S$
de dois dados honestos, cada $G(\omega^j) = \frac16\sum_{k=1}^6
\omega^{jk} = -\frac16$ para $j \neq 0$ (as sete raízes sétimas
da unidade somam zero), logo
\[
\P(7 \mid S) = \frac17\Bigl(1 +
6\cdot\frac1{36}\Bigr) = \frac16 ,
\]
confirmando a contagem do~\cref{ex:b2:genfun:twodice} ---
e o método escala para questões em que a contagem direta
não escala.
\end{remark}

\begin{theorem}[Somas aleatórias: identidade de Wald para funções
geradoras]\label{thm:b2:genfun:compound}
Sejam $(X_k)_{k\geq1}$ variáveis \omterm{def:b2:proba:independence}{independentes} com valores em $\N$, de mesma
\omterm{def:b2:randomvar:law}{lei} e \omterm{ex:b2:powerseries:fibonacci}{função geradora} $G_X$, e seja $N$ uma
variável com valores em $\N$ \omterm{def:b2:proba:independence}{independente} das $X_k$, de \omterm{ex:b2:powerseries:fibonacci}{função
geradora} $G_N$. Então a soma aleatória $S = X_1 + \dots + X_N$ (com
$S = 0$ quando $N = 0$) tem \omterm{ex:b2:powerseries:fibonacci}{função geradora}
\[
G_S = G_N \circ G_X .
\]
Em particular, se $N$ e $X_1$ têm \omterm{def:b2:randomvar:expectation}{esperança}, $\E(S) =
\E(N)\,\E(X_1)$.
\end{theorem}

\begin{proof}
Condicione a $N$ (probabilidade total,
\cref{thm:b2:proba:bayes}): para $\abs t \leq 1$,
\[
G_S(t) = \sum_{n=0}^\infty \P(N = n)\,
\E\bigl(t^{X_1 + \dots + X_n}\bigr)
= \sum_{n=0}^\infty \P(N = n)\,G_X(t)^n
= G_N\bigl(G_X(t)\bigr),
\]
usando a multiplicatividade para cada $n$ fixo e a \omterm{def:b2:series:summable}{somabilidade} de
toda a família dupla ($\abs{G_X(t)} \leq 1$). A troca de
somas é Fubini para \omterm{def:b2:series:summable}{famílias somáveis}
(\cref{ch:b2:series}). Derivando em $1^-$ pela regra da cadeia
e \cref{thm:b2:genfun:moments}: $\E(S) = G_N'(G_X(1))\,G_X'(1)
= G_N'(1)G_X'(1) = \E(N)\E(X_1)$.
\end{proof}

\begin{example}[Poisson composta: perdas anuais de
seguro]\label{ex:b2:genfun:compoundpoisson}
Uma seguradora recebe $N \sim \mathcal P(\lambda)$ sinistros num
ano, custando cada sinistro $X_k$ (unidades inteiras, i.i.d., com \omterm{ex:b2:powerseries:fibonacci}{função geradora}
$G_X$, média $\mu$, independentes de $N$). Pelo
\cref{thm:b2:genfun:compound}, a perda total $S$ tem
\[
G_S(t) = \eu^{\lambda(G_X(t) - 1)},
\qquad
\E(S) = \lambda\mu ,
\]
e, derivando duas vezes em $1^-$:
\[
V(S) = \lambda\,G_X''(1) + \lambda^2\mu^2 + \lambda\mu -
(\lambda\mu)^2 = \lambda\,\E(X^2) .
\]
A \omterm{def:b2:randomvar:variance}{variância} envolve o \emph{segundo} momento de um único
sinistro, não sua \omterm{def:b2:randomvar:variance}{variância}: uma soma de Poisson composta sente o
sinistro grande ocasional duas vezes --- uma pelo quantos, outra
pelo quão grande. Para $\lambda = 10$ sinistros de \omterm{def:b2:randomvar:law}{lei} geométrica
de média $2$ ($\E X^2 = 6$): $\E S = 20$, $V(S) = 60$, e
Chebyshev (\cref{ch:b2:randomvar}) já fornece margens de
solvência utilizáveis. Esse padrão de ``soma parada aleatoriamente'' é o
mesmo que moverá a recursão de ramificação do
\cref{prop:b2:genfun:branching}: a composição de \omterm{ex:b2:powerseries:fibonacci}{funções
geradoras} é a \omterm{def:b2:structures:algebra}{álgebra} das populações aleatórias.
\end{example}

\begin{remark}
A \omterm{def:b2:proba:independence}{independência} de $N$ em relação às parcelas não é decorativa.
Tome $X_k \in \{0, 2\}$ com probabilidades iguais e ponha $N =
X_1$ (flagrantemente dependente): então $S = X_1 + \dots + X_N$ vale
$0$ quando $X_1 = 0$, e $2 + X_2$ quando $X_1 = 2$, de modo que $\E(S) =
\frac12(2 + 1) = \frac32$, enquanto $\E(N)\E(X_1) = 1\cdot1 =
1$: a identidade de Wald falha. Quando o número de parcelas
pode \emph{reagir} às próprias parcelas, a estrutura limpa de
produto desmorona --- a teoria completa dessas regras de
``parada'' é o capítulo de martingais do volume do terceiro ano
de graduação.
\end{remark}

\section{Aproximação de Poisson}

\begin{theorem}[Lei dos eventos raros]\label{thm:b2:genfun:rareevents}
Seja $X_n \sim \mathcal{B}(n, p_n)$ com $n\,p_n \to \lambda > 0$.
Então, para todo $k \in \N$:
\[
\P(X_n = k)
\xrightarrow[n\to\infty]{}
e^{-\lambda}\frac{\lambda^k}{k!} :
\]
a \omterm{def:b2:randomvar:law}{lei} binomial de muitos \omterm{def:b2:proba:independence}{eventos independentes} raros converge à
\omterm{def:b2:randomvar:law}{lei} de Poisson de parâmetro $\lambda$.
\end{theorem}

\begin{proof}
Cálculo direto com $p_n = \frac{\lambda_n}{n}$, $\lambda_n
\to \lambda$:
\[
\P(X_n = k)
= \binom nk p_n^k(1 - p_n)^{n-k}
= \frac{n(n-1)\cdots(n-k+1)}{n^k}\cdot
\frac{\lambda_n^k}{k!}\,
\bigl(1 - \tfrac{\lambda_n}{n}\bigr)^{n-k} .
\]
Quando $n \to \infty$ com $k$ fixo: o primeiro fator tende a $1$
(produto de $k$ fatores $\to 1$); $\lambda_n^k \to \lambda^k$;
e $\bigl(1 - \frac{\lambda_n}{n}\bigr)^{n-k} =
\exp\bigl((n-k)\ln(1 - \frac{\lambda_n}{n})\bigr) \to
e^{-\lambda}$ pois $(n - k)\ln\bigl(1 - \frac{\lambda_n}{n}\bigr)
\sim -\lambda_n \to -\lambda$
(\cref{ch:b2:comparison}). Alternativamente, no nível das
\omterm{ex:b2:powerseries:fibonacci}{funções geradoras}: $G_{X_n}(t) = \bigl(1 +
\frac{\lambda_n(t-1)}{n}\bigr)^n \to e^{\lambda(t - 1)} =
G_{\mathcal{P}(\lambda)}(t)$ para cada $t \in [0, 1]$ fixo ---
convergência de \omterm{ex:b2:powerseries:fibonacci}{funções geradoras} que (para variáveis com valores em $\N$)
é equivalente à convergência de cada $\P(X_n = k)$;
veja o~\cref{exo:b2:genfun:9}.
\end{proof}

\begin{remark}
É por isso que as \omterm{def:b2:randomvar:law}{leis} de Poisson modelam contagens de \omterm{def:b2:proba:space}{eventos} raros --- erros de digitação por
página, decaimentos radioativos por segundo, acidentes por dia num
cruzamento: cada oportunidade é quase desprezível, as oportunidades são
muitas, e só a taxa média $\lambda$ sobrevive no limite.
\end{remark}

\begin{example}[Vendo o limite de Poisson
convergir]\label{ex:b2:genfun:poissonnumerics}
Fixe $\lambda = 2$ e faça $X_n \sim \mathcal B(n, 2/n)$. A
probabilidade de nenhum \omterm{def:b2:proba:space}{evento} é exatamente $\P(X_n = 0) = (1 - 2/n)^n$:
\[
n = 10:\ 0.107, \qquad
n = 20:\ 0.122, \qquad
n = 50:\ 0.130, \qquad
n = 100:\ 0.133,
\]
contra o limite $\eu^{-2} \approx 0.135$. A convergência
é monótona e de velocidade $O(1/n)$ --- desenvolvendo, $(1 -
2/n)^n = \eu^{-2}\bigl(1 - \tfrac2n + O(n^{-2})\bigr)$ ---
de modo que, para $n$ na casa das centenas, o modelo de Poisson já é
preciso até o terceiro dígito. Esse é o conteúdo prático
da lei dos eventos raros: quem modela nunca conhece $n$ e
$p$ separadamente (quantas micro-oportunidades de erro de digitação
uma página contém?), mas apenas seu produto $\lambda$, e a
\omterm{def:b2:randomvar:law}{lei} limite misericordiosamente não depende de mais nada.
\end{example}

\section{Processos de ramificação}

Considere uma população que parte de um único ancestral; cada indivíduo,
independentemente, tem um número aleatório de filhos com \omterm{def:b2:randomvar:law}{lei} $(p_k)_{k
\in \N}$ e \omterm{ex:b2:powerseries:fibonacci}{função geradora} $G$ (a \emph{\omterm{def:b2:randomvar:law}{distribuição}
de descendência}). Seja $Z_n$ o tamanho da geração $n$ ($Z_0 =
1$), e seja $m = G'(1) = \E(Z_1)$ o número médio de filhos.

\begin{proposition}\label{prop:b2:genfun:branching}
A \omterm{ex:b2:powerseries:fibonacci}{função geradora} de $Z_n$ é o $n$-ésimo iterado
$G_{Z_n} = G \circ G \circ \dots \circ G$ ($n$ vezes), e as
probabilidades de extinção $q_n = \P(Z_n = 0)$ satisfazem
\[
q_0 = 0, \qquad q_{n+1} = G(q_n),
\]
e crescem para a probabilidade $q$ de extinção eventual, que
é um ponto fixo de $G$.
\end{proposition}

\begin{proof}
A geração $n + 1$ é a soma aleatória da descendência dos $Z_n$
membros da geração $n$, com contagens \omterm{def:b2:proba:independence}{independentes} entre si
e de $Z_n$: o~\cref{thm:b2:genfun:compound} dá $G_{Z_{n+1}} =
G_{Z_n} \circ G$, e a indução a partir de $G_{Z_0}(t) = t$ dá o
iterado $n$ vezes --- que, pela associatividade da composição, pode
igualmente ser lido como $G_{Z_{n+1}} = G \circ G_{Z_n}$. Avaliando
essa segunda forma em $0$: $q_{n+1} = G_{Z_{n+1}}(0) =
G\bigl(G_{Z_n}(0)\bigr) = G(q_n)$. Os \omterm{def:b2:proba:space}{eventos} $\{Z_n = 0\}$ crescem (populações extintas permanecem
extintas), de modo que $q_n \uparrow q = \P\bigl(\bigcup_n\{Z_n = 0\}\bigr)$
pela \omterm{def:b2:metric:continuity}{continuidade} monótona (\cref{thm:b2:proba:continuity}), e a
\omterm{def:b2:metric:continuity}{continuidade} de $G$ em $[0, 1]$ transforma $q_{n+1} = G(q_n)$ em $q =
G(q)$ no limite.
\end{proof}

\begin{example}[Vendo a extinção convergir]
\label{ex:b2:genfun:cobwebnumerics}
Para a \omterm{def:b2:randomvar:law}{lei} de descendência $(p_0, p_1, p_2) = (\tfrac14,
\tfrac14, \tfrac12)$ de \cref{ex:b2:genfun:branchingexample},
$G(t) = \tfrac14 + \tfrac14t + \tfrac12t^2$ e a iteração
$q_{n+1} = G(q_n)$ dá
\[
q_1 = 0.25, \quad q_2 = 0.34375, \quad q_3 \approx 0.39502,
\quad q_4 \approx 0.42678, \quad q_5 \approx 0.44776,
\]
subindo rumo à probabilidade de extinção $q = \tfrac12$.
As diferenças $q - q_n$ valem $0.25$, $0.156$, $0.105$, $0.073$,
$0.052$: cada uma é aproximadamente $\tfrac34$ da anterior e,
de fato, o teorema do valor médio dá $q - q_{n+1} = G'(c_n)(q
- q_n)$ com $G'(q) = \tfrac14 + q = \tfrac34$. Duas morais: uma
linhagem ainda viva na geração $n$ tem, embutida
no mesmo cálculo, probabilidade $q - q_n$ de estar condenada
mais tarde; e a taxa de convergência da escada da
figura abaixo é a derivada no ponto fixo --- o
problema de fim de semana transforma ambas as observações em teoremas.
\end{example}

\begin{theorem}[Critério de extinção]\label{thm:b2:genfun:extinction}
Suponha $p_1 \neq 1$. A probabilidade de extinção $q$ é o
\emph{menor} ponto fixo de $G$ em $\intcc{0}{1}$, e:
\begin{itemize}
\item se $m \leq 1$ (subcrítico ou crítico), $q = 1$: a extinção
é certa;
\item se $m > 1$ (supercrítico), $q < 1$: a população sobrevive
para sempre com probabilidade positiva $1 - q$.
\end{itemize}
\end{theorem}

\begin{proof}
$G$ é \omterm{def:b2:affine:convex}{convexa} em $\intcc{0}{1}$ (série de potências com coeficientes
não negativos: $G'' \geq 0$), não decrescente, com $G(1) = 1$.

\emph{Menor ponto fixo:} seja $r \in \intcc{0}{1}$ um ponto fixo
qualquer. Então $q_0 = 0 \leq r$ e, indutivamente, $q_{n+1} = G(q_n)
\leq G(r) = r$ (monotonia): logo $q = \lim q_n \leq r$.

\emph{Caso $m \leq 1$:} suponha que $r < 1$ seja um ponto fixo. Pelo
teorema do valor médio em $[r, 1]$, existe $c \in \intoo{r}{1}$ com
$G'(c) = \frac{G(1) - G(r)}{1 - r} = \frac{1 - r}{1 - r} = 1$. Mas
$G'$ é não decrescente (convexidade) com $\lim_{t\to1^-}G'(t) = m
\leq 1$, logo $G' \leq 1$ em $\intoo{0}{1}$; a igualdade $G'(c) =
1$ força então $G'$ a ser constante igual a $1$ em $\intco{c}{1}$,
portanto $G'' = \sum n(n-1)p_nt^{n-2} \equiv 0$ ali. Uma série de potências
com coeficientes não negativos que se anula num intervalo tem todos
esses coeficientes nulos: $p_n = 0$ para $n \geq 2$, de modo que $G(t) = p_0
+ p_1t$ e $1 = G'(c) = p_1$ --- contradizendo a hipótese
$p_1 \neq 1$. Assim, $1$ é o único ponto fixo: $q = 1$.

\emph{Caso $m > 1$:} perto de $1$, $G(t) - t$ tem derivada $G'(t) -
1 \to m - 1 > 0$ quando $t \to 1^-$, logo $G(t) - t < G(1) - 1 = 0$ em
algum intervalo $\intoo{1 - \delta}{1}$: a função \omterm{def:b2:metric:continuity}{contínua}
$G(t) - t$ vale $\geq 0$ em $t = 0$ ($G(0) = p_0 \geq 0$) e é $< 0$
logo abaixo de $1$, de modo que ela se anula em algum $r < 1$ (teorema do valor
intermediário). O menor ponto fixo é então $q \leq r < 1$.
\end{proof}

\begin{remark}[Como ler o diagrama de teia]
Na figura, um movimento vertical aplica $G$ (de $(q_n, q_n)$
até $(q_n, G(q_n))$), e um movimento horizontal até a diagonal
converte saída em entrada: a escada \emph{é} a
recursão $q_{n+1} = G(q_n)$. A convexidade de $G$ e $G(1) = 1$
deixam apenas duas geometrias. Ou a curva fica acima da
diagonal em $\intco01$ (média $m \leq 1$): a escada não tem
onde parar antes de $1$. Ou a curva cruza em algum $q <
1$ ($m > 1$): a escada fica presa abaixo do cruzamento
e converge para ele, à taxa geométrica $G'(q) < 1$
quantificada no~\cref{ex:b2:genfun:cobwebnumerics}. Toda a
análise do teorema de extinção é visível nessa única
imagem --- e é por isso que vale a pena desenhá-la antes de
calcular.
\end{remark}

\begin{example}\label{ex:b2:genfun:branchingexample}
\omterm{def:b2:randomvar:law}{Lei} de descendência: nenhum filho, um filho, dois filhos com
probabilidades $\frac14, \frac14, \frac12$. Então $m = \frac14 + 1 =
\frac54 > 1$ e $G(t) = \frac14 + \frac14 t + \frac12 t^2$. Pontos
fixos: $\frac12 t^2 - \frac34 t + \frac14 = 0$, isto é, $2t^2 - 3t
+ 1 = (2t - 1)(t - 1) = 0$: $q = \frac12$. A linhagem se extingue
com probabilidade $\frac12$ --- e com probabilidade $\frac12$
ela vive para sempre.
\end{example}

\begin{remark}[Perspectivas dentro deste volume]
O capítulo é o entroncamento do livro, e cada ingrediente
chegou de um lugar com nome: a \omterm{def:b2:structures:algebra}{álgebra} de séries do
\cref{ch:b2:series} e do~\cref{ch:b2:powerseries}, a
probabilidade do~\cref{ch:b2:proba} (a \omterm{def:b2:metric:continuity}{continuidade} monótona
demonstra $q_n \uparrow q$) e do~\cref{ch:b2:randomvar} ($G_X =
\E(t^X)$ é uma \omterm{def:b2:randomvar:expectation}{esperança}, a multiplicatividade é o teorema
do produto), a convexidade do~\cref{ch:b2:realfun} via o
\cref{ch:b2:affine}. Até as patologias de cauda pesada se conectam:
a variável de São Petersburgo do capítulo anterior tem
$G(t) = \sum_k2^{-k}t^{2^k}$, uma série perfeitamente convergente
em $\intcc01$ cuja derivada em $1^-$ diverge ---
média infinita, visível num relance. Um só objeto, todas as ferramentas
do ano: um último capítulo à altura.
\end{remark}

\begin{remark}[Armadilhas comuns]
(i) As \omterm{ex:b2:powerseries:fibonacci}{funções geradoras} se aplicam apenas a variáveis com valores em $\N$:
para variáveis com sinal ou não inteiras, o objeto $\E(t^X)$
perde sua estrutura de série de potências (o terceiro ano o substitui por
transformadas adaptadas a $\R$). (ii) A primeira verificação de bom senso de
qualquer $G$ calculada é $G(1) = 1$; a segunda é que os
coeficientes sejam não negativos --- um coeficiente negativo
significa um deslize algébrico, não uma \omterm{def:b2:randomvar:law}{lei} nova. (iii) Em somas aleatórias,
a ordem da composição importa: $G_S = G_N \circ G_X$, sendo a função
\emph{externa} a que conta as parcelas; compor na outra
ordem não faz sentido ($G_X \circ G_N$ contaria itens
de itens). (iv) A multiplicatividade exige \omterm{def:b2:proba:independence}{independência} e
fontes distintas de aleatoriedade: $G_{2X}(t) = G_X(t^2)$, e não
$G_X(t)^2$. (v) Derivar em $1$ é uma operação
de bordo: quando o raio é exatamente $1$, como no
\cref{ex:b2:genfun:heavytail}, $G'(1^-)$ pode ser infinita,
e a formulação por limite monótono do teorema dos momentos não é
um preciosismo pedante, mas o enunciado honesto.
\end{remark}

\section*{Encerrando o volume}

A \omterm{ex:b2:powerseries:fibonacci}{função geradora} é um objeto final à altura deste livro:
ela é simultaneamente uma série de potências (\cref{ch:b2:powerseries}), uma
ferramenta de \omterm{def:b2:series:summable}{famílias somáveis} (\cref{ch:b2:series}), uma \omterm{def:b2:randomvar:expectation}{esperança}
(\cref{ch:b2:randomvar}), uma função \omterm{def:b2:affine:convex}{convexa} cuja geometria decide a
extinção (\cref{ch:b2:realfun}) e uma iteração de ponto fixo
(\cref{ch:b2:metric}). A matemática do segundo ano é um só assunto.
O volume do terceiro ano de graduação abrirá as portas deliberadamente deixadas
fechadas aqui: a integração de Lebesgue (quitando o teorema da
convergência dominada do~\cref{ch:b2:integration}), a
probabilidade em espaços não \omterm{def:b2:structures:countable}{enumeráveis} à luz da teoria da medida e a
demonstração completa do teorema da função inversa
(\cref{ch:b2:diffcalc}) no contexto da geometria diferencial.

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:genfun:1}
Calcule a \omterm{ex:b2:powerseries:fibonacci}{função geradora} da \omterm{def:b2:randomvar:law}{lei} uniforme em $\{1, 2,
\dots, 6\}$ (um dado honesto). Mostre que a soma de dois dados honestos
\emph{não} pode ser uniforme em $\{2, \dots, 12\}$: fatore $G_{X+Y}$
e conte raízes. \emph{(Uma soma uniforme forçaria $G_X(t)G_Y(t) =
\frac{t^2}{11}\sum_{k=0}^{10}t^k$, cujas raízes não nulas são as
raízes $11$-ésimas da unidade distintas de $1$ --- nenhuma delas real ---
enquanto $G_X/t$ e $G_Y/t$ são polinômios reais de grau $5$, cada um
com ao menos uma raiz real.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:genfun:2}
Usando \omterm{ex:b2:powerseries:fibonacci}{funções geradoras}, recupere $\E$ e $V$ para as \omterm{def:b2:randomvar:law}{leis} binomial
e geométrica (\cref{thm:b2:genfun:moments}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:genfun:3}
Dois dados viciados: é possível viciar dois dados (independentemente,
de modo idêntico ou não) para que sua soma seja uniforme em $\{2, \dots,
12\}$? \emph{(A mesma obstrução por fatoração do
\cref{exo:b2:genfun:1}: a resposta é não mesmo com viciamentos diferentes,
pois cada fator $G_X(t)/t$ tem grau ímpar $5$,
logo uma raiz real, enquanto o alvo não tem nenhuma.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:genfun:4}
Sejam $X_1, X_2, \dots$ variáveis de Bernoulli $\mathcal{B}(p)$ \omterm{def:b2:proba:independence}{independentes}
e $N \sim \mathcal{P}(\lambda)$ \omterm{def:b2:proba:independence}{independente} delas. Mostre,
via o~\cref{thm:b2:genfun:compound}, que $S = X_1 + \dots + X_N
\sim \mathcal{P}(\lambda p)$: um número de Poisson de itens, cada um mantido
com probabilidade $p$, deixa um número de Poisson --- o \emph{afinamento}.
Calcule também a \omterm{def:b2:randomvar:law}{lei} da contagem descartada e admire: ela é
$\mathcal{P}(\lambda(1-p))$, e pode-se mostrar que é independente
de $S$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:genfun:5}
(Binomial negativa) Seja $T_r$ o número de lançamentos para obter
$r$ caras (probabilidade de cara $p$). Escreva $T_r$ como soma de $r$
variáveis geométricas \omterm{def:b2:proba:independence}{independentes}, deduza
\[
G_{T_r}(t) = \Bigl(\frac{pt}{1 - (1-p)t}\Bigr)^{r},
\qquad
\E(T_r) = \frac rp,
\qquad
V(T_r) = \frac{r(1-p)}{p^2},
\]
e desenvolva $G_{T_r}$ para achar $\P(T_r = n) = \binom{n-1}{r-1}
p^r(1-p)^{n-r}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:genfun:6}
Para a \omterm{def:b2:randomvar:law}{lei} de descendência $p_0 = \frac18$, $p_1 = \frac38$, $p_2 =
\frac38$, $p_3 = \frac18$: calcule $m$, decida a
supercriticalidade e calcule exatamente a probabilidade de extinção $q$.
\emph{(Fatore a raiz $t = 1$ de $G(t) - t$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:7}
(Prole total) Num \omterm{pb:b2:genfun:1}{processo de ramificação} subcrítico ($m < 1$), seja
$Y = \sum_{n\geq0} Z_n$ o número total de indivíduos jamais
nascidos. Mostre que $\E(Y) = \sum_n m^n = \frac{1}{1 - m}$ (justifique a
troca de somas) e demonstre que a \omterm{ex:b2:powerseries:fibonacci}{função geradora}
$H = G_Y$ satisfaz a equação funcional $H(t) = t\,G(H(t))$.
\emph{(O ancestral, mais as proles totais de cada um de seus
filhos, que são cópias independentes de $Y$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:8}
Seja $X$ com \omterm{ex:b2:powerseries:fibonacci}{função geradora} $G$ de \omterm{def:b2:powerseries:radius}{raio de convergência} $>
1$. Demonstre a \emph{cota exponencial de cauda}: existem $C > 0$ e
$\rho \in \intoo{0}{1}$ com $\P(X \geq n) \leq C\rho^n$.
\emph{(Markov aplicado a $t^X$ para um $t > 1$ fixo dentro do
disco.)} Reciprocamente, mostre que, se $\P(X \geq n) \leq C\rho^n$ com
$\rho < 1$, o raio de $G$ é $\geq 1/\rho > 1$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:9}
(Teorema de \omterm{def:b2:metric:continuity}{continuidade}, caso elementar) Sejam $X, X_1, X_2, \dots$
variáveis com valores em $\N$ tais que $G_{X_n}(t) \to G_X(t)$ para todo $t \in
\intco{0}{1}$. Mostre que $\P(X_n = k) \to \P(X = k)$ para todo
$k$. \emph{(Indução em $k$: para $k = 0$ tome $t \to 0$ ---
com cuidado: fixe $t$ pequeno, use $\abs{\P(X_n = 0) - G_{X_n}(t)}
\leq \frac{t}{1-t}$, válido pois a cauda $\sum_{j \geq 1}p_jt^j
\leq \frac{t}{1 - t}$; depois diagonalize. Para o passo de indução,
considere $\frac{G(t) - \P(X = 0)}{t}$, a \omterm{ex:b2:powerseries:fibonacci}{função geradora} de
uma \omterm{def:b2:randomvar:law}{lei} deslocada.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:genfun:10}
(Truque da paridade) Mostre que, para uma variável $X$ com valores em $\N$,
\[
\P(X \text{ par}) = \frac{1 + G_X(-1)}{2} ,
\]
e calcule essa probabilidade para $X \sim \mathcal P(\lambda)$
e $X \sim \mathcal B(n, p)$. O que significa $G_X(-1) \to 0$
probabilisticamente?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:genfun:11}
(Dados de Sicherman) Verifique a fatoração da \omterm{ex:b2:powerseries:fibonacci}{função geradora}
do dado honesto
\[
\frac{t + t^2 + \dots + t^6}{6}
= \frac{t\,(1 + t)(1 + t + t^2)(1 - t + t^2)}{6},
\]
e mostre que os dois dados de faces $\{1, 2, 2, 3, 3, 4\}$
e $\{1, 3, 4, 5, 6, 8\}$ têm \omterm{ex:b2:powerseries:fibonacci}{funções geradoras}
$\frac{t(1+t)(1+t+t^2)}6$ e
$\frac{t(1+t)(1+t+t^2)(1-t+t^2)^2}6$, cujo produto é o de
dois dados padrão: esses dados exóticos produzem todo total $2,
\dots, 12$ com exatamente as probabilidades padrão.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:12}
(Esperando duas caras seguidas) Uma moeda com probabilidade de cara
$p$ é lançada até aparecerem duas caras consecutivas; seja $T$
o número de lançamentos (o jogo do~\cref{exo:b2:proba:6}).
Condicionando aos primeiros lançamentos, deduza um sistema linear para
as \omterm{ex:b2:powerseries:fibonacci}{funções geradoras} a partir dos estados ``nenhuma cara corrente''
e ``uma cara corrente'', e conclua que
\[
G_T(t) = \frac{p^2t^2}{1 - qt - pqt^2}
\qquad (q = 1 - p);
\]
verifique $G_T(1) = 1$ e $\E(T) = \dfrac{1 + p}{p^2}$ ($= 6$
para uma moeda honesta).
\end{exercise}

\section{Problema: o processo de Galton--Watson, resolvido}

\begin{problem}[{Problema de fim de semana --- taxas de crescimento, soluções
exatas, prole total e a estimativa crítica de
Kolmogorov}]\label{pb:b2:genfun:1}
O critério de extinção (\cref{thm:b2:genfun:extinction})
separa os \omterm{pb:b2:genfun:1}{processos de ramificação}\index{processo de ramificação} em
subcríticos, críticos e supercríticos --- mas nada diz
sobre \emph{taxas}: quão rápido morre uma linhagem condenada, quão
grande cresce uma sobrevivente. Este problema as calcula. Mantemos
a notação do capítulo: \omterm{def:b2:randomvar:law}{lei} de descendência $(p_k)$ com \omterm{ex:b2:powerseries:fibonacci}{função geradora} $G$,
média $m = G'(1)$, tamanhos de geração $Z_n$ ($Z_0 = 1$), iterados
$G_n = G_{Z_n}$, probabilidades de extinção $q_n = \P(Z_n = 0)
\uparrow q$; supomos sempre $p_1 \neq 1$ e, onde aparecem momentos
de segunda ordem, $G''(1) < \infty$, e escrevemos $\sigma^2 =
V(Z_1)$.

\medskip
\noindent\textbf{Parte I --- Momentos das gerações.}
\begin{enumerate}
  \item Mostre que $\E(Z_n) = m^n$ \emph{(regra da cadeia em $G_n = G
        \circ G_{n-1}$ em $1^-$, usando $G_{n-1}(1) = 1$ e
        \cref{thm:b2:genfun:moments})}.
  \item Estabeleça a recursão $G_n''(1) =
        G''(1)\,m^{2(n-1)} + m\,G_{n-1}''(1)$ e resolva-a:
        $G_n''(1) = G''(1)\,m^{n-1}\dfrac{m^n - 1}{m - 1}$
        para $m \neq 1$, e $G_n''(1) = n\,G''(1)$ para $m =
        1$.
  \item Deduza
        \[
        V(Z_n) = \sigma^2m^{n-1}\,\frac{m^n - 1}{m - 1}
        \quad (m \neq 1),
        \qquad
        V(Z_n) = n\,\sigma^2 \quad (m = 1).
        \]
  \item (Taxa subcrítica, cota superior) Para $m < 1$, mostre que
        $\P(Z_n > 0) \leq m^n$ \emph{(Markov na
        variável $Z_n$ com valores inteiros)}: a extinção é certa com uma
        taxa geométrica --- um refinamento quantitativo do
        critério do capítulo.
  \item (Taxa subcrítica, cota inferior) Usando
        Cauchy--Schwarz em $Z_n\mathbf 1_{Z_n > 0}$, mostre que
        \[
        \P(Z_n > 0) \geq \frac{\E(Z_n)^2}{\E(Z_n^2)}
        \geq c\,m^{n}
        \quad\text{com}\quad
        c = \Bigl(\frac{\sigma^2}{m(1-m)} + 1\Bigr)^{-1} :
        \]
        a taxa geométrica $m^n$ é exata a menos de constantes.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A família geométrica, resolvida
exatamente.} Seja a \omterm{def:b2:randomvar:law}{lei} de descendência geométrica em $\N$: $p_k =
qp^k$ ($k \geq 0$), com $0 < p < 1$, $q = 1 - p$.
\begin{enumerate}[resume]
  \item Calcule $G(t) = \dfrac{q}{1 - pt}$ e $m = \dfrac
        pq$; localize os três regimes em termos de $p$.
  \item Resolva $G(t) = t$: mostre que os pontos fixos são $1$ e
        $q/p = 1/m$, e recupere a probabilidade de extinção
        $q_{\mathrm{ext}} = \min(1, 1/m)$.
  \item Demonstre por indução as formas fechadas
        \[
        q_n = \frac{m^n - 1}{m^{n+1} - 1} \quad (m \neq 1),
        \qquad
        q_n = \frac{n}{n+1} \quad (m = 1).
        \]
  \item Deduza as taxas exatas: $1 - q_n \sim (1 - m)\,m^n$
        no caso subcrítico, e $q_{\mathrm{ext}} - q_n
        \sim \dfrac{m - 1}{m^{2}}\cdot m^{-n}$ no
        caso supercrítico; verifique que a razão de contração
        supercrítica é $G'(q_{\mathrm{ext}}) = 1/m$.
  \item Caso crítico ($p = \tfrac12$): calcule $\sigma^2 =
        2$ e note que $1 - q_n = \frac1{n+1}$: a sobrevivência decai
        como $\frac1n$ --- nem geométrica nem \omterm{def:b2:series:summable}{somável}.
  \item Ainda no caso crítico: demonstre por indução o iterado \omterm{def:b2:metric:complete}{completo}
        \[
        G_n(t) = \frac{n - (n-1)t}{n + 1 - nt},
        \]
        e deduza que, condicionado à sobrevivência, $Z_n$ é
        geométrico em $\N^*$ de parâmetro $\frac1{n+1}$:
        \[
        \P(Z_n = k \mid Z_n > 0) = \frac1{n+1}
        \Bigl(\frac{n}{n+1}\Bigr)^{k-1},
        \qquad
        \E(Z_n \mid Z_n > 0) = n + 1 .
        \]
        A linhagem média morre, mas as linhagens sobreviventes têm
        tamanho da ordem de $n$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Prole total.} Seja $Y =
\sum_{n\geq0}Z_n \in \N^* \cup \{\infty\}$ o número total
de indivíduos jamais nascidos, e $H(t) =
\sum_{k\geq1}\P(Y = k)t^k$.
\begin{enumerate}[resume]
  \item Justifique $\P(Y < \infty) = q_{\mathrm{ext}}$ e
        recorde do~\cref{exo:b2:genfun:7} a equação
        funcional $H(t) = t\,G(H(t))$ (cuja dedução não
        usou $m < 1$).
  \item (Ramificação binária) Para $p_0 = p_2 = \frac12$
        (crítico), resolva a equação funcional:
        \[
        H(t) = \frac{1 - \sqrt{1 - t^2}}{t},
        \]
        e desenvolva com \cref{ex:b2:powerseries:catalan} para
        obter
        \[
        \P(Y = 2k + 1) = \frac{C_k}{2^{2k+1}},
        \qquad C_k = \frac1{k+1}\binom{2k}k ;
        \]
        confira os valores $\P(Y = 1) = \frac12$ e $\P(Y = 3)
        = \frac18$ por contagem direta.
  \item Derivando a equação funcional em $1^-$,
        mostre que $\E(Y) = \frac{1}{1-m}$ para $m < 1$, enquanto a
        criticalidade força $\E(Y) = \infty$: a prole total
        crítica é finita quase certamente com média
        infinita.
  \item Com a assintótica do binomial central
        (\cref{ex:b2:comparison:centralbinomial}), mostre que
        \[
        \P(Y = 2k+1) \sim \frac{1}{2\sqrt\pi\,k^{3/2}},
        \]
        uma cauda pesada em $k^{-3/2}$, e deduza $\P(Y > n)
        \asymp n^{-1/2}$ (cotas superior e inferior dessa
        ordem bastam).
  \item Compare com o \omterm{pb:b2:proba:1}{passeio aleatório} honesto (o problema de fim
        de semana do~\cref{ch:b2:proba}): lá tempos de retorno certos mas
        de média infinita, aqui prole total certa mas
        de média infinita, ambos com leis locais em
        $n^{-3/2}$. Um parágrafo sobre por que a
        criticalidade produz essa assinatura.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- A estimativa de Kolmogorov na
criticalidade.} Suponha $m = 1$, $0 < \sigma^2 = G''(1) <
\infty$.
\begin{enumerate}[resume]
  \item Mostre que $G''$ se estende \omterm{def:b2:metric:continuity}{continuamente} a $\intcc01$
        \emph{(não negativa, crescente e com limite finito)} e
        deduza o desenvolvimento de Taylor em $1$:
        \[
        G(t) = t + b\,(1-t)^2 + o\bigl((1-t)^2\bigr),
        \qquad b = \frac{G''(1)}2 = \frac{\sigma^2}2 .
        \]
  \item Para $t \in \intco01$, ponha $h(t) = \dfrac1{1 - G(t)} -
        \dfrac1{1 - t}$. Mostre que
        \[
        h(t) = \frac{G(t) - t}{(1 - G(t))(1 - t)}
        \xrightarrow[t\to1^-]{} b .
        \]
  \item Telescope ao longo da iteração $q_{j+1} = G(q_j)$:
        \[
        \frac1{1 - q_n} = 1 + \sum_{j=0}^{n-1}h(q_j),
        \]
        e conclua, com um argumento de Cesàro, que
        \[
        \P(Z_n > 0) = 1 - q_n \sim \frac{2}{\sigma^2\,n}
        \]
        --- a \emph{estimativa de Kolmogorov}: todo
        \omterm{pb:b2:genfun:1}{processo de ramificação} crítico morre à taxa universal $1/n$,
        com apenas a constante lembrando a \omterm{def:b2:randomvar:law}{lei} de descendência.
  \item Confira a estimativa contra o caso geométrico
        crítico da questão 10.
  \item Deduza $\E(Z_n \mid Z_n > 0) = \dfrac{1}{1 - q_n}
        \sim \dfrac{\sigma^2 n}{2}$ \emph{(note que $\E(Z_n
        \mathbf 1_{Z_n>0}) = \E(Z_n) = 1$)} e confira-a
        contra a questão 11: condicionada à sobrevivência, a
        população cresce \emph{linearmente} --- a corda bamba
        crítica entre a morte e a explosão.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Aplicações e síntese.}
\begin{enumerate}[resume]
  \item (Epidemias, reações em cadeia) Para uma \omterm{def:b2:randomvar:law}{lei} de descendência
        de Poisson $\mathcal P(\lambda)$ --- cada caso infecta
        $\mathcal P(\lambda)$ novos casos --- escreva a
        equação de extinção $q = \eu^{\lambda(q-1)}$ e
        resolva-a numericamente para $\lambda = 1.5$ ($q \approx
        0.417$) e $\lambda = 2$ ($q \approx 0.203$):
        partindo de um único caso, um surto grande \emph{não}
        é certo nem mesmo quando $\lambda > 1$. Explique
        por que a iteração $q_{n+1} = \eu^{\lambda(q_n - 1)}$
        a partir de $q_0 = 0$ converge para a raiz certa.
  \item Partindo de $k$ ancestrais em vez de um, mostre que
        a probabilidade de extinção é $q^k$. Aplicação:
        com $\lambda = 1.5$, quantos casos iniciais tornam um
        surto ao menos $99\%$ provável?
  \item (Condicionando um processo supercrítico à extinção)
        Para $m > 1$ com probabilidade de extinção $q \in
        \intoo01$: demonstre primeiro, por convexidade, que $G'(q) < 1$
        no menor ponto fixo, e deduza
        $q_{\mathrm{ext}} - q_n = O\bigl(G'(q)^n\bigr)$
        (convergência geométrica, como exemplificou a questão 9).
        Mostre então que $\widehat G(t) = G(qt)/q$ é a \omterm{ex:b2:powerseries:fibonacci}{função geradora}
        de uma \omterm{def:b2:randomvar:law}{lei} de descendência legítima, de média $\widehat m =
        G'(q) < 1$: um processo companheiro subcrítico. Verifique
        na família geométrica: condicionar o processo
        supercrítico em $(p, q)$ à extinção troca
        $p$ e $q$. (O enunciado completo --- o processo condicionado
        \emph{é} o processo companheiro --- é demonstrado
        no volume do terceiro ano de graduação; aqui você verificou sua
        sombra em \omterm{ex:b2:powerseries:fibonacci}{funções geradoras}.)
  \item Síntese: monte a tabela da tricotomia --- para $m <
        1$, $m = 1$, $m > 1$: valor de $q$; taxa de $\P(Z_n >
        0)$ ou de $q - q_n$; $\E(Y)$; tamanho de uma geração
        sobrevivente. Enuncie, em uma frase por ferramenta, como
        a composição de \omterm{ex:b2:powerseries:fibonacci}{funções geradoras}, a convexidade, Taylor em $1^-$ e
        a média de Cesàro carregaram todo o problema, e o que
        o volume do terceiro ano de graduação acrescenta (o martingal $Z_n/m^n$ e
        a \omterm{def:b2:randomvar:law}{lei} limite exponencial de Yaglom).
\end{enumerate}
\end{problem}
