\chapter{मानदंडित सदिश समष्टियाँ}\label{ch:b2:nvs}

जब \omterm{def:b2:metric:def}{दूरिक समष्टि} स्वयं एक सदिश समष्टि हो और दूरी किसी \omterm{def:b2:nvs:norm}{मानदंड} से आती हो,
तब सांस्थिति और रैखिक बीजगणित आपस में खेलने लगते हैं: रैखिक प्रतिचित्रण
ठीक तभी \omterm{def:b2:metric:continuity}{संतत} होते हैं जब वे इकाई गोले पर परिबद्ध हों, परिमित विमा सभी
मानदंडों को एक-दूसरे से सहमत होने पर बाध्य कर देती है, और पूर्णता
निरपेक्ष अभिसारी श्रेणियों को अभिसारी बना देती है। परिमित और अनंत विमा
के बीच की खाई --- जो रीस की प्रमेय में स्फटिक बन जाती है --- इस अध्याय
का सबसे गहरा पाठ है।

आगे सर्वत्र $E, F$ $K = \R$ या $\C$ पर सदिश समष्टियाँ हैं।

\section{मानदंड}

\begin{definition}\label{def:b2:nvs:norm}
$E$ पर \emph{मानदंड}\index{मानदंड} ऐसा प्रतिचित्रण $\norm{\,\cdot\,} \colon E
\to \R_+$ है कि सभी $x, y \in E$,
$\lambda \in K$ के लिए:
\[
\norm x = 0 \iff x = 0,
\qquad
\norm{\lambda x} = \abs\lambda\,\norm x,
\qquad
\norm{x + y} \leq \norm x + \norm y .
\]
तब $d(x, y) = \norm{x - y}$ एक दूरी है, और \cref{ch:b2:metric} की पूरी सामग्री लागू हो जाती है। विपरीत
त्रिभुज असमिका $\bigl|\norm x - \norm y\bigr| \leq \norm{x - y}$ मानदंड को स्वयं $1$-लिप्शिट्स बना देती है; और योग
तथा अदिश गुणन \omterm{def:b2:metric:continuity}{संतत} होते हैं (आकलन $\norm{(x + y) - (x' + y')} \leq \norm{x -
x'} + \norm{y - y'}$ इत्यादि)।
\end{definition}

\begin{example}\label{ex:b2:nvs:examples}
$K^n$ पर:
\[
\norm{x}_1 = \sum_i \abs{x_i},
\qquad
\norm{x}_2 = \Bigl(\sum_i \abs{x_i}^2\Bigr)^{1/2},
\qquad
\norm{x}_\infty = \max_i \abs{x_i}
\]
($\norm\cdot_2$ कोशी--श्वार्ज़ से \omterm{def:b2:nvs:norm}{मानदंड} है, प्रथम वर्ष का खंड)। $C(\intcc{a}{b})$ पर:
\[
\norm f_\infty = \sup \abs f,
\qquad
\norm f_1 = \int_a^b \abs f,
\qquad
\norm f_2 = \Bigl(\int_a^b \abs f^2\Bigr)^{1/2},
\]
जिनमें अंतिम दो समाकल की यथार्थ धनात्मकता और समाकल कोशी--श्वार्ज़ (प्रथम
वर्ष का खंड) के कारण \omterm{def:b2:nvs:norm}{मानदंड} हैं। आव्यूहों पर: $\mathcal M_n(K) \simeq K^{n^2}$ का कोई भी \omterm{def:b2:nvs:norm}{मानदंड};
और नीचे आने वाले \omterm{thm:b2:nvs:continuouslinear}{संकारक मानदंड} संरचनात्मक दृष्टि से महत्त्वपूर्ण हैं।
\end{example}

\begin{definition}[तुल्य मानदंड]\label{def:b2:nvs:equivalent}
$E$ पर दो \omterm{def:b2:nvs:norm}{मानदंड} $N_1, N_2$ \emph{तुल्य}\index{तुल्य मानदंड} कहलाते हैं जब ऐसे
अचर $c, C > 0$ हों कि
\[
c\,N_1 \leq N_2 \leq C\, N_1 .
\]
तुल्य मानदंडों के \omterm{def:b2:metric:topology}{विवृत} समुच्चय एक ही होते हैं, अभिसारी और कोशी अनुक्रम
एक ही, \omterm{def:b2:metric:compact}{संहत} और \omterm{def:b2:metric:complete}{पूर्ण} उपसमुच्चय एक ही: अर्थात् वही विश्लेषण।
\end{definition}

\begin{example}[अनंत विमा में अतुल्यता]\label{ex:b2:nvs:nonequivalent}
$C(\intcc{0}{1})$ पर: $\norm f_1 \leq \norm f_\infty$ सदा, परंतु उलटा कोई परिबंध नहीं टिकता: $f_n(x) = x^n$ के लिए $\norm{f_n}_\infty = 1$ और
$\norm{f_n}_1 = \frac{1}{n+1} \to 0$ होते हैं। अतः $\norm\cdot_1$ के लिए $f_n \to 0$ है पर $\norm\cdot_\infty$ के लिए नहीं: दोनों \omterm{def:b2:nvs:norm}{मानदंड}
अभिसरण के बारे में ही असहमत हैं।
\end{example}

\begin{example}[विमा $n$ में स्पष्ट अचर]\label{ex:b2:nvs:constants}
$K^n$ पर तीनों क्लासिक \omterm{def:b2:nvs:norm}{मानदंड} तीखे अचरों के साथ \omterm{def:b2:nvs:equivalent}{तुल्य} हैं:
\[
\norm x_\infty \leq \norm x_2 \leq \norm x_1
\leq \sqrt n\,\norm x_2 \leq n\,\norm x_\infty ,
\]
जिनमें बीच वाला परिबंध $\norm x_1 \leq \sqrt n\norm x_2$ सर्व-एक सदिश के विरुद्ध कोशी--श्वार्ज़ से
आता है। चरम सदिश: $e_1$ पहली दो असमिकाओं को समता बना देता है और $(1, 1,
\dots, 1)$
अंतिम दो को। विमा $n$ अचरों में स्पष्ट दिखाई देती है --- यही अनंत विमा
में विफलता का परिमाणात्मक बीज है: $n \to \infty$ होने पर कोई एकसमान अचर नहीं बचता,
और ठीक यही \cref{ex:b2:nvs:nonequivalent} फलन समष्टियों पर दिखा देता है।
\end{example}

\section{संतत रैखिक प्रतिचित्रण}

\begin{theorem}[अभिलक्षण]\label{thm:b2:nvs:continuouslinear}
मानदंडित समष्टियों के बीच रैखिक प्रतिचित्रण $u \colon E \to F$ के लिए निम्नलिखित \omterm{def:b2:nvs:equivalent}{तुल्य}
हैं:
\begin{enumerate}
  \item $u$ \omterm{def:b2:metric:continuity}{संतत} है;
  \item $u$ $0$ पर \omterm{def:b2:metric:continuity}{संतत} है;
  \item $u$ संवृत इकाई गोले पर परिबद्ध है: $\sup_{\norm x \leq
        1} \norm{u(x)} < \infty$;
  \item कोई ऐसा $C \geq 0$ है कि सभी $x$ के लिए $\norm{u(x)} \leq C \norm x$;
  \item $u$ \omterm{def:b2:metric:continuity}{लिप्शिट्स} है।
\end{enumerate}
ऐसा न्यूनतम $C$ \emph{संकारक मानदंड}\index{संकारक मानदंड} $\vertiii{u} = \sup_{\norm x \leq 1}\norm{u(x)} = \sup_{x \neq 0}
\frac{\norm{u(x)}}{\norm x}$ कहलाता
है; वह \omterm{def:b2:metric:continuity}{संतत} रैखिक प्रतिचित्रणों की समष्टि $\mathcal{L}_c(E,
F)$ को मानदंडित समष्टि बना
देता है, जहाँ
\[
\vertiii{v \circ u} \leq \vertiii v\, \vertiii u .
\]
\end{theorem}

\begin{proof}
(1 $\Rightarrow$ 2) तुच्छ। (2 $\Rightarrow$ 3): $\varepsilon = 1$ के साथ $0$ पर संततता $\norm x \leq \delta
\Rightarrow \norm{u(x)} \leq 1$ वाला $\delta$ दे
देती है; और फिर समघातता $\norm x \leq 1$ वाले किसी भी $x$ को उस गोले में उतारकर वापस
ले आती है:
\[
\norm{u(x)} = \frac1\delta\,\norm{u(\delta x)} \leq
\frac1\delta ,
\]
क्योंकि $\norm{\delta x} \leq \delta$। (3 $\Rightarrow$ 4): $x \neq 0$ के लिए परिबंध $\frac{x}{\norm x}$ पर लगाइए। (4 $\Rightarrow$ 5):
$\norm{u(x) - u(y)} = \norm{u(x - y)} \leq
C\norm{x - y}$। (5 $\Rightarrow$ 1) ज्ञात है।

$\vertiii\cdot$ के लिए \omterm{def:b2:nvs:norm}{मानदंड} अभिगृहीत: समघातता और पृथक्करण स्पष्ट हैं ($\vertiii u = 0$ गोले
पर $u = 0$ को बाध्य कर देता है, अतः सर्वत्र); त्रिभुज असमिका $\norm{(u + v)(x)} \leq
\norm{u(x)} + \norm{v(x)}$ से; और
उपगुणात्मकता: $\norm{v(u(x))}
\leq \vertiii v\,\norm{u(x)} \leq \vertiii v \vertiii u \norm x$।
\end{proof}

\begin{example}\label{ex:b2:nvs:operatornorms}
$\bigl(C(\intcc{0}{1}), \norm\cdot_\infty\bigr)$ पर: मूल्यांकन $f \mapsto f(0)$ का \omterm{thm:b2:nvs:continuouslinear}{संकारक मानदंड} $1$ है; समाकलन $f \mapsto
\int_0^1 f$ का \omterm{def:b2:nvs:norm}{मानदंड} $1$
है; और प्रतिचित्रण $f \mapsto \int_0^1 t f(t)\dd t$ का \omterm{def:b2:nvs:norm}{मानदंड} $\int_0^1 t\,\dd t = \frac12$ (ऊपरी परिबंध समाकलों की त्रिभुज
असमिका से; और $f \equiv 1$ पर प्राप्त)। परंतु $(C^1, \norm\cdot_\infty)$ से $(C^0,
\norm\cdot_\infty)$ में अवकलन \omterm{def:b2:metric:continuity}{संतत}
\emph{नहीं} है: $\norm{\sin(nx)}_\infty
= 1$ जबकि अवकलज का उच्चतम \omterm{def:b2:nvs:norm}{मानदंड} $n$ है। अनंत विमा में
रैखिक होने से \omterm{def:b2:metric:continuity}{संतत} होना नहीं निकलता।
\end{example}

\begin{example}[एक ही अनुक्रम पर दो मानदंड, दो निर्णय]\label{ex:b2:nvs:sqrtnxn}
$C(\intcc01)$ पर मान लीजिए $g_n(x) = \sqrt{n}\,x^n$। तब
\[
\norm{g_n}_1 = \frac{\sqrt n}{n + 1} \longrightarrow 0,
\qquad
\norm{g_n}_2^2 = \frac{n}{2n + 1} \longrightarrow \frac12,
\qquad
\norm{g_n}_\infty = \sqrt n \longrightarrow \infty :
\]
अर्थात् एक अनुक्रम, तीन \omterm{def:b2:nvs:norm}{मानदंड}, तीन व्यवहार --- शून्य पर अभिसरण, कोई
अभिसरण नहीं (\omterm{def:b2:nvs:norm}{मानदंड} $\frac1{\sqrt2}$ पर स्थिर हो जाते हैं जबकि बिंदुवार सीमा $0$
है), और विस्फोट। $x = 1$ के पास संकेंद्रित होता द्रव्यमान $\norm\cdot_1$ को दिखाई नहीं
देता, $\norm\cdot_2$ को आधा दिखाई देता है, और $\norm\cdot_\infty$ के लिए वही प्रमुख है। अनंत विमा
में ``क्या यह अभिसरित होता है?'' किसी अनुक्रम के बारे में प्रश्न नहीं
है: वह अनुक्रम \emph{और} \omterm{def:b2:nvs:norm}{मानदंड}, दोनों के बारे में प्रश्न है।
\end{example}

\begin{method}[संकारक मानदंड का परिकलन]\label{met:b2:nvs:opnorm}
सदा दो चालों में। \emph{ऊपरी परिबंध:} त्रिभुज असमिकाओं, कोशी--श्वार्ज़
अथवा समाकल परिबंधों से $\norm{u(x)}$ का $C\norm x$ के द्वारा आकलन कीजिए --- इससे $\vertiii u
\leq C$
सिद्ध होता है। \emph{साक्षी:} या तो $\norm{u(x_0)} = C\norm{x_0}$ वाला कोई विशिष्ट $x_0 \neq 0$ दिखाइए
(परिबंध प्राप्त हो जाता है), अथवा इकाई सदिशों का ऐसा अनुक्रम $x_n$ जिसके
लिए $\norm{u(x_n)} \to C$ (परिबंध के पास पहुँचा जाता है)। दोनों चालें अनिवार्य हैं: अकेला
ऊपरी परिबंध केवल $\vertiii u \leq C$ देता है, और अकेला साक्षी केवल $\vertiii u \geq C$। अनंत विमा में
साक्षी को अनुक्रम होना पड़ सकता है --- उच्चतम का प्राप्त होना आवश्यक
नहीं (\cref{exo:b2:nvs:8})।
\end{method}

\begin{example}[विकर्ण संकारक हर मानदंड को एक-सा देखते हैं]\label{ex:b2:nvs:diagonalnorm}
$K^n$ पर $\norm\cdot_1, \norm\cdot_2,
\norm\cdot_\infty$ में से \emph{किसी भी} \omterm{def:b2:nvs:norm}{मानदंड} के साथ $D = \operatorname{diag}(d_1, \dots, d_n)$ के लिए: निर्देशांक-दर-निर्देशांक
$\abs{d_ix_i} \leq
\bigl(\max_j\abs{d_j}\bigr)\abs{x_i}$ से $\norm{Dx} \leq \max_j\abs{d_j}\,\norm x$; और $x = e_{j_0}$ (कोई महत्तम करने वाला सूचकांक) उसे प्राप्त कर लेता
है। अतः तीनों स्थितियों में $\vertiii D =
\max_j\abs{d_j}$: विकर्ण प्रतिचित्रणों के लिए सभी उचित
\omterm{def:b2:nvs:norm}{मानदंड} एक ही कहानी कहते हैं, अर्थात् सबसे बड़ा खिंचाव गुणक। संकारक
मानदंडों में जो कुछ कठिन है वह सब \emph{अविकर्ण} व्यवहार के बारे में है
--- और इसीलिए इस अध्याय की सप्ताहांत समस्या (प्रश्न 22) के अनुकूलित
\omterm{def:b2:nvs:norm}{मानदंड} पहले आव्यूह को विकर्ण बनाकर ही काम करते हैं।
\end{example}

\begin{example}[स्तंभ-योग: \cref{exo:b2:nvs:4} का $1$-मानदंड वाला जुड़वाँ]\label{ex:b2:nvs:columnsums}
$(\R^n, \norm\cdot_1)$ पर आव्यूह $A$ का \omterm{thm:b2:nvs:continuouslinear}{संकारक मानदंड} सबसे बड़ा निरपेक्ष \emph{स्तंभ}-योग
होता है। विधि चलाइए: $\norm x_1 \leq 1$ के लिए
\[
\norm{Ax}_1 = \sum_i\Bigl|\sum_j a_{ij}x_j\Bigr|
\leq \sum_j \abs{x_j}\sum_i\abs{a_{ij}}
\leq \Bigl(\max_j\sum_i\abs{a_{ij}}\Bigr)\norm x_1 ,
\]
और परिबंध किसी महत्तम करने वाले स्तंभ $j_0$ के लिए $x = e_{j_0}$ पर प्राप्त हो
जाता है --- कल्पना की जा सकने वाली सबसे सुथरी साक्षी। अतः $A =
\begin{psmallmatrix}1 & -2\\ 3 & 1\end{psmallmatrix}$ के लिए:
$\vertiii A_1 = \max(1 + 3,\ 2 + 1) = 4$, जबकि $\vertiii A_\infty = 4$ भी (पंक्तियाँ) --- यह यहाँ संयोग है, नियम नहीं: आव्यूह
की प्रविष्टियाँ असममित ढंग से बदल दीजिए और दोनों \omterm{def:b2:nvs:norm}{मानदंड} अलग हो जाएँगे।
$\norm\cdot_\infty$ के लिए पंक्तियाँ, $\norm\cdot_1$ के लिए स्तंभ: स्मरण-सूत्र यह है कि हर
\omterm{def:b2:nvs:norm}{मानदंड} के इकाई सदिश (क्रमशः चिह्न-प्रतिरूप और आधार सदिश) मेल खाते योग
चुन लेते हैं।
\end{example}

\begin{proposition}[द्विरैखिक प्रतिचित्रण]\label{prop:b2:nvs:bilinear}
कोई द्विरैखिक प्रतिचित्रण $b \colon E \times F \to G$ \omterm{def:b2:metric:continuity}{संतत} है यदि और केवल यदि किसी $C$ के लिए
$\norm{b(x,y)} \leq C\norm x\,\norm y$ हो; तब वह परिबद्ध समुच्चयों पर \omterm{def:b2:metric:continuity}{लिप्शिट्स} होता है। (उपपत्ति का ढर्रा
वही है; गुणनफल $(u, v) \mapsto v \circ u$ और आव्यूह गुणन मुख्य उदाहरण हैं।)
\end{proposition}

\begin{proof}
यदि परिबंध सत्य है:
\[
b(x,y) - b(x_0,y_0) = b(x - x_0,\, y) + b(x_0,\, y - y_0),
\]
अतः $\norm{b(x,y) - b(x_0,y_0)} \leq C\norm{x - x_0}\norm y +
C\norm{x_0}\norm{y - y_0}$: अर्थात् $(x_0, y_0)$ पर संततता, और जहाँ $\norm x, \norm y \leq R$ वहाँ \omterm{def:b2:metric:continuity}{लिप्शिट्स} परिबंध।
विलोमतः $(0,0)$ पर संततता $\norm x, \norm y \leq \delta$ पर $\norm{b(x,y)} \leq 1$ वाला $\delta$ दे देती है; अब दोनों चरों का
मापन बदलिए।
\end{proof}

\section{परिमित विमा}

\begin{theorem}[परिमित विमा में मानदंडों की तुल्यता]\label{thm:b2:nvs:finitedim}
परिमित-विमीय समष्टि पर \emph{सभी \omterm{def:b2:nvs:norm}{मानदंड} \omterm{def:b2:nvs:equivalent}{तुल्य} होते
हैं}।\index{मानदंडों की तुल्यता} फलतः परिमित विमा में: अभिसरण, विवृतता,
संहतता और पूर्णता मानदंड-निरपेक्ष धारणाएँ हैं; \omterm{def:b2:metric:compact}{संहत} $=$ संवृत और
परिबद्ध; समष्टि \omterm{def:b2:metric:complete}{पूर्ण} होती है; और परिमित-विमीय समष्टि \emph{से} जाने
वाला हर रैखिक (अथवा बहुरैखिक) प्रतिचित्रण \omterm{def:b2:metric:continuity}{संतत} होता है।
\end{theorem}

\begin{proof}
कोई आधार स्थिर कीजिए और $E \simeq K^n$ से पहचान लीजिए; अब किसी भी \omterm{def:b2:nvs:norm}{मानदंड} $N$ की
$\norm\cdot_\infty$ से तुलना कर लेना पर्याप्त है।

\emph{एक दिशा बीजगणित है:} $C = \sum N(e_i)$ के साथ $N(x) = N(\sum x_i e_i) \leq
\sum \abs{x_i} N(e_i) \leq C \norm x_\infty$। इससे यह भी दिखता है कि $N$
$(K^n, \norm\cdot_\infty)$ पर \omterm{def:b2:metric:continuity}{संतत} है (वह $C$-लिप्शिट्स है: $\abs{N(x) - N(y)} \leq N(x - y)$)।

\emph{दूसरी सांस्थिति है:} इकाई गोलक $S = \{x :
\norm{x}_\infty = 1\}$ $(K^n,
\norm\cdot_\infty)$ में संवृत और परिबद्ध है,
अतः \omterm{def:b2:metric:compact}{संहत} (\cref{thm:b2:metric:compactprops}~(2), जो $\C^n \simeq
\R^{2n}$ के लिए मान्य है)। \omterm{def:b2:metric:continuity}{संतत} फलन $N$ $S$ पर अपना
न्यूनतम $c$ प्राप्त कर लेता है; और $c > 0$ क्योंकि $N$ केवल $0 \notin S$ पर लुप्त
होता है। समघातता परिबंध को फैला देती है: सभी $x$ के लिए $N(x) \geq c \norm{x}_\infty$।

परिणाम: सारे कथन $(K^n,
\norm\cdot_\infty)$ पर सिमट आते हैं, जहाँ वे ज्ञात हैं (\cref{thm:b2:metric:rncomplete}, \cref{thm:b2:metric:compactprops});
और परिमित-विमीय $E$ से जाने वाला रैखिक $u$ $\norm{u(x)} \leq \sum\abs{x_i}
\norm{u(e_i)} \leq C'\norm{x}_\infty$ पूरा करता है: अर्थात्
\cref{thm:b2:nvs:continuouslinear} का परिबंध (4)।
\end{proof}

\begin{corollary}\label{cor:b2:nvs:closedsubspace}
किसी भी मानदंडित समष्टि की परिमित-विमीय उपसमष्टि संवृत होती है।
\end{corollary}

\begin{proof}
वह प्रेरित \omterm{def:b2:nvs:norm}{मानदंड} के लिए \omterm{def:b2:metric:complete}{पूर्ण} है (\cref{thm:b2:nvs:finitedim}), और \omterm{def:b2:metric:complete}{पूर्ण} उपसमुच्चय संवृत होते
हैं (\cref{def:b2:metric:complete})।
\end{proof}

\begin{example}[सममिति से परिकलित एक सर्वोत्तम सन्निकटन]\label{ex:b2:nvs:bestaffine}
$\bigl(C(\intcc{-1}{1}), \norm\cdot_\infty\bigr)$ में $f(x) = \abs x$ आनत फलनों $a + bx$ की (संवृत, द्वि-विमीय) उपसमष्टि से कितनी दूर
है? सममिति से, $a + bx$ के स्थान पर $a - bx$ रखने पर $\norm{f - (a \pm bx)}_\infty$ अपरिवर्तित रहता है, और
मध्यबिंदु $a$ कम से कम उतना ही अच्छा है (औसत पर त्रिभुज असमिका): अतः
केवल अचरों पर विचार करना पर्याप्त है। किसी अचर $a$ के लिए:
\[
\norm{\abs x - a}_\infty = \max\,(1 - a,\ a)
\geq \frac12 ,
\]
जो $a = \frac12$ पर न्यूनतम होता है: अतः दूरी $\frac12$ है, जो अचर $\frac12$ पर प्राप्त
होती है। त्रुटि-वक्र $\abs x -
\frac12$ पर ध्यान दीजिए: वह $x = -1, 0,
1$ पर बारी-बारी से $\pm\frac12$
तक पहुँचता है --- अर्थात् दो-प्राचली कुल से सर्वोत्तम सन्निकटन के लिए
बारी-बारी चिह्न वाले तीन चरम मान। यह \emph{समदोलन} प्रतिरूप संयोग नहीं
है; वह अनुकूलतम होने का हस्ताक्षर है, जिसे इस अध्याय की सप्ताहांत समस्या
चेबिशेव की प्रमेय में बदल देती है।
\end{example}

\begin{example}[संवृत बनाम सघन उपसमष्टियाँ]\label{ex:b2:nvs:closeddense}
$E = \bigl(C(\intcc{0}{1}), \norm\cdot_\infty\bigr)$ में: हर $\R_n[X]$ ($\intcc01$ तक प्रतिबंधित घात $\leq n$ के बहुपद) परिमित-विमीय है,
अतः \emph{संवृत} उपसमष्टि --- घात $\leq n$ के बहुपदों की एकसमान सीमा भी वैसी
ही होती है। परंतु उन सबका सम्मिलन $\R[X]$ $E$ में \emph{सघन} है (वाइरश्ट्रास
सन्निकटन प्रमेय, जो \cref{ch:b2:funcseq} में सिद्ध की गई है), और सघन उचित उपसमष्टियाँ
जितनी असंवृत हो सकती हैं उतनी होती हैं। सार: उपसमष्टियों की संवृतता
परिमित विमा का विशेषाधिकार है; संवृत मंज़िलों को चुनते जाने से सघन
गगनचुंबी इमारत बन सकती है।
\end{example}

\begin{theorem}[रीस]\label{thm:b2:nvs:riesz}
किसी मानदंडित समष्टि $E$ का संवृत इकाई गोला \omterm{def:b2:metric:compact}{संहत} है \emph{यदि और केवल
यदि} $\dim E < \infty$।\index{रीस की प्रमेय}
\end{theorem}

\begin{proof}
परिमित विमा: संवृत और परिबद्ध होना पर्याप्त है (\cref{thm:b2:nvs:finitedim})।

विलोमतः मान लीजिए $\dim E = \infty$। \emph{रीस की प्रमेयिका:} प्रत्येक उचित संवृत
उपसमष्टि $F \subsetneq E$ तथा $\varepsilon \in
\intoo{0}{1}$ के लिए ऐसा इकाई सदिश $x$ है कि $d(x, F) \geq 1 -
\varepsilon$। उपपत्ति: $y \notin F$
चुनिए, $\delta = d(y, F) > 0$ लीजिए ($F$ संवृत है), $\norm{y - f} \leq
\frac{\delta}{1 - \varepsilon}$ वाला $f \in F$ चुनिए, और $x = \frac{y - f}{\norm{y -
f}}$ रखिए: तब
किसी भी $g \in F$ के लिए
\[
\norm{x - g} = \frac{\norm{y - (f + \norm{y-f}\,g)}}{\norm{y - f}}
\geq \frac{\delta}{\norm{y-f}} \geq 1 - \varepsilon ,
\]
जहाँ अंश $y$ से $F$ के किसी बिंदु तक की दूरी है।

अब आगमन से इकाई सदिश $x_1, x_2, \dots$ बनाइए: $F_k =
\operatorname{Vect}(x_1, \dots, x_k)$ परिमित-विमीय है, अतः संवृत (\cref{cor:b2:nvs:closedsubspace})
और उचित; और $\varepsilon = \frac12$ के साथ रीस की प्रमेयिका $d(x_{k+1},
F_k) \geq \frac12$ वाला इकाई $x_{k+1}$ दे देती है।
यह अनुक्रम $p \neq q$ के लिए $\norm{x_p - x_q} \geq
\frac12$ पूरा करता है: अतः कोई अभिसारी उपअनुक्रम नहीं
--- अर्थात् इकाई गोला \omterm{def:b2:metric:compact}{संहत} नहीं।
\end{proof}

\begin{example}[विमा-संसूचक के रूप में रीस]\label{ex:b2:nvs:rieszdetector}
क्या $C(\intcc01)$ परिमित-विमीय है? रीस बिना कोई स्पष्ट अनंत स्वतंत्र कुल दिखाए
उत्तर दे देती है: अनुक्रम $f_n(x) = x^n$ संवृत इकाई गोले में है और $m > n$ के लिए
उपयुक्त बिंदुओं पर $\norm{f_n - f_m}_\infty \geq f_n(x_0) - f_m(x_0) > 0$ पूरा करता है --- जिसे इस अध्याय की सप्ताहांत
समस्या (प्रश्न 16) में सफ़ाई से परिमाणित किया गया है, जहाँ कोई उपअनुक्रम
परस्पर दूरी $\geq \frac14$ पर बना रहता है। कोई अभिसारी उपअनुक्रम नहीं, अतः गोला
\omterm{def:b2:metric:compact}{संहत} नहीं, अतः \cref{thm:b2:nvs:riesz} से $\dim C(\intcc01) = \infty$। इकाई गोले की संहतता एक पूर्ण द्विभाजन है: वह
परिमित विमा में सत्य है, अनंत विमा में विफल, और बीच में कुछ नहीं ---
अकेली ज्यामिति विमा का प्रकार पढ़ लेती है।
\end{example}

\section{बानाख समष्टियाँ}

\begin{definition}\label{def:b2:nvs:banach}
\emph{बानाख समष्टि}\index{बानाख समष्टि} \omterm{def:b2:metric:complete}{पूर्ण} मानदंडित समष्टि होती है।
उदाहरण: हर परिमित-विमीय मानदंडित समष्टि (\cref{thm:b2:nvs:finitedim}); $\bigl(C(\intcc{a}{b}),
\norm\cdot_\infty\bigr)$ (\cref{thm:b2:metric:rncomplete}); और $\mathcal{L}_c(E, F)$ जब
$F$ बानाख हो (उपपत्ति का वही ढर्रा जो \omterm{def:b2:metric:continuity}{संतत} फलनों के लिए था)।
अनुदाहरण: $\bigl(C(\intcc{0}{1}),
\norm\cdot_1\bigr)$ (\cref{exo:b2:nvs:7})।
\end{definition}

\begin{example}[समाकलन का संकारक मानदंड]\label{ex:b2:nvs:integrationnorm}
$\bigl(C(\intcc01), \norm\cdot_\infty\bigr)$ पर $T(f)(x) =
\int_0^x f(t)\,\dd t$ लीजिए (यह अंतःरूपांतरण है: $T(f)$ \omterm{def:b2:metric:continuity}{संतत} है)। \cref{met:b2:nvs:opnorm} चलाइए। ऊपरी
परिबंध:
\[
\abs{T(f)(x)} \leq \int_0^x\abs f \leq x\,\norm f_\infty \leq
\norm f_\infty ,
\]
अतः $\vertiii T \leq 1$। साक्षी: $f \equiv 1$ से $T(f)(x) =
x$ और $\norm{T(f)}_\infty = 1 = \norm f_\infty$ मिलता है: अतः प्राप्त, $\vertiii T = 1$। परंतु
ध्यान दीजिए कि $\vertiii{T^2} = \frac12 \neq
\vertiii T^2$: वास्तव में $T^2(f)(x) = \int_0^x(x - t)f(t)\dd t$ के लिए $\abs{T^2(f)(x)} \leq \frac{x^2}2\norm f_\infty$, जो फिर $f \equiv 1$ पर
प्राप्त होता है; और सामान्यतः $\vertiii{T^n} = \frac1{n!}$ --- अर्थात् उपगुणात्मक परिबंध $\vertiii T^n = 1$
एक क्रमगुणित जितना चूक जाता है। यही वह परिघटना है जिसे \cref{ch:b2:metric} की
सप्ताहांत समस्या की पुनरावृत्त-युक्ति रैखिक अवकल समीकरणों की वैश्विक
हल्यता में बदल देती है।
\end{example}

\begin{theorem}[बानाख समष्टियों में निरपेक्ष अभिसरण]\label{thm:b2:nvs:absoluteconvergence}
किसी \omterm{def:b2:nvs:banach}{बानाख समष्टि} में यदि $\sum \norm{u_n} < \infty$ हो तो $\sum u_n$ अभिसरित होती है, और $\norm{\sum u_n} \leq \sum\norm{u_n}$।
(मानदंडित समष्टियों में श्रेणियों का पूरा सिद्धांत \cref{ch:b2:series} है।)
\end{theorem}

\begin{proof}
आंशिक योग $S_N$: $q > p$ के लिए $\norm{S_q - S_p} \leq
\sum_{n=p+1}^{q}\norm{u_n}$, जो $0$ की ओर जाता है (मानदंडों की
वास्तविक श्रेणी के लिए कोशी कसौटी): अतः $(S_N)$ कोशी है, इसलिए अभिसारी। और
असमिका परिमित त्रिभुज असमिका से सीमा तक चली जाती है।
\end{proof}

\begin{example}[आव्यूह घातांकी, पहला परिचय]\label{ex:b2:nvs:matrixexp}
\emph{आव्यूह घातांकी}\index{आव्यूह घातांकी}: किसी भी उपगुणात्मक \omterm{def:b2:nvs:norm}{मानदंड}
के साथ $\mathcal{M}_n(K)$ ($\vertiii{AB}
\leq \vertiii A \vertiii B$) बानाख है (परिमित विमा)। तब प्रत्येक $A$ के लिए
\[
\eu^A = \sum_{k=0}^{\infty} \frac{A^k}{k!}
\]
निरपेक्ष रूप से अभिसरित होती है ($\vertiii{A^k/k!} \leq \vertiii A^k /k!$, जो योग्य है): अतः सुपरिभाषित।
\cref{ch:b2:diffeq} इसका क्रमबद्ध उपयोग करता है।
\end{example}

\begin{example}[एक नॉयमान श्रेणी जो समाप्त हो जाती है]\label{ex:b2:nvs:neumannfinite}
$A = \begin{psmallmatrix}0 & \frac12\\ 0 &
0\end{psmallmatrix}$ के लिए: \cref{ex:b2:nvs:examples} के मानदंडों पर बने किसी भी \omterm{thm:b2:nvs:continuouslinear}{संकारक मानदंड} में $\vertiii A < 1$, और
$A^2 = 0$, अतः गुणोत्तर श्रेणी सिमट जाती है:
\[
(I - A)^{-1} = \sum_{k \geq 0} A^k = I + A =
\begin{pmatrix}1 & \tfrac12\\ 0 & 1\end{pmatrix},
\]
जिसे $(I - A)(I + A) = I - A^2 = I$ से सत्यापित किया जा सकता है। शून्यंभाविता श्रेणी को ठीक वैसे
ही काट देती है जैसे उसने \cref{ch:b2:reduction} में घातांकी को काटा था; और यह उदाहरण
अपेक्षाएँ भी सँवार देता है: नॉयमान प्रतिलोम सामान्यतः अनंत श्रेणी है,
और बहुपद ठीक तभी जब विक्षोभ शून्यंभावी हो, तथा $N$ पदों के बाद त्रुटि
सदा गुणोत्तर पुच्छ $\vertiii A^{N+1}/(1 - \vertiii A)$ से परिबद्ध रहती है।
\end{example}

\begin{example}[किसी घूर्णन जनक का घातांकी]\label{ex:b2:nvs:rotationexp}
$A = \begin{psmallmatrix}0 & -\theta\\ \theta &
0\end{psmallmatrix}$ लीजिए। तब $A^2 = -\theta^2 I$, अतः घातें चार के आवर्त में चक्कर लगाती हैं, और
श्रेणी सम तथा विषम भागों में बँट जाती है:
\[
\eu^{A} = \sum_{k}\frac{A^k}{k!}
= \Bigl(\sum_{j}\frac{(-1)^j\theta^{2j}}{(2j)!}\Bigr) I
+ \Bigl(\sum_{j}\frac{(-1)^j\theta^{2j+1}}{(2j+1)!}\Bigr)
\frac{A}{\theta}
= \begin{pmatrix}
\cos\theta & -\sin\theta\\
\sin\theta & \cos\theta
\end{pmatrix},
\]
जहाँ सारे पुनर्विन्यास निरपेक्ष अभिसरण से अनुमत हैं। किसी प्रतिसममित
जनक का घातांकी एक घूर्णन होता है --- और यहाँ वह केवल श्रेणी से परिकलित
हुआ है, अवकल समीकरण $x' = Ax$ (\cref{ch:b2:diffeq}) के यह समझाने से तीन अध्याय पहले कि ऐसा
\emph{क्यों} है: $\eu^{tA}$ एकसमान वृत्तीय गति है। समापन दृष्टि: आव्यूह
श्रेणियों की सर्वसमिकाएँ ठीक वैसे ही सिद्ध होती हैं जैसे अदिश वाली, बस
कोई उपगुणात्मक \omterm{def:b2:nvs:norm}{मानदंड} निरपेक्ष अभिसरण प्रमाणित कर दे।
\end{example}

\begin{example}[उच्चतम मानदंड का अर्थ है एकसमान: शब्दकोश]\label{ex:b2:nvs:uniformdictionary}
कथन $\norm{f_n - f}_\infty \to 0$ \emph{ही} एकसमान अभिसरण है: एक ही संख्या $\sup_x\abs{f_n(x) - f(x)}$ हर बिंदु पर एक
साथ त्रुटि को परिबद्ध कर देती है। $\intcc{0}{1}$ पर $f_n(x) = x^n$ के साथ शब्दकोश काम पर:
बिंदुवार $\intco{0}{1}$ पर $f_n
\to 0$ और $f_n(1) = 1$; और \omterm{def:b2:nvs:norm}{मानदंड} में $\norm{f_n - 0}_\infty = 1 \not\to 0$, तथा वास्तव में बिंदुवार
सीमा असंतत है, इसलिए $C(\intcc01)$ में $\norm\cdot_\infty$-सीमा की पहुँच से बाहर (जो
एकसमान सीमाओं के अंतर्गत संवृत है, \cref{thm:b2:metric:rncomplete})। $\intcc{0}{a}$ पर $a < 1$: $\norm{f_n}_\infty = a^n \to 0$ ---
अर्थात् प्रांत सिकोड़ने से एकसमान अभिसरण लौट आता है। \cref{ch:b2:funcseq} का हर अभिसरण
कथन इसी एक \omterm{def:b2:nvs:norm}{मानदंड} के बारे में कथन है; और यह शब्दकोश ध्यान में रखने से
वह अध्याय आधा हो जाता है।
\end{example}

\begin{remark}[सामान्य भूलें]\label{rem:b2:nvs:pitfalls}
(क) \omterm{thm:b2:nvs:continuouslinear}{संकारक मानदंड} \emph{दोनों} चुने गए मानदंडों पर निर्भर करता है: उसी
आव्यूह का $\vertiii\cdot_\infty$ पंक्ति-योगों से मिलता है (\cref{exo:b2:nvs:4}) और $\vertiii\cdot_1$ उससे भिन्न होता है
(स्तंभ-योग); अंतर्निहित मानदंडों का नाम लिए बिना ``आव्यूह का'' \omterm{def:b2:nvs:norm}{मानदंड}
कहना निरर्थक है। (ख) $\vertiii{AB} \leq
\vertiii A\,\vertiii B$ एक असमिका है, और प्रायः यथार्थ --- घातें
परिबंध $\vertiii A^k$ के सुझाव से कहीं तेज़ी से सिकुड़ सकती हैं, और अनुकूलित
मानदंडों का पूरा उद्देश्य यही है (इस अध्याय की सप्ताहांत समस्या, प्रश्न
22)। (ग) ``रैखिक होने से \omterm{def:b2:metric:continuity}{संतत} होना निकलता है'' परिमित विमा का
विशेषाधिकार है: बहुपदों पर अवकलन रैखिक है और अपरिबद्ध (\cref{ex:b2:nvs:operatornorms})। (घ) $\sum u_n$ का
निरपेक्ष अभिसरण तभी काम आता है जब समष्टि \omterm{def:b2:metric:complete}{पूर्ण} हो (\cref{exo:b2:series:9} प्रतिउदाहरण बना
देता है)। (ङ) अनंत विमा में इकाई गोले पर उच्चतम सचमुच का उच्चतम होता
है: यह मत मान लीजिए कि वह प्राप्त हो जाता है (\cref{exo:b2:nvs:8})।
\end{remark}

\begin{remark}[इसी खंड के भीतर के परिप्रेक्ष्य]\label{rem:b2:nvs:perspectives}
अब तीन मुलाक़ातें तय हो चुकी हैं। \cref{ch:b2:series} के साथ: \omterm{def:b2:nvs:banach}{बानाख समष्टि} में निरपेक्ष
अभिसारी श्रेणियाँ अभिसरित होती हैं, अतः संकारकों की गुणोत्तर और घातांकी
श्रेणियाँ रोज़ के औज़ार बन जाती हैं --- $I - A$ का प्रतिलोमन, $\eu^{A}$ की
परिभाषा (\cref{ex:b2:series:neumann})। \cref{ch:b2:funcseq} और \cref{ch:b2:powerseries} के साथ: फलन-अनुक्रमों और घात श्रेणियों का
अभिसरण $\bigl(C, \norm\cdot_\infty\bigr)$ में अभिसरण ही है (\cref{ex:b2:nvs:uniformdictionary}), और अभिसरण त्रिज्या इस बारे में कथन
है कि कौन-सी गुणोत्तर श्रेणियाँ हावी हैं। \cref{ch:b2:fourier} के साथ: \omterm{def:b2:nvs:norm}{मानदंड} $\norm\cdot_2$ और
$\norm\cdot_\infty$ $C(\intcc{0}{1})$ पर सचमुच असहमत हैं (\cref{ex:b2:nvs:sqrtnxn}), और ठीक इसीलिए फूरिये श्रेणियों का
माध्य-वर्ग अभिसरण और एकसमान अभिसरण दो भिन्न प्रमेयें हैं जिनकी क़ीमतें
भी भिन्न हैं।
\end{remark}

\begin{remark}[यह अध्याय कहाँ काम आता है]
\omterm{thm:b2:nvs:continuouslinear}{संकारक मानदंड} और गुणोत्तर श्रेणी \cref{ch:b2:diffcalc} (प्रतिलोम फलन प्रमेय) के विक्षोभ
तर्कों तथा \cref{ch:b2:diffeq} के \omterm{ex:b2:nvs:matrixexp}{आव्यूह घातांकी} को चलाते हैं; \omterm{thm:b2:nvs:finitedim}{मानदंडों की तुल्यता} \cref{ch:b2:funcseq}
और उससे आगे के हर ``अपना मनपसंद \omterm{def:b2:nvs:norm}{मानदंड} चुन लीजिए'' वाले तर्क को चुपचाप
अधिकृत कर देती है; और रीस की प्रमेय की परिमित/अनंत खाई --- जिसे इस
अध्याय की सप्ताहांत समस्या परिमाणात्मक बना देती है --- यही कारण है कि
तृतीय वर्ष के खंड को नए औज़ार चाहिए (दुर्बल अभिसरण,
आर्ज़ेला--आस्कोली, हिल्बर्ट समष्टि प्रक्षेप) जहाँ यह खंड अभी भी अभिसारी
उपअनुक्रम निकाल लेता था।
\end{remark}

\begin{exercise}[$\star$]\label{exo:b2:nvs:1}
$\R^2$ पर $\norm\cdot_1$, $\norm\cdot_2$, $\norm\cdot_\infty$ के इकाई गोले खींचिए, और विमा $2$ में सर्वोत्तम
अचरों के साथ असमिकाएँ $\norm x_\infty
\leq \norm x_2 \leq \norm x_1 \leq 2\norm x_\infty$ सिद्ध कीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:nvs:2}
क्या $N(f) = \abs{f(0)} + \norm{f'}_\infty$ $C^1(\intcc{0}{1})$ पर \omterm{def:b2:nvs:norm}{मानदंड} है? उसकी $\norm{f}_\infty$ से तुलना कीजिए: एक असमिका सत्य है,
दूसरी विफल (उदाहरण दीजिए)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:nvs:3}
$\bigl(C(\intcc{0}{1}), \norm\cdot_\infty\bigr) \to \R$ पर $u(f) = \int_0^1 f(t)\,\eu^t\,\dd t$ का, और $(K^n, \norm\cdot_\infty)$ पर स्थानांतरण $S(x_1, x_2, \dots, x_n) = (x_2, \dots, x_n, 0)$ का \omterm{thm:b2:nvs:continuouslinear}{संकारक मानदंड} परिकलित
कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:nvs:4}
$(\R^n, \norm\cdot_\infty)$ पर सिद्ध कीजिए कि आव्यूह $A$ का \omterm{thm:b2:nvs:continuouslinear}{संकारक मानदंड} $\vertiii A_\infty = \max_i \sum_j \abs{a_{ij}}$ है (सबसे बड़ा
निरपेक्ष पंक्ति-योग)। इसे $\begin{pmatrix} 1 & -2\\
3 & 1\end{pmatrix}$ के लिए परिकलित कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:nvs:5}
सिद्ध कीजिए कि $GL_n(K)$ $\mathcal{M}_n(K)$ में \omterm{def:b2:metric:topology}{विवृत} है और $A
\mapsto A^{-1}$ उस पर \omterm{def:b2:metric:continuity}{संतत} है।
\emph{संकेत: विवृतता के लिए, यदि $\vertiii H < \frac{1}{\vertiii{A^{-1}}}$ तो $\vertiii{A^{-1}H} < 1$ के साथ $A + H = A(I +
A^{-1}H)$, और $\vertiii B < 1$ के लिए
$I + B$ प्रतिलोमनीय है, गुणोत्तर श्रेणी से (\cref{thm:b2:nvs:absoluteconvergence}); और संततता के लिए उसी
श्रेणी से $(A+H)^{-1} - A^{-1}$ को परिबद्ध कीजिए।}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:nvs:6}
मान लीजिए $\varphi$ किसी मानदंडित समष्टि $E$ पर रैखिक फलनिक है। सिद्ध कीजिए
कि $\varphi$ \omterm{def:b2:metric:continuity}{संतत} है यदि और केवल यदि $\ker\varphi$ संवृत हो। \emph{(यदि $\ker\varphi$ संवृत है और
$\varphi \neq 0$, तो $\varphi(a) = 1$ वाला $a$ चुनिए और $B(a, r) \cap \ker\varphi =
\emptyset$ वाला $r > 0$; फिर $a - \frac{h}{\varphi(h)}$ पर मापन-तर्क से
$\abs{\varphi(h)} \leq \frac{1}{r}\norm h$ निष्कर्ष निकालिए।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:nvs:7}
सिद्ध कीजिए कि $\bigl(C(\intcc{0}{1}), \norm\cdot_1\bigr)$ \omterm{def:b2:metric:complete}{पूर्ण} नहीं है: दिखाइए कि फलन $f_n$, जो $\bigl[\frac12 - \frac1n, \frac12\bigr]$ पर $0$ से
$1$ तक आनत ढलान हैं (पहले मान $0$, बाद में $1$), ऐसा कोशी अनुक्रम बनाते
हैं जिसकी कोई \omterm{def:b2:metric:continuity}{संतत} $\norm\cdot_1$-सीमा नहीं।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:nvs:8}
$\norm\cdot_\infty$ सहित $E = C(\intcc{0}{1})$ पर
\[
\varphi(f) = \sum_{n \geq 1} (-1)^n\, 2^{-n} f\bigl(\tfrac1n\bigr).
\]
लीजिए। सिद्ध कीजिए कि $\varphi$ सुपरिभाषित \omterm{def:b2:metric:continuity}{संतत} रैखिक फलनिक है और $\vertiii\varphi = 1$, परंतु
$\vertiii\varphi$ को परिभाषित करने वाला उच्चतम संवृत इकाई गोले पर \emph{प्राप्त नहीं
होता}। \emph{(ऊपरी परिबंध: त्रिभुज असमिका। \omterm{def:b2:nvs:norm}{मानदंड} $= 1$: ऐसे \omterm{def:b2:metric:continuity}{संतत} $f_K$
बनाइए कि $\norm{f_K}_\infty \leq 1$ और $n \leq K$ के लिए $f_K(\frac1n)
= (-1)^n$ --- बिंदु $\frac1n$ एक-दूसरे से अलग-थलग
हैं। अप्राप्ति: समता प्रत्येक $n$ के लिए $f(\frac1n) =
(-1)^n$ को बाध्य कर देती, जो $0$
पर $f$ की संततता के साथ असंगत है क्योंकि $\frac1n \to 0$।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:nvs:9}
मान लीजिए $E$ ऐसी मानदंडित समष्टि है जिसका संवृत इकाई गोला \omterm{def:b2:metric:compact}{संहत} है। \cref{thm:b2:nvs:riesz}
उद्धृत किए बिना पुनः सिद्ध कीजिए कि हर परिबद्ध अनुक्रम का कोई अभिसारी
उपअनुक्रम है, और सिद्ध कीजिए कि $E$ पर हर रैखिक फलनिक \omterm{def:b2:metric:continuity}{संतत} है यदि और
केवल यदि $\dim E < \infty$। \emph{(अनंत विमा के लिए, किसी रैखिकतः स्वतंत्र
सामान्यीकृत अनुक्रम पर स्वतंत्र रूप से मान देकर और फिर विस्तार करके कोई
असंतत फलनिक बनाइए --- बीजगणितीय संपूरक का अस्तित्व स्वीकार कर लीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:nvs:10}
$C(\intcc{0}{1})$ पर $\norm f_1 \leq \norm f_2 \leq \norm
f_\infty$ सिद्ध कीजिए \emph{(पहले के लिए कोशी--श्वार्ज़)}, और कुल $f_n(x) = x^n$
से दिखाइए कि कोई भी असमिका किसी अचर तक भी उलटी नहीं की जा सकती: अर्थात्
तीनों \omterm{def:b2:nvs:norm}{मानदंड} जोड़े-जोड़े में अतुल्य हैं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:nvs:11}
(किसी अधिसमतल से दूरी) मान लीजिए $\varphi$ किसी मानदंडित समष्टि $E$ पर अशून्य
\omterm{def:b2:metric:continuity}{संतत} रैखिक फलनिक है। सिद्ध कीजिए कि
\[
d\bigl(x, \ker\varphi\bigr) =
\frac{\abs{\varphi(x)}}{\vertiii\varphi}
\qquad (x \in E),
\]
और \cref{exo:b2:nvs:8} पर जाँचिए कि निम्नतम अधिसमतल के किसी बिंदु पर प्राप्त होना
आवश्यक नहीं।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:nvs:12}
$E = \R[X]$ (सभी बहुपद) पर $N_1(P) =
\sup_{\intcc{0}{1}}\abs P$ और $N_2(P) =
\sup_{\intcc{0}{2}}\abs P$ लीजिए। दिखाइए कि $N_1 \leq N_2$, परंतु $N_1$ और $N_2$
\omterm{def:b2:nvs:equivalent}{तुल्य} \emph{नहीं} हैं; इससे निष्कर्ष निकालिए कि तत्समक $(E, N_2) \to (E, N_1)$ ऐसा \omterm{def:b2:metric:continuity}{संतत}
रैखिक एकैकी आच्छादक प्रतिचित्रण है जिसका प्रतिलोम असंतत है। अंत में
दिखाइए कि $(E, N_1)$ \omterm{def:b2:metric:complete}{पूर्ण} नहीं है \emph{($\eu^x$ के टेलर आंशिक योग)}। तीनों
परिघटनाएँ परिमित विमा में असंभव हैं --- कारण बताइए।
\end{exercise}

\section{समस्या: सर्वोत्तम सन्निकटन और चेबिशेव की प्रमेय}

किसी दी हुई घात के बहुपदों से कोई फलन कितनी अच्छी तरह सन्निकटित किया जा
सकता है, और कौन-सा बहुपद उसे सबसे अच्छा करता है? अस्तित्व वाला पक्ष इसी
अध्याय का है: परिमित विमा में संहतता सर्वोत्तम सन्निकटनों को विद्यमान
बना देती है। और स्पष्ट पक्ष पर, एक अतुच्छ स्थिति नंगे हाथों पूरी तरह हल
की जा सकती है --- घात $n$ के सभी \emph{एकिक} बहुपदों में से $\intcc{-1}{1}$ पर सबसे
छोटे उच्चतम \omterm{def:b2:nvs:norm}{मानदंड} वाला बहुपद (सामान्यीकृत) चेबिशेव बहुपद है, जिसका
\omterm{def:b2:nvs:norm}{मानदंड} $2^{1-n}$ है: यही \emph{चेबिशेव की चरम प्रमेय} है। यह समस्या दोनों पक्ष
सिद्ध करती है, और फिर मापती है कि अनंत विमा में संहतता कितनी बुरी तरह
विफल होती है: $C(\intcc{0}{1})$ के इकाई गोले में परस्पर दूरी $1$ वाले बिंदुओं के अनंत
नक्षत्र बैठे हैं।

\begin{problem}[{सप्ताहांत समस्या --- चेबिशेव की चरम प्रमेय और
इकाई गोले की ज्यामिति}]
\label{pb:b2:nvs:1}
बिना अधोलेख वाले \omterm{def:b2:nvs:norm}{मानदंड} निर्दिष्ट खंड पर उच्चतम \omterm{def:b2:nvs:norm}{मानदंड} हैं।

\medskip
\noindent\textbf{भाग I --- मानदंडित समष्टियों में सर्वोत्तम
सन्निकटन।}
\begin{enumerate}
  \item मान लीजिए $F$ किसी मानदंडित समष्टि $E$ की परिमित-विमीय
        उपसमष्टि है और $x \in E$। सिद्ध कीजिए कि दूरी $d(x,
        F) = \inf_{f \in F}\norm{x - f}$ \emph{प्राप्त}
        होती है \emph{($F$ के किसी संवृत परिबद्ध उपसमुच्चय पर लाइए और
        \cref{thm:b2:nvs:finitedim} का उपयोग कीजिए)}।
  \item कोई \omterm{def:b2:nvs:norm}{मानदंड} \emph{यथार्थतः उत्तल} कहलाता है जब $\norm u = \norm v
        = 1$ और $u \neq v$ से
        $\bigl\Vert\frac{u +
        v}2\bigr\Vert < 1$ निकले। दिखाइए कि $\R^n$ पर $\norm\cdot_2$ यथार्थतः उत्तल है
        \emph{(समांतर चतुर्भुज सर्वसमिका)}, और यह कि $n \geq 2$ के लिए $\norm\cdot_1$
        तथा $\norm\cdot_\infty$ नहीं हैं।
  \item सिद्ध कीजिए कि यथार्थतः उत्तल \omterm{def:b2:nvs:norm}{मानदंड} के लिए प्रश्न 1 का
        सर्वोत्तम सन्निकटन \emph{अद्वितीय} होता है।
  \item $(\R^2, \norm\cdot_\infty)$ में रेखा $F =
        \operatorname{Vect}\bigl((1,0)\bigr)$ द्वारा $x = (0, 1)$ के सभी सर्वोत्तम सन्निकटन
        परिकलित कीजिए: न्यूनतमकारियों का एक पूरा अंतराल।
  \item $\bigl(C(\intcc{a}{b}), \norm\cdot_\infty\bigr)$ में दिखाइए कि \emph{अचरों} द्वारा $f$ का सर्वोत्तम
        सन्निकटन अद्वितीय है, $c^* = \frac{\max f
        + \min f}{2}$ के बराबर है, और दूरी $\frac{\max f - \min
        f}{2}$; और $\intcc{0}{1}$
        पर $f(x) = x^2$ के लिए दोनों परिकलित कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- चेबिशेव बहुपद।}
\begin{enumerate}[resume]
  \item दिखाइए कि ठीक एक बहुपद $T_n$ ऐसा है कि सभी $\theta$ के लिए $T_n(\cos\theta) = \cos n\theta$
        \emph{(कोसाइन योग-सूत्र से पुनरावृत्ति $T_{n+1} = 2XT_n - T_{n-1}$)}, कि $\deg T_n = n$, और यह कि
        $n \geq 1$ के लिए उसका अग्र गुणांक $2^{n-1}$ है।
  \item दिखाइए कि $\intcc{-1}{1}$ पर $\abs{T_n} \leq 1$, जिसमें $n + 1$ बिंदुओं $\eta_k =
        \cos\frac{k\pi}{n}$ ($k = 0, \dots, n$) पर $T_n(\eta_k) = (-1)^k$,
        और यह कि $T_n$ के मूल $n$ बिंदु $\cos\frac{(2k-1)\pi}{2n}$ हैं, जो $\eta_k$ के बीच-बीच
        में आते हैं।
  \item $T_2, T_3, T_4$ परिकलित कीजिए, और सीधे मान रखकर $\eta_0, \dots, \eta_3 = 1, \frac12, -\frac12,
        -1$ पर $T_3$ का एकांतरण
        सत्यापित कीजिए।
  \item $\abs x \geq 1$ के लिए सिद्ध कीजिए
        \[
        T_n(x) = \frac{\bigl(x + \sqrt{x^2 - 1}\bigr)^n +
        \bigl(x - \sqrt{x^2 - 1}\bigr)^n}{2},
        \]
        और स्थिर $x > 1$ के लिए $T_n(x) \sim \frac12\bigl(x + \sqrt{x^2 -
        1}\bigr)^n \to \infty$ ज्यामितीय ढंग से निष्कर्ष रूप में
        निकालिए।
  \item संयोजन नियम $T_m \circ T_n = T_{mn}$ सिद्ध कीजिए \emph{($\intcc{-1}{1}$ पर जाँचिए और बहुपदों
        की दृढ़ता का सहारा लीजिए)}।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- चेबिशेव की चरम प्रमेय।} $Q_n = 2^{1-n}T_n$ लिखिए (जो
प्रश्न 6 के अनुसार एकिक है)।
\begin{enumerate}[resume]
  \item मान लीजिए $P$ $\sup_{\intcc{-1}{1}}\abs P < 2^{1-n}$ के साथ घात $n \geq 1$ का एकिक बहुपद है। $D
        = Q_n - P$ का
        बिंदुओं $\eta_k$ पर मूल्यांकन करके और चिह्न-परिवर्तन गिनकर
        विरोधाभास निकालिए। निष्कर्ष:
        \[
        \sup_{\intcc{-1}{1}}\abs P \;\geq\; 2^{1-n}
        \qquad\text{घात } P \text{ के प्रत्येक एकिक } n.
        \]
  \item (समता की स्थिति) मान लीजिए $\sup_{\intcc{-1}{1}}\abs P =
        2^{1-n}$, जहाँ $P$ घात $n$ का एकिक है,
        और $D = Q_n
        - P \neq 0$ लीजिए। दिखाइए कि सभी $k$ के लिए $(-1)^kD(\eta_k) \geq 0$; दिखाइए कि $n$
        अंतरालों $\intcc{\eta_{k}}{\eta_{k-1}}$ में से हर एक में $D$ का कोई शून्य है, और यह कि
        दो क्रमागत अंतरालों में साझा शून्य कोई \emph{अभ्यंतरीय} बिंदु
        $\eta_k$ होता है जहाँ $D' = 0$ भी। निष्कर्ष निकालिए कि $D$ के बहुलता
        सहित $n$ शून्य हैं, अतः $D = 0$: अर्थात् न्यूनतमकारी ठीक $Q_n$ है
        --- यही \emph{चेबिशेव की चरम प्रमेय} है।
  \item प्रमेय को दूरी के रूप में फिर से कहिए: $\intcc{-1}{1}$ पर
        \[
        d_\infty\bigl(X^n,\ \R_{n-1}[X]\bigr) = 2^{1-n},
        \]
        और सर्वोत्तम सन्निकटन $X^n - Q_n$ अद्वितीय है; तथा आनत प्रतिस्थापन
        $x = \frac{1+t}2$ से दिखाइए कि $\intcc{0}{1}$ पर दूरी $2^{1-2n}$ हो जाती है।
  \item (अनुकूलतम अंतर्वेशन नोड) $n$ नोड $x_1, \dots,
        x_n \in \intcc{-1}{1}$ के लिए नोड-बहुपद $\omega(x) = \prod_i(x - x_i)$
        घात $n$ का एकिक है। प्रश्न 12 से निष्कर्ष निकालिए कि नोडों का
        कौन-सा चुनाव $\sup_{\intcc{-1}{1}}\abs\omega$ को न्यूनतम करता है --- जो क्लासिक अंतर्वेशन
        त्रुटि-परिबंध का नोड-निर्भर गुणक है --- और न्यूनतम मान दीजिए।
  \item प्रमेय की $n = 2$ वाली स्थिति हाथ से जाँचिए ($\inf_c \sup_{\intcc{-1}{1}}\abs{x^2 - c}$ सीधे ज्ञात
        कीजिए), और $n = 10$ के लिए $\intcc{0}{1}$ पर प्रश्न 13 की दूरी संख्यात्मक रूप
        से परिकलित कीजिए। उसका आकार $x^{10}$ के आलेख के बारे में क्या कहता
        है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- $C(\intcc{0}{1})$ का इकाई गोला।}
\begin{enumerate}[resume]
  \item मान लीजिए $g_k(x) = x^{2^k}$। दिखाइए कि $\norm{g_k}_\infty = 1$ और $j > k$ के लिए $\norm{g_k - g_j}_\infty \geq \frac14$ \emph{(उस बिंदु
        पर मूल्यांकन कीजिए जहाँ $x^{2^k} =
        \frac12$)}: अर्थात् बिना किसी अभिसारी
        उपअनुक्रम का एक स्पष्ट परिबद्ध अनुक्रम --- संवृत इकाई गोला
        \omterm{def:b2:metric:compact}{संहत} नहीं, और यह नंगे हाथों दिखा।
  \item (रीस की प्रमेयिका, तीखी की हुई) मान लीजिए $F$ किसी मानदंडित
        समष्टि $E$ की \emph{परिमित-विमीय} उचित उपसमष्टि है। प्रश्न 1 का
        उपयोग करके ऐसा इकाई सदिश $x$ बनाइए जिसके लिए ठीक $d(x, F) = 1$ हो ---
        केवल \cref{thm:b2:nvs:riesz} की प्रमेयिका वाला $\geq 1 -
        \varepsilon$ नहीं।
  \item इससे निष्कर्ष निकालिए: प्रत्येक अनंत-विमीय मानदंडित समष्टि में
        परस्पर दूरी $\geq 1$ वाले इकाई सदिशों का कोई अनुक्रम है, और उसी से
        रीस की प्रमेय पुनः निकालिए।
  \item $C(\intcc{0}{1})$ में ऐसा नक्षत्र स्पष्ट रूप से दिखाइए: $\bigl[\frac1{n+1}, \frac1n\bigr]$ पर आधारित तंबू
        फलन $h_n$, जिनका शिखर-मान $1$ हो। $\norm{h_n} = 1$ और $n
        \neq m$ के लिए $\norm{h_n - h_m} = 1$
        सत्यापित कीजिए, और ध्यान दीजिए कि $h_n \to 0$ बिंदुवार तो है पर
        एकसमान नहीं।
  \item (पूर्ण परिबद्धता विफल) दिखाइए कि $C(\intcc{0}{1})$ का संवृत इकाई गोला
        त्रिज्या $\frac13$ के परिमित गोलों से आच्छादित नहीं किया जा सकता
        \emph{(ऐसे हर गोले में अधिकतम एक $h_n$ होता है)} --- और इसकी
        तुलना \cref{thm:b2:metric:borellebesgue} की उपपत्ति के पूर्ण-परिबद्धता वाले चरण से कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- आव्यूहों पर \omterm{def:b2:nvs:norm}{मानदंड} काम पर, और संश्लेषण।}
\begin{enumerate}[resume]
  \item सिद्ध कीजिए कि $A \in
        \mathcal{M}_n(\C)$ का प्रत्येक \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\lambda$ हर \omterm{thm:b2:nvs:continuouslinear}{संकारक
        मानदंड} के लिए $\abs\lambda \leq
        \vertiii A$ पूरा करता है; \cref{exo:b2:nvs:4} लगाकर
        $\begin{psmallmatrix}1 & -2\\ 3 & 1\end{psmallmatrix}$
        के अभिलक्षणिक मानों को परिबद्ध कीजिए और उनके सच्चे मापांक से
        तुलना कीजिए।
  \item (अनुकूलित \omterm{def:b2:nvs:norm}{मानदंड}) मान लीजिए $A$ \omterm{def:b2:reduction:diag}{विकर्णनीय} है, $A =
        P\,\mathrm{diag}(\lambda_1, \dots, \lambda_n)\,P^{-1}$। दिखाइए कि
        $N_P(x) = \norm{P^{-1}x}_\infty$ ऐसा \omterm{def:b2:nvs:norm}{मानदंड} है जिसका \omterm{thm:b2:nvs:continuouslinear}{संकारक मानदंड} $\vertiii A_{N_P} =
        \max_i\abs{\lambda_i}$ पूरा करता है।
  \item इससे निष्कर्ष निकालिए: \omterm{def:b2:reduction:diag}{विकर्णनीय} $A$ के लिए $A^k \to 0$ यदि और केवल
        यदि सभी \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\abs{\lambda_i} < 1$ पूरा करें --- और \omterm{thm:b2:nvs:finitedim}{मानदंडों की
        तुल्यता} निष्कर्ष को मानदंड-निरपेक्ष बना देती है। $A =
        \frac14\begin{psmallmatrix}1 & 2\\ 2 &
        1\end{psmallmatrix}$ पर
        जाँचिए।
  \item (तुल्यता के अचर फट जाते हैं) $\R_n[X]$ पर $N_c(P) = \max_k \abs{a_k}$ (गुणांक) और $\norm{P}_{\intcc{0}{1}}$ की
        तुलना कीजिए: दोनों \omterm{def:b2:nvs:norm}{मानदंड} हैं, अतः हर स्थिर $n$ के लिए वे
        \omterm{def:b2:nvs:equivalent}{तुल्य} हैं; परंतु $\intcc{0}{1}$ पर प्रश्न 13 के एकिक न्यूनतमकारी का उपयोग
        करके दिखाइए कि $N_c \leq
        C_n\norm\cdot_{\intcc{0}{1}}$ में सर्वोत्तम अचर $C_n$ $C_n \geq
        2^{2n-1}$ पूरा करता है।
        एक वाक्य में निष्कर्ष निकालिए कि ``सभी \omterm{def:b2:nvs:norm}{मानदंड} \omterm{def:b2:nvs:equivalent}{तुल्य} हैं'' अनंत
        विमा में क्यों मर जाता है।
  \item (संश्लेषण) एक-एक वाक्य में: परिमित-विमीय गोलों की संहतता कहाँ
        काम आई (प्रश्न 1, 12); यथार्थ उत्तलता किस पर शासन करती है;
        प्रश्न 18--19 का नक्षत्र क्या नष्ट कर देता है; और प्रश्न 24 उस
        विफलता को कैसे परिमाणित करता है। शिखर का नाम लीजिए (चेबिशेव की
        चरम प्रमेय) और बताइए कि सर्वोत्तम सन्निकटन को अपना आधुनिक घर
        कहाँ मिलता है (हिल्बर्ट समष्टियों पर प्रक्षेप प्रमेय, तृतीय
        वर्ष का खंड, जहाँ संहतता का स्थान पूर्णता ले लेती है)।
\end{enumerate}
\end{problem}
