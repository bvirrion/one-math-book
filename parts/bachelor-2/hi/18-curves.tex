है कि वह कितना मुड़ता है। विमा $3$ में एक दूसरा निश्चर,
\emph{\omterm{thm:b2:curves:frenet3d}{ऐंठन}}, यह मापता है कि वक्र अपने समतल से बाहर कितना बल खाता है। इस
सब का लेखा रखने वाला यंत्र गतिशील \emph{फ्रेने ढाँचा} है।

\section{प्राचलित चाप}

\begin{definition}[प्राचलित चाप]\label{def:b2:curves:arc}
वर्ग $\mathcal{C}^k$ ($k \geq 1$) का \emph{प्राचलित चाप}\index{प्राचलित चाप}
किसी अंतराल $I$ पर वर्ग $\mathcal{C}^k$ का प्रतिचित्रण $\gamma \colon I \to \R^n$ है।
बिंदु $\gamma(t)$ \emph{नियमित} है यदि $\gamma'(t) \neq 0$, और चाप \emph{नियमित}
है यदि उसके सारे बिंदु नियमित हों। $\gamma'(t)$ से दिशित, $\gamma(t)$ से होकर
जाने वाली रेखा नियमित बिंदु पर \emph{स्पर्श रेखा}\index{स्पर्श रेखा} है।
\end{definition}

\begin{definition}[प्राचल परिवर्तन]\label{def:b2:curves:reparam}
वर्ग $\mathcal{C}^k$ का \emph{प्राचल परिवर्तन}\index{प्राचल परिवर्तन}
अंतरालों के बीच कोई $\mathcal{C}^k$ अवकल समाकृतिकता $\theta \colon J \to I$ है (हर जगह
$\theta' \neq 0$)। चाप $\gamma$ और $\gamma \circ \theta$ \emph{तुल्य} कहलाते हैं;
\emph{ज्यामितीय चाप}\index{ज्यामितीय चाप} (या वक्र) कोई तुल्यता वर्ग है।
प्राचल परिवर्तन के अंतर्गत निश्चर धारणाएँ --- पथ, \omterm{def:b2:curves:arc}{स्पर्श रेखा},
\omterm{def:b2:curves:length}{चाप-लंबाई}, \omterm{thm:b2:curves:frenet2d}{वक्रता} --- \emph{ज्यामितीय} कहलाती हैं।
\end{definition}

\begin{remark}
अकेला पथ \omterm{def:b2:curves:reparam}{ज्यामितीय चाप} को निर्धारित नहीं करता: $[0, 2\pi]$ पर तथा
$[0, 4\pi]$ पर प्राचलन $t \mapsto (\cos t, \sin t)$ का प्रतिबिंब एक ही है, पर वे वृत्त को
क्रमशः एक बार और दो बार तय करते हैं। \omterm{def:b2:curves:reparam}{ज्यामितीय चाप} फेरे की बहुलता और
अभिविन्यास याद रखता है, उसकी चाल नहीं।
\end{remark}

\begin{example}[चाल ज्यामितीय रूप से कुछ नहीं
बदलती]\label{ex:b2:curves:speedinvariance}
इकाई वृत्त का प्राचलन इस प्रकार कीजिए:
\[
\gamma(t) = (\cos t^2,\ \sin t^2),
\qquad t \in \intcc0{\sqrt{2\pi}} .
\]
चाल $\norm{\gamma'(t)} = 2t$ रैखिक रूप से बढ़ती है, फिर भी
\[
L = \int_0^{\sqrt{2\pi}}2t\,\dd t = 2\pi ,
\]
अर्थात् वही \omterm{def:b2:curves:length}{लंबाई} जो अचर चाल पर मिलती --- जैसा \cref{thm:b2:curves:lengthinv} वादा करती है,
\omterm{def:b2:curves:reparam}{प्राचल परिवर्तन} $t \mapsto t^2$ के द्वारा। \omterm{def:b2:curves:arc}{स्पर्श रेखा}, \cref{prop:b2:curves:kappaformula} से परिकलित
\omterm{thm:b2:curves:frenet2d}{वक्रता}, और हर दूसरी ज्यामितीय राशि भी मेल खाती है; केवल $t = 0$
एक नज़र का हक़दार है, जहाँ $\gamma'(0) = 0$ \emph{इस प्राचलन} को अनियमित बना
देता है यद्यपि पथ पूरा-पूरा वृत्त है। ज्यामितीय कथन बुरे प्राचलनों को
बुरी तरह सहते हैं: पहले पुनःप्राचलन कीजिए, निष्कर्ष बाद में
निकालिए।
\end{example}

\begin{example}\label{ex:b2:curves:cusp}
चाप $\gamma(t) = (t^2, t^3)$ $\mathcal{C}^\infty$ है पर नियमित नहीं: $\gamma'(0) = (0, 0)$। उसका पथ,
अर्धघनीय परवलय $y^2 = x^3$, मूल बिंदु पर एक \emph{उभयाग्र} रखता है:
प्राचलन की चिकनाई वहाँ ज्यामितीय विचित्रता नहीं रोकती जहाँ वेग लुप्त हो
जाता है। इसीलिए नियमितता की परिकल्पना $\gamma' \neq 0$ केवल सजावट नहीं है।
\end{example}

\section{चाप-लंबाई}

\begin{definition}[चाप-लंबाई]\label{def:b2:curves:length}
मान लीजिए $\gamma \colon [a, b] \to \R^n$ कोई $\mathcal{C}^1$ चाप है। उसकी
\emph{लंबाई}\index{चाप-लंबाई} है
\[
L(\gamma) = \int_a^b \norm{\gamma'(t)}\, \dd t ,
\]
जहाँ $\norm{\cdot}$ यूक्लिडीय \omterm{def:b2:nvs:norm}{मानदंड} है। $t_0$ पर आधारित \emph{चाप-लंबाई
फलन}\index{चाप-लंबाई फलन} $s(t) = \int_{t_0}^t
\norm{\gamma'(u)}\,\dd u$ है।
\end{definition}

\begin{theorem}[लंबाई ज्यामितीय है; बहुभुजीय
अभिलक्षणन]\label{thm:b2:curves:lengthinv}
\leavevmode
\begin{enumerate}
\item यदि $\theta \colon [c, d] \to [a, b]$ कोई $\mathcal{C}^1$
\omterm{def:b2:curves:reparam}{प्राचल परिवर्तन} हो, तो $L(\gamma \circ \theta) = L(\gamma)$।
\item $L(\gamma)$ अंतर्लिखित बहुभुजों की लंबाइयों का उच्चतम है:
\[
L(\gamma)
= \sup\Bigl\{\, \sum_{i=1}^{m} \norm{\gamma(t_i) -
\gamma(t_{i-1})} \;:\; a = t_0 < t_1 < \dots < t_m = b \,\Bigr\}.
\]
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} चरों के परिवर्तन $t = \theta(u)$ से (प्रथम वर्ष का
खंड; वैध है क्योंकि $\theta$ $\mathcal{C}^1$ एकदिष्ट है),
\[
\int_c^d \norm{(\gamma\circ\theta)'(u)}\,\dd u
= \int_c^d \norm{\gamma'(\theta(u))}\,\abs{\theta'(u)}\,\dd u
= \int_a^b \norm{\gamma'(t)}\,\dd t ,
\]
जहाँ हमने $(\gamma\circ\theta)' = \theta'\cdot
(\gamma'\circ\theta)$ का उपयोग किया, और यदि $\theta$ ह्रासमान हो, तो
$\theta'$ का चिह्न सीमाओं के उलट जाने में समा जाता है।

\emph{2.} किसी भी विभाजन के लिए $\gamma(t_i) - \gamma(t_{i-1}) =
\int_{t_{i-1}}^{t_i}\gamma'(t)\,\dd t$, अतः समाकलों की त्रिभुज
असमिका से $\norm{\gamma(t_i) - \gamma(t_{i-1})}
\leq \int_{t_{i-1}}^{t_i}\norm{\gamma'}$: हर बहुभुज $L(\gamma)$ से छोटा है, अतः $\sup \leq L(\gamma)$।

विपरीत असमिका के लिए मान लीजिए $\varepsilon > 0$। चूँकि $\gamma'$
\omterm{def:b2:metric:compact}{संहत} $[a,b]$ पर \omterm{def:b2:metric:continuity}{संतत} है, वह एकसमान \omterm{def:b2:metric:continuity}{संतत} है: ऐसा $\delta > 0$ है कि
$\norm{\gamma'(t) - \gamma'(u)} \leq
\varepsilon$ जब भी $\abs{t - u} \leq \delta$। पग $\leq \delta$ का कोई विभाजन
लीजिए। हर टुकड़े पर, $t \in [t_{i-1}, t_i]$ के लिए,
\[
\gamma(t_i) - \gamma(t_{i-1})
= \int_{t_{i-1}}^{t_i}\gamma'(t)\,\dd t
= (t_i - t_{i-1})\,\gamma'(t_{i-1}) + R_i,
\qquad
\norm{R_i} \leq \varepsilon\,(t_i - t_{i-1}),
\]
क्योंकि $R_i = \int_{t_{i-1}}^{t_i}(\gamma'(t) -
\gamma'(t_{i-1}))\,\dd t$। अतः
\[
\norm{\gamma(t_i) - \gamma(t_{i-1})}
\geq (t_i - t_{i-1})\norm{\gamma'(t_{i-1})}
- \varepsilon (t_i - t_{i-1}) .
\]
जोड़ने पर, और $\sum (t_i -
t_{i-1})\norm{\gamma'(t_{i-1})}$ की तुलना $\int_a^b\norm{\gamma'}$ से करने पर (जो \omterm{def:b2:metric:continuity}{संतत} फलन
$\norm{\gamma'}$ का रीमान योग है, और एकसमान संततता से पर्याप्त छोटे $\delta$
के लिए समाकल के $\varepsilon(b - a)$ के भीतर है), हमें $\geq
L(\gamma) - 2\varepsilon(b - a)$ \omterm{def:b2:curves:length}{लंबाई} का बहुभुज
मिलता है। $\varepsilon \to 0$ लेने पर दावा सिद्ध हो जाता है।
\end{proof}

\begin{example}[आर्किमिडीज़ और अंतर्लिखित
बहुभुज]\label{ex:b2:curves:archimedes}
इकाई वृत्त के लिए अंतर्लिखित सम $n$-भुज की \omterm{def:b2:curves:length}{लंबाई}
$L_n = 2n\sin\frac\pi n$ है, और प्रसार $\sin x = x -
\frac{x^3}6 + O(x^5)$ देता है
\[
L_n = 2\pi - \frac{\pi^3}{3n^2} + O\Bigl(\frac1{n^4}\Bigr) :
\]
अर्थात् \cref{thm:b2:curves:lengthinv} की बहुभुजीय लंबाइयाँ \emph{वर्गिक} रूप से
अभिसरण करती हैं। संख्यात्मक रूप से: $L_6 = 6$
(षट्भुज, जो कच्चा $\pi > 3$ देता है), जबकि $2\pi \approx
6.28319$ के सामने $L_{96} =
192\sin\frac{\pi}{96} \approx 6.28206$ --- त्रुटि $0.00113$ अनुमानित
$\pi^3/(3\cdot96^2) \approx 0.00112$ से मेल खाती है। इसीलिए आर्किमिडीज़ षट्भुज को पाँच बार
दुगुना करके $96$ भुजाओं तक ले जाकर हाथ से ही
$\pi$ को तीन अंकों तक घेर सके: हर दुगुनापन त्रुटि को चार से
भाग देता है। बहुभुजीय अभिलक्षणन का उच्चतम केवल सीमा में ही प्राप्त
नहीं होता; वह \emph{तेज़ी से} प्राप्त होता है, क्योंकि कोई चिकना वक्र
अपनी जीवाओं से केवल द्वितीय कोटि पर ही अलग होता है।
\end{example}

\begin{theorem}[चाप-लंबाई प्राचलन]\label{thm:b2:curves:arclength}
मान लीजिए $\gamma \colon I \to \R^n$ कोई नियमित $\mathcal{C}^k$ चाप है
($k \geq 1$)। \omterm{def:b2:curves:length}{चाप-लंबाई फलन} $s$ $I$ से किसी अंतराल $J$ पर $\mathcal{C}^k$
अवकल समाकृतिकता है, और $\tilde\gamma =
\gamma \circ s^{-1}$ हर जगह $\norm{\tilde\gamma'} = 1$ पूरा
करता है। प्राचल के स्थानांतरण तथा अभिविन्यास तक यह \emph{\omterm{def:b2:curves:length}{चाप-लंबाई}}
(या \emph{इकाई-चाल}) प्राचलन अद्वितीय है।
\end{theorem}

\begin{proof}
$s'(t) = \norm{\gamma'(t)} > 0$ और $s'$ $\mathcal{C}^{k-1}$ है
($\mathcal{C}^{k-1}$ प्रतिचित्रण $\gamma'$ का \omterm{def:b2:nvs:norm}{मानदंड} के साथ संयोजन,
जो $0$ से दूर चिकना है), अतः $s$ $\mathcal{C}^k$ है, दृढ़ता से
वर्धमान, $J = s(I)$ पर एकैकी आच्छादक, और विमा $1$ में प्रतिलोम
फलन प्रमेय (प्रथम वर्ष का खंड) से उसका प्रतिलोम $\mathcal{C}^k$ है। तब
\[
\tilde\gamma'(\sigma)
= \frac{\gamma'(t)}{s'(t)}
= \frac{\gamma'(t)}{\norm{\gamma'(t)}},
\qquad t = s^{-1}(\sigma),
\]
जो इकाई सदिश है। यदि $\hat\gamma = \gamma\circ\theta$ कोई दूसरा
इकाई-चाल प्राचलन हो, तो $\abs{\theta'} = 1$, अतः संततता से $\theta'
= \pm 1$ अचर, अर्थात् $\theta(u) = \pm u + c$।
\end{proof}

\begin{remark}
\omterm{def:b2:curves:length}{चाप-लंबाई} वह प्राचल है जो ज्यामिति को गतिकी से अलग करता है। किसी पथ
को किसी भी चाल-रूपरेखा के साथ तय किया जा सकता है --- यही गति की
भौतिकी है --- पर प्राचलन-निश्चर हर प्रश्न (आकार, मुड़ाव, चुंबन) की
एक विहित घड़ी होती है: तय की गई दूरी। इसीलिए नीचे के सारे \omterm{thm:b2:curves:frenet2d}{वक्रता}
सूत्र इकाई चाल पर \emph{परिभाषित} किए जाते हैं और फिर शृंखला नियम से
स्वेच्छ प्राचलनों में \emph{अनूदित} किए जाते हैं: अनुवाद के गुणनखंड
$v = s'$ की घातें हैं, और उन्हें ठीक-ठीक साधना ही \cref{prop:b2:curves:kappaformula} की पूरी
अंतर्वस्तु है।
\end{remark}

\begin{example}[वृत्त और कुंडली]\label{ex:b2:curves:helix}
वृत्त $\gamma(t) = (R\cos t, R\sin t)$ के लिए $\norm{\gamma'} =
R$, अतः $s = Rt$, और एक पूरे फेरे की \omterm{def:b2:curves:length}{लंबाई} $2\pi R$ है।
$a > 0$ वाली कुंडली $\gamma(t) = (a\cos t,\ a\sin t,\ bt)$ के लिए
$\norm{\gamma'(t)} = \sqrt{a^2 + b^2}$ अचर है: कुंडली अचर चाल पर तय होती है, और $s = t\sqrt{a^2 + b^2}$।
\end{example}

\begin{example}[ध्रुवीय निर्देशांकों में
चाप-लंबाई]\label{ex:b2:curves:polarlength}
ध्रुवीय वक्र $r = r(\theta)$ चाप $\gamma(\theta) =
(r\cos\theta,\ r\sin\theta)$ है, जहाँ
\[
\gamma'(\theta) = (r'\cos\theta - r\sin\theta,\
r'\sin\theta + r\cos\theta),
\qquad
\norm{\gamma'(\theta)}^2 = r'^2 + r^2
\]
(मिश्रित पद कट जाते हैं): अर्थात् ध्रुवीय लंबाई-अवयव
$\sqrt{r^2 + r'^2}\,\dd\theta$ है। कार्डिऑइड $r = 1 +
\cos\theta$ के लिए:
\[
r^2 + r'^2 = (1 + \cos\theta)^2 + \sin^2\theta = 2 +
2\cos\theta = 4\cos^2\tfrac\theta2,
\]
और $\intcc{-\pi}{\pi}$ पर अर्ध-कोण की कोसाइन अऋणात्मक है, अतः
\[
L = \int_{-\pi}^{\pi}2\cos\tfrac\theta2\,\dd\theta
= \Bigl[4\sin\tfrac\theta2\Bigr]_{-\pi}^{\pi}
= 4 - (-4) = 8 :
\]
\cref{exo:b2:curves:1} के चक्रज मेहराब की तरह, वृत्तों से बना कोई वक्र परिमेय
\omterm{def:b2:curves:length}{लंबाई} रखता है, और कहीं कोई $\pi$ नहीं। अर्ध-कोण गुणनखंडन
वृत्त-जनित वक्रों की लंबाइयों के लिए मानक युक्ति है; जब वह विफल हो
जाती है (दीर्घवृत्त), तब \omterm{def:b2:curves:length}{लंबाई} सचमुच एक नया फलन बन जाती है --- कोई
दीर्घवृत्तीय समाकल, जो प्रारंभिक संवृत रूपों से परे है।
\end{example}

\section{समतल में वक्रता}

इस अनुभाग भर चाप अभिविन्यस्त यूक्लिडीय समतल में $\mathcal{C}^2$ तथा
नियमित हैं। हम \omterm{def:b2:curves:length}{चाप-लंबाई} से प्राचलन करते हैं और इकाई स्पर्शी के लिए
$T(s) = \tilde\gamma'(s)$ लिखते हैं, तथा $T(s)$ के सीधे लांबिक इकाई सदिश के
लिए $N(s)$ (अर्थात् $T$ का $+\pi/2$ से घूर्णन)।

\begin{theorem}[समतल फ्रेने सूत्र]\label{thm:b2:curves:frenet2d}
मान लीजिए $\tilde\gamma$ अभिविन्यस्त समतल में कोई इकाई-चाल $\mathcal{C}^2$ चाप
है। ऐसा \omterm{def:b2:metric:continuity}{संतत} फलन $\kappa$ है, जो (बीजीय)
\emph{वक्रता}\index{वक्रता} कहलाता है, कि
\[
T'(s) = \kappa(s)\, N(s),
\qquad
N'(s) = -\kappa(s)\, T(s).
\]
\end{theorem}

\begin{proof}
चूँकि सब $s$ के लिए $\norm{T(s)}^2 = 1$, अदिश गुणन का अवकलन
$2\langle T'(s), T(s)\rangle = 0$ देता है: $T'(s)$ $T(s)$ के लांबिक है, अतः
$N(s)$ के संरेख (विमा $2$); $T'(s)
= \kappa(s) N(s)$ लिखिए, जहाँ $\kappa(s) = \langle T'(s), N(s)\rangle$
\omterm{def:b2:metric:continuity}{संतत} है। इसी प्रकार $N' \perp N$, अतः $N' = \lambda T$; और
$\langle T, N\rangle = 0$ का अवकलन $\langle T', N\rangle +
\langle T, N'\rangle = \kappa + \lambda = 0$ देता है।
\end{proof}

\begin{definition}\label{def:b2:curves:curvature}
जब $\kappa(s) \neq 0$, तब \emph{वक्रता
त्रिज्या}\index{वक्रता त्रिज्या} $R(s) =
1/\abs{\kappa(s)}$ है और \emph{वक्रता
केंद्र}\index{वक्रता केंद्र}
$\tilde\gamma(s) + \frac{1}{\kappa(s)} N(s)$ है; उस केंद्र तथा त्रिज्या $R(s)$
वाला वृत्त \emph{चुंबन वृत्त}\index{चुंबन वृत्त} है, जो $\tilde\gamma(s)$ पर
वक्र का सर्वोत्तम वृत्तीय सन्निकटन है।
\end{definition}

\begin{example}[चरघातांकी का चुंबन
वृत्त]\label{ex:b2:curves:oscexp}
बिंदु $(0, 1)$ पर $y = \eu^x$ के लिए: $f'(0) = f''(0) = 1$,
अतः नीचे दिए ग्राफ़-सूत्र से
\[
\kappa(0) = \frac{1}{(1 + 1)^{3/2}} = \frac1{2\sqrt2},
\qquad R = 2\sqrt2 .
\]
इकाई स्पर्शी $T = \frac{(1, 1)}{\sqrt2}$ है, सीधा अभिलंब
$N = \frac{(-1, 1)}{\sqrt2}$, और \omterm{def:b2:curves:curvature}{वक्रता केंद्र}
\[
(0, 1) + 2\sqrt2\cdot\frac{(-1, 1)}{\sqrt2} = (-2,\ 3) :
\]
है; \omterm{def:b2:curves:curvature}{चुंबन वृत्त} का समीकरण $(x + 2)^2 + (y - 3)^2 =
8$ है। ``सर्वोत्तम वृत्तीय सन्निकटन'' के दावे की जाँच के रूप में:
$(0,1)$ के निकट वृत्त के समीकरण को $y$ के लिए हल करके
प्रसारित करने पर $y = 1 + x + \frac{x^2}2 + O(x^3)$ मिलता है ---
ठीक $\eu^x$ का द्वितीय-कोटि टेलर प्रसार। \omterm{def:b2:curves:curvature}{चुंबन वृत्त} मान, ढाल
\emph{तथा} द्वितीय अवकलज तीनों से मेल खाता है; कोई साधारण स्पर्शी
वृत्त केवल पहले दो से मेल खाता।
\end{example}

\begin{example}[वृत्त की विकसिता उसका
केंद्र है]\label{ex:b2:curves:circleevolute}
वामावर्त तय किए गए त्रिज्या $R$ के वृत्त के लिए
$\kappa = 1/R$ और $N$ केंद्र की ओर संकेत करता है, अतः हर $s$
के लिए \omterm{def:b2:curves:curvature}{वक्रता केंद्र} $\tilde\gamma + \frac1\kappa N$ वृत्त का
केंद्र ही है: वृत्त का \omterm{def:b2:curves:curvature}{चुंबन वृत्त} स्वयं वही वृत्त है, और \omterm{thm:b2:curves:frenet2d}{वक्रता}
केंद्रों का बिंदुपथ एक बिंदु में सिमट जाता है। यह अपभ्रष्ट स्थिति
\cref{exo:b2:curves:6} का अंशांकन करती है: वहाँ विकसिता का
वेग $-\frac{\kappa'}{\kappa^2}N$ है, जो ठीक तभी सर्वत्र लुप्त
होता है जब $\kappa$ अचर हो।
\end{example}

\begin{proposition}[स्वेच्छ प्राचलन में
वक्रता]\label{prop:b2:curves:kappaformula}
किसी नियमित $\mathcal{C}^2$ समतल चाप $\gamma(t) = (x(t), y(t))$ के लिए
\[
\kappa(t)
= \frac{x'(t)\,y''(t) - y'(t)\,x''(t)}
       {\bigl(x'(t)^2 + y'(t)^2\bigr)^{3/2}} ,
\]
विशेष रूप से ग्राफ़ $y = f(x)$ के लिए $\kappa = \dfrac{y''}{(1 + y'^2)^{3/2}}$।
\end{proposition}

\begin{proof}
$v(t) = \norm{\gamma'(t)} = s'(t)$ लिखिए, अतः $\gamma' = vT$
(इकाई-चाल आँकड़ों का $s$ के साथ संयोजन)। अवकलन करने पर,
\[
\gamma'' = v'T + v\,T'\cdot s' = v'T + v^2\kappa N .
\]
अब $(\gamma', \gamma'')$ का \omterm{def:b2:linalg:det}{सारणिक} (विहित अभिविन्यस्त आधार में) लीजिए: चूँकि
$\det(T, T) = 0$ और $\det(T, N) = 1$,
\[
\det(\gamma', \gamma'') = \det(vT,\ v'T + v^2\kappa N)
= v^3\kappa .
\]
बायाँ पक्ष $x'y'' - y'x''$ है, और $v^3 = (x'^2 +
y'^2)^{3/2}$। ग्राफ़ की स्थिति प्राचलन $t \mapsto (t,
f(t))$ है।
\end{proof}

\begin{example}[वृत्त, रेखा, परवलय]\label{ex:b2:curves:kappaexamples}
रेखा की $\kappa = 0$ होती है (और विलोमतः: $T' = 0$ का अर्थ है $T$
अचर, अतः $\tilde\gamma(s) = \tilde\gamma(0) + sT$, यानी कोई रेखा)। वामावर्त तय
किए गए त्रिज्या $R$ के वृत्त की $\kappa = 1/R$ होती है:
$\gamma(t) = (R\cos t, R\sin t)$ के साथ सूत्र $\kappa =
R^2/R^3$ देता है। परवलय $y = x^2/2$ के लिए:
$\kappa(x) = 1/(1 +
x^2)^{3/2}$, जो शीर्ष पर अधिकतम है --- परवलय वहीं सबसे तीखा मुड़ा होता है
जहाँ वह पलटता है।
\end{example}

\begin{remark}[वक्रता के आसपास की सामान्य चूकें]
(क) समतल चाप की बीजीय \omterm{thm:b2:curves:frenet2d}{वक्रता} तब चिह्न बदलती है जब चाप का या समतल
का अभिविन्यास उलट दिया जाए: केवल
$\abs\kappa$ और $R = 1/\abs\kappa$ विशुद्ध रूप से ज्यामितीय हैं। दक्षिणावर्त
तय किए गए वृत्त की $\kappa = -1/R$ होती है। (ख) ग्राफ़-सूत्र
$\kappa = f''/(1 + f'^2)^{3/2}$ चुपचाप $x$ द्वारा
प्राचलन चुन लेता है; उसे ऐसे वक्र पर लगाना जो उस बिंदु के निकट
ग्राफ़ नहीं है (ऊर्ध्वाधर स्पर्शी), शास्त्रीय भूल है। (ग) जिस बिंदु
पर $\gamma' = 0$, वहाँ कुछ भी परिभाषित नहीं --- न $T$, न
$\kappa$ --- और पथ सचमुच टूट सकता है
(\cref{ex:b2:curves:cusp}); इकाई स्पर्शी का अवकलन करने से पहले
नियमितता सदा जाँचिए। (घ) अवकाश में परंपरा से $\kappa =
\norm{T'} \geq 0$: वहाँ कोई
चिह्न ग़लत करने को नहीं है, पर भुनाने को भी नहीं --- नति-परिवर्तन
जैसी सूचना \omterm{thm:b2:curves:frenet3d}{ऐंठन} में चली जाती है। (ङ) अंत में, $\kappa$
\emph{\omterm{def:b2:curves:length}{चाप-लंबाई} के सापेक्ष} अवकलज है: इकाई-चाल से भिन्न प्राचलन के
लिए \cref{prop:b2:curves:kappaformula} में गुणनखंड $v^3$ भूल जाना व्यवहार
में सबसे बार-बार होने वाली भूल है।
\end{remark}

\begin{theorem}[वक्रता वक्र को निर्धारित करती है]\label{thm:b2:curves:fundamental}
मान लीजिए $\kappa \colon J \to \R$ \omterm{def:b2:metric:continuity}{संतत} है। समतल में \omterm{thm:b2:curves:frenet2d}{वक्रता}
$\kappa$ वाला कोई इकाई-चाल $\mathcal{C}^2$ चाप विद्यमान है, और वह किसी
सीधे समदूरिक प्रतिचित्रण (घूर्णन के बाद स्थानांतरण) तक अद्वितीय है।
\end{theorem}

\begin{proof}
\emph{अस्तित्व।} $s_0 \in J$ स्थिर कीजिए और $\varphi(s) =
\int_{s_0}^s \kappa(u)\,\dd u$ रखिए, फिर
\[
\tilde\gamma(s) = \Bigl(\int_{s_0}^s \cos\varphi(u)\,\dd u,\
\int_{s_0}^s \sin\varphi(u)\,\dd u\Bigr).
\]
तब $T(s) = (\cos\varphi(s), \sin\varphi(s))$ इकाई सदिश है,
$N(s) = (-\sin\varphi, \cos\varphi)$, और
\[
T'(s) = \varphi'(s)\,(-\sin\varphi, \cos\varphi)
= \kappa(s)\,N(s) :
\]
अर्थात् चाप इकाई-चाल का है और उसकी \omterm{thm:b2:curves:frenet2d}{वक्रता} $\kappa$ है।

\emph{अद्वितीयता।} मान लीजिए $\gamma_1, \gamma_2$ एक ही \omterm{thm:b2:curves:frenet2d}{वक्रता} वाले इकाई-चाल
चाप हैं। हर इकाई स्पर्शी किसी कोण फलन तक उठ जाता है, और वह भी एक
स्पष्ट रचना से: $T_j$ को मापांक $1$ की सम्मिश्र
संख्या $z_j = a_j + \iu b_j$ के रूप में देखिए, $z_j(0) = \eu^{\iu\varphi_j(0)}$ के साथ
$\varphi_j(0)$ चुनिए, और रखिए
\[
\varphi_j(s) = \varphi_j(0) + \int_0^s\det\bigl(T_j,
T_j'\bigr)(u)\,\dd u .
\]
$\abs{z_j} = 1$ से: $\operatorname{Re}(\conj{z_j}\,z_j') =
0$, अतः $\conj{z_j}\,z_j' = \iu\det(T_j, T_j') =
\iu\,\varphi_j'$, अर्थात् $z_j' = \iu\varphi_j'z_j$; फिर
\[
\bigl(z_j\,\eu^{-\iu\varphi_j}\bigr)'
= \eu^{-\iu\varphi_j}\bigl(z_j' - \iu\varphi_j'z_j\bigr) = 0,
\]
अतः सर्वत्र $z_j = \eu^{\iu\varphi_j}$: $T_j =
(\cos\varphi_j, \sin\varphi_j)$, जहाँ $\varphi_j$ वर्ग
$\mathcal C^1$ का है। इसके अतिरिक्त $\det(T_j, T_j') = \det(T_j,
\kappa N_j) = \kappa$, अतः $\varphi_j' = \kappa$। इसलिए किसी
अचर $c$ के लिए $\varphi_2 = \varphi_1 + c$: $T_2$ ही $T_1$ है, जो नियत कोण $c$ से घुमाया गया है, अतः
समाकलन करने पर $\gamma_2 = \rho(\gamma_1) + w$, जहाँ $\rho$ कोण $c$ का
घूर्णन है और $w$ कोई अचर सदिश।
\end{proof}

\begin{remark}
यह \emph{ज्यामिति की किसी आधारभूत प्रमेय} का एकविमीय प्रतिरूप है:
स्थानीय निश्चरों का कोई पूरा समुच्चय (यहाँ, एक फलन) वस्तु को दृढ़ गति
तक वर्गीकृत कर देता है। नीचे दिए त्रिविमीय रूप को दो निश्चर चाहिए।
\end{remark}

\begin{example}[अचर वक्रता का अर्थ है
वृत्त]\label{ex:b2:curves:constantkappa}
अस्तित्व-सूत्र में $\kappa \equiv \kappa_0 > 0$ लीजिए:
$\varphi(s) = \kappa_0 s$ और
\[
\tilde\gamma(s) = \Bigl(\frac{\sin\kappa_0s}{\kappa_0},\
\frac{1 - \cos\kappa_0s}{\kappa_0}\Bigr) :
\]
अर्थात् $(0,
1/\kappa_0)$ पर केंद्रित त्रिज्या $1/\kappa_0$ का वृत्त, जो इकाई चाल पर तय होता है। प्रमेय के
अद्वितीयता वाले आधे भाग से, अचर \omterm{thm:b2:curves:frenet2d}{वक्रता} $\kappa_0$ वाला \emph{हर}
इकाई-चाल चाप त्रिज्या $1/\kappa_0$ के किसी वृत्त का टुकड़ा है
(या $\kappa_0 = 0$ होने पर कोई रेखा) --- यही \cref{ex:b2:curves:kappaexamples} के परिकलन
का विलोम है, और \cref{exo:b2:curves:9} की समतल स्थिति।
\end{example}

\begin{example}[वक्रता से वक्र का
पुनर्निर्माण]\label{ex:b2:curves:reconstruction}
किस इकाई-चाल वक्र की \omterm{def:b2:curves:curvature}{वक्रता त्रिज्या} $R(s) = 1 +
s^2$ है? $\kappa(s) =
\frac1{1+s^2}$ तथा $s_0 = 0$ के साथ अस्तित्व की उपपत्ति का अनुसरण
कीजिए: $\varphi(s) = \arctan s$, अतः
\[
T(s) = (\cos\arctan s,\ \sin\arctan s)
= \Bigl(\frac{1}{\sqrt{1+s^2}},\
\frac{s}{\sqrt{1+s^2}}\Bigr),
\]
और समाकलन करने पर
\[
\tilde\gamma(s) = \Bigl(\ln\bigl(s + \sqrt{1 + s^2}\bigr),\
\sqrt{1 + s^2} - 1\Bigr).
\]
$x = \ln(s + \sqrt{1+s^2})$ रखने पर, अर्थात् $s = \sinh x$, दूसरा
निर्देशांक $\cosh x - 1$ है: यानी वक्र \emph{कैटेनरी}
$y = \cosh x - 1$ है। इससे \cref{exo:b2:curves:3} के साथ घेरा पूरा हो जाता है, जहाँ हमने
$R = \cosh^2 x = 1 +
\sinh^2 x = 1 + s^2$ सीधे परिकलित किया था: आधारभूत प्रमेय
आश्वस्त करती है कि इस वक्रता-रूपरेखा वाला \emph{एकमात्र} वक्र, किसी
सीधे समदूरिक प्रतिचित्रण तक, कैटेनरी ही है।
\end{example}

\begin{remark}[वक्रता आगे कहाँ काम आती है]
\cref{prop:b2:curves:kappaformula} की उपपत्ति में मिला अपघटन $\gamma'' = v'T + v^2\kappa N$ हर
वक्रित गति की गतिकी है: स्पर्शीय बनाम अभिकेंद्री त्वरण। \omterm{thm:b2:curves:frenet2d}{वक्रता}
पृष्ठों के लिए (\cref{ch:b2:surfaces}) उन पर खींचे गए वक्रों की \omterm{thm:b2:curves:frenet2d}{वक्रता} के
द्वारा लौटती है, और इस अध्याय की सप्ताहांत समस्या का अन्वालोप-कलन ---
विकसिताएँ, दाहक वक्र --- तरंगाग्रों की ज्यामितीय प्रकाशिकी है। तृतीय
वर्ष का खंड $\R^n$ के उपबहुविधों के लिए आंतरिक दृष्टिकोण फिर से
उठाता है।
\end{remark}

\section{अवकाश में फ्रेने ढाँचा}

अब मान लीजिए $\tilde\gamma \colon J \to \R^3$ कोई इकाई-चाल
$\mathcal{C}^3$ चाप है जो \emph{द्विनियमित} है: सब $s$ के लिए
$T'(s) \neq 0$। तब $\kappa(s) = \norm{T'(s)} > 0$ \emph{\omterm{thm:b2:curves:frenet2d}{वक्रता}} को परिभाषित करता है
(अवकाश में कोई चिह्न नहीं: कोई अधिमान्य अभिलंब अभिविन्यास नहीं होता),
और हम रखते हैं:
\[
N(s) = \frac{T'(s)}{\kappa(s)}
\quad\text{(मुख्य अभिलंब)},
\qquad
B(s) = T(s) \wedge N(s)
\quad\text{(द्वि-अभिलंब)} ,
\]
जिससे $(T, N, B)$ कोई सीधा प्रसामान्य लांबिक ढाँचा बन जाता है, अर्थात्
\emph{फ्रेने ढाँचा}। $\tilde\gamma(s)$ से होकर जाने वाला, $T, N$ से \omterm{def:b2:structures:generated}{जनित}
समतल \emph{चुंबन समतल} है।

\begin{theorem}[अवकाश में फ्रेने सूत्र]\label{thm:b2:curves:frenet3d}
ऐसा \omterm{def:b2:metric:continuity}{संतत} फलन $\tau$ है, जो \emph{ऐंठन}\index{ऐंठन} कहलाता है, कि
\[
T' = \kappa N, \qquad
N' = -\kappa T + \tau B, \qquad
B' = -\tau N .
\]
\end{theorem}

\begin{proof}
पहला सूत्र $N$ की परिभाषा है। सदिशों
$T, N, B$ में से हर एक का \omterm{def:b2:nvs:norm}{मानदंड} अचर $1$ है और वे जोड़े-जोड़े में
लांबिक हैं; छह संबंधों $\langle X, Y\rangle = \delta_{XY}$ का अवकलन दिखाता
है कि आधार $(T, N, B)$ में $(T', N', B')$ का आव्यूह प्रतिसममित है: वस्तुतः
$\langle X', Y\rangle + \langle X, Y'\rangle = 0$ तथा
$\langle X', X\rangle = 0$। उसकी $(N, T)$ प्रविष्टि $\langle N', T\rangle =
-\langle N, T'\rangle = -\kappa$ है, और उसकी $(T, B)$-स्तंभ
प्रविष्टि $\langle T', B\rangle = \kappa\langle N, B\rangle = 0$। शेष स्वतंत्र प्रविष्टि को
$\tau = \langle N', B\rangle$ नाम देने पर ठीक तीनों प्रदर्शित सूत्र मिल जाते हैं:
प्रतिसममिति $\langle B', N\rangle = -\tau$ तथा
$\langle B', T\rangle = 0$ भर देती है। $\tau = \langle N', B\rangle$ की संततता स्पष्ट है,
क्योंकि $N'$ और $B$ \omterm{def:b2:metric:continuity}{संतत} हैं ($\tilde\gamma$ $\mathcal{C}^3$ है, अतः
$N = T'/\kappa$ $\mathcal{C}^1$ है)।
\end{proof}

\begin{example}[दार्बू सदिश]\label{ex:b2:curves:darboux}
तीनों फ्रेने सूत्र एक में सिमट जाते हैं। $\omega(s) =
\tau\,T + \kappa\,B$ रखिए (\emph{दार्बू सदिश})। $B
\wedge T = N$, $T \wedge N = B$, $N \wedge B = T$ का उपयोग करने पर:
\[
\omega \wedge T = \kappa\,N = T', \qquad
\omega \wedge N = \tau\,B - \kappa\,T = N', \qquad
\omega \wedge B = -\tau\,N = B' :
\]
अर्थात् ढाँचे का हर सदिश $X' = \omega \wedge X$ के अनुसार विकसित होता है, जो
कोणीय वेग सदिश $\omega$ वाले किसी \emph{तात्क्षणिक घूर्णन} का
गतिकीय हस्ताक्षर है। ढाँचा गतिशील अक्ष $\omega$ के परितः
$\norm\omega = \sqrt{\kappa^2 + \tau^2}$ दर से घूमता है; \omterm{thm:b2:curves:frenet2d}{वक्रता}
द्वि-अभिलंब के परितः घूर्णन का घटक है, \omterm{thm:b2:curves:frenet3d}{ऐंठन} स्पर्शी के परितः का
घटक। कुंडली के लिए $\omega$ बेलन के अक्ष के अनुदिश कोई अचर सदिश है
--- और ठीक इसीलिए कुंडली का ढाँचा \omterm{def:b2:funcseq:def}{एकसमान रूप से} पुरस्सरण करता है।
फ्रेने आव्यूह की प्रतिसममिति, जिसका \cref{exo:b2:curves:7} में उपयोग हुआ,
इसी अकेले ज्यामितीय तथ्य का आव्यूह रूप है।
\end{example}

\begin{proposition}[ऐंठन समतलता मापती है]\label{prop:b2:curves:torsion}
कोई द्विनियमित चाप किसी समतल में समाहित है तभी जब $\tau
\equiv 0$; उस स्थिति में वह समतल (अचर) चुंबन
समतल है।
\end{proposition}

\begin{proof}
यदि $\tau \equiv 0$, तो $B' = 0$, अतः $B$ कोई अचर इकाई
सदिश $B_0$ है, और
\[
\frac{\dd}{\dd s}\langle \tilde\gamma(s), B_0\rangle
= \langle T(s), B_0\rangle = 0 :
\]
अर्थात् $\langle \tilde\gamma, B_0\rangle$ अचर है, इसलिए चाप $B_0$ के लांबिक किसी समतल में
पड़ता है। विलोमतः, यदि चाप किसी समतल $P$ में पड़े, तो सब $s$
के लिए $T$ तथा $T'$ (अतः $N$) $P$ की दिशा के समांतर
हैं; अतः $B = T \wedge N$ $P$ के दो इकाई अभिलंबों में से एक है, और \omterm{def:b2:metric:continuity}{संतत}
होने के कारण अचर है; तब $N \neq 0$ के साथ $0 = B'
= -\tau N$ $\tau \equiv 0$ को बाध्य कर
देता है।
\end{proof}

\begin{example}[झुके हुए वृत्त की ऐंठन शून्य होती है]
चाप $\gamma(t) = \bigl(\cos t,\ \tfrac{\sin t}{\sqrt2},\
\tfrac{\sin t}{\sqrt2}\bigr)$ समतल $y = z$ में पड़ता है, और उस
समतल का इकाई वृत्त है (जाँचिए: $\norm{\gamma(t)} =
1$, और समतल का प्रसामान्य लांबिक आधार $(1,0,0)$,
$(0, \tfrac1{\sqrt2}, \tfrac1{\sqrt2})$ मानक प्राचलन दिखा देता है)। बिना
किसी फ्रेने परिकलन के \cref{prop:b2:curves:torsion} $\tau \equiv 0$ का अनुमान लगा देती है,
और नियत द्वि-अभिलंब समतल का इकाई अभिलंब $\pm(0,
\tfrac1{\sqrt2}, -\tfrac1{\sqrt2})$ ही होना
चाहिए। \omterm{thm:b2:curves:frenet3d}{ऐंठन} ``अवकाश में झुके होने'' को नहीं मापती; वह किसी समतल को
\emph{छोड़ने} को मापती है। केवल नीचे दी कुंडली का अशून्य $b$ ही
सच्ची \omterm{thm:b2:curves:frenet3d}{ऐंठन} पैदा करता है।
\end{example}

\begin{example}[कुंडली]\label{ex:b2:curves:helixfrenet}
कुंडली $\gamma(t) = (a\cos t, a\sin t, bt)$, $a > 0$, के लिए हमने
$c = \sqrt{a^2 + b^2}$ के साथ $s = ct$ परिकलित किया था। तब
\[
T = \frac1c(-a\sin t,\ a\cos t,\ b),
\qquad
T' \cdot \frac{\dd t}{\dd s}
= \frac{1}{c^2}(-a\cos t, -a\sin t, 0),
\]
अतः $\kappa = a/c^2 = a/(a^2 + b^2)$ और $N = (-\cos t, -\sin t,
0)$: मुख्य अभिलंब क्षैतिज रूप से अक्ष की ओर संकेत करता है।
आगे $B = T \wedge N = \frac1c(b\sin t, -b\cos t, a)$, और $B'
\frac{\dd t}{\dd s} = \frac{b}{c^2}(\cos t, \sin t, 0) = -\tau N$
देता है
\[
\boxed{\ \kappa = \frac{a}{a^2 + b^2}, \qquad
\tau = \frac{b}{a^2 + b^2}. \ }
\]
दोनों निश्चर स्थिर हैं --- और विलोमतः दिखाया जा सकता है कि अचर $\kappa > 0$
तथा अचर $\tau$ वाले एकमात्र द्विनियमित वक्र कुंडलियाँ हैं
($\tau = 0$ होने पर वृत्त)। चिह्नों पर ध्यान दीजिए: $b >
0$ धनात्मक \omterm{thm:b2:curves:frenet3d}{ऐंठन} वाली दक्षिणावर्ती कुंडली देता है।
\end{example}

\begin{example}[चाप-लंबाई के बिना वक्रता और ऐंठन; ऐंठा हुआ
घन वक्र]\label{ex:b2:curves:twistedcubic}
\omterm{def:b2:curves:length}{चाप-लंबाई} से पुनःप्राचलन प्रायः संवृत रूप में असंभव होता है, अतः
निश्चरों को कच्चे अवकलजों से ही निकालना पड़ता है। $v = \norm{\gamma'} = s'$ लिखिए; तब $\gamma' =
vT$, और \cref{prop:b2:curves:kappaformula} की उपपत्ति की तरह,
\[
\gamma'' = v'T + v^2\kappa N,
\qquad
\gamma' \wedge \gamma'' = v^3\kappa\,(T \wedge N) =
v^3\kappa\,B .
\]
\omterm{def:b2:nvs:norm}{मानदंड} लेने पर (अवकाश में $\kappa \geq 0$):
\[
\kappa = \frac{\norm{\gamma' \wedge \gamma''}}{v^3} .
\]
$\gamma''$ का एक बार और अवकलन करके $N' =
v(-\kappa T + \tau B)$ बदलने पर ($s$ के द्वारा शृंखला नियम), एकमात्र
$B$-घटक अंतिम पद से आता है:
\[
\gamma''' = \bigl(v'' - v^3\kappa^2\bigr)T +
\bigl(v'v\kappa + (v^2\kappa)'\bigr)N + v^3\kappa\tau\,B,
\]
जिससे, $\gamma' \wedge \gamma'' = v^3\kappa B$ के साथ युग्मन करने पर,
\[
\det(\gamma', \gamma'', \gamma''') = \langle\gamma' \wedge
\gamma'',\ \gamma'''\rangle = v^6\kappa^2\tau,
\qquad\text{अर्थात्}\qquad
\tau = \frac{\det(\gamma', \gamma'',
\gamma''')}{\norm{\gamma' \wedge \gamma''}^2} .
\]
अनुप्रयोग --- \emph{ऐंठा हुआ घन वक्र} $\gamma(t) = (t,\
t^2,\ t^3)$, $t = 0$ पर: $\gamma' = (1, 0, 0)$, $\gamma'' =
(0, 2, 0)$, $\gamma''' = (0, 0, 6)$, अतः $v = 1$,
\[
\gamma' \wedge \gamma'' = (0, 0, 2), \qquad \kappa(0) = 2,
\qquad
\det(\gamma', \gamma'', \gamma''') = 12, \qquad \tau(0) =
\frac{12}{4} = 3 .
\]
समापन दृष्टि: दोनों सूत्र ऐसे अनुपात हैं जिनमें चाल
$v$ ठीक उतनी ही मात्रा में कट जाती है जितनी चाहिए --- $\kappa$
प्रति इकाई \omterm{def:b2:curves:length}{लंबाई} किसी द्वितीय अवकलज की तरह मापित होता है, $\tau$
प्रति वर्ग क्षेत्रफल तीन अवकलजों के मिश्रित आयतन की तरह --- और
\emph{इसीलिए} वे ज्यामितीय हैं जबकि $\gamma''$ स्वयं नहीं।
\end{example}

\begin{remark}[अवकाश-वक्रों के लिए आधारभूत प्रमेय]
समतल की तरह, $\kappa > 0$ वाला युग्म $(\kappa, \tau)$ किसी द्विनियमित चाप को
$\R^3$ के सीधे समदूरिक प्रतिचित्रण तक निर्धारित कर देता है: फ्रेने
सूत्र ढाँचे $(T, N, B)$ के लिए कोई रैखिक अवकल निकाय बनाते हैं,
जिस पर \cref{ch:b2:diffeq} का कोशी--लिप्शिट्स सिद्धांत लागू होता है;
हल-ढाँचे का प्रसामान्य लांबिक होना सुरक्षित रहता है क्योंकि गुणांक
आव्यूह प्रतिसममित है (वही ग्राम-आव्यूह तर्क जो \cref{exo:b2:curves:7} में है),
और वक्र $T$ के समाकलन से पुनः प्राप्त हो जाता है। विवरण हम
पाठक के लिए एक पर्याप्त पर शिक्षाप्रद अभ्यास के रूप में छोड़ देते हैं।
\end{remark}

\section{स्थानीय अध्ययन: स्पर्श रेखा के सापेक्ष स्थिति}

\begin{proposition}[नियमित बिंदु पर स्थानीय
आकार]\label{prop:b2:curves:local}
मान लीजिए $\gamma$ $t_0$ पर वर्ग $\mathcal{C}^k$ का समतल चाप है, जहाँ
$p$ वह सबसे छोटा सूचकांक है जिसके लिए $\gamma^{(p)}(t_0) \neq 0$, और $q$ वह
सबसे छोटा सूचकांक $> p$ जिसके लिए $\gamma^{(q)}(t_0)$ $\gamma^{(p)}(t_0)$ के संरेख
नहीं है (मान लीजिए दोनों विद्यमान हैं, $q \leq k$)। $\gamma(t_0)$ पर
केंद्रित आधार $(u, v) = (\gamma^{(p)}(t_0), \gamma^{(q)}(t_0))$ में टेलर--युंग निर्देशांक
देता है
\[
X(t) \sim \frac{(t - t_0)^p}{p!},
\qquad
Y(t) \sim \frac{(t - t_0)^q}{q!} .
\]
स्थानीय चित्र केवल $p$ तथा $q$ की सम-विषमता पर निर्भर करता है:
\begin{center}
\begin{tabular}{c c l}
$p$ विषम, $q$ सम & \emph{साधारण बिंदु} & वक्र किसी रेखा को नहीं
काटता, \omterm{def:b2:curves:arc}{स्पर्श रेखा} के एक ही ओर\\
$p$ विषम, $q$ विषम & \emph{नति-परिवर्तन बिंदु} & वक्र अपनी
\omterm{def:b2:curves:arc}{स्पर्श रेखा} को काटता है\\
$p$ सम, $q$ विषम & \emph{प्रथम प्रकार का उभयाग्र} & दोनों शाखाएँ
\omterm{def:b2:curves:arc}{स्पर्श रेखा} के विपरीत ओर\\
$p$ सम, $q$ सम & \emph{द्वितीय प्रकार का उभयाग्र} & दोनों
शाखाएँ एक ही ओर\\
\end{tabular}
\end{center}
\end{proposition}

\begin{proof}
कोटि $q$ पर टेलर--युंग (फलन $\gamma$ $t_0$ के निकट
$\mathcal{C}^q$ है):
\[
\gamma(t) - \gamma(t_0)
= \sum_{j=p}^{q} \frac{(t-t_0)^j}{j!}\,\gamma^{(j)}(t_0)
+ o\bigl((t-t_0)^q\bigr).
\]
$p$ तथा $q$ के चुनाव से, $p
\leq j < q$ वाला हर $\gamma^{(j)}(t_0)$ $u$ के संरेख है; आधार $(u, v)$ में घटक इकट्ठे करने पर: $X(t) = \frac{(t-t_0)^p}{p!}(1 + o(1))$ और
$Y(t) = \frac{(t-t_0)^q}{q!}(1 + o(1))$। $t \gtrless t_0$ के लिए $X$ तथा
$Y$ की चिह्न-सारणी --- जिसे ठीक-ठीक सम-विषमता चलाती है ---
चारों चित्र दे देती है: उदाहरण के लिए यदि $p$ सम हो तो दोनों ओर
$X > 0$ (दोनों शाखाएँ दिशा $+u$ में निकलती हैं: उभयाग्र),
और \omterm{def:b2:curves:arc}{स्पर्श रेखा} का पक्ष ($\operatorname{sign} Y$) $q$ के विषम होने
पर पलट जाता है।
\end{proof}

\begin{example}
$t_0 = 0$ पर $\gamma(t) = (t^2, t^3)$ के लिए
(\cref{ex:b2:curves:cusp}): $\gamma'' (0)= (2, 0)$, $\gamma'''(0) =
(0, 6)$, अतः $p = 2$, $q = 3$: प्रथम प्रकार का उभयाग्र,
अर्धघनीय परवलय का जाना-पहचाना चित्र। $0$ पर $\gamma(t) = (t,
t^3)$ के लिए: $p = 1$, $q = 3$: नति-परिवर्तन --- घन वक्र अपनी
\omterm{def:b2:curves:arc}{स्पर्श रेखा} को काटता है।
\end{example}

\begin{remark}[इस खंड के भीतर के परिप्रेक्ष्य]
वक्र आगे के अध्यायों को तीन तरह से पोसते हैं। किसी पृष्ठ पर खींचे
जाने पर वे उसके स्पर्श समतल तथा उसका प्रथम आधारभूत रूप
(\cref{ch:b2:surfaces}) परिभाषित करते हैं, और उनकी लंबाइयाँ परिवेशी
दूरिक को प्रतिबंधित करके परिकलित होती हैं --- आगे वाला अध्याय काफ़ी
हद तक यही अध्याय है, सापेक्षित। सप्ताहांत समस्या का अन्वालोप-कलन
\cref{ch:b2:multint} में द्विक समाकलों से मिलता है, जहाँ \omterm{pb:b2:curves:1}{ऐस्ट्रॉयड} का
क्षेत्रफल ग्रीन के सूत्र (\cref{exo:b2:multint:5}) से फिर से परिकलित
होता है --- एक वक्र, दो सिद्धांत, मिलते-जुलते उत्तर। और फ्रेने निकाय
ने \cref{ch:b2:diffeq} के रैखिक अवकल समीकरणों का उपयोग पहले ही कर लिया
(अस्तित्व, अद्वितीयता, तथा \cref{exo:b2:curves:7} का लांबिकता-संरक्षण
तर्क): वक्रों की आधारभूत प्रमेय ज्यामितीय वस्त्र पहने हुए एक अवकल
समीकरण प्रमेय है।
\end{remark}

\begin{remark}[विधि: स्थानीय अध्ययन चलाना]
व्यवहार में यह वर्गीकरण चार पगों की दिनचर्या है।
\emph{एक}, $t_0$ पर तब तक अवकलन कीजिए जब तक पहला अशून्य
अवकलज न आ जाए: उसका सूचकांक $p$ है, उसका मान सदिश
$u$। \emph{दो}, तब तक अवकलन करते रहिए जब तक $u$ के संरेख
न होने वाला कोई अवकलज न आ जाए: सूचकांक $q$, सदिश $v$।
\emph{तीन}, सारणी में सम-विषमता $(p, q)$ पढ़िए।
\emph{चार}, चित्र खींचिए: यदि $p$ विषम हो तो वक्र $+u$ के
अनुदिश निकलता है (और यदि $p$ सम हो तो $u$ के अनुदिश और फिर
$u$ के अनुदिश वापस), $v$ के उस पक्ष पर जो $Y$ के चिह्न
से तय होता है। दो चेतावनियाँ। ढाँचा
$(u, v)$ सामान्यतः प्रसामान्य लांबिक \emph{नहीं} होता --- सारणी
\omterm{def:b2:curves:arc}{स्पर्श रेखा} के सापेक्ष स्थितियाँ बताती है, कोण या दूरियाँ नहीं, अतः
चित्र से \omterm{thm:b2:curves:frenet2d}{वक्रता} मत पढ़िए। और कोटि $p$ तथा $q$ के बीच $u$
के संरेख मध्यवर्ती अवकलज अनुमत हैं (वे केवल $X$ का प्रसार
खिसकाते हैं); जो \emph{नहीं} होना चाहिए वह है पहले अशून्य अवकलज पर
रुककर $q = p + 1$ का अनुमान लगा लेना: $\gamma(t)
= (t^2, t^4 + t^5)$ के लिए भोला अनुमान $q = 3$ ग़लत है, क्योंकि
$\gamma^{(3)}(0)$ अब भी $\gamma''(0)$ के संरेख है ---
और यही ठीक \cref{exo:b2:curves:5} है।
\end{remark}

\begin{problem}[{सप्ताहांत समस्या --- अन्वालोप यंत्र और
चार शास्त्रीय वक्र}]\label{pb:b2:curves:1}
रेखाओं का एकप्राचलीय कुल प्रायः समतल को \omterm{def:b2:funcseq:def}{एकसमान रूप से} नहीं ढँक
पाता: रेखाएँ किसी ऐसे वक्र के अनुदिश ढेर हो जाती हैं जो उन सबको
स्पर्श करता है --- उनका \emph{अन्वालोप}\index{अन्वालोप}। प्रकाश-किरणें
अन्वालोपों को \emph{दाहक वक्रों} के रूप में दिखा देती हैं --- कॉफ़ी के
प्याले में दिखने वाला चमकीला उभयाग्री वक्र। यह समस्या पहले व्यापक
\omterm{pb:b2:curves:1}{अन्वालोप} यंत्र बनाती है, फिर उसे चार बार चलाती है: सरकती सीढ़ी
(\omterm{pb:b2:curves:1}{ऐस्ट्रॉयड}), परवलय तथा चक्रज के अभिलंब (विकसिताएँ, और अंत में हाइगेंस
का लोलक), और कॉफ़ी-प्याले का दाहक वक्र (\omterm{pb:b2:curves:1}{नेफ्रॉयड})। सर्वत्र $D_t$
समीकरण $a(t)\,x + b(t)\,y = c(t)$ वाली रेखा को सूचित करता है, जहाँ $a, b, c$
ऐसे $\mathcal C^2$ फलन हैं कि $(a(t), b(t)) \neq (0,0)$,
और $\Delta(t) = a(t)b'(t) - a'(t)b(t)$।

\medskip
\noindent\textbf{भाग I --- \omterm{pb:b2:curves:1}{अन्वालोप} यंत्र।}
\begin{enumerate}
  \item मान लीजिए $\Delta(t) \neq 0$। दिखाइए कि
        \emph{अभिलक्षणिक निकाय}
        \[
        \begin{cases} a(t)\,x + b(t)\,y = c(t)\\
        a'(t)\,x + b'(t)\,y = c'(t)\end{cases}
        \]
        का अद्वितीय हल $E(t) = (x(t), y(t))$ है, जो
        $x = \dfrac{cb' - c'b}{\Delta}$, $y = \dfrac{ac' -
        a'c}{\Delta}$ से दिया जाता है।
  \item इसके अतिरिक्त मान लीजिए $E$ $t$ के निकट $\mathcal C^1$ है,
        जहाँ $E'(t) \neq 0$। निकाय के पहले समीकरण का
        अवकलन करके दिखाइए कि $a(t)\,x'(t) +
        b(t)\,y'(t) = 0$, और निष्कर्ष निकालिए कि वक्र $E$
        $D_t$ के किसी बिंदु से \emph{$D_t$ की दिशा के
        साथ} होकर जाता है: अर्थात् कुल $E$ को स्पर्श करता है,
        जिसे उसका \emph{\omterm{pb:b2:curves:1}{अन्वालोप}} कहते हैं।
  \item जाँच: बिंदुओं $(t, t^2/2)$ पर परवलय $y =
        x^2/2$ की स्पर्श रेखाएँ $tx - y =
        t^2/2$ हैं। सत्यापित कीजिए कि \omterm{pb:b2:curves:1}{अन्वालोप}
        यंत्र परवलय को ही लौटाता है।
  \item (अभिलंबों का \omterm{pb:b2:curves:1}{अन्वालोप}) मान लीजिए $\gamma$ इकाई-चाल का है,
        जहाँ $\kappa(s) \neq 0$। $\gamma(s)$ पर अभिलंब रेखा
        $\{M : \langle M - \gamma(s), T(s)
        \rangle = 0\}$ है। दिखाइए कि उसका अभिलक्षणिक निकाय
        $\langle M - \gamma(s), N(s)\rangle =
        1/\kappa(s)$ को बाध्य कर देता है, अतः अभिलक्षणिक बिंदु
        \omterm{def:b2:curves:curvature}{वक्रता केंद्र} है: \emph{अभिलंबों का \omterm{pb:b2:curves:1}{अन्वालोप} विकसिता है},
        जिससे \cref{exo:b2:curves:6} पुनः प्राप्त होती है। $\Delta(s) =
        \kappa(s)$ जाँचिए।
  \item दो अपभ्रंश। किरणपुंज $D_\theta : x\cos
        \theta + y\sin\theta = 0$ के लिए दिखाइए कि हर $\theta$
        के लिए अभिलक्षणिक बिंदु मूल बिंदु है (``\omterm{pb:b2:curves:1}{अन्वालोप}'' एक
        बिंदु में सिमट जाता है, और $E' = 0$: प्रश्न 2 लागू नहीं
        होता)। समांतर रेखाओं के किसी कुल ($a, b$ अचर) के लिए
        दिखाइए कि $\Delta \equiv 0$ और अभिलक्षणिक निकाय सामान्यतः
        असंगत है: कोई \omterm{pb:b2:curves:1}{अन्वालोप} नहीं।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- सरकती सीढ़ी और
\omterm{pb:b2:curves:1}{ऐस्ट्रॉयड}।} \omterm{def:b2:curves:length}{लंबाई} $1$ का कोई खंड इस तरह सरकता है कि उसका एक
सिरा $P_t =
(\cos t, 0)$ फ़र्श पर और दूसरा $Q_t = (0, \sin t)$ दीवार पर
रहे, $t \in \intoo0{\pi/2}$।
\begin{enumerate}[resume]
  \item दिखाइए कि रेखा $(P_tQ_t)$ का समीकरण $x\sin t +
        y\cos t = \sin t\cos t$ है, और \omterm{pb:b2:curves:1}{अन्वालोप} यंत्र अभिलक्षणिक बिंदु
        \[
        E(t) = (\cos^3 t,\ \sin^3 t) :
        \]
        देता है --- \emph{ऐस्ट्रॉयड}\index{ऐस्ट्रॉयड}, जिसका
        अंतर्निहित समीकरण $x^{2/3} + y^{2/3} = 1$ है (सममिति से
        अन्य चतुर्थांशों तक विस्तारित)।
  \item दिखाइए कि $E'(0) = 0$, और स्थानीय
        वर्गीकरण (\cref{prop:b2:curves:local}) का उपयोग करते हुए कि
        \omterm{pb:b2:curves:1}{ऐस्ट्रॉयड} का $(1, 0)$ पर प्रथम प्रकार का उभयाग्र है
        --- और इसी प्रकार उसके चारों अक्ष-बिंदुओं पर भी।
  \item $\norm{E'(t)} = \tfrac32\abs{\sin 2t}$ परिकलित कीजिए और
        निष्कर्ष निकालिए कि \omterm{pb:b2:curves:1}{ऐस्ट्रॉयड} की कुल \omterm{def:b2:curves:length}{लंबाई} $6$ है।
  \item सीढ़ी \omterm{pb:b2:curves:1}{ऐस्ट्रॉयड} को कहाँ छूती है? दिखाइए कि $E(t) =
        P_t + \sin^2 t\,(Q_t - P_t)$: स्पर्श बिंदु
        सीढ़ी को $\sin^2 t : \cos^2
        t$ अनुपात में बाँटता है, और सीढ़ी के सरकने के साथ
        उसे एक सिरे से दूसरे सिरे तक बुहार देता है।
  \item \omterm{pb:b2:curves:1}{ऐस्ट्रॉयड} से घिरा क्षेत्रफल परिकलित कीजिए: दिखाइए कि
        पहले चतुर्थांश का क्षेत्रफल $3\int_0^{\pi/2}\sin^4
        t\cos^2 t\,\dd t$ है, समाकल को
        रैखिकीकरण ($\sin^2 2t = \tfrac{1 - \cos 4t}2$) से मूल्यांकित कीजिए,
        और निष्कर्ष निकालिए कि कुल क्षेत्रफल $3\pi/8$ है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- परवलय की विकसिता।}
मान लीजिए $\gamma(t) = (t, t^2/2)$।
\begin{enumerate}[resume]
  \item दिखाइए कि $\gamma(t)$ पर अभिलंब रेखा का समीकरण
        $x + t\,y = t + t^3/2$ है।
  \item \omterm{pb:b2:curves:1}{अन्वालोप} यंत्र चलाइए: दिखाइए कि अभिलंबों का \omterm{pb:b2:curves:1}{अन्वालोप}
        \[
        E(t) = \Bigl(-t^3,\ 1 + \tfrac32 t^2\Bigr),
        \]
        है, जिसका अंतर्निहित समीकरण $x^2 = \tfrac8{27}(y - 1)^3$ है:
        यानी कोई अर्धघनीय परवलय।
  \item प्रश्न 4 से जाँच कीजिए: \omterm{def:b2:curves:curvature}{वक्रता केंद्र} $\gamma(t) + \frac1{\kappa(t)}N(t)$ को
        $\kappa(t) = (1 + t^2)^{-3/2}$
        से परिकलित कीजिए (\cref{ex:b2:curves:kappaexamples}) और वही बिंदु
        पुनः प्राप्त कीजिए।
  \item दिखाइए कि विकसिता का $(0, 1)$ पर प्रथम प्रकार का उभयाग्र
        है, जो शीर्ष पर \omterm{def:b2:curves:curvature}{वक्रता केंद्र} है --- वही बिंदु जहाँ
        $\kappa$ चरम है, जैसा \cref{exo:b2:curves:6} के सूत्र $c' = -\frac{\kappa'}{\kappa^2}N$ से
        अनुमानित है।
  \item परवलय के कितने अभिलंब किसी दिए गए बिंदु $(x_0, y_0)$ से
        होकर जाते हैं? दिखाइए कि उत्तर घन $\tfrac{t^3}2 + (1 - y_0)\,t - x_0 = 0$ से तय
        होता है; अक्ष की स्थिति $x_0 = 0$ को पूरी तरह निपटाइए
        ($y_0 < 1$ के लिए एक अभिलंब, $y_0 > 1$ के लिए तीन), और
        विकसिता की संक्रमण वक्र के रूप में व्याख्या कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- कॉफ़ी-प्याले का दाहक वक्र।} दिशा
$(1, 0)$ की समांतर किरणें दर्पण-वृत्त $x^2 + y^2 = 1$ के भीतरी भाग से
टकराती हैं; $P_\theta =
(\cos\theta, \sin\theta)$ पर पड़ने वाली किरण परावर्तन के नियम के अनुसार परावर्तित
होती है।
\begin{enumerate}[resume]
  \item अभिलंब (अर्थात् त्रिज्या) में दर्पण-सममिति से यह
        न्यायसंगत ठहराइए कि परावर्तित दिशा $v = u -
        2\langle u, n\rangle n$ है, जहाँ $u = (1,0)$, $n =
        (\cos\theta, \sin\theta)$, और $v =
        -(\cos2\theta, \sin2\theta)$ परिकलित कीजिए।
  \item दिखाइए कि परावर्तित किरण रेखा
        \[
        x\sin 2\theta - y\cos 2\theta = \sin\theta .
        \]
        पर पड़ती है।
  \item \omterm{pb:b2:curves:1}{अन्वालोप} यंत्र चलाइए ($\Delta = 2$): दिखाइए कि
        दाहक वक्र
        \[
        E(\theta) = \Bigl(\tfrac{3\cos\theta -
        \cos3\theta}4,\ \tfrac{3\sin\theta -
        \sin3\theta}4\Bigr),
        \]
        है --- \emph{नेफ्रॉयड}\index{नेफ्रॉयड}।
  \item $E'(\theta) = \tfrac32\sin\theta\,
        (\cos2\theta, \sin2\theta)$ परिकलित कीजिए; जाँचिए कि स्पर्शी
        दिशा परावर्तित किरण की दिशा है (प्रश्न 16), दोनों
        उभयाग्र $(\pm\tfrac12, 0)$ ढूँढ़िए, और दिखाइए कि परावर्तित
        किरण अक्ष $y = 0$ को $x = \frac1{2\cos\theta}$ पर काटती है ---
        अतः लगभग अक्षीय किरणें $x = \tfrac12$ पर फोकस होती हैं:
        त्रिज्या $R$ के दर्पण की फोकस दूरी $R/2$।
  \item दिखाइए कि \omterm{pb:b2:curves:1}{नेफ्रॉयड} की कुल \omterm{def:b2:curves:length}{लंबाई} $6$ है और $\theta = 0$ के
        निकट
        \[
        E(\theta) - \bigl(\tfrac12, 0\bigr) =
        \bigl(\tfrac34\theta^2 + o(\theta^2),\ \theta^3 +
        o(\theta^3)\bigr) :
        \]
        अर्थात् अक्ष के अनुदिश संकेत करता प्रथम प्रकार का उभयाग्र।
  \item एक अनुच्छेद में समझाइए कि दाहक वक्र चमकीला क्यों होता
        है: उसके ठीक बाहर के हर बिंदु से दो परावर्तित किरणें
        गुज़रती हैं, और उस पर पड़ने वाले हर बिंदु पर किरणें
        ``अपरिमित रूप से संकेंद्रित'' होती हैं (प्रतिचित्रण $(\theta,
        \text{किरण के अनुदिश दूरी}) \mapsto \R^2$ का क्रांतिक
        बिंदु ठीक \omterm{pb:b2:curves:1}{अन्वालोप} पर होता है)।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- हाइगेंस: चक्रज अपनी ही
विकसिता है।} मान लीजिए $\gamma(t) = (t - \sin t,\ 1 - \cos t)$, $t \in
\intoo0{2\pi}$, जो चक्रज का एक मेहराब है।
\begin{enumerate}[resume]
  \item $\kappa(t) = -\dfrac1{4\sin(t/2)}$ तथा \omterm{def:b2:curves:curvature}{वक्रता केंद्र}
        परिकलित कीजिए; दिखाइए कि विकसिता
        \[
        c(t) = (t + \sin t,\ \cos t - 1),
        \]
        है और प्रतिस्थापन $t = u + \pi$ उसे $(\pi, -2)$ से
        स्थानांतरित मूल चक्रज के रूप में प्रकट कर देता है:
        \emph{चक्रज की विकसिता कोई सर्वांगसम चक्रज होती है}
        (हाइगेंस)।
  \item सत्यापित कीजिए कि शीर्ष $t =
        \pi$ पर \omterm{def:b2:curves:curvature}{वक्रता त्रिज्या} $4$ के बराबर है, जो एक मेहराब की \omterm{def:b2:curves:length}{लंबाई} $8$ की
        आधी है (\cref{exo:b2:curves:1}); विकसिता का उभयाग्र सीधे
        शीर्ष के नीचे, $4$ दूरी पर, ढूँढ़िए।
  \item (डोरी गुणधर्म) मान लीजिए $\gamma$ इकाई-चाल का है, जहाँ
        $\kappa > 0$, $\kappa$ वर्ग $\mathcal C^1$ का है और
        $R = 1/\kappa$ दृढ़ता से एकदिष्ट है। $c' = R'N$ का उपयोग करते हुए
        दिखाइए कि $c(s_0)$ और $c(s_1)$ के बीच विकसिता की \omterm{def:b2:curves:length}{चाप-लंबाई}
        $\abs{R(s_1) - R(s_0)}$ है।
        व्याख्या: विकसिता से खोली गई कोई तनी हुई डोरी, जिसकी
        आरंभिक \omterm{def:b2:curves:length}{लंबाई} $R(s_0)$ है, अपने स्वतंत्र सिरे से मूल वक्र
        खींच देती है --- अतः हाइगेंस की घड़ी के दो चक्रजीय गालों
        के बीच झूलता लोलक कोई चक्रज ही बनाता है।
  \item संश्लेषण। भाग I के यंत्र ने \omterm{pb:b2:curves:1}{ऐस्ट्रॉयड}, कोई अर्धघनीय
        परवलय, कोई \omterm{pb:b2:curves:1}{नेफ्रॉयड} और कोई चक्रज उत्पन्न किए। इन चारों
        कुलों में से हर एक के लिए एक वाक्य में बताइए कि
        परिकल्पनाएँ $\Delta \neq 0$ तथा $E' \neq 0$ कहाँ टिकीं या टूटीं, और
        $E' \neq 0$ की हर विफलता ने किस ज्यामितीय घटना (उभयाग्र,
        फोकस, अपभ्रंश) का संकेत दिया। अभिलंबों के \omterm{pb:b2:curves:1}{अन्वालोप} पर
        \omterm{thm:b2:curves:frenet2d}{वक्रता} के चरम कहाँ प्रकट होने ही चाहिए, और क्यों?
\end{enumerate}
\end{problem}
