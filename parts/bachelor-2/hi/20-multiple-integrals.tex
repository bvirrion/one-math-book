\chapter{रेखा समाकल और बहुक समाकल}\label{ch:b2:multint}

यह अध्याय समाकलन को अंतरालों से आगे वक्रों तक तथा समतल और अवकाश के
प्रांतों तक फैलाता है। \omterm{def:b2:multint:lineint}{रेखा समाकल} किसी अभिविन्यस्त चाप के अनुदिश
किसी \emph{\omterm{def:b2:multint:lineint}{अवकल रूप}} $P\,\dd x + Q\,\dd y$ का समाकलन करते हैं; द्विक तथा त्रिक समाकल
द्वि- और त्रिविमीय प्रांतों पर फलनों का समाकलन करते हैं। दोनों
सिद्धांत \emph{ग्रीन--रीमान प्रमेय} में मिलते हैं, जो कलन की
द्विविमीय आधारभूत प्रमेय है; और सर्वत्र मुख्य परिकलन-औज़ार
\emph{चर-परिवर्तन सूत्र} है, जिसका विरूपण गुणक याकोबी \omterm{def:b2:linalg:det}{सारणिक} का
निरपेक्ष मान है।

\section{रेखा समाकल}

\begin{definition}[अवकल रूप; रेखा समाकल]\label{def:b2:multint:lineint}
मान लीजिए $U \subseteq \R^2$ \omterm{def:b2:metric:topology}{विवृत} है। $U$ पर घात $1$ तथा वर्ग $\mathcal{C}^0$ का
\emph{अवकल रूप}\index{अवकल रूप} कोई व्यंजक $\omega = P\,\dd x + Q\,\dd y$ है, जहाँ $P, Q \colon U \to \R$ \omterm{def:b2:metric:continuity}{संतत}
हैं --- औपचारिक रूप से, $U$ से $\R^2$ के \omterm{def:b2:linalg:dual}{द्वैत} $\omega(M) = P(M)\,e_1^* + Q(M)\,e_2^*$ में कोई \omterm{def:b2:metric:continuity}{संतत}
प्रतिचित्रण। किसी $\mathcal{C}^1$ चाप $\gamma \colon [a, b] \to U$, $\gamma(t) =
(x(t), y(t))$, के लिए $\gamma$ के अनुदिश
$\omega$ का \emph{रेखा समाकल}\index{रेखा समाकल} है
\[
\int_\gamma \omega
= \int_a^b \Bigl(P(\gamma(t))\,x'(t)
+ Q(\gamma(t))\,y'(t)\Bigr)\,\dd t .
\]
परिभाषाएँ अक्षरशः $\R^3$ तक (रूप $P\,\dd x +
Q\,\dd y + R\,\dd z$) तथा खंडशः $\mathcal{C}^1$ चापों तक
(टुकड़ों पर योग) फैल जाती हैं।
\end{definition}

\begin{proposition}[अपरिवर्त्यता और अभिविन्यास]\label{prop:b2:multint:lineinv}
\omterm{def:b2:multint:lineint}{रेखा समाकल} किसी वर्धमान $\mathcal{C}^1$ \omterm{def:b2:curves:reparam}{प्राचल परिवर्तन} के अंतर्गत अपरिवर्तित
रहता है, और ह्रासमान परिवर्तन के अंतर्गत चिह्न बदल देता है। इसलिए
वह केवल \emph{अभिविन्यस्त} \omterm{def:b2:curves:reparam}{ज्यामितीय चाप} पर ही निर्भर करता है।
\end{proposition}

\begin{proof}
यदि $\theta \colon [c, d] \to [a, b]$ कोई \omterm{def:b2:curves:reparam}{प्राचल परिवर्तन} हो और $\tilde\gamma = \gamma \circ \theta$, तो शृंखला नियम तथा
एकचर चर-परिवर्तन $t = \theta(u)$ से
\[
\int_{\tilde\gamma}\omega
= \int_c^d \bigl(P(\gamma(\theta(u)))\,x'(\theta(u))
+ Q(\gamma(\theta(u)))\,y'(\theta(u))\bigr)\,\theta'(u)\,\dd u
= \pm\int_a^b \bigl(Px' + Qy'\bigr)(t)\,\dd t ,
\]
जहाँ $\theta$ के वर्धमान होने पर चिह्न $+$ है ($\theta(c) = a$) और ह्रासमान होने
पर $-$ (सीमाएँ अदल-बदल जाती हैं)।
\end{proof}

\begin{example}[किसी बल का कार्य; परिसंचरण]\label{ex:b2:multint:work}
यदि $F = (P, Q)$ कोई बल क्षेत्र हो, तो $\int_\gamma P\dd x + Q\dd y =
\int_a^b \langle F(\gamma(t)), \gamma'(t)\rangle\,\dd t$ $\gamma$ के अनुदिश $F$ का
\emph{कार्य} है। वामावर्त इकाई वृत्त $\gamma(t) =
(\cos t, \sin t)$ के अनुदिश $\omega = -y\,\dd x +
x\,\dd y$ के लिए:
\[
\int_\gamma \omega
= \int_0^{2\pi}\bigl((-\sin t)(-\sin t)
+ \cos t\cos t\bigr)\,\dd t = 2\pi ,
\]
जो घिरे हुए \omterm{def:b2:multint:domain}{क्षेत्रफल} का दुगुना है --- ग्रीन--रीमान का पहला संकेत।
\end{example}

\begin{example}[एक समाकल, दो प्राचलन, एक
चिह्न-जाल]\label{ex:b2:multint:semicircle}
$(1, 0)$ से $(-1, 0)$ तक ऊपरी इकाई अर्धवृत्त के अनुदिश $\int_\gamma x\,\dd y$ परिकलित कीजिए।
$\gamma(t) =
(\cos t, \sin t)$, $t \in \intcc0\pi$ के साथ:
\[
\int_0^\pi\cos t\cdot\cos t\,\dd t = \frac\pi2 .
\]
ग्राफ़-प्राचलन $x \mapsto (x, \sqrt{1 -
x^2})$ के साथ, जहाँ $x$ $1$ से $-1$ तक जाता है (दिशा
पर ध्यान दीजिए!):
\[
\int_1^{-1}x\cdot\frac{-x}{\sqrt{1 - x^2}}\,\dd x
= \int_{-1}^{1}\frac{x^2}{\sqrt{1 - x^2}}\,\dd x
= \frac\pi2
\]
($x = \sin u$ उसे किसी \omterm{pb:b2:multint:1}{वालिस समाकल} तक सिमटा देता है)। वही मान, जैसा \cref{prop:b2:multint:lineinv}
आश्वस्त करती है --- पर केवल इसलिए कि दोनों बार यात्रा $(1,0)$ से $(-1,0)$
तक हुई; यात्रा उलटने पर चिह्न पलट जाता है। पथ को $x$-अक्ष के
अनुदिश बंद करने से (जहाँ $\dd y = 0$) कुछ नहीं जुड़ता, और कुल $\frac\pi2$
अर्धचक्रिका का \omterm{def:b2:multint:domain}{क्षेत्रफल} है: नीचे दिए ग्रीन--रीमान के
सीमा-क्षेत्रफल सूत्रों का पहला उदाहरण।
\end{example}

\begin{definition}[यथातथ और संवृत रूप]\label{def:b2:multint:exact}
वर्ग $\mathcal{C}^0$ का रूप $\omega = P\,\dd x + Q\,\dd y$ $U$ पर \emph{यथातथ}\index{यथातथ रूप} है यदि
कोई ऐसा $f \in \mathcal{C}^1(U)$ (कोई \emph{विभव}\index{विभव}) हो कि $\omega = \dd f$, अर्थात्
$P = f_x$ तथा $Q
= f_y$। कोई $\mathcal{C}^1$ रूप \emph{संवृत}\index{संवृत रूप} है यदि
$U$ पर $P_y = Q_x$ हो।
\end{definition}

\begin{theorem}[रेखा समाकलों के लिए आधारभूत
प्रमेय]\label{thm:b2:multint:ftc}
यदि $\omega = \dd f$ \omterm{def:b2:multint:exact}{यथातथ} हो और $\gamma$ $U$ में $A$ से $B$ तक कोई खंडशः $\mathcal{C}^1$ चाप
हो, तो
\[
\int_\gamma \omega = f(B) - f(A) .
\]
विशेष रूप से किसी \omterm{def:b2:multint:exact}{यथातथ रूप} का किसी भी संवृत चाप के अनुदिश समाकल
शून्य है, और हर \omterm{def:b2:multint:exact}{यथातथ} $\mathcal{C}^1$ रूप संवृत है।
\end{theorem}

\begin{proof}
शृंखला नियम (\cref{ch:b2:diffcalc}) से $\frac{\dd}{\dd t}f(\gamma(t)) = f_x(\gamma(t))x'(t) +
f_y(\gamma(t))y'(t)$, अतः \cref{def:b2:multint:lineint} का समाकल्य $t \mapsto
f(\gamma(t))$ का अवकलज है, और
कलन की आधारभूत प्रमेय हर टुकड़े पर परिणाम दे देती है; मध्यवर्ती मान
दूरबीन की तरह सिमट जाते हैं। \omterm{def:b2:multint:exact}{यथातथ} $\mathcal{C}^1$ रूपों की संवृतता श्वार्ज़
की प्रमेय है: $P_y =
f_{xy} = f_{yx} = Q_x$।
\end{proof}

\begin{example}[किसी विभव का पुनर्निर्माण]\label{ex:b2:multint:potential}
मान लीजिए $\R^2$ पर $\omega = y\,\eu^{xy}\,\dd x + (x\,\eu^{xy} + 2y)\,\dd y$। वह संवृत है: दोनों मिश्रित अवकलज $\eu^{xy}(1 + xy)$ के
बराबर हैं। कोई \omterm{def:b2:multint:exact}{विभव} ढूँढ़ने के लिए $y$ स्थिर रखकर $P$ का $x$ में
समाकलन कीजिए:
\[
f(x, y) = \int y\,\eu^{xy}\,\dd x = \eu^{xy} + c(y),
\]
फिर $f_y$ से मिलान करके $c$ समंजित कीजिए: $x\,\eu^{xy} + c'(y) =
x\,\eu^{xy} + 2y$ $c(y) = y^2$ देता है। अतः
$f(x,y) = \eu^{xy} +
y^2$, और $(0,0)$ से $(1,1)$ तक किसी भी खंडशः $\mathcal C^1$ चाप के लिए
\[
\int_\gamma\omega = f(1,1) - f(0,0) = (\eu + 1) - 1 = \eu ,
\]
जो पथ पर निर्भर नहीं --- दो-पग वाला नुस्ख़ा ($x$ में समाकलन कीजिए,
$y$ में सुधार कीजिए) उन प्रांतों पर \cref{thm:b2:multint:ftc} का व्यावहारिक विलोम है
जहाँ \omterm{def:b2:multint:exact}{संवृत रूप} \omterm{def:b2:multint:exact}{यथातथ} होते हैं।
\end{example}

\begin{example}[संवृत होने से यथातथ होना नहीं निकलता]\label{ex:b2:multint:angleform}
$U = \R^2 \setminus \{0\}$ पर \emph{कोण रूप}
\[
\omega = \frac{-y\,\dd x + x\,\dd y}{x^2 + y^2}
\]
संवृत है (सीधा परिकलन: $P_y$ तथा $Q_x$ दोनों $\frac{y^2 - x^2}{(x^2+y^2)^2}$ के बराबर हैं), पर
इकाई वृत्त के अनुदिश उसका समाकल $2\pi \neq 0$ है (वही परिकलन जो \cref{ex:b2:multint:work} में है,
$1$ से भाग देकर): अर्थात् $\omega$ $U$ पर \omterm{def:b2:multint:exact}{यथातथ} नहीं है। स्थानीय रूप
से ध्रुवीय कोण के किसी निर्धारण $\theta$ के लिए $\omega = \dd\theta$; विफलता वैश्विक
है --- छिद्र के चारों ओर कोण को \omterm{def:b2:metric:continuity}{संतत} रूप से परिभाषित नहीं किया जा
सकता। बिना छेद वाले प्रांतों पर यह रोग ग़ायब हो जाता है: किसी
\emph{तारकाकार} \omterm{def:b2:metric:topology}{विवृत} समुच्चय पर हर संवृत $\mathcal{C}^1$ रूप \omterm{def:b2:multint:exact}{यथातथ} होता है
(प्वांकारे की प्रमेयिका, \cref{exo:b2:multint:8})।
\end{example}

\section{द्विक समाकल}

हम एक चर में रीमान समाकलन का सिद्धांत मान लेते हैं (प्रथम वर्ष का
खंड, तथा \cref{ch:b2:integration}) और उसके दो-चर रूप का रेखाचित्र खींचते हैं। आयत $R = [a, b] \times [c, d]$
पर \omterm{def:b2:metric:continuity}{संतत} कोई फलन $f$ द्विक समाकल $\iint_R f$ रखता है, जो जालियों पर रीमान
योगों से ठीक वैसे ही परिभाषित होता है जैसे एक चर में, और पुनरावृत्ति
से परिकलित होता है:

\begin{theorem}[आयत पर फ़ूबिनी]\label{thm:b2:multint:fubini}
$R = [a,b] \times [c,d]$ पर \omterm{def:b2:metric:continuity}{संतत} $f$ के लिए,
\[
\iint_R f
= \int_a^b \Bigl(\int_c^d f(x, y)\,\dd y\Bigr)\dd x
= \int_c^d \Bigl(\int_a^b f(x, y)\,\dd x\Bigr)\dd y .
\]
\end{theorem}

\begin{proof}
$F(x) = \int_c^d f(x, y)\,\dd y$ रखिए। \omterm{def:b2:metric:compact}{संहत} $R$ पर $f$ की एकसमान संततता $F$ को \omterm{def:b2:metric:continuity}{संतत} बना देती
है (प्रभावी आकलन: $\abs{F(x) - F(x')} \leq (d - c)\sup_y\abs{f(x,y) - f(x',y)}$)। अब $[a,b]$ तथा $[c,d]$ को $n$ बराबर भागों में
बाँटिए, जिससे \omterm{def:b2:multint:domain}{क्षेत्रफल} $\Delta x\,\Delta y$ वाली कोशिकाओं $R_{ij}$ की जाली बनती है।
हर कोशिका पर एकचर समाकल की एकदिष्टता से (दो बार लगाकर) $\inf_{R_{ij}} f \cdot \Delta x \Delta y \leq
\int_{x_{i-1}}^{x_i}\int_{y_{j-1}}^{y_j} f(x,y)\,\dd y\,\dd x \leq
\sup_{R_{ij}} f \cdot \Delta x \Delta y$।
कोशिकाओं पर जोड़ने पर पुनरावृत्त समाकल $\int_a^b F$ जाली के निचले और ऊपरी
रीमान योगों के बीच दब जाता है; और एकसमान संततता से $n \to
\infty$ होने पर
दोनों योग उस साझे मान तक अभिसरण करते हैं जो $\iint_R f$ को परिभाषित करता
है। वही तर्क $x$ तथा $y$ की भूमिकाएँ बदलकर भी लागू होता है, अतः
दोनों पुनरावृत्त समाकल $\iint_R f$ के बराबर हैं।
\end{proof}

\begin{remark}
फ़ूबिनी की उपपत्ति में \omterm{def:b2:metric:compact}{संहत} आयत पर संततता सचमुच काम कर रही है: वही
वह एकसमान संततता देती है जो रीमान योगों को दबाती है। अधिक जंगली
समाकल्यों के लिए कथन सचमुच विफल हो जाता है --- ऐसे फलन हैं जिनके
दोनों पुनरावृत्त समाकल विद्यमान हैं पर भिन्न हैं। ईमानदार व्यापक
प्रमेय, जिसकी एकमात्र परिकल्पना समाकलनीयता है, लेबेग समाकल के लिए
फ़ूबिनी की प्रमेय है, जो तृतीय वर्ष के खंड में सिद्ध होती है; इस
अध्याय की हर बात उसी \omterm{def:b2:metric:continuity}{संतत} परिवेश में रहती है जहाँ ऊपर की प्रारंभिक
उपपत्ति पूरी है।
\end{remark}

\begin{definition}[प्रारंभिक प्रांत]\label{def:b2:multint:domain}
कोई प्रांत $D \subseteq \R^2$ \emph{$y$-प्रारंभिक} है यदि
\[
D = \{(x, y) : a \leq x \leq b,\
\varphi_1(x) \leq y \leq \varphi_2(x)\}
\]
हो, जहाँ $\varphi_1 \leq \varphi_2$ $[a,b]$ पर \omterm{def:b2:metric:continuity}{संतत} हैं ($x$-प्रारंभिक: सममित रूप से)। किसी
$y$-प्रारंभिक $D$ पर \omterm{def:b2:metric:continuity}{संतत} $f$ के लिए
\[
\iint_D f
= \int_a^b\Bigl(
\int_{\varphi_1(x)}^{\varphi_2(x)} f(x,y)\,\dd y\Bigr)\dd x ,
\]
और यह जाँचा जाता है (किसी सन्निकटन तर्क से $f$ का विस्तार करके, या
उपविभाजन करके) कि जब $D$ दोनों दिशाओं में प्रारंभिक हो तब दोनों
पुनरावृत्त समाकल मेल खाते हैं। परिमित रूप से कई प्रारंभिक टुकड़ों
में कटे प्रांत योज्यता से साधे जाते हैं, और $D$ का
\emph{क्षेत्रफल}\index{क्षेत्रफल!समतल प्रांत का} $\operatorname
{Area}(D) = \iint_D 1$ है।
\end{definition}

\begin{example}\label{ex:b2:multint:triangle}
त्रिभुज $D = \{0 \leq x \leq 1,\ 0 \leq y \leq x\}$ पर:
\[
\iint_D xy \,\dd x\,\dd y
= \int_0^1 x\Bigl(\int_0^x y\,\dd y\Bigr)\dd x
= \int_0^1 x\cdot\frac{x^2}{2}\,\dd x = \frac18 .
\]
क्रम बदलने पर ($x$ $y$ से $1$ तक): $\int_0^1
y\bigl(\int_y^1 x\,\dd x\bigr)\dd y = \int_0^1
y\,\frac{1 - y^2}{2}\,\dd y = \frac18$ --- वही मान, दूसरा
परिकलन: समाकलन का क्रम ठीक चुनना आधी कारीगरी है।
\end{example}

\begin{example}[जब केवल एक ही क्रम चलता है]\label{ex:b2:multint:swaporder}
$I = \displaystyle\int_0^1\!\!\int_x^1
\eu^{y^2}\,\dd y\,\dd x$ परिकलित कीजिए। जैसा लिखा है, भीतरी समाकल $\int\eu^{y^2}\dd y$ का कोई प्रारंभिक
प्रतिअवकलज नहीं है: परिकलन अटक जाता है। पर प्रांत त्रिभुज $0 \leq
x \leq y \leq 1$ है,
जो दोनों दिशाओं में प्रारंभिक है; क्रम बदलने पर,
\[
I = \int_0^1\!\!\int_0^y \eu^{y^2}\,\dd x\,\dd y
= \int_0^1 y\,\eu^{y^2}\,\dd y
= \Bigl[\tfrac12\eu^{y^2}\Bigr]_0^1 = \frac{\eu - 1}{2} .
\]
भीतरी चर $x$ समाकल्य में कहीं आया ही नहीं, अतः उसका पहले समाकलन
ठीक वही गुणक $y$ दे देता है जो बाहरी समाकल को तत्काल बना देता है।
सार: फ़ूबिनी केवल पुनरावृत्ति की अनुमति नहीं है --- वह \emph{चुनने}
की अनुमति है, और सही क्रम किसी असंभव समाकल को एक पंक्ति में बदल
सकता है। आरंभ करने से पहले प्रांत का चित्र सदा खींचिए और दोनों
वर्णन पढ़ लीजिए।
\end{example}

\begin{theorem}[चर-परिवर्तन]\label{thm:b2:multint:changevar}
मान लीजिए $\Phi \colon U' \to U$ $\R^2$ के \omterm{def:b2:metric:topology}{विवृत} समुच्चयों के बीच कोई $\mathcal{C}^1$ अवकल
समाकृतिकता है, $K \subseteq U$ प्रारंभिक टुकड़ों में कटा कोई \omterm{def:b2:metric:compact}{संहत} प्रांत है
जहाँ $K' = \Phi^{-1}(K)$, और $f$ $K$ पर \omterm{def:b2:metric:continuity}{संतत} है। तब
\[
\iint_K f(x, y)\,\dd x\,\dd y
= \iint_{K'} f\bigl(\Phi(u, v)\bigr)\,
\abs{\det J_\Phi(u, v)}\,\dd u\,\dd v .
\]
\end{theorem}
\admitted

\begin{remark}
पूरी उपपत्ति --- किसी महीन जाली पर $\Phi$ को उसके \omterm{def:b2:diffcalc:differential}{अवकल} से सन्निकट
करना और सीमांत कोशिकाओं को साधना --- लंबी है, यद्यपि गहरी नहीं; वह
वर्ष 3 के माप सिद्धांत में लेबेग सिद्धांत के परिणाम के रूप में पूरी
की जाती है। अंतर्ज्ञान वही चित्र है जो पृष्ठ-क्षेत्रफल के लिए पहले
काम आया: $(u, v)$ पर भुजा $\dd u$ का कोई छोटा वर्ग प्रथम कोटि तक $\Phi_u\,\dd u$ तथा
$\Phi_v\,\dd v$ से \omterm{def:b2:structures:generated}{जनित} समांतर चतुर्भुज पर भेजा जाता है, जिसका \omterm{def:b2:multint:domain}{क्षेत्रफल}
$\abs{\det J_\Phi}\,\dd u\,\dd v$
है (\cref{lem:b2:surfaces:lagrange})।
\end{remark}

\begin{remark}[विधि: चर-परिवर्तन का चुनाव]
तीन प्रतिवर्त अधिकांश स्थितियाँ ढँक लेते हैं। \emph{समाकल्य की
सममिति}: $x^2 + y^2$ ध्रुवीय की माँग करता है, कोई गुणनफल संरचना कार्तीय
अक्ष बनाए रखने की। \emph{सीमा का आकार}: सीमाएँ $u(x,y) = c_1$, $v(x,y) = c_2$ स्वयं
निर्देशांकों $(u, v)$ की भीख माँगती हैं, जैसा नीचे अतिपरवलयी क्षेत्र
वाले उदाहरण में --- प्रांत आयत बन जाता है, और यही पूरी जीत है।
\emph{रैखिक संरचना}: $x + y$ तथा $x - y$ वाले व्यंजक $45$-अंश के घूर्णन या
किसी अपरूपण (\cref{ex:b2:multint:affine}) को न्योता देते हैं। हर स्थिति में समाकलन से पहले
तीन ख़ाने भरने हैं: प्रतिचित्रण नए प्रांत का पुराने पर एकैकी
आच्छादन है; उसका याकोबी \emph{उसी दिशा में} परिकलित है जिसमें
वस्तुतः उपयोग हुआ (आसान हो तो अंत में प्रतिलोम ले लीजिए); और
याकोबी अपने निरपेक्ष मान के साथ प्रवेश करता है।
\end{remark}

\begin{example}[ध्रुवीय निर्देशांक]\label{ex:b2:multint:polar}
$\Phi(\rho, \alpha) = (\rho\cos\alpha,\ \rho\sin\alpha)$ का
\[
J_\Phi = \begin{pmatrix}
\cos\alpha & -\rho\sin\alpha\\
\sin\alpha & \rho\cos\alpha
\end{pmatrix},
\qquad \det J_\Phi = \rho ,
\]
है, अतः $\dd x\,\dd y = \rho\,\dd\rho\,\dd\alpha$। त्रिज्या $R$ की चक्रिका $D_R$ के लिए:
\[
\iint_{D_R} e^{-(x^2 + y^2)}\,\dd x\,\dd y
= \int_0^{2\pi}\!\!\int_0^R e^{-\rho^2}\rho\,\dd\rho\,\dd\alpha
= \pi\bigl(1 - e^{-R^2}\bigr)
\xrightarrow[R\to\infty]{} \pi .
\]
वर्ग $[-R, R]^2$ से तुलना करने पर (जो चक्रिकाओं $D_R$ तथा $D_{R\sqrt2}$ के बीच दबा
है, और सारे समाकल्य धनात्मक हैं) $\bigl(\int_{-\infty}^\infty e^{-x^2}\dd x\bigr)^2 = \pi$ मिलता है:
\[
\boxed{\ \int_{-\infty}^{+\infty} e^{-x^2}\,\dd x = \sqrt{\pi}\ }
\]
--- अर्थात् फिर गाउस समाकल, अब अपनी सबसे प्रसिद्ध उपपत्ति से
(\cref{ch:b2:integration} की एकचर व्युत्पत्ति से तुलना कीजिए)।
\end{example}

\begin{example}[आनत चर-परिवर्तन]\label{ex:b2:multint:affine}
$M$ प्रतिलोमनीय होने पर किसी \omterm{def:b2:affine:subspace}{आनत प्रतिचित्रण} $\Phi(u, v) = M(u, v)^{\mathsf T} + C$ के लिए याकोबी
अचर आव्यूह $M$ है: \omterm{def:b2:multint:domain}{क्षेत्रफल} अचर गुणक $\abs{\det M}$ से गुणित हो जाते हैं
--- \cref{ch:b2:affine} में किया गया वादा अब प्रमेय है। दो तात्कालिक उपयोग।
दीर्घवृत्त $\frac{x^2}{a^2} +
\frac{y^2}{b^2} \leq 1$ $(u, v) \mapsto (au, bv)$ के अंतर्गत इकाई चक्रिका का प्रतिबिंब है, अतः
उसका \omterm{def:b2:multint:domain}{क्षेत्रफल} $ab \cdot \pi$ है --- बिना किसी परिकलन के। और वर्ग $K = \intcc01^2$ पर
$f(x + y)$ के समाकल के लिए अपरूपण $\Phi(u, v) = (u - v, v)$ (\omterm{def:b2:linalg:det}{सारणिक} $1$) उसे किसी समांतर
चतुर्भुज पर $f(u)$ के समाकल में बदल देता है, जिसे फ़ूबिनी अचर $u$
पर काटता है: $f =
\exp$ के साथ,
\[
\iint_K \eu^{x+y}\,\dd x\,\dd y
= \Bigl(\int_0^1 \eu^x\,\dd x\Bigr)^2 = (\eu - 1)^2,
\]
जैसा गुणनफल संरचना पुष्ट करती है। समाकल्य के अनुकूल निर्देशांक
चुनना --- प्रांत के अनुकूल नहीं --- कारीगरी का दूसरा आधा भाग है।
\end{example}


\begin{example}[किसी वक्ररेखीय प्रांत के अनुकूल
निर्देशांक]\label{ex:b2:multint:hyperbolicregion}
मान लीजिए $D$ पहले चतुर्थांश का वह क्षेत्र है जो अतिपरवलयों $xy = 1$
तथा $xy = 3$ और रेखाओं $y = x$ तथा $y = 3x$ से घिरा है। निर्देशांकों $u = xy$, $v = y/x$
में प्रांत वर्ग $\intcc13 \times \intcc13$ बन जाता है; प्रतिलोम लेने पर,
\[
x = \sqrt{u/v}, \qquad y = \sqrt{uv},
\qquad
\det J = x_uy_v - x_vy_u = \frac{1}{2v}
\]
($x = u^{1/2}v^{-1/2}$, $y =
u^{1/2}v^{1/2}$ के साथ दो पंक्तियों का परिकलन)। इसलिए
\[
\operatorname{Area}(D)
= \int_1^3\!\!\int_1^3\frac{\dd u\,\dd v}{2v}
= 2\cdot\frac{\ln 3}{2} = \ln 3 \approx 1.10 .
\]
$D$ को कार्तीय निर्देशांकों में काटने का अर्थ है उसे अतिपरवलयी
तथा रैखिक सीमाओं वाले तीन टुकड़ों में बाँटना --- संभव, आनंदहीन, और
भूल-भरा। सार \cref{ex:b2:multint:affine} को पूरी शक्ति से दोहराता है: सीमा-समीकरण पढ़िए,
और \emph{उन्हीं को} निर्देशांक चुनने दीजिए; फिर याकोबी वक्ररेखीय
जाल की कोशिका का \omterm{def:b2:multint:domain}{क्षेत्रफल} बदल देता है, ठीक वैसे जैसे $\rho$ ने
ध्रुवीय निर्देशांकों के लिए किया।
\end{example}

\begin{example}[माध्य मान]\label{ex:b2:multint:average}
किसी प्रांत $D$ पर $f$ का \emph{माध्य मान} $\frac1{\operatorname{Area}(D)}\iint_Df$ है। नमूना:
त्रिज्या $R$ की चक्रिका के किसी \omterm{def:b2:funcseq:def}{एकसमान रूप से} चुने गए बिंदु के लिए
केंद्र से माध्य दूरी
\[
\frac{1}{\pi R^2}\int_0^{2\pi}\!\!\int_0^R
\rho\cdot\rho\,\dd\rho\,\dd\alpha
= \frac{2\pi R^3/3}{\pi R^2} = \frac{2R}3 ,
\]
है, $R/2$ नहीं: एकसमान \omterm{def:b2:multint:domain}{क्षेत्रफल} बड़ी त्रिज्याओं पर अधिक द्रव्यमान
रखता है (त्रिज्या $\rho$ पर वलय का भार $\rho$ के समानुपाती है), अतः
माध्य आधे के निशान से आगे बैठता है। इस गुणक को ठीक पाना ठीक वही
ध्रुवीय याकोबी है जो काम कर रहा है, और वही भारण अध्याय में आगे
परिकलित अर्धगोले के गुरुत्व केंद्र $\bar z = 3R/8$ को --- $R/2$ के बजाय ---
समझा देता है।
\end{example}

\section{ग्रीन--रीमान प्रमेय}

\begin{theorem}[ग्रीन--रीमान]\label{thm:b2:multint:green}
मान लीजिए $K \subseteq \R^2$ कोई ऐसा \omterm{def:b2:metric:compact}{संहत} प्रांत है जो दोनों दिशाओं में
प्रारंभिक है (अथवा खंडों के अनुदिश जोड़े गए ऐसे प्रांतों का कोई
परिमित संघ), जिसकी सीमा $\partial K$ कोई खंडशः $\mathcal{C}^1$ संवृत वक्र है और जो
\emph{वामावर्त} अभिविन्यस्त है (प्रांत बाईं ओर रहता है)। $K$ के
किसी प्रतिवेश पर वर्ग $\mathcal{C}^1$ के $P, Q$ के लिए:
\[
\oint_{\partial K} P\,\dd x + Q\,\dd y
= \iint_K \Bigl(\frac{\partial Q}{\partial x}
- \frac{\partial P}{\partial y}\Bigr)\,\dd x\,\dd y .
\]
\end{theorem}

\begin{proof}
पहले, $K$ को किसी खंड के अनुदिश दो टुकड़ों $K_1, K_2$ में काटने पर दोनों
पक्ष योज्य हैं: द्विक समाकल $\iint$ की योज्यता से जुड़ जाते हैं; और
सीमा-समाकलों के लिए, $K_1$ तथा $K_2$ की वामावर्त सीमाएँ भीतरी कटान को
एक-एक बार, पर \emph{विपरीत} दिशाओं में तय करती हैं, अतः योग
\[
\oint_{\partial K_1} + \oint_{\partial K_2}
= \oint_{\partial K} + (\text{कटान, दोनों दिशाओं में})
= \oint_{\partial K},
\]
में कटान के दोनों फेरे कट जाते हैं (\cref{prop:b2:multint:lineinv}) और केवल बाहरी सीमा बची
रह जाती है। परिमित रूप से कई कटान दोहराने पर किसी प्रारंभिक प्रांत
को साधना पर्याप्त है। हम किसी $y$-प्रारंभिक प्रांत $D = \{a \leq x \leq b,\ \varphi_1(x) \leq y
\leq \varphi_2(x)\}$ पर $\oint P\,\dd x = -\iint_K P_y$
सिद्ध करते हैं; सर्वसमिका $\oint Q\,\dd y = \iint_K Q_x$ \omterm{def:b2:quadratic:adjoint}{सममित} है ($x$-प्रारंभिक), और
प्रमेय उनका योग है।

द्विक समाकल फ़ूबिनी तथा एकचर आधारभूत प्रमेय से परिकलित कीजिए:
\[
\iint_D \frac{\partial P}{\partial y}\,\dd x\,\dd y
= \int_a^b \bigl(P(x, \varphi_2(x)) - P(x, \varphi_1(x))\bigr)
\,\dd x .
\]
अब $D$ की वामावर्त सीमा में ये हैं: बाएँ से दाएँ तय किया गया निचला
ग्राफ़ $y = \varphi_1(x)$, दाहिना ऊर्ध्वाधर खंड $x = b$ (ऊपर की ओर), \emph{दाएँ से
बाएँ} तय किया गया ऊपरी ग्राफ़ $y =
\varphi_2(x)$, और बायाँ ऊर्ध्वाधर खंड $x = a$
(नीचे की ओर)। ऊर्ध्वाधर खंडों के अनुदिश $x$ अचर है, अतः वे $\oint P\,\dd x$ में
$0$ का योगदान देते हैं; और $x$ से प्राचलित ग्राफ़ देते हैं
\[
\oint_{\partial D} P\,\dd x
= \int_a^b P(x, \varphi_1(x))\,\dd x
- \int_a^b P(x, \varphi_2(x))\,\dd x
= -\iint_D \frac{\partial P}{\partial y}\,\dd x\,\dd y .
\qedhere
\]
\end{proof}

\begin{corollary}[सीमा से क्षेत्रफल]\label{cor:b2:multint:area}
\cref{thm:b2:multint:green} की परिकल्पनाओं के अंतर्गत,
\[
\operatorname{Area}(K)
= \oint_{\partial K} x\,\dd y
= -\oint_{\partial K} y\,\dd x
= \frac12\oint_{\partial K} x\,\dd y - y\,\dd x .
\]
\end{corollary}

\begin{proof}
$(P, Q) = (0, x)$, $(-y, 0)$ तथा $\frac12(-y, x)$ पर ग्रीन--रीमान लगाइए: हर बार $Q_x - P_y = 1$।
\end{proof}

\begin{remark}[तीनों क्षेत्रफल-सूत्रों में से चुनना]
तीनों सीमा-सूत्र बराबर हैं, पर व्यवहार में अदल-बदल \omterm{def:b2:series:summable}{योग्य} नहीं।
$\oint x\,\dd y$ तब लीजिए जब प्राचलन $\dd y$ को सरल बना दे ($y$-अक्ष के ऊपर
ग्राफ़), $-\oint y\,\dd x$ सममित रूप से, और \omterm{def:b2:quadratic:adjoint}{सममित} अर्ध-योग तब जब प्राचलन $x$
तथा $y$ के साथ बराबर बरताव करे --- दीर्घवृत्त के लिए उसने अचर
समाकल्य दे दिया, बिना ज़रा भी त्रिकोणमितीय रैखिकीकरण के। बहुभुजीय
सीमाओं पर अर्ध-योग \cref{exo:b2:multint:12} का फ़ीता सूत्र बन जाता है, जो भू-मापकों की
कलनविधि है। और जब दिया गया प्राचलन सीमा को दक्षिणावर्त तय करे, तब
तीनों सूत्र \omterm{def:b2:multint:domain}{क्षेत्रफल} का \emph{ऋण} लौटाते हैं: कोई ऋणात्मक परिणाम
परिकलन की भूल नहीं, बल्कि अभिविन्यास की सूचना है --- चिह्न पलट
दीजिए, या प्राचलन।
\end{remark}

\begin{example}[दीर्घवृत्त का क्षेत्रफल]\label{ex:b2:multint:ellipsearea}
$x = a\cos t$, $y = b\sin t$, $t \in [0, 2\pi]$ के लिए:
\[
\operatorname{Area}
= \frac12\int_0^{2\pi}\bigl(a\cos t \cdot b\cos t
- b\sin t\cdot(-a\sin t)\bigr)\,\dd t
= \frac{ab}{2}\int_0^{2\pi}\dd t = \pi ab .
\]
\end{example}

\begin{example}[जाँच के रूप में
ग्रीन--रीमान]\label{ex:b2:multint:greencheck}
संवृत इकाई चक्रिका $D$ पर $P = -y^3$, $Q = x^3$ लीजिए। सीमा वाला पक्ष, $\gamma(t) = (\cos t, \sin t)$ के
साथ:
\[
\oint_{\partial D}P\,\dd x + Q\,\dd y
= \int_0^{2\pi}\bigl(\sin^4 t + \cos^4 t\bigr)\dd t
= 2\pi\cdot\Bigl(\frac38 + \frac38\Bigr) = \frac{3\pi}2 ,
\]
रैखिकीकरण ($\sin^4 + \cos^4 = \tfrac34 +
\tfrac14\cos4t$) से। भीतरी पक्ष:
\[
\iint_D(Q_x - P_y)\,\dd x\,\dd y
= \iint_D 3(x^2 + y^2)\,\dd x\,\dd y
= 3\int_0^{2\pi}\!\!\int_0^1\rho^3\,\dd\rho\,\dd\alpha
= \frac{3\pi}2 .
\]
वही संख्या, दो बिलकुल भिन्न परिकलन --- और व्यावहारिक उपयोग यही है:
ग्रीन की सर्वसमिका का जो पक्ष आसान हो वही परिकलन बन जाता है, दूसरा
सत्यापन। संवृत वक्रों के चारों ओर बहुपदीय क्षेत्रों के परिसंचरण के
लिए द्विक समाकल लगभग सदा आसान पक्ष होता है।
\end{example}

\begin{remark}
ग्रीन--रीमान \cref{ex:b2:multint:angleform} को समझा देती है: किसी \omterm{def:b2:multint:exact}{संवृत रूप} ($Q_x = P_y$) के लिए
$U$ में समाहित किसी भी प्रांत की सीमा के चारों ओर समाकल लुप्त हो
जाता है। कोण रूप केवल इसलिए \omterm{def:b2:multint:exact}{यथातथ} होने से चूक जाता है कि मूल बिंदु
का छिद्र इकाई वृत्त से घिरी चक्रिका को $U$ के भीतर पड़ने नहीं देता
--- संवृत रूपों के \omterm{def:b2:multint:lineint}{रेखा समाकल} प्रांत के छेद पकड़ लेते हैं। (और आगे
धकेलने पर यही प्रेक्षण दे राम सह-समरूपता बन जाता है।)
\end{remark}

\section{त्रिक समाकल}

सिद्धांत बिना किसी नए विचार के तीन चरों तक फैल जाता है: फ़ूबिनी $\iiint$
को तीन एकचर समाकलों तक सिमटा देता है (या तो \emph{काटकर}: क्षैतिज
फाँकों $K_z$ पर $\iiint_K f = \int\bigl(\iint_{K_z}
f\bigr)\dd z$, या \emph{चट्टा लगाकर}: ऊर्ध्वाधर छड़ों के
अनुदिश पहले $z$ में समाकलन करके), और चर-परिवर्तन सूत्र $3 \times 3$ याकोबी
के साथ टिका रहता है।

\begin{example}[बेलनी और गोलीय निर्देशांक]\label{ex:b2:multint:spherical}
\emph{बेलनी} $(x, y, z) = (\rho\cos\alpha, \rho\sin\alpha,
z)$: $\dd x\,\dd y\,\dd z = \rho\,\dd\rho\,\dd\alpha\,\dd z$। \emph{गोलीय} $(x, y, z) = (r\cos\theta\cos\varphi,\
r\sin\theta\cos\varphi,\ r\sin\varphi)$ ($\theta$ देशांतर, $\varphi \in [-\frac\pi2, \frac\pi2]$
अक्षांश): $3
\times 3$ \omterm{def:b2:linalg:det}{सारणिक} को अंतिम पंक्ति के अनुदिश खोलने पर,
\[
\det J = r^2\cos\varphi ,
\qquad
\dd x\,\dd y\,\dd z
= r^2\cos\varphi\;\dd r\,\dd\theta\,\dd\varphi .
\]
त्रिज्या $R$ के \omterm{pb:b2:multint:1}{गोलक का आयतन}:
\[
V = \int_0^R\!\!\int_0^{2\pi}\!\!\int_{-\pi/2}^{\pi/2}
r^2\cos\varphi\;\dd\varphi\,\dd\theta\,\dd r
= \frac{R^3}{3}\cdot 2\pi \cdot 2
= \boxed{\frac43\pi R^3} ,
\]
जिससे पिछली पुस्तकों के आयतन वाले अध्यायों में मान लिया गया सूत्र
अंततः चुका दिया जाता है।
\end{example}

\begin{example}[चतुष्फलक, दो बार]
\label{ex:b2:multint:tetra}
$T = \{x, y, z \geq 0,\ x + y + z \leq 1\}$ का आयतन, चट्टा लगाकर: त्रिभुज $x + y \leq 1$ में $(x, y)$ स्थिर रखने पर $z$
$\intcc0{1 - x - y}$ पर चलता है, अतः
\[
V = \int_0^1\!\!\int_0^{1-x}(1 - x - y)\,\dd y\,\dd x
= \int_0^1\frac{(1 - x)^2}{2}\,\dd x = \frac16 .
\]
काटकर: ऊँचाई $z$ पर परिच्छेद त्रिभुज $\{x, y
\geq 0,\ x + y \leq 1 - z\}$ है, जिसका \omterm{def:b2:multint:domain}{क्षेत्रफल} $\frac{(1-z)^2}2$
है, और फिर $V = \int_0^1\frac{(1-z)^2}2\,\dd z = \frac16$ --- दोनों परिकलन वही समाकल हैं, बस भिन्न क्रम में,
और फ़ूबिनी इससे अधिक कुछ दावा नहीं करती। मान $\frac16 =
\frac13\cdot\frac12\cdot1$ त्रिभुजाकार आधार
वाला शंकु सूत्र (\cref{ex:b2:multint:cone}) है, और $n$-विमीय रूप $1/n!$ ठीक इसी कटान से
सप्ताहांत समस्या में सिद्ध होता है।
\end{example}

\begin{example}[अर्धगोलक का गुरुत्व केंद्र]\label{ex:b2:multint:centroid}
त्रिज्या $R$ के ऊपरी अर्धगोलक $H$ ($z \geq 0$) के लिए गुरुत्व केंद्र की
ऊँचाई $\bar z = \frac1{V}\iiint_H z$ है, जहाँ $V =
\frac23\pi R^3$। गोलीय निर्देशांकों में ($z =
r\sin\varphi$, $\varphi \in \intcc0{\pi/2}$):
\[
\iiint_H z
= \int_0^R r^3\,\dd r\int_0^{2\pi}\dd\theta
\int_0^{\pi/2}\sin\varphi\cos\varphi\,\dd\varphi
= \frac{R^4}4\cdot2\pi\cdot\frac12 = \frac{\pi R^4}4 ,
\]
अतः
\[
\bar z = \frac{\pi R^4/4}{2\pi R^3/3} = \frac{3R}8 :
\]
अर्थात् किसी ठोस अर्धगोले का संतुलन बिंदु सपाट फलक से त्रिज्या के
तीन-आठवें भाग ऊपर बैठता है --- आधी ऊँचाई $R/2$ से नीचे, जैसा होना
ही चाहिए, क्योंकि ठोस आधार के पास मोटा है। हर गुरुत्व-केंद्र परिकलन
का यही आकार होता है: एक आघूर्ण समाकल, एक आयतन, एक अनुपात, और
ज्यामिति के सामने विश्वसनीयता की एक जाँच।
\end{example}

\begin{example}[काटकर आयतन: शंकु]\label{ex:b2:multint:cone}
आधार-क्षेत्रफल $A$ तथा ऊँचाई $h$ वाला कोई शंकु (शीर्ष ऊपर, आधार
$z = 0$ पर): ऊँचाई $z$ पर फाँक आधार का गुणक $(1 -
z/h)$ से मापन है, जिसका
\omterm{def:b2:multint:domain}{क्षेत्रफल} $A(1 - z/h)^2$ है। इसलिए
\[
V = \int_0^h A\Bigl(1 - \frac zh\Bigr)^2\dd z = \frac{Ah}{3} :
\]
अर्थात् स्कूली सूत्रों वाला एक-तिहाई, जो \emph{किसी भी} आधार-आकार
के लिए वैध है --- कटान उसे किसी वर्ग के समाकल में बदल देता है।
\end{example}

\begin{example}[समतल में समाकलनीयता की
देहलियाँ]\label{ex:b2:multint:threshold}
किन $\alpha > 0$ के लिए $\iint_{D}\rho^{-\alpha}\,\dd
x\,\dd y$ छिद्रित इकाई चक्रिका $D$ पर अभिसरण करता है
(वलयों $\varepsilon \leq \rho \leq 1$ पर सीमा)? ध्रुवीय निर्देशांकों में,
\[
\int_0^{2\pi}\!\!\int_\varepsilon^1\rho^{-\alpha}\,
\rho\,\dd\rho\,\dd\alpha
= 2\pi\int_\varepsilon^1\rho^{1-\alpha}\,\dd\rho ,
\]
जो $\varepsilon \to 0$ होने पर तभी अभिसरण करता है जब $1 - \alpha > -1$, अर्थात् $\alpha < 2$: विमा $2$
में क्रांतिक विचित्रता-घातांक स्वयं विमा ही है, क्योंकि याकोबी से
आया अतिरिक्त $\rho$ विचित्रता को एक घात नरम कर देता है। (इसी प्रकार
अवकाश में किसी बिंदु-विचित्रता के लिए $r^2$ के द्वारा $\alpha < 3$।) इस
प्रकार का त्रिज्य लेखा-जोखा ही वह तरीक़ा है जिससे वर्ष 3 के लेबेग
ढाँचे में समाकलनीयता एक नज़र में तय हो जाती है --- और यही कारण है
कि \cref{exo:b2:multint:7} में $\iiint 1/r$ बिना किसी मशक़्क़त के अभिसरित हुआ।
\end{example}

\begin{remark}[सामान्य चूकें]
(क) \emph{अभिविन्यास}: कोई \omterm{def:b2:multint:lineint}{रेखा समाकल} यात्रा की दिशा के साथ चिह्न
बदल देता है, और ग्रीन--रीमान सीमा को वामावर्त माँगती है (प्रांत
बाईं ओर); छेद वाले प्रांत के लिए भीतरी सीमा \emph{दक्षिणावर्त} तय
की जाती है। (ख) \emph{याकोबी निरपेक्ष मान के साथ प्रवेश करता है}:
चर-परिवर्तन कभी ऋणात्मक \omterm{def:b2:multint:domain}{क्षेत्रफल} नहीं देता, और $\abs{\det}$ भूल जाना
प्रायः ठीक तभी चिह्न पलट देता है जब प्रतिचित्रण अभिविन्यास उलटता
है। (ग) \emph{ध्रुवीय गुणक $\rho$}: $\dd x\,\dd y = \rho\,\dd\rho\,\dd\alpha$, $\dd\rho\,\dd\alpha$ नहीं --- पूरे अध्याय
की सबसे आम भूल; विमीय विश्लेषण उसे पकड़ लेता है, क्योंकि $\dd\rho\,\dd\alpha$ की
विमा किसी \omterm{def:b2:curves:length}{लंबाई} की है, \omterm{def:b2:multint:domain}{क्षेत्रफल} की नहीं। (घ) \emph{अनुचित द्विक
समाकल}: यहाँ बढ़ती चक्रिकाओं तथा बढ़ते वर्गों पर सीमाएँ इसलिए मेल
खाती हैं कि समाकल्य धनात्मक हैं (दबाव); चिह्न बदलने वाले समाकल्यों
के लिए सीमा निःशेषण पर निर्भर कर सकती है, और निरपेक्ष अभिसरण के
बिना कोई दावा नहीं किया जाता। (ङ) \emph{प्रांत बनाम समाकल्य}: किसी
अगुणनफल प्रांत पर कोई गुणनफल समाकल्य समाकल का गुणनखंडन \emph{नहीं}
करता --- गुणनखंडन को दोनों चाहिए, जैसा \cref{ex:b2:multint:polar} के वर्ग में।
\end{remark}

\begin{remark}[इस खंड के भीतर के परिप्रेक्ष्य]
यहाँ परिकलित गाउस समाकल प्रायिकता वाले अध्यायों में चुपचाप हर जगह
है: स्टर्लिंग सूत्र (\cref{thm:b2:comparison:stirling}) के भीतर का अचर $\sqrt\pi$ इसी अध्याय का समाकल
है, और स्टर्लिंग के द्वारा वह \cref{ch:b2:proba} में यादृच्छिक चहलक़दमी की
वापसी-प्रायिकताओं की $1/\sqrt{\pi n}$ अनंतस्पर्शी को नियत कर देता है। सप्ताहांत
समस्या के \omterm{pb:b2:multint:1}{वालिस समाकल} भी वहीं फिर प्रकट होते हैं, और वही
केंद्रीय-द्विपद आकलन चलाते हैं। दूसरी दिशा में, इस अध्याय के
\omterm{def:b2:multint:domain}{क्षेत्रफल} तथा आयतन अवयव \cref{ch:b2:surfaces} की ज्यामिति पूरी करते हैं, और ग्रीन का
सूत्र \cref{ch:b2:curves} के अन्वालोप-क्षेत्रफल फिर से परिकलित कर देता है
(\omterm{pb:b2:curves:1}{ऐस्ट्रॉयड}, \cref{exo:b2:multint:5} में)। एक अध्याय, तीन सेवाएँ: ज्यामिति के लिए माप,
प्रायिकता के लिए अचर, और चर-परिवर्तन का वह अनुशासन जिसे दोनों
बरतते हैं।
\end{remark}

\begin{exercise}[$\star$]\label{exo:b2:multint:1}
$\int_\gamma y^2\,\dd x + x\,\dd y$ इनके अनुदिश परिकलित कीजिए: (क) $(0,0)$ से $(1,1)$ तक का खंड; (ख) $(0,0)$
से $(1,1)$ तक परवलय-चाप $y = x^2$। क्या रूप \omterm{def:b2:multint:exact}{यथातथ} है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:multint:2}
दिखाइए कि $\omega = (2xy + y^3)\,\dd x + (x^2 + 3xy^2 + 1)\,\dd y$ $\R^2$ पर संवृत है, कोई \omterm{def:b2:multint:exact}{विभव} ढूँढ़िए, और $(0, 0)$ से $(1, 2)$ तक
किसी भी चाप के अनुदिश $\int_\gamma\omega$ परिकलित कीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:multint:3}
$\iint_D (x + y)\,\dd x\,\dd y$ परिकलित कीजिए, जहाँ $D$ $y = x^2$ तथा $y = x$ ($0 \leq x \leq 1$) से घिरा प्रांत है,
और यह समाकलन के दोनों क्रमों में कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:multint:4}
ध्रुवीय निर्देशांकों का उपयोग करते हुए $\iint_D \frac{\dd x\,\dd y}{(1 +
x^2 + y^2)^2}$ पूरे समतल पर (चक्रिकाओं
पर सीमा के रूप में) परिकलित कीजिए, तथा $\iint_{D'} xy\,\dd x\,\dd y$ चतुर्थांश चक्रिका $D' = \{x, y
\geq 0,\ x^2 + y^2 \leq 1\}$
पर।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:multint:5}
\cref{cor:b2:multint:area} का उपयोग करते हुए \omterm{pb:b2:curves:1}{ऐस्ट्रॉयड} $x = \cos^3 t$, $y =
\sin^3 t$, $t \in [0, 2\pi]$, से घिरा \omterm{def:b2:multint:domain}{क्षेत्रफल}
परिकलित कीजिए। \emph{($\sin^2 t\cos^2 t$ का रैखिकीकरण कीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:multint:6}
नीचे से परवलयज $z
= x^2 + y^2$ और ऊपर से समतल $z = 1$ से घिरे ठोस का आयतन दोनों
विधियों से परिकलित कीजिए: चट्टा लगाकर (इकाई चक्रिका पर $1 - x^2 - y^2$ का
समाकलन, ध्रुवीय निर्देशांक) और काटकर (क्षैतिज फाँकें त्रिज्या $\sqrt z$
की चक्रिकाएँ हैं)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:multint:7}
(गोलक का गुरुत्वाकर्षण --- न्यूटन की प्रमेय, विशेष स्थिति) दिखाइए
कि गोलीय खोल $a \leq r \leq
b$ का आयतन $\frac43\pi(b^3 - a^3)$ है और त्रिज्या $R$ के गोलक $B$ पर
$\iiint_{B}
\frac{\dd x\,\dd y\,\dd z}{r}$ परिकलित कीजिए ($r$ मूल बिंदु से दूरी)। \emph{(गोलीय
निर्देशांक।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:multint:8}
(प्वांकारे की प्रमेयिका, तारकाकार स्थिति) मान लीजिए $U$ $0$ के
सापेक्ष तारकाकार है (अर्थात् $M \in U \Rightarrow [0, M] \subseteq U$) और $\omega = P\dd x + Q\dd y$ $U$ पर कोई संवृत $\mathcal{C}^1$ रूप
है। परिभाषित कीजिए
\[
f(x, y) = \int_0^1 \bigl(x\,P(tx, ty) + y\,Q(tx, ty)\bigr)\dd t .
\]
समाकल चिह्न के भीतर अवकलन (\cref{ch:b2:integration}) तथा $P_y = Q_x$ का उपयोग करते हुए दिखाइए
कि $f_x = P$ और $f_y = Q$: अर्थात् किसी तारकाकार \omterm{def:b2:metric:topology}{विवृत} समुच्चय पर हर \omterm{def:b2:multint:exact}{संवृत
रूप} \omterm{def:b2:multint:exact}{यथातथ} है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:multint:9}
(द्विक समाकलन से डिरिक्ले समाकल) $[0, A] \times [0, \infty)$ पर
\[
\int_0^\infty\!\!\int_0^\infty e^{-xy}\sin x\;\dd y\,\dd x
\quad\text{बनाम}\quad
\int_0^\infty\!\!\int_0^\infty e^{-xy}\sin x\;\dd x\,\dd y
\]
को न्यायसंगत ठहराइए और भुनाइए: दिखाइए कि $\int_0^A \frac{\sin
x}{x}\dd x = \frac\pi2 - \int_0^\infty
e^{-Ay}\frac{y\sin A + \cos A}{1 + y^2}\dd y$ और $\int_0^\infty \frac{\sin x}{x}\,\dd x = \frac\pi2$ पुनः प्राप्त
कीजिए, तथा \cref{ch:b2:integration} की प्राचल-समाकल उपपत्ति से तुलना कीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:multint:10}
(विर्टिंगर के द्वारा समपरिमाप असमिका) मान लीजिए $\gamma$ \omterm{def:b2:curves:length}{लंबाई} $2\pi$ का
कोई सरल संवृत $\mathcal{C}^1$ वक्र है, जो $[0, 2\pi]$ पर \omterm{def:b2:curves:length}{चाप-लंबाई} से प्राचलित है और
\omterm{def:b2:multint:domain}{क्षेत्रफल} $A$ घेरता है। \cref{cor:b2:multint:area}, पारसेवाल तथा विर्टिंगर असमिका
(\cref{ch:b2:fourier} के अभ्यास) का उपयोग करते हुए $A \leq \pi$ सिद्ध कीजिए, जिसमें वृत्त
के लिए समता होती है। \emph{($\int_0^{2\pi} x(s)\dd s
= 0$ सामान्यीकृत कीजिए; $2A = \oint x\,\dd y - y\,\dd x$ लिखिए और
$2A = \int_0^{2\pi}(xy' - yx')\dd
s$ तथा $x^2 + y'^2 \geq 2xy'$ के द्वारा $2A \leq
\int (x^2 + y'^2)$ को ध्यान से परिबद्ध कीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:multint:11}
(गोलक के आघूर्ण) $\R^3$ में त्रिज्या $R$ के गोलक $B$ के लिए गोलीय
निर्देशांकों में $\iiint_B z^2\,\dd x\,\dd y\,\dd z$ परिकलित कीजिए, और सममिति से $\iiint_B (x^2 + y^2 +
z^2)\,\dd x\,\dd y\,\dd z$ निकालिए।
बाद वाले की खोल-परिकलन $\int_0^R r^2\cdot4\pi
r^2\,\dd r$ के सामने जाँच कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:multint:12}
(फ़ीता सूत्र) मान लीजिए $K$ कोई बहुभुज है जिसके शीर्ष वामावर्त
क्रम में $(x_1,
y_1), \dots, (x_m, y_m)$ हैं (सूचकांक $m$ मॉड्यूलो)। \cref{cor:b2:multint:area} से निष्कर्ष
निकालिए कि
\[
\operatorname{Area}(K)
= \frac12\sum_{i=1}^m
\bigl(x_iy_{i+1} - x_{i+1}y_i\bigr) ,
\]
और सूत्र को त्रिभुज $(0,0)$, $(1,0)$, $(0,1)$ पर जाँचिए।
\end{exercise}

\section{समस्या: विमा $n$ में गोलक का आयतन}

\begin{problem}[{सप्ताहांत समस्या --- $V_n =
\pi^{n/2}/\Gamma(\frac n2 + 1)$, और ऊँची विमाओं की विचित्रता}]\label{pb:b2:multint:1}
चक्रिका का \omterm{def:b2:multint:domain}{क्षेत्रफल} $\pi$ है, \omterm{pb:b2:multint:1}{गोलक का आयतन} $\frac43\pi$ --- और फिर? यह
समस्या हर $n$ के लिए $\R^n$ के इकाई \omterm{pb:b2:multint:1}{गोलक का आयतन} दो बार परिकलित करती
है (पहले वालिस समाकलों से चलने वाली कटान-पुनरावृत्ति से, फिर $\Gamma$
फलन तथा \cref{ex:b2:multint:polar} के गाउस समाकल के द्वारा), और फिर ज्यामिति पढ़ लेती है:
आयतन विमा पाँच पर चरम पर पहुँचते हैं और फिर शून्य की ओर दौड़ पड़ते
हैं, और किसी ऊँची-विमीय गोलक का लगभग सारा हिस्सा अपनी सीमा के पास
एक पतले खोल में छिपा रहता है।\index{गोलक का आयतन}\index{वालिस
समाकल} $\R^n$ के किसी गोलक पर \omterm{def:b2:metric:continuity}{संतत} फलन के लिए समाकल $n$-गुना
पुनरावृत्त समाकल समझा जाता है (एक-एक निर्देशांक काटते हुए, जैसा
अध्याय में $n
\leq 3$ के लिए); हम $0$ पर केंद्रित त्रिज्या $R$ के
संवृत गोलक के लिए $B_n(R)$ लिखते हैं, उसके आयतन के लिए $v_n(R)$, और $V_n =
v_n(1)$,
जहाँ परंपरा से $V_0 = 1$।

\medskip
\noindent\textbf{भाग I --- कटान-पुनरावृत्ति।}
\begin{enumerate}
  \item $n$ पुनरावृत्त समाकलों में से हर एक में $x_i = Ru_i$
        प्रतिस्थापित करके दिखाइए कि $v_n(R) = V_nR^n$।
  \item $B_n(1)$ को उसके अंतिम निर्देशांक के अनुदिश काटकर दिखाइए
        कि
        \[
        V_n = V_{n-1}\int_{-1}^{1}(1 -
        t^2)^{\frac{n-1}2}\,\dd t .
        \]
  \item $t = \sin\theta$ के साथ समाकल को किसी \omterm{pb:b2:multint:1}{वालिस समाकल} के रूप में
        पहचानिए: $\int_{-1}^1(1 -
        t^2)^{\frac{n-1}2}\dd t = 2W_n$, जहाँ $W_n =
        \int_0^{\pi/2}\cos^n\theta\,\dd\theta =
        \int_0^{\pi/2}\sin^n\theta\,\dd\theta$।
  \item दोनों वालिस सर्वसमिकाएँ सिद्ध कीजिए (खंडशः समाकलन
        कीजिए; फिर $nW_nW_{n-1}$ को दूरबीन की तरह सिकोड़िए):
        \[
        W_n = \frac{n-1}nW_{n-2} \quad (n \geq 2),
        \qquad
        W_nW_{n-1} = \frac{\pi}{2n} \quad (n \geq 1).
        \]
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- पुनरावृत्ति का हल।}
\begin{enumerate}[resume]
  \item प्रश्न 2 से 4 को दो-पग वाली पुनरावृत्ति में जोड़िए:
        \[
        V_n = \frac{2\pi}{n}\,V_{n-2}
        \qquad (n \geq 2).
        \]
  \item $k \geq 0$ के लिए \omterm{def:b2:multint:exact}{संवृत रूप} निकालिए:
        \[
        V_{2k} = \frac{\pi^k}{k!},
        \qquad
        V_{2k+1} = \frac{2^{k+1}\pi^k}{1\cdot3\cdot5\cdots
        (2k+1)} .
        \]
  \item $V_1, \dots, V_7$ की संख्यात्मक सारणी बनाइए। अनुपात $V_n/V_{n-2} = 2\pi/n$ तथा $2W_5$ और
        $2W_6$ के मानों का उपयोग करते हुए सिद्ध कीजिए कि अनुक्रम
        $(V_n)$ अपने अधिकतम $V_5 = \frac{8\pi^2}{15} \approx
        5.26$ तक बढ़ता है और उसके बाद घटता है।
  \item दिखाइए कि $V_n \to 0$ किसी भी गुणोत्तर अनुक्रम से तेज़ी से
        होता है, और $\sum_{n\geq1} V_n$ अभिसरण करता है: अर्थात् सारे इकाई
        गोलक मिलकर परिमित कुल आयतन रखते हैं।
  \item जनक सर्वसमिका
        \[
        \sum_{k\geq0} V_{2k}\,x^{2k} = \eu^{\pi x^2}
        \qquad (x \in \R),
        \]
        सिद्ध कीजिए और $\sum_{k\geq0}V_{2k} = \eu^\pi \approx
        23.14$ निकालिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- दूसरा रास्ता: $\Gamma$ और गाउस
समाकल।}
\begin{enumerate}[resume]
  \item फ़ूबिनी से दिखाइए (समाकल्य गुणनफल है) कि
        \[
        I_n = \int_{\R^n}\eu^{-\norm
        x^2}\dd x
        = \Bigl(\int_{-\infty}^{+\infty}
        \eu^{-t^2}\dd t\Bigr)^{\!n} = \pi^{n/2},
        \]
        अर्थात् $n$-विमीय गाउस समाकल, जिसे घनों $\intcc{-R}{R}^n$ पर सीमा के
        रूप में समझा जाता है।
  \item $\Gamma(s) = \int_0^\infty t^{s-1}\eu^{-t}\dd
        t$ (\cref{def:b2:integration:gamma}) स्मरण कीजिए। $\Gamma(s+1) = s\,\Gamma(s)$ (\cref{thm:b2:integration:gammaprops}) तथा $\Gamma(\tfrac12) = \sqrt\pi$ ($t = u^2$
        प्रतिस्थापित कीजिए और गाउस समाकल का सहारा लीजिए) से
        परिकलित कीजिए
        \[
        \Gamma(k + 1) = k!,
        \qquad
        \Gamma\Bigl(k + \frac32\Bigr) =
        \frac{1\cdot3\cdots(2k+1)}{2^{k+1}}\,\sqrt\pi .
        \]
  \item प्रश्न 5 की पुनरावृत्ति के द्वारा आगमन से अकेला सूत्र
        \[
        V_n = \frac{\pi^{n/2}}{\Gamma\bigl(\frac n2 +
        1\bigr)} \qquad (n \geq 1),
        \]
        सिद्ध कीजिए और जाँचिए कि वह प्रश्न 6 के दोनों \omterm{def:b2:multint:exact}{संवृत
        रूप} दोहरा देता है।
  \item दिखाइए कि $\int_0^\infty \eu^{-r^2}r^{n-1}\dd r =
        \tfrac12\Gamma\bigl(\tfrac n2\bigr)$ और सर्वसमिका
        \[
        I_n = n\,V_n\int_0^\infty \eu^{-r^2}\,r^{n-1}\,\dd r.
        \]
        निकालिए। उसकी व्याख्या कीजिए: $\R^n$ का गाउसी द्रव्यमान
        उन गोलीय खोलों के अनुदिश इकट्ठा किया जाता है जिनका
        त्रिज्या $r$ पर ``$(n-1)$-विमीय \omterm{def:b2:multint:domain}{क्षेत्रफल}'' $nV_nr^{n-1}$ है ---
        और चूँकि दोनों पक्ष अब स्वतंत्र रूप से सिद्ध हैं,
        व्याख्या मुफ़्त पड़ती है।
  \item $s_{n-1} = nV_n$ रखिए (इकाई गोले $S^{n-1}$ का \omterm{def:b2:multint:domain}{क्षेत्रफल}, $v_n(R) = \int_0^R
        s_{n-1}r^{n-1}\dd r$ के
        अनुरूप)। $s_0, \dots, s_3$ की सारणी बनाइए और $s_1 = 2\pi$, $s_2 = 4\pi$, $s_3 = 2\pi^2$
        जाँचिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- ऊँची विमाएँ विचित्र हैं।}
\begin{enumerate}[resume]
  \item $k!$ पर लगाए गए स्टर्लिंग सूत्र (\cref{thm:b2:comparison:stirling}) से सम $n = 2k$ के लिए
        दिखाइए:
        \[
        V_n \sim \frac{1}{\sqrt{\pi n}}
        \Bigl(\frac{2\pi\eu}{n}\Bigr)^{n/2}
        \qquad (n \to \infty, \ n \text{ सम}),
        \]
        और समझाइए कि वही अति-गुणोत्तर क्षय-परिबंध पुनरावृत्ति
        के द्वारा विषम $n$ तक क्यों फैल जाता है।
  \item इकाई गोलक आयतन $2^n$ के घन $\intcc{-1}1^n$ में बैठता है। $n = 2, 3, 10$ के
        लिए भरण-अनुपात $V_n/2^n$ परिकलित कीजिए, और दिखाइए कि वह $0$
        की ओर जाता है: ऊँची विमा में घन का लगभग सारा हिस्सा
        उसके कोनों में पड़ा रहता है।
  \item दिखाइए कि $v_n(1)$ का जितना अंश सीमा-गोले से $\varepsilon$ दूरी के
        भीतर पड़ता है वह $1 -
        (1 - \varepsilon)^n \to 1$ है; संख्यात्मक रूप से, $100$-विमीय
        गोलक का कितना अंश मोटाई $1\%$ के बाहरी खोल में पड़ता है?
  \item वालिस अनंतस्पर्शी $W_n \sim
        \sqrt{\dfrac{\pi}{2n}}$ सिद्ध कीजिए \emph{($(W_n)$ की
        एकदिष्टता, अनुपात $W_n/W_{n-2} \to 1$, तथा $W_nW_{n-1}
        = \frac\pi{2n}$)}, और सब $n$ के लिए
        निचला परिबंध $W_n \geq
        \sqrt{\dfrac{\pi}{2(n+1)}}$ भी।
  \item (किसी पट्टी पर संकेंद्रण) इकाई गोलक का जितना अंश पहले
        निर्देशांक में $\delta$ से आगे पड़ता है वह $\int_\delta^1(1 - x^2)^{\frac{n-1}2}\dd x \,\big/\,
        (2W_n)$ है। $1 - u \leq \eu^{-u}$
        तथा पुच्छ-परिबंध $\int_\delta^\infty \eu^{-a x^2}\dd x \leq
        \frac{\eu^{-a\delta^2}}{2a\delta}$ का उपयोग करते हुए दिखाइए कि यह
        अंश अधिक से अधिक
        \[
        \frac{\eu^{-(n-1)\delta^2/2}}{(n-1)\,\delta}
        \Big/ \sqrt{\frac{2\pi}{n+1}}
        \]
        है, और निष्कर्ष निकालिए: $\delta = s/\sqrt{n-1}$ के लिए गोलक का $O(\eu^{-s^2/2}/s)$ अंश
        छोड़कर बाक़ी सब पट्टी $\abs{x_1} \leq s/\sqrt{n-1}$ में पड़ता है। ऊँची विमा का
        गोलक, सांख्यिकीय रूप से, एक साथ हर दिशा में एक पतला
        पूआ है।
  \item प्रश्न 16 से 19 को एक अनुच्छेद में जोड़िए: $B_n(1)$ का
        आयतन कहाँ बैठता है (सीमा-गोले के पास, फिर भी केंद्र से
        होकर जाने वाले हर अतिसमतल की $O(1/\sqrt n)$ पट्टियों के भीतर),
        और ये दोनों कथन एक-दूसरे का खंडन क्यों नहीं करते।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- अन्य पिंड, और संश्लेषण।}
\begin{enumerate}[resume]
  \item (सरलक) मान लीजिए $\Delta_n = \{x \in \R^n : x_i \geq 0,\
        \sum x_i \leq 1\}$। कटान तथा आगमन से दिखाइए कि
        $\operatorname{vol}(\Delta_n) = \frac1{n!}$।
  \item (क्रॉस-पॉलीटोप) निष्कर्ष निकालिए कि $C_n = \{x :
        \sum\abs{x_i} \leq 1\}$ का आयतन $\frac{2^n}{n!}$
        है, और आयतनों के स्तर पर सैंडविच $C_n \subseteq B_n(1)
        \subseteq \intcc{-1}1^n$ सत्यापित कीजिए:
        $\frac{2^n}{n!} \leq V_n \leq 2^n$।
  \item $V_4$ तीसरी तरह से परिकलित कीजिए: $\R^4 = \R^2 \times
        \R^2$ काटिए, $(x, y)$-चक्रिका
        पर $(z, w)$-चक्रिका के \omterm{def:b2:multint:domain}{क्षेत्रफल} का ध्रुवीय निर्देशांकों में
        समाकलन कीजिए, और $V_4 = \frac{\pi^2}2$ पुनः प्राप्त कीजिए।
  \item (मुसीबत में मोंते कार्लो) घन $\intcc{-1}1^{20}$ में कोई बिंदु
        \omterm{def:b2:funcseq:def}{एकसमान रूप से} खींचा जाता है। दिखाइए कि उसके अंतर्लिखित
        गोलक में गिरने की प्रायिकता $V_{20}/2^{20} \approx 2.5\cdot10^{-8}$ है, अतः पहली सफलता की
        प्रत्याशा से पहले लगभग चार करोड़ खींचें चाहिए:
        अस्वीकृति-प्रतिचयन से $V_n$ का आकलन ऊँची विमा में ढह
        जाता है (विमीयता का अभिशाप)।
  \item संश्लेषण। दो स्वतंत्र व्युत्पत्तियाँ $V_n =
        \pi^{n/2}/\Gamma(\frac n2 + 1)$ पर मिलीं:
        बताइए कि हर एक ने इस अध्याय की कौन सी प्रमेय बरती
        (फ़ूबिनी, चर-परिवर्तन, ध्रुवीय गाउस समाकल), और कौन से
        एकचर निवेश (वालिस, $\Gamma$, स्टर्लिंग)। वर्ष 3 का खंड इस
        परिकलन को लेबेग सिद्धांत से कहाँ दोहराता है, और वह
        उसमें क्या जोड़ता है?
\end{enumerate}
\end{problem}
