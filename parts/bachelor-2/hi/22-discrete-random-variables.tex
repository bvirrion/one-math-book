\chapter{विविक्त यादृच्छिक चर}\label{ch:b2:randomvar}

\omterm{def:b2:randomvar:law}{यादृच्छिक चर} प्रायिकता के परिकलनों को \omterm{def:b2:proba:space}{घटनाओं} के बजाय फलनों के चारों
ओर व्यवस्थित कर देते हैं। \omterm{def:b2:structures:countable}{गणनीय} समष्टियों पर सिद्धांत को \cref{ch:b2:series} के
\omterm{def:b2:series:summable}{योग्य कुल} चलाते हैं: \omterm{def:b2:randomvar:expectation}{प्रत्याशा} \omterm{def:b2:proba:space}{प्रतिदर्श समष्टि} से अनुक्रमित किसी
कुल का योग है, और उसके सारे गुणधर्म --- रैखिकता, अंतरण, \omterm{def:b2:proba:independence}{स्वतंत्र}
चरों के लिए गुणनफल सूत्र --- \omterm{def:b2:series:summable}{योग्य} कुलों के विषय में प्रमेय हैं। यह
अध्याय मार्कोव, चेबिशेव, कोशी--श्वार्ज़ तथा जेनसन की मुख्य असमिकाएँ
सिद्ध करता है, और शास्त्रीय नियमों तथा बृहत् संख्याओं के दुर्बल
नियम पर समाप्त होता है, जिसकी उपपत्ति चेबिशेव के उपलब्ध होते ही दो
पंक्तियों की रह जाती है।

\section{यादृच्छिक चर और उनके नियम}

\begin{definition}[विविक्त यादृच्छिक चर; नियम]\label{def:b2:randomvar:law}
मान लीजिए $(\Omega, \P)$ कोई \omterm{def:b2:structures:countable}{गणनीय} \omterm{def:b2:proba:space}{प्रायिकता समष्टि} है। \emph{यादृच्छिक
चर}\index{यादृच्छिक चर} कोई प्रतिचित्रण $X \colon \Omega \to E$ है ($E$ कोई भी
समुच्चय; $E = \R$ होने पर \emph{वास्तविक} यादृच्छिक चर)। उसका
\emph{नियम}\index{नियम!यादृच्छिक चर का} (या
\emph{बंटन}\index{बंटन}) \omterm{def:b2:structures:countable}{गणनीय समुच्चय} $X(\Omega)$ पर वह \omterm{def:b2:proba:space}{प्रायिकता माप}
$\P_X$ है जो इस प्रकार परिभाषित है:
\[
\P_X(\{x\}) = \P(X = x)
= \P\bigl(\{\omega : X(\omega) = x\}\bigr) .
\]
\end{definition}

\begin{example}[शास्त्रीय नियम]\label{ex:b2:randomvar:laws}
\leavevmode
\begin{itemize}
\item \emph{बर्नूली} $\mathcal{B}(p)$: $X \in \{0, 1\}$, $\P(X =
1) = p$। किसी \omterm{def:b2:proba:space}{घटना} का सूचक।
\item \emph{द्विपद} $\mathcal{B}(n, p)$: $\P(X = k) =
\binom nk p^k(1-p)^{n-k}$, $0 \leq k \leq n$: $n$ \omterm{def:b2:proba:independence}{स्वतंत्र} बर्नूली प्रयासों
में सफलताओं की संख्या (हाई स्कूल खंड; नीचे \omterm{def:b2:proba:independence}{स्वतंत्र} चरों के योग के
द्वारा फिर से सिद्ध)।
\item \emph{ज्यामितीय} $\mathcal{G}(p)$: $\P(X = k) =
(1-p)^{k-1}p$, $k \in \N^*$: पहली सफलता की कोटि (\cref{ex:b2:proba:geometric})।
\item \emph{प्वासों} $\mathcal{P}(\lambda)$: $\P(X = k) =
e^{-\lambda}\frac{\lambda^k}{k!}$, $k \in \N$ --- जो चरघातांकी श्रेणी से
कोई \omterm{def:b2:proba:space}{प्रायिकता माप} है। विरल \omterm{def:b2:proba:space}{घटनाओं} का \omterm{def:b2:randomvar:law}{नियम} (\cref{ch:b2:genfun})।
\end{itemize}
\end{example}

\begin{remark}[कौन सा नियम किसका प्रतिरूप है]
चारों \omterm{def:b2:randomvar:law}{नियम} चार आदिम प्रश्नों के उत्तर देते हैं: बर्नूली, ``क्या वह
हुआ?''; द्विपद, ``$n$ प्रयासों में कितनी बार?''; ज्यामितीय, ``पहली
बार तक कितनी देर?''; प्वासों, ``दी गई दर पर कितनी घटनाएँ, जब
प्रयास बहुत हों और अलग-अलग असंभावित?''। प्रश्न को पहचान लेना
प्रतिरूपण का नौ-दसवाँ भाग है: सूचकों के योग द्विपद की ओर संकेत करते
हैं, प्रतीक्षा-समय ज्यामितीय की ओर, विरल-घटना गणनाएँ प्वासों की ओर
--- और द्विपद से प्वासों तक का मार्ग \cref{ch:b2:genfun} में विरल \omterm{def:b2:proba:space}{घटनाओं} के \omterm{def:b2:randomvar:law}{नियम} से
सुनिश्चित होता है।
\end{remark}

\begin{proposition}[ज्यामितीय नियम की
स्मृतिहीनता]\label{prop:b2:randomvar:memoryless}
यदि $X \sim \mathcal{G}(p)$, तो सब $m, n \in \N$ के लिए:
\[
\P(X > m + n \mid X > m) = \P(X > n) ,
\]
और $\N^*$ पर यह गुणधर्म रखने वाले एकमात्र \omterm{def:b2:randomvar:law}{नियम} ज्यामितीय \omterm{def:b2:randomvar:law}{नियम} हैं।
\end{proposition}

\begin{proof}
ज्यामितीय भारों को जोड़ने पर $\P(X > n) = (1-p)^n$। इसलिए
\[
\P(X > m + n \mid X > m)
= \frac{\P(X > m + n)}{\P(X > m)}
= \frac{(1-p)^{m+n}}{(1-p)^m} = (1-p)^n = \P(X > n).
\]
विलोमतः, यदि $G(n) = \P(X > n)$ $G(0) = 1$ के साथ $G(m + n) =
G(m)G(n)$ पूरा करे, तो आगमन से $G(n) = G(1)^n$; $q =
G(1) \in \intco{0}{1}$,
और $q = 0$, अन्यथा \omterm{def:b2:randomvar:law}{नियम} $\mathcal{G}(1 - q)$ है: $\P(X = k) = G(k-1) - G(k) =
q^{k-1}(1 - q)$।
\end{proof}

\begin{example}[कोई संख्या कभी ``बक़ाया'' नहीं होती]
छक्के की प्रतीक्षा करते हुए पासा फेंकिए: प्रतीक्षा-समय $X \sim
\mathcal G(1/6)$ है।
स्मृतिहीनता कहती है कि $10$ निष्फल फेंकों के बाद \emph{शेष}
प्रतीक्षा $X - 10$, $X > 10$ दिए जाने पर, फिर से $\mathcal G(1/6)$ है: अर्थात् सप्रतिबंध
प्रत्याशित प्रतीक्षा अब भी $6$ फेंक है, ठीक वैसे ही जैसे आरंभ में
थी। पासा याद नहीं रखता, और कोई छक्का कभी ``बक़ाया'' नहीं होता ---
जुआरी का भ्रम यही मान्यता है कि सप्रतिबंध \omterm{def:b2:randomvar:law}{नियम} खिसक जाना चाहिए था।
विलोमतः, प्रतिज्ञप्ति का अद्वितीयता वाला आधा भाग कहता है कि यही
उदासीनता ज्यामितीय प्रतीक्षा-समयों को \emph{अभिलक्षित} करती है:
जिस भी प्रतीक्षा-समय का पूर्वानुमान कभी नहीं बदलता वह ज्यामितीय
है। असली क़तारें तथा जीवनकाल प्रायः बदलते हैं, और ठीक इसी से पता
चलता है कि वे ज्यामितीय नहीं हैं।
\end{example}

\section{प्रत्याशा}

\begin{definition}[प्रत्याशा]\label{def:b2:randomvar:expectation}
$(\Omega, \P)$ पर किसी वास्तविक \omterm{def:b2:randomvar:law}{यादृच्छिक चर} $X$ की \emph{प्रत्याशा विद्यमान
है} यदि कुल $\bigl(X(\omega)\,\P(\{\omega\})\bigr)_{\omega \in \Omega}$ \omterm{def:b2:series:summable}{योग्य} हो (\cref{ch:b2:series}); तब उसकी
\emph{प्रत्याशा}\index{प्रत्याशा} है
\[
\E(X) = \sum_{\omega \in \Omega} X(\omega)\,\P(\{\omega\}) .
\]
\end{definition}

\begin{theorem}[अंतरण प्रमेय]\label{thm:b2:randomvar:transfer}
$X$ की \omterm{def:b2:randomvar:expectation}{प्रत्याशा} विद्यमान है तभी जब कुल $\bigl(x\,\P(X =
x)\bigr)_{x \in X(\Omega)}$ \omterm{def:b2:series:summable}{योग्य} हो, और तब
\[
\E(X) = \sum_{x \in X(\Omega)} x\,\P(X = x) .
\]
अधिक व्यापक रूप से, $f \colon X(\Omega) \to \R$ के लिए चर $f(X)$ की \omterm{def:b2:randomvar:expectation}{प्रत्याशा} विद्यमान है तभी
जब $\sum_x \abs{f(x)}\,\P(X = x) <
\infty$, और तब $\E(f(X)) = \sum_x f(x)\,\P(X = x)$।
\end{theorem}

\begin{proof}
$\Omega$ को स्तर समुच्चयों $\Omega_x = \{X = x\}$, $x
\in X(\Omega)$, में विभाजित कीजिए। \omterm{def:b2:series:summable}{योग्य} कुलों के
लिए पुलिंदों से योग वाली प्रमेय (\cref{ch:b2:series}) से कुल $(X(\omega)\P(\{\omega\}))_\omega$ \omterm{def:b2:series:summable}{योग्य} है तभी जब
हर पुलिंदा \omterm{def:b2:series:summable}{योग्य} हो (जो स्वतः है: $\sum_{\omega \in \Omega_x}\abs{x}\P(\{\omega\}) =
\abs x\,\P(X = x)$) \emph{और} पुलिंदा-योगों का
कुल $\bigl(x\,\P(X = x)\bigr)_x$ \omterm{def:b2:series:summable}{योग्य} हो --- और तब कुल योग मेल खाते हैं। $f(X)$ के लिए: सिद्ध
कथन को चर $Y =
f \circ X$ पर लगाइए, जिसके स्तर समुच्चय $\{Y = y\} =
\bigsqcup_{x : f(x) = y}\{X = x\}$ हैं; और फिर
पुलिंदों से दूसरा योग $\sum_y y\,\P(Y = y)$ को $\sum_x
f(x)\,\P(X = x)$ में बदल देता है, जहाँ पुलिंदे
अब मानों $x$ को उनके प्रतिबिंब $f(x)$ के अनुसार समूहित करते हैं,
और एक कुल की निरपेक्ष योग्यता दूसरे की योग्यता के तुल्य है।
\end{proof}

\begin{theorem}[प्रत्याशा के गुणधर्म]\label{thm:b2:randomvar:linearity}
\omterm{def:b2:randomvar:expectation}{प्रत्याशा} रखने वाले यादृच्छिक चरों के समुच्चय पर:
\begin{enumerate}
\item (रैखिकता) $\E(aX + bY) = a\,\E(X) + b\,\E(Y)$।
\item (धनात्मकता तथा एकदिष्टता) $X \geq 0 \Rightarrow \E(X)
\geq 0$; $X \leq Y \Rightarrow \E(X) \leq \E(Y)$; और $\abs{\E(X)} \leq \E(\abs X)$।
\item (प्रभुत्व) यदि $\abs X \leq Z$ हो और $Z$ की \omterm{def:b2:randomvar:expectation}{प्रत्याशा} विद्यमान हो, तो
$X$ की भी है।
\end{enumerate}
\end{theorem}

\begin{proof}
ये सब \omterm{def:b2:series:summable}{योग्य} कुलों के योगों के गुणधर्म हैं (\cref{ch:b2:series}): योग की रैखिकता,
पद-दर-पद धनात्मकता, तथा योग्यता की प्रभुत्व कसौटी। (ध्यान दीजिए कि
$\Omega$ पर दी गई \emph{परिभाषा} पर रैखिकता तत्काल है, जबकि अंतरण सूत्र
पर वह भद्दी पड़ती --- $\E$ को ऊपर की ओर परिभाषित करने का एक लाभ यही
है।)
\end{proof}

\begin{example}\label{ex:b2:randomvar:expectations}
$X \sim \mathcal{B}(n, p)$: $X = X_1 + \dots + X_n$ को बर्नूली सूचकों के योग के रूप में लिखकर और रैखिकता का
उपयोग करके $\E(X) = np$ --- किसी द्विपद गुणांक की ज़रूरत नहीं। $X \sim \mathcal{G}(p)$: $\E(X) =
\sum_{k\geq1}k(1-p)^{k-1}p = p\cdot\frac{1}{(1 - (1-p))^2} =
\frac1p$, जो
ज्यामितीय श्रेणी का उसकी चक्रिका के भीतर अवकलन करने से मिलता है
(\cref{ch:b2:powerseries})। $X \sim \mathcal{P}(\lambda)$: $\E(X)
= \sum_{k\geq1}k e^{-\lambda}\frac{\lambda^k}{k!} = \lambda
e^{-\lambda}\sum_{j\geq0}\frac{\lambda^j}{j!} = \lambda$।
\end{example}

\begin{example}[अंतरण, काम करता हुआ]\label{ex:b2:randomvar:transferex}
$X \sim \mathcal P(\lambda)$ के लिए $\E\bigl(\frac1{1+X}\bigr)$ परिकलित कीजिए --- $\frac1{1+X}$ का \omterm{def:b2:randomvar:law}{नियम} स्वयं भद्दा है, पर
अंतरण उसे कभी माँगता ही नहीं:
\[
\E\Bigl(\frac1{1+X}\Bigr)
= \sum_{k\geq0}\frac{1}{k+1}\,\eu^{-\lambda}
\frac{\lambda^k}{k!}
= \frac{\eu^{-\lambda}}{\lambda}\sum_{k\geq0}
\frac{\lambda^{k+1}}{(k+1)!}
= \frac{\eu^{-\lambda}}{\lambda}\bigl(\eu^\lambda - 1\bigr)
= \frac{1 - \eu^{-\lambda}}{\lambda} .
\]
दो सीख। परिकलन की दृष्टि से: किसी खिसकी हुई चरघातांकी श्रेणी को
पहचान लेना ही पूरा काम है --- अंतरण $f(X)$ की प्रत्याशाओं को श्रेणी
के हेर-फेर तक सिमटा देता है। संरचना की दृष्टि से: भोला
प्रतिस्थापन-मान $\frac1{1 + \E X} =
\frac1{1 + \lambda}$ होता, और सच्चा उत्तर उससे बड़ा है,
\[
\frac{1 - \eu^{-\lambda}}{\lambda} \geq
\frac{1}{1 + \lambda},
\]
ठीक वैसा ही जैसा उत्तल फलन $t
\mapsto \frac1{1+t}$ के लिए जेनसन असमिका माँगती है। \omterm{def:b2:affine:convex}{उत्तल}
प्रतिबिंबों की प्रत्याशाएँ भोले प्रतिस्थापन-मान से ऊपर बैठती हैं,
और अंतरण के साथ कोई श्रेणी-जाँच उस अमूर्त असमिका को मूर्त बना देती
है।
\end{example}

\begin{theorem}[स्वतंत्रता और गुणनफल]\label{thm:b2:randomvar:product}
\omterm{def:b2:randomvar:law}{यादृच्छिक चर} $X, Y$ \emph{\omterm{def:b2:proba:independence}{स्वतंत्र}} हैं यदि सब $x, y$ के लिए $\P(X = x, Y =
y) = \P(X = x)\P(Y = y)$ हो ---
तुल्य रूप से, सब $A,
B$ के लिए घटनाएँ $\{X \in A\}$ तथा $\{Y \in B\}$ \omterm{def:b2:proba:independence}{स्वतंत्र} हों। यदि
$X$ और $Y$ \omterm{def:b2:randomvar:expectation}{प्रत्याशा} रखने वाले \omterm{def:b2:proba:independence}{स्वतंत्र} वास्तविक चर हों, तो $XY$ की
\omterm{def:b2:randomvar:expectation}{प्रत्याशा} विद्यमान है और
\[
\E(XY) = \E(X)\,\E(Y) .
\]
\end{theorem}

\begin{proof}
दोनों सूत्रीकरणों की तुल्यता $(x, y) \in A \times B$ पर \omterm{def:b2:funcseq:def}{बिंदुवार} सर्वसमिका को जोड़ने
से निकलती है ($\sigma$-योज्यता दो बार)। गुणनफल के लिए: द्विक कुल $\bigl(xy\,\P(X = x)\P(Y = y)\bigr)_{(x,y)}$
\omterm{def:b2:series:summable}{योग्य} है, क्योंकि कुलों के लिए फ़ूबिनी (\cref{ch:b2:series}) से
\[
\sum_{x, y}\abs x \abs y\,\P(X{=}x)\P(Y{=}y)
= \Bigl(\sum_x \abs x \P(X{=}x)\Bigr)
\Bigl(\sum_y \abs y \P(Y{=}y)\Bigr) < \infty ;
\]
और स्वतंत्रता से यह कुल ठीक $\bigl(xy\,\P(X = x, Y =
y)\bigr)$ है, जिसका योग चर $(X, Y) \mapsto xy$ पर लगाए गए
अंतरण से $\E(XY)$ है; फ़ूबिनी फिर उस अचिह्नित योग का मूल्यांकन गुणनफल
$\E(X)\E(Y)$ के रूप में कर देती है।
\end{proof}

\begin{example}[गुणनफल, स्वतंत्रता के साथ और
बिना]\label{ex:b2:randomvar:productcontrast}
दो निष्पक्ष पासे फेंकिए। यदि $Y$ दूसरा पासा हो (जो पहले से
\omterm{def:b2:proba:independence}{स्वतंत्र} है), तो $\E(XY) = \E(X)\E(Y) = 3.5^2 = 12.25$। और यदि इसके बदले $Y = X$ हो (पासे का स्वयं से
``गुणनफल''), तो
\[
\E(X^2) = \frac{1 + 4 + 9 + 16 + 25 + 36}{6} = \frac{91}{6}
\approx 15.17 \neq 12.25 :
\]
अर्थात् दोनों परिदृश्यों में सीमांत \omterm{def:b2:randomvar:law}{नियम} वही, संयुक्त \omterm{def:b2:randomvar:law}{नियम} भिन्न,
और गुणनफल-प्रत्याशाएँ भिन्न। खुदवा लेने \omterm{def:b2:series:summable}{योग्य} सार: $\E(XY)$ \emph{युग्म}
का फलनिक है, दोनों सीमांतों का नहीं --- और अंतराल $\E(X^2) - \E(X)^2 \approx 2.92$ ठीक
कोनिग--हाइगेंस से पासे का \omterm{def:b2:randomvar:variance}{प्रसरण} $\frac{35}{12}$ है।
\end{example}

\section{प्रसरण, सहप्रसरण, और शास्त्रीय असमिकाएँ}

\begin{definition}[आघूर्ण, प्रसरण]\label{def:b2:randomvar:variance}
$X$ \emph{कोटि 2 का आघूर्ण} रखता है यदि $X^2$ की \omterm{def:b2:randomvar:expectation}{प्रत्याशा} विद्यमान
हो (तब प्रभुत्व से $X$ की भी: $\abs X \leq \frac{1 +
X^2}{2}$)। तब उसका
\emph{प्रसरण}\index{प्रसरण} तथा \emph{मानक
विचलन}\index{मानक विचलन} हैं
\[
V(X) = \E\bigl((X - \E(X))^2\bigr)
= \E(X^2) - \E(X)^2 ,
\qquad
\sigma(X) = \sqrt{V(X)} ,
\]
(दूसरा रूप --- \emph{कोनिग--हाइगेंस} सूत्र --- वर्ग खोलकर और
रैखिकता का उपयोग करके मिलता है:
\[
\E\bigl((X - \E X)^2\bigr)
= \E\bigl(X^2 - 2X\,\E X + \E(X)^2\bigr)
= \E(X^2) - 2\,\E(X)^2 + \E(X)^2 ,
\]
जहाँ बीच वाले पद में इसका उपयोग हुआ कि $\E X$ कोई अचर है)। द्वितीय
आघूर्ण रखने वाले $X, Y$ के लिए \emph{सहप्रसरण}\index{सहप्रसरण} है
\[
\operatorname{Cov}(X, Y)
= \E\bigl((X - \E X)(Y - \E Y)\bigr)
= \E(XY) - \E(X)\E(Y) .
\]
\end{definition}

\begin{theorem}[प्रसरण की औज़ार-पेटी]\label{thm:b2:randomvar:variancerules}
द्वितीय आघूर्ण रखने वाले चरों के लिए:
\begin{enumerate}
\item $V(aX + b) = a^2\,V(X)$;
\item $V(X + Y) = V(X) + V(Y) + 2\operatorname{Cov}(X, Y)$, और अधिक व्यापक रूप से
\[
V\Bigl(\sum_{i=1}^n X_i\Bigr)
= \sum_{i=1}^n V(X_i)
+ 2\sum_{i < j}\operatorname{Cov}(X_i, X_j) ;
\]
\item यदि $X, Y$ \omterm{def:b2:proba:independence}{स्वतंत्र} हों, तो $\operatorname{Cov}(X, Y) = 0$ (विलोम असत्य है), अतः \omterm{def:b2:proba:independence}{स्वतंत्र}
चरों के \omterm{def:b2:randomvar:variance}{प्रसरण} जुड़ जाते हैं।
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1} तथा \emph{2} वर्गों के प्रसार और रैखिकता हैं; गुणनफल $X_iX_j$
की प्रत्याशाएँ नीचे दी कोशी--श्वार्ज़ से (या $\abs{X_iX_j} \leq \frac{X_i^2 + X_j^2}{2}$ से) विद्यमान हैं।
\emph{3} केंद्रित चरों पर लगाई गई \cref{thm:b2:randomvar:product} है। विलोम का मानक
प्रतिउदाहरण: $\{-1, 0, 1\}$ पर एकसमान $X$ तथा $Y = X^2$ असहसंबद्ध हैं ($\E(XY) = \E(X^3) = 0 = \E X \cdot \E Y$) पर
स्पष्टतः परतंत्र।
\end{proof}

\begin{theorem}[मार्कोव तथा चेबिशेव
असमिकाएँ]\label{thm:b2:randomvar:markov}
\leavevmode
\begin{enumerate}
\item (मार्कोव) यदि $X \geq 0$ की \omterm{def:b2:randomvar:expectation}{प्रत्याशा} विद्यमान हो, तो हर $a
> 0$ के
लिए:
\[
\P(X \geq a) \leq \frac{\E(X)}{a} .
\]
\item (चेबिशेव) यदि $X$ द्वितीय आघूर्ण रखता हो, तो हर $\varepsilon > 0$ के लिए:
\[
\P\bigl(\abs{X - \E(X)} \geq \varepsilon\bigr)
\leq \frac{V(X)}{\varepsilon^2} .
\]
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} \omterm{def:b2:funcseq:def}{बिंदुवार} $a\,\mathbf{1}_{X \geq a} \leq X$ (उस \omterm{def:b2:proba:space}{घटना} पर जहाँ बायाँ पक्ष $a \leq X$ है; उससे
बाहर $0 \leq X$)। \omterm{def:b2:randomvar:expectation}{प्रत्याशा} लीजिए: एकदिष्टता से $a\,\P(X \geq a) \leq \E(X)$ तथा $\E(\mathbf{1}_A) = \P(A)$।
\emph{2.} अऋणात्मक चर $(X - \E
X)^2$ पर स्तर $a = \varepsilon^2$ पर मार्कोव लगाइए: \omterm{def:b2:proba:space}{घटना} $\{(X - \E X)^2
\geq \varepsilon^2\}$
ठीक $\{\abs{X - \E X} \geq
\varepsilon\}$ है।
\end{proof}

\begin{example}[असहसंबद्ध, पर आपस में चिपके
हुए]\label{ex:b2:randomvar:sumdiff}
दो निष्पक्ष पासे फेंकिए, $X$ तथा $Y$ \omterm{def:b2:proba:independence}{स्वतंत्र}, और $S = X +
Y$, $D = X - Y$ रखिए।
\omterm{def:b2:randomvar:variance}{सहप्रसरण} की द्विरैखिकता से
\[
\operatorname{Cov}(S, D) = V(X) - V(Y) +
\operatorname{Cov}(Y, X) - \operatorname{Cov}(X, Y) = V(X) -
V(Y) = 0 :
\]
अर्थात् योग तथा अंतर असहसंबद्ध हैं। \omterm{def:b2:proba:independence}{स्वतंत्र}? निश्चित रूप से नहीं:
$S = 12$ $D = 0$ को बाध्य कर देता है, जबकि बिना प्रतिबंधन के $\P(D = 0) =
\frac16$। सहसंबंध
किसी परतंत्रता के केवल \emph{रैखिक} भाग की जाँच करता है; यहाँ
परतंत्रता इस प्रतिबंध से ढोई जाती है कि $S$ तथा $D$ की सम-विषमता
एक ही हो, जो \omterm{def:b2:randomvar:variance}{सहप्रसरण} को अदृश्य है। (इस युग्म के लिए शून्य \omterm{def:b2:randomvar:variance}{सहप्रसरण}
को $V(X) = V(Y)$ चाहिए था: काम स्वतंत्रता ने नहीं, समान बंटनों ने किया।)
\end{example}

\begin{example}[मार्कोव कब यथातथ होती है]\label{ex:b2:randomvar:markovsharp}
मार्कोव असमिका ठीक तब समता होती है जब परिबंध $a\,\mathbf 1_{X\geq a} \leq X$ में कुछ भी बर्बाद
न हो: चर को केवल मान $0$ तथा $a$ ही लेने चाहिए। मूर्त रूप में, यदि
$\P(X = a) = \pi$ तथा $\P(X = 0) = 1 - \pi$, तो $\E(X)
= a\pi$ और
\[
\P(X \geq a) = \pi = \frac{\E(X)}{a} .
\]
यथार्थ पाठ: जिस जनसंख्या में औसत संपत्ति $100$ हो और संपत्ति या तो
$0$ हो या $10^6$, वहाँ करोड़पतियों का अनुपात ठीक $10^{-4}$ है ---
मार्कोव का परिबंध, अधिकतम असमानता से ठीक-ठीक छुआ हुआ। जब भी $X$
मध्यवर्ती मानों पर फैलता है, परिबंध कठोर होता है, प्रायः बहुत ही
अधिक; पर जैसा यह चरम स्थिति दिखाती है, अकेले माध्य से इससे बेहतर
कोई असमिका नहीं निकाली जा सकती।
\end{example}

\begin{example}[चेबिशेव तीखी है --- और कोई अतिरिक्त परिकल्पना
नहीं]\label{ex:b2:randomvar:chebsharp}
$\varepsilon > 0$, $q \in \intoc01$ स्थिर कीजिए, और मान लीजिए $X$ मान $\pm\varepsilon$ हर एक प्रायिकता
$\frac q2$ के साथ लेता है तथा $0$ प्रायिकता $1 - q$ के साथ। तब $\E(X) = 0$, $V(X) =
q\varepsilon^2$, और
\[
\P\bigl(\abs{X - \E X} \geq \varepsilon\bigr) = q
= \frac{V(X)}{\varepsilon^2} :
\]
अर्थात् चेबिशेव में समता। अतः केवल \omterm{def:b2:randomvar:variance}{प्रसरण} के उपयोग से असमिका सुधारी
नहीं जा सकती --- $1/\varepsilon^2$ क्षय द्वितीय-आघूर्ण सूचना का ठीक-ठीक मूल्य
है। तेज़ क्षय के लिए प्रबल परिकल्पनाएँ चाहिए: चर का परिबद्ध होना
\emph{चरघातांकी} संकेंद्रण ख़रीद लेता है, जैसा \cref{exo:b2:randomvar:7} झलक दिखाता है
और इस अध्याय की सप्ताहांत समस्या क्रमबद्ध रूप से विकसित करती है।
\end{example}

\begin{theorem}[कोशी--श्वार्ज़ और जेनसन]\label{thm:b2:randomvar:jensen}
\leavevmode
\begin{enumerate}
\item (कोशी--श्वार्ज़) यदि $X, Y$ द्वितीय आघूर्ण रखते हों, तो $XY$ की
\omterm{def:b2:randomvar:expectation}{प्रत्याशा} विद्यमान है और $\E(XY)^2 \leq \E(X^2)\,\E(Y^2)$; फलस्वरूप $\operatorname{Cov}(X,Y)^2 \leq V(X)V(Y)$।
\item (जेनसन) यदि $\varphi \colon I \to \R$ $X(\Omega)$ को समाहित करने वाले किसी अंतराल पर \omterm{def:b2:affine:convex}{उत्तल}
हो, और $X$, $\varphi(X)$ \omterm{def:b2:randomvar:expectation}{प्रत्याशा} रखते हों, तो
\[
\varphi\bigl(\E(X)\bigr) \leq \E\bigl(\varphi(X)\bigr) .
\]
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} $XY$ की योग्यता: $\abs{XY} \leq \frac{X^2 + Y^2}2$। प्रतिचित्रण $(X, Y) \mapsto \E(XY)$ द्वितीय आघूर्ण
वाले चरों की समष्टि पर कोई धनात्मक \omterm{def:b2:quadratic:def}{सममित द्विरैखिक रूप} है, अतः \cref{ch:b2:quadratic}
की अमूर्त कोशी--श्वार्ज़ असमिका लागू होती है (असमिका के लिए धनात्मक
\emph{अर्ध}-निश्चित होना पर्याप्त है)। उसे केंद्रित चरों पर लगाने
से \omterm{def:b2:randomvar:variance}{सहप्रसरण} परिबंध मिल जाता है।

\emph{2.} पहले, $m = \E(X)$ $I$ में पड़ता है: $I$ $X$ के सारे मानों को
समाहित करने वाला कोई अंतराल है, और \omterm{def:b2:randomvar:expectation}{प्रत्याशा} एकदिष्ट है, अतः $m$
$\inf X(\Omega)$ तथा $\sup X(\Omega)$ के बीच है। उत्तल फलनों की आधार-रेखा प्रमेय (\cref{ch:b2:realfun}) से
ऐसे $\alpha, \beta$ हैं कि सब $t \in I$ के लिए $\varphi(t) \geq \alpha t + \beta$ और $\varphi(m) = \alpha m + \beta$। तब $\Omega$ पर \omterm{def:b2:funcseq:def}{बिंदुवार}
$\varphi(X) \geq \alpha X + \beta$; \omterm{def:b2:randomvar:expectation}{प्रत्याशा} लेने पर,
\[
\E\bigl(\varphi(X)\bigr) \geq \alpha\,\E(X) + \beta
= \varphi\bigl(\E(X)\bigr). \qedhere
\]
\end{proof}

\begin{example}
$\varphi(t) = t^2$ वाली जेनसन $\E(X)^2 \leq \E(X^2)$ देती है --- यानी \omterm{def:b2:randomvar:variance}{प्रसरण} की धनात्मकता; और $\intoo{0}{\infty}$
पर $\varphi(t) = 1/t$ के साथ: $\frac{1}{\E X} \leq \E\bigl(\frac1X\bigr)$ --- अर्थात् हरात्मक माध्य समांतर माध्य से नीचे
है, अब यादृच्छिक रूप में।
\end{example}

\begin{remark}[सामान्य चूकें]
(क) $\E(XY) = \E(X)\E(Y)$ को स्वतंत्रता \emph{चाहिए} (या कम से कम शून्य \omterm{def:b2:randomvar:variance}{सहप्रसरण}):
$Y = X$ लेने पर जब भी $V(X) > 0$ हो तब $\E(X^2) \neq
\E(X)^2$ मिलता है। (ख) इसी प्रकार $V(X + X) =
4V(X)$, $2V(X)$
नहीं: \omterm{def:b2:randomvar:variance}{प्रसरण} केवल \omterm{def:b2:proba:independence}{स्वतंत्र} (या असहसंबद्ध) पदों पर ही जुड़ते हैं।
(ग) $\E(f(X))$ $f(\E(X))$ नहीं है; \omterm{def:b2:affine:convex}{उत्तल} $f$ के लिए जेनसन तो त्रुटि की दिशा भी
बता देती है, जैसा \cref{ex:b2:randomvar:transferex} में। (घ) अस्तित्व सचमुच एक परिकल्पना है:
$\P(K = k) = 2^{-k}$ ($k \geq 1$) वाले सेंट पीटर्सबर्ग चर $X = 2^K$ के लिए
\[
\sum_{k\geq1}2^k\cdot2^{-k} = \sum_{k\geq1}1 = \infty :
\]
$X$ लगभग निश्चित रूप से परिमित है फिर भी उसकी कोई \omterm{def:b2:randomvar:expectation}{प्रत्याशा} नहीं,
और खेल के लिए कोई न्यायसंगत प्रवेश-मूल्य विद्यमान नहीं है। $\E$ की
परिभाषा में योग्यता लेखा-जोखा की पंडिताई नहीं है --- वहीं भारी
पुच्छ पकड़ी जाती है। (ङ) अंत में, अंतरण प्रमेय को मानों पर योग के
किसी भी पुनर्विन्यास से पहले \emph{निरपेक्ष} योग्यता चाहिए (\cref{ch:b2:series})।
\end{remark}


\begin{example}[सौ उछालों पर
चेबिशेव]\label{ex:b2:randomvar:hundredtosses}
$X \sim \mathcal B(100, \frac12)$ के लिए: $\E X = 50$, $V(X) =
25$। $\varepsilon = 6$ के साथ चेबिशेव:
\[
\P(45 \leq X \leq 55) = \P(\abs{X - 50} < 6)
\geq 1 - \frac{25}{36} \approx 0.31 ,
\]
जबकि सटीक द्विपद योग $\approx 0.73$ देता है। आश्वस्त $31\%$ सच्चाई से बहुत दूर
है, पर उसे \emph{केवल} माध्य तथा \omterm{def:b2:randomvar:variance}{प्रसरण} चाहिए थे --- वही प्रमाणपत्र
$\E = 50$, $V = 25$ वाले किसी भी चर पर, चाहे वह कितना ही विचित्र हो, अक्षरशः
लागू होता है, और \cref{ex:b2:randomvar:chebsharp} दिखाता है कि कोई ऐसा चर उसे संतृप्त भी कर
देता है। सार्वभौमिकता की क़ीमत होती है; जब \omterm{def:b2:randomvar:law}{बंटन} सचमुच द्विपद हो, तब
सप्ताहांत समस्या के चरघातांकी औज़ार अधिकांश अंतर पाट देते हैं।
\end{example}

\begin{example}[किसी भाग का अपने पूर्ण के साथ
सहसंबंध]\label{ex:b2:randomvar:partwhole}
\omterm{def:b2:randomvar:variance}{प्रसरण} $\sigma^2 > 0$ वाले \omterm{def:b2:proba:independence}{स्वतंत्र} समरूप बँटे $X, Y$ के लिए कोई एक पद योग $S = X + Y$
के साथ कितना सहसंबद्ध है? परिकलित कीजिए
\[
\operatorname{Cov}(X, S) = \operatorname{Cov}(X, X) +
\operatorname{Cov}(X, Y) = \sigma^2 + 0 = \sigma^2,
\qquad V(S) = 2\sigma^2,
\]
अतः सहसंबंध गुणांक
\[
\rho(X, S) = \frac{\operatorname{Cov}(X,
S)}{\sigma(X)\,\sigma(S)}
= \frac{\sigma^2}{\sigma\cdot\sigma\sqrt2}
= \frac{1}{\sqrt2} \approx 0.707 ,
\]
है, चाहे साझा \omterm{def:b2:randomvar:law}{नियम} कोई भी हो --- पासे, सिक्के, प्वासों गणनाएँ।
$n$ पदों के साथ वही परिकलन $\rho(X_1, S_n) =
1/\sqrt n$ देता है: अर्थात् कुल पर हर
अलग-अलग पद का प्रभाव वर्गमूल की तरह पतला होता जाता है, जो
उतार-चढ़ावों के $\sqrt n$ मापक्रम की सहसंबंधीय छाया है। कोशी--श्वार्ज़
सदा $\abs\rho \leq 1$ आश्वस्त करती है; यहाँ परिबंध अपभ्रष्ट स्थिति $n = 1$ में
ठीक-ठीक छुआ जाता है और उसके बाद अनुमेय ढंग से क्षीण होता है।
\end{example}

\begin{example}[जेनसन से भारित
समांतर--गुणोत्तर माध्य]\label{ex:b2:randomvar:amgm}
मान लीजिए $Y$ धनात्मक मान $a_1, \dots, a_k$ प्रायिकताओं $\lambda_1, \dots, \lambda_k$ के साथ लेता है। फलन
$-\ln$ $\intoo0\infty$ पर \omterm{def:b2:affine:convex}{उत्तल} है, अतः जेनसन $-\ln\E(Y) \leq \E(-\ln Y)$ देती है, अर्थात्
\[
a_1^{\lambda_1}a_2^{\lambda_2}\cdots a_k^{\lambda_k}
\;\leq\; \lambda_1a_1 + \lambda_2a_2 + \dots + \lambda_ka_k :
\]
यानी भारित समांतर--गुणोत्तर असमिका, जिसमें समता तभी होती है जब $Y$
अचर हो। बराबर भार $\lambda_i = \frac1k$ शास्त्रीय समांतर--गुणोत्तर असमिका पुनः दे
देते हैं। प्रायिकता ने चुपचाप कोई विशुद्ध बीजगणितीय प्रमेय सिद्ध कर
दी: किसी प्रायिकता \omterm{def:b2:randomvar:law}{नियम} को चुनना उत्तल संचयों के लिए महज़ एक
लेखा-यंत्र है --- यानी \cref{ch:b2:affine} का बैरिकेंद्रीय दृष्टिकोण फिर से, अब
जेनसन को इंजन बनाकर।
\end{example}

\section{बृहत् संख्याओं का दुर्बल नियम}

\begin{theorem}[बृहत् संख्याओं का दुर्बल नियम]\label{thm:b2:randomvar:lln}
मान लीजिए $(X_k)_{k \geq 1}$ एक ही \omterm{def:b2:randomvar:law}{नियम} वाले युग्मशः \omterm{def:b2:proba:independence}{स्वतंत्र} \omterm{def:b2:randomvar:law}{यादृच्छिक चर} हैं,
जो द्वितीय आघूर्ण रखते हैं; $m = \E(X_1)$ तथा $S_n = X_1 + \dots + X_n$ लिखिए। तब हर $\varepsilon > 0$ के लिए:
\[
\P\Bigl(\,\Bigl|\frac{S_n}{n} - m\Bigr| \geq \varepsilon\Bigr)
\;\leq\; \frac{V(X_1)}{n\,\varepsilon^2}
\xrightarrow[n \to \infty]{} 0 .
\]
\end{theorem}

\begin{proof}
रैखिकता से $\E(S_n/n) = m$; \cref{thm:b2:randomvar:variancerules} से (युग्मशः स्वतंत्रता सहप्रसरणों को मार देती
है) $V(S_n) = n\,V(X_1)$, अतः $V(S_n/n) =
V(X_1)/n$। $S_n/n$ पर लगाई गई चेबिशेव असमिका परिबंध दे देती है।
\end{proof}

\begin{remark}
यही वह प्रमेय है जो प्रायिकता को बारंबारता से जोड़ती है: \omterm{def:b2:proba:independence}{स्वतंत्र}
पुनरावृत्तियों में किसी \omterm{def:b2:proba:space}{घटना} $A$ के सूचक $X_k$ के लिए $S_n/n$ $A$ की
प्रेक्षित बारंबारता है, और बृहत् संख्याओं का नियम कहता है कि वह
दर $\frac{p(1-p)}{n\varepsilon^2}$ से $\P(A)$ के आसपास संकेंद्रित होती है। \emph{प्रबल} \omterm{def:b2:randomvar:law}{नियम} ($S_n/n \to
m$
लगभग निश्चित रूप से) वर्ष 3 की प्रमेय है --- फिर भी चतुर्थ आघूर्णों
के लिए उसकी उपपत्ति पहुँच के भीतर है: देखिए \cref{exo:b2:randomvar:9}, जो चेबिशेव-प्रकार
के परिबंध पर बोरेल--कांतेली चलाता है। यही चेबिशेव आकलन \cref{ch:b2:funcseq} में
वाइरश्ट्रास सन्निकटन प्रमेय की बर्नस्टाइन-बहुपद उपपत्ति को भी चला
रहा था --- वहाँ की गणना-प्रमेयिका भेस बदले हुए बृहत् संख्याओं का
दुर्बल नियम ही \emph{थी}।
\end{remark}

\begin{example}[पचास कूपन इकट्ठे करना]
\label{ex:b2:randomvar:couponnumeric}
$n =
50$ अलग-अलग खिलौनों वाला \cref{exo:b2:randomvar:3} का कूपन संग्राहक: प्रत्याशित कुल
\[
\E(T_{50}) = 50\,H_{50} = 50\sum_{k=1}^{50}\frac1k
\approx 50 \times 4.499 \approx 225
\]
डिब्बे हैं --- भोले अनुमान $50$ का साढ़े चार गुना। पूरी कहानी
हरात्मक वृद्धि की है: पहले $25$ खिलौने लगभग $50\ln2 \approx 35$ डिब्बों में आ जाते
हैं, जबकि अकेला \emph{अंतिम} खिलौना औसतन $50$ डिब्बे माँगता है
(प्राचल $\frac1{50}$ की कोई ज्यामितीय प्रतीक्षा)। पूर्ति-समस्याओं पर उनके
अंतिम चरण का प्रभुत्व रहता है, और इसीलिए \cref{exo:b2:randomvar:12} माध्य $n\ln n$ के आसपास
कोटि $n$ के --- यानी उसी अंतिम ज्यामितीय प्रतीक्षा के आकार के ---
उतार-चढ़ाव पाता है।
\end{example}

\begin{example}[$n$ कितना बड़ा होना चाहिए?]\label{ex:b2:randomvar:llnnumeric}
प्रेक्षित बारंबारता को विश्वास $95\%$ के साथ $\P(A)$ के $\varepsilon = 0.01$ के भीतर बाँधने
के लिए चेबिशेव का परिबंध माँगता है
\[
\frac{p(1-p)}{n\varepsilon^2} \leq \frac{1}{4n\varepsilon^2}
\leq 0.05,
\qquad\text{अर्थात्}\qquad
n \geq \frac{1}{4\cdot0.05\cdot(0.01)^2} = 50\,000 .
\]
निर्भरता $\varepsilon$ में क्रूर है (वर्गिक) और विश्वास में नरम ($1/\alpha$ में
रैखिक)। दोनों लक्षण \emph{परिबंध} के गुणधर्म हैं, सच्चाई के नहीं:
सप्ताहांत समस्या की चरघातांकी असमिकाएँ विश्वास की क़ीमत $1/\alpha$ से
$\ln(1/\alpha)$ तक घटा देती हैं --- वहाँ वही विनिर्देश लगभग $18\,500$ प्रतिदर्श
माँगेगा --- जबकि $1/\varepsilon^2$ मापक्रम असली और असुधार्य है। किसी परिबंध का
कौन सा भाग ढीला है, यह जानना स्वयं परिबंध जितना ही उपयोगी है।
\end{example}

\begin{remark}[इस खंड के भीतर के परिप्रेक्ष्य]
आगे की ओर, यहाँ का सब कुछ \cref{ch:b2:genfun} को खिलाता है: $X$ के किसी चतुर फलन
की \omterm{def:b2:randomvar:expectation}{प्रत्याशा} $\E(t^X)$ पूरे \omterm{def:b2:randomvar:law}{नियम} को किसी घात श्रेणी में भर देती है,
आघूर्ण $1$ पर अवकलज बन जाते हैं, और यादृच्छिक योगों के लिए
वाल्ड-प्रकार की सर्वसमिकाएँ शाखन-प्रक्रम सिद्धांत ढोती हैं;
\omterm{def:b2:proba:independence}{स्वतंत्र} चरों के लिए गुणनफल प्रमेय जनक फलनों की गुणनात्मकता बन जाती
है। पीछे की ओर, \omterm{def:b2:randomvar:expectation}{प्रत्याशा} प्रायिकता-भारों वाला कोई \omterm{def:b2:affine:barycenter}{बैरिकेंद्र} है
(\cref{ch:b2:affine}), जेनसन असमिका उत्तल फलनों की आधार-रेखा ज्यामिति है (\cref{ch:b2:realfun}),
और इस अध्याय की सप्ताहांत समस्या की चरघातांकी-आघूर्ण विधि $\eu^{tX}$ पर
लगाई गई मार्कोव है --- एक असमिका, एक अच्छे चर-परिवर्तन से उन्नत,
तीन अध्यायों में फैली हुई।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:1}
$X \sim \mathcal{B}(n, p)$ (सूचकों के द्वारा), $X \sim \mathcal{P}(\lambda)$ (दिखाइए $V(X) =
\lambda$), तथा $X \sim \mathcal{G}(p)$ (दिखाइए $V(X) =
\frac{1-p}{p^2}$;
$\E(X(X-1))$ तथा ज्यामितीय श्रेणी का द्वितीय अवकलज बरतिए) के लिए $\E(X)$ तथा
$V(X)$ परिकलित कीजिए।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:2}
मान लीजिए $X \sim \mathcal{P}(\lambda)$ तथा $Y \sim \mathcal{P}(\mu)$ \omterm{def:b2:proba:independence}{स्वतंत्र} हैं। दिखाइए कि $X + Y \sim \mathcal{P}(\lambda + \mu)$ (भारों का
संवलन; द्विपद प्रमेय), और यह कि $X + Y = n$ दिए जाने पर $X$ का सप्रतिबंध
\omterm{def:b2:randomvar:law}{नियम} द्विपद $\mathcal{B}\bigl(n, \frac{\lambda}{\lambda + \mu}\bigr)$ है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:3}
(कूपन संग्राहक, \omterm{def:b2:randomvar:expectation}{प्रत्याशा}) कोई अनाज-ब्रांड हर डिब्बे में $n$ अलग
खिलौनों में से एक \omterm{def:b2:funcseq:def}{एकसमान रूप से} छिपाता है। मान लीजिए $T_n$ सारे $n$
खिलौने इकट्ठे करने के लिए आवश्यक डिब्बों की संख्या है। $T_n$ को
\omterm{def:b2:proba:independence}{स्वतंत्र} ज्यामितीय चरों के योग के रूप में लिखकर (जब $k$ अब भी शेष
हों तब किसी \emph{नए} खिलौने को देखने का समय), दिखाइए
\[
\E(T_n) = n\sum_{k=1}^{n}\frac{1}{k} \sim n\ln n
\]
(\cref{ch:b2:comparison} की श्रेणी--समाकल तुलना से तुल्य)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:4}
मान लीजिए $X \geq 0$ पूर्णांक-मान वाला है। $X =
\sum_{n\geq1}\mathbf{1}_{X \geq n}$ लिखकर और योगों की अदला-बदली
करके (अऋणात्मक कुलों के लिए फ़ूबिनी) \emph{पुच्छ सूत्र}
\[
\E(X) = \sum_{n=1}^{\infty} \P(X \geq n)
\]
सिद्ध कीजिए (जब कोई भी पक्ष परिमित हो)। ज्यामितीय \omterm{def:b2:randomvar:law}{नियम} के लिए $\E(X) = \frac1p$
पुनः प्राप्त कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:5}
(बिना प्रतिस्थापन प्रतिचयन अधिक संकेंद्रित है) किसी कलश में $N$
गेंदें हैं, जिनमें $M$ सफ़ेद हैं। बिना प्रतिस्थापन $n \leq N$ खींचिए और
मान लीजिए $X$ सफ़ेदों की गिनती करता है (\emph{अतिज्यामितीय} \omterm{def:b2:randomvar:law}{नियम})।
$Y_i$ को $i$-वीं खींच मानकर सूचकों $X = \sum_{i=1}^n Y_i$ का उपयोग करते हुए: दिखाइए कि
हर $Y_i$ प्राचल $p = M/N$ का बर्नूली है (सममिति!), निष्कर्ष निकालिए कि
प्रतिस्थापन वाली स्थिति की तरह ही ठीक-ठीक $\E(X) = np$, और दिखाइए कि $i
\neq j$
के लिए $\operatorname{Cov}(Y_i, Y_j) = -\frac{p(1-p)}{N-1} < 0$, अतः $V(X) = np(1-p)\frac{N - n}{N - 1} \leq np(1-p)$।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:6}
मान लीजिए $X$ द्वितीय आघूर्ण रखता है। दिखाइए कि $c \mapsto \E\bigl((X -
c)^2\bigr)$ ठीक $c = \E(X)$ पर
न्यूनतम है, और न्यूनतम मान $V(X)$। फिर दिखाइए कि $\P(X = \E(X)) = 1$ तभी जब $V(X) =
0$।
\emph{(दूसरे बिंदु के लिए: यदि $V(X) = 0$, तो $\varepsilon = 1/n$ के साथ चेबिशेव तथा
एकदिष्ट संततता \cref{thm:b2:proba:continuity} बरतिए।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:7}
(संकेंद्रण मार्कोव को हरा देता है) मान लीजिए $S_n \sim \mathcal{B}(n,
\frac12)$ ($n$ निष्पक्ष
उछालों में चित्तों की संख्या)। मार्कोव ($\P(S_n \geq \frac{3n}{4})$), चेबिशेव, तथा
चरघातांकी (चेर्नोफ़) विधि
\[
\P\Bigl(S_n \geq \frac{3n}4\Bigr)
\leq \E\bigl(e^{tS_n}\bigr)e^{-3nt/4}
= \Bigl(\frac{1 + e^t}{2}\Bigr)^n e^{-3nt/4}
\quad (t > 0),
\]
से मिले परिबंधों की तुलना कीजिए, और चरघातांकी रूप से छोटा परिबंध
पाने के लिए $t$ का इष्टतमीकरण कीजिए। \emph{($t = \ln 3$ पर: $\bigl(2\cdot 3^{-3/4}\bigr)^n \approx
(0.877)^n$ परिबद्ध
कीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:8}
(वाइरश्ट्रास फिर, प्रायिकतात्मक रूप से) मान लीजिए $f \colon [0,1] \to \R$ \omterm{def:b2:metric:continuity}{संतत} है और
$S_n \sim \mathcal{B}(n, x)$। दिखाइए कि \omterm{thm:b2:funcseq:weierstrass}{बर्नस्टाइन बहुपद} $B_nf(x) = \sum_{k=0}^n f\bigl(\frac
kn\bigr)\binom nk x^k(1-x)^{n-k}$ $\E\bigl[f\bigl(\frac{S_n}{n}\bigr)\bigr]$ के बराबर है, और \cref{ch:b2:funcseq} का
आकलन $\abs{B_nf(x) - f(x)} \leq \omega_f(\delta) +
\frac{2\norm f_\infty}{4n\delta^2}$ इसी प्रायिकतात्मक भाषा में फिर से व्युत्पन्न कीजिए ($\bigl|\frac{S_n}{n} -
x\bigr| \geq \delta$
पर बाँटिए और चेबिशेव बरतिए)।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:9}
(चतुर्थ आघूर्णों के अंतर्गत प्रबल नियम) मान लीजिए $(X_k)$ \omterm{def:b2:proba:independence}{स्वतंत्र},
समरूप बँटे, केंद्रित ($\E X_1 = 0$) चर हैं, जहाँ $\E(X_1^4) < \infty$। $\E(S_n^4)$ को खोलकर और बचे
हुए पदों को गिनकर (केवल $\E(X_i^4)$ तथा $\E(X_i^2X_j^2)$ वाले पद, $i
\neq j$) दिखाइए कि किसी
अचर $C$ के साथ $\E(S_n^4) \leq C n^2$। हर $\varepsilon > 0$ के लिए
$\sum_n \P\bigl(\abs{S_n/n} \geq \varepsilon\bigr) < \infty$
निकालिए (कोटि 4 पर मार्कोव) और बोरेल--कांतेली (\cref{thm:b2:proba:borelcantelli}) से निष्कर्ष
निकालिए कि किसी उपयुक्त सूत्रीकरण के अनुदिश लगभग निश्चित रूप से
$S_n/n \to 0$: अर्थात् \omterm{def:b2:proba:space}{घटना} $\bigcap_{j}\bigcup_N\bigcap_{n \geq N}\{\abs{S_n/n} <
\frac1j\}$ की प्रायिकता $1$ है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:10}
दो निष्पक्ष पासे फेंके जाते हैं; मान लीजिए $M$ दोनों परिणामों में
से बड़ा है। \cref{exo:b2:randomvar:4} के पुच्छ सूत्र (परिमित रूप) का उपयोग करते हुए
दिखाइए
\[
\E(M) = \sum_{k=1}^{6}\P(M \geq k)
= 6 - \sum_{j=0}^5\Bigl(\frac j6\Bigr)^2 = \frac{161}{36}
\approx 4.47 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:11}
मान लीजिए $F_n$ $\{1, \dots, n\}$ के किसी एकसमान यादृच्छिक क्रमचय के अचल बिंदुओं
की संख्या है ($n \geq 2$)। $F_n =
\sum_i\mathbf 1_{\sigma(i) = i}$ लिखकर $\E(F_n) = 1$ परिकलित कीजिए, $i \neq j$ के लिए $\operatorname{Cov}(\mathbf 1_{\sigma(i)=i}, \mathbf
1_{\sigma(j)=j}) = \frac1{n^2(n-1)}$,
और निष्कर्ष निकालिए $V(F_n) = 1$: अर्थात् औसतन एक चिट्ठी अचल रहती है, और
\omterm{def:b2:randomvar:variance}{प्रसरण} ठीक $1$ है, चाहे $n$ कुछ भी हो।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:12}
(कूपन संग्राहक, संकेंद्रण) \cref{exo:b2:randomvar:3} के परिवेश में ज्यामितीय चरणों की
स्वतंत्रता तथा $V(\mathcal
G(p)) = \frac{1-p}{p^2}$ (\cref{exo:b2:randomvar:1}; मान $\pi^2/6$ \cref{ex:b2:fourier:basel} है) का उपयोग करते हुए दिखाइए
\[
V(T_n) = \sum_{k=1}^n\frac{1 - k/n}{(k/n)^2}
\leq n^2\sum_{k=1}^n\frac{1}{k^2} \leq \frac{\pi^2}{6}n^2,
\]
चेबिशेव से निकालिए कि \emph{प्रायिकता में} $\dfrac{T_n}{n\ln n} \to 1$: अर्थात् संग्राहक का
कुल समय कोटि $n$ के उतार-चढ़ावों तक $n\ln n$ है।
\end{exercise}

\section{समस्या: संकेंद्रण की औज़ार-पेटी, मार्कोव से हॉफडिंग तक}

\begin{problem}[{सप्ताहांत समस्या --- हाथ से चरघातांकी संकेंद्रण, और किसी
मतसर्वेक्षण को कितने लोगों से पूछना चाहिए}]
\label{pb:b2:randomvar:1}
मार्कोव असमिका एक आघूर्ण की क़ीमत लेती है और $1/a$ क्षय ख़रीदती है;
चेबिशेव दो आघूर्णों की क़ीमत लेती है और $1/\varepsilon^2$ ख़रीदती है --- और \cref{ex:b2:randomvar:chebsharp}
दिखाता है कि उतने आघूर्ण इससे अधिक ख़रीद ही नहीं सकते। यह समस्या
सीढ़ी का शेष भाग चढ़ती है: सिक्का-उछालों के लिए अपनी \emph{सटीक}
दर वाली चरघातांकी (चेर्नोफ़) विधि, सारे परिबद्ध चरों के लिए
हॉफडिंग असमिका, और प्रतिफल --- मतसर्वेक्षणों, चुनाव-घोषणाओं तथा
सिक्का-परीक्षण के लिए स्पष्ट, ईमानदार प्रतिदर्श-आकार।
\index{संकेंद्रण असमिकाएँ}\index{हॉफडिंग
असमिका}\index{चेर्नोफ़ परिबंध} सर्वत्र $S_n \sim
\mathcal B(n, p)$ $n$ \omterm{def:b2:proba:independence}{स्वतंत्र} बर्नूली
चरों का योग है और $\widehat p_n = S_n/n$ आनुभविक बारंबारता।

\medskip
\noindent\textbf{भाग I --- निष्पक्ष सिक्के पर अंशांकन।}
यहाँ $p = \frac12$ तथा $a \in \intoo{\frac12}{1}$।
\begin{enumerate}
  \item स्तर $an$ पर मार्कोव: दिखाइए $\P(S_n \geq an) \leq
        \frac1{2a}$, जो ऐसा परिबंध है जो $0$
        की ओर जाता भी नहीं। मार्कोव इतना कहाँ खो देती है?
  \item चेबिशेव: $n/2$ के परितः निष्पक्ष द्विपद की सममिति का
        उपयोग करते हुए दिखाइए
        \[
        \P(S_n \geq an) = \tfrac12\,
        \P\bigl(\abs{S_n - \tfrac n2} \geq n(a -
        \tfrac12)\bigr)
        \leq \frac{1}{8n(a - 1/2)^2},
        \]
        अर्थात् $a = \frac34$ पर $\frac2n$: आख़िर बहुपदीय क्षय।
  \item (चेर्नोफ़, व्यापक स्तर) $\E(\eu^{tS_n}) = \bigl(\frac{1 + \eu^t}2\bigr)^n$ परिकलित कीजिए और $t > 0$ पर $\P(S_n \geq an) \leq
        \E(\eu^{tS_n})\eu^{-tan}$
        का इष्टतमीकरण कीजिए: दिखाइए कि इष्टतम $t$ $\ln\frac{a}{1-a}$ है और
        \[
        \P(S_n \geq an) \leq \eu^{-n\,I(a)},
        \qquad
        I(a) = \ln 2 + a\ln a + (1-a)\ln(1-a) > 0 .
        \]
        जाँचिए कि $a = \frac34$ \cref{exo:b2:randomvar:7} का परिबंध $\bigl(2\cdot3^{-3/4}\bigr)^n$ पुनः दे देता है।
  \item (घातांक सटीक है) मान लीजिए $k = an$ कोई पूर्णांक है। इस
        तथ्य से कि $\binom nk a^k(1-a)^{n-k}$ किसी प्रायिकता \omterm{def:b2:randomvar:law}{बंटन} के $n + 1$ पदों में
        सबसे बड़ा है, $H(a) = -a\ln a -
        (1-a)\ln(1-a)$ के साथ $\binom nk \geq
        \frac{\eu^{nH(a)}}{n+1}$ सिद्ध कीजिए, और मेल
        खाता निचला परिबंध
        \[
        \P(S_n \geq an) \geq \binom{n}{an}2^{-n}
        \geq \frac{\eu^{-n\,I(a)}}{n + 1} .
        \]
        निकालिए।
  \item $n = 100$, $a =
        \frac34$ पर तीनों परिबंधों की सारणी बनाइए: मार्कोव
        $\frac23$, चेबिशेव $0.02$, चेर्नोफ़ $\approx 2.1\cdot10^{-6}$ (सच्चा मान $\approx 2.8\cdot10^{-7}$
        है)। एक वाक्य में सार?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- हॉफडिंग असमिका।}
\begin{enumerate}[resume]
  \item (रादेमाखर स्थिति) हर एक प्रायिकता $\frac12$ वाले $\varepsilon = \pm1$ के लिए
        दोनों श्रेणियों की पद-दर-पद तुलना करके ($(2k)! \geq
        2^kk!$) सिद्ध
        कीजिए
        \[
        \E(\eu^{t\varepsilon}) = \cosh t \leq \eu^{t^2/2}
        \qquad (t \in \R)
        \]
  \item \omterm{def:b2:proba:independence}{स्वतंत्र} रादेमाखर चरों $\varepsilon_1, \dots, \varepsilon_n$ तथा हर $s >
        0$ के लिए निकालिए:
        \[
        \P\Bigl(\sum_{i=1}^n\varepsilon_i \geq s\Bigr)
        \leq \eu^{-s^2/(2n)} .
        \]
  \item निष्पक्ष सिक्कों में अनुवाद कीजिए ($X_i =
        \frac{1+\varepsilon_i}2$): $\P\bigl(\widehat p_n -
        \tfrac12 \geq \delta\bigr) \leq \eu^{-2n\delta^2}$, और गुणक
        $2$ वाला द्विपक्षीय रूप।
  \item (हॉफडिंग की प्रमेयिका) मान लीजिए $\E X =
        p$ वाला $X \in \intcc01$ है, और
        $\psi(t) = \ln\E(\eu^{tX})$। न्यायसंगत ठहराइए कि $\psi$ दो बार \omterm{def:b2:diffcalc:differential}{अवकलनीय} है, जहाँ
        \[
        \psi''(t) = \E_t(X^2) - \E_t(X)^2, \qquad
        \E_t(Y) := \frac{\E(Y\eu^{tX})}{\E(\eu^{tX})},
        \]
        जो अब भी $\intcc01$ में मान लेने वाले किसी पुनःभारित चर का
        \emph{\omterm{def:b2:randomvar:variance}{प्रसरण}} है; उसे $\frac14$ से परिबद्ध कीजिए (\cref{exo:b2:randomvar:6} का
        न्यूनतमता तर्क) और टेलर से निष्कर्ष निकालिए:
        \[
        \E\bigl(\eu^{t(X - p)}\bigr) \leq \eu^{t^2/8} .
        \]
  \item (हॉफडिंग असमिका) साझा माध्य $p$ वाले \omterm{def:b2:proba:independence}{स्वतंत्र} $X_i \in
        \intcc01$ के
        लिए निकालिए
        \[
        \P\bigl(\abs{\widehat p_n - p} \geq \delta\bigr)
        \leq 2\,\eu^{-2n\delta^2}
        \qquad (\delta > 0).
        \]
  \item चेबिशेव दर $\frac{p(1-p)}{n\delta^2}$ की हॉफडिंग की $2\eu^{-2n\delta^2}$ से तुलना कीजिए: हर
        एक को कौन सी परिकल्पना चाहिए, और $\delta = 0.03$, $p =
        \frac12$ पर लगभग
        किस $n$ से चरघातांकी परिबंध जीतने लगता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- किसी मतसर्वेक्षण को कितने लोगों से
पूछना चाहिए?} कोई मतसर्वेक्षण $n$ \omterm{def:b2:proba:independence}{स्वतंत्र}, \omterm{def:b2:funcseq:def}{एकसमान रूप से} चुने गए
मतदाताओं से पूछता है; हर एक ईमानदारी से उत्तर देता है; $p$ सच्चा
अंक है और $\widehat p_n$ सर्वेक्षण का आँकड़ा।
\begin{enumerate}[resume]
  \item दिखाइए कि मतसर्वेक्षण विश्वास $1 - \alpha$ के साथ $\pm\delta$ तक सटीक
        है (अर्थात् $\P(\abs{\widehat p_n - p} \geq \delta) \leq \alpha$), जैसे ही
        \[
        n \;\geq\; \frac{\ln(2/\alpha)}{2\,\delta^2} .
        \]
  \item मानक ``तीन अंक, पंचानबे प्रतिशत'' विनिर्देश ($\delta
        = 0.03$, $\alpha = 0.05$)
        के लिए आवश्यक $n$ परिकलित कीजिए: $n \geq 2050$; और एक अंक के
        लिए: $n \geq 18\,445$। इस चकित कर देने वाले तथ्य पर ध्यान दीजिए ---
        और उसे समझाइए --- कि उत्तर में जनसंख्या का आकार आता ही
        नहीं।
  \item प्रश्न 13 को चेबिशेव ($V(X_1) = p(1-p)
        \leq \frac14$) के साथ फिर कीजिए: तीन अंकों
        पर $n \geq \frac1{4\alpha\delta^2} =
        5556$। ध्यान दीजिए कि \emph{बिना} प्रतिस्थापन
        प्रतिचयन केवल मदद ही करता है (\cref{exo:b2:randomvar:5}: \omterm{def:b2:randomvar:variance}{प्रसरण} $\frac{N-n}{N-1}$ से
        सिकुड़ जाता है)।
  \item (चुनाव की घोषणा) किसी प्रत्याशी का सच्चा अंक $p
        = 0.52$ है।
        कितने मतदाताओं से पूछा जाना चाहिए ताकि $\P(\widehat p_n \leq \tfrac12) \leq 0.01$? दिखाइए
        $n
        \geq \frac{\ln 100}{2\cdot(0.02)^2} \approx 5757$ --- अर्थात् किसी काँटे की टक्कर की घोषणा किसी अंक
        के आकलन से कहीं अधिक महँगी पड़ती है।
  \item गणित जिसे ढँकता \emph{नहीं}: बरती गई प्रतिरूपण-मान्यताएँ
        सूचीबद्ध कीजिए (\omterm{def:b2:proba:independence}{स्वतंत्र} एकसमान प्रतिचयन, ईमानदार
        उत्तर, नियत $p$), और किसी छोटे अनुच्छेद में समझाइए कि
        असली सर्वेक्षण-त्रुटियों पर \emph{अभिनति} (असमान
        प्रतिचयन, उत्तर न देना) का प्रभुत्व क्यों रहता है,
        जिसे $n$ बढ़ाकर घटाया नहीं जा सकता।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- अधिक तीखा और अधिक सस्ता।}
\begin{enumerate}[resume]
  \item (माध्यों की माध्यिका: दो आघूर्णों से चरघातांकी क्षय)
        $km$ प्रतिदर्शों के बजट को $m$ आकार के $k$ \omterm{def:b2:proba:independence}{स्वतंत्र}
        समूहों में बाँटिए; मान लीजिए $\widehat p^{(1)}, \dots, \widehat
        p^{(k)}$ समूह-माध्य हैं और $M$
        उनकी माध्यिका। $m$ इस प्रकार चुनिए कि हर समूह $\P(\abs{\widehat p^{(i)} - p} \geq \delta) \leq
        \frac18$ पूरा
        करे (चेबिशेव: $m \geq \frac2{\delta^2}$ पर्याप्त है)। दिखाइए कि यदि $\abs{M - p} \geq \delta$ हो,
        तो कम से कम $k/2$ समूह चूकते हैं, और निकालिए
        \[
        \P(\abs{M - p} \geq \delta)
        \leq \binom{k}{\lceil k/2\rceil}\Bigl(\frac18
        \Bigr)^{k/2}
        \leq 2^k\cdot 8^{-k/2} = 2^{-k/2} :
        \]
        अर्थात् प्रसरणों से आगे कुछ भी बरते बिना चरघातांकी
        संकेंद्रण।
  \item (पेली--ज़िगमुंड) द्वितीय आघूर्ण वाले $X \geq 0$ के लिए $\P(X > 0) \geq \dfrac{\E(X)^2}{\E(X^2)}$
        सिद्ध कीजिए \emph{($X\mathbf 1_{X>0}$ पर कोशी--श्वार्ज़)}: यह उलटी
        दिशा का औज़ार है --- आघूर्ण \omterm{def:b2:proba:space}{घटनाओं} को घटित होने पर बाध्य
        भी कर सकते हैं।
  \item (हल्का पिंस्कर) $\intoo{\frac12}1$ पर $I(a) \geq 2\bigl(a -
        \tfrac12\bigr)^2$ दिखाइए \emph{(अंतर $\frac12$
        पर द्वितीय कोटि तक लुप्त होता है और उसका द्वितीय
        अवकलज $\frac1{a(1-a)} - 4 \geq
        0$ है)}: अर्थात् चेर्नोफ़ का सटीक घातांक
        हॉफडिंग के वर्गिक घातांक को सदा हरा देता है।
  \item $I\bigl(\tfrac12 + \delta\bigr) = 2\delta^2 +
        O(\delta^4)$ का प्रसार कीजिए और प्रश्न 4 के साथ जोड़िए: छोटे
        विचलनों के लिए हॉफडिंग घातांक $2n\delta^2$ अनंतस्पर्शी रूप से
        \emph{सटीक} है --- कोई भी विधि उसे बहुपदीय गुणकों से
        अधिक नहीं हरा सकती।
  \item औज़ार-पेटी की सारणी बनाइए: मार्कोव, चेबिशेव, \cref{exo:b2:randomvar:9} का
        चतुर्थ-आघूर्ण परिबंध, हॉफडिंग, तथा घातांक $I$ वाला
        चेर्नोफ़ --- हर एक के लिए एक पंक्ति में बताइए: आवश्यक
        परिकल्पना, प्राप्त क्षय, और इस समस्या का वह प्रश्न जहाँ
        वह सबसे तीखा था।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- लाभांश।}
\begin{enumerate}[resume]
  \item (सिक्के की जाँच) कोई सिक्का या तो निष्पक्ष है या $p = 0.55$
        के साथ पक्षपाती। आप उसे $n$ बार उछालते हैं और $\widehat p_n > 0.525$ होने
        पर ``पक्षपाती'' घोषित करते हैं। दिखाइए कि दोनों
        त्रुटि-प्रायिकताएँ अधिक से अधिक $\eu^{-2n(0.025)^2}$ हैं, और यह कि $n \geq 3685$
        उछाल दोनों को $1\%$ से नीचे आश्वस्त कर देते हैं।
  \item (विरल \omterm{def:b2:proba:space}{घटनाओं} को प्रसरण-सजग परिबंध चाहिए) मान लीजिए $p =
        0.01$
        और सापेक्ष विनिर्देश $\delta =
        p/2 = 0.005$, $\alpha = 0.05$ लीजिए। हॉफडिंग ($n \approx 74\,000$)
        तथा सच्चे \omterm{def:b2:randomvar:variance}{प्रसरण} $p(1-p)$ वाली चेबिशेव ($n
        \approx 7920$) द्वारा
        माँगे गए प्रतिदर्श-आकारों की तुलना कीजिए: अर्थात्
        प्रसरण-अंधा चरघातांकी परिबंध विनम्र द्वितीय आघूर्ण से
        हार जाता है। सार बताइए, और यह भी कि छूटा हुआ औज़ार (कोई
        प्रसरण-सजग चरघातांकी परिबंध; \cref{ch:b2:genfun} का प्वासों
        सन्निकटन) कहाँ से आएगा।
  \item (सिक्कों के लिए प्रबल नियम) $\sum_n
        2\eu^{-2n\delta^2} < \infty$ तथा बोरेल--कांतेली
        (\cref{thm:b2:proba:borelcantelli}) से सिद्ध कीजिए कि \omterm{def:b2:proba:independence}{स्वतंत्र} सिक्का-उछालों के लिए
        लगभग निश्चित रूप से $\widehat p_n \to p$: लगभग-निश्चित \omterm{def:b2:proba:space}{घटना} को \cref{exo:b2:randomvar:9}
        की तरह $\bigcap_j\bigcup_N\bigcap_{n\geq N}
        \{\abs{\widehat p_n - p} < \tfrac1j\}$ के रूप में सूत्रबद्ध कीजिए, और निष्कर्ष
        निकालिए। (वहाँ बरते गए चतुर्थ आघूर्ण की जगह यहाँ
        परिबद्धता ले लेती है।)
  \item संश्लेषण। पाँच वाक्यों में: सीढ़ी का हर पायदान (आघूर्ण
        एक, दो, चार; परिबद्ध चरघातांकी; सटीक घातांक) क्या
        क़ीमत लेता है और क्या ख़रीदता है; किसी भी आकार के देश
        के लिए $2050$ लोगों का सर्वेक्षण क्यों पर्याप्त है; और
        इनमें से किस परिबंध को वर्ष 3 का खंड केंद्रीय सीमा
        प्रमेय के सटीक अचरों में तीखा करेगा।
\end{enumerate}
\end{problem}
