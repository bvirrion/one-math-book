\chapter{प्रायिकता जनक फलन}\label{ch:b2:genfun}

\cref{ch:b2:powerseries} की घात श्रेणियाँ किसी प्रायिकतात्मक मिशन के साथ लौटती हैं: किसी
$\N$-मान वाले \omterm{def:b2:randomvar:law}{यादृच्छिक चर} से हम गुणांकों $\P(X = n)$ वाली घात श्रेणी जोड़
देते हैं। यह \emph{\omterm{ex:b2:powerseries:fibonacci}{जनक फलन}} \omterm{def:b2:proba:independence}{स्वतंत्र} चरों के योगों को गुणनफलों में,
आघूर्णों को $1$ पर अवकलजों में, और कठिन साहचर्यिक सर्वसमिकाओं को
एक-पंक्ति के गुणनों में बदल देता है। अध्याय पुस्तक को दो प्रदर्शनों
के साथ बंद करता है: विरल \omterm{def:b2:proba:space}{घटनाओं} का प्वासों सन्निकटन, और शाखन
प्रक्रमों की विलोपन कसौटी --- सचमुच अपरिमित एक प्रायिकतात्मक परिकलन,
जो पूरी तरह किसी उत्तल वक्र की ज्यामिति से हल हो जाता है।

\section{परिभाषा और मूल गुणधर्म}

\begin{definition}[प्रायिकता जनक फलन]\label{def:b2:genfun:pgf}
मान लीजिए $X$ कोई $\N$-मान वाला \omterm{def:b2:randomvar:law}{यादृच्छिक चर} है, $p_n = \P(X = n)$। $X$ का
\emph{प्रायिकता जनक फलन}\index{प्रायिकता जनक फलन} घात श्रेणी
\[
G_X(t) = \E\bigl(t^X\bigr) = \sum_{n=0}^{\infty} p_n\,t^n .
\]
का योग है।
\end{definition}

\begin{example}[पहले प्रतिवर्त]\label{ex:b2:genfun:firstreflexes}
किसी अचर चर $X = c$ का $G_X(t) = t^c$ होता है; किसी खिसकाव के लिए $G_{X+c}(t) = t^c\,G_X(t)$; और विशेष
बिंदुओं पर मूल्यांकन बिना किसी प्रसार के सूचना पढ़ लेता है: $G_X(0) = \P(X
= 0)$,
$G_X(1) = 1$, तथा $G_X(-1) = \P(X\text{ सम}) -
\P(X\text{ विषम})$, अर्थात् वही
सम-विषमता संतुलन जिसका \cref{exo:b2:genfun:10} में उपयोग हुआ। ये एक-पंक्ति वाले तथ्य
नीचे सर्वत्र चुपचाप बरते जाते हैं --- और मूल्यांकन $G_X(0)$ ठीक वही है
जिससे अध्याय के अंत में पुनरावृत्त जनक फलनों से विलोपन प्रायिकताएँ
निकाली जाएँगी।
\end{example}

\begin{proposition}[त्रिज्या और पहले गुणधर्म]\label{prop:b2:genfun:radius}
$G_X$ को परिभाषित करने वाली श्रेणी की \omterm{def:b2:powerseries:radius}{अभिसरण त्रिज्या} $\geq 1$ है; $G_X$
$\intcc{-1}{1}$ पर परिभाषित तथा \omterm{def:b2:metric:continuity}{संतत} है, $\intoo{-1}{1}$ पर $\mathcal{C}^\infty$ है, जहाँ $G_X(1) = 1$ तथा $\abs{G_X(t)} \leq 1$।
इसके अतिरिक्त $G_X$ $X$ का \omterm{def:b2:randomvar:law}{नियम} निर्धारित कर देता है:
\[
p_n = \frac{G_X^{(n)}(0)}{n!} .
\]
\end{proposition}

\begin{proof}
चूँकि $\sum p_n = 1$ अभिसरित होता है, पद $p_n\,1^n$ परिबद्ध हैं, अतः त्रिज्या $\geq 1$
है (आबेल की प्रमेयिका, \cref{ch:b2:powerseries}); $t = \pm1$ पर श्रेणी निरपेक्षतः अभिसरित
होती है ($\sum p_n = 1$ प्रभुत्व करता है); और इससे भी बेहतर, पूरे अंतराल $\intcc{-1}1$
पर
\[
\sup_{\abs t\leq1}\,\abs{p_nt^n} = p_n
\quad\text{जहाँ}\quad \sum_np_n < \infty :
\]
अर्थात् श्रेणी $\intcc{-1}1$ पर \emph{प्रसामान्य} रूप से अभिसरित होती है,
इसलिए उसका योग वहाँ \omterm{def:b2:metric:continuity}{संतत} है (\cref{thm:b2:funcseq:weierstrass,thm:b2:funcseq:continuity})। भीतर की चिकनाई तथा गुणांक सूत्र
घात श्रेणियों का सामान्य सिद्धांत है; और गुणांक पुनः प्राप्य होने
के कारण एक ही \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} वाले दो चरों का \omterm{def:b2:randomvar:law}{नियम} एक ही होता है।
\end{proof}

\begin{example}[शास्त्रीय नियम]\label{ex:b2:genfun:classical}
\leavevmode
\begin{itemize}
\item बर्नूली $\mathcal{B}(p)$: $G(t) = 1 - p + pt$।
\item द्विपद $\mathcal{B}(n, p)$: $G(t) = \sum_k \binom nk
(pt)^k(1-p)^{n-k} = (1 - p + pt)^n$ (द्विपद प्रमेय)।
\item ज्यामितीय $\mathcal{G}(p)$: $G(t) =
\sum_{k\geq1}(1-p)^{k-1}p\,t^k = \dfrac{pt}{1 - (1-p)t}$ (त्रिज्या $\frac{1}{1-p} > 1$)।
\item प्वासों $\mathcal{P}(\lambda)$: $G(t) =
\sum_k e^{-\lambda}\frac{(\lambda t)^k}{k!} = e^{\lambda(t - 1)}$ (त्रिज्या $\infty$)।
\end{itemize}
\end{example}

\begin{example}[जनक फलन का समाकलन]\label{ex:b2:genfun:integraltrick}
$1$ पर $G_X$ के अवकलज धनात्मक आघूर्ण देते हैं; \emph{समाकल} कोई
ऋणात्मक आघूर्ण देता है। $\int_0^1t^k\dd t
= \frac1{k+1}$ तथा पद-दर-पद समाकलन ($\intcc01$ पर प्रसामान्य
अभिसरण) से:
\[
\int_0^1G_X(t)\,\dd t = \sum_{k\geq0}\frac{\P(X =
k)}{k+1} = \E\Bigl(\frac1{1+X}\Bigr).
\]
$X \sim \mathcal P(\lambda)$ के लिए:
\[
\E\Bigl(\frac1{1+X}\Bigr) =
\int_0^1\eu^{\lambda(t-1)}\,\dd t = \frac{1 -
\eu^{-\lambda}}{\lambda},
\]
जिससे \cref{ex:b2:randomvar:transferex} का श्रेणी-परिकलन एक पंक्ति में पुनः मिल जाता है। \omterm{ex:b2:powerseries:fibonacci}{जनक फलन}
दोतरफ़ा यंत्र है: आघूर्णों $\E(X)$, $\E(X(X-1))$ के लिए $1$ पर अवकलन कीजिए,
$\E\bigl(\frac1{1+X}\bigr)$ के लिए $\intcc01$ पर समाकलन --- एक ही \omterm{def:b2:powerseries:analytic}{वैश्लेषिक} वस्तु, जिससे उसी
दिशा में पूछा जाता है जिसमें समस्या को चाहिए।
\end{example}

\begin{example}[ठीक एक त्रिज्या वाला कोई
नियम]\label{ex:b2:genfun:heavytail}
मान लीजिए $k \geq 1$ के लिए $\P(X = k) = \dfrac{6}{\pi^2k^2}$ --- जो बासेल सर्वसमिका (\cref{ex:b2:fourier:basel}) से कोई
प्रायिकता \omterm{def:b2:randomvar:law}{नियम} है। उसके \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} $G(t) = \frac6{\pi^2}\sum_{k\geq1}\frac{t^k}{k^2}$ की \omterm{def:b2:powerseries:radius}{अभिसरण त्रिज्या} ठीक $1$ है:
अर्थात् \cref{prop:b2:genfun:radius} का व्यापक परिबंध ``त्रिज्या $\geq
1$'' सुधारा नहीं जा सकता।
और माध्य
\[
\sum_{k\geq1}k\,\P(X = k) =
\frac6{\pi^2}\sum_{k\geq1}\frac1k = \infty :
\]
है। $G$ $\intcc{-1}1$ पर \omterm{def:b2:metric:continuity}{संतत} है, भीतर चिकना, पर उसका अवकलज $1^-$ पर फट
पड़ता है --- ग्राफ़ बिंदु $(1, 1)$ पर किसी ऊर्ध्वाधर स्पर्शी के साथ
पहुँचता है। भारी पुच्छ \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} पर, अकेले बिंदु $t = 1$ पर,
\emph{ज्यामितीय रूप से} दिख जाती हैं; और नीचे दी आघूर्ण प्रमेय इस
संगति को ठीक-ठीक बना देती है।
\end{example}

\begin{theorem}[जनक फलन से
आघूर्ण]\label{thm:b2:genfun:moments}
$X$ की \omterm{def:b2:randomvar:expectation}{प्रत्याशा} विद्यमान है तभी जब $G_X$ $1^-$ पर \omterm{def:b2:diffcalc:differential}{अवकलनीय} हो (बायाँ
अवकलज, परिमित), और तब $\E(X) = G_X'(1)$। इसी प्रकार $X$ द्वितीय आघूर्ण रखता है
तभी जब $G_X$ $1^-$ पर दो बार \omterm{def:b2:diffcalc:differential}{अवकलनीय} हो, और तब
\[
\E\bigl(X(X - 1)\bigr) = G_X''(1),
\qquad
V(X) = G_X''(1) + G_X'(1) - G_X'(1)^2 .
\]
\end{theorem}

\begin{proof}
$t \in \intoo{0}{1}$ के लिए चक्रिका के भीतर पद-दर-पद अवकलन $G_X'(t) = \sum_{n\geq1} np_n t^{n-1}$ देता है, जो
अऋणात्मक गुणांकों वाली कोई श्रेणी है: अतः $t \mapsto G_X'(t)$ $\intoo{0}{1}$ पर अनह्रासमान है,
और आंशिक योगों के एकदिष्ट अभिसरण से (या अऋणात्मक गुणांकों के लिए
आबेल की प्रमेय, \cref{ch:b2:powerseries}),
\[
\lim_{t \to 1^-} G_X'(t)
= \sum_{n\geq1} n\,p_n \in \intcc{0}{+\infty} ,
\]
जहाँ हर पक्ष ठीक तभी परिमित है जब दूसरा परिमित हो। परिमित होने पर
माध्यमान प्रमेय अंतर-भागफलों $\frac{G_X(1) -
G_X(t)}{1 - t}$ को $G_X'$ के मानों के बीच दबा देती
है, अतः $G_X$ $1^-$ पर \omterm{def:b2:diffcalc:differential}{अवकलनीय} है और $G_X'(1) = \sum np_n = \E(X)$ (अंतरण से)। द्वितीय-कोटि का
कथन उसी तर्क को एक स्तर ऊपर दोहराता है: $G''_X(t) = \sum_{n\geq2}n(n-1)p_nt^{n-2}$ $\intoo01$ पर अनह्रासमान है
और उसकी एकदिष्ट सीमा $\sum_nn(n-1)p_n = \E(X(X-1))$ है, जो ठीक तभी परिमित है जब $X$
द्वितीय आघूर्ण रखता हो। \omterm{def:b2:randomvar:variance}{प्रसरण} सूत्र फिर कोनिग--हाइगेंस से निकलता
है:
\[
V(X) = \E(X^2) - \E(X)^2 = \E\bigl(X(X-1)\bigr) + \E(X) -
\E(X)^2 = G''_X(1) + G'_X(1) - G'_X(1)^2 .
\]
\end{proof}

\begin{example}
प्वासों: $G'(t) = \lambda e^{\lambda(t-1)}$, अतः $\E(X) = \lambda$; $G''(1) = \lambda^2$, अतः $V(X) = \lambda^2 + \lambda - \lambda^2 =
\lambda$ --- अर्थात् \cref{ch:b2:randomvar} के परिकलन,
हर एक एक पंक्ति में।
\end{example}

\begin{example}[किसी प्वासों नियम का बहुलक]
\label{ex:b2:genfun:poissonmode}
$X \sim \mathcal
P(\lambda)$ के लिए $\P(X = k)$ कहाँ सबसे बड़ा है? क्रमागत भार अनुपात
\[
\frac{\P(X = k+1)}{\P(X = k)} = \frac{\lambda}{k + 1} ,
\]
के द्वारा तुलनीय हैं, जो $k < \lambda - 1$ रहने तक $1$ से अधिक है और $k > \lambda - 1$ होते ही
$1$ से नीचे गिर जाता है: अर्थात् भार पहले चढ़ते हैं फिर गिरते हैं,
और बहुलक $\floor\lambda$ है ($\lambda$ पूर्णांक होने पर $\lambda - 1$ तथा $\lambda$ के बीच
बराबरी: $\lambda = 3$ के लिए $\P(X = 2) = \P(X = 3) = \frac92\eu^{-3} \approx 0.224$)। गुणांकों पर अनुपात परीक्षण प्रायः किसी
विविक्त \omterm{def:b2:randomvar:law}{नियम} के गुणात्मक तथ्यों तक पहुँचने का सबसे तेज़ रास्ता होते
हैं --- किसी \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} की ज़रूरत नहीं, पर गुणांक तो \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} ही
\emph{हैं}, पद-दर-पद पढ़े हुए।
\end{example}

\section{स्वतंत्र चरों के योग}

\begin{theorem}[गुणनात्मकता]\label{thm:b2:genfun:product}
यदि $X$ तथा $Y$ \omterm{def:b2:proba:independence}{स्वतंत्र} $\N$-मान वाले \omterm{def:b2:randomvar:law}{यादृच्छिक चर} हों, तो
\[
G_{X + Y}(t) = G_X(t)\,G_Y(t)
\qquad (\abs t \leq 1),
\]
और आगमन से \omterm{def:b2:proba:independence}{स्वतंत्र} $X_1, \dots, X_n$ के लिए $G_{X_1 + \dots + X_n} = \prod_i G_{X_i}$।
\end{theorem}

\begin{proof}
दो उपपत्तियाँ, दोनों शिक्षाप्रद। \emph{प्रत्याशाओं के द्वारा:}
$t^X$ तथा $t^Y$ \omterm{def:b2:proba:independence}{स्वतंत्र} परिबद्ध चर हैं, अतः (\cref{thm:b2:randomvar:product})
\[
G_{X+Y}(t) = \E\bigl(t^{X+Y}\bigr)
= \E\bigl(t^X t^Y\bigr)
= \E\bigl(t^X\bigr)\E\bigl(t^Y\bigr) .
\]
\emph{कोशी गुणनफलों के द्वारा:} $X + Y$ का \omterm{def:b2:randomvar:law}{नियम} संवलन $\P(X + Y = n) = \sum_{k=0}^n \P(X = k)\P(Y = n - k)$ है, और
निरपेक्षतः अभिसारी श्रेणियों के लिए \omterm{thm:b2:series:fubini}{कोशी गुणनफल} प्रमेय (\cref{ch:b2:series}) दोनों
घात श्रेणियों को ठीक इसी संवलन के अनुदिश गुणा कर देती है।
\end{proof}

\begin{example}[शास्त्रीय नियमों की स्थिरता]\label{ex:b2:genfun:stability}
एक ही $p$ वाले \omterm{def:b2:proba:independence}{स्वतंत्र} द्विपद जुड़ जाते हैं: $(1 - p + pt)^m(1 -
p + pt)^n = (1 - p + pt)^{m+n}$, अतः $\mathcal{B}(m, p) +
\mathcal{B}(n, p) = \mathcal{B}(m + n, p)$ ---
विशेष रूप से $n$ \omterm{def:b2:proba:independence}{स्वतंत्र} बर्नूली चरों का योग द्विपद है, जिससे
सफलताओं की संख्या का \omterm{def:b2:randomvar:law}{नियम} फिर सिद्ध हो जाता है। \omterm{def:b2:proba:independence}{स्वतंत्र} प्वासों भी
जुड़ जाते हैं: $e^{\lambda(t-1)}e^{\mu(t-1)} = e^{(\lambda + \mu)(t-1)}$, अतः $\mathcal{P}(\lambda) + \mathcal{P}(\mu) = \mathcal{P}(\lambda +
\mu)$ --- यानी \cref{exo:b2:randomvar:2} का संवलन-परिकलन, अब बिना
किसी परिकलन के।
\end{example}

\begin{example}[दो पासे, एक बहुपद का
वर्ग]\label{ex:b2:genfun:twodice}
एक निष्पक्ष पासे के लिए $G(t) = \frac{t + t^2 + \dots + t^6}{6}$; और दोनों के योग के लिए
\[
G(t)^2 = \frac{1}{36}\bigl(t^2 + 2t^3 + 3t^4 + 4t^5 + 5t^6 +
6t^7 + 5t^8 + 4t^9 + 3t^{10} + 2t^{11} + t^{12}\bigr) :
\]
अर्थात् पासा-योगों का त्रिभुजीय \omterm{def:b2:randomvar:law}{नियम} ($7$ बहुलक है, प्रायिकता $\frac6{36} = \frac16$
के साथ), जो किसी बहुपद के वर्ग से पढ़ लिया जाता है --- और उस वर्ग
को जीवन में एक ही बार खोलना पड़ता है। संवलन सूत्र से ग्यारह अलग-अलग
गणना-तर्क चाहिए होते; \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} उन्हें एक साथ कर देता है, क्योंकि
बहुपदों को गुणा करना गुणांकों का संवलन \emph{ही है}। यह यांत्रिक
अनुवाद --- नियमों से गुणांक, योगों से गुणनफल --- इस अध्याय का पूरा
व्यापार-प्रतिरूप है, और \cref{exo:b2:genfun:11} उसे चौंकाने वाले सिशरमान पासों तक
धकेल देता है।
\end{example}

\begin{example}[तीन पासे और एक गुणांक
निष्कर्षण]\label{ex:b2:genfun:threedice}
तीन निष्पक्ष पासों के योग $S$ के लिए $\P(S = 10)$ $\bigl(\frac{t + \dots +
t^6}6\bigr)^3$ में $t^{10}$ का गुणांक है।
गुणनखंडन कीजिए और द्विपद तथा ज्यामितीय श्रेणी से प्रसार कीजिए:
\[
\Bigl(\frac{t(1 - t^6)}{6(1 - t)}\Bigr)^{\!3}
= \frac{t^3}{216}\,\bigl(1 - 3t^6 + 3t^{12} -
t^{18}\bigr)\sum_{j\geq0}\binom{j+2}2t^j .
\]
$t^{10}$ के गुणांक के लिए गुणनफल से $t^7$ चाहिए: पद $1$ के साथ $j = 7$, और पद
$-3t^6$ के साथ $j = 1$:
\[
\P(S = 10) = \frac{1}{216}\Bigl(\binom92 -
3\binom32\Bigr) = \frac{36 - 9}{216} = \frac{27}{216} =
\frac18 .
\]
$27$ त्रिकों की सीधी गणना भूल-भरी है; बीजगणित यांत्रिक है और किसी
भी संख्या के पासों तक फैल जाता है --- $(1 - t^6)^3$ में दिखता
समावेश--अपवर्जन स्थिति-विश्लेषण अपने आप कर रहा है।
\end{example}

\begin{example}[किसी नियम को उसके जनक फलन से
पढ़ना]\label{ex:b2:genfun:identify}
किस \omterm{def:b2:randomvar:law}{नियम} का $G(t) = \dfrac1{2 - t}$ है? घात श्रेणी में प्रसार कीजिए:
\[
\frac{1}{2 - t} = \frac12\cdot\frac1{1 - t/2}
= \sum_{k\geq0}\frac{t^k}{2^{k+1}} :
\]
अऋणात्मक गुणांक $G(1) = 1$ तक जुड़ते हैं, अतः यह सचमुच कोई \omterm{def:b2:randomvar:law}{नियम} है, $\N$
पर $\P(X = k) = 2^{-(k+1)}$ --- यानी $0$ से शुरू होने वाला कोई ज्यामितीय \omterm{def:b2:randomvar:law}{नियम}।
अद्वितीयता (\cref{prop:b2:genfun:radius}) से कोई दूसरा \omterm{def:b2:randomvar:law}{नियम} यह $G$ साझा नहीं करता। जनक
फलनों से \omterm{def:b2:randomvar:law}{नियम} पहचानना अभ्यास \omterm{def:b2:series:summable}{योग्य} कौशल है: सप्ताहांत समस्या का
क्रांतिक शाखन पुनरावर्त $G_n(t) = \frac{n - (n-1)t}{n+1 - nt}$ इसी तरह उत्तरजीविता पर प्रतिबंधित
किसी ज्यामितीय \omterm{def:b2:randomvar:law}{नियम} के रूप में उघड़ता है।
\end{example}

\begin{remark}
स्थिरता एक ही दिशा में चलती है: \omterm{def:b2:proba:independence}{स्वतंत्र} प्वासों के योग प्वासों
होते हैं, पर अंतर नहीं --- $X - Y$ ऋणात्मक मान लेता है, अतः उसका कोई
\omterm{ex:b2:powerseries:fibonacci}{जनक फलन} है ही नहीं, और उसका \omterm{def:b2:randomvar:law}{नियम} (स्केलम \omterm{def:b2:randomvar:law}{बंटन}) इस अध्याय की
औज़ार-पेटी से बाहर पड़ता है। इसी प्रकार $p \neq p'$ वाला $\mathcal B(m, p) + \mathcal B(n, p')$ द्विपद
\emph{नहीं} है: गुणनफल $(1 - p +
pt)^m(1 - p' + p't)^n$ के मूलों के दो भिन्न स्थान हैं, जबकि हर
द्विपद \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} का एक ही पुनरावृत्त मूल होता है। मूल-प्रतिरूपों से
स्थिरता पढ़ लेना इस बात की छोटी-सी झलक है कि बहुपद कितनी संरचना
समेटे रहता है।
\end{remark}

\begin{remark}[एकत्व-मूल छननी]
$-1$ पर मूल्यांकन सम को विषम से अलग करता है; सारे $m$-वें एकत्व-मूलों
पर मूल्यांकन हर अवशेष वर्ग को अलग कर देता है: $\omega = \eu^{2\iu\pi/m}$ के साथ
\[
\P(X \equiv r \bmod m)
= \frac1m\sum_{j=0}^{m-1}\omega^{-jr}\,G_X(\omega^j),
\]
क्योंकि $j$ पर $\omega^{j(k-r)}$ का औसत लेने पर $k
\equiv r$ होने पर $1$ मिलता है और
अन्यथा $0$। नमूना लाभांश: दो निष्पक्ष पासों के योग $S$ के लिए $j \neq 0$
हेतु हर $G(\omega^j) = \frac16\sum_{k=1}^6
\omega^{jk} = -\frac16$ (सातों सातवें एकत्व-मूलों का योग शून्य है), अतः
\[
\P(7 \mid S) = \frac17\Bigl(1 +
6\cdot\frac1{36}\Bigr) = \frac16 ,
\]
जो \cref{ex:b2:genfun:twodice} की गिनती की पुष्टि करता है --- और यह विधि उन प्रश्नों तक
भी फैलती है जहाँ सीधी गणना नहीं फैलती।
\end{remark}

\begin{theorem}[यादृच्छिक योग: जनक फलनों के लिए वाल्ड की
सर्वसमिका]\label{thm:b2:genfun:compound}
मान लीजिए $(X_k)_{k\geq1}$ एक ही \omterm{def:b2:randomvar:law}{नियम} तथा \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} $G_X$ वाले \omterm{def:b2:proba:independence}{स्वतंत्र} $\N$-मान
चर हैं, और $N$ कोई $\N$-मान वाला चर है जो $X_k$ से \omterm{def:b2:proba:independence}{स्वतंत्र} है, जिसका
\omterm{ex:b2:powerseries:fibonacci}{जनक फलन} $G_N$ है। तब यादृच्छिक योग $S = X_1 + \dots + X_N$ ($N = 0$ होने पर $S = 0$) का \omterm{ex:b2:powerseries:fibonacci}{जनक फलन}
\[
G_S = G_N \circ G_X .
\]
है। विशेष रूप से, यदि $N$ तथा $X_1$ की प्रत्याशाएँ विद्यमान हों, तो
$\E(S) =
\E(N)\,\E(X_1)$।
\end{theorem}

\begin{proof}
$N$ पर प्रतिबंधित कीजिए (पूर्ण प्रायिकता, \cref{thm:b2:proba:bayes}): $\abs t \leq 1$ के लिए
\[
G_S(t) = \sum_{n=0}^\infty \P(N = n)\,
\E\bigl(t^{X_1 + \dots + X_n}\bigr)
= \sum_{n=0}^\infty \P(N = n)\,G_X(t)^n
= G_N\bigl(G_X(t)\bigr),
\]
जहाँ हर नियत $n$ के लिए गुणनात्मकता तथा पूरे द्विक कुल की योग्यता
($\abs{G_X(t)} \leq 1$) का उपयोग हुआ। योगों की अदला-बदली \omterm{def:b2:series:summable}{योग्य} कुलों के लिए फ़ूबिनी
(\cref{ch:b2:series}) है। शृंखला नियम तथा \cref{thm:b2:genfun:moments} से $1^-$ पर अवकलन करने पर: $\E(S) = G_N'(G_X(1))\,G_X'(1)
= G_N'(1)G_X'(1) = \E(N)\E(X_1)$।
\end{proof}

\begin{example}[संयुक्त प्वासों: वार्षिक बीमा-हानियाँ]\label{ex:b2:genfun:compoundpoisson}
कोई बीमाकर्ता वर्ष भर में $N \sim \mathcal P(\lambda)$ दावे पाता है, और हर दावे की क़ीमत
$X_k$ है (पूर्णांक इकाइयाँ, \omterm{def:b2:proba:independence}{स्वतंत्र} समरूप बँटे, \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} $G_X$, माध्य
$\mu$, $N$ से \omterm{def:b2:proba:independence}{स्वतंत्र})। \cref{thm:b2:genfun:compound} से कुल हानि $S$ का
\[
G_S(t) = \eu^{\lambda(G_X(t) - 1)},
\qquad
\E(S) = \lambda\mu ,
\]
है, और $1^-$ पर दो बार अवकलन करने पर:
\[
V(S) = \lambda\,G_X''(1) + \lambda^2\mu^2 + \lambda\mu -
(\lambda\mu)^2 = \lambda\,\E(X^2) .
\]
\omterm{def:b2:randomvar:variance}{प्रसरण} में किसी अकेले दावे का \emph{द्वितीय} आघूर्ण आता है, उसका
\omterm{def:b2:randomvar:variance}{प्रसरण} नहीं: कोई संयुक्त प्वासों योग कभी-कभार आने वाले बड़े दावे को
दो बार महसूस करता है --- एक बार कितने के द्वारा, एक बार कितने बड़े
के द्वारा। माध्य $2$ ($\E X^2 = 6$) वाले ज्यामितीय \omterm{def:b2:randomvar:law}{नियम} के $\lambda = 10$ दावों के
लिए: $\E S = 20$, $V(S) = 60$, और चेबिशेव (\cref{ch:b2:randomvar}) पहले ही काम लायक़
शोधन-क्षमता सीमांत दे देती है। यह ``यादृच्छिक रूप से रोका गया
योग'' प्रतिरूप वही है जो \cref{prop:b2:genfun:branching} की शाखन-पुनरावृत्ति चलाएगा: जनक
फलनों का संयोजन यादृच्छिक जनसंख्याओं का बीजगणित है।
\end{example}

\begin{remark}
$N$ का पदों से \omterm{def:b2:proba:independence}{स्वतंत्र} होना सजावट नहीं है। बराबर प्रायिकताओं के
साथ $X_k \in \{0, 2\}$ लीजिए और $N =
X_1$ रखिए (जो खुलेआम परतंत्र है): तब $S = X_1 + \dots + X_N$ $X_1 = 0$ होने
पर $0$ है, और $X_1 = 2$ होने पर $2 + X_2$, अतः $\E(S) =
\frac12(2 + 1) = \frac32$, जबकि $\E(N)\E(X_1) = 1\cdot1 =
1$: यानी वाल्ड की
सर्वसमिका विफल हो जाती है। जब पदों की संख्या को स्वयं पदों पर
\emph{प्रतिक्रिया} करने की छूट मिल जाती है, तब स्वच्छ गुणनफल संरचना
ढह जाती है --- ऐसे ``रोकन'' नियमों का पूरा सिद्धांत वर्ष 3 के खंड
का मार्टिंगेल अध्याय है।
\end{remark}

\section{प्वासों सन्निकटन}

\begin{theorem}[विरल घटनाओं का नियम]\label{thm:b2:genfun:rareevents}
मान लीजिए $n\,p_n \to \lambda > 0$ वाला $X_n \sim \mathcal{B}(n, p_n)$ है। तब हर $k \in \N$ के लिए:
\[
\P(X_n = k)
\xrightarrow[n\to\infty]{}
e^{-\lambda}\frac{\lambda^k}{k!} :
\]
अर्थात् बहुत-सी विरल \omterm{def:b2:proba:independence}{स्वतंत्र} \omterm{def:b2:proba:space}{घटनाओं} का द्विपद \omterm{def:b2:randomvar:law}{नियम} प्राचल $\lambda$ के
प्वासों \omterm{def:b2:randomvar:law}{नियम} तक अभिसरित होता है।
\end{theorem}

\begin{proof}
$p_n = \frac{\lambda_n}{n}$, $\lambda_n
\to \lambda$ के साथ सीधा परिकलन:
\[
\P(X_n = k)
= \binom nk p_n^k(1 - p_n)^{n-k}
= \frac{n(n-1)\cdots(n-k+1)}{n^k}\cdot
\frac{\lambda_n^k}{k!}\,
\bigl(1 - \tfrac{\lambda_n}{n}\bigr)^{n-k} .
\]
$k$ नियत रखकर $n \to \infty$ होने पर: पहला गुणक $1$ की ओर जाता है ($\to 1$ वाले
$k$ गुणकों का गुणनफल); $\lambda_n^k \to \lambda^k$; और $\bigl(1 - \frac{\lambda_n}{n}\bigr)^{n-k} =
\exp\bigl((n-k)\ln(1 - \frac{\lambda_n}{n})\bigr) \to
e^{-\lambda}$, क्योंकि $(n - k)\ln\bigl(1 - \frac{\lambda_n}{n}\bigr)
\sim -\lambda_n \to -\lambda$ (\cref{ch:b2:comparison})।
वैकल्पिक रूप से, जनक फलनों के स्तर पर: हर नियत $t \in [0, 1]$ के लिए $G_{X_n}(t) = \bigl(1 +
\frac{\lambda_n(t-1)}{n}\bigr)^n \to e^{\lambda(t - 1)} =
G_{\mathcal{P}(\lambda)}(t)$ ---
अर्थात् जनक फलनों का अभिसरण, जो ($\N$-मान वाले चरों के लिए) हर
$\P(X_n = k)$ के अभिसरण के तुल्य है; देखिए \cref{exo:b2:genfun:9}।
\end{proof}

\begin{remark}
इसीलिए प्वासों \omterm{def:b2:randomvar:law}{नियम} विरल \omterm{def:b2:proba:space}{घटनाओं} की गिनतियों का प्रतिरूप बनते हैं ---
प्रति पृष्ठ मुद्रण-दोष, प्रति सेकंड रेडियोसक्रिय क्षय, किसी चौराहे
पर प्रति दिन दुर्घटनाएँ: हर अवसर लगभग नगण्य है, अवसर बहुत हैं, और
सीमा में केवल माध्य दर $\lambda$ बची रहती है।
\end{remark}

\begin{example}[प्वासों सीमा को अभिसरित होते
देखना]\label{ex:b2:genfun:poissonnumerics}
$\lambda = 2$ स्थिर कीजिए और $X_n \sim \mathcal B(n, 2/n)$ लीजिए। किसी \omterm{def:b2:proba:space}{घटना} के न होने की प्रायिकता ठीक
$\P(X_n = 0) = (1 - 2/n)^n$ है:
\[
n = 10:\ 0.107, \qquad
n = 20:\ 0.122, \qquad
n = 50:\ 0.130, \qquad
n = 100:\ 0.133,
\]
सीमा $\eu^{-2} \approx 0.135$ के सामने। अभिसरण एकदिष्ट है और चाल $O(1/n)$ की --- प्रसार करने
पर $(1 -
2/n)^n = \eu^{-2}\bigl(1 - \tfrac2n + O(n^{-2})\bigr)$ --- अतः सैकड़ों में $n$ के लिए प्वासों प्रतिरूप पहले ही तीसरे
अंक तक सटीक है। विरल \omterm{def:b2:proba:space}{घटनाओं} के \omterm{def:b2:randomvar:law}{नियम} का व्यावहारिक सार यही है:
प्रतिरूपकार $n$ तथा $p$ को अलग-अलग कभी नहीं जानता (किसी पृष्ठ में
मुद्रण-दोष के कितने सूक्ष्म अवसर होते हैं?), केवल उनका गुणनफल $\lambda$
जानता है, और सीमा नियम कृपापूर्वक और किसी चीज़ पर निर्भर नहीं करता।
\end{example}

\section{शाखन प्रक्रम}

एक पूर्वज से शुरू होने वाली किसी जनसंख्या पर विचार कीजिए; हर
व्यक्ति के, \omterm{def:b2:proba:independence}{स्वतंत्र} रूप से, \omterm{def:b2:randomvar:law}{नियम} $(p_k)_{k
\in \N}$ तथा \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} $G$
(\emph{संतति \omterm{def:b2:randomvar:law}{बंटन}}) वाली यादृच्छिक संख्या में बच्चे होते हैं। मान
लीजिए $Z_n$ पीढ़ी $n$ का आकार है ($Z_0 =
1$), और $m = G'(1) = \E(Z_1)$ माध्य संतति-संख्या।

\begin{proposition}\label{prop:b2:genfun:branching}
$Z_n$ का \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} $n$-वाँ पुनरावर्त $G_{Z_n} = G \circ G \circ \dots \circ G$ है ($n$ बार), और विलोपन
प्रायिकताएँ $q_n = \P(Z_n = 0)$ पूरा करती हैं
\[
q_0 = 0, \qquad q_{n+1} = G(q_n),
\]
तथा बढ़कर अंततः विलोपन की प्रायिकता $q$ तक जाती हैं, जो $G$ का कोई
अचल बिंदु है।
\end{proposition}

\begin{proof}
पीढ़ी $n + 1$ पीढ़ी $n$ के $Z_n$ सदस्यों की संतति का यादृच्छिक योग है,
जहाँ गिनतियाँ एक-दूसरे से तथा $Z_n$ से \omterm{def:b2:proba:independence}{स्वतंत्र} हैं: \cref{thm:b2:genfun:compound} $G_{Z_{n+1}} =
G_{Z_n} \circ G$ देता है,
और $G_{Z_0}(t) = t$ से आगमन $n$-गुना पुनरावर्त दे देता है --- जिसे, संयोजन की
साहचर्यता से, समान रूप से $G_{Z_{n+1}} = G \circ G_{Z_n}$ भी पढ़ा जा सकता है। इस दूसरे रूप का
$0$ पर मूल्यांकन: $q_{n+1} = G_{Z_{n+1}}(0) =
G\bigl(G_{Z_n}(0)\bigr) = G(q_n)$। घटनाएँ $\{Z_n = 0\}$ बढ़ती हैं (विलुप्त जनसंख्या
विलुप्त ही रहती है), अतः एकदिष्ट संततता (\cref{thm:b2:proba:continuity}) से $q_n \uparrow q = \P\bigl(\bigcup_n\{Z_n = 0\}\bigr)$, और $[0, 1]$
पर $G$ की संततता सीमा में $q_{n+1} = G(q_n)$ को $q =
G(q)$ में बदल देती है।
\end{proof}

\begin{example}[विलोपन को अभिसरित होते देखना]
\label{ex:b2:genfun:cobwebnumerics}
\cref{ex:b2:genfun:branchingexample} के संतति \omterm{def:b2:randomvar:law}{नियम} $(p_0, p_1, p_2) = (\tfrac14,
\tfrac14, \tfrac12)$ के लिए $G(t) = \tfrac14 + \tfrac14t + \tfrac12t^2$, और पुनरावृत्ति $q_{n+1} = G(q_n)$ देती है
\[
q_1 = 0.25, \quad q_2 = 0.34375, \quad q_3 \approx 0.39502,
\quad q_4 \approx 0.42678, \quad q_5 \approx 0.44776,
\]
जो विलोपन प्रायिकता $q = \tfrac12$ की ओर चढ़ता है। अंतराल $q - q_n$ ये हैं: $0.25$,
$0.156$, $0.105$, $0.073$, $0.052$: हर एक पिछले का लगभग $\tfrac34$ है, और वस्तुतः
माध्यमान प्रमेय $G'(q) = \tfrac14 + q = \tfrac34$ के साथ $q - q_{n+1} = G'(c_n)(q
- q_n)$ दे देती है। दो सीख: पीढ़ी $n$
पर अब भी जीवित किसी वंश की, उसी परिकलन में अंतर्निहित, बाद में
मरने की प्रायिकता $q - q_n$ है; और नीचे के चित्र में सीढ़ी की अभिसरण
दर अचल बिंदु पर अवकलज है --- सप्ताहांत समस्या दोनों प्रेक्षणों को
प्रमेयों में बदल देती है।
\end{example}

\begin{theorem}[विलोपन कसौटी]\label{thm:b2:genfun:extinction}
मान लीजिए $p_1 \neq 1$। विलोपन प्रायिकता $q$ $\intcc{0}{1}$ में $G$ का
\emph{सबसे छोटा} अचल बिंदु है, और:
\begin{itemize}
\item यदि $m \leq 1$ (अवक्रांतिक या क्रांतिक), तो $q = 1$: विलोपन निश्चित है;
\item यदि $m > 1$ (अधिक्रांतिक), तो $q < 1$: जनसंख्या धनात्मक प्रायिकता
$1 - q$ के साथ सदा जीवित रहती है।
\end{itemize}
\end{theorem}

\begin{proof}
$G$ $\intcc{0}{1}$ पर \omterm{def:b2:affine:convex}{उत्तल} है (अऋणात्मक गुणांकों वाली घात श्रेणी: $G'' \geq 0$),
अनह्रासमान है, और $G(1) = 1$।

\emph{सबसे छोटा अचल बिंदु:} मान लीजिए $r \in \intcc{0}{1}$ कोई अचल बिंदु है। तब
$q_0 = 0 \leq r$, और आगमन से (एकदिष्टता से) $q_{n+1} = G(q_n)
\leq G(r) = r$: अतः $q = \lim q_n \leq r$।

\emph{स्थिति $m \leq 1$:} मान लीजिए $r < 1$ कोई अचल बिंदु है। $[r, 1]$ पर
माध्यमान प्रमेय से ऐसा $c \in \intoo{r}{1}$ है कि $G'(c) = \frac{G(1) - G(r)}{1 - r} = \frac{1 - r}{1 - r} = 1$। पर $G'$ अनह्रासमान है
(उत्तलता), जहाँ $\lim_{t\to1^-}G'(t) = m
\leq 1$, अतः $\intoo{0}{1}$ पर $G' \leq 1$; और तब समता $G'(c) =
1$ $G'$ को $\intco{c}{1}$
पर $1$ के बराबर अचर होने पर बाध्य कर देती है, इसलिए वहाँ $G'' = \sum n(n-1)p_nt^{n-2} \equiv 0$।
जिस अऋणात्मक-गुणांक घात श्रेणी के सारे गुणांक किसी अंतराल पर लुप्त
हों उसके ये सारे गुणांक शून्य होते हैं: $n \geq 2$ के लिए $p_n = 0$, अतः $G(t) = p_0
+ p_1t$
तथा $1 = G'(c) = p_1$ --- जो परिकल्पना $p_1 \neq 1$ के विरुद्ध है। अतः $1$ अकेला अचल
बिंदु है: $q = 1$।

\emph{स्थिति $m > 1$:} $1$ के निकट $G(t) - t$ का अवकलज $t \to 1^-$ होने पर $G'(t) -
1 \to m - 1 > 0$ है,
अतः किसी अंतराल $\intoo{1 - \delta}{1}$ पर $G(t) - t < G(1) - 1 = 0$: \omterm{def:b2:metric:continuity}{संतत} फलन $G(t) - t$ $t = 0$ पर $\geq 0$ है ($G(0) = p_0 \geq 0$) और
$1$ से ठीक नीचे $< 0$, अतः वह किसी $r < 1$ पर लुप्त हो जाता है
(मध्यमान प्रमेय)। तब सबसे छोटा अचल बिंदु $q \leq r < 1$ है।
\end{proof}

\begin{remark}[मकड़जाल कैसे पढ़ें]
चित्र में कोई ऊर्ध्वाधर चाल $G$ लगाती है ($(q_n, q_n)$ से ऊपर $(q_n, G(q_n))$ तक), और
विकर्ण तक कोई क्षैतिज चाल निर्गत को निवेश में बदल देती है: सीढ़ी
पुनरावृत्ति $q_{n+1} = G(q_n)$ \emph{ही है}। $G$ की उत्तलता तथा $G(1) = 1$ केवल दो
ज्यामितियाँ छोड़ते हैं। या तो वक्र $\intco01$ पर विकर्ण के ऊपर बना रहता
है (माध्य $m \leq 1$): तब सीढ़ी को $1$ से पहले कहीं रुकने की जगह नहीं। या
वक्र किसी $q <
1$ पर काटता है ($m > 1$): तब सीढ़ी उस कटान के नीचे फँस
जाती है और उसी तक अभिसरित होती है, गुणोत्तर दर $G'(q) < 1$ से, जिसे \cref{ex:b2:genfun:cobwebnumerics}
में परिमाणित किया गया है। विलोपन प्रमेय का सारा विश्लेषण इसी एक
चित्र में दिख जाता है --- और इसीलिए परिकलन से पहले उसे खींच लेना
सार्थक है।
\end{remark}

\begin{example}\label{ex:b2:genfun:branchingexample}
संतति \omterm{def:b2:randomvar:law}{नियम}: प्रायिकताओं $\frac14, \frac14, \frac12$ के साथ कोई बच्चा नहीं, एक बच्चा, दो
बच्चे। तब $m = \frac14 + 1 =
\frac54 > 1$ और $G(t) = \frac14 + \frac14 t + \frac12 t^2$। अचल बिंदु: $\frac12 t^2 - \frac34 t + \frac14 = 0$, अर्थात् $2t^2 - 3t
+ 1 = (2t - 1)(t - 1) = 0$: $q = \frac12$। वंश
प्रायिकता $\frac12$ के साथ मिट जाता है --- और प्रायिकता $\frac12$ के साथ वह
सदा जीवित रहता है।
\end{example}

\begin{remark}[इस खंड के भीतर के परिप्रेक्ष्य]
यह अध्याय पुस्तक का चौराहा है, और हर सामग्री किसी नामित जगह से आई
है: श्रेणी-बीजगणित \cref{ch:b2:series} तथा \cref{ch:b2:powerseries} से, प्रायिकता \cref{ch:b2:proba} से (एकदिष्ट
संततता $q_n \uparrow q$ सिद्ध करती है) तथा \cref{ch:b2:randomvar} से ($G_X =
\E(t^X)$ कोई \omterm{def:b2:randomvar:expectation}{प्रत्याशा} है, और
गुणनात्मकता गुणनफल प्रमेय है), उत्तलता \cref{ch:b2:affine} के द्वारा \cref{ch:b2:realfun} से। भारी
पुच्छ वाली विसंगतियाँ भी जुड़ती हैं: पिछले अध्याय के सेंट
पीटर्सबर्ग चर का $G(t) = \sum_k2^{-k}t^{2^k}$ है, जो $\intcc01$ पर पूरी तरह अभिसारी कोई श्रेणी है
पर जिसका $1^-$ पर अवकलज अपसरित हो जाता है --- अपरिमित माध्य, एक ही
नज़र में दिखता हुआ। एक वस्तु, और वर्ष भर का हर औज़ार: अंतिम अध्याय
के लिए उपयुक्त।
\end{remark}

(क) \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} केवल $\N$-मान वाले चरों पर लागू होते हैं: चिह्नित या

है, किसी नए \omterm{def:b2:randomvar:law}{नियम} का नहीं। (ग) यादृच्छिक योगों में संयोजन का क्रम

(घ) गुणनात्मकता को स्वतंत्रता तथा यादृच्छिकता के भिन्न स्रोत चाहिए:
$G_{2X}(t) = G_X(t^2)$, $G_X(t)^2$ नहीं। (ङ) $1$ पर अवकलन कोई सीमांत संक्रिया है: जब

नहीं, बल्कि ईमानदार कथन है।
\end{remark}

\sim \mathcal{P}(\lambda p)$: अर्थात् वस्तुओं की कोई प्वासों संख्या,

ही छोड़ती है --- यानी \emph{विरलन}। छोड़ी गई गिनती का \omterm{def:b2:randomvar:law}{नियम} भी
परिकलित कीजिए और सराहिए: वह $\mathcal{P}(\lambda(1-p))$ है, और दिखाया जा सकता है कि वह $S$
से \omterm{def:b2:proba:independence}{स्वतंत्र} है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:genfun:5}
(ऋणात्मक द्विपद) मान लीजिए $T_r$ $r$ चित्त पाने के लिए आवश्यक उछालों
की संख्या है (चित्त की प्रायिकता $p$)। $T_r$ को $r$ \omterm{def:b2:proba:independence}{स्वतंत्र}
ज्यामितीय चरों के योग के रूप में लिखिए, निकालिए
\[
G_{T_r}(t) = \Bigl(\frac{pt}{1 - (1-p)t}\Bigr)^{r},
\qquad
\E(T_r) = \frac rp,
\qquad
V(T_r) = \frac{r(1-p)}{p^2},
\]
और $\P(T_r = n) = \binom{n-1}{r-1}
p^r(1-p)^{n-r}$ ढूँढ़ने के लिए $G_{T_r}$ का प्रसार कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:genfun:6}
संतति \omterm{def:b2:randomvar:law}{नियम} $p_0 = \frac18$, $p_1 = \frac38$, $p_2 =
\frac38$, $p_3 = \frac18$ के लिए: $m$ परिकलित कीजिए,
अधिक्रांतिकता तय कीजिए, और विलोपन प्रायिकता $q$ ठीक-ठीक परिकलित
कीजिए। \emph{($G(t) - t$ का मूल $t = 1$ बाहर निकाल लीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:7}
(कुल संतति) किसी अवक्रांतिक \omterm{pb:b2:genfun:1}{शाखन प्रक्रम} ($m < 1$) में मान लीजिए $Y = \sum_{n\geq0} Z_n$
अब तक जन्मे कुल व्यक्तियों की संख्या है। दिखाइए $\E(Y) = \sum_n m^n = \frac{1}{1 - m}$ (योगों की
अदला-बदली न्यायसंगत ठहराइए), और सिद्ध कीजिए कि \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} $H = G_Y$ फलनिक
समीकरण $H(t) = t\,G(H(t))$ पूरा करता है। \emph{(पूर्वज, साथ में उसके हर बच्चे की
कुल संतति, जो $Y$ की \omterm{def:b2:proba:independence}{स्वतंत्र} प्रतिलिपियाँ हैं।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:8}
मान लीजिए $X$ का \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} $G$ है, जिसकी \omterm{def:b2:powerseries:radius}{अभिसरण त्रिज्या} $>
1$ है।
\emph{चरघातांकी पुच्छ परिबंध} सिद्ध कीजिए: ऐसे $C > 0$ तथा $\rho \in \intoo{0}{1}$ हैं कि
$\P(X \geq n) \leq C\rho^n$। \emph{(चक्रिका के भीतर किसी नियत $t > 1$ के लिए $t^X$ पर लगाई गई
मार्कोव।)} विलोमतः दिखाइए कि यदि $\rho < 1$ के साथ $\P(X \geq n) \leq C\rho^n$, तो $G$ की
त्रिज्या $\geq 1/\rho > 1$ है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:9}
(संततता प्रमेय, प्रारंभिक स्थिति) मान लीजिए $X, X_1, X_2, \dots$ $\N$-मान वाले हैं और
हर $t \in
\intco{0}{1}$ के लिए $G_{X_n}(t) \to G_X(t)$। दिखाइए कि हर $k$ के लिए $\P(X_n = k) \to \P(X = k)$। \emph{($k$ पर
आगमन: $k = 0$ के लिए $t \to 0$ लीजिए --- ध्यान से: छोटा $t$ स्थिर कीजिए,
$\abs{\P(X_n = 0) - G_{X_n}(t)}
\leq \frac{t}{1-t}$ बरतिए, जो वैध है क्योंकि पुच्छ $\sum_{j \geq 1}p_jt^j
\leq \frac{t}{1 - t}$; फिर विकर्णन कीजिए।
आगमन-चरण के लिए $\frac{G(t) - \P(X = 0)}{t}$ पर विचार कीजिए, जो किसी खिसके हुए \omterm{def:b2:randomvar:law}{नियम} का \omterm{ex:b2:powerseries:fibonacci}{जनक
फलन} है।)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:genfun:10}
(सम-विषमता की युक्ति) दिखाइए कि किसी $\N$-मान वाले चर $X$ के लिए
\[
\P(X \text{ सम}) = \frac{1 + G_X(-1)}{2} ,
\]
और यह प्रायिकता $X \sim \mathcal P(\lambda)$ तथा $X \sim \mathcal B(n, p)$ के लिए परिकलित कीजिए। $G_X(-1) \to 0$ का
प्रायिकतात्मक अर्थ क्या है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:genfun:11}
(सिशरमान पासे) निष्पक्ष पासे के \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} का गुणनखंडन
\[
\frac{t + t^2 + \dots + t^6}{6}
= \frac{t\,(1 + t)(1 + t + t^2)(1 - t + t^2)}{6},
\]
सत्यापित कीजिए और दिखाइए कि फलकों $\{1, 2, 2, 3, 3, 4\}$ तथा $\{1, 3, 4, 5, 6, 8\}$ वाले दोनों पासों के
\omterm{ex:b2:powerseries:fibonacci}{जनक फलन} $\frac{t(1+t)(1+t+t^2)}6$ तथा $\frac{t(1+t)(1+t+t^2)(1-t+t^2)^2}6$ हैं, जिनका गुणनफल दो मानक पासों के \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} के
बराबर है: अर्थात् ये विचित्र पासे हर योग $2,
\dots, 12$ ठीक मानक प्रायिकताओं
के साथ उत्पन्न करते हैं।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:12}
(लगातार दो चित्तों की प्रतीक्षा) चित्त की प्रायिकता $p$ वाला कोई
सिक्का तब तक उछाला जाता है जब तक लगातार दो चित्त न आ जाएँ; मान
लीजिए $T$ उछालों की संख्या है (\cref{exo:b2:proba:6} का खेल)। पहले उछालों पर
प्रतिबंधित करके ``कोई चालू चित्त नहीं'' तथा ``एक चालू चित्त''
अवस्थाओं से जनक फलनों के लिए कोई रैखिक निकाय व्युत्पन्न कीजिए, और
निष्कर्ष निकालिए
\[
G_T(t) = \frac{p^2t^2}{1 - qt - pqt^2}
\qquad (q = 1 - p);
\]
$G_T(1) = 1$ तथा $\E(T) = \dfrac{1 + p}{p^2}$ जाँचिए (निष्पक्ष सिक्के के लिए $= 6$)।
\end{exercise}

\section{समस्या: गाल्टन--वाट्सन प्रक्रम, हल किया हुआ}

\begin{problem}[{सप्ताहांत समस्या --- वृद्धि दरें, सटीक हल, कुल संतति, और
कोल्मोगोरोव का क्रांतिक आकलन}]\label{pb:b2:genfun:1}
विलोपन कसौटी (\cref{thm:b2:genfun:extinction}) शाखन प्रक्रमों\index{शाखन प्रक्रम} को
अवक्रांतिक, क्रांतिक तथा अधिक्रांतिक में बाँट देती है --- पर वह
\emph{दरों} के विषय में कुछ नहीं कहती: कोई अभिशप्त वंश कितनी तेज़ी
से मरता है, कोई बचा हुआ वंश कितना बड़ा होता है। यह समस्या उन्हें
परिकलित करती है। हम अध्याय का संकेतन बनाए रखते हैं: \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} $G$
वाला संतति \omterm{def:b2:randomvar:law}{नियम} $(p_k)$, माध्य $m = G'(1)$, पीढ़ी-आकार $Z_n$ ($Z_0 = 1$), पुनरावर्त
$G_n = G_{Z_n}$, विलोपन प्रायिकताएँ $q_n = \P(Z_n = 0)
\uparrow q$; हम सदा $p_1 \neq 1$ मानते हैं, और जहाँ
द्वितीय आघूर्ण आते हैं वहाँ $G''(1) < \infty$, तथा हम $\sigma^2 =
V(Z_1)$ लिखते हैं।

\medskip
\noindent\textbf{भाग I --- पीढ़ियों के आघूर्ण।}
\begin{enumerate}
  \item दिखाइए $\E(Z_n) = m^n$ \emph{($1^-$ पर $G_n = G
        \circ G_{n-1}$ पर शृंखला नियम, $G_{n-1}(1) = 1$ तथा
        \cref{thm:b2:genfun:moments} बरतकर)}।
  \item पुनरावृत्ति $G_n''(1) =
        G''(1)\,m^{2(n-1)} + m\,G_{n-1}''(1)$ स्थापित कीजिए और उसे हल कीजिए:
        $G_n''(1) = G''(1)\,m^{n-1}\dfrac{m^n - 1}{m - 1}$
        $m \neq 1$ के लिए, तथा $m =
        1$ के लिए $G_n''(1) = n\,G''(1)$।
  \item निकालिए
        \[
        V(Z_n) = \sigma^2m^{n-1}\,\frac{m^n - 1}{m - 1}
        \quad (m \neq 1),
        \qquad
        V(Z_n) = n\,\sigma^2 \quad (m = 1).
        \]
  \item (अवक्रांतिक दर, ऊपरी परिबंध) $m < 1$ के लिए दिखाइए $\P(Z_n > 0) \leq m^n$
        \emph{(पूर्णांक-मान वाले $Z_n$ पर मार्कोव)}: अर्थात्
        विलोपन किसी गुणोत्तर दर के साथ निश्चित है --- अध्याय
        की कसौटी का परिमाणात्मक परिष्कार।
  \item (अवक्रांतिक दर, निचला परिबंध) $Z_n\mathbf 1_{Z_n > 0}$ पर कोशी--श्वार्ज़ का
        उपयोग करते हुए दिखाइए
        \[
        \P(Z_n > 0) \geq \frac{\E(Z_n)^2}{\E(Z_n^2)}
        \geq c\,m^{n}
        \quad\text{जहाँ}\quad
        c = \Bigl(\frac{\sigma^2}{m(1-m)} + 1\Bigr)^{-1} :
        \]
        अर्थात् गुणोत्तर दर $m^n$ अचरों तक सटीक है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- ज्यामितीय कुल, ठीक-ठीक हल किया
हुआ।} मान लीजिए संतति \omterm{def:b2:randomvar:law}{नियम} $\N$ पर ज्यामितीय है: $p_k =
qp^k$ ($k \geq 0$), जहाँ
$0 < p < 1$, $q = 1 - p$।
\begin{enumerate}[resume]
  \item $G(t) = \dfrac{q}{1 - pt}$ तथा $m = \dfrac
        pq$ परिकलित कीजिए; $p$ के पदों में तीनों
        व्यवस्थाएँ ढूँढ़िए।
  \item $G(t) = t$ हल कीजिए: दिखाइए कि अचल बिंदु $1$ तथा $q/p = 1/m$ हैं, और
        विलोपन प्रायिकता $q_{\mathrm{ext}} = \min(1, 1/m)$ पुनः प्राप्त कीजिए।
  \item आगमन से \omterm{def:b2:multint:exact}{संवृत रूप} सिद्ध कीजिए
        \[
        q_n = \frac{m^n - 1}{m^{n+1} - 1} \quad (m \neq 1),
        \qquad
        q_n = \frac{n}{n+1} \quad (m = 1).
        \]
  \item सटीक दरें निकालिए: अवक्रांतिक स्थिति में $1 - q_n \sim (1 - m)\,m^n$, और
        अधिक्रांतिक स्थिति में $q_{\mathrm{ext}} - q_n
        \sim \dfrac{m - 1}{m^{2}}\cdot m^{-n}$; जाँचिए कि अधिक्रांतिक
        संकुचन अनुपात $G'(q_{\mathrm{ext}}) = 1/m$ है।
  \item क्रांतिक स्थिति ($p = \tfrac12$): $\sigma^2 =
        2$ परिकलित कीजिए और ध्यान
        दीजिए कि $1 - q_n = \frac1{n+1}$: अर्थात् उत्तरजीविता $\frac1n$ की तरह क्षय
        करती है --- न गुणोत्तर, न \omterm{def:b2:series:summable}{योग्य}।
  \item अब भी क्रांतिक: आगमन से पूरा पुनरावर्त
        \[
        G_n(t) = \frac{n - (n-1)t}{n + 1 - nt},
        \]
        सिद्ध कीजिए और निकालिए कि उत्तरजीविता पर प्रतिबंधित
        करने पर $Z_n$ $\N^*$ पर प्राचल $\frac1{n+1}$ का ज्यामितीय है:
        \[
        \P(Z_n = k \mid Z_n > 0) = \frac1{n+1}
        \Bigl(\frac{n}{n+1}\Bigr)^{k-1},
        \qquad
        \E(Z_n \mid Z_n > 0) = n + 1 .
        \]
        औसत वंश मर जाता है, पर बचे हुए वंशों का आकार कोटि $n$
        का होता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- कुल संतति।} मान लीजिए $Y =
\sum_{n\geq0}Z_n \in \N^* \cup \{\infty\}$ अब तक जन्मे
कुल व्यक्तियों की संख्या है, और $H(t) =
\sum_{k\geq1}\P(Y = k)t^k$।
\begin{enumerate}[resume]
  \item $\P(Y < \infty) = q_{\mathrm{ext}}$ न्यायसंगत ठहराइए, और \cref{exo:b2:genfun:7} से फलनिक समीकरण $H(t) = t\,G(H(t))$
        स्मरण कीजिए (जिसकी व्युत्पत्ति में $m < 1$ का उपयोग नहीं
        हुआ था)।
  \item (द्विआधारी शाखन) $p_0 = p_2 = \frac12$ (क्रांतिक) के लिए फलनिक समीकरण हल
        कीजिए:
        \[
        H(t) = \frac{1 - \sqrt{1 - t^2}}{t},
        \]
        और \cref{ex:b2:powerseries:catalan} से प्रसार करके
        \[
        \P(Y = 2k + 1) = \frac{C_k}{2^{2k+1}},
        \qquad C_k = \frac1{k+1}\binom{2k}k ;
        \]
        पाइए; मान $\P(Y = 1) = \frac12$ तथा $\P(Y = 3)
        = \frac18$ सीधी गणना से जाँचिए।
  \item फलनिक समीकरण का $1^-$ पर अवकलन करके दिखाइए कि $m < 1$ के
        लिए $\E(Y) = \frac{1}{1-m}$, जबकि क्रांतिकता $\E(Y) = \infty$ को बाध्य कर देती है:
        अर्थात् क्रांतिक कुल संतति लगभग निश्चित रूप से परिमित
        है पर उसकी \omterm{def:b2:randomvar:expectation}{प्रत्याशा} अपरिमित है।
  \item केंद्रीय द्विपद अनंतस्पर्शी (\cref{ex:b2:comparison:centralbinomial}) के साथ दिखाइए
        \[
        \P(Y = 2k+1) \sim \frac{1}{2\sqrt\pi\,k^{3/2}},
        \]
        अर्थात् कोई भारी $k^{-3/2}$ पुच्छ, और निकालिए $\P(Y > n)
        \asymp n^{-1/2}$ (इसी कोटि
        के ऊपरी तथा निचले परिबंध पर्याप्त हैं)।
  \item निष्पक्ष \omterm{pb:b2:proba:1}{यादृच्छिक चहलक़दमी} (\cref{ch:b2:proba} की सप्ताहांत समस्या)
        से तुलना कीजिए: वहाँ निश्चित पर अपरिमित-माध्य वापसी-समय,
        यहाँ निश्चित पर अपरिमित-माध्य कुल संतति, और दोनों में
        $n^{-3/2}$ स्थानीय नियम। क्रांतिकता यह हस्ताक्षर क्यों उत्पन्न
        करती है, इस पर एक अनुच्छेद।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- क्रांतिकता पर कोल्मोगोरोव का आकलन।}
मान लीजिए $m = 1$, $0 < \sigma^2 = G''(1) <
\infty$।
\begin{enumerate}[resume]
  \item दिखाइए कि $G''$ $\intcc01$ तक \omterm{def:b2:metric:continuity}{संतत} रूप से फैल जाता है
        \emph{(अऋणात्मक वर्धमान, परिमित सीमा के साथ)} और $1$ पर
        टेलर प्रसार निकालिए:
        \[
        G(t) = t + b\,(1-t)^2 + o\bigl((1-t)^2\bigr),
        \qquad b = \frac{G''(1)}2 = \frac{\sigma^2}2 .
        \]
  \item $t \in \intco01$ के लिए $h(t) = \dfrac1{1 - G(t)} -
        \dfrac1{1 - t}$ रखिए। दिखाइए
        \[
        h(t) = \frac{G(t) - t}{(1 - G(t))(1 - t)}
        \xrightarrow[t\to1^-]{} b .
        \]
  \item पुनरावृत्ति $q_{j+1} = G(q_j)$ के अनुदिश दूरबीन की तरह सिकोड़िए:
        \[
        \frac1{1 - q_n} = 1 + \sum_{j=0}^{n-1}h(q_j),
        \]
        और किसी चेज़ारो तर्क से निष्कर्ष निकालिए कि
        \[
        \P(Z_n > 0) = 1 - q_n \sim \frac{2}{\sigma^2\,n}
        \]
        --- यही \emph{कोल्मोगोरोव का आकलन} है: हर क्रांतिक \omterm{pb:b2:genfun:1}{शाखन
        प्रक्रम} सार्वभौमिक दर $1/n$ से मरता है, और केवल अचर ही
        संतति \omterm{def:b2:randomvar:law}{नियम} को याद रखता है।
  \item आकलन को प्रश्न 10 की क्रांतिक ज्यामितीय स्थिति के सामने
        जाँचिए।
  \item $\E(Z_n \mid Z_n > 0) = \dfrac{1}{1 - q_n}
        \sim \dfrac{\sigma^2 n}{2}$ निकालिए \emph{(ध्यान दीजिए $\E(Z_n
        \mathbf 1_{Z_n>0}) = \E(Z_n) = 1$)}, और उसे प्रश्न
        11 के सामने जाँचिए: उत्तरजीविता पर प्रतिबंधित करने पर
        जनसंख्या \emph{रैखिक रूप से} बढ़ती है --- मृत्यु और
        विस्फोट के बीच की क्रांतिक रस्सी।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- अनुप्रयोग और संश्लेषण।}
\begin{enumerate}[resume]
  \item (महामारियाँ, शृंखला अभिक्रियाएँ) किसी प्वासों संतति \omterm{def:b2:randomvar:law}{नियम}
        $\mathcal P(\lambda)$ के लिए --- जहाँ हर रोगी $\mathcal P(\lambda)$ नए रोगियों को संक्रमित
        करता है --- विलोपन समीकरण $q = \eu^{\lambda(q-1)}$ लिखिए और उसे $\lambda = 1.5$ ($q \approx
        0.417$)
        तथा $\lambda = 2$ ($q \approx 0.203$) के लिए संख्यात्मक रूप से हल कीजिए: एक
        रोगी से शुरू करने पर कोई बड़ा प्रकोप $\lambda > 1$ होने पर भी
        निश्चित \emph{नहीं} है। समझाइए कि $q_0 = 0$ से पुनरावृत्ति
        $q_{n+1} = \eu^{\lambda(q_n - 1)}$ सही मूल तक क्यों अभिसरित होती है।
  \item एक के बजाय $k$ पूर्वजों से शुरू करने पर दिखाइए कि
        विलोपन प्रायिकता $q^k$ है। अनुप्रयोग: $\lambda = 1.5$ के साथ,
        कितने आरंभिक रोगी किसी प्रकोप को कम से कम $99\%$ संभावित
        बना देते हैं?
  \item (किसी अधिक्रांतिक प्रक्रम को विलोपन पर प्रतिबंधित करना)
        विलोपन प्रायिकता $q \in
        \intoo01$ वाले $m > 1$ के लिए: पहले उत्तलता से
        सिद्ध कीजिए कि सबसे छोटे अचल बिंदु पर $G'(q) < 1$, और निकालिए
        $q_{\mathrm{ext}} - q_n = O\bigl(G'(q)^n\bigr)$
        (गुणोत्तर अभिसरण, जैसा प्रश्न 9 ने दिखाया)। फिर दिखाइए
        कि $\widehat G(t) = G(qt)/q$ किसी सच्चे संतति \omterm{def:b2:randomvar:law}{नियम} का \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} है, जिसका माध्य
        $\widehat m =
        G'(q) < 1$ है: अर्थात् कोई अवक्रांतिक साथी प्रक्रम। ज्यामितीय
        कुल पर सत्यापित कीजिए: अधिक्रांतिक $(p, q)$-प्रक्रम को
        विलोपन पर प्रतिबंधित करना $p$ तथा $q$ की अदला-बदली कर
        देता है। (पूरा कथन --- कि प्रतिबंधित प्रक्रम साथी
        प्रक्रम \emph{ही है} --- वर्ष 3 के खंड में सिद्ध होता
        है; यहाँ आपने उसकी जनक-फलन छाया सत्यापित की है।)
  \item संश्लेषण: त्रिभाजन की सारणी बनाइए --- $m <
        1$, $m = 1$, $m > 1$
        के लिए: $q$ का मान; $\P(Z_n >
        0)$ की या $q - q_n$ की दर; $\E(Y)$; किसी
        बची हुई पीढ़ी का आकार। हर औज़ार के लिए एक वाक्य में
        बताइए कि जनक फलनों के संयोजन, उत्तलता, $1^-$ पर टेलर, तथा
        चेज़ारो औसतन ने पूरी समस्या को कैसे ढोया, और वर्ष 3 का
        खंड इसमें क्या जोड़ता है (मार्टिंगेल $Z_n/m^n$ तथा याग्लोम
        का चरघातांकी सीमा नियम)।
\end{enumerate}
\end{problem}
