\chapter{अवकल समीकरण}\label{ch:b2:diffeq}

प्रथम वर्ष ने वे रैखिक समीकरण हल किए थे जिनके सूत्र मिल जाते हैं। यह
अध्याय वह देता है जो सूत्र नहीं दे सकते: \emph{कोशी--लिप्शिट्स प्रमेय} ---
$y' = f(t, y)$ के लिए अस्तित्व और अद्वितीयता --- जो बानाख अचल बिंदु प्रमेय से सिद्ध
है, ठीक वैसे ही जैसा \cref{ch:b2:metric} में वचन दिया गया था; और फिर \emph{रैखिक निकायों}
$X' = A(t)X + B(t)$ का पूरा सिद्धांत, जिसमें \omterm{ex:b2:nvs:matrixexp}{आव्यूह घातांकी} और \omterm{def:b2:diffeq:wronskian}{व्रोंस्कियन} परिकलन के इंजन
हैं।

\section{कोशी--लिप्शिट्स प्रमेय}

\begin{theorem}[कोशी--लिप्शिट्स, वैश्विक लिप्शिट्स रूप]\label{thm:b2:diffeq:cauchylipschitz}
मान लीजिए $I$ कोई खंड है, $f \colon I \times \R^n \to \R^n$ \omterm{def:b2:metric:continuity}{संतत} है और अपने दूसरे चर में पहले चर के
सापेक्ष \omterm{def:b2:funcseq:def}{एकसमान रूप से} \emph{\omterm{def:b2:metric:continuity}{लिप्शिट्स}}: अर्थात् सभी $t
\in I$ के लिए $\norm{f(t, y) - f(t, z)} \leq k\,\norm{y - z}$। तब
प्रत्येक $(t_0, y_0) \in I \times \R^n$ के लिए कोशी समस्या\index{कोशी--लिप्शिट्स प्रमेय}
\[
y' = f(t, y), \qquad y(t_0) = y_0
\]
का ठीक एक हल $y \colon I \to \R^n$ है, जो $C^1$ वर्ग का है।
\end{theorem}

\begin{proof}
\emph{पुनःकथन।} कोई \omterm{def:b2:metric:continuity}{संतत} $y$ इस समस्या को हल करता है तभी जब वह समाकल
समीकरण
\[
y(t) = y_0 + \int_{t_0}^{t} f\bigl(s, y(s)\bigr)\,\dd s
=: T(y)(t)
\]
पूरा करे (दोनों दिशाओं में कलन की मूल प्रमेय; और समाकल समीकरण का \omterm{def:b2:metric:continuity}{संतत} हल
स्वतः $C^1$ होता है)।

\emph{पुनःमानदंडन के बाद एक संकुचन।} \omterm{def:b2:nvs:banach}{बानाख समष्टि} $E =
C(I, \R^n)$ पर \emph{भारित}
\omterm{def:b2:nvs:norm}{मानदंड}
\[
N(y) = \sup_{t \in I}\; \eu^{-2k\abs{t - t_0}}\,\norm{y(t)} ,
\]
लीजिए (जो उच्चतम \omterm{def:b2:nvs:norm}{मानदंड} के तुल्य है: भार खंड $I$ पर ऊपर और नीचे से परिबद्ध
है, अतः $E$ \omterm{def:b2:metric:complete}{पूर्ण} बनी रहती है), और $y, z \in
E$ तथा (मान लीजिए) $t \geq t_0$ के लिए आकलन
कीजिए:
\[
\norm{T(y)(t) - T(z)(t)}
\leq \int_{t_0}^{t} k\,\norm{y(s) - z(s)}\,\dd s
\leq k\,N(y - z)\int_{t_0}^{t} \eu^{2k(s - t_0)}\dd s
\leq \frac{N(y-z)}{2}\,\eu^{2k(t - t_0)} .
\]
$\eu^{-2k(t - t_0)}$ से गुणा करके उच्चतम लीजिए (स्थिति $t
< t_0$ \omterm{def:b2:quadratic:adjoint}{सममित} है): $N\bigl(T(y) - T(z)\bigr) \leq \frac12 N(y -
z)$: अर्थात् $T$
\omterm{def:b2:metric:complete}{पूर्ण} $(E, N)$ का $\frac12$-संकुचन है। बानाख अचल बिंदु प्रमेय (\cref{thm:b2:metric:banach}) अद्वितीय अचल
बिंदु दे देती है: वही अद्वितीय हल है।
\end{proof}

\begin{remark}
केवल $C^1$ (स्थानीय रूप से \omterm{def:b2:metric:continuity}{लिप्शिट्स}) $f$ के लिए प्रमेय \emph{स्थानीय} रूप
से सत्य है, जहाँ किसी महत्तम \omterm{def:b2:metric:topology}{विवृत} अंतराल पर कोई महत्तम हल होता है; और हल
परिमित समय में फट सकते हैं ($y' = y^2$, $y(0) = 1$: $y(t)
= \frac{1}{1-t}$, और $t = 1$ पर ग़ायब)। पूरा खंड
वैश्विक \omterm{def:b2:metric:continuity}{लिप्शिट्स} परिकल्पना ही ख़रीदती है। दो परिणाम गढ़ लेने \omterm{def:b2:series:summable}{योग्य} हैं:
\omterm{def:b2:metric:continuity}{लिप्शिट्स} क्षेत्र वाले किसी अवकल समीकरण के हल-वक्र \emph{कभी नहीं} कटते;
और किसी रैखिक समांगी समीकरण का कहीं भी लुप्त होने वाला एकमात्र हल शून्य
फलन है।
\end{remark}

\begin{example}[अद्वितीयता एक प्रमेय है: एक रिसता हुआ क्षेत्र]\label{ex:b2:diffeq:leaky}
$y(0) = 0$ के साथ $y' = 2\sqrt{\abs y}$ लीजिए। शून्य फलन उसे हल करता है; और
\[
y(t) = \begin{cases} 0 & t \leq 0,\\ t^2 & t \geq 0,
\end{cases}
\]
भी, जो $C^1$ है (दोनों टुकड़ों का अवकलज जोड़-बिंदु पर $0$ है) तथा $t > 0$ के लिए
$y'(t) = 2t = 2\sqrt{t^2}$ पूरा करता है --- वस्तुतः उड़ान में देर करके \emph{हर} छोड़ने-समय $c \geq 0$
के लिए एक हल मिल जाता है: अर्थात् एक ही आरंभिक आँकड़े से होकर अपरिमित हल।
\cref{thm:b2:diffeq:cauchylipschitz} के साथ कोई विरोधाभास नहीं: $y = 0$ के पास
\[
\frac{\abs{2\sqrt y - 2\sqrt z}}{\abs{y - z}}
= \frac{2}{\sqrt y + \sqrt z} \longrightarrow +\infty ,
\]
अर्थात् क्षेत्र $y$ में \omterm{def:b2:metric:continuity}{लिप्शिट्स} नहीं है, और प्रमेय चुप है। समापन
दृष्टि: भौतिक पाठ यह है कि गुरुत्व के अधीन ख़ाली होती बाल्टी को उलटा
चलाया जाए --- ख़ाली अवस्था से यह नहीं बताया जा सकता कि वह भरनी कब शुरू
हुई थी; अवकल समीकरणों का नियतिवाद ठीक \omterm{def:b2:metric:continuity}{लिप्शिट्स} शर्त है, प्रकृति का नियम
नहीं।
\end{example}

\section{रैखिक निकाय}

\begin{theorem}[रैखिक निकायों की संरचना]\label{thm:b2:diffeq:linear}
मान लीजिए $A \colon I \to \mathcal{M}_n(\R)$ और $B \colon I \to \R^n$ किसी अंतराल $I$ पर \omterm{def:b2:metric:continuity}{संतत} हैं। प्रत्येक $(t_0, X_0)$ के लिए
समस्या
\[
X' = A(t)X + B(t), \qquad X(t_0) = X_0
\]
का \emph{पूरे} $I$ पर ठीक एक हल है। समांगी निकाय ($B = 0$) के हल ठीक $n$ विमा
की सदिश समष्टि $\mathcal{S}_H$ बनाते हैं, और मूल्यांकन $X \mapsto X(t_0)$ तुल्याकारिता $\mathcal{S}_H \to \R^n$ है; और
सामान्य हल $=$ विशिष्ट $+$ समांगी।
\end{theorem}

\begin{proof}
$t_0$ वाले प्रत्येक खंड $J \subseteq I$ पर: $f(t, X) =
A(t)X + B(t)$ \omterm{def:b2:metric:continuity}{संतत} है और $X$ में अचर $k
= \sup_J \vertiii{A(t)}$ के साथ
\omterm{def:b2:metric:continuity}{लिप्शिट्स} (जो परिमित है: खंड पर \omterm{def:b2:metric:continuity}{संतत}): अतः \cref{thm:b2:diffeq:cauchylipschitz} $J$ पर लागू होती है; और $J$
को $I$ में फैलाते जाने पर अद्वितीयता उन हलों को $I$ पर एक ही हल में जोड़
देती है। हल-समुच्चय तथा मूल्यांकन प्रतिचित्रण की रैखिकता स्पष्ट है; और
मूल्यांकन अस्तित्व से आच्छादक तथा अद्वितीयता से एकैकी है: $\dim \mathcal{S}_H = n$। आनत संरचना
प्रथम वर्ष का अक्षरशः वही तर्क है।
\end{proof}

\begin{example}[मूल्यांकन तुल्याकारिता, मूर्त रूप में]\label{ex:b2:diffeq:evaliso}
$y'' + y = 0$ के लिए, जिसे $X = (y,
y')$ के साथ निकाय $X' =
\begin{pmatrix} 0 & 1\\ -1 & 0\end{pmatrix}X$ के रूप में देखा गया है: प्रमेय कहती
है कि हल-समष्टि एक समतल है और $X \mapsto X(0) = (y(0), y'(0))$ $\R^2$ पर तुल्याकारिता है। हल $\cos$ और $\sin$
$(1, 0)$ तथा $(0, 1)$ --- अर्थात् $\R^2$ का विहित आधार --- पर मूल्यांकित होते हैं, अतः
वे हल-समष्टि का आधार बनाते हैं, और \emph{हर} हल
\[
y(t) = y(0)\cos t + y'(0)\sin t ,
\]
है, जहाँ गुणांक सीधे आरंभिक आँकड़ों से पढ़ लिए जाते हैं, कोई रैखिक निकाय
हल नहीं करना पड़ता। समापन दृष्टि: वह \omterm{def:b2:diffeq:wronskian}{मूलभूत निकाय} चुनना जिसके आरंभिक मान
विहित आधार हों (यहाँ $\cos, \sin$), ठीक $\eu^{tA}$ के स्तंभ चुनना है; और मूल्यांकन
तुल्याकारिता ही वह कारण है कि आरंभिक शर्तें प्रक्षेप-पथों का प्राचलन कर
देती हैं --- रैखिक समीकरणों के लिए ``नियतिवादी गतिकी'' की ज्यामितीय
सामग्री यही है।
\end{example}

\begin{definition}[व्रोंस्कियन]\label{def:b2:diffeq:wronskian}
समांगी निकाय के हलों $X_1, \dots, X_n$ के लिए \emph{व्रोंस्कियन}\index{व्रोंस्कियन} $W(t) = \det\bigl(X_1(t),
\dots, X_n(t)\bigr)$
होता है। ऊपर की तुल्याकारिता से या तो $W$ सर्वथा लुप्त होता है (कुल परस्पर
जुड़ा है) या कभी नहीं (कोई \emph{मूलभूत निकाय}\index{मूलभूत निकाय}); और
परिमाणात्मक रूप में $W' = \operatorname{tr}\bigl(A(t)\bigr)
W$, अतः
\[
W(t) = W(t_0)\,\exp\Bigl(\int_{t_0}^{t}
\operatorname{tr} A(s)\,\dd s\Bigr)
\quad \text{(ल्यूविल का सूत्र)}.
\]
\end{definition}

\begin{example}[किसी ऑयलर समीकरण पर ल्यूविल की जाँच]\label{ex:b2:diffeq:eulerwronskian}
$\intoo{0}{\infty}$ पर समीकरण $t^2y'' + ty' - y = 0$ के हल $y_1(t) = t$ और $y_2(t) = \frac1t$ हैं (प्रतिस्थापित कीजिए)। उनका
\omterm{def:b2:diffeq:wronskian}{व्रोंस्कियन}:
\[
W(t) = \det\begin{pmatrix} t & \tfrac1t\\[2pt]
1 & -\tfrac{1}{t^2}\end{pmatrix}
= -\frac1t - \frac1t = -\frac2t ,
\]
जो कभी शून्य नहीं: अर्थात् \omterm{def:b2:diffeq:wronskian}{मूलभूत निकाय}। अब ल्यूविल जाँचिए: सामान्यीकृत
रूप $y'' + \frac1t\,y' - \frac{1}{t^2}\,y = 0$ में सहचर आव्यूह $A(t) = \begin{pmatrix} 0 & 1\\ \frac{1}{t^2}
& -\frac1t\end{pmatrix}$ का अनुरेख $-\frac1t$ है, अतः
\[
W(t) = W(1)\exp\Bigl(-\int_1^t\frac{\dd s}{s}\Bigr)
= -2\,\eu^{-\ln t} = -\frac2t . \checkmark
\]
समापन दृष्टि: ल्यूविल किसी हल के ज्ञात होने से पहले ही \omterm{def:b2:diffeq:wronskian}{व्रोंस्कियन} का
\emph{आकार} बता देती है --- यहाँ कि $W$ $\frac{c}{t}$ होना चाहिए; और यही
कोटि-घटाने की विधि (\cref{prop:b2:diffeq:secondorder}) को चलाता है, जहाँ $y_1$ और \omterm{def:b2:diffeq:wronskian}{व्रोंस्कियन} का रूप
जानने से $y_2$ एक ही समाकलन में मिल जाता है।
\end{example}

\begin{proof}[ल्यूविल के सूत्र की उपपत्ति]
$M' = AM$ सहित $W(t) = \det M(t)$। \omterm{def:b2:linalg:det}{सारणिक} को स्तंभों के बहुरैखिक फलन के रूप में अवकलित करने
पर,
\[
W'(t) = \sum_j \det(X_1, \dots, X_j', \dots, X_n)
= \sum_j \det(X_1, \dots, AX_j, \dots, X_n) .
\]
अब प्रतिचित्रण $(C_1, \dots, C_n) \mapsto \sum_j \det(C_1, \dots,
AC_j, \dots, C_n)$ $n$-रैखिक और एकांतर है (दो बराबर स्तंभों $C_i = C_k$ के साथ पद
$j \notin \{i, k\}$ सीधे लुप्त हो जाते हैं, और पद $j = i$ तथा $j = k$ एक स्तंभ-अदला-बदली के बाद
जोड़े-जोड़े में कट जाते हैं): अतः अद्वितीयता प्रमेय (\cref{thm:b2:linalg:detspace}) से वह $c \cdot \det$ है,
जहाँ $c$ विहित स्तंभों पर पढ़ा जाता है: $c = \sum_j \det(e_1, \dots, Ae_j, \dots,
e_n) = \sum_j a_{jj} = \operatorname{tr} A$। इसलिए $W' =
\operatorname{tr}\bigl(A(t)\bigr)W$: अर्थात् एक
अदिश रैखिक अवकल समीकरण, जिसे प्रथम वर्ष का सूत्र हल कर देता है।
\end{proof}

\section{अचर गुणांक: आव्यूह घातांकी}

\begin{theorem}\label{thm:b2:diffeq:matrixexp}
$A \in \mathcal{M}_n(\R)$ (अथवा $\C$) के लिए घातांकी $\eu^{tA}
= \sum_k \frac{(tA)^k}{k!}$ (\cref{ex:b2:nvs:matrixexp}) ये पूरा करता है: $t \mapsto \eu^{tA}$ $C^1$ है
(वस्तुतः $C^\infty$) जहाँ\index{आव्यूह घातांकी}
\[
\frac{\dd}{\dd t}\,\eu^{tA} = A\,\eu^{tA} = \eu^{tA}A ,
\qquad
\eu^{(s+t)A} = \eu^{sA}\,\eu^{tA},
\qquad
(\eu^{A})^{-1} = \eu^{-A} ;
\]
और $AB = BA$ होने \emph{पर} $\eu^{A + B} = \eu^A\eu^B$। कोशी समस्या $X' = AX$, $X(0) = X_0$ का अद्वितीय हल $X(t) =
\eu^{tA}X_0$ है;
और स्रोत के साथ \emph{अचरों का विचरण} सूत्र सत्य रहता है:
\[
X(t) = \eu^{(t - t_0)A}X_0 + \int_{t_0}^{t} \eu^{(t-s)A}B(s)\,\dd
s .
\]
\end{theorem}

\begin{proof}
\emph{अवकलनीयता:} श्रेणी $\sum \frac{t^kA^k}{k!}$ और उसकी पद-व्युत्पन्न श्रेणी $\sum \frac{t^{k-1}A^k}{(k-1)!} = A\sum
\frac{(tA)^{k-1}}{(k-1)!}$ हर खंड पर
\omterm{def:b2:funcseq:series}{सामान्य रूप से} अभिसरित होती हैं (\omterm{def:b2:nvs:norm}{मानदंड} $\leq \frac{(\abs t\,\vertiii A)^k}{k!}$): अतः पद-दर-पद अवकलन कीजिए
(\cref{thm:b2:funcseq:seriestransfer}, सदिश-मान वाली)। और दोनों क्रम $A\eu^{tA}$ तथा $\eu^{tA}A$ सहमत हैं क्योंकि हर आंशिक
योग $A$ के साथ क्रमविनिमेय है।

\emph{समूह नियम:} क्रमविनिमेय $A, B$ के लिए दोनों घातांकी श्रेणियों का \omterm{thm:b2:series:fubini}{कोशी
गुणनफल} द्विपद प्रमेय से ठीक वैसे ही पुनर्संगठित होता है जैसा \cref{ex:b2:series:exp} में
(बानाख बीजगणित में निरपेक्ष अभिसरण उसे न्यायोचित ठहराता है): $\eu^{A+B} = \eu^A\eu^B$; और $B = sA$
के साथ यह एक-प्राचली समूह नियम देता है, तथा $B = -A$ प्रतिलोम।

\emph{कोशी समस्या:} $X(t) = \eu^{tA}X_0$ उसे हल करता है (अवकलन कीजिए); और अद्वितीयता \cref{thm:b2:diffeq:linear}
से। अचरों का विचरण: $Y(t) = \eu^{-tA}X(t)$ रखिए; अवकलन करने पर $Y' = \eu^{-tA}(X' -
AX) = \eu^{-tA}B(t)$; $t_0$ से $t$ तक समाकलन
कीजिए और $\eu^{tA}$ से वापस गुणा कीजिए।
\end{proof}

\begin{method}[$\eu^{tA}$ का परिकलन]\label{met:b2:diffeq:computeexp}
$A$ (\cref{ch:b2:reduction}) तक ले आइए: यदि $A = PDP^{-1}$ विकर्ण हो, तो $\eu^{t\lambda_i}$ के विकर्ण $\eu^{tD}$ के साथ $\eu^{tA} = P\,\eu^{tD}P^{-1}$;
और सामान्य रूप में डनफ़र्ड $A = D + N$ (क्रमविनिमेय) काम में लीजिए: $\eu^{tA} = \eu^{tD}\,\eu^{tN}$, जहाँ $\eu^{tN}$
$t$ का \emph{बहुपद} है (शून्यंभाविता श्रेणी को काट देती है)। और सम्मिश्र
\omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} युग्मित होकर घूर्णन-गुणा-घातांकी खंड बना देते हैं (\cref{exo:b2:diffeq:5})।
\end{method}

\begin{remark}[सामान्य भूलें]
\emph{(क) क्रमविनिमेयता के बिना $\eu^{A+B} \neq \eu^A\eu^B$:} $A = \begin{pmatrix} 0 & 1\\ 0 & 0\end{pmatrix}$, $B =
\begin{pmatrix} 0 & 0\\ 1 & 0\end{pmatrix}$ लीजिए। तब $\eu^A = I +
A$, $\eu^B = I + B$
(शून्यंभाविता), अतः $(A+B)^2 = I$ का उपयोग करके
\[
\eu^A\eu^B = \begin{pmatrix} 2 & 1\\ 1 & 1\end{pmatrix},
\qquad\text{जबकि}\qquad
\eu^{A+B} = \cosh(1)\,I + \sinh(1)\,(A + B)
= \begin{pmatrix} \cosh 1 & \sinh 1\\ \sinh 1 & \cosh 1
\end{pmatrix},
\]
और $\cosh 1 \approx 1.54 \neq 2$। \cref{thm:b2:diffeq:matrixexp} का समूह नियम एक सच्ची परिकल्पना उठाए चलता है। \emph{(ख)
रैखिक मैदान पर अरैखिक अंतर्ज्ञान:} \omterm{def:b2:metric:continuity}{संतत} गुणांकों वाले किसी \emph{रैखिक}
निकाय के हल पूरे अंतराल पर बसते हैं (\cref{thm:b2:diffeq:linear}) --- और यदि कोई संभावित हल $I$
के भीतर फट जाए, तो या तो समीकरण रैखिक नहीं था या परिकलन ग़लत है; इसके
विपरीत अरैखिक समीकरणों के लिए बिना तर्क के वैश्विकता का वचन कभी मत दीजिए
($y' = y^2$)। \emph{(ग) अज्ञात से भाग देना:} $y' = y(1-y)$ में चर अलग करने से अचर हल $0$
और $1$ चुपचाप छूट जाते हैं --- और ठीक वही कला-रेखा को संगठित करते हैं
(\cref{exo:b2:diffeq:3}); पहले अचर हल सूचीबद्ध कीजिए। \emph{(घ) आरंभिक आँकड़े सदिश तय करते
हैं, अदिश नहीं:} $n$-वीं कोटि के अदिश समीकरण को $n$ शर्तें चाहिए ($t_0$
पर $y, y', \dots$); और केवल $y(t_0)$ मिलाने से $(n-1)$-प्राचली कुल बचा रह जाता है, जो
``खोए हुए'' अचरों का क्लासिक स्रोत है।
\end{remark}

\begin{example}[डनफ़र्ड से एक $3\times3$ घातांकी]\label{ex:b2:diffeq:dunford3}
$A = \begin{pmatrix} 2 & 1 & 0\\ 0 & 2 &
0\\ 0 & 0 & 3\end{pmatrix}$ के लिए $X' = AX$ हल कीजिए। खंडों से डनफ़र्ड: $D = \operatorname{diag}(2, 2, 3)$ तथा $N = E_{12}$ सहित $A = D + N$, जो
क्रमविनिमेय हैं ($N$ अभिलक्षणिक-मान-$2$ वाले खंड के भीतर बसता है), और $N^2 =
0$:
\[
\eu^{tA} = \eu^{tD}\,\eu^{tN}
= \begin{pmatrix}
\eu^{2t} & t\,\eu^{2t} & 0\\
0 & \eu^{2t} & 0\\
0 & 0 & \eu^{3t}
\end{pmatrix} .
\]
सामान्य हल स्तंभ-दर-स्तंभ पढ़ा जाता है: $X(t) =
\bigl(\eu^{2t}(x_0 + ty_0),\ \eu^{2t}y_0,\
\eu^{3t}z_0\bigr)$। तर्कसंगति की जाँचें: $t = 0$ पर
आव्यूह $I$ है; उसका \omterm{def:b2:linalg:det}{सारणिक} $\eu^{7t} =
\eu^{t\operatorname{tr}A}$ है, जैसा ल्यूविल माँगती है; और गुणक $t$
ठीक वहीं प्रकट होता है जहाँ \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $2$ दोषयुक्त है। समापन दृष्टि:
बहुपद गुणा घातांकी कोई रटा हुआ अनुमान नहीं है --- वे छिन्न श्रेणी $\eu^{tN}$ हैं,
और उनकी घात शून्यंभाविता सूचकांक से परिबद्ध रहती है, कभी उससे अधिक नहीं।
\end{example}

\begin{method}[$X' = AX + B(t)$ का हल, आरंभ से अंत तक]\label{met:b2:diffeq:solvelinear}
\begin{enumerate}
  \item $A$ का \omterm{def:b2:reduction:eigen}{वर्णक्रम}; फिर \cref{met:b2:diffeq:computeexp} के द्वारा $\eu^{tA}$ (विकर्णन कीजिए; अथवा \cref{ex:b2:diffeq:dunford3}
        जैसा डनफ़र्ड; अथवा $A^2 = -I$ जैसी कोई बहुपद युक्ति)।
  \item कोई विशिष्ट हल: अचरों का विचरण $\int_{t_0}^t\eu^{(t-s)A}B(s)\dd s$ सदा चलता है; और
        घातांकी-बहुपद $B$ के लिए उसी आकार का कोई प्रस्ताव (\omterm{pb:b2:diffeq:1}{अनुनाद} पर घात
        बढ़ाकर, \cref{exo:b2:diffeq:10}) अधिक तेज़ है।
  \item सामान्य हल $= \eu^{(t-t_0)A}X_0 +$ विशिष्ट; और आरंभिक आँकड़े \emph{अंत में}, पूरे
        सूत्र पर बिठाइए।
  \item तर्कसंगति की जाँचें: $X(t_0)$ सही हो; समांगी भाग की वृद्धि अभिलक्षणिक
        मानों के वास्तविक भागों से मेल खाए (\cref{exo:b2:diffeq:8}); और किसी मूलभूत आव्यूह
        का $\det$ ल्यूविल का पालन करे।
\end{enumerate}
\end{method}

\begin{example}[एक कला-चित्र]\label{ex:b2:diffeq:phaseportrait}
$A = \begin{pmatrix} 0 & 1\\ -1 & 0\end{pmatrix}$ के साथ $X' = AX$: $A^2 = -I$, अतः श्रेणी

प्रथम कोटि के वस्त्रों में संनादी दोलक $x'' + x = 0$। \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\pm\iu$ काल्पनिक
अक्ष पर: अर्थात् एक \emph{केंद्र}। और सामान्य रूप में $A$ के अभिलक्षणिक

\section{चर गुणांकों वाली द्वितीय कोटि}

\begin{proposition}\label{prop:b2:diffeq:secondorder}
समीकरण $y'' + a(t)y' + b(t)y = c(t)$ ($I$ पर $a, b, c$ \omterm{def:b2:metric:continuity}{संतत}) $X = (y, y')$ के लिए निकाय $X' = A(t)X + B(t)$ है: अतः किसी भी आरंभिक
आँकड़े $(y(t_0),
y'(t_0))$ के लिए हल पूरे $I$ पर विद्यमान और अद्वितीय हैं; और समांगी हल
एक समतल बनाते हैं। और यदि कोई अलुप्त समांगी हल $y_1$ ज्ञात हो, तो
\emph{कोटि घटाकर} दूसरा स्वतंत्र हल मिल जाता है: $y = y_1 z$ रखने पर समांगी समीकरण
$z'$ के लिए प्रथम कोटि का समीकरण बन जाता है, जो समाकलनों से हल हो जाता है।
\end{proposition}

\begin{proof}
निकाय-रूप और \cref{thm:b2:diffeq:linear} सारी संरचनात्मक बातें दे देते हैं। कोटि घटाना: $y = y_1z$
प्रतिस्थापित करने पर
\[
y_1 z'' + (2y_1' + a y_1)z' + \underbrace{(y_1'' + ay_1' +
by_1)}_{=\,0}\,z = 0 :
\]
जो $u = z'$ में प्रथम कोटि का रैखिक समीकरण है और प्रथम वर्ष के सूत्र से हल हो
जाता है; $u$ का समाकलन $z$ देता है, अतः $y_2 = y_1 z$, जो $z$ के अनचर होने पर $y_1$ से
स्वतंत्र है।
\end{proof}

\begin{example}\label{ex:b2:diffeq:lowering}
$\intoo{0}{\infty}$ पर $t^2y'' - 2y = 0$: $y_1 = t^2$ एक हल है। $y = t^2z$ प्रतिस्थापित कीजिए: $y' = t^2z' + 2tz$ और $y''
= t^2z'' + 4tz' + 2z$ से,
\[
t^2y'' - 2y = t^4 z'' + 4t^3z' = 0,
\qquad\text{अर्थात्}\qquad \frac{z''}{z'} = -\frac4t :
\]
$z' = t^{-4}$ (किसी अचर तक), $z = -\frac{1}{3t^3}$, और $y_2
= t^2z = -\frac{1}{3t}$। सामान्य हल: $y = \alpha t^2 +
\frac{\beta}{t}$।
\end{example}

$A = \begin{pmatrix} \alpha & -\beta\\ \beta &
\alpha\end{pmatrix}$ के लिए $\eu^{tA} =
\eu^{\alpha t}\begin{pmatrix} \cos\beta t & -\sin\beta t\\
\sin\beta t & \cos\beta t\end{pmatrix}$ दो तरह से सिद्ध कीजिए --- सर्पिल प्रक्षेप-पथ: श्रेणी के
द्वारा ($A = \alpha I + \beta J$, $J^2 =
-I$ लिखिए), और सम्मिश्र पहचान $z' = (\alpha +
\iu\beta)z$ के द्वारा।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:6}
(ग्रोनवाल की प्रमेयिका) मान लीजिए $u$ \omterm{def:b2:metric:continuity}{संतत} अऋणात्मक है और $\intco{t_0}{T}$ पर $u(t)
\leq C + k\int_{t_0}^{t} u(s)\,\dd s$।
सिद्ध कीजिए कि $u(t) \leq C\,\eu^{k(t - t_0)}$ \emph{($v(t) =
\eu^{-kt}\int_{t_0}^t u$ का अवकलन कीजिए)}। इससे कोशी--लिप्शिट्स की
अद्वितीयता पुनः निकालिए, और भिन्न आरंभिक आँकड़ों वाले दो हलों के लिए \omterm{def:b2:metric:continuity}{संतत}
निर्भरता $\norm{y(t) -
z(t)} \leq \norm{y_0 - z_0}\,\eu^{k\abs{t - t_0}}$ भी।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:7}
यह जानते हुए कि $\intoo{0}{\pi}$ पर $y_1(t) = \frac{\sin t}{t}$ $ty'' + 2y' + ty =
0$ को हल करता है, कोटि घटाकर दूसरा स्वतंत्र
हल ज्ञात कीजिए, और सामान्य हल दीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffeq:8}
मान लीजिए $A \in \mathcal{M}_n(\C)$ के सारे अभिलक्षणिक मानों के वास्तविक भाग (यथार्थतः) ऋणात्मक
हैं। सिद्ध कीजिए कि $X' = AX$ का हर हल $t \to +\infty$ होने पर $0$ की ओर जाता है, और वह भी
घातांकी दर से: किसी $\alpha > 0$ के लिए $\norm{X(t)}
\leq C\,\eu^{-\alpha t}$। \emph{(त्रिभुजन कीजिए; और त्रिभुजाकार
निकाय को अंतिम पंक्ति से ऊपर की ओर निपटाइए, अथवा डनफ़र्ड काम में लीजिए:
$\norm{\eu^{tD}} \leq \eu^{-\alpha' t}$ तथा $t$ में बहुपद $\eu^{tN}$ के साथ $\eu^{tA} = \eu^{tD}\eu^{tN}$।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffeq:9}
(रैखिक वृद्धि के लिए परिमित समय में कोई पलायन नहीं) मान लीजिए $f$ \omterm{def:b2:metric:continuity}{संतत} है,
$\intco{0}{\infty} \times \R^n$ पर $\norm{f(t, y)} \leq a\norm y + b$, और $y$ में स्थानीय रूप से \omterm{def:b2:metric:continuity}{लिप्शिट्स}। समाकल रूप पर ग्रोनवाल
(\cref{exo:b2:diffeq:6}) का उपयोग करके सिद्ध कीजिए कि महत्तम हल वैश्विक हैं (पूरे $\intco{0}{\infty}$ पर
परिभाषित)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:diffeq:10}
$y'' - 3y' + 2y = \eu^{t}$ हल कीजिए: पहले समांगी हल, फिर $\alpha t\,\eu^{t}$ रूप का कोई विशिष्ट हल \emph{(भोला
अनुमान $\alpha\eu^t$ क्यों विफल हो जाता है?)}; सामान्य हल, और $y(0) = y'(0) = 0$ वाला हल।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:11}
जॉर्डन खंड
\[
A = \begin{pmatrix} \lambda & 1 & 0\\ 0 & \lambda & 1\\
0 & 0 & \lambda\end{pmatrix},
\]
के लिए $\eu^{tA}$ परिकलित कीजिए और $X' = AX$ के सारे हल बताइए: अर्थात् बहुपद सदिशों
गुणा घातांकी, जिनकी घात अधिकतम $2$ है। बहुपद की घात कहाँ से आती है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffeq:12}
(आवर्ती प्रणोदन, आवर्ती प्रतिक्रिया) मान लीजिए $A \in
\mathcal{M}_n(\R)$ और $B \colon \R \to \R^n$ \omterm{def:b2:metric:continuity}{संतत} तथा
$T$-आवर्ती है।
\begin{enumerate}
  \item दिखाइए कि $X' = AX + B(t)$ का कोई हल $T$-आवर्ती है यदि और केवल यदि $X(T) = X(0)$
        \emph{($X(\cdot + T)$ और $X$ की तुलना कीजिए)}।
  \item दिखाइए कि $\eu^{TA}$ के \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\eu^{T\lambda}$, $\lambda \in \operatorname{Sp}A$ हैं \emph{($\C$ पर
        त्रिभुजन कीजिए)}। इससे निष्कर्ष निकालिए: यदि $A$ का कोई
        \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\frac{2\iu\pi}{T}\Z$ में न हो, तो $I - \eu^{TA}$ प्रतिलोमनीय है।
  \item उसी परिकल्पना के अंतर्गत सिद्ध कीजिए कि निकाय का ठीक एक
        $T$-आवर्ती हल है, जहाँ
        \[
        X(0) = \bigl(I - \eu^{TA}\bigr)^{-1}
        \int_0^{T}\eu^{(T-s)A}B(s)\,\dd s .
        \]
        संनादी दोलक के लिए अपवर्जित स्थिति किसके अनुरूप है? (सप्ताहांत
        समस्या उत्तर देती है: \omterm{pb:b2:diffeq:1}{अनुनाद}।)
\end{enumerate}
\end{exercise}

\section{समस्या: दोलन, अनुनाद, और स्टुर्म की तुलना प्रमेयें}

\begin{problem}\label{pb:b2:diffeq:1}
एक ही समीकरण यांत्रिक और विद्युत जगत पर शासन करता है:
\[
x'' + 2\zeta\omega\,x' + \omega^2 x = F(t),
\qquad \omega > 0,\ \zeta \geq 0 .
\]
यह समस्या उसका पूरा अध्ययन करती है --- समतलीय रैखिक निकायों के
अनुरेख--सारणिक वर्गीकरण के द्वारा, तीनों अवमंदन-प्रणालियाँ, आवर्ती
प्रणोदन के लिए स्थायी-अवस्था प्रतिक्रिया अपने \emph{अनुनाद}\index{अनुनाद}
शिखर तथा अनुनाद-प्रलय के साथ --- और फिर अचर गुणांक छोड़कर \emph{स्टुर्म
की पृथक्करण और तुलना प्रमेयों}\index{स्टुर्म तुलना प्रमेय} तक जाती है, जो
$y'' + q(t)y = 0$ के हलों के शून्यों को बिना किसी सूत्र के नियंत्रित कर लेती हैं।

\medskip
\noindent\textbf{भाग I --- अनुरेख--सारणिक समतल।} मान लीजिए $A \in \mathcal M_2(\R)$, $\tau = \operatorname{tr}A$, $\delta
= \det A$,
$\Delta = \tau^2 - 4\delta$।
\begin{enumerate}
  \item दिखाइए कि $A$ के \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\frac{\tau\pm\sqrt\Delta}{2}$ हैं और वर्गीकरण कीजिए: विपरीत
        चिह्नों वाले दो वास्तविक \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} तभी जब $\delta < 0$; समान चिह्न
        वाले वास्तविक \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} तभी जब $\delta > 0$, $\Delta
        \geq 0$ ($\tau$ का चिह्न); और
        अवास्तविक संयुग्मी युग्म तभी जब $\Delta < 0$ (वास्तविक भाग $\frac\tau2$)।
  \item (काठी, $\delta < 0$) अभिलक्षणिक मानों $\mu < 0 <
        \lambda$ तथा अभिलक्षणिक सदिशों $v_\pm$ के
        साथ सामान्य हल लिखिए और प्रक्षेप-पथों का वर्णन कीजिए: दो स्थायी
        और दो अस्थायी किरणें, और शेष सारी कक्षाएँ दोनों की अनंतस्पर्शी।
        $0$ के अतिरिक्त कोई हल पूरे $\R$ पर परिबद्ध क्यों नहीं रह सकता?
  \item (नोड, $\delta > 0$, $\Delta > 0$) $\mu < \lambda
        < 0$ के लिए दिखाइए कि हर अशून्य हल $0$ की ओर
        जाता है और तीव्र अक्ष पर पड़ी कक्षाओं को छोड़कर सारी कक्षाएँ
        \emph{मंद अभिलक्षणिक दिशा के स्पर्शी होकर} पहुँचती हैं \emph{($\eu^{\mu t}$
        और $\eu^{\lambda t}$ की तुलना कीजिए)}।
  \item (सर्पिल और केंद्र, $\Delta < 0$) \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\alpha \pm \iu\beta$ लिखकर \cref{exo:b2:diffeq:5} का उपयोग
        कीजिए (किसी वास्तविक आधार-परिवर्तन के बाद, जिसे उस व्यापकता में
        स्वीकार कर लीजिए अथवा भाग II के निकायों के लिए सिद्ध कीजिए, और वे
        ही आगे काम आते हैं) और कक्षाओं का वर्णन कीजिए: $\alpha =
        \frac\tau2 < 0$ के लिए
        अभिसरित होते सर्पिल, $\tau > 0$ के लिए अपसरित होते, और $\tau = 0$ के लिए बंद
        वक्र (केंद्र)।
  \item (सीमा-स्थितियाँ) दोहरे \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} ($\Delta =
        0$) के लिए: दिखाइए कि
        शून्यंभावी $N
        = A - \lambda I$ के साथ $\eu^{tA} = \eu^{\lambda t}(I + tN)$, और तारा ($N = 0$) तथा अनुचित नोड ($N \neq 0$)
        में भेद कीजिए। भाग I का सारांश इस अध्याय के चित्र वाले
        अनुरेख--सारणिक चित्र में दीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- अवमंदित दोलक।} अब $F =
0$: $x'' + 2\zeta\omega x' + \omega^2x = 0$, अर्थात् $A = \begin{pmatrix} 0 & 1\\ -\omega^2 &
-2\zeta\omega\end{pmatrix}$ के
साथ $X' = AX$।
\begin{enumerate}[resume]
  \item $\tau, \delta, \Delta$ परिकलित कीजिए और तीनों प्रणालियों को अनुरेख--सारणिक समतल में
        रखिए: \emph{अल्प-अवमंदित} $0 < \zeta < 1$ (स्थायी सर्पिल), \emph{क्रांतिक
        अवमंदित} $\zeta = 1$ (दोहरा \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान}), \emph{अति-अवमंदित} $\zeta > 1$
        (स्थायी नोड); और $\zeta = 0$ केंद्र है।
  \item तीनों प्रणालियाँ स्पष्ट रूप से हल कीजिए:
        \[
        \zeta < 1:\ \eu^{-\zeta\omega t}\bigl(a\cos\omega_d t
        + b\sin\omega_dt\bigr),\ \omega_d =
        \omega\sqrt{1-\zeta^2};
        \qquad
        \zeta = 1:\ (a + bt)\,\eu^{-\omega t};
        \]
        $\zeta > 1$: दो वास्तविक घातांकी। छद्म-आवर्तकाल $\frac{2\pi}{\omega_d}$ परिभाषित कीजिए और
        दिखाइए कि $\abs x$ के क्रमागत महत्तम मानों का अनुपात अचर $\eu^{-2\pi\zeta/\sqrt{1-\zeta^2}}$ है
        (लघुगणकीय ह्रासांक)।
  \item (दरवाज़ा-बंद करने वाला सिद्धांत) $\zeta \geq 1$ के लिए क्षय-दर सबसे मंद
        \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\lambda_{\mathrm{slow}} = -\omega\bigl(\zeta -
        \sqrt{\zeta^2-1}\bigr)$ से शासित होती है। दिखाइए कि $\abs{\lambda_{\mathrm{slow}}} =
        \frac{\omega}{\zeta + \sqrt{\zeta^2 - 1}}$ $\zeta \geq 1$ का
        \emph{ह्रासमान} फलन है: अर्थात् क्रांतिक अवमंदन $\zeta = 1$ विश्राम तक
        सबसे तेज़ अदोलनकारी वापसी देता है।
  \item (ऊर्जा) $E(t) = \frac12x'^2 +
        \frac12\omega^2x^2$ लीजिए। सिद्ध कीजिए कि $E' = -2\zeta\omega\,x'^2
        \leq 0$, और इससे निष्कर्ष
        निकालिए कि $\zeta > 0$ के लिए समीकरण का कोई अशून्य आवर्ती हल नहीं है
        \emph{(कोई आवर्तकाल $E$ को अचर होने पर बाध्य कर देता, अतः $x' \equiv 0$)}।
  \item दो वाक्यों में समझाइए कि केंद्र $\zeta = 0$ \emph{संरचनात्मक रूप से
        नाज़ुक} क्यों है: कोई भी $\zeta > 0$, चाहे कितना ही छोटा, आवर्तिता नष्ट
        कर देता है --- और अनुरेख--सारणिक समतल में यह कहाँ दिखता है
        (केंद्र-रेखा का अभ्यंतर रिक्त है)।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- प्रणोदित दोलन और \omterm{pb:b2:diffeq:1}{अनुनाद}।} अब $F, \gamma > 0$ के साथ $F(t) = F\cos(\gamma t)$।
\begin{enumerate}[resume]
  \item ($\zeta > 0$: स्थायी अवस्था) $x_p =
        \Re\bigl(z\,\eu^{\iu\gamma t}\bigr)$ खोजिए: दिखाइए
        \[
        z = \frac{F}{\omega^2 - \gamma^2 +
        2\iu\zeta\omega\gamma},
        \qquad
        A(\gamma) := \abs z = \frac{F}{\sqrt{(\omega^2 -
        \gamma^2)^2 + 4\zeta^2\omega^2\gamma^2}} ,
        \]
        और $\tan\varphi =
        \frac{2\zeta\omega\gamma}{\omega^2-\gamma^2}$ के साथ $x_p = A(\gamma)\cos(\gamma t - \varphi)$ लिखिए।
  \item दिखाइए कि \emph{हर} हल $x_p$ जमा भाग II का कोई क्षणिक पद है, और वह
        $0$ की ओर जाता है: अर्थात् आरंभिक आँकड़े चाहे जो हों, निकाय
        स्थायी अवस्था पर जम जाता है --- आयाम $A(\gamma)$, कला-पश्चता $\varphi$।
  \item (\omterm{pb:b2:diffeq:1}{अनुनाद} वक्र) $A$ को महत्तम कीजिए: दिखाइए कि $A(\gamma)$ का कोई
        अभ्यंतरीय महत्तम तभी है जब $\zeta <
        \frac{1}{\sqrt2}$, और वह
        \[
        \gamma_* = \omega\sqrt{1 - 2\zeta^2},
        \qquad
        A(\gamma_*) =
        \frac{F}{2\zeta\omega^2\sqrt{1-\zeta^2}} ,
        \]
        पर है; तथा छोटे $\zeta$ के लिए वह शिखर स्थैतिक प्रतिक्रिया $A(0) = \frac F{\omega^2}$ को
        गुणक $\approx \frac{1}{2\zeta}$ से बढ़ा देता है।
  \item ($\zeta = 0$, \omterm{pb:b2:diffeq:1}{अनुनाद} से हटकर) $\gamma \neq \omega$ के लिए दिखाइए कि $x(0) = x'(0) = 0$ वाला हल
        \[
        x(t) = \frac{F}{\omega^2 -
        \gamma^2}\bigl(\cos\gamma t - \cos\omega t\bigr)
        = \frac{2F}{\omega^2-\gamma^2}
        \sin\frac{(\omega-\gamma)t}{2}
        \sin\frac{(\omega+\gamma)t}{2} :
        \]
        है: परिबद्ध, और जब $\gamma$ $\omega$ के निकट हो तब \emph{विस्पंद} ---
        अर्थात् किसी मंद लिफ़ाफ़े के नीचे तेज़ दोलन।
  \item ($\zeta = 0$, \omterm{pb:b2:diffeq:1}{अनुनाद}) $\gamma = \omega$ के लिए दिखाइए कि $x_p(t) = \frac{F}{2\omega}\,t\sin(\omega t)$ एक हल है, और $\gamma \to \omega$ होने
        पर उसे प्रश्न 14 की सीमा के रूप में पुनः प्राप्त कीजिए: आयाम सदा
        के लिए रैखिक रूप से बढ़ता है --- अर्थात् अनुनाद-प्रलय।
  \item (फूरिये संबंध) कोई भी आवर्ती प्रणोदन संनादियों में विघटित होता है
        (फूरिये अध्याय); और रैखिकता से स्थायी अवस्था संनादी प्रतिक्रियाओं
        का योग है। आवृत्ति $\omega = 3$ वाले अनवमंदित दोलक को \cref{exo:b2:fourier:1}-प्रकार की वर्ग
        तरंग (जिसकी संनादियाँ सभी विषम पूर्णांकों पर हैं) से प्रणोदित
        किया जाए, तो कौन-सी संनादी \omterm{pb:b2:diffeq:1}{अनुनाद} करेगी? एक वाक्य में बताइए कि
        अभियंता वर्ग तरंगों से क्यों डरते हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- स्टुर्म की प्रमेयें।} किसी अंतराल $I$ पर $y''
+ q(t)\,y = 0$
लीजिए, जहाँ $q$ \omterm{def:b2:metric:continuity}{संतत} है। (कोई भी समीकरण $y'' + ay' + by = 0$ प्रतिस्थापन $y = u\exp\bigl(-\frac12\int a\bigr)$ से इसी
सामान्य रूप में आ जाता है; और प्रश्न 21 उस युक्ति का एक रूपांतर क्रिया में
दिखाता है।)
\begin{enumerate}[resume]
  \item दो हलों $y_1, y_2$ के लिए दिखाइए कि \omterm{def:b2:diffeq:wronskian}{व्रोंस्कियन} $W = y_1y_2' - y_1'y_2$ \emph{अचर} है और वह
        शून्य है तभी जब हल अनुपाती हों; और यह भी कि किसी अशून्य हल के
        केवल \emph{सरल, विलगित} शून्य होते हैं।
  \item (स्टुर्म पृथक्करण) मान लीजिए $y_1, y_2$ स्वतंत्र हल हैं और $a < b$ $y_1$ के दो
        क्रमागत शून्य। सिद्ध कीजिए कि $y_2$ $\intoo{a}{b}$ में ठीक एक बार लुप्त होता
        है \emph{(अचर $W$ का $a$ तथा $b$ पर मूल्यांकन कीजिए: वहाँ $W = y_1'y_2$,
        और $y_1'(a)$, $y_1'(b)$ के चिह्न विपरीत हैं)}: अर्थात् स्वतंत्र हलों के
        शून्य एक-दूसरे के बीच में आते हैं।
  \item (स्टुर्म तुलना) मान लीजिए $I$ पर $q_1 \leq q_2$, $y'' + q_1y = 0$ सहित $y \neq
        0$, $z'' + q_2z
        = 0$ सहित
        $z \neq 0$, और $a < b$ $y$ के क्रमागत शून्य। दिखाइए कि $z$ $\intcc{a}{b}$ में लुप्त
        होता है --- और यदि $\intoo ab$ पर कहीं $q_1 < q_2$ हो तो यथार्थतः भीतर \emph{(यदि
        $\intoo ab$ पर $z \neq
        0$ हो, तो $y, z$ के स्थिर चिह्नों के साथ $(yz' - y'z)' = (q_1 -
        q_2)yz$ का अध्ययन
        कीजिए और सीमा-मानों की तुलना कीजिए)}।
  \item \emph{अंतराल-परिबंध} निष्कर्ष रूप में निकालिए: यदि $I$ पर $0 < m^2 \leq
        q(t) \leq M^2$,
        तो $y'' + qy = 0$ के किसी अशून्य हल के किन्हीं दो क्रमागत शून्य $a < b$ पूरा
        करते हैं
        \[
        \frac{\pi}{M} \;\leq\; b - a \;\leq\; \frac{\pi}{m}
        \]
        \emph{($u'' + M^2u = 0$ तथा $u'' + m^2u
        = 0$ से तुलना कीजिए, जिनके शून्य क्रमशः $\frac\pi M$ और
        $\frac\pi m$ दूरी पर हैं)}। इसे संनादी दोलक पर जाँचिए।
  \item $ty'' + 2y' + ty = 0$ (\cref{exo:b2:diffeq:7}) को $u = ty$ से $u'' + u =
        0$ में बदलिए, उसके हल $\frac{\sin t}t$, $\frac{\cos t}{t}$ तत्काल
        पुनः प्राप्त कीजिए, और निष्कर्ष निकालिए कि हर अशून्य हल के शून्य
        ठीक $\pi$ दूरी पर हैं: यही स्टुर्म का जगत-दर्शन है --- शून्यों पर
        गुणांक $q$ का शासन है, सूत्र हों या न हों।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- द्यूआमेल और परिबद्धता की सीमा।}
\begin{enumerate}[resume]
  \item (दोलक के लिए द्यूआमेल) दिखाइए कि \omterm{def:b2:metric:continuity}{संतत} $F$ के लिए $x(0) = x'(0) = 0$ सहित $x'' + \omega^2x =
        F(t)$ का
        हल
        \[
        x(t) = \frac1\omega\int_0^t\sin\bigl(\omega(t -
        s)\bigr)F(s)\,\dd s ,
        \]
        है, और उससे $F(s) = F\cos(\omega s)$ लेकर प्रश्न 15 का अनुनादी हल पुनः निकालिए
        \emph{(गुणनफल-से-योग)}।
  \item ($\zeta > 0$: परिबद्ध निवेश, परिबद्ध निर्गम) दिखाइए कि $\zeta > 0$ और \emph{किसी
        भी} परिबद्ध \omterm{def:b2:metric:continuity}{संतत} $F$ के लिए अवमंदित समीकरण का हर हल $\intco{0}{\infty}$ पर
        परिबद्ध है \emph{(अचरों का विचरण और \cref{exo:b2:diffeq:8} का घातांकी क्षय $\vertiii{\eu^{tA}} \leq
        C\eu^{-\alpha t}$)}।
  \item ($\zeta = 0$) दिखाइए कि बिना अवमंदन के परिबद्ध आवर्ती प्रणोदन सारे हलों
        को परिबद्ध रखता है, सिवाय ठीक \omterm{pb:b2:diffeq:1}{अनुनाद} पर ($\gamma =
        \omega$, प्रश्न 15 बनाम
        प्रश्न 14): अर्थात् परिबद्धता की सीमा को एकसमान स्थायित्व अवमंदन
        ही बनाता है।
  \item संश्लेषण। एक-एक वाक्य में: (क) अनुरेख--सारणिक समतल भाग I और II को
        कैसे संगठित करता है और प्रणोदन (भाग III) उससे कहाँ बाहर निकल जाता
        है; (ख) $\gamma_*$, $A(\gamma_*)$ तथा गुणक $\frac1{2\zeta}$ का भौतिक अर्थ; (ग) स्टुर्म की
        प्रमेयें वह क्या कहती हैं जो स्पष्ट सूत्र नहीं कह सकते; (घ) इस
        समस्या के कौन-से दो परिणाम पुस्तक का शेष भाग चुपचाप फिर काम में
        लेगा (ल्यूविल-अचर \omterm{def:b2:diffeq:wronskian}{व्रोंस्कियन}; परिबद्ध-निवेश स्थायित्व)।
\end{enumerate}
\end{problem}
