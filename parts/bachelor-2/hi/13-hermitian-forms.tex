\chapter{हर्मीशियन रूप}\label{ch:b2:hermitian}

सम्मिश्र सदिश समष्टियों की अपनी अंतर्गुणन-ज्यामिति होती है, एक मोड़ के
साथ: एक चर में रैखिकता, दूसरे में \emph{संयुग्मी}-रैखिकता। इस मोड़ को
स्वीकार करने का लाभ वास्तविक से भी स्वच्छ \omterm{def:b2:reduction:eigen}{वर्णक्रम} सिद्धांत है ---
\omterm{def:b2:hermitian:adjoint}{हर्मीशियन} अंतःरूपांतरणों के \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} वास्तविक होते हैं, \omterm{def:b2:hermitian:adjoint}{एकात्मक} के
मापांक एक, और दोनों प्रसामान्य लांबिक आधारों में \omterm{def:b2:reduction:diag}{विकर्णनीय} होते हैं। यह
छोटा अध्याय \cref{ch:b2:quadratic} का यूक्लिडीय कार्यक्रम $\C$ पर चलाता है।

\section{हर्मीशियन अंतर्गुणन}

\begin{definition}\label{def:b2:hermitian:def}
किसी सम्मिश्र सदिश समष्टि $E$ पर \emph{हर्मीशियन
अंतर्गुणन}\index{हर्मीशियन अंतर्गुणन} ऐसा प्रतिचित्रण $\langle\cdot,\cdot\rangle
\colon E \times E \to \C$ है जो दूसरे चर
में रैखिक हो, संयुग्मी-सममित हो ($\langle y, x\rangle = \conj{\langle x,
y\rangle}$ --- अतः पहले चर में संयुग्मी-रैखिक),
और धनात्मक निश्चित ($x \neq 0$ के लिए $\langle x, x\rangle > 0$)। $\C^n$ पर मानक उदाहरण:
\[
\langle x, y \rangle = \sum_{i=1}^{n} \conj{x_i}\, y_i ;
\]
और \omterm{def:b2:metric:continuity}{संतत} फलनों पर $\langle f, g\rangle = \int_a^b \conj
f\,g$। \omterm{def:b2:nvs:norm}{मानदंड}: $\norm x = \sqrt{\langle x,x\rangle}$; और इस प्रकार सज्जित परिमित-विमीय
सम्मिश्र समष्टि \emph{हर्मीशियन समष्टि}\index{हर्मीशियन समष्टि} कहलाती है।
\end{definition}

\begin{example}[पहले परिकलन]\label{ex:b2:hermitian:firstcomp}
$\C^2$ में $x = (1+\iu,\ 2-\iu)$ और $y = (\iu,\ 1)$ लीजिए। तब
\[
\norm x^2 = \abs{1+\iu}^2 + \abs{2-\iu}^2 = 2 + 5 = 7,
\qquad
\norm y^2 = 1 + 1 = 2,
\]
और पहली कोटि का संयुग्म लेने पर,
\[
\langle x, y\rangle
= \conj{(1+\iu)}\,\iu + \conj{(2-\iu)}\cdot1
= (1-\iu)\iu + (2+\iu)
= (\iu + 1) + (2 + \iu) = 3 + 2\iu .
\]
कोशी--श्वार्ज़ सही निकलता है: $\abs{\langle x, y\rangle}^2 = 9 +
4 = 13 \leq 14 = \norm x^2\norm y^2$ --- और समता के निकट, क्योंकि $x$ $y$ के
किसी गुणज के निकट है। यह भी ध्यान दीजिए कि $\langle y, x\rangle = \conj{3 + 2\iu} = 3 - 2\iu$: अर्थात् संयुग्मी-सममितता
क्रिया में, और यही कारण है कि $\langle x,
x\rangle$ सदा वास्तविक होता है।
\end{example}

\begin{theorem}[कोशी--श्वार्ज़, सम्मिश्र]\label{thm:b2:hermitian:cs}
$\abs{\langle x, y\rangle} \leq \norm x \norm y$, और समता तभी जब $x, y$ रैखिकतः परतंत्र हों; तथा $\norm\cdot$ एक \omterm{def:b2:nvs:norm}{मानदंड} है। इसके
अतिरिक्त प्रसामान्य लांबिक आधार विद्यमान होते हैं (ग्राम--श्मिट अक्षरशः
चलता है), जहाँ
\[
x = \sum_i \langle e_i, x\rangle\, e_i,
\qquad
\norm x^2 = \sum_i \abs{\langle e_i, x\rangle}^2 .
\]
\end{theorem}

\begin{proof}
$y \neq 0$ और $t \in \C$ के लिए: $0 \leq \norm{x - ty}^2 = \norm x^2
- 2\Re\bigl(\conj t\langle y, x\rangle\bigr) + \abs t^2\norm y^2$। $t = \frac{\langle y, x\rangle}{\norm y^2}$ चुनिए:
\[
0 \leq \norm x^2 - \frac{\abs{\langle y, x\rangle}^2}{\norm y^2},
\]
जो वही असमिका है; और समता $x = ty$ को बाध्य कर देती है। त्रिभुज असमिका, पूरी:
\[
\norm{x + y}^2 = \norm x^2 + 2\,\Re\langle x, y\rangle
+ \norm y^2
\leq \norm x^2 + 2\,\abs{\langle x, y\rangle} + \norm y^2
\leq \bigl(\norm x + \norm y\bigr)^2 ,
\]
जिसमें $\Re z \leq \abs z$ और फिर कोशी--श्वार्ज़ का उपयोग हुआ; समघातता और पृथक्करण तत्काल
हैं, अतः $\norm\cdot$ एक \omterm{def:b2:nvs:norm}{मानदंड} है। ग्राम--श्मिट: वास्तविक स्थिति की तरह, बस
संयुग्म परिभाषा के अनुसार लगाए जाते हैं (परिपाटी पर ध्यान दीजिए: हमारे
गुणन \emph{पहले} खाँचे में संयुग्मी-रैखिक हैं, अतः निर्देशांक $\langle e_i, x\rangle$ होते
हैं; और अगला उदाहरण कलनविधि को एक बार पूरा चलाता है)।
\end{proof}

\begin{example}[सम्मिश्र ग्राम--श्मिट, पूरा चलाया हुआ]\label{ex:b2:hermitian:gramschmidt}
$\C^2$ के आधार $v_1 = (1, \iu)$, $v_2 = (0, 1)$ को प्रसामान्य लांबिक बनाइए। पहला सदिश: $\norm{v_1}^2 = \abs1^2 + \abs\iu^2 = 2$, अतः $e_1 = \frac{1}{\sqrt2}(1, \iu)$।
अब $v_2$ का प्रक्षेप --- पहले खाँचे में संयुग्म के साथ:
\[
\langle e_1, v_2\rangle
= \frac{1}{\sqrt2}\bigl(\conj{1}\cdot0 +
\conj{\iu}\cdot1\bigr)
= \frac{-\iu}{\sqrt2} ,
\qquad
v_2 - \langle e_1, v_2\rangle e_1
= (0,1) + \frac{\iu}{2}\,(1, \iu)
= \Bigl(\frac\iu2,\ \frac12\Bigr) .
\]
उसका \omterm{def:b2:nvs:norm}{मानदंड} $\sqrt{\frac14 + \frac14} = \frac{1}{\sqrt2}$ है: $e_2 = \frac{1}{\sqrt2}(\iu, 1)$। जाँच: $\langle e_1,
e_2\rangle = \frac12(\conj1\cdot\iu + \conj\iu\cdot1) =
\frac12(\iu - \iu) = 0$। नए आधार में $v_2$ के निर्देशांक:
$\langle e_2, v_2\rangle = \frac{1}{\sqrt2}$ सहित $v_2 = \langle e_1, v_2\rangle e_1 + \langle e_2, v_2\rangle
e_2$ --- क्रम का ध्यान रखिए: $\langle v_2, e_1\rangle$ \emph{संयुग्मी} गुणांक देता।
समापन दृष्टि: कलनविधि अक्षरशः यूक्लिडीय वाली है; एकमात्र जाल यह है कि
संयुग्म कहाँ पड़ता है, और $\norm{v_2 -
\text{प्रक्षेप}}^2 > 0$ का परिकलन चुपचाप
धनात्मकता का उपयोग करता है --- वही अभिगृहीत जो पूरी ज्यामिति चलाता है।
\end{example}

\section{सहसंलग्न, हर्मीशियन और एकात्मक अंतःरूपांतरण}

\begin{definition}\label{def:b2:hermitian:adjoint}
$u \in \mathcal{L}(E)$ का \emph{सहसंलग्न} $u^*$ $\langle u^*(x), y\rangle = \langle x, u(y)\rangle$ से परिभाषित होता है; और किसी प्रसामान्य
लांबिक आधार में $\operatorname{Mat}(u^*) = \conj{A}^{\mathsf T} =:
A^{\dagger}$ (संयुग्मी \omterm{def:b2:linalg:transpose}{परिवर्त}) --- वास्तव में यदि $B =
(b_{ij})$ प्रसामान्य
लांबिक आधार $(e_i)$ में $u^*$ का आव्यूह हो, तो $b_{ij} = \langle e_i, u^*(e_j)\rangle$, और परिभाषा वाली सर्वसमिका
देती है
\[
\conj{b_{ij}}
= \langle u^*(e_j), e_i\rangle
= \langle e_j, u(e_i)\rangle = a_{ji} ,
\qquad\text{अतः}\qquad
B = \conj{A}^{\mathsf T} .
\] $u$ \emph{हर्मीशियन}\index{हर्मीशियन
अंतःरूपांतरण} कहलाता है जब $u^* = u$ ($A^\dagger = A$), \emph{एकात्मक}\index{एकात्मक
समूह} जब $u^*u =
\mathrm{id}$ ($A^\dagger A = I$: समूह $U(n)$), और \emph{प्रसामान्य} जब $u^*u = uu^*$।
\end{definition}

\begin{proposition}\label{prop:b2:hermitian:eigenvalues}
किसी \omterm{def:b2:hermitian:adjoint}{हर्मीशियन अंतःरूपांतरण} के \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} \emph{वास्तविक} होते हैं;
किसी \omterm{def:b2:hermitian:adjoint}{एकात्मक} अंतःरूपांतरण के अभिलक्षणिक मानों का \emph{मापांक $1$} होता
है; और दोनों स्थितियों में भिन्न अभिलक्षणिक मानों की अभिलक्षणिक समष्टियाँ
लांबिक होती हैं।
\end{proposition}

\begin{proof}
\omterm{def:b2:hermitian:adjoint}{हर्मीशियन}, $u(x) = \lambda x$, $x \neq 0$:
\[
\lambda \norm x^2 = \langle x, u(x)\rangle = \langle u(x), x\rangle
= \conj{\langle x, u(x)\rangle} = \conj\lambda\,\norm x^2 ,
\]
अतः $\lambda \in \R$। \omterm{def:b2:hermitian:adjoint}{एकात्मक}: $\norm{u(x)} = \norm x$ ($u^*u =
\mathrm{id}$ से), अतः $\abs\lambda\norm x = \norm x$। लांबिकता (\omterm{def:b2:hermitian:adjoint}{हर्मीशियन} स्थिति):
वास्तविक $\lambda \neq \mu$ वाले अभिलक्षणिक सदिशों के लिए $\lambda\langle x, y\rangle = \langle u(x),
y\rangle = \langle x, u(y)\rangle = \mu\langle x, y\rangle$। और \omterm{def:b2:hermitian:adjoint}{एकात्मक} स्थिति, पूरी:
$u(x) = \lambda x$ के लिए $\lambda \neq
\mu$ (दोनों इकाई-मापांक) सहित $u(y) = \mu y$,
\[
\langle x, y\rangle = \langle u(x), u(y)\rangle
= \conj\lambda\mu\,\langle x, y\rangle ,
\]
और $\conj\lambda\mu = \frac{\mu}{\lambda} \neq 1$: गुणक $1$ नहीं है, अतः $\langle x, y\rangle = 0$।
\end{proof}

\begin{example}[एक विषम-हर्मीशियन आव्यूह, विकर्णित]\label{ex:b2:hermitian:skewexample}
$A = \begin{pmatrix} 0 & -2\\ 2 & 0\end{pmatrix}$ $A^\dagger = A^{\mathsf T} = -A$ पूरा करता है: अर्थात् विषम-हर्मीशियन (और वास्तविक प्रतिसममित भी
--- $\R$ पर उसका कोई \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} है ही नहीं)। \omterm{def:b2:reduction:charpoly}{अभिलक्षणिक बहुपद} $X^2 + 4$:
\omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\pm2\iu$, जो विशुद्ध काल्पनिक हैं, ठीक जैसा \cref{exo:b2:hermitian:9} \omterm{def:b2:funcseq:series}{सामान्य रूप से}
बताती है। \omterm{def:b2:reduction:eigen}{अभिलक्षणिक सदिश}: $(A - 2\iu I)v = 0$ $v_1 =
\frac{1}{\sqrt2}(1, \iu)$ देता है, और $-2\iu$ के लिए $v_2 = \frac{1}{\sqrt2}(1,
-\iu)$; और वे
लांबिक हैं:
\[
\langle v_1, v_2\rangle
= \tfrac12\bigl(\conj{1}\cdot1 +
\conj{\iu}\cdot(-\iu)\bigr)
= \tfrac12(1 - 1) = 0 .
\]
अतः \omterm{def:b2:hermitian:adjoint}{एकात्मक} $U =
(v_1\ v_2)$ सहित $A = U\operatorname{diag}(2\iu, -2\iu)\,U^\dagger$। समापन दृष्टि: $H = -\iu A =
\begin{pmatrix} 0 & 2\iu\\ -2\iu & 0\end{pmatrix}$ वास्तविक \omterm{def:b2:reduction:eigen}{वर्णक्रम} $\{\pm2\}$
और \emph{उन्हीं} अभिलक्षणिक सदिशों के साथ \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} है --- अर्थात्
\omterm{def:b2:hermitian:adjoint}{हर्मीशियन} और विषम-हर्मीशियन अंतःरूपांतरणों के बीच का एकैकी आच्छादक
प्रतिचित्रण $u \mapsto \iu u$ (\cref{exo:b2:hermitian:9}), आव्यूह-दर-आव्यूह देखा हुआ; और $\R$ पर वही $A$
बिना किसी \omterm{def:b2:reduction:eigen}{अभिलक्षणिक सदिश} का घूर्णन-मापन है, तथा केवल $\C$ तक जाने पर ही
उसका प्रसामान्य रूप प्रकट होता है।
\end{example}

\begin{theorem}[हर्मीशियन वर्णक्रमीय प्रमेय]\label{thm:b2:hermitian:spectral}
किसी \omterm{def:b2:hermitian:def}{हर्मीशियन समष्टि} के प्रत्येक \omterm{def:b2:hermitian:adjoint}{हर्मीशियन अंतःरूपांतरण} के अभिलक्षणिक
सदिशों का कोई प्रसामान्य लांबिक आधार होता है (वास्तविक अभिलक्षणिक मानों के
साथ): $A^\dagger = A$ से $U \in U(n)$ तथा वास्तविक विकर्ण $D$ सहित $A = U D U^{\dagger}$ निकलता
है।\index{वर्णक्रमीय प्रमेय!हर्मीशियन}
\end{theorem}

\begin{proof}
$\C$ पर \omterm{def:b2:reduction:charpoly}{अभिलक्षणिक बहुपद} पूरी तरह गुणनखंडित हो जाता है: अतः कोई
\omterm{def:b2:reduction:eigen}{अभिलक्षणिक सदिश} $e_1$ विद्यमान है (\cref{ch:b2:reduction}) --- यहाँ किसी संहतता-तर्क की
आवश्यकता नहीं, और यह $\C$ का एक लाभ है। उसे सामान्यीकृत कीजिए। उसका लांबिक
पूरक $F = e_1^\perp$ स्थायी है: $x \perp e_1$ के लिए
\[
\langle e_1, u(x)\rangle = \langle u(e_1), x\rangle
= \lambda_1\langle e_1, x\rangle = 0
\]
($\lambda_1$ वास्तविक)। प्रतिबंधन \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} है; अब विमा पर आगमन कीजिए और जोड़
दीजिए। विस्तार से: प्रतिबंधन $u|_F$ \omterm{def:b2:hermitian:def}{हर्मीशियन समष्टि} $F$ (विमा $n - 1$) का
अंतःरूपांतरण है जिसे $u$ से $\langle u|_F(x), y\rangle = \langle x, u|_F(y)\rangle$ विरासत में मिला है; आगमन परिकल्पना $F$
के अभिलक्षणिक सदिशों का प्रसामान्य लांबिक आधार $(e_2, \dots, e_n)$ देती है, और $(e_1, e_2, \dots, e_n)$ $E$
में प्रसामान्य लांबिक है ($e_1 \perp
F$) तथा $u$ के अभिलक्षणिक सदिशों से बना है।
आव्यूह में अनुवाद: $U$ के स्तंभ $e_i$ हैं, $U^\dagger U = I$ उनकी प्रसामान्य लांबिकता
व्यक्त करता है, और $AU = UD$ अभिलक्षणिक-मान समीकरण इकट्ठे करता है, जिससे
वास्तविक विकर्ण $D$ सहित $A = UDU^\dagger$ (\cref{prop:b2:hermitian:eigenvalues})।
\end{proof}

\begin{example}\label{ex:b2:hermitian:pauli}
$A = \begin{pmatrix} 0 & -\iu\\ \iu & 0\end{pmatrix}$ \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} है ($A^\dagger = A$): $\chi_A = X^2 - 1$ से \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\pm 1$ (वास्तविक, जैसा वचन
था), और प्रसामान्य लांबिक \omterm{def:b2:reduction:eigen}{अभिलक्षणिक सदिश} $\frac{1}{\sqrt2}(1, \iu)^{\mathsf T}$ तथा $\frac{1}{\sqrt2}(1,
-\iu)^{\mathsf T}$। (भौतिकी वाले $A$
को पाउली आव्यूह के रूप में जानते हैं; और \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} वर्णक्रमों का वास्तविक
होना ही वह कारण है जिससे क्वांटम प्रेक्षणीय राशियाँ \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} संकारकों से
प्रतिरूपित की जाती हैं।)
\end{example}

\begin{example}[एक धनात्मक निश्चित हर्मीशियन आव्यूह, किया हुआ]\label{ex:b2:hermitian:pdrun}
$A = \begin{pmatrix} 2 & 1-\iu\\ 1+\iu & 3\end{pmatrix}$: यह \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} है, क्योंकि विकर्ण वास्तविक है और विकर्णेतर
प्रविष्टियाँ संयुग्मी। \omterm{def:b2:reduction:charpoly}{अभिलक्षणिक बहुपद}:
\[
(2-\lambda)(3-\lambda) - \abs{1-\iu}^2
= \lambda^2 - 5\lambda + 4
= (\lambda - 1)(\lambda - 4) :
\]
\omterm{def:b2:reduction:eigen}{वर्णक्रम} $\{1, 4\}$, जो वास्तविक और धनात्मक है --- अतः $A$ धनात्मक निश्चित है।
\omterm{def:b2:reduction:eigen}{अभिलक्षणिक सदिश}: $\lambda = 4$ के लिए निकाय $(A -
4I)v = 0$ $v_4 = (1 - \iu,\ 2)$ देता है (दूसरी पंक्ति जाँचिए:
$(1+\iu)(1-\iu) - 2 = 0$); और $\lambda = 1$ के लिए $v_1 = (1 - \iu,\
-1)$। लांबिकता, पहले खाँचे में संयुग्म के साथ:
\[
\langle v_4, v_1\rangle
= \conj{(1-\iu)}\,(1-\iu) + \conj{2}\,(-1)
= 2 - 2 = 0 . \checkmark
\]
सामान्यीकरण ($\norm{v_4}^2 = 2 + 4 = 6$, $\norm{v_1}^2 = 2 + 1
= 3$) करने पर $A =
U\operatorname{diag}(4,1)U^\dagger$ सहित \omterm{def:b2:hermitian:adjoint}{एकात्मक} $U = \bigl(\frac{v_4}{\sqrt6}\
\frac{v_1}{\sqrt3}\bigr)$ मिलता है। समापन
दृष्टि: रैले-पाठ तत्काल है --- $\C^2$ के इकाई गोलक पर $\langle x, Ax\rangle$ $\intcc{1}{4}$ में घूमता है
और दोनों अभिलक्षणिक सदिशों पर प्राप्त होता है; और यही उस कूराँ--फिशर
सिद्धांत का $n = 2$ बीज है जो सप्ताहांत समस्या में खड़ा किया गया है।
अभिलक्षणिक सदिशों से हटकर भी रूप के वास्तविक होने की जाँच कीजिए: $x = (1, \iu)$ पर
\[
Ax = \bigl(2 + (1-\iu)\iu,\ (1+\iu) + 3\iu\bigr)
= (3 + \iu,\ 1 + 4\iu),
\]
\[
\langle x, Ax\rangle = \conj{1}\,(3+\iu) +
\conj{\iu}\,(1+4\iu)
= (3 + \iu) + (-\iu)(1 + 4\iu) = 3 + \iu - \iu + 4 = 7 \in
\R ,
\]
जैसा \cref{prop:b2:hermitian:eigenvalues} की उपपत्ति की कार्यविधि ($A^\dagger = A$ के विरुद्ध संयुग्मी-सममितता)
प्रत्येक $x$ के लिए गारंटी देती है।
\end{example}

\begin{example}[एक एकात्मक आव्यूह विकर्णित]\label{ex:b2:hermitian:unitary}
$U = \frac{1}{\sqrt2}\begin{pmatrix} 1 & \iu\\ \iu & 1
\end{pmatrix}$ (\cref{exo:b2:hermitian:2} से \omterm{def:b2:hermitian:adjoint}{एकात्मक})। उसका \omterm{def:b2:reduction:charpoly}{अभिलक्षणिक बहुपद} $\bigl(X - \frac{1}{\sqrt2}\bigr)^2
+ \frac12$ है, जिसके मूल
\[
\lambda_\pm = \frac{1 \pm \iu}{\sqrt2} = \eu^{\pm\iu\pi/4},
\]
हैं, और वे इकाई-मापांक हैं जैसा \cref{prop:b2:hermitian:eigenvalues} ने वचन दिया था; तथा प्रसामान्य लांबिक
\omterm{def:b2:reduction:eigen}{अभिलक्षणिक सदिश} $\frac{1}{\sqrt2}(1, \pm1)$ हैं। अतः $U =
V\operatorname{diag}(\eu^{\iu\pi/4},
\eu^{-\iu\pi/4})V^\dagger$: अर्थात् उपयुक्त आधार में $U$ $\pm\frac\pi4$ से दो
समतलीय घूर्णनों का युग्म है --- किसी वास्तविक घूर्णन आव्यूह का कोई
वास्तविक \omterm{def:b2:reduction:eigen}{अभिलक्षणिक सदिश} नहीं होता, पर $\C$ पर वह दो इकाई-मापांक अदिशों
में बँट जाता है। समापन दृष्टि: \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} \omterm{def:b2:reduction:eigen}{वर्णक्रम} वास्तविक रेखा पर बसते
हैं, \omterm{def:b2:hermitian:adjoint}{एकात्मक} \omterm{def:b2:reduction:eigen}{वर्णक्रम} इकाई वृत्त पर; दोनों उसी प्रसामान्यता की परछाइयाँ
हैं, और \cref{exo:b2:hermitian:6} का केली रूपांतरण एक चित्र को दूसरे पर भेज देता है।
\end{example}

\begin{example}[केली रूपांतरण, परिकलित]\label{ex:b2:hermitian:cayleyrun}
$H = \begin{pmatrix} 0 & 1\\
1 & 0\end{pmatrix}$ पर \cref{exo:b2:hermitian:6} चलाइए (\omterm{def:b2:hermitian:adjoint}{हर्मीशियन}, \omterm{def:b2:reduction:eigen}{वर्णक्रम} $\{1, -1\}$, प्रसामान्य लांबिक \omterm{def:b2:reduction:eigen}{अभिलक्षणिक
सदिश} $\frac{1}{\sqrt2}(1, \pm1)$)। अभिलक्षणिक आधार में सब कुछ अदिश है: रूपांतरण $\lambda
\mapsto \frac{\lambda - \iu}{\lambda + \iu}$ भेजता है
\[
1 \longmapsto \frac{1 - \iu}{1 + \iu} = -\iu,
\qquad
-1 \longmapsto \frac{-1 - \iu}{-1 + \iu} = \iu ,
\]
(हर के संयुग्म से गुणा कीजिए), अतः $U = (H -
\iu I)(H + \iu I)^{-1}$ वही \omterm{def:b2:hermitian:adjoint}{एकात्मक} है जिसके उन्हीं
अभिलक्षणिक सदिशों पर \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\mp\iu$ हैं:
\[
U = \frac{1}{2}\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
\begin{pmatrix} -\iu & 0\\ 0 & \iu\end{pmatrix}
\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
= \begin{pmatrix} 0 & -\iu\\ -\iu & 0\end{pmatrix} .
\]
जाँच: $U^\dagger U = I$, और $1 \notin \operatorname{Sp}U =
\{\pm\iu\}$, जैसा सिद्धांत वचन देता है। समापन दृष्टि: वास्तविक
रेखा इकाई वृत्त में से बिंदु $1$ हटाकर बने समुच्चय पर जाती है ---
\omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} दर \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान}, केली रूपांतरण म्यूबियस प्रतिचित्रण $\frac{\lambda-\iu}{\lambda+\iu}$
है, और आव्यूह बस अपने वर्णक्रमों के पीछे चलते हैं।
\end{example}

\begin{remark}[सामान्य भूलें]
\emph{(क) रेखा कहाँ पड़ती है:} यह पुस्तक \emph{पहले} खाँचे का संयुग्म
लेती है, अतः निर्देशांक $\langle e_i, x\rangle$ और $\langle\lambda x, y\rangle = \conj\lambda\langle x,
y\rangle$ होते हैं; अनेक ग्रंथ इसके बदले दूसरे
खाँचे का संयुग्म लेते हैं --- सूत्रों की तुलना करने से पहले अनुवाद कर
लीजिए, अन्यथा $\iu$ के चिह्न चुपचाप ग़लत हो जाएँगे। \emph{(ख) सम्मिश्र
\omterm{def:b2:quadratic:def}{ध्रुवण} अधिक शक्तिशाली है:} $\C$ पर यदि \emph{सभी} $x$ के लिए $\langle x, u(x)\rangle = 0$ हो, तो
$u = 0$ ($x + y$ और $x + \iu y$ का प्रसार कीजिए: $\langle x, u(y)\rangle$ के वास्तविक और काल्पनिक दोनों भाग
लुप्त हो जाते हैं); जबकि $\R$ पर यह विफल है --- $\frac\pi2$ से घूर्णन सर्वत्र
$\langle x, u(x)\rangle = 0$ पूरा करता है। फलतः केवल $\C$ पर ही ``सभी $x$ के लिए $\langle x, u(x)\rangle \in \R$'' पहले ही
$u$ के \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} होने को बाध्य कर देता है। \emph{(ग) वास्तविक प्रसामान्य
\omterm{def:b2:reduction:diag}{विकर्णनीय} नहीं होता:} \cref{ex:b2:hermitian:skewexample} का आव्यूह प्रसामान्य है पर उसका कोई वास्तविक
\omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} नहीं; \omterm{def:b2:hermitian:adjoint}{एकात्मक} विकर्णन $\C$ \emph{पर} प्रमेय है, और $\R$ पर
केवल खंड-समानयन मिलता है। \emph{(घ) एकात्मकता की जाँच:} $U^\dagger U = I$ का अर्थ है कि
\emph{स्तंभ} \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} गुणन के लिए प्रसामान्य लांबिक हैं --- $UU^{\mathsf T}$ जाँचना,
अथवा स्तंभ-गुणनों में संयुग्म भूल जाना, ग़लत आव्यूह को प्रमाणित करने के
दो क्लासिक तरीक़े हैं।
\end{remark}

\begin{example}[समदूरिक प्रतिचित्रण ठीक एकात्मक ही हैं]\label{ex:b2:hermitian:isometry}
\omterm{def:b2:nvs:norm}{मानदंड} बनाए रखना एकात्मकता से दुर्बल दिखता है, पर $\C$ पर वह दुर्बल है
नहीं: यदि सभी $x$ के लिए $\norm{u(x)} = \norm x$ हो, तो $u^*u =
\mathrm{id}$। वास्तव में $v = u^*u - \mathrm{id}$ \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} है और
प्रत्येक $x$ के लिए $\langle x, v(x)\rangle = \norm{u(x)}^2 - \norm x^2 =
0$ पूरा करता है; और सम्मिश्र \omterm{def:b2:quadratic:def}{ध्रुवण} (ऊपर की भूल (ख))
से जिस प्रतिचित्रण का ``विकर्ण'' सर्वथा लुप्त हो वह शून्य होता है: $v = 0$।
मूर्त रूप में \omterm{def:b2:quadratic:def}{ध्रुवण} इस प्रकार चलता है
\[
0 = \langle x + y, v(x+y)\rangle
= \langle x, v(y)\rangle + \langle y, v(x)\rangle,
\qquad
0 = \langle x + \iu y, v(x + \iu y)\rangle
= \iu\langle x, v(y)\rangle - \iu\langle y, v(x)\rangle ,
\]
और दोनों पंक्तियाँ मिलकर सभी $x, y$ के लिए $\langle x, v(y)\rangle = 0$ को बाध्य कर देती हैं। समापन
दृष्टि: यही कारण है कि ``\omterm{def:b2:hermitian:adjoint}{एकात्मक}'' होना केवल लंबाइयाँ मापकर जाँचा जा सकता
है --- और इसी कठोरता का उपयोग फूरिये अध्याय करेगा, जहाँ ऊर्जा $\norm f_2$ बनाए
रखना (पारसेवाल) गुणांकों के सारे अंतर्गुणन बनाए रखने के बराबर है।
\end{example}

\begin{remark}[इसी खंड के भीतर के परिप्रेक्ष्य]
यहाँ बनाया गया \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} यंत्र लगभग तुरंत काम में आ जाता है। फूरिये
अध्याय अनंत विमा में \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} ज्यामिति \emph{ही} है: चरघातांकी $(e_n)$ $\langle f, g\rangle =
\frac{1}{2\pi}\int\conj fg$
के लिए प्रसामान्य लांबिक कुल हैं, बेसेल असमिका इस अध्याय की \cref{thm:b2:hermitian:cs} का
प्रक्षेप-आकलन है, और पारसेवाल उसकी सीमांत समता। परिमित फूरिये रूपांतरण
(\cref{exo:b2:hermitian:10}) जब भी संवलन का विकर्णन करना हो तब फिर प्रकट होता है। और इस अध्याय
की सप्ताहांत समस्या --- कूराँ--फिशर, वाइल, अंतर्गुंथन --- वह अभिलक्षणिक-मान
स्थायित्व देती है जिसका सहारा अवकल समीकरणों वाला अध्याय तब लेता है जब वह
कहता है कि किसी निकाय के छोटे विक्षोभ उसकी आवृत्तियों को थोड़ा ही हिलाते
हैं। और पीछे की ओर, यहाँ का सब कुछ द्विघात-रूप वाले अध्याय का सम्मिश्र
दर्पण है: दोनों शब्दकोश साथ-साथ रखिए ($A^{\mathsf T} \leftrightarrow
A^\dagger$, लांबिक $\leftrightarrow$ \omterm{def:b2:hermitian:adjoint}{एकात्मक}, और रैले
दोनों में वास्तविक)।
\end{remark}

\begin{remark}[प्रसामान्य अंतःरूपांतरण]
$\C$ पर निर्णायक कथन यह है: $u$ \emph{\omterm{def:b2:hermitian:adjoint}{एकात्मक} रूप से \omterm{def:b2:reduction:diag}{विकर्णनीय} है यदि और
केवल यदि वह प्रसामान्य हो} ($u^*u = uu^*$) --- और यह एक साथ \omterm{def:b2:hermitian:adjoint}{हर्मीशियन}, \omterm{def:b2:hermitian:adjoint}{एकात्मक} तथा
विषम-हर्मीशियन प्रतिचित्रणों को समेट लेता है। उपपत्ति ऊपर वाले तर्क का
एक सुखद सुदृढ़ीकरण है (\cref{exo:b2:hermitian:8})। इसके विपरीत $\R$ पर प्रसामान्यता केवल
खंड-विकर्णन ख़रीदती है (घूर्णन खंड): सम्मिश्र ज्यामिति सचमुच सरल है।
\end{remark}

\begin{remark}[यह कहाँ काम आता है]
\omterm{def:b2:hermitian:adjoint}{हर्मीशियन} \omterm{def:b2:reduction:eigen}{वर्णक्रम} सिद्धांत क्वांटम यांत्रिकी का गणित है: प्रेक्षणीय
राशियाँ \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} संकारकों से प्रतिरूपित होती हैं (वास्तविक \omterm{def:b2:reduction:eigen}{वर्णक्रम} =
मापे जा सकने वाले मान), और समय-विकास \omterm{def:b2:hermitian:adjoint}{एकात्मक} संकारकों से (\omterm{def:b2:nvs:norm}{मानदंड} का बना
रहना = प्रायिकता का संरक्षण)। इस पुस्तक के भीतर फूरिये अध्याय चरघातांकियों
की प्रसामान्य लांबिकता पर टिका है --- जो \omterm{def:b2:hermitian:def}{हर्मीशियन अंतर्गुणन} का कथन है
--- और \omterm{def:b2:structures:generated}{चक्रीय} आव्यूहों का विकर्णन (\cref{exo:b2:hermitian:10}) परिमित फूरिये रूपांतरण ही है।
सप्ताहांत समस्या अभिलक्षणिक मानों का चरमीय कलन विकसित करती है
(कूराँ--फिशर, वाइल, अंतर्गुंथन), जो संख्यात्मक विश्लेषण और गणितीय भौतिकी
की रोज़ की रोटी है; और तृतीय वर्ष का खंड उसे हिल्बर्ट समष्टियों पर \omterm{def:b2:metric:compact}{संहत}
स्वसंलग्न संकारकों तक बढ़ा देता है।
\end{remark}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:5}
सिद्ध कीजिए कि $U(n)$ \omterm{def:b2:metric:compact}{संहत} है, और यह कि अभिलक्षणिक-मान प्रतिचित्रण आच्छादक
है: प्रत्येक इकाई-मापांक $\lambda$ किसी न किसी \omterm{def:b2:hermitian:adjoint}{एकात्मक} आव्यूह से मिलता है। सिद्ध
कीजिए कि $U \in
U(n)$ के लिए $\det U \in \mathbb{U}$ (इकाई वृत्त)।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:6}
(केली रूपांतरण) मान लीजिए $H$ \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} है। सिद्ध कीजिए कि $H + \iu I$
प्रतिलोमनीय है और $U = (H - \iu I)(H + \iu I)^{-1}$ $1 \notin \operatorname{Sp}(U)$ सहित \omterm{def:b2:hermitian:adjoint}{एकात्मक} है। \emph{(वर्णक्रमीय ढंग से काम
कीजिए: $H$ के किसी अभिलक्षणिक आधार पर सब कुछ अदिश है।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:7}
\omterm{def:b2:hermitian:adjoint}{हर्मीशियन} धनात्मक निश्चित $A$ के लिए ($x \neq 0$ के लिए $\langle x, Ax\rangle > 0$) सिद्ध कीजिए कि $\operatorname{Sp}(A) \subseteq
\intoo{0}{\infty}$,
कि किसी \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} धनात्मक निश्चित $B$ के लिए $A = B^2$, और कि $\det A > 0$।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:hermitian:8}
(प्रसामान्य अंतःरूपांतरणों के लिए वर्णक्रमीय प्रमेय) मान लीजिए $u$ किसी
\omterm{def:b2:hermitian:def}{हर्मीशियन समष्टि} पर प्रसामान्य है।
\begin{enumerate}
  \item सभी $x$ के लिए $\norm{u(x)} = \norm{u^*(x)}$ सिद्ध कीजिए, और $\ker(u - \lambda) = \ker(u^* - \conj\lambda)$ निष्कर्ष रूप में निकालिए।
  \item सिद्ध कीजिए कि भिन्न अभिलक्षणिक मानों के लिए $u$ की अभिलक्षणिक
        समष्टियाँ लांबिक हैं, और किसी \omterm{def:b2:reduction:eigen}{अभिलक्षणिक समष्टि} का लांबिक पूरक
        $u$-स्थायी है।
  \item आगमन से निष्कर्ष निकालिए कि $u$ \omterm{def:b2:hermitian:adjoint}{एकात्मक} रूप से \omterm{def:b2:reduction:diag}{विकर्णनीय} है; और
        विलोमतः भी।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:hermitian:9}
कोई अंतःरूपांतरण \emph{विषम-हर्मीशियन} कहलाता है जब $u^* = -u$। सिद्ध कीजिए कि
उसके \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} विशुद्ध काल्पनिक होते हैं, कि $u \mapsto \iu
u$ \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} से
विषम-हर्मीशियन अंतःरूपांतरणों पर एकैकी आच्छादक प्रतिचित्रण है, और यह कि
विषम-हर्मीशियन अंतःरूपांतरण \omterm{def:b2:hermitian:adjoint}{एकात्मक} रूप से \omterm{def:b2:reduction:diag}{विकर्णनीय} होते हैं (\cref{exo:b2:hermitian:8})।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:10}
(परिमित फूरिये रूपांतरण) मान लीजिए $S$ $\C^n$ का \omterm{def:b2:structures:generated}{चक्रीय} स्थानांतरण है: $S(x_0, x_1, \dots, x_{n-1}) = (x_{n-1}, x_0, \dots,
x_{n-2})$,
और $\omega = \eu^{2\iu\pi/n}$।
\begin{enumerate}
  \item दिखाइए कि $S$ \omterm{def:b2:hermitian:adjoint}{एकात्मक} है, और सदिश $f_k =
        \frac{1}{\sqrt n}\bigl(1, \omega^k, \omega^{2k}, \dots,
        \omega^{(n-1)k}\bigr)$, $0 \leq k < n$, अभिलक्षणिक सदिशों का
        प्रसामान्य लांबिक आधार बनाते हैं: $Sf_k = \omega^{-k}
        f_k$।
  \item इससे निष्कर्ष निकालिए कि प्रत्येक \emph{\omterm{def:b2:structures:generated}{चक्रीय}} आव्यूह $C =
        \sum_{j=0}^{n-1} c_jS^j$
        प्रसामान्य है और उसी आधार से विकर्णित होता है, जिसके \omterm{def:b2:reduction:eigen}{अभिलक्षणिक
        मान} $\widehat c(k) = \sum_j
        c_j\,\omega^{-jk}$ हैं।
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:hermitian:11}
मान लीजिए $P$ किसी \omterm{def:b2:hermitian:def}{हर्मीशियन समष्टि} का वर्गसम ($P^2 = P$) अंतःरूपांतरण है।
सिद्ध कीजिए कि $P$ $\operatorname{im} P$ पर \emph{लांबिक} प्रक्षेप है यदि और केवल यदि $P^* = P$।
$\C v$ ($\norm v = 1$) पर लांबिक प्रक्षेप का आव्यूह दीजिए, और प्रसामान्य लांबिक आधार
$(v_1, \dots, v_k)$ वाली किसी उपसमष्टि पर भी।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:hermitian:12}
(अंतर्वेशन से वर्णक्रमीय प्रक्षेपक) मान लीजिए $A$ \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} है और उसके
\emph{भिन्न} \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} $\lambda_1, \dots, \lambda_p$ हैं तथा अभिलक्षणिक-समष्टि अपघटन $E = \bigoplus_i E_i$।
लाग्रांज बहुपद $L_i(X) = \prod_{j\neq i}\frac{X -
\lambda_j}{\lambda_i - \lambda_j}$ परिभाषित कीजिए। सिद्ध कीजिए कि $P_i = L_i(A)$ $E_i$ पर लांबिक
प्रक्षेप है, कि $i \neq j$ के लिए $P_iP_j = 0$, $\sum_i P_i = I$, और $A = \sum_i \lambda_iP_i$ (अर्थात् \emph{वर्णक्रमीय
अपघटन}); और किसी भी बहुपद $f$ के लिए $f(A)$ को $P_i$ के पदों में व्यक्त
कीजिए।
\end{exercise}

\section{समस्या: कूराँ--फिशर, वाइल, और अभिलक्षणिक मानों का कलन}

\begin{problem}\label{pb:b2:hermitian:1}
किसी \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} आव्यूह के \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} केवल किसी बहुपद के मूल नहीं हैं:
वे \emph{अनुकूलन समस्याओं} के हल हैं। यही चरमीय दृष्टिकोण --- \omterm{pb:b2:hermitian:1}{रैले भागफल}
और \emph{कूराँ--फिशर न्यूनतम-महत्तम प्रमेय}\index{कूराँ--फिशर प्रमेय} ---
अभिलक्षणिक मानों को तुलनीय, स्थायी और परिकलनीय बना देता है, और यह समस्या
उसकी क्लासिक फ़सलें काटती है: \emph{वाइल की विक्षोभ असमिकाएँ}\index{वाइल
की असमिका}, \emph{कोशी अंतर्गुंथन}, शूर तथा काय फान की अनुरेख असमिकाएँ,
आव्यूह वर्गमूल की एकदिष्टता, और विविक्त लाप्लासीय का \omterm{def:b2:reduction:eigen}{वर्णक्रम}। आगे सर्वत्र
$A, B,
E$ $E = \C^n$ पर \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} हैं जिनके \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} ह्रासमान क्रम $\lambda_1(A) \geq \dots \geq \lambda_n(A)$ में
सूचीबद्ध हैं, और $x \neq 0$ के लिए $R_A(x) = \frac{\langle x, Ax\rangle}{\langle x, x\rangle}$ \emph{रैले भागफल}\index{रैले भागफल} है।

\medskip
\noindent\textbf{भाग I --- \omterm{pb:b2:hermitian:1}{रैले भागफल} और न्यूनतम-महत्तम।} कोई प्रसामान्य
लांबिक अभिलक्षणिक आधार $(e_1, \dots, e_n)$, $Ae_i =
\lambda_ie_i$ स्थिर कीजिए।
\begin{enumerate}
  \item दिखाइए कि $R_A(x)$ वास्तविक है, और
        \[
        \lambda_n \leq R_A(x) \leq \lambda_1
        \qquad (x \neq 0),
        \]
        जहाँ दोनों परिबंध प्राप्त होते हैं: $\lambda_1 = \max R_A$, $\lambda_n = \min R_A$।
  \item दिखाइए कि $R_A$ के क्रांतिक बिंदु ठीक $A$ के \omterm{def:b2:reduction:eigen}{अभिलक्षणिक सदिश} हैं
        \emph{(स्वेच्छ $v$ के लिए $t = 0$ पर $t \mapsto R_A(x +
        tv)$ का प्रसार कीजिए, फिर $v$ के
        स्थान पर $\iu v$ रखिए)}।
  \item मान लीजिए $V_k = \operatorname{Vect}(e_1, \dots, e_k)$ और $W_k = \operatorname{Vect}(e_k, \dots, e_n)$। दिखाइए
        \[
        \min_{x \in V_k\setminus\{0\}} R_A(x) = \lambda_k
        = \max_{x \in W_k\setminus\{0\}} R_A(x) .
        \]
  \item \emph{कूराँ--फिशर प्रमेय} सिद्ध कीजिए: $1 \leq
        k \leq n$ के लिए
        \[
        \lambda_k = \max_{\dim V = k}\;\min_{x \in
        V\setminus\{0\}} R_A(x)
        = \min_{\dim W = n-k+1}\;\max_{x \in
        W\setminus\{0\}} R_A(x)
        \]
        \emph{(विमा $k$ वाले किसी भी $V$ के लिए: ग्रासमान से $V \cap W_k \neq
        \{0\}$, अतः
        $\min_V R_A \leq \lambda_k$; और प्रश्न 3 दिखाता है कि परिबंध प्राप्त हो जाता है)}।
  \item (एकदिष्टता) $B - A$ के धनात्मक अर्धनिश्चित होने पर $A \leq B$ लिखिए।
        प्रश्न 4 से निष्कर्ष निकालिए: $A \leq
        B$ से प्रत्येक $k$ के लिए $\lambda_k(A) \leq \lambda_k(B)$।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- वाइल की असमिकाएँ।}
\begin{enumerate}[resume]
  \item दिखाइए कि $\dim V
        + \dim W > n$ वाली उपसमष्टियाँ $V, W \subseteq \C^n$ अतुच्छ रूप से प्रतिच्छेद करती
        हैं, और सामान्यीकरण कीजिए: $\dim(V_1 \cap V_2 \cap V_3) \geq \dim V_1 + \dim V_2 +
        \dim V_3 - 2n$।
  \item \emph{वाइल की असमिका} सिद्ध कीजिए: $i + j - 1 \leq n$ के लिए
        \[
        \lambda_{i+j-1}(A + B) \leq \lambda_i(A) +
        \lambda_j(B)
        \]
        \emph{(प्रश्न 3 की उपसमष्टियों $W_i(A)$, $W_j(B)$ और $V_{i+j-1}(A+B)$ का प्रतिच्छेदन
        लीजिए और विमाएँ गिनिए)}।
  \item $\vertiii{E}_2 = \max_{\norm x = 1}\norm{Ex}$ परिभाषित कीजिए और \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} $E$ के लिए $\vertiii E_2 = \max_k\abs{\lambda_k(E)}$ दिखाइए। इससे
        \emph{वाइल की विक्षोभ प्रमेय} निष्कर्ष रूप में निकालिए:
        \[
        \bigl|\lambda_k(A + E) - \lambda_k(A)\bigr| \leq
        \vertiii{E}_2
        \qquad (1 \leq k \leq n) :
        \]
        अर्थात् हर \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} आव्यूह का $1$-लिप्शिट्स फलन है।
  \item (कोटि-एक विक्षोभ) मान लीजिए $P$ कोटि $1$ का \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} धनात्मक
        अर्धनिश्चित है। $\lambda_1(A) \leq \lambda_1(A+P)$ के साथ
        \[
        \lambda_k(A) \leq \lambda_k(A + P) \leq
        \lambda_{k-1}(A) \qquad (2 \leq k \leq n),
        \]
        दिखाइए: अर्थात् नए \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} पुरानों के बीच
        \emph{अंतर्गुंथित} हो जाते हैं।
  \item प्रश्न 8 को संख्यात्मक रूप से जाँचिए: $A = \begin{pmatrix} 2 &
        \iu\\ -\iu & 2\end{pmatrix}$ (\cref{exo:b2:hermitian:4}: \omterm{def:b2:reduction:eigen}{वर्णक्रम} $\{3, 1\}$)
        और
        $E = \begin{pmatrix} 0 & 1\\ 1 & 0\end{pmatrix}$
        (\omterm{def:b2:reduction:eigen}{वर्णक्रम} $\{1, -1\}$): $A +
        E$ का \omterm{def:b2:reduction:eigen}{वर्णक्रम} तथा असमिका के दोनों पक्ष
        परिकलित कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- अंतर्गुंथन और अनुरेख असमिकाएँ।}
\begin{enumerate}[resume]
  \item (कोशी अंतर्गुंथन) मान लीजिए $B$ $A$ का अग्र $(n-1)
        \times(n-1)$ मुख्य उपआव्यूह
        है। सिद्ध कीजिए
        \[
        \lambda_{k+1}(A) \leq \lambda_k(B) \leq \lambda_k(A)
        \qquad (1 \leq k \leq n-1)
        \]
        \emph{($\C^{n-1} \subseteq \C^n$ मानिए; उस पर $R_B$ $R_A$ का प्रतिबंधन है; और दोनों स्तरों
        पर कूराँ--फिशर लगाइए)}।
  \item इसे दोहराइए: आकार $n - m$ के किसी मुख्य उपआव्यूह $B$ के लिए $\lambda_{k+m}(A) \leq \lambda_k(B) \leq
        \lambda_k(A)$।
  \item (शूर) मान लीजिए $d_1 \geq d_2 \geq \dots \geq d_n$ $A$ की क्रमबद्ध विकर्ण प्रविष्टियाँ हैं।
        प्रत्येक $k$ के लिए सिद्ध कीजिए:
        \[
        \sum_{i=1}^{k} d_i \leq \sum_{i=1}^{k}\lambda_i(A),
        \]
        और $k = n$ पर समता (अनुरेख) \emph{(चुनी गई $k$ विकर्ण प्रविष्टियाँ
        कोई मुख्य $k\times k$ उपआव्यूह बनाती हैं; उसके अनुरेख को प्रश्न 12 से
        परिबद्ध कीजिए)}।
  \item (काय फान) सिद्ध कीजिए
        \[
        \sum_{i=1}^{k}\lambda_i(A)
        = \max\Bigl\{\sum_{i=1}^{k}\langle x_i, Ax_i\rangle
        : (x_1, \dots, x_k) \text{ प्रसामान्य लांबिक}\Bigr\} .
        \]
  \item प्रश्न 11 और 13 को
        \[
        A = \begin{pmatrix} 2 & 1 & 0\\ 1 & 2 & 1\\
        0 & 1 & 2\end{pmatrix}
        \qquad
        \bigl(\operatorname{Sp} = \{2 - \sqrt2,\ 2,\
        2+\sqrt2\}\bigr)
        \]
        पर उसके अग्र $2\times2$ खंड (\omterm{def:b2:reduction:eigen}{वर्णक्रम} $\{1,
        3\}$) तथा उसके विकर्ण के विरुद्ध
        सत्यापित कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- लोएव्नर क्रम।} $A \leq B$ का अर्थ अब भी $B - A$ का
धनात्मक अर्धनिश्चित होना है; और इस भाग के सारे आव्यूह \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} हैं।
\begin{enumerate}[resume]
  \item दिखाइए: $A \leq B$ से $i$, $\operatorname{tr} A \leq \operatorname{tr} B$ के लिए $a_{ii} \leq b_{ii}$, और \emph{प्रत्येक} सम्मिश्र
        आव्यूह $C$ के लिए $C^\dagger AC \leq C^\dagger BC$।
  \item दिखाइए कि वर्ग करना एकदिष्ट \emph{नहीं} है:
        \[
        A = \begin{pmatrix} 1 & 0\\ 0 & 0\end{pmatrix},
        \qquad
        B = \begin{pmatrix} 2 & 1\\ 1 & 1\end{pmatrix},
        \]
        के लिए $0 \leq A \leq B$ जाँचिए पर $A^2 \not\leq B^2$।
  \item सिद्ध कीजिए कि वर्गमूल एकदिष्ट \emph{है}: $0 \leq
        A \leq B$ से $\sqrt A \leq \sqrt B$ \emph{($\mu$
        को $\sqrt B - \sqrt A$ का कोई \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} लीजिए जिसका इकाई \omterm{def:b2:reduction:eigen}{अभिलक्षणिक सदिश}
        $v$ हो; $\langle v, (B - A)v\rangle =
        \mu\bigl(\langle v, \sqrt B\,v\rangle + \langle v,
        \sqrt A\,v\rangle\bigr)$ परिकलित कीजिए और चर्चा कीजिए)}।
  \item सिद्ध कीजिए कि धनात्मक निश्चित आव्यूहों पर प्रतिलोमन प्रतिएकदिष्ट
        है: $0 < A \leq B$ से $B^{-1} \leq A^{-1}$ \emph{($A^{-1/2}$ से \omterm{def:b2:quadratic:def}{सर्वांगसमता} करके $I \leq M
        \Rightarrow M^{-1} \leq I$ तक ले
        आइए, जो किसी वर्णक्रमीय आधार में अदिश है)}।
  \item मान लीजिए $A, B$ धनात्मक निश्चित हैं। दिखाइए कि $AB$ (जो सामान्यतः
        \omterm{def:b2:hermitian:adjoint}{हर्मीशियन} नहीं!) के \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान} वास्तविक और धनात्मक हैं, और
        \[
        \lambda_{\max}(AB) \leq
        \lambda_{\max}(A)\,\lambda_{\max}(B)
        \]
        \emph{($\sqrt A$ से संयुग्मन कीजिए: $AB \sim \sqrt
        A\,B\sqrt A$)}।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- विविक्त लाप्लासीय, करके दिखाया गया।} मान
लीजिए $T_n$ $n \times n$ त्रिविकर्णी आव्यूह है जिसके विकर्ण पर $2$ और दोनों
सटे विकर्णों पर $-1$ है।
\begin{enumerate}[resume]
  \item $\theta_k = \frac{k\pi}{n+1}$ के साथ सत्यापित कीजिए कि सदिश $v_k = \bigl(\sin(j\theta_k)\bigr)_{1\leq j\leq
        n}$ $T_nv_k = (2 - 2\cos\theta_k)\,v_k$ पूरा करते हैं
        \emph{(गुणनफल-से-योग सर्वसमिका; सीमा-पंक्तियाँ $j = 1, n$ जाँच लीजिए)}।
        निष्कर्ष निकालिए:
        \[
        \operatorname{Sp}(T_n) = \Bigl\{4\sin^2
        \frac{k\pi}{2(n+1)} : 1 \leq k \leq n\Bigr\},
        \]
        सब सरल, सब धनात्मक: अतः $T_n$ धनात्मक निश्चित है।
  \item (एक विभव) किसी वास्तविक विकर्ण $D =
        \operatorname{diag}(d_1, \dots, d_n)$ के लिए \omterm{def:b2:reduction:eigen}{वर्णक्रम} को दोनों ओर से
        घेरिए: प्रत्येक $k$ के लिए
        \[
        \lambda_k(T_n) + \min_i d_i \;\leq\;
        \lambda_k(T_n + D) \;\leq\; \lambda_k(T_n) + \max_i
        d_i .
        \]
  \item $T_3$ और $T_2$ के बीच कोशी अंतर्गुंथन स्पष्ट रूप से जाँचिए (\omterm{def:b2:reduction:eigen}{वर्णक्रम}
        $\{2 \pm \sqrt2, 2\}$ और $\{1,
        3\}$), और उसका अर्थ बताइए: $T_2$ वही $T_3$ है जिसमें पथ का
        एक अंत्यबिंदु हटा दिया गया हो।
  \item दिखाइए कि $n \to
        \infty$ होने पर चरम \omterm{def:b2:reduction:eigen}{अभिलक्षणिक मान}
        \[
        \lambda_{\min}(T_n) = 4\sin^2\frac{\pi}{2(n+1)}
        \sim \frac{\pi^2}{(n+1)^2},
        \qquad
        \lambda_{\max}(T_n) \to 4 ,
        \]
        पूरा करते हैं, अतः अवस्था संख्या $\kappa_n =
        \lambda_{\max}/\lambda_{\min}$ $\frac{4(n+1)^2}{\pi^2}$ की तरह बढ़ती है:
        अर्थात् किसी दूसरे अवकलज का उत्तरोत्तर बारीक जाल पर विविक्तीकरण
        स्वभाव से ही दुरवस्थित है।
  \item संश्लेषण। एक-एक वाक्य में: (क) अभिलक्षणिक मानों को स्थायी बनाने
        वाला \omterm{def:b2:reduction:charpoly}{अभिलक्षणिक बहुपद} नहीं बल्कि चरमीय अभिलक्षण क्यों है (प्रश्न
        8--9); (ख) किन प्रश्नों में केवल $\lambda_1 = \max R_A$ लगा और किनमें पूरा
        न्यूनतम-महत्तम चाहिए था; (ग) लोएव्नर क्रम कहानी में क्या जोड़ता
        है; (घ) ये औज़ार फिर कहाँ प्रकट होते हैं (प्रश्न 24 के दृढ़ता
        आव्यूहों का संख्यात्मक विश्लेषण; क्वांटम विक्षोभ सिद्धांत; और
        तृतीय वर्ष के खंड में \omterm{def:b2:metric:compact}{संहत} स्वसंलग्न संकारकों के लिए
        न्यूनतम-महत्तम सिद्धांत)।
\end{enumerate}
\end{problem}
