\chapter{समुच्चय और संरचनाएँ}\label{ch:b2:structures}

यह आरंभिक अध्याय प्रथम वर्ष के खंड में रखी गई नींव को रोज़ काम आने वाले
औज़ारों में ढालता है: समुच्चयों और विभागों का कलन, अनंत समुच्चयों की
तुलना (गणनीयता, कैंटर--बर्नस्टाइन), तथा समूहों और वलयों का संरचनात्मक
सिद्धांत --- लाग्रांज प्रमेय, अपने चिह्न सहित सममित समूह, गुणजावलियाँ और
चीनी शेषफल प्रमेय। यहाँ की हर बात पुस्तक के शेष भाग में लगातार काम आती
है: चिह्न से सारणिक बनता है (\cref{ch:b2:linalg}), \omterm{def:b2:structures:quotientring}{विभाग वलय} अंकगणित चलाते हैं, और
गणनीयता सांस्थिति तथा प्रायिकता दोनों के नीचे बिछी है।

\section{समुच्चय, प्रतिचित्रण, विभाग}

प्रथम वर्ष के खंड में स्थापित समुच्चयों, प्रतिचित्रणों तथा तुल्यता और
क्रम संबंधों की भाषा हम स्वतंत्र रूप से प्रयोग करते हैं। दो परिष्कार
अलग से कथन के योग्य हैं।

\begin{proposition}[कुलों के प्रतिबिंब और पूर्वप्रतिबिंब]\label{prop:b2:structures:images}
मान लीजिए $f \colon E \to F$, और मान लीजिए $(A_i)_{i \in I}$, $(B_j)_{j \in J}$
क्रमशः $E$ तथा $F$ के उपसमुच्चयों के कुल हैं। तब
\[
f^{-1}\Bigl(\bigcup_j B_j\Bigr) = \bigcup_j f^{-1}(B_j),
\qquad
f^{-1}\Bigl(\bigcap_j B_j\Bigr) = \bigcap_j f^{-1}(B_j),
\qquad
f^{-1}(F \setminus B) = E \setminus f^{-1}(B),
\]
\[
f\Bigl(\bigcup_i A_i\Bigr) = \bigcup_i f(A_i),
\qquad
f\Bigl(\bigcap_i A_i\Bigr) \subseteq \bigcap_i f(A_i)
\quad (\text{एकैकी होने पर समता, } f).
\]
\end{proposition}

\begin{proof}
प्रत्येक सर्वसमिका परिभाषाओं को खोलने भर से मिल जाती है; उदाहरण के लिए, सभी $j$ के लिए $x \in
f^{-1}(\bigcap B_j) \iff f(x) \in B_j$, और यही सभी $j$ के लिए $\iff x \in
f^{-1}(B_j)$। प्रतिबिंब संबंधी
सर्वसमिकाएँ तथा प्रतिच्छेदन की स्थिति में समता का टूटना (और एकैकीपन से
उसका सुधार) प्रथम वर्ष के खंड में दो समुच्चयों के लिए सिद्ध किए जा चुके
हैं; कुलों के लिए तर्क अक्षरशः वही रहते हैं।
\end{proof}

\begin{example}[जहाँ प्रतिबिंब का अंतर्भाव यथार्थतः कठोर है]\label{ex:b2:structures:imagestrict}
लीजिए $f \colon \R \to \R$, $f(x) = x^2$, जहाँ $A_1 =
\intcc{-1}{0}$ और $A_2 = \intcc{0}{1}$। तब
\[
f(A_1 \cap A_2) = f(\{0\}) = \{0\},
\qquad
f(A_1) \cap f(A_2) = \intcc{0}{1} \cap \intcc{0}{1} =
\intcc{0}{1} :
\]
इससे \cref{prop:b2:structures:images} का अंतर्भाव उतना ही कठोर है
जितना हो सकता है --- एक ही मान के दो पूर्वप्रतिबिंब बिंदु $\pm x$ भिन्न-भिन्न $A_i$ में बैठते हैं। एकैकीपन ठीक इसी विभाजन को रोकता है,
और इसीलिए पूर्वप्रतिबिंब (जो बिंदुओं को कभी मिलाते नहीं) चारों
सर्वसमिकाएँ बिना शर्त पूरी करते हैं, जबकि प्रतिबिंब प्रतिच्छेदन वाली
सर्वसमिका खो देते हैं। पूरी पुस्तक के लिए अंगूठे का नियम: समुच्चय
संक्रियाओं में से \emph{पूर्वप्रतिबिंब} बेधड़क निकाल ले जाइए; प्रतिबिंबों
को सँभालकर बरतिए।
\end{example}

\begin{definition}[विभाग समुच्चय]\label{def:b2:structures:quotient}
मान लीजिए $\mathcal{R}$, $E$ पर एक तुल्यता संबंध है। \emph{विभाग
समुच्चय}\index{विभाग समुच्चय} $E/\mathcal{R}$ तुल्यता वर्गों का समुच्चय है; आच्छादन $\pi \colon E \to E/\mathcal{R}$,
$x \mapsto \mathrm{cl}(x)$, \emph{विहित प्रक्षेप}\index{विहित प्रक्षेप} कहलाता है।

\emph{सार्वत्रिक गुणधर्म (गुणनखंडन):} यदि $f \colon E \to F$ संबंध $\mathcal{R}$ के साथ
\emph{संगत} है (अर्थात् $x \mathbin{\mathcal{R}}
y \implies f(x) = f(y)$), तो $f = \overline f \circ \pi$ को पूरा करने वाला ठीक एक प्रतिचित्रण $\overline f
\colon E/\mathcal{R} \to F$ विद्यमान है।
\end{definition}

\begin{proof}[सार्वत्रिक गुणधर्म की उपपत्ति]
अद्वितीयता: शर्त $f = \overline f \circ \pi$ का अर्थ है
\[
\overline f\bigl(\mathrm{cl}(x)\bigr) = f(x)
\qquad (x \in E),
\]
और चूँकि $\pi$ आच्छादक है, $E/\mathcal{R}$ का हर अवयव किसी न किसी $\mathrm{cl}(x)$ के रूप में
आता है: इस प्रकार $\overline f$ के सभी मान बँध जाते हैं। अस्तित्व: उपर्युक्त
प्रदर्शन को ही $\overline f$ की \emph{परिभाषा} मान लीजिए; संगतता के कारण वह
असंदिग्ध है --- यदि $\mathrm{cl}(x) =
\mathrm{cl}(y)$, तो $x \mathbin{\mathcal{R}} y$, अतः $f(x) =
f(y)$, और दोनों संभावित मान
मिल जाते हैं --- और रचना से ही वह $f$ का गुणनखंडन देता है। श्रम-विभाजन
पर ध्यान दीजिए: $\pi$ की आच्छादकता अद्वितीयता देती है, संगतता अस्तित्व।
\end{proof}

\begin{example}
$\Z/n\Z$ सर्वांगसमता मॉड $n$ द्वारा $\Z$ का विभाग है; प्रथम वर्ष के खंड
में की गई सुपरिभाषितता की जाँचें सार्वत्रिक गुणधर्म के ही उदाहरण थीं।
विभाग ``प्रतिनिधियों पर संगत रचनाओं'' को सच्चे प्रतिचित्रणों में बदल देते
हैं --- आगे हम इसका निरंतर उपयोग करेंगे।
\end{example}

\section{गणनीयता और गणनसंख्या}

\begin{definition}[समशक्तता, गणनीयता]\label{def:b2:structures:countable}
दो समुच्चय \emph{समशक्त}\index{समशक्तता} कहलाते हैं जब उनके बीच कोई
एकैकी आच्छादक प्रतिचित्रण हो। कोई समुच्चय \emph{गणनीय}\index{गणनीय
समुच्चय} कहलाता है जब वह $\N$ के समशक्त हो (कुछ लेखक परिमित समुच्चयों को
भी सम्मिलित करते हैं; ``परिमित या गणनीय'' के लिए हम \emph{अधिकतम गणनीय}
कहते हैं)।
\end{definition}

\begin{proposition}[स्थायित्व के गुणधर्म]\label{prop:b2:structures:countablestable}
\begin{enumerate}
  \item $\N$ का प्रत्येक अनंत उपसमुच्चय \omterm{def:b2:structures:countable}{गणनीय} है; कोई समुच्चय अधिकतम
        \omterm{def:b2:structures:countable}{गणनीय} है यदि और केवल यदि वह $\N$ में एकैकी रूप से जाता है, यदि
        और केवल यदि वह रिक्त है अथवा $\N$ का आच्छादक प्रतिबिंब है।
  \item $\N \times \N$ \omterm{def:b2:structures:countable}{गणनीय} है; दो अधिकतम \omterm{def:b2:structures:countable}{गणनीय} समुच्चयों का गुणन भी अधिकतम
        \omterm{def:b2:structures:countable}{गणनीय} है।
  \item अधिकतम \omterm{def:b2:structures:countable}{गणनीय} समुच्चयों का अधिकतम \omterm{def:b2:structures:countable}{गणनीय} सम्मिलन अधिकतम \omterm{def:b2:structures:countable}{गणनीय}
        होता है।
  \item $\Z$ और $\Q$ \omterm{def:b2:structures:countable}{गणनीय} हैं।
\end{enumerate}
\end{proposition}

\begin{proof}
(1) किसी अनंत $A \subseteq \N$ को बार-बार न्यूनतम लेकर सूचीबद्ध कीजिए: $a_0 =
\min A$, $a_{k+1} = \min\,(A \setminus \{a_0, \dots, a_k\})$
($A$ अनंत होने से ये रिक्त नहीं होते); प्रतिचित्रण $k \mapsto a_k$ निरंतर
वर्धमान, एकैकी तथा $A$ पर आच्छादक है (हर $a \in A$ $A$ के केवल परिमित
अवयवों से बड़ा होता है, अतः उस तक पहुँचा जाता है)। यदि $E$ $\varphi$ के
द्वारा $\N$ में एकैकी रूप से जाता है, तो $E$ $\varphi(E) \subseteq \N$ के \omterm{def:b2:structures:countable}{समशक्त} है:
अर्थात् परिमित या \omterm{def:b2:structures:countable}{गणनीय}। यदि $s \colon \N \to
E$ आच्छादक है, तो $x \mapsto \min s^{-1}(\{x\})$ $E$ को
$\N$ में एकैकी रूप से भेजता है।

(2) प्रतिचित्रण $(p, q) \mapsto 2^p(2q + 1) - 1$ एक एकैकी आच्छादक प्रतिचित्रण $\N^2
\to \N$ है (अनन्य
गुणनखंडन से हर धनात्मक पूर्णांक का विषम--सम विभाजन $2^p m$ अद्वितीय है,
जहाँ $m$ विषम है)। गुणन के लिए: एकैकी प्रतिचित्रणों का संयोजन लीजिए।

(3) आच्छादनों $s_n \colon \N \to E_n$ सहित समुच्चय $E_n$ दिए हों (कोई $E_n$ परिमित हो तो
भी हानि नहीं: मान दोहरा दीजिए), तो प्रतिचित्रण $(n, k)
\mapsto s_n(k)$ \omterm{def:b2:structures:countable}{गणनीय} $\N^2$ से
$\bigcup E_n$ पर एक आच्छादन है।

(4) $\Z = \N \cup (-\N^*)$: \omterm{def:b2:structures:countable}{गणनीय} सम्मिलन। $\Q$ $\Z \times \N^*$ का आच्छादक प्रतिबिंब है
(भिन्न वाला प्रतिचित्रण), अतः अधिकतम \omterm{def:b2:structures:countable}{गणनीय}, और अनंत भी।
\end{proof}

\begin{example}[एक युग्मन फलन, कार्य करते हुए]\label{ex:b2:structures:pairing}
उपपत्ति में आया एकैकी आच्छादक प्रतिचित्रण $(p, q) \mapsto 2^p(2q + 1) - 1$ काम करते हुए देखने
योग्य है। उसके पहले कुछ मान:
\[
\begin{array}{c|ccccc}
 & q = 0 & q = 1 & q = 2 & q = 3 & q = 4\\
\hline
p = 0 & 0 & 2 & 4 & 6 & 8\\
p = 1 & 1 & 5 & 9 & 13 & 17\\
p = 2 & 3 & 11 & 19 & 27 & 35\\
p = 3 & 7 & 23 & 39 & 55 & 71
\end{array}
\]
पंक्ति $p$ उन पूर्णांकों $n$ को इकट्ठा करती है जिनके लिए $n + 1$ ठीक $2^p$
से विभाज्य है: हर प्राकृत संख्या ठीक एक बार प्रकट होती है। विसंकेतन
संकेतन जितना ही स्पष्ट है: $n = 43$ के लिए गुणनखंडन $n + 1
= 44 = 2^2\cdot 11 = 2^2(2\cdot5 + 1)$ कीजिए, अतः $(p, q) = (2, 5)$।
समापन दृष्टि: गणनीयता की उपपत्तियाँ प्रायः भेस बदले हुए
\emph{कलनविधियाँ} होती हैं --- यहाँ, ``दो के गुणनखंड बाहर निकालिए''।
\end{example}

\begin{example}[बीजीय संख्याएँ गणनीय हैं]\label{ex:b2:structures:algebraic}
कोई सम्मिश्र संख्या \emph{बीजीय} कहलाती है जब वह परिमेय गुणांकों वाले
किसी अशून्य बहुपद का मूल हो। बीजीय संख्याओं का समुच्चय $\overline\Q$ \omterm{def:b2:structures:countable}{गणनीय} है:
$\Q$ पर घात $\leq d$ के बहुपद $\Q^{d+1}$ में एकैकी रूप से जाते हैं, जो \omterm{def:b2:structures:countable}{गणनीय}
समुच्चयों का परिमित गुणन है (\cref{prop:b2:structures:countablestable}~(2)); $d$ पर सम्मिलन लेने से अशून्य
परिमेय बहुपदों की गणना $P_0, P_1,
P_2, \dots$ के रूप में हो जाती है; प्रत्येक $P_k$ के मूल
परिमित हैं; और
\[
\overline\Q = \bigcup_{k \in \N}\ \{\text{मूल } P_k\}
\]
परिमित समुच्चयों का \omterm{def:b2:structures:countable}{गणनीय} सम्मिलन है (\cref{prop:b2:structures:countablestable}~(3)), तथा अनंत है क्योंकि
इसमें $\Q$ समाहित है। $\R$ की अगणनीयता (नीचे \cref{thm:b2:structures:cantor}) के साथ मिलाकर यह
सिद्ध करता है --- एक भी उदाहरण दिखाए बिना --- कि अबीजीय संख्याएँ
विद्यमान हैं और अगणनीय बहुमत बनाती हैं: यही कैंटर का 1874 का गणना-तर्क
है, केवल गणनसंख्या से अस्तित्व।
\end{example}

\begin{theorem}[कैंटर; $\R$ की अगणनीयता]\label{thm:b2:structures:cantor}
\begin{enumerate}
  \item प्रत्येक समुच्चय $E$ के लिए कोई आच्छादन $E \to
        \mathcal{P}(E)$ नहीं होता।
  \item $\R$ \omterm{def:b2:structures:countable}{गणनीय} \emph{नहीं} है।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) प्रथम वर्ष के खंड में सिद्ध हो चुका है (विकर्ण समुच्चय $D = \{x : x
\notin f(x)\}$)।

(2) मान लीजिए $(x_n)_{n \in \N}$ $\R$ की गणना करता है। ऐसे अंतःस्थ खंड $I_0 \supseteq I_1 \supseteq \dots$ बनाइए
जिनके लिए $\abs{I_n} = 3^{-n}$ तथा $x_n \notin I_n$ हो: वर्तमान खंड को तीन संवृत तिहाइयों में
बाँटिए; कम से कम एक तिहाई $x_n$ से बचती है (कोई बिंदु तीनों में से
अधिकतम दो को छूता है)। अंतःस्थ खंडों की प्रमेय (सटे हुए अंत्यबिंदु) $\ell \in \bigcap_n I_n$
देती है; परंतु किसी $N$ के लिए $\ell = x_N$, और $x_N
\notin I_N$: विरोधाभास।
\end{proof}

\begin{theorem}[कैंटर--बर्नस्टाइन]\label{thm:b2:structures:cantorbernstein}
यदि $E$ $F$ में एकैकी रूप से जाता है और $F$ $E$ में, तो $E$ और $F$
\omterm{def:b2:structures:countable}{समशक्त} हैं।\index{कैंटर--बर्नस्टाइन प्रमेय}
\end{theorem}

\begin{proof}
मान लीजिए $f \colon E \to F$ और $g \colon F \to E$ एकैकी प्रतिचित्रण हैं। प्रत्येक बिंदु के लिए
(चाहे वह $E$ का हो या $F$ का) क्रमिक पूर्वप्रतिबिंबों की उसकी
\emph{पूर्वज-शृंखला} $x \mapsto g^{-1}(x) \mapsto
f^{-1}(g^{-1}(x)) \mapsto \dots$ खींचिए --- जब तक वर्तमान बिंदु संबंधित एकैकी
प्रतिचित्रण के प्रतिबिंब में है तब तक हर पग परिभाषित है, और एकैकीपन से
अद्वितीय भी। तीन परस्पर अपवर्जी परिणतियाँ होती हैं: शृंखला $E \setminus g(F)$ के किसी
बिंदु पर रुक जाती है (\emph{$E$ में उद्गम}), $F \setminus f(E)$ के किसी बिंदु पर रुक
जाती है (\emph{$F$ में उद्गम}), अथवा कभी नहीं रुकती। इससे $E = E_E \cup E_F \cup
E_\infty$ और $F = F_E \cup F_F \cup F_\infty$
का उद्गम के अनुसार विभाजन हो जाता है।

अब देखिए: $f$ $E_E$ को $F_E$ \emph{पर} भेजता है --- $f(x)$ की शृंखला $x$ की
शृंखला के आगे एक पग जोड़ने से बनती है, अतः उद्गम एक ही रहते हैं; और हर
$y \in F_E$ की शृंखला में कम से कम एक पग होता है (उसका उद्गम $E$ में है), अतः
$x \in E_E$ के साथ $y = f(x)$। यही तर्क एकैकी आच्छादक प्रतिचित्रण $f \colon E_\infty \to F_\infty$ और $g \colon F_F
\to E_F$ देता
है। जोड़ने पर,
\[
h(x) =
\begin{cases}
f(x) & \text{यदि } x \in E_E \cup E_\infty,\\
g^{-1}(x) & \text{यदि } x \in E_F,
\end{cases}
\]
$E$ से $F = F_E \cup F_\infty \cup F_F$ पर एकैकी आच्छादक प्रतिचित्रण बन जाता है: यह टुकड़ों में
एकैकी आच्छादक है, और लक्ष्य के तीनों टुकड़े असंयुक्त हैं।
\end{proof}

\begin{example}\label{ex:b2:structures:cbexample}
$\intoo{0}{1}$ और $\intcc{0}{1}$ \omterm{def:b2:structures:countable}{समशक्त} हैं: एक दिशा में तत्समक एकैकी है, दूसरी दिशा में
$x \mapsto \frac{x + 1}{3}$; और प्रमेय (अनिवार्यतः असंतत) एकैकी आच्छादक प्रतिचित्रण गढ़ देती
है। इसी प्रकार $\R$, $\intoo{0}{1}$ ($\tanh$ प्रकार के एकैकी आच्छादक प्रतिचित्रणों से)
तथा $\mathcal{P}(\N)$ (द्विआधारी प्रसार, \cref{exo:b2:structures:3}) सब \omterm{def:b2:structures:countable}{समशक्त} हैं: यही ``सांतत्यक की
गणनसंख्या'' है।
\end{example}

\begin{example}[खंड और वर्ग]\label{ex:b2:structures:squareseg}
$\intcc{0}{1}$ और $\intcc{0}{1}^2$ \omterm{def:b2:structures:countable}{समशक्त} हैं --- गणनसंख्या को विमा दिखाई ही नहीं देती। एक
दिशा का एकैकी प्रतिचित्रण तुच्छ है: $x
\mapsto (x, 0)$। दूसरी दिशा के लिए $(x, y)$ को उस
वास्तविक संख्या पर भेजिए जिसके दशमलव अंक $x$ और $y$ के अंकों को एक के
बाद एक पिरोकर बनते हैं,
\[
(0.x_1x_2x_3\dots,\ 0.y_1y_2y_3\dots)
\;\longmapsto\; 0.x_1y_1x_2y_2x_3y_3\dots,
\]
जहाँ प्रत्येक निर्देशांक के लिए वह प्रसार चुना जाता है जो अंत तक सब $9$
न हो: इस परिपाटी से प्रतिबिंब के अंक $x$ और $y$ के अंक तय कर देते
हैं, अतः प्रतिचित्रण एकैकी है (आच्छादक होना आवश्यक नहीं --- प्रतिबिंबों
के, मान लीजिए, विषम स्थानों के अंक अंततः $9$ कभी नहीं होते --- और यह
ठीक ही है)। कैंटर--बर्नस्टाइन (\cref{thm:b2:structures:cantorbernstein}) इससे सच्चा एकैकी आच्छादक
प्रतिचित्रण जोड़ देती है। संततता की आशा तो निरर्थक है: दोनों के बीच
संतत एकैकी आच्छादक प्रतिचित्रण असंभव है --- दूरिक अध्याय बताएँगे क्यों
(संबद्धता रेखा को समतल से अलग करती है, \cref{ch:b2:metric})।
\end{example}

\section{समूह}

\begin{definition}[जनित उपसमूह; कोटि]\label{def:b2:structures:generated}
मान लीजिए $G$ एक समूह है और $A \subseteq G$। $A$ द्वारा \emph{जनित} उपसमूह, जिसे
$\langle A \rangle$ लिखा जाता है, $A$ को समाहित करने वाला सबसे छोटा उपसमूह है ---
मूर्त रूप में, $A$ के अवयवों तथा उनके प्रतिलोमों के सभी परिमित गुणनफल।
कोई समूह \emph{चक्रीय}\index{चक्रीय समूह} कहलाता है जब वह एक ही अवयव से
जनित हो: $\langle a\rangle = \{a^k : k \in \Z\}$। $a \in G$ की \emph{कोटि}\index{कोटि!अवयव की} $\operatorname{ord}(a) = \abs{\langle a \rangle}$ है (जो अनंत
भी हो सकती है); परिमित होने पर वह न्यूनतम ऐसा $n \geq 1$ है जिसके लिए $a^n = e$,
और $a^k = e \iff \operatorname{ord}(a) \mid k$।
\end{definition}

\begin{proof}[कोटि के अभिलक्षण की उपपत्ति]
यदि किसी $a^m = e$ के लिए $m \geq 1$, तो $n \geq 1$ को न्यूनतम ऐसा लीजिए जिसके लिए $a^n
= e$।
तब अवयव $e, a, \dots, a^{n-1}$ परस्पर भिन्न हैं ($0 \leq i < j < n$ के साथ $a^{i} = a^{j}$ से $a^{j-i} = e$ मिलता, जो
न्यूनतमता के विरुद्ध है), और यूक्लिडीय विभाजन $k = nq + r$ से हर $a^k$ उन्हीं में
से किसी एक पर सिमट जाता है: अतः $\langle a\rangle$ में ठीक $n$ अवयव हैं, और $a^k = a^r = e \iff r = 0 \iff n \mid k$।
यदि कोई भी घात तत्समक न हो, तो सभी $a^k$ ($k \in \Z$) भिन्न हैं (वही विभाजन
वाला तर्क) और \omterm{def:b2:structures:generated}{कोटि} अनंत है।
\end{proof}

\begin{theorem}[लाग्रांज]\label{thm:b2:structures:lagrange}
मान लीजिए $G$ एक परिमित समूह है और $H$ उसका उपसमूह। तब $\abs H$ $\abs G$ को
विभाजित करता है।\index{लाग्रांज प्रमेय} विशेष रूप से प्रत्येक अवयव की
\omterm{def:b2:structures:generated}{कोटि} $\abs G$ को विभाजित करती है, और सभी $a \in G$ के लिए $a^{\abs G} = e$।
\end{theorem}

\begin{proof}
संबंध $x \sim y \iff x^{-1}y \in H$ एक तुल्यता संबंध है (स्वतुल्य: $e \in H$; सममित: प्रतिलोम लीजिए;
संक्रामक: गुणनफल लीजिए)। $x$ का वर्ग \emph{वाम सहसमुच्चय} $xH = \{xh : h \in H\}$ है, और
$h \mapsto xh$ एक एकैकी आच्छादक प्रतिचित्रण $H \to xH$ है (प्रतिलोम $y \mapsto
x^{-1}y$): अतः सभी वर्गों
में $\abs H$ अवयव हैं। वर्ग $G$ का विभाजन करते हैं (प्रथम वर्ष के खंड की
सामान्य विभाजन प्रमेय), अतः $\abs G =
\abs H \times (\text{सहसमुच्चयों की संख्या})$। किसी
अवयव के लिए: यही $H = \langle a\rangle$ पर लगाइए; तब $a^{\abs G} = (a^{\operatorname{ord}
a})^{\abs G / \operatorname{ord} a} = e$।
\end{proof}

\begin{example}[सहसमुच्चय काम करते हुए: $\mathfrak{S}_3$ के भीतर $A_3$]\label{ex:b2:structures:cosets}
लीजिए $G = \mathfrak{S}_3$ (\omterm{def:b2:structures:generated}{कोटि} $6$) और $H = A_3 =
\{\mathrm{id},\ (1\,2\,3),\ (1\,3\,2)\}$। वाम सहसमुच्चय ये हैं:
\[
H = \{\mathrm{id},\ (1\,2\,3),\ (1\,3\,2)\},
\qquad
(1\,2)H = \{(1\,2),\ (2\,3),\ (1\,3)\} :
\]
अर्थात् तीन-तीन अवयवों के दो वर्ग $G$ का विभाजन करते हैं, ठीक वैसे ही
जैसी गणना $\abs G = \abs H \times (\text{सहसमुच्चयों की संख्या})$ माँगती है --- और स्पष्टतः
यह सम और विषम क्रमचयों का विभाजन है। ध्यान दीजिए कि $(1\,3)H = (1\,2)H$, यद्यपि $(1\,3) \neq
(1\,2)$:
सहसमुच्चय \emph{वर्ग} हैं, अपने प्रतिनिधियों से नामांकित नहीं, और $x^{-1}y \in H$
ही एकमात्र वैध तुलना है। दो वर्गों वाला यह चित्र चिह्न के लिए सामान्य
चित्र है: $A_n$ और उसका अकेला साथी सहसमुच्चय $\mathfrak{S}_n$ को आधा-आधा बाँट देते
हैं, और इसी से सप्ताहांत समस्या पहुँच योग्य पहेली-स्थितियाँ गिनती है।
\end{example}

\begin{proposition}[चक्रीय समूह]\label{prop:b2:structures:cyclic}
मान लीजिए $G = \langle a \rangle$ \omterm{def:b2:structures:generated}{कोटि} $n$ का \omterm{def:b2:structures:generated}{चक्रीय समूह} है।
\begin{enumerate}
  \item $\overline k \mapsto
        a^k$ के द्वारा $G$ $(\Z/n\Z, +)$ के तुल्याकारी है।
  \item $G$ का प्रत्येक उपसमूह \omterm{def:b2:structures:generated}{चक्रीय} है; प्रत्येक भाजक $d \mid n$ के लिए
        \omterm{def:b2:structures:generated}{कोटि} $d$ का ठीक एक उपसमूह है, अर्थात् $\langle
        a^{n/d}\rangle$।
  \item $a^k$ $G$ को \omterm{def:b2:structures:generated}{जनित} करता है यदि और केवल यदि $\gcd(k, n) = 1$: अतः $G$ के
        $\varphi(n)$ जनक हैं (ऑयलर फलन\index{ऑयलर फलन})।
\end{enumerate}
\end{proposition}

\begin{proof}
(1) $\Z$ से $G$ पर प्रतिचित्रण $k \mapsto a^k$ मॉड $n$ सर्वांगसमता के साथ संगत है
(\omterm{def:b2:structures:generated}{कोटि} के अभिलक्षण से $a^{k} = a^{k'} \iff n \mid k - k'$); सार्वत्रिक गुणधर्म (\cref{def:b2:structures:quotient}) से $\Z/n\Z$ से एक
सुपरिभाषित एकैकी आच्छादक समाकारिता प्राप्त होती है।

(2) मान लीजिए $H \leq G$ अतुच्छ है और $m$ न्यूनतम ऐसा $\geq 1$ है जिसके लिए $a^m \in
H$।
यूक्लिडीय विभाजन से $H = \langle a^m\rangle$ मिलता है ($a^k \in
H$ के लिए: $k = mq + r$ से $a^r \in H$ आवश्यक हो
जाता है, अतः $r = 0$), तथा $m \mid n$ ($m$ से $n$ का विभाजन कीजिए: $a^{n \bmod m} \in H$)। तब
$\abs H = n/m$; और $m = n/d$ लेने पर प्रत्येक भाजक $d$ साकार हो जाता है।
अद्वितीयता: \omterm{def:b2:structures:generated}{कोटि} $d$ का कोई भी उपसमूह, उपर्युक्त के अनुसार, $n/m = d$
सहित $\langle a^m \rangle$ के रूप का होता है --- अतः $m = n/d$ निर्धारित है और उपसमूह भी।

(3) हमारा दावा है कि $\operatorname{ord}(a^k) = \frac{n}{\gcd(k, n)}$। लिखिए $d = \gcd(k, n)$। किसी भी $m \geq 1$ के लिए \cref{def:b2:structures:generated} की \omterm{def:b2:structures:generated}{कोटि}
का अभिलक्षण तुल्यताओं की यह शृंखला देता है
\[
(a^k)^m = e
\iff n \mid km
\iff \frac{n}{d} \,\Big|\, \frac{k}{d}\,m
\iff \frac{n}{d} \,\Big|\, m ,
\]
जिसमें अंतिम पग गाउस की प्रमेयिका से आता है, क्योंकि $\frac nd$ और $\frac kd$
सहअभाज्य हैं। ऐसा न्यूनतम $m$ $\frac nd$ है: $\operatorname{ord}(a^k) = \frac n{\gcd(k,n)}$, जो $n$ के बराबर है यदि और
केवल यदि $\gcd(k, n) = 1$। $n$ के सापेक्ष ऐसे $\varphi(n)$ वर्ग $k$ होते हैं।
\end{proof}

\section{सममित समूह}

\begin{definition}\label{def:b2:structures:sn}
$\mathfrak{S}_n$ $\intint{1}{n}$ के क्रमचयों का समूह है (\omterm{def:b2:structures:generated}{कोटि} $n!$)। एक \emph{चक्र}\index{चक्र}
$(a_1\,a_2\,\cdots\,a_k)$ $a_1 \mapsto a_2 \mapsto \dots \mapsto a_k \mapsto a_1$ को भेजता है और शेष सब कुछ अचर छोड़ देता है; $k$ उसकी
\emph{लंबाई} है, और $2$-चक्र \emph{पार्यय}\index{पार्यय} कहलाता है। दो
चक्र \emph{असंयुक्त} कहलाते हैं जब उनके वाहक (अनचर बिंदु) असंयुक्त हों।
\end{definition}

\begin{theorem}[चक्र अपघटन]\label{thm:b2:structures:cycles}
प्रत्येक क्रमचय $\sigma \neq \mathrm{id}$ परस्पर असंयुक्त चक्रों का गुणनफल है, और गुणनखंडों
के क्रम को छोड़कर यह अद्वितीय है। असंयुक्त \omterm{def:b2:structures:sn}{चक्र} क्रमविनिमेय होते हैं,
और $\operatorname{ord}(\sigma)$ लंबाइयों का लघुत्तम समापवर्त्य है।
\end{theorem}

\begin{proof}
$\sigma$ के वाहक पर ``कक्षा'' संबंध लीजिए: $x \sim y$ यदि और केवल यदि किसी $k \in \Z$ के
लिए $y = \sigma^k(x)$ --- यह एक तुल्यता संबंध है। प्रत्येक वर्ग $\{x, \sigma(x), \dots, \sigma^{k-1}(x)\}$ (परिमित होने से
पुनरावृत्तियाँ \omterm{def:b2:structures:sn}{चक्र} बनाती हैं --- एकैकीपन के कारण पहली पुनरावृत्ति $x$
पर ही लौटनी चाहिए) \omterm{def:b2:structures:sn}{चक्र} $(x\ \sigma(x)\
\cdots\ \sigma^{k-1}(x))$ वहन करता है, और $\sigma$ इन्हीं चक्रों का
गुणनफल है: प्रत्येक कक्षा पर केवल संगत \omterm{def:b2:structures:sn}{चक्र} क्रिया करता है।
अद्वितीयता: असंयुक्त चक्रों में कोई भी गुणनखंडन ठीक इन्हीं कक्षाओं को
पुनः उत्पन्न करता है ($x$ से होकर जाने वाला \omterm{def:b2:structures:sn}{चक्र} $(x\ \sigma(x)\ \cdots)$ ही होना चाहिए)।
असंयुक्त \omterm{def:b2:structures:sn}{चक्र} क्रमविनिमेय हैं क्योंकि वे असंयुक्त बिंदुओं को हिलाते
हैं; और \omterm{def:b2:structures:generated}{कोटि} वाला कथन इसलिए मिलता है कि $\sigma^m = \mathrm{id}$ तभी होता है जब हर \omterm{def:b2:structures:sn}{चक्र} की
$m$-वीं घात तत्समक हो (असंयुक्तता से), अर्थात् जब हर लंबाई $m$ को
विभाजित करे।
\end{proof}

\begin{example}[चक्र-प्रकार एक जनगणना के रूप में]\label{ex:b2:structures:cycletype}
$\mathfrak{S}_9$ के कितने क्रमचयों का चक्र-प्रकार $(4, 3, 2)$ है --- अर्थात् एक $4$-चक्र,
एक $3$-चक्र और एक \omterm{def:b2:structures:sn}{पार्यय}? वाहक तथा \omterm{def:b2:structures:generated}{चक्रीय} क्रम चुनिए:
\[
\frac{9!}{4\cdot 3\cdot 2}
= \frac{362\,880}{24} = 15\,120 :
\]
नौ प्रतीकों को एक पंक्ति में रखिए ($9!$ प्रकार से), पहले चार, अगले तीन
और अंतिम दो को चक्रों में बाँधिए, और प्रत्येक बंधन के भीतर के घूर्णनों
से भाग दीजिए (जो क्रमशः $4$, $3$ और $2$ हैं) क्योंकि वे वही क्रमचय
देते हैं। (यहाँ चक्रों की \emph{लंबाइयाँ} भिन्न हैं, अतः और भाग नहीं
देना पड़ता; समान लंबाइयाँ होतीं तो समान बंधनों के क्रमचयों से भी भाग
देना पड़ता।) ऐसे प्रत्येक क्रमचय की \omterm{def:b2:structures:generated}{कोटि} $\operatorname{lcm}(4,3,2) = 12$ और चिह्न $(-1)^3(-1)^2(-1)^1 = +1$ होता है
(\cref{thm:b2:structures:cycles} तथा नीचे दी गई चिह्न-प्रमेय से)। $9$ का एक विभाजन, एक संयुग्मता
वर्ग, एक जनगणना --- $\mathfrak{S}_n$ का संचयशास्त्र विभाजनों का अंकगणित ही है।
\end{example}

\begin{theorem}[चिह्न]\label{thm:b2:structures:signature}
ठीक एक समूह समाकारिता $\varepsilon \colon
\mathfrak{S}_n \to \{\pm 1\}$ ऐसी है ($n \geq 2$ के लिए) जो पार्ययों पर मान $-1$
लेती है: यही \emph{चिह्न}\index{चिह्न} है। इसके अतिरिक्त $\varepsilon(\sigma) = (-1)^{I(\sigma)}$, जहाँ $I(\sigma)$
\emph{व्युत्क्रमों} की संख्या है (अर्थात् ऐसे युग्म $i < j$ जिनके लिए
$\sigma(i) >
\sigma(j)$); $k$-चक्र का चिह्न $(-1)^{k-1}$ होता है, और \emph{एकांतर समूह} $A_n = \ker\varepsilon$ की
\omterm{def:b2:structures:generated}{कोटि} $\frac{n!}{2}$ है।
\end{theorem}

\begin{proof}
\emph{अस्तित्व।} $\sigma \in \mathfrak{S}_n$ के लिए रखिए
\[
\varepsilon(\sigma)
= \prod_{1 \leq i < j \leq n}
\frac{\sigma(j) - \sigma(i)}{j - i} .
\]
गुणनखंडों के निरपेक्ष मान गुणा होकर $1$ देते हैं (अक्रमित युग्म $\{\sigma(i), \sigma(j)\}$
सभी युग्मों पर घूमते हैं), अतः $\varepsilon(\sigma) = (-1)^{I(\sigma)} \in \{\pm1\}$। समाकारिता: $\sigma, \tau$ के लिए,
\[
\varepsilon(\sigma\tau)
= \prod_{i<j} \frac{\sigma(\tau(j)) - \sigma(\tau(i))}{j - i}
= \prod_{i<j} \frac{\sigma(\tau(j)) - \sigma(\tau(i))}{\tau(j) -
\tau(i)} \cdot \prod_{i<j} \frac{\tau(j) - \tau(i)}{j - i}
= \varepsilon(\sigma)\,\varepsilon(\tau),
\]
जहाँ बीच का गुणनफल युग्मों $\{\tau(i), \tau(j)\}$ के अनुसार पुनः सूचीबद्ध करने पर $\varepsilon(\sigma)$
हो जाता है (प्रत्येक अक्रमित युग्म एक ही बार आता है, और अंश तथा हर का
चिह्न साथ-साथ बदलता है)। $a < b$ सहित \omterm{def:b2:structures:sn}{पार्यय} $\tau = (a\,b)$ में व्युत्क्रमों की
संख्या विषम होती है; ठीक-ठीक गिनने पर, $\tau(i) > \tau(j)$ वाले व्युत्क्रमित युग्म
$(i, j)$, $i < j$ ये हैं:
\[
(a, j) \ \text{, जहाँ } a < j < b, \qquad
(i, b) \ \text{, जहाँ } a < i < b, \qquad
(a, b) \ \text{ स्वयं},
\]
अर्थात् कुल $(b - a - 1) + (b - a - 1) + 1 = 2(b - a) - 1$, जो विषम है। (वैकल्पिक रूप से: सीधे $(1\,2)$ की जाँच
कीजिए, जिसमें एक ही व्युत्क्रम है, और फिर संयुग्मन कीजिए --- संयुग्मियों
के चिह्न बराबर होते हैं क्योंकि $\varepsilon$ क्रमविनिमेय समूह में जाने वाली
समाकारिता है।) अतः $\varepsilon((a\,b)) = (-1)^{2(b-a)-1} = -1$।

\emph{अद्वितीयता।} \omterm{def:b2:structures:sn}{पार्यय} $\mathfrak{S}_n$ को \omterm{def:b2:structures:generated}{जनित} करते हैं (कोई भी \omterm{def:b2:structures:sn}{चक्र} $(a_1\cdots a_k) = (a_1\,a_k)(a_1\,a_{k-1})\cdots(a_1\,a_2)$ है, और
\cref{thm:b2:structures:cycles} शेष काम कर देता है); और $\{\pm1\}$ में जाने वाली समाकारिता जनकों पर अपने
मानों से पूर्णतः निर्धारित हो जाती है।

\emph{परिणाम।} ऊपर की चक्र-सर्वसमिका $k$-चक्र को $k - 1$ पार्ययों के रूप
में लिखती है: अतः चिह्न $(-1)^{k-1}$। $A_n$: समाकारिता $\varepsilon$ आच्छादक है ($n \geq 2$ के
लिए \omterm{def:b2:structures:sn}{पार्यय} विद्यमान हैं), और दोनों ``सहसमुच्चय'' $A_n$ तथा $(1\,2)A_n$ \omterm{def:b2:structures:countable}{समशक्त}
हैं और $\mathfrak{S}_n$ का विभाजन करते हैं (लाग्रांज वाला तर्क): अतः $\abs{A_n} =
\frac{n!}{2}$।
\end{proof}

\begin{example}\label{ex:b2:structures:snexample}
$\sigma = \begin{pmatrix} 1&2&3&4&5&6\\ 3&6&5&4&1&2 \end{pmatrix}
= (1\,3\,5)(2\,6)$: \omterm{def:b2:structures:generated}{कोटि} $\operatorname{lcm}(3,2) = 6$, चिह्न $(-1)^{2}\cdot(-1)^{1} = -1$। फेंटने की सम-विषमता जाँचने का सबसे तेज़
साधन चिह्न ही है --- और \cref{ch:b2:linalg} में सारणिक का इंजन भी वही है।
\end{example}

\begin{example}[एक ही चिह्न तक तीन रास्ते]\label{ex:b2:structures:threeroads}
मान लीजिए $\sigma \in \mathfrak{S}_5$ $1, 2, 3, 4, 5$ को $3, 5, 4,
1, 2$ पर भेजता है। \emph{चक्रों से:} $1 \mapsto 3 \mapsto 4 \mapsto 1$ और $2
\mapsto 5 \mapsto 2$,
अतः $\sigma = (1\,3\,4)(2\,5)$ तथा $\varepsilon(\sigma) = (-1)^{2}(-1)^{1} = -1$। \emph{व्युत्क्रमों से:} मानों की सूची $3, 5, 4, 1, 2$ में
क्रम-भंग करने वाले युग्म $(3,1)$, $(3,2)$, $(5,4)$, $(5,1)$, $(5,2)$, $(4,1)$, $(4,2)$ हैं:
कुल सात, और $(-1)^7 = -1$। \emph{पार्ययों से:} $\sigma = (1\,4)(1\,3)(2\,5)$, तीन गुणनखंड, अतः $(-1)^3 = -1$।
तीन परिकलन, एक ही सम-विषमता: \cref{thm:b2:structures:signature} की अद्वितीयता इसकी गारंटी देती है कि
कोई भी लेखा-पद्धति इन्हें कभी असहमत नहीं कर सकती --- और यही $\varepsilon$ को
एक अपरिवर्त्य के रूप में उपयोगी बनाता है (सप्ताहांत समस्या देखिए)।
\end{example}

\begin{remark}[यहाँ से चिह्न कहाँ जाता है]
चिह्न आगे की तीन फसलों का बीज है: \cref{ch:b2:linalg} में यह सारणिक और उसका
गुणनफल-नियम बनाता है; संचयात्मक पहेलियों के लिए यह सम-विषमता के
अपरिवर्त्य देता है (इस अध्याय की सप्ताहांत समस्या इसी से पंद्रह पहेली
हल करती है); और इससे परिभाषित एकांतर समूह $A_n$ तृतीय वर्ष के खंड में
केंद्रीय हो जाते हैं, जहाँ $n \geq 5$ के लिए उनकी सरलता यह समझाती है कि घात
$5$ के समीकरण करणियों द्वारा हल क्यों नहीं होते।
\end{remark}

\section{वलय, गुणजावलियाँ, विभाग}

\begin{definition}[गुणजावली]\label{def:b2:structures:ideal}
मान लीजिए $A$ एक क्रमविनिमेय वलय है। \emph{गुणजावली}\index{गुणजावली}
$I
\subseteq A$ एक ऐसा योज्य उपसमूह है कि सभी $a \in A$, $x \in I$ के लिए $a x \in I$ हो। वलय
समाकारिताओं के केंद्रक गुणजावलियाँ होते हैं; $I = A$ यदि और केवल यदि $1 \in I$,
यदि और केवल यदि $I$ में कोई इकाई है। $x$ द्वारा \emph{\omterm{def:b2:structures:generated}{जनित}} गुणजावली
$xA = \{xa\}$ है (एक \emph{मुख्य} गुणजावली)।
\end{definition}

\begin{theorem}[{$\Z$ तथा $K[X]$ की गुणजावलियाँ}]\label{thm:b2:structures:principal}
$\Z$ की प्रत्येक \omterm{def:b2:structures:ideal}{गुणजावली} किसी अद्वितीय $n \in \N$ के लिए $n\Z$ है; और $K[X]$ ($K$
एक क्षेत्र) की प्रत्येक \omterm{def:b2:structures:ideal}{गुणजावली} किसी अद्वितीय एकिक (अथवा शून्य) $P$
के लिए $P\,K[X]$ है। फलतः दोनों वलयों में महत्तम समापवर्तक विद्यमान होते हैं
और उनके साथ बेज़ू संबंध भी: $x\Z +
y\Z = \gcd(x,y)\Z$, और बहुपदों के लिए भी वैसे ही।
\end{theorem}

\begin{proof}
$\Z$ के लिए यह प्रथम वर्ष के खंड की उपसमूह-प्रमेय ही थी (\omterm{def:b2:structures:ideal}{गुणजावली} विशेष
रूप से उपसमूह होती है, और $n\Z$ एक \omterm{def:b2:structures:ideal}{गुणजावली} है)। $K[X]$ के लिए: मान लीजिए
$I \neq \{0\}$ एक \omterm{def:b2:structures:ideal}{गुणजावली} है और $P \in I$ उसमें न्यूनतम घात का अशून्य अवयव, जिसे एकिक
बना लिया गया है। $F \in I$ के लिए यूक्लिडीय विभाजन $F = PQ + R$ से $\deg R < \deg P$ सहित $R = F - PQ \in I$
मिलता है: न्यूनतमता से $R
= 0$ आवश्यक हो जाता है, अतः $I = P\,K[X]$। अद्वितीयता:
दो एकिक जनक एक-दूसरे को विभाजित करते हैं। बेज़ू के कथन \omterm{def:b2:structures:ideal}{गुणजावली} $x\Z +
y\Z$
(अथवा उसके बहुपद-रूप) की महत्तम समापवर्तक की मुख्य \omterm{def:b2:structures:ideal}{गुणजावली} के साथ
समता ही हैं --- प्रथम वर्ष में प्रयुक्त महत्तम समापवर्तक की वही
परिभाषा, जिसे अब गुणजावलियों का कथन मान लिया गया है।
\end{proof}

\begin{example}[बहुपदों का महत्तम समापवर्तक, दो प्रकार से]\label{ex:b2:structures:polygcd}
$\Q[X]$ में $\gcd(X^3 - 1,\ X^2 - 1)$ परिकलित कीजिए। \emph{यूक्लिड से:}
\[
X^3 - 1 = X\,(X^2 - 1) + (X - 1),
\qquad
X^2 - 1 = (X + 1)(X - 1) + 0 ,
\]
अतः महत्तम समापवर्तक $X - 1$ है, और पीछे प्रतिस्थापन करने से बेज़ू संबंध
मिलता है
\[
X - 1 = 1\cdot(X^3 - 1) - X\cdot(X^2 - 1).
\]
\emph{गुणजावलियों से:} \omterm{def:b2:structures:ideal}{गुणजावली} $(X^3 - 1)\Q[X] + (X^2 - 1)\Q[X]$ मुख्य है (\cref{thm:b2:structures:principal}); उसमें $X -
1$
समाहित है (उपर्युक्त प्रदर्शन) और वह स्वयं $(X - 1)\Q[X]$ में समाहित है (दोनों
जनक $1$ पर शून्य होते हैं, अतः $X - 1$ के गुणज हैं): इसलिए एकिक जनक
$X - 1$ है। समापन दृष्टि: \omterm{def:b2:structures:ideal}{गुणजावली} वाली दृष्टि महत्तम समापवर्तक को
\emph{बिना भाग दिए} पहचान लेती है --- उभयनिष्ठ मूल \omterm{def:b2:structures:ideal}{गुणजावली} का पता दे
देते हैं, और यूक्लिड केवल उसकी पुष्टि करता है।
\end{example}

\begin{definition}[विभाग वलय $\Z/n\Z$, पुनरावलोकन]\label{def:b2:structures:quotientring}
$A$ की किसी \omterm{def:b2:structures:ideal}{गुणजावली} $I$ के लिए संबंध $x \sim y \iff x - y \in I$ एक तुल्यता है जो $+$ और $\times$
के साथ संगत है; \omterm{def:b2:structures:quotient}{विभाग समुच्चय} $A/I$ को वलय संरचना विरासत में मिलती है
--- यही \emph{विभाग वलय}\index{विभाग वलय} है --- जिससे $\pi \colon A \to A/I$ केंद्रक $I$
वाली समाकारिता बन जाती है। $A = \Z$, $I = n\Z$ के लिए यह प्रथम वर्ष के खंड का
$\Z/n\Z$ है, अब अपने सार्वत्रिक गुणधर्म सहित: $I$ को मारने वाली कोई भी
समाकारिता $A/I$ से होकर गुणनखंडित होती है।
\end{definition}

\begin{theorem}[चीनी शेषफल प्रमेय, वलय-रूप]\label{thm:b2:structures:crt}
यदि $\gcd(m, n) = 1$, तो प्रतिचित्रण
\[
\Z/mn\Z \longrightarrow \Z/m\Z \times \Z/n\Z,
\qquad
\overline{x} \longmapsto (x \bmod m,\; x \bmod n)
\]
एक वलय तुल्याकारिता है।\index{चीनी शेषफल प्रमेय} फलतः सहअभाज्य $m, n$ के
लिए $\varphi(mn) = \varphi(m)\varphi(n)$, और
\[
\varphi(n) = n \prod_{p \mid n} \Bigl(1 - \frac 1p\Bigr)
\quad (p \text{ अभाज्य}).
\]
\end{theorem}

\begin{proof}
यह प्रतिचित्रण सुपरिभाषित वलय समाकारिता है (संगतताएँ तत्काल दिखती
हैं)। एकैकीपन: $m$ मॉड तथा $n$ मॉड $x \equiv 0$ होने पर, $\gcd(m,n) = 1$ के साथ, $mn \mid x$
आवश्यक हो जाता है (गाउस)। आच्छादकता: दोनों पक्षों में $mn$ अवयव हैं,
अतः एकैकीपन ही पर्याप्त है (परिमित और समान गणनसंख्याएँ) --- अथवा
स्पष्ट रूप से: बेज़ू संबंध $um +
vn = 1$ से \[
x = b\,um + a\,vn
\] का वर्ग $(a \bmod m,\ b \bmod n)$ पर जाता है, क्योंकि
$vn = 1 - um \equiv 1
\pmod m$ से $x \equiv a \pmod m$ हो जाता है, और $n$ के सापेक्ष सममित रूप से --- यही
विधि \cref{ex:b2:structures:crtinverse} में संख्यात्मक रूप से प्रयुक्त हुई थी। इकाइयाँ इकाइयों के
युग्मों से संगत होती हैं (गुणन वलय की इकाइयाँ इकाइयों के युग्म ही
हैं), अतः $\varphi(mn) =
\varphi(m)\varphi(n)$। अभाज्य घात के लिए $\varphi(p^k) = p^k -
p^{k-1}$ ($p^k$ के सापेक्ष अनिकाइयाँ
$p$ के गुणज हैं); और गुणात्मकता से गुणनफल-सूत्र जुड़ जाता है।
\end{proof}

\begin{example}[चीनी तुल्याकारिता का प्रतिलोम]\label{ex:b2:structures:crtinverse}
लीजिए $m = 8$, $n = 9$। तुल्याकारिता का प्रतिलोम दो \emph{वर्गसम} अवयवों से
स्पष्ट हो जाता है: ऐसे $u \equiv 1 \pmod 8$, $u \equiv 0 \pmod 9$ खोजिए कि $v \equiv 0 \pmod 8$, $v \equiv 1 \pmod
9$ हों। $u = 9k \equiv 1 \pmod 8$ से: $k \equiv 1$,
अतः $u = 9$; $v = 8k \equiv 1 \pmod 9$ से: $-k \equiv 1$, $k \equiv 8$, अतः $v =
64$। तब $72$ के सापेक्ष $x = 9a + 64b$ का
वर्ग $x \equiv a \pmod 8$, $x \equiv b \pmod 9$ का अद्वितीय हल है: $a =
3$, $b = 5$ के लिए $27 + 320 = 347 \equiv 59 \pmod{72}$ मिलता है
--- ठीक वही मध्यवर्ती मान जो \cref{exo:b2:structures:8} में प्रतिस्थापन से मिला था। समापन
दृष्टि: $u$ और $v$ $72$ के सापेक्ष $u + v \equiv 1$, $uv \equiv 0$, $u^2 \equiv u$, $v^2
\equiv v$ पूरा करते
हैं; वे $(1, 0)$ और $(0,
1)$ के प्रतिबिंब हैं, और हर चीनी अपघटन मूलतः $1$ का
लंबकोणीय वर्गसम अवयवों में अपघटन ही है।
\end{example}

\begin{theorem}[ऑयलर; फर्मा का पुनरावलोकन]\label{thm:b2:structures:euler}
$\Z/n\Z$ की इकाइयाँ \omterm{def:b2:structures:generated}{कोटि} $\varphi(n)$ का समूह बनाती हैं; अतः $\gcd(a, n) = 1$ के लिए:
\[
a^{\varphi(n)} \equiv 1 \pmod n
\qquad (\text{ऑयलर प्रमेय\index{ऑयलर प्रमेय}}),
\]
और फर्मा की लघु प्रमेय $n = p$ के अभाज्य होने की स्थिति है, जो अब लाग्रांज
से एक ही पंक्ति दूर है।
\end{theorem}

\begin{proof}
प्रतिलोमनीय वर्ग ठीक वही हैं जो $n$ के सहअभाज्य पूर्णांकों के हैं
(प्रथम वर्ष का खंड): उनकी संख्या $\varphi(n)$ है, और वे गुणन के अंतर्गत समूह
बनाते हैं। लाग्रांज (\cref{thm:b2:structures:lagrange}) से: प्रत्येक अवयव की समूह-कोटि वाली घात
तत्समक होती है।
\end{proof}

\begin{example}[बिना जनक का इकाई समूह]\label{ex:b2:structures:units15}
समूह $(\Z/15\Z)^*$ में $\varphi(15) = \varphi(3)\varphi(5)
= 8$ अवयव हैं। क्या वह \omterm{def:b2:structures:generated}{चक्रीय} है? चीनी तुल्याकारिता $(\Z/15\Z)^* \simeq (\Z/3\Z)^* \times (\Z/5\Z)^*$
से कोटियाँ परिकलित कीजिए ($15$ के सापेक्ष कोई इकाई इकाइयों का युग्म
होती है): दोनों गुणनखंडों की कोटियाँ $2$ और $4$ हैं, अतः हर अवयव की
\omterm{def:b2:structures:generated}{कोटि} $\operatorname{lcm}(2, 4) = 4 < 8$ को विभाजित करती है --- कोई भी अवयव जनक नहीं है। मूर्त रूप
में:
\[
2^4 = 16 \equiv 1, \qquad
4^2 = 16 \equiv 1, \qquad
7^4 \equiv 1, \qquad
11^2 = 121 \equiv 1, \qquad
14^2 \equiv 1 \pmod{15} :
\]
कोटियाँ $4, 2, 4, 2, 2$ हैं, $8$ कभी नहीं। इसकी तुलना \cref{exo:b2:structures:10} से कीजिए: $p$
अभाज्य होने पर $(\Z/p\Z)^*$ \emph{\omterm{def:b2:structures:generated}{चक्रीय} है}, क्योंकि वहाँ इकाई समूह एक क्षेत्र
के भीतर बैठता है। ऑयलर प्रमेय घातांक $\varphi(15) = 8$ के साथ अब भी लागू होती है,
परंतु यहाँ सच्चा सार्वत्रिक घातांक $4$ है --- ऑयलर एक ऊपरी सीमा देता
है, सदा सबसे तीखी नहीं।
\end{example}

\begin{definition}[बीजगणित]\label{def:b2:structures:algebra}
\emph{$K$-बीजगणित}\index{बीजगणित} ऐसी $K$-सदिश समष्टि $A$ है जिस पर
वलय संरचना हो और जिसका गुणन $K$-द्विरैखिक हो। उदाहरण: $K[X]$, $\mathcal{M}_n(K)$, $\mathcal{L}(E)$,
फलन समष्टियाँ $\mathcal{F}(X, K)$, तथा $\R$-बीजगणित के रूप में $\C$। बीजगणितों की
समाकारिताएँ रैखिक वलय समाकारिताएँ होती हैं; और $K[X]$ से $\mathcal{L}(E)$ (अथवा
$\mathcal{M}_n(K)$) में जाने वाली \emph{मूल्यांकन} $P \mapsto P(u)$ केंद्रीय उदाहरण है, जो \cref{ch:b2:reduction}
को चलाती है।
\end{definition}

\begin{example}[एक मूल्यांकन समाकारिता और उसका केंद्रक]\label{ex:b2:structures:evaluation}
लीजिए $A = \begin{pmatrix}0 & 1\\ 0 & 0\end{pmatrix}$ और मूल्यांकन $\varepsilon_A \colon \R[X] \to \mathcal{M}_2(\R)$, $P \mapsto P(A)$। चूँकि $A^2 = 0$,
\[
P(A) = P(0)\,I + P'(0)\,A =
\begin{pmatrix} P(0) & P'(0)\\ 0 & P(0)\end{pmatrix},
\]
($P$ के केवल अचर और रैखिक पद बचते हैं)। अतः $\ker\varepsilon_A = \{P : P(0) = P'(0) = 0\} = X^2\,\R[X]$: एक मुख्य \omterm{def:b2:structures:ideal}{गुणजावली},
ठीक जैसा \cref{thm:b2:structures:principal} बताती है, जो केंद्रक में न्यूनतम घात के एकिक $X^2$ से \omterm{def:b2:structures:generated}{जनित}
है --- यही $A$ का \emph{न्यूनतम बहुपद} है, \cref{ch:b2:reduction} का नायक। प्रतिबिंब
दो-विमीय क्रमविनिमेय \omterm{def:b2:structures:algebra}{बीजगणित} $\{aI + bA\}$ है: मूल्यांकन समाकारिताएँ
अनंत-विमीय $\R[X]$ को छोटे, परिकलनीय बीजगणितों पर सिकोड़ देती हैं।
\end{example}

\begin{remark}[परिप्रेक्ष्य: सुनने योग्य तीन स्वरलहरियाँ]\label{rem:b2:structures:perspectives}
इस अध्याय के तीन संरचनात्मक विचार पूरे खंड में लौटते हैं, हर बार अधिक
भारी वादन के साथ। \emph{विभाग से होकर गुणनखंडन} (\cref{def:b2:structures:quotient}): यहाँ वह $\Z/n\Z$
बनाता है, \cref{ch:b2:linalg} में रैखिक निकायों के हल-समुच्चयों पर प्रतिचित्रण
परिभाषित करता है, और ``वर्गों पर सुपरिभाषित'' वाले हर तर्क के नीचे
चुपचाप बैठा रहता है। \emph{अपरिवर्त्य}: चिह्न $\{\pm1\}$ में जाने वाली ऐसी
समाकारिता है जिससे कोई वैध चाल बच नहीं सकती --- यही तर्क सारणिक का
गुणनफल-नियम (\cref{ch:b2:linalg}), अनुरेख की सादृश्यता के अंतर्गत अपरिवर्त्यता, तथा
\cref{ch:b2:diffeq} की संरक्षित राशियाँ देता है। \emph{किसी संरचना के सहारे गिनना}:
लाग्रांज सहसमुच्चयों से गिनता है, विमा आधारों से गिनी जाती है (\cref{ch:b2:linalg}),
बहुलता बहुपद-घातों से गिनी जाती है (\cref{ch:b2:reduction}); जब भी कोई सीमा चमत्कारी लगे,
समझिए कोई विभाजन या श्रेणीकरण भीतर गिनती कर रहा है।
\end{remark}

\begin{remark}[सामान्य भूलें]\label{rem:b2:structures:pitfalls}
चार क्लासिक भूलें। (क) विभाग पर परिभाषित प्रतिचित्रण की
\emph{सुपरिभाषितता} जाँचनी ही पड़ती है: ``$\overline x \mapsto$ ($x$ पर सूत्र)'' तभी वैध
है जब सूत्र वर्गों पर अचर हो --- \cref{def:b2:structures:quotient} की संगतता कोई औपचारिकता नहीं है।
(ख) $\operatorname{ord}(ab) =
\operatorname{lcm}(\operatorname{ord}a, \operatorname{ord}b)$ सामान्यतः \emph{असत्य} है, यहाँ तक कि क्रमविनिमेय अवयवों के
लिए भी ($a$ और $a^{-1}$); सही ``सहअभाज्य और क्रमविनिमेय'' कथन \cref{exo:b2:structures:4} देती है,
और क्रमचयों वाला सही रूप असंयुक्त \omterm{def:b2:structures:sn}{चक्र} देते हैं। (ग) गणनीयता \omterm{def:b2:structures:countable}{गणनीय}
\emph{सम्मिलनों} और परिमित \emph{गुणनों} में बची रहती है, \omterm{def:b2:structures:countable}{गणनीय} गुणनों
में नहीं: $\{0,1\}^{\N}$ अगणनीय है (\cref{exo:b2:structures:3}) यद्यपि हर गुणनखंड में केवल दो अवयव हैं।
(घ) कैंटर--बर्नस्टाइन को दोनों दिशाओं में केवल एकैकी प्रतिचित्रण
चाहिए, परंतु जो एकैकी आच्छादक प्रतिचित्रण वह बनाती है वह प्रायः असंतत
और अस्पष्ट होता है --- सूत्र की आशा मत रखिए (\cref{ex:b2:structures:cbexample})।
\end{remark}

\begin{remark}[यह अध्याय कहाँ काम आता है]
लगभग सर्वत्र। चिह्न सारणिक बनाता है (\cref{ch:b2:linalg}); मूल्यांकन समाकारिता $P \mapsto P(u)$
तथा $K[X]$ की मुख्य गुणजावलियाँ न्यूनतम बहुपद और \cref{ch:b2:reduction} के केंद्रक-अपघटन
उत्पन्न करती हैं; गणनीयता वह मंच है जिस पर \cref{ch:b2:proba} अभिनय करता है (\omterm{def:b2:structures:countable}{गणनीय}
समष्टियों पर प्रायिकता) और वही कारण है कि सांस्थिति बार-बार \omterm{def:b2:structures:countable}{गणनीय} सघन
समुच्चय गढ़ती रहती है (\cref{ch:b2:metric})। विभाग की रचना $A/I$ तृतीय वर्ष के खंड में
क्षेत्र $K[X]/(P)$ बनाने के लिए फिर से लगाई जाती है और उनसे गाल्वा सिद्धांत
खड़ा होता है: यहाँ सिद्ध किया गया सार्वत्रिक गुणधर्म वहाँ शब्दशः
प्रयुक्त होता है।
\end{remark}

निम्नलिखित में से कौन-से समुच्चय \omterm{def:b2:structures:countable}{गणनीय} हैं? $\N$ के परिमित उपसमुच्चयों

\end{exercise}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:structures:3}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:structures:4}
मान लीजिए $G$ एक समूह है और $a, b \in G$ उसके ऐसे क्रमविनिमेय अवयव हैं जिनकी

सिद्ध कीजिए कि $A_n$ ($n \geq 3$) $3$-चक्रों से \omterm{def:b2:structures:generated}{जनित} है। \emph{(दो पार्ययों का

\begin{exercise}[$\star\star$]\label{exo:b2:structures:8}

\end{exercise}

\section{समस्या: पंद्रह पहेली}

पंद्रह पहेली $4 \times 4$ का एक पट है जिसमें $1$ से $15$ तक अंकित पंद्रह
सरकने वाली गोटियाँ और एक रिक्त खाना होता है; एक चाल में रिक्त खाने से
सटी किसी गोटी को उसमें सरका दिया जाता है। 1890 के दशक में सैम लॉयड ने
उसे यह घोषणा करके लोकप्रिय बनाया कि जो कोई गोटियाँ $14$ और $15$ आपस में
बदलकर शेष हर गोटी को उसके स्थान पर लौटा देगा उसे \$1000 मिलेंगे। वह
पुरस्कार कभी किसी ने नहीं लिया, और यह सप्ताहांत समस्या इसके कारण के
दोनों आधे सिद्ध करती है: \cref{thm:b2:structures:signature} का चिह्न लॉयड की अदला-बदली को रोकता है,
और --- कठिनतर, रचनात्मक आधा --- चिह्न जो कुछ भी अनुमत करता है वह
\emph{सब} वास्तव में हल किया जा सकता है। पूरा कथन जॉनसन--स्टोरी प्रमेय
(1879) है।

\begin{problem}[{सप्ताहांत समस्या --- जॉनसन--स्टोरी हलनीयता
प्रमेय}]
\label{pb:b2:structures:1}
खानों को पठन-क्रम में (बाएँ से दाएँ, ऊपर से नीचे) $1$ से $16$ तक अंकित
कीजिए, ताकि $k
= 4(i - 1) + j$ के साथ खाना $k$ पंक्ति $i$ और स्तंभ $j$ में पड़े।
खाना $16$ (नीचे दाहिना) रिक्त खाने का \emph{घर} है; रिक्त खाने को हम
सोलहवीं गोटी मानते हैं, जिसे $b$ लिखा जाता है और संख्या $16$ से पहचाना
जाता है। \emph{विन्यास} एक एकैकी आच्छादक प्रतिचित्रण $\sigma \colon \intint1{16}
\to \intint1{16}$ है, अर्थात्
खाना $\mapsto$ वस्तु; और \emph{हल किया हुआ} विन्यास $\sigma = \mathrm{id}$ है। आगे सर्वत्र
$\varepsilon$ \cref{thm:b2:structures:signature} का चिह्न है, और दो खाने \emph{सटे हुए} कहलाते हैं जब उनकी
एक भुजा साझी हो।

\medskip
\noindent\textbf{भाग I --- विन्यास, चालें, चिह्न।}
\begin{enumerate}
  \item तर्क दीजिए कि विन्यास ठीक $\mathfrak{S}_{16}$ के अवयव हैं, अतः उनकी संख्या
        $16! =
        20\,922\,789\,888\,000$ है; और यह भी कि किसी दिए गए विन्यास से वैध चालों की
        संख्या $2$, $3$ या $4$ होती है, इसके अनुसार कि रिक्त खाना
        कोने में, किनारे पर या भीतर पड़ता है।
  \item मान लीजिए $\sigma$ एक विन्यास है, $p = \sigma^{-1}(16)$ रिक्त का खाना है, और $c$
        $p$ से सटा कोई खाना। दिखाइए कि $c$ की गोटी को $p$ में
        सरकाने से $\tau =
        (p\ c)$ सहित विन्यास $\sigma' = \sigma \circ \tau$ बनता है, और इससे निष्कर्ष
        निकालिए कि हर चाल चिह्न पलट देती है: $\varepsilon(\sigma') = -\varepsilon(\sigma)$।
  \item पट को शतरंजी रंग दीजिए: पंक्ति $i$, स्तंभ $j$ वाले खाने
        $k$ के लिए $\chi(k) = (-1)^{i+j}$। दिखाइए कि हर चाल $\chi(\text{रिक्त का
        खाना})$ पलट देती है, और इससे निष्कर्ष निकालिए कि रिक्त को
        उसके आरंभिक खाने पर लौटाने वाली चालों की कोई भी शृंखला सम
        लंबाई की होती है।
  \item दिखाइए कि
        \[
        I(\sigma) = \varepsilon(\sigma)\,
        \chi\bigl(\sigma^{-1}(16)\bigr)
        \]
        हर वैध चाल के अंतर्गत अपरिवर्त्य है, और $I(\mathrm{id})$ परिकलित कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- लॉयड का इनाम: अपरिवर्त्य काम पर।}
\begin{enumerate}[resume]
  \item लॉयड का विन्यास $\sigma_L$ हल किए हुए विन्यास से केवल इतना भिन्न है
        कि खाने $14$ और $15$ में गोटियाँ $15$ और $14$ रखी हैं। $I(\sigma_L)$
        परिकलित कीजिए और निष्कर्ष निकालिए कि चालों की कोई भी शृंखला
        $\sigma_L$ को हल किए हुए विन्यास से नहीं जोड़ती: लॉयड के \$1000 पर
        कभी संकट था ही नहीं।
  \item दिखाइए कि ठीक आधे विन्यास $I =
        +1$ को संतुष्ट करते हैं: $\abs{\{\sigma : I(\sigma) = +1\}} = 16!/2$।
        \emph{(रिक्त का खाना स्थिर रखकर, दो अन्य खानों के एक निश्चित
        \omterm{def:b2:structures:sn}{पार्यय} से संयोजन करके विन्यासों को युग्मित कीजिए।)}
  \item दिखाइए कि हर चाल किसी वैध चाल से पलटी जा सकती है, कि ``$\sigma'$
        $\sigma$ से वैध चालों द्वारा प्राप्य है'' एक तुल्यता संबंध है, और
        कि हल किए हुए विन्यास का वर्ग $R$ $R \subseteq \{I = +1\}$ को संतुष्ट करता है।
        निष्कर्ष निकालिए कि कम से कम दो वर्ग हैं।
  \item मान लीजिए रिक्त अपने घर में है: $\sigma(16) = 16$। दिखाइए कि $I(\sigma) = \varepsilon(\rho)$, जहाँ
        $\rho \in
        \mathfrak{S}_{15}$ खानों $1, \dots, 15$ तक $\sigma$ का प्रतिबंधन है; और यह भी कि कोई भी
        विन्यास वैध चालों से ऐसे विन्यास तक ले जाया जा सकता है जिसमें
        रिक्त घर में हो। निष्कर्ष निकालिए: $R = \{I = +1\}$ सिद्ध करने के लिए
        इतना ही पर्याप्त है कि घर से भिन्न पंद्रह खानों का प्रत्येक
        \emph{सम} क्रमचय ऐसी चालों की शृंखला से साकार किया जाए जो
        रिक्त के घर में रहते हुए आरंभ और समाप्त होती हो।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- रिक्त के भ्रमण और कार्यक्रम समूह।}
\emph{कार्यक्रम} वैध चालों की ऐसी परिमित शृंखला है जो रिक्त के घर में
रहते हुए किसी विन्यास से आरंभ होती है और जिसका अंतिम विन्यास भी रिक्त
को घर में ही रखता है। उसका \emph{प्रभाव} खानों का वह क्रमचय $\pi$ है जो
इस प्रकार परिभाषित होता है: खाने $x$ की वस्तु खाने $\pi(x)$ में जा पहुँचती
है।
\begin{enumerate}[resume]
  \item दिखाइए कि $\sigma$ से चलाया गया कार्यक्रम $\sigma
        \circ \pi^{-1}$ पर समाप्त होता है;
        कि दो कार्यक्रम क्रमशः चलाने पर उनके प्रभाव संयोजित होते हैं;
        और कि सभी प्रभावों का समुच्चय $H$ $\mathfrak{S}_{15}$ (खानों $1, \dots, 15$ के क्रमचय)
        का उपसमूह है जो एकांतर समूह $A_{15}$ में समाहित है।
  \item (प्राथमिक भ्रमण) रिक्त के घर में रहते हुए उसे नीचे-दाहिने $2 \times 2$
        खंड के चारों ओर सरकाइए: खाने $16 \to 12 \to 11 \to 15 \to 16$। दिखाइए कि प्रभाव $3$-चक्र
        $(11\ 12\ 15)$ है, और उल्टा भ्रमण $(11\ 15\ 12)$ देता है। दोनों $H$ में हैं।
  \item (महाभ्रमण) सत्यापित कीजिए कि
        \[
        16 \to 15 \to 14 \to 13 \to 9 \to 5 \to 1 \to 2 \to 3
        \to 4 \to 8 \to 7 \to 6 \to 10 \to 11 \to 12 \to 16
        \]
        सोलहों खानों से होकर जाने वाला बंद पथ है (केवल सटे हुए पग),
        और उसका प्रभाव $15$-चक्र
        \[
        \zeta = (15\ 12\ 11\ 10\ 6\ 7\ 8\ 4\ 3\ 2\ 1\ 5\ 9\ 13\
        14) .
        \]
        है। उसके \omterm{def:b2:structures:generated}{चक्रीय} क्रम को $x_0 = 15$, $x_1 = 12$, $x_2 = 11$, \dots, $x_{14}
        = 14$ लिखकर
        जाँचिए कि प्रश्न 10 का उल्टा प्राथमिक भ्रमण ठीक $(x_0\ x_1\
        x_2)$ है।
  \item किसी भी $\mathfrak{S}_n$ में संयुग्मन सूत्र सिद्ध कीजिए: क्रमचय $g$ और
        $3$-चक्र के लिए
        \[
        g\,(a\ b\ c)\,g^{-1} = \bigl(g(a)\ g(b)\ g(c)\bigr),
        \]
        और ध्यान दीजिए कि $H$ समूह होने के कारण अपने ही अवयवों से
        संयुग्मन के अंतर्गत संवृत है।
  \item इससे निष्कर्ष निकालिए कि $H$ में महाभ्रमण के सभी पंद्रह
        \emph{क्रमागत} $3$-चक्र समाहित हैं:
        \[
        s_t = (x_t\ x_{t+1}\ x_{t+2}) \qquad (t \in \Z/15\Z,
        \text{ सूचकांक मॉड } 15).
        \]
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- एकांतर समूह को \omterm{def:b2:structures:generated}{जनित} करना।}
\begin{enumerate}[resume]
  \item (प्रमेयिका A) मान लीजिए $s$ और $t$ ऐसे $3$-चक्र हैं जिनके
        वाहक ठीक दो बिंदु साझा करते हैं, मान लीजिए वाहक $\{a, b, c\}$ और $\{b, c, d\}$
        हैं। दिखाइए कि आवश्यकता होने पर $s$ या $t$ के स्थान पर उसका
        प्रतिलोम रख देने से (जिससे \omterm{def:b2:structures:generated}{जनित} उपसमूह पर कोई अंतर नहीं पड़ता)
        गुणनफल $st$ दोहरा \omterm{def:b2:structures:sn}{पार्यय} बन जाता है; दिखाइए कि $A_4$ में \omterm{def:b2:structures:generated}{कोटि}
        $6$ का कोई उपसमूह नहीं है \emph{(सूचकांक $2$ का उपसमूह हर वर्ग
        समाहित करता है; वर्गों में $3$-चक्र गिनिए)}; और निष्कर्ष
        निकालिए कि $\langle s, t\rangle$ चार अक्षरों $\{a, b, c, d\}$ का पूरा एकांतर समूह है।
  \item (प्रमेयिका B) मान लीजिए $X$ $k \geq 4$ अक्षरों का समुच्चय है,
        $w
        \notin X$, और $G$ किसी $\mathfrak{S}_n$ का ऐसा उपसमूह है जिसमें $X$ के सभी
        सम क्रमचय तथा $u, v \in X$ सहित एक $3$-चक्र $(u\ v\ w)$ समाहित हैं।
        दिखाइए कि सभी भिन्न $a, b \in X$ के लिए $X$ का कोई \emph{सम} क्रमचय
        $g$ ऐसा है कि $g(u) = a$, $g(v) = b$ हों, और इससे $(a\ b\ w) \in G$ निष्कर्ष
        निकालिए।
  \item इससे निष्कर्ष निकालिए कि प्रमेयिका B का समूह $G$ $X \cup \{w\}$ के हर
        सम क्रमचय को समाहित करता है \emph{(\cref{exo:b2:structures:6} का उपयोग कीजिए:
        $3$-चक्र \omterm{def:b2:structures:generated}{जनित} करते हैं)}। फिर प्रश्न 13 के क्रमागत
        $3$-चक्रों $s_0, s_1, \dots, s_{12}$ के सहारे प्रमेयिकाओं A और B को शृंखलाबद्ध
        करके सिद्ध कीजिए कि $\langle s_0, \dots, s_{12}\rangle = A_{15}$।
  \item निष्कर्ष निकालिए कि $H = A_{15}$: \emph{पंद्रह गोटियों का प्रत्येक सम
        पुनर्विन्यास किसी कार्यक्रम से साध्य है}, और $H$ में $15!/2 = 653\,837\,184\,000$
        अवयव हैं।
  \item (जॉनसन--स्टोरी प्रमेय, 1879) प्रश्न 6, 7, 8 और 17 को जोड़िए:
        हल किए हुए विन्यास से प्राप्य विन्यास \emph{ठीक} वे $16!/2 =
        10\,461\,394\,944\,000$
        विन्यास हैं जिनके लिए $I = +1$; और प्राप्यता के ठीक \emph{दो}
        वर्ग हैं, हल किए हुए विन्यास का वर्ग और लॉयड के $\sigma_L$ का वर्ग।
        \emph{(दूसरे बिंदु के लिए गोटियों $14$ और $15$ के नाम आपस में
        बदल दीजिए: दिखाइए कि $\sigma \mapsto (14\ 15) \circ
        \sigma$ चाल-शृंखलाओं को चाल-शृंखलाओं पर
        भेजता है और $\{I = +1\}$ को $\{I = -1\}$ से बदल देता है।)}
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- कसौटियाँ, रूपांतर, और ऊपर से दिखता दृश्य।}
\begin{enumerate}[resume]
  \item (व्यावहारिक कसौटी) पंद्रह गोटियों को उनके खानों के पठन-क्रम
        में पढ़िए, रिक्त को छोड़ते हुए, और मान लीजिए $N$ इस सूची के
        व्युत्क्रमों की संख्या है; मान लीजिए $r$ रिक्त की पंक्ति है,
        जो \emph{नीचे} से गिनी गई हो। दिखाइए कि $I(\sigma) = (-1)^{N + r + 1}$, अतः $\sigma$ हल्य है
        यदि और केवल यदि $N + r$ विषम है।
  \item (समूह क्रियाएँ) समुच्चय $X$ पर समूह $G$ की \emph{क्रिया} एक
        ऐसा प्रतिचित्रण $G \times X \to X$, $(g, x) \mapsto g \cdot
        x$ है जिसके लिए $e \cdot x = x$ और $g \cdot (h \cdot x) =
        (gh) \cdot x$ हों; $x$
        की \emph{कक्षा} $G \cdot x$ है, और क्रिया \emph{मुक्त} कहलाती है जब
        $g \cdot x = x$ से $g =
        e$ आवश्यक हो जाए। दिखाइए कि $h \cdot \sigma = \sigma \circ h^{-1}$ रिक्त-घर विन्यासों
        के समुच्चय पर $H$ की एक मुक्त क्रिया परिभाषित करता है, कि
        उसकी कक्षाएँ ठीक कार्यक्रमों द्वारा परस्पर प्राप्यता के वर्ग
        हैं, और कक्षाओं की गिनती से पुनः प्राप्त कीजिए कि ये विन्यास
        ठीक $15!\,/\,\abs H = 2$ वर्गों में बँटते हैं।
  \item ($3 \times 3$ वाली बाधा) दिखाइए कि $3 \times 3$ पट पर ऐसा \emph{कोई} बंद पथ
        नहीं है जो हर खाने पर ठीक एक बार जाए: आठ पहेली के लिए भाग III
        की महाभ्रमण-युक्ति विफल हो जाती है। \emph{(नौ खानों को शतरंजी
        रंग दीजिए।)}
  \item (मरम्मत) $3 \times 3$ पट पर, जिसके खाने पठन-क्रम में $1$ से $9$ तक
        हैं और घर $9$ है: परिमाप-भ्रमण $9 \to 8 \to 7 \to 4 \to 1 \to 2 \to 3
        \to 6 \to 9$ (केंद्र $5$ को अचर
        छोड़ने वाला $7$-चक्र $\zeta'$) तथा कोने-भ्रमण $9 \to 6 \to 5 \to 8 \to 9$ (केंद्र से
        होकर जाने वाला $3$-चक्र) के प्रभाव परिकलित कीजिए। बाद वाले
        को $\zeta'$ की घातों से संयुग्मित करके तथा प्रमेयिकाएँ A और B
        शृंखलाबद्ध करके सिद्ध कीजिए कि आठ पहेली का कार्यक्रम समूह पूरा
        $A_8$ है, अतः $9! =
        362\,880$ विन्यासों में से ठीक $9!/2 = 181\,440$ हल्य हैं।
  \item (एक दुर्बल पट) अब पट को $n
        \geq 4$ खानों का एक ही \omterm{def:b2:structures:sn}{चक्र} मानिए जिस पर
        $n - 1$ गोटियाँ हैं। दिखाइए कि गोटियों का \omterm{def:b2:structures:generated}{चक्रीय} क्रम अपरिवर्त्य
        है, कि प्राप्यता के प्रत्येक वर्ग में ठीक $n(n - 1)$ विन्यास हैं
        \emph{(वर्ग \omterm{def:b2:structures:generated}{कोटि} $\operatorname{lcm}(n, n-1) = n(n-1)$ के किसी \omterm{def:b2:structures:generated}{चक्रीय समूह} की कक्षाएँ हैं)},
        और कि कुल $(n - 2)!$ वर्ग हैं --- $n \geq 5$ के लिए $2$ से कहीं अधिक:
        पतले पट पर सम-विषमता वाला अपरिवर्त्य लगभग कुछ नहीं पकड़ता, और
        राज ज्यामिति का चलता है।
  \item प्रश्न 19 की कसौटी से दो निर्णय: पूरी तरह उलटा पट (गोटियाँ
        $15, 14, \dots, 1$ खानों $1$ से $15$ तक, रिक्त घर में) और वह पट जिसमें
        रिक्त खाने $1$ में है और उसके बाद गोटियाँ $15, 14, \dots, 1$ खानों $2$ से
        $16$ तक हैं। इनमें से कौन-सा हल्य है?
  \item (संश्लेषण) उपपत्ति के दो स्वतंत्र स्तंभ हैं: एक
        \emph{अपरिवर्त्य} ($I$, जो चिह्न समाकारिता से बना है) जो
        दिखाता है कि अधिकतम आधे विन्यास प्राप्य हैं, और एक
        \emph{स्पष्ट जनन} प्रमेय ($H = A_{15}$) जो दिखाती है कि कम से कम आधे
        प्राप्य हैं। एक-एक वाक्य में बताइए कि निम्नलिखित कहाँ आए:
        $\varepsilon$ का समाकारिता होना; लाग्रांज प्रमेय; $3$-चक्रों द्वारा
        $A_n$ का जनन; संयुग्मन। एक पंक्ति में महासिद्धांत कहिए।
\end{enumerate}
\end{problem}
