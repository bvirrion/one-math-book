\chapter{समाकलन}\label{ch:b2:integration}

प्रथम वर्ष के खंड ने समाकल किसी खंड पर बनाया था। यह अध्याय उसे किसी भी
अंतराल तक बढ़ाता है (\omterm{def:b2:integration:improper}{अनुचित समाकल}, तुलना के पूरे औज़ार-बक्से के साथ), और
फिर \emph{प्राचल पर निर्भर} समाकलों का अध्ययन करता है --- समाकल-चिह्न के
भीतर संततता और अवकलन --- जिसे प्रभावी अभिसरण प्रमेय चलाती है, और वही इस
अध्याय का एकमात्र परिणाम है जिसे विश्वास पर लिया गया है। $\Gamma$ फलन चलता
हुआ उदाहरण भी है और गणित के आधे विशिष्ट फलनों का प्रवेशद्वार भी।

\section{किसी भी अंतराल पर समाकल}

\begin{definition}\label{def:b2:integration:improper}
मान लीजिए $f$ $\intco{a}{b}$ पर खंडशः \omterm{def:b2:metric:continuity}{संतत} है ($b \in \R$ अथवा $+\infty$)। समाकल तब
\emph{अभिसरित}\index{अनुचित समाकल} होता है जब $\lim_{x \to b^-} \int_a^x f$ विद्यमान हो; और तब उस
सीमा को $\int_a^b f$ लिखा जाता है। (इसी प्रकार $\intoc{a}{b}$ पर, और $\intoo{a}{b}$ पर किसी अभ्यंतरीय
बिंदु पर बाँटकर --- शाल के कारण चुनाव से कोई फ़र्क़ नहीं पड़ता।) समाकल
\emph{\omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसरित} तब होता है जब $\int_a^b \abs f$ अभिसरित हो; और निरपेक्ष
अभिसरण से अभिसरण निकलता है, कोशी कसौटी
\[
\Bigl| \int_x^{y} f \Bigr| \leq \int_x^{y} \abs f
\]
तथा $\R$ की पूर्णता से (प्रतिअवकलज में कोशी गुणधर्म होता है)। विस्तार से:
मान लीजिए $F(x) = \int_a^x f$ और $G(x) = \int_a^x \abs
f$। यदि $\int^b\abs f$ अभिसरित हो, तो $b^-$ पर $G$ की कोई सीमा
है, अतः प्रत्येक $\varepsilon > 0$ के लिए ऐसा $c < b$ है कि $c \leq x \leq y < b$ होने पर $G(y) - G(x)
\leq \varepsilon$; और
प्रदर्शन इस कोशी गुणधर्म को $F$ पर स्थानांतरित कर देता है। तब किसी भी
अनुक्रम $x_n
\to b^-$ के लिए मान $F(x_n)$ वास्तविक संख्याओं का कोशी अनुक्रम बनाते हैं,
जो पूर्णता से अभिसारी है; और दो ऐसे अनुक्रमों को गूँथकर देखने पर सीमा सब
के लिए एक ही निकलती है: अतः $b^-$ पर $F$ की सीमा है।
\end{definition}

\begin{theorem}[धनात्मक तुलना का औज़ार-बक्सा]\label{thm:b2:integration:comparison}
$\intco{a}{b}$ पर खंडशः \omterm{def:b2:metric:continuity}{संतत} $f, g \geq 0$ के लिए:
\begin{enumerate}
  \item $\int_a^b f$ \omterm{def:b2:integration:improper}{अभिसरित} होता है यदि और केवल यदि प्रतिअवकलज $x \mapsto \int_a^x
        f$ परिबद्ध हो;
  \item $f \leq g$: $\int g$ के अभिसरण से $\int f$ का अभिसरण निकलता है; और अपसरण दूसरी
        दिशा में स्थानांतरित होता है;
  \item $b$ पर $f \sim g$: दोनों समाकलों की प्रकृति एक ही है;
  \item संदर्भ पैमाने: $+\infty$ पर $\int^{\infty}
        \frac{\dd t}{t^\alpha}$ \omterm{def:b2:integration:improper}{अभिसरित} होता है तभी जब $\alpha > 1$, और $\int^\infty \frac{\dd t}{t(\ln t)^\beta}$
        तभी जब $\beta > 1$; और किसी परिमित अंत्यबिंदु $b$ पर $\int^b \frac{\dd t}{(b -
        t)^\alpha}$ \omterm{def:b2:integration:improper}{अभिसरित} होता
        है तभी जब $\alpha < 1$।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) प्रतिअवकलज $F(x) = \int_a^x f$ अवर्धमान नहीं है ($f \geq
0$)। यदि वह परिबद्ध हो, तो $\ell = \sup_{x < b}F$
परिमित है और $F(x) \to \ell$: $\varepsilon > 0$ दिया हो तो कोई $F(x_0) > \ell -
\varepsilon$, और एकदिष्टता $x_0 \leq x < b$ के लिए $F(x) \in
\intoc{\ell - \varepsilon}{\ell}$
को फँसा लेती है। और यदि वह अपरिबद्ध हो, तो $F \to +\infty$: अर्थात् अपसरण।

(2) $f \leq g$ से: सभी $x$ के लिए $\int_a^x f \leq \int_a^x g$; और यदि $\int^b g$ \omterm{def:b2:integration:improper}{अभिसरित} हो, तो दाहिना पक्ष
परिबद्ध है, अतः बायाँ भी, और (1) निष्कर्ष दे देता है। प्रतिधनात्मक कथन
अपसरण को दूसरी दिशा में स्थानांतरित कर देता है।

(3) $b$ पर $f \sim g$ से $c < b$ मिलता है, जहाँ
\[
\tfrac12\,g(t) \;\leq\; f(t) \;\leq\; 2\,g(t)
\qquad (c \leq t < b) :
\]
और $\intco{c}{b}$ पर दोनों दिशाओं में (2) लगाने से दोनों समाकलों की प्रकृति एक ही
हो जाती है; और आरंभिक टुकड़ा $\intcc{a}{c}$ उचित समाकल है, जो कुछ नहीं बदलता।

(4) स्पष्ट प्रतिअवकलज: $\alpha \neq 1$ और $\beta \neq 1$ के लिए
\[
\int_c^x \frac{\dd t}{t^\alpha}
= \frac{x^{1-\alpha} - c^{1-\alpha}}{1 - \alpha},
\qquad
\int_c^x \frac{\dd t}{t(\ln t)^\beta}
= \frac{(\ln x)^{1-\beta} - (\ln c)^{1-\beta}}{1 - \beta},
\]
और अपवर्जित स्थितियों में लघुगणक: $x \to
+\infty$ होने पर परिबद्ध ठीक तभी जब $\alpha > 1$,
अथवा क्रमशः $\beta > 1$। किसी परिमित अंत्यबिंदु पर प्रतिस्थापन $u = b - t$ उसे पैमाने $\int_0 u^{-\alpha}\,\dd u$
पर ला देता है, जो परिबद्ध है तभी जब $\alpha < 1$। हर बार (1) लगाइए।
\end{proof}

\begin{example}[दो अभ्यास, अंत तक किए हुए]\label{ex:b2:integration:warmups}
\emph{(क)} $\displaystyle\int_0^1 \ln t\,\dd t$: समाकल्य $0^+$ पर फट जाता है, परंतु वहाँ $\abs{\ln t} = o\bigl(t^{-1/2}\bigr)$ (लघुगणक घातों
से हार जाते हैं), और $\int_0 t^{-1/2}$ \omterm{def:b2:integration:improper}{अभिसरित} होता है: अतः निरपेक्ष अभिसरण। और मान,
$\intcc{\varepsilon}{1}$ पर खंडशः समाकलन से:
\[
\int_\varepsilon^1 \ln t\,\dd t
= \bigl[t\ln t - t\bigr]_\varepsilon^1
= -1 - \varepsilon\ln\varepsilon + \varepsilon
\xrightarrow[\varepsilon\to0^+]{} -1 .
\]
\emph{(ख)} $\displaystyle\int_0^\infty \frac{\ln t}{1 +
t^2}\,\dd t$: दोनों सिरों पर परेशानी है, अतः $1$ पर बाँटिए। $0$ के
पास: $\abs{\ln t}$, जो (क) की तरह समाकलनीय है; और $\infty$ के पास: $\frac{\ln
t}{1+t^2} = o(t^{-3/2})$: अर्थात्
\omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसारी। प्रतिस्थापन $t = \frac1u$ $\intoo{0}{1}$ को $\intoo{1}{\infty}$ पर भेज देता है
और
\[
\int_0^1 \frac{\ln t}{1+t^2}\,\dd t
= \int_1^{\infty} \frac{-\ln u}{1 + u^{-2}}\cdot
\frac{\dd u}{u^2}
= -\int_1^\infty \frac{\ln u}{1+u^2}\,\dd u :
\]
अतः दोनों आधे कट जाते हैं और समाकल $0$ है। समापन दृष्टि: $t \mapsto \frac1t$ के
अंतर्गत सममिति एक पन्ने भर के परिकलन के बराबर है --- यही युक्ति \cref{exo:b2:integration:3} को
पहले ही चला चुकी है।
\end{example}

\begin{example}[एक मान, तीन समाकल]\label{ex:b2:integration:onecos}
$I = \displaystyle\int_0^{\infty} \frac{1 - \cos
t}{t^2}\,\dd t$ का अध्ययन कीजिए। $0$ पर: $1 - \cos t \sim \frac{t^2}2$, अतः समाकल्य मान $\frac12$ से संततता के साथ
आगे बढ़ जाता है --- वहाँ कोई विचित्रता है ही नहीं। $\infty$ पर: $0 \leq \frac{1 - \cos t}{t^2}
\leq \frac{2}{t^2}$: अर्थात्
निरपेक्ष अभिसरण (\cref{thm:b2:integration:comparison})। मान: $u = 1 - \cos t$, $v' =
t^{-2}$ के साथ $\intcc{\varepsilon}{M}$ पर खंडशः समाकलन कीजिए:
\[
\int_\varepsilon^M \frac{1 - \cos t}{t^2}\,\dd t
= \Bigl[-\frac{1 - \cos t}{t}\Bigr]_\varepsilon^M
+ \int_\varepsilon^M \frac{\sin t}{t}\,\dd t .
\]
कोष्ठक दोनों सिरों पर लुप्त हो जाता है ($\frac{1 -
\cos\varepsilon}{\varepsilon} \sim \frac\varepsilon2$; $M$ पर अंश परिबद्ध), और
समाकल डिरिक्ले मान $\frac\pi2$ की ओर जाता है (\cref{exo:b2:integration:10}): अतः $I = \frac\pi2$। समापन दृष्टि:
$1 - \cos t = 2\sin^2\frac t2$ और $u =
\frac t2$ के साथ
\[
I = \int_0^\infty \frac{2\sin^2 u}{(2u)^2}\;2\,\dd u
= \int_0^\infty \Bigl(\frac{\sin u}{u}\Bigr)^{\!2}\dd u :
\]
अर्थात् तीनों क्लासिक $\int_0^\infty\frac{\sin t}{t}\dd t$, $\int_0^\infty\bigl(\frac{\sin t}{t}\bigr)^2\dd t$ (\cref{exo:b2:integration:11}) और $I$ सबका मान $\frac\pi2$ ही है, जो
खंडशः समाकलन और प्रतिस्थापन के द्वारा एक-दूसरे को सौंपा जाता रहता है ---
और उनमें से केवल पहला अर्ध-अभिसारी है: खंडशः समाकलन ने सरल समाकल्य के
बदले निरपेक्ष अभिसरण दे दिया।
\end{example}

\begin{example}[एक अर्ध-अभिसारी समाकल]\label{ex:b2:integration:sint}
$\displaystyle\int_1^{\infty} \frac{\sin t}{t}\,\dd t$ \omterm{def:b2:integration:improper}{अभिसरित} होता है: खंडशः समाकलन कीजिए,
\[
\int_1^x \frac{\sin t}{t}\dd t
= \Bigl[\frac{-\cos t}{t}\Bigr]_1^x - \int_1^x \frac{\cos
t}{t^2}\dd t ,
\]
जहाँ कोष्ठक की सीमा है और अंतिम समाकल \omterm{def:b2:series:def}{निरपेक्ष रूप से} \omterm{def:b2:integration:improper}{अभिसरित} होता है
($\abs{\cos t}/t^2 \leq t^{-2}$)। परंतु \omterm{def:b2:series:def}{निरपेक्ष रूप से} नहीं: $\abs{\sin t} \geq \sin^2 t$ से,
\[
\int_1^x \frac{\abs{\sin t}}{t}\,\dd t
\;\geq\; \int_1^x \frac{\sin^2t}{t}\,\dd t
= \underbrace{\int_1^x \frac{\dd t}{2t}}_{=\ \frac12\ln x
\ \to\ \infty}
\;-\; \underbrace{\int_1^x \frac{\cos 2t}{2t}\,\dd
t}_{\text{अभिसारी}} ,
\]
जहाँ अंतिम समाकल ऊपर वाले \emph{उसी} खंडशः समाकलन से \omterm{def:b2:integration:improper}{अभिसरित} होता है
(कोष्ठक में $\sin 2t$ के साथ): अर्थात् अपसारी टुकड़े में से अभिसारी घटाने पर
अपसरण। अतः $\int_1^\infty\frac{\sin t}{t}\dd t$ \omterm{def:b2:series:def}{निरपेक्ष रूप से} \omterm{def:b2:integration:improper}{अभिसरित} हुए बिना \omterm{def:b2:integration:improper}{अभिसरित} होता है ---
अर्थात् एकांतर श्रेणी का समाकल-रूप, जिसमें एकांतर जाँच की भूमिका खंडशः
समाकलन निभाता है।
\end{example}

\section{अभिसरण प्रमेय}

\begin{theorem}[प्रभावी अभिसरण]\label{thm:b2:integration:dominated}
मान लीजिए $(f_n)$ किसी अंतराल $I$ पर खंडशः \omterm{def:b2:metric:continuity}{संतत} हैं और बिंदुवार खंडशः \omterm{def:b2:metric:continuity}{संतत}
$f$ पर \omterm{def:b2:integration:improper}{अभिसरित} होते हैं, तथा कोई स्थिर समाकलनीय $\varphi \geq 0$ ($\int_I \varphi < \infty$) ऐसा है
कि\index{प्रभावी अभिसरण प्रमेय}
\[
\abs{f_n(t)} \leq \varphi(t) \qquad (t \in I,\ n \in \N).
\]
तब सभी $\int_I f_n$ और $\int_I f$ \omterm{def:b2:series:def}{निरपेक्ष रूप से} \omterm{def:b2:integration:improper}{अभिसरित} होते हैं, और
\[
\int_I f_n \xrightarrow[n \to \infty]{} \int_I f .
\]
\end{theorem}
\admitted

\begin{remark}
ईमानदार उपपत्ति तृतीय वर्ष के लेबेग समाकलन सिद्धांत की है; परंतु कथन का
उपयोग अब से लगातार होगा। \emph{प्रभुत्व} वाली परिकल्पना ही सारी बात है:
अकेला बिंदुवार अभिसरण पर्याप्त नहीं ($f_n = n\,\mathbf{1}_{\intoo{0}{1/n}}$, सरकते उभार: $\int f_n = 1 \not\to 0 = \int f$)। यह प्रमेय
\omterm{def:b2:metric:continuity}{संतत} प्राचल ($f_\lambda$, $\lambda \to
\lambda_0$) के लिए भी सत्य है, सीमाओं के अनुक्रम-अभिलक्षण के
द्वारा।
\end{remark}

\begin{example}[प्रभुत्व से एक गाउसीय सीमा]\label{ex:b2:integration:wallislimit}
$\displaystyle\lim_{n\to\infty} I_n$ परिकलित कीजिए, जहाँ $I_n =
\int_0^\infty \Bigl(1 + \frac{t^2}{n}\Bigr)^{\!-n}\dd t$। बिंदुवार $(1 + t^2/n)^n \to \eu^{t^2}$ (चक्रवृद्धि ब्याज वाली सीमा),
अतः समाकल्य $\eu^{-t^2}$ की ओर जाते हैं। प्रभुत्व: $u \geq 0$ के लिए अनुक्रम $n \mapsto (1 + u/n)^n$
अवर्धमान नहीं है ($1, 1 + \frac un, \dots, 1 +
\frac un$ गुणनखंडों $n + 1$ पर समांतर--गुणोत्तर माध्य $(1 + \frac u{n+1})^{n+1} \geq (1 + \frac
un)^n$ देता
है), अतः $n \geq 2$ के लिए:
\[
\Bigl(1 + \frac{t^2}{n}\Bigr)^{\!-n}
\leq \Bigl(1 + \frac{t^2}{2}\Bigr)^{\!-2},
\]
अर्थात् समाकलनीय प्रभुत्वकारी (अनंत पर $\sim 4t^{-4}$)। प्रभावी अभिसरण:
\[
I_n \xrightarrow[n\to\infty]{} \int_0^\infty \eu^{-t^2}\dd t
= \frac{\sqrt\pi}{2}
\]
(\cref{exo:b2:integration:8} का गाउसीय समाकल)। समापन जाँच: प्रतिस्थापन $t = \sqrt n\tan\theta$ $I_n$ को यथार्थतः
परिकलित कर देता है, $I_n = \sqrt n\int_0^{\pi/2}\cos^{2n-2}\theta\,\dd\theta
= \sqrt n\,W_{2n-2}$, और वालिस अनंतस्पर्शिता $W_m \sim
\sqrt{\pi/(2m)}$ (\cref{lem:b2:comparison:wallis}) फिर $\sqrt
n\,W_{2n-2} \to \frac{\sqrt\pi}2$
देती है: अर्थात् इस और पिछले अध्याय के दोनों स्तंभ सहमत हैं।
\end{example}

\begin{example}[प्रभावी अभिसरण, संतत प्राचल]
\label{ex:b2:integration:arctanlimit}
परिकलित कीजिए
\[
\lim_{x\to+\infty}\int_0^\infty
\frac{\arctan(xt)}{1+t^2}\,\dd t .
\]
प्रत्येक $t > 0$ के लिए $x \to \infty$ होने पर $\arctan(xt) \to \frac\pi2$; और प्रभुत्व
\[
\Bigl|\frac{\arctan(xt)}{1+t^2}\Bigr|
\leq \frac{\pi/2}{1+t^2},
\qquad\text{समाकलनीय, और इससे स्वतंत्र: } x,
\]
सभी $x$ के लिए सत्य है। \cref{thm:b2:integration:dominated} के संतत-प्राचल रूप से (अनुक्रम-अभिलक्षण: हर
$x_n \to \infty$ के अनुदिश जाँचिए),
\[
\int_0^\infty\frac{\arctan(xt)}{1+t^2}\,\dd t
\xrightarrow[x\to+\infty]{}
\frac\pi2\int_0^\infty\frac{\dd t}{1+t^2}
= \frac{\pi^2}{4} .
\]
समापन दृष्टि: अकेला बिंदु $t = 0$, जहाँ बिंदुवार सीमा $\frac\pi2$ के बदले $0$ है,
कुछ नहीं बदलता --- सीमा फलन केवल अपने समाकल के द्वारा प्रवेश करता है, और
यह प्रमेय की एक चुपचाप दया है।
\end{example}

\section{प्राचल वाले समाकल}

\begin{theorem}[समाकल-चिह्न के भीतर संततता]\label{thm:b2:integration:continuity}
मान लीजिए $f \colon A \times I \to \R$ ($A$ कोई \omterm{def:b2:metric:def}{दूरिक समष्टि}, $I$ कोई अंतराल) ऐसा है कि: हर $x$
के लिए $t \mapsto f(x, t)$ खंडशः \omterm{def:b2:metric:continuity}{संतत} हो; हर $t$ के लिए $x \mapsto f(x, t)$ \omterm{def:b2:metric:continuity}{संतत} हो; और कोई
\emph{प्रभुत्व} $\abs{f(x,t)} \leq \varphi(t)$ हो ($\varphi$ $I$ पर समाकलनीय और $x$ से स्वतंत्र)। तब
\[
F(x) = \int_I f(x, t)\,\dd t
\]
परिभाषित है और $A$ पर \omterm{def:b2:metric:continuity}{संतत}।\index{प्राचल समाकल}
\end{theorem}

\begin{proof}
परिभाषितता: प्रभुत्व निरपेक्ष अभिसरण दे देता है। $x_0$ पर संततता: किसी भी
अनुक्रम $x_n \to x_0$ के लिए फलन $g_n(t) =
f(x_n, t)$ स्थिर प्रभुत्व $\varphi$ के अंतर्गत बिंदुवार $f(x_0, t)$
पर \omterm{def:b2:integration:improper}{अभिसरित} होते हैं ($x$ में संततता): अतः प्रभावी अभिसरण $F(x_n) \to F(x_0)$ दे देती है;
और संततता के अनुक्रम-अभिलक्षण (\cref{def:b2:metric:continuity}) से निष्कर्ष मिल जाता है।
\end{proof}

\begin{theorem}[समाकल-चिह्न के भीतर अवकलन]\label{thm:b2:integration:leibnizrule}
मान लीजिए $f \colon J \times I \to \R$ ($J$ प्राचलों का कोई अंतराल) ऐसा है कि: हर $x$ के लिए $t \mapsto f(x,t)$
$I$ पर समाकलनीय हो; हर $t$ के लिए $x \mapsto
f(x,t)$ $C^1$ वर्ग का हो, और आंशिक अवकलज
$\frac{\partial f}{\partial x}$ $t$ में खंडशः \omterm{def:b2:metric:continuity}{संतत} तथा प्रभुत्व-युक्त हो: $\bigl|\frac{\partial f}{\partial x}(x,t)\bigr| \leq
\psi(t)$, जहाँ $\psi$
समाकलनीय। तब $F(x) = \int_I f(x,t)\dd t$ $J$ पर $C^1$ है और
\[
F'(x) = \int_I \frac{\partial f}{\partial x}(x, t)\,\dd t .
\]
\end{theorem}

\begin{proof}
$x$ और $h_n \to 0$ स्थिर कीजिए। अंतर-भागफल
\[
\frac{F(x + h_n) - F(x)}{h_n}
= \int_I \frac{f(x + h_n, t) - f(x, t)}{h_n}\,\dd t
\]
के समाकल्य बिंदुवार $\frac{\partial f}{\partial
x}(x, t)$ पर \omterm{def:b2:integration:improper}{अभिसरित} होते हैं और $\psi(t)$ से प्रभुत्व-युक्त
हैं: क्योंकि स्थिर $t$ पर $x$ में मध्यमान असमिका लगाने से
\[
\Bigl|\frac{f(x + h_n, t) - f(x,t)}{h_n}\Bigr|
\leq \sup_{\xi} \Bigl|\frac{\partial f}{\partial x}(\xi, t)\Bigr|
\leq \psi(t) .
\]
प्रभावी अभिसरण भागफलों की सीमा $\int_I \frac{\partial
f}{\partial x}(x,t)\dd t$ दे देती है: अर्थात् $F$ अवकलनीय है और
उसका अवकलज घोषित रूप वाला है, जो $\frac{\partial
f}{\partial x}$ पर लगाई गई \cref{thm:b2:integration:continuity} के अनुसार \omterm{def:b2:metric:continuity}{संतत} भी
है।
\end{proof}

\begin{example}[किसी सूत्र के विरुद्ध जाँचा गया एक प्राचल समाकल]\label{ex:b2:integration:arctanparam}
$x > 0$ के लिए $F(x) = \displaystyle\int_0^\infty \frac{\dd t}{t^2 + x}$ लीजिए। प्रत्येक $\intcc{a}{b} \subset \intoo{0}{\infty}$ पर समाकल्य $\frac{1}{t^2 + a}$ से प्रभुत्व-युक्त है,
जो समाकलनीय और $x$ से स्वतंत्र है: अतः $F$ \omterm{def:b2:metric:continuity}{संतत} है (\cref{thm:b2:integration:continuity})। और यहाँ प्रमेय
को किसी स्पष्ट मान के विरुद्ध जाँचा जा सकता है:
\[
F(x) = \Bigl[\frac{1}{\sqrt x}\arctan\frac{t}{\sqrt
x}\Bigr]_0^\infty = \frac{\pi}{2\sqrt x} ,
\]
जो स्पष्टतः \omterm{def:b2:metric:continuity}{संतत} है। अब समाकल के भीतर अवकलन कीजिए: $x$-अवकलज $-\frac{1}{(t^2+x)^2}$ $\intcc ab$
पर $\frac{1}{(t^2+a)^2}$ से प्रभुत्व-युक्त है, जो समाकलनीय है: अतः \cref{thm:b2:integration:leibnizrule} देता है
\[
F'(x) = -\int_0^\infty \frac{\dd t}{(t^2 + x)^2}
\qquad\text{जबकि}\qquad
F'(x) = -\frac{\pi}{4}\,x^{-3/2} ,
\]
और इस प्रकार हमने एक नया समाकल मुफ़्त में \emph{परिकलित} कर लिया:
$\int_0^\infty\frac{\dd t}{(t^2+x)^2} = \frac{\pi}{4x^{3/2}}$। समापन दृष्टि: किसी ज्ञात \omterm{thm:b2:integration:continuity}{प्राचल समाकल} का अवकलन नए सूत्रों की
फ़ैक्टरी है --- इसे दोहराने से प्रत्येक $n$ के लिए $\int_0^\infty\frac{\dd t}{(t^2+1)^n}$ मिल जाता है,
बिना किसी त्रिकोणमितीय प्रतिस्थापन के।
\end{example}

\begin{method}[किसी अनुचित समाकल का अध्ययन]\label{met:b2:integration:study}
$\int_a^b f$ दिया हो तो:
\begin{enumerate}
  \item परेशानी का पता लगाइए: वे अंत्यबिंदु (अथवा अभ्यंतरीय बिंदु)
        सूचीबद्ध कीजिए जहाँ $f$ अपरिबद्ध है या अंतराल अनंत है, और इस
        प्रकार बाँटिए कि हर टुकड़े में ठीक एक परेशानी वाला सिरा हो।
  \item यदि उस सिरे के पास $f$ का चिह्न अचर हो, तो कोई समतुल्य ढूँढ़िए
        और \cref{thm:b2:integration:comparison} के संदर्भ पैमानों से तुलना कीजिए।
  \item यदि $f$ दोलन करता हो, तो पहले $\abs f$ जाँचिए (निरपेक्ष अभिसरण)। और
        यदि $\int\abs f$ अपसरित हो, तो खंडशः समाकलन से दोलन के बदले क्षय लेने की
        कोशिश कीजिए, जैसा \cref{ex:b2:integration:sint} में; और $\abs{\sin t} \geq \sin^2t$ जैसे निम्न-आकलन सच्ची
        अर्ध-अभिसारिता पकड़ लेते हैं।
  \item केवल प्रकृति नहीं, मान भी चाहिए तो: खंडशः समाकलन, प्रतिस्थापन,
        अथवा कोई प्राचल (किसी सरल समाकल का अवकलन कीजिए, जैसा \cref{ex:b2:integration:arctanparam} और \cref{ex:b2:integration:laplace}
        में)।
  \item किसी भी परिकलित मान की तर्कसंगति जाँचें: किसी कच्चे परिबंध के
        विरुद्ध चिह्न और मोटा आकार ($\int_0^\infty
        \eu^{-t^2}\dd t \in \intoo{0}{1 + \int_1^\infty
        \eu^{-t}}$, अतः $\frac{\sqrt\pi}{2} \approx 0.886$ विश्वसनीय है); और
        मापन के अंतर्गत विमीय संगति ($t \mapsto \lambda t$ से दोनों पक्षों का मापन एक ही
        ढंग से बदलना चाहिए --- खोए हुए गुणक का सबसे तेज़ संसूचक)।
\end{enumerate}
\end{method}

\begin{remark}[सामान्य भूलें]
तीन बार-बार होने वाली भूलें। \emph{(क) प्राचल पर निर्भर प्रभुत्वकारी:}
प्रभुत्व $\abs{f(x,t)} \leq \varphi(t)$ विचाराधीन समुच्चय पर $x$ में एकसमान होना चाहिए; वह प्रायः
खंडों $\intcc ab$ पर सत्य होता है पर वैश्विक रूप से नहीं --- $\int_0^\infty\eu^{-xt}\dd t$ के लिए सभी $x > 0$
के लिए मान्य कोई समाकलनीय प्रभुत्वकारी है ही नहीं, फिर भी $x \geq a > 0$ पर
प्रभुत्व पूरी \omterm{def:b2:metric:topology}{विवृत} अर्धरेखा पर काम करने के लिए पर्याप्त है, क्योंकि
संततता और अवकलज स्थानीय धारणाएँ हैं। \emph{(ख) चिह्न वाले समाकल्यों की
तुलना:} तुलना का औज़ार-बक्सा \emph{अऋणात्मक} फलनों के लिए है; अपसारी $\int g$
के साथ $\abs f \leq g$ से कुछ भी निष्कर्ष नहीं निकाला जा सकता --- $\int_1^\infty\frac{\sin t}t\,\dd t$ \omterm{def:b2:integration:improper}{अभिसरित} होता
है यद्यपि $\frac1t$ से हर तुलना विफल रहती है। \emph{(ग) आधी परेशानी भूल जाना:}
$\intoo{0}{\infty}$ पर सदा दोनों सिरों का अलग-अलग अध्ययन कीजिए; $\int_0^\infty\frac{\dd t}{t}$ \emph{दोनों} पर
अपसरित होता है, और अभिसारी दिखने वाला कोई विभाजन चुपचाप दो अनंतों को काट
सकता है। सुरक्षित पलटा \cref{met:b2:integration:study} की जाँच-सूची है।
\end{remark}

\begin{example}[एक बर्ट्रांड सीमा-स्थिति, अंक तक]\label{ex:b2:integration:bertrandboundary}
\cref{thm:b2:integration:comparison} का पैमाना $\int^\infty\frac{\dd t}{t(\ln t)^\beta}$ ठीक घात-पैमानों के किनारे पर बैठा है; और उसकी
सीमा-स्थितियाँ एक पूरे परिकलन की हक़दार हैं। $\beta = 2$ के लिए:
\[
\int_\eu^{\infty}\frac{\dd t}{t(\ln t)^2}
= \Bigl[-\frac{1}{\ln t}\Bigr]_\eu^{\infty}
= 0 - (-1) = 1 ,
\]
अर्थात् एक अभिसारी समाकल जिसका मान सुखद रूप से यथार्थ है; जबकि $\beta = 1$ के
लिए
\[
\int_\eu^{x}\frac{\dd t}{t\ln t}
= \bigl[\ln\ln t\bigr]_\eu^{x} = \ln\ln x
\longrightarrow \infty ,
\]
अर्थात् अपसारी --- पर इतनी सुस्ती से कि $\ln\ln x = 10$ तक पहुँचने के लिए $x = \eu^{\eu^{10}} \approx 10^{9566}$ चाहिए।
समापन दृष्टि: ``हर घात $t^{-1-\varepsilon}$ \omterm{def:b2:integration:improper}{अभिसरित} होती है'' और ``$t^{-1}$ अपसरित होता है''
के बीच लघुगणकीय पैमानों की एक अनंत सीढ़ी बसती है, और हर सीढ़ी पिछली को
परिष्कृत करती है; प्रतिस्थापन $u = \ln t$ हर पायदान को पिछले पर ढहा देता है,
और इसीलिए बर्ट्रांड कसौटियाँ रीमान वाली कसौटियों की एक स्तर ऊपर की गूँज
हैं।
\end{example}

\begin{remark}[इसी खंड के भीतर के परिप्रेक्ष्य]
इस अध्याय के औज़ार अब सर्वत्र होने वाले हैं। प्रभावी अभिसरण अगले अध्याय
के सन्निकट तत्समकों के पीछे का इंजन है (सरकते कर्नेल, बर्नस्टाइन और
फेयेर दोनों); और समाकल-चिह्न के भीतर संततता तथा अवकलन फूरिये अध्याय में
फूरिये गुणांकों का कलन उत्पन्न करते हैं, जहाँ हर $c_n(f)$ भेस बदला हुआ \omterm{thm:b2:integration:continuity}{प्राचल
समाकल} है। $\Gamma$ फलन दो बार लौटता है: बहु समाकलों वाले अध्याय में, जहाँ कोई
दोहरा समाकल ऑयलर का बीटा--गामा सूत्र पूरी तरह सिद्ध कर देता है, और
प्रायिकता के अध्यायों में, जहाँ $\Gamma$-प्रकार के समाकल मानक घनत्वों को
सामान्यीकृत करते हैं और उनके आघूर्ण परिकलित करते हैं। और अर्ध-अभिसारी $\int\frac{\sin t}{t}$
फूरिये अध्याय के गिब्स अचर के रूप में फिर उभरता है --- वही समाकल, जो किसी
उछाल पर आंशिक योगों का अतिक्रमण मापता है।
\end{remark}

\begin{definition}[$\Gamma$ फलन]\label{def:b2:integration:gamma}
$x > 0$ के लिए:\index{गामा फलन}
\[
\Gamma(x) = \int_0^{\infty} t^{x-1}\,\eu^{-t}\,\dd t ,
\]
जो दोनों सिरों पर अभिसारी है ($x > 0$ के लिए $0^+$ पर $t^{x-1}$ समाकलनीय; और $\infty$ पर
घातांकी क्षय)।
\end{definition}

\begin{theorem}\label{thm:b2:integration:gammaprops}
$\Gamma$ $\intoo{0}{+\infty}$ पर \omterm{def:b2:metric:continuity}{संतत} है, फलनात्मक समीकरण
\[
\Gamma(x + 1) = x\,\Gamma(x),
\qquad \Gamma(1) = 1,
\qquad\text{अतः}\qquad \Gamma(n + 1) = n! ,
\]
पूरा करता है, और $C^1$ वर्ग का है (वस्तुतः $C^\infty$) जहाँ $\Gamma'(x) =
\int_0^\infty t^{x-1}\eu^{-t}\ln t\,\dd t$।
\end{theorem}

\begin{proof}
फलनात्मक समीकरण: $\intcc{\varepsilon}{M}$ पर खंडशः समाकलन कीजिए और सिरों को जाने दीजिए: $\int t^{x}\eu^{-t} = [-t^x\eu^{-t}] + x\int
t^{x-1}\eu^{-t}$,
जिसमें सीमा-पद लुप्त हो जाते हैं --- वास्तव में $\varepsilon \to
0^+$ होने पर $\varepsilon^x\eu^{-\varepsilon} \to 0$ क्योंकि
$x > 0$, और $M \to \infty$ होने पर $M^x\eu^{-M} \to 0$ क्योंकि घातांकी हर घात को हरा देता है; और दोनों
छिन्न समाकल \cref{def:b2:integration:gamma} में स्थापित अभिसरण के कारण अपने अनुचित मानों पर \omterm{def:b2:integration:improper}{अभिसरित}
होते हैं। $\Gamma(1) = \int
\eu^{-t} = 1$; और आगमन क्रमगुणित दे देता है।

$\intcc{a}{b} \subset \intoo{0}{\infty}$ पर संततता: $t^{x-1}\eu^{-t}$ को $\varphi(t) = (t^{a-1} + t^{b-1})\eu^{-t}$ से प्रभुत्व दीजिए, जो समाकलनीय और $x \in \intcc{a}{b}$ से
स्वतंत्र है: अतः \cref{thm:b2:integration:continuity} ऐसे हर खंड पर लागू होती है, इसलिए पूरी अर्धरेखा पर।
अवकलनीयता: $x$-अवकलज $t^{x-1}\eu^{-t}\ln t$ $\intcc{a}{b}$ पर $(t^{a-1}
+ t^{b-1})\eu^{-t}\,\abs{\ln t}$ से प्रभुत्व-युक्त है, जो अब भी
समाकलनीय है: अतः \cref{thm:b2:integration:leibnizrule}; और इसे दोहराने से सारे अवकलज मिल जाते हैं (हर बार
$\ln t$ की एक घात जुड़ती है, जो हानिरहित है)।
\end{proof}

\begin{example}[अर्ध-पूर्णांक क्रमगुणित]\label{ex:b2:integration:halffactorial}
फलनात्मक समीकरण और $\Gamma\bigl(\frac12\bigr) =
\sqrt\pi$ (जो \cref{exo:b2:integration:8} से एक प्रतिस्थापन दूर है: परिभाषा वाले
समाकल में $t = u^2$ रखिए) सारे अर्ध-पूर्णांक मान उत्पन्न कर देते हैं:
\[
\Gamma\Bigl(\frac32\Bigr) = \frac12\,\Gamma\Bigl(\frac12\Bigr)
= \frac{\sqrt\pi}{2},
\qquad
\Gamma\Bigl(\frac52\Bigr) = \frac32\cdot\frac{\sqrt\pi}{2}
= \frac{3\sqrt\pi}{4},
\qquad
\Gamma\Bigl(\frac72\Bigr) = \frac{15\sqrt\pi}{8} .
\]
और चूँकि $\Gamma(n+1) = n!$, यह कहना उचित है कि ``$\frac12! =
\frac{\sqrt\pi}{2} \approx 0.886$'': अर्थात् क्रमगुणित का
अंतर्वेशन हो गया, और अंतर्वेशी वक्र $0! = 1$ तथा $1! = 1$ के बीच $1$ से नीचे डुबकी
लगाता है (उसका न्यूनतम $x \approx 1.4616$ पर $\approx 0.8856$ सप्ताहांत समस्या के भाग I के
उत्तलता-चित्र से मेल खाता है)। समापन दृष्टि: समाकल $\int_0^\infty t^{x-1}\eu^{-t}\dd t$ में कुछ भी
पूर्णांकों को विशेषाधिकार नहीं देता --- क्रमगुणित का विविक्त होना गिनती
की दुर्घटना थी, और $\sqrt\pi$ वही है जो $1$ और $1$ के बीच बसता है।
\end{example}

\begin{remark}[$\Gamma$ यहाँ से कहाँ जाता है]
इस अध्याय की सप्ताहांत समस्या $\Gamma$ के चारों ओर पूरा ऑयलर-कलन खड़ा करती है:
\omterm{pb:b2:integration:1}{बीटा फलन}, उसके खंडशः-समाकलन वाले पुनरावर्तन, बीटा मानों के रूप में वालिस
समाकल, और गाउस का सीमा सूत्र। बहु समाकलों वाला अध्याय सभी कोटियों के लिए
ऑयलर का बीटा--गामा सूत्र किसी दोहरे समाकल से सिद्ध करता है; और प्रायिकता
के अध्यायों में $\Gamma$ सबसे सामान्य घनत्वों के सामान्यीकरण में तथा
प्रतीक्षा-समयों के आघूर्णों में फिर मिलता है। तृतीय वर्ष का खंड $\Gamma$ को
लेबेग की नींव पर फिर से खड़ा करता है, बोर--मोलेरुप की अद्वितीयता प्रमेय
सिद्ध करता है, और प्रभावी अभिसरण के द्वारा स्टर्लिंग सूत्र को पूर्णांकों
से बढ़ाकर पूरी वास्तविक अर्धरेखा तक ले जाता है।
\end{remark}

\begin{example}[अवकलन से एक क्लासिक परिकलन]\label{ex:b2:integration:laplace}
$x \in \R$ के लिए $F(x) = \int_0^{\infty} \eu^{-t^2}\cos(xt)\,\dd
t$ लीजिए (जो \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसारी है, $\eu^{-t^2}$ से
प्रभुत्व-युक्त)। \cref{thm:b2:integration:leibnizrule} से ($x$-अवकलज पर $t\,\eu^{-t^2}$ का प्रभुत्व, जो समाकलनीय है):
\[
F'(x) = -\int_0^\infty t\,\eu^{-t^2}\sin(xt)\,\dd t
= \Bigl[\frac{\eu^{-t^2}}{2}\sin(xt)\Bigr]_0^\infty
- \frac x2\int_0^\infty \eu^{-t^2}\cos(xt)\,\dd t
= -\frac x2\,F(x),
\]
($u' = t\eu^{-t^2}$ के साथ खंडशः)। अवकल समीकरण $F' =
-\frac x2 F$ का समाकलन $F(x) = F(0)\,\eu^{-x^2/4}$ देता है: अर्थात्
गाउसीय-प्रकार का समाकल स्वयं को पुनः उत्पन्न कर लेता है। अचर $F(0) =
\int_0^\infty \eu^{-t^2}\dd t = \frac{\sqrt\pi}{2}$ \cref{exo:b2:integration:8}
में परिकलित हुआ है --- और दोहरे समाकलन से फिर \cref{ch:b2:multint} में भी।
\end{example}

कीजिए; और $0$ पर संततता तथा $\infty$ पर सीमा का उपयोग करके निचोड़ लगाइए।)}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:8}
(प्राचल की युक्ति से गाउसीय समाकल) $x \geq 0$ के लिए

G(x) = \Bigl(\int_0^x \eu^{-t^2}\dd t\Bigr)^{\!2},

\]
रखिए। सिद्ध कीजिए कि $G' + H' = 0$ (समाकल के भीतर $H$ का अवकलन कीजिए और परिणामी

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:10}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:12}
(गाउसीय पुच्छ) $x > 0$ के लिए $T(x) = \displaystyle
\int_x^\infty \eu^{-t^2}\dd t$ रखिए।
\begin{enumerate}
  \item $\eu^{-t^2} = \frac{1}{-2t}\cdot(-2t\,\eu^{-t^2})$ लिखकर दो बार खंडशः समाकलन कीजिए और प्राप्त कीजिए
        \[
        T(x) = \eu^{-x^2}\Bigl(\frac{1}{2x} -
        \frac{1}{4x^3}\Bigr) +
        \frac34\int_x^\infty \frac{\eu^{-t^2}}{t^4}\,\dd t .
        \]
  \item शेषपद को परिबद्ध कीजिए: $0 \leq \frac34\int_x^\infty
        t^{-4}\eu^{-t^2}\dd t \leq \frac{3}{8x^5}\,\eu^{-x^2}$, और कोष्ठकन निष्कर्ष रूप में
        निकालिए
        \[
        \eu^{-x^2}\Bigl(\frac{1}{2x} - \frac{1}{4x^3}\Bigr)
        \leq T(x) \leq \frac{\eu^{-x^2}}{2x},
        \qquad\text{अतः}\qquad
        T(x) \sim \frac{\eu^{-x^2}}{2x} \quad (x \to +\infty).
        \]
  \item खंडशः समाकलन दोहराने से मिलने वाली पूरी एकांतर श्रेणी किसी स्थिर
        $x$ के लिए कभी \omterm{def:b2:integration:improper}{अभिसरित} क्यों नहीं हो सकती? \emph{(गुणांकों $1\cdot3\cdots(2k-1)$
        की वृद्धि की तुलना घातों $(2x^2)^k$ से कीजिए।)}
\end{enumerate}
\end{exercise}

\section{समस्या: ऑयलर के समाकल --- बीटा, गामा और गाउस का सीमा सूत्र}

\begin{problem}\label{pb:b2:integration:1}
\cref{def:b2:integration:gamma} का $\Gamma$ फलन ऑयलर के समाकल-कलन का एक आधा है; दूसरा आधा \emph{बीटा
फलन}\index{बीटा फलन} है
\[
B(x, y) = \int_0^1 t^{x-1}(1 - t)^{y-1}\,\dd t .
\]
यह समस्या युग्म $(\Gamma, B)$ को केवल इसी अध्याय के औज़ारों से विकसित करती है ---
खंडशः समाकलन, प्रतिस्थापन, प्रभावी अभिसरण --- और अर्ध-पूर्णांकों पर
\emph{ऑयलर के बीटा--गामा सूत्र} $B(x,y) = \frac{\Gamma(x)\Gamma(y)}
{\Gamma(x+y)}$ तथा $\Gamma$ के लिए \emph{गाउस के सीमा
सूत्र}\index{गाउस का सीमा सूत्र} पर पहुँचती है। रास्ते में \cref{lem:b2:comparison:wallis} के वालिस
समाकल बीटा मानों के रूप में फिर प्रकट होते हैं, और लजांद्र का द्विगुणन
सूत्र अपने आप निकल आता है।

\medskip
\noindent\textbf{भाग I --- $\Gamma$ की सूक्ष्म संरचना।}
\begin{enumerate}
  \item स्मरण कीजिए कि $\Gamma(x) = \int_0^\infty t^{x-1}\eu^{-t}\dd
        t$ ठीक $x > 0$ के लिए क्यों \omterm{def:b2:integration:improper}{अभिसरित} होता है, और
        दिखाइए
        \[
        \Gamma(x) \sim \frac1x \qquad (x \to 0^+)
        \]
        \emph{(फलनात्मक समीकरण तथा $1$ पर $\Gamma$ की संततता)}।
  \item सिद्ध कीजिए $\Gamma\bigl(\tfrac12\bigr) = \sqrt\pi$ \emph{($t = u^2$ प्रतिस्थापित कीजिए और \cref{exo:b2:integration:8} का सहारा
        लीजिए)}, और $\int_\R \eu^{-u^2/2}\dd u = \sqrt{2\pi}$ निष्कर्ष रूप में निकालिए।
  \item $n \in \N$ के लिए आगमन से दिखाइए:
        \[
        \Gamma\Bigl(n + \frac12\Bigr)
        = \frac{(2n)!}{4^n\,n!}\,\sqrt\pi .
        \]
  \item $\Gamma''(x) = \int_0^\infty
        t^{x-1}\eu^{-t}(\ln t)^2\dd t > 0$ को न्यायोचित ठहराइए, और इससे निष्कर्ष निकालिए कि $\Gamma$
        यथार्थतः उत्तल है, किसी $x_0 \in \intoo{1}{2}$ पर अद्वितीय न्यूनतम प्राप्त करता है
        \emph{($\Gamma(1) = \Gamma(2)
        = 1$ और रोल)}, $\intoo{0}{x_0}$ पर घटता है और $\intoo{x_0}{\infty}$ पर बढ़ता है।
  \item दिखाइए कि $\Gamma$ हर घात को हरा देता है: प्रत्येक $k \in
        \N$ के लिए $x \to +\infty$
        होने पर $x^k = o\bigl(\Gamma(x)\bigr)$ \emph{($x$ को पूर्णांकों के बीच निचोड़िए और प्रश्न
        4 की एकदिष्टता के साथ $\Gamma(n+1)
        = n!$ का उपयोग कीजिए)}।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- \omterm{pb:b2:integration:1}{बीटा फलन}, खंडशः समाकलन से।}
\begin{enumerate}[resume]
  \item दिखाइए कि $B(x,y)$ ठीक $x > 0$ और $y >
        0$ के लिए \omterm{def:b2:integration:improper}{अभिसरित} होता है, और यह कि
        $B(x,y) = B(y,x)$।
  \item $B(x, 1) = \frac1x$ परिकलित कीजिए, और $x, y > 0$ के लिए खंडशः समाकलन से सिद्ध कीजिए:
        \[
        B(x, y+1) = \frac{y}{x}\,B(x+1, y) .
        \]
  \item विभाजन $t^{x-1}(1-t)^{y-1} =
        t^{x}(1-t)^{y-1} + t^{x-1}(1-t)^{y}$ से $B(x,y) =
        B(x+1,y) + B(x,y+1)$ निष्कर्ष निकालिए, और उसे प्रश्न 7 के साथ
        मिलाकर \emph{अवरोहण संबंध} बनाइए
        \[
        B(x, y+1) = \frac{y}{x+y}\,B(x,y),
        \qquad
        B(x+1, y) = \frac{x}{x+y}\,B(x,y) .
        \]
  \item पूर्णांक $m, n \geq 1$ के लिए निष्कर्ष निकालिए:
        \[
        B(m, n) = \frac{(m-1)!\,(n-1)!}{(m+n-1)!}
        = \frac{1}{(m+n-1)\binom{m+n-2}{m-1}} .
        \]
  \item \emph{एक पूर्णांक कोटि के लिए ऑयलर का सूत्र} सिद्ध कीजिए:
        प्रत्येक $x > 0$ और $n \in \N^*$ के लिए
        \[
        B(x, n) = \frac{\Gamma(x)\,\Gamma(n)}{\Gamma(x + n)}
        \]
        \emph{($n$ पर आगमन: दोनों पक्ष $n
        = 1$ पर $\frac1x$ के बराबर हैं और
        एक ही अवरोहण संबंध का पालन करते हैं)}।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- बीटा मानों के रूप में वालिस समाकल।}
\begin{enumerate}[resume]
  \item त्रिकोणमितीय रूप पाने के लिए $t = \sin^2\theta$ प्रतिस्थापित कीजिए
        \[
        B(x, y) = 2\int_0^{\pi/2}
        \sin^{2x-1}\theta\,\cos^{2y-1}\theta\,\dd\theta .
        \]
  \item वालिस समाकल $W_n = \int_0^{\pi/2}\sin^n
        \theta\,\dd\theta$ के लिए $W_n = \frac12\,B\bigl(\frac{n+1}2, \frac12\bigr)$ निष्कर्ष रूप में निकालिए, और केवल
        प्रश्न 8 के अवरोहण संबंधों से \cref{lem:b2:comparison:wallis} की पुनरावृत्ति $W_n =
        \frac{n-1}{n}W_{n-2}$ पुनः
        प्राप्त कीजिए।
  \item $B\bigl(\frac12, \frac12\bigr) = 2W_0 = \pi$ परिकलित कीजिए और उसे $\Gamma\bigl(\frac12\bigr)^2/\Gamma(1)$ के विरुद्ध जाँचिए: अर्थात् ऑयलर का
        सूत्र $\bigl(\frac12,
        \frac12\bigr)$ पर सत्य है।
  \item पुनरावृत्ति से बंद रूप $W_{2n} = \frac\pi2\,
        \frac{(2n)!}{4^n(n!)^2}$ निकालिए, और सत्यापित कीजिए
        \[
        B\Bigl(n + \frac12, \frac12\Bigr)
        = \frac{\Gamma\bigl(n + \frac12\bigr)\Gamma\bigl(
        \frac12\bigr)}{\Gamma(n+1)} .
        \]
        फिर अवरोहण संबंधों के साथ आगमन से निष्कर्ष निकालिए कि जब भी $2x$
        और $2y$ धनात्मक पूर्णांक हों तब ऑयलर का सूत्र $B(x,y) =
        \frac{\Gamma(x)\Gamma(y)}{\Gamma(x+y)}$ सत्य होता है।
  \item तीसरा क्लासिक रूप पाने के लिए $u = \frac{t}{1-t}$ प्रतिस्थापित कीजिए
        \[
        B(x,y) = \int_0^\infty
        \frac{u^{x-1}}{(1+u)^{x+y}}\,\dd u ,
        \]
        और स्थिति $x = y = \frac12$ सीधे जाँचिए ($u = v^2$ उसे $\int_0^\infty\frac{2\,\dd v}{1+v^2}$ पर ला देता है)।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- गाउस का सीमा सूत्र।}
\begin{enumerate}[resume]
  \item $x > 0$ और $n \in \N^*$ के लिए $n$ क्रमिक खंडशः समाकलनों से सिद्ध कीजिए:
        \[
        \int_0^n \Bigl(1 - \frac tn\Bigr)^{\!n} t^{x-1}\,\dd t
        = \frac{n!\;n^x}{x(x+1)\cdots(x+n)} .
        \]
  \item \cref{exo:b2:integration:7} (प्रभावी अभिसरण) के साथ \emph{गाउस का सीमा सूत्र} निष्कर्ष
        रूप में निकालिए:
        \[
        \Gamma(x) = \lim_{n\to\infty}
        \frac{n!\;n^x}{x(x+1)\cdots(x+n)}
        \qquad (x > 0).
        \]
  \item लघुगणक लेकर दिखाइए कि $x > 0$ के लिए:
        \[
        \ln\Gamma(x) = -\ln x - \gamma x +
        \sum_{k=1}^{\infty}\Bigl(\frac xk -
        \ln\Bigl(1 + \frac xk\Bigr)\Bigr),
        \]
        जहाँ $\gamma$ ऑयलर अचर है (\cref{ex:b2:comparison:harmonic}); और श्रेणी के अभिसरण को न्यायोचित
        ठहराइए \emph{(व्यापक पद $\sim \frac{x^2}{2k^2}$ है)}।
  \item $x = \frac12$ पर गाउस का सूत्र तथा केंद्रीय द्विपद अनंतस्पर्शिता $\binom{2n}{n} \sim
        \frac{4^n}{\sqrt{\pi n}}$
        (\cref{ex:b2:comparison:centralbinomial}) काम में लेकर $\Gamma\bigl(\frac12\bigr) = \sqrt\pi$ पुनः परिकलित कीजिए: स्टर्लिंग का अचर
        और गाउसीय समाकल दो भेस में एक ही संख्या हैं।
  \item जाँचिए कि गाउस का सूत्र फलनात्मक समीकरण फिर से सिद्ध कर देता है:
        यथार्थ सर्वसमिका
        \[
        \frac{n!\,n^{x+1}}{(x+1)\cdots(x+n+1)}
        = \frac{n!\,n^{x}}{x(x+1)\cdots(x+n)}\cdot
        \frac{n\,x}{x+n+1},
        \]
        से $\Gamma(x+1) = x\,\Gamma(x)$ फिर निष्कर्ष निकालिए। (गाउस का सूत्र $\Gamma$ को सीधे
        निर्धारित कर देता है; और तृतीय वर्ष का खंड उससे तीखी
        बोर--मोलेरुप प्रमेय सिद्ध करता है: अकेले फलनात्मक समीकरण और
        लघु-उत्तलता ही $\Gamma$ को बाँध देते हैं।)
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- लाभ।}
\begin{enumerate}[resume]
  \item $a > 0$ के लिए दिखाइए $\int_0^\infty \eu^{-t^a}\dd t =
        \Gamma\bigl(1 + \frac1a\bigr)$, और $a \to +\infty$ होने पर सीमा प्रभावी अभिसरण से
        परिकलित कीजिए \emph{(बिंदुवार सीमा $\mathbf 1_{t < 1}$; $\intoc{0}{1}$ पर $1$ से और उससे
        आगे $a \geq
        2$ के लिए $\eu^{-t^2}$ से प्रभुत्व दीजिए)}। उत्तर को $\Gamma$ की संततता
        के विरुद्ध जाँचिए।
  \item $n \geq 1$ के लिए दिखाइए
        \[
        \int_0^1 \frac{\dd t}{\sqrt{1 - t^n}}
        = \frac1n\,B\Bigl(\frac1n, \frac12\Bigr),
        \]
        और मान $2$ ($n = 1$) तथा $\frac\pi2$ ($n
        = 2$) पुनः प्राप्त कीजिए। ($n = 4$ के
        लिए यह \emph{लेम्निस्केट अचर} है, जिसका कोई प्रारंभिक बंद रूप
        नहीं है; उसकी कहानी दीर्घवृत्तीय समाकलों के सिद्धांत की है।)
  \item (आघूर्ण) $x > 0$ और $k \in \N$ के लिए दिखाइए
        \[
        \frac{1}{\Gamma(x)}\int_0^\infty
        t^{k}\,t^{x-1}\eu^{-t}\,\dd t
        = \frac{\Gamma(x+k)}{\Gamma(x)}
        = x(x+1)\cdots(x+k-1),
        \]
        अर्थात् आरोही क्रमगुणित; और जाँचिए कि $x = 1$ $k!$ देता है।
        (प्रायिकता के अध्यायों में यह किसी मानक प्रतीक्षा-समय घनत्व का
        $k$-वाँ आघूर्ण है।)
  \item \emph{सभी} $x > 0$ के लिए मान्य बीटा सर्वसमिका सिद्ध कीजिए:
        \[
        B(x, x) = 2^{1-2x}\,B\Bigl(x, \frac12\Bigr)
        \]
        \emph{($t = \frac{1+s}2$ प्रतिस्थापित कीजिए, $s$ में सममिति का लाभ उठाइए,
        फिर $s = \sqrt v$ रखिए)}। इससे $2x \in \N^*$ के लिए \emph{लजांद्र का द्विगुणन
        सूत्र} निष्कर्ष रूप में निकालिए
        \[
        \Gamma(x)\,\Gamma\Bigl(x + \frac12\Bigr)
        = 2^{1-2x}\,\sqrt\pi\;\Gamma(2x),
        \]
        और उसे प्रश्न 3 के द्वारा $x = n$ पर सीधे सत्यापित कीजिए।
        (सामान्य $x$ के लिए वह उसी सर्वसमिका से निकलता है, बशर्ते
        ऑयलर का सूत्र सभी कोटियों के लिए ज्ञात हो --- अर्थात् बहु
        समाकलों वाले अध्याय की दोहरे-समाकल वाली उपपत्ति।)
  \item संश्लेषण। एक-एक वाक्य में: (क) खंडशः समाकलन ने पूरा भाग II कहाँ
        उठाया; (ख) प्रभावी अभिसरण भाग IV और V में कहाँ आई; (ग) तुलना
        वाले अध्याय से कौन-से अनंतस्पर्शी निवेश आयात हुए; (घ) ऑयलर के
        सूत्र $B(x,y) = \Gamma(x)\Gamma(y)/\Gamma(x+y)$ का अब क्या सिद्ध हो चुका है, और दोहरे समाकल के
        निपटाने के लिए क्या शेष है।
\end{enumerate}
\end{problem}
