\chapter{फलनों के अनुक्रम और श्रेणियाँ}\label{ch:b2:funcseq}

जब फलन किसी फलन पर अभिसरित होते हैं, तब कौन-से गुणधर्म सीमा तक जाने में
बच रहते हैं? \omterm{def:b2:funcseq:def}{बिंदुवार अभिसरण} लगभग कुछ भी सुरक्षित नहीं रखता;
\emph{एकसमान} अभिसरण --- अर्थात् उच्चतम \omterm{def:b2:nvs:norm}{मानदंड} में अभिसरण --- संततता,
खंडों पर समाकल, और एक मोड़ के साथ अवकलज भी सुरक्षित रखता है। यह अध्याय
तीनों स्थानांतरण प्रमेयें, उनके श्रेणी-रूप, और उन पर मुकुट के रूप में
वाइरश्ट्रास सन्निकटन प्रमेय सिद्ध करता है, जिसे बर्नस्टाइन के सुंदर
प्रायिकतात्मक बहुपदों से सिद्ध किया गया है।

\section{बिंदुवार और एकसमान अभिसरण}

\begin{definition}\label{def:b2:funcseq:def}
मान लीजिए $f_n, f \colon X \to \R$ (अथवा $\C$, या कोई मानदंडित समष्टि), और $X$ कोई समुच्चय।
$(f_n)$ $f$ पर \emph{बिंदुवार}\index{बिंदुवार अभिसरण} अभिसरित होता है जब
प्रत्येक $x$ के लिए $f_n(x) \to
f(x)$ हो; और \emph{एकसमान रूप से}\index{एकसमान
अभिसरण} तब जब
\[
\norm{f_n - f}_\infty = \sup_{x \in X}\, \abs{f_n(x) - f(x)}
\xrightarrow[n \to \infty]{} 0 .
\]
एकसमान से बिंदुवार निकलता है; और $C(\intcc{a}{b})$ पर एकसमान अभिसरण ठीक \cref{ch:b2:nvs} की \omterm{def:b2:nvs:banach}{बानाख
समष्टि} $\bigl(C(\intcc{a}{b}), \norm\cdot_\infty\bigr)$ में अभिसरण है।
\end{definition}

\begin{example}\label{ex:b2:funcseq:xn}
$\intcc{0}{1}$ पर $f_n(x) = x^n$ \emph{असंतत} सीमा $f = \mathbf{1}_{\{1\}}$ पर \omterm{def:b2:funcseq:def}{बिंदुवार} अभिसरित होता है; और यह
अभिसरण एकसमान नहीं है: $\norm{f_n - f}_\infty \geq f_n\bigl(1 -
\tfrac1n\bigr) = (1 - \tfrac1n)^n \to \eu^{-1} \neq 0$। जबकि $a < 1$ वाले $\intcc{0}{a}$ पर वह एकसमान \emph{है}
($\sup = a^n \to
0$): अर्थात् एकसमानता जितनी अनुक्रम का गुणधर्म है उतनी ही प्रांत का
भी।
\end{example}

\begin{example}[दो सीमाएँ जो क्रम बदलने से इनकार करती हैं]\label{ex:b2:funcseq:doublelimit}
पूरा अध्याय सीमाओं की अदला-बदली के बारे में है, अतः यह रही सबसे छोटी
संभव विफलता। $n, m \geq 1$ के लिए $a_{n,m} = \dfrac{n}{n+m}$ लीजिए। तब
\[
\lim_{m\to\infty}\Bigl(\lim_{n\to\infty}a_{n,m}\Bigr)
= \lim_{m\to\infty} 1 = 1,
\qquad
\lim_{n\to\infty}\Bigl(\lim_{m\to\infty}a_{n,m}\Bigr)
= \lim_{n\to\infty} 0 = 0 :
\]
दोनों क्रमिक सीमाएँ विद्यमान हैं और वे भिन्न हैं। इस अध्याय की हर
स्थानांतरण प्रमेय दो सीमाओं की अदला-बदली का लाइसेंस है --- $\lim_n$ के साथ $\lim_{x\to a}$
(संततता), $\int$ (समाकलन), $\frac{\dd}{\dd x}$ (अवकलन) --- और \omterm{def:b2:funcseq:def}{एकसमान अभिसरण} ठीक वह शुल्क है
जो इस अदला-बदली को क़ानूनी बना देता है। समापन दृष्टि: जब भी कोई ``उपपत्ति''
चुपचाप दो सीमा-संक्रियाएँ आपस में बदल दे, तब उसके सामने रखने के लिए यही
दो-पंक्ति की सारणी प्रतिउदाहरण है; और \cref{exo:b2:funcseq:2} के सरकते उभार वही परिघटना हैं,
बस समाकल-चिह्न पहने हुए।
\end{example}

\section{तीनों स्थानांतरण प्रमेयें}

\begin{theorem}[संततता]\label{thm:b2:funcseq:continuity}
यदि हर $f_n$ $a$ पर \omterm{def:b2:metric:continuity}{संतत} हो और $a$ के किसी प्रतिवेश पर $f_n \to f$ \omterm{def:b2:funcseq:def}{एकसमान रूप से}
अभिसरित हो, तो $f$ $a$ पर \omterm{def:b2:metric:continuity}{संतत} है। \omterm{def:b2:metric:continuity}{संतत} फलनों की एकसमान सीमा \omterm{def:b2:metric:continuity}{संतत} होती
है।
\end{theorem}

\begin{proof}
वही $3\varepsilon$ तर्क जो \cref{thm:b2:metric:rncomplete} में पहले ही प्रयुक्त हुआ: $\norm{f_n -
f}_\infty \leq \varepsilon$ वाला $n$ चुनिए, फिर
$a$ पर $f_n$ की संततता से $\delta$; और $\abs{x - a} \leq \delta$ के लिए
\[
\abs{f(x) - f(a)} \leq \abs{f(x) - f_n(x)} + \abs{f_n(x) - f_n(a)}
+ \abs{f_n(a) - f(a)} \leq 3\varepsilon . \qedhere
\]
\end{proof}

\begin{example}[एकसमानता ठीक वहीं विफल होती है जहाँ सीमा टूटती है]\label{ex:b2:funcseq:threshold}
$\intcc{0}{2}$ पर $f_n(x) = \dfrac{x^n}{1 + x^n}$ लीजिए। \omterm{def:b2:funcseq:def}{बिंदुवार} सीमा तीन टुकड़ों वाला फलन है:
\[
f(x) = \begin{cases} 0 & 0 \leq x < 1,\\[2pt]
\tfrac12 & x = 1,\\[2pt]
1 & 1 < x \leq 2, \end{cases}
\]
जो $1$ पर असंतत है, अतः \cref{thm:b2:funcseq:continuity} के अनुसार अभिसरण $\intcc{0}{2}$ पर एकसमान हो ही नहीं
सकता। परंतु देहली से बचने वाले संवृत टुकड़ों पर वह एकसमान है: $0 \leq x \leq
a < 1$ के लिए
\[
\sup_{\intcc{0}{a}}\abs{f_n - 0}
= \frac{a^n}{1 + a^n} \leq a^n \to 0 ,
\]
और $1 < b \leq x \leq 2$ के लिए
\[
\sup_{\intcc{b}{2}}\abs{f_n - 1}
= \frac{1}{1 + b^n} \leq b^{-n} \to 0 ,
\]
जिनके दोनों उच्चतम $u \mapsto
\frac{u}{1+u}$ तथा $x \mapsto x^n$ की एकदिष्टता से परिकलित हुए। समापन
दृष्टि: एकसमानता की विफलता सीमा की असंततता पर स्थानीयकृत है --- वही
ज्यामिति जो \cref{ex:b2:funcseq:xn} में थी, और यही कारण है कि ``भीतर के हर खंड पर एकसमान''
वाला अनुशासन पूरे अध्याय में लौटता रहता है।
\end{example}

\begin{theorem}[किसी खंड पर समाकलन]\label{thm:b2:funcseq:integration}
यदि $\intcc{a}{b}$ पर $f_n \to f$ \omterm{def:b2:funcseq:def}{एकसमान रूप से} अभिसरित हो, जहाँ $f_n$ खंडशः \omterm{def:b2:metric:continuity}{संतत} है (और $f$
भी), तो
\[
\int_a^b f_n \longrightarrow \int_a^b f .
\]
\end{theorem}

\begin{proof}
समाकलों के लिए रैखिकता और त्रिभुज असमिका देती हैं
\[
\Bigl|\int_a^b f_n - \int_a^b f\Bigr|
= \Bigl|\int_a^b (f_n - f)\Bigr|
\leq \int_a^b\abs{f_n - f}
\leq (b - a)\,\norm{f_n - f}_\infty \longrightarrow 0 .
\]
लंबाई गुणक $(b - a)$ वहीं है जहाँ खंड की संहतता प्रवेश करती है: असंहत अंतरालों
पर वही आकलन बेकार परिबंध $\infty\cdot0$ उत्पन्न करता है, और प्रभुत्व के बिना निष्कर्ष
सचमुच विफल हो जाता है --- चपटे उभार $f_n =
\frac1n\mathbf 1_{\intcc{0}{n}}$ $\intco{0}{\infty}$ पर $0$ पर \omterm{def:b2:funcseq:def}{एकसमान रूप से}
अभिसरित होते हैं फिर भी $\int f_n = 1$ बनाए रखते हैं (नीचे भूलों वाली टिप्पणी), और
\cref{ch:b2:integration} की टिप्पणी के सरकते उभार \omterm{def:b2:funcseq:def}{बिंदुवार अभिसरण} के साथ वही करते हैं: \omterm{def:b2:funcseq:def}{एकसमान
अभिसरण} ऊँचाइयाँ नियंत्रित करता है, चौड़ाइयाँ कभी नहीं।
\end{proof}

\begin{theorem}[अवकलन]\label{thm:b2:funcseq:differentiation}
मान लीजिए $f_n$ किसी अंतराल $I$ पर $C^1$ हैं, और: $(f_n')$ $I$ पर (अथवा $I$ के हर
खंड पर) किसी $g$ पर \emph{एकसमान} रूप से अभिसरित होते हैं, तथा $(f_n(x_0))$ किसी
एक बिंदु $x_0$ पर अभिसरित होता है। तब $(f_n)$ किसी $C^1$ फलन $f$ पर (खंडों पर
\omterm{def:b2:funcseq:def}{एकसमान रूप से}) अभिसरित होता है और $f'
= g$: अर्थात् सीमा का अवकलन किया जा
सकता है।
\end{theorem}

\begin{proof}
$f(x) = \lim f_n(x_0) + \int_{x_0}^x g$ परिभाषित कीजिए: यह वैध है, क्योंकि $g$ \omterm{def:b2:metric:continuity}{संतत} है --- वास्तव में $g$
\omterm{def:b2:metric:continuity}{संतत} $f_n'$ की खंडों पर \emph{एकसमान} सीमा है, अतः \cref{thm:b2:funcseq:continuity} लागू होती है, और
\omterm{def:b2:metric:continuity}{संतत} फलन का समाकल सुपरिभाषित है, तथा कलन की मूल प्रमेय से
\[
f'(x) = g(x) \qquad (x \in I) :
\]
अर्थात् संभावित सीमा $C^1$ है और उसका अवकलज \emph{रचना से ही} सही है, किसी
भी अभिसरण के सिद्ध होने से पहले। मूल प्रमेय से फिर $f_n(x) = f_n(x_0) + \int_{x_0}^x
f_n'$; और घटाने पर
\[
\abs{f_n(x) - f(x)} \leq \abs{f_n(x_0) - \lim f_n(x_0)}
+ \abs{x - x_0}\,\norm{f_n' - g}_{\infty} ,
\]
जो हर खंड पर \omterm{def:b2:funcseq:def}{एकसमान रूप से} $0$ की ओर जाता है। और $f$ रचना से $f' = g$ सहित
$C^1$ है।
\end{proof}

\begin{example}[परिकल्पना अवकलजों पर क्यों बैठती है]\label{ex:b2:funcseq:sqrtabs}
$\R$ पर $F_n(x) = \sqrt{x^2 + \frac1n}$ लीजिए। हर $F_n$ $C^1$ है (वस्तुतः $C^\infty$), और $\abs x$ पर अभिसरण पूरे $\R$
पर एकसमान है:
\[
0 \leq F_n(x) - \abs x
= \frac{(x^2 + \frac1n) - x^2}{\sqrt{x^2+\frac1n} + \abs x}
= \frac{1/n}{\sqrt{x^2 + \frac1n} + \abs x}
\leq \frac{1/n}{1/\sqrt n} = \frac{1}{\sqrt n} .
\]
फिर भी सीमा $\abs x$ $0$ पर अवकलनीय नहीं है: \emph{फलनों} का \omterm{def:b2:funcseq:def}{एकसमान अभिसरण},
चाहे कितना ही तेज़ हो, अवकलनीयता स्थानांतरित नहीं करता। विफलता अवकलजों पर
दिखाई देती है:
\[
F_n'(x) = \frac{x}{\sqrt{x^2 + \frac1n}}
\longrightarrow \begin{cases} 1 & x > 0,\\ 0 & x = 0,\\
-1 & x < 0, \end{cases}
\]
अर्थात् असंतत \omterm{def:b2:funcseq:def}{बिंदुवार} सीमा, इसलिए $0$ के पास $(F_n')$ \omterm{def:b2:funcseq:def}{एकसमान रूप से}
अभिसरित नहीं हो सकता (फिर \cref{thm:b2:funcseq:continuity})। समापन दृष्टि: \cref{thm:b2:funcseq:differentiation} जान-बूझकर $f_n$ का
नहीं बल्कि $f_n'$ का \omterm{def:b2:funcseq:def}{एकसमान अभिसरण} मानती है --- और यही उदाहरण उसका कारण
है।
\end{example}

\section{फलनों की श्रेणियाँ}

\begin{definition}\label{def:b2:funcseq:series}
फलनों की कोई श्रेणी $\sum u_n$ \omterm{def:b2:funcseq:def}{बिंदुवार}/\omterm{def:b2:funcseq:def}{एकसमान रूप से} तब अभिसरित होती है जब
उसके आंशिक योग वैसा करें। और वह ($X$ पर) \emph{सामान्य रूप
से}\index{सामान्य अभिसरण} तब अभिसरित होती है जब $\sum
\norm{u_n}_\infty < \infty$। सामान्य अभिसरण से
\omterm{def:b2:funcseq:def}{एकसमान अभिसरण} निकलता है (परिबद्ध फलनों की \omterm{def:b2:nvs:banach}{बानाख समष्टि} में: \cref{thm:b2:nvs:absoluteconvergence}), और उससे
\omterm{def:b2:funcseq:def}{बिंदुवार}; और दोनों निहितार्थ यथार्थ हैं।
\end{definition}

\begin{example}[एक श्रेणी, तीन निर्णय]\label{ex:b2:funcseq:threemodes}
$\intco{0}{1}$ पर $u_n(x) = \frac{x^n}{n}$ लीजिए। \emph{\omterm{def:b2:funcseq:def}{बिंदुवार}:} प्रत्येक $x \in \intco01$ के लिए अभिसारी (गुणोत्तर
श्रेणी से तुलना)। \emph{$\intcc{0}{a}$, $a < 1$ पर सामान्य:} $\norm{u_n}_{\infty,\intcc0a} = \frac{a^n}{n}$, जो \omterm{def:b2:series:summable}{योग्य} है। \emph{$\intco{0}{1}$
पर सामान्य नहीं:} $\norm{u_n}_{\infty,\intco01} = \frac1n$, और $\sum\frac1n$ अपसरित होती है। \emph{$\intco{0}{1}$ पर एकसमान तक
नहीं:} $1$ के पास शेषपद अड़ जाता है,
\[
R_N(x) = \sum_{n>N}\frac{x^n}{n}
\geq \sum_{n=N+1}^{2N}\frac{x^n}{n}
\geq \frac{N\,x^{2N}}{2N} = \frac{x^{2N}}{2}
\xrightarrow[x\to1^-]{} \frac12 ,
\]
अतः प्रत्येक $N$ के लिए $\sup_{\intco01}\abs{R_N} \geq \frac12$। समापन दृष्टि: चारों निर्णय शांति से साथ
रहते हैं --- योग $-\ln(1-x)$ $\intco{0}{1}$ पर \omterm{def:b2:metric:continuity}{संतत} है क्योंकि संततता को केवल \emph{हर
बिंदु के पास} एकसमानता चाहिए, अर्थात् खंडों $\intcc0a$ पर; और किनारे पर फट जाना
योग का अधिकार है।
\end{example}

\begin{theorem}[श्रेणियों के लिए स्थानांतरण]\label{thm:b2:funcseq:seriestransfer}
यदि $\sum u_n$ संबंधित समुच्चय पर \omterm{def:b2:funcseq:def}{एकसमान रूप से} (जैसे \omterm{def:b2:funcseq:series}{सामान्य रूप से}) अभिसरित
हो: तो $a$ पर सभी $u_n$ की संततता योग तक जाती है; किसी खंड पर समाकलन
पद-दर-पद किया जा सकता है; और यदि $\sum u_n(x_0)$ अभिसरित हो जबकि $\sum u_n'$ खंडों पर \omterm{def:b2:funcseq:def}{एकसमान
रूप से} अभिसरित हो, तो योग $C^1$ है और उसका अवकलज $\sum u_n'$ है।
\end{theorem}

\begin{proof}
सब कुछ संगत प्रमेय का आंशिक योगों $S_N = \sum_{n\leq N}u_n$ पर प्रयोग है, जो संबंधित नियमितता
वाले फलनों के परिमित योग हैं। संततता: हर $S_N$ $a$ पर \omterm{def:b2:metric:continuity}{संतत} है और $S_N \to \sum u_n$
एकसमान: अतः \cref{thm:b2:funcseq:continuity}। समाकलन: खंड पर
\[
\int_a^b \sum_{n\geq0} u_n
= \lim_N \int_a^b S_N
= \lim_N \sum_{n=0}^{N}\int_a^b u_n
= \sum_{n\geq0}\int_a^b u_n
\]
जिसमें पहली समता \cref{thm:b2:funcseq:integration} से और दूसरी समाकल की रैखिकता से आती है। अवकलन: $S_N$
$C^1$ हैं, $S_N(x_0)$ अभिसरित होता है, और $S_N' =
\sum_{n\leq N}u_n'$ खंडों पर \omterm{def:b2:funcseq:def}{एकसमान रूप से} अभिसरित होता
है: अतः \cref{thm:b2:funcseq:differentiation} से मिलता है कि योग $C^1$ है और उसका अवकलज $\lim S_N' = \sum u_n'$।
\end{proof}

\begin{remark}[सामान्य भूलें]
चार जाल, और चारों उत्तर-पुस्तिकाओं में देखे गए। \emph{(क) आधी जाँची गई
उच्चतम:} किसी सुविधाजनक अनुक्रम $x_n$ के अनुदिश $f_n$ का मूल्यांकन $\norm{f_n - f}_\infty$ को
केवल \emph{नीचे} से परिबद्ध करता है --- एकसमानता खंडित करने के लिए
पर्याप्त (जैसा \cref{ex:b2:funcseq:xn} में), सिद्ध करने के लिए कभी नहीं; सिद्ध करने के लिए
\emph{सभी} $x$ के लिए मान्य परिकलन से उच्चतम को परिबद्ध कीजिए। \emph{(ख)
ग़लत समुच्चय पर एकसमानता:} सामान्य या \omterm{def:b2:funcseq:def}{एकसमान अभिसरण} प्रायः हर $\intcc{-a}{a}$ अथवा
$\intco\delta\infty$ पर सत्य होता है पर \omterm{def:b2:metric:topology}{विवृत} सम्मिलन पर विफल; और यह कोई बाधा नहीं है ---
संततता तथा अवकलनीयता स्थानीय हैं, अतः \cref{ex:b2:funcseq:zeta} का खंड-दर-खंड अनुशासन उन्हें
पूरे \omterm{def:b2:metric:topology}{विवृत} समुच्चय पर दे देता है। \emph{(ग) खंड से भिन्न पर समाकलन:} \cref{thm:b2:funcseq:integration}
खंडों के बारे में कथन है; $\intco0\infty$ पर \omterm{def:b2:funcseq:def}{एकसमान अभिसरण} द्रव्यमान को अनंत की ओर
भाग जाने से नहीं रोकता ($f_n =
\frac1n\mathbf 1_{\intcc{0}{n}}$ $0$ पर \omterm{def:b2:funcseq:def}{एकसमान रूप से} अभिसरित होते हैं, और
$\int f_n = 1$) --- वहाँ प्रभावी अभिसरण काम में लीजिए। \emph{(घ) सीमा का अवकलन:}
\cref{ex:b2:funcseq:sqrtabs}; परिकल्पना $(f_n')$ पर है, और $(f_n)$ के अभिसरण की कोई भी दर उसकी जगह नहीं
ले सकती।
\end{remark}

\begin{example}[रीमान $\zeta$ फलन]\label{ex:b2:funcseq:zeta}
$\zeta(s) = \sum_{n\geq1} n^{-s}$ प्रत्येक अर्धरेखा $\intco{a}{+\infty}$, $a > 1$ पर \omterm{def:b2:funcseq:series}{सामान्य रूप से} अभिसरित होती है ($\norm{n^{-s}}_\infty =
n^{-a}$, जो
\omterm{def:b2:series:summable}{योग्य} है): अतः $\zeta$ $\intoo{1}{+\infty}$ पर \omterm{def:b2:metric:continuity}{संतत} है; और पद-दर-पद अवकलन करने पर (व्युत्पन्न
श्रेणी $\sum -\ln n\; n^{-s}$ भी $\intco{a}{\infty}$ पर \omterm{def:b2:funcseq:series}{सामान्य रूप से} अभिसरित होती है) $\zeta$ $C^1$ है --- और
इसे दोहराने पर $C^\infty$ --- जहाँ $\zeta'(s) = -\sum \frac{\ln
n}{n^s}$। अनुशासन पर ध्यान दीजिए: \omterm{def:b2:funcseq:series}{सामान्य
अभिसरण} \emph{उप}-अर्धरेखाओं पर जाँचा जाता है, कभी स्वयं \omterm{def:b2:metric:topology}{विवृत} $\intoo{1}{\infty}$ पर
नहीं, जहाँ वह विफल हो जाता है।
\end{example}

\begin{example}[एक लघुगणकीय श्रेणी, अंत तक की हुई]\label{ex:b2:funcseq:logseries}
$\intoo{0}{\infty}$ पर $F(x) = \sum_{n\geq1} \frac{\eu^{-nx}}{n}$ लीजिए। $\intco{\delta}
\infty$ पर हर पद $\frac{\eu^{-n\delta}}{n} \leq \eu^{-n\delta}$ से परिबद्ध है, जो अभिसारी गुणोत्तर
श्रेणी है: अतः प्रत्येक $\intco\delta\infty$ पर \omterm{def:b2:funcseq:series}{सामान्य अभिसरण}, इसलिए $F$ $\intoo{0}{\infty}$ पर \omterm{def:b2:metric:continuity}{संतत} है।
व्युत्पन्न श्रेणी $\sum -\eu^{-nx}$ भी $\intco\delta\infty$ पर उसी तरह \omterm{def:b2:funcseq:series}{सामान्य रूप से} अभिसारी है ($\norm{\eu^{-nx}}_{\infty,\intco\delta\infty} =
\eu^{-n\delta}$),
अतः $F$ गुणोत्तर अवकलज के साथ $C^1$ है:
\[
F'(x) = -\sum_{n\geq1}\eu^{-nx}
= \frac{-\eu^{-x}}{1 - \eu^{-x}}
= \frac{-1}{\eu^{x} - 1} .
\]
और इसे दोहराने पर $F$ $C^\infty$ है। $F'$ का समाकलन कीजिए (दोनों $F$ और $x
\mapsto -\ln(1 - \eu^{-x})$
$+\infty$ पर लुप्त होते हैं और $\intoo0\infty$ पर उनके अवकलज एक ही हैं):
\[
F(x) = -\ln\bigl(1 - \eu^{-x}\bigr),
\]
अर्थात् $t = \eu^{-x}$ पर लघुगणकीय श्रेणी। समापन दृष्टि: $x
\to 0^+$ होने पर $F(x) = -\ln(x + O(x^2)) = \ln\frac1x + O(x)$ ---
श्रेणी सीमा पर लघुगणकीय ढंग से अपसरित होती है, ठीक उसी हरात्मक श्रेणी की
तरह जो वह $x = 0$ पर बन जाती है; और $\intco\delta\infty$ पर \omterm{def:b2:funcseq:series}{सामान्य अभिसरण} पर $\intoo0\infty$ पर नहीं,
यही उसका लक्षण है।
\end{example}

\begin{method}[एकसमान अभिसरण सिद्ध या खंडित करना]\label{met:b2:funcseq:uniform}
$X$ पर \omterm{def:b2:funcseq:def}{बिंदुवार} $f_n \to f$ के लिए:
\begin{enumerate}
  \item $\norm{f_n - f}_\infty$ परिकलित या परिबद्ध कीजिए: फलन $x \mapsto \abs{f_n(x) - f(x)}$ का अध्ययन कीजिए (अवकलज,
        एकदिष्टता) ताकि उसका महत्तम मिल जाए; और सभी $x$ के लिए मान्य
        कोई ऐसा परिबंध जो $0$ की ओर जाता हो, एकसमानता सिद्ध कर देता है।
  \item \emph{खंडित} करने के लिए: ऐसे बिंदु $x_n$ दिखाइए जिनके लिए $\abs{f_n(x_n) - f(x_n)} \not\to 0$
        (प्रायः $x_n$ सरकते उभार का पीछा करता है, जैसा \cref{exo:b2:funcseq:1} में); अथवा
        किसी स्थानांतरण प्रमेय को प्रतिधनात्मक रूप में उद्धृत कीजिए ---
        \omterm{def:b2:metric:continuity}{संतत} फलनों की असंतत सीमा (\cref{ex:b2:funcseq:threshold}), या किसी खंड पर $\int f_n
        \not\to \int f$।
  \item श्रेणियों के लिए पहले \omterm{def:b2:funcseq:series}{सामान्य अभिसरण} आज़माइए ($\sum\sup\abs{u_n} < \infty$); और यदि वह
        वैश्विक रूप से विफल हो, तो उन उपखंडों पर जाँचिए जो मायने रखते
        हैं (\cref{ex:b2:funcseq:threemodes}); और यदि वह सर्वत्र विफल हो, तब भी एकांतर शेषपद
        परिबंध (\cref{exo:b2:funcseq:4}) अथवा खंडशः योग के द्वारा \omterm{def:b2:funcseq:def}{एकसमान अभिसरण} संभव है।
\end{enumerate}
\end{method}

\section{वाइरश्ट्रास सन्निकटन प्रमेय}

\begin{theorem}[वाइरश्ट्रास, बर्नस्टाइन के द्वारा]\label{thm:b2:funcseq:weierstrass}
प्रत्येक \omterm{def:b2:metric:continuity}{संतत} $f \colon \intcc{0}{1} \to \R$ बहुपदों की एकसमान सीमा है --- और स्पष्ट रूप से, अपने
\emph{बर्नस्टाइन बहुपदों}\index{बर्नस्टाइन बहुपद}\index{वाइरश्ट्रास
सन्निकटन प्रमेय} की:
\[
B_n(f)(x) = \sum_{k=0}^{n} f\Bigl(\frac kn\Bigr)\binom nk x^k
(1-x)^{n-k} .
\]
\end{theorem}

\begin{proof}
$x \in \intcc{0}{1}$ स्थिर कीजिए और $p_k(x) = \binom nk x^k(1 -
x)^{n-k}$ रखिए। तीन द्विपद सर्वसमिकाएँ, जो $y = 1 - x$ पर $(x + y)^n$ तथा
उसके दो $x$-अवकलजों का मूल्यांकन करने से मिलती हैं:
\[
\sum_k p_k = 1,
\qquad
\sum_k k\,p_k = nx,
\qquad
\sum_k k(k-1) p_k = n(n-1)x^2 .
\]
विस्तार से: $y = 1-x$ पर $(x+y)^n = \sum_k\binom nk x^ky^{n-k}$ पहली है; $x$ में अवकलन करने पर
\[
n(x+y)^{n-1} = \sum_k k\binom nk x^{k-1}y^{n-k} ,
\]
और फिर $x$ से गुणा करके $y = 1 - x$ रखने पर दूसरी मिलती है; और दो बार अवकलन
करके $x^2$ से गुणा करने पर तीसरी। $(k - nx)^2 = k(k-1) + k(1 - 2nx) +
n^2x^2$ को खोलकर तीनों को मिलाने पर
\[
\sum_k (k - nx)^2 p_k
= n(n-1)x^2 + nx(1 - 2nx) + n^2x^2
= nx(1 - x) \leq \frac n4 ,
\]
अर्थात् \emph{प्रसरण सर्वसमिका}।

अब $\sum p_k = 1$ का उपयोग करते हुए आकलन कीजिए:
\[
\abs{B_n(f)(x) - f(x)}
\leq \sum_{k} \Bigl| f\Bigl(\frac kn\Bigr) - f(x)\Bigr|\, p_k(x)
= \Sigma_{\text{निकट}} + \Sigma_{\text{दूर}} ,
\]
जिसमें विभाजन इस आधार पर है कि $\abs{\frac kn - x} \leq \delta$ है या नहीं। $\varepsilon > 0$ दिया हो तो $f$ की
एकसमान संततता (हाइने) ऐसा $\delta$ देती है कि $\Sigma_{\text{निकट}} \leq \varepsilon$। और दूर
वाले योग के लिए, $M = \norm f_\infty$ के साथ: प्रसरण सर्वसमिका तथा चेबिशेव की गिनती-युक्ति
से
\[
\Sigma_{\text{दूर}} \leq 2M \sum_{\abs{k - nx} > n\delta} p_k
\leq 2M\,\frac{\sum_k (k - nx)^2 p_k}{n^2\delta^2}
\leq \frac{2M}{4 n \delta^2} \cdot 1
= \frac{M}{2n\delta^2}
\xrightarrow[n\to\infty]{} 0 ,
\]
और यह $x$ में एकसमान है। अतः बड़े $n$ के लिए $\norm{B_n(f) - f}_\infty \leq \varepsilon +
\frac{M}{2n\delta^2} \leq 2\varepsilon$।
\end{proof}

\begin{remark}
आनत प्रतिस्थापन से प्रमेय किसी भी खंड $\intcc{a}{b}$ पर सत्य हो जाती है। वह $\R$ पर
विफल है ($\R$ पर बहुपदों की एकसमान सीमा बहुपद ही होती है: \cref{exo:b2:funcseq:8})। और
प्रायिकतात्मक पाठ --- $B_n(f)(x)$ किसी द्विपद औसत पर $f$ का प्रत्याशित मान है,
और प्रसरण परिबंध चेबिशेव की असमिका है --- \cref{ch:b2:genfun} में ईमानदारी से लिखा गया
है।
\end{remark}

\begin{remark}[यह कहाँ काम आता है]
वाइरश्ट्रास सन्निकटन क्लासिक विश्लेषण की सघनता-प्रमेय है: वह $C(\intcc ab)$ को
वियोज्य बना देती है, समाकल सर्वसमिकाओं को केवल बहुपदों पर जाँच लेने देती
है (आघूर्ण समस्याएँ), और फूरिये अध्याय में फेयेर कर्नेल से सिद्ध किए गए
त्रिकोणमितीय रूप के नीचे बैठी है। इस अध्याय की सप्ताहांत समस्या

मापांक से शासित अभिसरण दरें --- और फिर वह अलग करती है कि वास्तव में काम
किस चीज़ ने किया, कोरोवकिन की प्रमेय में: धनात्मकता और तीन परीक्षण फलन।
तृतीय वर्ष का खंड सघनता वाले कथन को किसी भी उपबीजगणित (स्टोन--वाइरश्ट्रास)
तथा \omterm{def:b2:metric:compact}{संहत} समष्टियों तक सामान्यीकृत कर देता है।
\end{remark}

\begin{example}[बहुभुजीय सन्निकटन, दर सहित]\label{ex:b2:funcseq:polygonal}
$\intcc{0}{1}$ पर $L$-लिप्शिट्स $f$ के लिए मान लीजिए $I_nf$ नोडों $\frac kn$ पर खंडशः आनत
अंतर्वेशी है। किसी कोष्ठ $\intcc{\frac kn}{\frac{k+1}n}$ पर $f(x)$ और $I_nf(x)$ दोनों उन चरम मानों के बीच
पड़ते हैं जो दिए गए दो नोड-मानों के साथ कोई $L$-लिप्शिट्स फलन ले सकता है,
अतः उस कोष्ठ के $x$ के लिए $x_k = \frac kn$ लिखने पर:
\[
\abs{I_nf(x) - f(x)}
\leq \abs{I_nf(x) - f(x_k)} + \abs{f(x_k) - f(x)}
\leq L\,\abs{x - x_k} + L\,\abs{x - x_k}
\leq \frac{2L}{n}
\]
(अंतर्वेशी स्वयं उस कोष्ठ पर $L$-लिप्शिट्स है: उसकी ढाल $f$ का
अंतर-भागफल है)। अतः $\norm{I_nf -
f}_\infty \leq \frac{2L}{n}$: अर्थात् \omterm{def:b2:metric:continuity}{लिप्शिट्स} फलनों का बहुभुजीय सन्निकटन
गति $\frac1n$ से अभिसरित होता है --- जो उसी वर्ग के लिए बर्नस्टाइन के $\frac{1}{\sqrt n}$ से
\emph{तेज़} है (सप्ताहांत समस्या, भाग II)। समापन दृष्टि: बहुभुज अंतर्वेशन
करता है पर चिकना नहीं है, बर्नस्टाइन चिकना है पर सुस्त; सन्निकटक की
नियमितता और गति के बीच कोई मुफ़्त भोजन नहीं है --- और इस समझौते को
सप्ताहांत समस्या के संतृप्ति-परिणाम परिशुद्ध कर देते हैं।
\end{example}

\begin{remark}[इसी खंड के भीतर के परिप्रेक्ष्य]
यहाँ से आगे \omterm{def:b2:funcseq:def}{एकसमान अभिसरण} इस पुस्तक का श्रमिक है। घात-श्रेणी वाला अध्याय
पूरा-का-पूरा \omterm{def:b2:metric:compact}{संहत} उपचक्रिकाओं पर \omterm{def:b2:funcseq:series}{सामान्य अभिसरण} पर चलता है --- वहाँ की हर
पद-दर-पद प्रमेय इस अध्याय की स्थानांतरण प्रमेयों की विशेष स्थिति है।
फूरिये अध्याय एक मंज़िल ऊपर बसता है: उसके आंशिक योग $S_N$ ठीक वहीं विफल होते
हैं जहाँ यह अध्याय चेतावनी देता है (उछालों पर \omterm{def:b2:funcseq:def}{बिंदुवार} पर एकसमान नहीं), और
उसके फेयेर माध्य उसी $3\varepsilon$ यांत्रिकी से सफल होते हैं जिसने \cref{thm:b2:funcseq:continuity} सिद्ध की थी।
अवकल समीकरणों वाला अध्याय $\eu^{tA}$ को किसी सामान्य अभिसारी श्रेणी से परिभाषित
करके पद-दर-पद अवकलन करता है --- अर्थात् आव्यूह प्रविष्टियों पर अक्षरशः
लगाई गई \cref{thm:b2:funcseq:seriestransfer}। आगे पुस्तक में जब भी संदेह हो कि ``हम ऐसा क्यों कर सकते
हैं'', तो उत्तर प्रायः इसी अध्याय की कोई प्रमेय होती है।
\end{remark}

\begin{exercise}[$\star\star\star$]\label{exo:b2:funcseq:9}

\[
W(x) = \sum_{n=0}^{\infty} \Bigl(\frac{3}{4}\Bigr)^{\!n}
\varphi(4^n x) .
\]
सिद्ध कीजिए: (क) $W$ $\R$ पर \omterm{def:b2:metric:continuity}{संतत} है (\omterm{def:b2:funcseq:series}{सामान्य अभिसरण}); (ख) प्रत्येक $x$

\infty
\]

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:funcseq:12}
(श्रेणी से एक वोल्टेरा समाकल समीकरण) $f \in
C(\intcc{0}{1})$ के लिए $Tf(x) = \int_0^x f(t)\,\dd t$ परिभाषित कीजिए।
\begin{enumerate}
  \item आगमन से दिखाइए कि $n \geq 1$ के लिए:
        \[
        T^n f(x) = \int_0^x \frac{(x -
        t)^{n-1}}{(n-1)!}\,f(t)\,\dd t,
        \qquad
        \norm{T^n f}_\infty \leq \frac{\norm f_\infty}{n!} .
        \]
  \item इससे निष्कर्ष निकालिए कि $S = \sum_{n\geq0} T^n f$ $\intcc{0}{1}$ पर \omterm{def:b2:funcseq:series}{सामान्य रूप से} अभिसरित होती
        है और समाकल समीकरण $S = f
        + TS$ को हल करती है।
  \item सत्यापित कीजिए कि $S(x) = f(x) + \int_0^x \eu^{x-t}f(t)\,\dd

        अद्वितीयता सिद्ध कीजिए \emph{(यदि $S = TS$ तो $\norm{S}_\infty \leq \norm{T^nS}_\infty \to 0$)}: फिर योग का बंद

\end{enumerate}
\end{exercise}

\section{समस्या: सन्निकटन की दरें और कोरोवकिन की प्रमेय}

\begin{problem}\label{pb:b2:funcseq:1}
\cref{thm:b2:funcseq:weierstrass} की बर्नस्टाइन-उपपत्ति में दो ख़ज़ाने छिपे हैं। पहला, वह
\emph{परिमाणात्मक} है: $B_nf
\to f$ कितनी तेज़ी से होता है, यह $f$ के संततता
मापांक से शासित होता है, और तीखी दर $\abs{x - \frac12}$ पर प्राप्त होती है। दूसरा, वह
\emph{संरचनात्मक} है: सारी बात इतनी ही थी कि $B_n$ एक धनात्मक रैखिक संकारक
है जो $1$, $x$, $x^2$ पर ठीक व्यवहार करता है --- और उसी अवलोकन को अलग कर
देने पर वह \emph{कोरोवकिन की प्रमेय}\index{कोरोवकिन की प्रमेय} है। यह
समस्या दोनों सिद्ध करती है और वोरोनोव्स्काया की यथार्थ अनंतस्पर्शिता पर
समाप्त होती है। आगे सर्वत्र $f \in C(\intcc{0}{1})$, $M = \norm f_\infty$, $p_k(x) = \binom
nk x^k(1-x)^{n-k}$, और $e_j$ का अर्थ $x \mapsto x^j$ है।

\medskip
\noindent\textbf{भाग I --- बर्नस्टाइन संकारक।}
\begin{enumerate}
  \item दिखाइए कि $B_n$ रैखिक और \emph{धनात्मक} ($f \geq 0
        \Rightarrow B_nf \geq 0$) है, अतः एकदिष्ट
        ($f \leq g
        \Rightarrow B_nf \leq B_ng$), जहाँ $\norm{B_nf}_\infty \leq \norm f_\infty$; और यह कि $B_nf$ दोनों अंत्यबिंदुओं पर $f$ का
        अंतर्वेशन करता है।
  \item सर्वसमिकाएँ $B_n e_0 = e_0$, $B_n e_1 =
        e_1$ और $B_n e_2 = e_2 + \frac{e_1 - e_2}{n}$ पुनः निकालिए \emph{($(x + y)^n$ का दो बार
        अवकलन कीजिए और $y = 1 -
        x$ रखिए)}।
  \item प्रसरण सर्वसमिका $\sum_k \bigl(\frac kn -
        x\bigr)^2 p_k(x) = \frac{x(1-x)}{n}$ निष्कर्ष रूप में निकालिए और
        कोशी--श्वार्ज़ से प्रथम-आघूर्ण परिबंध:
        \[
        \sum_{k=0}^{n}\Bigl|\frac kn - x\Bigr|\,p_k(x)
        \leq \sqrt{\frac{x(1-x)}{n}} \leq \frac{1}{2\sqrt n} .
        \]
  \item दिखाइए कि यदि $f$ उत्तल हो तो $\intcc{0}{1}$ पर $B_nf \geq f$ \emph{(भार $p_k(x)$ के
        लिए परिमित जेनसन असमिका)}।
  \item (चेबिशेव का गिनती-परिबंध, पुनः कहा गया) $\delta > 0$ के लिए दिखाइए
        \[
        \sum_{\abs{k/n - x} > \delta} p_k(x)
        \leq \frac{x(1-x)}{n\delta^2}
        \leq \frac{1}{4n\delta^2} ,
        \]
        और प्रायिकतात्मक पाठ दीजिए: $B_nf(x)$ $f$ का औसत किसी ऐसे द्विपद
        प्रतिदर्श-माध्य पर लेता है जो $x$ पर संकेंद्रित होता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- दरें: संततता मापांक।} $\delta > 0$ के लिए $\omega(\delta) = \sup\{\abs{f(s) - f(t)}
: s, t \in \intcc{0}{1},\ \abs{s - t} \leq \delta\}$ रखिए।
\begin{enumerate}[resume]
  \item दिखाइए: $\omega$ परिमित है, अवर्धमान नहीं है, $\delta \to 0^+$ होने पर $\omega(\delta) \to 0$
        (हाइने), उपयोज्य है ($\omega(\delta_1 + \delta_2) \leq
        \omega(\delta_1) + \omega(\delta_2)$), और सभी $\lambda > 0$ के लिए $\omega(\lambda\delta) \leq (1 +
        \lambda)\,\omega(\delta)$।
  \item प्रत्येक $\delta > 0$ के लिए मुख्य आकलन सिद्ध कीजिए:
        \[
        \abs{B_nf(x) - f(x)}
        \leq \sum_k \omega\Bigl(\Bigl|\frac kn -
        x\Bigr|\Bigr)p_k(x)
        \leq \Bigl(1 + \frac1\delta\sum_k\Bigl|\frac kn -
        x\Bigr|p_k(x)\Bigr)\,\omega(\delta) .
        \]
  \item $\delta = n^{-1/2}$ चुनिए और \emph{परिमाणात्मक वाइरश्ट्रास प्रमेय} निष्कर्ष
        रूप में निकालिए:
        \[
        \norm{B_nf - f}_\infty \leq
        \frac32\,\omega\Bigl(\frac{1}{\sqrt n}\Bigr)
        \xrightarrow[n\to\infty]{} 0 .
        \]
  \item दरें निष्कर्ष रूप में निकालिए: $L$-लिप्शिट्स $f$ के लिए $\norm{B_nf - f}_\infty \leq
        \frac{3L}{2\sqrt n}$,
        और $\alpha$-हॉल्डर $f$ के लिए $\leq
        \frac32 C n^{-\alpha/2}$ ($\abs{f(s) - f(t)} \leq C\abs{s-t}^\alpha$)।
  \item (तीखा उदाहरण --- एक द्विपद सर्वसमिका) $m
        \geq 1$ के लिए सिद्ध कीजिए
        \[
        \sum_{k=m+1}^{2m} (k - m)\binom{2m}{k}
        = \frac{m}{2}\binom{2m}{m},
        \qquad\text{अतः}\qquad
        \sum_{k=0}^{2m}\abs{k - m}\binom{2m}{k}
        = m\binom{2m}{m}
        \]
        \emph{($k\binom{2m}k = 2m\binom{2m-1}{k-1}$ तथा द्विपद पंक्ति की सममिति का उपयोग कीजिए, जो $\sum_{j=m}^{2m-1}\binom{2m-1}{j} = 2^{2m-2}$
        देती है)}।
  \item $f(t) = \abs{t - \frac12}$ के लिए यथार्थ मान और उसकी अनंतस्पर्शिता निष्कर्ष रूप में
        निकालिए (केंद्रीय द्विपद, \cref{ex:b2:comparison:centralbinomial}):
        \[
        B_{2m}f\Bigl(\frac12\Bigr) - f\Bigl(\frac12\Bigr)
        = \frac{\binom{2m}{m}}{2\cdot4^{m}}
        \;\sim\; \frac{1}{2\sqrt{\pi m}} :
        \]
        अर्थात् प्रश्न 8 की दर $\omega(n^{-1/2})$ (किसी अचर तक) प्राप्त हो जाती है ---
        केवल \omterm{def:b2:metric:continuity}{संतत} $f$ के लिए बर्नस्टाइन का $n^{-1/2}$ ईमानदार है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- कोरोवकिन की प्रमेय।} मान लीजिए $(L_n)$ $C(\intcc{0}{1})$ से
उसी में जाने वाले \emph{धनात्मक रैखिक} संकारकों का ऐसा अनुक्रम है कि $j = 0, 1, 2$
के लिए $L_ne_j \to e_j$ \omterm{def:b2:funcseq:def}{एकसमान रूप से}।
\begin{enumerate}[resume]
  \item दिखाइए कि कोई धनात्मक रैखिक $L$ एकदिष्ट होता है और \omterm{def:b2:funcseq:def}{बिंदुवार} $\abs{Lf} \leq L\abs f$
        पूरा करता है।
  \item दिखाइए: प्रत्येक $\varepsilon > 0$ के लिए ऐसा $\delta > 0$ है कि \emph{सभी} $s, x \in \intcc{0}{1}$ के लिए:
        \[
        \abs{f(s) - f(x)} \leq \varepsilon +
        \frac{2M}{\delta^2}(s - x)^2
        \]
        \emph{($\abs{s - x} \leq \delta$ को हाइने से और $\abs{s-x} > \delta$ को कच्चे परिबंध $2M$ से निपटाइए)}।
  \item $x$ स्थिर कीजिए, प्रश्न 13 की असमिका पर चर $s$ में $L_n$ लगाइए,
        और निकालिए
        \[
        \abs{L_nf(x) - f(x)\,L_ne_0(x)}
        \leq \varepsilon\,L_ne_0(x) + \frac{2M}{\delta^2}
        \bigl(L_ne_2(x) - 2x\,L_ne_1(x) + x^2 L_ne_0(x)\bigr).
        \]
  \item दिखाइए कि $\sup_x \bigl(L_ne_2(x) - 2x\,L_ne_1(x) + x^2
        L_ne_0(x)\bigr) \to 0$, फिर \emph{कोरोवकिन की प्रमेय} जोड़िए:
        \emph{प्रत्येक} $f \in C(\intcc{0}{1})$ के लिए $L_nf \to f$ \omterm{def:b2:funcseq:def}{एकसमान रूप से}।
  \item जाँचिए कि $(B_n)$ कोरोवकिन की परिकल्पनाएँ पूरा करता है: अर्थात्
        तीसरी बार वाइरश्ट्रास, केवल तीन एकपदियों से।
  \item मान लीजिए $I_n$ नोडों $\frac kn$ पर खंडशः आनत अंतर्वेशन संकारक है।
        दिखाइए कि $I_n$ धनात्मक रैखिक है, $I_ne_0 = e_0$, $I_ne_1 = e_1$, और
        $\norm{I_ne_2 - e_2}_\infty = \frac{1}{4n^2}$
        \emph{(हर कोष्ठ पर $t^2$ के आनत अंतर्वेशन की त्रुटि $(t - a)(b - t)$ है)}।
        कोरोवकिन से निष्कर्ष निकालिए: प्रत्येक \omterm{def:b2:metric:continuity}{संतत} $f$ के लिए बहुभुजीय
        अंतर्वेशी \omterm{def:b2:funcseq:def}{एकसमान रूप से} अभिसरित होते हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- लाभ: सघनता, आघूर्ण, अवकलज।}
\begin{enumerate}[resume]
  \item दिखाइए कि \emph{परिमेय} गुणांकों वाले बहुपद $\bigl(C(\intcc{0}{1}),
        \norm\cdot_\infty\bigr)$ में सघन हैं:
        अर्थात् यह \omterm{def:b2:nvs:banach}{बानाख समष्टि} वियोज्य है।
  \item (आघूर्ण फलन को निर्धारित कर देते हैं) मान लीजिए $f \in
        C(\intcc{0}{1})$ ऐसा है कि
        प्रत्येक $n \in \N$ के लिए $\int_0^1 f(t)\,t^n \dd t = 0$। दिखाइए कि हर बहुपद के लिए $\int_0^1 f P = 0$, फिर $\int_0^1 f^2 = 0$,
        फिर $f = 0$।
  \item अवकलज सर्वसमिका सिद्ध कीजिए
        \[
        (B_nf)'(x) = n\sum_{k=0}^{n-1}\Bigl(
        f\Bigl(\frac{k+1}{n}\Bigr) -
        f\Bigl(\frac kn\Bigr)\Bigr)\,
        \binom{n-1}{k}x^k(1-x)^{n-1-k}
        \]
        \emph{($p_k$ का अवकलन कीजिए और पुनः सूचीबद्ध कीजिए --- एक \omterm{thm:b2:series:abel}{आबेल
        योग})}।
  \item मान लीजिए $f$ $C^1$ है। हर वृद्धि में मध्यमान प्रमेय लगाकर और
        $B_{n-1}(f')$ से तुलना करके दिखाइए कि $\intcc{0}{1}$ पर \omterm{def:b2:funcseq:def}{एकसमान रूप से} $(B_nf)' \to f'$। इससे
        निष्कर्ष निकालिए: $f \in C^1$ के लिए ऐसे बहुपद हैं जो अपने अवकलजों
        \emph{सहित} $f$ पर अभिसरित होते हैं।
  \item मान लीजिए $f$ $C^2$ है। $x$ पर टेलर--लाग्रांज से दिखाइए
        \[
        \abs{B_nf(x) - f(x)} \leq
        \frac{\norm{f''}_\infty}{2}\cdot\frac{x(1-x)}{n}
        \leq \frac{\norm{f''}_\infty}{8n} :
        \]
        अर्थात् चिकनाई दर को $n^{-1/2}$ से बढ़ाकर $n^{-1}$ कर देती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- संतृप्ति: वोरोनोव्स्काया की प्रमेय।}
\begin{enumerate}[resume]
  \item चौथे आघूर्ण की सर्वसमिका सिद्ध कीजिए
        \[
        \sum_k (k - nx)^4 p_k(x)
        = nx(1-x)\bigl(1 + 3(n-2)x(1-x)\bigr) \leq n^2
        \quad (n \geq 1)
        \]
        \emph{($k^4$ का अवरोही क्रमगुणितों $k(k-1)\cdots$ में प्रसार कीजिए और प्रश्न
        2 की अवकलन-युक्ति दो बार और लगाइए)}।
  \item (वोरोनोव्स्काया) मान लीजिए $f$ $C^2$ है और $x \in
        \intcc{0}{1}$। $\eta$ परिबद्ध तथा
        $t \to x$ होने पर $\eta(t) \to 0$ के साथ $f(t) = f(x) + f'(x)(t-x) +
        \frac{f''(x)}2(t-x)^2 + \eta(t)(t-x)^2$ लिखकर सिद्ध कीजिए
        \[
        n\bigl(B_nf(x) - f(x)\bigr)
        \xrightarrow[n\to\infty]{}
        \frac{x(1-x)}{2}\,f''(x)
        \]
        \emph{($\eta$ वाले योग को $\abs{t - x} \leq
        \delta$ पर बाँटिए; दूर वाले भाग को प्रश्न
        23 से नियंत्रित कीजिए)}। अतः प्रश्न 22 की त्रुटि कोटि \emph{और}
        अचर दोनों में यथार्थ है: $B_n$ $\frac1n$ पर \emph{संतृप्त} हो जाता है,
        चाहे $f$ कितना ही चिकना क्यों न हो --- इसकी तुलना \cref{exo:b2:funcseq:7} से
        कीजिए।
  \item संश्लेषण। एक-एक वाक्य में: (क) अकेली धनात्मकता ने क्या दिलाया
        (भाग I और III); (ख) $\intcc{0}{1}$ की संहतता हर भाग में कहाँ आई; (ग)
        कोरोवकिन की प्रमेय में तीन परीक्षण फलन क्यों पर्याप्त हैं; (घ)
        बर्नस्टाइन कौन-सा समझौता करता है (उबड़-खाबड़ $f$ के लिए मज़बूत
        $n^{-1/2}$, पर चिकने $f$ के लिए $\frac1n$ की छत), और इस पुस्तक का कौन-सा
        अध्याय त्रिकोणमितीय बहुपदों के साथ वही खेल खेलेगा।
\end{enumerate}
\end{problem}
