\chapter{फलनों की तुलना}\label{ch:b2:comparison}

अनंतस्पर्शी विश्लेषण --- अर्थात् किसी जटिल राशि के स्थान पर एक सरल राशि
और नियंत्रित त्रुटि रख देने की कला --- प्रथम वर्ष के खंड में टेलर
प्रसारों के साथ आरंभ हुआ था। यह अध्याय उसे अपने आप में एक अनुशासन बना
देता है: सामान्य पैमानों पर प्रसार, श्रेणी--समाकल तुलना उसकी पूरी
अनंतस्पर्शी शक्ति के साथ, स्टर्लिंग सूत्र (पूरी तरह सिद्ध), और अंतर्निहित
रूप से परिभाषित अनुक्रमों का क्रमबद्ध अध्ययन। ये तकनीकें अनंतस्पर्शी
विश्लेषण की रोज़ की रोटी हैं, और आगे का हर वह अध्याय जो किसी भी चीज़ का
आकलन करता है --- श्रेणियाँ, समाकल, प्रायिकताएँ --- इसी थाली से खाता है।

\section{तुलना संबंध और पैमाने}

\begin{definition}\label{def:b2:comparison:landau}
किसी बिंदु $a$ के पास ($a \in \R$ अथवा $\pm\infty$), फलनों के लिए (अथवा अनुक्रमों के
लिए, $n \to \infty$ के साथ): $f = o(g)$, $f = O(g)$, $f \sim g$ --- प्रथम वर्ष के खंड की तरह। $a$ पर
\emph{तुलना-पैमाना}\index{तुलना-पैमाना} धनात्मक फलनों का ऐसा कुल है जो
जोड़े-जोड़े में तुलनीय हो और $o(\cdot)$ के द्वारा पूर्णतः क्रमित हो --- और $+\infty$
पर मानक पैमाना
\[
x^{\alpha} (\ln x)^{\beta}
\qquad (\alpha, \beta \in \R),
\]
है, जो $(\alpha, \beta)$ में शब्दकोश-क्रम में क्रमित है और आवश्यकता पड़ने पर
घातांकी $\eu^{\gamma x}$ से परिष्कृत किया जाता है।
\end{definition}

\begin{definition}[अनंतस्पर्शी प्रसार]\label{def:b2:comparison:expansion}
$f$ का \emph{अनंतस्पर्शी प्रसार}\index{अनंतस्पर्शी प्रसार}
\[
f = c_1 \varphi_1 + c_2\varphi_2 + \dots + c_k \varphi_k +
o(\varphi_k)
\qquad (\varphi_{i+1} = o(\varphi_i) \text{ पैमाने में})
\]
तब होता है जब क्रमिक शेषपद प्रदर्शित आकलन पूरे करें। तब गुणांक अद्वितीय
होते हैं: $c_1 = \lim f/\varphi_1$, और आगमन से $c_{i+1} = \lim\,(f - \sum_{j \leq i}
c_j\varphi_j)/\varphi_{i+1}$।
\end{definition}

\begin{example}
टेलर प्रसार $a$ पर पैमाने $(x -
a)^k$ के अनुदिश \omterm{def:b2:comparison:expansion}{अनंतस्पर्शी प्रसार} ही हैं। परंतु
यह धारणा यथार्थतः व्यापक है: $+\infty$ पर
\[
\frac{1}{x - \ln x}
= \frac1x \cdot \frac{1}{1 - \frac{\ln x}{x}}
= \frac1x + \frac{\ln x}{x^2} + o\Bigl(\frac{\ln x}{x^2}\Bigr),
\]
यह मिश्रित पैमाने पर प्रसार है --- यहाँ कोई टेलर प्रमेय लागू नहीं होती,
केवल गुणोत्तर प्रसार और $o$ का कलन।
\end{example}

\begin{example}[मानक पैमाना सचमुच क्रमित है]\label{ex:b2:comparison:scaleorder}
\cref{def:b2:comparison:landau} के शब्दकोश-क्रम वाले दावे को हर स्थिति में एक पंक्ति की उपपत्ति
चाहिए। $+\infty$ पर $x^{\alpha}(\ln x)^{\beta}$ और $x^{\alpha'}(\ln x)^{\beta'}$ की तुलना कीजिए। यदि $\alpha < \alpha'$: तो अनुपात $x^{\alpha - \alpha'}(\ln x)^{\beta - \beta'} \to 0$ है,
क्योंकि $x$ की कोई ऋणात्मक घात $\ln x$ की किसी भी घात को कुचल देती है ($x =
\eu^t$
रखिए: प्रथम वर्ष के खंड की घातांकी-बनाम-बहुपद सीमा से $\eu^{(\alpha - \alpha')t}\,t^{\beta - \beta'} \to 0$)। और यदि $\alpha = \alpha'$
तथा $\beta < \beta'$: तो अनुपात सीधे $(\ln x)^{\beta - \beta'} \to 0$ है। अतः युग्म $(\alpha, \beta)$, शब्दकोश-क्रम में
क्रमित, पैमाने को $o(\cdot)$ के द्वारा क्रमित कर देते हैं --- और प्रतिस्थापन
$x = \eu^t$ मिश्रित घात-लघुगणक तुलनाओं के लिए हर नाप में फिट होने वाली युक्ति
है।
\end{example}

\begin{example}[एक चिड़ियाघर की क्रम-सूची]\label{ex:b2:comparison:ranking}
पैमाने \emph{क्रमित} होने ही चाहिए; और यह रहा मानक अभ्यास। $+\infty$ पर $n^{10}$,
$\eu^{\sqrt{\ln n}\,\cdot\,\sqrt n}$, $2^n$ और $n^{\ln n}$ की तुलना लघुगणक लेकर कीजिए:
\[
10\ln n
\;\ll\; (\ln n)^2
\;\ll\; \sqrt{n\ln n}
\;\ll\; n\ln 2 ,
\]
जहाँ $a_n \ll b_n$ का अर्थ $a_n = o(b_n)$ है; और दूसरी प्रविष्टि $\ln(n^{\ln n})$ है। घातांकी इन यथार्थ
अंतरों को बनाए रखते हैं (यदि $\ln u_n - \ln v_n \to -\infty$ तो $u_n/v_n \to 0$), अतः
\[
n^{10} = o\bigl(n^{\ln n}\bigr),
\qquad
n^{\ln n} = o\bigl(\eu^{\sqrt{n\ln n}}\bigr),
\qquad
\eu^{\sqrt{n\ln n}} = o(2^n) .
\]
सार दो बार: सदा लघुगणकों के द्वारा तुलना कीजिए (लघुगणकों के अंतर से, न
कि लघुगणकों के अनुपात से), और $\ln u_n \sim \ln v_n$ से $u_n \sim v_n$ कभी निष्कर्ष मत निकालिए ---
युग्म $n^{10}$ तथा $n^{\ln n}$ का $\ln$-अनुपात $\infty$ की ओर जाता है, परंतु $2^n$ और
$4^n$ का $\ln$-अनुपात ठीक $2$ है और फिर भी वे अत्यंत असमतुल्य हैं।
\end{example}

\section{श्रेणी--समाकल तुलना, अनंतस्पर्शी रूप में}

\begin{theorem}\label{thm:b2:comparison:seriesintegral}
मान लीजिए $f$ $\intco{1}{+\infty}$ पर \omterm{def:b2:metric:continuity}{संतत}, धनात्मक और \emph{ह्रासमान} है।
\begin{enumerate}
  \item यदि $\int_1^{\infty} f$ अभिसरित हो, तो शेषपद पूरा करते हैं
        \[
        \int_{n+1}^{\infty} f \;\leq\; \sum_{k > n} f(k) \;\leq\;
        \int_{n}^{\infty} f .
        \]
  \item यदि $\int_1^\infty f$ अपसरित हो, तो आंशिक योग किसी अचर $C$ के लिए $\sum_{k=1}^{n} f(k) = \int_1^n f + C + o(1)$ पूरा
        करते हैं: अर्थात् अंतर $\sum_{k \leq n} f(k) -
        \int_1^n f$ \emph{अभिसरित} होता है।
\end{enumerate}
\end{theorem}

\begin{proof}
कोष्ठकन $f(k+1) \leq \int_k^{k+1} f \leq f(k)$ (ह्रास से) प्रथम वर्ष की युक्ति थी; $k \geq n+1$ अथवा $k \geq n$ पर योग
लेने से (1) मिल जाता है। (2) के लिए $u_k = f(k) - \int_k^{k+1} f$ रखिए: कोष्ठकन से $0 \leq u_k \leq f(k) - f(k+1)$, अतः $\sum u_k$
के आंशिक योग क्रमिक निरसन $f(1) - f(n+1) \leq
f(1)$ से परिबद्ध हैं: इसलिए श्रेणी अभिसरित होती
है। इसके अतिरिक्त अनुक्रम $\bigl(\int_n^{n+1}
f\bigr)_n$ अवर्धमान ($f$ घटता है) और अऋणात्मक है,
अतः अभिसारी। और
\[
\sum_{k=1}^{n} f(k) - \int_1^n f
= \sum_{k=1}^{n} u_k + \int_n^{n+1} f ,
\]
लिखने पर $n \to \infty$ होने पर दाहिना पक्ष अभिसरित होता है: अर्थात् अंतर किसी अचर
$C$ पर अभिसरित होता है, जो कथन (2) है।
\end{proof}

\begin{example}[हरात्मक प्रसार]\label{ex:b2:comparison:harmonic}
$f(t) = \frac1t$ के लिए: $H_n = \ln n + \gamma + o(1)$, जिससे ऑयलर अचर (प्रथम वर्ष का खंड) एक स्वच्छतर उपपत्ति
के साथ पुनः मिल जाता है। एक कोटि और आगे बढ़ने पर (\cref{exo:b2:comparison:3}):
\[
H_n = \ln n + \gamma + \frac{1}{2n} + o\Bigl(\frac1n\Bigr).
\]
$n = 10$ पर संख्याएँ लाभ को दिखा देती हैं: $H_{10} =
2.928968\dots$ और $\ln 10 = 2.302585\dots$, अतः $\gamma$ का कच्चा आकलन
$H_{10} - \ln 10 = 0.626383$ है, जो $0.049$ की चूक है; और सुधार $\frac1{20}$ घटाने पर $0.576383$ मिलता है, जो $\gamma = 0.577216$
से केवल $8.3\cdot10^{-4}$ की चूक पर है --- और वह स्वयं प्रसार का अगला पद $\frac{1}{12\cdot100}$ है,
जैसा सप्ताहांत समस्या (प्रश्न 8) सिद्ध करती है।
\end{example}

\begin{example}[स्टर्लिंग के बिना एक कच्चा $\ln(n!)$]\label{ex:b2:comparison:lnfactorial}
अकेला कोष्ठकन यंत्र ही $\ln(n!)$ का पता लगा देता है। चूँकि $\ln$ बढ़ता है,
\[
\int_{k-1}^{k}\ln t\,\dd t \;\leq\; \ln k \;\leq\;
\int_{k}^{k+1}\ln t\,\dd t ,
\]
और $k = 2, \dots, n$ पर योग लेने से ($\int_1^n\ln = n\ln n
- n + 1$ के साथ):
\[
n\ln n - n + 1 \;\leq\; \ln(n!) \;\leq\; (n+1)\ln(n+1) - n .
\]
दोनों बाड़ें $n\ln n - n + O(\ln n)$ हैं: अतः $\ln(n!) = n\ln
n - n + O(\ln n)$, और विशेष रूप से $\ln(n!) \sim n\ln n$। स्टर्लिंग जो जोड़ता
है वे अगली दो सीढ़ियाँ हैं --- $\frac12\ln n$ और अचर $\ln\sqrt{2\pi}$ --- और उनकी क़ीमत \cref{thm:b2:comparison:stirling}
का सूक्ष्मतर क्रमिक निरसन है। कौन-सा औज़ार कितनी परिशुद्धता ख़रीदता है,
यह जानना अनंतस्पर्शी कारीगरी का आधा हिस्सा है।
\end{example}

\begin{example}[निरसन प्रसार माँगता है]\label{ex:b2:comparison:cancellation}
$\sqrt{n^2 + n} - n$ की सीमा परिकलित कीजिए। दोनों पद $\sim n$ हैं, और ``$\sim n - n$'' निरर्थक है:
समतुल्यों को घटाया नहीं जा सकता। इसके बदले प्रसार कीजिए:
\[
\sqrt{n^2 + n} - n
= n\Bigl(\sqrt{1 + \tfrac1n} - 1\Bigr)
= n\Bigl(\frac{1}{2n} - \frac{1}{8n^2} +
O\Bigl(\frac1{n^3}\Bigr)\Bigr)
= \frac12 - \frac{1}{8n} + O\Bigl(\frac1{n^2}\Bigr) :
\]
सीमा $\frac12$, और उपहार में पहुँचने की गति $\frac1{8n}$ भी। इस कार्यविधि का कोई नाम
होना चाहिए: दो बड़ी समतुल्य राशियों का अंतर पूरा-का-पूरा उनके \emph{अगले}
पदों में रहता है, अतः उस पहली कोटि तक प्रसार करना पड़ता है जहाँ दोनों
पक्ष भिन्न होते हैं --- और शेषपद साथ ले चलना पड़ता है ताकि प्रमाणित हो
सके कि उस कोटि पर और कुछ नहीं बचा।
\end{example}

\begin{example}[एक अपसारी तुलना, कर के दिखाई गई]\label{ex:b2:comparison:loglog}
$\intco{2}{+\infty}$ पर $f(t) = \frac{1}{t\ln t}$ के लिए (जो \omterm{def:b2:metric:continuity}{संतत}, धनात्मक और ह्रासमान है): $\int_2^x f = \ln\ln x -
\ln\ln 2 \to \infty$, अतः \cref{thm:b2:comparison:seriesintegral}~(2)
से किसी अचर $C$ के लिए
\[
\sum_{k=2}^{n}\frac{1}{k\ln k} = \ln\ln n + C + o(1)
\]
दो सीख। पहली, अपसरण वास्तविक तो है पर हिमनद जितना धीमा: आंशिक योग $4$
से पहली बार $n
\approx \eu^{\eu^{4 - C}}$ के आसपास आगे निकलता है, जो खगोलीय रूप से बड़ा है।
दूसरी, \emph{आकार} $\ln\ln n$ किसी प्रतिअवकलज ने दिया, अनुमान से नहीं आया:
एकदिष्ट पदों के लिए समाकल ही योग करने का विहित यंत्र है, और अचर $C$ ---
ऑयलर के $\gamma$ की तरह --- आरंभिक पदों की स्मृति है।
\end{example}

\section{स्टर्लिंग सूत्र}

\begin{lemma}[वालिस समाकल, पुनरावलोकन]\label{lem:b2:comparison:wallis}
मान लीजिए $W_n = \int_0^{\pi/2} \sin^n t\,\dd t$। तब $n \geq 1$ के लिए $nW_nW_{n-1} =
\frac\pi2$, $(W_n)$ घटता है, और $W_n \sim
\sqrt{\dfrac{\pi}{2n}}$।
\end{lemma}

\begin{proof}
खंडशः समाकलन से $nW_n = (n-1)W_{n-2}$ मिलता है ($n \geq 2$), अतः $nW_nW_{n-1}$ $n$ में अचर है और $1 \cdot W_1 W_0 =
\frac\pi2$ के
बराबर। ह्रास: $\intcc{0}{\frac\pi2}$ पर $\sin^{n+1} \leq \sin^n$। निचोड़, विस्तार से: एकदिष्टता से $W_{n+1} \leq W_n \leq W_{n-1}$ मिलता है,
और $W_{n-1}
> 0$ से भाग देने पर
\[
\frac{n}{n+1} = \frac{W_{n+1}}{W_{n-1}} \leq
\frac{W_n}{W_{n-1}} \leq 1 ,
\]
जिसमें बाईं ओर की सर्वसमिका सूचकांक $n + 1$ पर पुनरावृत्ति से आती है। दोनों
परिबंध $1$ की ओर जाते हैं: $W_n \sim W_{n-1}$, जिससे
\[
nW_n^2 \sim nW_nW_{n-1} = \frac\pi2
\qquad\Longrightarrow\qquad
W_n \sim \sqrt{\frac{\pi}{2n}} .
\]
\end{proof}

\begin{example}[पहले कुछ वालिस समाकल]\label{ex:b2:comparison:wallisvalues}
$W_0 = \frac\pi2$, $W_1 = 1$ और पुनरावृत्ति $nW_n =
(n-1)W_{n-2}$ से:
\[
W_2 = \frac\pi4, \qquad
W_3 = \frac23, \qquad
W_4 = \frac{3\pi}{16}, \qquad
W_5 = \frac{8}{15}, \qquad
W_6 = \frac{5\pi}{32}.
\]
सम सूचकांक $\pi$ लिए चलते हैं, विषम परिमेय होते हैं --- बंद रूपों के दो
गुँथे हुए गुणनफल। संख्यात्मक रूप से $W_6
\approx 0.4909$ बनाम अनंतस्पर्शी $\sqrt{\pi/12} \approx 0.5116$: $n = 6$ पर
समतुल्य पहले से ही $5\%$ के भीतर है, और गुणनफल सर्वसमिका हर $n$ पर यथार्थ
है: $6\,W_6W_5 = 6\cdot\frac{5\pi}{32}\cdot\frac8{15} =
\frac\pi2$। ऐसी छोटी तालिकाएँ बीजगणित की चूक को अनंतस्पर्शी तर्क में फैलने
से पहले पकड़ने का सबसे सस्ता उपाय हैं।
\end{example}

\begin{theorem}[स्टर्लिंग]\label{thm:b2:comparison:stirling}
\[
n! \;\sim\; \sqrt{2\pi n}\, \Bigl(\frac{n}{\eu}\Bigr)^{\!n}
.\index{स्टर्लिंग सूत्र}
\]
\end{theorem}

\begin{proof}
\emph{चरण 1: किसी अचर $C >
0$ के लिए $n! \sim C \sqrt n\, (n/\eu)^n$।} रखिए
\[
d_n = \ln(n!) - \Bigl(n + \frac12\Bigr)\ln n + n .
\]
तब $\ln(1 + \frac1n)$ के टेलर प्रसार से
\[
d_n - d_{n+1}
= \Bigl(n + \frac12\Bigr) \ln\frac{n+1}{n} - 1
= \Bigl(n + \frac12\Bigr)\Bigl(\frac1n - \frac{1}{2n^2} +
\frac{1}{3n^3} + o\bigl(n^{-3}\bigr)\Bigr) - 1
= \frac{1}{12n^2} + o\Bigl(\frac{1}{n^2}\Bigr),
\]
अतः श्रेणी $\sum (d_n
- d_{n+1})$ निरपेक्ष रूप से अभिसरित होती है ($\sum
n^{-2}$ से तुलना), इसलिए
$(d_n)$ अभिसरित होता है, मान लीजिए $d$ पर; और घातांकी लेने पर $C = \eu^{d}$ सहित
$n! \sim
C\sqrt n\,(n/\eu)^n$।

\emph{चरण 2: वालिस के द्वारा $C = \sqrt{2\pi}$।} बंद रूप $W_{2p}
= \frac{(2p)!}{4^p (p!)^2}\cdot\frac\pi2$ (पुनरावृत्ति से, प्रथम
वर्ष का परिकलन \cref{lem:b2:comparison:wallis} के परिवेश में दोहराया गया) चरण 1 के साथ मिलकर देता
है:
\[
W_{2p} \sim \frac{C\sqrt{2p}\,(2p/\eu)^{2p}}
{4^p\,\bigl(C\sqrt p\,(p/\eu)^p\bigr)^2}\cdot\frac{\pi}{2}
= \frac{\sqrt{2p}}{C\,p}\cdot\frac{\pi}{2}
= \frac{\pi}{C}\cdot\frac{1}{\sqrt{2p}} .
\]
$W_{2p} \sim \sqrt{\frac{\pi}{4p}}$ से तुलना करने पर (\cref{lem:b2:comparison:wallis}): $\frac{\pi}{C\sqrt{2p}} =
\sqrt{\frac{\pi}{4p}}\,(1 + o(1))$ $C = \pi
\sqrt{\frac{4p}{2p\,\pi}} = \sqrt{2\pi}$ को बाध्य कर देता है।
\end{proof}

\begin{example}[केंद्रीय द्विपद गुणांक]\label{ex:b2:comparison:centralbinomial}
\[
\binom{2n}{n} = \frac{(2n)!}{(n!)^2}
\sim \frac{\sqrt{4\pi n}\,(2n/\eu)^{2n}}{2\pi n\,(n/\eu)^{2n}}
= \frac{4^n}{\sqrt{\pi n}} :
\]
अर्थात् किसी सममित यादृच्छिक पथ के समय $2n$ पर $0$ लौट आने की प्रायिकता
$\sim \frac{1}{\sqrt{\pi n}}$ है --- और यह \cref{ch:b2:randomvar} की पूर्वघोषणा है।
\end{example}

\begin{remark}[इसी खंड के भीतर के परिप्रेक्ष्य]\label{rem:b2:comparison:perspectives}
आगे का हर परिमाणात्मक अध्याय इसी अध्याय की भाषा बोलता है। \cref{ch:b2:series} श्रेणियों
को पदों की पैमाने $n^{-\alpha}(\ln n)^{-\beta}$ से तुलना करके वर्गीकृत करता है --- और उसकी
सप्ताहांत समस्या उस सीमांत का पूरा मानचित्र बना देती है। \cref{ch:b2:integration} अनुचित
समाकलों के लिए वही करता है, \omterm{def:b2:metric:continuity}{संतत} चर में उसी पैमाने के साथ। \cref{ch:b2:powerseries} $\limsup\abs{a_n}^{1/n}$ से
अभिसरण त्रिज्याएँ परिकलित करता है, जो $n$-वें मूलों के समतुल्य का अभ्यास
है और जहाँ स्टर्लिंग मानक कुंजी है ($\sqrt[n]{n!}
\sim \frac n\eu$, \cref{exo:b2:comparison:4})। और प्रायिकता के अध्याय
स्टर्लिंग को सीधे भुना लेते हैं: द्विपद गुणांकों के लिए \cref{ch:b2:randomvar} के स्थानीय
आकलन अक्षरशः \cref{ex:b2:comparison:centralbinomial} और \cref{ex:b2:comparison:lopsided} हैं। यहाँ अनंतस्पर्शिता कोई एक अध्याय नहीं है;
वह पूरे खंड का लहजा है।
\end{remark}

\begin{method}[क्रमिक परिष्करण की जाँच-सूची]\label{met:b2:comparison:checklist}
किसी परिष्कृत प्रसार पर भरोसा करने से पहले चार बिंदुओं की जाँच कीजिए।
(1) \emph{पहले अस्तित्व:} किसी भी प्रसार से पहले मूल या अनुक्रम को
पक्का कर लीजिए (एकदिष्टता, मध्यमान प्रमेय) --- बिना निर्देश्य के प्रतीक
बहुत सुंदर प्रसार देते हैं और कुछ भी अर्थ नहीं रखते। (2) \emph{एक चक्कर
में एक ही कोटि:} हर प्रतिस्थापन पर उतना ही भरोसा किया जा सकता है जितनी
कोटि का आकलन डाला गया था; एक ही चक्कर से दो नए पद निकालना ग़लत गुणांकों
का क्लासिक स्रोत है। (3) \emph{शेषपद साथ चलते हैं:} हर बीजगणितीय पग में
$o(\cdot)$ साथ ले चलिए और अवशोषण (बड़े शेषपदों में छोटे पदों का समा जाना) अंत
में, स्पष्ट रूप से होने दीजिए। (4) \emph{संख्यात्मक जाँच:} $n$ के किसी
एक ईमानदार मान पर मूल्यांकन कीजिए; गुणांक की भूल बीजगणितीय पुनर्व्युत्पत्ति
में आश्चर्यजनक रूप से अक्सर बच निकलती है, और अंकगणित में लगभग कभी नहीं।
\end{method}

\begin{remark}[सामान्य भूलें]\label{rem:b2:comparison:pitfalls}
(क) समतुल्य \emph{जोड़ने} में बुरे हैं: $u_n \sim n + \ln n$ और $v_n \sim -n$ से $u_n + v_n \sim \ln
n$ निष्कर्ष
\emph{नहीं} निकाला जा सकता; निरसन स्पष्ट शेषपदों वाले प्रसार माँगते हैं,
नंगे समतुल्य कभी नहीं। (ख) किसी समतुल्यता का घातांकी कभी मत लीजिए: $n + 1 \sim n$
परंतु $\eu^{n+1} \not\sim \eu^n$; सुरक्षित दिशा $+\infty$ की ओर जाने वाले समतुल्यों का लघुगणक लेना
है (इस अध्याय की सप्ताहांत समस्या, प्रश्न 24)। (ग) \omterm{def:b2:comparison:expansion}{अनंतस्पर्शी प्रसार}
किसी \emph{पैमाने} से जुड़ा होता है: $f = \frac1x + o\bigl(\frac1{x^2}\bigr)$ लिखना $f =
\frac1x + o\bigl(\frac1x\bigr)$ से अधिक दावा करता है,
और दोनों को मिला देने से आगे का बीजगणित अमान्य हो जाता है। (घ) क्रमिक
परिष्करण में शेषपद सहित \emph{पूरा} वर्तमान प्रसार प्रतिस्थापित कीजिए ---
बीच में कोई $o(\cdot)$ गिरा देने से विश्वसनीय दिखने वाले पर ग़लत गुणांक निकलते
हैं। (ङ) श्रेणी--समाकल तुलना को एकदिष्टता चाहिए: दोलनशील पदों के लिए वह
सीधे विफल हो जाती है ($\sum\frac{\sin k}k$, \cref{ch:b2:series} की तुलना कीजिए)।
\end{remark}

\begin{example}[संख्याओं में स्टर्लिंग]\label{ex:b2:comparison:stirlingnum}
$n = 10$ पर: सूत्र $\sqrt{20\pi}\,(10/\eu)^{10}
\approx 3\,598\,696$ देता है जबकि $10! = 3\,628\,800$: अर्थात् सापेक्ष त्रुटि $8.3\cdot10^{-3}$, जो
$n = 10$ पर किसी ``अनंतस्पर्शी'' कथन के लिए उल्लेखनीय है। इस त्रुटि की एक
संरचना है --- यथार्थ परिष्करण $n!
= \sqrt{2\pi n}\,(n/\eu)^n\bigl(1 + \frac1{12n} +
O(n^{-2})\bigr)$ --- जिसका पहला सुधार $\frac1{120} \approx
8.3\cdot10^{-3}$ देखी गई खाई
को लगभग ठीक-ठीक समझा देता है। सप्ताहांत समस्या का ऑयलर--मैक्लॉरिन यंत्र
ठीक ऐसे ही सुधार-पदों का क्रमबद्ध स्रोत है।
\end{example}

\begin{remark}[यह अध्याय कहाँ काम आता है]
अनंतस्पर्शी तुलना आगे की हर परिमाणात्मक बात का व्याकरण है: \cref{ch:b2:series} की
अभिसरण-जाँचें और बर्ट्रांड परिदृश्य, \cref{ch:b2:integration} की समाकलनीयता कसौटियाँ, \cref{ch:b2:powerseries} के
अभिसरण-त्रिज्या परिकलन, और \cref{ch:b2:randomvar} की सीमा प्रमेयें (जहाँ स्टर्लिंग
डी म्वाव्र--लाप्लास आकलन चलाता है)। तृतीय वर्ष का खंड जिस एक विचार को हम
यहाँ हाथ से सिद्ध करते हैं --- मुख्य पद निकालिए, शेष को परिबद्ध कीजिए
--- उसी को लाप्लास विधि और प्रभावी अभिसरण में औद्योगिक रूप दे देता है।
\end{remark}

\begin{example}[अपने ही से तुलना किया गया एक समाकल: $\int_2^x \frac{\dd t}{\ln t}$]\label{ex:b2:comparison:liworked}
तुलना का औज़ार-बक्सा समाकलों पर भी चलता है। मान लीजिए $F(x) =
\int_2^x\frac{\dd t}{\ln t}$ (समाकल्य $\intco2\infty$
पर \omterm{def:b2:metric:continuity}{संतत} है)। खंडशः समाकलन कीजिए:
\[
F(x) = \Bigl[\frac{t}{\ln t}\Bigr]_2^x +
\int_2^x\frac{\dd t}{(\ln t)^2}
= \frac{x}{\ln x} + O\Bigl(\int_2^x\frac{\dd t}{(\ln
t)^2}\Bigr) + O(1),
\]
और शेष समाकल $o\bigl(\frac{x}{\ln x}\bigr)$ है: उसे $\sqrt x$ पर बाँटिए और
\[
\int_2^{\sqrt x}\frac{\dd t}{(\ln t)^2} \leq \sqrt x
\qquad\text{तथा}\qquad
\int_{\sqrt x}^{x}\frac{\dd t}{(\ln t)^2} \leq
\frac{x}{(\ln\sqrt x)^2} = \frac{4x}{(\ln x)^2} .
\]
से परिबद्ध कीजिए। अतः $F(x) \sim \frac{x}{\ln x}$। जिन पाठकों ने इस अध्याय की सप्ताहांत समस्या
में अभाज्य संख्या प्रमेय देखी है वे $F$ को पहचान लेंगे: यह लघुगणकीय
समाकल है, $\pi(x)$ का बेहतर आकलक, और यह परिकलन दिखा देता है कि वह $\frac{x}{\ln x}$ से
प्रथम कोटि तक सहमत है।
\end{example}

\begin{example}[किसी असंतुलित द्विपद पर स्टर्लिंग]\label{ex:b2:comparison:lopsided}
वही तीन-क्रमगुणित वाली विधि, जो \cref{ex:b2:comparison:centralbinomial} के लिए थी, $\binom{3n}{n} = \frac{(3n)!}{n!\,(2n)!}$ के लिए देती है:
\[
\binom{3n}{n} \sim
\frac{\sqrt{6\pi n}\,(3n/\eu)^{3n}}
{\sqrt{2\pi n}\,(n/\eu)^{n}\cdot\sqrt{4\pi n}\,(2n/\eu)^{2n}}
= \sqrt{\frac{3}{4\pi n}}\,
\Bigl(\frac{27}{4}\Bigr)^{\!n} .
\]
घातांकी दर $\frac{27}4 = \frac{3^3}{2^2}$ सूचना सिद्धांत की एन्ट्रॉपी संकेतन में $\eu^{3n\,H(1/3)}$ है: असंतुलित
द्विपद केंद्रीय $4^n$ प्रति दो पग से यथार्थतः धीमे बढ़ते हैं --- यहाँ $(27/4)^{1/3} \approx 1.89 < 2$
प्रति पग। संचय और प्रायिकता की हर द्विपद अनंतस्पर्शिता (\cref{ch:b2:randomvar}) यही एक
परिकलन है, बस भार भिन्न होते हैं।
\end{example}

\section{अंतर्निहित रूप से परिभाषित अनुक्रम}

\begin{method}\label{met:b2:comparison:implicit}
किसी समीकरण $F(x, n) =
0$ के हलों $x_n$ की अनंतस्पर्शिता ज्ञात करने के
लिए:\index{अंतर्निहित अनुक्रम}
\begin{enumerate}
  \item \emph{स्थानीयकरण}: किसी निश्चित अंतराल में $x_n$ के अस्तित्व और
        अद्वितीयता को सिद्ध कीजिए (एकदिष्टता, मध्यमान प्रमेय), और उसका
        कच्चा व्यवहार ज्ञात कीजिए (सीमा, वृद्धि की कोटि)।
  \item \emph{क्रमिक परिष्करण}: कच्चे रूप $x_n = (\text{मुख्य
        पद})(1 + \varepsilon_n)$ को समीकरण में प्रतिस्थापित कीजिए और $\varepsilon_n$ की अगली
        कोटि के लिए हल कीजिए; और इसे दोहराइए, हर चक्कर में एक कोटि
        परिष्कृत होती है।
\end{enumerate}
\end{method}

\begin{example}\label{ex:b2:comparison:tan}
$n \geq 1$ के लिए समीकरण $\tan x = x$ का $\intoo{n\pi - \frac\pi2}{n\pi + \frac\pi2}$ में ठीक एक हल $x_n$ है (वहाँ फलन $\tan x - x$ $-\infty$ से
$+\infty$ तक बढ़ता है, क्योंकि उसका अवकलज $\tan^2 x \geq 0$ है)। \emph{कच्चा:} $y_n \in \intoo{0}{\pi}$ के साथ
$x_n = n\pi +
\frac\pi2 - y_n$; और चूँकि $x_n \to
\infty$ तथा $\tan x_n = x_n \to +\infty$, $x_n$ अनंतस्पर्शी रेखा तक बाईं ओर से पहुँचता
है: $y_n \to 0$। \emph{परिष्करण:} $\tan x_n =
\cot y_n = \frac{1}{\tan y_n} \sim \frac{1}{y_n}$, और समीकरण $\cot y_n = x_n \sim n\pi$ $y_n \sim \frac{1}{n\pi}$ देता है। अतः
\[
x_n = n\pi + \frac\pi2 - \frac{1}{n\pi} +
o\Bigl(\frac1n\Bigr),
\]
और यह प्रक्रिया किसी भी कोटि तक चलती रहती है (\cref{exo:b2:comparison:6})।
\end{example}

\begin{example}[विधि का दूसरा चक्कर]\label{ex:b2:comparison:xplusln}
$x + \ln x = n$ को अनंतस्पर्शी रूप से हल कीजिए। \emph{स्थानीयकरण:} $x
\mapsto x + \ln x$ $\intoo{0}{+\infty}$ पर $-\infty$
से $+\infty$ तक बढ़ता है: अतः अद्वितीय मूल $x_n$, और $x_n \to \infty$। \emph{कच्चा:} $\ln x_n = o(x_n)$ से
$x_n \sim n$ मिलता है। \emph{परिष्करण:} $x_n = n - \ln x_n$ और $\ln x_n = \ln n
+ o(1)$ से (समतुल्यों के लघुगणक,
दोनों पक्ष $\to \infty$):
\[
x_n = n - \ln n + o(1) ;
\]
एक चक्कर और, $\ln x_n = \ln\bigl(n - \ln n + o(1)\bigr) =
\ln n - \frac{\ln n}{n} + o\bigl(\frac{\ln n}n\bigr)$ के साथ:
\[
x_n = n - \ln n + \frac{\ln n}{n} +
o\Bigl(\frac{\ln n}{n}\Bigr).
\]
($n = 100$ पर जाँच: मूल $x \approx 95.4415$ है; तीन-पद सूत्र $100 - 4.6052 + 0.0461 = 95.4409$ देता है और दो-पद वाला $95.3948$
--- हर चक्कर पूर्वानुमानित कोटि जीतता है।) वही \omterm{def:b2:structures:sn}{चक्र}, तीसरा परिदृश्य:
\cref{met:b2:comparison:implicit} की विधि को इससे कोई मतलब नहीं कि समीकरण कैसा दिखता है, केवल यह
चाहिए कि हर चक्कर प्रमुख अज्ञात को अलग कर दे।
\end{example}

\begin{exercise}[$\star\star$]\label{exo:b2:comparison:5}
$n \geq 2$ के लिए सिद्ध कीजिए कि $x^n + x = 1$ का अद्वितीय हल $x_n
\in \intoo{0}{1}$ है, कि $x_n \to 1$, और

x_n = 1 - \frac{\ln n}{n} + o\Bigl(\frac{\ln n}{n}\Bigr).
\]
\emph{($x_n^n = 1 - x_n$ से: लघुगणक लीजिए और $x_n = 1 - \varepsilon_n$ के साथ क्रमिक परिष्करण कीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:comparison:6}
\cref{ex:b2:comparison:tan} को एक कोटि और आगे बढ़ाइए:
\[
x_n = n\pi + \frac\pi2 - \frac{1}{n\pi} + \frac{1}{2n^2\pi} +
o\Bigl(\frac{1}{n^2}\Bigr).
\]
\emph{($\cot y_n = x_n$ यथार्थ रूप से लिखिए, $\cot y = \frac1y -
\frac y3 + o(y)$ और $x_n = n\pi(1 + \frac{1}{2n} - \dots)$ का प्रसार कीजिए, और पहचानिए।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:comparison:7}
$\lim_{n\to\infty} \dfrac{1}{n!}\sum_{k=0}^{n} k!$ निर्धारित कीजिए \emph{(अंतिम दो को छोड़कर सभी पदों के योग को परिबद्ध
कीजिए)}, और इससे \omterm{def:b2:comparison:expansion}{अनंतस्पर्शी प्रसार} $\sum_{k \leq n} k! = n!\bigl(1 + \frac1n +
O(n^{-2})\bigr)$ निष्कर्ष रूप में निकालिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:comparison:8}
मान लीजिए $u_0 > 0$ और $u_{n+1} = u_n + \dfrac{1}{u_n}$। सिद्ध कीजिए कि $u_n
\to \infty$, फिर कि $u_n \sim \sqrt{2n}$ \emph{($u_n^2$ का
अध्ययन कीजिए: उसकी वृद्धियाँ $2 + u_n^{-2}$ हैं; योग लीजिए)}, और परिष्कृत कीजिए:
\[
u_n = \sqrt{2n}\Bigl(1 + \frac{\ln n}{8n} +
o\Bigl(\frac{\ln n}{n}\Bigr)\Bigr).
\]
\emph{($u_n^2 = 2n + \sum_{k<n} u_k^{-2} + u_0^2$ और $u_k^2
\sim 2k$ से: \cref{thm:b2:comparison:seriesintegral} के अनुसार योग $\sim \frac12\ln n$ है।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:comparison:9}
(एक मरोड़ वाला रीमान योग) निम्नलिखित का अनंतस्पर्शी व्यवहार निर्धारित
कीजिए
\[
S_n = \sum_{k=1}^{n} \frac{1}{n + k\ln n} .
\]
\emph{($n$ बाहर निकालिए: $S_n = \frac1n\sum_k \bigl(1 +
\frac{k\ln n}{n}\bigr)^{-1}$; धीरे बदलते प्राचल $t = \ln n$ वाले रीमान-प्रकार के
योग को पहचानिए, $\int_0^1
\frac{\dd u}{1 + tu} = \frac{\ln(1+t)}{t}$ परिकलित कीजिए, और $S_n \sim
\frac{\ln\ln n}{\ln n}$ निष्कर्ष निकालिए।)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:comparison:10}
सर्वसमिका $(\ln n)^{\ln n} = n^{\ln\ln n}$ सिद्ध कीजिए, फिर निम्नलिखित को अनंत पर बढ़ते $o(\cdot)$ क्रम में
उपपत्ति सहित सजाइए: $n^2$, $(\ln n)^{\ln n}$, $2^n$, $n!$, $n^n$।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:comparison:11}
($\sum 1/k^2$ की पुच्छ, दो पद) यथार्थ क्रमिक निरसन $\sum_{k > n} \frac{1}{k(k+1)} = \frac{1}{n+1}$ और अपघटन $\frac1{k^2} = \frac{1}{k(k+1)} +
\frac{1}{k^2(k+1)}$ का उपयोग
करके सिद्ध कीजिए
\[
\sum_{k > n} \frac{1}{k^2}
= \frac1n - \frac{1}{2n^2} + O\Bigl(\frac{1}{n^3}\Bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:comparison:12}
मान लीजिए $u_0 = \frac12$ और $u_{n+1} = u_n + \eu^{-u_n}$। सिद्ध कीजिए कि $u_n \to \infty$, फिर --- $v_n = \eu^{u_n}$ रखकर और $v_{n+1} = v_n + 1 + \frac{1}{2v_n} + O\bigl(v_n^{-2}\bigr)$
दिखाकर --- स्थापित कीजिए
\[
u_n = \ln n + \frac{\ln n}{2n} + O\Bigl(\frac1n\Bigr).
\]
\end{exercise}

\section{समस्या: क्रमिक परिष्करण, ऑयलर--मैक्लॉरिन से अभाज्यों तक}

कोई अंतर्निहित या संचित राशि अपनी अनंतस्पर्शिता एक ही बार में शायद ही
सौंपती है; उसे चक्करों में निकालना पड़ता है, और हर चक्कर पिछले आकलन को
परिभाषित करने वाले संबंध में वापस डाल देता है। यह सप्ताहांत समस्या उस
\omterm{def:b2:structures:sn}{चक्र} को नए समीकरणों पर प्रशिक्षित करती है, प्रथम कोटि का
\emph{ऑयलर--मैक्लॉरिन सूत्र} सिद्ध करती है (श्रेणी--समाकल तुलना का समलंब
संस्करण, कठोर त्रुटि-पट्टियों के साथ), $x\ln x = n$ को प्रतिलोमित करती है, और
विधि का सबसे प्रसिद्ध चेक भुनाती है: स्वीकृत अभाज्य संख्या प्रमेय से
$n$-वें अभाज्य का अनंतस्पर्शी नियम $p_n \sim n\ln n$।

\begin{problem}[{सप्ताहांत समस्या --- ऑयलर--मैक्लॉरिन सुधार और
$n$-वें अभाज्य की अनंतस्पर्शिता}]
\label{pb:b2:comparison:1}

\medskip
\noindent\textbf{भाग I --- एक नए समीकरण पर परिष्करण-चक्र।}
\begin{enumerate}
  \item \cref{def:b2:comparison:expansion} का अद्वितीयता वाला दावा सिद्ध कीजिए: यदि उसी पैमाने पर
        $f = \sum_{i\leq
        k} c_i\varphi_i + o(\varphi_k) = \sum_{i \leq k}
        c_i'\varphi_i + o(\varphi_k)$, तो सभी $i$ के लिए $c_i = c_i'$। फिर पाठ के मिश्रित उदाहरण को एक
        सीढ़ी और ऊपर ले जाइए:
        \[
        \frac{1}{x - \ln x} = \frac1x + \frac{\ln x}{x^2} +
        \frac{(\ln x)^2}{x^3} + o\Bigl(\frac{(\ln
        x)^2}{x^3}\Bigr) \qquad (x \to +\infty),
        \]
        और समझाइए कि कोई पद $\frac{c}{x^2}$ क्यों नहीं आता।
  \item दिखाइए कि प्रत्येक $n \geq 1$ के लिए समीकरण $\eu^x + x =
        n$ का ठीक एक वास्तविक हल
        $x_n$ है, और यह कि $x_n \sim \ln n$ के साथ $x_n
        \to +\infty$।
  \item दो बार क्रमिक परिष्करण कीजिए:
        \[
        x_n = \ln n - \frac{\ln n}{n} - \frac{(\ln n)^2}{2n^2} +
        o\Bigl(\frac{(\ln n)^2}{n^2}\Bigr).
        \]
  \item $n = 1000$ पर संख्यात्मक जाँच कीजिए: $x_{1000}
        \approx 6.90083$ की तुलना प्रश्न 3 के
        एक-, दो- और तीन-पद वाले मानों से पाँच दशमलव स्थानों तक कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- ऑयलर--मैक्लॉरिन, प्रथम कोटि।}
\begin{enumerate}[resume]
  \item समलंब कर्नेल सर्वसमिका सिद्ध कीजिए: $\intcc{0}{1}$ पर $C^2$ वर्ग के $g$ के
        लिए
        \[
        \int_0^1 g(t)\,\dd t = \frac{g(0) + g(1)}{2}
        - \frac12\int_0^1 t(1 - t)\,g''(t)\,\dd t
        \]
        \emph{($\frac12 t(1-t)g''$ का दो बार खंडशः समाकलन कीजिए)}।
  \item मान लीजिए $f$ $\intco{1}{+\infty}$ पर $C^2$ है और $\int_1^\infty \abs{f''} < \infty$। दिखाइए कि
        \[
        E_n = \sum_{k=1}^{n} f(k) - \int_1^n f -
        \frac{f(1) + f(n)}{2}
        \]
        किसी अचर $E$ पर अभिसरित होता है, और पुच्छ-परिबंध $\abs{E - E_n} \leq \frac18\int_n^\infty\abs{f''}$ के साथ:
        यही प्रथम कोटि तक \emph{ऑयलर--मैक्लॉरिन सूत्र} है।
  \item इसे $f(t) = \frac1t$ पर लगाइए: सिद्ध कीजिए
        \[
        H_n = \ln n + \gamma + \frac{1}{2n} + \varepsilon_n,
        \qquad \abs{\varepsilon_n} \leq \frac{1}{8n^2},
        \]
        जिससे \cref{exo:b2:comparison:3} और मज़बूत हो जाता है (अचर को \cref{ex:b2:comparison:harmonic} से तुलना करके
        $\gamma$ के साथ पहचानिए)।
  \item अगला गुणांक निकालिए: दिखाइए कि $\varepsilon_n =
        -\frac{1}{12n^2} + o\bigl(\frac1{n^2}\bigr)$ \emph{($E_n$ की वृद्धियाँ
        $\frac12\int_0^1t(1-t)f''(n+t)\dd t =
        \frac1{12}f''(n) + o(f''(n))$ हैं; \cref{thm:b2:comparison:seriesintegral} के साथ पुच्छ का योग लीजिए)}।
  \item प्रश्न 6 को $f = \ln$ पर लगाइए: तीन पंक्तियों में $d_n = \ln n! - (n +
        \frac12)\ln n + n$ का अभिसरण
        (अर्थात् \cref{thm:b2:comparison:stirling} का चरण 1) पुनः निकालिए, और उपहार में त्रुटि-दर
        $d_n = d + O\bigl(\frac1n\bigr)$ भी।
  \item प्रश्न 6 को $f(t) = \frac{1}{\sqrt t}$ पर लगाइए: दिखाइए कि किसी अचर $c$ के लिए
        \[
        \sum_{k=1}^{n}\frac1{\sqrt k} = 2\sqrt n + c +
        \frac{1}{2\sqrt n} + O\Bigl(\frac{1}{n^{3/2}}\Bigr)
        \]
        और $n =
        10^4$ पर सभी पदों का मूल्यांकन कीजिए (अचर $c \approx -1.4604$ है)।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- प्रतिलोमन: समीकरण $x\ln x =
n$।}
\begin{enumerate}[resume]
  \item दिखाइए कि $n \geq 1$ के लिए $x\ln x = n$ का ठीक एक हल $x_n \in
        \intco{1}{+\infty}$ है, कि $x_n \to
        \infty$, और कि
        $\ln x_n \sim \ln n$।
  \item एक-पद प्रतिलोमन $x_n \sim
        \dfrac{n}{\ln n}$ निष्कर्ष रूप में निकालिए, फिर एक बार और
        क्रमिक परिष्करण कीजिए:
        \[
        \ln x_n = \ln n - \ln\ln n + o(1),
        \qquad
        x_n = \frac{n}{\ln n}\Bigl(1 + \frac{\ln\ln n}{\ln n}
        + o\Bigl(\frac{\ln\ln n}{\ln n}\Bigr)\Bigr).
        \]
  \item $n = 10^6$ पर जाँचिए: सच्चा मूल $x \approx
        87\,848$ है; एक-पद ($\approx 72\,382$) और दो-पद ($\approx 86\,140$)
        मानों से तुलना कीजिए, और धीमे लाभ का कारण समझाइए (प्रसार-प्राचल
        $\frac{\ln\ln
        n}{\ln n}$ है, जो $n = 10^6$ पर केवल $\approx 0.19$ है)।
  \item अब हम अभाज्य संख्या प्रमेय \emph{स्वीकार} कर लेते हैं: $\leq x$ तक
        के अभाज्यों की संख्या $\pi(x)$ $x \to \infty$ होने पर $\pi(x) \sim
        \frac{x}{\ln x}$ पूरा करती है
        (तृतीय वर्ष के खंड में ईमानदारी से सिद्ध)। $n$-वें अभाज्य के
        लिए $p_n$ लिखते हुए $\pi(p_n) = n$ को न्यायोचित ठहराइए, और प्रश्न 11--12
        का प्रतिलोमन चलाकर सिद्ध कीजिए
        \[
        p_n \sim n \ln n .
        \]
  \item लाभ: (क) दिखाइए कि $\sum_{k \leq n} p_k \sim
        \frac{n^2\ln n}{2}$ \emph{($\sum k\ln k$ की $\int t\ln t\,\dd t$ से तुलना कीजिए)};
        (ख) $100$ अंकों वाले किसी एकसमान यादृच्छिक पूर्णांक के अभाज्य
        होने की लगभग संभावना परिकलित कीजिए ($\ln 10^{100} \approx 230.26$: लगभग $230$ में एक)।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- विधि का निर्यात: $x\tan x =
1$।}
\begin{enumerate}[resume]
  \item दिखाइए कि प्रत्येक $n \geq 1$ के लिए समीकरण $\tan x =
        \frac1x$ का $\intoo{n\pi}{\,n\pi + \frac\pi2}$ में ठीक एक हल
        $x_n$ है, और यह कि $z_n = x_n
        - n\pi \to 0^+$।
  \item एक पद: $z_n \sim \dfrac{1}{n\pi}$।
  \item दिखाइए कि $z_n$ के प्रसार में कोई $\frac{c}{n^2}$ पद \emph{नहीं} है: $z_n = \frac1{n\pi} +
        O\bigl(\frac{1}{n^3}\bigr)$।
  \item तीन पद: $\arctan u = u - \frac{u^3}3 +
        O(u^5)$ और $\frac1{x_n} = \frac{1}{n\pi} -
        \frac{z_n}{(n\pi)^2} + O(n^{-3}\cdot z_n^2)$ का उपयोग करके सिद्ध कीजिए
        \[
        x_n = n\pi + \frac{1}{n\pi} -
        \frac{4}{3\pi^3 n^3} + o\Bigl(\frac{1}{n^3}\Bigr).
        \]
  \item $n = 3$ पर जाँचिए: सच्चा मूल $x_3 \approx 9.5293344$; एक-पद और तीन-पद मानों की तुलना
        कीजिए, और एक वाक्य में पाठ के $\tan x = x$ (\cref{ex:b2:comparison:tan}) से अंतर बताइए: हर
        अनुक्रम अपनी खिड़की में कहाँ बैठता है, और क्यों।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- एक गतिक परिष्करण, खेल के नियम, संश्लेषण।}
\begin{enumerate}[resume]
  \item मान लीजिए $u_0 \in \intoo{0}{\pi}$ और $u_{n+1} = \sin u_n$। दिखाइए कि $u_n \to 0$ ह्रासमान रूप से, और $\dfrac{1}{u_{n+1}^2} - \dfrac{1}{u_n^2}$ की
        सीमा परिकलित कीजिए \emph{($\sin u = u - \frac{u^3}6 + o(u^3)$ के द्वारा $\sin^{-2}$ का प्रसार कीजिए)}।
  \item इससे चेज़ारो माध्यों (प्रथम वर्ष का खंड) के द्वारा क्लासिक
        परिणाम निष्कर्ष रूप में निकालिए
        \[
        u_n \sim \sqrt{\frac{3}{n}} .
        \]
  \item (प्रमाणित संख्यात्मकी) प्रश्न 7 के कठोर परिबंध का उपयोग करके
        दिखाइए कि $n
        = 10^6$ पर $\ln n + \gamma + \frac1{2n}$ का मूल्यांकन $H_{10^6}$ देता है जिसमें त्रुटि
        अधिकतम $1.25\cdot10^{-13}$ है --- अर्थात् दस लाख पदों का योग तीन पदों से तेरह
        अंकों तक परिकलित।
  \item (खेल के नियम) उपपत्ति अथवा प्रतिउदाहरण के साथ सिद्ध या खंडित
        कीजिए: (क) यदि $u_n \sim v_n \to +\infty$ तो $\ln u_n \sim \ln v_n$; (ख) यदि $u_n \sim v_n$ तो $\eu^{u_n} \sim \eu^{v_n}$; (ग) यदि $+\infty$
        पर $f \sim g$ हो ($f, g$ अवकलनीय) तो $f' \sim g'$।
  \item (संश्लेषण) एक-एक वाक्य में: भाग I, III, IV में प्रयुक्त \cref{met:b2:comparison:implicit}
        का परिष्करण-चक्र; समलंब सुधार \cref{thm:b2:comparison:seriesintegral} में क्या जोड़ता है; $x\ln x$ का
        प्रतिलोमन ठीक क्यों $\pi(x)$ से $p_n$ तक का पुल है; और प्रश्न 24 के
        किस नियम ने किस पग की रक्षा की। दोनों शिखर नाम लीजिए:
        ऑयलर--मैक्लॉरिन सूत्र (प्रथम कोटि), और $n$-वें अभाज्य का
        अनंतस्पर्शी नियम।
\end{enumerate}
\end{problem}
