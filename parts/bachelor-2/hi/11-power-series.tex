\chapter{घात श्रेणियाँ}\label{ch:b2:powerseries}

घात श्रेणियाँ गणित की सबसे सुव्यवहृत फलन-श्रेणियाँ हैं: अपनी अभिसरण
चक्रिका के भीतर वे \omterm{def:b2:metric:compact}{संहत} उपसमुच्चयों पर \omterm{def:b2:funcseq:series}{सामान्य रूप से} अभिसरित होती हैं,
बिना दूसरी बार सोचे पद-दर-पद अवकलित और समाकलित की जा सकती हैं, और उनके योग
--- \emph{\omterm{def:b2:powerseries:analytic}{वैश्लेषिक फलन}} --- अपने गुणांकों से निर्धारित हो जाते हैं। यह
अध्याय वह पूरा पैकेट सिद्ध करता है और प्रथम वर्ष के खंड की हर टेलर श्रेणी
ईमानदारी से वापस ले आता है; और \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} बीजगणितीय लाभ के साथ उसे समाप्त
करते हैं।

\section{अभिसरण त्रिज्या}

\begin{lemma}[आबेल]\label{lem:b2:powerseries:abel}
यदि किसी $z_0 \neq 0$ के लिए अनुक्रम $(a_n z_0^n)$ परिबद्ध हो, तो $\sum a_n z^n$ प्रत्येक $\abs z < \abs
{z_0}$ के लिए
\omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसरित होती है, और प्रत्येक चक्रिका $\abs z \leq r < \abs{z_0}$ पर \omterm{def:b2:funcseq:series}{सामान्य रूप
से}।
\end{lemma}

\begin{proof}
$\abs{a_n z_0^n} \leq M$ और $\abs z \leq r$ के साथ:
\[
\abs{a_n z^n} = \abs{a_n z_0^n}\,\Bigl|\frac{z}{z_0}\Bigr|^n
\leq M\Bigl(\frac{r}{\abs{z_0}}\Bigr)^{\!n},
\]
अर्थात् अभिसारी गुणोत्तर परिबंध, जो उस चक्रिका पर एकसमान है।
\end{proof}

\begin{definition}[अभिसरण त्रिज्या]\label{def:b2:powerseries:radius}
$\sum a_n z^n$ की \emph{अभिसरण त्रिज्या}\index{अभिसरण त्रिज्या}
\[
R = \sup\{r \geq 0 : (a_n r^n) \text{ परिबद्ध}\} \in
\intcc{0}{+\infty} .
\]
है। \cref{lem:b2:powerseries:abel} के अनुसार: $\abs
z < R$ के लिए निरपेक्ष अभिसरण (\omterm{def:b2:metric:compact}{संहत} उपचक्रिकाओं पर
सामान्य), और $\abs z > R$ के लिए अपसरण --- वस्तुतः अपरिबद्ध पद। सीमा-वृत्त पर कुछ
भी हो सकता है (\cref{exo:b2:powerseries:2})। व्यवहार में $R$ $\abs{a_n}\abs z^n$ पर डालांबेर की अनुपात जाँच से
अथवा तुलना से परिकलित होती है।
\end{definition}

\begin{example}[बिना अनुपात जाँच के एक त्रिज्या]\label{ex:b2:powerseries:sinn}
$\sum \sin(n)\,z^n$ की त्रिज्या क्या है? अनुपात $\abs{\sin(n+1)/\sin n}$ की कोई सीमा नहीं है, परंतु परिभाषा
सीधे काम कर जाती है। \emph{$R \geq 1$:} $\abs{\sin n} \leq 1$, अतः $(\sin
n\cdot r^n)$ प्रत्येक $r < 1$ के लिए ---
वस्तुतः $r =
1$ के लिए --- परिबद्ध है। \emph{$R \leq 1$:} इतना ही पर्याप्त है कि
$\sin n \not\to 0$। मान लीजिए $\sin n \to 0$; तब योग-सूत्र
\[
\sin(n+1) = \sin n\cos 1 + \cos n\sin 1
\]
$\cos n \to 0$ को बाध्य कर देता (उसे $\cos n$ के लिए हल कीजिए: $\sin 1 \neq
0$), जो $\sin^2 n + \cos^2 n = 1$ के विरुद्ध
है। अतः पद $\sin(n)\,1^n$ $0$ की ओर नहीं जाते: श्रेणी $z =
1$ पर अपसरित होती है, और
$R \leq 1$। निष्कर्ष: $R = 1$। समापन दृष्टि: त्रिज्या $\abs{a_n}r^n$ के परिबद्ध होने के बारे
में कथन है --- अनुपातों की किसी सीमा की कभी आवश्यकता नहीं, और परिबद्धता के
तर्क वे स्थितियाँ निपटा देते हैं जिन्हें अनुपात जाँच छू भी नहीं सकती (\cref{exo:b2:powerseries:1}
के दोलनशील गुणांकों से तुलना कीजिए)।
\end{example}

\begin{proposition}[संक्रियाएँ]\label{prop:b2:powerseries:operations}
मान लीजिए $\sum a_nz^n$, $\sum b_nz^n$ की त्रिज्याएँ $R_a, R_b$ हैं। तब $\abs z < \min(R_a, R_b)$ के लिए:
\[
\sum (a_n + b_n)z^n = \sum a_nz^n + \sum b_nz^n,
\qquad
\Bigl(\sum a_nz^n\Bigr)\Bigl(\sum b_nz^n\Bigr) = \sum c_n z^n,
\quad c_n = \sum_{k=0}^{n} a_kb_{n-k},
\]
और दोनों श्रेणियों की त्रिज्या $\geq \min(R_a, R_b)$ है। (गुणनफल \omterm{thm:b2:series:fubini}{कोशी गुणनफल} है, जो
निरपेक्ष अभिसरण तथा \cref{thm:b2:series:fubini} के कारण वैध है।)
\end{proposition}

\begin{proof}
योग वाला सूत्र अभिसारी श्रेणियों की रैखिकता है, और जब भी $a_nr^n$ तथा $b_nr^n$
दोनों परिबद्ध हों तब $(a_n + b_n)r^n$ भी परिबद्ध है: अतः त्रिज्या $\geq \min(R_a, R_b)$। गुणनफल के लिए
$\abs z
< \min(R_a, R_b)$ स्थिर कीजिए: वहाँ दोनों श्रेणियाँ \emph{\omterm{def:b2:series:def}{निरपेक्ष रूप से}} अभिसरित होती
हैं (\cref{lem:b2:powerseries:abel}), अतः दोहरे सूचकांक वाला कुल $(a_kz^k\,b_lz^l)_{k,l}$ \omterm{def:b2:series:summable}{योग्य} है, और \cref{thm:b2:series:fubini} किसी भी समूहन
की अनुमति दे देता है। $k
+ l = n$ के अनुसार समूहबद्ध करने पर:
\[
\Bigl(\sum_k a_kz^k\Bigr)\Bigl(\sum_l b_lz^l\Bigr)
= \sum_{n\geq0}\Bigl(\sum_{k+l=n}a_kb_l\Bigr)z^n
= \sum_{n\geq0}c_nz^n ,
\]
जो ऐसे प्रत्येक $z$ के लिए \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसारी है: अतः गुणनफल श्रेणी
की त्रिज्या भी $\geq \min(R_a, R_b)$ है।
\end{proof}

\begin{example}[एक कोशी वर्ग, प्रतिजाँच सहित]\label{ex:b2:powerseries:cauchysquare}
गुणोत्तर श्रेणी का वर्ग लीजिए: $\abs x < 1$ के लिए $\bigl(\sum x^k\bigr)^2$ में $x^n$ का गुणांक $c_n = \sum_{k+l=n}
1\cdot1 = n + 1$ है,
अतः
\[
\frac{1}{(1-x)^2} = \sum_{n\geq0}(n+1)\,x^n .
\]
पद-दर-पद अवकलन (नीचे \cref{thm:b2:powerseries:calculus}) से प्रतिजाँच: $\frac{1}{1-x} = \sum x^n$ का अवकलन $\frac{1}{(1-x)^2} = \sum
nx^{n-1} = \sum(n+1)x^n$ देता है ---
अर्थात् दो असंबद्ध कार्यविधियों से वही श्रेणी। समापन दृष्टि: जब कोई
गुणांक-सर्वसमिका रहस्यमय लगे, तब इन दोनों इंजनों (संवलन अथवा अवकलन) में से
कोई एक प्रायः उसे एक ही पंक्ति में उत्पन्न कर देता है; और सप्ताहांत समस्या
का $\sum\binom{2k}k\binom{2n-2k}{n-k} = 4^n$ वाला प्रश्न संवलन इंजन को पूरी शक्ति से चलाता है।
\end{example}

\begin{example}[$\frac{1}{1-x}$ से गुणा करने पर गुणांकों का योग बन जाता है]\label{ex:b2:powerseries:partialsums}
गुणोत्तर श्रेणी के विरुद्ध \omterm{thm:b2:series:fubini}{कोशी गुणनफल} का एक स्मरणीय अर्थ है: त्रिज्या
$R > 0$ वाली किसी भी $\sum a_nx^n$ तथा $\abs x <
\min(R, 1)$ के लिए
\[
\frac{1}{1-x}\sum_{n\geq0}a_nx^n
= \sum_{n\geq0}\Bigl(\sum_{k=0}^{n}a_k\Bigr)x^n :
\]
अर्थात् $\frac{1}{1-x}$ से गुणा करने पर गुणांकों के स्थान पर उनके आंशिक योग आ जाते हैं
(सर्व-एक अनुक्रम के साथ संवलन)। उदाहरण: $s_n =
\sum_{k\leq n}\frac{1}{k!}$ के साथ $\dfrac{\eu^x}{1-x} = \sum_n s_n x^n$, जो $\eu$ के आंशिक
योग हैं --- इसकी तुलना \cref{exo:b2:powerseries:11} से कीजिए, जहाँ $\eu^{-x}$ के साथ वही गुणनफल
अव्यवस्थापन-गणना संकेतित करता है। समापन दृष्टि: घात श्रेणियों पर संक्रियाएँ
भेस बदली हुई गुणांक-अनुक्रमों पर संक्रियाएँ हैं ($\frac1{1-x}$ से गुणा: योग; $x$ से
गुणा: स्थानांतरण; अवकलन: $n$ से गुणा और स्थानांतरण) --- और जनक फलनों वाला
अध्याय इस शब्दकोश को धाराप्रवाह पढ़ेगा।
\end{example}

\section{योग की नियमितता}

\begin{theorem}[पद-दर-पद कलन]\label{thm:b2:powerseries:calculus}
मान लीजिए $f(x) = \sum_{n\geq0} a_n x^n$ की त्रिज्या $R > 0$ है (वास्तविक चर $x \in \intoo{-R}{R}$)।
\begin{enumerate}
  \item व्युत्पन्न श्रेणी $\sum n\,a_n x^{n-1}$ की त्रिज्या \emph{वही} $R$ है, और $f$ $f'(x) = \sum_{n \geq 1}
        n a_n x^{n-1}$
        सहित $C^1$ है। इसे दोहराने पर $f$ $C^\infty$ है और
        \[
        a_n = \frac{f^{(n)}(0)}{n!} :
        \]
        और घात श्रेणी के गुणांक अद्वितीय होते हैं ($0$ के पास एक ही योग
        वाली दो श्रेणियों के गुणांक बराबर होते हैं)।
  \item पद-दर-पद प्रतिअवकलज: $\sum \frac{a_n}{n+1}x^{n+1}$ की त्रिज्या $R$ है और अवकलज $f$।
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{वही त्रिज्या:} यदि $(a_nr^n)$ परिबद्ध हो और $r' < r$, तो $n\abs{a_n} r'^{\,n-1} = \frac{n}{r'}\abs{a_nr^n}
\bigl(\frac{r'}{r}\bigr)^n$ परिबद्ध है (वस्तुतः
$\to 0$: गुणोत्तर $n$ को हरा देता है), अतः $R' \geq R$; और विलोमतः $\abs{a_n x^n} \leq \abs x
\cdot n\abs{a_n}\abs x^{n-1}$ $R \geq R'$ देता है।

\emph{अवकलन:} $\intcc{-r}{r}$, $r < R$ पर व्युत्पन्न श्रेणी \omterm{def:b2:funcseq:series}{सामान्य रूप से} अभिसरित होती है
(त्रिज्या-परिकलन से $n\abs{a_n}r^{n-1}$ \omterm{def:b2:series:summable}{योग्य}); और मूल श्रेणी $x = 0$ पर अभिसरित होती है: अतः
श्रेणियों के लिए अवकलन प्रमेय (\cref{thm:b2:funcseq:seriestransfer}) ऐसे हर खंड पर, इसलिए $\intoo{-R}{R}$ पर लागू होती
है। $k$ बार दोहराकर $0$ पर मूल्यांकन कीजिए: स्पष्ट रूप से $k$-वीं
व्युत्पन्न श्रेणी
\[
f^{(k)}(x) = \sum_{n\geq k} n(n-1)\cdots(n-k+1)\,a_n\,x^{n-k},
\]
है, और $x = 0$ पर $n > k$ वाला हर पद लुप्त हो जाता है, केवल अचर पद $k(k-1)\cdots1\cdot a_k$ बचता है:
$f^{(k)}(0) =
k!\,a_k$। गुणांकों की अद्वितीयता इससे निकल आती है: $0$ के पास एक ही योग वाली
दो घात श्रेणियों के $0$ पर अवकलज एक ही होते हैं, अतः $a_k$ भी। प्रतिअवकलज:
उसी परिकलन से वही त्रिज्या, और पद-दर-पद अवकलन करके वापस।
\end{proof}

\begin{example}[किसी बिंदु पर श्रेणी का मूल्यांकन]\label{ex:b2:powerseries:evaluate}
$\sum_{n\geq1}\dfrac{n^2}{2^n}$ क्या है? वह \cref{exo:b2:powerseries:3} का $x = \frac12 < 1 = R$ पर, अर्थात् चक्रिका के \emph{भीतर} मूल्यांकित
योग $\sum n^2x^n$ है, जहाँ बंद रूप निकालने में प्रयुक्त हर प्रबंधन वैध था:
\[
\sum_{n\geq1} n^2x^n = \frac{x(1+x)}{(1-x)^3}
\quad\Longrightarrow\quad
\sum_{n\geq1}\frac{n^2}{2^n}
= \frac{\frac12\cdot\frac32}{(\frac12)^3}
= \frac{3/4}{1/8} = 6 .
\]
वही इंजन, दूसरी घुंडियाँ: $x = \frac13$ $\sum\frac{n^2}{3^n} = \frac{\frac13\cdot\frac43}{(2/3)^3} =
\frac32$ देता है। समापन दृष्टि: घात-श्रेणी
सर्वसमिका कोई एक सूत्र नहीं, एक मशीन है --- एक ही व्युत्पत्ति सभी $\abs q < 1$ के
लिए हर संख्यात्मक श्रेणी $\sum n^2q^n$ का दाम एक साथ तय कर देती है; और इसी तरह जनक
फलनों वाला अध्याय प्रत्याशाएँ और प्रसरण थोक में परिकलित करेगा।
\end{example}

\begin{example}[क्लासिक, इस बार ईमानदारी से]\label{ex:b2:powerseries:classics}
$\displaystyle\frac{1}{1 - x} = \sum x^n$ ($R = 1$); पद-दर-पद समाकलन करने पर (\cref{thm:b2:powerseries:calculus}~(2)):
\[
-\ln(1 - x) = \sum_{n\geq1} \frac{x^n}{n},
\qquad
\arctan x = \sum_{n \geq 0} \frac{(-1)^n x^{2n+1}}{2n+1}
\quad (\abs x < 1),
\]
और दूसरा दो पगों में: गुणोत्तर श्रेणी में $-x^2$ प्रतिस्थापित करके $\frac{1}{1+x^2} = \sum(-1)^nx^{2n}$
पाइए (त्रिज्या $1$, क्योंकि $\abs{x^2} < 1 \iff \abs x < 1$), फिर $0$ पर लुप्त होने वाला पद-दर-पद
प्रतिअवकलज लीजिए; दोनों पक्ष एक ही फलन के प्रतिअवकलज हैं और $0$ पर उनका
मान एक ही है, अतः वे $\intoo{-1}{1}$ पर बराबर हैं। और $\exp$: श्रेणी $E(x) = \sum \frac{x^n}{n!}$ ($R = \infty$) $E' = E$,
$E(0) = 1$ पूरा करती है, जो पद-दर-पद अवकलन से मिलता है, अतः प्रथम वर्ष की
अद्वितीयता से $E = \exp$। प्रथम वर्ष के खंड का हर ``मानक प्रसार'' अब अपनी पूरी
घात श्रेणी के बारे में प्रमेय है।
\end{example}

\begin{example}[चक्रिका के भीतर से परिकलित एक लघुगणक]\label{ex:b2:powerseries:lntwo}
$-\ln(1-x) = \sum\frac{x^n}{n}$ का अभ्यंतरीय बिंदु $x = \frac12$ पर मूल्यांकन:
\[
\sum_{n\geq1}\frac{1}{n\,2^n} = \ln 2 ,
\]
अर्थात् $\ln 2$ का तेज़ी से अभिसरित होने वाला निरूपण (दस पद ही $\ln 2 = 0.69314\ldots$ के मुक़ाबले
$0.69306\ldots$ दे देते हैं), जो केवल सीमा पर उपलब्ध एकांतर श्रेणी $1 - \frac12 + \frac13 -
\dots$ से कहीं बेहतर
है। समापन दृष्टि: जब भी कोई अचर किनारे पर और चक्रिका के यथार्थतः भीतर,
दोनों जगह पहुँच में हो, तब संख्यात्मक दृष्टि से भीतर वाला जीतता है ---
हरात्मक क्षय के मुक़ाबले गुणोत्तर क्षय।
\end{example}

\begin{example}[अवकलन त्रिज्या बचाता है, सीमा नहीं]\label{ex:b2:powerseries:boundaryloss}
श्रेणी $\sum_{n\geq1}\frac{x^n}{n^2}$ की त्रिज्या $1$ है और वह \emph{दोनों} अंत्यबिंदुओं पर अभिसरित
होती है ($\sum\frac1{n^2}$ और उसका एकांतर जुड़वाँ)। उसकी व्युत्पन्न श्रेणी
\[
\sum_{n\geq1}\frac{x^{n-1}}{n} ,
\]
की त्रिज्या वही $1$ है --- जैसा \cref{thm:b2:powerseries:calculus} गारंटी देती है --- परंतु अब वह $x = 1$ पर
अपसरित होती है (हरात्मक श्रेणी) जबकि $x = -1$ पर अब भी अभिसरित (एकांतर)। एक और
अवकलन $\sum_{n\geq2}\frac{n-1}{n}x^{n-2}$ देता है, जो दोनों सिरों पर अपसारी है (पद $0$ की ओर नहीं जाते)।
समापन दृष्टि: हर अवकलन गुणांकों को $n$ से गुणा कर देता है, जो त्रिज्या
कभी नहीं हिलाता (गुणोत्तर बहुपद को हरा देता है) पर सीमा-क्षय की एक कोटि खा
जाता है; पद-दर-पद कलन भीतर का खेल है, और किनारे पर जो कुछ होता है उसकी फिर
से जाँच करनी ही पड़ती है --- और सप्ताहांत समस्या का आबेल--टाउबर सिद्धांत
ठीक वही पुनर्जाँच है।
\end{example}

\begin{example}[शेषों के अनुसार श्रेणी को बाँटना --- अंत तक किया हुआ]\label{ex:b2:powerseries:quarter}
$f(x) = \sum_{n\geq0} \dfrac{x^{4n}}{(4n)!}$ का बंद रूप परिकलित कीजिए। $\cosh x = \sum \frac{x^{2m}}{(2m)!}$ और $\cos x =
\sum \frac{(-1)^m x^{2m}}{(2m)!}$ दोनों की त्रिज्या $\infty$ है, अतः
उनका औसत पद-दर-पद परिकलित किया जा सकता है:
\[
\frac{\cosh x + \cos x}{2}
= \sum_{m\geq0}\frac{1 + (-1)^m}{2}\,\frac{x^{2m}}{(2m)!}
= \sum_{m \text{ सम}}\frac{x^{2m}}{(2m)!}
= \sum_{n\geq0}\frac{x^{4n}}{(4n)!} = f(x) .
\]
छननी $\frac{1+(-1)^m}{2}$ ठीक सम $m$ रख लेती है: यह इकाई-मूल छननी का वास्तविक अवतार है
(सम्मिश्र रूप, $\iu^n$ के साथ, $4$ के सापेक्ष शेष एक ही झटके में निकाल लेता
है)। समापन जाँच: $f$ $f(0) = 1$, $f'(0) = f''(0) = f'''(0) = 0$ के साथ $f'''' = f$ हल करता है --- श्रेणी का चार
बार अवकलन कीजिए (\cref{thm:b2:powerseries:calculus}) और उसे स्वयं को पुनः उत्पन्न करते देखिए; और $\frac{\cosh + \cos}{2}$
वही आँकड़े पूरा करता है।
\end{example}

\begin{definition}[वैश्लेषिक फलन]\label{def:b2:powerseries:analytic}
$f$ $x_0$ पर \emph{वैश्लेषिक}\index{वैश्लेषिक फलन} कहलाता है जब वह किसी
प्रतिवेश पर $(x - x_0)$ की किसी घात श्रेणी का योग हो; और किसी अंतराल पर तब जब वह
हर बिंदु पर ऐसा हो। घात श्रेणियों के योग अपनी चक्रिका के भीतर वैश्लेषिक
होते हैं (प्रसार का पुनर्विन्यास --- पुनःकेंद्रण के लिए इस स्तर पर
स्वीकृत, और $x_0 = 0$ वाली स्थिति \cref{thm:b2:powerseries:calculus} है)। वैश्लेषिक से $C^\infty$ निकलता है; पर विलोम
\emph{विफल} है: चपटा फलन $\eu^{-1/x^2}$ (\cref{exo:b2:powerseries:7})।
\end{definition}

\begin{example}[पुनःकेंद्रण, और दूरी के रूप में त्रिज्या]\label{ex:b2:powerseries:recenter}
$f(x) = \frac{1}{1-x}$ का $x_0 = \frac12$ के परितः प्रसार कीजिए: $x = \frac12 + h$ लिखने पर,
\[
\frac{1}{1 - x} = \frac{1}{\frac12 - h}
= \frac{2}{1 - 2h}
= \sum_{n\geq0} 2^{n+1}\,h^n
= \sum_{n\geq0} 2^{n+1}\Bigl(x - \frac12\Bigr)^{\!n},
\]
जो $\abs{2h} < 1$ के लिए मान्य है, अर्थात् $\abs{x - \frac12} < \frac12$ के लिए। नई त्रिज्या ठीक नए केंद्र से
विचित्रता $x = 1$ तक की दूरी है: अर्थात् पुनःकेंद्रण चक्रिका को निकटतम बाधा के
अनुसार सिकोड़ (या बढ़ा) देता है। समापन दृष्टि: वैश्लेषिकता की परिभाषा के
पीछे यही चित्र है --- एक ही फलन, अनेक स्थानीय घात श्रेणियाँ, और हर एक उस
सबसे बड़ी चक्रिका पर बसती है जो परेशानी से बच निकलती है; और तृतीय वर्ष का
खंड ``त्रिज्या $=$ निकटतम सम्मिश्र विचित्रता तक की दूरी'' वाले अनुमान को
प्रमेय बना देता है।
\end{example}

\begin{remark}[सामान्य भूलें]
\emph{(क) अनुपात जाँच पर्याप्त है, आवश्यक नहीं:} जब $\abs{a_{n+1}/a_n}$ की कोई सीमा न हो
(\cref{ex:b2:powerseries:sinn}, \cref{exo:b2:powerseries:1}), तब परिभाषा पर लौटिए: $R = \sup\{r : (a_nr^n)$ परिबद्ध$\}$। \emph{(ख) सीमा कोई भी
चीज़ मुफ़्त पार नहीं करती:} पद-दर-पद अवकलन और समाकलन \omterm{def:b2:metric:topology}{विवृत} चक्रिका के
\emph{भीतर} की प्रमेयें हैं; $\abs x = R$ पर हर श्रेणी की फिर से जाँच करनी पड़ती
है (और सप्ताहांत समस्या का पूरा विषय यही है)। \emph{(ग) योग की त्रिज्या:}
$\min(R_a, R_b)$ केवल निचला परिबंध है --- निरसन उसे बढ़ा सकते हैं ($a_n = 1,
b_n = -1$: योग सर्वथा $0$,
त्रिज्या $\infty$)। \emph{(घ) $C^\infty$ \omterm{def:b2:powerseries:analytic}{वैश्लेषिक} नहीं होता:} कोई अभिसारी टेलर
श्रेणी \emph{ग़लत} फलन पर अभिसरित हो सकती है (\cref{exo:b2:powerseries:7}); अतः $f(x) = \sum
\frac{f^{(n)}(0)}{n!}x^n$ लिखने से
पहले उसे सिद्ध कीजिए --- किसी अवकल समीकरण (\cref{met:b2:powerseries:ode}), किसी शेषपद आकलन, अथवा
किसी समाकल सूत्र के द्वारा।
\end{remark}

\begin{remark}[यह कहाँ काम आता है]
घात श्रेणियाँ आगे के तीन अध्यायों की श्रमिक हैं: अवकल समीकरणों वाला अध्याय
$\sum a_nx^n$ डालकर रैखिक साधारण अवकल समीकरण हल करता है (नीचे वाला विधि-बक्सा,
औद्योगिक रूप में); जनक फलनों वाला अध्याय प्रायिकताओं की सर्वसमिकाओं को घात
श्रेणियों की सर्वसमिकाओं में और वापस बदल देता है; और तृतीय वर्ष का खंड
सम्मिश्र चर को औपचारिक बना देता है, जहाँ वैश्लेषिकता सम्मिश्र अवकलनीयता के
तुल्य हो जाती है और ऊपर की ``स्वीकृत पुनःकेंद्रण'' को उसकी ईमानदार उपपत्ति
मिल जाती है। और सप्ताहांत समस्या ठीक उसी एक जगह की छानबीन करती है जहाँ इस
अध्याय की प्रमेयें चुप रहती हैं: स्वयं सीमा $\abs x = R$ पर।
\end{remark}

\begin{method}[अवकल समीकरण के द्वारा प्रसार]\label{met:b2:powerseries:ode}
किसी फलन $f$ का घात श्रेणी में प्रसार करने के लिए: बहुपद गुणांकों वाला
कोई ऐसा रैखिक अवकल समीकरण ढूँढ़िए जिसे $f$ संतुष्ट करता हो; $\sum a_nx^n$ डालिए;
गुणांक पहचानकर $(a_n)$ के लिए पुनरावृत्ति पाइए; उसे हल कीजिए, और त्रिज्या तथा
आरंभिक शर्तें जाँच लीजिए। उदाहरण --- द्विपद श्रेणी: $f(x) = (1+x)^\alpha$ $(1+x)f' = \alpha f$, $f(0)
= 1$ पूरा करता
है; डालने पर $(n+1)a_{n+1} = (\alpha - n)a_n$ मिलता है, अतः $a_n =
\binom{\alpha}{n}$, त्रिज्या $1$ (अनुपात जाँच), और योग,
जो उन्हीं आरंभिक मानों के साथ वही अवकल समीकरण पूरा करता है, रैखिक अवकल
समीकरणों की अद्वितीयता प्रमेय (प्रथम वर्ष का खंड) से $(1 + x)^\alpha$ के बराबर है।
\end{method}

\begin{example}[किसी प्रणोदित समीकरण पर यह विधि]\label{ex:b2:powerseries:odeforced}
घात श्रेणी से $y' = y + x$, $y(0) = 0$ हल कीजिए। $y =
\sum a_nx^n$ डालने पर:
\[
\sum_{n\geq0}(n+1)a_{n+1}x^n
= \sum_{n\geq0}a_nx^n + x ,
\]
और गुणांक पहचानने पर: $a_1 = a_0 = 0$, $2a_2 = a_1 + 1
= 1$, तथा $n \geq 2$ के लिए $(n+1)a_{n+1} = a_n$। अतः $a_2 =
\frac{1}{2!}$, और आगमन से
प्रत्येक $n \geq 2$ के लिए $a_n = \frac{1}{n!}$: त्रिज्या $\infty$, और
\[
y(x) = \sum_{n\geq2}\frac{x^n}{n!} = \eu^x - 1 - x .
\]
जाँच: $y' = \eu^x - 1 = y + x$ और $y(0) = 0$। समापन दृष्टि: पुनरावृत्ति \emph{ही} समीकरण है,
गुणांक-दर-गुणांक; और प्रणोदन पद केवल आरंभ के परिमित गुणांकों को विक्षुब्ध
करता है, जिसके बाद समांगी ढर्रा सँभाल लेता है --- ``विशिष्ट हल जमा समांगी
हल'' की विविक्त परछाईं।
\end{example}

\section{जनक फलन}

\begin{example}[फिबोनाच्ची]\label{ex:b2:powerseries:fibonacci}
मान लीजिए $F(x) = \sum_{n\geq0} F_n x^n$ (फिबोनाच्ची संख्याएँ, $F_0 = 0$, $F_1 = 1$)। पुनरावृत्ति $F_{n+2} = F_{n+1} + F_n$ $x^{n+2}$ से
गुणा करके योग लेने पर बदल जाती है
\[
F(x) - x = x\,F(x) + x^2 F(x)
\quad\Longrightarrow\quad
F(x) = \frac{x}{1 - x - x^2} ,
\]
में, जो वहाँ मान्य है जहाँ श्रेणी अभिसरित होती है। त्रिज्या $\frac{1}{\varphi}$ है: $F_n \sim
\frac{\varphi^n}{\sqrt5}$
से (बिने, अगला उदाहरण --- अथवा कच्चा आगमन $F_n \leq 2^n$ और पुनरावृत्ति) अनुपात जाँच
देती है
\[
\frac{F_{n+1}\abs x^{n+1}}{F_n\abs x^n}
\longrightarrow \varphi\abs x ,
\qquad\text{अभिसरण तभी जब } \abs x < \frac1\varphi
\approx 0.618 .
\] $\frac{x}{1 - x - x^2}$ पर आंशिक भिन्न और गुणोत्तर श्रेणी
बिने का सूत्र पुनः निकाल देते हैं --- \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} रैखिक पुनरावृत्तियों को
औद्योगिक रूप दे देते हैं।\index{जनक फलन}
\end{example}

\begin{example}[बिने का सूत्र, करके दिखाया गया]\label{ex:b2:powerseries:binet}
मान लीजिए $\varphi = \frac{1+\sqrt5}{2}$ और $\psi =
\frac{1-\sqrt5}{2}$, जो $X^2 = X + 1$ के मूल हैं; और चूँकि $\varphi
+ \psi = 1$ तथा $\varphi\psi = -1$,
\[
1 - x - x^2 = (1 - \varphi x)(1 - \psi x) .
\]
आंशिक भिन्न: $\frac{x}{(1-\varphi x)(1-\psi x)} =
\frac{A}{1 - \varphi x} + \frac{B}{1 - \psi x}$ खोजने पर अचर पद $A + B = 0$ देता है और $x$-गुणांक $-A\psi - B\varphi
= 1$, अतः $A(\varphi - \psi) = 1$:
$A = \frac{1}{\sqrt5} = -B$। दो गुणोत्तर श्रेणियों के बाद,
\[
F(x) = \frac{1}{\sqrt5}\sum_{n\geq0}
\bigl(\varphi^n - \psi^n\bigr)x^n
\quad\Longrightarrow\quad
F_n = \frac{\varphi^n - \psi^n}{\sqrt5}
\]
गुणांकों की अद्वितीयता (\cref{thm:b2:powerseries:calculus}) से। और चूँकि $\abs\psi < 1$, पद $\frac{\psi^n}{\sqrt5}$ का निरपेक्ष मान
$< \frac12$ है: अतः $F_n$ $\frac{\varphi^n}{\sqrt5}$ के \emph{निकटतम पूर्णांक} है। समापन दृष्टि: $F$ की
त्रिज्या $\frac1\varphi$ प्रबल मूल का व्युत्क्रम है --- गुणांकों की वृद्धि और \omterm{def:b2:powerseries:radius}{अभिसरण
त्रिज्या} एक ही सूचना हैं, बस विपरीत दिशाओं में पढ़ी हुई।
\end{example}

\begin{example}[कातालान संख्याएँ]\label{ex:b2:powerseries:catalan}
\omterm{ex:b2:powerseries:catalan}{कातालान संख्याएँ} $C_n$ (त्रिभुजनों की, कोष्ठकनों की, डिक पथों की संख्या,
\dots) $C_0 = 1$ और $C_{n+1} = \sum_{k=0}^n
C_kC_{n-k}$ पूरा करती हैं। तब \omterm{ex:b2:powerseries:fibonacci}{जनक फलन} $C(x) = \sum C_nx^n$ (\omterm{thm:b2:series:fubini}{कोशी गुणनफल}!) पूरा करता
है
\[
C(x) = 1 + x\,C(x)^2
\quad\Longrightarrow\quad
C(x) = \frac{1 - \sqrt{1 - 4x}}{2x} ,
\]
और $C(0) = 1$ वाला मूल चुनने पर: द्विघात $xC^2 - C + 1 = 0$ को हल करने पर दो उम्मीदवार $\frac{1 \pm
\sqrt{1-4x}}{2x}$
मिलते हैं, और $x \to 0$ होने पर ``$+$'' वाला मूल $\frac1x$ की तरह फट जाता है जबकि
``$-$'' वाला $1$ की ओर जाता है ($\sqrt{1-4x} = 1 - 2x + O(x^2)$ का प्रसार कीजिए) --- अतः केवल ऋण
चिह्न ही $C_0 = 1$ वाली घात श्रेणी उठा सकता है। द्विपद श्रेणी से $\sqrt{1 - 4x}$ का प्रसार
करने पर बंद रूप
\[
C_n = \frac{1}{n+1}\binom{2n}{n} ,
\]
मिलता है, जो \cref{exo:b2:powerseries:8} में करके दिखाया गया है।\index{कातालान संख्याएँ}
\end{example}

\begin{remark}[औपचारिक श्रेणी बनाम अभिसारी श्रेणी]
ऊपर का हर जनक-फलन परिकलन गुणांकों की अद्वितीयता का सहारा लेकर समाप्त होता
है, और वह प्रमेय किसी धनात्मक त्रिज्या वाली चक्रिका के \emph{भीतर} बसती
है: अतः $F_n$ या $C_n$ ``पढ़ लेने'' से पहले $R > 0$ जानना ही पड़ता है। कोई कच्चा
पूर्वगामी परिबंध पर्याप्त है --- फिबोनाच्ची के लिए $F_n \leq 2^n$ (तत्काल आगमन) $R \geq
\frac12$
देता है; और $C_n \leq 4^n$ (हर कातालान संख्या पथों के उपसमुच्चय गिनती है) $R \geq \frac14$ देता
है। पैमाने के पतित सिरे से सावधान रहिए: $\sum n!\,x^n$ की त्रिज्या $0$ है, और उसे
फलन की तरह बरतना निरर्थक है --- ऐसी श्रेणियों वाली सर्वसमिकाएँ गुणांकों
के \emph{औपचारिक} कलन की हैं, जो अपने ही (भिन्न) नियमों वाला विशुद्ध
बीजगणितीय खेल है। इस स्तर पर: पहले धनात्मक त्रिज्या पक्की कीजिए, फिर उसके
भीतर खुलकर परिकलन कीजिए।
\end{remark}

\begin{remark}[इसी खंड के भीतर के परिप्रेक्ष्य]
घात श्रेणियाँ इस पुस्तक की दो महान प्रसार-मशीनों में से एक हैं; दूसरी
हार्मोनिक अध्यायों की फूरिये श्रेणी है, और दोनों की तुलना शिक्षाप्रद है।
घात श्रेणी कठोर है: उसके गुणांक बाध्य हैं ($a_n = f^{(n)}(0)/n!$), उसका अभिसरण निर्मम है
(भीतर सामान्य, बाहर निराशाजनक), और उसका योग \omterm{def:b2:powerseries:analytic}{वैश्लेषिक} है --- अर्थात् अनंत
रूप से कठोर (\cref{def:b2:powerseries:analytic})। फूरिये श्रेणी लचीली है: वह केवल खंडशः-चिकने संकेत भी
निरूपित कर लेती है, पर उसकी क़ीमत चिकनाई की सीमा पर अभिसरण के नाज़ुक
प्रश्न हैं। दोनों सिद्धांत इस अध्याय की सप्ताहांत समस्या में मिलते हैं:
यहाँ सीमा-वृत्त के लिए विकसित चेज़ारो और \omterm{thm:b2:series:abel}{आबेल योग} फूरिये अध्याय में फेयेर
तथा प्वासों कर्नेल के रूप में लौटते हैं। इस बीच अवकल समीकरणों वाला अध्याय
घात श्रेणियों का सीधा उपभोग करता है ($\eu^{tA}$, श्रेणी-हल), और जनक फलनों वाला
अध्याय \cref{ex:b2:powerseries:fibonacci} की युक्ति को प्रायिकताओं के लिए क्रमबद्ध कलन बना देता है।
\end{remark}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:b2:powerseries:1}
अभिसरण त्रिज्याएँ: $\sum \dfrac{n^2}{2^n}z^n$; $\;\sum \dfrac{z^n}{\binom{2n}{n}}$; $\;\sum z^{n!}$; $\;\sum \bigl(2 + (-1)^n\bigr)^n z^n$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:powerseries:2}
दिखाइए कि $\sum z^n$, $\sum \frac{z^n}{n}$, $\sum \frac{z^n}{n^2}$ सबकी त्रिज्या $1$ है परंतु $z = 1$ और $z = -1$ पर उनका
व्यवहार भिन्न है: अपसरण/अपसरण, अपसरण/अभिसरण, अभिसरण/अभिसरण।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:powerseries:3}
$\abs x < 1$ के लिए योग परिकलित कीजिए:
\[
\sum_{n\geq0} n x^n,
\qquad
\sum_{n\geq0} n^2 x^n,
\qquad
\sum_{n\geq0} \frac{x^{2n+1}}{2n+1} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:powerseries:4}
$0$ पर घात श्रेणी में प्रसार कीजिए, त्रिज्या सहित: $\dfrac{1}{(1-x)(2-x)}$ (आंशिक भिन्न); $\;\ln(1 + x + x^2)$
\emph{($1 + x + x^2
= \frac{1 - x^3}{1 - x}$ लिखिए)}।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:powerseries:5}
सिद्ध कीजिए कि $\abs x < 1$ के लिए $f(x) = \sum_{n\geq1} H_n x^n = -\dfrac{\ln(1 -
x)}{1 - x}$, जहाँ $H_n$ हरात्मक संख्या है \emph{($\sum x^n$ और
$\sum \frac{x^n}{n}$ का \omterm{thm:b2:series:fubini}{कोशी गुणनफल})}।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:powerseries:6}
\omterm{ex:b2:powerseries:fibonacci}{जनक फलन} से पुनरावृत्ति $u_0 = 1$, $u_{n+1} =
2u_n + n$ हल कीजिए: $U(x) = \sum u_nx^n$ का बंद रूप परिकलित कीजिए,
उसका अपघटन कीजिए, और $u_n = 2^{n+1} - n - 1$ पढ़ लीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:powerseries:7}
$x \neq 0$ के लिए $f(x) = \eu^{-1/x^2}$ लीजिए, और $f(0) = 0$। सिद्ध कीजिए कि $f$ $\R$ पर $C^\infty$ है और सभी
$n$ के लिए $f^{(n)}(0) = 0$ \emph{(आगमन से दिखाइए कि बहुपदों $P_n$ के लिए $f^{(n)}(x) = P_n\bigl(\frac1x\bigr)
\eu^{-1/x^2}$, और
वृद्धि-तुलना का उपयोग कीजिए)}। निष्कर्ष निकालिए कि $f$ $0$ पर \omterm{def:b2:powerseries:analytic}{वैश्लेषिक}
नहीं है: $0$ पर उसकी टेलर श्रेणी अभिसरित तो होती है --- पर ग़लत फलन पर।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:powerseries:8}
\cref{ex:b2:powerseries:catalan} पूरा कीजिए: द्विपद श्रेणी से $\sqrt{1 - 4x}$ का प्रसार कीजिए और दिखाइए
\[
\binom{1/2}{n+1}(-4)^{n+1} = -\frac{2}{n+1}\binom{2n}{n},
\]
और $C_n = \frac{1}{n+1}\binom{2n}{n}$ निष्कर्ष रूप में निकालिए; $C(x)$ की \omterm{def:b2:powerseries:radius}{अभिसरण त्रिज्या} तथा स्टर्लिंग के
द्वारा $C_n$ की अनंतस्पर्शिता निर्धारित कीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:powerseries:9}
(आबेल की त्रिज्यीय सीमा प्रमेय, विशेष स्थिति) मान लीजिए $\sum a_n$ अभिसरित होती
है। सिद्ध कीजिए कि $\lim_{x \to 1^-} \sum_{n} a_n x^n = \sum_n
a_n$। \emph{(\omterm{thm:b2:series:abel}{आबेल योग}: आंशिक योग $A_n$ और $A =
\lim A_n$ के साथ $\sum a_nx^n = (1 - x)\sum A_n x^n$
लिखिए; फिर $\sum
a_nx^n - A = (1-x)\sum (A_n - A)x^n$, और योग को किसी बड़े $N$ पर बाँटिए।)} अनुप्रयोग: घात
श्रेणी से पुनः सिद्ध $\sum \frac{(-1)^{n-1}}{n} = \ln 2$ और $\sum
\frac{(-1)^n}{2n+1} = \frac\pi4$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:powerseries:10}
दिखाइए कि $0 < \abs x < 1$ के लिए $\displaystyle\sum_{n\geq1}\frac{x^n}{n(n+1)} = 1 +
\frac{1-x}{x}\,\ln(1-x)$, त्रिज्या निर्धारित कीजिए, और जाँचिए कि अभिसरण
$\intcc{-1}{1}$ पर सामान्य है; तथा सत्यापित कीजिए कि संततता से $x = 1$ पर पूर्वानुमानित
मान क्रमिक निरसन योग $\sum
\frac{1}{n(n+1)} = 1$ से मेल खाता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:powerseries:11}
(अव्यवस्थापन) मान लीजिए $D_n$ $n$ वस्तुओं के उन क्रमचयों की संख्या है
जिनका कोई अचर बिंदु न हो ($D_0 = 1$)। $\{1, \dots, n\}$ के क्रमचयों को उनके अचर-बिंदु
समुच्चय के अनुसार छाँटने पर $n! = \sum_{k=0}^{n}\binom nk D_{n-k}$ मिलता है। $\frac{x^n}{n!}$ से गुणा कीजिए, योग लीजिए,
और \omterm{thm:b2:series:fubini}{कोशी गुणनफल} पहचानकर घातांकी \omterm{ex:b2:powerseries:fibonacci}{जनक फलन}
\[
\sum_{n\geq0} D_n\,\frac{x^n}{n!} = \frac{\eu^{-x}}{1-x}
\qquad (\abs x < 1),
\]
प्राप्त कीजिए; फिर बंद रूप $\dfrac{D_n}{n!} =
\sum_{k=0}^{n}\dfrac{(-1)^k}{k!}$ और सीमा $\dfrac{D_n}{n!}
\to \eu^{-1}$ पढ़ लीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:powerseries:12}
\cref{met:b2:powerseries:ode} की द्विपद श्रेणी से सिद्ध कीजिए
\[
\frac{1}{\sqrt{1 - 4x}} = \sum_{n\geq0}\binom{2n}{n}x^n
\qquad \Bigl(\abs x < \frac14\Bigr),
\]
और वर्ग करके ($\frac{1}{1-4x} = \sum 4^nx^n$ के विरुद्ध \omterm{thm:b2:series:fubini}{कोशी गुणनफल}) संवलन सर्वसमिका निष्कर्ष रूप
में निकालिए
\[
\sum_{k=0}^{n}\binom{2k}{k}\binom{2n-2k}{n-k} = 4^n .
\]
\end{exercise}

\section{समस्या: आबेल, टाउबर, और अभिसरण की सीमा}

\begin{problem}\label{pb:b2:powerseries:1}
अभिसरण चक्रिका के भीतर सब कुछ आसान है; घात श्रेणियों का सारा नाटक सीमा
\emph{पर} होता है। यह समस्या वास्तविक चर में सीमा का सिद्धांत बनाती है:
आबेल की प्रमेय अपने एकसमान रूप में, टाउबर की शर्त के अंतर्गत उसका विलोम,
योगकरण विधियों की चेज़ारो--आबेल पदानुक्रम (फ्रोबेनियस की प्रमेय सहित),
सीमा तक पद-दर-पद समाकलन जिसके लाभ में क्लासिक अचर मिलते हैं, और अंत में
\omterm{def:b2:powerseries:analytic}{वैश्लेषिक} फलनों की कठोरता --- तादात्म्य प्रमेय।\index{आबेल की
प्रमेय}\index{टाउबर की प्रमेय} आगे सर्वत्र $(a_n)$ कोई वास्तविक अनुक्रम है,
$f(x) = \sum_{n\geq0}
a_nx^n$, और $A_n = a_0 + \dots + a_n$।

\medskip
\noindent\textbf{भाग I --- आबेल की प्रमेय, एकसमान रूप में।} इस भाग में
मान लीजिए $\sum a_n$ अभिसरित होती है, और $r_n =
\sum_{k\geq n} a_k$ रखिए (अतः $r_n \to 0$ और $a_n = r_n -
r_{n+1}$)।
\begin{enumerate}
  \item खंडशः योग से सिद्ध कीजिए कि सभी $0 \leq x
        \leq 1$ और $N \leq M$ के लिए:
        \[
        \Bigl|\sum_{n=N}^{M} a_n x^n\Bigr|
        \leq 2\sup_{n \geq N}\,\abs{r_n} .
        \]
  \item इससे निष्कर्ष निकालिए कि $\sum a_nx^n$ $\intcc{0}{1}$ पर \emph{एकसमान} रूप से
        अभिसरित होती है, कि उसका योग वहाँ \omterm{def:b2:metric:continuity}{संतत} है, और \cref{exo:b2:powerseries:9} की त्रिज्यीय
        सीमा पुनः प्राप्त कीजिए: $x
        \to 1^-$ होने पर $f(x) \to \sum a_n$।
  \item (कोशी गुणनफलों के लिए आबेल की प्रमेय) मान लीजिए $\sum a_n = A$, $\sum b_n = B$, और
        \omterm{thm:b2:series:fubini}{कोशी गुणनफल} $\sum
        c_n$, $c_n = \sum_{k} a_kb_{n-k}$, \emph{अभिसरित} होता है और उसका योग $C$
        है। सिद्ध कीजिए कि $C = AB$ \emph{(चक्रिका के भीतर गुणनफल सर्वसमिका
        \cref{prop:b2:powerseries:operations} से सत्य है; अब $x \to
        1^-$ लीजिए)}।
  \item दिखाइए कि परिकल्पना मायने रखती है: $a_n = b_n =
        \frac{(-1)^n}{\sqrt{n+1}}$ के लिए दोनों श्रेणियाँ
        अभिसरित होती हैं, फिर भी $\abs{c_n} \geq \frac{2(n+1)}{n+2} \geq 1$ \emph{(हर गुणनखंड $\sqrt{(k+1)(n-k+1)}$ को
        समांतर--गुणोत्तर माध्य से परिबद्ध कीजिए)}: अर्थात् दो अभिसारी
        श्रेणियों का \omterm{thm:b2:series:fubini}{कोशी गुणनफल} अपसरित हो सकता है।
  \item (\cref{exo:b2:powerseries:5} का एक लाभ) दिखाइए कि $\intoo{-1}{1}$ पर $\bigl(\ln(1-x)\bigr)^2 = 2\sum_{n\geq1}
        \frac{H_n}{n+1}x^{n+1}$, जाँचिए कि $\bigl(\frac{H_n}{n+1}\bigr)_{n\geq1}$ घटकर
        $0$ तक जाता है, और आबेल के साथ निष्कर्ष निकालिए:
        \[
        \sum_{n\geq1} (-1)^{n+1}\,\frac{H_n}{n+1}
        = \frac{(\ln 2)^2}{2} .
        \]
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- टाउबर का विलोम।} $\sum
a_n$ को $L$ पर
\emph{आबेल-योग्य} कहिए जब $x \to
1^-$ होने पर $f(x) \to L$।
\begin{enumerate}[resume]
  \item दिखाइए कि $\sum (-1)^n$ $\frac12$ पर आबेल-योग्य है फिर भी अपसारी: अर्थात् आबेल
        की प्रमेय का कोई अशर्त विलोम नहीं है।
  \item (चेज़ारो प्रमेयिका) यदि $u_n \to 0$ तो $\frac{u_1 + \dots
        + u_N}{N} \to 0$ \emph{(योग को किसी स्थिर $m$
        पर बाँटिए)}।
  \item अब मान लीजिए $n\,a_n \to 0$ और $f(x) \to L$। $x_N =
        1 - \frac1N$ के साथ दोनों आकलन सिद्ध कीजिए
        \[
        \Bigl|\sum_{n=0}^{N} a_n\bigl(1 - x_N^n\bigr)\Bigr|
        \leq \frac{1}{N}\sum_{n=1}^{N} n\,\abs{a_n},
        \qquad
        \Bigl|\sum_{n>N} a_n x_N^n\Bigr|
        \leq \sup_{n>N}\bigl(n\abs{a_n}\bigr)
        \]
        \emph{(पहले के लिए $1 - x^n \leq n(1-x)$; और दूसरे के लिए $\abs{a_n} \leq \frac{1}{N}\sup_{m>N}
        m\abs{a_m}$ तथा $\sum x_N^n \leq N$)}।
  \item \emph{टाउबर की प्रमेय} निष्कर्ष रूप में निकालिए: यदि $n\,a_n \to 0$ हो और
        $\sum a_n$ $L$ पर आबेल-योग्य हो, तो $\sum a_n$ $L$ पर अभिसरित होती है।
  \item (धनात्मक गुणांकों के लिए सरल टाउबरीय परिणाम) यदि $a_n
        \geq 0$ हो और $f$
        $\intco{0}{1}$ पर परिबद्ध हो, तो दिखाइए कि $\sum a_n$ अभिसरित होती है और $\sum a_n = \lim_{x\to1^-}
        f(x)$
        \emph{($\sum_{n\leq N}a_nx^n \leq f(x)$ को परिबद्ध कीजिए और $x \to 1^-$ लीजिए, फिर आबेल का उपयोग
        कीजिए)}।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- चेज़ारो माध्य और फ्रोबेनियस की प्रमेय।}
$\sum a_n$ को $L$ पर \emph{चेज़ारो-योग्य} कहिए जब $\sigma_N = \frac{A_0 + \dots + A_{N-1}}{N} \to L$।
\begin{enumerate}[resume]
  \item दिखाइए कि अभिसारी श्रेणी अपने योग पर चेज़ारो-योग्य होती है
        \emph{($A_n - L$ पर लगाया गया प्रश्न 7)}।
  \item $\sum(-1)^n$ का चेज़ारो मान परिकलित कीजिए और जाँचिए कि वह प्रश्न 6 के
        आबेल मान $\frac12$ से मेल खाता है।
  \item $S_n = A_0 + \dots + A_n = (n+1)\,\sigma_{n+1}$ के साथ $0 \leq x < 1$ के लिए दोनों सर्वसमिकाएँ सिद्ध कीजिए:
        \[
        f(x) = (1-x)^2\sum_{n\geq0}(n+1)\,\sigma_{n+1}x^n,
        \qquad
        (1-x)^2\sum_{n\geq0}(n+1)x^n = 1 .
        \]
  \item (फ्रोबेनियस) निष्कर्ष निकालिए: यदि $\sigma_N \to L$ हो तो $x \to 1^-$ होने पर $f(x) \to
        L$ ---
        अर्थात् चेज़ारो-योग्य होने से आबेल-योग्य होना निकलता है, और उसी
        मान पर \emph{(दोनों सर्वसमिकाएँ घटाइए और योग को किसी बड़े $N$ पर
        बाँटिए, जैसा \cref{exo:b2:powerseries:9} में)}।
  \item दिखाइए कि पदानुक्रम
        \[
        \text{अभिसारी} \;\Longrightarrow\;
        \text{चेज़ारो-योग्य} \;\Longrightarrow\;
        \text{आबेल-योग्य}
        \]
        दोनों तीरों पर यथार्थ है: पहले के लिए प्रश्न 6; और दूसरे के लिए
        दिखाइए कि $\sum(-1)^n(n+1)$ $\frac14$ पर आबेल-योग्य है ($f$ परिकलित कीजिए) पर
        चेज़ारो-योग्य नहीं (सम और विषम $N$ के लिए $\sigma_N$ अलग-अलग परिकलित
        कीजिए)।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- सीमा तक समाकलन।}
\begin{enumerate}[resume]
  \item मान लीजिए $\sum a_nx^n$ $\intco{0}{1}$ पर अभिसरित होती है और $\sum \frac{a_n}{n+1}$ अभिसरित होती है।
        सिद्ध कीजिए कि \omterm{def:b2:integration:improper}{अनुचित समाकल} $\int_0^1 f$ विद्यमान है और
        \[
        \int_0^1 \Bigl(\sum_{n\geq0} a_nx^n\Bigr)\dd x
        = \sum_{n\geq0}\frac{a_n}{n+1}
        \]
        \emph{(प्रतिअवकलज $F(x) = \sum\frac{a_n}{n+1}x^{n+1}$ भाग I के अनुसार $1$ पर \omterm{def:b2:metric:continuity}{संतत} है)}।
  \item $\eta = \sum_{n\geq1}\frac{(-1)^{n-1}}{n^2}$ लीजिए। दिखाइए कि $\int_0^1 \frac{\ln(1+x)}{x}\dd x = \eta$, और \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसारी $\sum \frac1{n^2}$ में सम
        तथा विषम सूचकांक बाँटकर $\eta =
        \frac12\sum_{n\geq1}\frac{1}{n^2}$। (फूरिये अध्याय की सप्ताहांत समस्या
        $\sum\frac1{n^2} = \frac{\pi^2}{6}$ का मूल्यांकन करती है।)
  \item सिद्ध कीजिए
        \[
        \sum_{n\geq0}\frac{(-1)^n}{3n+1}
        = \int_0^1\frac{\dd x}{1+x^3}
        = \frac13\Bigl(\ln 2 + \frac{\pi}{\sqrt3}\Bigr)
        \]
        \emph{(श्रेणी लाइब्नित्स से अभिसरित होती है; प्रश्न 16 के साथ
        गुणोत्तर श्रेणी $\sum(-1)^nx^{3n}$ का समाकलन कीजिए; फिर आंशिक भिन्न: $\frac{1}{1+x^3} =
        \frac{1/3}{1+x} + \frac{(2-x)/3}{x^2-x+1}$)}।
  \item $(1-t)^{-1/2}$ (\cref{exo:b2:powerseries:12}) की द्विपद श्रेणी से निकालिए
        \[
        \arcsin x = \sum_{n\geq0}
        \frac{\binom{2n}{n}}{4^n(2n+1)}\,x^{2n+1}
        \quad(\abs x < 1),
        \qquad\text{फिर}\qquad
        \sum_{n\geq0}\frac{\binom{2n}{n}}{4^n(2n+1)}
        = \frac\pi2 ,
        \]
        और सीमा-मान को $\intcc{-1}{1}$ पर \emph{सामान्य} अभिसरण से न्यायोचित
        ठहराइए ($\binom{2n}n4^{-n} \sim \frac{1}{\sqrt{\pi n}}$, \cref{ex:b2:comparison:centralbinomial} का उपयोग कीजिए) --- यहाँ आबेल की भी आवश्यकता
        नहीं पड़ती।
  \item (सीमा पर कातालान) दिखाइए कि $\sum C_n 4^{-n} =
        2$: \cref{ex:b2:powerseries:catalan} की कातालान श्रेणी अपनी
        त्रिज्या $\frac14$ \emph{पर} अभिसरित होती है (\cref{exo:b2:powerseries:8} की
        अनंतस्पर्शिता), उसका योग $\intcc{0}{\frac14}$ पर \omterm{def:b2:metric:continuity}{संतत} है, और बंद रूप की वहाँ
        सीमा $2$ है।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- कठोरता: तादात्म्य प्रमेय।}
\begin{enumerate}[resume]
  \item (विलगित शून्य) मान लीजिए $f = \sum a_nx^n$ की त्रिज्या $R >
        0$ है और सारे $a_n = 0$ शून्य
        नहीं हैं; और $a_m \neq 0$ वाला न्यूनतम सूचकांक $m$ लीजिए। दिखाइए कि $f(x) = x^m g(x)$,
        जहाँ $g$ त्रिज्या $R$ वाली कोई घात श्रेणी है और $g(0) = a_m \neq 0$, और इससे
        निष्कर्ष निकालिए कि $0$ के किसी छिद्रित प्रतिवेश में $f$ का कोई
        शून्य नहीं है।
  \item (तादात्म्य प्रमेय) मान लीजिए $f, h$ $0$ के पास घात श्रेणियों के
        योग हैं और $(x_k)$ \emph{अशून्य} बिंदुओं का ऐसा अनुक्रम है कि $x_k \to 0$
        और $f(x_k) = h(x_k)$। सिद्ध कीजिए कि $f$ और $h$ के गुणांक एक ही हैं, अतः वे
        $0$ के पास मेल खाते हैं।
  \item $0$ के पास \omterm{def:b2:powerseries:analytic}{वैश्लेषिक} वे \emph{सभी} फलन $f$ ज्ञात कीजिए जिनके
        लिए
        \[
        f\Bigl(\frac1k\Bigr) = \frac{k^2}{k^2+1}
        \qquad\text{सभी बड़े पूर्णांक } k .
        \text{ के लिए}\]
  \item दिखाइए कि किसी \omterm{def:b2:metric:topology}{विवृत} अंतराल $I$ पर \omterm{def:b2:powerseries:analytic}{वैश्लेषिक फलन}, जो किसी
        उपअंतराल पर लुप्त हो, $I$ पर सर्वथा लुप्त होता है \emph{(जिन
        बिंदुओं के चारों ओर $f$ सर्वथा लुप्त है उनका समुच्चय \omterm{def:b2:metric:topology}{विवृत} है,
        और संचय-बिंदुओं पर तादात्म्य प्रमेय लगाने से $I$ में संवृत भी)}।
        निष्कर्ष निकालिए कि $\R$ पर \omterm{def:b2:metric:compact}{संहत} वाहक वाला कोई अशून्य \omterm{def:b2:powerseries:analytic}{वैश्लेषिक
        फलन} नहीं है --- जबकि $C^\infty$ उभार फलन विद्यमान हैं (\cref{exo:b2:powerseries:7} निर्माण-खंड
        दे देता है): अर्थात् वैश्लेषिकता कठोर है, चिकनाई ढीली।
  \item संश्लेषण। एक-एक वाक्य में: (क) आबेल की प्रमेय \cref{lem:b2:powerseries:abel} के
        सामान्य-अभिसरण पैकेट में क्या जोड़ती है; (ख) विलोम ठीक किन
        परिकल्पनाओं के अंतर्गत सत्य है (टाउबर) और बीच का पायदान कौन-सा है
        (फ्रोबेनियस); (ग) भाग IV का कोई एक सीमा-अचर जिसे आप अब किसी मित्र
        के लिए दो पंक्तियों में निकाल सकते हैं; (घ) इस पुस्तक में चेज़ारो
        माध्य कहाँ फिर प्रकट होंगे, और बिलकुल भिन्न प्रकार की श्रेणियों
        के लिए।
\end{enumerate}
\end{problem}
