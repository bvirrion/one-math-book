\chapter{दूरिक समष्टियों की सांस्थिति}\label{ch:b2:metric}

वास्तविक रेखा की सांस्थिति (प्रथम वर्ष का खंड) लगभग एक भी शब्द बदले बिना
किसी भी ऐसे समुच्चय पर सामान्यीकृत हो जाती है जिस पर कोई दूरी दी गई हो।
लाभ बहुत बड़ा है: फलनों, आव्यूहों और वक्रों के अनुक्रम --- सब दूरिक
समष्टियों के बिंदु बन जाते हैं, और यहाँ सिद्ध किए गए तीनों स्तंभ ---
बानाख अचल बिंदु प्रमेय के साथ पूर्णता, संहतता, संबद्धता --- उन सब पर एक
समान लागू होते हैं। यह अध्याय पुस्तक के पूरे विश्लेषण-भाग की रीढ़ है।

\section{दूरिक समष्टियाँ}

\begin{definition}\label{def:b2:metric:def}
\emph{दूरिक समष्टि}\index{दूरिक समष्टि} एक समुच्चय $X$ है जिस पर ऐसा
प्रतिचित्रण $d
\colon X \times X \to \R_+$ दिया हो कि सभी $x, y, z$ के लिए:
\[
d(x,y) = 0 \iff x = y,
\qquad
d(x,y) = d(y,x),
\qquad
d(x,z) \leq d(x,y) + d(y,z).
\]
गोले: $B(a, r) = \{x : d(a,x) < r\}$ (\omterm{def:b2:metric:topology}{विवृत}), $\overline B(a,r) =
\{x : d(a,x) \leq r\}$ (संवृत)। कोई उपसमुच्चय $A \subseteq X$ प्रेरित दूरी के साथ
स्वयं दूरिक समष्टि बन जाता है।
\end{definition}

\begin{example}\label{ex:b2:metric:examples}
$\abs{x - y}$ सहित $\R$; $\R^n$ निम्नलिखित में से किसी भी दूरी के साथ
\[
d_1(x,y) = \sum_i \abs{x_i - y_i},
\quad
d_2(x,y) = \Bigl(\sum_i (x_i - y_i)^2\Bigr)^{1/2},
\quad
d_\infty(x,y) = \max_i \abs{x_i - y_i};
\]
\omterm{def:b2:metric:continuity}{संतत} फलनों का समुच्चय $C(\intcc{a}{b})$ \emph{उच्चतम दूरी} $d_\infty(f, g) = \sup_{\intcc{a}{b}} \abs{f -
g}$ के साथ (जो परिमित है:
$f - g$ परिबद्ध होता है); और कोई भी समुच्चय \emph{विविक्त} दूरी के साथ
($x \neq y$ के लिए $d(x,y) = 1$)। मानदंडों से आने वाली दूरियाँ \cref{ch:b2:nvs} का विषय हैं।
\end{example}

\begin{definition}[दूरिक समष्टि की सांस्थिति]\label{def:b2:metric:topology}
$U \subseteq X$ \emph{विवृत}\index{विवृत समुच्चय!दूरिक समष्टि में} कहलाता है जब $U$
का हर बिंदु किसी ऐसे गोले का केंद्र हो जो $U$ में समाहित है; और $F$
\emph{संवृत} कहलाता है जब उसका पूरक विवृत हो। प्रतिवेश, अभ्यंतर,
संवृतक, सघनता और परिसीमा ठीक वैसे ही परिभाषित होते हैं जैसे वास्तविक
रेखा पर (प्रथम वर्ष का खंड), बस अंतरालों के स्थान पर गोले आ जाते हैं;
और वहाँ सिद्ध किए गए कथन --- विवृत समुच्चयों के सम्मिलन/प्रतिच्छेदन,
अभ्यंतर और संवृतक के अभिलक्षण, सबसे छोटे संवृत अधिसमुच्चय के रूप में
संवृतक --- उन्हीं उपपत्तियों के साथ यहाँ भी चलते हैं। विवृत गोले विवृत
होते हैं और संवृत गोले संवृत (त्रिभुज असमिका)।
\end{definition}

\begin{example}[एक ही समुच्चय पर अभ्यंतर, संवृतक और परिसीमा]\label{ex:b2:metric:intclosure}
$\R$ में मान लीजिए $A = \intoc{0}{1} \cup \{2\}$। अभ्यंतर: $\intoo{0}{1}$ --- किसी भी $x \in \intoo01$ के चारों ओर छोटा
गोला $A$ में रहता है; $1$ के चारों ओर हर गोला $\intoo{1-r}{1+r}$ दाईं ओर से $A$ के
बाहर निकल जाता है, अतः $1$ अभ्यंतरीय नहीं है; और अलग-थलग पड़ा $2$ भी
अभ्यंतरीय नहीं। संवृतक: $\intcc{0}{1} \cup \{2\}$ (बिंदु $0$ $A$ की सीमा है, और कुछ नहीं
जुड़ता)। परिसीमा (संवृतक में से अभ्यंतर घटाकर): $\{0, 1, 2\}$। स्मरण रखने योग्य
असमरूपताएँ: कोई अंत्यबिंदु समुच्चय का सदस्य हो सकता है और फिर भी
अभ्यंतरीय न हो ($1$), संलग्न हो सकता है और सदस्य न हो ($0$), और
अलग-थलग बिंदु स्वयं अपनी परिसीमा होता है ($2$)। यही लेखा-जोखा किसी भी
\omterm{def:b2:metric:def}{दूरिक समष्टि} में अक्षरशः चलता है, अंतरालों के स्थान पर गोले लेकर।
\end{example}

\begin{definition}[सीमाएँ, संततता]\label{def:b2:metric:continuity}
$X$ में $x_n \to x$ तब कहा जाता है जब $d(x_n, x) \to 0$। दूरिक समष्टियों के बीच कोई
प्रतिचित्रण $f \colon X \to Y$ $a$ पर \emph{संतत}\index{संतत} कहलाता है जब
\[
\forall \varepsilon > 0,\ \exists\delta > 0,\quad
d_X(x, a) \leq \delta \implies d_Y\bigl(f(x), f(a)\bigr) \leq
\varepsilon ;
\]
और समतुल्य रूप से ($\R$ वाली वही उपपत्ति), जब हर अनुक्रम $x_n \to a$ के लिए
$f(x_n) \to f(a)$ हो। $f$ अचर $k$ के साथ
\emph{लिप्शिट्स}\index{लिप्शिट्स} कहलाता है जब सदा $d_Y(f(x), f(y)) \leq k\, d_X(x, y)$ हो --- तब वह
एकसमान संतत होता है, अतः संतत भी।
\end{definition}

\begin{theorem}[संततता का वैश्विक अभिलक्षण]\label{thm:b2:metric:globalcontinuity}
$f \colon X \to Y$ (हर बिंदु पर) \omterm{def:b2:metric:continuity}{संतत} है यदि और केवल यदि हर \omterm{def:b2:metric:topology}{विवृत} समुच्चय का
पूर्वप्रतिबिंब \omterm{def:b2:metric:topology}{विवृत} हो --- और यदि और केवल यदि हर संवृत समुच्चय का
पूर्वप्रतिबिंब संवृत हो।
\end{theorem}

\begin{proof}
($\Rightarrow$) मान लीजिए $V \subseteq Y$ \omterm{def:b2:metric:topology}{विवृत} है और $a \in f^{-1}(V)$: तो कोई गोला $B(f(a), \varepsilon) \subseteq V$; और $a$ पर संततता
$f(B(a, \delta)) \subseteq B(f(a),
\varepsilon)$ सहित $\delta$ देती है, अतः $B(a, \delta) \subseteq f^{-1}(V)$।

($\Leftarrow$) $a$ और $\varepsilon$ दिए हों: $f^{-1}\bigl(B(f(a),
\varepsilon)\bigr)$ \omterm{def:b2:metric:topology}{विवृत} है और उसमें $a$ है, अतः उसमें कोई
गोला $B(a, \delta)$ भी है: यही $a$ पर संततता की परिभाषा है। संवृत समुच्चयों के
लिए: पूरक लीजिए (\cref{prop:b2:structures:images})।
\end{proof}

\section{पूर्ण समष्टियाँ}

\begin{definition}\label{def:b2:metric:complete}
कोई अनुक्रम $(x_n)$ \emph{कोशी} कहलाता है जब $N \to \infty$ होने पर $\sup_{p, q \geq N} d(x_p,
x_q) \to 0$। कोई \omterm{def:b2:metric:def}{दूरिक
समष्टि} \emph{पूर्ण}\index{पूर्ण दूरिक समष्टि} कहलाती है जब उसका हर कोशी
अनुक्रम अभिसरित हो। अभिसारी $\Rightarrow$ सदा कोशी; पूर्ण समष्टियों के संवृत
उपसमुच्चय पूर्ण होते हैं, और किसी भी समष्टि के पूर्ण उपसमुच्चय संवृत
(वही उपपत्तियाँ जो $\R$ पर थीं: प्रथम वर्ष का खंड)।
\end{definition}

\begin{example}[कोशी, पर बिना सीमा के]\label{ex:b2:metric:cauchynolimit}
सामान्य दूरी वाले $X = \Q$ में $\sqrt2$ के दशमलव छेदन
\[
x_0 = 1,\quad x_1 = 1.4,\quad x_2 = 1.41,\quad x_3 = 1.414,
\quad\dots
\]
$\abs{x_p - x_q} \leq 10^{-\min(p,q)}$ पूरा करते हैं: अर्थात् $\Q$ में कोशी। $\Q$ में कोई सीमा $\R$ में भी
सीमा होती, अर्थात् $\sqrt2 \notin \Q$: अतः $X$ में कोई सीमा नहीं है। अपूर्णता ऐसी ही
``प्रेत सीमाओं'' की उपस्थिति है; और प्रथम वर्ष के खंड में $\R$ की
पूर्णता ठीक इसीलिए गढ़ी गई थी कि हर कोशी अनुक्रम को घर मिल जाए।
\end{example}

\begin{theorem}\label{thm:b2:metric:rncomplete}
$\R^n$ (\cref{ex:b2:metric:examples} की तीनों दूरियों में से कोई भी) और $\bigl(C(\intcc{a}{b}),
d_\infty\bigr)$ \omterm{def:b2:metric:complete}{पूर्ण} हैं।
\end{theorem}

\begin{proof}
$\R^n$: कोई कोशी अनुक्रम हर निर्देशांक में कोशी होता है (तीनों दूरियों के
लिए हर $\abs{x_i - y_i} \leq d(x,y)$), अतः हर निर्देशांक अभिसरित होता है ($\R$ की पूर्णता, प्रथम
वर्ष का खंड), और निर्देशांक-दर-निर्देशांक अभिसरण से $d_\infty$ के लिए अभिसरण
निकलता है (निर्देशांक परिमित हैं), अतः तीनों के लिए (तीनों दूरियाँ अचर
गुणकों के भीतर एक-दूसरे पर हावी हैं: $d_\infty
\leq d_2 \leq d_1 \leq n\,d_\infty$)।

$C(\intcc{a}{b})$: मान लीजिए $(f_n)$ $d_\infty$-कोशी है। प्रत्येक $x$ के लिए $(f_n(x))$ $\R$ में कोशी है
($\abs{f_p(x) - f_q(x)} \leq
d_\infty(f_p, f_q)$): अतः वह किसी $f(x)$ पर अभिसरित होता है। $q \to \infty$ होने पर $\abs{f_p(x) - f_q(x)} \leq \varepsilon$ में सीमा
लेने पर ($p, q
\geq N_\varepsilon$, सभी $x$ के लिए मान्य): सभी $x$ के लिए $\abs{f_p(x) -
f(x)} \leq \varepsilon$, अर्थात्
$d_\infty(f_p, f) \leq
\varepsilon$: अर्थात् एकसमान अभिसरण। सीमा \omterm{def:b2:metric:continuity}{संतत} है: $\varepsilon$ दिया हो तो $\sup\abs{f_p - f} \leq \varepsilon$ वाला
$p$ चुनिए, फिर $a$ पर $f_p$ की संततता और तीन-पद विभाजन
\[
\abs{f(x) - f(a)} \leq \abs{f(x) - f_p(x)} + \abs{f_p(x) - f_p(a)}
+ \abs{f_p(a) - f(a)} \leq 3\varepsilon
\]
का उपयोग $a$ के निकट $x$ के लिए कीजिए। (यह ``$3\varepsilon$ तर्क'' \cref{ch:b2:funcseq} की
एकसमान-सीमा प्रमेय के रूप में लौटता है।)
\end{proof}

\begin{example}[संततता से पहचाने गए विवृत और संवृत समुच्चय]\label{ex:b2:metric:recognize}
वैश्विक अभिलक्षण (\cref{thm:b2:metric:globalcontinuity}) सांस्थितिक लेखा-जोखा का रोज़मर्रा का औज़ार है।
$\R^2$ में: समुच्चय $\{(x, y) : x^2 + y^2
< 1,\ y > x^3\}$ \omterm{def:b2:metric:topology}{विवृत} है --- वह \omterm{def:b2:metric:continuity}{संतत} $g(x,y) =
x^2 + y^2$ और $h(x, y) = y - x^3$ के लिए $g^{-1}(\intoo{-\infty}{1})
\cap h^{-1}(\intoo{0}{+\infty})$ है,
अर्थात् दो \omterm{def:b2:metric:topology}{विवृत} पूर्वप्रतिबिंबों का प्रतिच्छेदन। $\bigl(C(\intcc01), d_\infty\bigr)$ में: $f(0) = f(1)$ और $\int_0^1 f = 0$
वाले फलनों का समुच्चय संवृत है --- वह $\R^2$ में जाने वाले \omterm{def:b2:metric:continuity}{संतत}
प्रतिचित्रण $f \mapsto
\bigl(f(0) - f(1),\ \int_0^1 f\bigr)$ के अंतर्गत $\{(0,0)\}$ का पूर्वप्रतिबिंब है (हर निर्देशांक
$1$-लिप्शिट्स है, जैसा \cref{exo:b2:metric:3} में)। यह विधि कभी चित्र नहीं बनाती: कोई
\omterm{def:b2:metric:continuity}{संतत} प्रतिचित्रण दिखाइए, समुच्चय को पूर्वप्रतिबिंब की तरह पढ़िए, और
प्रमेय उद्धृत कीजिए।
\end{example}

\begin{example}[अनंत शर्तों से परिभाषित एक संवृत समुच्चय]\label{ex:b2:metric:lipclosed}
$\bigl(C(\intcc01), d_\infty\bigr)$ में $1$-लिप्शिट्स फलनों का समुच्चय
\[
L = \{f : \abs{f(x) - f(y)} \leq \abs{x - y}
\ \text{ सभी } x, y\}
\]
संवृत है, यद्यपि वह अगणनीय रूप से अनेक शर्तों से काटा गया है: प्रत्येक
स्थिर युग्म $(x, y)$ के लिए प्रतिचित्रण $f \mapsto \abs{f(x) - f(y)} - \abs{x - y}$ \omterm{def:b2:metric:continuity}{संतत} है (मूल्यांकन
$1$-लिप्शिट्स होते हैं), अतः हर एकल शर्त कोई संवृत समुच्चय परिभाषित
करती है, और $L$ इसी कुल का \emph{प्रतिच्छेदन} है --- और संवृत
समुच्चयों का कोई भी प्रतिच्छेदन संवृत होता है। यही खाका एकदिष्ट फलनों,
उत्तल फलनों तथा किसी स्थिर $g$ से परिबद्ध फलनों के लिए भी संवृतता
प्रमाणित कर देता है: एकसमान सीमाएँ हर वह गुणधर्म विरासत में पाती हैं जो
बिंदुवार संवृत प्रतिबंधों के किसी कुल के रूप में व्यक्त हो सके। और
एकसमान सीमाएँ जो अपने आप विरासत में \emph{नहीं} पातीं --- जैसे
अवकलनीयता --- ठीक वही \cref{ch:b2:funcseq} को मेहनत करके साधना पड़ता है।
\end{example}

\begin{theorem}[बानाख अचल बिंदु प्रमेय]\label{thm:b2:metric:banach}
मान लीजिए $X$ अरिक्त \omterm{def:b2:metric:complete}{पूर्ण दूरिक समष्टि} है और $f \colon X \to X$ एक \emph{संकुचन}:
अर्थात् अचर $k < 1$ के साथ \omterm{def:b2:metric:continuity}{लिप्शिट्स}। तब $f$ का अद्वितीय अचल बिंदु $\ell$ है,
और हर कक्षा $x_{n+1} = f(x_n)$ $\ell$ पर अभिसरित होती है, जहाँ\index{बानाख अचल बिंदु
प्रमेय}
\[
d(x_n, \ell) \leq \frac{k^n}{1 - k}\, d(x_1, x_0) .
\]
\end{theorem}

\begin{proof}
अद्वितीयता: दो अचल बिंदुओं के बीच की दूरी अपने ही $\leq k$ गुने के बराबर हो
जाती है। अस्तित्व: आगमन से $d(x_{n+1}, x_n) \leq k^n d(x_1, x_0)$, अतः $q > p$ के लिए
\[
d(x_q, x_p) \leq \sum_{j=p}^{q-1} d(x_{j+1}, x_j)
\leq d(x_1, x_0) \sum_{j \geq p} k^j
= \frac{k^p}{1-k}\, d(x_1, x_0) \xrightarrow[p\to\infty]{} 0 :
\]
अर्थात् कोशी, इसलिए किसी $\ell$ पर अभिसारी; और $f$ की संततता $x_{n+1} = f(x_n)$ को सीमा
तक ले जाती है: $\ell = f(\ell)$। त्रुटि-परिबंध $q \to \infty$ के साथ यही प्रदर्शित आकलन है।
\end{proof}

\begin{example}[एक समाकल समीकरण]\label{ex:b2:metric:integralequation}
$X = C(\intcc{0}{1})$ पर (जो \omterm{def:b2:metric:complete}{पूर्ण} है, \cref{thm:b2:metric:rncomplete}), $T(f)(x) = 1 + \frac12
\int_0^x f(t)\,\dd t$ लीजिए। $f, g \in X$ के लिए:
\[
\abs{T(f)(x) - T(g)(x)} \leq \frac12 \int_0^x \abs{f - g} \leq
\frac12\, d_\infty(f, g),
\]
अतः $T$ एक $\frac12$-संकुचन है: उसका अद्वितीय \omterm{def:b2:metric:continuity}{संतत} अचल बिंदु है --- अर्थात्
$f' = \frac f2$, $f(0) = 1$ का हल, जो $\eu^{x/2}$ है। यही योजना औद्योगिक रूप ले लेने पर \cref{ch:b2:diffeq} की
कोशी--लिप्शिट्स प्रमेय बन जाती है।
\end{example}

\begin{example}[एक संख्यात्मक अचल बिंदु: $x = \cos x$]\label{ex:b2:metric:cosfix}
\omterm{def:b2:metric:complete}{पूर्ण} $X = \intcc{0}{1}$ पर प्रतिचित्रण $f = \cos$ $X$ को $\intcc{\cos 1}{1} \subseteq X$ में भेजता है और संकुचन है:
मध्यमान असमिका से,
\[
\abs{\cos x - \cos y} \leq \bigl(\sup_{\intcc01}\abs{\sin}\bigr)
\abs{x - y} = (\sin 1)\abs{x - y},
\qquad \sin 1 \approx 0.841 < 1 .
\]
बानाख: $\intcc01$ में $x = \cos x$ का अद्वितीय हल (अतः $\R$ में भी: कोई भी वास्तविक अचल
बिंदु $\intcc{-1}{1}$ में पड़ता है, और एक बार प्रतिचित्रण लगाने पर $\intcc{\cos 1}{1}$ में), और
पुनरावृत्ति $x_{n+1} = \cos x_n$ किसी भी आरंभ से उस पर अभिसरित होती है: $x_\infty
\approx 0.739085$, वही
प्रसिद्ध संख्या जो परिकलक की कोसाइन कुंजी बार-बार दबाने से मिलती है।
त्रुटि-परिबंध $(\sin1)^n/(1 - \sin1)$ क्षय की भविष्यवाणी करता है --- अर्थात् लगभग $13$
दबावों में एक अंक; और इस अध्याय की सप्ताहांत समस्या (प्रश्न 14) का
पश्चगामी परिबंध हर पग को हाथों-हाथ प्रमाणित कर देता है।
\end{example}

\section{संहतता}

\begin{definition}\label{def:b2:metric:compact}
कोई \omterm{def:b2:metric:def}{दूरिक समष्टि} $X$ \emph{संहत}\index{संहत दूरिक समष्टि} कहलाती है जब
$X$ के हर अनुक्रम का कोई उपअनुक्रम $X$ \emph{के भीतर} अभिसरित हो
(बोल्ज़ानो--वाइरश्ट्रास गुणधर्म)। कोई उपसमुच्चय संहत कहलाता है जब वह
प्रेरित दूरी के साथ ऐसा हो।
\end{definition}

\begin{theorem}[पहले गुणधर्म]\label{thm:b2:metric:compactprops}
\begin{enumerate}
  \item \omterm{def:b2:metric:compact}{संहत} उपसमुच्चय संवृत और परिबद्ध होता है; और \omterm{def:b2:metric:compact}{संहत} समष्टि का
        संवृत उपसमुच्चय \omterm{def:b2:metric:compact}{संहत} होता है।
  \item $\R^n$ में विलोम भी सत्य है: \omterm{def:b2:metric:compact}{संहत} $\iff$ संवृत और परिबद्ध।
  \item \omterm{def:b2:metric:compact}{संहत} समष्टि का \omterm{def:b2:metric:continuity}{संतत} प्रतिबिंब \omterm{def:b2:metric:compact}{संहत} होता है; और अरिक्त \omterm{def:b2:metric:compact}{संहत}
        समष्टि पर \omterm{def:b2:metric:continuity}{संतत} वास्तविक फलन परिबद्ध होता है तथा अपनी सीमाएँ
        प्राप्त कर लेता है।
  \item (हाइने) \omterm{def:b2:metric:compact}{संहत} समष्टि पर \omterm{def:b2:metric:continuity}{संतत} प्रतिचित्रण एकसमान \omterm{def:b2:metric:continuity}{संतत} होता है।
  \item गुणन: यदि $X, Y$ \omterm{def:b2:metric:compact}{संहत} हैं तो $X \times Y$ भी \omterm{def:b2:metric:compact}{संहत} है ($d\bigl((x,y),(x',y')\bigr) = d(x,x') + d(y,y')$ के साथ)।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) वही तर्क जो रेखा पर थे (प्रथम वर्ष का खंड): अनंत की ओर भागते हुए%
अथवा बाहर अभिसरित होते हुए अनुक्रम का कोई उपअनुक्रम भीतर अभिसरित नहीं
होता; और दूसरे दावे के लिए परिवेशी \omterm{def:b2:metric:compact}{संहत} में उपअनुक्रम निकालिए और
संवृतता का उपयोग कीजिए।

(2) $\R^n$ में परिबद्ध अनुक्रमों के घटक-दर-घटक अभिसारी उपअनुक्रम होते हैं:
पहले निर्देशांक पर उपअनुक्रम निकालिए ($\R$ पर
बोल्ज़ानो--वाइरश्ट्रास), फिर उसी उपअनुक्रम में से दूसरे पर, और इसी तरह
($n$ क्रमिक निष्कर्षण); और संवृतता सीमा को भीतर रोक लेती है।

(3) $(f(x_n))$ दिया हो तो $x_{\varphi(n)} \to x \in X$ निकालिए; संततता $f(x_{\varphi(n)}) \to f(x) \in f(X)$ दे देती है। वास्तविक
स्थिति: $f(X) \subseteq \R$ की संहतता उसे संवृत और परिबद्ध बना देती है, और $\sup f(X) \in f(X)$ (किसी
समुच्चय का उच्चतम उससे संलग्न होता है, और $f(X)$ संवृत है)।

(4) प्रथम वर्ष की उपपत्ति अक्षरशः यहाँ आ जाती है; दूरिक वेश में वह यह
है। मान लीजिए $f \colon X \to Y$ \omterm{def:b2:metric:compact}{संहत} $X$ पर \omterm{def:b2:metric:continuity}{संतत} है परंतु एकसमान \omterm{def:b2:metric:continuity}{संतत} नहीं: तो कोई
$\varepsilon > 0$ ऐसा है कि प्रत्येक $n$ के लिए ऐसे बिंदु मिल जाते हैं जिनके लिए
\[
d_X(x_n, y_n) \leq \frac{1}{n+1}
\qquad\text{तथा}\qquad
d_Y\bigl(f(x_n), f(y_n)\bigr) > \varepsilon .
\]
$x_{\varphi(n)} \to a \in X$ निकालिए; तब $y_{\varphi(n)} \to
a$ भी (परस्पर दूरियाँ $0$ की ओर जाती हैं)। $a$ पर
संततता दोनों प्रतिबिंब अनुक्रमों को $f(a)$ पर भेज देती है, अतः $d_Y\bigl(f(x_{\varphi(n)}), f(y_{\varphi(n)})\bigr) \to 0$ ---
जो एकसमान अंतराल $> \varepsilon$ के विरुद्ध है। संहतता ने ठीक एक ही चीज़ दी:
गुच्छन बिंदु $a$, जिस पर साधारण संततता लगाई जा सके।

(5) पहले $X$-निर्देशांकों पर उपअनुक्रम निकालिए, फिर $Y$-निर्देशांकों
पर।
\end{proof}

\begin{example}[हाइने प्रमेय, संहतता के साथ और उसके बिना]\label{ex:b2:metric:heinework}
$\intcc{0}{1}$ पर फलन $x \mapsto x^2$ एकसमान \omterm{def:b2:metric:continuity}{संतत} है --- हाइने बिना किसी परिकलन के यह कह देती
है, परंतु सीधा आकलन शिक्षाप्रद है:
\[
\abs{x^2 - y^2} = \abs{x + y}\,\abs{x - y} \leq 2\abs{x - y},
\]
अतः $\delta = \varepsilon/2$ \emph{सभी बिंदुओं के लिए एक साथ} काम करता है। $\R$ पर वही फलन
एकसमान \omterm{def:b2:metric:continuity}{संतत} नहीं है: $x_n = n$ और $y_n = n + \frac1n$ लेने पर अंतर $\abs{x_n - y_n}
= \frac1n \to 0$ रहता है जबकि $\abs{x_n^2 - y_n^2} = 2 + \frac1{n^2}
\geq 2$:
अर्थात् कोई एक $\delta$ $\varepsilon = 1$ के लिए काम नहीं आता। कार्यविधि दिखाई देती है:
स्थानीय \omterm{def:b2:metric:continuity}{लिप्शिट्स} अचर $\abs{x + y}$ \omterm{def:b2:metric:compact}{संहत} पर परिबद्ध है और $\R$ पर अपरिबद्ध ---
और हाइने प्रमेय ठीक यही कहती है कि संहतता ऐसे स्थानीय अचरों पर एकसमान
ढक्कन लगा देती है।
\end{example}

\begin{method}[किसी समुच्चय के संहत होने को सिद्ध करना]\label{met:b2:metric:compactness}
तीन रास्ते, बारंबारता के क्रम में। (1) \emph{परिवेशी पहचान:} $\R^n$ में
(अथवा किसी भी परिमित-विमीय मानदंडित समष्टि में, \cref{ch:b2:nvs}) संवृत होना जाँचिए
--- प्रायः पूर्वप्रतिबिंब के रूप में, \cref{ex:b2:metric:recognize} --- और परिबद्ध होना भी। (2)
\emph{विरासत:} किसी ज्ञात \omterm{def:b2:metric:compact}{संहत} समष्टि का संवृत उपसमुच्चय \omterm{def:b2:metric:compact}{संहत} होता है;
संहतों का परिमित सम्मिलन या गुणन \omterm{def:b2:metric:compact}{संहत} होता है; \omterm{def:b2:metric:compact}{संहत} का \omterm{def:b2:metric:continuity}{संतत} प्रतिबिंब
\omterm{def:b2:metric:compact}{संहत} होता है। (3) \emph{नंगे हाथ:} किसी भी अनुक्रम में से अभिसारी
उपअनुक्रम निकालिए --- प्रायः निर्देशांक-दर-निर्देशांक क्रमिक निष्कर्षण
से। और \emph{असंहतता} सिद्ध करने के लिए एक ही साक्षी पर्याप्त है: ऐसा
अनुक्रम जिसका कोई अभिसारी उपअनुक्रम न हो, और प्रायः वह परस्पर दूरी $\geq \varepsilon$
वाले बिंदुओं से बनता है।
\end{method}

\begin{example}[समुच्चयों के बीच दूरी: संहतता अपनी रोटी कमाती है]\label{ex:b2:metric:setdistance}
किसी \omterm{def:b2:metric:def}{दूरिक समष्टि} में मान लीजिए $K$ \omterm{def:b2:metric:compact}{संहत} है, $F$ संवृत है और $K \cap F = \emptyset$। तब
\[
d(K, F) = \inf\,\{d(x, y) : x \in K,\ y \in F\} > 0 :
\]
क्योंकि फलन $x \mapsto d(x, F)$ \omterm{def:b2:metric:continuity}{संतत} है (\cref{exo:b2:metric:11}) और $K$ पर धनात्मक ($d(x, F) = 0$ से $x \in \overline F = F$ हो जाता),
अतः वह \omterm{def:b2:metric:compact}{संहत} $K$ पर धनात्मक निम्नतम प्राप्त कर लेता है (\cref{thm:b2:metric:compactprops}~(3))।
संहतता सजावट नहीं है: दो \emph{संवृत} समुच्चयों के लिए निम्नतम बिना
प्राप्त हुए लुप्त हो सकता है --- $\R^2$ में अतिपरवलय $F_1 = \{xy
= 1\}$ और अक्ष $F_2 = \{y = 0\}$
असंयुक्त संवृत समुच्चय हैं और $d(F_1, F_2) = 0$ (बिंदु $(n, \frac1n)$ अक्ष के पास पहुँचते
जाते हैं)। अनंत की ओर भाग निकलना ठीक वही है जिसे संहतता रोकती है।
\end{example}

\begin{theorem}[बोरेल--लेबेग]\label{thm:b2:metric:borellebesgue}
कोई \omterm{def:b2:metric:def}{दूरिक समष्टि} $X$ \omterm{def:b2:metric:compact}{संहत} है यदि और केवल यदि \omterm{def:b2:metric:topology}{विवृत} समुच्चयों द्वारा $X$
के हर आच्छादन का कोई \emph{परिमित} उपआच्छादन हो।\index{बोरेल--लेबेग
प्रमेय}
\end{theorem}

\begin{proof}
($\Leftarrow$) मान लीजिए $(x_n)$ का कोई अभिसारी उपअनुक्रम नहीं है। हमारा दावा है कि
प्रत्येक $x \in X$ के लिए कोई ऐसा गोला $B(x, r_x)$ है जिसमें केवल परिमित सूचकांकों
$n$ के लिए $x_n$ हो: अन्यथा हर गोला $B(x, \frac1{k+1})$ अपरिमित पदों को समाहित करता, और
सूचकांक
\[
\varphi(0) < \varphi(1) < \varphi(2) < \cdots
\quad\text{ जहाँ }\quad
x_{\varphi(k)} \in B\Bigl(x, \frac{1}{k+1}\Bigr)
\]
चुनते जाने से (जो हर पग पर ठीक इसलिए संभव है कि अपरिमित उम्मीदवार बचे
रहते हैं) $x$ पर अभिसरित होने वाला उपअनुक्रम बन जाता। गोले $B(x, r_x)$ $X$
को आच्छादित करते हैं; और यदि उनमें से परिमित ही $X$ को आच्छादित कर
लेते, तो सूचकांक समुच्चय $\N$ परिमित समुच्चयों का परिमित सम्मिलन होता:
जो असंगत है।

($\Rightarrow$) दो चरण। \emph{लेबेग संख्या:} किसी \omterm{def:b2:metric:compact}{संहत} $X$ के \omterm{def:b2:metric:topology}{विवृत} आच्छादन $(U_i)$
के लिए कोई ऐसा $\rho > 0$ होता है कि त्रिज्या $\rho$ का हर गोला किसी न किसी $U_i$
में पड़े। अन्यथा प्रत्येक $n$ के लिए ऐसा $x_n$ चुनिए कि $B(x_n, \frac{1}{n+1})$ किसी भी $U_i$
में न हो; $x_{\varphi(n)} \to x \in U_{i_0} \supseteq B(x, r)$ निकालिए; तब बड़े $n$ के लिए $B(x_{\varphi(n)}, \frac{1}{\varphi(n)+1}) \subseteq B(x, r)
\subseteq U_{i_0}$: विरोधाभास।
\emph{पूर्ण परिबद्धता:} प्रत्येक $\rho > 0$ के लिए त्रिज्या $\rho$ के परिमित
गोले $X$ को आच्छादित कर लेते हैं। अन्यथा आगमन से $B(x_0, \rho) \cup
\dots \cup B(x_n, \rho)$ के बाहर $x_{n+1}$
चुनते जाइए: उस अनुक्रम की परस्पर दूरियाँ $\geq
\rho$ हैं, अतः कोई कोशी --- और
इसलिए कोई अभिसारी --- उपअनुक्रम नहीं: विरोधाभास। दोनों मिलाइए: $X$ को
त्रिज्या $\rho$ (लेबेग संख्या) के परिमित गोलों से आच्छादित कीजिए, जिनमें
से हर एक किसी न किसी $U_i$ के भीतर है: यही परिमित उपआच्छादन है।
\end{proof}

\begin{example}[एक $\varepsilon$-जाल, गिनकर]\label{ex:b2:metric:epsnet}
पूर्ण परिबद्धता (\cref{thm:b2:metric:borellebesgue} की उपपत्ति से) $\intcc{0}{1}$ पर बहुत मूर्त है: $\varepsilon > 0$ के लिए
$\varepsilon,
3\varepsilon, 5\varepsilon, \dots$ पर केंद्रित त्रिज्या $\varepsilon$ के $\lceil
\frac{1}{2\varepsilon}\rceil$ गोले उसे आच्छादित कर लेते हैं ---
लगभग $\frac1{2\varepsilon}$ गोले, और कोई भी आच्छादन उनमें से $\frac{1}{2\varepsilon}$ से कम में काम नहीं चला
सकता (हर गोला अधिकतम $2\varepsilon$ लंबाई ढकता है)। $\intcc01^2$ में यह गिनती वर्ग हो
जाती है, अर्थात् कोटि $\varepsilon^{-2}$: आच्छादन-संख्याएँ विमा $d$ में $\varepsilon^{-d}$ की तरह
बढ़ती हैं --- यह संहतता का एक परिमाणात्मक चेहरा है, और यही कारण है कि
\cref{ch:b2:nvs} के अनंत-विमीय इकाई गोले (जहाँ कोई परिमित $\frac13$-जाल है ही नहीं)
\omterm{def:b2:metric:compact}{संहत} नहीं हो सकते।
\end{example}

\begin{example}[आच्छादनों पर संहतता पढ़ना]\label{ex:b2:metric:covers}
अर्ध-विवृत अंतराल $\intoc{0}{1}$ \omterm{def:b2:metric:topology}{विवृत} समुच्चयों $U_n = \intoo{\frac1n}{2}$, $n \geq 1$ से आच्छादित है; किसी
भी परिमित उपकुल का कोई महत्तम सूचकांक $N$ होता है और वह $\intoc{0}{\frac1N}$ छोड़ देता
है: अतः कोई परिमित उपआच्छादन नहीं, इसलिए $\intoc{0}{1}$ \omterm{def:b2:metric:compact}{संहत} नहीं --- और अनुक्रम
वाली परिभाषा इसे $x_n = \frac1n$ के द्वारा देख लेती है, जिसकी सीमा $0$ भाग निकलती
है। दूसरी ओर, केवल एक बिंदु $0$ जोड़ देने से दोनों निदान एक साथ सुधर
जाते हैं: $\intcc{0}{1}$ पर ऐसे हर आच्छादन में कोई ऐसा समुच्चय होना ही पड़ता है
जिसमें $0$ हो, और वह पूरा एक आरंभिक खंड निगल लेता है, तथा परिमित
समुच्चय शेष पूरा कर देते हैं। \cref{thm:b2:metric:borellebesgue} की दोनों भाषाएँ सदा साथ ही विफल या
सफल होती हैं --- आच्छादन भाग निकलना ठीक वहीं पकड़ते हैं जहाँ अनुक्रम
पकड़ते हैं।
\end{example}

\begin{remark}[इसी खंड के भीतर के परिप्रेक्ष्य]\label{rem:b2:metric:perspectives}
यह अध्याय पूरे खंड की भार वहन करने वाली दीवार है; देखिए कि हर स्तंभ
कहाँ भार उठाता है। \emph{पूर्णता}: कोशी कसौटी बानाख समष्टियों में
श्रेणियों की अभिसरण-जाँच बन जाती है (\cref{ch:b2:series}), \cref{ch:b2:funcseq} में एकसमान अभिसरण ठीक
\omterm{def:b2:metric:complete}{पूर्ण} $\bigl(C, d_\infty\bigr)$ में अभिसरण ही है, और कोशी--लिप्शिट्स (\cref{ch:b2:diffeq}) बानाख अचल बिंदु
प्रमेय का समाकल-समीकरण वाला वेश है। \emph{संहतता}: वह मानदंडों की
तुल्यता सिद्ध करती है (\cref{ch:b2:nvs}), \cref{ch:b2:diffcalc} के अनुकूलन में चरम मानों की प्राप्ति,
और \cref{ch:b2:nvs} की सप्ताहांत समस्या में सर्वोत्तम सन्निकटनों का अस्तित्व।
\emph{संबद्धता}: वह स्थानीय कथनों को वैश्विक बना देती है --- अवकल
समीकरणों के हलों की अद्वितीयता, वक्रों पर मध्यमान प्रमेय (\cref{ch:b2:curves}), और
$GL_n(\R)$ के वे दो घटक जिन्हें दिक्विन्यास सिद्धांत (\cref{ch:b2:multint}) अलग-अलग रखेगा।
\end{remark}

\begin{remark}[सामान्य भूलें]\label{rem:b2:metric:pitfalls}
(क) ``संवृत और परिबद्ध होने से संहतता निकलती है'' $\R^n$ के बारे में
प्रमेय है, दूरिक समष्टियों के बारे में नहीं: विविक्त दूरी वाला कोई अनंत
समुच्चय अपने भीतर संवृत और परिबद्ध है फिर भी \omterm{def:b2:metric:compact}{संहत} नहीं (\cref{exo:b2:metric:4}), और $C(\intcc01)$
का संवृत इकाई गोला भी विफल हो जाता है (\cref{ch:b2:nvs})। (ख) पूर्णता \emph{दूरी}
का गुणधर्म है, सांस्थिति का नहीं: $d(x,y) = \abs{\arctan x - \arctan y}$ के साथ $\R$ में अभिसारी अनुक्रम
वही हैं फिर भी वह अपूर्ण है (\cref{exo:b2:metric:1})। (ग) \omterm{def:b2:metric:continuity}{संतत} एकैकी आच्छादक प्रतिचित्रण
का समाकृतिकता होना आवश्यक नहीं --- \cref{exo:b2:metric:7} का वृत्त-प्राचलन देखिए; और
उद्गम की संहतता इसे सुधार देती है। (घ) बानाख की प्रमेय को $k < 1$
\emph{एकसमान रूप से} चाहिए: असंहत समष्टि पर अकेली शर्त $d(f(x), f(y)) < d(x,y)$ कुछ भी
सुनिश्चित नहीं करती (\cref{exo:b2:metric:5})। (ङ) संबद्धता से सामान्यतः पथ-संबद्धता नहीं
निकलती --- परंतु इस पुस्तक में मिलने वाले मानदंडित समष्टियों के \omterm{def:b2:metric:topology}{विवृत}
उपसमुच्चयों के लिए दोनों एक ही हैं (\cref{ch:b2:nvs})।
\end{remark}

\begin{remark}[यह अध्याय कहाँ काम आता है]
विश्लेषण-भाग में सर्वत्र। $C(\intcc{a}{b})$ की पूर्णता \cref{ch:b2:funcseq} की अभिसरण प्रमेयों और \cref{ch:b2:diffeq}
के कोशी--लिप्शिट्स सिद्धांत को चलाती है (इस अध्याय की सप्ताहांत समस्या
स्थानीय पिकार--लिंडलोफ़ प्रमेय अभी सिद्ध कर देती है); संहतता परिमित विमा
में मानदंडों की तुल्यता देती है (\cref{ch:b2:nvs}) और \cref{ch:b2:diffcalc} में चरम मानों का
अस्तित्व; और संबद्धता \cref{ch:b2:realfun} के मध्यमान तर्कों तथा अवकल समीकरणों के लिए
अद्वितीयता के वैश्विकीकरण के नीचे बैठी है। तृतीय वर्ष के खंड में फलन
समष्टियों की संहतता (आर्ज़ेला--आस्कोली प्रमेय) और बेयर श्रेणी प्रमेय
(यहाँ \cref{exo:b2:metric:12}) रोज़ काम आने वाले औज़ार बन जाते हैं।
\end{remark}

\section{संबद्धता}

\begin{definition}\label{def:b2:metric:connected}
$X$ \emph{संबद्ध}\index{संबद्ध समष्टि} कहलाती है जब उसका दो अरिक्त
\omterm{def:b2:metric:topology}{विवृत} उपसमुच्चयों में विभाजन संभव न हो --- और समतुल्य रूप से, जब उसके
एकमात्र ऐसे उपसमुच्चय जो \omterm{def:b2:metric:topology}{विवृत} भी हों और संवृत भी, $\emptyset$ और $X$ हों। $X$
\emph{पथ-संबद्ध} कहलाती है जब उसके किन्हीं दो बिंदुओं को कोई \omterm{def:b2:metric:continuity}{संतत}
प्रतिचित्रण $\gamma \colon \intcc{0}{1} \to X$ जोड़ता हो।
\end{definition}

\begin{theorem}\label{thm:b2:metric:connectedness}
\begin{enumerate}
  \item $\R$ के \omterm{def:b2:metric:connected}{संबद्ध} उपसमुच्चय ठीक अंतराल हैं।
  \item \omterm{def:b2:metric:connected}{संबद्ध समष्टि} का \omterm{def:b2:metric:continuity}{संतत} प्रतिबिंब \omterm{def:b2:metric:connected}{संबद्ध} होता है --- और इसी से
        सामान्य मध्यमान प्रमेय: \omterm{def:b2:metric:connected}{संबद्ध समष्टि} पर \omterm{def:b2:metric:continuity}{संतत} वास्तविक फलन
        अपने किन्हीं दो मानों के बीच का हर मान लेता है।
  \item \omterm{def:b2:metric:connected}{पथ-संबद्ध} $\Rightarrow$ \omterm{def:b2:metric:connected}{संबद्ध}। (विलोम सामान्यतः विफल है; मानदंडित
        समष्टियों के \omterm{def:b2:metric:topology}{विवृत} उपसमुच्चयों के लिए वह सत्य है, \cref{ch:b2:nvs}।)
\end{enumerate}
\end{theorem}

\begin{proof}
(1) कोई अनंतराल $A$ अपने दो बिंदुओं के बीच का कोई $z$ छोड़ देता है:
$A = (A \cap \intoo{-\infty}{z}) \cup (A \cap \intoo{z}{+\infty})$
उसे दो अरिक्त ($A$ में) \omterm{def:b2:metric:topology}{विवृत} टुकड़ों में बाँट देता है। विलोमतः मान
लीजिए $I$ अंतराल है और $I = U \cup V$ अरिक्त सापेक्षतः \omterm{def:b2:metric:topology}{विवृत} समुच्चयों में उसका
विभाजन; $a \in U$, $b \in V$ चुनिए, मान लीजिए $a < b$, और $s = \sup\,(U \cap \intcc{a}{b})$ रखिए, जो $\intcc{a}{b}
\subseteq I$ का बिंदु
है। यदि $s \in U$: तो $s \neq b$, और $U$ की सापेक्ष विवृतता $s$ के चारों ओर
पूरा एक अंतराल ($I$ से प्रतिच्छेदित) $U$ के भीतर रख देती है --- अतः
$U \cap \intcc{a}{b}$ के बिंदु $s$ से बड़े हो जाते हैं, जो उच्चतम के विरुद्ध है। और यदि
$s \in V$: तो $V$ की सापेक्ष विवृतता कोई अंतराल $\intoo{s - r}{s + r} \cap I$ $V$ के भीतर रख देती
है; परंतु उच्चतम $U \cap \intcc{a}{b}$ से संलग्न है, जिसे उस अंतराल से मिलना ही पड़ेगा
--- और यह $U \cap V = \emptyset$ के विरुद्ध है। (यह $I$ के भीतर चलाया गया $\R$ वाला
प्रथम वर्ष का विवृत-संवृत तर्क है।)

(2) यदि $f(X) = U' \cup V'$ अरिक्त सापेक्षतः \omterm{def:b2:metric:topology}{विवृत} समुच्चयों में बँट जाता है, तो $X = f^{-1}(U') \cup f^{-1}(V')$
$X$ को बाँट देता है (\cref{thm:b2:metric:globalcontinuity})। मध्यमान प्रमेय: $f(X) \subseteq \R$ \omterm{def:b2:metric:connected}{संबद्ध} है, अतः (1) से
वह एक अंतराल है।

(3) मान लीजिए $X = U \cup V$, जहाँ दोनों अरिक्त \omterm{def:b2:metric:topology}{विवृत} हैं, और $a \in U$ को $b \in V$ से किसी
पथ $\gamma$ द्वारा जोड़िए: तब $\gamma^{-1}(U),
\gamma^{-1}(V)$ $\intcc{0}{1}$ को बाँट देते हैं, जो (1) के विरुद्ध
है।
\end{proof}

\begin{example}\label{ex:b2:metric:glnr}
$GL_n(\R)$ \omterm{def:b2:metric:connected}{संबद्ध} नहीं है: $\det$ \omterm{def:b2:metric:continuity}{संतत} है (प्रविष्टियों में बहुपद) और $\R^*$ पर
जाता है, जो \omterm{def:b2:metric:connected}{संबद्ध} नहीं; और $\R_+^*$ तथा $\R_-^*$ के पूर्वप्रतिबिंब $GL_n(\R)$ को बाँट
देते हैं। (वास्तव में हर टुकड़ा \omterm{def:b2:metric:connected}{पथ-संबद्ध} है --- यह हमारी आवश्यकता से
परे एक सुखद अभ्यास है।) इसके विपरीत $GL_n(\C)$ \omterm{def:b2:metric:connected}{पथ-संबद्ध} \emph{है}: \cref{exo:b2:metric:10}।
\end{example}

\begin{example}[केवल संबद्धता से एक अचल बिंदु]\label{ex:b2:metric:ivtfixed}
प्रत्येक \omterm{def:b2:metric:continuity}{संतत} $f \colon \intcc01 \to \intcc01$ का कोई अचल बिंदु होता है --- न संकुचन की परिकल्पना,
न पुनरावृत्ति। $g(x) = f(x) - x$ लीजिए, जो \omterm{def:b2:metric:connected}{संबद्ध} $\intcc01$ पर \omterm{def:b2:metric:continuity}{संतत} है:
\[
g(0) = f(0) \geq 0,
\qquad
g(1) = f(1) - 1 \leq 0 ,
\]
और मध्यमान प्रमेय (\cref{thm:b2:metric:connectedness}~(2)) $g$ का कोई शून्य दे देती है, अर्थात् $f$
का अचल बिंदु। बानाख (\cref{thm:b2:metric:banach}) से तुलना कीजिए: यहाँ अस्तित्व सांस्थितिक और
मुफ़्त है, परंतु अद्वितीयता और कलनविधि खो जाती हैं --- $f =
\mathrm{id}$ का हर बिंदु
अचल है, और असंकुचनकारी $f$ की पुनरावृत्ति सदा चक्कर काटती रह सकती है।
इस अध्याय की दोनों अचल बिंदु प्रमेयें भिन्न प्रश्नों का उत्तर भिन्न
मुद्राओं में देती हैं।
\end{example}

\begin{example}[$\R$ और $\R^2$ समाकृतिक नहीं हैं]\label{ex:b2:metric:rnotr2}
संबद्धता एक सांस्थितिक अंगुलिचिह्न है। मान लीजिए $h \colon
\R^2 \to \R$ कोई समाकृतिकता है
(अर्थात् \omterm{def:b2:metric:continuity}{संतत} प्रतिलोम वाला \omterm{def:b2:metric:continuity}{संतत} एकैकी आच्छादक प्रतिचित्रण)। एक बिंदु
$a \in \R^2$ हटा दीजिए: प्रतिबंधन $h \colon \R^2\setminus\{a\} \to
\R\setminus\{h(a)\}$ तब भी समाकृतिकता है। परंतु $\R^2$ में से एक
बिंदु हटाने पर वह \omterm{def:b2:metric:connected}{पथ-संबद्ध} रहता है --- किन्हीं दो बिंदुओं को किसी खंड
से जोड़िए, और यदि $a$ सीधा रास्ता रोक ले तो किसी सहायक बिंदु से होकर
दूसरे खंड का चक्कर लगाइए --- अतः \omterm{def:b2:metric:connected}{संबद्ध} (\cref{thm:b2:metric:connectedness}~(3)); जबकि $\R$ में से एक
बिंदु हटाने पर वह दो अरिक्त \omterm{def:b2:metric:topology}{विवृत} अर्धरेखाओं में बँट जाता है: \omterm{def:b2:metric:connected}{संबद्ध}
नहीं। और \omterm{def:b2:metric:continuity}{संतत} प्रतिचित्रण संबद्धता सुरक्षित रखते हैं: विरोधाभास। समतल
और रेखा \emph{सांस्थितिक समष्टियों के रूप में} सचमुच भिन्न हैं --- और
अकेली गणनसंख्या (जहाँ \cref{exo:b2:structures:3} जैसी एकैकी आच्छादक संगतियाँ विद्यमान हैं!)
इसे देखने के लिए बहुत मोटी है।
\end{example}

\section{समस्या: पिकार पुनरावृत्ति}

पूर्णता और संकुचन मिलकर एक हल करने की मशीन बनते हैं: उसमें अचल बिंदु की
समस्या के रूप में लिखा हुआ समीकरण डालिए, और वह अस्तित्व, अद्वितीयता,
कलनविधि तथा त्रुटि-पट्टियाँ लौटा देती है। यह सप्ताहांत समस्या उस मशीन
को समीकरण $y' = f(t,
y)$ पर पूरी शक्ति से चलाती है: हम स्थानीय
\emph{पिकार--लिंडलोफ़ प्रमेय} सिद्ध करते हैं (\cref{ch:b2:diffeq} के कोशी--लिप्शिट्स
सिद्धांत का अरैखिक हृदय), प्रतिउदाहरणों के द्वारा हर परिकल्पना को अपनी
रोटी कमाते देखते हैं, और विशुद्ध दूरिक लाभ बटोरते हैं --- आँकड़ों पर
\omterm{def:b2:metric:continuity}{संतत} निर्भरता, केप्लर समीकरण, और कैंटर समुच्चय की स्वसमरूपता।

\begin{problem}[{सप्ताहांत समस्या --- पिकार--लिंडलोफ़
प्रमेय}]
\label{pb:b2:metric:1}
आगे सर्वत्र $t_0 \in \R$, $y_0 \in \R$, $a, b > 0$, और $f$ आयत $R = \intcc{t_0 - a}{t_0 +
a} \times \intcc{y_0 - b}{y_0 + b}$ पर ऐसा \omterm{def:b2:metric:continuity}{संतत} फलन है जो $M =
\sup_R\,\abs f$ से
परिबद्ध है और अपने दूसरे चर में \emph{$L$-लिप्शिट्स} है: जब दोनों बिंदु
$R$ में हों तब $\abs{f(t, y) - f(t, z)} \leq L\abs{y - z}$। रखिए
\[
h = \min\Bigl(a, \frac bM\Bigr) \quad (\text{ जहाँ } h = a
\text{ यदि } M = 0), \qquad I = \intcc{t_0 - h}{t_0 + h}.
\]

\medskip
\noindent\textbf{भाग I --- पूर्ण मंच।}
\begin{enumerate}
  \item \cref{def:b2:metric:complete} में उद्धृत दोनों कथन सिद्ध कीजिए: \omterm{def:b2:metric:complete}{पूर्ण दूरिक समष्टि} का
        संवृत उपसमुच्चय \omterm{def:b2:metric:complete}{पूर्ण} होता है, और किसी भी \omterm{def:b2:metric:def}{दूरिक समष्टि} का
        \omterm{def:b2:metric:complete}{पूर्ण} उपसमुच्चय संवृत। इससे निष्कर्ष निकालिए कि $\bigl(C(I), d_\infty\bigr)$ का हर
        संवृत उपसमुच्चय \omterm{def:b2:metric:complete}{पूर्ण दूरिक समष्टि} है।
  \item दिखाइए कि प्रतिचित्रण $C(I) \to C(I)$, $y \mapsto \bigl(t
        \mapsto \int_{t_0}^{t}y(s)\,\dd s\bigr)$, $d_\infty$ के लिए $h$-लिप्शिट्स है।
  \item (अचल बिंदु प्रतिचित्रणों से कम हिलते हैं) मान लीजिए $g \colon X
        \to X$ किसी
        \omterm{def:b2:metric:def}{दूरिक समष्टि} का $k$-संकुचन है जिसका अचल बिंदु $\ell_g$ है, और $\widetilde g \colon X \to X$
        \emph{कोई भी} ऐसा प्रतिचित्रण जिसका अचल बिंदु $\ell_{\widetilde g}$ है। सिद्ध
        कीजिए
        \[
        d(\ell_g, \ell_{\widetilde g}) \leq
        \frac{d\bigl(g(\ell_{\widetilde g}),
        \widetilde g(\ell_{\widetilde g})\bigr)}{1 - k}
        \leq \frac{\sup_{x \in X} d\bigl(g(x), \widetilde
        g(x)\bigr)}{1 - k}.
        \]
  \item (पुनरावृत्त की युक्ति) मान लीजिए $X$ अरिक्त \omterm{def:b2:metric:complete}{पूर्ण} है और $g
        \colon X \to X$
        ऐसा प्रतिचित्रण --- जिसे \omterm{def:b2:metric:continuity}{संतत} नहीं माना गया --- कि उसका कोई
        पुनरावृत्त $g^m$ $k$-संकुचन हो। सिद्ध कीजिए कि $g$ का
        अद्वितीय अचल बिंदु $\ell$ है और \emph{हर} कक्षा $x_{n+1} = g(x_n)$ $\ell$ पर
        अभिसरित होती है। \emph{($g$ के अचल बिंदु $g^m$ के अचल बिंदु
        होते हैं; विलोमतः $g(\ell)$ $g^m$ का अचल बिंदु है; और कक्षा को $m$
        के सापेक्ष शेषों के अनुसार बाँटिए।)}
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- पिकार--लिंडलोफ़ प्रमेय।}
\begin{enumerate}[resume]
  \item दिखाइए कि कोई फलन $y \colon I \to \intcc{y_0 - b}{y_0
        + b}$ $I$ पर $y(t_0) = y_0$ तथा $y' = f(t, y)$ के साथ $C^1$ है यदि और
        केवल यदि वह \omterm{def:b2:metric:continuity}{संतत} हो और समाकल समीकरण
        \[
        y(t) = y_0 + \int_{t_0}^{t} f\bigl(s, y(s)\bigr)\dd s
        \qquad (t \in I).
        \]
        को संतुष्ट करे।
  \item मान लीजिए $X_h = \{y \in C(I) : \abs{y(t) - y_0} \leq b
        \text{ पर } I\}$ और $T$ को $T(y)(t) =
        y_0 + \int_{t_0}^{t}f(s, y(s))\dd s$ से परिभाषित
        कीजिए। दिखाइए कि $X_h$ $C(I)$ का अरिक्त संवृत उपसमुच्चय है, अतः
        \omterm{def:b2:metric:complete}{पूर्ण} है, और यह कि $T$ $X_h$ को $X_h$ में भेजता है --- यहीं
        $h \leq
        b/M$ काम आता है।
  \item दिखाइए कि $X_h$ पर $d_\infty\bigl(T(y), T(z)\bigr) \leq
        Lh\,d_\infty(y, z)$: यदि $Lh < 1$ हो, तो बानाख की प्रमेय
        पहले ही निष्कर्ष दे देती है। अगले चरण में हम यह लघुता वाली
        शर्त हटा देंगे।
  \item $n$ पर आगमन से सिद्ध कीजिए:
        \[
        \abs{T^n(y)(t) - T^n(z)(t)} \leq
        \frac{\bigl(L\abs{t - t_0}\bigr)^n}{n!}\,
        d_\infty(y, z) \qquad (y, z \in X_h,\ t \in I),
        \]
        अतः $T$ का कोई पुनरावृत्त संकुचन है। प्रश्न 4 से निष्कर्ष
        निकालिए (\emph{पिकार--लिंडलोफ़ प्रमेय}): कोशी समस्या $y' = f(t,y)$,
        $y(t_0) = y_0$ का $I = \intcc{t_0 - h}{t_0 + h}$ पर $\intcc{y_0 - b}{y_0 + b}$ में मान लेने वाला ठीक एक हल है।
  \item दिखाइए कि ``$\intcc{y_0 -
        b}{y_0 + b}$ में मान लेते हुए'' वाला प्रतिबंध स्वतः लग
        जाता है: $I$ पर परिभाषित कोशी समस्या का कोई भी हल, जिसका
        आलेख $R$ में आरंभ होता है, $\intcc{y_0 - b}{y_0 + b}$ में ही रहता है
        \emph{(पहला निर्गम समय लीजिए और $\abs{y(t) - y_0}$ को $M\abs{t -
        t_0}$ से परिबद्ध
        कीजिए)}। अतः अद्वितीयता $I$ पर के सभी हलों में सत्य है।
  \item मशीन को $y' = y$, $y(0) = 1$ पर चलाइए, और अचर $y^{(0)} \equiv 1$ से आरंभ कीजिए: पिकार
        पुनरावृत्त $y^{(n)}$ परिकलित कीजिए, उन्हें पहचानिए, और अभिसरण का
        वर्णन कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- हर परिकल्पना अपनी रोटी कमाती है।}
\begin{enumerate}[resume]
  \item (\omterm{def:b2:metric:continuity}{लिप्शिट्स} विफल, अद्वितीयता विफल) $y' =
        2\sqrt{\abs y}$, $y(0) = 0$ के लिए: जाँचिए कि
        $y \equiv 0$, और यह भी कि प्रत्येक $c \geq 0$ के लिए फलन $t \leq c$ के लिए $y_c(t) = 0$ तथा
        $t > c$ के लिए $y_c(t) = (t - c)^2$ --- ये सब $\R$ पर $C^1$ हल हैं। $y \mapsto
        2\sqrt{\abs y}$ ठीक कहाँ
        \omterm{def:b2:metric:continuity}{लिप्शिट्स} नहीं है?
  \item (स्थानीयता वास्तविक है) $y' = y^2$, $y(0) = 1$ के लिए: स्पष्ट रूप से हल
        कीजिए, अस्तित्व का महत्तम अंतराल दीजिए, और आयत के सभी चुनावों
        ($a$ बड़ा, $b$ स्वतंत्र) पर प्रमेय जो सर्वोत्तम $h$
        प्रमाणित कर सकती है वह परिकलित कीजिए: दिखाइए कि $h_{\max} = \sup_{b>0} \frac{b}{(1+b)^2} = \frac14$, जबकि
        सच्चा हल $\intoo{-\infty}{1}$ पर रहता है।
  \item (पूर्णता सजावट नहीं है) सामान्य दूरी वाले $X = \Q \cap
        \intcc{1}{2}$ पर $g(x) =
        \frac x2 + \frac1x$ लीजिए।
        दिखाइए कि $g(X) \subseteq X$, कि $g$ $\frac12$-संकुचन है \emph{(मध्यमान
        असमिका)}, और कि $g$ का $X$ में कोई अचल बिंदु नहीं है। बानाख
        की प्रमेय की कौन-सी परिकल्पना विफल हो रही है, और पूर्णक में
        अचल बिंदु क्या है?
  \item (त्रुटि-पट्टियाँ) किसी \omterm{def:b2:metric:complete}{पूर्ण} समष्टि पर $k$-संकुचन $g$ के लिए
        \emph{पश्चगामी} आकलन $d(x_n,
        \ell) \leq \frac{k}{1-k}\,d(x_n, x_{n-1})$ सिद्ध कीजिए। $\intcc{1}{2}$ पर हीरोन
        प्रतिचित्रण $g(x) = \frac x2 + \frac 1x$ (अचल बिंदु $\sqrt2$) के लिए, $x_0 = \frac32$ से आरंभ करके:
        यथार्थता $10^{-6}$ के लिए \emph{पूर्वगामी} परिबंध $\frac{k^n}{1-k}d(x_1, x_0)$ कितने पग
        माँगता है, और वास्तव में कितने पग पर्याप्त हैं? ($x_1, x_2, x_3$ और उनकी
        त्रुटियाँ परिकलित कीजिए; संकुचन-परिबंध ईमानदार है पर
        निराशावादी --- हीरोन द्विघातीय गति से अभिसरित होता है।)
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- \omterm{def:b2:metric:continuity}{संतत} निर्भरता।} इस भाग में $Lh < 1$, अतः $T$
स्वयं $X_h$ पर संकुचन है (प्रश्न 7); और आरंभिक मान $y_0$ वाले हल के लिए
$y[\,y_0\,]$ लिखिए।
\begin{enumerate}[resume]
  \item (आरंभिक मान पर निर्भरता) मान लीजिए $z_0$ कोई अन्य आरंभिक मान है
        और $\abs{z_0 - y_0}$ इतना छोटा है कि दोनों समस्याएँ आयत में समा जाएँ। प्रश्न
        3 का उपयोग करके सिद्ध कीजिए
        \[
        d_\infty\bigl(y[y_0], y[z_0]\bigr) \leq
        \frac{\abs{y_0 - z_0}}{1 - Lh}.
        \]
  \item (शृंखलाबद्ध करके लंबे अंतराल) मान लीजिए हल किसी लंबे खंड पर
        विद्यमान हैं जिसे $m$ क्रमागत टुकड़ों में काटा गया है और
        प्रत्येक पर पिछला परिबंध $Lh \leq
        \frac12$ के साथ लागू होता है। दिखाइए कि
        विचलन प्रति टुकड़ा अधिकतम $2$ गुना बढ़ता है, अतः कुल मिलाकर
        $d_\infty \leq 2^m\abs{y_0 - z_0}$ --- अर्थात् लंबाई में घातांकीय परिबंध, जो ग्रोनवाल
        प्रमेयिका (\cref{ch:b2:diffeq}) के $\eu^{L\abs{t - t_0}}$ की विविक्त परछाईं है।
  \item (क्षेत्र पर निर्भरता) मान लीजिए $g$ $R$ पर एक और क्षेत्र है,
        जो $y$ में $L$-लिप्शिट्स भी है, और $\sup_R \abs{f - g}
        \leq \varepsilon$। सिद्ध कीजिए कि
        संगत हल $d_\infty \leq
        \frac{\varepsilon h}{1 - Lh}$ पूरा करते हैं: अर्थात् प्रतिरूपण-त्रुटि रैखिक
        रूप से फैलती है।
  \item (प्राचल) यदि क्षेत्रों का कोई कुल $f_\lambda$ $y$ में एकसमान रूप से
        $L$-लिप्शिट्स हो और $\sup_R\abs{f_\lambda
        - f_\mu} \leq C\abs{\lambda - \mu}$, तो निष्कर्ष निकालिए कि $\lambda \mapsto y_\lambda$ प्राचल
        समष्टि से $\bigl(C(I), d_\infty\bigr)$ में \omterm{def:b2:metric:continuity}{लिप्शिट्स} है।
  \item (निकायों की कोई क़ीमत नहीं) समझाइए कि भाग I, II और IV $\R^n$ में
        मान लेने वाले $y$ के लिए अक्षरशः क्यों सत्य रहते हैं (\cref{ex:b2:metric:examples} की
        किसी भी दूरी पर बनी उच्चतम दूरियाँ), और फिर निकाय $y' = Ay$, $y(0) = (c_1,
        c_2)$,
        $A = \begin{psmallmatrix} 0 & 1\\ 0 &
        0\end{psmallmatrix}$ के लिए सभी पिकार पुनरावृत्त परिकलित कीजिए: दिखाइए कि एक
        ही पग के बाद पुनरावृत्ति यथार्थ हल पर स्थिर हो जाती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- दूरिक लाभ और संश्लेषण।}
\begin{enumerate}[resume]
  \item (तत्समक का विक्षोभ) मान लीजिए $X$ मानदंडित समष्टि जैसा कोई
        पूर्ण मंच है: $X = C(I)$ अथवा $\R^n$ लीजिए। यदि $\eta \colon X \to X$ $k < 1$ के साथ
        $k$-लिप्शिट्स हो, तो सिद्ध कीजिए कि $x \mapsto x + \eta(x)$ $X$ का ऐसा एकैकी
        आच्छादक प्रतिचित्रण है जिसका प्रतिलोम $\frac1{1-k}$-लिप्शिट्स है
        \emph{(प्रत्येक $y$ के लिए $x \mapsto y - \eta(x)$ पर बानाख लगाइए)}। यही
        प्रतिलोम फलन प्रमेय (\cref{ch:b2:diffcalc}) का दूरिक हृदय है।
  \item (केप्लर समीकरण) $0 \leq e < 1$ और $m \in \R$ के लिए सिद्ध कीजिए कि $x = m + e\sin x$ का ठीक
        एक हल है, कि पुनरावृत्ति $x_{n+1} = m + e\sin x_n$ किसी भी आरंभ से उस पर अभिसरित
        होती है, और आकलन कीजिए: $e = \frac12$, $m = 1$ के लिए पूर्वगामी परिबंध से
        त्रुटि $\leq
        10^{-3}$ की गारंटी के लिए कितनी पुनरावृत्तियाँ चाहिए? (हल
        $x
        \approx 1.4987$ है।)
  \item (कैंटर समुच्चय एक अचल बिंदु है) $\R$ पर $S_1(x) = \frac
        x3$ और $S_2(x) = \frac x3 + \frac23$ लीजिए,
        और मान लीजिए $C$ \cref{exo:b2:metric:8} का कैंटर समुच्चय है। सिद्ध कीजिए कि
        $C = S_1(C) \cup S_2(C)$, और एक वाक्य में समझाइए कि कोई \emph{अन्य} अरिक्त \omterm{def:b2:metric:compact}{संहत}
        समुच्चय यह समीकरण क्यों नहीं पूरा करता (प्रतिचित्रण $A \mapsto S_1(A) \cup S_2(A)$ \omterm{def:b2:metric:compact}{संहत}
        समुच्चयों के बीच की एक दूरी --- हाउसडॉर्फ दूरी, जिसे तृतीय वर्ष
        के खंड में ईमानदारी से गढ़ा गया है --- के लिए संकुचन है)।
  \item (संबद्धता अद्वितीयता को वैश्विक बनाती है) मान लीजिए $f$ किसी
        \omterm{def:b2:metric:topology}{विवृत} समुच्चय पर $y$ में स्थानीय रूप से \omterm{def:b2:metric:continuity}{लिप्शिट्स} है, और $y, z$
        किसी साझे अंतराल $J$ पर $y' = f(t, y)$ के दो ऐसे हल हैं कि $y(t_0) = z(t_0)$। सिद्ध
        कीजिए कि $J$ पर $y = z$: दिखाइए कि $\{t \in J : y(t) = z(t)\}$ अरिक्त है, $J$ में
        संवृत है, और $J$ में \omterm{def:b2:metric:topology}{विवृत} है (स्थानीय अद्वितीयता से), और फिर
        अंतरालों की संबद्धता (\cref{thm:b2:metric:connectedness}) का उपयोग कीजिए।
  \item (रैखिक समीकरणों के लिए कोई लघुता नहीं) किसी खंड $\intcc{A}{B}$ पर \omterm{def:b2:metric:continuity}{संतत}
        $\alpha, \beta$ के साथ $y' =
        \alpha(t)y + \beta(t)$ के लिए प्रश्न 8 को अनुकूलित करके दिखाइए कि
        क्रमगुणित परिबंध \emph{पूरे} खंड पर लागू होता है, अतः वहाँ
        अस्तित्व और अद्वितीयता वैश्विक हैं --- यह \cref{ch:b2:diffeq} की
        कोशी--लिप्शिट्स प्रमेय की अदिश स्थिति है, जिसमें लंबाई $B - A$ पर
        कोई प्रतिबंध नहीं।
  \item (संश्लेषण) एक-एक वाक्य में: पूर्णता ने क्या दिया; संकुचन ने
        क्या दिया; सादी बानाख की तुलना में पुनरावृत्त की युक्ति से क्या
        मिला; संबद्धता कहाँ आई; और भाग III का कौन-सा प्रतिउदाहरण
        कौन-सी परिकल्पना की रक्षा करता है। शिखर प्रमेय का नाम लीजिए, और
        बताइए कि $f$ के केवल \omterm{def:b2:metric:continuity}{संतत} होने पर संकुचन का स्थान क्या लेता
        है (पियानो प्रमेय, फलन समष्टियों में संहतता के द्वारा ---
        तृतीय वर्ष के खंड की आर्ज़ेला--आस्कोली)।
\end{enumerate}
\end{problem}
