\chapter{अनुक्रम और श्रेणियाँ}\label{ch:b2:series}

संख्यात्मक श्रेणियों का सिद्धांत (प्रथम वर्ष का खंड) यहाँ तीन दिशाओं में
परिपक्व होता है: बानाख समष्टियों में मान लेने वाली श्रेणियाँ, जहाँ सारा
काम निरपेक्ष अभिसरण करता है; वास्तविक श्रेणियों के लिए सूक्ष्मतर जाँचें
(\omterm{thm:b2:series:abel}{आबेल योग}); और \emph{\omterm{def:b2:series:summable}{योग्य कुल}} --- अर्थात् पदों के क्रम से मुक्त किया
गया योग --- दोहरे योगों के लिए फ़ूबिनी प्रमेय तथा \omterm{thm:b2:series:fubini}{कोशी गुणनफल} के साथ। ये
औज़ार आगे के सारे फलन-श्रेणी अध्यायों को उठाए चलते हैं।

\section{मानदंडित समष्टियों में श्रेणियाँ}

\begin{definition}\label{def:b2:series:def}
किसी मानदंडित समष्टि $E$ में अनुक्रम $(u_n)$ के लिए श्रेणी $\sum u_n$ तब अभिसरित
होती है जब उसके आंशिक योग अभिसरित हों; और वह \emph{निरपेक्ष रूप
से}\index{निरपेक्ष अभिसरण!बानाख समष्टि में} तब अभिसरित होती है जब $\sum \norm{u_n} < \infty$।
किसी \emph{बानाख} समष्टि में निरपेक्ष अभिसरण से अभिसरण निकलता है (\cref{thm:b2:nvs:absoluteconvergence});
और अपूर्ण समष्टि में यह विफल हो सकता है (\cref{exo:b2:series:9})।
\end{definition}

\begin{example}\label{ex:b2:series:neumann}
$\mathcal{M}_n(K)$ में (अथवा $\mathcal{L}_c(E)$ में, जहाँ $E$ बानाख है): $\vertiii A < 1$ के लिए \emph{नॉयमान
श्रेणी} $\sum A^k$ \omterm{def:b2:series:def}{निरपेक्ष रूप से} $(I - A)^{-1}$ पर अभिसरित होती है (जो \cref{exo:b2:nvs:5} में सिद्ध की
गई है); और प्रत्येक $A$ के लिए $\sum \frac{A^k}{k!}$ \omterm{def:b2:series:def}{निरपेक्ष रूप से} $\eu^A$ पर (\cref{ex:b2:nvs:matrixexp})।
संकारक-मान वाली गुणोत्तर और घातांकी श्रेणियाँ अपने अदिश नमूनों की तरह
व्यवहार करती हैं --- और बानाख ढाँचे का पूरा उद्देश्य यही है।
\end{example}

\section{आबेल योग}

\begin{theorem}[आबेल का योग और उसकी जाँच]\label{thm:b2:series:abel}
(खंडशः योग) अदिश $a_n$ और सदिश $b_n$ के लिए, $B_n
= \sum_{k=0}^{n} b_k$ के साथ:\index{आबेल योग}
\[
\sum_{n=0}^{N} a_n b_n
= a_N B_N - \sum_{n=0}^{N-1} (a_{n+1} - a_n) B_n .
\]
(आबेल की जाँच) यदि $(a_n)$ $0$ तक घटता हुआ वास्तविक अनुक्रम हो और आंशिक योग
$B_n$ \emph{परिबद्ध} हों (किसी \omterm{def:b2:nvs:banach}{बानाख समष्टि} में), तो $\sum a_n b_n$ अभिसरित होती है।
\end{theorem}

\begin{proof}
सर्वसमिका, पग-दर-पग: $B_{-1} = 0$ के साथ $b_n = B_n -
B_{n-1}$ लिखिए और बाँटिए,
\[
\sum_{n=0}^{N} a_nb_n
= \sum_{n=0}^{N} a_nB_n - \sum_{n=0}^{N} a_nB_{n-1}
= \sum_{n=0}^{N} a_nB_n - \sum_{n=0}^{N-1} a_{n+1}B_{n} ,
\]
दूसरे योग को $n \mapsto n + 1$ के अनुसार पुनः सूचीबद्ध कीजिए ($B_{-1}$ वाला पद लुप्त हो
जाता है); और साझा परिसर $0 \leq n \leq N-1$ इकट्ठा करने पर $a_NB_N$ तथा $\sum_{n\leq N-1}(a_n - a_{n+1})B_n$ बचते हैं: यही
कथित सूत्र है। यह विविक्त खंडशः समाकलन है, जिसमें $(B_n)$ $(b_n)$ का प्रतिअवकलज
है और अंतर $a_{n+1}
- a_n$ $(a_n)$ का अवकलज। जाँच के लिए, $\norm{B_n}
\leq M$ के साथ: सीमा-पद $a_N B_N \to 0$;
और श्रेणी $\sum (a_n -
a_{n+1})B_n$ \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसरित होती है, क्योंकि
\[
\sum_n \norm{(a_{n+1} - a_n)B_n} \leq M \sum_n (a_n - a_{n+1})
= M a_0 < \infty
\]
(क्रमिक निरसन, $a_n \downarrow 0$)। सर्वसमिका के दोनों टुकड़े अभिसरित होते हैं, अतः
$\sum a_n b_n$ भी।
\end{proof}

\begin{example}\label{ex:b2:series:sinn}
$\sum \frac{\sin n}{n}$ अभिसरित होती है: $a_n = \frac1n \downarrow 0$ और $B_n = \sum_{k=1}^{n} \sin k$ परिबद्ध है --- वास्तव में $B_n =
\Im\sum_{k \leq n} \eu^{\iu k} = \Im\,\frac{\eu^{\iu}(\eu^{\iu n} -
1)}{\eu^{\iu} - 1}$, जिसका
मापांक $\leq \frac{2}{\abs{\eu^{\iu} - 1}}$ है। वह \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसरित \emph{नहीं} होती ($\abs{\sin n} \geq \sin^2 n
= \frac{1 - \cos 2n}{2}$, और $\sum \frac{1 - \cos 2n}{2n}$
अपसरित होती है क्योंकि उसी आबेल जाँच से $\sum \frac{\cos 2n}{n}$ अभिसरित होती है जबकि $\sum \frac{1}{2n}$
अपसरित)। एकांतर श्रेणी की जाँच $b_n = (-1)^n$ वाली विशेष स्थिति है।
\end{example}

\begin{example}[अभिसरण वृत्त पर आबेल]\label{ex:b2:series:abelboundary}
$\abs z = 1$ वाले किन सम्मिश्र $z$ के लिए $\sum_{n \geq 1}
\frac{z^n}{n}$ अभिसरित होती है? $z = 1$ पर वह हरात्मक
श्रेणी है: अपसारी। और वृत्त पर $z \neq 1$ के लिए आबेल की जाँच $a_n = \frac1n \downarrow 0$ तथा $b_n = z^n$ के
साथ लागू होती है, जिसके आंशिक योग $N$ से स्वतंत्र रूप से परिबद्ध हैं:
\[
\Bigl|\sum_{n=1}^{N} z^n\Bigr|
= \Bigl|\frac{z(z^N - 1)}{z - 1}\Bigr|
\leq \frac{2}{\abs{z - 1}} .
\]
अतः अभिसारी --- यद्यपि \omterm{def:b2:series:def}{निरपेक्ष रूप से} कभी नहीं ($\sum\frac1n$)। एक ही श्रेणी,
और व्यवहारों का पूरा वृत्त: एक ही बिंदु पर अपसरण, और अन्यत्र सर्वत्र
अर्ध-अभिसरण। यह घात श्रेणियों का मानक सीमा-व्यवहार है (\cref{ch:b2:powerseries}), जो यहाँ
नंगे हाथों मिल गया; $z = -1$ पर उससे एकांतर हरात्मक श्रेणी पुनः मिलती है,
और $z = \eu^{\iu\theta}$ पर उसके वास्तविक तथा काल्पनिक भाग \cref{exo:b2:series:4} की श्रेणियाँ $\sum\frac{\cos
n\theta}{n}$ और $\sum\frac{\sin n\theta}{n}$
हैं।
\end{example}

\begin{example}[एक जाल बिछाई हुई एकांतर श्रेणी]\label{ex:b2:series:trap}
क्या $\sum_{n\geq2} \dfrac{(-1)^n}{\sqrt n + (-1)^n}$ अभिसरित होती है? चिह्न बारी-बारी बदलते हैं और पद $0$ की ओर जाते
हैं --- फिर भी एकांतर जाँच लागू \emph{नहीं} होती: मापांक $\frac{1}{\sqrt n + (-1)^n}$ ह्रासमान
नहीं हैं (वे हर विषम $n$ पर उछल जाते हैं)। इसके बदले प्रसार कीजिए:
\[
\frac{(-1)^n}{\sqrt n + (-1)^n}
= \frac{(-1)^n}{\sqrt n}\cdot
\frac{1}{1 + \frac{(-1)^n}{\sqrt n}}
= \frac{(-1)^n}{\sqrt n} - \frac{1}{n}
+ O\Bigl(\frac{1}{n^{3/2}}\Bigr).
\]
पहला टुकड़ा अभिसरित होता है (एकांतर जाँच, $\frac1{\sqrt n}\downarrow0$ पर ईमानदारी से लगाई गई),
तीसरा \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसरित होता है --- परंतु बीच वाला टुकड़ा अपसारी
हरात्मक श्रेणी है: अतः योग $-\infty$ की ओर \emph{अपसरित} होता है। समापन दृष्टि:
जब एकदिष्टता विफल हो, तब तक प्रसार कीजिए जब तक हर टुकड़ा या तो \omterm{def:b2:series:def}{निरपेक्ष
रूप से} अभिसारी हो या कोई स्वच्छ जाँच-स्थिति; छिपा हुआ $-\frac1n$
चिह्न-गिनती को दिखाई ही नहीं देता।
\end{example}

\begin{remark}[सामान्य भूलें]\label{rem:b2:series:pitfalls}
(क) ``पद $0$ की ओर जाते हैं'' से कुछ भी सिद्ध नहीं होता: हरात्मक श्रेणी
अपसरित होती है। (ख) एकांतर जाँच को \emph{ह्रासमान} मापांक चाहिए --- \cref{ex:b2:series:trap}
विहित प्रतिउदाहरण है, और \cref{exo:b2:series:1} की तीसरी श्रेणी उसका अभ्यास। (ग) सप्रतिबंध
अभिसारी श्रेणियों का पुनर्विन्यास नहीं किया जा सकता (\cref{ex:b2:series:rearrange}), और उनके \omterm{thm:b2:series:fubini}{कोशी
गुणनफल} अपसरित हो सकते हैं: $\sum\frac{(-1)^n}{\sqrt{n+1}}$ का वर्ग लेने पर विकर्ण पद
\[
\abs{c_k} = \sum_{m=0}^{k}
\frac{1}{\sqrt{(m+1)(k-m+1)}}
\geq (k+1)\cdot\frac{2}{k+2} \longrightarrow 2 \neq 0
\]
पूरा करते हैं (समांतर--गुणोत्तर माध्य असमिका से हर गुणनखंड अधिकतम $\frac{k+2}2$
है), अतः $\sum c_k$ अपसरित होती है --- कम से कम एक गुणनखंड का निरपेक्ष अभिसरण
(\cref{exo:b2:series:8}) विलासिता नहीं है। (घ) योग्यता परिभाषा से ही \emph{निरपेक्ष}
परिबंधों के बारे में है: सप्रतिबंध \omterm{def:b2:series:summable}{योग्य कुल} जैसी कोई चीज़ होती ही नहीं।
\end{remark}

\section{योग्य कुल}

\begin{definition}\label{def:b2:series:summable}
मान लीजिए $I$ कोई \omterm{def:b2:structures:countable}{गणनीय} सूचकांक समुच्चय है। \emph{अऋणात्मक वास्तविक}
संख्याओं का कुल $(u_i)_{i \in I}$ \emph{योग्य}\index{योग्य कुल} कहलाता है जब उसके
परिमित आंशिक योग परिबद्ध हों; और उसका योग
\[
\sum_{i \in I} u_i = \sup_{F \subseteq I \text{ परिमित}} \sum_{i
\in F} u_i \in \intcc{0}{+\infty} .
\]
है। वास्तविक अथवा सम्मिश्र (या बानाख सदिश) संख्याओं का कुल तब योग्य
कहलाता है जब $(\norm{u_i})$ योग्य हो; और तब उसका योग धनात्मक/ऋणात्मक (अथवा
वास्तविक/काल्पनिक) भागों में बाँटकर परिभाषित होता है --- समतुल्य रूप से,
$I$ की सभी गणनाओं $\sigma$ पर $\sum_{n} u_{\sigma(n)}$ के साझे मान के रूप में (नीचे देखिए)।
\end{definition}

\begin{method}[कौन-सी जाँच चुनें]\label{met:b2:series:choosing}
$\sum u_n$ के सामने, इसी क्रम में: (1) यदि $u_n \not\to 0$, तो अपसरण, बात ख़त्म। (2) यदि
पदों का चिह्न अचर हो, तो तुलना कीजिए: कोई समतुल्य ढूँढ़िए (\cref{ch:b2:comparison}) और उसे
रीमान--बर्ट्रांड मानचित्र पर रखिए। (3) यदि चिह्न \emph{ह्रासमान}
मापांकों के साथ बारी-बारी बदलते हों, तो एकांतर जाँच; और यदि मापांक
एकदिष्ट न हों, तो पद का तब तक प्रसार कीजिए जब तक हर टुकड़ा \omterm{def:b2:series:def}{निरपेक्ष रूप
से} अभिसारी या कोई स्वच्छ जाँच-स्थिति न हो जाए (\cref{ex:b2:series:trap})। (4) यदि चिह्न का
प्रतिरूप दोलनशील पर संरचित हो ($\sin
n\theta$, $\eu^{\iu n\theta}$, आव्यूह घातें), तो परिबद्ध
आंशिक योगों के साथ आबेल जाँच। (5) निरपेक्ष अभिसरण सबसे पहले जाँच लेना
सदा लाभकारी है: वह अधिक मज़बूत है, क्रम-प्रतिरोधी है, और \omterm{thm:b2:series:fubini}{कोशी गुणनफल} तथा
फ़ूबिनी का ताला खोल देता है।
\end{method}

\begin{example}[विकर्ण गिनती से योग्यता]\label{ex:b2:series:diagonalcount}
किन $s > 0$ के लिए कुल $\bigl((m + n)^{-s}\bigr)_{m, n \geq 1}$ \omterm{def:b2:series:summable}{योग्य} है? परिमित आंशिक योगों को विकर्णों $m + n = k$ के
अनुसार समूहबद्ध कीजिए: विकर्ण $k$ पर $k - 1$ युग्म हैं और हर एक $k^{-s}$ का
योगदान देता है, अतः परिमित योग ठीक
\[
\sum_{k \geq 2} \frac{k - 1}{k^{s}} ,
\]
से परिबद्ध हैं (और उसे पूरा भी कर देते हैं) --- यह धनात्मक पदों वाली
श्रेणी है जो $k^{1-s}$ के समतुल्य है: अतः \omterm{def:b2:series:summable}{योग्य} तभी जब $s - 1 > 1$, अर्थात् $s > 2$।
द्विविमीय सूचकांक पूरी एक घात खा जाता है: पदों का कोई समतल किसी रेखा से
``एक विमा अधिक अपसारी'' होता है --- और योग्यता का निर्णय सूचकांक समुच्चय
की गिनती-ज्यामिति करती है, अलग-अलग पदों का आकार नहीं। (इसी जनगणना से यह
भी दिखता है कि $\bigl((m^2 + n^2)^{-1}\bigr)$ \omterm{def:b2:series:summable}{योग्य} नहीं है: विकर्ण $m + n = k$ पर हर पद कम से कम $k^{-2}$ है,
और $(k-1)\cdot k^{-2}$ हरात्मक श्रेणी की तरह जुड़ता है।)
\end{example}

\begin{theorem}[योग्यता और क्रम]\label{thm:b2:series:rearrangement}
\begin{enumerate}
  \item अऋणात्मक कुलों के लिए योग किसी भी गणना के अंतर्गत अपरिवर्त्य
        रहता है: हर एकैकी आच्छादक प्रतिचित्रण $\sigma \colon \N \to I$ के लिए $\sum_{i} u_i = \sum_{n=0}^{\infty}
        u_{\sigma(n)}$।
  \item कोई वास्तविक या सम्मिश्र श्रेणी $\sum u_n$ \emph{क्रमविनिमेय रूप से
        अभिसारी} है (अर्थात् हर पुनर्विन्यास उसी योग पर अभिसरित होता
        है) यदि और केवल यदि वह \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसारी हो।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) प्रत्येक आंशिक योग $\sum_{n \leq N} u_{\sigma(n)}$ कुल का कोई परिमित आंशिक योग है (अतः उच्चतम
से $\leq$); और प्रत्येक परिमित $F$ किसी $\{\sigma(0), \dots, \sigma(N)\}$ में समाहित है (अतः उच्चतम
श्रेणी की सीमा से $\leq$)। दोनों परिबंध मिल जाते हैं।

(2) यदि $\sum\abs{u_n} < \infty$: तो किसी भी पुनर्विन्यास $\sigma$ और $\varepsilon > 0$ के लिए $\sum_{n > N}\abs{u_n} \leq
\varepsilon$ वाला $N$
चुनिए; और जिस कोटि के आगे $\sigma$ $\intint{0}{N}$ को पूरा कर चुका है उससे आगे
पुनर्विन्यस्त आंशिक योग मूल सीमा से अधिकतम $\varepsilon$ भिन्न होते हैं: अर्थात्
वही योग। और यदि $\sum
\abs{u_n} = \infty$ हो पर $\sum u_n$ अभिसरित हो (वास्तविक स्थिति; सम्मिश्र
निर्देशांक-दर-निर्देशांक निकल आती है): तो धनात्मक और ऋणात्मक दोनों भाग
अपसरित होते हैं, और पुनर्विन्यास से किसी भी नियत सीमा तक पहुँचा जा सकता
है --- यह रीमान की प्रमेय है, जिसे \cref{exo:b2:series:5} में करके दिखाया गया है --- अतः
क्रमविनिमेय अभिसरण विफल हो जाता है।
\end{proof}

\begin{example}[रंगे हाथ पकड़ा गया एक पुनर्विन्यास]\label{ex:b2:series:rearrange}
एकांतर हरात्मक श्रेणी का योग $\sum_{n\geq1}\frac{(-1)^{n-1}}{n} = \ln 2$ है (प्रथम वर्ष का खंड)। उसे ``एक
धनात्मक, दो ऋणात्मक'' के रूप में पुनर्विन्यस्त कीजिए:
\[
1 - \frac12 - \frac14 + \frac13 - \frac16 - \frac18 + \frac15 -
\cdots
\]
तीन-तीन के हर खंड को समूहबद्ध करने पर
\[
\frac{1}{2k-1} - \frac{1}{4k-2} - \frac1{4k}
= \frac{1}{4k-2} - \frac{1}{4k}
= \frac12\Bigl(\frac{1}{2k-1} - \frac1{2k}\Bigr),
\]
अतः पुनर्विन्यस्त श्रेणी $\frac12\ln 2$ पर अभिसरित होती है --- अर्थात् मूल योग का
आधा, और वह भी ठीक उन्हीं पदों से। जो श्रेणियाँ \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसारी
नहीं हैं वे अपने पदों का क्रम याद रखती हैं; और \omterm{def:b2:series:summable}{योग्य कुल} ठीक वे हैं जो
नहीं रखते।
\end{example}

\begin{example}[समूहबद्ध करना सुरक्षित है, समूह तोड़ना नहीं]\label{ex:b2:series:grouping}
किसी \emph{अभिसारी} श्रेणी के क्रमागत पदों को समूहबद्ध करने से योग कभी
नहीं बदलता: समूहबद्ध आंशिक योग मूल आंशिक योगों का उपअनुक्रम बनाते हैं।
उलटी क्रिया वर्जित है:
\[
(1 - 1) + (1 - 1) + (1 - 1) + \cdots = 0 + 0 + \cdots = 0,
\]
फिर भी बिना समूह वाली $1 - 1 + 1 - 1 + \cdots$ अपसरित होती है (आंशिक योग $1$ और $0$ के बीच
दोलन करते हैं)। समूह तोड़ना केवल किसी क्षतिपूरक परिकल्पना के साथ वैध है
--- उदाहरण के लिए, पद $0$ की ओर जाते हों और समूहों की लंबाइयाँ परिबद्ध
हों: तब दो समूहबद्ध आंशिक योगों के बीच मूल योग अधिकतम परिबद्ध संख्या में
$o(1)$ पदों जितना ही भटकते हैं, और अभिसरण वापस स्थानांतरित हो जाता है। ठीक
यही वह शर्त है जिसके अंतर्गत \cref{ex:b2:series:rearrange} का खंड-परिकलन उपपत्ति है, हाथ की सफ़ाई
नहीं।
\end{example}

\begin{theorem}[कुलों के लिए फ़ूबिनी; कोशी गुणनफल]\label{thm:b2:series:fubini}
मान लीजिए $(u_{m,n})_{(m,n) \in \N^2}$ कोई \omterm{def:b2:series:summable}{योग्य} दोहरा कुल है (अर्थात् $\sup_F \sum_F \abs{u_{m,n}} < \infty$)। तब\index{श्रेणियों
के लिए फ़ूबिनी प्रमेय}
\[
\sum_{(m,n)} u_{m,n}
= \sum_{m=0}^{\infty}\Bigl(\sum_{n=0}^{\infty} u_{m,n}\Bigr)
= \sum_{n=0}^{\infty}\Bigl(\sum_{m=0}^{\infty} u_{m,n}\Bigr)
= \sum_{k=0}^{\infty}\Bigl(\sum_{m+n=k} u_{m,n}\Bigr),
\]
और सभी भीतरी श्रेणियाँ (\omterm{def:b2:series:def}{निरपेक्ष रूप से}) अभिसरित होती हैं। विशेष रूप से
यदि $\sum
a_m$ और $\sum b_n$ \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसरित हों, तो उनका \emph{कोशी
गुणनफल}\index{कोशी गुणनफल} \omterm{def:b2:series:def}{निरपेक्ष रूप से} अभिसरित होता है, जहाँ
\[
\Bigl(\sum_m a_m\Bigr)\Bigl(\sum_n b_n\Bigr)
= \sum_{k=0}^{\infty} c_k,
\qquad
c_k = \sum_{m=0}^{k} a_m b_{k-m} .
\]
\end{theorem}

\begin{proof}
\emph{अऋणात्मक स्थिति।} हर समूहन (पंक्तियों, स्तंभों अथवा विकर्णों के
अनुसार) वही उच्चतम परिकलित करता है: युग्मों का कोई भी परिमित समुच्चय
पंक्तियों के किसी परिमित खंड में समाहित होता है (जिससे हर समूहबद्ध योग
नीचे से परिमित आंशिक योगों और ऊपर से कुल योग से परिबद्ध हो जाता है), और
शेष काम आंशिक योगों का एकदिष्ट अभिसरण कर देता है --- मूर्त रूप में,
पंक्तियों के लिए $\sum_{m \leq M}\sum_{n \leq N} u_{m,n} \leq S$ देता है, और पहले $N
\to \infty$ फिर $M \to \infty$ लेने पर $\sum_m \sum_n u_{m,n} \leq S$; विलोमतः
हर परिमित $F$ ऐसे ही किसी आयत में बैठता है, अतः $S \leq
\sum_m\sum_n u_{m,n}$। विकर्ण: वही दोनों
परिबंध, बस आयतों के स्थान पर त्रिभुज।

\emph{सामान्य स्थिति।} धनात्मक और ऋणात्मक (वास्तविक और काल्पनिक) भागों
में बाँटिए, जिनमें से हर एक \omterm{def:b2:series:summable}{योग्य} अऋणात्मक कुल है; चारों समूहन हर भाग पर
सहमत हैं, अतः अंतर पर भी; और भीतरी श्रेणियों का निरपेक्ष अभिसरण $\abs{u_{m,n}}$ पर
लगाई गई अऋणात्मक स्थिति से आता है।

\emph{\omterm{thm:b2:series:fubini}{कोशी गुणनफल}।} कुल $u_{m,n} = a_m b_n$ \omterm{def:b2:series:summable}{योग्य} है: $\abs{a_mb_n}$ के परिमित आंशिक योग $\bigl(\sum\abs{a_m}\bigr)\bigl(\sum\abs{b_n}\bigr)$ से
परिबद्ध हैं। पंक्तियाँ $\bigl(\sum a_m\bigr)\bigl(\sum b_n\bigr)$ देती हैं; और विकर्ण $\sum_k
c_k$।
\end{proof}

\begin{example}[घातांकी सर्वसमिका, ईमानदारी से]\label{ex:b2:series:exp}
$a, b \in \C$ के लिए (अथवा क्रमविनिमेय आव्यूहों के लिए):
\[
\Bigl(\sum_m \frac{a^m}{m!}\Bigr)\Bigl(\sum_n
\frac{b^n}{n!}\Bigr)
= \sum_k \sum_{m+n=k} \frac{a^m b^n}{m!\,n!}
= \sum_k \frac{(a + b)^k}{k!},
\]
जो हर विकर्ण पर द्विपद प्रमेय से आता है: $\eu^a \eu^b = \eu^{a+b}$ --- अर्थात् $\exp$ का
फलनात्मक समीकरण केवल श्रेणी से व्युत्पन्न। (क्रमविनिमेयता का उपयोग
द्विपद वाले पग में हुआ है; और अक्रमविनिमेय आव्यूहों के लिए यह सर्वसमिका
सचमुच विफल हो जाती है, \cref{ch:b2:diffeq}।)
\end{example}

\begin{example}[परिकलन के यंत्र के रूप में कोशी गुणनफल]\label{ex:b2:series:squaresum}
गुणोत्तर श्रेणी और \cref{exo:b2:series:2} के $\sum_{n
\geq 1} nz^n = \frac{z}{(1-z)^2}$ ($\abs z < 1$) से, एक और \omterm{thm:b2:series:fubini}{कोशी गुणनफल} दूसरा आघूर्ण
पूरा कर देता है। $\sum_m
mz^m$ को $\sum_n z^n$ से गुणा कीजिए: विकर्ण गुणांक $\sum_{m=0}^k
m = \frac{k(k+1)}2$ है, अतः
\[
\frac{z}{(1-z)^3} = \sum_{k\geq0}\frac{k(k+1)}{2}\,z^k ,
\]
और सर्वसमिका $n^2 = 2\cdot\frac{n(n+1)}2 - n$ जोड़ देती है
\[
\sum_{n\geq1} n^2z^n = \frac{2z}{(1-z)^3} - \frac{z}{(1-z)^2}
= \frac{z(1+z)}{(1-z)^3} .
\]
$z = \frac12$ पर: $\sum_{n\geq1}\frac{n^2}{2^n} =
\frac{\frac12\cdot\frac32}{\frac18} = 6$ --- अर्थात् कहीं भी अवकलन किए बिना एक बंद मान, केवल
निरपेक्ष अभिसारी श्रेणियों को बहुपदों की तरह गुणा करके। इन्हीं
सर्वसमिकाओं का क्रमिक निरसन हर $\sum n^dz^n$ परिकलित कर देता है, और प्रायिकता के
जानकार इसमें गुणोत्तर वितरण का दूसरा क्रमगुणित आघूर्ण पहचान लेंगे (\cref{ch:b2:genfun})।
\end{example}

\begin{example}[एक दोहरे योग का मूल्यांकन]\label{ex:b2:series:doublesum}
वास्तविक $s > 1$ के लिए $\zeta(s) = \sum_{n\geq1} n^{-s}$ लीजिए। दोहरे योग से भाजक गिनने पर --- $(m,n) \in (\N^*)^2$ पर
कुल $(m^{-s}n^{-s})$ \omterm{def:b2:series:summable}{योग्य} है (अभिसारी धनात्मक श्रेणियों का गुणन) --- और गुणनफल $q = mn$
के अनुसार समूहबद्ध करने पर:
\[
\zeta(s)^2 = \sum_{m,n} \frac{1}{(mn)^s}
= \sum_{q=1}^{\infty} \frac{d(q)}{q^s},
\]
जहाँ $d(q)$ $q$ के भाजकों की संख्या है। \omterm{def:b2:series:summable}{योग्य कुल} संचयशास्त्र को विश्लेषण
में बदल देते हैं।
\end{example}

\begin{example}[एक फ़ूबिनी मूल्यांकन: $\sum_n (\zeta(n) - 1) = 1$]\label{ex:b2:series:zetasum}
पूर्णांक $n \geq 2$ के लिए $\zeta(n) - 1 = \sum_{k \geq 2} k^{-n}$। दोहरा कुल $(k^{-n})_{k, n \geq 2}$ \omterm{def:b2:series:summable}{योग्य} है: पहले गुणोत्तर स्तंभों
का योग लेने पर
\[
\sum_{k\geq2}\sum_{n\geq2} \frac{1}{k^n}
= \sum_{k\geq2} \frac{1/k^2}{1 - 1/k}
= \sum_{k\geq2} \frac{1}{k(k-1)} = 1
\]
(क्रमिक निरसन), और सारे पद धनात्मक हैं, अतः \cref{thm:b2:series:fubini} पंक्तियों के अनुसार योग
लेने की अनुमति दे देता है:
\[
\sum_{n\geq2}\bigl(\zeta(n) - 1\bigr) = 1 .
\]
अपरिमित $\zeta$-मान, जिनमें से हर एक अबीजीय-सा दिखता है, अपनी पुच्छों सहित
जुड़कर ठीक $1$ दे देते हैं। समापन दृष्टि: जब किसी दोहरे योग के पद
धनात्मक हों, तब उसे उसी क्रम में परिकलित कीजिए जिसमें वह सिमट जाए ---
यहाँ स्तंभ गुणोत्तर हैं, पंक्तियाँ रहस्यमय, और फ़ूबिनी उस सिमटने को
स्थानांतरित कर देती है।
\end{example}

\begin{example}[गुणोत्तर श्रेणी एक समीकरण हल करती है]\label{ex:b2:series:neumannsolve}
\omterm{def:b2:nvs:banach}{बानाख समष्टि} $\bigl(C(\intcc01),
\norm\cdot_\infty\bigr)$ में $x - K(x) = y$ हल कीजिए, जहाँ $K(f)$ अचर फलन $\frac12\int_0^1 f$ है। \omterm{thm:b2:nvs:continuouslinear}{संकारक
मानदंड} $\vertiii K \leq \frac12 < 1$ है, अतः नॉयमान श्रेणी लागू होती है (\cref{ex:b2:series:neumann}): $x = \sum_{n\geq0} K^n(y)$। पुनरावृत्त
परिकलित कीजिए: $K(y) = \frac12\int_0^1 y$ (एक अचर), और किसी अचर $c$ पर $K$ लगाने से $\frac c2$
मिलता है, अतः $n \geq
1$ के लिए $K^n(y) = \frac{1}{2^{n-1}}\cdot\frac12\int_0^1 y$। गुणोत्तर अचरों का योग लेने पर:
\[
x = y + \Bigl(\int_0^1 y\Bigr)
\sum_{n\geq1}\frac{1}{2^n} = y + \int_0^1 y .
\]
जाँच: $x - K(x) = y + \int y - \frac12\bigl(\int y + \int
y\bigr) = y$। एक अनंत श्रेणी, एक परिमित उत्तर, और एक पंक्ति का सत्यापन ---
गुणोत्तर श्रेणी प्रतिलोमन की कलनविधि है, केवल अभिसरण का कथन नहीं।
\end{example}

\begin{example}[आंशिक भिन्नों से क्रमिक निरसन]\label{ex:b2:series:telescope}
यथार्थ योग विरल हैं; और उनका मुख्य आपूर्तिकर्ता क्रमिक निरसन है। अपघटन
कीजिए
\[
\frac{1}{n(n+1)(n+2)}
= \frac{1}{2}\Bigl(\frac{1}{n(n+1)} -
\frac{1}{(n+1)(n+2)}\Bigr),
\]
(साझे हर पर लाकर जाँच लीजिए), अतः आंशिक योग सिमट जाते हैं:
\[
\sum_{n=1}^{N}\frac{1}{n(n+1)(n+2)}
= \frac12\Bigl(\frac{1}{1\cdot2} -
\frac{1}{(N+1)(N+2)}\Bigr)
\longrightarrow \frac14 .
\]
यही ढर्रा --- पद को किसी स्पष्ट $(u_n)$ के लिए $c(u_n - u_{n+1})$ के रूप में लिखना --- \cref{exo:b2:series:10}
(चापज्या) को हल कर चुका है और हर $\sum\frac{1}{n(n+1)\cdots(n +
k)} = \frac{1}{k\cdot k!}$ परिकलित कर देता है। इस स्तर पर जब
कोई यथार्थ योग विद्यमान होता है, तब प्रायः पद के भीतर कोई दूरबीन छिपी
होती है।
\end{example}

\begin{remark}[इसी खंड के भीतर के परिप्रेक्ष्य]\label{rem:b2:series:perspectives}
आगे के तीन अध्याय सीधे ग्राहक हैं। \cref{ch:b2:funcseq} के लिए: $\sum f_n$ का सामान्य अभिसरण
\omterm{def:b2:nvs:banach}{बानाख समष्टि} $\bigl(C, \norm\cdot_\infty\bigr)$ में $\sum\norm{f_n}_\infty$ का निरपेक्ष अभिसरण ही है --- अर्थात् इस अध्याय
का \cref{thm:b2:nvs:absoluteconvergence} वेश बदलकर। \cref{ch:b2:powerseries} के लिए: अभिसरण चक्रिका के भीतर सब कुछ निरपेक्ष और
\omterm{def:b2:series:summable}{योग्य} है, अतः \omterm{thm:b2:series:fubini}{कोशी गुणनफल} और पुनर्विन्यास खुलकर चलते हैं (इसीलिए घात
श्रेणियाँ बहुपदों की तरह गुणा होती हैं); और सीमा पर आबेल की जाँच मोर्चा
सँभाल लेती है (\cref{ex:b2:series:abelboundary})। \cref{ch:b2:genfun} के लिए: प्रायिकता जनक फलन ऐसी घात श्रेणियाँ हैं
जिनके सारे प्रबंधन --- स्वतंत्र चरों के योग के लिए गुणनफल, संयुक्त
वितरणों के लिए दोहरे योग --- \cref{thm:b2:series:fubini} से अधिकृत हैं। \omterm{def:b2:series:summable}{योग्य कुल} आगे आने वाले
विश्लेषण का विधि-विभाग हैं।
\end{remark}

\begin{remark}[यह अध्याय कहाँ काम आता है]
जिस भी चीज़ में अनंत योग है वह यहीं से होकर जाती है: घात श्रेणियाँ (\cref{ch:b2:powerseries})
भेस बदले हुए \omterm{def:b2:series:summable}{योग्य कुल} हैं, फूरिये गुणांक कोशी गुणनफलों से गुणा होते हैं
और पारसेवाल से पुनर्विन्यस्त (\cref{ch:b2:fourier}), और प्रायिकता जनक फलन (\cref{ch:b2:genfun})
प्रत्याशाओं पर लगाई गई फ़ूबिनी प्रमेय ही हैं। तृतीय वर्ष का खंड \omterm{def:b2:series:summable}{योग्य}
कुलों को गणन माप पर लेबेग समाकलन में समो लेता है --- जहाँ \cref{thm:b2:series:fubini}
फ़ूबिनी--तोनेली प्रमेय की एक विशेष स्थिति बन जाती है।
\end{remark}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:5}
(रीमान पुनर्विन्यास) मान लीजिए $\sum u_n$ कोई अभिसारी पर \omterm{def:b2:series:def}{निरपेक्ष रूप से}
अभिसारी न होने वाली वास्तविक श्रेणी है, और $\ell \in \R$। सिद्ध कीजिए कि $\sum u_n$ का
कोई पुनर्विन्यास $\ell$ पर अभिसरित होता है। \emph{(दिखाइए कि धनात्मक और
ऋणात्मक पदों की दोनों उपश्रेणियाँ अपसरित होती हैं; फिर लालची ढंग से
बारी-बारी लीजिए: $\ell$ से आगे निकलने तक धनात्मक पद, फिर उससे नीचे गिरने
तक ऋणात्मक, और इसी तरह; और पद $0$ की ओर जाते हैं, जो $\ell$ पर अभिसरण को
बाध्य कर देता है।)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:6}
सिद्ध कीजिए कि कुल $\Bigl(\dfrac{x^{m+n}}{m!\,n!}\Bigr)_{(m,n)\in\N^2}$ प्रत्येक $x \in \R$ के लिए \omterm{def:b2:series:summable}{योग्य} है, और दोहरे योग को
विकर्णों $m + n =
k$ के अनुदिश समूहबद्ध करके सर्वसमिका $(\eu^x)^2 =
\eu^{2x}$ पुनः निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:7}
सिद्ध कीजिए कि कुल $\bigl(\frac{1}{m^2 n^2}\bigr)_{m,n \geq 1}$ \omterm{def:b2:series:summable}{योग्य} है, और $\gcd$ के अनुसार समूहबद्ध करने पर, $q = \gcd(m,n)$
के साथ,
\[
\zeta(2)^2 = \sum_{q\geq1} \frac{1}{q^4}
\sum_{\substack{a,b \geq 1\\ \gcd(a,b)=1}} \frac{1}{a^2b^2}
= \zeta(4) \cdot S,
\]
जहाँ $S = \sum_{\gcd(a,b)=1} \frac{1}{a^2b^2}$: इससे $S =
\zeta(2)^2/\zeta(4)$ निष्कर्ष निकालिए। \emph{(हर युग्म $(m,n)$ $\gcd(a,b) = 1$ के साथ
अद्वितीय रूप से $(qa, qb)$ लिखा जाता है।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:8}
(गुणनफलों पर आबेल की प्रमेय, हल्का रूप) मान लीजिए $\sum a_n$ \omterm{def:b2:series:def}{निरपेक्ष रूप से}

अभिसरित होता है, जहाँ $\sum c_n = (\sum
a_n)(\sum b_n)$। \emph{($C_N = \sum_{k\leq N} c_k = \sum_n
a_n B_{N-n}$ लिखिए, जहाँ $B$ $b$ के आंशिक योग
हैं; फिर $(B_m)$ की परिबद्धता और $(a_n)$ की निरपेक्ष पुच्छ का उपयोग करके $n \leq N/2$
के अनुसार बाँटिए।)}
\end{exercise}

\end{exercise}

\section{समस्या: ऑयलर का $\zeta(2) = \pi^2/6$, कोशी के कोस्पर्शज्या योग से}

ऑयलर की सबसे प्रसिद्ध सर्वसमिका $1 + \frac14 + \frac19 + \cdots =
\frac{\pi^2}6$ की एक पूरी तरह प्रारंभिक उपपत्ति है,
जो कोशी की देन है: डी म्वाव्र का सूत्र ऐसा बहुपद देता है जिसके मूल
संख्याएँ $\cot^2\frac{k\pi}{2n+1}$ हैं, वियेत उन मूलों का यथार्थ योग दे देता है, और निचोड़
$\cot^2\theta < \frac{1}{\theta^2} <
1 + \cot^2\theta$ $\sum\frac1{k^2}$ के आंशिक योगों को दो स्पष्ट परिमेय परिबंधों के बीच कुचल देता है।
हम यह उपपत्ति पूरी चलाते हैं, उसी विधि से $\zeta(4) = \frac{\pi^4}{90}$ निकालते हैं, फिर
बर्ट्रांड श्रेणी के साथ अभिसरण और अपसरण के बीच के पूरे सीमांत का मानचित्र
बनाते हैं --- और सिद्ध करते हैं कि उस सीमांत पर सबसे धीमी अभिसारी श्रेणी
जैसी कोई चीज़ है ही नहीं।

\begin{problem}[{सप्ताहांत समस्या --- $\zeta(2) = \pi^2/6$ और बर्ट्रांड
परिदृश्य}]
\label{pb:b2:series:1}
आगे सर्वत्र $k
= 1, \dots, n$ के लिए $n \geq 1$ और $\theta_k = \dfrac{k\pi}{2n+1}$; ध्यान दीजिए कि $0 < \theta_k < \frac\pi2$।

\medskip
\noindent\textbf{भाग I --- कोस्पर्शज्या सर्वसमिका।}
\begin{enumerate}
  \item डी म्वाव्र का सूत्र $(\cos\theta +
        \iu\sin\theta)^m = \cos m\theta + \iu\sin m\theta$ ($m
        \in \N$) सिद्ध कीजिए, और $m = 2n + 1$ के लिए
        निष्कर्ष रूप में निकालिए
        \[
        \sin\bigl((2n{+}1)\theta\bigr) = \sum_{j=0}^{n}
        (-1)^j\binom{2n+1}{2j+1}\cos^{2(n-j)}\theta\,
        \sin^{2j+1}\theta .
        \]
  \item इससे निष्कर्ष निकालिए कि $\theta \in \intoo{0}{\frac\pi2}$ के लिए
        \[
        \sin\bigl((2n{+}1)\theta\bigr) =
        \sin^{2n+1}\theta\; P_n(\cot^2\theta),
        \qquad
        P_n(x) = \sum_{j=0}^{n}
        (-1)^j\binom{2n+1}{2j+1}x^{\,n-j},
        \]
        जो $2n
        + 1$ अग्र गुणांक वाला घात $n$ का बहुपद है।
  \item दिखाइए कि $x_k = \cot^2\theta_k$, $k = 1, \dots, n$, $P_n$ के $n$ \emph{भिन्न} मूल हैं --- अतः
        वही सब मूल हैं।
  \item वियेत से यथार्थ सर्वसमिका सिद्ध कीजिए
        \[
        \sum_{k=1}^{n} \cot^2\frac{k\pi}{2n+1}
        = \frac{n(2n-1)}{3}.
        \]
  \item $\displaystyle\sum_{k=1}^{n}
        \frac{1}{\sin^2\theta_k} = \frac{2n(n+1)}{3}$ भी निष्कर्ष रूप में निकालिए।
  \item निचोड़ सिद्ध कीजिए: $\theta \in
        \intoo{0}{\frac\pi2}$ के लिए $\cot^2\theta < \dfrac1{\theta^2} <
        \dfrac{1}{\sin^2\theta}$ \emph{($\sin\theta < \theta <
        \tan\theta$ से)}।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- निचोड़ बंद होता है: ऑयलर की प्रमेय।}
\begin{enumerate}[resume]
  \item प्रश्न 6 का $\theta =
        \theta_k$ के साथ $k = 1, \dots, n$ पर योग लेकर स्थापित कीजिए
        \[
        \frac{n(2n-1)}{3} \;<\;
        \frac{(2n+1)^2}{\pi^2}\sum_{k=1}^{n}\frac1{k^2}
        \;<\; \frac{2n(n+1)}{3}.
        \]
  \item निष्कर्ष निकालिए (\emph{ऑयलर की प्रमेय, कोशी की उपपत्ति से}):
        \[
        \zeta(2) = \sum_{k=1}^{\infty}\frac{1}{k^2} =
        \frac{\pi^2}{6}.
        \]
  \item इस निचोड़ से एक दर निकालिए: दिखाइए
        \[
        \Bigl|\sum_{k=1}^{n}\frac1{k^2} - \frac{\pi^2}6\Bigr|
        = O\Bigl(\frac1n\Bigr),
        \]
        जो \cref{exo:b2:comparison:11} की यथार्थ पुच्छ $\sum_{k>n}k^{-2} =
        \frac1n - \frac{1}{2n^2} + O(n^{-3})$ के अनुरूप है।
  \item यंत्र को एक मंज़िल ऊपर चलाइए: $P_n$ के दूसरे वियेत फलन का उपयोग
        करके दिखाइए
        \[
        \sum_{k=1}^{n}\cot^4\theta_k =
        \Bigl(\frac{n(2n-1)}3\Bigr)^{\!2} -
        \frac{2n(2n-1)(2n-2)(2n-3)}{60}
        \;\sim\; \frac{8n^4}{45},
        \]
        और $\cot^4 < \theta^{-4} < (1 +
        \cot^2)^2$ से निचोड़कर $\zeta(4) = \dfrac{\pi^4}{90}$ प्राप्त कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- लाभ।}
\begin{enumerate}[resume]
  \item $\zeta(2) = \frac{\pi^2}6$ से निष्कर्ष निकालिए:
        \[
        \sum_{k\geq0}\frac{1}{(2k+1)^2} = \frac{\pi^2}{8},
        \qquad
        \sum_{k\geq1}\frac{(-1)^{k-1}}{k^2} = \frac{\pi^2}{12}.
        \]
  \item \cref{exo:b2:series:7} के साथ मिलाइए: $S =
        \sum_{\gcd(a,b)=1}\frac{1}{a^2b^2} =
        \frac{\zeta(2)^2}{\zeta(4)} = \frac52$ परिकलित कीजिए, और $\frac{1}{\zeta(2)} = \frac{6}{\pi^2} \approx 0.608$ को सहअभाज्य
        युग्मों के घनत्व के रूप में समझाइए (अनुमान ईमानदारी से कहिए:
        कठोर गिनती तृतीय वर्ष के खंड का विषय है)।
  \item (प्रमाणित त्वरण) प्रश्न 9 का पुच्छ-सूत्र $\sum_{k\leq n}k^{-2} + \frac1n - \frac1{2n^2} =
        \frac{\pi^2}6 + O(n^{-3})$ देता है। $\zeta(2)$ के छह
        अंकों के लिए आवश्यक श्रम की तुलना कीजिए: सीधा योग बनाम $n = 100$ पर
        सुधारा हुआ योग (जहाँ त्रुटि $1.7\cdot10^{-7}$ है)।
  \item प्रश्न 4 को $n = 1$ और $n = 2$ पर हाथ से जाँचिए (मान $\cot^2\frac\pi3 = \frac13$ और $\cot^2\frac\pi5 + \cot^2\frac{2\pi}5 = 2$),
        और इसके लिए $\cos\frac\pi5 = \frac{1+\sqrt5}4$ अथवा कोई संख्यात्मक मूल्यांकन काम में लीजिए।
  \item सहचर सर्वसमिका
        \[
        \sum_{k=1}^{n}\tan^2\frac{k\pi}{2n+1} = n(2n+1)
        \]
        सिद्ध कीजिए \emph{(संख्याएँ $\tan^2\theta_k$ उलटे बहुपद $x^nP_n(1/x)$ के मूल हैं)},
        और उसे $n
        = 1$ पर सत्यापित कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- बर्ट्रांड परिदृश्य।} $\alpha,
\beta \in \R$ के लिए
\emph{बर्ट्रांड श्रेणी}
\[
\sum_{n \geq 3} \frac{1}{n^{\alpha}(\ln n)^{\beta}} .
\]
लीजिए।
\begin{enumerate}[resume]
  \item दिखाइए कि $\alpha > 1$ के लिए श्रेणी अभिसरित होती है, $\beta$ चाहे जो हो
        \emph{($n^{-(1+\alpha)/2}$ से तुलना कीजिए)}।
  \item दिखाइए कि $\alpha < 1$ के लिए वह अपसरित होती है, $\beta$ चाहे जो हो।
  \item $\alpha = 1$ के लिए: $f(t) =
        \frac{1}{t(\ln t)^\beta}$ के साथ श्रेणी--समाकल तुलना (\cref{thm:b2:comparison:seriesintegral}) का उपयोग करके
        सिद्ध कीजिए कि अभिसरण तभी होता है जब $\beta
        > 1$।
  \item सीमांत को दोहराइए: दिखाइए कि $\sum\frac{1}{n\ln n\,\ln\ln
        n}$ अपसरित होती है जबकि $\sum\frac{1}{n\ln n\,(\ln\ln n)^2}$
        अभिसरित।
  \item दो जाल: इनकी प्रकृति निर्धारित कीजिए
        \[
        \sum_n \frac{1}{n^{1 + 1/\ln n}}
        \qquad\text{तथा}\qquad
        \sum_n \frac{1}{n^{1 + 1/\ln\ln n}}
        \]
        \emph{($n^{1/\ln n}$ यथार्थ रूप से परिकलित कीजिए; $n^{1/\ln\ln n}$ की तुलना $\ln n$ की
        हर घात से कीजिए)}।
  \item (सबसे धीमी अभिसारी श्रेणी होती ही नहीं) मान लीजिए $\sum a_n$ $a_n > 0$
        वाली कोई भी अभिसारी श्रेणी है, और $R_n = \sum_{k
        \geq n}a_k$ उसकी पुच्छ। सिद्ध कीजिए
        कि $\sum
        \frac{a_n}{\sqrt{R_n}}$ फिर भी \emph{अभिसरित} होती है \emph{(क्रमिक निरसन $2(\sqrt{R_n} -
        \sqrt{R_{n+1}})$
        से तुलना कीजिए)}, यद्यपि $\frac{a_n/\sqrt{R_n}}{a_n}
        \to \infty$: अर्थात् हर अभिसारी श्रेणी पर कोई
        दूसरी अभिसारी श्रेणी यथार्थतः हावी हो जाती है। अभिसरण का सीमांत
        कोई वक्र नहीं, कोहरा है।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- प्रतिजाँचें और संश्लेषण।}
\begin{enumerate}[resume]
  \item (कोशी संघनन) सिद्ध कीजिए: धनात्मक ह्रासमान $(a_n)$ के लिए $\sum a_n$
        अभिसरित होती है यदि और केवल यदि $\sum 2^k a_{2^k}$ अभिसरित हो। इससे प्रश्न 18
        का सीमांत पुनः निकालिए।
  \item (सुस्ती की क़ीमत) $\sum\frac1{n(\ln n)^2}$ के लिए पुच्छ को किसी समाकल से परिबद्ध
        कीजिए और दिखाइए कि $N = 10^6$ तक योग लेने पर भी त्रुटि $0.07$ से बड़ी रह
        जाती है: अर्थात् सिद्धांत से प्रमाणित अभिसरण संख्यात्मकी के लिए
        बेकार हो सकता है --- और प्रश्न 13 से इसका अंतर देखिए।
  \item एक-एक पंक्ति के औचित्य के साथ वर्गीकृत कीजिए: $\sum\frac1{n\ln n}$, $\sum\frac1{n^{1.01}}$, $\sum\frac{(\ln n)^{100}}{n^{1.001}}$,
        $\sum\frac1{n(\ln n)(\ln\ln n)^{3}}$।
  \item (संश्लेषण) एक-एक वाक्य में: डी म्वाव्र ने किसी त्रिकोणमितीय
        सर्वसमिका को ऐसे बहुपद में कैसे बदला जिसके मूलों का योग परिकलनीय
        था; निचोड़ को समतुल्यों के बदले \emph{यथार्थ} अंत्यबिंदु
        सर्वसमिकाएँ कहाँ चाहिए थीं; \cref{ch:b2:comparison} के किस औज़ार ने भाग IV चलाया;
        और प्रश्न 21 ``सार्वत्रिक तुलना-जाँच'' के स्वप्न के बारे में
        क्या कहता है। दोनों शिखर नाम लीजिए: ऑयलर का $\zeta(2) = \frac{\pi^2}6$ (और उसकी ऊपरी
        मंज़िल $\zeta(4) = \frac{\pi^4}{90}$), तथा बर्ट्रांड वर्गीकरण। और यह भी लिखिए कि $\zeta(2)$
        फिर कहाँ सिद्ध होगा: \cref{ch:b2:fourier} में पारसेवाल से --- एक प्रमेय, दो
        सभ्यताएँ।
\end{enumerate}
\end{problem}
