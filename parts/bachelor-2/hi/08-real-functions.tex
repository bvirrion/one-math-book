\chapter{एक वास्तविक चर के फलन}\label{ch:b2:realfun}

इससे पहले कि विश्लेषण फलनों के फलनों की ओर बढ़े (\cref{ch:b2:funcseq}), एक-चर वाले
परिदृश्य को प्रथम वर्ष की आवश्यकता से अधिक बारीकी से जान लेना लाभकारी
है: कोई एकदिष्ट फलन कितना असंतत हो सकता है, कोई उत्तल फलन कितना नियमित
होना ही पड़ता है, और अवकलजों को कौन-से विशेष गुणधर्म प्राप्त हैं
(डार्बू)। ये संरचनात्मक परिणाम छोटे हैं, तीखे हैं, और परीक्षकों को बहुत
प्रिय।

\section{एकदिष्ट फलन}

\begin{theorem}[एकदिष्ट फलनों की नियमितता]\label{thm:b2:realfun:monotone}
मान लीजिए $f \colon I \to \R$ किसी अंतराल पर वर्धमान है।
\begin{enumerate}
  \item प्रत्येक अभ्यंतरीय बिंदु $a$ पर एकपक्षीय सीमाएँ विद्यमान होती
        हैं:
        \[
        f(a^-) = \sup_{x < a} f(x) \;\leq\; f(a) \;\leq\;
        f(a^+) = \inf_{x > a} f(x) ;
        \]
        अर्थात् हर असंततता कोई \emph{उछाल} है।
  \item $f$ की असंततताओं का समुच्चय अधिकतम \omterm{def:b2:structures:countable}{गणनीय} होता
        है।\index{एकदिष्ट फलन!असंततताएँ}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) समुच्चय $\{f(x) : x < a\}$ अरिक्त है और ऊपर से $f(a)$ द्वारा परिबद्ध: उसका उच्चतम $s$
$x \to a^-$ होने पर $f(x) \to s$ पूरा करता है ($\varepsilon$ दिया हो तो कोई $f(x_0) > s - \varepsilon$, और एकदिष्टता $x \in
\intoo{x_0}{a}$
के लिए $f(x) \in \intoc{s - \varepsilon}{s}$ को फँसा लेती है)। दाईं ओर सममित रूप से।

(2) हर असंततता $a$ के साथ अरिक्त \omterm{def:b2:metric:topology}{विवृत} अंतराल $J_a = \intoo{f(a^-)}{f(a^+)}$ (एक सच्चा उछाल) जोड़
दीजिए। $a < b$ असंततताओं के लिए $J_a$ और $J_b$ असंयुक्त हैं: बीच के किसी भी $c$
के लिए $f(a^+) \leq f(c)
\leq f(b^-)$। हर $J_a$ में कोई परिमेय संख्या है; और भिन्न असंततताओं को भिन्न
परिमेय मिलती हैं: अर्थात् असंतति-समुच्चय का $\Q$ में एकैकी प्रतिचित्रण,
जो \omterm{def:b2:structures:countable}{गणनीय} है (\cref{prop:b2:structures:countablestable})।
\end{proof}

\begin{example}
यह परिबंध तीखा है: $\Q \cap
\intoo{0}{1}$ की कोई गणना $(r_n)$ स्थिर कीजिए और $f(x) = \sum_{n : r_n \leq x} 2^{-n}$ रखिए (यह
योग्य-कुल वाली परिभाषा है, \cref{def:b2:series:summable})। तब $f$ $\intcc{0}{1}$ पर वर्धमान है और ठीक $\intoo{0}{1}$
की हर परिमेय संख्या पर असंतत ($r_n$ पर उछाल $2^{-n}$): अर्थात् कोई एकदिष्ट
फलन किसी सघन \omterm{def:b2:structures:countable}{गणनीय समुच्चय} पर असंतत हो \emph{सकता} है।
\end{example}

\begin{example}[उछाल चढ़ाई से भारी नहीं हो सकते]\label{ex:b2:realfun:jumpbudget}
$\intcc{a}{b}$ पर वर्धमान $f$ के लिए उछालों का एक बजट होता है: यदि $a < c_1 < \dots < c_m < b$ उछालों $s_i = f(c_i^+) - f(c_i^-) > 0$
वाली असंततताएँ हों, तो बीच-बीच के बिंदु $a < c_1 < t_1 < c_2 < \dots$ चुनकर और हर टुकड़े पर
एकदिष्टता का उपयोग करके
\[
\sum_{i=1}^{m} s_i \;\leq\; f(b) - f(a) :
\]
अर्थात् कुल चढ़ाई कुल उछाल को परिबद्ध कर देती है। परिणाम: प्रत्येक $k$
के लिए अधिकतम $k\,\bigl(f(b) - f(a)\bigr)$ असंततताओं का उछाल $\geq \frac1k$ होता है --- यह \cref{thm:b2:realfun:monotone}~(2) का
परिमाणात्मक परिष्कार है, क्योंकि असंतति-समुच्चय $k$ पर इन्हीं परिमित
समुच्चयों का \omterm{def:b2:structures:countable}{गणनीय} सम्मिलन है। ऊपर के परिमेय-उछाल वाले फलन पर यह बजट
पूरा-का-पूरा ख़र्च होता है: उछाल $2^{-n}$ स्पष्ट विस्तारित अर्थ में $1 = f(1^+) - f(0^-)$ तक
जुड़ जाते हैं। एकदिष्ट फलन सघन रूप से उछल सकते हैं, पर केवल एक कड़े
भत्ते के भीतर।
\end{example}

\section{उत्तल फलन}

\begin{lemma}[ढाल असमिका]\label{lem:b2:realfun:slopes}
मान लीजिए $f$ $I$ पर उत्तल है और $I$ में $x < y < z$। तब
\[
\frac{f(y) - f(x)}{y - x}
\;\leq\; \frac{f(z) - f(x)}{z - x}
\;\leq\; \frac{f(z) - f(y)}{z - y} :
\]
अर्थात् जीवाओं की ढालें दोनों अंत्यबिंदुओं में बढ़ती हैं।\index{उत्तल
फलन!ढाल असमिका}
\end{lemma}

\begin{proof}
$y = \frac{z - y}{z - x}\,x + \frac{y - x}{z - x}\,z$ लिखिए: यह उत्तल संचय है, क्योंकि दोनों गुणांक धनात्मक हैं और उनका
योग $1$ है। उत्तलता देती है
\[
f(y) \;\leq\; \frac{z-y}{z-x}\,f(x) + \frac{y-x}{z-x}\,f(z).
\]
बाईं असमिका के लिए दोनों पक्षों में से $f(x)$ घटाइए और $\frac{z-y}{z-x} - 1 = -\frac{y-x}{z-x}$ का उपयोग कीजिए:
\[
f(y) - f(x) \leq \frac{y - x}{z - x}\bigl(f(z) - f(x)\bigr),
\]
और $y - x > 0$ से भाग दीजिए। दाईं असमिका के लिए इसके बदले $f(z)$ में से घटाइए:
\[
f(z) - f(y) \geq f(z) - \frac{z-y}{z-x}f(x) -
\frac{y-x}{z-x}f(z)
= \frac{z - y}{z - x}\bigl(f(z) - f(x)\bigr),
\]
और $z - y > 0$ से भाग दीजिए। दोनों प्रदर्शित पग वही बैरिकेंद्रीय सर्वसमिका हैं,
बस अलग-अलग अंत्यबिंदु के सापेक्ष पढ़ी गई।
\end{proof}

\begin{theorem}[उत्तल फलनों की नियमितता]\label{thm:b2:realfun:convexreg}
मान लीजिए $f$ किसी अंतराल $I$ पर उत्तल है।
\begin{enumerate}
  \item प्रत्येक अभ्यंतरीय बिंदु पर $f$ के परिमित एकपक्षीय अवकलज $f'_g \leq f'_d$
        होते हैं; दोनों बिंदु के वर्धमान फलन हैं; और विशेष रूप से $f$
        $I$ के अभ्यंतर पर \omterm{def:b2:metric:continuity}{संतत} है (पर अंत्यबिंदुओं पर शायद नहीं)।
  \item $f$ हर \emph{अवलंब रेखा} के ऊपर रहता है: अभ्यंतरीय $a$ और किसी
        भी $m \in \intcc{f'_g(a)}{f'_d(a)}$ के लिए
        \[
        f(x) \geq f(a) + m(x - a) \qquad (x \in I).
        \]
  \item (जेनसन, भारित) $x_i \in I$ तथा भार $\lambda_i
        \geq 0$, $\sum\lambda_i = 1$ के लिए:
        \[
        f\Bigl(\sum_i \lambda_i x_i\Bigr) \leq \sum_i \lambda_i
        f(x_i) .\index{जेनसन असमिका}
        \]
\end{enumerate}
\end{theorem}

\begin{proof}
(1) कोई अभ्यंतरीय $a$ स्थिर कीजिए। \cref{lem:b2:realfun:slopes} से ढाल $\tau(h) = \frac{f(a + h) - f(a)}{h}$ $h$ का वर्धमान फलन है
(दोनों ओर, और $h_- < 0 <
h_+$ के लिए $\tau(h_-) \leq \tau(h_+)$)। अतः $h \to 0^-$ होने पर $\tau$ की परिमित सीमा है
(वर्धमान, और ऊपर से किसी भी दाईं ढाल से परिबद्ध) --- यही $f'_g(a)$ है --- और
$h \to 0^+$ होने पर भी ($f'_d(a)$), जहाँ $f'_g(a) \leq f'_d(a)$। परिमित एकपक्षीय अवकलज $a$ पर संततता
को बाध्य कर देते हैं। बिंदु में एकदिष्टता: अभ्यंतरीय $a < b$ के लिए $f'_d(a) \leq \frac{f(b) - f(a)}{b -
a} \leq f'_g(b)$,
फिर ढाल असमिका से।

(2) $x > a$ के लिए: $\frac{f(x) - f(a)}{x - a} \geq f'_d(a) \geq m$; और $x < a$ के लिए: $\frac{f(a) - f(x)}{a - x} \leq f'_g(a) \leq m$। दोनों पुनर्व्यवस्थित होकर दावा
दे देते हैं।

(3) बिंदुओं की संख्या पर आगमन, ठीक प्रथम वर्ष के खंड की तरह (दो-बिंदु
वाली स्थिति परिभाषा ही है) --- अथवा एक ही झटके में: (2) को $a = \sum\lambda_i x_i$ पर लगाइए
और बिंदुओं $x_i$ पर अवलंब-रेखा असमिकाओं का भार $\lambda_i$ के साथ औसत लीजिए: $\sum_i
\lambda_i f(x_i) \geq f(a) + m\sum_i\lambda_i(x_i - a) = f(a)$।
\end{proof}

\begin{example}[अंत्यबिंदु पर असंततता]
$\intcc{0}{1}$ पर फलन $f(0) = 1$, और $x > 0$ के लिए $f(x) = 0$, उत्तल है परंतु अंत्यबिंदु $0$ पर
असंतत: अर्थात् कथन (1) तीखा है।
\end{example}

\begin{example}[कोने और अवलंब रेखाओं का पूला]\label{ex:b2:realfun:corner}
$a = 0$ पर $f(x) = \abs x$ के लिए: एकपक्षीय अवकलज $f'_g(0) = -1$ और $f'_d(0) = +1$ हैं, और \cref{thm:b2:realfun:convexreg}~(2)
\emph{प्रत्येक} ढाल $m \in \intcc{-1}{1}$ के लिए एक अवलंब रेखा दे देती है:
\[
\abs x \geq m\,x \qquad (x \in \R,\ -1 \leq m \leq 1),
\]
जिनमें से हर एक ठीक किसी अर्धरेखा पर अथवा $0$ पर समता है। कोई उत्तल फलन
$a$ पर ठीक तभी अवकलनीय होता है जब यह पूला सिमटकर एक ही रेखा बन जाए
($f'_g(a) = f'_d(a)$); और कोनों पर स्पर्श रेखाओं का पूरा अंतराल होता है। यह पूला उत्तल
अनुकूलन के \emph{उपअवकलज} का परिमित-विमीय बीज है --- और यही कारण है कि
उत्तल फलन इतने मज़बूत हैं: जहाँ अवकलज विफल हो जाता है वहाँ भी अवलंब
ज्यामिति बची रहती है, और जेनसन की उपपत्ति को इतना ही चाहिए था।
\end{example}

\begin{example}[घात माध्य असमिका]\label{ex:b2:realfun:powermeans}
$0 < p < q$ तथा $1$ तक जुड़ने वाले भार $\lambda_i$ के साथ धनात्मक $x_i$ के लिए, बिंदुओं
$x_i^p$ पर उत्तल $t \mapsto t^{q/p}$ पर जेनसन लगाने से:
\[
\Bigl(\sum \lambda_i x_i^{p}\Bigr)^{1/p}
\leq \Bigl(\sum \lambda_i x_i^{q}\Bigr)^{1/q} :
\]
अर्थात् घात माध्य घातांक के साथ बढ़ते हैं --- जिसमें समांतर--वर्गमाध्य
असमिका समाहित है, और सीमा $p \to 0$ (\cref{exo:b2:realfun:6}) में एक बार फिर समांतर--गुणोत्तर माध्य
असमिका भी।
\end{example}

\begin{example}[महत्तम एन्ट्रॉपी]\label{ex:b2:realfun:entropy}
किसी प्रायिकता सदिश $(p_1, \dots, p_n)$ (धनात्मक, और $1$ तक जुड़ने वाला) के लिए
एन्ट्रॉपी $H(p) = -\sum_i p_i\ln p_i$ पूरा करती है
\[
H(p) \leq \ln n ,
\qquad\text{और समता तभी जब सभी }
i .
\text{ के लिए } p_i = \frac1n \]
उपपत्ति: भार $p_i$ के साथ बिंदुओं $\frac{1}{p_i}$ पर \emph{अवतल} $\ln$ पर जेनसन
(\cref{thm:b2:realfun:convexreg}~(3)) लगाइए:
\[
H(p) = \sum_i p_i\ln\frac{1}{p_i}
\leq \ln\Bigl(\sum_i p_i\,\frac1{p_i}\Bigr) = \ln n ,
\]
और समता सारे बिंदुओं $\frac1{p_i}$ को बराबर होने पर बाध्य कर देती है (यथार्थ
अवतलता), अर्थात् $p$ एकसमान। समतुल्य रूप से यह एकसमान $q$ के साथ \cref{exo:b2:realfun:7}
है। अनिश्चितता तब महत्तम होती है जब अज्ञान एकसमान रूप से फैला हो --- और
यही वह चरमीय सिद्धांत है जो कूटलेखन, सांख्यिकीय यांत्रिकी तथा \cref{ch:b2:randomvar} की
एन्ट्रॉपी-उपस्थितियों के पीछे है।
\end{example}

\begin{method}[किसी असमिका के पीछे का उत्तल फलन खोजना]\label{met:b2:realfun:spotting}
अधिकांश क्लासिक असमिकाएँ भेस बदले हुए जेनसन हैं; और किसी एक का भेस
उतारने के लिए: (1) इस प्रकार सामान्यीकृत कीजिए कि कोई \emph{भारित औसत}
प्रकट हो जाए (भार धनात्मक और $1$ तक जुड़ने वाले --- आवश्यकता हो तो कुल
द्रव्यमान से भाग दीजिए); (2) देखिए कि औसत के भीतर और बाहर कौन-सा फलन
लगाया गया है: दावा ``$f(\text{औसत}) \leq$ $f$ का औसत'' उत्तल $f$ का नाम
बता देता है; (3) उत्तलता को दूसरे अवकलज से प्रमाणित कीजिए और समता को
यथार्थता के द्वारा सँभालिए; (4) यदि कोई औसत दिखाई ही न दे, तो पहले
लघुगणक लीजिए --- गुणनफल और घातें औसत बन जाते हैं, और $\ln$ की अवतलता
समांतर--गुणोत्तर माध्य, यंग तथा उनके सगे-संबंधियों को उठा लेती है (इस
अध्याय की सप्ताहांत समस्या उन सब पर चरण 1--4 चलाती है)। और यदि लघुगणक भी
कोई औसत न दिखाएँ, तो असमिका को ढालों की एकदिष्टता के रूप में पढ़िए (\cref{lem:b2:realfun:slopes})
--- \cref{exo:b2:realfun:9} जैसे अधियोज्यता-कथन वहीं रहते हैं।
\end{method}

\begin{remark}[सामान्य भूलें]\label{rem:b2:realfun:pitfalls}
(क) उत्तलता गुणनफल के अंतर्गत सुरक्षित नहीं रहती: $x$ और $(x - 1)^2$ $\intcc{0}{2}$ पर
उत्तल हैं, परंतु उनके गुणनफल $x(x-1)^2$ का दूसरा अवकलज $6x - 4$ है, जो $\intco{0}{\frac23}$ पर
ऋणात्मक है --- अतः उत्तल नहीं; और न ही उत्तलता एकदिष्टता के बिना संयोजन
के अंतर्गत सुरक्षित रहती है (\cref{exo:b2:realfun:10})। (ख) अवतल फलनों के लिए जेनसन पलट जाती
है: आधी क्लासिक असमिकाएँ अवतल $\ln$ वाला रूप हैं; और उत्तल रूप को $\ln$ पर
लगाना समांतर--गुणोत्तर माध्य असमिका को \emph{उलटा} सिद्ध करने का सबसे
तेज़ रास्ता है। (ग) अकेली मध्यबिंदु उत्तलता से उत्तलता नहीं निकलती ---
संततता (अथवा केवल परिबद्धता) चाहिए (\cref{exo:b2:realfun:8}); और विकृत प्रतिउदाहरण इस
पुस्तक के अभिगृहीतों से परे रहते हैं। (घ) किसी \emph{\omterm{def:b2:metric:topology}{विवृत}} अंतराल पर
उत्तल फलन \omterm{def:b2:metric:continuity}{संतत} होता है, बल्कि स्थानीय रूप से \omterm{def:b2:metric:continuity}{लिप्शिट्स} भी (\cref{exo:b2:realfun:12});
अंत्यबिंदुओं पर कुछ भी मुफ़्त नहीं। (ङ) अवकलज डार्बू का पालन करते हैं पर
उनका \omterm{def:b2:metric:continuity}{संतत} होना आवश्यक नहीं (\cref{ex:b2:realfun:oscillation}): ``$f'$ में कोई उछाल नहीं'' का अर्थ
कभी ``$f'$ \omterm{def:b2:metric:continuity}{संतत} है'' नहीं होता।
\end{remark}

\section{डार्बू गुणधर्म}

\begin{theorem}[डार्बू]\label{thm:b2:realfun:darboux}
मान लीजिए $f$ किसी अंतराल $I$ पर अवकलनीय है। तब $f'$ अपने किन्हीं दो
मानों के बीच का हर मान लेता है --- यद्यपि $f'$ का \omterm{def:b2:metric:continuity}{संतत} होना आवश्यक नहीं
है।\index{डार्बू की प्रमेय}
\end{theorem}

\begin{proof}
मान लीजिए $I$ में $a < b$ है और $v$ $f'(a)$ तथा $f'(b)$ के यथार्थतः बीच में है, मान
लीजिए $f'(a) < v < f'(b)$। फलन $g(x) = f(x) - vx$ अवकलनीय है और $g'(a) < 0 < g'(b)$: $\intcc{a}{b}$ पर उसका न्यूनतम (जो प्राप्त
होता है: \omterm{def:b2:metric:compact}{संहत} पर संततता) न $a$ पर है ($a$ के ठीक बाद $g$ $g(a)$ से नीचे
उतरता है) और न $b$ पर ($b$ के ठीक पहले $g$ $g(b)$ से नीचे है): अतः वह
अभ्यंतरीय है, और वहाँ $g'(c)
= 0$, अर्थात् $f'(c) = v$। (यह प्रथम वर्ष का तारांकित
अभ्यास था; सिद्धांत में उसका स्थान यहाँ है।)
\end{proof}

\begin{example}[कौन-से फलन अवकलज होते हैं?]\label{ex:b2:realfun:whichderivatives}
डार्बू की प्रमेय अनस्तित्व की मशीन है। महत्तम पूर्णांक फलन $\lfloor x\rfloor$ $\R$ पर
किसी भी फलन का अवकलज नहीं है: वह मान $0$ और $1$ लेता है पर $\intcc{0}{1}$ पर $\frac12$
छोड़ देता है, और अवकलजों के लिए \cref{thm:b2:realfun:darboux} इसे वर्जित करती है। यही निर्णय हर
उछाल वाले फलन पर पड़ता है --- चिह्न फलन, हेविसाइड, सारे सोपान फलन ---
चाहे वे कितने ही निर्दोष क्यों न दिखें; उनके ``प्रतिअवकलज'' (चिह्न के
लिए $\abs x$, इत्यादि) केवल उछाल से दूर विद्यमान होते हैं और वहाँ कोने के साथ
गाँठ बना लेते हैं। इसके विपरीत \cref{ex:b2:realfun:oscillation} का उग्र रूप से असंतत $f'$ अवकलज
\emph{है} --- उसकी असंततता दोलन है, जिसे डार्बू सह लेती है। दोनों
व्यवहारों के बीच की सीमा ठीक नीचे वाला ``कोई उछाल नहीं'' उपप्रमेय है।
\end{example}

\begin{corollary}\label{cor:b2:realfun:nojumps}
किसी अवकलज में उछाल-असंततताएँ नहीं होतीं: यदि $f'(a^-)$ और $f'(a^+)$ विद्यमान हों,
तो वे $f'(a)$ के बराबर होते हैं। किसी अवकलज की असंततताएँ सदा दोलन-प्रकार की
होती हैं ($0$ पर $x^2\sin\frac1x$ का अवकलज, प्रथम वर्ष का खंड)।
\end{corollary}

\begin{proof}
यदि $f'(a^+) = \lim_{x\to a^+} f'(x)$ विद्यमान हो और $f'(a)$ से भिन्न हो, तो उनके यथार्थतः बीच के मान $f'$
किसी दाएँ प्रतिवेश पर छोड़ देता --- जो अंतरालों $\intcc{a}{a + h}$ पर डार्बू के विरुद्ध
है। (वैकल्पिक रूप से: मध्यमान प्रमेय $f'(a) = \lim_{h\to0^+} \frac{f(a+h)-f(a)}{h} = f'(a^+)$ को बाध्य कर देती है, क्योंकि
अंतर-भागफल किसी मध्यवर्ती बिंदु पर $f'$ का मान है।) बाईं ओर भी वैसा ही।
\end{proof}

\begin{example}[विहित दोलनशील अवकलज]\label{ex:b2:realfun:oscillation}
$x \neq 0$ के लिए $f(x) = x^2\sin\frac1x$ और $f(0) = 0$ लीजिए। $0$ पर: $\bigl|\frac{f(h) - f(0)}{h}\bigr| = \abs{h\sin\frac1h}
\leq \abs h \to 0$, अतः $f'(0) = 0$ विद्यमान है। और $0$
से दूर,
\[
f'(x) = 2x\sin\frac1x - \cos\frac1x ,
\]
जिसका पहला पद $0$ की ओर जाता है जबकि $\cos\frac1x$ हर अंतराल $\intoo{0}{\delta}$ पर $\intcc{-1}{1}$ में
दोलन करता है: अतः सीमा $f'(0^+)$ विद्यमान नहीं है। इस प्रकार $f'$ सर्वत्र
परिभाषित है पर $0$ पर असंतत --- और ठीक जैसा \cref{cor:b2:realfun:nojumps} बताती है, यह असंततता
दोलन है, उछाल नहीं: हर $\intoo{0}{\delta}$ पर $f'$ $0$ के चारों ओर पूरा एक अंतराल छान
मारता है। अवकलज उग्र हो सकते हैं, पर केवल डार्बू के अनुकूल ढंग से।
\end{example}

\begin{remark}[यह अध्याय कहाँ काम आता है]
उत्तलता असमिका-उद्योग का इंजन है: इस अध्याय की सप्ताहांत समस्या उसी से
यंग, हॉल्डर, मिंकोव्स्की और घात-माध्य शृंखला बना देती है, जिन्हें \cref{ch:b2:nvs} का
\omterm{def:b2:nvs:norm}{मानदंड} सिद्धांत और \cref{ch:b2:integration} के समाकल आकलन उपभोग करते हैं; और जेनसन प्रायिकता
में \cref{ch:b2:randomvar} की आघूर्ण-असमिकाओं के रूप में लौटती है। एकदिष्ट नियमितता \cref{ch:b2:integration} में
लौटती है (एकदिष्ट फलन समाकलनीय होते हैं) और तृतीय वर्ष के खंड में
एकदिष्ट फलनों की लगभग-सर्वत्र अवकलनीयता के रूप में --- जहाँ ``\omterm{def:b2:structures:countable}{गणनीय}
उछाल'' लेबेग के सिद्धांत का पहला चरण बन जाता है।
\end{remark}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:4}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:5}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:6}
(सीमा के रूप में गुणोत्तर माध्य) धनात्मक $x_i$ तथा $1$ तक जुड़ने वाले भार
$\lambda_i$ के लिए $x_i^p = \eu^{p\ln x_i} = 1 + p\ln x_i + O(p^2)$ के द्वारा सिद्ध कीजिए
\[
\lim_{p \to 0^+} \Bigl(\sum_i \lambda_i x_i^p\Bigr)^{1/p}
= \prod_i x_i^{\lambda_i} ,
\]
और \cref{ex:b2:realfun:powermeans} से भारित समांतर--गुणोत्तर माध्य असमिका निष्कर्ष रूप में निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:7}
(एन्ट्रॉपी असमिका) $t \mapsto t\ln t$ की यथार्थ उत्तलता का उपयोग करके सिद्ध कीजिए कि
$\sum p_i = \sum q_i = 1$ वाले धनात्मक $p_i, q_i$ के लिए:
\[
\sum_i p_i \ln\frac{p_i}{q_i} \geq 0 ,
\]
और समता तभी जब $p = q$। \emph{($\varphi(t) = t\ln t$ के साथ बाएँ पक्ष को $\sum q_i\,
\varphi\bigl(\frac{p_i}{q_i}\bigr)$ के रूप में लिखिए
और भार $q_i$ के साथ जेनसन लगाइए।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:realfun:8}
(मध्यबिंदु उत्तलता) $f \colon I \to \R$ \emph{मध्यबिंदु उत्तल} कहलाता है जब सदा $f\bigl(\frac{x+y}{2}\bigr) \leq \frac{f(x) + f(y)}{2}$ हो।
सिद्ध कीजिए कि \emph{\omterm{def:b2:metric:continuity}{संतत}} मध्यबिंदु उत्तल फलन उत्तल होता है। \emph{($m$
पर आगमन से द्विआधारी भार $\frac{k}{2^m}$ के लिए उत्तलता असमिका स्थापित कीजिए, फिर
सघनता और संततता के द्वारा सीमा लीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:realfun:9}
मान लीजिए $f$ $\intco{0}{+\infty}$ पर उत्तल है और $f(0) \leq 0$। सिद्ध कीजिए कि $x \mapsto \frac{f(x)}{x}$ $\intoo{0}{+\infty}$ पर वर्धमान
है, और इससे निष्कर्ष निकालिए कि $f(0) =
0$ वाले उत्तल $f$ के लिए $x, y \geq 0$ होने पर
$f(x + y) \geq f(x) + f(y)$ (अधियोज्यता)।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:realfun:10}
मान लीजिए $f$ $I$ पर उत्तल है और $g$ $f(I)$ को समाहित करने वाले किसी अंतराल
पर उत्तल तथा \emph{वर्धमान}। सिद्ध कीजिए कि $g \circ f$ उत्तल है, और प्रतिउदाहरण
से दिखाइए कि $g$ की एकदिष्टता हटाई नहीं जा सकती।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:11}
(एर्मीत--आदामार) मान लीजिए $f$ $\intcc{a}{b}$ पर उत्तल और \omterm{def:b2:metric:continuity}{संतत} है। सिद्ध कीजिए
\[
f\Bigl(\frac{a+b}{2}\Bigr) \;\leq\; \frac{1}{b -
a}\int_a^b f(t)\,\dd t \;\leq\; \frac{f(a) + f(b)}{2} .
\]
\emph{(बाएँ: मध्यबिंदु पर अवलंब रेखा का समाकलन कीजिए। दाएँ: $f$ को जीवा
से परिबद्ध कीजिए।)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:realfun:12}
सिद्ध कीजिए कि किसी \emph{\omterm{def:b2:metric:topology}{विवृत}} अंतराल $I$ पर उत्तल फलन स्थानीय रूप से
\omterm{def:b2:metric:continuity}{लिप्शिट्स} होता है: प्रत्येक खंड $\intcc{a}{b} \subseteq I$ तथा $\intcc{a - \delta}{b + \delta}
\subseteq I$ वाले हाशिये $\delta > 0$ के लिए $\intcc{a}{b}$
पर $f$ का प्रतिबंधन \omterm{def:b2:metric:continuity}{लिप्शिट्स} है, और उसका अचर
$\max\Bigl(\bigl|\frac{f(a) - f(a - \delta)}{\delta}\bigr|,
\bigl|\frac{f(b + \delta) - f(b)}{\delta}\bigr|\Bigr)$
\emph{(ढाल असमिका से हर जीवा-ढाल को इन दोनों के बीच फँसाइए)}।
\end{exercise}

\section{समस्या: उत्तलता का औज़ार-बक्सा}

एक ही परिभाषा --- आलेख के ऊपर जीवा --- क्लासिक असमिकाओं का पूरा
औज़ार-बक्सा उत्पन्न कर देती है। यह सप्ताहांत समस्या उसे तार्किक क्रम में
बनाती है: पहले उत्तलता की कसौटियाँ और यथार्थ जेनसन, फिर यंग, हॉल्डर तथा
मिंकोव्स्की ($p$-मानदंडों के जन्म-प्रमाणपत्र), न्यूनतम से महत्तम तक पूरी
घात-माध्य शृंखला, और दो मुकुट-लाभ --- कार्लेमान की असमिका, तथा द्वैतता के
रूप में पढ़ी गई हॉल्डर। सब कुछ सिद्ध किया गया है; कुछ भी आयातित नहीं।

\begin{problem}[{सप्ताहांत समस्या --- यंग, हॉल्डर, मिंकोव्स्की
और घात-माध्य शृंखला}]
\label{pb:b2:realfun:1}
आगे सर्वत्र $p, q > 1$ \emph{संयुग्मी घातांक} हैं: $\frac1p +
\frac1q = 1$; सदिश $a = (a_1, \dots, a_n) \in \R^n$ हैं; और भार
$\lambda_i > 0$ $\sum_i\lambda_i = 1$ पूरा करते हैं।

\medskip
\noindent\textbf{भाग I --- कसौटियाँ और यथार्थ जेनसन।}
\begin{enumerate}
  \item मान लीजिए $f$ किसी अंतराल $I$ पर अवकलनीय है। सिद्ध कीजिए कि $f$
        उत्तल है यदि और केवल यदि $f'$ वर्धमान हो \emph{(एक दिशा ढाल
        असमिका \cref{lem:b2:realfun:slopes} में सीमा लेकर; दूसरी मध्यमान प्रमेय से)}। इससे $C^2$
        कसौटी $f'' \geq 0$ निष्कर्ष रूप में निकालिए।
  \item मान लीजिए $I$ पर $f''> 0$। सिद्ध कीजिए कि $f$ \emph{यथार्थतः} उत्तल
        है ($x \neq y$ तथा $\lambda \in \intoo01$ के लिए यथार्थ असमिका), और यह कि यथार्थतः उत्तल
        फलन जेनसन असमिका (\cref{thm:b2:realfun:convexreg}~(3)) में समता \emph{केवल} तब देता है जब
        सारे $x_i$ एक ही हों।
  \item औज़ार-बक्से का कच्चा माल प्रमाणित कीजिए: $-\ln$ $\intoo{0}{+\infty}$ पर यथार्थतः
        उत्तल है; $t \mapsto t^r$ वहाँ $r > 1$ के लिए यथार्थतः उत्तल और $0 < r < 1$ के लिए
        यथार्थतः अवतल है; और $\exp$ $\R$ पर यथार्थतः उत्तल है।
  \item (यंग की असमिका) $a, b \geq 0$ के लिए सिद्ध कीजिए
        \[
        ab \;\leq\; \frac{a^p}{p} + \frac{b^q}{q},
        \]
        और समता तभी जब $a^p = b^q$ \emph{($\ln$ की अवतलता को भार $\frac1p, \frac1q$ के साथ
        दोनों बिंदुओं $a^p, b^q$ पर लगाइए)}।
  \item $\ln$ की अवतलता से एक ही पंक्ति में भारित समांतर--गुणोत्तर माध्य
        असमिका पुनः निकालिए:
        \[
        \prod_i x_i^{\lambda_i} \leq \sum_i\lambda_ix_i \qquad
        (x_i > 0),
        \]
        समता की स्थिति सहित; और \cref{exo:b2:realfun:6} के सीमा-मार्ग से तुलना कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- हॉल्डर और मिंकोव्स्की।} $\norm{a}_p = \bigl(\sum_i \abs{a_i}^p\bigr)^{1/p}$ और $\norm{a}_\infty = \max_i\abs{a_i}$ लिखिए।
\begin{enumerate}[resume]
  \item (हॉल्डर) सिद्ध कीजिए
        \[
        \sum_{i=1}^{n}\abs{a_ib_i} \;\leq\;
        \norm a_p\,\norm b_q ,
        \]
        और समता तभी जब सदिश $(\abs{a_i}^p)$ तथा $(\abs{b_i}^q)$ अनुपाती हों \emph{($\norm a_p = \norm b_q = 1$ को
        सामान्यीकृत कीजिए और पद-दर-पद यंग लगाइए)}।
  \item विशेष स्थितियाँ पहचानिए: $p = q = 2$ (कोशी--श्वार्ज़), और अंत्य युग्म
        $(p, q) = (1,
        \infty)$: $\sum\abs{a_ib_i} \leq
        \norm a_1\norm b_\infty$ कहिए और सिद्ध कीजिए।
  \item (मिंकोव्स्की) $p \geq 1$ के लिए सिद्ध कीजिए
        \[
        \norm{a + b}_p \leq \norm a_p + \norm b_p
        \]
        \emph{($\abs{a_i + b_i}^p \leq \abs{a_i +
        b_i}^{p-1}(\abs{a_i} + \abs{b_i})$ लिखिए और हर गुणनफल पर हॉल्डर लगाइए)}। निष्कर्ष:
        प्रत्येक $p \in \intco{1}{+\infty}$ के लिए $\norm\cdot_p$ $\R^n$ पर \omterm{def:b2:nvs:norm}{मानदंड} है, जिससे \cref{ch:b2:nvs} का चित्र
        पूरा हो जाता है।
  \item समाकल रूप: $\intcc{a}{b}$ पर \omterm{def:b2:metric:continuity}{संतत} $f, g$ के लिए $\norm f_p = \bigl(\int_a^b\abs
        f^p\bigr)^{1/p}$ के लिए हॉल्डर और
        मिंकोव्स्की कहिए और सिद्ध कीजिए \emph{(वही उपपत्तियाँ, बस समता
        की चर्चा के लिए समाकल की यथार्थ धनात्मकता के साथ)}।
  \item $1
        \leq p \leq q$ के लिए एकदिष्टता $\norm a_q \leq \norm a_p$ सिद्ध कीजिए, $p \to \infty$ होने पर सीमा $\norm a_p \to
        \norm a_\infty$,
        और तीखे अचर के साथ उलटी तुलना भी:
        \[
        \norm a_p \leq n^{\frac1p - \frac1q}\,\norm a_q
        \]
        \emph{(अचर सदिश के विरुद्ध हॉल्डर)}। हर समता को साकार करने वाले
        सदिश पहचानिए।
  \item (अंतर्वेशन) $\frac1r = \frac\theta p +
        \frac{1-\theta}q$ वाले $1 \leq p < r < q$ और $\theta \in
        \intoo01$ के लिए सिद्ध कीजिए
        \[
        \norm a_r \leq \norm a_p^{\theta}\,
        \norm a_q^{1-\theta}
        \]
        \emph{($\abs{a_i}^{\theta
        r}\abs{a_i}^{(1-\theta)r}$ पर घातांक $\frac{p}{\theta
        r}$ और $\frac{q}{(1-\theta)r}$ के साथ हॉल्डर लगाइए)}।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- घात-माध्य शृंखला, पूरी।} $p \neq 0$ के लिए $M_p = \bigl(\sum_i\lambda_i
x_i^p\bigr)^{1/p}$
($x_i > 0$) रखिए, और $M_0 = \prod_i
x_i^{\lambda_i}$।
\begin{enumerate}[resume]
  \item सिद्ध कीजिए कि $p \mapsto M_p$ पूरे $\R^*$ पर वर्धमान है: $p < q < 0$ को व्युत्क्रम
        सर्वसमिका $M_{-p}(x) = M_p(1/x)^{-1}$ से निपटाइए, और $p < 0 < q$ के लिए $M_p \leq M_0 \leq M_q$ दिखाकर $0$ के पार
        पुल बनाइए \emph{($\ln$ की अवतलता $x_i^q$ पर लगाइए, और ऋणात्मक घातांकों
        के लिए उलटी असमिका)}।
  \item सीमाएँ सिद्ध कीजिए: $p \to
        +\infty$ होने पर $M_p \to \max_i x_i$, और $p \to -\infty$ होने पर $M_p \to \min_i x_i$।
  \item समान भारों के लिए शृंखला $\min \leq \mathrm{HM} \leq
        \mathrm{GM} \leq \mathrm{AM} \leq \mathrm{QM} \leq
        \max$ लिखकर क्लासिक परिणाम सिद्ध कीजिए:
        धनात्मक $a_1, \dots, a_n$ के लिए
        \[
        \Bigl(\sum_i a_i\Bigr)\Bigl(\sum_i\frac1{a_i}\Bigr)
        \geq n^2 .
        \]
  \item माध्यों को मानदंडों से जोड़िए: समान भारों $\lambda_i =
        \frac1n$ के लिए $M_p(x) = n^{-1/p}\norm x_p$।
        दोनों एकदिष्टताओं में मेल बिठाइए --- माध्य $p$ के साथ
        \emph{बढ़ते} हैं जबकि \omterm{def:b2:nvs:norm}{मानदंड} \emph{घटते} हैं (प्रश्न 10) ---
        गुणक $n^{-1/p}$ के बारे में एक ही वाक्य में।
  \item प्रश्न 14 की पूरी शृंखला पर समता की स्थितियाँ निर्धारित कीजिए
        (धनात्मक भार): कहीं भी समता सारे $x_i$ को बराबर होने पर बाध्य कर
        देती है --- यथार्थ उत्तलता का प्रतिफल।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- लाभ।}
\begin{enumerate}[resume]
  \item (घुंडी वाला यंग) $a, b \geq 0$ और $\varepsilon >
        0$ के लिए सिद्ध कीजिए
        \[
        ab \leq \varepsilon\,\frac{a^p}{p} +
        \varepsilon^{-q/p}\,\frac{b^q}{q},
        \]
        और श्रमिक स्थिति $ab \leq \varepsilon a^2 +
        \frac{b^2}{4\varepsilon}$: यही वह अवशोषण-युक्ति है जो पूरे विश्लेषण
        में काम आती है।
  \item (कार्लेमान की ओर) मान लीजिए $c_k = \frac{(k+1)^k}{k^{k-1}}$। क्रमिक निरसन सर्वसमिका $\prod_{k=1}^{n}c_k =
        (n+1)^n$
        सिद्ध कीजिए, और संख्याओं $c_ka_k$ पर समांतर--गुणोत्तर माध्य लगाकर
        निष्कर्ष निकालिए
        \[
        (a_1a_2\cdots a_n)^{1/n} \leq
        \frac{1}{n(n+1)}\sum_{k=1}^{n} c_k a_k
        \qquad (a_k > 0).
        \]
  \item (कार्लेमान की असमिका) $n$ पर योग लीजिए, योग का क्रम बदलिए
        (धनात्मक \omterm{def:b2:series:summable}{योग्य कुल}, \cref{thm:b2:series:fubini}), और $\sum_{n \geq
        k}\frac{1}{n(n+1)} = \frac1k$ तथा $c_k/k = \bigl(1 +
        \frac1k\bigr)^k < \eu$ का उपयोग करके
        निष्कर्ष निकालिए: धनात्मक पदों वाली प्रत्येक अभिसारी $\sum a_k$ के लिए
        \[
        \sum_{n=1}^{\infty}(a_1a_2\cdots a_n)^{1/n}
        \;\leq\; \eu\sum_{k=1}^{\infty}a_k .
        \]
  \item $\intcc{0}{1}$ पर \omterm{def:b2:metric:continuity}{संतत} और धनात्मक $f$ के लिए सिद्ध कीजिए
        \[
        \Bigl(\int_0^1 f\Bigr)\Bigl(\int_0^1\frac1f\Bigr)
        \geq 1,
        \]
        और समता तभी जब $f$ अचर हो \emph{($\sqrt f\cdot\frac1{\sqrt
        f}$ पर कोशी--श्वार्ज़)}।
  \item (गोलों की ज्यामिति) मिंकोव्स्की की समता-स्थिति का उपयोग करके
        दिखाइए कि $1 < p < \infty$ के लिए $\norm\cdot_p$ के इकाई गोलक में कोई खंड नहीं है
        (\omterm{def:b2:nvs:norm}{मानदंड} \cref{pb:b2:nvs:1} के अर्थ में यथार्थतः उत्तल है), जबकि $p = 1$ तथा $p =
        \infty$
        के लिए है: सपाट टुकड़े दिखाइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग V --- द्वैतता और संश्लेषण।}
\begin{enumerate}[resume]
  \item (द्वैतता के रूप में हॉल्डर) सिद्ध कीजिए कि प्रत्येक $a \in
        \R^n$ के लिए
        \[
        \norm a_p = \max_{\norm b_q \leq 1}\ \sum_i a_ib_i ,
        \]
        और महत्तम करने वाला $b$ स्पष्ट रूप से दिखाइए। ($p$-मानदंड
        $q$-मानदंड का \omterm{def:b2:linalg:dual}{द्वैत} है --- $L^p$ द्वैतता का परिमित-विमीय बीज।)
  \item (आघूर्ण) मान लीजिए $X$ कोई ऐसा यादृच्छिक चर है जो परिमित
        धनात्मक मान $x_i$ प्रायिकताओं $\lambda_i$ के साथ लेता है। प्रश्न 12 को इस
        रूप में फिर से कहिए: $r \mapsto
        \E[X^r]^{1/r}$ वर्धमान है --- अर्थात् आघूर्ण
        (ल्यापुनोव) असमिका, जो \cref{ch:b2:randomvar} में फिर काम आएगी।
  \item नामित औज़ारों से दो-दो पंक्तियों में हल कीजिए: (क) धनात्मक $a, b, c$
        के लिए $a^3 + b^3 + c^3 \geq
        \frac{(a+b+c)^3}{9}$; (ख) धनात्मक $x_1, \dots,
        x_n$ के लिए $\bigl(\sum_i\sqrt{x_i}\bigr)^2 \leq
        n\sum_i x_i$।
  \item (संश्लेषण) पाँच वाक्यों में वंशावली खींचिए: जीवा वाली परिभाषा से
        ढाल प्रमेयिका तक; ढालों से अवलंब रेखाओं तक और जेनसन तक; $\ln$ की
        अवतलता से यंग तक, हॉल्डर तक, मिंकोव्स्की तक और $p$-मानदंडों
        तक; जेनसन से घात-माध्य शृंखला तक और आघूर्णों तक; समांतर--गुणोत्तर
        माध्य से कार्लेमान तक। शिखर नाम लीजिए (हॉल्डर--मिंकोव्स्की;
        कार्लेमान), और बताइए कि औज़ार-बक्सा कहाँ जा रहा है: तृतीय वर्ष के
        खंड की $L^p$ समष्टियाँ, जिनके अभिगृहीत ठीक प्रश्न 6 और 8 हैं।
\end{enumerate}
\end{problem}
