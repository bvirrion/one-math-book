\chapter{Formas cuadráticas}\label{ch:b2:quadratic}

Una \omterm{def:b2:quadratic:def}{forma cuadrática} es la sombra algebraica de una geometría: las
partes de signatura nula aplanan, las positivas curvan en un sentido y
las negativas en el otro. Este capítulo reduce toda \omterm{def:b2:quadratic:def}{forma cuadrática}
real a una suma de cuadrados con signos $\pm$ (Gauss), demuestra que los
recuentos de signos son intrínsecos (Sylvester) y corona después la
geometría euclídea con el \emph{teorema espectral}: los
\omterm{def:b2:quadratic:adjoint}{endomorfismos simétricos} se diagonalizan en bases ortonormales, el
teorema más utilizado del álgebra lineal aplicada.

\section{Formas bilineales y cuadráticas}

\begin{definition}\label{def:b2:quadratic:def}
Una \emph{forma bilineal \omterm{def:b2:quadratic:adjoint}{simétrica}} sobre un espacio vectorial real
$E$ es una aplicación bilineal $\varphi \colon E \times E \to \R$
con $\varphi(x, y) = \varphi(y, x)$; la \emph{forma cuadrática}
asociada\index{forma cuadrática} es $q(x) = \varphi(x, x)$. La forma
$\varphi$ se recupera de $q$ por
\emph{polarización}\index{identidad de polarización}:
\[
\varphi(x, y) = \tfrac12\bigl(q(x + y) - q(x) - q(y)\bigr).
\]
En una base $(e_i)$, la \emph{matriz} de $\varphi$ es la matriz
\omterm{def:b2:quadratic:adjoint}{simétrica} $B = (\varphi(e_i, e_j))$, con $q(x) = X^{\mathsf T} B
X$; un cambio de base de matriz $P$ sustituye $B$ por $P^{\mathsf T}
B P$ (\emph{congruencia}\index{matrices congruentes}, ¡no
semejanza!). El \emph{rango} de $q$ es $\operatorname{rk} B$
(invariante: la congruencia multiplica por matrices invertibles).
\end{definition}

\begin{example}
Sobre $\R^2$: $q(x, y) = x^2 + 4xy + y^2$ tiene matriz
$\begin{pmatrix} 1 & 2\\ 2 & 1\end{pmatrix}$. Un producto escalar es
exactamente una \omterm{def:b2:quadratic:def}{forma bilineal simétrica} cuya \omterm{def:b2:quadratic:def}{forma cuadrática} es
definida positiva; este capítulo estudia el caso general, de signo
indefinido.
\end{example}

\begin{example}[La congruencia en acción]\label{ex:b2:quadratic:congruence}
Tomemos $q(x, y) = x^2 + 4xy + y^2$ (de matriz $B = \begin{pmatrix}
1 & 2\\ 2 & 1\end{pmatrix}$) y la nueva base $e_1' = (1, 1)$, $e_2'
= (1, -1)$, es decir, $P = \begin{pmatrix} 1 & 1\\ 1 &
-1\end{pmatrix}$. Entonces
\[
P^{\mathsf T}BP
= \begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
\begin{pmatrix} 1 & 2\\ 2 & 1\end{pmatrix}
\begin{pmatrix} 1 & 1\\ 1 & -1\end{pmatrix}
= \begin{pmatrix} 6 & 0\\ 0 & -2\end{pmatrix} :
\]
en las coordenadas $(u, v)$ respecto de la nueva base, $q = 6u^2 -
2v^2$; compruébese: $x = u + v$, $y = u - v$ da $x^2 + 4xy + y^2 =
6u^2 - 2v^2$ directamente. Obsérvese que las nuevas entradas
diagonales $6, -2$ \emph{no} son los \omterm{def:b2:reduction:eigen}{valores propios} $3, -1$ de
$B$: la \omterm{def:b2:quadratic:def}{congruencia} reescala, y solo la semejanza conserva los
\omterm{def:b2:reduction:eigen}{espectros}; pero los signos coinciden, como exige el teorema de
Sylvester. (La base de aquí es ortogonal pero no ortonormal;
normalizándola por $\frac{1}{\sqrt2}$ se dividiría la diagonal entre
$2$ y se recuperarían los \omterm{def:b2:reduction:eigen}{valores propios}.)
\end{example}

\begin{example}[Los determinantes de Gram miden áreas]\label{ex:b2:quadratic:gramarea}
Para $v_1, v_2$ en un espacio euclídeo, la matriz de Gram $G =
\bigl(\langle v_i, v_j\rangle\bigr)$ empaqueta longitudes y ángulo;
su \omterm{def:b2:linalg:det}{determinante} empaqueta el \emph{área}:
\[
\det G = \norm{v_1}^2\norm{v_2}^2 - \langle v_1, v_2\rangle^2
= \norm{v_1}^2\norm{v_2}^2\bigl(1 - \cos^2\theta\bigr)
= \bigl(\norm{v_1}\,\norm{v_2}\sin\theta\bigr)^2 ,
\]
el cuadrado del área del paralelogramo sobre $v_1, v_2$; y
Cauchy--Schwarz es exactamente el enunciado $\det G \geq 0$. Ejemplo
resuelto: $v_1 = (1, 2, 2)$, $v_2 = (2, 1, -2)$ en $\R^3$:
\[
G = \begin{pmatrix} 9 & 0\\ 0 & 9 \end{pmatrix},
\qquad
\det G = 81 :
\]
los vectores son ortogonales de longitud $3$ y generan un
paralelogramo (aquí, un cuadrado) de área $\sqrt{81} = 9$. Moraleja:
no se ha usado ningún producto vectorial ni magia de la dimensión
$3$; $\sqrt{\det G}$ mide el volumen $k$-dimensional en
\emph{cualquier} dimensión, punto de partida de la parte I del
problema de fin de semana y de las integrales de área de superficie
posteriores de este volumen.
\end{example}

\section{Reducción de Gauss e inercia de Sylvester}

\begin{theorem}[Reducción de Gauss]\label{thm:b2:quadratic:gauss}
Toda \omterm{def:b2:quadratic:def}{forma cuadrática} $q$ sobre un espacio real de dimensión finita
puede escribirse\index{reducción de Gauss}
\[
q = \sum_{i=1}^{s} \ell_i^2 - \sum_{j=1}^{t} m_j^2 ,
\]
donde $\ell_1, \dots, \ell_s, m_1, \dots, m_t$ son formas lineales
linealmente independientes; equivalentemente, alguna base hace
diagonal la matriz de $q$, con entradas $+1$ ($s$ veces), $-1$ ($t$
veces) y $0$.
\end{theorem}

\begin{proof}
Inducción sobre el número de variables, en coordenadas: $q(x_1,
\dots, x_n)$.

\emph{Caso 1: aparece algún cuadrado}, digamos que el coeficiente
$a$ de $x_1^2$ es no nulo. Agrúpense todos los términos en $x_1$ y
complétese el cuadrado:
\[
q = a\Bigl(x_1 + \frac{1}{a}\,\lambda(x_2, \dots,
x_n)\Bigr)^{\!2} + q_1(x_2, \dots, x_n),
\]
donde $\lambda$ es lineal y $q_1$ cuadrática en las variables
restantes: se ha separado una forma independiente (esta contiene
$x_1$ y las demás no), la inducción se aplica a $q_1$, y los signos
$\pm$ provienen del signo de $a$ tras reescalar por $\sqrt{\abs a}$.

\emph{Caso 2: no hay cuadrados, pero sí algún término cruzado},
digamos $b\,x_1x_2$ con $b \neq 0$. Úsese la identidad
\[
x_1x_2 = \tfrac14\bigl((x_1 + x_2)^2 - (x_1 - x_2)^2\bigr)
\]
tras agrupar: escribiendo $q = b\,x_1x_2 + x_1\alpha + x_2\beta +
q_2$ (con $\alpha, \beta, q_2$ en las demás variables), se comprueba
que
\[
q = \frac{b}{4}\Bigl[\Bigl(x_1 + x_2 + \frac{\alpha +
\beta}{b}\Bigr)^{2} - \Bigl(x_1 - x_2 + \frac{\beta -
\alpha}{b}\Bigr)^{2}\Bigr] + \widetilde q ,
\]
con $\widetilde q$ libre de $x_1, x_2$: se separan dos formas
independientes y la inducción remata.

Independencia de las formas recogidas: ordénense los lotes según se
producen. Las formas del primer lote contienen $x_1$ (caso 1) o
$x_1, x_2$ (caso 2); todas las posteriores están libres de esas
variables. Supongamos que se anula una combinación lineal de todas
las formas recogidas. Leyendo el coeficiente de $x_1$ (y de $x_2$):
solo aporta el primer lote, y dentro de él las una o dos formas son
visiblemente independientes ($\ell$ sola, o bien $\ell \pm m$ con
$\ell, m$ independientes): se anulan los coeficientes del primer
lote. Elimínese el lote y repítase: por inducción sobre los lotes,
todos los coeficientes se anulan; la familia entera es libre, con
triangularidad explícita.
\end{proof}

\begin{theorem}[Ley de inercia de Sylvester]\label{thm:b2:quadratic:sylvester}
El par $(s, t)$ del~\cref{thm:b2:quadratic:gauss} depende solo de
$q$ y no de la reducción: es la \emph{signatura}\index{signatura de
una forma cuadrática} de $q$. Además,
\[
s = \max\{\dim F : q|_F \text{ es definida positiva}\},
\]
y simétricamente para $t$.
\end{theorem}

\begin{proof}
Sea $q = \sum_{i \leq s}\ell_i^2 - \sum_{j\leq t} m_j^2$ y sea $F_+$
la envoltura de los vectores preduales sobre los que las
$(\ell_i)$ se restringen a coordenadas; en concreto: complétese la
familia independiente $(\ell_1, \dots, \ell_s, m_1, \dots, m_t)$
hasta una base del \omterm{def:b2:linalg:dual}{dual} $E^*$ y sea $(u_1, \dots, u_n)$ la base de
$E$ de la que esas son las formas coordenadas (la base predual:
$\ell_i(u_k) = \delta_{ik}$ para $k \leq s$, y las formas
posteriores se anulan sobre los vectores anteriores). Pongamos $F_+
= \operatorname{Vect}(u_1, \dots, u_s)$: para $x = \sum_{i\leq
s}x_iu_i \in F_+$,
\[
\ell_i(x) = x_i,
\qquad m_j(x) = 0,
\qquad\text{luego}\qquad
q(x) = \sum_{i\leq s}x_i^2 > 0 \quad (x \neq 0) :
\]
$q|_{F_+}$ es definida positiva y el máximo de la fórmula es $\geq
s$. Recíprocamente, sea $F$ un subespacio cualquiera con $q|_F$
definida positiva, y $G = \{x : \ell_1(x) = \dots = \ell_s(x) =
0\}$, de codimensión $\leq s$; sobre $G$, $q(x) = -\sum m_j^2 \leq
0$. Entonces $F \cap G = \{0\}$ (un vector no nulo allí tendría $q >
0$ y $q \leq 0$), de modo que $\dim F \leq \dim E - \dim G \leq s$.
Por tanto, el máximo vale $s$ para \emph{toda} reducción: $s$ es
intrínseco, y $t = \operatorname{rk} q - s$ también.
\end{proof}

\begin{example}\label{ex:b2:quadratic:gaussexample}
$q(x, y, z) = xy + yz + zx$ (sin cuadrados). Con $x_1 = x$, $x_2 =
y$: $q = xy + z(x + y)$, y la identidad de los dos cuadrados da
\[
q = \tfrac14(x + y + 2z)^2 - \tfrac14(x - y)^2 - z^2 ,
\]
(desarróllese para comprobar). Tres formas independientes:
signatura $(1, 2)$, rango $3$. Una dirección positiva y dos
negativas: la geometría de ``cono de luz'' de esta forma.
\end{example}

\begin{example}[Una forma degenerada, reducida por completo]\label{ex:b2:quadratic:degenerate}
$q(x, y, z) = xy + yz$ sobre $\R^3$: no hay cuadrados, así que caso
2 con la agrupación $q = y(x + z)$. La identidad de los dos
cuadrados sobre el producto de las formas independientes $y$ y $x +
z$:
\[
q = \frac14\bigl(y + x + z\bigr)^2 -
\frac14\bigl(y - x - z\bigr)^2 .
\]
Las dos formas lineales $y + x + z$ e $y - x - z$ son independientes
(su diferencia es $2(x+z)$ y su suma $2y$), de modo que Sylvester
permite leer: signatura $(1, 1)$, rango $2$; \emph{degenerada}. El
núcleo de la forma polar se halla resolviendo $\varphi(v, \cdot) =
0$: con matriz $\frac12\begin{pmatrix} 0&1&0\\ 1&0&1\\
0&1&0\end{pmatrix}$, el núcleo es $\{y = 0,\ x + z = 0\} = \R\,(1,
0, -1)$, la dirección a lo largo de la cual $q$ no ve nada.
Moraleja: la deficiencia de rango aparece en Gauss como ``quedarse
sin variables''; la reducción produjo solo dos cuadrados en tres
dimensiones, y la dimensión que falta es exactamente el núcleo.
\end{example}

\begin{example}[Una forma, dos caminos hacia la signatura]\label{ex:b2:quadratic:tworoads}
$q(x, y, z) = 2x^2 + 2y^2 + 2z^2 + 2xy + 2yz$, de matriz
$\begin{pmatrix} 2 & 1 & 0\\ 1 & 2 & 1\\ 0 & 1 & 2
\end{pmatrix}$. \emph{Camino 1, Gauss:} complétense cuadrados en
orden,
\[
q = 2\Bigl(x + \frac y2\Bigr)^{\!2} + \frac32 y^2 + 2yz + 2z^2
= 2\Bigl(x + \frac y2\Bigr)^{\!2}
+ \frac32\Bigl(y + \frac{2z}{3}\Bigr)^{\!2} + \frac43 z^2 :
\]
tres cuadrados positivos sobre formas independientes, signatura
$(3, 0)$: definida positiva. \emph{Camino 2, \omterm{def:b2:reduction:eigen}{valores propios}:} la
matriz es la tridiagonal $2I + N$ con $N$ la matriz de vecinos; sus
\omterm{def:b2:reduction:eigen}{valores propios} son $2 + \sqrt2$, $2$ y $2 - \sqrt2$ (compruébense
los \omterm{def:b2:reduction:eigen}{vectores propios} $(1, \pm\sqrt2, 1)$ y $(1, 0, -1)$), todos
positivos: mismo veredicto, por el~\cref{cor:b2:quadratic:principalaxes}. Gauss es más rápido; los
\omterm{def:b2:reduction:eigen}{valores propios} dicen más (dan los ejes principales y los valores
extremos de $q$ sobre la esfera). Moraleja: los pivotes positivos
$2, \frac32, \frac43$ de Gauss son exactamente los cocientes
$\frac{\Delta_k}{\Delta_{k-1}}$ de menores principales dominantes
($\Delta_1 = 2$, $\Delta_2 = 3$, $\Delta_3 = 4$); el problema de fin
de semana lo demuestra en general.
\end{example}

\begin{method}[Calcular una signatura: tres vías]\label{met:b2:quadratic:signature}
\begin{enumerate}
  \item \emph{Gauss} (siempre funciona; la más rápida a mano):
        complétense cuadrados en orden, con el caso 2 cuando no haya
        ningún cuadrado disponible, y cuéntense los signos.
        Compruébese que las formas lineales recogidas son
        independientes; menos formas que variables significa que hay
        núcleo (\cref{ex:b2:quadratic:degenerate}).
  \item \emph{Menores dominantes} (para tests de definición): todos
        los $\Delta_k > 0$ si y solo si es definida positiva
        (\cref{exo:b2:quadratic:8}); los pivotes
        $\Delta_k/\Delta_{k-1}$ dan incluso los coeficientes de
        Gauss (problema de fin de semana). Falla en silencio si
        algún $\Delta_k = 0$: recúrrase entonces a la vía 1.
  \item \emph{\omterm{def:b2:reduction:eigen}{Valores propios}} (la más informativa, la más cara):
        signos del \omterm{def:b2:reduction:eigen}{espectro}
        (\cref{cor:b2:quadratic:principalaxes}); da además los ejes
        principales y los valores extremos de $q$ sobre la esfera
        unidad. Es preferible cuando de todos modos se necesita la
        estructura propia.
\end{enumerate}
\end{method}

\section{El teorema espectral}

Sea ahora $E$ euclídeo (con producto escalar
$\langle\cdot,\cdot\rangle$, volumen del primer año).

\begin{definition}[Adjunto; endomorfismos simétricos]\label{def:b2:quadratic:adjoint}
Para $u \in \mathcal{L}(E)$, el \emph{adjunto}\index{adjunto} $u^*$
es el único endomorfismo con
\[
\langle u(x), y\rangle = \langle x, u^*(y)\rangle
\qquad (x, y \in E);
\]
en una base ortonormal, $\operatorname{Mat}(u^*) =
\operatorname{Mat}(u)^{\mathsf T}$. $u$ es
\emph{simétrico}\index{endomorfismo simétrico} (autoadjunto) cuando
$u^* = u$; equivalentemente, cuando su matriz en una base ortonormal
es simétrica.
\end{definition}

\begin{proof}[Existencia y unicidad del adjunto]
Para $y$ fijo, la forma $x \mapsto \langle u(x), y\rangle$ es
lineal, luego (dimensión finita) es de la forma $\langle x,
z_y\rangle$ para un único $z_y$, y la aplicación $y \mapsto z_y =:
u^*(y)$ es lineal por unicidad. La identificación matricial: léase
$\langle u(e_i), e_j\rangle$ de las dos maneras.
\end{proof}

\begin{example}[El adjunto depende del producto escalar]\label{ex:b2:quadratic:weightedadjoint}
Sobre $\R^2$, tomemos el producto escalar \emph{con pesos} $\langle
x, y\rangle_D = x_1y_1 + 2x_2y_2$ (de matriz $D =
\operatorname{diag}(1,2)$) y $u$ de matriz $A = \begin{pmatrix} 0 &
1\\ 0 & 0\end{pmatrix}$ en la base canónica. De $\langle u(x),
y\rangle_D = (Ax)^{\mathsf T}Dy = x^{\mathsf T}(A^{\mathsf T}D)y$ y
$\langle x, u^*(y)\rangle_D = x^{\mathsf T}(DA^*)y$, la matriz del
adjunto es
\[
A^* = D^{-1}A^{\mathsf T}D
= \begin{pmatrix} 1 & 0\\ 0 & \tfrac12\end{pmatrix}
\begin{pmatrix} 0 & 0\\ 1 & 0\end{pmatrix}
\begin{pmatrix} 1 & 0\\ 0 & 2\end{pmatrix}
= \begin{pmatrix} 0 & 0\\ \tfrac12 & 0\end{pmatrix}
\neq A^{\mathsf T} .
\]
Comprobación de sensatez en $x = (1,0)$, $y = (0,1)$:
\[
\langle u(x), y\rangle_D = \langle (0,0), y\rangle_D = 0,
\quad
\langle x, u^*(y)\rangle_D
= \langle(1,0), (0,\tfrac12)\rangle_D = 0 ;
\]
y en $x = (0,1)$, $y = (1,0)$:
\[
\langle u(x), y\rangle_D = \langle(1,0),(1,0)\rangle_D = 1,
\quad
\langle x, u^*(y)\rangle_D =
\langle(0,1),(0,\tfrac12)\rangle_D = 1 .
\] Moraleja: ``$\operatorname{Mat}(u^*) =
\operatorname{Mat}(u)^{\mathsf T}$'' es un enunciado sobre bases
\emph{ortonormales} únicamente; en general interviene la métrica
$D$, exactamente como en la reducción simultánea del problema de fin
de semana.
\end{example}

\begin{theorem}[Teorema espectral]\label{thm:b2:quadratic:spectral}
Sea $u$ un \omterm{def:b2:quadratic:adjoint}{endomorfismo simétrico} de un espacio euclídeo $E$.
Entonces $E$ tiene una \emph{base ortonormal de \omterm{def:b2:reduction:eigen}{vectores propios}}
de $u$; todos los \omterm{def:b2:reduction:eigen}{valores propios} son reales y los \omterm{def:b2:reduction:eigen}{subespacios
propios} de \omterm{def:b2:reduction:eigen}{valores propios} distintos son ortogonales. Forma
matricial: toda matriz real \omterm{def:b2:quadratic:adjoint}{simétrica} $A$ se escribe
\[
A = P\,D\,P^{\mathsf T},
\qquad P \text{ ortogonal } (P^{\mathsf T} P = I),\ D
\text{ diagonal}.\index{teorema espectral}
\]
\end{theorem}

\begin{proof}
\emph{Existe un \omterm{def:b2:reduction:eigen}{vector propio}.} La función $x \mapsto \langle u(x),
x\rangle$ es \omterm{def:b2:metric:continuity}{continua} sobre la esfera unidad $S$ de $E$, que es
\omterm{def:b2:metric:compact}{compacta} (dimensión finita, \cref{thm:b2:nvs:finitedim}): alcanza su
máximo $\lambda$ en cierto $a \in S$. Afirmamos que $u(a) = \lambda
a$. Para todo $y \perp a$ con $\norm y = 1$ y todo $t \in \R$, el
vector $x_t = \frac{a + ty}{\sqrt{1 + t^2}}$ está en $S$ ($\norm{a +
ty}^2 = 1 + t^2$ por Pitágoras); desarrollando la función
maximizada,
\[
g(t) = \langle u(x_t), x_t\rangle
= \frac{\langle u(a), a\rangle + 2t\langle u(a), y\rangle +
t^2\langle u(y), y\rangle}{1 + t^2}
\]
(la simetría de $u$ ha fundido los dos términos cruzados: $\langle
u(a), y\rangle = \langle a, u(y)\rangle = \langle u(y), a\rangle$).
$g$ es una función derivable de $t$ con un máximo en $t = 0$; la
regla del cociente en $0$ da
\[
g'(0) = \frac{2\langle u(a), y\rangle\cdot 1 - \langle u(a),
a\rangle\cdot 0}{1} = 2\langle u(a), y\rangle = 0 .
\]
Así pues, $u(a)$ es ortogonal a todo el hiperplano $a^\perp$: $u(a)
\in (a^{\perp})^{\perp} = \R a$, es decir, $u(a) = \mu a$; y $\mu =
\langle u(a), a\rangle = \lambda$.

\emph{Inducción.} El complemento ortogonal $F = a^\perp$ es estable
por $u$: para $x \perp a$, $\langle u(x), a\rangle = \langle x,
u(a)\rangle = \lambda\langle x, a\rangle = 0$. La restricción $u|_F$
es \omterm{def:b2:quadratic:adjoint}{simétrica} para el producto escalar inducido; por inducción
sobre la dimensión, $F$ tiene una base ortonormal de \omterm{def:b2:reduction:eigen}{vectores
propios}, y se antepone $a$.

\emph{Complementos.} Los \omterm{def:b2:reduction:eigen}{valores propios} son los números reales
$\langle u(e), e\rangle$ sobre la base propia. Ortogonalidad de los
\omterm{def:b2:reduction:eigen}{subespacios propios}: de $u(x) = \lambda x$ y $u(y) = \mu y$ resulta
$\lambda\langle x, y\rangle = \langle u(x), y\rangle = \langle x,
u(y)\rangle = \mu\langle x, y\rangle$, luego $\langle x, y\rangle =
0$ cuando $\lambda \neq \mu$. Forma matricial: las columnas de $P$
son la base ortonormal de \omterm{def:b2:reduction:eigen}{vectores propios}.
\end{proof}

\begin{example}[Una ejecución espectral completa]\label{ex:b2:quadratic:fullrun}
Diagonalicemos ortogonalmente $A = \begin{pmatrix} 1 & 2\\ 2 &
1\end{pmatrix}$. \omterm{def:b2:reduction:charpoly}{Polinomio característico} $(1 - \lambda)^2 - 4$:
\omterm{def:b2:reduction:eigen}{valores propios} $3$ y $-1$. \omterm{def:b2:reduction:eigen}{Vectores propios}: $(A - 3I)v = 0$ da
$v_1 = \frac{1}{\sqrt2}(1,1)$, y $(A + I)v = 0$ da $v_2 =
\frac{1}{\sqrt2}(1,-1)$; ortogonales, como garantiza el~\cref{thm:b2:quadratic:spectral} sin cálculo alguno. Con $P = (v_1\
v_2)$ (una rotación de $\frac\pi4$):
\[
P^{\mathsf T}AP = \begin{pmatrix} 3 & 0\\ 0 & -1
\end{pmatrix},
\qquad
x^2 + 4xy + y^2 = 3u^2 - v^2
\quad\text{en el sistema girado} .
\]
Así pues, la forma del~\cref{exo:b2:quadratic:1} es de tipo
hiperbólico: signatura $(1,1)$, coherente con su
\omterm{thm:b2:quadratic:gauss}{reducción de Gauss} $(x + 2y)^2 - 3y^2$; cuadrados distintos, misma
signatura, como exige Sylvester. Moraleja: Gauss dio la respuesta
más deprisa, pero la vía espectral informa además de que sobre la
circunferencia unidad $q$ recorre exactamente $\intcc{-1}{3}$,
alcanzados a lo largo de $v_2$ y de $v_1$: el trabajo extra compra
geometría.
\end{example}

\begin{corollary}[Ejes principales; tests de positividad]\label{cor:b2:quadratic:principalaxes}
\begin{enumerate}
  \item Toda \omterm{def:b2:quadratic:def}{forma cuadrática} $q$ sobre un espacio euclídeo se
        diagonaliza en alguna base \emph{ortonormal}: $q(x) = \sum_i
        \lambda_i x_i^2$, con los $\lambda_i$ los \omterm{def:b2:reduction:eigen}{valores propios}
        de la matriz \omterm{def:b2:quadratic:adjoint}{simétrica} de $q$; y la signatura cuenta los
        \omterm{def:b2:reduction:eigen}{valores propios} positivos y negativos.
  \item Una matriz \omterm{def:b2:quadratic:adjoint}{simétrica} es semidefinida positiva (o definida
        positiva) si y solo si todos sus \omterm{def:b2:reduction:eigen}{valores propios} son $\geq 0$
        (o $> 0$); y entonces los valores extremos del cociente de
        Rayleigh son
        \[
        \min_{\norm x = 1} \langle Ax, x\rangle = \lambda_{\min},
        \qquad
        \max_{\norm x = 1} \langle Ax, x\rangle = \lambda_{\max} .
        \]
\end{enumerate}
\end{corollary}

\begin{proof}
(1) Escríbase $q(x) = \langle A x, x\rangle$ con $A$ \omterm{def:b2:quadratic:adjoint}{simétrica} (la
matriz de $q$ en una base ortonormal) y diagonalícese $A$ por el
teorema espectral: para $x = \sum x_ie_i$ en la base ortonormal de
\omterm{def:b2:reduction:eigen}{vectores propios},
\[
q(x) = \Bigl\langle \sum_i \lambda_ix_ie_i,\ \sum_j
x_je_j\Bigr\rangle = \sum_i \lambda_i x_i^2
\]
(la ortonormalidad mata los términos cruzados). Los signos $\pm$ de
los $\lambda_i$ cuentan la signatura por Sylvester: reescalar cada
coordenada por $\sqrt{\abs{\lambda_i}}$ exhibe una
\omterm{thm:b2:quadratic:gauss}{reducción de Gauss} con formas independientes.

(2) En la base de \omterm{def:b2:reduction:eigen}{vectores propios}, $\langle Ax, x\rangle = \sum
\lambda_i x_i^2$, atrapado entre $\lambda_{\min}\norm x^2$ y
$\lambda_{\max}\norm x^2$, con igualdad en los \omterm{def:b2:reduction:eigen}{vectores propios}
correspondientes; la positividad de todos los \omterm{def:b2:reduction:eigen}{valores propios}
equivale, pues, a la positividad de la forma.
\end{proof}

\begin{example}\label{ex:b2:quadratic:spectralexample}
$A = \begin{pmatrix} 2 & 1\\ 1 & 2 \end{pmatrix}$: \omterm{def:b2:reduction:eigen}{valores propios}
$3$ (con \omterm{def:b2:reduction:eigen}{vector propio} $\frac{1}{\sqrt2}(1,1)$) y $1$ (con
$\frac{1}{\sqrt2}(1,-1)$). La \omterm{def:b2:quadratic:def}{forma cuadrática} $2x^2 + 2xy + 2y^2$
pasa a ser $3X^2 + Y^2$ en el sistema ortonormal girado: los ejes
principales de una elipse, calculados. La \omterm{thm:b2:quadratic:gauss}{reducción de Gauss}
alcanza también una forma diagonal, pero solo el teorema espectral
la alcanza \emph{sin distorsionar longitudes}.
\end{example}

\begin{example}[Una elipse plenamente identificada]\label{ex:b2:quadratic:ellipserun}
¿Qué curva es $5x^2 + 4xy + 2y^2 = 6$? La matriz $\begin{pmatrix} 5
& 2\\ 2 & 2\end{pmatrix}$ tiene \omterm{def:b2:reduction:charpoly}{polinomio característico} $\lambda^2
- 7\lambda + 6 = (\lambda - 1)(\lambda - 6)$: \omterm{def:b2:reduction:eigen}{valores propios} $1$ y
$6$, ambos positivos; es una elipse. \omterm{def:b2:reduction:eigen}{Vectores propios} ortonormales:
para $\lambda = 1$, resuélvase $\begin{pmatrix} 4 & 2\\ 2 &
1\end{pmatrix}v = 0$: $v_1 = \frac{1}{\sqrt5}(1, -2)$; y para
$\lambda = 6$: $v_2 = \frac{1}{\sqrt5}(2, 1)$. En las coordenadas
giradas $(X, Y)$ a lo largo de $(v_2, v_1)$, la ecuación pasa a ser
\[
6X^2 + Y^2 = 6,
\qquad\text{es decir,}\qquad
X^2 + \frac{Y^2}{6} = 1 :
\]
semiejes $1$ (a lo largo de $v_2$) y $\sqrt6$ (a lo largo de
$v_1$). Moraleja: la forma aproximada salió gratis ---$\det = 6 > 0$
y traza positiva anuncian una elipse antes de calcular ningún
vector propio---, pero solo el teorema espectral entrega las
direcciones y las longitudes de los ejes, es decir, la geometría
real.
\end{example}

\begin{example}[Extremos sobre la esfera, leídos en el espectro]\label{ex:b2:quadratic:sphereextremes}
¿Cuáles son los valores extremos de $q(x,y,z) = 2xy + 2yz + 2zx$
sobre la esfera unidad? Su matriz (la de unos fuera de la diagonal
del~\cref{exo:b2:quadratic:2}) tiene \omterm{def:b2:reduction:eigen}{valores propios} $2$ y $-1$
(doble), de modo que por el~\cref{cor:b2:quadratic:principalaxes}~(2):
\[
\max_{\norm v = 1} q(v) = 2
\ \text{ en } v = \tfrac{1}{\sqrt3}(1,1,1),
\qquad
\min_{\norm v = 1} q(v) = -1
\ \text{ sobre la circunferencia } x + y + z = 0 .
\]
Sin cálculo diferencial y sin multiplicadores de Lagrange: el
teorema espectral resuelve esta optimización con restricciones de
golpe, y exhibe el maximizador. Moraleja: compárese con el método de
los multiplicadores del capítulo de cálculo diferencial, que halla
los mismos puntos críticos con más trabajo; para objetivos
\emph{cuadráticos} sobre esferas, los \omterm{def:b2:reduction:eigen}{espectros} son el camino real
(el problema de fin de semana del capítulo hermítico construye toda
la teoría de Courant--Fischer sobre esta observación).
\end{example}

\begin{remark}[Errores frecuentes]
\emph{(i) La \omterm{def:b2:quadratic:def}{congruencia} no es la semejanza:} un cambio de base
para una forma actúa por $P^{\mathsf T}BP$, no por $P^{-1}BP$; los
\omterm{def:b2:reduction:eigen}{valores propios} \emph{no} son invariantes de una \omterm{def:b2:quadratic:def}{forma cuadrática}
($I$ y $4I$ son congruentes mediante $P = 2I$), y solo lo son sus
signos (Sylvester). Hablar de los \omterm{def:b2:reduction:eigen}{valores propios} de una forma solo
es lícito una vez fijado un producto escalar. \emph{(ii) Las
entradas positivas no demuestran nada:} $\begin{pmatrix} 1 & 2\\ 2 &
1\end{pmatrix}$ tiene todas sus entradas positivas y signatura
$(1,1)$ ($\det = -3$); y recíprocamente, una matriz definida
positiva puede tener entradas negativas fuera de la diagonal (el~\cref{ex:b2:quadratic:tworoads} desplazado: $2I - N$ sirve
igualmente). Úsese el~\cref{met:b2:quadratic:signature}.
\emph{(iii) Cuadrados dependientes:} escribir $q = \ell_1^2 -
\ell_2^2$ no dice nada si $\ell_1, \ell_2$ son proporcionales: $x^2
+ 2xy + y^2 = (x+y)^2$ tiene rango $1$ y no $2$; compruébese siempre
la independencia antes de leer la signatura. \emph{(iv) Extremos
sobre la esfera sin \omterm{def:b2:metric:compact}{compacidad}:} las cotas de Rayleigh del~\cref{cor:b2:quadratic:principalaxes} se alcanzan porque la esfera
es \omterm{def:b2:metric:compact}{compacta}; sobre la bola \omterm{def:b2:metric:topology}{abierta} o sobre todo el espacio, una
forma indefinida no tiene ni máximo ni mínimo.
\end{remark}

\begin{remark}[Dónde se usa]
El teorema espectral es el resultado más exportado de este libro: la
estadística diagonaliza con él las matrices de covarianza (análisis
de componentes principales), la mecánica extrae de él los modos
normales de oscilación (la reducción simultánea del problema de fin
de semana), el análisis numérico construye sobre él las
descomposiciones de Cholesky y en valores singulares (mismo
problema), y el capítulo siguiente lo transporta a los espacios
hermíticos complejos. El volumen del tercer año demuestra su avatar
en dimensión infinita para operadores \omterm{def:b2:metric:compact}{compactos}
autoadjuntos, donde el argumento de \omterm{def:b2:metric:compact}{compacidad} de la demostración
en dimensión finita se convierte en toda la historia.
\end{remark}

\begin{remark}[Perspectivas dentro de este volumen]
Las \omterm{def:b2:quadratic:def}{formas cuadráticas} atraviesan el resto del libro 4 bajo tres
disfraces. Como \emph{hessianas}: el capítulo de cálculo diferencial
clasifica los puntos críticos por la signatura de la forma de
segundo orden, de modo que la invariancia de Sylvester es lo que
hace de ``punto de silla'' una palabra bien definida. Como
\emph{energías}: los osciladores del capítulo de ecuaciones
diferenciales llevan la energía cuadrática $\frac12x'^2 +
\frac12\omega^2x^2$, y la reducción simultánea del problema de fin
de semana de este capítulo es exactamente la extracción de los modos
normales. Como \emph{geometría}: las cónicas de este capítulo crecen
hasta las cuádricas de los capítulos de geometría, donde la segunda
forma fundamental de una superficie ---una \omterm{def:b2:quadratic:def}{forma cuadrática} sobre
cada plano tangente--- tiene una signatura que decide si la
superficie se curva como un cuenco o como una silla de montar. El
capítulo hermítico, a continuación, vuelve a tocar toda la partitura
sobre $\C$.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:quadratic:1}
Reduce por Gauss y da el rango y la signatura de
\[
q_1(x,y) = x^2 + 4xy + y^2,
\qquad
q_2(x,y,z) = x^2 + 2y^2 + 3z^2 + 2xy + 2yz .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:quadratic:2}
Diagonaliza ortogonalmente $A = \begin{pmatrix} 0 & 1 & 1\\ 1 & 0 &
1\\ 1 & 1 & 0\end{pmatrix}$ (los \omterm{def:b2:reduction:eigen}{valores propios} salen del cálculo
del~\cref{ch:b2:reduction}; hazlo ahora con base ortonormal) y
reduce a ejes principales la forma $q(x,y,z) = 2xy + 2yz + 2zx$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:quadratic:3}
Demuestra que $u^{**} = u$, que $(u \circ v)^* = v^* \circ u^*$ y
que $\ker u^* = (\operatorname{im} u)^{\perp}$. Deduce
$\operatorname{rk} u^* = \operatorname{rk} u$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:4}
Sea $A$ real \omterm{def:b2:quadratic:adjoint}{simétrica} con $A^3 = A$. Demuestra que $A^2$ es la
matriz de una proyección ortogonal. Más en general, relaciona las
descomposiciones espectrales de $A$ y de $P(A)$ para un polinomio
$P$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:5}
Demuestra que $O(n) = \{P : P^{\mathsf T}P = I\}$ es un subconjunto
\omterm{def:b2:metric:compact}{compacto} de $\mathcal{M}_n(\R)$ \emph{(cerrado: imagen recíproca de
$\{I\}$ por una aplicación \omterm{def:b2:metric:continuity}{continua}; acotado: las columnas son
vectores unitarios)}. ¿Es \omterm{def:b2:metric:connected}{conexo}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:6}
(Raíz cuadrada) Sea $A$ \omterm{def:b2:quadratic:adjoint}{simétrica} semidefinida positiva. Construye
una $B$ \omterm{def:b2:quadratic:adjoint}{simétrica} semidefinida positiva con $B^2 = A$ y demuestra
que es \emph{única} \emph{(existencia: tómense raíces cuadradas de
los \omterm{def:b2:reduction:eigen}{valores propios} en una base espectral; unicidad: una candidata
$B$ conmuta con $A = B^2$, luego conserva sus \omterm{def:b2:reduction:eigen}{subespacios propios};
redúzcase al caso escalar sobre cada uno)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:7}
Para $A$ real \omterm{def:b2:quadratic:adjoint}{simétrica}, demuestra que $\vertiii{A}_2 :=
\sup_{\norm x_2 = 1}\norm{Ax}_2 = \max_i \abs{\lambda_i}$ (radio
espectral) y calcula $\vertiii{A}_2$ para $A = \begin{pmatrix} 1 &
2\\ 2 & 1\end{pmatrix}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:quadratic:8}
(Criterio de Sylvester) Sea $A$ real \omterm{def:b2:quadratic:adjoint}{simétrica} con menores
principales dominantes $\Delta_1, \dots, \Delta_n$
(\omterm{def:b2:linalg:det}{determinantes} de los bloques superiores izquierdos). Demuestra que
$A$ es definida positiva si y solo si todos los $\Delta_k > 0$.
\emph{(Para $\Rightarrow$: las restricciones de una forma definida
son definidas, y el \omterm{def:b2:linalg:det}{determinante de una matriz} definida positiva ---el
producto de sus valores propios--- es positivo. Para $\Leftarrow$:
inducción sobre $n$; el bloque superior izquierdo de tamaño $n-1$ es
definido positivo, diagonalícese la forma sobre ese subespacio y
complétese el cuadrado en la última variable; el signo de la última
entrada diagonal lo gobierna $\det A = \Delta_n > 0$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:quadratic:9}
(Courant--Fischer, segundo \omterm{def:b2:reduction:eigen}{valor propio}) Sea $u$ \omterm{def:b2:quadratic:adjoint}{simétrico} con
\omterm{def:b2:reduction:eigen}{valores propios} $\lambda_1 \geq \lambda_2 \geq \dots \geq
\lambda_n$. Demuestra que
\[
\lambda_2 = \min_{\substack{H \text{ hiperplano}}}\;
\max_{\substack{x \in H,\ \norm x = 1}} \langle u(x), x\rangle .
\]
\emph{(Para $\leq$: todo hiperplano corta al $2$-plano \omterm{def:b2:structures:generated}{generado} por
los dos \omterm{def:b2:reduction:eigen}{vectores propios} superiores. Para $\geq$: tómese $H =
(e_1)^{\perp}$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:10}
Determina el rango y la signatura de $q(x_1, \dots, x_n) = \sum_{i <
j} x_ix_j$ sobre $\R^n$ ($n \geq 2$), de dos maneras: mediante la
identidad algebraica $2q = \bigl(\sum x_i\bigr)^2 - \sum x_i^2$
junto con la restricción de $q$ al hiperplano $\sum x_i = 0$; y
calculando los \omterm{def:b2:reduction:eigen}{valores propios} de su matriz $\frac12(J - I)$, donde
$J$ es la matriz de unos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:quadratic:11}
Sean $A$ y $B$ reales \omterm{def:b2:quadratic:adjoint}{simétricas} con $B$ semidefinida positiva.
Demuestra que
\[
\lambda_{\min}(A)\operatorname{tr} B
\;\leq\;
\operatorname{tr}(AB)
\;\leq\;
\lambda_{\max}(A)\operatorname{tr} B .
\]
\emph{(Escribe $B = C^{\mathsf T}C$ y $\operatorname{tr}(AB) =
\sum_i \langle A c_i, c_i\rangle$ sobre las columnas $c_i$ de
$C^{\mathsf T}$.)} En particular, $\operatorname{tr}(AB) \geq 0$
cuando ambas son semidefinidas positivas.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:quadratic:12}
Sobre $E = \mathcal{M}_n(\R)$, consideremos $q(M) =
\operatorname{tr}(M^2)$.
\begin{enumerate}
  \item Prueba que $q$ es una \omterm{def:b2:quadratic:def}{forma cuadrática} de forma polar
        $\varphi(M, N) = \operatorname{tr}(MN)$.
  \item Prueba que las matrices \omterm{def:b2:quadratic:adjoint}{simétricas} y las antisimétricas
        forman subespacios $\varphi$-ortogonales sobre los que $q$
        es, respectivamente, definida positiva y definida negativa
        \emph{(calcula $\operatorname{tr}(M^2)$ entrada a entrada en
        cada caso)}.
  \item Concluye que $q$ tiene signatura $\bigl(\frac{n(n+1)}{2},
        \frac{n(n-1)}{2}\bigr)$ y rango $n^2$.
\end{enumerate}
\end{exercise}

\section{Problema: Cholesky, Hadamard y la descomposición polar}

\begin{problem}\label{pb:b2:quadratic:1}
El teorema espectral es un microscopio; este problema lo usa como
fábrica. A partir de las matrices de Gram fabricamos la
\emph{factorización de Cholesky}\index{factorización de Cholesky} (e
identificamos los pivotes de Gauss como cocientes de menores),
demostramos después la \emph{desigualdad de
Hadamard}\index{desigualdad de Hadamard} sobre \omterm{def:b2:linalg:det}{determinantes},
construimos la \emph{descomposición polar}\index{descomposición
polar} $A = QS$ y la \emph{descomposición en valores singulares},
clasificamos las cónicas del plano y terminamos con la reducción
simultánea de dos formas: el teorema que hay tras los modos normales
de oscilación. En todo el problema, $E = \R^n$ con su producto
escalar estándar.

\medskip
\noindent\textbf{Parte I --- Matrices de Gram y Cholesky.} Para
vectores $v_1, \dots, v_n \in E$, su \emph{matriz de Gram} es $G =
\bigl(\langle v_i, v_j\rangle\bigr)_{i,j}$.
\begin{enumerate}
  \item Prueba que $G$ es \omterm{def:b2:quadratic:adjoint}{simétrica} semidefinida positiva, y
        definida positiva si y solo si $(v_1, \dots, v_n)$ es
        linealmente independiente \emph{(calcula $X^{\mathsf
        T}GX$)}.
  \item Recíprocamente, prueba que toda $A$ \omterm{def:b2:quadratic:adjoint}{simétrica} semidefinida
        positiva es una matriz de Gram: $A = C^{\mathsf T}C$ para
        cierta $C$ (úsese la raíz cuadrada del~\cref{exo:b2:quadratic:6}), con $C$ invertible si y solo si
        $A$ es definida.
  \item Deduce que una $A$ semidefinida positiva cumple
        $\abs{a_{ij}} \leq \sqrt{a_{ii}\,a_{jj}}$ para todos $i, j$
        \emph{(restrínjase a dos coordenadas)}: la desigualdad de
        Cauchy--Schwarz releída matricialmente.
  \item (Cholesky) Sea $A$ definida positiva. Demuestra que existe
        una \emph{única} matriz triangular superior $T$ con entradas
        diagonales positivas tal que
        \[
        A = T^{\mathsf T}\,T
        \]
        \emph{(existencia: aplíquese Gram--Schmidt a vectores que
        realicen $A$ como matriz de Gram; unicidad: si $T_1^{\mathsf
        T}T_1 = T_2^{\mathsf T}T_2$, pruébese que $T_1T_2^{-1}$ es
        ortogonal y triangular con diagonal positiva, luego $I$)}.
  \item Prueba que los menores principales dominantes cumplen
        $\Delta_k = (t_{11}\cdots t_{kk})^2$ y deduce que los
        pivotes producidos por la \omterm{thm:b2:quadratic:gauss}{reducción de Gauss} de una forma
        definida positiva, tomados en el orden natural de las
        variables, son
        \[
        d_k = \frac{\Delta_k}{\Delta_{k-1}}
        \qquad (\Delta_0 = 1) :
        \]
        los menores del criterio de Sylvester
        (\cref{exo:b2:quadratic:8}) y los pivotes de Gauss son el
        mismo dato. Compruébalo sobre el~\cref{ex:b2:quadratic:tworoads}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La desigualdad de Hadamard.}
\begin{enumerate}[resume]
  \item Sea $A$ definida positiva. Demuestra que
        \[
        \det A \leq a_{11}\,a_{22}\cdots a_{nn}
        \]
        \emph{(normalícese: $B = DAD$ con $D =
        \operatorname{diag}(a_{ii}^{-1/2})$ tiene diagonal de unos;
        acótese $\det B = \prod \mu_i$ mediante la desigualdad entre
        medias contra $\operatorname{tr} B = n$)}.
  \item Prueba que hay igualdad si y solo si $A$ es diagonal.
  \item Deduce la \emph{desigualdad de Hadamard}: para toda matriz
        cuadrada real $M$ de columnas $c_1, \dots, c_n$,
        \[
        \abs{\det M} \leq \prod_{i=1}^{n}\norm{c_i}_2 ,
        \]
        con igualdad (para $M$ invertible) si y solo si las columnas
        son ortogonales dos a dos \emph{(aplíquense las preguntas
        6--7 a $M^{\mathsf T}M$)}.
  \item Dividendos geométricos y combinatorios: interpreta la
        pregunta 8 como ``el volumen de un paralelepípedo es a lo
        sumo el producto de las longitudes de sus aristas'', y
        prueba que una matriz con todas sus entradas en
        $\intcc{-1}{1}$ cumple $\abs{\det M} \leq n^{n/2}$. (Las
        matrices que alcanzan esa cota ---las matrices de
        Hadamard--- existen para $n = 1, 2$ y para muchos múltiplos
        de $4$; si existen para \emph{todos} los múltiplos de $4$ es
        un famoso problema \omterm{def:b2:metric:topology}{abierto}.)
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- \omterm{pb:b2:quadratic:1}{Descomposición polar} y valores
singulares.}
\begin{enumerate}[resume]
  \item Sea $A$ invertible. Prueba que $A^{\mathsf T}A$ es definida
        positiva y que
        \[
        S = \sqrt{A^{\mathsf T}A}
        \quad\text{(la raíz cuadrada del~\cref{exo:b2:quadratic:6})},
        \qquad Q = AS^{-1}
        \]
        dan una factorización $A = QS$ con $Q$ ortogonal y $S$
        definida positiva.
  \item Demuestra que esta factorización de una $A$ invertible es
        única.
  \item Extiende la existencia a una $A$ arbitraria: elíjanse
        $\varepsilon_k \to 0$ con $A + \varepsilon_k I$ invertible,
        escríbase $A + \varepsilon_kI = Q_kS_k$ y úsese la
        \omterm{def:b2:metric:compact}{compacidad} de $O(n)$ (\cref{exo:b2:quadratic:5}) para
        extraer $Q_k \to Q$; prueba que $S_k = Q_k^{\mathsf T}(A +
        \varepsilon_kI)$ converge a cierta $S$ semidefinida positiva
        con $A = QS$ y $S = \sqrt{A^{\mathsf T}A}$. ¿Dónde falla la
        unicidad para $A$ singular?
  \item (Descomposición en valores singulares) Deduce que toda $A$
        real cuadrada se escribe
        \[
        A = U\,\Sigma\,V^{\mathsf T},
        \qquad U, V \in O(n),\quad
        \Sigma = \operatorname{diag}(\sigma_1, \dots,
        \sigma_n),\ \sigma_i \geq 0 ,
        \]
        donde los $\sigma_i$ (los \emph{valores singulares}) son los
        \omterm{def:b2:reduction:eigen}{valores propios} de $\sqrt{A^{\mathsf T}A}$.
  \item Tres consecuencias: $\vertiii{A}_2 = \sigma_{\max}$ para
        \emph{toda} $A$ real (generalizando el~\cref{exo:b2:quadratic:7}); $\abs{\det A} =
        \sigma_1\cdots\sigma_n$; y la imagen de la esfera unidad por
        una $A$ invertible es un elipsoide de semiejes $\sigma_1,
        \dots, \sigma_n$ a lo largo de las columnas de $U$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Cónicas, por el teorema espectral.}
Una cónica del plano es el conjunto de ceros de $f(x) = q(x) +
\langle b, x\rangle + c$, con $q \neq 0$ una \omterm{def:b2:quadratic:def}{forma cuadrática} de
matriz $A$, $b \in \R^2$ y $c \in \R$.
\begin{enumerate}[resume]
  \item Reduce $f$ mediante una rotación (ejes principales,
        \cref{cor:b2:quadratic:principalaxes}) seguida de una
        traslación, y clasifica las formas posibles no vacías y no
        degeneradas según la signatura de $q$: elipse ($\det A >
        0$), hipérbola ($\det A < 0$), parábola ($\det A = 0$, rango
        $1$, con el término lineal no absorbido).
  \item Ejecuta la reducción completa para
        \[
        x^2 + 4xy + y^2 + 2x - 2y = 4 :
        \]
        coordenadas giradas, ecuación reducida, naturaleza y centro
        de la cónica.
  \item (Cónicas con centro) Supongamos $\det A \neq 0$. Prueba que
        el \emph{centro} es $x_0 = -\frac12 A^{-1}b$ y que la
        \omterm{def:b2:quadratic:def}{congruencia} por $\begin{pmatrix} I & x_0\\ 0 &
        1\end{pmatrix}$ de la matriz $3\times3$ $\widetilde Q =
        \begin{pmatrix} A & b/2 \\ b^{\mathsf T}/2 &
        c\end{pmatrix}$ da
        \[
        \det\widetilde Q = f(x_0)\,\det A :
        \]
        la cónica con centro es degenerada (un punto o dos rectas)
        exactamente cuando $\det\widetilde Q = 0$.
  \item Verifica la pregunta 17 sobre el ejemplo de la pregunta 16:
        calcula $x_0$, $f(x_0)$ y $\det\widetilde Q$, y concluye de
        nuevo que la cónica es una hipérbola no degenerada.
  \item (Un haz de cuádricas) Para $\lambda \in \R$, clasifica la
        superficie
        \[
        x^2 + y^2 + z^2 + 2\lambda(xy + yz + zx) = 1
        \]
        mediante los \omterm{def:b2:reduction:eigen}{valores propios} de su matriz \emph{(estructura
        de unos: \omterm{def:b2:reduction:eigen}{valores propios} $1 + 2\lambda$ y $1 - \lambda$
        doble)}: esfera o elipsoide, cilindro, par de planos,
        hiperboloides de una y de dos hojas, según $\lambda$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dos formas a la vez: reducción
simultánea.}
\begin{enumerate}[resume]
  \item Sea $q$ definida positiva y $q'$ una \omterm{def:b2:quadratic:def}{forma cuadrática}
        arbitraria sobre $E$. Demuestra que existe una base de $E$
        ortonormal para $q$ y ortogonal para $q'$: en ella, $q =
        \sum x_i^2$ y $q' = \sum \mu_i x_i^2$ \emph{(úsese $q$ como
        producto escalar y aplíquese el teorema espectral al
        endomorfismo que representa $q'$)}.
  \item Forma matricial: para $A$ definida positiva y $B$
        \omterm{def:b2:quadratic:adjoint}{simétrica}, existe una $P$ invertible con $P^{\mathsf T}AP =
        I$ y $P^{\mathsf T}BP = \operatorname{diag}(\mu_1, \dots,
        \mu_n)$, donde los $\mu_i$ son las raíces de $\det(B - \mu
        A) = 0$.
  \item Ejecútalo por completo para
        \[
        A = \begin{pmatrix} 2 & 1\\ 1 & 1 \end{pmatrix},
        \qquad
        B = \begin{pmatrix} 0 & 1\\ 1 & 0 \end{pmatrix} :
        \]
        los \omterm{def:b2:reduction:eigen}{valores propios} generalizados $\mu_\pm$ y unos vectores
        que diagonalicen ambas formas.
  \item Prueba que no puede prescindirse de la definición positiva:
        para
        \[
        A = \begin{pmatrix} 1 & 0\\ 0 & -1 \end{pmatrix},
        \qquad
        B = \begin{pmatrix} 0 & 1\\ 1 & 0 \end{pmatrix},
        \]
        ninguna base diagonaliza ambas formas \emph{(si $P$
        diagonalizara ambas, $\det(B - \mu A)$ se escindiría con
        raíces reales; calcúlese)}.
  \item Prueba que los $\mu_i$ de la pregunta 21 son los
        \omterm{def:b2:reduction:eigen}{valores propios} de $A^{-1}B$, y que $A^{-1}B$, sin ser
        \omterm{def:b2:quadratic:adjoint}{simétrica} en general, es siempre \omterm{def:b2:reduction:diag}{diagonalizable} con
        \omterm{def:b2:reduction:eigen}{valores propios} reales \emph{(conjúguese por $\sqrt A$)}.
  \item Síntesis. Una frase para cada punto: (i) el único teorema en
        el que se apoyó cada parte; (ii) qué resultados de las
        partes I--III sobreviven para matrices
        \emph{semi}definidas positivas y cuáles necesitan la
        definición estricta; (iii) el sistema físico cuyas pequeñas
        oscilaciones diagonalizan las preguntas 20--22 (las energías
        cinética y potencial como las dos formas) y qué significan
        allí los $\mu_i$.
\end{enumerate}
\end{problem}
