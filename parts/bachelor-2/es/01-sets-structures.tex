\section{Conjuntos, aplicaciones, cocientes}

Usamos con toda libertad el lenguaje de conjuntos, aplicaciones y
relaciones de equivalencia y de \omterm{def:b2:structures:generated}{orden}
establecido en el volumen del primer año. Dos mejoras merecen un
enunciado propio.

\begin{proposition}[Imágenes e imágenes recíprocas de familias]\label{prop:b2:structures:images}
Sea $f \colon E \to F$ y sean $(A_i)_{i \in I}$, $(B_j)_{j \in J}$
familias de subconjuntos de $E$ y de $F$, respectivamente. Entonces
\[
f^{-1}\Bigl(\bigcup_j B_j\Bigr) = \bigcup_j f^{-1}(B_j),
\qquad
f^{-1}\Bigl(\bigcap_j B_j\Bigr) = \bigcap_j f^{-1}(B_j),
\qquad
f^{-1}(F \setminus B) = E \setminus f^{-1}(B),
\]
\[
f\Bigl(\bigcup_i A_i\Bigr) = \bigcup_i f(A_i),
\qquad
f\Bigl(\bigcap_i A_i\Bigr) \subseteq \bigcap_i f(A_i)
\quad (\text{con igualdad si } f \text{ es inyectiva}).
\]
\end{proposition}

\begin{proof}
Cada identidad se obtiene desplegando las definiciones; por ejemplo,
$x \in f^{-1}(\bigcap B_j) \iff f(x) \in B_j$ para todo $j$ $\iff x
\in f^{-1}(B_j)$ para todo $j$. Las identidades sobre imágenes y el
fallo de la igualdad en el caso de la intersección (con el remedio de
la inyectividad) se demostraron en el volumen del primer año para dos
conjuntos; los argumentos son idénticos para familias.
\end{proof}

\begin{example}[Cuándo la inclusión de imágenes es estricta]\label{ex:b2:structures:imagestrict}
Tomemos $f \colon \R \to \R$, $f(x) = x^2$, con $A_1 =
\intcc{-1}{0}$ y $A_2 = \intcc{0}{1}$. Entonces
\[
f(A_1 \cap A_2) = f(\{0\}) = \{0\},
\qquad
f(A_1) \cap f(A_2) = \intcc{0}{1} \cap \intcc{0}{1} =
\intcc{0}{1} :
\]
la inclusión de la~\cref{prop:b2:structures:images} es todo lo
estricta que puede ser: las dos preimágenes $\pm x$ de un mismo
valor viven en $A_i$ distintos. La inyectividad es justamente lo
que prohíbe ese desdoblamiento, y por eso las imágenes recíprocas
(que nunca funden puntos) cumplen las cuatro identidades sin
condiciones, mientras que las imágenes pierden la de la
intersección. Regla práctica para todo el libro: las
\emph{imágenes recíprocas} atraviesan las operaciones
conjuntistas sin más; con las imágenes hay que ir con cuidado.
\end{example}

\begin{definition}[Conjunto cociente]\label{def:b2:structures:quotient}
Sea $\mathcal{R}$ una relación de equivalencia sobre $E$. El
\emph{conjunto cociente}\index{conjunto cociente} $E/\mathcal{R}$ es
el conjunto de las clases de equivalencia; la sobreyección $\pi
\colon E \to E/\mathcal{R}$, $x \mapsto \mathrm{cl}(x)$, es la
\emph{proyección canónica}\index{proyección canónica}.

\emph{Propiedad universal (factorización):} si $f \colon E \to F$ es
\emph{compatible} con $\mathcal{R}$ (es decir, $x
\mathbin{\mathcal{R}} y \implies f(x) = f(y)$), existe una única
aplicación $\overline f \colon E/\mathcal{R} \to F$ tal que $f =
\overline f \circ \pi$.
\end{definition}

\begin{proof}[Demostración de la propiedad universal]
Unicidad: la condición $f = \overline f \circ \pi$ se lee
\[
\overline f\bigl(\mathrm{cl}(x)\bigr) = f(x)
\qquad (x \in E),
\]
y, como $\pi$ es sobreyectiva, todo elemento de $E/\mathcal{R}$
es de la forma $\mathrm{cl}(x)$: los valores de $\overline f$
quedan todos forzados. Existencia: tomemos la fórmula anterior
como \emph{definición} de $\overline f$; es inequívoca
precisamente por la compatibilidad ---si $\mathrm{cl}(x) =
\mathrm{cl}(y)$, entonces $x \mathbin{\mathcal{R}} y$, luego
$f(x) = f(y)$ y los dos valores candidatos coinciden--- y
factoriza $f$ por construcción. Obsérvese el reparto de papeles:
la sobreyectividad de $\pi$ da la unicidad, la compatibilidad da
la existencia.
\end{proof}

\begin{example}
$\Z/n\Z$ es el cociente de $\Z$ por la congruencia módulo $n$; las
comprobaciones de buena definición del volumen del primer año eran
casos particulares de la propiedad universal. Los cocientes
convierten las ``construcciones compatibles sobre representantes''
en aplicaciones legítimas; lo usaremos sin cesar más abajo.
\end{example}

\section{Numerabilidad y cardinalidad}

\begin{definition}[Equipotencia, numerabilidad]\label{def:b2:structures:countable}
Dos conjuntos son \emph{equipotentes}\index{equipotencia} cuando
existe una biyección entre ellos. Un conjunto es
\emph{numerable}\index{conjunto numerable} cuando es equipotente a
$\N$ (algunos autores incluyen los conjuntos finitos; nosotros
decimos \emph{a lo sumo numerable} para ``finito o numerable'').
\end{definition}

\begin{proposition}[Propiedades de estabilidad]\label{prop:b2:structures:countablestable}
\begin{enumerate}
  \item Todo subconjunto infinito de $\N$ es \omterm{def:b2:structures:countable}{numerable}; un conjunto es a lo
        sumo \omterm{def:b2:structures:countable}{numerable} si y solo si se inyecta en $\N$, si y solo si es
        vacío o imagen sobreyectiva de $\N$.
  \item $\N \times \N$ es \omterm{def:b2:structures:countable}{numerable}; un producto de dos conjuntos a lo
        sumo \omterm{def:b2:structures:countable}{numerables} es a lo sumo \omterm{def:b2:structures:countable}{numerable}.
  \item Una unión a lo sumo \omterm{def:b2:structures:countable}{numerable} de conjuntos a lo sumo
        \omterm{def:b2:structures:countable}{numerables} es a lo sumo \omterm{def:b2:structures:countable}{numerable}.
  \item $\Z$ y $\Q$ son \omterm{def:b2:structures:countable}{numerables}.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Enumeremos un $A \subseteq \N$ infinito por mínimos sucesivos:
$a_0 = \min A$, $a_{k+1} = \min\,(A \setminus \{a_0, \dots, a_k\})$
(conjunto no vacío, pues $A$ es infinito); la aplicación $k \mapsto
a_k$ es estrictamente creciente, inyectiva y sobreyectiva sobre $A$
(cada $a \in A$ supera solo a un número finito de elementos de $A$,
luego se alcanza). Si $E$ se inyecta en $\N$ mediante $\varphi$,
entonces $E$ es \omterm{def:b2:structures:countable}{equipotente} a $\varphi(E) \subseteq \N$: finito o
\omterm{def:b2:structures:countable}{numerable}. Si $s \colon \N \to E$ es sobreyectiva, entonces $x
\mapsto \min s^{-1}(\{x\})$ inyecta $E$ en $\N$.

(2) La aplicación $(p, q) \mapsto 2^p(2q + 1) - 1$ es una biyección
$\N^2 \to \N$ (todo entero positivo admite una única descomposición
$2^p m$ con $m$ impar, por la factorización única). Productos:
compóngase con inyecciones.

(3) Dados conjuntos $E_n$ con sobreyecciones $s_n \colon \N \to E_n$
(inofensivo cuando algún $E_n$ es finito: se repiten valores), la
aplicación $(n, k) \mapsto s_n(k)$ es una sobreyección del \omterm{def:b2:structures:countable}{conjunto
numerable} $\N^2$ sobre $\bigcup E_n$.

(4) $\Z = \N \cup (-\N^*)$: unión \omterm{def:b2:structures:countable}{numerable}. $\Q$ es imagen
sobreyectiva de $\Z \times \N^*$ (la aplicación fracción), luego es a
lo sumo \omterm{def:b2:structures:countable}{numerable}, y es infinito.
\end{proof}

\begin{example}[Una función de emparejamiento, calculada]\label{ex:b2:structures:pairing}
La biyección $(p, q) \mapsto 2^p(2q + 1) - 1$ de la demostración
merece verse en funcionamiento. Sus primeros valores:
\[
\begin{array}{c|ccccc}
 & q = 0 & q = 1 & q = 2 & q = 3 & q = 4\\
\hline
p = 0 & 0 & 2 & 4 & 6 & 8\\
p = 1 & 1 & 5 & 9 & 13 & 17\\
p = 2 & 3 & 11 & 19 & 27 & 35\\
p = 3 & 7 & 23 & 39 & 55 & 71
\end{array}
\]
La fila $p$ reúne los enteros $n$ para los que $n + 1$ es
divisible exactamente por $2^p$: cada número natural aparece una
sola vez. Descodificar es tan explícito como codificar: para $n =
43$ se factoriza $n + 1 = 44 = 2^2\cdot 11 = 2^2(2\cdot5 + 1)$,
luego $(p, q) = (2, 5)$. La moraleja: las demostraciones de
\omterm{def:b2:structures:countable}{numerabilidad} son a menudo
\emph{algoritmos} disfrazados; aquí, ``sacar factor común los
doses''.
\end{example}

\begin{example}[Los números algebraicos son numerables]\label{ex:b2:structures:algebraic}
Un número complejo es \emph{algebraico} cuando anula algún
polinomio no nulo con coeficientes racionales. El conjunto
$\overline\Q$ de los números algebraicos es \omterm{def:b2:structures:countable}{numerable}: los polinomios
de grado $\leq d$ sobre $\Q$ se inyectan en $\Q^{d+1}$, producto
finito de \omterm{def:b2:structures:countable}{conjuntos numerables}
(\cref{prop:b2:structures:countablestable}~(2)); la unión sobre
$d$ enumera los polinomios racionales no nulos como $P_0, P_1,
P_2, \dots$; cada $P_k$ tiene un número finito de raíces; y
\[
\overline\Q = \bigcup_{k \in \N}\ \{\text{raíces de } P_k\}
\]
es una unión \omterm{def:b2:structures:countable}{numerable} de conjuntos finitos
(\cref{prop:b2:structures:countablestable}~(3)), infinita porque
contiene a $\Q$. Junto con la no \omterm{def:b2:structures:countable}{numerabilidad} de $\R$
(\cref{thm:b2:structures:cantor} más abajo), esto demuestra ---sin
exhibir ni uno solo--- que los números trascendentes existen y
forman una mayoría no \omterm{def:b2:structures:countable}{numerable}: el argumento de conteo de Cantor
de 1874, existencia por pura cardinalidad.
\end{example}

\begin{theorem}[Cantor; no numerabilidad de $\R$]\label{thm:b2:structures:cantor}
\begin{enumerate}
  \item Para todo conjunto $E$ no existe ninguna sobreyección $E \to
        \mathcal{P}(E)$.
  \item $\R$ \emph{no} es \omterm{def:b2:structures:countable}{numerable}.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) se demostró en el volumen del primer año (el conjunto diagonal
$D = \{x : x \notin f(x)\}$).

(2) Supongamos que $(x_n)_{n \in \N}$ enumera $\R$. Construyamos
segmentos encajados $I_0 \supseteq I_1 \supseteq \dots$ con
$\abs{I_n} = 3^{-n}$ y $x_n \notin I_n$: se divide el segmento actual
en tres tercios cerrados; al menos uno de ellos evita $x_n$ (un punto
pertenece a lo sumo a dos de los tres). El teorema de los segmentos
encajados (extremos adyacentes) proporciona $\ell \in \bigcap_n I_n$;
pero $\ell = x_N$ para algún $N$, y $x_N \notin I_N$: contradicción.
\end{proof}

\begin{theorem}[Cantor--Bernstein]\label{thm:b2:structures:cantorbernstein}
Si $E$ se inyecta en $F$ y $F$ se inyecta en $E$, entonces $E$ y $F$
son \omterm{def:b2:structures:countable}{equipotentes}.\index{teorema de Cantor--Bernstein}
\end{theorem}

\begin{proof}
Sean $f \colon E \to F$ y $g \colon F \to E$ inyectivas. Para cada
punto (de $E$ o de $F$), recorramos su \emph{cadena de antecesores}
de preimágenes sucesivas, $x \mapsto g^{-1}(x) \mapsto
f^{-1}(g^{-1}(x)) \mapsto \dots$: cada paso está definido mientras el
punto actual esté en la imagen de la inyección correspondiente, y
entonces es único por inyectividad. Hay tres destinos mutuamente
excluyentes: la cadena se detiene en un punto de $E \setminus g(F)$
(\emph{origen en $E$}), se detiene en un punto de $F \setminus f(E)$
(\emph{origen en $F$}) o no se detiene nunca. Esto reparte $E = E_E
\cup E_F \cup E_\infty$ y $F = F_E \cup F_F \cup F_\infty$ según el
origen.

Obsérvese ahora que $f$ aplica $E_E$ \emph{sobre} $F_E$: la cadena de
$f(x)$ es la de $x$ con un paso más al principio, luego los orígenes
coinciden; y todo $y \in F_E$ tiene una cadena con al menos un paso
(su origen está en $E$), así que $y = f(x)$ con $x \in E_E$. El mismo
argumento da biyecciones $f \colon E_\infty \to F_\infty$ y $g \colon
F_F \to E_F$. Pegando,
\[
h(x) =
\begin{cases}
f(x) & \text{si } x \in E_E \cup E_\infty,\\
g^{-1}(x) & \text{si } x \in E_F,
\end{cases}
\]
es una biyección de $E$ sobre $F = F_E \cup F_\infty \cup F_F$: es
biyectiva en cada trozo y las tres piezas de llegada son disjuntas.
\end{proof}

\begin{example}\label{ex:b2:structures:cbexample}
$\intoo{0}{1}$ y $\intcc{0}{1}$ son \omterm{def:b2:structures:countable}{equipotentes}: la identidad inyecta
en un sentido y $x \mapsto \frac{x + 1}{3}$ en el otro; el teorema
fabrica la biyección (forzosamente discontinua). Del mismo modo,
$\R$, $\intoo{0}{1}$ (mediante biyecciones de tipo $\tanh$) y
$\mathcal{P}(\N)$ (desarrollos binarios, \cref{exo:b2:structures:3})
son todos \omterm{def:b2:structures:countable}{equipotentes}: ``el cardinal del continuo''.
\end{example}

\begin{example}[El segmento y el cuadrado]\label{ex:b2:structures:squareseg}
$\intcc{0}{1}$ y $\intcc{0}{1}^2$ son \omterm{def:b2:structures:countable}{equipotentes}: la dimensión es
invisible para el cardinal. Una inyección es trivial: $x \mapsto
(x, 0)$. Para la otra, enviemos $(x, y)$ al número real cuyas
cifras decimales intercalan las de $x$ y las de $y$,
\[
(0.x_1x_2x_3\dots,\ 0.y_1y_2y_3\dots)
\;\longmapsto\; 0.x_1y_1x_2y_2x_3y_3\dots,
\]
eligiendo para cada coordenada el desarrollo que no termina en
infinitos $9$: con ese convenio las cifras de la imagen
determinan las de $x$ y las de $y$, de modo que la aplicación es
inyectiva (no tiene por qué ser sobreyectiva ---ninguna imagen
tiene, por ejemplo, todas las cifras de posición impar iguales a
$9$ a partir de un punto---, y eso no importa).
Cantor--Bernstein (\cref{thm:b2:structures:cantorbernstein})
ensambla una biyección auténtica. La continuidad, eso sí, es
inalcanzable: una biyección continua entre ambos es imposible, y
los capítulos métricos explican por qué (la conexidad distingue
la recta del plano, \cref{ch:b2:metric}).
\end{example}

\section{Grupos}

\begin{definition}[Subgrupo generado; orden]\label{def:b2:structures:generated}
Sea $G$ un grupo y $A \subseteq G$. El subgrupo \emph{generado} por
$A$, escrito $\langle A \rangle$, es el menor subgrupo que contiene a
$A$; en concreto, el conjunto de todos los productos finitos de
elementos de $A$ y de sus inversos. Un grupo es
\emph{cíclico}\index{grupo cíclico} cuando está generado por un solo
elemento: $\langle a\rangle = \{a^k : k \in \Z\}$. El
\emph{orden}\index{orden!de un elemento} de $a \in G$ es
$\operatorname{ord}(a) = \abs{\langle a \rangle}$ (posiblemente
infinito); cuando es finito, es el menor $n \geq 1$ con $a^n = e$, y
$a^k = e \iff \operatorname{ord}(a) \mid k$.
\end{definition}

\begin{proof}[Demostración de la caracterización del orden]
Si algún $a^m = e$ con $m \geq 1$, sea $n \geq 1$ el menor con $a^n =
e$. Los elementos $e, a, \dots, a^{n-1}$ son distintos dos a dos
($a^{i} = a^{j}$ con $0 \leq i < j < n$ da $a^{j-i} = e$, en contra
de la minimalidad), y toda potencia $a^k$ se reduce a uno de ellos
mediante la división euclídea $k = nq + r$: $\langle a\rangle$ tiene
exactamente $n$ elementos, y $a^k = a^r = e \iff r = 0 \iff n \mid
k$. Si ninguna potencia es trivial, todas las $a^k$ ($k \in \Z$) son
distintas (mismo argumento de división) y el \omterm{def:b2:structures:generated}{orden} es infinito.
\end{proof}

\begin{theorem}[Lagrange]\label{thm:b2:structures:lagrange}
Sea $G$ un grupo finito y $H$ un subgrupo. Entonces $\abs H$ divide a
$\abs G$.\index{teorema de Lagrange} En particular, el \omterm{def:b2:structures:generated}{orden} de todo
elemento divide a $\abs G$, y $a^{\abs G} = e$ para todo $a \in G$.
\end{theorem}

\begin{proof}
La relación $x \sim y \iff x^{-1}y \in H$ es de equivalencia
(reflexiva: $e \in H$; simétrica: por inversos; transitiva: por
productos). La clase de $x$ es la \emph{clase lateral izquierda} $xH
= \{xh : h \in H\}$, y $h \mapsto xh$ es una biyección $H \to xH$ (de
inversa $y \mapsto x^{-1}y$): todas las clases tienen $\abs H$
elementos. Las clases forman una partición de $G$ (el teorema general
de partición del volumen del primer año), luego $\abs G = \abs H
\times (\text{número de clases})$. Para un elemento: aplíquese lo
anterior a $H = \langle a\rangle$; entonces $a^{\abs G} =
(a^{\operatorname{ord} a})^{\abs G / \operatorname{ord} a} = e$.
\end{proof}

\begin{example}[Clases laterales en acción: $A_3$ dentro de $\mathfrak{S}_3$]\label{ex:b2:structures:cosets}
Tomemos $G = \mathfrak{S}_3$ (de \omterm{def:b2:structures:generated}{orden} $6$) y $H = A_3 =
\{\mathrm{id},\ (1\,2\,3),\ (1\,3\,2)\}$. Las clases laterales
izquierdas son
\[
H = \{\mathrm{id},\ (1\,2\,3),\ (1\,3\,2)\},
\qquad
(1\,2)H = \{(1\,2),\ (2\,3),\ (1\,3)\} :
\]
dos clases de tres elementos que reparten $G$, exactamente como
exige el recuento $\abs G = \abs H \times (\text{número de
clases})$, y visiblemente la partición en permutaciones pares e
impares. Nótese que $(1\,3)H = (1\,2)H$ aunque $(1\,3) \neq
(1\,2)$: las clases laterales son \emph{clases}, no van
etiquetadas por sus representantes, y $x^{-1}y \in H$ es la única
comparación legítima. Esta imagen de dos clases es la general para
la signatura: $A_n$ y su única clase acompañante parten
$\mathfrak{S}_n$ por la mitad, y así es como el problema de fin de
semana cuenta las posiciones alcanzables del rompecabezas.
\end{example}

\begin{example}\label{ex:b2:structures:lagrangeapp}
Dos dividendos inmediatos. \emph{Los grupos de \omterm{def:b2:structures:generated}{orden} primo son
\omterm{def:b2:structures:generated}{cíclicos}:} si $\abs G = p$ es primo y $a \neq e$, entonces
$\operatorname{ord}(a)$ divide a $p$ y no vale $1$, luego vale $p$:
$\langle a\rangle = G$. \emph{El retículo de subgrupos de
$\Z/12\Z$:} por la~\cref{prop:b2:structures:cyclic} de más abajo hay
exactamente un subgrupo por cada divisor de $12$ ---de órdenes $1, 2,
3, 4, 6, 12$, \omterm{def:b2:structures:generated}{generados} respectivamente por $\overline 0$,
$\overline 6$, $\overline 4$, $\overline 3$, $\overline 2$,
$\overline 1$---. La advertencia final: el \emph{recíproco} del
teorema de Lagrange es falso en general; $A_4$ tiene \omterm{def:b2:structures:generated}{orden} $12$ pero
ningún subgrupo de \omterm{def:b2:structures:generated}{orden} $6$, como demostramos en el problema de fin
de semana de este capítulo (\cref{pb:b2:structures:1}, pregunta 14).
Lagrange restringe los órdenes posibles; no los garantiza.
\end{example}

\begin{proposition}[Grupos cíclicos]\label{prop:b2:structures:cyclic}
Sea $G = \langle a \rangle$ \omterm{def:b2:structures:generated}{cíclico} de \omterm{def:b2:structures:generated}{orden} $n$.
\begin{enumerate}
  \item $G$ es isomorfo a $(\Z/n\Z, +)$, mediante $\overline k \mapsto
        a^k$.
  \item Todo subgrupo de $G$ es \omterm{def:b2:structures:generated}{cíclico}; para cada divisor $d \mid n$
        hay exactamente un subgrupo de \omterm{def:b2:structures:generated}{orden} $d$, a saber $\langle
        a^{n/d}\rangle$.
  \item $a^k$ genera $G$ si y solo si $\gcd(k, n) = 1$: $G$ tiene
        $\varphi(n)$ generadores (función de Euler\index{función
        indicatriz de Euler}).
\end{enumerate}
\end{proposition}

\begin{proof}
(1) La aplicación $k \mapsto a^k$ de $\Z$ sobre $G$ es compatible con
la congruencia módulo $n$ ($a^{k} = a^{k'} \iff n \mid k - k'$, por
la caracterización del \omterm{def:b2:structures:generated}{orden}); la propiedad universal
(\cref{def:b2:structures:quotient}) proporciona un morfismo biyectivo
bien definido desde $\Z/n\Z$.

(2) Sea $H \leq G$ no trivial y $m$ el menor entero $\geq 1$ con $a^m
\in H$. La división euclídea muestra que $H = \langle a^m\rangle$
(para $a^k \in H$: de $k = mq + r$ se sigue $a^r \in H$, luego $r =
0$), y $m \mid n$ (divídase $n$ entre $m$: $a^{n \bmod m} \in H$).
Entonces $\abs H = n/m$; tomando $m = n/d$ se realiza cada divisor
$d$. Unicidad: todo subgrupo de \omterm{def:b2:structures:generated}{orden} $d$ es, por lo anterior, de la
forma $\langle a^m \rangle$ con $n/m = d$; así pues $m = n/d$ queda
forzado y el subgrupo queda determinado.

(3) Afirmamos que $\operatorname{ord}(a^k) = \frac{n}{\gcd(k, n)}$.
Escribamos $d = \gcd(k, n)$. Para todo $m \geq 1$, la
caracterización del \omterm{def:b2:structures:generated}{orden} de la~\cref{def:b2:structures:generated} da
la cadena de equivalencias
\[
(a^k)^m = e
\iff n \mid km
\iff \frac{n}{d} \,\Big|\, \frac{k}{d}\,m
\iff \frac{n}{d} \,\Big|\, m ,
\]
donde el último paso es el lema de Gauss, ya que $\frac nd$ y
$\frac kd$ son primos entre sí. El menor $m$ así es $\frac nd$:
$\operatorname{ord}(a^k) = \frac n{\gcd(k,n)}$, que vale $n$ si y
solo si $\gcd(k, n) = 1$. Hay $\varphi(n)$ clases $k$ módulo $n$
en esas condiciones.
\end{proof}

\section{El grupo simétrico}

\begin{definition}\label{def:b2:structures:sn}
$\mathfrak{S}_n$ es el grupo de las permutaciones de $\intint{1}{n}$
(de \omterm{def:b2:structures:generated}{orden} $n!$). Un \emph{ciclo}\index{ciclo}
$(a_1\,a_2\,\cdots\,a_k)$ aplica $a_1 \mapsto a_2 \mapsto \dots
\mapsto a_k \mapsto a_1$ y deja fijo todo lo demás; $k$ es su
\emph{longitud}, y un ciclo de longitud $2$ es una
\emph{transposición}\index{transposición}. Dos ciclos son
\emph{disjuntos} cuando lo son sus soportes (los puntos no fijos).
\end{definition}

\begin{theorem}[Descomposición en ciclos]\label{thm:b2:structures:cycles}
Toda permutación $\sigma \neq \mathrm{id}$ es producto de
\omterm{def:b2:structures:sn}{ciclos} disjuntos dos a dos, de manera única salvo el orden de los
factores. Los \omterm{def:b2:structures:sn}{ciclos} disjuntos conmutan, y $\operatorname{ord}(\sigma)$
es el mcm de las longitudes.
\end{theorem}

\begin{proof}
Consideremos la relación de ``órbita'' sobre el soporte de $\sigma$:
$x \sim y$ si y solo si $y = \sigma^k(x)$ para algún $k \in \Z$; es
una relación de equivalencia. Cada clase $\{x, \sigma(x), \dots,
\sigma^{k-1}(x)\}$ (finita, de modo que los iterados
cierran el \omterm{def:b2:structures:sn}{ciclo}: la primera repetición ha de volver a $x$ por
inyectividad) lleva asociado el \omterm{def:b2:structures:sn}{ciclo} $(x\ \sigma(x)\ \cdots\
\sigma^{k-1}(x))$, y $\sigma$ es el producto de esos
\omterm{def:b2:structures:sn}{ciclos}: sobre cada órbita solo actúa el \omterm{def:b2:structures:sn}{ciclo} correspondiente.
Unicidad: toda factorización en \omterm{def:b2:structures:sn}{ciclos} disjuntos reproduce
exactamente las órbitas (el \omterm{def:b2:structures:sn}{ciclo} que pasa por $x$ ha de ser $(x\
\sigma(x)\ \cdots)$). Los \omterm{def:b2:structures:sn}{ciclos} disjuntos conmutan porque mueven
puntos disjuntos; el enunciado sobre el \omterm{def:b2:structures:generated}{orden} se sigue de que
$\sigma^m = \mathrm{id}$ si y solo si lo es la potencia $m$-ésima de
cada \omterm{def:b2:structures:sn}{ciclo} (por disjunción), si y solo si cada longitud divide a
$m$.
\end{proof}

\begin{example}[El tipo de ciclos como recuento]\label{ex:b2:structures:cycletype}
¿Cuántas permutaciones de $\mathfrak{S}_9$ tienen el tipo de
\omterm{def:b2:structures:sn}{ciclos} $(4, 3, 2)$ ---un \omterm{def:b2:structures:sn}{ciclo} de longitud $4$, uno de longitud
$3$ y una transposición---? Se eligen los soportes y los órdenes
\omterm{def:b2:structures:generated}{cíclicos}:
\[
\frac{9!}{4\cdot 3\cdot 2}
= \frac{362\,880}{24} = 15\,120 :
\]
se alinean los nueve símbolos en fila ($9!$ maneras), se agrupan
los cuatro primeros, los tres siguientes y los dos últimos en
\omterm{def:b2:structures:sn}{ciclos} y se divide por las rotaciones dentro de cada grupo
($4$, $3$ y $2$ respectivamente), que dan la misma permutación.
(Aquí las \emph{longitudes} de los \omterm{def:b2:structures:sn}{ciclos} son distintas, luego no
hay que dividir más; con longitudes iguales habría que dividir
además por las permutaciones de los grupos iguales.) Toda
permutación de este tipo tiene \omterm{def:b2:structures:generated}{orden} $\operatorname{lcm}(4,3,2) =
12$ y signatura $(-1)^3(-1)^2(-1)^1 = +1$
(\cref{thm:b2:structures:cycles} y el teorema de la signatura de
más abajo). Una partición de $9$, una clase de conjugación, un
recuento: la combinatoria de $\mathfrak{S}_n$ es la aritmética de
las particiones.
\end{example}

\begin{theorem}[Signatura]\label{thm:b2:structures:signature}
Existe exactamente un morfismo de grupos $\varepsilon \colon
\mathfrak{S}_n \to \{\pm 1\}$ (para $n \geq 2$) que vale $-1$ sobre
las \omterm{def:b2:structures:sn}{transposiciones}: la \emph{signatura}\index{signatura}. Además,
$\varepsilon(\sigma) = (-1)^{I(\sigma)}$, donde $I(\sigma)$ es el
número de \emph{inversiones} (pares $i < j$ con $\sigma(i) >
\sigma(j)$); un \omterm{def:b2:structures:sn}{ciclo} de longitud $k$ tiene signatura $(-1)^{k-1}$, y
el \emph{grupo alternado} $A_n = \ker\varepsilon$ tiene \omterm{def:b2:structures:generated}{orden}
$\frac{n!}{2}$.
\end{theorem}

\begin{proof}
\emph{Existencia.} Para $\sigma \in \mathfrak{S}_n$ pongamos
\[
\varepsilon(\sigma)
= \prod_{1 \leq i < j \leq n}
\frac{\sigma(j) - \sigma(i)}{j - i} .
\]
Los valores absolutos de los factores se multiplican para dar $1$
(los pares no ordenados $\{\sigma(i), \sigma(j)\}$ recorren todos los
pares), luego $\varepsilon(\sigma) = (-1)^{I(\sigma)} \in \{\pm1\}$.
Es morfismo: para $\sigma, \tau$,
\[
\varepsilon(\sigma\tau)
= \prod_{i<j} \frac{\sigma(\tau(j)) - \sigma(\tau(i))}{j - i}
= \prod_{i<j} \frac{\sigma(\tau(j)) - \sigma(\tau(i))}{\tau(j) -
\tau(i)} \cdot \prod_{i<j} \frac{\tau(j) - \tau(i)}{j - i}
= \varepsilon(\sigma)\,\varepsilon(\tau),
\]
donde el producto central vale $\varepsilon(\sigma)$ tras
reindexar por los pares $\{\tau(i), \tau(j)\}$ (cada par no
ordenado aparece una vez, y numerador y denominador cambian de
signo a la vez). Una \omterm{def:b2:structures:sn}{transposición} $\tau = (a\,b)$ con $a < b$
tiene un número impar de inversiones; contémoslas exactamente:
los pares invertidos $(i, j)$, con $i < j$ y $\tau(i) > \tau(j)$,
son
\[
(a, j) \ \text{con } a < j < b, \qquad
(i, b) \ \text{con } a < i < b, \qquad
(a, b) \ \text{mismo},
\]
es decir $(b - a - 1) + (b - a - 1) + 1 = 2(b - a) - 1$, un
número impar. (Alternativamente: compruébese $(1\,2)$
directamente, con una sola inversión, y conjúguese ---los
conjugados tienen la misma signatura, pues $\varepsilon$ es un
morfismo hacia un grupo abeliano---.) Por tanto
$\varepsilon((a\,b)) = (-1)^{2(b-a)-1} = -1$.

\emph{Unicidad.} Las \omterm{def:b2:structures:sn}{transposiciones} generan
$\mathfrak{S}_n$: todo \omterm{def:b2:structures:sn}{ciclo} cumple
\[
(a_1\cdots a_k) = (a_1\,a_k)(a_1\,a_{k-1})\cdots(a_1\,a_2),
\]
y el~\cref{thm:b2:structures:cycles} remata. Un morfismo hacia
$\{\pm1\}$ queda determinado por sus valores sobre los generadores.

\emph{Consecuencias.} La identidad anterior escribe un \omterm{def:b2:structures:sn}{ciclo} de
longitud $k$ como producto de $k - 1$ \omterm{def:b2:structures:sn}{transposiciones}: signatura
$(-1)^{k-1}$. En cuanto a $A_n$: el morfismo $\varepsilon$ es
sobreyectivo (hay \omterm{def:b2:structures:sn}{transposiciones} para $n \geq 2$), y las dos
``clases laterales'' $A_n$ y $(1\,2)A_n$ son \omterm{def:b2:structures:countable}{equipotentes} y parten
$\mathfrak{S}_n$ (argumento de Lagrange): $\abs{A_n} =
\frac{n!}{2}$.
\end{proof}

\begin{example}\label{ex:b2:structures:snexample}
$\sigma = \begin{pmatrix} 1&2&3&4&5&6\\ 3&6&5&4&1&2 \end{pmatrix}
= (1\,3\,5)(2\,6)$: \omterm{def:b2:structures:generated}{orden} $\operatorname{lcm}(3,2) = 6$, signatura
$(-1)^{2}\cdot(-1)^{1} = -1$. La signatura es la comprobación de
paridad más rápida sobre barajaduras, y el motor del determinante
en el~\cref{ch:b2:linalg}.
\end{example}

\begin{example}[Tres caminos hacia un mismo signo]\label{ex:b2:structures:threeroads}
Sea $\sigma \in \mathfrak{S}_5$ la permutación que envía $1, 2, 3,
4, 5$ a $3, 5, 4, 1, 2$. \emph{Por \omterm{def:b2:structures:sn}{ciclos}:} $1 \mapsto 3 \mapsto
4 \mapsto 1$ y $2 \mapsto 5 \mapsto 2$, luego $\sigma =
(1\,3\,4)(2\,5)$ y $\varepsilon(\sigma) = (-1)^{2}(-1)^{1} = -1$.
\emph{Por inversiones:} en la lista de valores $3, 5, 4, 1, 2$ los
pares desordenados son $(3,1)$, $(3,2)$, $(5,4)$, $(5,1)$,
$(5,2)$, $(4,1)$, $(4,2)$: siete, y $(-1)^7 = -1$. \emph{Por
\omterm{def:b2:structures:sn}{transposiciones}:} $\sigma = (1\,4)(1\,3)(2\,5)$, tres factores,
$(-1)^3 = -1$. Tres cálculos, una misma paridad: la unicidad del~\cref{thm:b2:structures:signature} garantiza que ningún sistema de
recuento puede hacerlos discrepar, y eso es exactamente lo que
convierte a $\varepsilon$ en un invariante utilizable (véase el
problema de fin de semana).
\end{example}

\begin{remark}[Adónde va la signatura a partir de aquí]
La signatura es la semilla de tres cosechas posteriores: construye
el determinante y su regla del producto en el~\cref{ch:b2:linalg};
alimenta invariantes de paridad para rompecabezas combinatorios (el
problema de fin de semana de este capítulo resuelve con ella el
juego del quince); y los grupos alternados $A_n$ que define pasan a
ocupar un lugar central en el volumen del tercer año, donde su
simplicidad para $n \geq 5$ explica que las ecuaciones de grado $5$
no se resuelvan por radicales.
\end{remark}

\section{Anillos, ideales, cocientes}

\begin{definition}[Ideal]\label{def:b2:structures:ideal}
Sea $A$ un anillo conmutativo. Un \emph{ideal}\index{ideal} $I
\subseteq A$ es un subgrupo aditivo tal que $a x \in I$ para todos
$a \in A$, $x \in I$. Los núcleos de morfismos de anillos son
ideales; $I = A$ si y solo si $1 \in I$, si y solo si $I$ contiene
una unidad. El ideal \emph{\omterm{def:b2:structures:generated}{generado}} por $x$ es $xA = \{xa\}$ (un
ideal \emph{principal}).
\end{definition}

\begin{theorem}[{Ideales de $\Z$ y de $K[X]$}]\label{thm:b2:structures:principal}
Todo \omterm{def:b2:structures:ideal}{ideal} de $\Z$ es de la forma $n\Z$ para un único $n \in \N$;
todo \omterm{def:b2:structures:ideal}{ideal} de $K[X]$ ($K$ un cuerpo) es de la forma $P\,K[X]$ para un
único $P$ mónico (o nulo). En consecuencia, existen máximos comunes
divisores en ambos anillos, con relaciones de Bézout: $x\Z + y\Z =
\gcd(x,y)\Z$, y análogamente para polinomios.
\end{theorem}

\begin{proof}
Para $\Z$ esto era el teorema sobre subgrupos del volumen del primer
año (un \omterm{def:b2:structures:ideal}{ideal} es en particular un subgrupo, y $n\Z$ es un \omterm{def:b2:structures:ideal}{ideal}).
Para $K[X]$: sea $I \neq \{0\}$ un \omterm{def:b2:structures:ideal}{ideal} y $P \in I$ no nulo de grado
mínimo, normalizado mónico. Para $F \in I$, la división euclídea $F =
PQ + R$ da $R = F - PQ \in I$ con $\deg R < \deg P$: la minimalidad
obliga a $R = 0$, luego $I = P\,K[X]$. Unicidad: dos generadores
mónicos se dividen mutuamente. Los enunciados de Bézout expresan la
igualdad del \omterm{def:b2:structures:ideal}{ideal} $x\Z + y\Z$ (o de su análogo polinómico) con el
\omterm{def:b2:structures:ideal}{ideal} principal del mcd ---la propia definición de mcd usada en el
primer año, reconocida ahora como un enunciado sobre ideales---.
\end{proof}

\begin{example}[Un mcd de polinomios, de dos maneras]\label{ex:b2:structures:polygcd}
Calculemos $\gcd(X^3 - 1,\ X^2 - 1)$ en $\Q[X]$. \emph{Por
Euclides:}
\[
X^3 - 1 = X\,(X^2 - 1) + (X - 1),
\qquad
X^2 - 1 = (X + 1)(X - 1) + 0 ,
\]
luego el mcd es $X - 1$, y remontando la división se obtiene la
relación de Bézout
\[
X - 1 = 1\cdot(X^3 - 1) - X\cdot(X^2 - 1).
\]
\emph{Por \omterm{def:b2:structures:ideal}{ideales}:} el \omterm{def:b2:structures:ideal}{ideal} $(X^3 - 1)\Q[X] + (X^2 - 1)\Q[X]$
es principal (\cref{thm:b2:structures:principal}); contiene a $X -
1$ (por la fórmula anterior) y está contenido en $(X - 1)\Q[X]$
(ambos generadores se anulan en $1$, luego son múltiplos de $X -
1$): el generador mónico es $X - 1$. La moraleja: el punto de
vista de los \omterm{def:b2:structures:ideal}{ideales} identifica el mcd \emph{sin dividir} ---las
raíces comunes localizan el \omterm{def:b2:structures:ideal}{ideal} y Euclides se limita a
certificarlo---.
\end{example}

\begin{definition}[El anillo cociente $\Z/n\Z$, revisitado]\label{def:b2:structures:quotientring}
Para un \omterm{def:b2:structures:ideal}{ideal} $I$ de $A$, la relación $x \sim y \iff x - y \in I$ es
una equivalencia compatible con $+$ y $\times$; el \omterm{def:b2:structures:quotient}{conjunto cociente}
$A/I$ hereda una estructura de anillo ---el \emph{anillo
cociente}\index{anillo cociente}--- que hace de $\pi \colon A \to
A/I$ un morfismo de núcleo $I$. Para $A = \Z$ e $I = n\Z$ esto es el
$\Z/n\Z$ del volumen del primer año, ahora con su propiedad
universal: todo morfismo que anula $I$ factoriza a través de $A/I$.
\end{definition}

\begin{theorem}[Teorema chino del resto, forma anular]\label{thm:b2:structures:crt}
Si $\gcd(m, n) = 1$, la aplicación
\[
\Z/mn\Z \longrightarrow \Z/m\Z \times \Z/n\Z,
\qquad
\overline{x} \longmapsto (x \bmod m,\; x \bmod n)
\]
es un isomorfismo de anillos.\index{teorema chino del resto} En
consecuencia, $\varphi(mn) = \varphi(m)\varphi(n)$ para $m, n$
primos entre sí, y
\[
\varphi(n) = n \prod_{p \mid n} \Bigl(1 - \frac 1p\Bigr)
\quad (p \text{ primo}).
\]
\end{theorem}

\begin{proof}
La aplicación es un morfismo de anillos bien definido (las
compatibilidades son inmediatas). Inyectividad: si $x \equiv 0$
módulo $m$ y módulo $n$ con $\gcd(m,n) = 1$, entonces $mn \mid x$
(Gauss). Sobreyectividad: ambos lados tienen $mn$ elementos, así que
basta la inyectividad (cardinales finitos iguales); o
explícitamente, a partir de una relación de Bézout $um + vn = 1$, la
clase de
\[
x = b\,um + a\,vn
\]
se aplica en $(a \bmod m,\ b \bmod n)$, pues $vn = 1 - um \equiv
1 \pmod m$ hace que $x \equiv a \pmod m$, y simétricamente módulo
$n$ ---la receta usada numéricamente en el~\cref{ex:b2:structures:crtinverse}---. Las unidades se corresponden
con los pares de unidades (las unidades de un anillo producto son
los pares de unidades), luego $\varphi(mn) = \varphi(m)\varphi(n)$.
Para una potencia de primo, $\varphi(p^k) = p^k - p^{k-1}$ (los no
invertibles módulo $p^k$ son los múltiplos de $p$); la
multiplicatividad ensambla la fórmula del producto.
\end{proof}

\begin{example}[Inversión del isomorfismo chino]\label{ex:b2:structures:crtinverse}
Tomemos $m = 8$, $n = 9$. La inversa del isomorfismo se hace
explícita mediante dos \emph{idempotentes}: buscamos $u \equiv 1
\pmod 8$, $u \equiv 0 \pmod 9$ y $v \equiv 0 \pmod 8$, $v \equiv 1
\pmod 9$. De $u = 9k \equiv 1 \pmod 8$ resulta $k \equiv 1$, luego
$u = 9$; de $v = 8k \equiv 1 \pmod 9$ resulta $-k \equiv 1$, $k
\equiv 8$, luego $v = 64$. Entonces la clase de $x = 9a + 64b$
módulo $72$ es la única solución de $x \equiv a \pmod 8$, $x
\equiv b \pmod 9$: para $a = 3$, $b = 5$ se obtiene $27 + 320 =
347 \equiv 59 \pmod{72}$, exactamente el valor intermedio hallado
por sustitución en el~\cref{exo:b2:structures:8}. La moraleja: $u$
y $v$ cumplen $u + v \equiv 1$, $uv \equiv 0$, $u^2 \equiv u$,
$v^2 \equiv v$ módulo $72$; son las imágenes de $(1, 0)$ y $(0,
1)$, y toda descomposición china es, en el fondo, una
descomposición de $1$ en idempotentes ortogonales.
\end{example}

\begin{theorem}[Euler; Fermat revisitado]\label{thm:b2:structures:euler}
Las unidades de $\Z/n\Z$ forman un grupo de \omterm{def:b2:structures:generated}{orden} $\varphi(n)$; por
tanto, si $\gcd(a, n) = 1$,
\[
a^{\varphi(n)} \equiv 1 \pmod n
\qquad (\text{teorema de Euler\index{teorema de Euler}}),
\]
y el pequeño teorema de Fermat es el caso $n = p$ primo, ahora a una
línea de Lagrange.
\end{theorem}

\begin{proof}
Las clases invertibles son exactamente las de los enteros primos con
$n$ (volumen del primer año): hay $\varphi(n)$, y forman un grupo
para la multiplicación. Por Lagrange
(\cref{thm:b2:structures:lagrange}), todo elemento elevado al \omterm{def:b2:structures:generated}{orden}
del grupo da el neutro.
\end{proof}

\begin{example}[Un grupo de unidades sin generador]\label{ex:b2:structures:units15}
El grupo $(\Z/15\Z)^*$ tiene $\varphi(15) = \varphi(3)\varphi(5)
= 8$ elementos. ¿Es \omterm{def:b2:structures:generated}{cíclico}? Calculemos órdenes con el
isomorfismo chino $(\Z/15\Z)^* \simeq (\Z/3\Z)^* \times
(\Z/5\Z)^*$ (una unidad módulo $15$ es un par de unidades): los
factores tienen órdenes $2$ y $4$, luego el \omterm{def:b2:structures:generated}{orden} de todo elemento
divide a $\operatorname{lcm}(2, 4) = 4 < 8$; ninguno genera. En
concreto:
\[
2^4 = 16 \equiv 1, \qquad
4^2 = 16 \equiv 1, \qquad
7^4 \equiv 1, \qquad
11^2 = 121 \equiv 1, \qquad
14^2 \equiv 1 \pmod{15} :
\]
órdenes $4, 2, 4, 2, 2$ y nunca $8$. Compárese con el~\cref{exo:b2:structures:10}: $(\Z/p\Z)^*$ \emph{sí} es \omterm{def:b2:structures:generated}{cíclico} para
$p$ primo, porque allí el grupo de unidades vive dentro de un
cuerpo. El teorema de Euler sigue valiendo con exponente
$\varphi(15) = 8$, pero el verdadero exponente universal aquí es
$4$: Euler da una cota superior, no siempre la óptima.
\end{example}

\begin{definition}[Álgebra]\label{def:b2:structures:algebra}
Una \emph{$K$-álgebra}\index{álgebra} es un $K$-espacio vectorial $A$
dotado de una estructura de anillo cuya multiplicación es
$K$-bilineal. Ejemplos: $K[X]$, $\mathcal{M}_n(K)$, $\mathcal{L}(E)$,
los espacios de funciones $\mathcal{F}(X, K)$, $\C$ como
$\R$-álgebra. Los morfismos de álgebras son los morfismos de anillos
lineales; la \emph{evaluación} $P \mapsto P(u)$ de $K[X]$ en
$\mathcal{L}(E)$ (o en $\mathcal{M}_n(K)$) es el ejemplo central, y
es el motor del~\cref{ch:b2:reduction}.
\end{definition}

\begin{example}[Un morfismo de evaluación y su núcleo]\label{ex:b2:structures:evaluation}
Tomemos $A = \begin{pmatrix}0 & 1\\ 0 & 0\end{pmatrix}$ y la
evaluación $\varepsilon_A \colon \R[X] \to \mathcal{M}_2(\R)$, $P
\mapsto P(A)$. Como $A^2 = 0$,
\[
P(A) = P(0)\,I + P'(0)\,A =
\begin{pmatrix} P(0) & P'(0)\\ 0 & P(0)\end{pmatrix},
\]
(solo sobreviven los términos constante y lineal de $P$). Por
tanto $\ker\varepsilon_A = \{P : P(0) = P'(0) = 0\} = X^2\,\R[X]$:
un \omterm{def:b2:structures:ideal}{ideal} principal, exactamente como predice el~\cref{thm:b2:structures:principal}, \omterm{def:b2:structures:generated}{generado} por el polinomio mónico
$X^2$ de menor grado del núcleo ---el \emph{polinomio mínimo} de
$A$, protagonista del~\cref{ch:b2:reduction}---. La imagen es el
\omterm{def:b2:structures:algebra}{álgebra} conmutativa de dimensión dos $\{aI + bA\}$: los
morfismos de evaluación comprimen el $\R[X]$ de dimensión infinita
sobre \omterm{def:b2:structures:algebra}{álgebras} pequeñas y calculables.
\end{example}

\begin{remark}[Perspectivas: tres melodías que conviene escuchar]\label{rem:b2:structures:perspectives}
Tres ideas estructurales de este capítulo reaparecen a lo largo del
volumen, cada vez con orquestación más densa. \emph{La factorización
a través de un cociente} (\cref{def:b2:structures:quotient}):
construye aquí $\Z/n\Z$, define aplicaciones sobre los conjuntos de
soluciones de sistemas lineales en el~\cref{ch:b2:linalg} y subyace
en silencio a todo argumento del tipo ``está bien definido sobre las
clases''. \emph{Los invariantes}: la signatura es un morfismo hacia
$\{\pm1\}$ que ningún movimiento legal puede esquivar; la misma
lógica da la regla del producto del determinante
(\cref{ch:b2:linalg}), la invariancia de la traza por semejanza y
las cantidades conservadas del~\cref{ch:b2:diffeq}. \emph{El
recuento contra una estructura}: Lagrange cuenta mediante clases
laterales, la dimensión cuenta mediante bases (\cref{ch:b2:linalg}),
la multiplicidad cuenta mediante grados de polinomios
(\cref{ch:b2:reduction}); siempre que una cota parezca milagrosa,
alguna partición o graduación está haciendo el recuento.
\end{remark}

\begin{remark}[Errores frecuentes]\label{rem:b2:structures:pitfalls}
Cuatro clásicos. (i) Una aplicación definida sobre un cociente debe
comprobarse \emph{bien definida}: ``$\overline x \mapsto$ (fórmula
en $x$)'' solo es legítimo si la fórmula es constante sobre las
clases ---la compatibilidad de la~\cref{def:b2:structures:quotient}, y no un mero formalismo---. (ii)
La igualdad $\operatorname{ord}(ab) =
\operatorname{lcm}(\operatorname{ord}a, \operatorname{ord}b)$ es
\emph{falsa} en general, incluso para elementos que conmutan ($a$ y
$a^{-1}$); el~\cref{exo:b2:structures:4} da el enunciado correcto
para órdenes primos entre sí, y los \omterm{def:b2:structures:sn}{ciclos} disjuntos la versión
correcta para permutaciones. (iii) La \omterm{def:b2:structures:countable}{numerabilidad} se conserva por
\emph{uniones} \omterm{def:b2:structures:countable}{numerables} y por \emph{productos} finitos, pero no por
productos \omterm{def:b2:structures:countable}{numerables}: $\{0,1\}^{\N}$ no es \omterm{def:b2:structures:countable}{numerable}
(\cref{exo:b2:structures:3}) aunque cada factor tenga dos
elementos. (iv) Cantor--Bernstein solo necesita inyecciones en
ambos sentidos, pero la biyección que construye suele ser
discontinua y no explícita: no hay que esperar una fórmula
(\cref{ex:b2:structures:cbexample}).
\end{remark}

\begin{remark}[Dónde se usa este capítulo]
Casi en todas partes. La signatura construye los determinantes
(\cref{ch:b2:linalg}); el morfismo de evaluación $P \mapsto P(u)$ y
los \omterm{def:b2:structures:ideal}{ideales} principales de $K[X]$ producen los polinomios mínimos y
las descomposiciones en núcleos del~\cref{ch:b2:reduction}; la
\omterm{def:b2:structures:countable}{numerabilidad} es el escenario sobre el que actúa el~\cref{ch:b2:proba} (probabilidad sobre espacios \omterm{def:b2:structures:countable}{numerables}) y la razón
de que la topología no deje de producir conjuntos densos \omterm{def:b2:structures:countable}{numerables}
(\cref{ch:b2:metric}). La construcción del cociente $A/I$ se
redespliega en el volumen del tercer año para construir cuerpos
$K[X]/(P)$ y, a partir de ellos, la teoría de Galois: la propiedad
universal demostrada aquí se usa allí palabra por palabra.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:structures:1}
¿Cuáles de los siguientes conjuntos son \omterm{def:b2:structures:countable}{numerables}? El conjunto de
los subconjuntos finitos de $\N$; el conjunto de \emph{todos} los
subconjuntos de $\N$; $\R \setminus \Q$; el conjunto de los
polinomios con coeficientes racionales; el conjunto de las
sucesiones de $0$ y de $1$ que son nulas a partir de un cierto
índice.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:structures:2}
En $\mathfrak{S}_7$, sean $\sigma = (1\,4\,2\,6)(3\,5)$ y $\tau =
(2\,3\,7)$. Calcula $\sigma\tau$ y $\tau\sigma$ en forma de
\omterm{def:b2:structures:sn}{ciclos} disjuntos, los órdenes y las signaturas de las cuatro
permutaciones, y $\sigma^{2026}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:structures:3}
Construye inyecciones explícitas que muestren que $\mathcal{P}(\N)$,
$\intcc{0}{1}$ y el conjunto $\{0,1\}^{\N}$ de las sucesiones
binarias son \omterm{def:b2:structures:countable}{equipotentes} dos a dos \emph{(desarrollos binarios en
ambos sentidos; Cantor--Bernstein absorbe la molestia de la doble
representación)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:structures:4}
Sea $G$ un grupo y $a, b \in G$ dos elementos que conmutan, de
órdenes finitos $m$ y $n$ primos entre sí. Demuestra que
$\operatorname{ord}(ab) = mn$. Muestra con un ejemplo en
$\mathfrak{S}_3$ que la conmutación es esencial.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:structures:5}
Sea $G$ un grupo finito de \omterm{def:b2:structures:generated}{orden} par. Demuestra que $G$ contiene un
elemento de \omterm{def:b2:structures:generated}{orden} $2$. \emph{(Emparéjese cada elemento con su
inverso y cuéntense los que quedan emparejados consigo mismos.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:structures:6}
Demuestra que $A_n$ ($n \geq 3$) está \omterm{def:b2:structures:generated}{generado} por los \omterm{def:b2:structures:sn}{ciclos} de
longitud $3$. \emph{(Un producto de dos \omterm{def:b2:structures:sn}{transposiciones} es un \omterm{def:b2:structures:sn}{ciclo}
de longitud $3$ o un producto de dos \omterm{def:b2:structures:sn}{ciclos} de longitud $3$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:structures:7}
Determina todos los morfismos de grupos: de $(\Q, +)$ en $(\Z, +)$;
de $(\Z/n\Z, +)$ en $(\Z/m\Z, +)$ \emph{(cuéntalos: hay
$\gcd(m,n)$)}; de $(\Q, +)$ en $(\Q_+^*, \times)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:structures:8}
Usando el teorema chino del resto, calcula $\varphi(360)$, halla
todos los $x$ con $x \equiv 3 \pmod 8$, $x \equiv 5 \pmod 9$ y $x
\equiv 2 \pmod 5$, y calcula las dos últimas cifras de $3^{2026}$
\emph{(Euler módulo $100$; atención: trabaja módulo $4$ y módulo
$25$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:structures:9}
Demuestra que todo dominio de integridad finito es un cuerpo. Deduce
que $\Z/n\Z$ es un cuerpo si y solo si $n$ es primo (de nuevo).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:structures:10}
(Un clásico) Sea $K$ un cuerpo y $G$ un subgrupo \emph{finito} de
$(K^*, \times)$. Demuestra que $G$ es \omterm{def:b2:structures:generated}{cíclico}.
\emph{Indicación: sea $m$ el \omterm{def:b2:structures:generated}{orden} máximo entre los elementos de
$G$; prueba que el \omterm{def:b2:structures:generated}{orden} de todo elemento divide a $m$ (aplicando el~\cref{exo:b2:structures:4} a partes coprimas adecuadas), de modo que
todo $G$ satisface $x^m = 1$; cuenta después las raíces de $X^m -
1$.} En particular, $(\Z/p\Z)^*$ es \omterm{def:b2:structures:generated}{cíclico}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:structures:11}
Demuestra que el grupo $(\Q, +)$ no es \omterm{def:b2:structures:generated}{cíclico} y, peor aún, que ni
siquiera es finitamente \omterm{def:b2:structures:generated}{generado}. Demuestra en cambio que todo
subgrupo finitamente \omterm{def:b2:structures:generated}{generado} de $(\Q, +)$ es \omterm{def:b2:structures:generated}{cíclico}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:structures:12}
(Criterio de Dedekind) Demuestra que todo conjunto infinito contiene
un subconjunto \omterm{def:b2:structures:countable}{numerable} y deduce que un conjunto $E$ es infinito si y
solo si es \omterm{def:b2:structures:countable}{equipotente} a un subconjunto propio de sí mismo. \emph{(Para
la implicación directa, desplaza un subconjunto \omterm{def:b2:structures:countable}{numerable} un paso;
para la recíproca, recuerda el principio del palomar.)}
\end{exercise}

\section{Problema: el juego del quince}

El juego del quince es una bandeja de $4 \times 4$ con quince fichas
deslizantes numeradas del $1$ al $15$ y una casilla vacía; un
movimiento desliza a la casilla vacía una de las fichas contiguas a
ella. En la década de 1890 Sam Loyd popularizó el juego ofreciendo
1000 dólares a quien lograra intercambiar las fichas $14$ y $15$
devolviendo todas las demás a su sitio. Nadie cobró nunca, y este
problema de fin de semana demuestra las dos mitades de la razón: la
signatura del~\cref{thm:b2:structures:signature} prohíbe el
intercambio de Loyd y ---la mitad más difícil, la constructiva---
\emph{todo} lo que la signatura permite es de verdad realizable. El
enunciado completo es el teorema de Johnson--Story (1879).

\begin{problem}[{Problema de fin de semana --- el teorema de
resolubilidad de Johnson--Story}]
\label{pb:b2:structures:1}
Numeremos las casillas del $1$ al $16$ en orden de lectura (de
izquierda a derecha y de arriba abajo), de modo que la casilla $k$
ocupe la fila $i$ y la columna $j$ con $k = 4(i - 1) + j$. La
casilla $16$ (abajo a la derecha) es la \emph{casa} de la casilla
vacía; trataremos la casilla vacía como una decimosexta ficha,
escrita $b$ e identificada con el número $16$. Una
\emph{configuración} es una biyección $\sigma \colon \intint1{16}
\to \intint1{16}$, casilla $\mapsto$ contenido; la configuración
\emph{resuelta} es $\sigma = \mathrm{id}$. En todo el problema,
$\varepsilon$ es la signatura del~\cref{thm:b2:structures:signature} y dos casillas son
\emph{contiguas} cuando comparten un lado de la bandeja.

\medskip
\noindent\textbf{Parte I --- Configuraciones, movimientos,
signaturas.}
\begin{enumerate}
  \item Justifica que las configuraciones son exactamente los
        elementos de $\mathfrak{S}_{16}$, de modo que hay $16! =
        20\,922\,789\,888\,000$, y que el número de movimientos
        legales desde una configuración dada es $2$, $3$ o $4$,
        según que la casilla vacía esté en una esquina, en un
        borde o en el interior.
  \item Sea $\sigma$ una configuración, $p = \sigma^{-1}(16)$ la
        casilla del hueco y $c$ una casilla contigua a $p$. Prueba
        que deslizar la ficha de $c$ hasta $p$ produce la
        configuración $\sigma' = \sigma \circ \tau$ con $\tau =
        (p\ c)$, y deduce que todo movimiento cambia el signo de
        la signatura: $\varepsilon(\sigma') =
        -\varepsilon(\sigma)$.
  \item Colorea la bandeja como un tablero de ajedrez: $\chi(k) =
        (-1)^{i+j}$ para la casilla $k$ situada en la fila $i$ y
        la columna $j$. Prueba que todo movimiento cambia el signo
        de $\chi(\text{casilla del hueco})$ y deduce que una
        sucesión de movimientos que devuelve el hueco a su casilla
        de partida tiene longitud par.
  \item Prueba que
        \[
        I(\sigma) = \varepsilon(\sigma)\,
        \chi\bigl(\sigma^{-1}(16)\bigr)
        \]
        es invariante por todo movimiento legal, y calcula
        $I(\mathrm{id})$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El premio de Loyd: el invariante en
acción.}
\begin{enumerate}[resume]
  \item La configuración $\sigma_L$ de Loyd coincide con la
        resuelta salvo que las casillas $14$ y $15$ contienen las
        fichas $15$ y $14$. Calcula $I(\sigma_L)$ y concluye que
        ninguna sucesión de movimientos une $\sigma_L$ con la
        configuración resuelta: los 1000 dólares de Loyd nunca
        corrieron peligro.
  \item Prueba que exactamente la mitad de las configuraciones
        cumplen $I = +1$: $\abs{\{\sigma : I(\sigma) = +1\}} =
        16!/2$. \emph{(Para una casilla vacía fija, emparéjense
        las configuraciones componiendo con una \omterm{def:b2:structures:sn}{transposición}
        fija de otras dos casillas.)}
  \item Prueba que todo movimiento se deshace mediante un
        movimiento legal, que ``$\sigma'$ es alcanzable desde
        $\sigma$ mediante movimientos legales'' es una relación de
        equivalencia, y que la clase $R$ de la configuración
        resuelta cumple $R \subseteq \{I = +1\}$. Concluye que hay
        al menos dos clases.
  \item Supongamos que el hueco está en casa: $\sigma(16) = 16$.
        Prueba que $I(\sigma) = \varepsilon(\rho)$, donde $\rho
        \in \mathfrak{S}_{15}$ es la restricción de $\sigma$ a las
        casillas $1, \dots, 15$, y que toda configuración puede
        llevarse mediante movimientos legales a otra con el hueco
        en casa. Concluye: para demostrar que $R = \{I = +1\}$
        basta con realizar toda permutación \emph{par} de las
        quince casillas distintas de la casa mediante una sucesión
        de movimientos que empiece y termine con el hueco en casa.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Recorridos del hueco y grupo de
programas.} Un \emph{programa} es una sucesión finita de
movimientos legales, iniciada en una configuración con el hueco en
casa, cuya configuración final tiene también el hueco en casa. Su
\emph{efecto} es la permutación $\pi$ de las casillas definida
por: el contenido de la casilla $x$ acaba en la casilla $\pi(x)$.
\begin{enumerate}[resume]
  \item Prueba que un programa ejecutado desde $\sigma$ termina en
        $\sigma \circ \pi^{-1}$; que ejecutar dos programas
        seguidos compone sus efectos; y que el conjunto $H$ de
        todos los efectos es un subgrupo de $\mathfrak{S}_{15}$
        (permutaciones de las casillas $1, \dots, 15$) contenido
        en el grupo alternado $A_{15}$.
  \item (El recorrido elemental) Con el hueco en casa, hazlo
        circular por el bloque $2 \times 2$ inferior derecho:
        casillas $16 \to 12 \to 11 \to 15 \to 16$. Prueba que el
        efecto es el \omterm{def:b2:structures:sn}{ciclo} $(11\ 12\ 15)$ y que el recorrido
        inverso da $(11\ 15\ 12)$. Ambos pertenecen a $H$.
  \item (El gran recorrido) Comprueba que
        \[
        16 \to 15 \to 14 \to 13 \to 9 \to 5 \to 1 \to 2 \to 3
        \to 4 \to 8 \to 7 \to 6 \to 10 \to 11 \to 12 \to 16
        \]
        es un camino cerrado que pasa por las dieciséis casillas
        (con pasos entre casillas contiguas únicamente), y que su
        efecto es el \omterm{def:b2:structures:sn}{ciclo} de longitud $15$
        \[
        \zeta = (15\ 12\ 11\ 10\ 6\ 7\ 8\ 4\ 3\ 2\ 1\ 5\ 9\ 13\
        14) .
        \]
        Escribiendo $x_0 = 15$, $x_1 = 12$, $x_2 = 11$, \dots,
        $x_{14} = 14$ para su \omterm{def:b2:structures:generated}{orden} \omterm{def:b2:structures:generated}{cíclico}, comprueba que el
        recorrido elemental inverso de la pregunta 10 es
        exactamente $(x_0\ x_1\ x_2)$.
  \item Demuestra la fórmula de conjugación en cualquier
        $\mathfrak{S}_n$: para una permutación $g$ y un \omterm{def:b2:structures:sn}{ciclo} de
        longitud $3$,
        \[
        g\,(a\ b\ c)\,g^{-1} = \bigl(g(a)\ g(b)\ g(c)\bigr),
        \]
        y observa que $H$, por ser un grupo, es estable por
        conjugación por sus propios elementos.
  \item Deduce que $H$ contiene los quince \omterm{def:b2:structures:sn}{ciclos} de longitud $3$
        \emph{consecutivos} del gran recorrido:
        \[
        s_t = (x_t\ x_{t+1}\ x_{t+2}) \qquad (t \in \Z/15\Z,
        \text{ índices módulo } 15).
        \]
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Generación del grupo alternado.}
\begin{enumerate}[resume]
  \item (Lema A) Sean $s$ y $t$ dos \omterm{def:b2:structures:sn}{ciclos} de longitud $3$ cuyos
        soportes comparten exactamente dos puntos, digamos $\{a,
        b, c\}$ y $\{b, c, d\}$. Prueba que, tras sustituir $s$ o
        $t$ por su inverso si hace falta (lo cual no cambia el
        subgrupo \omterm{def:b2:structures:generated}{generado}), el producto $st$ es una doble
        \omterm{def:b2:structures:sn}{transposición}; prueba que $A_4$ no contiene ningún
        subgrupo de \omterm{def:b2:structures:generated}{orden} $6$ \emph{(un subgrupo de índice $2$
        contiene todos los cuadrados; cuenta los \omterm{def:b2:structures:sn}{ciclos} de
        longitud $3$ que son cuadrados)}; y concluye que $\langle
        s, t\rangle$ es todo el grupo alternado de las cuatro
        letras $\{a, b, c, d\}$.
  \item (Lema B) Sea $X$ un conjunto de $k \geq 4$ letras, $w
        \notin X$, y sea $G$ un subgrupo de algún
        $\mathfrak{S}_n$ que contiene todas las permutaciones
        pares de $X$ y un \omterm{def:b2:structures:sn}{ciclo} de longitud $3$ de la forma $(u\
        v\ w)$ con $u, v \in X$. Prueba que para todos $a, b \in
        X$ distintos existe una permutación \emph{par} $g$ de $X$
        con $g(u) = a$, $g(v) = b$, y deduce que $(a\ b\ w) \in
        G$.
  \item Deduce que el grupo $G$ del lema B contiene todas las
        permutaciones pares de $X \cup \{w\}$ \emph{(usa el~\cref{exo:b2:structures:6}: los \omterm{def:b2:structures:sn}{ciclos} de longitud $3$
        generan)}. Después, encadenando los lemas A y B a lo largo
        de los \omterm{def:b2:structures:sn}{ciclos} consecutivos $s_0, s_1, \dots, s_{12}$ de la
        pregunta 13, demuestra que $\langle s_0, \dots,
        s_{12}\rangle = A_{15}$.
  \item Concluye que $H = A_{15}$: \emph{toda reordenación par de
        las quince fichas es realizable mediante un programa}, y
        $H$ tiene $15!/2 = 653\,837\,184\,000$ elementos.
  \item (El teorema de Johnson--Story, 1879) Ensambla las
        preguntas 6, 7, 8 y 17: las configuraciones alcanzables
        desde la resuelta son \emph{exactamente} las $16!/2 =
        10\,461\,394\,944\,000$ configuraciones con $I = +1$; y la
        alcanzabilidad tiene exactamente \emph{dos} clases, la de
        la configuración resuelta y la de la $\sigma_L$ de Loyd.
        \emph{(Para el segundo punto, reetiqueta las fichas $14$ y
        $15$: prueba que $\sigma \mapsto (14\ 15) \circ \sigma$
        transforma sucesiones de movimientos en sucesiones de
        movimientos e intercambia $\{I = +1\}$ con $\{I = -1\}$.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Criterios, variantes y la vista desde
arriba.}
\begin{enumerate}[resume]
  \item (El criterio práctico) Lee las quince fichas en el orden
        de lectura de sus casillas, saltándote el hueco, y sea $N$
        el número de inversiones de esa lista; sea $r$ la fila del
        hueco contada desde \emph{abajo}. Prueba que $I(\sigma) =
        (-1)^{N + r + 1}$, de modo que $\sigma$ es resoluble si y
        solo si $N + r$ es impar.
  \item (Acciones de grupo) Una \emph{acción} de un grupo $G$
        sobre un conjunto $X$ es una aplicación $G \times X \to
        X$, $(g, x) \mapsto g \cdot x$, con $e \cdot x = x$ y $g
        \cdot (h \cdot x) = (gh) \cdot x$; la \emph{órbita} de $x$
        es $G \cdot x$, y la acción es \emph{libre} cuando $g
        \cdot x = x$ obliga a $g = e$. Prueba que $h \cdot \sigma
        = \sigma \circ h^{-1}$ define una acción libre de $H$
        sobre el conjunto de configuraciones con el hueco en casa,
        que sus órbitas son exactamente las clases de
        alcanzabilidad mutua mediante programas, y recupera del
        recuento de órbitas que esas configuraciones se reparten
        en exactamente $15!\,/\,\abs H = 2$ clases.
  \item (La obstrucción $3 \times 3$) Prueba que el tablero de $3
        \times 3$ \emph{no} admite ningún camino cerrado que visite
        cada casilla exactamente una vez: la estrategia del gran
        recorrido de la parte III falla para el juego del ocho.
        \emph{(Colorea las nueve casillas como un tablero de
        ajedrez.)}
  \item (La reparación) En el tablero de $3 \times 3$ con casillas
        del $1$ al $9$ en orden de lectura y casa $9$: calcula los
        efectos del recorrido por el perímetro $9 \to 8 \to 7 \to
        4 \to 1 \to 2 \to 3 \to 6 \to 9$ (un \omterm{def:b2:structures:sn}{ciclo} $\zeta'$ de
        longitud $7$ que deja fijo el centro $5$) y del recorrido
        de esquina $9 \to 6 \to 5 \to 8 \to 9$ (un \omterm{def:b2:structures:sn}{ciclo} de
        longitud $3$ que pasa por el centro). Conjugando el
        segundo por las potencias de $\zeta'$ y encadenando los
        lemas A y B, demuestra que el grupo de programas del juego
        del ocho es todo $A_8$, y por tanto que exactamente $9!/2
        = 181\,440$ de las $9! = 362\,880$ configuraciones son
        resolubles.
  \item (Un tablero pobre) Sea ahora el tablero un único
        \omterm{def:b2:structures:sn}{ciclo} de $n \geq 4$ casillas con $n - 1$ fichas. Prueba
        que el \omterm{def:b2:structures:generated}{orden} \omterm{def:b2:structures:generated}{cíclico} de las fichas es invariante, que cada
        clase de alcanzabilidad tiene exactamente $n(n - 1)$
        configuraciones \emph{(las clases son las órbitas de un
        \omterm{def:b2:structures:generated}{grupo cíclico} de \omterm{def:b2:structures:generated}{orden} $\operatorname{lcm}(n, n-1) =
        n(n-1)$)}, y que hay $(n - 2)!$ clases ---para $n \geq 5$,
        muchas más que $2$---: en un tablero estrecho el
        invariante de paridad casi no captura nada y manda la
        geometría.
  \item Dos veredictos mediante el criterio de la pregunta 19: la
        bandeja completamente invertida (fichas $15, 14, \dots, 1$
        en las casillas $1$ a $15$, hueco en casa) y la bandeja
        con el hueco en la casilla $1$ seguido de las fichas $15,
        14, \dots, 1$ en las casillas $2$ a $16$. ¿Cuál de las dos
        es resoluble?
  \item (Síntesis) La demostración tiene dos pilares
        independientes: un \emph{invariante} ($I$, construido a
        partir del morfismo signatura) que muestra que a lo sumo
        la mitad de las configuraciones son alcanzables, y un
        teorema de \emph{generación explícita} ($H = A_{15}$) que
        muestra que al menos la mitad lo son. Di, en una frase
        cada uno, dónde intervinieron: la propiedad de morfismo de
        $\varepsilon$; el teorema de Lagrange; la generación de
        $A_n$ por los \omterm{def:b2:structures:sn}{ciclos} de longitud $3$; la conjugación.
        Enuncia el metaprincipio en una línea.
\end{enumerate}
\end{problem}
