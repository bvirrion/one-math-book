\chapter{Sucesiones y series}\label{ch:b2:series}

La teoría de las series numéricas (volumen del primer año) madura aquí
en tres direcciones: las series con valores en \omterm{def:b2:nvs:banach}{espacios de Banach},
donde el trabajo lo hace la convergencia absoluta; los criterios más
finos para series reales (\omterm{thm:b2:series:abel}{sumación de Abel}); y las \emph{\omterm{def:b2:series:summable}{familias
sumables}} ---la suma liberada del orden de los términos---, con el
teorema de Fubini para sumas dobles y el \omterm{thm:b2:series:fubini}{producto de Cauchy}. Estas
herramientas sostienen todos los capítulos de series de funciones que
vienen a continuación.

\section{Series en espacios normados}

\begin{definition}\label{def:b2:series:def}
Para una sucesión $(u_n)$ de un espacio normado $E$, la serie $\sum
u_n$ converge cuando lo hacen sus sumas parciales; converge
\emph{absolutamente}\index{convergencia absoluta!en un espacio de
Banach} cuando $\sum \norm{u_n} < \infty$. En un espacio de
\emph{Banach}, la convergencia absoluta implica la convergencia
(\cref{thm:b2:nvs:absoluteconvergence}); en un espacio
no \omterm{def:b2:metric:complete}{completo} esto puede fallar (\cref{exo:b2:series:9}).
\end{definition}

\begin{example}\label{ex:b2:series:neumann}
En $\mathcal{M}_n(K)$ (o en $\mathcal{L}_c(E)$ con $E$ de Banach):
para $\vertiii A < 1$, la \emph{serie de Neumann} $\sum A^k$
converge \omterm{def:b2:series:def}{absolutamente} a $(I - A)^{-1}$ (demostrado en el~\cref{exo:b2:nvs:5}); y $\sum \frac{A^k}{k!}$ converge
\omterm{def:b2:series:def}{absolutamente} a $\eu^A$ para toda $A$
(\cref{ex:b2:nvs:matrixexp}). Las series geométrica y exponencial
con valores en operadores se comportan como sus modelos escalares:
ese es todo el sentido del marco de Banach.
\end{example}

\section{Sumación de Abel}

\begin{theorem}[Sumación de Abel y criterio de Abel]\label{thm:b2:series:abel}
(Sumación por partes) Para escalares $a_n$ y vectores $b_n$, con
$B_n = \sum_{k=0}^{n} b_k$:\index{sumación de Abel}
\[
\sum_{n=0}^{N} a_n b_n
= a_N B_N - \sum_{n=0}^{N-1} (a_{n+1} - a_n) B_n .
\]
(Criterio de Abel) Si $(a_n)$ es una sucesión real decreciente
hacia $0$ y las sumas parciales $B_n$ están \emph{acotadas} (en un
\omterm{def:b2:nvs:banach}{espacio de Banach}), entonces $\sum a_n b_n$ converge.
\end{theorem}

\begin{proof}
La identidad, paso a paso: con $B_{-1} = 0$, escribamos $b_n = B_n -
B_{n-1}$ y separemos,
\[
\sum_{n=0}^{N} a_nb_n
= \sum_{n=0}^{N} a_nB_n - \sum_{n=0}^{N} a_nB_{n-1}
= \sum_{n=0}^{N} a_nB_n - \sum_{n=0}^{N-1} a_{n+1}B_{n} ,
\]
reindexando la segunda suma mediante $n \mapsto n + 1$ (el término
en $B_{-1}$ se anula); agrupando el rango común $0 \leq n \leq N-1$
queda $a_NB_N$ más $\sum_{n\leq N-1}(a_n - a_{n+1})B_n$: la fórmula
del enunciado. Es la integración por partes discreta, con $(B_n)$
como primitiva de $(b_n)$ y la diferencia $a_{n+1} - a_n$ como
derivada de $(a_n)$. Para el criterio, con $\norm{B_n} \leq M$: el
término de frontera $a_N B_N \to 0$; y la serie $\sum (a_n -
a_{n+1})B_n$ converge \omterm{def:b2:series:def}{absolutamente}, pues
\[
\sum_n \norm{(a_{n+1} - a_n)B_n} \leq M \sum_n (a_n - a_{n+1})
= M a_0 < \infty
\]
(suma telescópica, con $a_n \downarrow 0$). Ambas piezas de la
identidad convergen, luego también lo hace $\sum a_n b_n$.
\end{proof}

\begin{example}\label{ex:b2:series:sinn}
$\sum \frac{\sin n}{n}$ converge: $a_n = \frac1n \downarrow 0$ y
$B_n = \sum_{k=1}^{n} \sin k$ está acotada; en efecto, $B_n =
\Im\sum_{k \leq n} \eu^{\iu k} = \Im\,\frac{\eu^{\iu}(\eu^{\iu n} -
1)}{\eu^{\iu} - 1}$, de módulo $\leq \frac{2}{\abs{\eu^{\iu} - 1}}$.
\emph{No} converge \omterm{def:b2:series:def}{absolutamente} ($\abs{\sin n} \geq \sin^2 n =
\frac{1 - \cos 2n}{2}$, y $\sum \frac{1 - \cos 2n}{2n}$ diverge,
puesto que $\sum \frac{\cos 2n}{n}$ converge por el mismo criterio
de Abel mientras que $\sum \frac{1}{2n}$ diverge). El criterio de
las series alternadas es el caso particular $b_n = (-1)^n$.
\end{example}

\begin{example}[Abel sobre la circunferencia de convergencia]\label{ex:b2:series:abelboundary}
¿Para qué complejos $z$ con $\abs z = 1$ converge $\sum_{n \geq 1}
\frac{z^n}{n}$? En $z = 1$ es la serie armónica: divergente. Para
$z \neq 1$ sobre la circunferencia se aplica el criterio de Abel
con $a_n = \frac1n \downarrow 0$ y $b_n = z^n$, cuyas sumas
parciales están acotadas independientemente de $N$:
\[
\Bigl|\sum_{n=1}^{N} z^n\Bigr|
= \Bigl|\frac{z(z^N - 1)}{z - 1}\Bigr|
\leq \frac{2}{\abs{z - 1}} .
\]
Convergente, aunque nunca \omterm{def:b2:series:def}{absolutamente} (por $\sum\frac1n$). Una
sola serie, toda una circunferencia de comportamientos:
divergencia en un único punto y semiconvergencia en todos los
demás. Este es el comportamiento habitual de las series de
potencias en la frontera (\cref{ch:b2:powerseries}), encontrado
aquí a mano; en $z = -1$ recupera la serie armónica alternada, y
en $z = \eu^{\iu\theta}$ sus partes real e imaginaria son las
series $\sum\frac{\cos n\theta}{n}$ y $\sum\frac{\sin
n\theta}{n}$ del~\cref{exo:b2:series:4}.
\end{example}

\begin{example}[Una serie alternada con trampa]\label{ex:b2:series:trap}
¿Converge $\sum_{n\geq2} \dfrac{(-1)^n}{\sqrt n + (-1)^n}$? Los
signos alternan y los términos tienden a $0$ y, sin embargo, el
criterio de las alternadas \emph{no} se aplica: los módulos
$\frac{1}{\sqrt n + (-1)^n}$ no son decrecientes (dan un salto
hacia arriba en cada $n$ impar). Desarrollemos en su lugar:
\[
\frac{(-1)^n}{\sqrt n + (-1)^n}
= \frac{(-1)^n}{\sqrt n}\cdot
\frac{1}{1 + \frac{(-1)^n}{\sqrt n}}
= \frac{(-1)^n}{\sqrt n} - \frac{1}{n}
+ O\Bigl(\frac{1}{n^{3/2}}\Bigr).
\]
La primera pieza converge (criterio de las alternadas, aplicado
honestamente a $\frac1{\sqrt n}\downarrow0$), la tercera converge
\omterm{def:b2:series:def}{absolutamente}, pero la pieza central es la serie armónica
divergente: la suma \emph{diverge} a $-\infty$. La moraleja: cuando
falla la monotonía, hay que desarrollar hasta que cada pieza sea o
bien \omterm{def:b2:series:def}{absolutamente} convergente o bien un caso limpio de aplicación
de un criterio; el $-\frac1n$ escondido es invisible para un
recuento de signos.
\end{example}

\begin{remark}[Errores frecuentes]\label{rem:b2:series:pitfalls}
(i) ``Los términos tienden a $0$'' no demuestra nada: la serie
armónica diverge. (ii) El criterio de las alternadas exige módulos
\emph{decrecientes}; el~\cref{ex:b2:series:trap} es el
contraejemplo canónico, y la tercera serie del~\cref{exo:b2:series:1} el ejercicio de rigor. (iii) Las series
condicionalmente convergentes no pueden reordenarse
(\cref{ex:b2:series:rearrange}), y sus \omterm{thm:b2:series:fubini}{productos de Cauchy} pueden
diverger: para el cuadrado de $\sum\frac{(-1)^n}{\sqrt{n+1}}$, los
términos diagonales cumplen
\[
\abs{c_k} = \sum_{m=0}^{k}
\frac{1}{\sqrt{(m+1)(k-m+1)}}
\geq (k+1)\cdot\frac{2}{k+2} \longrightarrow 2 \neq 0
\]
(cada factor es a lo sumo $\frac{k+2}2$ por la desigualdad entre
medias), de modo que $\sum c_k$ diverge: la convergencia absoluta
de al menos uno de los factores (\cref{exo:b2:series:8}) no es un
lujo. (iv) La \omterm{def:b2:series:summable}{sumabilidad} se refiere por definición a cotas
\emph{absolutas}: no existe tal cosa como una \omterm{def:b2:series:summable}{familia sumable}
condicionalmente.
\end{remark}

\section{Familias sumables}

\begin{definition}\label{def:b2:series:summable}
Sea $I$ un conjunto de índices \omterm{def:b2:structures:countable}{numerable}. Una familia $(u_i)_{i \in
I}$ de \emph{reales no negativos} es
\emph{sumable}\index{familia sumable} cuando las sumas parciales
finitas están acotadas; su suma es
\[
\sum_{i \in I} u_i = \sup_{F \subseteq I \text{ finito}} \sum_{i
\in F} u_i \in \intcc{0}{+\infty} .
\]
Una familia de reales o de complejos (o de vectores de un Banach) es
sumable cuando lo es $(\norm{u_i})$; su suma se define entonces
separando las partes positiva y negativa (o real e imaginaria) o,
equivalentemente, como el valor común de $\sum_{n} u_{\sigma(n)}$
sobre todas las enumeraciones $\sigma$ de $I$ (véase más abajo).
\end{definition}

\begin{method}[Cómo elegir un criterio]\label{met:b2:series:choosing}
Ante $\sum u_n$, por este orden: (1) si $u_n \not\to 0$, hay
divergencia; se acabó. (2) Si los términos tienen signo constante,
compárese: hállese un equivalente (\cref{ch:b2:comparison}) y
sitúese en el mapa de Riemann--Bertrand. (3) Si los signos alternan
con módulos \emph{decrecientes}, el criterio de las alternadas; si
los módulos no son monótonos, desarróllese el término hasta que
cada pieza sea \omterm{def:b2:series:def}{absolutamente} convergente o un caso limpio
(\cref{ex:b2:series:trap}). (4) Si el patrón de signos es
oscilatorio pero estructurado ($\sin n\theta$, $\eu^{\iu
n\theta}$, potencias de matrices), el criterio de Abel con sumas
parciales acotadas. (5) La convergencia absoluta siempre merece
comprobarse primero: es más fuerte, es inmune al orden y desbloquea
los \omterm{thm:b2:series:fubini}{productos de Cauchy} y el teorema de Fubini.
\end{method}

\begin{example}[Sumabilidad por recuento diagonal]\label{ex:b2:series:diagonalcount}
¿Para qué $s > 0$ es \omterm{def:b2:series:summable}{sumable} la familia $\bigl((m +
n)^{-s}\bigr)_{m, n \geq 1}$? Agrupemos las sumas parciales finitas
por diagonales $m + n = k$: la diagonal $k$ contiene $k - 1$ pares,
cada uno aportando $k^{-s}$, de modo que las sumas finitas están
acotadas exactamente por (y agotan)
\[
\sum_{k \geq 2} \frac{k - 1}{k^{s}} ,
\]
serie de términos positivos equivalente a $k^{1-s}$: \omterm{def:b2:series:summable}{sumable} si y
solo si $s - 1 > 1$, es decir, $s > 2$. El índice bidimensional se
come una potencia entera: un plano de términos es ``una dimensión
más divergente'' que una recta; quien decide la \omterm{def:b2:series:summable}{sumabilidad} es la
geometría de recuento del conjunto de índices, no el tamaño de cada
término. (El mismo censo muestra que $\bigl((m^2 +
n^2)^{-1}\bigr)$ no es \omterm{def:b2:series:summable}{sumable}: sobre la diagonal $m + n = k$ cada
término es al menos $k^{-2}$, y $(k-1)\cdot k^{-2}$ suma como la
serie armónica.)
\end{example}

\begin{theorem}[Sumabilidad y orden]\label{thm:b2:series:rearrangement}
\begin{enumerate}
  \item Para familias no negativas, la suma es invariante por
        cualquier enumeración: $\sum_{i} u_i = \sum_{n=0}^{\infty}
        u_{\sigma(n)}$ para toda biyección $\sigma \colon \N \to I$.
  \item Una serie real o compleja $\sum u_n$ es
        \emph{conmutativamente convergente} (toda reordenación
        converge, con la misma suma) si y solo si converge
        \omterm{def:b2:series:def}{absolutamente}.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Toda suma parcial $\sum_{n \leq N} u_{\sigma(n)}$ es una suma
parcial finita de la familia (luego $\leq$ el supremo); y todo $F$
finito está contenido en algún $\{\sigma(0), \dots, \sigma(N)\}$
(luego el supremo es $\leq$ el límite de la serie). Las dos cotas
coinciden.

(2) Si $\sum\abs{u_n} < \infty$: para toda reordenación $\sigma$ y
todo $\varepsilon > 0$, elíjase $N$ con $\sum_{n > N}\abs{u_n} \leq
\varepsilon$; a partir del índice en que $\sigma$ ha agotado
$\intint{0}{N}$, las sumas parciales reordenadas distan del límite
original a lo sumo $\varepsilon$: misma suma. Si $\sum \abs{u_n} =
\infty$ pero $\sum u_n$ converge (caso real; el complejo se sigue
coordenada a coordenada): las partes positiva y negativa divergen
ambas, y se puede reordenar para alcanzar cualquier límite prefijado
---teorema de Riemann, realizado en el~\cref{exo:b2:series:5}---,
de modo que falla la convergencia conmutativa.
\end{proof}

\begin{example}[Una reordenación pillada in fraganti]\label{ex:b2:series:rearrange}
La serie armónica alternada suma
$\sum_{n\geq1}\frac{(-1)^{n-1}}{n} = \ln 2$ (volumen del primer
año). Reordenémosla como ``un positivo, dos negativos'':
\[
1 - \frac12 - \frac14 + \frac13 - \frac16 - \frac18 + \frac15 -
\cdots
\]
Agrupando cada bloque de tres,
\[
\frac{1}{2k-1} - \frac{1}{4k-2} - \frac1{4k}
= \frac{1}{4k-2} - \frac{1}{4k}
= \frac12\Bigl(\frac{1}{2k-1} - \frac1{2k}\Bigr),
\]
de modo que la serie reordenada converge a $\frac12\ln 2$: la mitad
de la suma original, con exactamente los mismos términos. Las
series convergentes no \omterm{def:b2:series:def}{absolutamente} recuerdan el orden de sus
términos; las \omterm{def:b2:series:summable}{familias sumables} son precisamente las que no.
\end{example}

\begin{example}[Agrupar es seguro; desagrupar, no]\label{ex:b2:series:grouping}
Agrupar términos consecutivos de una serie \emph{convergente} no
altera nunca la suma: las sumas parciales agrupadas forman una
subsucesión de las originales. La operación inversa está
prohibida:
\[
(1 - 1) + (1 - 1) + (1 - 1) + \cdots = 0 + 0 + \cdots = 0,
\]
y sin embargo la versión desagrupada $1 - 1 + 1 - 1 + \cdots$
diverge (sus sumas parciales oscilan entre $1$ y $0$). Desagrupar
solo es legítimo con una hipótesis compensatoria; por ejemplo, que
los términos tiendan a $0$ con longitudes de grupo acotadas:
entonces, entre dos sumas parciales agrupadas, las originales se
desvían a lo sumo en una suma de un número acotado de términos
$o(1)$, y la convergencia se transfiere de vuelta. Esa es
exactamente la cláusula bajo la cual el cálculo por bloques del~\cref{ex:b2:series:rearrange} es una demostración y no un juego de
manos.
\end{example}

\begin{theorem}[Fubini para familias; productos de Cauchy]\label{thm:b2:series:fubini}
Sea $(u_{m,n})_{(m,n) \in \N^2}$ una familia doble \omterm{def:b2:series:summable}{sumable} (es
decir, con $\sup_F \sum_F \abs{u_{m,n}} < \infty$).
Entonces\index{teorema de Fubini para series}
\[
\sum_{(m,n)} u_{m,n}
= \sum_{m=0}^{\infty}\Bigl(\sum_{n=0}^{\infty} u_{m,n}\Bigr)
= \sum_{n=0}^{\infty}\Bigl(\sum_{m=0}^{\infty} u_{m,n}\Bigr)
= \sum_{k=0}^{\infty}\Bigl(\sum_{m+n=k} u_{m,n}\Bigr),
\]
convergiendo todas las series interiores (\omterm{def:b2:series:def}{absolutamente}). En
particular, si $\sum a_m$ y $\sum b_n$ convergen \omterm{def:b2:series:def}{absolutamente}, su
\emph{producto de Cauchy}\index{producto de Cauchy} converge
\omterm{def:b2:series:def}{absolutamente} y
\[
\Bigl(\sum_m a_m\Bigr)\Bigl(\sum_n b_n\Bigr)
= \sum_{k=0}^{\infty} c_k,
\qquad
c_k = \sum_{m=0}^{k} a_m b_{k-m} .
\]
\end{theorem}

\begin{proof}
\emph{Caso no negativo.} Cada agrupación (por filas, por columnas o
por diagonales) calcula el mismo supremo: todo conjunto finito de
pares está contenido en un bloque finito de filas (lo que acota cada
suma agrupada por debajo mediante sumas parciales finitas y por
encima mediante el total), y la convergencia monótona de las sumas
parciales hace el resto; en concreto, para filas: de $\sum_{m \leq
M}\sum_{n \leq N} u_{m,n} \leq S$ se sigue, haciendo $N \to \infty$
y después $M \to \infty$, que $\sum_m \sum_n u_{m,n} \leq S$; y
recíprocamente, todo $F$ finito cabe en un rectángulo así, luego $S
\leq \sum_m\sum_n u_{m,n}$. Diagonales: las mismas dos cotas con
triángulos en lugar de rectángulos.

\emph{Caso general.} Sepárense las partes positiva y negativa (real
e imaginaria), cada una familia no negativa \omterm{def:b2:series:summable}{sumable}; las cuatro
agrupaciones coinciden en cada parte y, por tanto, en la diferencia;
la convergencia absoluta de las series interiores procede del caso
no negativo aplicado a $\abs{u_{m,n}}$.

\emph{\omterm{thm:b2:series:fubini}{Producto de Cauchy}.} La familia $u_{m,n} = a_m b_n$ es
\omterm{def:b2:series:summable}{sumable}: las sumas parciales finitas de $\abs{a_mb_n}$ están
acotadas por $\bigl(\sum\abs{a_m}\bigr)\bigl(\sum\abs{b_n}\bigr)$.
Las filas dan $\bigl(\sum a_m\bigr)\bigl(\sum b_n\bigr)$; las
diagonales dan $\sum_k c_k$.
\end{proof}

\begin{example}[La identidad exponencial, honestamente]\label{ex:b2:series:exp}
Para $a, b \in \C$ (o matrices que conmutan):
\[
\Bigl(\sum_m \frac{a^m}{m!}\Bigr)\Bigl(\sum_n
\frac{b^n}{n!}\Bigr)
= \sum_k \sum_{m+n=k} \frac{a^m b^n}{m!\,n!}
= \sum_k \frac{(a + b)^k}{k!},
\]
por el teorema del binomio en cada diagonal: $\eu^a \eu^b =
\eu^{a+b}$, la ecuación funcional de $\exp$ deducida solo de la
serie. (La conmutación se usa en el paso del binomio; para matrices
que no conmutan la identidad falla de verdad,
\cref{ch:b2:diffeq}.)
\end{example}

\begin{example}[Los productos de Cauchy como herramienta de cálculo]\label{ex:b2:series:squaresum}
A partir de la serie geométrica y del $\sum_{n \geq 1} nz^n =
\frac{z}{(1-z)^2}$ ($\abs z < 1$) del~\cref{exo:b2:series:2}, un
\omterm{thm:b2:series:fubini}{producto de Cauchy} más remata el segundo momento. Multipliquemos
$\sum_m mz^m$ por $\sum_n z^n$: el coeficiente diagonal es
$\sum_{m=0}^k m = \frac{k(k+1)}2$, luego
\[
\frac{z}{(1-z)^3} = \sum_{k\geq0}\frac{k(k+1)}{2}\,z^k ,
\]
y la identidad $n^2 = 2\cdot\frac{n(n+1)}2 - n$ ensambla
\[
\sum_{n\geq1} n^2z^n = \frac{2z}{(1-z)^3} - \frac{z}{(1-z)^2}
= \frac{z(1+z)}{(1-z)^3} .
\]
En $z = \frac12$: $\sum_{n\geq1}\frac{n^2}{2^n} =
\frac{\frac12\cdot\frac32}{\frac18} = 6$, un valor cerrado sin
derivar en ningún momento, solo multiplicando como polinomios unas
series \omterm{def:b2:series:def}{absolutamente} convergentes. El mismo encadenamiento de
identidades calcula todo $\sum n^dz^n$, y quien se dedique a la
probabilidad reconocerá el segundo momento factorial de la ley
geométrica (\cref{ch:b2:genfun}).
\end{example}

\begin{example}[Evaluación de una suma doble]\label{ex:b2:series:doublesum}
Para $s > 1$ real, sea $\zeta(s) = \sum_{n\geq1} n^{-s}$. Contando
divisores mediante sumación doble ---la familia $(m^{-s}n^{-s})$
sobre $(m,n) \in (\N^*)^2$ es \omterm{def:b2:series:summable}{sumable} (producto de series
positivas convergentes)--- y agrupando por el producto $q = mn$:
\[
\zeta(s)^2 = \sum_{m,n} \frac{1}{(mn)^s}
= \sum_{q=1}^{\infty} \frac{d(q)}{q^s},
\]
donde $d(q)$ es el número de divisores de $q$. Las \omterm{def:b2:series:summable}{familias
sumables} convierten la combinatoria en análisis.
\end{example}

\begin{example}[Una evaluación por Fubini: $\sum_n (\zeta(n) - 1) = 1$]\label{ex:b2:series:zetasum}
Para $n \geq 2$ entero, $\zeta(n) - 1 = \sum_{k \geq 2} k^{-n}$. La
familia doble $(k^{-n})_{k, n \geq 2}$ es \omterm{def:b2:series:summable}{sumable}: sumando primero
las columnas geométricas,
\[
\sum_{k\geq2}\sum_{n\geq2} \frac{1}{k^n}
= \sum_{k\geq2} \frac{1/k^2}{1 - 1/k}
= \sum_{k\geq2} \frac{1}{k(k-1)} = 1
\]
(telescópica), y todos los términos son positivos, así que el~\cref{thm:b2:series:fubini} autoriza a sumar por filas en su lugar:
\[
\sum_{n\geq2}\bigl(\zeta(n) - 1\bigr) = 1 .
\]
Los infinitos valores de $\zeta$, cada uno de aspecto
trascendente, tienen colas que suman exactamente $1$. Moraleja:
cuando una suma doble tiene términos positivos, calcúlese en el
orden que se derrumbe; aquí las columnas son geométricas, las filas
son misteriosas, y Fubini transfiere el derrumbe.
\end{example}

\begin{example}[La serie geométrica resuelve una ecuación]\label{ex:b2:series:neumannsolve}
En el \omterm{def:b2:nvs:banach}{espacio de Banach} $\bigl(C(\intcc01),
\norm\cdot_\infty\bigr)$, resolvamos $x - K(x) = y$, donde $K(f)$
es la función constante $\frac12\int_0^1 f$. La \omterm{thm:b2:nvs:continuouslinear}{norma de operador}
cumple $\vertiii K \leq \frac12 < 1$, de modo que se aplica la
serie de Neumann (\cref{ex:b2:series:neumann}): $x = \sum_{n\geq0}
K^n(y)$. Calculemos las iteradas: $K(y) = \frac12\int_0^1 y$ (una
constante) y, aplicando $K$ a una constante $c$, se obtiene $\frac
c2$, de modo que $K^n(y) = \frac{1}{2^{n-1}}\cdot\frac12\int_0^1 y$
para $n \geq 1$. Sumando las constantes geométricas:
\[
x = y + \Bigl(\int_0^1 y\Bigr)
\sum_{n\geq1}\frac{1}{2^n} = y + \int_0^1 y .
\]
Comprobación: $x - K(x) = y + \int y - \frac12\bigl(\int y + \int
y\bigr) = y$. Una serie infinita, una respuesta finita y una
verificación de una línea: la serie geométrica es un algoritmo de
inversión, no solo un enunciado de convergencia.
\end{example}

\begin{example}[Telescopar por fracciones simples]\label{ex:b2:series:telescope}
La sumación exacta es rara; las sumas telescópicas son su principal
proveedor. Descompongamos
\[
\frac{1}{n(n+1)(n+2)}
= \frac{1}{2}\Bigl(\frac{1}{n(n+1)} -
\frac{1}{(n+1)(n+2)}\Bigr),
\]
(compruébese reduciendo a común denominador), de modo que las sumas
parciales se derrumban:
\[
\sum_{n=1}^{N}\frac{1}{n(n+1)(n+2)}
= \frac12\Bigl(\frac{1}{1\cdot2} -
\frac{1}{(N+1)(N+2)}\Bigr)
\longrightarrow \frac14 .
\]
El mismo patrón ---escribir el término como $c(u_n - u_{n+1})$ para
una $(u_n)$ explícita--- resolvió el~\cref{exo:b2:series:10}
(arcotangentes) y calcula todo $\sum\frac{1}{n(n+1)\cdots(n + k)} =
\frac{1}{k\cdot k!}$. Cuando a este nivel existe una suma exacta,
por lo general hay un telescopio escondido en el término.
\end{example}

\begin{remark}[Perspectivas dentro de este volumen]\label{rem:b2:series:perspectives}
Tres capítulos posteriores son clientes directos. Para el~\cref{ch:b2:funcseq}: la convergencia normal de $\sum f_n$ es la
convergencia absoluta de $\sum\norm{f_n}_\infty$ en el
\omterm{def:b2:nvs:banach}{espacio de Banach} $\bigl(C, \norm\cdot_\infty\bigr)$, es decir, el~\cref{thm:b2:nvs:absoluteconvergence} disfrazado. Para el~\cref{ch:b2:powerseries}: dentro del disco de convergencia todo es
absoluto y \omterm{def:b2:series:summable}{sumable}, de modo que los \omterm{thm:b2:series:fubini}{productos de Cauchy} y las
reordenaciones circulan libremente (por eso las series de potencias
se multiplican como polinomios); en la frontera toma el relevo el
criterio de Abel (\cref{ex:b2:series:abelboundary}). Para el~\cref{ch:b2:genfun}: las funciones generatrices de probabilidad son
series de potencias cuyas manipulaciones ---productos para sumas de
variables independientes, sumas dobles para leyes compuestas---
están todas autorizadas por el~\cref{thm:b2:series:fubini}. Las
\omterm{def:b2:series:summable}{familias sumables} son la asesoría jurídica del análisis que viene.
\end{remark}

\begin{remark}[Dónde se usa este capítulo]
Todo lo que lleve una suma infinita pasa por aquí: las series de
potencias (\cref{ch:b2:powerseries}) son \omterm{def:b2:series:summable}{familias sumables}
disfrazadas, los coeficientes de Fourier se multiplican mediante
\omterm{thm:b2:series:fubini}{productos de Cauchy} y se reordenan mediante Parseval
(\cref{ch:b2:fourier}), y las funciones generatrices de
probabilidad (\cref{ch:b2:genfun}) son el teorema de Fubini
aplicado a esperanzas. El volumen del tercer año absorbe las
\omterm{def:b2:series:summable}{familias sumables} dentro de la integración de Lebesgue respecto de
la medida de contar, donde el~\cref{thm:b2:series:fubini} pasa a ser
un caso particular del teorema de Fubini--Tonelli.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:series:1}
Carácter de $\sum \dfrac{\cos n}{n}$; $\;\sum \dfrac{(-1)^n}{\ln
n}$; $\;\sum \dfrac{(-1)^n}{n^{3/4} + \cos n}$ \emph{(desarrolla
como en la trampa del primer año: el criterio de las alternadas
necesita monotonía)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:series:2}
Demuestra que, para $\abs z < 1$, $\sum_{n\geq1} n z^{n} =
\dfrac{z}{(1-z)^2}$, mediante el \omterm{thm:b2:series:fubini}{producto de Cauchy} de $\sum z^n$
consigo misma.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:3}
(Lema de tipo Kronecker) Sea $\sum b_n$ una serie real convergente.
Demuestra, mediante \omterm{thm:b2:series:abel}{sumación de Abel}, que $\dfrac{1}{n}\sum_{k=1}^{n}
k\,b_k \to 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:4}
Estudia la convergencia de $\sum \dfrac{\sin(n\theta)}{n^\alpha}$
($\theta \in \R$, $\alpha > 0$): ¿para qué $(\theta, \alpha)$ es
\omterm{def:b2:series:def}{absolutamente} convergente, semiconvergente o divergente?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:5}
(Reordenación de Riemann) Sea $\sum u_n$ una serie real convergente
pero no \omterm{def:b2:series:def}{absolutamente} convergente, y sea $\ell \in \R$. Demuestra que
alguna reordenación de $\sum u_n$ converge a $\ell$. \emph{(Prueba
que ambas subseries de términos positivos y negativos divergen; y
alterna después con avidez: toma términos positivos hasta superar
$\ell$, luego negativos hasta bajar de él, y así sucesivamente; los
términos tienden a $0$, lo que fuerza la convergencia a $\ell$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:6}
Demuestra que la familia
$\Bigl(\dfrac{x^{m+n}}{m!\,n!}\Bigr)_{(m,n)\in\N^2}$ es \omterm{def:b2:series:summable}{sumable}
para todo $x \in \R$, y vuelve a deducir la identidad $(\eu^x)^2 =
\eu^{2x}$ agrupando la suma doble a lo largo de las diagonales $m +
n = k$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:7}
Demuestra que la familia $\bigl(\frac{1}{m^2 n^2}\bigr)_{m,n \geq
1}$ es \omterm{def:b2:series:summable}{sumable} y que, agrupando por el mcd, con $q = \gcd(m,n)$,
\[
\zeta(2)^2 = \sum_{q\geq1} \frac{1}{q^4}
\sum_{\substack{a,b \geq 1\\ \gcd(a,b)=1}} \frac{1}{a^2b^2}
= \zeta(4) \cdot S,
\]
donde $S = \sum_{\gcd(a,b)=1} \frac{1}{a^2b^2}$: deduce $S =
\zeta(2)^2/\zeta(4)$. \emph{(Todo par $(m,n)$ se escribe de manera
única como $(qa, qb)$ con $\gcd(a,b) = 1$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:8}
(Teorema de Abel sobre productos, versión ligera) Supongamos que
$\sum a_n$ converge \omterm{def:b2:series:def}{absolutamente} y que $\sum b_n$ converge.
Demuestra que su \omterm{thm:b2:series:fubini}{producto de Cauchy} $\sum c_n$ converge, con $\sum
c_n = (\sum a_n)(\sum b_n)$. \emph{(Escribe $C_N = \sum_{k\leq N}
c_k = \sum_n a_n B_{N-n}$, con $B$ las sumas parciales de $b$;
sepáralo según $n \leq N/2$ o no, usando la acotación de $(B_m)$ y
la cola absoluta de $(a_n)$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:9}
En el espacio $E$ (no \omterm{def:b2:metric:complete}{completo}) de las sucesiones reales nulas a
partir de un índice, con la \omterm{def:b2:nvs:norm}{norma} del supremo, exhibe una serie
\omterm{def:b2:series:def}{absolutamente} convergente que no converja en $E$. \emph{(Prueba con
$u_n = 2^{-n} e_n$, siendo $(e_n)$ las sucesiones canónicas.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:10}
Verifica la identidad $\arctan(n+1) - \arctan(n) =
\arctan\dfrac{1}{n^2 + n + 1}$ y deduce el valor exacto de
\[
\sum_{n=1}^{\infty} \arctan\frac{1}{n^2 + n + 1} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:series:11}
Determina el carácter (con equivalentes) de
\[
\sum_n \bigl(\sqrt{n+1} - \sqrt n\bigr)^{\alpha}
\ (\alpha > 0), \qquad
\sum_n \Bigl(1 - \cos\frac1n\Bigr), \qquad
\sum_n \Bigl(\eu - \Bigl(1 + \frac1n\Bigr)^{\!n}\Bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:series:12}
Sea $(a_n)$ positiva y \emph{decreciente} con $\sum a_n$
convergente. Demuestra que $n\,a_n \to 0$ \emph{(acota $n a_{2n}$
por una cola)}. Prueba que el recíproco es falso y que la monotonía
es esencial, con contraejemplos explícitos.
\end{exercise}

\section{Problema: el $\zeta(2) = \pi^2/6$ de Euler, por la suma de
cotangentes de Cauchy}

La identidad más célebre de Euler, $1 + \frac14 + \frac19 + \cdots =
\frac{\pi^2}6$, admite una demostración completamente elemental,
debida a Cauchy: la fórmula de De Moivre produce un polinomio cuyas
raíces son los números $\cot^2\frac{k\pi}{2n+1}$, las fórmulas de
Vieta suman esas raíces de forma exacta, y el encaje $\cot^2\theta <
\frac{1}{\theta^2} < 1 + \cot^2\theta$ aplasta las sumas parciales de
$\sum\frac1{k^2}$ entre dos cotas racionales explícitas. Recorremos la
demostración completa, extraemos $\zeta(4) = \frac{\pi^4}{90}$ por el
mismo método y cartografiamos después toda la frontera entre
convergencia y divergencia con las series de Bertrand, demostrando
además que esa frontera no alberga ninguna serie convergente más
lenta que todas las demás.

\begin{problem}[{Problema de fin de semana --- $\zeta(2) = \pi^2/6$ y
el panorama de Bertrand}]
\label{pb:b2:series:1}
En todo el problema, $n \geq 1$ y $\theta_k = \dfrac{k\pi}{2n+1}$
para $k = 1, \dots, n$; obsérvese que $0 < \theta_k < \frac\pi2$.

\medskip
\noindent\textbf{Parte I --- La identidad de las cotangentes.}
\begin{enumerate}
  \item Demuestra la fórmula de De Moivre $(\cos\theta +
        \iu\sin\theta)^m = \cos m\theta + \iu\sin m\theta$ ($m \in
        \N$) y deduce, para $m = 2n + 1$,
        \[
        \sin\bigl((2n{+}1)\theta\bigr) = \sum_{j=0}^{n}
        (-1)^j\binom{2n+1}{2j+1}\cos^{2(n-j)}\theta\,
        \sin^{2j+1}\theta .
        \]
  \item Deduce que, para $\theta \in \intoo{0}{\frac\pi2}$,
        \[
        \sin\bigl((2n{+}1)\theta\bigr) =
        \sin^{2n+1}\theta\; P_n(\cot^2\theta),
        \qquad
        P_n(x) = \sum_{j=0}^{n}
        (-1)^j\binom{2n+1}{2j+1}x^{\,n-j},
        \]
        polinomio de grado $n$ con coeficiente director $2n + 1$.
  \item Prueba que los $x_k = \cot^2\theta_k$, $k = 1, \dots, n$,
        son $n$ raíces \emph{distintas} de $P_n$ y, por tanto, todas
        ellas.
  \item Mediante las fórmulas de Vieta, demuestra la identidad
        exacta
        \[
        \sum_{k=1}^{n} \cot^2\frac{k\pi}{2n+1}
        = \frac{n(2n-1)}{3}.
        \]
  \item Deduce también que $\displaystyle\sum_{k=1}^{n}
        \frac{1}{\sin^2\theta_k} = \frac{2n(n+1)}{3}$.
  \item Demuestra el encaje $\cot^2\theta < \dfrac1{\theta^2} <
        \dfrac{1}{\sin^2\theta}$ para $\theta \in
        \intoo{0}{\frac\pi2}$ \emph{(a partir de $\sin\theta <
        \theta < \tan\theta$)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El encaje se cierra: el teorema de
Euler.}
\begin{enumerate}[resume]
  \item Sumando la pregunta 6 sobre $k = 1, \dots, n$ con $\theta =
        \theta_k$, establece
        \[
        \frac{n(2n-1)}{3} \;<\;
        \frac{(2n+1)^2}{\pi^2}\sum_{k=1}^{n}\frac1{k^2}
        \;<\; \frac{2n(n+1)}{3}.
        \]
  \item Concluye (\emph{el teorema de Euler, por la demostración de
        Cauchy}):
        \[
        \zeta(2) = \sum_{k=1}^{\infty}\frac{1}{k^2} =
        \frac{\pi^2}{6}.
        \]
  \item Extrae una velocidad del encaje: prueba que
        \[
        \Bigl|\sum_{k=1}^{n}\frac1{k^2} - \frac{\pi^2}6\Bigr|
        = O\Bigl(\frac1n\Bigr),
        \]
        en coherencia con la cola exacta $\sum_{k>n}k^{-2} =
        \frac1n - \frac{1}{2n^2} + O(n^{-3})$ del~\cref{exo:b2:comparison:11}.
  \item Pon la máquina un piso más arriba: usando la segunda función
        de Vieta de $P_n$, prueba que
        \[
        \sum_{k=1}^{n}\cot^4\theta_k =
        \Bigl(\frac{n(2n-1)}3\Bigr)^{\!2} -
        \frac{2n(2n-1)(2n-2)(2n-3)}{60}
        \;\sim\; \frac{8n^4}{45},
        \]
        y encaja con $\cot^4 < \theta^{-4} < (1 + \cot^2)^2$ para
        obtener $\zeta(4) = \dfrac{\pi^4}{90}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Dividendos.}
\begin{enumerate}[resume]
  \item Deduce de $\zeta(2) = \frac{\pi^2}6$:
        \[
        \sum_{k\geq0}\frac{1}{(2k+1)^2} = \frac{\pi^2}{8},
        \qquad
        \sum_{k\geq1}\frac{(-1)^{k-1}}{k^2} = \frac{\pi^2}{12}.
        \]
  \item Combina con el~\cref{exo:b2:series:7}: calcula $S =
        \sum_{\gcd(a,b)=1}\frac{1}{a^2b^2} =
        \frac{\zeta(2)^2}{\zeta(4)} = \frac52$ e interpreta
        $\frac{1}{\zeta(2)} = \frac{6}{\pi^2} \approx 0.608$ como la
        densidad de los pares de enteros primos entre sí (enuncia la
        heurística con honestidad: el recuento riguroso es asunto
        del volumen del tercer año).
  \item (Aceleración certificada) La fórmula de la cola de la
        pregunta 9 da $\sum_{k\leq n}k^{-2} + \frac1n -
        \frac1{2n^2} = \frac{\pi^2}6 + O(n^{-3})$. Compara el
        trabajo necesario para seis cifras de $\zeta(2)$: sumación
        directa frente a la suma corregida en $n = 100$ (donde el
        error es $1.7\cdot10^{-7}$).
  \item Comprueba la pregunta 4 a mano para $n = 1$ y $n = 2$ (los
        valores $\cot^2\frac\pi3 = \frac13$ y $\cot^2\frac\pi5 +
        \cot^2\frac{2\pi}5 = 2$), usando $\cos\frac\pi5 =
        \frac{1+\sqrt5}4$ o una evaluación numérica.
  \item Demuestra la identidad compañera
        \[
        \sum_{k=1}^{n}\tan^2\frac{k\pi}{2n+1} = n(2n+1)
        \]
        \emph{(los números $\tan^2\theta_k$ son las raíces del
        polinomio invertido $x^nP_n(1/x)$)}, y verifícala en $n =
        1$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El panorama de Bertrand.} Para
$\alpha, \beta \in \R$, consideremos la \emph{serie de Bertrand}
\[
\sum_{n \geq 3} \frac{1}{n^{\alpha}(\ln n)^{\beta}} .
\]
\begin{enumerate}[resume]
  \item Prueba que para $\alpha > 1$ la serie converge sea cual sea
        $\beta$ \emph{(compara con $n^{-(1+\alpha)/2}$)}.
  \item Prueba que para $\alpha < 1$ diverge sea cual sea $\beta$.
  \item Para $\alpha = 1$: usando la comparación serie--integral
        (\cref{thm:b2:comparison:seriesintegral}) con $f(t) =
        \frac{1}{t(\ln t)^\beta}$, demuestra que converge si y solo
        si $\beta > 1$.
  \item Itera la frontera: prueba que $\sum\frac{1}{n\ln n\,\ln\ln
        n}$ diverge mientras que $\sum\frac{1}{n\ln n\,(\ln\ln
        n)^2}$ converge.
  \item Dos trampas: determina el carácter de
        \[
        \sum_n \frac{1}{n^{1 + 1/\ln n}}
        \qquad\text{y}\qquad
        \sum_n \frac{1}{n^{1 + 1/\ln\ln n}}
        \]
        \emph{(calcula $n^{1/\ln n}$ exactamente; compara
        $n^{1/\ln\ln n}$ con toda potencia de $\ln n$)}.
  \item (No hay serie convergente más lenta) Sea $\sum a_n$ una
        serie convergente cualquiera con $a_n > 0$, y sean $R_n =
        \sum_{k \geq n}a_k$ sus colas. Demuestra que $\sum
        \frac{a_n}{\sqrt{R_n}}$ \emph{sigue convergiendo}
        \emph{(compara con la suma telescópica $2(\sqrt{R_n} -
        \sqrt{R_{n+1}})$)}, pese a que
        $\frac{a_n/\sqrt{R_n}}{a_n} \to \infty$: toda serie
        convergente está estrictamente dominada por otra serie
        convergente. La frontera de la convergencia no es una curva,
        sino una niebla.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Comprobaciones cruzadas y síntesis.}
\begin{enumerate}[resume]
  \item (Condensación de Cauchy) Demuestra que, para $(a_n)$
        positiva y decreciente, $\sum a_n$ converge si y solo si
        $\sum 2^k a_{2^k}$ converge. Vuelve a deducir de ahí la
        frontera de la pregunta 18.
  \item (El coste de la lentitud) Para $\sum\frac1{n(\ln n)^2}$,
        acota la cola mediante una integral y prueba que sumar hasta
        $N = 10^6$ deja todavía un error mayor que $0.07$: una
        convergencia certificada por la teoría puede ser inútil para
        el cálculo numérico; contrástese con la pregunta 13.
  \item Clasifica (con justificaciones de una línea):
        $\sum\frac1{n\ln n}$, $\sum\frac1{n^{1.01}}$,
        $\sum\frac{(\ln n)^{100}}{n^{1.001}}$ y
        $\sum\frac1{n(\ln n)(\ln\ln n)^{3}}$.
  \item (Síntesis) Una frase para cada punto: cómo De Moivre
        convirtió una identidad trigonométrica en un polinomio con
        sumas de raíces calculables; dónde necesitó el encaje
        identidades \emph{exactas} en los extremos en lugar de
        equivalentes; qué herramienta del~\cref{ch:b2:comparison}
        movió la parte IV; y qué dice la pregunta 21 sobre el sueño
        de un ``criterio universal de comparación''. Nombra las dos
        cumbres: el $\zeta(2) = \frac{\pi^2}6$ de Euler (y su piso
        superior, $\zeta(4) = \frac{\pi^4}{90}$) y la clasificación
        de Bertrand. Señala dónde se volverá a demostrar $\zeta(2)$:
        mediante Parseval en el~\cref{ch:b2:fourier}; un teorema,
        dos civilizaciones.
\end{enumerate}
\end{problem}
