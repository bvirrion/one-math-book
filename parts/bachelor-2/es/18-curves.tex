\chapter{Curvas}\label{ch:b2:curves}

La geometría se vuelve ahora diferencial. Una curva es un punto que se
mueve por el espacio; el cálculo nos da su velocidad y su aceleración,
y la geometría pregunta qué es intrínseco, es decir, independiente de
la rapidez con que recorramos la trayectoria. Las respuestas son la
\emph{\omterm{def:b2:curves:length}{longitud de arco}}, que mide la propia trayectoria, y la
\emph{\omterm{thm:b2:curves:frenet2d}{curvatura}}, que mide cuánto se dobla. En dimensión $3$ hay un
segundo invariante, la \emph{\omterm{thm:b2:curves:frenet3d}{torsión}}, que mide cuánto se retuerce la
curva fuera de su plano. El instrumento contable de todo esto es el
\emph{triedro de Frenet} móvil.

\section{Arcos parametrizados}

\begin{definition}[Arco parametrizado]\label{def:b2:curves:arc}
Un \emph{arco parametrizado}\index{arco parametrizado} de clase
$\mathcal{C}^k$ ($k \geq 1$) es una aplicación $\gamma \colon I \to
\R^n$ de clase $\mathcal{C}^k$ sobre un intervalo $I$. Un punto
$\gamma(t)$ es \emph{regular} si $\gamma'(t) \neq 0$, y el arco es
\emph{regular} si lo son todos sus puntos. La recta que pasa por
$\gamma(t)$ y tiene dirección $\gamma'(t)$ es la \emph{recta
tangente}\index{recta tangente} en un punto regular.
\end{definition}

\begin{definition}[Cambio de parámetro]\label{def:b2:curves:reparam}
Un \emph{cambio de parámetro}\index{cambio de parámetro} de clase
$\mathcal{C}^k$ es un difeomorfismo $\mathcal{C}^k$ $\theta \colon J
\to I$ entre intervalos ($\theta' \neq 0$ en todas partes). Los arcos
$\gamma$ y $\gamma \circ \theta$ se llaman \emph{equivalentes}; un
\emph{arco geométrico}\index{arco geométrico} (o curva) es una clase
de equivalencia. Las nociones invariantes por cambio de parámetro
---la trayectoria, la \omterm{def:b2:curves:arc}{recta tangente}, la \omterm{def:b2:curves:length}{longitud de arco}, la
curvatura--- se llaman \emph{geométricas}.
\end{definition}

\begin{remark}
La trayectoria por sí sola no determina el \omterm{def:b2:curves:reparam}{arco geométrico}: las
parametrizaciones $t \mapsto (\cos t, \sin t)$ sobre $[0, 2\pi]$ y
sobre $[0, 4\pi]$ tienen la misma imagen, pero recorren la
circunferencia una y dos veces. Un \omterm{def:b2:curves:reparam}{arco geométrico} recuerda la
multiplicidad y la orientación del recorrido, no su rapidez.
\end{remark}

\begin{example}[La rapidez no cambia nada
geométrico]\label{ex:b2:curves:speedinvariance}
Parametricemos la circunferencia unidad mediante
\[
\gamma(t) = (\cos t^2,\ \sin t^2),
\qquad t \in \intcc0{\sqrt{2\pi}} .
\]
La rapidez $\norm{\gamma'(t)} = 2t$ crece linealmente y, sin
embargo,
\[
L = \int_0^{\sqrt{2\pi}}2t\,\dd t = 2\pi ,
\]
la misma \omterm{def:b2:curves:length}{longitud} que a rapidez constante, como promete el~\cref{thm:b2:curves:lengthinv} mediante el \omterm{def:b2:curves:reparam}{cambio de parámetro} $t
\mapsto t^2$. La \omterm{def:b2:curves:arc}{recta tangente}, la \omterm{thm:b2:curves:frenet2d}{curvatura} calculada con la~\cref{prop:b2:curves:kappaformula} y cualquier otra magnitud
geométrica coinciden igualmente; solo $t = 0$ merece un vistazo,
pues $\gamma'(0) = 0$ hace irregular \emph{esta parametrización}
aunque la trayectoria sea una circunferencia perfecta. Los
enunciados geométricos toleran mal las malas parametrizaciones:
primero reparametrizar, luego concluir.
\end{example}

\begin{example}\label{ex:b2:curves:cusp}
El arco $\gamma(t) = (t^2, t^3)$ es $\mathcal{C}^\infty$ pero no
regular: $\gamma'(0) = (0, 0)$. Su trayectoria, la parábola
semicúbica $y^2 = x^3$, tiene una \emph{cúspide} en el origen: la
regularidad de la parametrización no impide una singularidad
geométrica donde se anula la velocidad. Por eso la hipótesis de
regularidad $\gamma' \neq 0$ no es cosmética.
\end{example}

\section{Longitud de arco}

\begin{definition}[Longitud de arco]\label{def:b2:curves:length}
Sea $\gamma \colon [a, b] \to \R^n$ un arco $\mathcal{C}^1$. Su
\emph{longitud}\index{longitud de arco} es
\[
L(\gamma) = \int_a^b \norm{\gamma'(t)}\, \dd t ,
\]
donde $\norm{\cdot}$ es la \omterm{def:b2:nvs:norm}{norma} euclídea. La \emph{función longitud
de arco}\index{función longitud de arco} basada en $t_0$ es $s(t) =
\int_{t_0}^t \norm{\gamma'(u)}\,\dd u$.
\end{definition}

\begin{theorem}[La longitud es geométrica; caracterización
poligonal]\label{thm:b2:curves:lengthinv}
\leavevmode
\begin{enumerate}
\item Si $\theta \colon [c, d] \to [a, b]$ es un
\omterm{def:b2:curves:reparam}{cambio de parámetro} $\mathcal{C}^1$, entonces $L(\gamma \circ \theta)
= L(\gamma)$.
\item $L(\gamma)$ es el supremo de las longitudes de los polígonos
inscritos:
\[
L(\gamma)
= \sup\Bigl\{\, \sum_{i=1}^{m} \norm{\gamma(t_i) -
\gamma(t_{i-1})} \;:\; a = t_0 < t_1 < \dots < t_m = b \,\Bigr\}.
\]
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} Por el cambio de variables $t = \theta(u)$ (volumen del
primer año, válido porque $\theta$ es $\mathcal{C}^1$ y monótona),
\[
\int_c^d \norm{(\gamma\circ\theta)'(u)}\,\dd u
= \int_c^d \norm{\gamma'(\theta(u))}\,\abs{\theta'(u)}\,\dd u
= \int_a^b \norm{\gamma'(t)}\,\dd t ,
\]
donde hemos usado $(\gamma\circ\theta)' = \theta'\cdot
(\gamma'\circ\theta)$ y, si $\theta$ es decreciente, el signo de
$\theta'$ queda absorbido por la inversión de los límites.

\emph{2.} Para toda subdivisión, $\gamma(t_i) - \gamma(t_{i-1}) =
\int_{t_{i-1}}^{t_i}\gamma'(t)\,\dd t$, de modo que, por la
desigualdad triangular para integrales, $\norm{\gamma(t_i) -
\gamma(t_{i-1})} \leq \int_{t_{i-1}}^{t_i}\norm{\gamma'}$: todo
polígono es más corto que $L(\gamma)$, luego $\sup \leq L(\gamma)$.

Para la desigualdad recíproca, sea $\varepsilon > 0$. Como
$\gamma'$ es \omterm{def:b2:metric:continuity}{continua} sobre el \omterm{def:b2:metric:compact}{compacto} $[a,b]$, es uniformemente
continua: hay un $\delta > 0$ con $\norm{\gamma'(t) - \gamma'(u)}
\leq \varepsilon$ siempre que $\abs{t - u} \leq \delta$. Tómese una
subdivisión de paso $\leq \delta$. En cada trozo, para $t \in
[t_{i-1}, t_i]$,
\[
\gamma(t_i) - \gamma(t_{i-1})
= \int_{t_{i-1}}^{t_i}\gamma'(t)\,\dd t
= (t_i - t_{i-1})\,\gamma'(t_{i-1}) + R_i,
\qquad
\norm{R_i} \leq \varepsilon\,(t_i - t_{i-1}),
\]
pues $R_i = \int_{t_{i-1}}^{t_i}(\gamma'(t) -
\gamma'(t_{i-1}))\,\dd t$. De ahí,
\[
\norm{\gamma(t_i) - \gamma(t_{i-1})}
\geq (t_i - t_{i-1})\norm{\gamma'(t_{i-1})}
- \varepsilon (t_i - t_{i-1}) .
\]
Sumando, y comparando $\sum (t_i -
t_{i-1})\norm{\gamma'(t_{i-1})}$ con $\int_a^b\norm{\gamma'}$ (una
suma de Riemann de la función \omterm{def:b2:metric:continuity}{continua} $\norm{\gamma'}$, a distancia
$\varepsilon(b - a)$ de la integral si $\delta$ es lo bastante
pequeño, de nuevo por la \omterm{def:b2:metric:continuity}{continuidad} uniforme), obtenemos un polígono
de \omterm{def:b2:curves:length}{longitud} $\geq L(\gamma) - 2\varepsilon(b - a)$. Haciendo
$\varepsilon \to 0$ queda demostrada la afirmación.
\end{proof}

\begin{example}[Arquímedes y los polígonos
inscritos]\label{ex:b2:curves:archimedes}
Para la circunferencia unidad, el $n$-ágono regular inscrito tiene
\omterm{def:b2:curves:length}{longitud} $L_n = 2n\sin\frac\pi n$, y el desarrollo $\sin x = x -
\frac{x^3}6 + O(x^5)$ da
\[
L_n = 2\pi - \frac{\pi^3}{3n^2} + O\Bigl(\frac1{n^4}\Bigr) :
\]
las longitudes poligonales del~\cref{thm:b2:curves:lengthinv}
convergen \emph{cuadráticamente}. Numéricamente: $L_6 = 6$ (el
hexágono, que da el burdo $\pi > 3$), mientras que $L_{96} =
192\sin\frac{\pi}{96} \approx 6.28206$ frente a $2\pi \approx
6.28319$; el error $0.00113$ concuerda con el $\pi^3/(3\cdot96^2)
\approx 0.00112$ previsto. Por eso Arquímedes, duplicando el
hexágono cinco veces hasta $96$ lados, pudo acotar $\pi$ con tres
cifras a mano: cada duplicación divide el error entre cuatro. El
supremo de la caracterización poligonal no solo se alcanza en el
límite; se alcanza \emph{deprisa}, porque una curva regular se
separa de sus cuerdas solo a segundo orden.
\end{example}

\begin{theorem}[Parametrización por longitud de arco]\label{thm:b2:curves:arclength}
Sea $\gamma \colon I \to \R^n$ un arco $\mathcal{C}^k$ regular ($k
\geq 1$). La \omterm{def:b2:curves:length}{función longitud de arco} $s$ es un difeomorfismo
$\mathcal{C}^k$ de $I$ sobre un intervalo $J$, y $\tilde\gamma =
\gamma \circ s^{-1}$ cumple $\norm{\tilde\gamma'} = 1$ en todas
partes. Salvo traslación del parámetro y orientación, esta
parametrización \emph{por \omterm{def:b2:curves:length}{longitud de arco}} (o \emph{de rapidez
unidad}) es única.
\end{theorem}

\begin{proof}
$s'(t) = \norm{\gamma'(t)} > 0$ y $s'$ es $\mathcal{C}^{k-1}$
(composición de la aplicación $\mathcal{C}^{k-1}$ $\gamma'$ con la
\omterm{def:b2:nvs:norm}{norma}, regular fuera de $0$), de modo que $s$ es $\mathcal{C}^k$,
estrictamente creciente, una biyección sobre $J = s(I)$, y su
inversa es $\mathcal{C}^k$ por el teorema de la función inversa en
dimensión $1$ (volumen del primer año). Entonces
\[
\tilde\gamma'(\sigma)
= \frac{\gamma'(t)}{s'(t)}
= \frac{\gamma'(t)}{\norm{\gamma'(t)}},
\qquad t = s^{-1}(\sigma),
\]
un vector \omterm{def:b2:hermitian:adjoint}{unitario}. Si $\hat\gamma = \gamma\circ\theta$ es otra
parametrización de rapidez unidad, entonces $\abs{\theta'} = 1$,
luego $\theta' = \pm 1$ constante (por \omterm{def:b2:metric:continuity}{continuidad}), es decir,
$\theta(u) = \pm u + c$.
\end{proof}

\begin{remark}
La \omterm{def:b2:curves:length}{longitud de arco} es el parámetro que separa la geometría de la
dinámica. Una trayectoria puede recorrerse con cualquier perfil de
rapidez ---la física del movimiento---, pero toda cuestión
invariante por parametrización (forma, flexión, osculación) tiene un
reloj canónico: la distancia recorrida. Por eso todas las fórmulas
de \omterm{thm:b2:curves:frenet2d}{curvatura} de más abajo se \emph{definen} a rapidez unidad y
después se \emph{traducen} a parametrizaciones arbitrarias mediante
la regla de la cadena: los factores de traducción son potencias de $v
= s'$, y seguirles la pista correctamente es todo el contenido de la~\cref{prop:b2:curves:kappaformula}.
\end{remark}

\begin{example}[Circunferencia y hélice]\label{ex:b2:curves:helix}
Para la circunferencia $\gamma(t) = (R\cos t, R\sin t)$,
$\norm{\gamma'} = R$, luego $s = Rt$ y la \omterm{def:b2:curves:length}{longitud} de una vuelta
completa es $2\pi R$. Para la hélice $\gamma(t) = (a\cos t,\ a\sin
t,\ bt)$ con $a > 0$, $\norm{\gamma'(t)} = \sqrt{a^2 + b^2}$ es
constante: la hélice se recorre a rapidez constante, y $s =
t\sqrt{a^2 + b^2}$.
\end{example}

\begin{example}[Longitud de arco en coordenadas
polares]\label{ex:b2:curves:polarlength}
Una curva polar $r = r(\theta)$ es el arco $\gamma(\theta) =
(r\cos\theta,\ r\sin\theta)$, con
\[
\gamma'(\theta) = (r'\cos\theta - r\sin\theta,\
r'\sin\theta + r\cos\theta),
\qquad
\norm{\gamma'(\theta)}^2 = r'^2 + r^2
\]
(los términos cruzados se cancelan): el elemento polar de
\omterm{def:b2:curves:length}{longitud} es $\sqrt{r^2 + r'^2}\,\dd\theta$. Para la cardioide $r =
1 + \cos\theta$:
\[
r^2 + r'^2 = (1 + \cos\theta)^2 + \sin^2\theta = 2 +
2\cos\theta = 4\cos^2\tfrac\theta2,
\]
y sobre $\intcc{-\pi}{\pi}$ el coseno del ángulo mitad es no
negativo, así que
\[
L = \int_{-\pi}^{\pi}2\cos\tfrac\theta2\,\dd\theta
= \Bigl[4\sin\tfrac\theta2\Bigr]_{-\pi}^{\pi}
= 4 - (-4) = 8 :
\]
como el arco de cicloide del~\cref{exo:b2:curves:1}, una curva
construida a partir de circunferencias tiene \omterm{def:b2:curves:length}{longitud} racional, sin
ningún $\pi$ por ninguna parte. La factorización con el ángulo mitad
es el truco estándar para las longitudes de curvas \omterm{def:b2:structures:generated}{generadas} por
circunferencias; cuando falla (la elipse), la \omterm{def:b2:curves:length}{longitud} es una
función genuinamente nueva: una integral elíptica, más allá de las
formas cerradas elementales.
\end{example}

\section{Curvatura en el plano}

En toda esta sección, los arcos son $\mathcal{C}^2$ y regulares en el
plano euclídeo orientado. Parametrizamos por \omterm{def:b2:curves:length}{longitud de arco} y
escribimos $T(s) = \tilde\gamma'(s)$ para la tangente unitaria, y
$N(s)$ para el vector \omterm{def:b2:hermitian:adjoint}{unitario} directamente ortogonal a $T(s)$
(rotación de $T$ por $+\pi/2$).

\begin{theorem}[Fórmulas de Frenet en el plano]\label{thm:b2:curves:frenet2d}
Sea $\tilde\gamma$ un arco $\mathcal{C}^2$ de rapidez unidad en el
plano orientado. Existe una función \omterm{def:b2:metric:continuity}{continua} $\kappa$, la
\emph{curvatura}\index{curvatura} (algebraica), tal que
\[
T'(s) = \kappa(s)\, N(s),
\qquad
N'(s) = -\kappa(s)\, T(s).
\]
\end{theorem}

\begin{proof}
Como $\norm{T(s)}^2 = 1$ para todo $s$, derivar el producto escalar
da $2\langle T'(s), T(s)\rangle = 0$: $T'(s)$ es ortogonal a $T(s)$,
luego colineal con $N(s)$ (dimensión $2$); escríbase $T'(s) =
\kappa(s) N(s)$ con $\kappa(s) = \langle T'(s), N(s)\rangle$,
\omterm{def:b2:metric:continuity}{continua}. Igualmente, $N' \perp N$, así que $N' = \lambda T$; y
derivar $\langle T, N\rangle = 0$ da $\langle T', N\rangle + \langle
T, N'\rangle = \kappa + \lambda = 0$.
\end{proof}

\begin{definition}\label{def:b2:curves:curvature}
Cuando $\kappa(s) \neq 0$, el \emph{radio de curvatura}\index{radio
de curvatura} es $R(s) = 1/\abs{\kappa(s)}$ y el \emph{centro de
curvatura}\index{centro de curvatura} es $\tilde\gamma(s) +
\frac{1}{\kappa(s)} N(s)$; la circunferencia con ese centro y radio
$R(s)$ es la \emph{circunferencia osculatriz}\index{circunferencia
osculatriz}, la mejor aproximación circular de la curva en
$\tilde\gamma(s)$.
\end{definition}

\begin{example}[La circunferencia osculatriz de la
exponencial]\label{ex:b2:curves:oscexp}
Para $y = \eu^x$ en el punto $(0, 1)$: $f'(0) = f''(0) = 1$, así
que, por la fórmula para grafos de más abajo,
\[
\kappa(0) = \frac{1}{(1 + 1)^{3/2}} = \frac1{2\sqrt2},
\qquad R = 2\sqrt2 .
\]
La tangente unitaria es $T = \frac{(1, 1)}{\sqrt2}$, la normal
directa $N = \frac{(-1, 1)}{\sqrt2}$, y el \omterm{def:b2:curves:curvature}{centro de curvatura} es
\[
(0, 1) + 2\sqrt2\cdot\frac{(-1, 1)}{\sqrt2} = (-2,\ 3) :
\]
la \omterm{def:b2:curves:curvature}{circunferencia osculatriz} tiene ecuación $(x + 2)^2 + (y - 3)^2
= 8$. Como comprobación de que es ``la mejor aproximación
circular'': resolver la ecuación de la circunferencia en $y$ cerca de
$(0,1)$ y desarrollar da $y = 1 + x + \frac{x^2}2 + O(x^3)$,
exactamente el desarrollo de Taylor de segundo orden de $\eu^x$. La
\omterm{def:b2:curves:curvature}{circunferencia osculatriz} ajusta el valor, la pendiente \emph{y} la
derivada segunda; una circunferencia tangente ordinaria solo
ajustaría las dos primeras.
\end{example}

\begin{example}[La evoluta de una circunferencia es su
centro]\label{ex:b2:curves:circleevolute}
Para la circunferencia de radio $R$ recorrida en sentido
antihorario, $\kappa = 1/R$ y $N$ apunta hacia el centro, de modo que
el \omterm{def:b2:curves:curvature}{centro de curvatura} $\tilde\gamma + \frac1\kappa N$ es el centro
de la circunferencia, para todo $s$: la \omterm{def:b2:curves:curvature}{circunferencia osculatriz}
de una circunferencia es ella misma, y el lugar de los \omterm{def:b2:curves:curvature}{centros de
curvatura} colapsa en un punto. Este caso degenerado calibra el~\cref{exo:b2:curves:6}: allí la velocidad de la evoluta es
$-\frac{\kappa'}{\kappa^2}N$, que se anula idénticamente justo
cuando $\kappa$ es constante.
\end{example}

\begin{proposition}[Curvatura en una parametrización
arbitraria]\label{prop:b2:curves:kappaformula}
Para un arco plano $\mathcal{C}^2$ regular $\gamma(t) = (x(t),
y(t))$,
\[
\kappa(t)
= \frac{x'(t)\,y''(t) - y'(t)\,x''(t)}
       {\bigl(x'(t)^2 + y'(t)^2\bigr)^{3/2}} ,
\]
en particular $\kappa = \dfrac{y''}{(1 + y'^2)^{3/2}}$ para un grafo
$y = f(x)$.
\end{proposition}

\begin{proof}
Escríbase $v(t) = \norm{\gamma'(t)} = s'(t)$, de modo que $\gamma' =
vT$ (componiendo los datos de rapidez unidad con $s$). Derivando,
\[
\gamma'' = v'T + v\,T'\cdot s' = v'T + v^2\kappa N .
\]
Tómese ahora el \omterm{def:b2:linalg:det}{determinante} (en la base canónica orientada) de
$(\gamma', \gamma'')$: como $\det(T, T) = 0$ y $\det(T, N) = 1$,
\[
\det(\gamma', \gamma'') = \det(vT,\ v'T + v^2\kappa N)
= v^3\kappa .
\]
El miembro izquierdo es $x'y'' - y'x''$, y $v^3 = (x'^2 +
y'^2)^{3/2}$. El caso del grafo es la parametrización $t \mapsto (t,
f(t))$.
\end{proof}

\begin{example}[Circunferencia, recta, parábola]\label{ex:b2:curves:kappaexamples}
Una recta tiene $\kappa = 0$ (y recíprocamente: $T' = 0$ significa
$T$ constante, luego $\tilde\gamma(s) = \tilde\gamma(0) + sT$, una
recta). La circunferencia de radio $R$ recorrida en sentido
antihorario tiene $\kappa = 1/R$: con $\gamma(t) = (R\cos t, R\sin
t)$, la fórmula da $\kappa = R^2/R^3$. Para la parábola $y = x^2/2$:
$\kappa(x) = 1/(1 + x^2)^{3/2}$, máxima en el vértice; la parábola se
dobla más bruscamente donde da la vuelta.
\end{example}

\begin{remark}[Errores frecuentes en torno a la curvatura]
(i) La \omterm{thm:b2:curves:frenet2d}{curvatura} algebraica de un arco plano cambia de signo al
invertir la orientación del arco o del plano: solo $\abs\kappa$ y $R
= 1/\abs\kappa$ son puramente geométricos. Una circunferencia
recorrida en sentido horario tiene $\kappa = -1/R$. (ii) La fórmula
para grafos $\kappa = f''/(1 + f'^2)^{3/2}$ elige en silencio la
parametrización por $x$; aplicarla a una curva que no sea un grafo
cerca del punto (tangente vertical) es el error clásico. (iii) En un
punto donde $\gamma' = 0$ no hay nada definido ---ni $T$ ni
$\kappa$--- y la trayectoria puede romperse de verdad
(\cref{ex:b2:curves:cusp}); compruébese siempre la regularidad antes
de derivar la tangente unitaria. (iv) En el espacio, $\kappa =
\norm{T'} \geq 0$ por convenio: no hay signo que equivocar, pero
tampoco signo que explotar; la información de tipo inflexión se
traslada a la \omterm{thm:b2:curves:frenet3d}{torsión}. (v) Por último, $\kappa$ es una derivada
\emph{respecto de la \omterm{def:b2:curves:length}{longitud de arco}}: para una parametrización que
no sea de rapidez unidad, olvidar el factor $v^3$ de la~\cref{prop:b2:curves:kappaformula} es el error más frecuente en la
práctica.
\end{remark}

\begin{theorem}[La curvatura determina la curva]\label{thm:b2:curves:fundamental}
Sea $\kappa \colon J \to \R$ \omterm{def:b2:metric:continuity}{continua}. Existe un arco $\mathcal{C}^2$
de rapidez unidad en el plano con \omterm{thm:b2:curves:frenet2d}{curvatura} $\kappa$, y es único
salvo isometría directa (una rotación seguida de una traslación).
\end{theorem}

\begin{proof}
\emph{Existencia.} Fíjese $s_0 \in J$ y póngase $\varphi(s) =
\int_{s_0}^s \kappa(u)\,\dd u$; después,
\[
\tilde\gamma(s) = \Bigl(\int_{s_0}^s \cos\varphi(u)\,\dd u,\
\int_{s_0}^s \sin\varphi(u)\,\dd u\Bigr).
\]
Entonces $T(s) = (\cos\varphi(s), \sin\varphi(s))$ es un vector
\omterm{def:b2:hermitian:adjoint}{unitario}, $N(s) = (-\sin\varphi, \cos\varphi)$, y
\[
T'(s) = \varphi'(s)\,(-\sin\varphi, \cos\varphi)
= \kappa(s)\,N(s) :
\]
el arco es de rapidez unidad y de \omterm{thm:b2:curves:frenet2d}{curvatura} $\kappa$.

\emph{Unicidad.} Sean $\gamma_1, \gamma_2$ arcos de rapidez unidad
con la misma \omterm{thm:b2:curves:frenet2d}{curvatura}. Cada tangente unitaria se levanta a una
función ángulo, mediante una construcción explícita: véase $T_j$
como el número complejo $z_j = a_j + \iu b_j$ de módulo $1$, elíjase
$\varphi_j(0)$ con $z_j(0) = \eu^{\iu\varphi_j(0)}$ y póngase
\[
\varphi_j(s) = \varphi_j(0) + \int_0^s\det\bigl(T_j,
T_j'\bigr)(u)\,\dd u .
\]
De $\abs{z_j} = 1$ se sigue $\operatorname{Re}(\conj{z_j}\,z_j') =
0$, luego $\conj{z_j}\,z_j' = \iu\det(T_j, T_j') = \iu\,\varphi_j'$,
es decir, $z_j' = \iu\varphi_j'z_j$; entonces
\[
\bigl(z_j\,\eu^{-\iu\varphi_j}\bigr)'
= \eu^{-\iu\varphi_j}\bigl(z_j' - \iu\varphi_j'z_j\bigr) = 0,
\]
de modo que $z_j = \eu^{\iu\varphi_j}$ en todas partes: $T_j =
(\cos\varphi_j, \sin\varphi_j)$ con $\varphi_j$ de clase $\mathcal
C^1$. Además, $\det(T_j, T_j') = \det(T_j, \kappa N_j) = \kappa$,
luego $\varphi_j' = \kappa$. Por tanto, $\varphi_2 = \varphi_1 + c$
para una constante $c$: $T_2$ es $T_1$ girado el ángulo fijo $c$, de
modo que, integrando, $\gamma_2 = \rho(\gamma_1) + w$, donde $\rho$
es la rotación de ángulo $c$ y $w$ un vector constante.
\end{proof}

\begin{remark}
Este es el prototipo unidimensional de un \emph{teorema fundamental
de la geometría}: un conjunto \omterm{def:b2:metric:complete}{completo} de invariantes locales (aquí,
una función) clasifica el objeto salvo movimiento rígido. La versión
tridimensional de más abajo necesita dos invariantes.
\end{remark}

\begin{example}[Curvatura constante significa
circunferencia]\label{ex:b2:curves:constantkappa}
Tómese $\kappa \equiv \kappa_0 > 0$ en la fórmula de existencia:
$\varphi(s) = \kappa_0 s$ y
\[
\tilde\gamma(s) = \Bigl(\frac{\sin\kappa_0s}{\kappa_0},\
\frac{1 - \cos\kappa_0s}{\kappa_0}\Bigr) :
\]
la circunferencia de radio $1/\kappa_0$ centrada en $(0,
1/\kappa_0)$, recorrida a rapidez unidad. Por la mitad de unicidad
del teorema, \emph{todo} arco de rapidez unidad de \omterm{thm:b2:curves:frenet2d}{curvatura}
constante $\kappa_0$ es un trozo de circunferencia de radio
$1/\kappa_0$ (o una recta si $\kappa_0 = 0$): el recíproco del
cálculo del~\cref{ex:b2:curves:kappaexamples}, y el caso plano del~\cref{exo:b2:curves:9}.
\end{example}

\begin{example}[Reconstruir una curva a partir de su
curvatura]\label{ex:b2:curves:reconstruction}
¿Qué curva de rapidez unidad tiene \omterm{def:b2:curves:curvature}{radio de curvatura} $R(s) = 1 +
s^2$? Siguiendo la demostración de la existencia con $\kappa(s) =
\frac1{1+s^2}$ y $s_0 = 0$: $\varphi(s) = \arctan s$, luego
\[
T(s) = (\cos\arctan s,\ \sin\arctan s)
= \Bigl(\frac{1}{\sqrt{1+s^2}},\
\frac{s}{\sqrt{1+s^2}}\Bigr),
\]
e integrando,
\[
\tilde\gamma(s) = \Bigl(\ln\bigl(s + \sqrt{1 + s^2}\bigr),\
\sqrt{1 + s^2} - 1\Bigr).
\]
Poniendo $x = \ln(s + \sqrt{1+s^2})$, es decir, $s = \sinh x$, la
segunda coordenada es $\cosh x - 1$: la curva es la
\emph{catenaria} $y = \cosh x - 1$. Esto cierra el círculo con el~\cref{exo:b2:curves:3}, donde calculamos $R = \cosh^2 x = 1 + \sinh^2
x = 1 + s^2$ directamente: el teorema fundamental garantiza que la
catenaria es la \emph{única} curva con este perfil de \omterm{thm:b2:curves:frenet2d}{curvatura},
salvo isometría directa.
\end{example}

\begin{remark}[Dónde se usa la curvatura a continuación]
La descomposición $\gamma'' = v'T + v^2\kappa N$ obtenida en la
demostración de la~\cref{prop:b2:curves:kappaformula} es la
cinemática de todo movimiento curvo: aceleración tangencial frente a
centrípeta. La \omterm{thm:b2:curves:frenet2d}{curvatura} regresa con las superficies
(\cref{ch:b2:surfaces}) a través de la \omterm{thm:b2:curves:frenet2d}{curvatura} de las curvas
trazadas sobre ellas, y el cálculo de \omterm{pb:b2:curves:1}{envolventes} del problema de fin
de semana de este capítulo ---evolutas, cáusticas--- es la óptica
geométrica de los frentes de onda. El volumen del tercer año retoma
el punto de vista intrínseco para las subvariedades de $\R^n$.
\end{remark}

\section{Triedro de Frenet en el espacio}

Sea ahora $\tilde\gamma \colon J \to \R^3$ un arco $\mathcal{C}^3$ de
rapidez unidad y \emph{birregular}: $T'(s) \neq 0$ para todo $s$.
Entonces $\kappa(s) = \norm{T'(s)} > 0$ define la
\emph{\omterm{thm:b2:curves:frenet2d}{curvatura}} (sin signo en el espacio: no hay orientación normal
privilegiada), y ponemos:
\[
N(s) = \frac{T'(s)}{\kappa(s)}
\quad\text{(normal principal)},
\qquad
B(s) = T(s) \wedge N(s)
\quad\text{(binormal)} ,
\]
de modo que $(T, N, B)$ es un triedro ortonormal directo, el
\emph{triedro de Frenet}. El plano que pasa por $\tilde\gamma(s)$ y
está \omterm{def:b2:structures:generated}{generado} por $T, N$ es el \emph{plano osculador}.

\begin{theorem}[Fórmulas de Frenet en el espacio]\label{thm:b2:curves:frenet3d}
Existe una función \omterm{def:b2:metric:continuity}{continua} $\tau$, la \emph{torsión}\index{torsión},
con
\[
T' = \kappa N, \qquad
N' = -\kappa T + \tau B, \qquad
B' = -\tau N .
\]
\end{theorem}

\begin{proof}
La primera fórmula es la definición de $N$. Cada uno de los vectores
$T, N, B$ tiene \omterm{def:b2:nvs:norm}{norma} constante $1$ y son ortogonales dos a dos;
derivar las seis relaciones $\langle X, Y\rangle = \delta_{XY}$
muestra que la matriz de $(T', N', B')$ en la base $(T, N, B)$ es
antisimétrica: en efecto, $\langle X', Y\rangle + \langle X,
Y'\rangle = 0$ y $\langle X', X\rangle = 0$. Su entrada $(N, T)$ es
$\langle N', T\rangle = -\langle N, T'\rangle = -\kappa$, y la
entrada de la columna $(T, B)$ es $\langle T', B\rangle =
\kappa\langle N, B\rangle = 0$. Llamando $\tau = \langle N',
B\rangle$ a la entrada libre restante se obtienen exactamente las
tres fórmulas mostradas: la antisimetría completa $\langle B',
N\rangle = -\tau$ y $\langle B', T\rangle = 0$. La \omterm{def:b2:metric:continuity}{continuidad} de
$\tau = \langle N', B\rangle$ es clara, pues $N'$ y $B$ son
\omterm{def:b2:metric:continuity}{continuas} ($\tilde\gamma$ es $\mathcal{C}^3$, luego $N = T'/\kappa$ es
$\mathcal{C}^1$).
\end{proof}

\begin{example}[El vector de Darboux]\label{ex:b2:curves:darboux}
Las tres fórmulas de Frenet se comprimen en una. Póngase $\omega(s)
= \tau\,T + \kappa\,B$ (el \emph{vector de Darboux}). Usando $B
\wedge T = N$, $T \wedge N = B$, $N \wedge B = T$:
\[
\omega \wedge T = \kappa\,N = T', \qquad
\omega \wedge N = \tau\,B - \kappa\,T = N', \qquad
\omega \wedge B = -\tau\,N = B' :
\]
cada vector del triedro evoluciona según $X' = \omega \wedge X$, la
firma cinemática de una \emph{rotación instantánea} de vector de
velocidad angular $\omega$. El triedro gira a razón de $\norm\omega =
\sqrt{\kappa^2 + \tau^2}$ en torno al eje móvil $\omega$; la
\omterm{thm:b2:curves:frenet2d}{curvatura} es la componente del giro en torno a la binormal, y la
\omterm{thm:b2:curves:frenet3d}{torsión}, la componente en torno a la tangente. Para la hélice,
$\omega$ es un vector constante a lo largo del eje del cilindro, y
por eso el triedro de la hélice precesa de manera uniforme. La
antisimetría de la matriz de Frenet, explotada en el~\cref{exo:b2:curves:7}, es la forma matricial de este único hecho
geométrico.
\end{example}

\begin{proposition}[La torsión mide la planitud]\label{prop:b2:curves:torsion}
Un arco birregular está contenido en un plano si y solo si $\tau
\equiv 0$; y en ese caso el plano es el plano osculador
(constante).
\end{proposition}

\begin{proof}
Si $\tau \equiv 0$, entonces $B' = 0$, luego $B$ es un vector
\omterm{def:b2:hermitian:adjoint}{unitario} constante $B_0$, y
\[
\frac{\dd}{\dd s}\langle \tilde\gamma(s), B_0\rangle
= \langle T(s), B_0\rangle = 0 :
\]
$\langle \tilde\gamma, B_0\rangle$ es constante, de modo que el arco
está en un plano ortogonal a $B_0$. Recíprocamente, si el arco está
en un plano $P$, entonces $T$ y $T'$ (luego $N$) son paralelos a la
dirección de $P$ para todo $s$; así que $B = T \wedge N$ es una de
las dos normales unitarias de $P$ y, al ser \omterm{def:b2:metric:continuity}{continua}, es constante;
entonces $0 = B' = -\tau N$ con $N \neq 0$ fuerza $\tau \equiv 0$.
\end{proof}

\begin{example}[Una circunferencia inclinada tiene torsión nula]
El arco $\gamma(t) = \bigl(\cos t,\ \tfrac{\sin t}{\sqrt2},\
\tfrac{\sin t}{\sqrt2}\bigr)$ está en el plano $y = z$ y es la
circunferencia unidad de ese plano (compruébese: $\norm{\gamma(t)} =
1$, y la base ortonormal del plano $(1,0,0)$, $(0,
\tfrac1{\sqrt2}, \tfrac1{\sqrt2})$ exhibe la parametrización
estándar). Sin ningún cálculo de Frenet, la~\cref{prop:b2:curves:torsion} predice $\tau \equiv 0$, y la binormal
fija debe ser la normal \omterm{def:b2:hermitian:adjoint}{unitaria} del plano $\pm(0, \tfrac1{\sqrt2},
-\tfrac1{\sqrt2})$. La \omterm{thm:b2:curves:frenet3d}{torsión} no mide estar ``inclinado en el
espacio''; mide \emph{abandonar} un plano. Solo el $b$ no nulo de la
hélice de más abajo produce \omterm{thm:b2:curves:frenet3d}{torsión} genuina.
\end{example}

\begin{example}[La hélice]\label{ex:b2:curves:helixfrenet}
Para la hélice $\gamma(t) = (a\cos t, a\sin t, bt)$, con $a > 0$,
calculamos $s = ct$ con $c = \sqrt{a^2 + b^2}$. Entonces
\[
T = \frac1c(-a\sin t,\ a\cos t,\ b),
\qquad
T' \cdot \frac{\dd t}{\dd s}
= \frac{1}{c^2}(-a\cos t, -a\sin t, 0),
\]
luego $\kappa = a/c^2 = a/(a^2 + b^2)$ y $N = (-\cos t, -\sin t,
0)$: la normal principal apunta horizontalmente hacia el eje.
Después, $B = T \wedge N = \frac1c(b\sin t, -b\cos t, a)$, y $B'
\frac{\dd t}{\dd s} = \frac{b}{c^2}(\cos t, \sin t, 0) = -\tau N$
da
\[
\boxed{\ \kappa = \frac{a}{a^2 + b^2}, \qquad
\tau = \frac{b}{a^2 + b^2}. \ }
\]
Ambos invariantes son constantes; y puede probarse, recíprocamente,
que las únicas curvas birregulares con $\kappa > 0$ constante y
$\tau$ constante son las hélices (circunferencias cuando $\tau = 0$).
Obsérvense los signos: $b > 0$ da una hélice dextrógira, de
\omterm{thm:b2:curves:frenet3d}{torsión} positiva.
\end{example}

\begin{example}[Curvatura y torsión sin longitud de arco: la cúbica
alabeada]\label{ex:b2:curves:twistedcubic}
Reparametrizar por \omterm{def:b2:curves:length}{longitud de arco} suele ser imposible en forma
cerrada, así que los invariantes deben extraerse de las derivadas en
bruto. Escríbase $v = \norm{\gamma'} = s'$; entonces $\gamma' = vT$
y, como en la demostración de la~\cref{prop:b2:curves:kappaformula},
\[
\gamma'' = v'T + v^2\kappa N,
\qquad
\gamma' \wedge \gamma'' = v^3\kappa\,(T \wedge N) =
v^3\kappa\,B .
\]
Tomando \omterm{def:b2:nvs:norm}{normas} ($\kappa \geq 0$ en el espacio):
\[
\kappa = \frac{\norm{\gamma' \wedge \gamma''}}{v^3} .
\]
Derivando $\gamma''$ una vez más y convirtiendo $N' = v(-\kappa T +
\tau B)$ (regla de la cadena a través de $s$), la única componente en
$B$ procede del último término:
\[
\gamma''' = \bigl(v'' - v^3\kappa^2\bigr)T +
\bigl(v'v\kappa + (v^2\kappa)'\bigr)N + v^3\kappa\tau\,B,
\]
de modo que, emparejando con $\gamma' \wedge \gamma'' = v^3\kappa B$,
\[
\det(\gamma', \gamma'', \gamma''') = \langle\gamma' \wedge
\gamma'',\ \gamma'''\rangle = v^6\kappa^2\tau,
\qquad\text{es decir,}\qquad
\tau = \frac{\det(\gamma', \gamma'',
\gamma''')}{\norm{\gamma' \wedge \gamma''}^2} .
\]
Aplicación a la \emph{cúbica alabeada} $\gamma(t) = (t,\ t^2,\ t^3)$
en $t = 0$: $\gamma' = (1, 0, 0)$, $\gamma'' = (0, 2, 0)$, $\gamma'''
= (0, 0, 6)$, luego $v = 1$,
\[
\gamma' \wedge \gamma'' = (0, 0, 2), \qquad \kappa(0) = 2,
\qquad
\det(\gamma', \gamma'', \gamma''') = 12, \qquad \tau(0) =
\frac{12}{4} = 3 .
\]
Moraleja: ambas fórmulas son cocientes en los que la rapidez $v$ se
cancela exactamente en el grado necesario ---$\kappa$ escala como una
derivada segunda por unidad de longitud, y $\tau$ como el volumen
mixto de tres derivadas por área al cuadrado---, y \emph{por eso} son
geométricas mientras que $\gamma''$ no lo es.
\end{example}

\begin{remark}[Teorema fundamental para las curvas del espacio]
Como en el plano, el par $(\kappa, \tau)$ con $\kappa > 0$ determina
un arco birregular salvo isometría directa de $\R^3$: las fórmulas de
Frenet forman un sistema diferencial lineal para el triedro $(T, N,
B)$, al que se aplica la teoría de Cauchy--Lipschitz del~\cref{ch:b2:diffeq}; la ortonormalidad del triedro solución se
conserva porque la matriz de coeficientes es antisimétrica (el mismo
argumento de la matriz de Gram del~\cref{exo:b2:curves:7}), y la
curva se recupera integrando $T$. Dejamos los detalles al lector como
ejercicio sustancial pero instructivo.
\end{remark}

\section{Estudio local: posición respecto de la tangente}

\begin{proposition}[Forma local en un punto
regular]\label{prop:b2:curves:local}
Sea $\gamma$ un arco plano de clase $\mathcal{C}^k$ en $t_0$, con $p$
el menor índice tal que $\gamma^{(p)}(t_0) \neq 0$ y $q$ el menor
índice $> p$ tal que $\gamma^{(q)}(t_0)$ no sea colineal con
$\gamma^{(p)}(t_0)$ (suponiendo que ambos existen, $q \leq k$). En la
base $(u, v) = (\gamma^{(p)}(t_0), \gamma^{(q)}(t_0))$ centrada en
$\gamma(t_0)$, Taylor--Young da las coordenadas
\[
X(t) \sim \frac{(t - t_0)^p}{p!},
\qquad
Y(t) \sim \frac{(t - t_0)^q}{q!} .
\]
La imagen local depende solo de las paridades de $p$ y $q$:
\begin{center}
\begin{tabular}{c c p{5.1cm}}
$p$ impar, $q$ par & \emph{punto ordinario} & la curva no cruza
ninguna recta, queda a un lado de la tangente\\
$p$ impar, $q$ impar & \emph{punto de inflexión} & la curva cruza su
tangente\\
$p$ par, $q$ impar & \emph{cúspide de primera especie} & las dos
ramas a lados opuestos de la tangente\\
$p$ par, $q$ par & \emph{cúspide de segunda especie} & las dos ramas
al mismo lado\\
\end{tabular}
\end{center}
\end{proposition}

\begin{proof}
Taylor--Young de orden $q$ (la función $\gamma$ es $\mathcal{C}^q$
cerca de $t_0$):
\[
\gamma(t) - \gamma(t_0)
= \sum_{j=p}^{q} \frac{(t-t_0)^j}{j!}\,\gamma^{(j)}(t_0)
+ o\bigl((t-t_0)^q\bigr).
\]
Por la elección de $p$ y $q$, cada $\gamma^{(j)}(t_0)$ con $p \leq j
< q$ es colineal con $u$; recogiendo componentes en la base $(u, v)$:
$X(t) = \frac{(t-t_0)^p}{p!}(1 + o(1))$ e $Y(t) =
\frac{(t-t_0)^q}{q!}(1 + o(1))$. La tabla de signos de $X$ e $Y$ para
$t \gtrless t_0$ ---gobernada exactamente por las paridades--- da las
cuatro imágenes: por ejemplo, si $p$ es par, $X > 0$ a ambos lados
(las dos ramas salen en la dirección $+u$: una cúspide), y el lado de
la \omterm{def:b2:curves:arc}{recta tangente} ($\operatorname{signo} Y$) se invierte con $q$
impar.
\end{proof}

\begin{example}
Para $\gamma(t) = (t^2, t^3)$ en $t_0 = 0$
(\cref{ex:b2:curves:cusp}): $\gamma'' (0)= (2, 0)$, $\gamma'''(0) =
(0, 6)$, luego $p = 2$, $q = 3$: una cúspide de primera especie, la
imagen familiar de la parábola semicúbica. Para $\gamma(t) = (t,
t^3)$ en $0$: $p = 1$, $q = 3$: inflexión; la cúbica cruza su
tangente.
\end{example}

\begin{remark}[Perspectivas dentro de este volumen]
Las curvas alimentan los capítulos siguientes de tres maneras.
Trazadas sobre una superficie, definen sus planos tangentes y su
primera forma fundamental (\cref{ch:b2:surfaces}), y sus \omterm{def:b2:curves:length}{longitudes}
se calculan restringiendo la métrica ambiente: el capítulo que viene
es, en buena medida, este capítulo relativizado. El cálculo de
\omterm{pb:b2:curves:1}{envolventes} del problema de fin de semana se encuentra con las
integrales dobles en el~\cref{ch:b2:multint}, donde el área de la
\omterm{pb:b2:curves:1}{astroide} se recalcula con la fórmula de Green
(\cref{exo:b2:multint:5}): una curva, dos teorías, respuestas que
coinciden. Y el sistema de Frenet ya usó las ecuaciones diferenciales
lineales del~\cref{ch:b2:diffeq} (existencia, unicidad y el argumento
de conservación de la ortogonalidad del~\cref{exo:b2:curves:7}): el
teorema fundamental de las curvas es un teorema de ecuaciones
diferenciales vestido de geometría.
\end{remark}

\begin{remark}[Método: ejecutar el estudio local]
En la práctica, la clasificación es una rutina de cuatro pasos.
\emph{Uno}, derívese en $t_0$ hasta que aparezca la primera derivada
no nula: su índice es $p$ y su valor el vector $u$. \emph{Dos}, sígase
derivando hasta que aparezca una derivada no colineal con $u$:
índice $q$, vector $v$. \emph{Tres}, léanse las paridades $(p, q)$ en
la tabla. \emph{Cuatro}, dibújese: la curva sale a lo largo de $+u$
si $p$ es impar (a lo largo de $u$ y luego de vuelta a lo largo de
$u$ si $p$ es par), al lado de $v$ que dicte el signo de $Y$. Dos
advertencias. La base $(u, v)$ en general \emph{no} es ortonormal: la
tabla describe posiciones relativas a la \omterm{def:b2:curves:arc}{recta tangente}, no ángulos
ni distancias, así que no se lea la \omterm{thm:b2:curves:frenet2d}{curvatura} en el dibujo. Y se
permiten derivadas intermedias colineales con $u$ entre los rangos
$p$ y $q$ (solo desplazan el desarrollo de $X$); lo que \emph{no}
debe hacerse es detenerse en la primera derivada no nula y suponer $q
= p + 1$: para $\gamma(t) = (t^2, t^4 + t^5)$, la suposición ingenua
$q = 3$ es falsa, pues $\gamma^{(3)}(0)$ sigue siendo colineal con
$\gamma''(0)$; esto es precisamente el~\cref{exo:b2:curves:5}.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:curves:1}
Calcula la \omterm{def:b2:curves:length}{longitud} de un arco de la cicloide $\gamma(t) = (t -
\sin t,\ 1 - \cos t)$, $t \in [0, 2\pi]$. \emph{(Usa $1 - \cos t =
2\sin^2(t/2)$.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:curves:2}
Calcula la \omterm{thm:b2:curves:frenet2d}{curvatura} de la elipse $\gamma(t) = (a\cos t,\ b\sin t)$
($a > b > 0$) y localiza los puntos de \omterm{thm:b2:curves:frenet2d}{curvatura} máxima y mínima.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:curves:3}
Prueba que la \omterm{def:b2:curves:length}{longitud de arco} del grafo de $f(x) = \cosh x$ sobre
$[0, x]$ vale $\sinh x$, y calcula la \omterm{thm:b2:curves:frenet2d}{curvatura} de esa curva (la
\emph{catenaria}). Comprueba que $R(x) = 1/\kappa(x) = \cosh^2 x$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:curves:4}
(Espiral logarítmica) Sea $\gamma(t) = e^{t}(\cos t,\ \sin t)$, $t
\in \R$. Prueba que el ángulo entre $\gamma(t)$ y $\gamma'(t)$ es
constante, calcula la \omterm{def:b2:curves:length}{longitud de arco} de $\gamma$ sobre $(-\infty,
0]$ (¡finita!) y la \omterm{thm:b2:curves:frenet2d}{curvatura}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:curves:5}
Determina $p$, $q$ y la forma local (ordinaria, inflexión, cúspide)
de $\gamma(t) = (t^2,\ t^4 + t^5)$ en $t = 0$, y de $\gamma(t) =
(t^3,\ t^4)$ en $t = 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:curves:6}
Sea $\gamma$ un arco plano de rapidez unidad con $\kappa(s) > 0$ para
todo $s$, y sea $c(s) = \gamma(s) + \frac{1}{\kappa(s)}N(s)$ el
\omterm{def:b2:curves:curvature}{centro de curvatura} (la curva $c$ es la \emph{evoluta}). Suponiendo
que $\kappa$ es $\mathcal{C}^1$, prueba que $c'(s) =
-\frac{\kappa'(s)}{\kappa(s)^2}N(s)$: la evoluta es tangente a las
rectas normales de $\gamma$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:curves:7}
Sea $A(s)$ una familia \omterm{def:b2:metric:continuity}{continua} de matrices antisimétricas $3 \times
3$ y sea $F' = F A$ una solución matricial con $F(s_0)$ ortogonal.
Prueba que $F(s)$ es ortogonal para todo $s$. \emph{(Deriva $G = F
F^{\mathsf T}$ y usa la unicidad en Cauchy--Lipschitz.)} Explica su
relevancia para el sistema de Frenet.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:curves:8}
(\omterm{thm:b2:curves:frenet2d}{Curvatura} total de una curva cerrada convexa) Sea $\tilde\gamma$
un arco plano cerrado $\mathcal{C}^2$ de rapidez unidad y de
\omterm{def:b2:curves:length}{longitud} $L$ (de modo que $\tilde\gamma(s + L) = \tilde\gamma(s)$),
recorrido una vez en sentido antihorario. Usando la función ángulo
$\varphi$ con $T = (\cos\varphi, \sin\varphi)$ del~\cref{thm:b2:curves:fundamental}, explica por qué $\varphi(L) -
\varphi(0)$ es un múltiplo de $2\pi$, y prueba que $\int_0^L
\kappa(s)\,\dd s = \varphi(L) - \varphi(0)$. (Para una
circunferencia de radio $R$: $\int \kappa = \frac1R \cdot 2\pi R =
2\pi$. El teorema de las tangentes giratorias afirma el valor $2\pi$
para toda curva cerrada simple; no se pide demostrarlo.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:curves:9}
Prueba que una curva birregular del espacio con $\kappa > 0$
constante y $\tau = 0$ es (un arco de) una circunferencia de radio
$1/\kappa$. \emph{(Usa la~\cref{prop:b2:curves:torsion} y prueba
después que el centro $\gamma + \frac1\kappa N$ es constante.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:curves:10}
Calcula la \omterm{def:b2:curves:length}{longitud de arco} de la parábola $y = x^2/2$ sobre
$\intcc0a$ y prueba que vale
\[
\tfrac12\Bigl(a\sqrt{1 + a^2} + \ln\bigl(a + \sqrt{1 +
a^2}\bigr)\Bigr).
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:curves:11}
Sea $\gamma$ un arco $\mathcal C^2$ regular de $\R^n$ cuyas
\omterm{def:b2:curves:arc}{rectas tangentes} pasan todas por un punto fijo $P$. Demuestra que la
trayectoria de $\gamma$ está contenida en una recta.
\emph{(Parametriza por \omterm{def:b2:curves:length}{longitud de arco}, escribe $\gamma(s) +
\lambda(s)T(s) = P$ y deriva.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:curves:12}
(Teorema fundamental para las curvas del espacio) Sean $\kappa > 0$ y
$\tau$ funciones \omterm{def:b2:metric:continuity}{continuas} sobre un intervalo $J$. Lleva a cabo el
programa de la observación que sigue al~\cref{ex:b2:curves:helixfrenet}: (a) prueba que el sistema lineal $F'
= FA(s)$, con $A(s)$ la matriz antisimétrica de Frenet construida a
partir de $\kappa, \tau$ y $F(s_0)$ un triedro ortonormal directo,
tiene una única solución global, que sigue siendo un triedro
ortonormal directo; (b) construye una curva birregular de rapidez
unidad con \omterm{thm:b2:curves:frenet2d}{curvatura} $\kappa$ y \omterm{thm:b2:curves:frenet3d}{torsión} $\tau$; (c) demuestra la
unicidad salvo isometría directa de $\R^3$.
\end{exercise}

\section{Problema: envolventes --- la astroide, dos evolutas y una
cáustica}

\begin{problem}[{Problema de fin de semana --- la máquina de
envolventes y cuatro curvas clásicas}]\label{pb:b2:curves:1}
Una familia uniparamétrica de rectas no suele cubrir el plano de
manera uniforme: las rectas se amontonan a lo largo de una curva
tangente a todas ellas, su \emph{envolvente}\index{envolvente}. Los
rayos de luz hacen visibles las \omterm{pb:b2:curves:1}{envolventes} en forma de
\emph{cáusticas}: la curva brillante con cúspides que se ve en una
taza de café. Este problema construye la máquina general de
\omterm{pb:b2:curves:1}{envolventes} y después la ejecuta cuatro veces: la escalera que
resbala (\omterm{pb:b2:curves:1}{astroide}), las normales de la parábola y de la cicloide
(evolutas, con el péndulo de Huygens al final) y la cáustica de la
taza de café (\omterm{pb:b2:curves:1}{nefroide}). En todo el problema, $D_t$ denota la recta
de ecuación $a(t)\,x + b(t)\,y = c(t)$, donde $a, b, c$ son funciones
$\mathcal C^2$ con $(a(t), b(t)) \neq (0,0)$, y $\Delta(t) =
a(t)b'(t) - a'(t)b(t)$.

\medskip
\noindent\textbf{Parte I --- La máquina de \omterm{pb:b2:curves:1}{envolventes}.}
\begin{enumerate}
  \item Supongamos $\Delta(t) \neq 0$. Prueba que el \emph{sistema
        característico}
        \[
        \begin{cases} a(t)\,x + b(t)\,y = c(t)\\
        a'(t)\,x + b'(t)\,y = c'(t)\end{cases}
        \]
        tiene una única solución $E(t) = (x(t), y(t))$, dada por $x
        = \dfrac{cb' - c'b}{\Delta}$, $y = \dfrac{ac' -
        a'c}{\Delta}$.
  \item Supongamos además que $E$ es $\mathcal C^1$ cerca de $t$ con
        $E'(t) \neq 0$. Derivando la primera ecuación del sistema,
        prueba que $a(t)\,x'(t) + b(t)\,y'(t) = 0$, y concluye que
        la curva $E$ pasa por un punto de $D_t$ \emph{con la
        dirección de} $D_t$: la familia es tangente a $E$, que se
        llama su \emph{\omterm{pb:b2:curves:1}{envolvente}}.
  \item Comprobación de sensatez: las \omterm{def:b2:curves:arc}{rectas tangentes} de la parábola
        $y = x^2/2$ en los puntos $(t, t^2/2)$ son $tx - y = t^2/2$.
        Verifica que la máquina de \omterm{pb:b2:curves:1}{envolventes} devuelve la propia
        parábola.
  \item (\omterm{pb:b2:curves:1}{Envolvente} de las normales) Sea $\gamma$ de rapidez unidad
        con $\kappa(s) \neq 0$. La recta normal en $\gamma(s)$ es
        $\{M : \langle M - \gamma(s), T(s) \rangle = 0\}$. Prueba
        que su sistema característico fuerza $\langle M - \gamma(s),
        N(s)\rangle = 1/\kappa(s)$, y por tanto que el punto
        característico es el \omterm{def:b2:curves:curvature}{centro de curvatura}: \emph{la
        \omterm{pb:b2:curves:1}{envolvente} de las normales es la evoluta}, lo que recupera el~\cref{exo:b2:curves:6}. Comprueba que $\Delta(s) =
        \kappa(s)$.
  \item Dos degeneraciones. Para el haz $D_\theta : x\cos \theta +
        y\sin\theta = 0$, prueba que el punto característico es el
        origen para todo $\theta$ (la ``\omterm{pb:b2:curves:1}{envolvente}'' colapsa en un
        punto, y $E' = 0$: la pregunta 2 no se aplica). Para una
        familia de rectas paralelas ($a, b$ constantes), prueba que
        $\Delta \equiv 0$ y que el sistema característico es en
        general incompatible: no hay \omterm{pb:b2:curves:1}{envolvente}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La escalera que resbala y la
\omterm{pb:b2:curves:1}{astroide}.} Un segmento de \omterm{def:b2:curves:length}{longitud} $1$ resbala con un extremo $P_t =
(\cos t, 0)$ sobre el suelo y el otro $Q_t = (0, \sin t)$ sobre la
pared, $t \in \intoo0{\pi/2}$.
\begin{enumerate}[resume]
  \item Prueba que la recta $(P_tQ_t)$ tiene ecuación $x\sin t +
        y\cos t = \sin t\cos t$, y que la máquina de \omterm{pb:b2:curves:1}{envolventes} da
        el punto característico
        \[
        E(t) = (\cos^3 t,\ \sin^3 t) :
        \]
        la \emph{astroide}\index{astroide}, de ecuación implícita
        $x^{2/3} + y^{2/3} = 1$ (extendida a los demás cuadrantes
        por simetría).
  \item Prueba que $E'(0) = 0$ y, usando la clasificación local
        (\cref{prop:b2:curves:local}), que la \omterm{pb:b2:curves:1}{astroide} tiene una
        cúspide de primera especie en $(1, 0)$, y lo mismo en sus
        cuatro puntos sobre los ejes.
  \item Calcula $\norm{E'(t)} = \tfrac32\abs{\sin 2t}$ y deduce que
        la \omterm{def:b2:curves:length}{longitud} total de la \omterm{pb:b2:curves:1}{astroide} es $6$.
  \item ¿Dónde toca la escalera a la \omterm{pb:b2:curves:1}{astroide}? Prueba que $E(t) =
        P_t + \sin^2 t\,(Q_t - P_t)$: el punto de contacto divide la
        escalera en la razón $\sin^2 t : \cos^2 t$, y la recorre de
        un extremo al otro a medida que la escalera resbala.
  \item Calcula el área encerrada por la \omterm{pb:b2:curves:1}{astroide}: prueba que el
        área del primer cuadrante es $3\int_0^{\pi/2}\sin^4 t\cos^2
        t\,\dd t$, evalúa la integral por linealización ($\sin^2 2t
        = \tfrac{1 - \cos 4t}2$) y concluye que el área total es
        $3\pi/8$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La evoluta de la parábola.} Sea
$\gamma(t) = (t, t^2/2)$.
\begin{enumerate}[resume]
  \item Prueba que la recta normal en $\gamma(t)$ tiene ecuación $x
        + t\,y = t + t^3/2$.
  \item Ejecuta la máquina de \omterm{pb:b2:curves:1}{envolventes}: prueba que la \omterm{pb:b2:curves:1}{envolvente}
        de las normales es
        \[
        E(t) = \Bigl(-t^3,\ 1 + \tfrac32 t^2\Bigr),
        \]
        de ecuación implícita $x^2 = \tfrac8{27}(y - 1)^3$: una
        parábola semicúbica.
  \item Contrasta con la pregunta 4: calcula el \omterm{def:b2:curves:curvature}{centro de curvatura}
        $\gamma(t) + \frac1{\kappa(t)}N(t)$ a partir de $\kappa(t) =
        (1 + t^2)^{-3/2}$ (\cref{ex:b2:curves:kappaexamples}) y
        recupera el mismo punto.
  \item Prueba que la evoluta tiene una cúspide de primera especie
        en $(0, 1)$, el \omterm{def:b2:curves:curvature}{centro de curvatura} en el vértice: el punto
        donde $\kappa$ es extremal, como predice la fórmula $c' =
        -\frac{\kappa'}{\kappa^2}N$ del~\cref{exo:b2:curves:6}.
  \item ¿Cuántas normales de la parábola pasan por un punto dado
        $(x_0, y_0)$? Prueba que la respuesta la gobierna la cúbica
        $\tfrac{t^3}2 + (1 - y_0)\,t - x_0 = 0$; trata por completo
        el caso del eje $x_0 = 0$ (una normal para $y_0 < 1$, tres
        para $y_0 > 1$) e interpreta la evoluta como la curva de
        transición.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La cáustica de la taza de café.} Unos
rayos paralelos de dirección $(1, 0)$ inciden sobre el interior de la
circunferencia espejo $x^2 + y^2 = 1$; el rayo que llega a $P_\theta
= (\cos\theta, \sin\theta)$ se refleja según la ley de la reflexión.
\begin{enumerate}[resume]
  \item A partir de la simetría especular respecto de la normal (el
        radio), justifica que la dirección reflejada es $v = u -
        2\langle u, n\rangle n$ con $u = (1,0)$ y $n = (\cos\theta,
        \sin\theta)$, y calcula $v = -(\cos2\theta, \sin2\theta)$.
  \item Prueba que el rayo reflejado está sobre la recta
        \[
        x\sin 2\theta - y\cos 2\theta = \sin\theta .
        \]
  \item Ejecuta la máquina de \omterm{pb:b2:curves:1}{envolventes} ($\Delta = 2$): prueba que
        la cáustica es
        \[
        E(\theta) = \Bigl(\tfrac{3\cos\theta -
        \cos3\theta}4,\ \tfrac{3\sin\theta -
        \sin3\theta}4\Bigr),
        \]
        la \emph{nefroide}\index{nefroide}.
  \item Calcula $E'(\theta) = \tfrac32\sin\theta\, (\cos2\theta,
        \sin2\theta)$; comprueba que la dirección tangente es la
        dirección del rayo reflejado (pregunta 16), localiza las dos
        cúspides $(\pm\tfrac12, 0)$ y prueba que el rayo reflejado
        corta el eje $y = 0$ en $x = \frac1{2\cos\theta}$: los rayos
        casi axiales enfocan, pues, en $x = \tfrac12$, la distancia
        focal $R/2$ de un espejo de radio $R$.
  \item Prueba que la \omterm{pb:b2:curves:1}{nefroide} tiene \omterm{def:b2:curves:length}{longitud} total $6$ y que cerca
        de $\theta = 0$,
        \[
        E(\theta) - \bigl(\tfrac12, 0\bigr) =
        \bigl(\tfrac34\theta^2 + o(\theta^2),\ \theta^3 +
        o(\theta^3)\bigr) :
        \]
        una cúspide de primera especie, apuntando a lo largo del
        eje.
  \item Explica en un párrafo por qué la cáustica es brillante: por
        cada punto justo fuera de la cáustica pasan dos rayos
        reflejados, y en cada punto de ella los rayos están
        ``infinitamente concentrados'' (la aplicación $(\theta,
        \text{distancia a lo largo del rayo}) \mapsto \R^2$ tiene un
        punto crítico exactamente sobre la \omterm{pb:b2:curves:1}{envolvente}).
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Huygens: la cicloide es su propia
evoluta.} Sea $\gamma(t) = (t - \sin t,\ 1 - \cos t)$, $t \in
\intoo0{2\pi}$, un arco de cicloide.
\begin{enumerate}[resume]
  \item Calcula $\kappa(t) = -\dfrac1{4\sin(t/2)}$ y el
        \omterm{def:b2:curves:curvature}{centro de curvatura}; prueba que la evoluta es
        \[
        c(t) = (t + \sin t,\ \cos t - 1),
        \]
        y que la sustitución $t = u + \pi$ la exhibe como la
        cicloide original trasladada por $(\pi, -2)$: \emph{la
        evoluta de una cicloide es una cicloide congruente}
        (Huygens).
  \item Comprueba que el \omterm{def:b2:curves:curvature}{radio de curvatura} en el ápice $t = \pi$
        vale $4$, la mitad de la \omterm{def:b2:curves:length}{longitud} $8$ de un arco
        (\cref{exo:b2:curves:1}); localiza la cúspide de la evoluta
        directamente bajo el ápice, a distancia $4$.
  \item (La propiedad del hilo) Sea $\gamma$ de rapidez unidad con
        $\kappa > 0$, $\kappa$ de clase $\mathcal C^1$ y $R =
        1/\kappa$ estrictamente monótona. Usando $c' = R'N$, prueba
        que la \omterm{def:b2:curves:length}{longitud de arco} de la evoluta entre $c(s_0)$ y
        $c(s_1)$ vale $\abs{R(s_1) - R(s_0)}$. Interpretación: un
        hilo tenso desenrollado de la evoluta, de \omterm{def:b2:curves:length}{longitud} $R(s_0)$
        al principio, hace que su extremo libre trace la curva
        original; de ahí que un péndulo que oscile entre dos
        mejillas cicloidales del reloj de Huygens describa una
        cicloide.
  \item Síntesis. La máquina de la parte I produjo la \omterm{pb:b2:curves:1}{astroide}, una
        parábola semicúbica, una \omterm{pb:b2:curves:1}{nefroide} y una cicloide. Para cada
        una de las cuatro familias, indica en una frase dónde se
        cumplieron o fallaron las hipótesis $\Delta \neq 0$ y $E'
        \neq 0$, y qué acontecimiento geométrico (cúspide, foco,
        degeneración) señalaba cada fallo de $E' \neq 0$. ¿Dónde han
        de aparecer los extremos de la \omterm{thm:b2:curves:frenet2d}{curvatura} sobre la
        \omterm{pb:b2:curves:1}{envolvente} de las normales, y por qué?
\end{enumerate}
\end{problem}
