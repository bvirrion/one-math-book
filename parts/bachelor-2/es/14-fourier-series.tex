\chapter{Series de Fourier}\label{ch:b2:fourier}

¿Puede reconstruirse toda señal periódica a partir de senos y cosenos
puros? El audaz ``sí'' de Fourier creó un siglo de análisis. Este
capítulo demuestra los dos pilares al alcance de este nivel: el
\emph{teorema de Dirichlet} (reconstrucción \omterm{def:b2:funcseq:def}{puntual} para funciones $C^1$
a trozos, mediante el \omterm{lem:b2:fourier:kernel}{núcleo de Dirichlet}) y la \emph{identidad de
Parseval} (la energía de una señal es la suma de las energías de sus
armónicos), y cosecha las series numéricas clásicas, con la de Basilea
$\sum 1/n^2 = \pi^2/6$ a la cabeza.

En todo el capítulo, las funciones son $2\pi$-periódicas, \omterm{def:b2:metric:continuity}{continuas} a
trozos y con valores complejos; $\mathcal{C}$ denota las \omterm{def:b2:metric:continuity}{continuas}.

\section{Coeficientes de Fourier}

\begin{definition}\label{def:b2:fourier:coefficients}
Los \emph{coeficientes de Fourier}\index{coeficientes de Fourier} de
$f$ son
\[
c_n(f) = \frac{1}{2\pi}\int_{-\pi}^{\pi} f(t)\,\eu^{-\iu n t}\,\dd
t \qquad (n \in \Z),
\]
y los coeficientes en forma real son $a_n = c_n + c_{-n}$, $b_n =
\iu(c_n - c_{-n})$, de modo que las \emph{sumas parciales de
Fourier}\index{sumas parciales de Fourier} son
\[
S_N(f)(t) = \sum_{n=-N}^{N} c_n(f)\,\eu^{\iu nt}
= \frac{a_0}{2} + \sum_{n=1}^{N}\bigl(a_n\cos nt + b_n \sin
nt\bigr).
\]
Sobre $\mathcal{C}$ definimos el \omterm{def:b2:hermitian:def}{producto escalar hermítico} $\langle
f, g\rangle = \frac{1}{2\pi}\int_{-\pi}^{\pi}\conj f\,g$: las
exponenciales $e_n(t) = \eu^{\iu nt}$ son \emph{ortonormales}
($\langle e_m, e_n\rangle = \delta_{mn}$, por cálculo directo) y
$c_n(f) = \langle e_n, f\rangle$: el análisis de Fourier es geometría
\omterm{def:b2:hermitian:adjoint}{hermítica} (\cref{ch:b2:hermitian}) en dimensión infinita.
\end{definition}

\begin{proposition}[Desigualdad de Bessel]\label{prop:b2:fourier:bessel}
$S_N(f)$ es la proyección ortogonal de $f$ sobre el espacio
$\mathcal{T}_N$ de los polinomios trigonométricos de grado $\leq N$,
y\index{desigualdad de Bessel}
\[
\sum_{n=-N}^{N} \abs{c_n(f)}^2 \leq \norm f_2^2 =
\frac{1}{2\pi}\int_{-\pi}^{\pi} \abs f^2 :
\]
la serie $\sum \abs{c_n}^2$ converge y $c_n(f) \to 0$ cuando $\abs n
\to \infty$ (Riemann--Lebesgue para coeficientes).
\end{proposition}

\begin{proof}
$f - S_N(f)$ es ortogonal a cada $e_k$ con $\abs k \leq N$ ($\langle
e_k, f - S_N f\rangle = c_k - c_k = 0$): $S_Nf$ es la proyección
ortogonal sobre $\mathcal{T}_N = \operatorname{Vect}(e_{-N}, \dots,
e_N)$ (el teorema de la proyección del volumen del primer año,
palabra por palabra en el marco \omterm{def:b2:hermitian:adjoint}{hermítico}). Pitágoras: $\norm f_2^2 =
\norm{S_Nf}_2^2 + \norm{f - S_Nf}_2^2 \geq \norm{S_Nf}_2^2 =
\sum_{\abs n \leq N} \abs{c_n}^2$; hágase $N \to \infty$.
\end{proof}

\begin{example}[La mejor aproximación, medida]\label{ex:b2:fourier:bestapprox}
¿Con qué precisión aproximan los polinomios trigonométricos de grado
bajo a la señal en diente de sierra $f(t) = t$ (sobre
$\intoo{-\pi}{\pi}$) en media cuadrática? Por la~\cref{prop:b2:fourier:bessel}, la mejor aproximación de grado $N$
\emph{es} $S_N(f)$, con error cuadrático
\[
\norm{f - S_Nf}_2^2 = \norm f_2^2 -
\sum_{\abs n\leq N}\abs{c_n}^2 .
\]
Aquí $\norm f_2^2 = \frac{1}{2\pi}\int_{-\pi}^\pi t^2\dd t =
\frac{\pi^2}{3}$, y de $b_n = \frac{2(-1)^{n+1}}{n}$
(\cref{ex:b2:fourier:basel}) resulta $\abs{c_n}^2 + \abs{c_{-n}}^2 =
\frac{b_n^2}{2} = \frac{2}{n^2}$. Por tanto,
\[
\norm{f - S_Nf}_2^2
= \frac{\pi^2}{3} - \sum_{n=1}^{N}\frac{2}{n^2}
\qquad\text{: numéricamente } 1.29,\ 0.79,\ 0.57,\ 0.44
\]
para $N = 1, 2, 3, 4$: decreciente, pero despacio; la cola
$\sum_{n>N}\frac2{n^2} \sim \frac2N$ la gobierna el lento decaimiento
$\frac1n$ de los coeficientes, que a su vez es la firma del salto (el~\cref{exo:b2:fourier:6} leído al revés). Moraleja: Parseval convierte
la calidad de la aproximación en la cola de una serie numérica, y
predice, antes de dibujar nada, que los saltos hacen que las series de
Fourier converjan de mala gana.
\end{example}

\begin{method}[Calcular coeficientes de Fourier con eficacia]\label{met:b2:fourier:coefficients}
Antes de integrar nada:
\begin{enumerate}
  \item \emph{Paridad:} si $f$ es par, $b_n = 0$; si es impar, $a_n
        = 0$, y las integrales supervivientes se reducen a
        $\frac2\pi \int_0^\pi$: la mitad de trabajo y el doble de
        fiabilidad.
  \item \emph{Los polinomios trigonométricos ya están hechos:}
        linealícense los productos ($\cos^3$, $\sin^2\cos$, \dots) y
        léanse los coeficientes (\cref{exo:b2:fourier:9}); la
        ortonormalidad hace redundante cualquier integración
        posterior.
  \item \emph{Exponenciales complejas para las exponenciales:} ante
        factores $\eu^{at}$ u oscilaciones amortiguadas, calcúlese
        $c_n$ directamente; una integral de $\eu^{(a - \iu n)t}$
        gana a dos integraciones por partes
        (\cref{exo:b2:fourier:10}).
  \item \emph{Derívese un desarrollo conocido:} si se conocen los
        coeficientes de $f'$ y $f$ es \omterm{def:b2:metric:continuity}{continua}, $c_n(f) =
        \frac{c_n(f')}{\iu n}$ ($n \neq 0$) recupera todos salvo
        $c_0$, que es la media; a menudo es la vía más rápida, y es
        legítima exactamente bajo las hipótesis del~\cref{thm:b2:fourier:parseval}~(1).
\end{enumerate}
\end{method}

\section{El teorema de Dirichlet}

\begin{lemma}[Núcleo de Dirichlet]\label{lem:b2:fourier:kernel}
$S_N(f)(x) = \frac{1}{2\pi}\int_{-\pi}^{\pi} f(x + u)\,D_N(u)\,\dd
u$, donde\index{núcleo de Dirichlet}
\[
D_N(u) = \sum_{n=-N}^{N} \eu^{\iu nu}
= \frac{\sin\bigl((N + \frac12)u\bigr)}{\sin\frac u2}
\quad (u \notin 2\pi\Z),
\qquad
\frac{1}{2\pi}\int_{-\pi}^{\pi} D_N = 1 .
\]
\end{lemma}

\begin{proof}
Insértese la definición de $c_n$ en $S_N$ e intercámbiense suma e
integral (es legítimo: la suma es finita):
\[
S_N(f)(x)
= \sum_{n=-N}^{N}\Bigl(\frac{1}{2\pi}\int_{-\pi}^{\pi}
f(t)\,\eu^{-\iu nt}\dd t\Bigr)\eu^{\iu nx}
= \frac{1}{2\pi}\int_{-\pi}^{\pi} f(t)
\sum_{n=-N}^{N}\eu^{\iu n(x-t)}\,\dd t ;
\]
sustitúyase $u = t - x$ y deslícese el segmento de integración de
vuelta a $\intcc{-\pi}{\pi}$ mediante la $2\pi$-periodicidad del
integrando; el rango \omterm{def:b2:quadratic:adjoint}{simétrico} de índices hace que $\sum_n\eu^{-\iu
nu} = D_N(u)$. La forma cerrada: suma geométrica de razón $\eu^{\iu
u}$,
\[
D_N(u) = \eu^{-\iu Nu}\,\frac{\eu^{\iu(2N+1)u} - 1}{\eu^{\iu u} -
1}
= \frac{\eu^{\iu(N + \frac12)u} - \eu^{-\iu(N+\frac12)u}}
{\eu^{\iu u/2} - \eu^{-\iu u/2}} ,
\]
que es el cociente de senos. Su media vale $1$: solo contribuye $n =
0$.
\end{proof}

\begin{theorem}[Lema de Riemann--Lebesgue]\label{thm:b2:fourier:riemannlebesgue}
Para $g$ \omterm{def:b2:metric:continuity}{continua} a trozos sobre un segmento, $\int_a^b
g(t)\sin(\lambda t + \varphi)\,\dd t \to 0$ cuando $\lambda \to
+\infty$.\index{lema de Riemann--Lebesgue}
\end{theorem}

\begin{proof}
Aproxímese $g$ \omterm{def:b2:funcseq:def}{uniformemente} por funciones escalonadas (no hace falta
el~\cref{thm:b2:funcseq:weierstrass}: basta la aproximación
elemental de las funciones \omterm{def:b2:metric:continuity}{continuas} a trozos por funciones
escalonadas) e intégrese cada escalón explícitamente: cada trozo
aporta $O\bigl(\frac1\lambda\bigr)$, y el error de aproximación
aporta $\varepsilon(b - a)$. Este argumento se desarrolló por
completo como último ejercicio del capítulo de integración del
volumen del primer año; para trozos $C^1$ puede integrarse en su
lugar por partes y acotarse por $\frac C\lambda$.
\end{proof}

\begin{example}[¿Con qué rapidez mueren los coeficientes?]\label{ex:b2:fourier:decaytable}
Riemann--Lebesgue dice que los coeficientes tienden a $0$; su
\emph{ritmo} es un medidor de regularidad. Tres especímenes de este
capítulo y de sus ejercicios:
\[
\text{onda cuadrada: } b_n = \frac{4}{\pi n}\ (n\ \text{impar}),
\qquad
\abs t : a_n = \frac{-4}{\pi n^2}\ (n\ \text{impar}),
\qquad
\abs{\sin t} : a_{2k} = \frac{-4}{\pi(4k^2-1)} .
\]
Un salto en $f$ (onda cuadrada, diente de sierra) deja coeficientes
de orden $\frac1n$: no hay \omterm{def:b2:funcseq:series}{convergencia normal}, y sí sobreoscilación
de Gibbs en los saltos. La \omterm{def:b2:metric:continuity}{continuidad} con un pico ---un salto solo
en $f'$--- mejora el orden a $\frac{1}{n^2}$: \omterm{def:b2:funcseq:series}{convergencia normal} y
reconstrucción uniforme. En general, $k$ derivadas compran $c_n =
O(n^{-k})$ (\cref{exo:b2:fourier:6}) y, recíprocamente, un
\omterm{def:b2:reduction:eigen}{espectro} que decae más deprisa que toda potencia fuerza a $f$ a ser
$C^\infty$ (derívese término a término, ahora legítimamente).
Moraleja: la regularidad de la señal y el decaimiento del
\omterm{def:b2:reduction:eigen}{espectro} son la misma información; un ingeniero lee una en la
pendiente del otro sin llegar a dibujar la función.
\end{example}

\begin{theorem}[Dirichlet]\label{thm:b2:fourier:dirichlet}
Sea $f$ $2\pi$-periódica y $C^1$ a trozos. Entonces, para todo $x$,
\[
S_N(f)(x) \xrightarrow[N \to \infty]{}
\frac{f(x^+) + f(x^-)}{2}
\]
(la media de los límites laterales); en particular, $S_N(f)(x) \to
f(x)$ en todo punto de \omterm{def:b2:metric:continuity}{continuidad}.\index{teorema de Dirichlet}
\end{theorem}


\begin{proof}
Por el lema del núcleo y su media unidad, partiendo la integral en
las mitades $u > 0$ y $u < 0$ (cada una de media $\frac12$):
\begin{align*}
S_N(f)(x) - \frac{f(x^+) + f(x^-)}{2}
&= \frac{1}{2\pi}\int_{0}^{\pi} \bigl(f(x+u) -
f(x^+)\bigr)D_N(u)\,\dd u\\
&\quad+ \frac{1}{2\pi}\int_{-\pi}^{0}\bigl(f(x+u) -
f(x^-)\bigr)D_N(u)\,\dd u .
\end{align*}
Tratemos la primera (la segunda es \omterm{def:b2:quadratic:adjoint}{simétrica}). Escribamos
\[
\bigl(f(x + u) - f(x^+)\bigr)\,D_N(u)
= \underbrace{\frac{f(x+u) - f(x^+)}{\sin\frac u2}}_{g(u)}\,
\sin\Bigl(\Bigl(N + \frac12\Bigr)u\Bigr) .
\]
La función $g$ es \omterm{def:b2:metric:continuity}{continua} a trozos sobre $\intoc{0}{\pi}$ y tiene
\emph{límite finito en $0^+$}: escribiendo
\[
g(u) = \frac{f(x+u) - f(x^+)}{u}\cdot\frac{u}{\sin\frac u2} ,
\]
el primer factor tiende a $f'(x^+)$ (derivabilidad lateral,
consecuencia de ser $C^1$ a trozos) y el segundo a $2$ (el límite
estándar $\frac{\sin v}{v} \to 1$ con $v = \frac u2$): existe
$g(0^+) = 2f'(x^+)$. Así pues, $g$ se extiende de manera \omterm{def:b2:metric:continuity}{continua} a
trozos a $\intcc{0}{\pi}$, y Riemann--Lebesgue
(\cref{thm:b2:fourier:riemannlebesgue}) manda la integral a $0$. En
esto consiste precisamente la hipótesis: sin derivadas laterales, el
factor $\frac{1}{\sin(u/2)}$ estalla en $0$ más deprisa de lo que
Riemann--Lebesgue puede compensar, y la \omterm{def:b2:funcseq:def}{convergencia puntual} puede
fallar de verdad para $f$ meramente \omterm{def:b2:metric:continuity}{continua}; la laguna que el
teorema de Fejér (problema de fin de semana) cierra promediando.
\end{proof}

\begin{example}[Dirichlet en un salto]\label{ex:b2:fourier:jumpvalue}
Para el diente de sierra $f(t) = t$ sobre $\intoo{-\pi}{\pi}$
(\cref{ex:b2:fourier:basel} más abajo), la extensión periódica salta
en $t = \pi$ de $f(\pi^-) = \pi$ a $f(\pi^+) = -\pi$. Dirichlet
promete allí el valor $\frac{\pi + (-\pi)}{2} = 0$, y, en efecto,
todos los términos de $\sum \frac{2(-1)^{n+1}}{n}\sin nt$ se anulan
en $t = \pi$: la serie converge educadamente al punto medio,
ignorando ambos valores laterales. Si en cambio se evalúa en $t =
\frac\pi2$ (un punto de \omterm{def:b2:metric:continuity}{continuidad}), esa misma serie se convierte en
el $\frac\pi4$ de Leibniz. Una serie, dos comportamientos:
exactamente las dos cláusulas del teorema.
\end{example}

\begin{theorem}[Convergencia normal para $C^1$; Parseval]\label{thm:b2:fourier:parseval}
\begin{enumerate}
  \item Si $f$ es \omterm{def:b2:metric:continuity}{continua}, $2\pi$-periódica y $C^1$ a trozos,
        entonces $c_n(f') = \iu n\,c_n(f)$, la serie de Fourier de
        $f$ converge \emph{\omterm{def:b2:funcseq:series}{normalmente}} sobre $\R$ y su suma es
        $f$.
  \item (Parseval) Para toda $f$ $2\pi$-periódica \omterm{def:b2:metric:continuity}{continua} a
        trozos:\index{identidad de Parseval}
        \[
        \frac{1}{2\pi}\int_{-\pi}^{\pi}\abs{f}^2
        = \sum_{n=-\infty}^{\infty} \abs{c_n(f)}^2
        = \frac{\abs{a_0}^2}{4} + \frac12\sum_{n\geq1}
        \bigl(\abs{a_n}^2 + \abs{b_n}^2\bigr).
        \]
        \emph{(Se demuestra aquí para $f$ \omterm{def:b2:metric:continuity}{continua} y $C^1$ a
        trozos; se admite en general.)}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Integración por partes en cada trozo $C^1$ (los términos de
frontera se cancelan por \omterm{def:b2:metric:continuity}{continuidad} y periodicidad): $c_n(f') = \iu n
c_n(f)$. Entonces, por Cauchy--Schwarz sobre las dos \omterm{def:b2:series:summable}{familias
sumables} al cuadrado (\cref{prop:b2:fourier:bessel} aplicada a
$f'$):
\[
\sum_{n \neq 0} \abs{c_n(f)} = \sum_{n\neq0}
\frac{\abs{c_n(f')}}{\abs n}
\leq \Bigl(\sum \abs{c_n(f')}^2\Bigr)^{1/2}
\Bigl(\sum_{n\neq0}\frac{1}{n^2}\Bigr)^{1/2} < \infty :
\]
\omterm{def:b2:funcseq:series}{convergencia normal} de la serie de Fourier. Su suma es \omterm{def:b2:metric:continuity}{continua} y
coincide con $f$ en todo punto por Dirichlet
(\cref{thm:b2:fourier:dirichlet}: $f$ es \omterm{def:b2:metric:continuity}{continua}): la serie converge
a $f$ \omterm{def:b2:funcseq:def}{uniformemente}.

(2) Para tal $f$: $S_N f \to f$ \omterm{def:b2:funcseq:def}{uniformemente}, luego $\norm{f -
S_Nf}_2 \leq \norm{f - S_Nf}_\infty \to 0$, y Pitágoras ($\norm
f_2^2 = \sum_{\abs n \leq N}\abs{c_n}^2 + \norm{f - S_Nf}_2^2$) pasa
al límite. La forma real es contabilidad con $a_n$ y $b_n$.
\end{proof}

\begin{example}[La cota de la cola en el caso $C^1$, cuantificada]\label{ex:b2:fourier:tailbound}
La demostración del~\cref{thm:b2:fourier:parseval}~(1) esconde una
estimación utilizable. Para $f$ \omterm{def:b2:metric:continuity}{continua} y $C^1$ a trozos, el mismo
Cauchy--Schwarz aplicado solo a la cola da
\[
\sum_{\abs n > N}\abs{c_n(f)}
= \sum_{\abs n > N}\frac{\abs{c_n(f')}}{\abs n}
\leq \Bigl(\sum_{\abs n > N}\abs{c_n(f')}^2\Bigr)^{\!1/2}
\Bigl(\sum_{\abs n>N}\frac{1}{n^2}\Bigr)^{\!1/2}
\leq \norm{f'}_2\,\sqrt{\frac{2}{N}} ,
\]
usando Bessel para $f'$ y $\sum_{n>N}n^{-2} \leq \frac1N$. Así pues,
el error uniforme de las sumas parciales obedece
\[
\norm{f - S_Nf}_\infty
\leq \sum_{\abs n>N}\abs{c_n(f)}
\leq \norm{f'}_2\,\sqrt{\frac2N} .
\]
Para $f(t) = \abs t$: $\norm{f'}_2 = 1$ (la derivada vale $\pm1$), de
modo que diez términos ya reconstruyen $\abs t$ \omterm{def:b2:funcseq:def}{uniformemente} con
error $\sqrt{0.2} \approx 0.45$, y $N = 10^4$ con error
$0.015$. Moraleja: una derivada compra el ritmo uniforme
$\frac{1}{\sqrt N}$; comparado con el mundo de coeficientes $\frac1n$
de la onda cuadrada (sin \omterm{def:b2:funcseq:def}{convergencia uniforme} en absoluto), el
diccionario del~\cref{ex:b2:fourier:decaytable} adquiere números.
\end{example}

\begin{example}[Basilea y compañía]\label{ex:b2:fourier:basel}
Sea $f(t) = t$ sobre $\intoo{-\pi}{\pi}$, extendida
$2\pi$-periódicamente (un diente de sierra, $C^1$ a trozos).
Calculando, $a_n = 0$ (por imparidad) y
\[
b_n = \frac{1}{\pi}\int_{-\pi}^{\pi} t\sin nt\,\dd t
= \frac{2(-1)^{n+1}}{n} .
\]
Dirichlet en $t = \frac\pi2$ recupera el $\frac\pi4 = 1 - \frac13 +
\frac15 - \dots$ de Leibniz; y Parseval da
\[
\frac{1}{2\pi}\int_{-\pi}^{\pi} t^2\,\dd t = \frac{\pi^2}{3}
= \frac12\sum_{n\geq1}\frac{4}{n^2}
\quad\Longrightarrow\quad
\boxed{\;\sum_{n\geq1}\frac{1}{n^2} = \frac{\pi^2}{6}\;}
\]
la suma de Basilea de Euler, en dos líneas. La función $f(t) = t^2$
da análogamente $\sum \frac1{n^4} = \frac{\pi^4}{90}$
(\cref{exo:b2:fourier:3}).
\end{example}

\begin{example}[Un desarrollo completo con comprobación incorporada: $\abs{\sin t}$]\label{ex:b2:fourier:abssin}
La función $f(t) = \abs{\sin t}$ es \omterm{def:b2:metric:continuity}{continua}, par, $\pi$-periódica
(luego $2\pi$-periódica) y $C^1$ a trozos. La paridad mata los $b_n$;
$a_0 = \frac1\pi\int_{-\pi}^{\pi} \abs{\sin t}\dd t = \frac4\pi$; y
para $n \geq 1$, la fórmula de producto a suma da
\[
a_n = \frac2\pi\int_0^\pi \sin t\cos nt\,\dd t
= \frac{1}{\pi}\int_0^\pi\bigl(\sin(1+n)t +
\sin(1-n)t\bigr)\dd t
= \frac2\pi\cdot\frac{1 + \cos n\pi}{1 - n^2}
\]
para $n \neq 1$ (y $a_1 = 0$ directamente): cero para $n$ impar, y
$a_{2k} = \frac{-4}{\pi(4k^2-1)}$. Por el~\cref{thm:b2:fourier:parseval}~(1), la convergencia es normal, y
\[
\abs{\sin t} = \frac{2}{\pi} - \frac{4}{\pi}
\sum_{k\geq1}\frac{\cos(2kt)}{4k^2 - 1}
\qquad (t \in \R) .
\]
Comprobación incorporada en $t = 0$: la identidad exige
$\sum_{k\geq1}\frac{1}{4k^2-1} = \frac12$, lo que confirma un
telescopaje:
\[
\sum_{k\geq1}\frac{1}{4k^2-1}
= \frac12\sum_{k\geq1}\Bigl(\frac{1}{2k-1} -
\frac{1}{2k+1}\Bigr) = \frac12 . \checkmark
\]
Moraleja: el \omterm{def:b2:reduction:eigen}{espectro} de $\abs{\sin}$ vive solo en las frecuencias
\emph{pares}; rectificar un seno duplica su contenido frecuencial, y
por eso los rectificadores de onda completa zumban a $100$ o $120$
hercios, el doble de la frecuencia de la red.
\end{example}

\begin{example}[Parseval como máquina de calcular]\label{ex:b2:fourier:zetafourodd}
Parseval convierte desarrollos en series numéricas al por mayor.
Apliquémoslo a $f(t) = \abs t$ (\cref{exo:b2:fourier:2}: $a_0 = \pi$,
$a_n = \frac{-4}{\pi n^2}$ para $n$ impar, y los demás nulos):
\[
\frac{1}{2\pi}\int_{-\pi}^{\pi}t^2\,\dd t = \frac{\pi^2}{3}
= \frac{a_0^2}{4} + \frac12\sum_{n \text{ impar}} a_n^2
= \frac{\pi^2}{4} +
\frac{8}{\pi^2}\sum_{n\text{ impar}}\frac{1}{n^4} ,
\]
de donde
\[
\sum_{n\text{ impar}}\frac{1}{n^4}
= \frac{\pi^2}{8}\Bigl(\frac{\pi^2}{3} -
\frac{\pi^2}{4}\Bigr) = \frac{\pi^4}{96} .
\]
Contraste con el~\cref{exo:b2:fourier:3}: partir $\sum\frac1{n^4}$ en
sus partes impar y par da la contabilidad al estilo $\zeta$ $S =
S_{\mathrm{impar}} + \frac{S}{16}$, luego $S =
\frac{16}{15}\cdot\frac{\pi^4}{96} = \frac{\pi^4}{90}$, exactamente
el valor hallado allí con otra función. Dos desarrollos, un número:
la coherencia es la isometría de Parseval en acción. Moraleja: cada
nuevo desarrollo de Fourier es una máquina de generar identidades
entre series; la parte II del problema de fin de semana explica por
qué la máquina nunca puede contradecirse.
\end{example}

\begin{example}[Traslación y modulación]\label{ex:b2:fourier:shiftrules}
Dos reglas de una línea generan muchos desarrollos a partir de uno.
Para $a \in \R$, sustituyendo $s = t - a$:
\[
c_n\bigl(f(\cdot - a)\bigr)
= \frac{1}{2\pi}\int_{-\pi}^{\pi}f(t - a)\eu^{-\iu nt}\dd t
= \eu^{-\iu na}\,c_n(f)
\qquad\text{(trasladar modula el espectro)},
\]
y, directamente de la definición,
\[
c_n\bigl(\eu^{\iu kt}f\bigr) = c_{n-k}(f)
\qquad\text{(modular traslada el espectro)}.
\]
Ejemplo resuelto: el diente de sierra desplazado en $\pi$, $g(t) =
f(t - \pi)$ con $f(t) = t$, tiene coeficientes $b_n$ iguales a
$(-1)^n\cdot\frac{2(-1)^{n+1}}{n} = -\frac2n$: el desarrollo $g \sim
-2\sum\frac{\sin nt}{n}$ del diente de sierra que salta en $0$ en vez
de en $\pi$, sin recalcular ninguna integral. Moraleja: los
desplazamientos temporales solo giran fases, nunca amplitudes
($\abs{c_n}$ es invariante por traslación), y por eso la energía
(Parseval) y la clase de convergencia son propiedades de la
\emph{forma} de la señal, no de dónde empieza el reloj.
\end{example}

\begin{remark}[Errores frecuentes]
\emph{(i) Tres convergencias, tres monedas:} la \omterm{def:b2:funcseq:def}{puntual} (Dirichlet:
exige $C^1$ a trozos y paga el \emph{punto medio} en cada salto,
nunca el valor lateral), la uniforme (exige un límite \omterm{def:b2:metric:continuity}{continuo};
imposible a través de un salto, y Gibbs es el síntoma visible) y la
de la media cuadrática (Parseval: la más robusta, ciega a los puntos
aislados). Dígase siempre cuál se está afirmando. \emph{(ii) Nada de
derivar término a término por defecto:} derivar término a término la
serie del diente de sierra del~\cref{ex:b2:fourier:basel} produce
$\sum 2(-1)^{n+1}\cos nt$, cuyos términos ni siquiera tienden a $0$;
los teoremas de transferencia del capítulo de sucesiones de funciones
exigen la \omterm{def:b2:funcseq:def}{convergencia uniforme} de la serie \emph{derivada}, que el
salto destruye. Primero la regularidad, después la derivación (el~\cref{exo:b2:fourier:6} es el diccionario). \emph{(iii) Sumas
parciales \omterm{def:b2:quadratic:adjoint}{simétricas}:} el teorema de Dirichlet se refiere a $S_N =
\sum_{-N}^{N}$; reordenar o sumar primero un lado puede convertir la
divergencia en convergencia y viceversa. \emph{(iv) Deriva de
normalización:} los convenios difieren de un libro a otro
($\frac{1}{2\pi}$ o $\frac1\pi$ delante, periodo $2\pi$ o $1$); los
invariantes fiables son las relaciones de ortonormalidad:
recalcúlese $\langle e_m, e_n\rangle$ en el convenio en uso antes de
fiarse de ninguna fórmula.
\end{remark}

\begin{remark}[Dónde se usa]
Parseval es el germen de la teoría $L^2$ de las series de Fourier: el
volumen del tercer año completa el cuadro (las exponenciales son una
base de Hilbert de $L^2$, y la aplicación $f \mapsto (c_n)$ es una
isometría biyectiva). Dentro de este volumen, el problema de fin de
semana demuestra el \emph{teorema de Fejér} ---las medias de
Cesàro de la serie de Fourier convergen \omterm{def:b2:funcseq:def}{uniformemente} para toda
$f$ periódica continua---, que asciende a teorema el Parseval general
admitido, da el teorema de Weierstrass trigonométrico y rinde dos
dividendos espectaculares: el teorema de equidistribución de Weyl y
la desigualdad isoperimétrica. La matemática aplicada lee este
capítulo a diario: \omterm{def:b2:reduction:eigen}{espectros} de señales, armónicos de sistemas
vibrantes y la transformada rápida de Fourier (cuyo avatar finito fue
el~\cref{exo:b2:hermitian:10}).
\end{remark}

\begin{remark}[Perspectivas dentro de este volumen]
Tres capítulos conversan con este. Hacia atrás: el capítulo
\omterm{def:b2:hermitian:adjoint}{hermítico} suministró la geometría (familias ortonormales,
proyecciones, Bessel) y el de sucesiones de funciones, el análisis
(\omterm{def:b2:funcseq:def}{convergencia uniforme}, teoremas de transferencia, identidades
aproximadas: el núcleo de Fejér es a las series de Fourier lo que
los \omterm{thm:b2:funcseq:weierstrass}{polinomios de Bernstein} fueron a Weierstrass). De lado: la teoría
de la frontera del capítulo de series de potencias vuelve mediante la
\omterm{thm:b2:series:abel}{sumación de Abel}, realizada aquí por el núcleo de Poisson
(\cref{exo:b2:fourier:12}), donde el radio $r$ del disco hace de
parámetro de sumación. Hacia delante: el capítulo de ecuaciones
diferenciales descompone en armónicos las excitaciones periódicas y
alimenta con cada uno la respuesta en frecuencia del oscilador; hay
resonancia cuando un modo de Fourier de la entrada coincide con una
frecuencia propia, razón por la cual su problema de fin de semana y
el de este capítulo son dos mitades de una misma historia.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:fourier:1}
Calcula los \omterm{def:b2:fourier:coefficients}{coeficientes de Fourier} de la onda cuadrada ($f = -1$
sobre $\intoo{-\pi}{0}$ y $+1$ sobre $\intoo{0}{\pi}$), enuncia la
conclusión de Dirichlet en $t = \frac\pi2$ y en el salto $t = 0$, y
recupera la serie de Leibniz.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:fourier:2}
Desarrolla en serie de Fourier $f(t) = \abs t$ ($\abs t \leq \pi$,
$2\pi$-periódica); justifica la convergencia \emph{normal}; evalúa en
$t = 0$ para obtener $\sum_{k\geq0}\frac{1}{(2k+1)^2} =
\frac{\pi^2}{8}$ y vuelve a deducir de ahí la suma de Basilea.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:fourier:3}
Desarrolla $f(t) = t^2$ ($\abs t \leq \pi$) y deduce
\[
\sum_{n\geq1}\frac{(-1)^{n+1}}{n^2} = \frac{\pi^2}{12},
\qquad
\sum_{n\geq1}\frac{1}{n^4} = \frac{\pi^4}{90}
\quad\text{(Parseval)}.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:4}
Sea $f$ \omterm{def:b2:metric:continuity}{continua} $2\pi$-periódica con $c_n(f) = 0$ para todo $n$.
Demuestra que $f = 0$ \emph{(Parseval; ¿para qué clase se demuestra
aquí? justifica que la \omterm{def:b2:metric:continuity}{continuidad} y el ser $C^1$ a trozos pueden
suprimirse admitiendo el Parseval general, o esboza el argumento de
densidad)}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:5}
Para $\alpha \notin \Z$, desarrolla $f(t) = \cos(\alpha t)$ ($\abs t
\leq \pi$) y deduce el desarrollo en fracciones simples de la
cotangente:
\[
\pi\cot(\pi\alpha) = \frac{1}{\alpha} + \sum_{n\geq1}
\frac{2\alpha}{\alpha^2 - n^2} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:6}
Demuestra que si $f$ es $2\pi$-periódica y de clase $C^k$ con
$f^{(k)}$ \omterm{def:b2:metric:continuity}{continua} a trozos, entonces $c_n(f) = O\bigl(\abs
n^{-k}\bigr)$: regularidad de la señal $=$ decaimiento de su
\omterm{def:b2:reduction:eigen}{espectro}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:fourier:7}
(Desigualdad de Wirtinger) Sea $f$ de clase $C^1$,
$2\pi$-periódica y con $\int_{-\pi}^{\pi} f = 0$. Demuestra que
\[
\int_{-\pi}^{\pi} \abs{f}^2 \leq \int_{-\pi}^{\pi} \abs{f'}^2 ,
\]
con igualdad si y solo si $f(t) = a\cos t + b\sin t$. \emph{(Parseval
en ambos miembros; compara $\abs{c_n}^2$ con $n^2\abs{c_n}^2$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:fourier:8}
(La constante de Gibbs) Para la onda cuadrada del~\cref{exo:b2:fourier:1}, evalúa la suma parcial en $x_N =
\frac{\pi}{2N}$: escribiendo $u_k = \frac{(2k+1)\pi}{2N}$ y $\Delta u
= \frac{\pi}{N}$, prueba que
\[
S_{2N-1}\Bigl(\frac{\pi}{2N}\Bigr)
= \frac{4}{\pi}\sum_{k=0}^{N-1} \frac{\sin u_k}{2k+1}
= \frac{2}{\pi}\sum_{k=0}^{N-1} \frac{\sin u_k}{u_k}\,\Delta u
\xrightarrow[N\to\infty]{}
\frac{2}{\pi}\int_0^{\pi}\frac{\sin u}{u}\,\dd u \approx 1.179 :
\]
una suma de Riemann de $\frac{2}{\pi}\cdot\frac{\sin u}{u}$ sobre
$\intcc{0}{\pi}$ en los puntos medios. Concluye que la
sobreoscilación por encima del valor de salto $1$ no se desvanece
cuando $N \to \infty$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:fourier:9}
Desarrolla $\cos^3 t$ y $\sin^2 t\,\cos t$ en serie de Fourier
\emph{(linealiza; un polinomio trigonométrico es su propia serie de
Fourier, por la unicidad de los coeficientes)}. ¿Cuánto valen $c_n$,
$a_n$ y $b_n$ en cada caso?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:10}
Sean $a > 0$ y $f(t) = \eu^{at}$ sobre $\intoc{-\pi}{\pi}$,
extendida $2\pi$-periódicamente. Calcula
\[
c_n(f) = \frac{(-1)^n\sinh(a\pi)}{\pi(a - \iu n)} ,
\]
aplica el teorema de Dirichlet en el salto $t = \pi$ y deduce el
desarrollo en fracciones simples de la cotangente hiperbólica:
\[
\coth(\pi a) = \frac{1}{\pi a} + \sum_{n\geq1}
\frac{2a}{\pi(a^2 + n^2)} .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fourier:11}
(Convolución) Para $f, g$ \omterm{def:b2:metric:continuity}{continuas} y $2\pi$-periódicas definimos
\[
(f * g)(x) = \frac{1}{2\pi}\int_{-\pi}^{\pi} f(x - t)\,g(t)\,
\dd t .
\]
Prueba que $f * g = g * f$, que $c_n(f*g) = c_n(f)\,c_n(g)$
\emph{(para intercambiar las dos integrales de un integrando
\omterm{def:b2:metric:continuity}{continuo}, compara las dos funciones del límite superior: ambas se
anulan en el extremo izquierdo y tienen la misma derivada, por
\omterm{def:b2:metric:continuity}{continuidad} y derivación bajo el signo integral)}, y que $S_N(f) = f
* D_N$ para el \omterm{lem:b2:fourier:kernel}{núcleo de Dirichlet}. (Las medias de Fejér del
problema de fin de semana son también convoluciones, $\sigma_N(f) = f
* F_N$.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:fourier:12}
(Núcleo de Poisson: medias de Abel de las series de Fourier) Para $0
\leq r < 1$, pongamos $P_r(t) = \sum_{n\in\Z} r^{\abs n}\eu^{\iu
nt}$.
\begin{enumerate}
  \item Suma las dos series geométricas y prueba que
        \[
        P_r(t) = \frac{1 - r^2}{1 - 2r\cos t + r^2} > 0,
        \qquad
        \frac{1}{2\pi}\int_{-\pi}^{\pi}P_r = 1 .
        \]
  \item Prueba que para $\delta \leq \abs t \leq \pi$: $P_r(t) \leq
        \frac{1 - r^2}{1 - 2r\cos\delta + r^2} \to 0$ cuando $r \to
        1^-$, \omterm{def:b2:funcseq:def}{uniformemente}.
  \item Deduce que, para toda $f$ \omterm{def:b2:metric:continuity}{continua} $2\pi$-periódica, las
        \emph{medias de Abel} $(f * P_r)(x) = \sum_n r^{\abs
        n}c_n(f)\,\eu^{\iu nx}$ convergen a $f$ \omterm{def:b2:funcseq:def}{uniformemente}
        cuando $r \to 1^-$: la hermana de parámetro \omterm{def:b2:metric:continuity}{continuo} del
        teorema de Fejér, y la encarnación en Fourier de la
        \omterm{thm:b2:series:abel}{sumación de Abel} del capítulo de series de potencias.
\end{enumerate}
\end{exercise}

\section{Problema: el teorema de Fejér y sus dividendos}

\begin{problem}\label{pb:b2:fourier:1}
El teorema de Dirichlet necesita que $f$ sea $C^1$ a trozos; para $f$
meramente \omterm{def:b2:metric:continuity}{continua}, las sumas parciales $S_N(f)$ pueden portarse mal.
El descubrimiento de Fejér: sus \emph{medias de Cesàro} nunca lo
hacen. El motor es la positividad del núcleo de Fejér, y la cosecha
es inmensa: aproximación trigonométrica uniforme (Weierstrass),
unicidad de los \omterm{def:b2:fourier:coefficients}{coeficientes de Fourier}, Parseval para toda función
de este capítulo (lo que elimina el ``admitido'' del~\cref{thm:b2:fourier:parseval}), el teorema de equidistribución de
Weyl y ---coronando un siglo de geometría--- la desigualdad
isoperimétrica.\index{teorema de Fejer@teorema de
Fejér}\index{equidistribución}\index{desigualdad isoperimétrica} En
todo el problema, $f$ es $2\pi$-periódica y \omterm{def:b2:metric:continuity}{continua} a trozos, y
\[
\sigma_N(f) = \frac{S_0(f) + S_1(f) + \dots +
S_{N-1}(f)}{N} .
\]

\medskip
\noindent\textbf{Parte I --- El núcleo de Fejér.}
\begin{enumerate}
  \item Prueba que $\sigma_N(f)(x) =
        \frac{1}{2\pi}\int_{-\pi}^{\pi}f(x+u)\,F_N(u)\,\dd u$ con
        $F_N = \frac{D_0 + \dots + D_{N-1}}{N}$, y que
        $\frac{1}{2\pi}\int_{-\pi}^{\pi}F_N = 1$.
  \item Demuestra la forma cerrada, para $u \notin 2\pi\Z$:
        \[
        F_N(u) = \frac{1}{N}\,
        \frac{\sin^2\bigl(\frac{Nu}{2}\bigr)}
        {\sin^2\bigl(\frac u2\bigr)} \;\geq\; 0
        \]
        \emph{(suma $\sin\bigl((n+\frac12)u\bigr)$ como parte
        imaginaria de una serie geométrica)}.
  \item Prueba la estimación de concentración: para $0 < \delta \leq
        \abs u \leq \pi$,
        \[
        F_N(u) \leq \frac{1}{N\sin^2\frac\delta2}
        \xrightarrow[N\to\infty]{} 0
        \quad\text{uniformemente} :
        \]
        $(F_N)$ es una identidad aproximada positiva.
  \item (Teorema de Fejér) Demuestra que si $f$ es \omterm{def:b2:metric:continuity}{continua} y
        $2\pi$-periódica, entonces $\sigma_N(f) \to f$
        \emph{\omterm{def:b2:funcseq:def}{uniformemente}} sobre $\R$ \emph{(parte la integral de
        $\bigl(f(x+u) - f(x)\bigr)F_N(u)$ en $\abs u = \delta$; usa
        Heine y las preguntas 1--3)}.
  \item Para $f$ \omterm{def:b2:metric:continuity}{continua} a trozos, prueba la versión \omterm{def:b2:funcseq:def}{puntual}
        $\sigma_N(f)(x) \to \frac{f(x^+) + f(x^-)}{2}$ en todo $x$, y
        la cota uniforme $\norm{\sigma_N(f)}_\infty \leq \norm
        f_\infty$ \emph{(¡positividad!)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Weierstrass, unicidad, Parseval.}
\begin{enumerate}[resume]
  \item (Weierstrass trigonométrico) Deduce que toda función
        \omterm{def:b2:metric:continuity}{continua} $2\pi$-periódica es límite uniforme de polinomios
        trigonométricos.
  \item (Unicidad) Deduce que una $f$ \omterm{def:b2:metric:continuity}{continua} con $c_n(f) = 0$ para
        todo $n$ es idénticamente nula: dos funciones periódicas
        \omterm{def:b2:metric:continuity}{continuas} con los mismos \omterm{def:b2:fourier:coefficients}{coeficientes de Fourier} coinciden
        (\cref{exo:b2:fourier:4}, ahora sin nada admitido).
  \item (Parseval, caso \omterm{def:b2:metric:continuity}{continuo}) Usando la propiedad de proyección
        de $S_N$ (\cref{prop:b2:fourier:bessel}) y que $\sigma_Nf \in
        \mathcal T_{N-1} \subseteq \mathcal T_N$, demuestra que
        \[
        \norm{f - S_Nf}_2 \leq \norm{f - \sigma_Nf}_2
        \leq \norm{f - \sigma_Nf}_\infty
        \xrightarrow[N\to\infty]{} 0 ,
        \]
        y concluye la identidad de Parseval para toda $f$
        $2\pi$-periódica \emph{\omterm{def:b2:metric:continuity}{continua}}.
  \item (Parseval, caso \omterm{def:b2:metric:continuity}{continuo} a trozos) Dadas $f$ \omterm{def:b2:metric:continuity}{continua} a
        trozos y $\varepsilon > 0$, construye una $g$ periódica
        \omterm{def:b2:metric:continuity}{continua} con $\norm{f - g}_2 \leq \varepsilon$
        \emph{(sustituye $f$ por una interpolación afín en
        intervalos minúsculos alrededor de los saltos)} y deduce que
        $\norm{f - S_Nf}_2 \to 0$ \emph{(usa Bessel: $\norm{S_Nh}_2
        \leq \norm h_2$)}: Parseval vale en toda la generalidad
        enunciada en el~\cref{thm:b2:fourier:parseval}; el
        ``admitido'' ha desaparecido.
  \item (Sin Gibbs para Fejér) Contrasta con el~\cref{exo:b2:fourier:8}: prueba que, para la onda cuadrada
        $f$, $\abs{\sigma_N(f)} \leq 1$ en todas partes y para todo
        $N$; promediar al modo de Cesàro borra la sobreoscilación
        que persigue a $S_N$. Explica en una frase qué propiedad de
        $F_N$ es la responsable.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Ritmos.}
\begin{enumerate}[resume]
  \item Demuestra las dos cotas del núcleo, para $0 < \abs u \leq
        \pi$:
        \[
        F_N(u) \leq N,
        \qquad
        F_N(u) \leq \frac{\pi^2}{N u^2}
        \]
        \emph{(para la primera, $\abs{\sin N\theta} \leq
        N\abs{\sin\theta}$ por inducción; para la segunda,
        $\sin\frac u2 \geq \frac{u}{\pi}$ sobre
        $\intcc{0}{\pi}$)}.
  \item Deduce la estimación del primer momento
        \[
        \frac{1}{2\pi}\int_{-\pi}^{\pi}\abs u\,F_N(u)\,\dd u
        \;\leq\; \frac{C\,\ln N}{N}
        \qquad (N \geq 2)
        \]
        para una constante explícita \emph{(parte en $\abs u =
        \frac1N$)}.
  \item Concluye: si $f$ es $L$-lipschitziana y $2\pi$-periódica,
        entonces
        \[
        \norm{\sigma_N f - f}_\infty \leq \frac{C\,L\ln N}{N} .
        \]
  \item (Saturación) Calcula $\sigma_N(e_1)$ para $e_1(t) = \eu^{\iu
        t}$ y prueba que $\norm{\sigma_Ne_1 - e_1}_\infty =
        \frac1N$: incluso para las funciones más regulares, Fejér
        no converge más deprisa que $\frac1N$; el análogo exacto de
        la saturación de Bernstein del problema de fin de semana del
        capítulo de sucesiones de funciones.
  \item (Localización) Prueba que si $f$ (\omterm{def:b2:metric:continuity}{continua} a trozos) se anula
        sobre $\intoo{x - \delta}{x + \delta}$, entonces
        $\sigma_N(f)(x) \to 0$, por muy salvaje que sea $f$ en el
        resto: la convergencia de las medias en $x$ solo ve a $f$
        cerca de $x$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El teorema de equidistribución de
Weyl.} Una sucesión $(x_n)_{n\geq1}$ de $\intco{0}{1}$ está
\emph{equidistribuida} cuando, para todo intervalo $\intcc{a}{b}
\subseteq \intcc{0}{1}$,
\[
\frac{\#\{n \leq N : x_n \in \intcc ab\}}{N}
\xrightarrow[N\to\infty]{} b - a .
\]
\begin{enumerate}[resume]
  \item Prueba que $(x_n)$ está equidistribuida en cuanto
        $\frac1N\sum_{n\leq N}f(x_n) \to \int_0^1 f$ para toda $f$
        $1$-periódica \emph{\omterm{def:b2:metric:continuity}{continua}} \emph{(encaja la indicatriz
        de $\intcc ab$ entre dos funciones \omterm{def:b2:metric:continuity}{continuas} afines a trozos
        cuyas integrales difieran en $\varepsilon$)}.
  \item (Criterio de Weyl, suficiencia) Supongamos que
        \[
        \frac{1}{N}\sum_{n=1}^{N}\eu^{2\iu\pi kx_n}
        \xrightarrow[N\to\infty]{} 0
        \qquad\text{para todo } k \in \Z\setminus\{0\} .
        \]
        Prueba que $\frac1N\sum f(x_n) \to \int_0^1f$ primero para
        los polinomios trigonométricos y después para toda $f$
        $1$-periódica \omterm{def:b2:metric:continuity}{continua} gracias a la pregunta 6 (trasladada
        al periodo $1$): con la pregunta 16, $(x_n)$ está
        equidistribuida.
  \item Sea $\alpha$ irracional y $x_n = \{n\alpha\}$ (la parte
        fraccionaria). Acota la suma geométrica
        \[
        \Bigl|\sum_{n=1}^{N}\eu^{2\iu\pi kn\alpha}\Bigr|
        \leq \frac{2}{\abs{1 - \eu^{2\iu\pi k\alpha}}}
        \qquad (k \neq 0),
        \]
        y concluye el \emph{teorema de Weyl}: $(\{n\alpha\})$ está
        equidistribuida en $\intco{0}{1}$.
  \item Deduce que $(\{n\alpha\})$ es densa en $\intcc{0}{1}$ para
        $\alpha$ irracional, y explica en una frase por qué la
        equidistribución es estrictamente más fuerte que la
        densidad.
  \item (Dígitos iniciales) Demuestra que la proporción de enteros
        $n \leq N$ tales que $2^n$ tiene por primer dígito (decimal)
        un $1$ tiende a $\log_{10}2 \approx 0.301$ \emph{(que el
        primer dígito sea $1$ significa $\{n\log_{10}2\} \in
        \intco{0}{\log_{10}2}$; prueba que $\log_{10}2$ es
        irracional)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- La desigualdad isoperimétrica.} Sea
$\Gamma$ una curva cerrada simple de clase $C^1$, de longitud $L$,
que encierra un área con signo $A$, parametrizada por longitud de
arco reescalada: $z(t) = x(t) + \iu y(t)$, $2\pi$-periódica, con
$\abs{z'(t)} = \frac{L}{2\pi}$ constante; el área encerrada es
\[
A = \frac12\int_0^{2\pi}\bigl(x\,y' - y\,x'\bigr)\dd t
= \frac{1}{2}\,\Im\int_0^{2\pi}\conj{z}\,z'\,\dd t
\]
(tomada aquí como definición del área con signo; el capítulo de
integrales múltiples demuestra que coincide con la intuitiva,
mediante la fórmula de Green).
\begin{enumerate}[resume]
  \item Desarrolla $z(t) = \sum_{n\in\Z}c_n\eu^{\iu nt}$ (la serie de
        una función $C^1$, \omterm{def:b2:funcseq:series}{normalmente} convergente) y demuestra,
        aplicando Parseval a $z'$, que
        \[
        \frac{L^2}{2\pi} = \int_0^{2\pi}\abs{z'}^2\dd t
        = 2\pi\sum_{n\in\Z}n^2\abs{c_n}^2 .
        \]
  \item Demuestra igualmente que $A = \pi\sum_{n\in\Z}
        n\,\abs{c_n}^2$ \emph{(Parseval en su forma polarizada:
        $\frac{1}{2\pi}\int\conj f g = \sum \conj{c_n(f)}c_n(g)$,
        aplicada a $f = z$, $g = z'$)}.
  \item (Hurwitz) Concluye que
        \[
        L^2 - 4\pi A = 4\pi^2\sum_{n\in\Z}
        (n^2 - n)\abs{c_n}^2 \;\geq\; 0 ,
        \]
        con igualdad si y solo si $z(t) = c_0 + c_1\eu^{\iu t}$, es
        decir, una circunferencia. La desigualdad isoperimétrica:
        entre las curvas cerradas de longitud $L$, solo la
        circunferencia encierra un área $\frac{L^2}{4\pi}$.
  \item Comprobaciones de sensatez: verifica la igualdad para la
        circunferencia de radio $R$ y la desigualdad estricta para
        el cuadrado de lado $a$; explica por qué $n^2 - n \geq 0$
        para todo entero $n$, incluidos los negativos, y dónde se
        usó que la parametrización tiene velocidad constante.
  \item Síntesis. Una frase para cada punto: (i) la única propiedad
        de $F_N$ de la que fluyen las partes I--III, y de la que
        carece $D_N$; (ii) cómo se relaciona aquí la sumación de
        Cesàro con el problema de fin de semana del capítulo de
        series de potencias (Frobenius); (iii) qué dividendo usó
        solo Weierstrass (pregunta 6) y cuál necesitó el Parseval
        \omterm{def:b2:metric:complete}{completo}; (iv) una frase sobre lo que añade el volumen del
        tercer año (la \omterm{def:b2:metric:complete}{completitud} de $L^2$: las series de Fourier
        como base de Hilbert).
\end{enumerate}
\end{problem}
