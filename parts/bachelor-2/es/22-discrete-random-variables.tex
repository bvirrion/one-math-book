\chapter{Variables aleatorias discretas}\label{ch:b2:randomvar}

Las \omterm{def:b2:randomvar:law}{variables aleatorias} organizan los cálculos de probabilidad en
torno a funciones y no a \omterm{def:b2:proba:space}{sucesos}. Sobre espacios \omterm{def:b2:structures:countable}{numerables}, la
teoría se apoya en las \omterm{def:b2:series:summable}{familias sumables} del~\cref{ch:b2:series}: la
\omterm{def:b2:randomvar:expectation}{esperanza} es la suma de una familia indexada por el
\omterm{def:b2:proba:space}{espacio muestral}, y todas sus propiedades ---linealidad, transferencia,
fórmula del producto para variables independientes--- son teoremas sobre
\omterm{def:b2:series:summable}{familias sumables}. El capítulo demuestra las desigualdades clave de
Markov, Chebyshev, Cauchy--Schwarz y Jensen, y termina con las
\omterm{def:b2:randomvar:law}{leyes} clásicas y la ley débil de los grandes números, cuya demostración
ocupa dos líneas en cuanto se dispone de Chebyshev.

\section{Variables aleatorias y sus leyes}

\begin{definition}[Variable aleatoria discreta; ley]\label{def:b2:randomvar:law}
Sea $(\Omega, \P)$ un \omterm{def:b2:proba:space}{espacio de probabilidad} \omterm{def:b2:structures:countable}{numerable}. Una
\emph{variable aleatoria}\index{variable aleatoria} es una aplicación
$X \colon \Omega \to E$ ($E$ un conjunto cualquiera; variable
aleatoria \emph{real} cuando $E = \R$). Su \emph{ley}\index{ley!de
una variable aleatoria} (o \emph{distribución}\index{distribución})
es la \omterm{def:b2:proba:space}{medida de probabilidad} $\P_X$ sobre el \omterm{def:b2:structures:countable}{conjunto numerable}
$X(\Omega)$ definida por
\[
\P_X(\{x\}) = \P(X = x)
= \P\bigl(\{\omega : X(\omega) = x\}\bigr) .
\]
\end{definition}

\begin{example}[Las leyes clásicas]\label{ex:b2:randomvar:laws}
\leavevmode
\begin{itemize}
\item \emph{Bernoulli} $\mathcal{B}(p)$: $X \in \{0, 1\}$, $\P(X =
1) = p$. La indicatriz de un \omterm{def:b2:proba:space}{suceso}.
\item \emph{Binomial} $\mathcal{B}(n, p)$: $\P(X = k) = \binom nk
p^k(1-p)^{n-k}$, $0 \leq k \leq n$: el número de éxitos en $n$
ensayos de Bernoulli \omterm{def:b2:proba:independence}{independientes} (volumen de secundaria; se
vuelve a demostrar más abajo mediante sumas de variables
\omterm{def:b2:proba:independence}{independientes}).
\item \emph{Geométrica} $\mathcal{G}(p)$: $\P(X = k) =
(1-p)^{k-1}p$, $k \in \N^*$: el rango del primer éxito
(\cref{ex:b2:proba:geometric}).
\item \emph{Poisson} $\mathcal{P}(\lambda)$: $\P(X = k) =
e^{-\lambda}\frac{\lambda^k}{k!}$, $k \in \N$; una
\omterm{def:b2:proba:space}{medida de probabilidad} por la serie exponencial. La ley de los
sucesos raros (\cref{ch:b2:genfun}).
\end{itemize}
\end{example}

\begin{remark}[Qué modela cada ley]
Las cuatro \omterm{def:b2:randomvar:law}{leyes} responden a cuatro preguntas primitivas: la de
Bernoulli, ``¿ocurrió?''; la binomial, ``¿cuántas veces en $n$
intentos?''; la geométrica, ``¿cuánto hasta la primera vez?''; y la
de Poisson, ``¿cuántos \omterm{def:b2:proba:space}{sucesos} a una tasa dada, cuando los intentos
son muchos e individualmente improbables?''. Reconocer la pregunta es
nueve décimas partes de la modelización: las sumas de indicatrices
apuntan a la binomial, los tiempos de espera a la geométrica y los
recuentos de \omterm{def:b2:proba:space}{sucesos} raros a la de Poisson, con el paso de la
binomial a la de Poisson precisado por la ley de los sucesos raros en
el~\cref{ch:b2:genfun}.
\end{remark}

\begin{proposition}[Ausencia de memoria de la ley
geométrica]\label{prop:b2:randomvar:memoryless}
Si $X \sim \mathcal{G}(p)$, entonces, para todos $m, n \in \N$:
\[
\P(X > m + n \mid X > m) = \P(X > n) ,
\]
y las \omterm{def:b2:randomvar:law}{leyes} geométricas son las únicas \omterm{def:b2:randomvar:law}{leyes} sobre $\N^*$ con esta
propiedad.
\end{proposition}

\begin{proof}
Sumando los pesos geométricos, $\P(X > n) = (1-p)^n$. De ahí,
\[
\P(X > m + n \mid X > m)
= \frac{\P(X > m + n)}{\P(X > m)}
= \frac{(1-p)^{m+n}}{(1-p)^m} = (1-p)^n = \P(X > n).
\]
Recíprocamente, si $G(n) = \P(X > n)$ cumple $G(m + n) = G(m)G(n)$
con $G(0) = 1$, entonces $G(n) = G(1)^n$ por inducción; con $q =
G(1) \in \intco{0}{1}$, o bien $q = 0$, o bien la \omterm{def:b2:randomvar:law}{ley} es
$\mathcal{G}(1 - q)$: $\P(X = k) = G(k-1) - G(k) = q^{k-1}(1 - q)$.
\end{proof}

\begin{example}[Ningún número está nunca ``a punto de salir'']
Lánzase un dado esperando un seis: el tiempo de espera es $X \sim
\mathcal G(1/6)$. La ausencia de memoria dice que, tras $10$ tiradas
infructuosas, la espera \emph{restante} $X - 10$, dado $X > 10$, es
de nuevo $\mathcal G(1/6)$: la espera esperada condicionada sigue
siendo de $6$ tiradas, exactamente como al principio. El dado no
recuerda, y ningún seis está nunca ``a punto de salir''; la falacia
del jugador es la creencia de que la \omterm{def:b2:randomvar:law}{ley} condicionada debería
haberse desplazado. Recíprocamente, la mitad de unicidad de la
proposición dice que esa indiferencia \emph{caracteriza} los tiempos
de espera geométricos: todo tiempo de espera cuya previsión nunca se
actualiza es geométrico. Las colas y los tiempos de vida reales sí
suelen actualizarse, que es precisamente cómo se detecta que no son
geométricos.
\end{example}

\section{Esperanza}

\begin{definition}[Esperanza]\label{def:b2:randomvar:expectation}
Una \omterm{def:b2:randomvar:law}{variable aleatoria} real $X$ sobre $(\Omega, \P)$ \emph{tiene
esperanza} si la familia
$\bigl(X(\omega)\,\P(\{\omega\})\bigr)_{\omega \in \Omega}$ es
\omterm{def:b2:series:summable}{sumable} (\cref{ch:b2:series}); y su \emph{esperanza}\index{esperanza}
es entonces
\[
\E(X) = \sum_{\omega \in \Omega} X(\omega)\,\P(\{\omega\}) .
\]
\end{definition}

\begin{theorem}[Teorema de transferencia]\label{thm:b2:randomvar:transfer}
$X$ tiene \omterm{def:b2:randomvar:expectation}{esperanza} si y solo si la familia $\bigl(x\,\P(X =
x)\bigr)_{x \in X(\Omega)}$ es \omterm{def:b2:series:summable}{sumable}, y entonces
\[
\E(X) = \sum_{x \in X(\Omega)} x\,\P(X = x) .
\]
Más en general, para $f \colon X(\Omega) \to \R$, la variable $f(X)$
tiene \omterm{def:b2:randomvar:expectation}{esperanza} si y solo si $\sum_x \abs{f(x)}\,\P(X = x) <
\infty$, y entonces $\E(f(X)) = \sum_x f(x)\,\P(X = x)$.
\end{theorem}

\begin{proof}
Pártase $\Omega$ en los conjuntos de nivel $\Omega_x = \{X = x\}$, $x
\in X(\Omega)$. Por el teorema de sumación por paquetes para
\omterm{def:b2:series:summable}{familias sumables} (\cref{ch:b2:series}), la familia
$(X(\omega)\P(\{\omega\}))_\omega$ es \omterm{def:b2:series:summable}{sumable} si y solo si lo es cada
paquete (automático: $\sum_{\omega \in \Omega_x}\abs{x}\P(\{\omega\})
= \abs x\,\P(X = x)$) \emph{y} la familia de sumas de paquetes
$\bigl(x\,\P(X = x)\bigr)_x$ es \omterm{def:b2:series:summable}{sumable}; y entonces las sumas totales
coinciden. Para $f(X)$: aplíquese el enunciado ya demostrado a la
variable $Y = f \circ X$, cuyos conjuntos de nivel son $\{Y = y\} =
\bigsqcup_{x : f(x) = y}\{X = x\}$; una segunda sumación por paquetes
convierte $\sum_y y\,\P(Y = y)$ en $\sum_x f(x)\,\P(X = x)$,
agrupando ahora los paquetes los valores $x$ según su imagen $f(x)$,
con la \omterm{def:b2:series:summable}{sumabilidad} absoluta de una familia equivalente a la de la
otra.
\end{proof}

\begin{theorem}[Propiedades de la esperanza]\label{thm:b2:randomvar:linearity}
Sobre el conjunto de las \omterm{def:b2:randomvar:law}{variables aleatorias} con \omterm{def:b2:randomvar:expectation}{esperanza}:
\begin{enumerate}
\item (Linealidad) $\E(aX + bY) = a\,\E(X) + b\,\E(Y)$.
\item (Positividad y monotonía) $X \geq 0 \Rightarrow \E(X) \geq 0$;
$X \leq Y \Rightarrow \E(X) \leq \E(Y)$; y $\abs{\E(X)} \leq
\E(\abs X)$.
\item (Dominación) Si $\abs X \leq Z$ y $Z$ tiene \omterm{def:b2:randomvar:expectation}{esperanza},
entonces $X$ también.
\end{enumerate}
\end{theorem}

\begin{proof}
Todas son propiedades de las sumas de \omterm{def:b2:series:summable}{familias sumables}
(\cref{ch:b2:series}): la linealidad de la suma, la positividad
término a término y el criterio de dominación para la \omterm{def:b2:series:summable}{sumabilidad}.
(Nótese que la linealidad es inmediata sobre la \emph{definición}
como suma sobre $\Omega$, mientras que resultaría incómoda sobre la
fórmula de transferencia; una ventaja de definir $\E$ aguas arriba.)
\end{proof}

\begin{example}\label{ex:b2:randomvar:expectations}
$X \sim \mathcal{B}(n, p)$: escribiendo $X = X_1 + \dots + X_n$ como
suma de indicatrices de Bernoulli y usando la linealidad, $\E(X) =
np$; sin necesidad de coeficientes binomiales. $X \sim
\mathcal{G}(p)$: $\E(X) = \sum_{k\geq1}k(1-p)^{k-1}p =
p\cdot\frac{1}{(1 - (1-p))^2} = \frac1p$, derivando la serie
geométrica dentro de su disco (\cref{ch:b2:powerseries}). $X \sim
\mathcal{P}(\lambda)$: $\E(X) = \sum_{k\geq1}k
e^{-\lambda}\frac{\lambda^k}{k!} = \lambda
e^{-\lambda}\sum_{j\geq0}\frac{\lambda^j}{j!} = \lambda$.
\end{example}

\begin{example}[La transferencia en acción]\label{ex:b2:randomvar:transferex}
Para $X \sim \mathcal P(\lambda)$, calculemos
$\E\bigl(\frac1{1+X}\bigr)$; la \omterm{def:b2:randomvar:law}{ley} de $\frac1{1+X}$ es incómoda,
pero la transferencia no la pide nunca:
\[
\E\Bigl(\frac1{1+X}\Bigr)
= \sum_{k\geq0}\frac{1}{k+1}\,\eu^{-\lambda}
\frac{\lambda^k}{k!}
= \frac{\eu^{-\lambda}}{\lambda}\sum_{k\geq0}
\frac{\lambda^{k+1}}{(k+1)!}
= \frac{\eu^{-\lambda}}{\lambda}\bigl(\eu^\lambda - 1\bigr)
= \frac{1 - \eu^{-\lambda}}{\lambda} .
\]
Dos lecciones. En lo computacional: reconocer una serie exponencial
desplazada es todo el trabajo; la transferencia reduce las
\omterm{def:b2:randomvar:expectation}{esperanzas} de $f(X)$ a manipulación de series. En lo estructural:
el valor ingenuo de sustituir sería $\frac1{1 + \E X} = \frac1{1 +
\lambda}$, y la respuesta verdadera es mayor,
\[
\frac{1 - \eu^{-\lambda}}{\lambda} \geq
\frac{1}{1 + \lambda},
\]
exactamente como exige la desigualdad de Jensen para la función
convexa $t \mapsto \frac1{1+t}$. Las \omterm{def:b2:randomvar:expectation}{esperanzas} de imágenes
\omterm{def:b2:affine:convex}{convexas} quedan por encima del valor ingenuo de sustituir, y la
transferencia más una comprobación con series hace concreta la
desigualdad abstracta.
\end{example}

\begin{theorem}[Independencia y productos]\label{thm:b2:randomvar:product}
Dos \omterm{def:b2:randomvar:law}{variables aleatorias} $X, Y$ son \emph{\omterm{def:b2:proba:independence}{independientes}} si $\P(X =
x, Y = y) = \P(X = x)\P(Y = y)$ para todos $x, y$; equivalentemente,
si los \omterm{def:b2:proba:space}{sucesos} $\{X \in A\}$ y $\{Y \in B\}$ son \omterm{def:b2:proba:independence}{independientes} para
todos $A, B$. Si $X$ e $Y$ son variables reales \omterm{def:b2:proba:independence}{independientes} con
\omterm{def:b2:randomvar:expectation}{esperanza}, entonces $XY$ tiene \omterm{def:b2:randomvar:expectation}{esperanza} y
\[
\E(XY) = \E(X)\,\E(Y) .
\]
\end{theorem}

\begin{proof}
La equivalencia de las dos formulaciones se sigue sumando la
identidad \omterm{def:b2:funcseq:def}{puntual} sobre $(x, y) \in A \times B$ ($\sigma$-aditividad
dos veces). Para el producto: la familia doble $\bigl(xy\,\P(X =
x)\P(Y = y)\bigr)_{(x,y)}$ es \omterm{def:b2:series:summable}{sumable}, pues por Fubini para familias
(\cref{ch:b2:series})
\[
\sum_{x, y}\abs x \abs y\,\P(X{=}x)\P(Y{=}y)
= \Bigl(\sum_x \abs x \P(X{=}x)\Bigr)
\Bigl(\sum_y \abs y \P(Y{=}y)\Bigr) < \infty ;
\]
y por la \omterm{def:b2:proba:independence}{independencia}, esta familia es exactamente $\bigl(xy\,\P(X =
x, Y = y)\bigr)$, cuya suma es $\E(XY)$ por transferencia aplicada a
la variable $(X, Y) \mapsto xy$; Fubini evalúa de nuevo la suma sin
signos como el producto $\E(X)\E(Y)$.
\end{proof}

\begin{example}[Productos, con y sin
independencia]\label{ex:b2:randomvar:productcontrast}
Lánzense dos dados equilibrados. Si $Y$ es el segundo dado
(\omterm{def:b2:proba:independence}{independiente} del primero), $\E(XY) = \E(X)\E(Y) = 3.5^2 = 12.25$.
Si, en cambio, $Y = X$ (el ``producto'' de un dado consigo mismo),
\[
\E(X^2) = \frac{1 + 4 + 9 + 16 + 25 + 36}{6} = \frac{91}{6}
\approx 15.17 \neq 12.25 :
\]
las mismas \omterm{def:b2:randomvar:law}{leyes} marginales en ambos escenarios, \omterm{def:b2:randomvar:law}{leyes} conjuntas
distintas y \omterm{def:b2:randomvar:expectation}{esperanzas} del producto distintas. La moraleja, digna
de grabarse: $\E(XY)$ es un funcional del \emph{par}, no de las dos
marginales; y la diferencia $\E(X^2) - \E(X)^2 \approx 2.92$ es, por
König--Huygens, precisamente la \omterm{def:b2:randomvar:variance}{varianza} $\frac{35}{12}$ del dado.
\end{example}

\section{Varianza, covarianza y las desigualdades clásicas}

\begin{definition}[Momentos, varianza]\label{def:b2:randomvar:variance}
$X$ tiene \emph{momento de orden 2} si $X^2$ tiene \omterm{def:b2:randomvar:expectation}{esperanza}
(entonces $X$ también, por dominación: $\abs X \leq \frac{1 +
X^2}{2}$). Su \emph{varianza}\index{varianza} y su \emph{desviación
típica}\index{desviación típica} son entonces
\[
V(X) = \E\bigl((X - \E(X))^2\bigr)
= \E(X^2) - \E(X)^2 ,
\qquad
\sigma(X) = \sqrt{V(X)} ,
\]
(la segunda forma ---la fórmula de \emph{König--Huygens}---
desarrollando el cuadrado y usando la linealidad:
\[
\E\bigl((X - \E X)^2\bigr)
= \E\bigl(X^2 - 2X\,\E X + \E(X)^2\bigr)
= \E(X^2) - 2\,\E(X)^2 + \E(X)^2 ,
\]
donde el término central usa que $\E X$ es una constante). Para $X,
Y$ con momentos de orden dos, la \emph{covarianza}\index{covarianza}
es
\[
\operatorname{Cov}(X, Y)
= \E\bigl((X - \E X)(Y - \E Y)\bigr)
= \E(XY) - \E(X)\E(Y) .
\]
\end{definition}

\begin{theorem}[Caja de herramientas de la varianza]\label{thm:b2:randomvar:variancerules}
Para variables con momentos de orden dos:
\begin{enumerate}
\item $V(aX + b) = a^2\,V(X)$;
\item $V(X + Y) = V(X) + V(Y) + 2\operatorname{Cov}(X, Y)$ y, más en
general,
\[
V\Bigl(\sum_{i=1}^n X_i\Bigr)
= \sum_{i=1}^n V(X_i)
+ 2\sum_{i < j}\operatorname{Cov}(X_i, X_j) ;
\]
\item si $X, Y$ son \omterm{def:b2:proba:independence}{independientes}, $\operatorname{Cov}(X, Y) = 0$
(el recíproco es falso), de modo que las \omterm{def:b2:randomvar:variance}{varianzas} de variables
\omterm{def:b2:proba:independence}{independientes} se suman.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1} y \emph{2} son desarrollos de cuadrados más linealidad; los
productos $X_iX_j$ tienen \omterm{def:b2:randomvar:expectation}{esperanza} por Cauchy--Schwarz más abajo (o
por $\abs{X_iX_j} \leq \frac{X_i^2 + X_j^2}{2}$). \emph{3} es el~\cref{thm:b2:randomvar:product} aplicado a las variables centradas.
Un contraejemplo estándar del recíproco: $X$ uniforme sobre $\{-1, 0,
1\}$ e $Y = X^2$ están incorreladas ($\E(XY) = \E(X^3) = 0 = \E X
\cdot \E Y$), pero son claramente dependientes.
\end{proof}

\begin{theorem}[Desigualdades de Markov y de
Chebyshev]\label{thm:b2:randomvar:markov}
\leavevmode
\begin{enumerate}
\item (Markov) Si $X \geq 0$ tiene \omterm{def:b2:randomvar:expectation}{esperanza}, entonces, para todo $a
> 0$:
\[
\P(X \geq a) \leq \frac{\E(X)}{a} .
\]
\item (Chebyshev) Si $X$ tiene momento de orden dos, entonces, para
todo $\varepsilon > 0$:
\[
\P\bigl(\abs{X - \E(X)} \geq \varepsilon\bigr)
\leq \frac{V(X)}{\varepsilon^2} .
\]
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} Punto a punto, $a\,\mathbf{1}_{X \geq a} \leq X$ (sobre el
\omterm{def:b2:proba:space}{suceso}, el miembro izquierdo es $a \leq X$; fuera de él, $0 \leq
X$). Tómense \omterm{def:b2:randomvar:expectation}{esperanzas}: $a\,\P(X \geq a) \leq \E(X)$ por monotonía y
$\E(\mathbf{1}_A) = \P(A)$. \emph{2.} Aplíquese Markov a la variable
no negativa $(X - \E X)^2$ al nivel $a = \varepsilon^2$: el \omterm{def:b2:proba:space}{suceso}
$\{(X - \E X)^2 \geq \varepsilon^2\}$ es exactamente $\{\abs{X - \E
X} \geq \varepsilon\}$.
\end{proof}

\begin{example}[Incorreladas pero pegadas la una a la
otra]\label{ex:b2:randomvar:sumdiff}
Lánzense dos dados equilibrados, $X$ e $Y$ \omterm{def:b2:proba:independence}{independientes}, y póngase
$S = X + Y$, $D = X - Y$. Por la bilinealidad de la \omterm{def:b2:randomvar:variance}{covarianza},
\[
\operatorname{Cov}(S, D) = V(X) - V(Y) +
\operatorname{Cov}(Y, X) - \operatorname{Cov}(X, Y) = V(X) -
V(Y) = 0 :
\]
la suma y la diferencia están incorreladas. ¿\omterm{def:b2:proba:independence}{Independientes}?
Desde luego que no: $S = 12$ fuerza $D = 0$, mientras que $\P(D = 0)
= \frac16$ incondicionalmente. La correlación solo pone a prueba la
parte \emph{lineal} de una dependencia; aquí, la dependencia la lleva
la restricción de que $S$ y $D$ tengan la misma paridad, invisible
para la \omterm{def:b2:randomvar:variance}{covarianza}. (Para este par, la \omterm{def:b2:randomvar:variance}{covarianza} nula necesitaba
$V(X) = V(Y)$: fueron las distribuciones idénticas, y no la
\omterm{def:b2:proba:independence}{independencia}, las que hicieron el trabajo.)
\end{example}

\begin{example}[Cuándo es exacta la de Markov]\label{ex:b2:randomvar:markovsharp}
La desigualdad de Markov es una igualdad precisamente cuando no se
desperdicia nada en la cota $a\,\mathbf 1_{X\geq a} \leq X$: la
variable ha de tomar solo los valores $0$ y $a$. En concreto, si
$\P(X = a) = \pi$ y $\P(X = 0) = 1 - \pi$, entonces $\E(X) = a\pi$ y
\[
\P(X \geq a) = \pi = \frac{\E(X)}{a} .
\]
Una lectura realista: en una población cuya riqueza media es $100$ y
donde la riqueza es $0$ o $10^6$, la proporción de millonarios es
exactamente $10^{-4}$; la cota de Markov, alcanzada exactamente por
una desigualdad maximal. Siempre que $X$ se reparta por valores
intermedios, la cota es estricta, a menudo desaforadamente; pero,
como muestra el caso extremo, de la sola media no puede extraerse
ninguna desigualdad mejor.
\end{example}

\begin{example}[Chebyshev es óptima, sin hipótesis
adicionales]\label{ex:b2:randomvar:chebsharp}
Fíjense $\varepsilon > 0$ y $q \in \intoc01$, y sea $X$ la variable
que toma los valores $\pm\varepsilon$ con probabilidad $\frac q2$
cada uno y $0$ con probabilidad $1 - q$. Entonces $\E(X) = 0$, $V(X)
= q\varepsilon^2$, y
\[
\P\bigl(\abs{X - \E X} \geq \varepsilon\bigr) = q
= \frac{V(X)}{\varepsilon^2} :
\]
igualdad en Chebyshev. Así pues, la desigualdad no puede mejorarse
usando solo la \omterm{def:b2:randomvar:variance}{varianza}; el decaimiento $1/\varepsilon^2$ es el
precio exacto de la información de segundo momento. Un decaimiento
más rápido exige hipótesis más fuertes: la acotación de la variable
compra concentración \emph{exponencial}, como anticipa el~\cref{exo:b2:randomvar:7} y desarrolla sistemáticamente el problema
de fin de semana de este capítulo.
\end{example}

\begin{theorem}[Cauchy--Schwarz y Jensen]\label{thm:b2:randomvar:jensen}
\leavevmode
\begin{enumerate}
\item (Cauchy--Schwarz) Si $X, Y$ tienen momentos de orden dos, $XY$
tiene \omterm{def:b2:randomvar:expectation}{esperanza} y $\E(XY)^2 \leq \E(X^2)\,\E(Y^2)$; en
consecuencia, $\operatorname{Cov}(X,Y)^2 \leq V(X)V(Y)$.
\item (Jensen) Si $\varphi \colon I \to \R$ es convexa sobre un
intervalo que contiene $X(\Omega)$, y $X$ y $\varphi(X)$ tienen
\omterm{def:b2:randomvar:expectation}{esperanza}, entonces
\[
\varphi\bigl(\E(X)\bigr) \leq \E\bigl(\varphi(X)\bigr) .
\]
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{1.} \omterm{def:b2:series:summable}{Sumabilidad} de $XY$: $\abs{XY} \leq \frac{X^2 + Y^2}2$. La
aplicación $(X, Y) \mapsto \E(XY)$ es una
\omterm{def:b2:quadratic:def}{forma bilineal simétrica} positiva sobre el espacio de las variables
con momento de orden dos, así que se aplica la desigualdad de
Cauchy--Schwarz abstracta del~\cref{ch:b2:quadratic} (para la
desigualdad basta con que sea \emph{semi}definida positiva).
Aplicándola a las variables centradas se obtiene la cota de la
\omterm{def:b2:randomvar:variance}{covarianza}.

\emph{2.} Primero, $m = \E(X)$ está en $I$: $I$ es un intervalo que
contiene todos los valores de $X$, y la \omterm{def:b2:randomvar:expectation}{esperanza} es monótona, de
modo que $m$ está entre $\inf X(\Omega)$ y $\sup X(\Omega)$. Por el
teorema de la recta de apoyo para funciones convexas
(\cref{ch:b2:realfun}), hay $\alpha, \beta$ con $\varphi(t) \geq
\alpha t + \beta$ para todo $t \in I$ y $\varphi(m) = \alpha m +
\beta$. Entonces, punto a punto sobre $\Omega$, $\varphi(X) \geq
\alpha X + \beta$; tomando \omterm{def:b2:randomvar:expectation}{esperanzas},
\[
\E\bigl(\varphi(X)\bigr) \geq \alpha\,\E(X) + \beta
= \varphi\bigl(\E(X)\bigr). \qedhere
\]
\end{proof}

\begin{example}
Jensen con $\varphi(t) = t^2$ da $\E(X)^2 \leq \E(X^2)$: la
positividad de la \omterm{def:b2:randomvar:variance}{varianza}; y con $\varphi(t) = 1/t$ sobre
$\intoo{0}{\infty}$: $\frac{1}{\E X} \leq \E\bigl(\frac1X\bigr)$; la
media armónica está por debajo de la aritmética, ahora en forma
aleatoria.
\end{example}

\begin{remark}[Errores frecuentes]
(i) $\E(XY) = \E(X)\E(Y)$ \emph{exige} \omterm{def:b2:proba:independence}{independencia} (o, al menos,
\omterm{def:b2:randomvar:variance}{covarianza} nula): tomando $Y = X$ resulta $\E(X^2) \neq \E(X)^2$
siempre que $V(X) > 0$. (ii) Igualmente, $V(X + X) = 4V(X)$, y no
$2V(X)$: las \omterm{def:b2:randomvar:variance}{varianzas} solo se suman entre sumandos \omterm{def:b2:proba:independence}{independientes} (o
incorrelados). (iii) $\E(f(X))$ no es $f(\E(X))$; para $f$ convexa,
Jensen dice incluso en qué dirección va el error, como en el~\cref{ex:b2:randomvar:transferex}. (iv) La existencia es una
hipótesis real: para la variable de San Petersburgo $X = 2^K$ con
$\P(K = k) = 2^{-k}$ ($k \geq 1$),
\[
\sum_{k\geq1}2^k\cdot2^{-k} = \sum_{k\geq1}1 = \infty :
\]
$X$ es finita casi seguramente y, sin embargo, no tiene \omterm{def:b2:randomvar:expectation}{esperanza}, y
no existe ningún precio de entrada justo para el juego. La
\omterm{def:b2:series:summable}{sumabilidad} en la definición de $\E$ no es pedantería contable: es
donde se detectan las colas pesadas. (v) Por último, el teorema de
transferencia necesita la \omterm{def:b2:series:summable}{sumabilidad} \emph{absoluta} antes de que
sea legítimo reordenar la suma sobre los valores
(\cref{ch:b2:series}).
\end{remark}


\begin{example}[Chebyshev sobre cien
lanzamientos]\label{ex:b2:randomvar:hundredtosses}
Para $X \sim \mathcal B(100, \frac12)$: $\E X = 50$ y $V(X) = 25$.
Chebyshev con $\varepsilon = 6$:
\[
\P(45 \leq X \leq 55) = \P(\abs{X - 50} < 6)
\geq 1 - \frac{25}{36} \approx 0.31 ,
\]
mientras que la suma binomial exacta da $\approx 0.73$. El $31\,\%$
garantizado queda lejos de la verdad, pero solo necesitó la media y
la \omterm{def:b2:randomvar:variance}{varianza}; el mismo certificado se aplica palabra por palabra a
cualquier variable con $\E = 50$ y $V = 25$, por exótica que sea, y
el~\cref{ex:b2:randomvar:chebsharp} muestra que alguna variable así
lo satura. La universalidad tiene un precio; cuando la
\omterm{def:b2:randomvar:law}{distribución} es genuinamente binomial, las herramientas exponenciales
del problema de fin de semana cierran casi toda la brecha.
\end{example}

\begin{example}[La correlación de una parte con su
todo]\label{ex:b2:randomvar:partwhole}
Para $X, Y$ \omterm{def:b2:proba:independence}{independientes} e idénticamente distribuidas con \omterm{def:b2:randomvar:variance}{varianza}
$\sigma^2 > 0$, ¿cuán correlacionado está un sumando con la suma $S =
X + Y$? Calculemos
\[
\operatorname{Cov}(X, S) = \operatorname{Cov}(X, X) +
\operatorname{Cov}(X, Y) = \sigma^2 + 0 = \sigma^2,
\qquad V(S) = 2\sigma^2,
\]
de modo que el coeficiente de correlación es
\[
\rho(X, S) = \frac{\operatorname{Cov}(X,
S)}{\sigma(X)\,\sigma(S)}
= \frac{\sigma^2}{\sigma\cdot\sigma\sqrt2}
= \frac{1}{\sqrt2} \approx 0.707 ,
\]
sea cual sea la \omterm{def:b2:randomvar:law}{ley} común: dados, monedas, recuentos de Poisson. Con
$n$ sumandos, el mismo cálculo da $\rho(X_1, S_n) = 1/\sqrt n$: la
influencia de cada término individual sobre el total se diluye como
una raíz cuadrada, que es la sombra correlacional de la escala
$\sqrt n$ de las fluctuaciones. Cauchy--Schwarz garantiza $\abs\rho
\leq 1$ siempre; aquí la cota se alcanza exactamente en el caso
degenerado $n = 1$ y decae de manera previsible después.
\end{example}

\begin{example}[La desigualdad de las medias ponderada, desde
Jensen]\label{ex:b2:randomvar:amgm}
Sea $Y$ la variable que toma los valores positivos $a_1, \dots, a_k$
con probabilidades $\lambda_1, \dots, \lambda_k$. La función $-\ln$
es convexa sobre $\intoo0\infty$, de modo que Jensen da $-\ln\E(Y)
\leq \E(-\ln Y)$, es decir,
\[
a_1^{\lambda_1}a_2^{\lambda_2}\cdots a_k^{\lambda_k}
\;\leq\; \lambda_1a_1 + \lambda_2a_2 + \dots + \lambda_ka_k :
\]
la desigualdad aritmético-geométrica ponderada, con igualdad si y
solo si $Y$ es constante. Con pesos iguales $\lambda_i = \frac1k$ se
recupera la desigualdad de las medias clásica. La probabilidad ha
demostrado calladamente un teorema puramente algebraico: elegir una
\omterm{def:b2:randomvar:law}{ley} de probabilidad no es más que un instrumento contable para
combinaciones \omterm{def:b2:affine:convex}{convexas}; el punto de vista baricéntrico del~\cref{ch:b2:affine} una vez más, ahora con Jensen de motor.
\end{example}

\section{La ley débil de los grandes números}

\begin{theorem}[Ley débil de los grandes números]\label{thm:b2:randomvar:lln}
Sean $(X_k)_{k \geq 1}$ \omterm{def:b2:randomvar:law}{variables aleatorias} \omterm{def:b2:proba:independence}{independientes} dos a dos
con la misma \omterm{def:b2:randomvar:law}{ley} y con momento de orden dos; escríbanse $m =
\E(X_1)$ y $S_n = X_1 + \dots + X_n$. Entonces, para todo
$\varepsilon > 0$:
\[
\P\Bigl(\,\Bigl|\frac{S_n}{n} - m\Bigr| \geq \varepsilon\Bigr)
\;\leq\; \frac{V(X_1)}{n\,\varepsilon^2}
\xrightarrow[n \to \infty]{} 0 .
\]
\end{theorem}

\begin{proof}
Por linealidad, $\E(S_n/n) = m$; por el~\cref{thm:b2:randomvar:variancerules} (la \omterm{def:b2:proba:independence}{independencia} dos a dos mata
las \omterm{def:b2:randomvar:variance}{covarianzas}), $V(S_n) = n\,V(X_1)$, de modo que $V(S_n/n) =
V(X_1)/n$. La desigualdad de Chebyshev aplicada a $S_n/n$ da la
cota.
\end{proof}

\begin{remark}
Este es el teorema que conecta la probabilidad con la frecuencia:
para $X_k$ la indicatriz de un \omterm{def:b2:proba:space}{suceso} $A$ en repeticiones
\omterm{def:b2:proba:independence}{independientes}, $S_n/n$ es la frecuencia observada de $A$, y la ley de
los grandes números dice que se concentra en torno a $\P(A)$ a un
ritmo $\frac{p(1-p)}{n\varepsilon^2}$. La \omterm{def:b2:randomvar:law}{ley} \emph{fuerte} ($S_n/n
\to m$ casi seguramente) es un teorema del tercer año; su
demostración con momentos de orden cuatro sí está al alcance, no
obstante: véase el~\cref{exo:b2:randomvar:9}, que ejecuta
Borel--Cantelli sobre la cota de tipo Chebyshev. La misma estimación
de Chebyshev impulsó la demostración con \omterm{thm:b2:funcseq:weierstrass}{polinomios de Bernstein} del
teorema de aproximación de Weierstrass del~\cref{ch:b2:funcseq}: el
lema de conteo de allí \emph{era} la ley débil de los grandes números
disfrazada.
\end{remark}

\begin{example}[Coleccionar cincuenta cromos]
\label{ex:b2:randomvar:couponnumeric}
El coleccionista de cromos del~\cref{exo:b2:randomvar:3} con $n = 50$
juguetes distintos: el total esperado es
\[
\E(T_{50}) = 50\,H_{50} = 50\sum_{k=1}^{50}\frac1k
\approx 50 \times 4.499 \approx 225
\]
cajas; cuatro veces y media la conjetura ingenua $50$. El crecimiento
armónico es toda la historia: los primeros $25$ juguetes llegan en
unas $50\ln2 \approx 35$ cajas, mientras que el \emph{último} juguete
cuesta él solo $50$ cajas de media (una espera geométrica de
parámetro $\frac1{50}$). Los problemas de compleción están dominados
por su final de partida, y por eso el~\cref{exo:b2:randomvar:12}
encuentra fluctuaciones de orden $n$ ---el tamaño de esa última
espera geométrica--- en torno a la media $n\ln n$.
\end{example}

\begin{example}[¿Cuán grande ha de ser $n$?]\label{ex:b2:randomvar:llnnumeric}
Para fijar la frecuencia observada a menos de $\varepsilon = 0.01$ de
$\P(A)$ con una confianza del $95\,\%$, la cota de Chebyshev exige
\[
\frac{p(1-p)}{n\varepsilon^2} \leq \frac{1}{4n\varepsilon^2}
\leq 0.05,
\qquad\text{es decir,}\qquad
n \geq \frac{1}{4\cdot0.05\cdot(0.01)^2} = 50\,000 .
\]
La dependencia es brutal en $\varepsilon$ (cuadrática) y suave en la
confianza (lineal en $1/\alpha$). Ambos rasgos son propiedades de la
\emph{cota}, no de la verdad: las desigualdades exponenciales del
problema de fin de semana rebajan el precio de la confianza de
$1/\alpha$ a $\ln(1/\alpha)$ ---la misma especificación costará allí
unas $18\,500$ muestras---, mientras que la escala $1/\varepsilon^2$
es genuina e inmejorable. Saber qué parte de una cota está floja es
tan útil como la cota misma.
\end{example}

\begin{remark}[Perspectivas dentro de este volumen]
Hacia delante, todo lo de aquí alimenta el~\cref{ch:b2:genfun}: la
\omterm{def:b2:randomvar:expectation}{esperanza} $\E(t^X)$ de una astuta función de $X$ empaqueta toda la
\omterm{def:b2:randomvar:law}{ley} en una serie de potencias, los momentos se convierten en
derivadas en $1$, y las identidades de tipo Wald para sumas
aleatorias sostienen la teoría de los procesos de ramificación; el
teorema del producto para variables \omterm{def:b2:proba:independence}{independientes} se convierte en la
multiplicatividad de las \omterm{ex:b2:powerseries:fibonacci}{funciones generatrices}. Hacia atrás, la
\omterm{def:b2:randomvar:expectation}{esperanza} es un \omterm{def:b2:affine:barycenter}{baricentro} con pesos de probabilidad
(\cref{ch:b2:affine}), la desigualdad de Jensen es la geometría de
las rectas de apoyo de las funciones convexas
(\cref{ch:b2:realfun}), y el método de los momentos exponenciales del
problema de fin de semana de este capítulo es Markov aplicado a
$\eu^{tX}$: una desigualdad, mejorada por un buen cambio de variable,
abarcando tres capítulos.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:1}
Calcula $\E(X)$ y $V(X)$ para $X \sim \mathcal{B}(n, p)$ (mediante
indicatrices), para $X \sim \mathcal{P}(\lambda)$ (prueba que $V(X) =
\lambda$) y para $X \sim \mathcal{G}(p)$ (prueba que $V(X) =
\frac{1-p}{p^2}$; usa $\E(X(X-1))$ y la derivada segunda de la serie
geométrica).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:2}
Sean $X \sim \mathcal{P}(\lambda)$ e $Y \sim \mathcal{P}(\mu)$
\omterm{def:b2:proba:independence}{independientes}. Prueba que $X + Y \sim \mathcal{P}(\lambda + \mu)$
(convolución de los pesos; teorema del binomio), y que la \omterm{def:b2:randomvar:law}{ley}
condicionada de $X$ dado $X + Y = n$ es la binomial
$\mathcal{B}\bigl(n, \frac{\lambda}{\lambda + \mu}\bigr)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:3}
(Coleccionista de cromos, \omterm{def:b2:randomvar:expectation}{esperanza}) Una marca de cereales esconde
uno de $n$ juguetes distintos, uniformemente, en cada caja. Sea $T_n$
el número de cajas necesarias para reunir los $n$ juguetes.
Escribiendo $T_n$ como suma de variables geométricas \omterm{def:b2:proba:independence}{independientes}
(el tiempo hasta ver un juguete \emph{nuevo} cuando aún faltan $k$),
prueba que
\[
\E(T_n) = n\sum_{k=1}^{n}\frac{1}{k} \sim n\ln n
\]
(equivalente por la comparación serie-integral del~\cref{ch:b2:comparison}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:4}
Sea $X \geq 0$ con valores enteros. Demuestra la \emph{fórmula de la
cola}
\[
\E(X) = \sum_{n=1}^{\infty} \P(X \geq n)
\]
(cuando alguno de los dos miembros es finito), escribiendo $X =
\sum_{n\geq1}\mathbf{1}_{X \geq n}$ e intercambiando las sumaciones
(Fubini para familias no negativas). Recupera $\E(X) = \frac1p$ para
la \omterm{def:b2:randomvar:law}{ley} geométrica.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:5}
(El muestreo sin reemplazamiento está más concentrado) Una urna
contiene $N$ bolas, $M$ de ellas blancas. Extráiganse $n \leq N$ sin
reemplazamiento y sea $X$ el número de blancas (\omterm{def:b2:randomvar:law}{ley}
\emph{hipergeométrica}). Usando indicatrices $X = \sum_{i=1}^n Y_i$
con $Y_i$ la $i$-ésima extracción: prueba que cada $Y_i$ es de
Bernoulli de parámetro $p = M/N$ (¡simetría!), concluye que $\E(X) =
np$ exactamente igual que con reemplazamiento, y prueba que
$\operatorname{Cov}(Y_i, Y_j) = -\frac{p(1-p)}{N-1} < 0$ para $i \neq
j$, de donde $V(X) = np(1-p)\frac{N - n}{N - 1} \leq np(1-p)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:6}
Sea $X$ con momento de orden dos. Prueba que $c \mapsto \E\bigl((X -
c)^2\bigr)$ es mínima exactamente en $c = \E(X)$, con mínimo $V(X)$.
Prueba después que $\P(X = \E(X)) = 1$ si y solo si $V(X) = 0$.
\emph{(Para el segundo punto: si $V(X) = 0$, usa Chebyshev con
$\varepsilon = 1/n$ y la \omterm{def:b2:metric:continuity}{continuidad} monótona,
\cref{thm:b2:proba:continuity}.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:7}
(La concentración gana a Markov) Sea $S_n \sim \mathcal{B}(n,
\frac12)$ (número de caras en $n$ lanzamientos equilibrados). Compara
las cotas que dan Markov (para $\P(S_n \geq \frac{3n}{4})$),
Chebyshev y el método exponencial (de Chernoff):
\[
\P\Bigl(S_n \geq \frac{3n}4\Bigr)
\leq \E\bigl(e^{tS_n}\bigr)e^{-3nt/4}
= \Bigl(\frac{1 + e^t}{2}\Bigr)^n e^{-3nt/4}
\quad (t > 0),
\]
y optimiza en $t$ para obtener una cota exponencialmente pequeña.
\emph{(En $t = \ln 3$: cota $\bigl(2\cdot 3^{-3/4}\bigr)^n \approx
(0.877)^n$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:8}
(Weierstrass otra vez, probabilísticamente) Sean $f \colon [0,1] \to
\R$ \omterm{def:b2:metric:continuity}{continua} y $S_n \sim \mathcal{B}(n, x)$. Prueba que el \omterm{thm:b2:funcseq:weierstrass}{polinomio
de Bernstein} $B_nf(x) = \sum_{k=0}^n f\bigl(\frac kn\bigr)\binom nk
x^k(1-x)^{n-k}$ vale $\E\bigl[f\bigl(\frac{S_n}{n}\bigr)\bigr]$, y
vuelve a deducir la estimación $\abs{B_nf(x) - f(x)} \leq
\omega_f(\delta) + \frac{2\norm f_\infty}{4n\delta^2}$ del~\cref{ch:b2:funcseq} en este lenguaje probabilista (pártase según
$\bigl|\frac{S_n}{n} - x\bigr| \geq \delta$ y úsese Chebyshev).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:9}
(\omterm{def:b2:randomvar:law}{Ley} fuerte con momentos de orden cuatro) Sean $(X_k)$
\omterm{def:b2:proba:independence}{independientes}, idénticamente distribuidas y centradas ($\E X_1 = 0$),
con $\E(X_1^4) < \infty$. Desarrollando $\E(S_n^4)$ y contando los
términos que sobreviven (solo los de $\E(X_i^4)$ y $\E(X_i^2X_j^2)$,
con $i \neq j$), prueba que $\E(S_n^4) \leq C n^2$ para una constante
$C$. Deduce que $\sum_n \P\bigl(\abs{S_n/n} \geq \varepsilon\bigr) <
\infty$ para cada $\varepsilon > 0$ (Markov de orden 4) y concluye
con Borel--Cantelli (\cref{thm:b2:proba:borelcantelli}) que $S_n/n
\to 0$ casi seguramente, en la formulación adecuada: el \omterm{def:b2:proba:space}{suceso}
$\bigcap_{j}\bigcup_N\bigcap_{n \geq N}\{\abs{S_n/n} < \frac1j\}$
tiene probabilidad $1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:randomvar:10}
Se lanzan dos dados equilibrados; sea $M$ el mayor de los dos
resultados. Usando la fórmula de la cola del~\cref{exo:b2:randomvar:4} (versión finita), prueba que
\[
\E(M) = \sum_{k=1}^{6}\P(M \geq k)
= 6 - \sum_{j=0}^5\Bigl(\frac j6\Bigr)^2 = \frac{161}{36}
\approx 4.47 .
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:randomvar:11}
Sea $F_n$ el número de puntos fijos de una permutación uniformemente
aleatoria de $\{1, \dots, n\}$ ($n \geq 2$). Escribiendo $F_n =
\sum_i\mathbf 1_{\sigma(i) = i}$, calcula $\E(F_n) = 1$,
$\operatorname{Cov}(\mathbf 1_{\sigma(i)=i}, \mathbf
1_{\sigma(j)=j}) = \frac1{n^2(n-1)}$ para $i \neq j$, y concluye que
$V(F_n) = 1$: de media queda una letra fija, con \omterm{def:b2:randomvar:variance}{varianza}
exactamente $1$, sea cual sea $n$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:randomvar:12}
(Coleccionista de cromos, concentración) En el marco del~\cref{exo:b2:randomvar:3}, prueba que
\[
V(T_n) = \sum_{k=1}^n\frac{1 - k/n}{(k/n)^2}
\leq n^2\sum_{k=1}^n\frac{1}{k^2} \leq \frac{\pi^2}{6}n^2,
\]
usando la \omterm{def:b2:proba:independence}{independencia} de las etapas geométricas y $V(\mathcal G(p))
= \frac{1-p}{p^2}$ (\cref{exo:b2:randomvar:1}; el valor $\pi^2/6$ es
el~\cref{ex:b2:fourier:basel}). Deduce con Chebyshev que
$\dfrac{T_n}{n\ln n} \to 1$ \emph{en probabilidad}: el tiempo total
del coleccionista es $n\ln n$ salvo fluctuaciones de orden $n$.
\end{exercise}

\section{Problema: la caja de herramientas de la concentración, de
Markov a Hoeffding}

\begin{problem}[{Problema de fin de semana --- concentración
exponencial a mano, y a cuánta gente ha de preguntar una
encuesta}]
\label{pb:b2:randomvar:1}
La desigualdad de Markov cuesta un momento y compra un decaimiento
$1/a$; Chebyshev cuesta dos momentos y compra $1/\varepsilon^2$; y el~\cref{ex:b2:randomvar:chebsharp} muestra que eso es todo lo que esos
momentos pueden comprar. Este problema sube el resto de la escalera:
el método exponencial (de Chernoff) con su ritmo \emph{exacto} para
lanzamientos de moneda, la desigualdad de Hoeffding para todas las
variables acotadas, y la recompensa: tamaños de muestra explícitos y
honestos para encuestas, adjudicaciones electorales y contraste de
monedas. \index{desigualdades de concentración}\index{desigualdad de
Hoeffding}\index{cota de Chernoff} En todo el problema, $S_n \sim
\mathcal B(n, p)$ es una suma de $n$ variables de Bernoulli
\omterm{def:b2:proba:independence}{independientes} y $\widehat p_n = S_n/n$ la frecuencia empírica.

\medskip
\noindent\textbf{Parte I --- Calibración sobre la moneda
equilibrada.} Aquí $p = \frac12$ y $a \in \intoo{\frac12}{1}$.
\begin{enumerate}
  \item Markov al nivel $an$: prueba que $\P(S_n \geq an) \leq
        \frac1{2a}$, una cota que ni siquiera tiende a $0$. ¿Dónde
        pierde tanto Markov?
  \item Chebyshev: usando la simetría de la binomial equilibrada
        respecto de $n/2$, prueba que
        \[
        \P(S_n \geq an) = \tfrac12\,
        \P\bigl(\abs{S_n - \tfrac n2} \geq n(a -
        \tfrac12)\bigr)
        \leq \frac{1}{8n(a - 1/2)^2},
        \]
        es decir, $\frac2n$ en $a = \frac34$: por fin, decaimiento
        polinómico.
  \item (Chernoff, nivel general) Calcula $\E(\eu^{tS_n}) =
        \bigl(\frac{1 + \eu^t}2\bigr)^n$ y optimiza $\P(S_n \geq an)
        \leq \E(\eu^{tS_n})\eu^{-tan}$ en $t > 0$: prueba que el $t$
        óptimo es $\ln\frac{a}{1-a}$ y que
        \[
        \P(S_n \geq an) \leq \eu^{-n\,I(a)},
        \qquad
        I(a) = \ln 2 + a\ln a + (1-a)\ln(1-a) > 0 .
        \]
        Comprueba que $a = \frac34$ recupera la cota
        $\bigl(2\cdot3^{-3/4}\bigr)^n$ del~\cref{exo:b2:randomvar:7}.
  \item (El exponente es exacto) Sea $k = an$ un entero. A partir de
        que $\binom nk a^k(1-a)^{n-k}$ es el mayor de los $n + 1$
        términos de una \omterm{def:b2:randomvar:law}{distribución} de probabilidad, demuestra que
        $\binom nk \geq \frac{\eu^{nH(a)}}{n+1}$ con $H(a) = -a\ln a
        - (1-a)\ln(1-a)$, y deduce la cota inferior correspondiente
        \[
        \P(S_n \geq an) \geq \binom{n}{an}2^{-n}
        \geq \frac{\eu^{-n\,I(a)}}{n + 1} .
        \]
  \item Tabula las tres cotas en $n = 100$, $a = \frac34$: Markov
        $\frac23$, Chebyshev $0.02$, Chernoff $\approx
        2.1\cdot10^{-6}$ (el valor verdadero es $\approx
        2.8\cdot10^{-7}$). ¿La moraleja, en una frase?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La desigualdad de Hoeffding.}
\begin{enumerate}[resume]
  \item (Caso de Rademacher) Para $\varepsilon = \pm1$ con
        probabilidad $\frac12$ cada uno, demuestra que
        \[
        \E(\eu^{t\varepsilon}) = \cosh t \leq \eu^{t^2/2}
        \qquad (t \in \R)
        \]
        comparando las dos series término a término ($(2k)! \geq
        2^kk!$).
  \item Deduce, para variables de Rademacher \omterm{def:b2:proba:independence}{independientes}
        $\varepsilon_1, \dots, \varepsilon_n$ y todo $s > 0$:
        \[
        \P\Bigl(\sum_{i=1}^n\varepsilon_i \geq s\Bigr)
        \leq \eu^{-s^2/(2n)} .
        \]
  \item Tradúcelo a monedas equilibradas ($X_i =
        \frac{1+\varepsilon_i}2$): $\P\bigl(\widehat p_n - \tfrac12
        \geq \delta\bigr) \leq \eu^{-2n\delta^2}$, y la versión
        bilateral con un factor $2$.
  \item (Lema de Hoeffding) Sea $X \in \intcc01$ con $\E X = p$, y
        $\psi(t) = \ln\E(\eu^{tX})$. Justifica que $\psi$ es dos
        veces \omterm{def:b2:diffcalc:differential}{diferenciable} con
        \[
        \psi''(t) = \E_t(X^2) - \E_t(X)^2, \qquad
        \E_t(Y) := \frac{\E(Y\eu^{tX})}{\E(\eu^{tX})},
        \]
        una \emph{\omterm{def:b2:randomvar:variance}{varianza}} de una variable reponderada que sigue
        tomando valores en $\intcc01$; acótala por $\frac14$ (el
        argumento de minimalidad del~\cref{exo:b2:randomvar:6}) y
        concluye por Taylor:
        \[
        \E\bigl(\eu^{t(X - p)}\bigr) \leq \eu^{t^2/8} .
        \]
  \item (Desigualdad de Hoeffding) Para $X_i \in \intcc01$
        \omterm{def:b2:proba:independence}{independientes} con media común $p$, deduce que
        \[
        \P\bigl(\abs{\widehat p_n - p} \geq \delta\bigr)
        \leq 2\,\eu^{-2n\delta^2}
        \qquad (\delta > 0).
        \]
  \item Compara el ritmo de Chebyshev $\frac{p(1-p)}{n\delta^2}$ con
        el $2\eu^{-2n\delta^2}$ de Hoeffding: ¿qué hipótesis exige
        cada uno, y a partir de qué $n$ (aproximadamente) gana la
        cota exponencial en $\delta = 0.03$, $p = \frac12$?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- ¿A cuánta gente ha de preguntar una
encuesta?} Una encuesta pregunta a $n$ votantes elegidos
uniformemente y de manera \omterm{def:b2:proba:independence}{independiente}; cada uno responde con
sinceridad; $p$ es el resultado verdadero y $\widehat p_n$ la cifra
de la encuesta.
\begin{enumerate}[resume]
  \item Prueba que la encuesta es precisa hasta $\pm\delta$ con
        confianza $1 - \alpha$ (es decir, $\P(\abs{\widehat p_n - p}
        \geq \delta) \leq \alpha$) en cuanto
        \[
        n \;\geq\; \frac{\ln(2/\alpha)}{2\,\delta^2} .
        \]
  \item Calcula el $n$ necesario para la especificación estándar de
        ``tres puntos, noventa y cinco por ciento'' ($\delta = 0.03$,
        $\alpha = 0.05$): $n \geq 2050$; y para un punto: $n \geq
        18\,445$. Obsérvese ---y explíquese--- el hecho llamativo de
        que la respuesta no involucra el tamaño de la población.
  \item Rehaz la pregunta 13 con Chebyshev ($V(X_1) = p(1-p) \leq
        \frac14$): $n \geq \frac1{4\alpha\delta^2} = 5556$ con tres
        puntos. Nótese que muestrear \emph{sin} reemplazamiento solo
        ayuda (\cref{exo:b2:randomvar:5}: la \omterm{def:b2:randomvar:variance}{varianza} se encoge en
        $\frac{N-n}{N-1}$).
  \item (Adjudicar una elección) El resultado verdadero de un
        candidato es $p = 0.52$. ¿A cuántos votantes hay que
        encuestar para que $\P(\widehat p_n \leq \tfrac12) \leq
        0.01$? Prueba que $n \geq \frac{\ln 100}{2\cdot(0.02)^2}
        \approx 5757$: adjudicar una carrera reñida cuesta mucho más
        que estimar un resultado.
  \item Lo que la matemática \emph{no} cubre: enumera las hipótesis
        de modelización usadas (muestreo uniforme \omterm{def:b2:proba:independence}{independiente},
        respuestas sinceras, $p$ fijo) y explica en un párrafo breve
        por qué los errores reales de las encuestas están dominados
        por el \emph{sesgo} (muestreo no uniforme, falta de
        respuesta), que ningún aumento de $n$ reduce.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Más afiladas y más baratas.}
\begin{enumerate}[resume]
  \item (Mediana de medias: decaimiento exponencial a partir de dos
        momentos) Repártase un presupuesto de $km$ muestras en $k$
        grupos independientes de $m$; sean $\widehat p^{(1)}, \dots,
        \widehat p^{(k)}$ las medias de los grupos y $M$ su mediana.
        Elíjase $m$ de modo que cada grupo cumpla $\P(\abs{\widehat
        p^{(i)} - p} \geq \delta) \leq \frac18$ (Chebyshev: basta $m
        \geq \frac2{\delta^2}$). Prueba que si $\abs{M - p} \geq
        \delta$, entonces al menos $k/2$ grupos yerran, y deduce que
        \[
        \P(\abs{M - p} \geq \delta)
        \leq \binom{k}{\lceil k/2\rceil}\Bigl(\frac18
        \Bigr)^{k/2}
        \leq 2^k\cdot 8^{-k/2} = 2^{-k/2} :
        \]
        concentración exponencial usando nada más allá de las
        \omterm{def:b2:randomvar:variance}{varianzas}.
  \item (Paley--Zygmund) Para $X \geq 0$ con momento de orden dos,
        demuestra que $\P(X > 0) \geq \dfrac{\E(X)^2}{\E(X^2)}$
        \emph{(Cauchy--Schwarz sobre $X\mathbf 1_{X>0}$)}: la
        herramienta en sentido inverso; los momentos también pueden
        forzar a los \omterm{def:b2:proba:space}{sucesos} a ocurrir.
  \item (Pinsker en versión ligera) Prueba que $I(a) \geq 2\bigl(a -
        \tfrac12\bigr)^2$ sobre $\intoo{\frac12}1$ \emph{(la
        diferencia se anula hasta el segundo orden en $\frac12$ y su
        derivada segunda es $\frac1{a(1-a)} - 4 \geq 0$)}: el
        exponente exacto de Chernoff siempre gana al cuadrático de
        Hoeffding.
  \item Desarrolla $I\bigl(\tfrac12 + \delta\bigr) = 2\delta^2 +
        O(\delta^4)$ y combínalo con la pregunta 4: para
        desviaciones pequeñas, el exponente $2n\delta^2$ de Hoeffding
        es asintóticamente \emph{exacto}; ningún método puede
        mejorarlo en más que factores polinómicos.
  \item Redacta la tabla de la caja de herramientas: para Markov,
        Chebyshev, la cota de cuarto momento del~\cref{exo:b2:randomvar:9}, Hoeffding y Chernoff con
        exponente $I$, indica en una línea cada uno: hipótesis
        requerida, decaimiento obtenido y la pregunta de este
        problema donde resultó más afilado.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dividendos.}
\begin{enumerate}[resume]
  \item (Contrastar una moneda) Una moneda es equilibrada o está
        sesgada con $p = 0.55$. Se lanza $n$ veces y se declara
        ``sesgada'' cuando $\widehat p_n > 0.525$. Prueba que ambas
        probabilidades de error son a lo sumo
        $\eu^{-2n(0.025)^2}$, y que $n \geq 3685$ lanzamientos
        garantizan que ambas queden por debajo del $1\,\%$.
  \item (Los \omterm{def:b2:proba:space}{sucesos} raros necesitan una cota consciente de la
        \omterm{def:b2:randomvar:variance}{varianza}) Sea $p = 0.01$ y tómese la especificación relativa
        $\delta = p/2 = 0.005$, $\alpha = 0.05$. Compara los tamaños
        de muestra que exigen Hoeffding ($n \approx 74\,000$) y
        Chebyshev con la \omterm{def:b2:randomvar:variance}{varianza} verdadera $p(1-p)$ ($n \approx
        7920$): la cota exponencial, ciega a la \omterm{def:b2:randomvar:variance}{varianza}, pierde
        frente al humilde momento de orden dos. Enuncia la moraleja y
        di de dónde vendrá la herramienta que falta (una cota
        exponencial consciente de la \omterm{def:b2:randomvar:variance}{varianza}; la aproximación de
        Poisson del~\cref{ch:b2:genfun}).
  \item (\omterm{def:b2:randomvar:law}{Ley} fuerte para monedas) A partir de $\sum_n
        2\eu^{-2n\delta^2} < \infty$ y de Borel--Cantelli
        (\cref{thm:b2:proba:borelcantelli}), demuestra que $\widehat
        p_n \to p$ casi seguramente para lanzamientos de moneda
        \omterm{def:b2:proba:independence}{independientes}: formula el \omterm{def:b2:proba:space}{suceso} casi seguro como
        $\bigcap_j\bigcup_N\bigcap_{n\geq N} \{\abs{\widehat p_n - p}
        < \tfrac1j\}$, como en el~\cref{exo:b2:randomvar:9}, y
        concluye. (La acotación sustituye al cuarto momento usado
        allí.)
  \item Síntesis. En cinco frases: qué cuesta y qué compra cada
        peldaño de la escalera (momentos uno, dos y cuatro;
        exponencial acotada; exponente exacto); por qué encuestar a
        $2050$ personas basta para un país de cualquier tamaño; y
        cuál de estas cotas afilará el volumen del tercer año hasta
        las constantes exactas del teorema central del límite.
\end{enumerate}
\end{problem}
