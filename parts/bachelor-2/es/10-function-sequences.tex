\chapter{Sucesiones y series de funciones}\label{ch:b2:funcseq}

Cuando unas funciones convergen a una función, ¿qué propiedades
sobreviven al paso al límite? La \omterm{def:b2:funcseq:def}{convergencia puntual} no conserva casi
nada; la convergencia \emph{uniforme} ---la convergencia en la
\omterm{def:b2:nvs:norm}{norma} del supremo--- conserva la \omterm{def:b2:metric:continuity}{continuidad}, las integrales sobre
segmentos y, con un matiz, las derivadas. Este capítulo demuestra los
tres teoremas de transferencia y sus versiones para series, y los
corona con el teorema de aproximación de Weierstrass, demostrado
mediante los preciosos polinomios probabilísticos de Bernstein.

\section{Convergencia puntual y uniforme}

\begin{definition}\label{def:b2:funcseq:def}
Sean $f_n, f \colon X \to \R$ (o $\C$, o un espacio normado), con $X$
un conjunto cualquiera. $(f_n)$ converge a $f$
\emph{puntualmente}\index{convergencia puntual} cuando $f_n(x) \to
f(x)$ para todo $x$, y \emph{uniformemente}\index{convergencia
uniforme} cuando
\[
\norm{f_n - f}_\infty = \sup_{x \in X}\, \abs{f_n(x) - f(x)}
\xrightarrow[n \to \infty]{} 0 .
\]
La convergencia uniforme implica la puntual; y sobre
$C(\intcc{a}{b})$, la convergencia uniforme es exactamente la
convergencia en el \omterm{def:b2:nvs:banach}{espacio de Banach} $\bigl(C(\intcc{a}{b}),
\norm\cdot_\infty\bigr)$ del~\cref{ch:b2:nvs}.
\end{definition}

\begin{example}\label{ex:b2:funcseq:xn}
Sobre $\intcc{0}{1}$, $f_n(x) = x^n$ converge \omterm{def:b2:funcseq:def}{puntualmente} al límite
\emph{discontinuo} $f = \mathbf{1}_{\{1\}}$; la convergencia no es
uniforme: $\norm{f_n - f}_\infty \geq f_n\bigl(1 - \tfrac1n\bigr) =
(1 - \tfrac1n)^n \to \eu^{-1} \neq 0$. Sobre $\intcc{0}{a}$ con $a <
1$ \emph{sí} lo es ($\sup = a^n \to 0$): la uniformidad es una
propiedad del dominio tanto como de la sucesión.
\end{example}

\begin{example}[Dos límites que se niegan a conmutar]\label{ex:b2:funcseq:doublelimit}
Todo el capítulo trata de intercambiar límites, así que he aquí el
fallo más pequeño posible. Sea $a_{n,m} = \dfrac{n}{n+m}$ para $n, m
\geq 1$. Entonces
\[
\lim_{m\to\infty}\Bigl(\lim_{n\to\infty}a_{n,m}\Bigr)
= \lim_{m\to\infty} 1 = 1,
\qquad
\lim_{n\to\infty}\Bigl(\lim_{m\to\infty}a_{n,m}\Bigr)
= \lim_{n\to\infty} 0 = 0 :
\]
ambos límites iterados existen y son distintos. Todo teorema de
transferencia de este capítulo es una licencia para conmutar dos
límites ---$\lim_n$ con $\lim_{x\to a}$ (\omterm{def:b2:metric:continuity}{continuidad}), con $\int$
(integración), con $\frac{\dd}{\dd x}$ (derivación)---, y la
\omterm{def:b2:funcseq:def}{convergencia uniforme} es precisamente el peaje que legaliza la
conmutación. Moraleja: siempre que una ``demostración'' intercambie
en silencio dos operaciones de paso al límite, este cuadro de dos
líneas es el contraejemplo que conviene oponerle; las jorobas
deslizantes del~\cref{exo:b2:funcseq:2} son el mismo fenómeno vestido
con un signo integral.
\end{example}

\section{Los tres teoremas de transferencia}

\begin{theorem}[Continuidad]\label{thm:b2:funcseq:continuity}
Si cada $f_n$ es \omterm{def:b2:metric:continuity}{continua} en $a$ y $f_n \to f$ uniformemente en un
entorno de $a$, entonces $f$ es \omterm{def:b2:metric:continuity}{continua} en $a$. Un límite uniforme
de funciones \omterm{def:b2:metric:continuity}{continuas} es \omterm{def:b2:metric:continuity}{continuo}.
\end{theorem}

\begin{proof}
El argumento de los $3\varepsilon$ ya empleado en el~\cref{thm:b2:metric:rncomplete}: tómese $n$ con $\norm{f_n -
f}_\infty \leq \varepsilon$ y después $\delta$ a partir de la
\omterm{def:b2:metric:continuity}{continuidad} de $f_n$ en $a$; para $\abs{x - a} \leq \delta$,
\[
\abs{f(x) - f(a)} \leq \abs{f(x) - f_n(x)} + \abs{f_n(x) - f_n(a)}
+ \abs{f_n(a) - f(a)} \leq 3\varepsilon . \qedhere
\]
\end{proof}

\begin{example}[La uniformidad falla justo donde se rompe el límite]\label{ex:b2:funcseq:threshold}
Sobre $\intcc{0}{2}$, sea $f_n(x) = \dfrac{x^n}{1 + x^n}$. El límite
\omterm{def:b2:funcseq:def}{puntual} es una función de tres piezas:
\[
f(x) = \begin{cases} 0 & 0 \leq x < 1,\\[2pt]
\tfrac12 & x = 1,\\[2pt]
1 & 1 < x \leq 2, \end{cases}
\]
discontinua en $1$, de modo que por el~\cref{thm:b2:funcseq:continuity} la convergencia no puede ser
uniforme sobre $\intcc{0}{2}$. Sobre las piezas cerradas que evitan
el umbral sí lo es: para $0 \leq x \leq a < 1$,
\[
\sup_{\intcc{0}{a}}\abs{f_n - 0}
= \frac{a^n}{1 + a^n} \leq a^n \to 0 ,
\]
y para $1 < b \leq x \leq 2$,
\[
\sup_{\intcc{b}{2}}\abs{f_n - 1}
= \frac{1}{1 + b^n} \leq b^{-n} \to 0 ,
\]
ambos supremos calculados por la monotonía de $u \mapsto
\frac{u}{1+u}$ y de $x \mapsto x^n$. Moraleja: el fallo de la
uniformidad está localizado en la discontinuidad del límite ---la
misma geometría que en el~\cref{ex:b2:funcseq:xn}--- y es la razón de
que la disciplina de ``uniforme sobre cada segmento interior'' se
repita a lo largo de todo el capítulo.
\end{example}

\begin{theorem}[Integración sobre un segmento]\label{thm:b2:funcseq:integration}
Si $f_n \to f$ \omterm{def:b2:funcseq:def}{uniformemente} sobre $\intcc{a}{b}$, con $f_n$ \omterm{def:b2:metric:continuity}{continuas}
a trozos (y $f$ también), entonces
\[
\int_a^b f_n \longrightarrow \int_a^b f .
\]
\end{theorem}

\begin{proof}
La linealidad y la desigualdad triangular para integrales dan
\[
\Bigl|\int_a^b f_n - \int_a^b f\Bigr|
= \Bigl|\int_a^b (f_n - f)\Bigr|
\leq \int_a^b\abs{f_n - f}
\leq (b - a)\,\norm{f_n - f}_\infty \longrightarrow 0 .
\]
El factor de longitud $(b - a)$ es donde entra la \omterm{def:b2:metric:compact}{compacidad} del
segmento: sobre intervalos no \omterm{def:b2:metric:compact}{compactos} esa misma estimación
produce la cota inútil $\infty\cdot0$, y la conclusión falla de
verdad sin dominación ---las jorobas planas $f_n =
\frac1n\mathbf 1_{\intcc{0}{n}}$ convergen \omterm{def:b2:funcseq:def}{uniformemente} a $0$ sobre
$\intco{0}{\infty}$ y sin embargo mantienen $\int f_n = 1$
(observación sobre errores frecuentes de más abajo), y las jorobas
deslizantes de la observación del~\cref{ch:b2:integration} hacen lo
mismo con la convergencia puntual---: la \omterm{def:b2:funcseq:def}{convergencia uniforme}
controla alturas, nunca anchuras.
\end{proof}

\begin{theorem}[Derivación]\label{thm:b2:funcseq:differentiation}
Sean $f_n$ de clase $C^1$ sobre un intervalo $I$, con $(f_n')$
convergente \emph{\omterm{def:b2:funcseq:def}{uniformemente}} sobre $I$ (o sobre cada segmento de
$I$) a cierta $g$, y $(f_n(x_0))$ convergente en un punto $x_0$.
Entonces $(f_n)$ converge (\omterm{def:b2:funcseq:def}{uniformemente} sobre segmentos) a una
función $f$ de clase $C^1$, y $f' = g$: puede derivarse el límite.
\end{theorem}

\begin{proof}
Definamos $f(x) = \lim f_n(x_0) + \int_{x_0}^x g$: es legítimo,
puesto que $g$ es \omterm{def:b2:metric:continuity}{continua} ---en efecto, $g$ es el límite
\emph{uniforme} sobre segmentos de las funciones \omterm{def:b2:metric:continuity}{continuas} $f_n'$,
así que se aplica el~\cref{thm:b2:funcseq:continuity}---, y la
integral de una función \omterm{def:b2:metric:continuity}{continua} está bien definida y cumple, por el
teorema fundamental del cálculo,
\[
f'(x) = g(x) \qquad (x \in I) :
\]
el límite candidato es de clase $C^1$ con la derivada correcta
\emph{por construcción}, antes de demostrar convergencia alguna. Por
el teorema fundamental de nuevo, $f_n(x) = f_n(x_0) + \int_{x_0}^x
f_n'$; restando,
\[
\abs{f_n(x) - f(x)} \leq \abs{f_n(x_0) - \lim f_n(x_0)}
+ \abs{x - x_0}\,\norm{f_n' - g}_{\infty} ,
\]
que tiende a $0$ \omterm{def:b2:funcseq:def}{uniformemente} sobre cada segmento. Y $f$ es de
clase $C^1$ con $f' = g$ por construcción.
\end{proof}

\begin{example}[Por qué la hipótesis recae sobre las derivadas]\label{ex:b2:funcseq:sqrtabs}
Sea $F_n(x) = \sqrt{x^2 + \frac1n}$ sobre $\R$. Cada $F_n$ es de
clase $C^1$ (de hecho $C^\infty$), y la convergencia a $\abs x$ es
uniforme en todo $\R$:
\[
0 \leq F_n(x) - \abs x
= \frac{(x^2 + \frac1n) - x^2}{\sqrt{x^2+\frac1n} + \abs x}
= \frac{1/n}{\sqrt{x^2 + \frac1n} + \abs x}
\leq \frac{1/n}{1/\sqrt n} = \frac{1}{\sqrt n} .
\]
Y sin embargo el límite $\abs x$ no es derivable en $0$: la
\omterm{def:b2:funcseq:def}{convergencia uniforme} de las \emph{funciones}, por rápida que sea, no
transfiere ninguna derivabilidad. El fallo se ve en las derivadas:
\[
F_n'(x) = \frac{x}{\sqrt{x^2 + \frac1n}}
\longrightarrow \begin{cases} 1 & x > 0,\\ 0 & x = 0,\\
-1 & x < 0, \end{cases}
\]
un límite \omterm{def:b2:funcseq:def}{puntual} discontinuo, de modo que $(F_n')$ no puede
converger \omterm{def:b2:funcseq:def}{uniformemente} cerca de $0$
(\cref{thm:b2:funcseq:continuity} otra vez). Moraleja: el~\cref{thm:b2:funcseq:differentiation} supone deliberadamente la
\omterm{def:b2:funcseq:def}{convergencia uniforme} de las $f_n'$ y no la de las $f_n$; este
ejemplo es la razón.
\end{example}

\section{Series de funciones}

\begin{definition}\label{def:b2:funcseq:series}
Una serie de funciones $\sum u_n$ converge
\omterm{def:b2:funcseq:def}{puntual}/\omterm{def:b2:funcseq:def}{uniformemente} cuando lo hacen sus sumas parciales. Converge
\emph{normalmente}\index{convergencia normal} (sobre $X$) cuando
$\sum \norm{u_n}_\infty < \infty$. La convergencia normal implica la
uniforme (en el \omterm{def:b2:nvs:banach}{espacio de Banach} de las funciones acotadas:
\cref{thm:b2:nvs:absoluteconvergence}), que implica la \omterm{def:b2:funcseq:def}{puntual};
ambas implicaciones son estrictas.
\end{definition}

\begin{example}[Una serie, tres veredictos]\label{ex:b2:funcseq:threemodes}
Tomemos $u_n(x) = \frac{x^n}{n}$ sobre $\intco{0}{1}$.
\emph{\omterm{def:b2:funcseq:def}{Puntualmente}:} converge para todo $x \in \intco01$
(comparación con la serie geométrica). \emph{\omterm{def:b2:funcseq:series}{Normalmente} sobre
$\intcc{0}{a}$, $a < 1$:} $\norm{u_n}_{\infty,\intcc0a} =
\frac{a^n}{n}$, \omterm{def:b2:series:summable}{sumable}. \emph{No \omterm{def:b2:funcseq:series}{normalmente} sobre
$\intco{0}{1}$:} $\norm{u_n}_{\infty,\intco01} = \frac1n$, y
$\sum\frac1n$ diverge. \emph{Ni siquiera \omterm{def:b2:funcseq:def}{uniformemente} sobre
$\intco{0}{1}$:} el resto se resiste cerca de $1$,
\[
R_N(x) = \sum_{n>N}\frac{x^n}{n}
\geq \sum_{n=N+1}^{2N}\frac{x^n}{n}
\geq \frac{N\,x^{2N}}{2N} = \frac{x^{2N}}{2}
\xrightarrow[x\to1^-]{} \frac12 ,
\]
de modo que $\sup_{\intco01}\abs{R_N} \geq \frac12$ para todo $N$.
Moraleja: los cuatro veredictos conviven en paz; la suma
$-\ln(1-x)$ es \omterm{def:b2:metric:continuity}{continua} sobre $\intco{0}{1}$ porque la \omterm{def:b2:metric:continuity}{continuidad}
solo necesita uniformidad \emph{cerca de cada punto}, es decir,
sobre los segmentos $\intcc0a$; explotar en el borde es un derecho de
la suma.
\end{example}

\begin{theorem}[Transferencia para series]\label{thm:b2:funcseq:seriestransfer}
Si $\sum u_n$ converge \omterm{def:b2:funcseq:def}{uniformemente} (por ejemplo, \omterm{def:b2:funcseq:series}{normalmente})
sobre el conjunto pertinente: la \omterm{def:b2:metric:continuity}{continuidad} de todas las $u_n$ en $a$
pasa a la suma; la integración sobre un segmento puede hacerse
término a término; y si $\sum u_n(x_0)$ converge y $\sum u_n'$
converge \omterm{def:b2:funcseq:def}{uniformemente} sobre segmentos, la suma es de clase $C^1$ con
derivada $\sum u_n'$.
\end{theorem}

\begin{proof}
Todo es el teorema correspondiente aplicado a las sumas parciales
$S_N = \sum_{n\leq N}u_n$, que son sumas finitas de funciones con la
regularidad pertinente. \omterm{def:b2:metric:continuity}{Continuidad}: cada $S_N$ es \omterm{def:b2:metric:continuity}{continua} en $a$ y
$S_N \to \sum u_n$ \omterm{def:b2:funcseq:def}{uniformemente}:
\cref{thm:b2:funcseq:continuity}. Integración: sobre el segmento,
\[
\int_a^b \sum_{n\geq0} u_n
= \lim_N \int_a^b S_N
= \lim_N \sum_{n=0}^{N}\int_a^b u_n
= \sum_{n\geq0}\int_a^b u_n
\]
por el~\cref{thm:b2:funcseq:integration} (primera igualdad) y la
linealidad de la integral (segunda). Derivación: las $S_N$ son de
clase $C^1$, $S_N(x_0)$ converge y $S_N' = \sum_{n\leq N}u_n'$
converge \omterm{def:b2:funcseq:def}{uniformemente} sobre segmentos: el~\cref{thm:b2:funcseq:differentiation} da que la suma es $C^1$ con
derivada $\lim S_N' = \sum u_n'$.
\end{proof}

\begin{remark}[Errores frecuentes]
Cuatro trampas, todas vistas en exámenes. \emph{(i) Supremos
comprobados a medias:} evaluar $f_n$ a lo largo de una sucesión
$x_n$ bien elegida solo acota $\norm{f_n - f}_\infty$ por
\emph{debajo}, lo bastante para refutar la uniformidad (como en el~\cref{ex:b2:funcseq:xn}) pero nunca para demostrarla; para
demostrarla hay que acotar el supremo mediante un cálculo válido
para \emph{todo} $x$. \emph{(ii) Uniformidad sobre el conjunto
equivocado:} la \omterm{def:b2:funcseq:series}{convergencia normal} o uniforme suele valer sobre
cada $\intcc{-a}{a}$ o $\intco\delta\infty$ y fallar sobre la unión
\omterm{def:b2:metric:topology}{abierta}; eso no es un obstáculo, porque la \omterm{def:b2:metric:continuity}{continuidad} y la
derivabilidad son locales, de modo que la disciplina segmento a
segmento del~\cref{ex:b2:funcseq:zeta} las da sobre todo el \omterm{def:b2:metric:topology}{abierto}.
\emph{(iii) Integrar sobre algo que no es un segmento:} el~\cref{thm:b2:funcseq:integration} es un enunciado sobre segmentos;
sobre $\intco0\infty$, la \omterm{def:b2:funcseq:def}{convergencia uniforme} no impide que la masa
escape al infinito ($f_n = \frac1n\mathbf 1_{\intcc{0}{n}}$ converge
\omterm{def:b2:funcseq:def}{uniformemente} a $0$ con $\int f_n = 1$); allí hay que usar la
convergencia dominada. \emph{(iv) Derivar el límite:}
\cref{ex:b2:funcseq:sqrtabs}; la hipótesis de derivación recae sobre
$(f_n')$, y ninguna velocidad de convergencia de $(f_n)$ puede
sustituirla.
\end{remark}

\begin{example}[La función $\zeta$ de Riemann]\label{ex:b2:funcseq:zeta}
$\zeta(s) = \sum_{n\geq1} n^{-s}$ converge \omterm{def:b2:funcseq:series}{normalmente} sobre toda
semirrecta $\intco{a}{+\infty}$ con $a > 1$ ($\norm{n^{-s}}_\infty =
n^{-a}$, \omterm{def:b2:series:summable}{sumable}): $\zeta$ es \omterm{def:b2:metric:continuity}{continua} sobre $\intoo{1}{+\infty}$;
y derivando término a término (la serie derivada $\sum -\ln n\;
n^{-s}$ también converge \omterm{def:b2:funcseq:series}{normalmente} sobre $\intco{a}{\infty}$),
$\zeta$ es de clase $C^1$ ---y, iterando, $C^\infty$--- con
$\zeta'(s) = -\sum \frac{\ln n}{n^s}$. Obsérvese la disciplina: la
\omterm{def:b2:funcseq:series}{convergencia normal} se comprueba sobre \emph{sub}semirrectas, nunca
sobre el \omterm{def:b2:metric:topology}{abierto} $\intoo{1}{\infty}$ mismo, donde falla.
\end{example}

\begin{example}[Una serie logarítmica, hasta el final]\label{ex:b2:funcseq:logseries}
Sea $F(x) = \sum_{n\geq1} \frac{\eu^{-nx}}{n}$ sobre
$\intoo{0}{\infty}$. Cada término está acotado sobre
$\intco{\delta}\infty$ por $\frac{\eu^{-n\delta}}{n} \leq
\eu^{-n\delta}$, término de una serie geométrica convergente:
\omterm{def:b2:funcseq:series}{convergencia normal} sobre cada $\intco\delta\infty$, luego $F$ es
\omterm{def:b2:metric:continuity}{continua} sobre $\intoo{0}{\infty}$. La serie derivada $\sum
-\eu^{-nx}$ es igualmente \omterm{def:b2:funcseq:series}{normalmente} convergente sobre
$\intco\delta\infty$ ($\norm{\eu^{-nx}}_{\infty,\intco\delta\infty} =
\eu^{-n\delta}$), de modo que $F$ es $C^1$ con derivada
geométrica:
\[
F'(x) = -\sum_{n\geq1}\eu^{-nx}
= \frac{-\eu^{-x}}{1 - \eu^{-x}}
= \frac{-1}{\eu^{x} - 1} .
\]
Iterando, $F$ es de clase $C^\infty$. Integrando $F'$ (tanto $F$ como
$x \mapsto -\ln(1 - \eu^{-x})$ se anulan en $+\infty$ y tienen la
misma derivada sobre $\intoo0\infty$):
\[
F(x) = -\ln\bigl(1 - \eu^{-x}\bigr),
\]
la serie logarítmica en $t = \eu^{-x}$. Moraleja: cuando $x \to
0^+$, $F(x) = -\ln(x + O(x^2)) = \ln\frac1x + O(x)$; la serie diverge
logarítmicamente en la frontera, exactamente igual que la serie
armónica en la que se convierte en $x = 0$, y la \omterm{def:b2:funcseq:series}{convergencia normal}
sobre $\intco\delta\infty$ pero no sobre $\intoo0\infty$ es el
síntoma.
\end{example}

\begin{method}[Cómo demostrar o refutar la convergencia uniforme]\label{met:b2:funcseq:uniform}
Para $f_n \to f$ \omterm{def:b2:funcseq:def}{puntualmente} sobre $X$:
\begin{enumerate}
  \item Calcúlese o acótese $\norm{f_n - f}_\infty$: estúdiese la
        función $x \mapsto \abs{f_n(x) - f(x)}$ (derivada,
        monotonía) para localizar su máximo; una cota válida para
        todo $x$ que tienda a $0$ demuestra la uniformidad.
  \item Para \emph{refutarla}: exhíbanse puntos $x_n$ con
        $\abs{f_n(x_n) - f(x_n)} \not\to 0$ (a menudo $x_n$ sigue la
        joroba móvil, como en el~\cref{exo:b2:funcseq:1}); o
        invóquese un teorema de transferencia en contrarrecíproco:
        un límite discontinuo de funciones \omterm{def:b2:metric:continuity}{continuas}
        (\cref{ex:b2:funcseq:threshold}), o $\int f_n \not\to \int
        f$ sobre un segmento.
  \item Para series, pruébese primero la \omterm{def:b2:funcseq:series}{convergencia normal}
        ($\sum\sup\abs{u_n} < \infty$); si falla globalmente,
        compruébese sobre los subsegmentos que importan
        (\cref{ex:b2:funcseq:threemodes}); y si falla en todas
        partes, todavía puede haber \omterm{def:b2:funcseq:def}{convergencia uniforme} vía la cota
        del resto alternado (\cref{exo:b2:funcseq:4}) o la sumación
        por partes.
\end{enumerate}
\end{method}

\section{El teorema de aproximación de Weierstrass}

\begin{theorem}[Weierstrass, vía Bernstein]\label{thm:b2:funcseq:weierstrass}
Toda función \omterm{def:b2:metric:continuity}{continua} $f \colon \intcc{0}{1} \to \R$ es límite
uniforme de polinomios; explícitamente, de sus \emph{polinomios de
Bernstein}\index{polinomios de Bernstein}\index{teorema de
aproximación de Weierstrass}
\[
B_n(f)(x) = \sum_{k=0}^{n} f\Bigl(\frac kn\Bigr)\binom nk x^k
(1-x)^{n-k} .
\]
\end{theorem}

\begin{proof}
Fijemos $x \in \intcc{0}{1}$ y pongamos $p_k(x) = \binom nk x^k(1 -
x)^{n-k}$. Tres identidades binomiales, obtenidas evaluando $(x +
y)^n$ y sus dos derivadas en $x$ en $y = 1 - x$:
\[
\sum_k p_k = 1,
\qquad
\sum_k k\,p_k = nx,
\qquad
\sum_k k(k-1) p_k = n(n-1)x^2 .
\]
En detalle: $(x+y)^n = \sum_k\binom nk x^ky^{n-k}$ en $y = 1-x$ es la
primera; derivando en $x$,
\[
n(x+y)^{n-1} = \sum_k k\binom nk x^{k-1}y^{n-k} ,
\]
y multiplicando después por $x$ y poniendo $y = 1 - x$ se obtiene la
segunda; derivando dos veces y multiplicando por $x^2$ se obtiene la
tercera. Desarrollando $(k - nx)^2 = k(k-1) + k(1 - 2nx) + n^2x^2$ y
combinando las tres:
\[
\sum_k (k - nx)^2 p_k
= n(n-1)x^2 + nx(1 - 2nx) + n^2x^2
= nx(1 - x) \leq \frac n4 ,
\]
la \emph{identidad de la varianza}.

Estimemos ahora, usando $\sum p_k = 1$:
\[
\abs{B_n(f)(x) - f(x)}
\leq \sum_{k} \Bigl| f\Bigl(\frac kn\Bigr) - f(x)\Bigr|\, p_k(x)
= \Sigma_{\text{cerca}} + \Sigma_{\text{lejos}} ,
\]
separando según que $\abs{\frac kn - x} \leq \delta$ o no. Dado
$\varepsilon > 0$, la \omterm{def:b2:metric:continuity}{continuidad} uniforme de $f$ (Heine) proporciona
un $\delta$ con $\Sigma_{\text{cerca}} \leq \varepsilon$. Para la
suma lejana, con $M = \norm f_\infty$: por la identidad de la
varianza y el truco de recuento de Chebyshev,
\[
\Sigma_{\text{lejos}} \leq 2M \sum_{\abs{k - nx} > n\delta} p_k
\leq 2M\,\frac{\sum_k (k - nx)^2 p_k}{n^2\delta^2}
\leq \frac{2M}{4 n \delta^2} \cdot 1
= \frac{M}{2n\delta^2}
\xrightarrow[n\to\infty]{} 0 ,
\]
uniformemente en $x$. Así pues, $\norm{B_n(f) - f}_\infty \leq
\varepsilon + \frac{M}{2n\delta^2} \leq 2\varepsilon$ para $n$
grande.
\end{proof}

\begin{remark}
Por sustitución afín, el teorema vale sobre cualquier segmento
$\intcc{a}{b}$. Falla sobre $\R$ (un límite uniforme de polinomios
sobre $\R$ es un polinomio: \cref{exo:b2:funcseq:8}). La lectura
probabilística ---$B_n(f)(x)$ es la esperanza de $f$ en una media
binomial, y la cota de la varianza es la desigualdad de Chebyshev---
se formaliza en el~\cref{ch:b2:genfun}.
\end{remark}

\begin{remark}[Dónde se usa]
La aproximación de Weierstrass es el teorema de densidad del
análisis clásico: hace separable $C(\intcc ab)$, permite comprobar
identidades integrales solo sobre polinomios (problemas de momentos)
y sostiene la versión trigonométrica demostrada en el capítulo de
Fourier mediante el núcleo de Fejér. El problema de fin de semana de
este capítulo extrae el contenido cuantitativo de la demostración de
Bernstein ---velocidades de convergencia gobernadas por el módulo de
continuidad--- y aísla después lo que de verdad la hacía funcionar,
en el teorema de Korovkin: positividad más tres funciones de prueba.
El volumen del tercer año generaliza el enunciado de densidad a
subálgebras arbitrarias (Stone--Weierstrass) y a espacios
\omterm{def:b2:metric:compact}{compactos}.
\end{remark}

\begin{example}[Aproximación poligonal, con velocidad]\label{ex:b2:funcseq:polygonal}
Para $f$ $L$-lipschitziana sobre $\intcc{0}{1}$, sea $I_nf$ el
interpolante afín a trozos en los nodos $\frac kn$. Sobre una celda
$\intcc{\frac kn}{\frac{k+1}n}$, tanto $f(x)$ como $I_nf(x)$ quedan
entre los valores extremos que puede tomar una función
$L$-lipschitziana dados los dos valores nodales, de modo que para
$x$ en la celda, escribiendo $x_k = \frac kn$:
\[
\abs{I_nf(x) - f(x)}
\leq \abs{I_nf(x) - f(x_k)} + \abs{f(x_k) - f(x)}
\leq L\,\abs{x - x_k} + L\,\abs{x - x_k}
\leq \frac{2L}{n}
\]
(el interpolante es él mismo $L$-lipschitziano sobre la celda: su
pendiente es un cociente incremental de $f$). Por tanto,
$\norm{I_nf - f}_\infty \leq \frac{2L}{n}$: la aproximación poligonal
de las funciones \omterm{def:b2:metric:continuity}{lipschitzianas} converge a velocidad $\frac1n$,
\emph{más deprisa} que el $\frac{1}{\sqrt n}$ de Bernstein para la
misma clase (problema de fin de semana, parte II). Moraleja: el
polígono interpola pero no es suave, y Bernstein es suave pero
lento; no hay nada gratis entre la regularidad del aproximante y la
velocidad, compromiso que los resultados de saturación del problema
de fin de semana precisan.
\end{example}

\begin{remark}[Perspectivas dentro de este volumen]
La \omterm{def:b2:funcseq:def}{convergencia uniforme} es el caballo de batalla de este libro a
partir de aquí. El capítulo de series de potencias funciona
enteramente con la \omterm{def:b2:funcseq:series}{convergencia normal} sobre subdiscos \omterm{def:b2:metric:compact}{compactos}:
todo teorema término a término de allí es un caso particular de los
teoremas de transferencia de este capítulo. El capítulo de Fourier
vive un piso más arriba: sus sumas parciales $S_N$ fallan justo donde
este capítulo advierte que podrían hacerlo (\omterm{def:b2:funcseq:def}{puntualmente} pero no
uniformemente en los saltos), y sus medias de Fejér triunfan por la
misma mecánica de los $3\varepsilon$ que demostró el~\cref{thm:b2:funcseq:continuity}. El capítulo de ecuaciones
diferenciales define $\eu^{tA}$ mediante una serie \omterm{def:b2:funcseq:series}{normalmente}
convergente y la deriva término a término: literalmente el~\cref{thm:b2:funcseq:seriestransfer} aplicado a las entradas de las
matrices. Cuando más adelante surja la duda de ``¿por qué podemos
hacer esto?'', la respuesta suele ser un teorema de este capítulo.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b2:funcseq:1}
Estudia la \omterm{def:b2:funcseq:def}{convergencia puntual} y la uniforme sobre $\intcc{0}{1}$
y, según convenga, sobre $\intcc{0}{a}$ ($a < 1$) o

f_n(x) = \frac{x}{1 + nx},
\qquad
g_n(x) = n x\,\eu^{-n x^2},
\qquad
h_n(x) = x^n(1 - x^n).
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:funcseq:2}
Demuestra que $\displaystyle\int_0^1 g_n \not\to \int_0^1 \lim g_n$
para las $g_n$ del~\cref{exo:b2:funcseq:1}, y reconcílialo con el~\cref{thm:b2:funcseq:integration}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:funcseq:3}
Demuestra que $S(x) = \sum_{n\geq1} \dfrac{x^n}{n^2}$ es \omterm{def:b2:metric:continuity}{continua}
sobre $\intcc{-1}{1}$ y que $S$ es de clase $C^1$ sobre
$\intoo{-1}{1}$ con $S'(x) = -\frac{\ln(1-x)}{x}$ para $0 < \abs x <
1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:funcseq:4}
Sea $F(x) = \sum_{n \geq 0} \dfrac{(-1)^n}{n + x}$ sobre
$\intoo{0}{+\infty}$. Demuestra la \omterm{def:b2:funcseq:def}{convergencia uniforme} (no normal)
sobre $\intco{\delta}{\infty}$ mediante la cota del resto de las
series alternadas, la \omterm{def:b2:metric:continuity}{continuidad} y la ecuación funcional $F(x) +
F(x + 1) = \frac1x$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:funcseq:5}
(Dini) Sean $f_n \colon K \to \R$ \omterm{def:b2:metric:continuity}{continuas} sobre un
\omterm{def:b2:metric:def}{espacio métrico} \emph{\omterm{def:b2:metric:compact}{compacto}}, con $f_n \to f$
\emph{\omterm{def:b2:funcseq:def}{puntualmente}}, $f$ \omterm{def:b2:metric:continuity}{continua} y $(f_n(x))$ \emph{decreciente} en
$n$ para cada $x$. Demuestra que la convergencia es uniforme.
\emph{(Dado $\varepsilon$, los \omterm{def:b2:metric:topology}{abiertos} $U_n = \{x : f_n(x) - f(x) <
\varepsilon\}$ crecen y recubren $K$; extráigase un subrecubrimiento
finito, \cref{thm:b2:metric:borellebesgue}.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:funcseq:6}
Demuestra que $\displaystyle\lim_{n\to\infty} \int_0^1
\frac{n\,f(x)}{1 + n^2x^2}\,\dd x = \frac{\pi}{2} f(0)$ para toda $f$
\omterm{def:b2:metric:continuity}{continua} sobre $\intcc{0}{1}$. \emph{(Sustituye $u = nx$; separa
$f(0)$; domina.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:funcseq:7}
Calcula explícitamente los \omterm{thm:b2:funcseq:weierstrass}{polinomios de Bernstein} de $f(x) = x^2$ y
verifica el error uniforme $\norm{B_n f - f}_\infty =
O\bigl(\frac1n\bigr)$ que predice la demostración del~\cref{thm:b2:funcseq:weierstrass}; aquí vale exactamente $\frac{x(1 -
x)}{n}$ en cada punto.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:funcseq:8}
Demuestra que si unos polinomios $P_n$ convergen \omterm{def:b2:funcseq:def}{uniformemente}
\emph{en todo $\R$} a $f$, entonces $f$ es un polinomio. \emph{(Para
$m, n$ grandes, $P_n - P_m$ es un polinomio acotado sobre $\R$ y por
tanto constante; así que la sucesión se estabiliza salvo
constantes.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:funcseq:9}
(Una función \omterm{def:b2:metric:continuity}{continua} y no derivable en ningún punto ---guiado) Sea
$\varphi$ la distancia al entero más próximo ($1$-periódica, con
$\norm{\varphi}_\infty = \frac12$ y $1$-lipschitziana) y
\[
W(x) = \sum_{n=0}^{\infty} \Bigl(\frac{3}{4}\Bigr)^{\!n}
\varphi(4^n x) .
\]
Demuestra: (a) $W$ es \omterm{def:b2:metric:continuity}{continua} sobre $\R$ (\omterm{def:b2:funcseq:series}{convergencia normal});
(b) para todo $x$ y todo $m$, eligiendo $h_m = \pm\frac12\cdot
4^{-m}$ con el signo que haga a $\varphi$ afín sobre el segmento de
$4^m x$ a $4^m(x + h_m)$, el cociente incremental cumple
\[
\Bigl|\frac{W(x + h_m) - W(x)}{h_m}\Bigr| \geq 3^m -
\sum_{n<m} 3^n \geq \frac{3^m + 1}{2} \xrightarrow[m\to\infty]{}
\infty
\]
(los términos $n > m$ se anulan por periodicidad; el término $n = m$
aporta exactamente $3^m$; y los términos $n < m$ están acotados por
la propiedad \omterm{def:b2:metric:continuity}{lipschitziana}). Concluye que $W$ no es derivable en
ningún punto.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:funcseq:10}
Sea $u_n(x) = (-1)^n x^n(1 - x)$ sobre $\intcc{0}{1}$. Prueba que
$\sum u_n$ converge \omterm{def:b2:funcseq:def}{puntualmente} sobre $\intcc{0}{1}$ y calcula su
suma; prueba que la convergencia es uniforme sobre $\intcc{0}{1}$
\emph{(acota el resto $R_N(x) = \sum_{n > N} u_n(x)$, una cola
geométrica, por su primer término y maximiza $x^{N+1}(1-x)$)} pero
\emph{no} normal \emph{(calcula $\norm{u_n}_\infty$)}: la
\omterm{def:b2:funcseq:def}{convergencia uniforme} es estrictamente más débil que la
normal. Contrasta con $\sum x^n(1-x)$, cuya suma es discontinua en
$1$: allí falla incluso la uniformidad.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:funcseq:11}
Sea $f_n \to f$ \omterm{def:b2:funcseq:def}{uniformemente} sobre un \omterm{def:b2:metric:def}{espacio métrico} $X$, con cada
$f_n$ \omterm{def:b2:metric:continuity}{continua}, y sea $x_n \to x$ en $X$. Demuestra que $f_n(x_n) \to
f(x)$. Muestra con un ejemplo sobre $X = \intcc{0}{1}$ que la
\omterm{def:b2:funcseq:def}{convergencia puntual} no basta, ni siquiera con $f$ \omterm{def:b2:metric:continuity}{continua}
\emph{(usa las jorobas $g_n$ del~\cref{exo:b2:funcseq:1} y $x_n =
\frac{1}{\sqrt{2n}}$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:funcseq:12}
(Una ecuación integral de Volterra por series) Para $f \in
C(\intcc{0}{1})$, definamos $Tf(x) = \int_0^x f(t)\,\dd t$.
\begin{enumerate}
  \item Prueba por inducción que, para $n \geq 1$,
        \[
        T^n f(x) = \int_0^x \frac{(x -
        t)^{n-1}}{(n-1)!}\,f(t)\,\dd t,
        \qquad
        \norm{T^n f}_\infty \leq \frac{\norm f_\infty}{n!} .
        \]
  \item Deduce que $S = \sum_{n\geq0} T^n f$ converge \omterm{def:b2:funcseq:series}{normalmente}
        sobre $\intcc{0}{1}$ y resuelve la ecuación integral $S = f
        + TS$.
  \item Verifica que $S(x) = f(x) + \int_0^x \eu^{x-t}f(t)\,\dd t$
        resuelve la misma ecuación y demuestra la unicidad de las
        soluciones \omterm{def:b2:metric:continuity}{continuas} \emph{(si $S = TS$, entonces
        $\norm{S}_\infty \leq \norm{T^nS}_\infty \to 0$)}: concluye
        la forma cerrada de la suma.
\end{enumerate}
\end{exercise}

\section{Problema: velocidades de aproximación y teorema de Korovkin}

\begin{problem}\label{pb:b2:funcseq:1}
La demostración de Bernstein del~\cref{thm:b2:funcseq:weierstrass} esconde dos tesoros. Primero, es
\emph{cuantitativa}: la rapidez con que $B_nf \to f$ está gobernada
por el módulo de \omterm{def:b2:metric:continuity}{continuidad} de $f$, con la velocidad óptima alcanzada
por $\abs{x - \frac12}$. Segundo, es \emph{estructural}: lo único
que importaba era que $B_n$ es un operador lineal positivo que se
porta bien sobre $1$, $x$ y $x^2$; esa observación, aislada, es el
\emph{teorema de Korovkin}\index{teorema de Korovkin}. Este problema
demuestra ambas cosas y cierra con la asintótica exacta de
Voronovskaya. En todo él, $f \in C(\intcc{0}{1})$, $M = \norm
f_\infty$, $p_k(x) = \binom nk x^k(1-x)^{n-k}$ y $e_j$ denota $x
\mapsto x^j$.

\medskip
\noindent\textbf{Parte I --- El operador de Bernstein.}
\begin{enumerate}
  \item Prueba que $B_n$ es lineal y \emph{positivo} ($f \geq 0
        \Rightarrow B_nf \geq 0$), y por tanto monótono ($f \leq g
        \Rightarrow B_nf \leq B_ng$), con $\norm{B_nf}_\infty \leq
        \norm f_\infty$, y que $B_nf$ interpola $f$ en ambos
        extremos.
  \item Vuelve a deducir las identidades $B_n e_0 = e_0$, $B_n e_1
        = e_1$ y $B_n e_2 = e_2 + \frac{e_1 - e_2}{n}$
        \emph{(deriva $(x + y)^n$ dos veces y pon $y = 1 - x$)}.
  \item Deduce la identidad de la varianza $\sum_k \bigl(\frac kn -
        x\bigr)^2 p_k(x) = \frac{x(1-x)}{n}$ y, por
        Cauchy--Schwarz, la cota del primer momento
        \[
        \sum_{k=0}^{n}\Bigl|\frac kn - x\Bigr|\,p_k(x)
        \leq \sqrt{\frac{x(1-x)}{n}} \leq \frac{1}{2\sqrt n} .
        \]
  \item Prueba que si $f$ es convexa, entonces $B_nf \geq f$ sobre
        $\intcc{0}{1}$ \emph{(desigualdad de Jensen finita para los
        pesos $p_k(x)$)}.
  \item (La cota de recuento de Chebyshev, reformulada) Para
        $\delta > 0$, prueba que
        \[
        \sum_{\abs{k/n - x} > \delta} p_k(x)
        \leq \frac{x(1-x)}{n\delta^2}
        \leq \frac{1}{4n\delta^2} ,
        \]
        y da la lectura probabilística: $B_nf(x)$ promedia $f$ sobre
        una media muestral binomial que se concentra en $x$.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Velocidades: el módulo de
\omterm{def:b2:metric:continuity}{continuidad}.} Para $\delta > 0$, pongamos $\omega(\delta) =
\sup\{\abs{f(s) - f(t)} : s, t \in \intcc{0}{1},\ \abs{s - t} \leq
\delta\}$.
\begin{enumerate}[resume]
  \item Prueba que $\omega$ es finito y no decreciente, que
        $\omega(\delta) \to 0$ cuando $\delta \to 0^+$ (Heine), que
        es subaditivo ($\omega(\delta_1 + \delta_2) \leq
        \omega(\delta_1) + \omega(\delta_2)$) y que
        $\omega(\lambda\delta) \leq (1 + \lambda)\,\omega(\delta)$
        para todo $\lambda > 0$.
  \item Demuestra la estimación maestra, para todo $\delta > 0$:
        \[
        \abs{B_nf(x) - f(x)}
        \leq \sum_k \omega\Bigl(\Bigl|\frac kn -
        x\Bigr|\Bigr)p_k(x)
        \leq \Bigl(1 + \frac1\delta\sum_k\Bigl|\frac kn -
        x\Bigr|p_k(x)\Bigr)\,\omega(\delta) .
        \]
  \item Elige $\delta = n^{-1/2}$ y concluye el \emph{teorema de
        Weierstrass cuantitativo}:
        \[
        \norm{B_nf - f}_\infty \leq
        \frac32\,\omega\Bigl(\frac{1}{\sqrt n}\Bigr)
        \xrightarrow[n\to\infty]{} 0 .
        \]
  \item Deduce las velocidades: $\norm{B_nf - f}_\infty \leq
        \frac{3L}{2\sqrt n}$ para $f$ $L$-lipschitziana, y $\leq
        \frac32 C n^{-\alpha/2}$ para $f$ $\alpha$-hölderiana
        ($\abs{f(s) - f(t)} \leq C\abs{s-t}^\alpha$).
  \item (El ejemplo óptimo ---una identidad binomial) Para $m \geq
        1$, demuestra que
        \[
        \sum_{k=m+1}^{2m} (k - m)\binom{2m}{k}
        = \frac{m}{2}\binom{2m}{m},
        \qquad\text{de donde}\qquad
        \sum_{k=0}^{2m}\abs{k - m}\binom{2m}{k}
        = m\binom{2m}{m}
        \]
        \emph{(usa $k\binom{2m}k = 2m\binom{2m-1}{k-1}$ y la
        simetría de la línea binomial, que da
        $\sum_{j=m}^{2m-1}\binom{2m-1}{j} = 2^{2m-2}$)}.
  \item Para $f(t) = \abs{t - \frac12}$, deduce el valor exacto y su
        asintótica (binomial central,
        \cref{ex:b2:comparison:centralbinomial}):
        \[
        B_{2m}f\Bigl(\frac12\Bigr) - f\Bigl(\frac12\Bigr)
        = \frac{\binom{2m}{m}}{2\cdot4^{m}}
        \;\sim\; \frac{1}{2\sqrt{\pi m}} :
        \]
        la velocidad $\omega(n^{-1/2})$ de la pregunta 8 se alcanza
        (salvo constante); para $f$ meramente \omterm{def:b2:metric:continuity}{continua}, el
        $n^{-1/2}$ de Bernstein es honesto.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El teorema de Korovkin.} Sea $(L_n)$
una sucesión de operadores \emph{lineales positivos} de
$C(\intcc{0}{1})$ en sí mismo tal que $L_ne_j \to e_j$
\omterm{def:b2:funcseq:def}{uniformemente} para $j = 0, 1, 2$.
\begin{enumerate}[resume]
  \item Prueba que un operador lineal positivo $L$ es monótono y
        cumple $\abs{Lf} \leq L\abs f$ \omterm{def:b2:funcseq:def}{puntualmente}.
  \item Prueba que para todo $\varepsilon > 0$ existe $\delta > 0$
        tal que, para \emph{todos} $s, x \in \intcc{0}{1}$,
        \[
        \abs{f(s) - f(x)} \leq \varepsilon +
        \frac{2M}{\delta^2}(s - x)^2
        \]
        \emph{(trata $\abs{s - x} \leq \delta$ con Heine y $\abs{s-x}
        > \delta$ con la cota tosca $2M$)}.
  \item Fija $x$, aplica $L_n$ a la desigualdad de la pregunta 13 en
        la variable $s$ y deduce
        \[
        \abs{L_nf(x) - f(x)\,L_ne_0(x)}
        \leq \varepsilon\,L_ne_0(x) + \frac{2M}{\delta^2}
        \bigl(L_ne_2(x) - 2x\,L_ne_1(x) + x^2 L_ne_0(x)\bigr).
        \]
  \item Prueba que $\sup_x \bigl(L_ne_2(x) - 2x\,L_ne_1(x) + x^2
        L_ne_0(x)\bigr) \to 0$ y ensambla el \emph{teorema de
        Korovkin}: $L_nf \to f$ \omterm{def:b2:funcseq:def}{uniformemente} para \emph{toda} $f
        \in C(\intcc{0}{1})$.
  \item Comprueba que $(B_n)$ satisface las hipótesis de Korovkin:
        Weierstrass por tercera vez, a partir de tres monomios.
  \item Sea $I_n$ el operador de interpolación afín a trozos en los
        nodos $\frac kn$. Prueba que $I_n$ es lineal positivo, que
        $I_ne_0 = e_0$, $I_ne_1 = e_1$ y $\norm{I_ne_2 -
        e_2}_\infty = \frac{1}{4n^2}$ \emph{(en cada celda, el error
        de la interpolación afín de $t^2$ es $(t - a)(b - t)$)}.
        Concluye por Korovkin que los interpolantes poligonales
        convergen \omterm{def:b2:funcseq:def}{uniformemente} para toda $f$ \omterm{def:b2:metric:continuity}{continua}.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Dividendos: densidad, momentos,
derivadas.}
\begin{enumerate}[resume]
  \item Prueba que los polinomios con coeficientes
        \emph{racionales} son densos en $\bigl(C(\intcc{0}{1}),
        \norm\cdot_\infty\bigr)$: este \omterm{def:b2:nvs:banach}{espacio de Banach} es
        separable.
  \item (Los momentos determinan la función) Sea $f \in
        C(\intcc{0}{1})$ con $\int_0^1 f(t)\,t^n \dd t = 0$ para
        todo $n \in \N$. Prueba que $\int_0^1 f P = 0$ para todo
        polinomio, después que $\int_0^1 f^2 = 0$ y, finalmente, que
        $f = 0$.
  \item Demuestra la identidad de la derivada
        \[
        (B_nf)'(x) = n\sum_{k=0}^{n-1}\Bigl(
        f\Bigl(\frac{k+1}{n}\Bigr) -
        f\Bigl(\frac kn\Bigr)\Bigr)\,
        \binom{n-1}{k}x^k(1-x)^{n-1-k}
        \]
        \emph{(deriva $p_k$ y reindexa: una \omterm{thm:b2:series:abel}{sumación de Abel})}.
  \item Supongamos $f$ de clase $C^1$. Usando el teorema del valor
        medio en cada incremento y comparando con $B_{n-1}(f')$,
        prueba que $(B_nf)' \to f'$ \omterm{def:b2:funcseq:def}{uniformemente} sobre
        $\intcc{0}{1}$. Deduce que, para $f \in C^1$, hay
        polinomios que convergen a $f$ \emph{junto con} sus
        derivadas.
  \item Supongamos $f$ de clase $C^2$. Mediante Taylor--Lagrange en
        $x$, prueba que
        \[
        \abs{B_nf(x) - f(x)} \leq
        \frac{\norm{f''}_\infty}{2}\cdot\frac{x(1-x)}{n}
        \leq \frac{\norm{f''}_\infty}{8n} :
        \]
        la suavidad mejora la velocidad de $n^{-1/2}$ a $n^{-1}$.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Saturación: el teorema de
Voronovskaya.}
\begin{enumerate}[resume]
  \item Demuestra la identidad del cuarto momento
        \[
        \sum_k (k - nx)^4 p_k(x)
        = nx(1-x)\bigl(1 + 3(n-2)x(1-x)\bigr) \leq n^2
        \quad (n \geq 1)
        \]
        \emph{(desarrolla $k^4$ en factoriales descendentes
        $k(k-1)\cdots$ y usa dos veces más el truco de derivación de
        la pregunta 2)}.
  \item (Voronovskaya) Sea $f$ de clase $C^2$ y $x \in
        \intcc{0}{1}$. Escribiendo $f(t) = f(x) + f'(x)(t-x) +
        \frac{f''(x)}2(t-x)^2 + \eta(t)(t-x)^2$ con $\eta$ acotada y
        $\eta(t) \to 0$ cuando $t \to x$, demuestra que
        \[
        n\bigl(B_nf(x) - f(x)\bigr)
        \xrightarrow[n\to\infty]{}
        \frac{x(1-x)}{2}\,f''(x)
        \]
        \emph{(separa la suma de $\eta$ en $\abs{t - x} \leq
        \delta$; controla la parte lejana con la pregunta 23)}. Así
        pues, el error de la pregunta 22 es exacto en orden \emph{y}
        en constante: $B_n$ \emph{satura} en $\frac1n$ por suave que
        sea $f$; compárese con el~\cref{exo:b2:funcseq:7}.
  \item Síntesis. Una frase para cada punto: (i) qué compró la
        positividad por sí sola (partes I y III); (ii) dónde
        intervino la \omterm{def:b2:metric:compact}{compacidad} de $\intcc{0}{1}$ en cada parte;
        (iii) por qué bastan tres funciones de prueba en el teorema
        de Korovkin; (iv) qué compromiso acepta Bernstein (un
        $n^{-1/2}$ robusto para $f$ rugosa, pero un techo de
        $\frac1n$ para $f$ suave) y qué capítulo de este libro
        jugará la misma partida con polinomios trigonométricos.
\end{enumerate}
\end{problem}
