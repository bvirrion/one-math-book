\chapter{Espacios afines}\label{ch:b2:affine}

Los espacios vectoriales tienen un punto privilegiado ---el origen---
que la geometría no quiere. Un \emph{\omterm{def:b2:affine:def}{espacio afín}} es un espacio
vectorial que ha olvidado su origen: los puntos y los vectores pasan a
ser especies distintas, relacionadas por traslación. Este capítulo
breve construye el diccionario (puntos, \omterm{def:b2:affine:barycenter}{baricentros}, subespacios y
\omterm{def:b2:affine:subspace}{aplicaciones afines}), la visión \omterm{def:b2:affine:subspace}{afín} de la convexidad y las
herramientas de clasificación que se usarán en los capítulos de
geometría siguientes.

\section{Puntos y vectores}

\begin{definition}\label{def:b2:affine:def}
Un \emph{espacio afín}\index{espacio afín} dirigido por un espacio
vectorial real $E$ es un conjunto no vacío $\mathcal{E}$ dotado de
una aplicación $(A, B) \mapsto \vect{AB} \in E$ que cumple
\[
\vect{AB} + \vect{BC} = \vect{AC}
\quad \text{(Chasles)},
\qquad
\text{para cada } A,\ B \mapsto \vect{AB} \text{ es una biyección }
\mathcal{E} \to E .
\]
Se escribe $B = A + u$ para el único punto con $\vect{AB} = u$. La
\emph{dimensión} de $\mathcal{E}$ es $\dim E$. Todo espacio vectorial
es un espacio \omterm{def:b2:affine:subspace}{afín} sobre sí mismo ($\vect{AB} = B - A$), y toda
elección de un origen $O \in \mathcal{E}$ identifica $\mathcal{E}$
con $E$ mediante $M \mapsto \vect{OM}$.
\end{definition}

\begin{example}[Un espacio afín sin origen
natural]\label{ex:b2:affine:planeexample}
El plano de soluciones $\mathcal E = \{(x, y, z) \in \R^3 : x + y +
z = 1\}$ no es un subespacio vectorial ($0 \notin \mathcal E$), pero
sí es un \omterm{def:b2:affine:def}{espacio afín} dirigido por $E = \{x + y + z = 0\}$: para $A,
B \in \mathcal E$, la diferencia $\vect{AB} = B - A$ cae en $E$ (las
sumas se cancelan), Chasles se hereda de $\R^3$ y $B \mapsto
\vect{AB}$ es biyectiva sobre $E$. Ningún punto de $\mathcal E$ está
distinguido: cualquier elección de ``origen'' $O \in \mathcal E$
sirve igual de bien, y todas las identificaciones $M \mapsto
\vect{OM}$ difieren en traslaciones. Esta es la situación típica: los
conjuntos de soluciones de los problemas lineales no homogéneos
(sistemas lineales, ecuaciones diferenciales lineales del~\cref{ch:b2:diffeq}) son \omterm{def:b2:affine:subspace}{afines}, nunca lineales, y el lema ``solución
particular más núcleo'' es exactamente el enunciado $\mathcal F = A +
F$ de la definición siguiente.
\end{example}

\begin{definition}[Baricentro]\label{def:b2:affine:barycenter}
Sean $(A_i, \lambda_i)_{i \leq k}$ puntos con pesos y
$\sum\lambda_i \neq 0$. El \emph{baricentro}\index{baricentro} $G =
\operatorname{bar}\bigl((A_i, \lambda_i)\bigr)$ es el único punto
con
\[
\sum_i \lambda_i\, \vect{GA_i} = 0
\qquad\text{o, equivalentemente,}\qquad
\vect{OG} = \frac{1}{\sum\lambda_i}\sum_i \lambda_i\,\vect{OA_i}
\quad (\text{para cualquier } O).
\]
Los baricentros son \emph{asociativos} (todo subgrupo de puntos
puede sustituirse por su baricentro parcial con el peso sumado) e
invariantes al reescalar todos los pesos.
\end{definition}

\begin{proof}[Demostración de la existencia y de las fórmulas]
Fíjese $O$ y escríbase $s = \sum_i\lambda_i \neq 0$. Por Chasles,
\[
\sum_i\lambda_i\,\vect{GA_i} = 0
\iff \sum_i\lambda_i\bigl(\vect{GO} + \vect{OA_i}\bigr) = 0
\iff s\,\vect{OG} = \sum_i\lambda_i\,\vect{OA_i},
\]
lo que determina $G = O + \frac1s\sum_i\lambda_i\vect{OA_i}$ de
manera única. \emph{Independencia de $O$:} para otro origen $O'$,
\[
\frac1s\sum_i\lambda_i\,\vect{O'A_i}
= \frac1s\sum_i\lambda_i\bigl(\vect{O'O} + \vect{OA_i}\bigr)
= \vect{O'O} + \frac1s\sum_i\lambda_i\,\vect{OA_i}
= \vect{O'G} :
\]
el mismo punto $G$. \emph{Asociatividad:} pártase el conjunto de
índices como $I \sqcup J$ con $s_I = \sum_{i\in I}\lambda_i \neq 0$,
y sea $G_I$ el \omterm{def:b2:affine:barycenter}{baricentro} de $(A_i, \lambda_i)_{i\in I}$, de modo
que $\sum_{i\in I}\lambda_i\vect{OA_i} = s_I\,\vect{OG_I}$.
Entonces
\[
s\,\vect{OG}
= \sum_{i\in I}\lambda_i\vect{OA_i} +
\sum_{j\in J}\lambda_j\vect{OA_j}
= s_I\,\vect{OG_I} + \sum_{j\in J}\lambda_j\vect{OA_j} :
\]
$G$ es el \omterm{def:b2:affine:barycenter}{baricentro} de $(G_I, s_I)$ junto con $(A_j,
\lambda_j)_{j\in J}$, como se afirmaba. \emph{Reescalado:}
sustituir cada $\lambda_i$ por $t\lambda_i$ ($t \neq 0$) multiplica
$s$ y la suma ponderada por $t$, y deja $\vect{OG}$ inalterado.
\end{proof}

\begin{remark}[Errores frecuentes]
Dos trampas rodean la definición. Primera: si los pesos suman
\emph{cero}, no hay \omterm{def:b2:affine:barycenter}{baricentro}; la aplicación $O \mapsto
\sum\lambda_i\vect{OA_i}$ es entonces independiente de $O$ y define
un \emph{vector}, no un punto; por ejemplo, $(A, -1;\ B, 1)$
codifica $\vect{AB}$. Llevar la cuenta de cuál de los dos objetos
produce un cálculo es la mitad de la higiene baricéntrica. Segunda:
los pesos solo tienen sentido salvo un factor común no nulo; las
fórmulas del tipo ``las coordenadas de $G$ son $\lambda_1, \dots,
\lambda_k$'' presuponen una normalización (habitualmente
$\sum\lambda_i = 1$), y olvidarse de normalizar es la fuente
estándar de razones equivocadas sobre una figura.
\end{remark}

\begin{definition}[Subespacios afines; aplicaciones afines]\label{def:b2:affine:subspace}
Un \emph{subespacio afín} es un conjunto $\mathcal{F} = A + F = \{A
+ u : u \in F\}$ con $F$ un subespacio vectorial (su
\emph{dirección}); equivalentemente, un conjunto no vacío estable
por \omterm{def:b2:affine:barycenter}{baricentros}. Los subespacios afines de $\R^n$ son exactamente
los conjuntos de soluciones de los sistemas lineales $MX = B$
(primer año: solución particular más núcleo). Una aplicación $f
\colon \mathcal{E} \to \mathcal{E}'$ es \emph{afín}\index{aplicación
afín} cuando conserva los \omterm{def:b2:affine:barycenter}{baricentros}; equivalentemente, cuando
\[
f(A + u) = f(A) + \varphi(u)
\]
para una (única) aplicación lineal $\varphi = \vec f$, la
\emph{parte lineal}. Aplicaciones afines de $\R^n$: $X \mapsto MX +
C$. Las composiciones son afines con las partes lineales
compuestas; $f$ es biyectiva si y solo si lo es $\vec f$.
\end{definition}

\begin{proof}[Demostración de la equivalencia para las aplicaciones]
Si $f(A + u) = f(A) + \varphi(u)$: para un \omterm{def:b2:affine:barycenter}{baricentro} $G$ de $(A_i,
\lambda_i)$, desarrollando todos los puntos desde $A$, $f(G) = f(A)
+ \varphi(\vect{AG})$ y $\varphi(\vect{AG}) =
\frac{\sum\lambda_i\varphi(\vect{AA_i})}{\sum\lambda_i}$: $f(G)$ es
el \omterm{def:b2:affine:barycenter}{baricentro} de las imágenes. Recíprocamente, fíjese $A$ y
defínase $\varphi(u) = \vect{f(A)\,f(A + u)}$. \emph{Homogeneidad:}
$A + tu = \operatorname{bar}\bigl(A, 1-t;\ A + u, t\bigr)$ para todo
$t$ real, de modo que la conservación de los \omterm{def:b2:affine:barycenter}{baricentros} (con pesos
reales arbitrarios, como se ha supuesto) da $\varphi(tu) =
t\,\varphi(u)$ directamente. \emph{Aditividad:} $A + u + v =
\operatorname{bar}\bigl(A + 2u, \tfrac12;\ A + 2v, \tfrac12\bigr)$,
luego $\varphi(u + v) = \tfrac12\varphi(2u) + \tfrac12\varphi(2v) =
\varphi(u) + \varphi(v)$, usando la homogeneidad. Por tanto,
$\varphi$ es lineal.
\end{proof}

\begin{remark}
La demostración usó \omterm{def:b2:affine:barycenter}{baricentros} con pesos \emph{reales
arbitrarios}: el paso de homogeneidad toma $t$ fuera de
$\intcc01$. Si solo se supone que una aplicación conserva los
\omterm{def:b2:affine:barycenter}{baricentros} de pesos no negativos ---equivalentemente, los puntos
medios y los segmentos---, la linealidad de la aplicación vectorial
ya no sale gratis: solo se obtiene $\Q$-linealidad, y hace falta una
hipótesis de \omterm{def:b2:metric:continuity}{continuidad} para concluir, exactamente como en el~\cref{exo:b2:affine:5}. Distinguir ``conserva todos los
\omterm{def:b2:affine:barycenter}{baricentros}'' de ``conserva las combinaciones \omterm{def:b2:affine:convex}{convexas}'' es una
sutileza pequeña pero real del vocabulario \omterm{def:b2:affine:subspace}{afín}.
\end{remark}

\begin{example}[Geometría baricéntrica clásica]\label{ex:b2:affine:median}
El centroide de un triángulo $ABC$ es el \omterm{def:b2:affine:barycenter}{baricentro} $G =
\operatorname{bar}(A,1; B,1; C,1)$. La asociatividad con el punto
medio $A' = \operatorname{bar}(B, 1; C, 1)$ muestra que
\[
G = \operatorname{bar}(A, 1;\ A', 2) :
\]
$G$ está sobre la mediana $AA'$ a dos tercios de ella; y lo mismo
para las otras dos medianas: las tres medianas son concurrentes, en
una línea de cálculo baricéntrico.
\end{example}

\begin{example}[Las bimedianas de un
cuadrilátero]\label{ex:b2:affine:bimedians}
Sea $ABCD$ un cuadrilátero cualquiera (¡plano o no!) y considérense
sus \emph{bimedianas}: los segmentos que unen los puntos medios de
lados opuestos, $M_{AB}M_{CD}$ y $M_{BC}M_{DA}$. Introdúzcase el
\omterm{def:b2:affine:barycenter}{baricentro} $G$ de $(A,1; B,1; C,1; D,1)$ y agrúpense los pesos de
dos maneras:
\[
G = \operatorname{bar}\bigl(M_{AB}, 2;\ M_{CD}, 2\bigr)
= \operatorname{bar}\bigl(M_{BC}, 2;\ M_{DA}, 2\bigr) :
\]
$G$ es el punto medio de \emph{ambas} bimedianas, de modo que las dos
bimedianas siempre se bisecan mutuamente, y el cuadrilátero de los
cuatro puntos medios es un paralelogramo (sus diagonales son las
bimedianas). Sin análisis de casos, sin coordenadas, y el argumento
sobrevive sin cambios para un cuadrilátero alabeado de $\R^3$, donde
una demostración basada en un dibujo ya sería delicada: a la
asociatividad no le importa la dimensión.
\end{example}

\begin{example}[Clasificar una aplicación afín, de principio a
fin]\label{ex:b2:affine:dilation}
Sea $f(x, y) = (2x - 1,\ 3y - 4)$ sobre $\R^2$. Su parte lineal es
$\varphi = \operatorname{diag}(2, 3)$, cuyo \omterm{def:b2:reduction:eigen}{espectro} $\{2, 3\}$
evita el $1$: por el criterio del punto fijo demostrado más abajo
(\cref{prop:b2:affine:fixedpoint}), $f$ tiene exactamente un punto
fijo, hallado resolviendo
\[
x = 2x - 1, \qquad y = 3y - 4
\qquad\Longrightarrow\qquad \Omega = (1, 2).
\]
Recentrando en $\Omega$ (póngase $x = 1 + u$, $y = 2 + v$):
\[
f(1 + u,\ 2 + v) = (1 + 2u,\ 2 + 3v) :
\]
en el sistema de referencia de $\Omega$, $f$ \emph{es} su parte
lineal, una dilatación anisótropa que estira por $2$ en horizontal y
por $3$ en vertical desde el centro $(1, 2)$. La lección general: una
\omterm{def:b2:affine:subspace}{aplicación afín} es ``aplicación lineal más datos de posición'', y
los datos de posición colapsan en un único origen bien elegido
siempre que $1$ no sea \omterm{def:b2:reduction:eigen}{valor propio}. Recíprocamente, trasladar mal el
origen \emph{crea} los términos constantes: la geometría \omterm{def:b2:affine:subspace}{afín} es el
arte de elegir dónde poner el $0$.
\end{example}

\begin{remark}[Método: concurrencia y alineación mediante
baricentros]
El~\cref{ex:b2:affine:median} es un caso de una receta general. Para
demostrar que tres cevianas de un triángulo son concurrentes,
exhíbase un \emph{único} sistema con pesos $(A, \alpha; B, \beta; C,
\gamma)$ y úsese la asociatividad de tres maneras: agrupar $(B, C)$
muestra que el \omterm{def:b2:affine:barycenter}{baricentro} está sobre la ceviana desde $A$; agrupar
$(C, A)$, sobre la ceviana desde $B$; y agrupar $(A, B)$, sobre la
tercera. Para las medianas, el sistema $(A, 1; B, 1; C, 1)$ hace todo
el trabajo; para cevianas que cortan los lados en razones
prescritas, los pesos se leen en las razones. Para demostrar que tres
puntos están \emph{alineados}, escríbase uno como \omterm{def:b2:affine:barycenter}{baricentro} de los
otros dos (\cref{exo:b2:affine:2}), o úsese el criterio del
\omterm{def:b2:linalg:det}{determinante} del~\cref{exo:b2:affine:11}. Ambas recetas sustituyen el
ingenio geométrico por contabilidad de pesos: exactamente para eso
sirve el cálculo baricéntrico.
\end{remark}

\section{Convexidad, en clave afín}

\begin{definition}\label{def:b2:affine:convex}
Un subconjunto $C$ de un \omterm{def:b2:affine:def}{espacio afín} es \emph{convexo}\index{conjunto
convexo} cuando contiene todo \omterm{def:b2:affine:barycenter}{baricentro} de pesos \emph{no
negativos} de sus puntos; equivalentemente, todo segmento
$\intcc{A}{B} = \{\operatorname{bar}(A, 1-t; B, t) : t \in
\intcc{0}{1}\}$ entre sus puntos. La \emph{envoltura
convexa}\index{envoltura convexa} $\operatorname{conv}(S)$ es el
conjunto de todos los \omterm{def:b2:affine:barycenter}{baricentros} con pesos no negativos de puntos
de $S$: el menor conjunto convexo que contiene a $S$.
\end{definition}

\begin{example}[Los epigrafos son conjuntos
convexos]\label{ex:b2:affine:epigraph}
La región $C = \{(x, y) : y \geq x^2\}$ por encima de la parábola es
\omterm{def:b2:affine:convex}{convexa}: para $(x_1, y_1), (x_2, y_2) \in C$ y $t \in \intcc01$, la
desigualdad de convexidad de la función cuadrado da
\[
\bigl((1-t)x_1 + tx_2\bigr)^2 \leq (1-t)x_1^2 + tx_2^2
\leq (1-t)y_1 + ty_2 ,
\]
de modo que el \omterm{def:b2:affine:barycenter}{baricentro} se queda por encima de la parábola. El
cálculo es general: $\{y \geq f(x)\}$ es \omterm{def:b2:affine:convex}{convexo} exactamente cuando
$f$ es una función convexa; los \emph{conjuntos} \omterm{def:b2:affine:convex}{convexos} y las
\emph{funciones} convexas (\cref{ch:b2:realfun}) son dos caras de una
misma noción, y los epigrafos son el diccionario. Esta es la razón
geométrica de que las funciones convexas tengan rectas de apoyo, el
hecho que demostrará la desigualdad de Jensen en el~\cref{ch:b2:randomvar}.
\end{example}

\begin{example}[Generadores redundantes de una envoltura
convexa]\label{ex:b2:affine:squarehull}
Consideremos el conjunto
\[
S = \{(0,0), (2,0), (2,2), (0,2), (1,1)\} .
\]
El quinto punto es el \omterm{def:b2:affine:barycenter}{baricentro}
\[
(1,1) = \operatorname{bar}\bigl((0,0), \tfrac12;\ (2,2),
\tfrac12\bigr),
\]
así que ya está en la envoltura de los otros cuatro:
$\operatorname{conv}(S)$ es el cuadrado con las cuatro esquinas como
vértices. En general, un punto de $S$ que sea \omterm{def:b2:affine:barycenter}{baricentro} de pesos no
negativos de los \emph{demás} puntos de $S$ puede suprimirse sin
cambiar la envoltura; los puntos que nunca pueden suprimirse (aquí,
las cuatro esquinas) son los \emph{puntos extremos} de la envoltura.
Determinarlos es un puro cálculo de \omterm{def:b2:affine:barycenter}{baricentros}: $(2,0)$, por
ejemplo, no puede escribirse como $\operatorname{bar}$ de los puntos
restantes con pesos no negativos, porque la primera coordenada
forzaría a poner todo el peso en los puntos con $x = 2$, y entonces
la segunda coordenada falla. Las cuestiones de convexidad se reducen,
una y otra vez, a resolver pequeños sistemas con pesos.
\end{example}

\begin{theorem}[Carathéodory]\label{thm:b2:affine:caratheodory}
En un \omterm{def:b2:affine:def}{espacio afín} de dimensión $n$, todo punto de
$\operatorname{conv}(S)$ es \omterm{def:b2:affine:barycenter}{baricentro} de a lo sumo $n + 1$ puntos de
$S$.\index{teorema de Carathéodory}
\end{theorem}

\begin{proof}
Sea $G = \operatorname{bar}(A_0, \lambda_0; \dots; A_k, \lambda_k)$
con $\lambda_i > 0$, $\sum\lambda_i = 1$ y $k + 1 > n + 1$ puntos.
Los $k$ vectores $\vect{A_0A_i}$ ($i \geq 1$) son ligados ($k > n$):
$\sum_{i\geq1}\mu_i \vect{A_0A_i} = 0$ de manera no trivial; poniendo
$\mu_0 = -\sum_{i\geq1}\mu_i$, obtenemos pesos $(\mu_i)$ con
$\sum\mu_i = 0$ y $\sum \mu_i\,\vect{OA_i} = 0$ (para cualquier $O$),
no todos nulos. Entonces, para todo real $t$, los pesos $\lambda_i -
t\mu_i$ siguen sumando $1$ y, como $\sum_i\mu_i\vect{OA_i} = 0$,
\[
\sum_i(\lambda_i - t\mu_i)\,\vect{OA_i}
= \sum_i\lambda_i\,\vect{OA_i} :
\]
producen el \emph{mismo} punto $G$. Deslicemos ahora $t$ desde $0$:
algún $\mu_i$ es positivo (suman cero y no son todos nulos), de modo
que
\[
t^* = \min\Bigl\{\frac{\lambda_i}{\mu_i} : \mu_i > 0\Bigr\}
\]
está bien definido y es positivo. En $t = t^*$: para los índices con
$\mu_i > 0$, $\lambda_i - t^*\mu_i \geq 0$ por minimalidad, con
igualdad en un índice minimizante; y para los índices con $\mu_i \leq
0$, $\lambda_i - t^*\mu_i \geq \lambda_i > 0$. Todos los pesos siguen
siendo no negativos y al menos uno ha muerto: $G$ queda reescrito
como \omterm{def:b2:affine:barycenter}{baricentro} de menos puntos. Itérese mientras queden más de $n +
1$ puntos.
\end{proof}

\begin{example}
En el plano ($n = 2$): todo punto de la \omterm{def:b2:affine:convex}{envoltura convexa} de un
conjunto finito está en un triángulo con vértices en el conjunto; el
contenido geométrico de Carathéodory, usado por igual en optimización
y en probabilidad (mezclas).
\end{example}

\begin{example}[Ejecutando el algoritmo de
Carathéodory]\label{ex:b2:affine:caratheodoryrun}
Escribamos el centro del cuadrado del~\cref{ex:b2:affine:squarehull} con sus cuatro esquinas $A_1 =
(0,0)$, $A_2 = (2,0)$, $A_3 = (2,2)$, $A_4 = (0,2)$:
\[
(1,1) = \operatorname{bar}\bigl(A_1, \tfrac14;\ A_2,
\tfrac14;\ A_3, \tfrac14;\ A_4, \tfrac14\bigr),
\]
cuatro puntos en dimensión $2$: uno de más. La receta de la
demostración pide unos pesos $(\mu_i)$ con $\sum\mu_i = 0$ y
$\sum\mu_i\vect{OA_i} = 0$: aquí sirve $\mu = (1, -1, 1, -1)$ (las
dos diagonales comparten su punto medio). Deslizar $\lambda_i \mapsto
\lambda_i - t\mu_i$ mantiene fijo el \omterm{def:b2:affine:barycenter}{baricentro} para todo $t$; el
valor admisible extremo $t = \frac14$ deja los pesos en $(0,
\tfrac12, 0, \tfrac12)$, matando $A_1$ y $A_3$ a la vez:
\[
(1,1) = \operatorname{bar}\bigl(A_2, \tfrac12;\ A_4,
\tfrac12\bigr),
\]
una representación con dos puntos, aún mejor que los tres que
garantiza el teorema, porque el centro resulta estar sobre un
segmento entre generadores. El algoritmo es enteramente mecánico:
hállese una dependencia, deslícese hasta que muera un peso y
repítase.
\end{example}

\section{Herramientas de clasificación afín}

\begin{proposition}[Puntos fijos de las aplicaciones afines]\label{prop:b2:affine:fixedpoint}
Sea $f$ un endomorfismo \omterm{def:b2:affine:subspace}{afín} de un \omterm{def:b2:affine:def}{espacio afín} de dimensión finita,
con parte lineal $\varphi$. Si $1 \notin
\operatorname{Sp}(\varphi)$, entonces $f$ tiene exactamente un punto
fijo $\Omega$ y, en la vectorialización en $\Omega$, $f$ \emph{es} su
parte lineal. (Las traslaciones, con $\varphi = \mathrm{id}$ y sin
punto fijo, son la obstrucción básica.)
\end{proposition}

\begin{proof}
Fíjese $O$ y escríbase $f(O + x) = f(O) + \varphi(x)$. El punto $O +
x$ es fijo si y solo si $O + x = f(O) + \varphi(x)$, es decir,
\[
(\mathrm{id} - \varphi)(x) = \vect{O f(O)} .
\]
En dimensión finita, $\mathrm{id} - \varphi$ es invertible si y solo
si $0$ no es \omterm{def:b2:reduction:eigen}{valor propio} de $\mathrm{id} - \varphi$, o sea, si y solo
si $1 \notin \operatorname{Sp}\varphi$; y en ese caso la ecuación
mostrada tiene exactamente una solución $x^*$, lo que da el único
punto fijo $\Omega = O + x^*$. Recentrado: para todo vector $u$,
\[
f(\Omega + u) = f(\Omega) + \varphi(u) = \Omega + \varphi(u),
\]
así que en el sistema de referencia de origen $\Omega$ la aplicación
se lee $u \mapsto \varphi(u)$: puramente lineal. Cuando $1 \in
\operatorname{Sp}\varphi$, o bien no existe ningún punto fijo (la
ecuación mostrada puede no tener solución, como para una traslación),
o bien existe todo un subespacio \omterm{def:b2:affine:subspace}{afín} de ellos (súmese a una solución
cualquier \omterm{def:b2:reduction:eigen}{vector propio} de \omterm{def:b2:reduction:eigen}{valor propio} $1$): la unicidad es
exactamente la condición espectral.
\end{proof}

\begin{example}[Isometrías del plano, completadas]\label{ex:b2:affine:isometries}
Una isometría \omterm{def:b2:affine:subspace}{afín} del plano euclídeo tiene parte lineal en $O(2)$:
una rotación $R_\theta$ o una simetría axial (volumen del primer
año). Si $\theta \neq 0$: $1 \notin \operatorname{Sp} R_\theta$,
luego la aplicación es una \emph{rotación en torno a un único
centro} (\cref{prop:b2:affine:fixedpoint}). Si la parte lineal es una
simetría axial: o bien es una simetría respecto de un eje (existen
puntos fijos), o bien una \emph{simetría con deslizamiento} (una
simetría axial compuesta con una traslación a lo largo del eje, sin
punto fijo). Junto con las traslaciones, esta es la clasificación
completa de las isometrías del plano.
\end{example}

\begin{remark}[Las isometrías del plano, de un vistazo]
Recogiendo los casos: identidad; traslaciones ($\vec f =
\mathrm{id}$, sin punto fijo salvo en el caso trivial); rotaciones
(parte lineal $R_\theta$ con $\theta \neq 0$: un centro); simetrías
axiales (parte lineal una simetría, con una recta de puntos fijos); y
simetrías con deslizamiento (misma parte lineal, sin punto fijo).
Cuatro familias más la identidad, cada una reconocible con solo dos
datos: la parte lineal y el conjunto de puntos fijos; el patrón de la~\cref{prop:b2:affine:fixedpoint} llevado hasta el final.
\end{remark}

\begin{example}[Una simetría con deslizamiento, pillada en
flagrante]\label{ex:b2:affine:glide}
Sea $f(x, y) = (y + 1,\ x + 1)$. La parte lineal $(x, y) \mapsto (y,
x)$ es la simetría respecto de la diagonal $y = x$, de modo que $1
\in \operatorname{Sp}\vec f$ y la~\cref{prop:b2:affine:fixedpoint} calla. Los puntos fijos exigirían $x
= y + 1$ e $y = x + 1$ a la vez: imposible; no hay ninguno, así que
$f$ no es una simetría axial. Elevar al cuadrado zanja la
clasificación:
\[
f\bigl(f(x, y)\bigr) = f(y + 1,\ x + 1) = (x + 2,\ y + 2),
\]
la traslación de vector $(2, 2)$: $f$ es la \emph{simetría con
deslizamiento} de eje la recta $y = x$ (desplazada como corresponda:
el punto medio de $M$ y $f(M)$ siempre está sobre $y = x + {}$una
constante, aquí $y = x$, como se comprueba en $M = (0, 0) \mapsto
(1,1)$) y de vector de deslizamiento $(1, 1)$, la mitad de $f \circ
f$. Compárese con el~\cref{exo:b2:affine:6}, donde la misma parte
lineal pero otra constante produjo una simetría axial honesta: cuando
está presente el \omterm{def:b2:reduction:eigen}{valor propio} $1$, el término constante lo decide
todo.
\end{example}

\begin{example}[Las recurrencias afines son dinámica
afín]\label{ex:b2:affine:recursion}
La recurrencia clásica $u_{n+1} = au_n + b$ ($a \neq 1$) itera la
\omterm{def:b2:affine:subspace}{aplicación afín} $f(x) = ax + b$ de la recta, cuya parte lineal $a$
evita el \omterm{def:b2:reduction:eigen}{valor propio} $1$: hay un único punto fijo $\omega =
\frac{b}{1-a}$, y recentrar ahí (el caso unidimensional de la
proposición anterior) convierte $f$ en la multiplicación por $a$:
\[
u_{n+1} - \omega = a\,(u_n - \omega)
\qquad\Longrightarrow\qquad
u_n = \omega + a^n(u_0 - \omega) .
\]
Para $u_{n+1} = \frac{u_n}2 + 3$: $\omega = 6$ y $u_n = 6 + (u_0 -
6)2^{-n} \to 6$. La receta que se enseña para tales recurrencias en
el~\cref{ch:b2:series} ---``réstese el punto fijo''--- es exactamente
la vectorialización de una \omterm{def:b2:affine:subspace}{aplicación afín} en su punto fijo; y la
convergencia para $\abs a < 1$ es el fenómeno de contracción que el~\cref{ch:b2:metric} convirtió en el teorema del punto fijo de Banach.
Una idea, tres capítulos.
\end{example}

\begin{example}[Hallar el centro de una rotación]\label{ex:b2:affine:rotationcenter}
Sea $f(x, y) = (-y + 2,\ x)$. La parte lineal es $\varphi(x, y) =
(-y, x)$: la rotación de ángulo $\frac\pi2$, cuyo \omterm{def:b2:reduction:eigen}{espectro} $\{\iu,
-\iu\}$ evita el $1$. Por la~\cref{prop:b2:affine:fixedpoint} hay
exactamente un punto fijo: $x = -y + 2$ e $y = x$ dan $x = 1$, $y =
1$, luego $\Omega = (1, 1)$, y $f$ \emph{es} la rotación de centro
$(1, 1)$ y ángulo $\frac\pi2$. La lección general: cuando $1 \notin
\operatorname{Sp}\vec f$, clasificar $f$ cuesta un sistema lineal; la
geometría está enteramente en la parte lineal, y la aritmética,
enteramente en localizar el centro.
\end{example}

\begin{remark}[Dónde se usa el lenguaje afín a continuación]
Los \omterm{def:b2:affine:barycenter}{baricentros} y las \omterm{def:b2:affine:subspace}{aplicaciones afines} son la gramática de los
capítulos de geometría que vienen: las rectas y los planos tangentes
son objetos \omterm{def:b2:affine:subspace}{afines} (\cref{ch:b2:curves,ch:b2:surfaces}), un cambio de
variables \omterm{def:b2:affine:subspace}{afín} multiplica áreas y volúmenes por $\abs{\det \vec f}$
(\cref{ch:b2:multint}), y la esperanza es un \omterm{def:b2:affine:barycenter}{baricentro} con pesos
dados por una ley de probabilidad, razón por la cual la convexidad
gobierna la desigualdad de Jensen (\cref{ch:b2:randomvar}). En el
volumen del tercer año, el mismo vocabulario de convexidad sostiene
el estudio de las \omterm{def:b2:nvs:norm}{normas} $L^p$ y de las desigualdades integrales.
\end{remark}

\begin{remark}[Perspectivas dentro de este volumen]
De este capítulo salen dos hilos. El hilo \emph{\omterm{def:b2:affine:subspace}{afín}}: las rectas
tangentes (\cref{ch:b2:curves}) y los planos tangentes
(\cref{ch:b2:surfaces}) son subespacios \omterm{def:b2:affine:subspace}{afines} adosados a objetos no
lineales, y la clasificación de las cuádricas del capítulo de
superficies funciona con la ecuación del centro $A\Omega = -b$ de
este capítulo. El hilo \emph{\omterm{def:b2:affine:convex}{convexo}} es más largo: la convexidad de
los semiplanos y de los discos impulsa la teoría de Helly del
problema de fin de semana; la convexidad de funciones da la
desigualdad de Jensen (\cref{ch:b2:randomvar}); y el teorema final
del libro ---el criterio de extinción de los procesos de ramificación
(\cref{ch:b2:genfun})--- lo decide la posición de una curva convexa
respecto de la diagonal, una imagen que pertenece tanto a este
capítulo como a la probabilidad. Los \omterm{def:b2:affine:barycenter}{baricentros} también vuelven
allí: una esperanza es un \omterm{def:b2:affine:barycenter}{baricentro} con pesos de probabilidad.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:affine:1}
En $\R^3$, ¿son subespacios \omterm{def:b2:affine:subspace}{afines} los siguientes conjuntos? Da sus
direcciones y dimensiones. $\{x + y + z = 1\}$; $\;\{x + y + z = 1,\
x - z = 3\}$; $\;\{x^2 + y^2 = 1\}$; y el conjunto de soluciones de
$MX = B$ para un sistema compatible dado.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:affine:2}
Demuestra que tres puntos distintos $A, B, C$ de un \omterm{def:b2:affine:def}{espacio afín} están
alineados si y solo si $C$ es \omterm{def:b2:affine:barycenter}{baricentro} de $A$ y $B$, y si y solo si
los vectores $\vect{AB}, \vect{AC}$ son ligados. Deduce la
contabilidad de pesos al estilo de Menelao: si $C =
\operatorname{bar}(A, 1 - t; B, t)$, localiza $C$ para $t =
\frac12$, $t = 2$ y $t = -1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:affine:3}
Sea $f$ la \omterm{def:b2:affine:subspace}{aplicación afín} de $\R^2$ dada por $f(X) = MX + C$ con $M
= \frac12\begin{pmatrix} 1 & 1\\ 1 & 1\end{pmatrix}$ y $C = (1,
0)^{\mathsf T}$. Determina la imagen de $f$, sus puntos fijos (si los
hay) y $f \circ f$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:4}
(La asociatividad en acción) En un triángulo $ABC$, sean $I, J, K$
los puntos que dividen $BC$, $CA$, $AB$ en las razones $\vect{BI} =
\frac13\vect{BC}$, $\vect{CJ} = \frac13\vect{CA}$, $\vect{AK} =
\frac13\vect{AB}$. Expresa $I, J, K$ como \omterm{def:b2:affine:barycenter}{baricentros} y calcula el
\omterm{def:b2:affine:barycenter}{baricentro} de $(I,1;J,1;K,1)$: ¿qué encuentras y por qué era
previsible?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:5}
Demuestra que una aplicación $f \colon \R^n \to \R^n$ que conserve
los \emph{puntos medios} ($f\bigl(\frac{A+B}{2}\bigr) = \frac{f(A) +
f(B)}{2}$) y sea \emph{\omterm{def:b2:metric:continuity}{continua}} es \omterm{def:b2:affine:subspace}{afín}. \emph{(Prueba que la
aplicación vectorial $u \mapsto f(O + u) - f(O)$ es aditiva mediante
los puntos medios, después $\Q$-homogénea y luego $\R$-homogénea por
\omterm{def:b2:metric:continuity}{continuidad}: la misma estrategia de densidad usada para la ecuación
funcional de Cauchy en el volumen del primer año; vuelve a deducir
aquí los pasos necesarios.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:6}
Clasifica la \omterm{def:b2:affine:subspace}{aplicación afín} $f(x, y) = (y + 1,\; x - 1)$ del plano
euclídeo: parte lineal, puntos fijos y naturaleza geométrica (¿simetría
axial?, ¿con deslizamiento?). Calcula $f \circ f$ y concluye.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:affine:7}
(Radon) Sean $A_1, \dots, A_{n+2}$ puntos de un \omterm{def:b2:affine:def}{espacio afín} de
dimensión $n$. Demuestra que pueden repartirse en dos grupos
disjuntos cuyas \omterm{def:b2:affine:convex}{envolturas convexas} se cortan. \emph{(Como en la
demostración de Carathéodory, halla pesos $\mu_i$, no todos nulos,
con $\sum\mu_i = 0$ y $\sum\mu_i\vect{OA_i} = 0$; separa los pesos
positivos de los negativos y normaliza ambos lados.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:affine:8}
Sea $f$ un endomorfismo \omterm{def:b2:affine:subspace}{afín} de $\R^n$ con $f \circ f = f$.
Demuestra que $f$ es la proyección \omterm{def:b2:affine:subspace}{afín} sobre el subespacio \omterm{def:b2:affine:subspace}{afín}
$\operatorname{Fix}(f) = \operatorname{im} f$ en la dirección
$\ker\vec f$, y que, recíprocamente, todas esas proyecciones son
idempotentes. \emph{(Prueba primero que $\operatorname{im} f$ consta
de puntos fijos.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:affine:9}
Sea $G = \operatorname{bar}(A, 1;\ B, 2;\ C, 3)$ en un triángulo
$ABC$. Usando la asociatividad, prueba que la recta $AG$ corta a $BC$
en $M = \operatorname{bar}(B, 2;\ C, 3)$ y localiza $G$ sobre el
segmento $\intcc AM$; localiza igualmente la intersección de $BG$ con
$CA$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:10}
Para $\lambda \neq 0$, la \emph{homotecia} $h_{\Omega, \lambda}$ es
la \omterm{def:b2:affine:subspace}{aplicación afín} que fija $\Omega$ y tiene parte lineal
$\lambda\,\mathrm{id}$. Demuestra que la composición $h_{\Omega',
\mu} \circ h_{\Omega, \lambda}$ es una homotecia de razón
$\lambda\mu$ cuando $\lambda\mu \neq 1$, y una traslación cuando
$\lambda\mu = 1$; en el caso $\lambda = \mu = -1$ (dos simetrías
centrales), calcula el vector de traslación.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:affine:11}
(Menelao) En un triángulo $ABC$, sean $A' \in (BC)$, $B' \in (CA)$ y
$C' \in (AB)$, todos distintos de los vértices, y defínanse $\alpha,
\beta, \gamma$ por $\vect{A'B} = \alpha\,\vect{A'C}$, $\vect{B'C} =
\beta\,\vect{B'A}$, $\vect{C'A} = \gamma\,\vect{C'B}$. Demuestra que
$A', B', C'$ están alineados si y solo si $\alpha\beta\gamma = 1$.
\emph{(Escribe cada punto como \omterm{def:b2:affine:barycenter}{baricentro} de dos vértices; prueba que
tres puntos están alineados si y solo si sus filas de coordenadas
baricéntricas respecto de $(A, B, C)$ forman una matriz $3 \times 3$
singular.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:affine:12}
Demuestra que la \omterm{def:b2:affine:convex}{envoltura convexa} de un subconjunto \omterm{def:b2:metric:compact}{compacto} $K$ de
$\R^n$ es \omterm{def:b2:metric:compact}{compacta}. \emph{(Por el~\cref{thm:b2:affine:caratheodory}, $\operatorname{conv}(K)$ es la
imagen de un conjunto \omterm{def:b2:metric:compact}{compacto} por una aplicación \omterm{def:b2:metric:continuity}{continua}.)} Prueba
con un ejemplo en $\R^2$ que la \omterm{def:b2:affine:convex}{envoltura convexa} de un conjunto
\emph{cerrado} no tiene por qué ser cerrada.
\end{exercise}

\begin{problem}[{Problema de fin de semana --- El teorema de Helly y
dos de sus dividendos}]\label{pb:b2:affine:1}
El lema de Radon (\cref{exo:b2:affine:7}) dice que $n + 2$ puntos de
un \omterm{def:b2:affine:def}{espacio afín} de dimensión $n$ siempre se reparten en dos grupos con
\omterm{def:b2:affine:convex}{envolturas convexas} que se cortan. Este problema convierte ese único
hecho de álgebra lineal en una cadena de teoremas de geometría
combinatoria: el teorema de intersección de Helly, el teorema del
punto central (una mediana bidimensional) y el teorema de
recubrimiento de Jung. En todo el problema, el plano es $\R^2$ con su
estructura euclídea habitual, y $\det$ es el \omterm{def:b2:linalg:det}{determinante} en la base
canónica.\index{teorema de Helly}\index{lema de
Radon}\index{teorema de Jung}\index{punto central}

\medskip
\noindent\textbf{Parte I --- Coordenadas baricéntricas.} Unos puntos
$A_0, \dots, A_k$ son \emph{afínmente independientes} cuando los
vectores $\vect{A_0A_1}, \dots, \vect{A_0A_k}$ son linealmente
independientes.
\begin{enumerate}
  \item Prueba que la independencia \omterm{def:b2:affine:subspace}{afín} no depende de la elección
        del punto base $A_0$, y que equivale a lo siguiente: siempre
        que dos familias de pesos, cada una de suma $1$, definan el
        mismo \omterm{def:b2:affine:barycenter}{baricentro} de $(A_0, \dots, A_k)$, los pesos
        coinciden.
  \item Sea $(A, B, C)$ afínmente independiente en el plano. Prueba
        que todo punto $M$ admite una única terna $(\alpha, \beta,
        \gamma)$ con $\alpha + \beta + \gamma = 1$ y $M =
        \operatorname{bar}(A, \alpha; B, \beta; C, \gamma)$: sus
        \emph{coordenadas baricéntricas}.
  \item Demuestra las fórmulas con \omterm{def:b2:linalg:det}{determinantes}
        \[
        \alpha = \frac{\det(\vect{MB},
        \vect{MC})}{\det(\vect{AB}, \vect{AC})},
        \qquad
        \beta = \frac{\det(\vect{MC},
        \vect{MA})}{\det(\vect{AB}, \vect{AC})},
        \qquad
        \gamma = \frac{\det(\vect{MA},
        \vect{MB})}{\det(\vect{AB}, \vect{AC})} :
        \]
        las coordenadas baricéntricas son cocientes de áreas con
        signo.
  \item Las rectas $BC$, $CA$, $AB$ son las rectas coordenadas
        $\{\alpha = 0\}$, $\{\beta = 0\}$, $\{\gamma = 0\}$. Prueba
        que $M$ está en el triángulo cerrado $\operatorname{conv}\{A,
        B, C\}$ si y solo si $\alpha, \beta, \gamma \geq 0$, y que
        las tres rectas cortan el plano en exactamente siete
        regiones, clasificadas por los signos de $(\alpha, \beta,
        \gamma)$ (siendo imposible el patrón de signos $(-, -, -)$).
  \item Sea $u \colon \R^2 \to \R$ una \omterm{def:b2:affine:subspace}{aplicación afín} (una
        \emph{forma \omterm{def:b2:affine:subspace}{afín}}). Prueba que $u(M) = \alpha\,u(A) +
        \beta\,u(B) + \gamma\,u(C)$, que los conjuntos de nivel de
        una forma \omterm{def:b2:affine:subspace}{afín} no constante son rectas, que toda recta se
        obtiene así, y que los semiplanos cerrados $\{u \geq c\}$ son
        \omterm{def:b2:affine:convex}{convexos}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Particiones de Radon, afinadas.} Una
familia de $n + 2$ puntos de $\R^n$ está en \emph{posición general}
cuando cualesquiera $n + 1$ de ellos son afínmente independientes.
Una \emph{dependencia \omterm{def:b2:affine:subspace}{afín}} de $(A_1, \dots, A_{n+2})$ es una familia
$(\mu_i)$ con $\sum_i \mu_i = 0$ y $\sum_i \mu_i\,\vect{OA_i} = 0$
para un origen $O$ (y por tanto para todos).
\begin{enumerate}[resume]
  \item Calcula una dependencia \omterm{def:b2:affine:subspace}{afín} no nula de los cuatro puntos
        $A_1 = (0,0)$, $A_2 = (3,0)$, $A_3 = (0,3)$, $A_4 = (1,1)$;
        da la partición de Radon y el punto de Radon.
  \item Prueba que, para puntos en posición general, el espacio
        vectorial de las dependencias \omterm{def:b2:affine:subspace}{afines} tiene dimensión
        exactamente $1$, y que una dependencia no nula \emph{no}
        tiene ningún coeficiente nulo.
  \item Deduce que la partición de Radon de $n + 2$ puntos en
        posición general es única (salvo intercambio de los dos
        bloques), siendo cada bloque el conjunto de índices donde
        $\mu_i$ tiene un signo fijo.
  \item Para cuatro puntos del plano en posición general, prueba la
        dicotomía: o bien la partición es de tipo $(1, 3)$ ---un
        punto interior al triángulo de los otros tres---, o bien es
        de tipo $(2, 2)$: los cuatro puntos están en posición
        \omterm{def:b2:affine:convex}{convexa} y los segmentos que unen las dos parejas (las
        diagonales) se cortan en el punto de Radon.
  \item Realiza la pregunta 6 para el cuadrado unidad $(0,0)$,
        $(1,0)$, $(1,1)$, $(0,1)$: dependencia, partición y punto de
        Radon.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- El teorema de Helly en el plano.}
\begin{enumerate}[resume]
  \item Sean $C_1, C_2, C_3, C_4$ subconjuntos \omterm{def:b2:affine:convex}{convexos} de $\R^2$
        tales que cualesquiera tres de ellos tengan un punto común.
        Tómese $x_i \in \bigcap_{j \neq i} C_j$ y aplíquese el lema
        de Radon a $x_1, \dots, x_4$: prueba que el punto de Radon
        pertenece a los cuatro conjuntos. \emph{(Para cada $k$, el
        bloque que no contiene $x_k$ consta de puntos de $C_k$.)}
  \item (Helly) Sean $C_1, \dots, C_m$ ($m \geq 3$) subconjuntos
        \omterm{def:b2:affine:convex}{convexos} de $\R^2$ tales que cualesquiera tres se corten.
        Demuestra que $\bigcap_{i=1}^m C_i \neq \emptyset$, por
        inducción sobre $m$: sustitúyanse $C_{m-1}$ y $C_m$ por
        $C_{m-1} \cap C_m$ y compruébese la hipótesis para la nueva
        familia usando la pregunta 11.
  \item Tres contraejemplos, uno por hipótesis: (a) los tres lados
        cerrados de un triángulo se cortan dos a dos, pero no tienen
        punto común (el $3$ no puede rebajarse a $2$); (b) los cuatro
        conjuntos $S_i = \{x_1, \dots, x_4\} \setminus \{x_i\}$,
        para cuatro puntos en posición general, cumplen la hipótesis
        de intersección triple pero no la conclusión (la convexidad
        importa); (c) los semiplanos cerrados $H_k =
        \intco{k}{+\infty} \times \R$, $k \in \N$, se cortan dos a
        dos y tres a tres, pero $\bigcap_k H_k = \emptyset$ (las
        familias infinitas necesitan \omterm{def:b2:metric:compact}{compacidad}).
  \item (Helly \omterm{def:b2:metric:compact}{compacto}) Sea $(K_i)_{i \in I}$ una familia
        arbitraria de subconjuntos \omterm{def:b2:affine:convex}{convexos} \omterm{def:b2:metric:compact}{compactos} de $\R^2$
        tales que cualesquiera tres se corten. Usando la pregunta 12
        y la propiedad de Borel--Lebesgue
        (\cref{thm:b2:metric:borellebesgue}), prueba que $\bigcap_{i
        \in I} K_i \neq \emptyset$.
  \item (Primer dividendo) Sea $S$ un conjunto finito de puntos del
        plano y $r > 0$. Prueba que si cada tres puntos de $S$ caben
        en algún disco cerrado de radio $r$, entonces $S$ cabe en un
        único disco cerrado de radio $r$. \emph{(Aplica Helly a los
        discos $\overline D(p, r)$, $p \in S$.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- El teorema del punto central.} Un
\emph{punto central} de un conjunto finito $S$ de $n$ puntos del
plano es un punto $c$ (no necesariamente de $S$) tal que todo
semiplano cerrado que contenga a $c$ contiene al menos $n/3$ puntos
de $S$.
\begin{enumerate}[resume]
  \item (Dimensión $1$) Para reales $x_1 \leq \dots \leq x_n$, prueba
        que la mediana $c = x_{\lceil n/2 \rceil}$ cumple que toda
        semirrecta cerrada que contenga a $c$ contiene al menos $n/2$
        de los $x_i$.
  \item (Lema de conteo) Si $A, B, C$ son subconjuntos de $S$ con
        $\abs A, \abs B, \abs C > \tfrac{2n}3$, prueba que $A \cap B
        \cap C \neq \emptyset$.
  \item Sea $m = \floor{2n/3} + 1$ y sea $\mathcal F$ la familia
        (finita) de las \omterm{def:b2:affine:convex}{envolturas convexas} $\operatorname{conv}(T)$,
        con $T \subseteq S$ y $\abs T = m$. Prueba que cualesquiera
        tres miembros de $\mathcal F$ tienen un punto común y deduce
        de Helly un punto $c$ común a todos ellos.
  \item Demuestra que ese $c$ es un punto central de $S$: el
        \emph{teorema del punto central}. \emph{(Si un semiplano
        cerrado que pasa por $c$ contuviera menos de $n/3$ puntos, su
        complementario \omterm{def:b2:metric:topology}{abierto} contendría un conjunto $T$ de $m$
        puntos, y $\operatorname{conv}(T)$ evitaría a $c$.)}
  \item Optimalidad: sea $n = 3k$ y colóquense $k$ puntos en cada uno
        de tres discos de radio pequeño $\varepsilon$ centrados en
        los vértices de un triángulo grande. Prueba que para
        \emph{todo} punto $c$ del plano hay algún semiplano cerrado
        que contiene a $c$ y a lo sumo $n/3$ puntos de $S$, de modo
        que la constante $1/3$ no puede mejorarse. \emph{(Entre las
        tres direcciones de $c$ a los centros de los discos, dos
        forman un ángulo de a lo sumo $2\pi/3$.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- El teorema de Jung y síntesis.}
\begin{enumerate}[resume]
  \item (Lema del triángulo) Sean $P, Q, R$ tres puntos con
        distancias mutuas $\leq 1$. Prueba que caben en un disco
        cerrado de radio $1/\sqrt3$. \emph{(Si algún ángulo es $\geq
        \pi/2$, tómese el disco que tiene por diámetro el lado más
        largo, usando la fórmula de la mediana $\norm{RM}^2 =
        \tfrac12\norm{RP}^2 + \tfrac12\norm{RQ}^2 -
        \tfrac14\norm{PQ}^2$; si el triángulo es acutángulo, acótese
        el circunradio $a/(2\sin \widehat A)$ mediante su mayor
        ángulo, que está en $\intco{\pi/3}{\pi/2}$.)}
  \item (Jung) Deduce que todo subconjunto \omterm{def:b2:metric:compact}{compacto} del plano de
        diámetro $\leq 1$ está contenido en un disco cerrado de radio
        $1/\sqrt3$.
  \item Optimalidad: para el triángulo equilátero $A_1A_2A_3$ de lado
        $1$ con centroide $G$, demuestra la identidad de Leibniz
        $\sum_i \norm{\vect{OA_i}}^2 = 3\norm{\vect{OG}}^2 + \sum_i
        \norm{\vect{GA_i}}^2$ para todo punto $O$, y concluye que
        todo disco que contenga los tres vértices tiene radio $\geq
        1/\sqrt3$, con igualdad solo para el disco circunscrito.
  \item (Helly en $\R^n$) Enuncia y demuestra el teorema de Helly en
        $\R^n$: si un número finito de \omterm{def:b2:affine:convex}{conjuntos convexos} es tal que
        cualesquiera $n + 1$ de ellos se cortan, entonces todos se
        cortan. \emph{(El lema de Radon del~\cref{exo:b2:affine:7} se ocupa de $n + 2$ conjuntos; después,
        hágase inducción como en la pregunta 12.)}
  \item Síntesis. Ensambla la cadena
        \[
        \text{dependencia afín} \Rightarrow \text{Radon}
        \Rightarrow \text{Helly} \Rightarrow
        \text{punto central y Jung},
        \]
        indicando en una frase cada cosa: dónde entra el álgebra
        lineal, dónde entran los signos de los pesos, dónde entra la
        convexidad y qué único paso usó la dimensión del plano. ¿En
        qué se convierten las constantes $3$ (en Helly), $1/3$ (punto
        central) y $1/\sqrt3$ (Jung) en $\R^n$? (Enúncialo sin
        demostración.)
\end{enumerate}
\end{problem}
