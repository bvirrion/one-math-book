\chapter{Integrales curvilíneas e integrales múltiples}\label{ch:b2:multint}

Este capítulo extiende la integración de los intervalos a las curvas y
a los dominios del plano y del espacio. Las \omterm{def:b2:multint:lineint}{integrales curvilíneas}
integran una \emph{\omterm{def:b2:multint:lineint}{forma diferencial}} $P\,\dd x + Q\,\dd y$ a lo largo
de un arco orientado; las integrales dobles y triples integran
funciones sobre dominios de dimensión dos y tres. Las dos teorías se
encuentran en el \emph{teorema de Green--Riemann}, el teorema
fundamental del cálculo en dimensión dos, y la principal herramienta de
cálculo en todo el capítulo es la \emph{fórmula del cambio de
variables}, cuyo factor de distorsión es el valor absoluto del
\omterm{def:b2:linalg:det}{determinante} jacobiano.

\section{Integrales curvilíneas}

\begin{definition}[Forma diferencial; integral curvilínea]\label{def:b2:multint:lineint}
Sea $U \subseteq \R^2$ un \omterm{def:b2:metric:topology}{abierto}. Una \emph{forma
diferencial}\index{forma diferencial} de grado $1$ y clase
$\mathcal{C}^0$ sobre $U$ es una expresión $\omega = P\,\dd x +
Q\,\dd y$ con $P, Q \colon U \to \R$ \omterm{def:b2:metric:continuity}{continuas}; formalmente, una
aplicación \omterm{def:b2:metric:continuity}{continua} de $U$ en el \omterm{def:b2:linalg:dual}{dual} de $\R^2$, $\omega(M) =
P(M)\,e_1^* + Q(M)\,e_2^*$. Para un arco $\mathcal{C}^1$ $\gamma
\colon [a, b] \to U$, $\gamma(t) = (x(t), y(t))$, la \emph{integral
curvilínea}\index{integral curvilínea} de $\omega$ a lo largo de
$\gamma$ es
\[
\int_\gamma \omega
= \int_a^b \Bigl(P(\gamma(t))\,x'(t)
+ Q(\gamma(t))\,y'(t)\Bigr)\,\dd t .
\]
Las definiciones se extienden palabra por palabra a $\R^3$ (formas
$P\,\dd x + Q\,\dd y + R\,\dd z$) y a los arcos $\mathcal{C}^1$ a
trozos (sumando sobre los trozos).
\end{definition}

\begin{proposition}[Invariancia y orientación]\label{prop:b2:multint:lineinv}
La \omterm{def:b2:multint:lineint}{integral curvilínea} no cambia bajo un \omterm{def:b2:curves:reparam}{cambio de parámetro}
$\mathcal{C}^1$ creciente, y cambia de signo bajo uno decreciente.
Por tanto, solo depende del \omterm{def:b2:curves:reparam}{arco geométrico} \emph{orientado}.
\end{proposition}

\begin{proof}
Si $\theta \colon [c, d] \to [a, b]$ es un \omterm{def:b2:curves:reparam}{cambio de parámetro} y
$\tilde\gamma = \gamma \circ \theta$, entonces, por la regla de la
cadena y el cambio de variables de una variable $t = \theta(u)$,
\[
\int_{\tilde\gamma}\omega
= \int_c^d \bigl(P(\gamma(\theta(u)))\,x'(\theta(u))
+ Q(\gamma(\theta(u)))\,y'(\theta(u))\bigr)\,\theta'(u)\,\dd u
= \pm\int_a^b \bigl(Px' + Qy'\bigr)(t)\,\dd t ,
\]
con signo $+$ si $\theta$ es creciente ($\theta(c) = a$) y $-$ si es
decreciente (los límites se intercambian).
\end{proof}

\begin{example}[Trabajo de una fuerza; circulación]\label{ex:b2:multint:work}
Si $F = (P, Q)$ es un campo de fuerzas, $\int_\gamma P\dd x + Q\dd y
= \int_a^b \langle F(\gamma(t)), \gamma'(t)\rangle\,\dd t$ es el
\emph{trabajo} de $F$ a lo largo de $\gamma$. Para $\omega = -y\,\dd
x + x\,\dd y$ a lo largo de la circunferencia unidad recorrida en
sentido antihorario $\gamma(t) = (\cos t, \sin t)$:
\[
\int_\gamma \omega
= \int_0^{2\pi}\bigl((-\sin t)(-\sin t)
+ \cos t\cos t\bigr)\,\dd t = 2\pi ,
\]
el doble del \omterm{def:b2:multint:domain}{área} encerrada: un primer indicio de Green--Riemann.
\end{example}

\begin{example}[Una integral, dos parametrizaciones, una trampa de
signo]\label{ex:b2:multint:semicircle}
Calculemos $\int_\gamma x\,\dd y$ a lo largo de la semicircunferencia
unidad superior de $(1, 0)$ a $(-1, 0)$. Con $\gamma(t) = (\cos t,
\sin t)$, $t \in \intcc0\pi$:
\[
\int_0^\pi\cos t\cdot\cos t\,\dd t = \frac\pi2 .
\]
Con la parametrización por el grafo $x \mapsto (x, \sqrt{1 - x^2})$,
con $x$ de $1$ a $-1$ (¡atención al sentido!):
\[
\int_1^{-1}x\cdot\frac{-x}{\sqrt{1 - x^2}}\,\dd x
= \int_{-1}^{1}\frac{x^2}{\sqrt{1 - x^2}}\,\dd x
= \frac\pi2
\]
($x = \sin u$ la reduce a una \omterm{pb:b2:multint:1}{integral de Wallis}). El mismo valor,
como garantiza la~\cref{prop:b2:multint:lineinv}; pero solo porque
ambos recorridos van de $(1,0)$ a $(-1,0)$: invertir el sentido
cambia el signo. Cerrar el camino a lo largo del eje $x$ (donde $\dd
y = 0$) no añade nada, y el total $\frac\pi2$ es el \omterm{def:b2:multint:domain}{área} del
semidisco: el primer caso de las fórmulas de
\omterm{def:b2:multint:domain}{área} por la frontera de Green--Riemann de más abajo.
\end{example}

\begin{definition}[Formas exactas y cerradas]\label{def:b2:multint:exact}
La forma $\omega = P\,\dd x + Q\,\dd y$ de clase $\mathcal{C}^0$ es
\emph{exacta}\index{forma exacta} sobre $U$ si existe $f \in
\mathcal{C}^1(U)$ (un \emph{potencial}\index{potencial}) con $\omega
= \dd f$, es decir, $P = f_x$ y $Q = f_y$. Una forma $\mathcal{C}^1$
es \emph{cerrada}\index{forma cerrada} si $P_y = Q_x$ sobre $U$.
\end{definition}

\begin{theorem}[Teorema fundamental para las integrales
curvilíneas]\label{thm:b2:multint:ftc}
Si $\omega = \dd f$ es \omterm{def:b2:multint:exact}{exacta} y $\gamma$ es un arco $\mathcal{C}^1$ a
trozos en $U$ que va de $A$ a $B$, entonces
\[
\int_\gamma \omega = f(B) - f(A) .
\]
En particular, la integral de una \omterm{def:b2:multint:exact}{forma exacta} a lo largo de
cualquier arco cerrado es nula, y toda forma $\mathcal{C}^1$
\omterm{def:b2:multint:exact}{exacta} es cerrada.
\end{theorem}

\begin{proof}
$\frac{\dd}{\dd t}f(\gamma(t)) = f_x(\gamma(t))x'(t) +
f_y(\gamma(t))y'(t)$ por la regla de la cadena
(\cref{ch:b2:diffcalc}), de modo que el integrando de la~\cref{def:b2:multint:lineint} es la derivada de $t \mapsto
f(\gamma(t))$, y el teorema fundamental del cálculo da el resultado
en cada trozo; los valores intermedios telescopian. Que las formas
$\mathcal{C}^1$ \omterm{def:b2:multint:exact}{exactas} sean cerradas es el teorema de Schwarz: $P_y
= f_{xy} = f_{yx} = Q_x$.
\end{proof}

\begin{example}[Reconstruir un potencial]\label{ex:b2:multint:potential}
Sea $\omega = y\,\eu^{xy}\,\dd x + (x\,\eu^{xy} + 2y)\,\dd y$ sobre
$\R^2$. Es cerrada: ambas derivadas cruzadas valen $\eu^{xy}(1 +
xy)$. Para hallar un \omterm{def:b2:multint:exact}{potencial}, intégrese $P$ en $x$ con $y$ fijo:
\[
f(x, y) = \int y\,\eu^{xy}\,\dd x = \eu^{xy} + c(y),
\]
y ajústese después $c$ igualando $f_y$: $x\,\eu^{xy} + c'(y) =
x\,\eu^{xy} + 2y$ da $c(y) = y^2$. Así pues, $f(x,y) = \eu^{xy} +
y^2$, y para todo arco $\mathcal C^1$ a trozos de $(0,0)$ a $(1,1)$,
\[
\int_\gamma\omega = f(1,1) - f(0,0) = (\eu + 1) - 1 = \eu ,
\]
independientemente del camino; la receta en dos pasos (integrar en
$x$, corregir en $y$) es el recíproco práctico del~\cref{thm:b2:multint:ftc} sobre los dominios donde las
\omterm{def:b2:multint:exact}{formas cerradas} son \omterm{def:b2:multint:exact}{exactas}.
\end{example}

\begin{example}[Cerrada no implica exacta]\label{ex:b2:multint:angleform}
Sobre $U = \R^2 \setminus \{0\}$, la \emph{forma ángulo}
\[
\omega = \frac{-y\,\dd x + x\,\dd y}{x^2 + y^2}
\]
es cerrada (cálculo directo: tanto $P_y$ como $Q_x$ valen $\frac{y^2
- x^2}{(x^2+y^2)^2}$), pero su integral a lo largo de la
circunferencia unidad vale $2\pi \neq 0$ (el mismo cálculo del~\cref{ex:b2:multint:work}, dividido por $1$): $\omega$ no es
\omterm{def:b2:multint:exact}{exacta} sobre $U$. Localmente, $\omega = \dd\theta$ para una
determinación $\theta$ del ángulo polar; el fallo es global: el
ángulo no puede definirse de manera \omterm{def:b2:metric:continuity}{continua} alrededor del pinchazo.
Sobre dominios sin agujeros la patología desaparece: en un \omterm{def:b2:metric:topology}{abierto}
\emph{estrellado}, toda forma $\mathcal{C}^1$ cerrada es \omterm{def:b2:multint:exact}{exacta}
(lema de Poincaré, \cref{exo:b2:multint:8}).
\end{example}

\section{Integrales dobles}

Damos por sabida la teoría de la integral de Riemann en una variable
(volumen del primer año, y \cref{ch:b2:integration}) y esbozamos su
versión de dos variables. Una función $f$ \omterm{def:b2:metric:continuity}{continua} sobre un rectángulo
$R = [a, b] \times [c, d]$ tiene una integral doble $\iint_R f$,
definida por sumas de Riemann sobre retículas exactamente igual que en
una variable, y calculada por iteración:

\begin{theorem}[Fubini sobre un rectángulo]\label{thm:b2:multint:fubini}
Para $f$ \omterm{def:b2:metric:continuity}{continua} sobre $R = [a,b] \times [c,d]$,
\[
\iint_R f
= \int_a^b \Bigl(\int_c^d f(x, y)\,\dd y\Bigr)\dd x
= \int_c^d \Bigl(\int_a^b f(x, y)\,\dd x\Bigr)\dd y .
\]
\end{theorem}

\begin{proof}
Póngase $F(x) = \int_c^d f(x, y)\,\dd y$. La \omterm{def:b2:metric:continuity}{continuidad} uniforme de
$f$ sobre el \omterm{def:b2:metric:compact}{compacto} $R$ hace \omterm{def:b2:metric:continuity}{continua} a $F$ (estimación dominada:
$\abs{F(x) - F(x')} \leq (d - c)\sup_y\abs{f(x,y) - f(x',y)}$).
Subdivídanse ahora $[a,b]$ y $[c,d]$ en $n$ partes iguales, lo que da
una retícula de celdas $R_{ij}$ de \omterm{def:b2:multint:domain}{área} $\Delta x\,\Delta y$. En cada
celda, $\inf_{R_{ij}} f \cdot \Delta x \Delta y \leq
\int_{x_{i-1}}^{x_i}\int_{y_{j-1}}^{y_j} f(x,y)\,\dd y\,\dd x \leq
\sup_{R_{ij}} f \cdot \Delta x \Delta y$ por la monotonía de la
integral de una variable (aplicada dos veces). Sumando sobre las
celdas, la integral iterada $\int_a^b F$ queda encajada entre las
sumas de Riemann inferior y superior de la retícula; y por la
\omterm{def:b2:metric:continuity}{continuidad} uniforme, ambas sumas convergen al valor común que define
$\iint_R f$ cuando $n \to \infty$. El mismo argumento se aplica con
los papeles de $x$ e $y$ intercambiados, así que ambas integrales
iteradas valen $\iint_R f$.
\end{proof}

\begin{remark}
La \omterm{def:b2:metric:continuity}{continuidad} sobre un rectángulo \omterm{def:b2:metric:compact}{compacto} hace un trabajo real en la
demostración de Fubini: proporciona la \omterm{def:b2:metric:continuity}{continuidad} uniforme que encaja
las sumas de Riemann. Para integrandos más salvajes, el enunciado
falla de verdad: hay funciones cuyas dos integrales iteradas existen
y difieren. El teorema general honesto, con la integrabilidad como
única hipótesis, es el teorema de Fubini para la integral de
Lebesgue, demostrado en el volumen del tercer año; todo lo de este
capítulo se queda en el marco \omterm{def:b2:metric:continuity}{continuo}, donde la demostración
elemental anterior es completa.
\end{remark}

\begin{definition}[Dominios elementales]\label{def:b2:multint:domain}
Un dominio $D \subseteq \R^2$ es \emph{$y$-elemental} si
\[
D = \{(x, y) : a \leq x \leq b,\
\varphi_1(x) \leq y \leq \varphi_2(x)\}
\]
con $\varphi_1 \leq \varphi_2$ \omterm{def:b2:metric:continuity}{continuas} sobre $[a,b]$
($x$-elemental: simétricamente). Para $f$ \omterm{def:b2:metric:continuity}{continua} sobre un $D$
$y$-elemental,
\[
\iint_D f
= \int_a^b\Bigl(
\int_{\varphi_1(x)}^{\varphi_2(x)} f(x,y)\,\dd y\Bigr)\dd x ,
\]
y se comprueba (extendiendo $f$ por un argumento de aproximación, o
subdividiendo) que cuando $D$ es elemental en ambas direcciones las
dos integrales iteradas coinciden. Los dominios que se cortan en un
número finito de trozos elementales se tratan por aditividad, y el
\emph{área}\index{área!de un dominio plano} de $D$ es
$\operatorname{Area}(D) = \iint_D 1$.
\end{definition}

\begin{example}\label{ex:b2:multint:triangle}
Sobre el triángulo $D = \{0 \leq x \leq 1,\ 0 \leq y \leq x\}$:
\[
\iint_D xy \,\dd x\,\dd y
= \int_0^1 x\Bigl(\int_0^x y\,\dd y\Bigr)\dd x
= \int_0^1 x\cdot\frac{x^2}{2}\,\dd x = \frac18 .
\]
Intercambiando el orden ($x$ de $y$ a $1$): $\int_0^1
y\bigl(\int_y^1 x\,\dd x\bigr)\dd y = \int_0^1 y\,\frac{1 -
y^2}{2}\,\dd y = \frac18$; el mismo valor, otro cálculo: elegir bien
el orden de integración es la mitad del oficio.
\end{example}

\begin{example}[Cuando solo funciona un orden]\label{ex:b2:multint:swaporder}
Calculemos $I = \displaystyle\int_0^1\!\!\int_x^1
\eu^{y^2}\,\dd y\,\dd x$. Tal como está escrita, la integral interior
$\int\eu^{y^2}\dd y$ no tiene primitiva elemental: el cálculo se
atasca. Pero el dominio es el triángulo $0 \leq x \leq y \leq 1$, que
es elemental en ambas direcciones; intercambiando el orden,
\[
I = \int_0^1\!\!\int_0^y \eu^{y^2}\,\dd x\,\dd y
= \int_0^1 y\,\eu^{y^2}\,\dd y
= \Bigl[\tfrac12\eu^{y^2}\Bigr]_0^1 = \frac{\eu - 1}{2} .
\]
La variable interior $x$ no aparecía en ninguna parte del integrando,
así que integrarla primero produjo exactamente el factor $y$ que hace
inmediata la integral exterior. Moraleja: Fubini no es solo una
licencia para iterar, sino una licencia para \emph{elegir}, y el
orden correcto puede convertir una integral imposible en una línea.
Dibújese siempre el dominio y léanse sus dos descripciones antes de
empezar.
\end{example}

\begin{theorem}[Cambio de variables]\label{thm:b2:multint:changevar}
Sea $\Phi \colon U' \to U$ un difeomorfismo $\mathcal{C}^1$ entre
\omterm{def:b2:metric:topology}{abiertos} de $\R^2$, sea $K \subseteq U$ un dominio \omterm{def:b2:metric:compact}{compacto}
descompuesto en trozos elementales con $K' = \Phi^{-1}(K)$, y sea $f$
\omterm{def:b2:metric:continuity}{continua} sobre $K$. Entonces
\[
\iint_K f(x, y)\,\dd x\,\dd y
= \iint_{K'} f\bigl(\Phi(u, v)\bigr)\,
\abs{\det J_\Phi(u, v)}\,\dd u\,\dd v .
\]
\end{theorem}
\admitted

\begin{remark}
La demostración completa ---aproximar $\Phi$ por su \omterm{def:b2:diffcalc:differential}{diferencial} sobre
una retícula fina y controlar las celdas de la frontera--- es larga
aunque no profunda; se hace por entero en la teoría de la medida del
tercer año, como consecuencia de la teoría de Lebesgue. La heurística
es la imagen ya usada para el \omterm{def:b2:multint:domain}{área} de una superficie: un cuadradito
de lado $\dd u$ en $(u, v)$ se aplica, a primer orden, sobre el
paralelogramo \omterm{def:b2:structures:generated}{generado} por $\Phi_u\,\dd u$ y $\Phi_v\,\dd v$, cuya
\omterm{def:b2:multint:domain}{área} es $\abs{\det J_\Phi}\,\dd u\,\dd v$
(\cref{lem:b2:surfaces:lagrange}).
\end{remark}

\begin{remark}[Método: elegir el cambio de variables]
Tres reflejos cubren la mayoría de los casos. \emph{Simetría del
integrando}: $x^2 + y^2$ pide coordenadas polares, y una estructura
de producto pide conservar los ejes cartesianos. \emph{Forma de la
frontera}: unas fronteras $u(x,y) = c_1$, $v(x,y) = c_2$ están
suplicando las propias coordenadas $(u, v)$, como en el ejemplo de la
región hiperbólica de más abajo: el dominio se convierte en un
rectángulo, que es toda la victoria. \emph{Estructura lineal}: las
expresiones en $x + y$ y $x - y$ invitan a la rotación de $45$ grados
o a una cizalla (\cref{ex:b2:multint:affine}). En todos los casos,
tres casillas que marcar antes de integrar: que la aplicación sea una
biyección del dominio nuevo sobre el viejo; que su jacobiano se
calcule \emph{en el sentido que realmente se usa} (invirtiendo al
final si resulta más fácil); y que el jacobiano entre con su valor
absoluto.
\end{remark}

\begin{example}[Coordenadas polares]\label{ex:b2:multint:polar}
$\Phi(\rho, \alpha) = (\rho\cos\alpha,\ \rho\sin\alpha)$ tiene
\[
J_\Phi = \begin{pmatrix}
\cos\alpha & -\rho\sin\alpha\\
\sin\alpha & \rho\cos\alpha
\end{pmatrix},
\qquad \det J_\Phi = \rho ,
\]
de modo que $\dd x\,\dd y = \rho\,\dd\rho\,\dd\alpha$. Para el disco
$D_R$ de radio $R$:
\[
\iint_{D_R} e^{-(x^2 + y^2)}\,\dd x\,\dd y
= \int_0^{2\pi}\!\!\int_0^R e^{-\rho^2}\rho\,\dd\rho\,\dd\alpha
= \pi\bigl(1 - e^{-R^2}\bigr)
\xrightarrow[R\to\infty]{} \pi .
\]
Comparando con el cuadrado $[-R, R]^2$ (encajado entre los discos
$D_R$ y $D_{R\sqrt2}$, con todos los integrandos positivos) resulta
$\bigl(\int_{-\infty}^\infty e^{-x^2}\dd x\bigr)^2 = \pi$:
\[
\boxed{\ \int_{-\infty}^{+\infty} e^{-x^2}\,\dd x = \sqrt{\pi}\ }
\]
la integral de Gauss otra vez, ahora con su demostración más famosa
(compárese con la deducción de una variable del~\cref{ch:b2:integration}).
\end{example}

\begin{example}[Cambios de variables afines]\label{ex:b2:multint:affine}
Para una \omterm{def:b2:affine:subspace}{aplicación afín} $\Phi(u, v) = M(u, v)^{\mathsf T} + C$ con
$M$ invertible, el jacobiano es la matriz constante $M$: las \omterm{def:b2:multint:domain}{áreas}
quedan multiplicadas por el factor constante $\abs{\det M}$; la
promesa hecha en el~\cref{ch:b2:affine} es ya un teorema. Dos usos
inmediatos. La elipse $\frac{x^2}{a^2} + \frac{y^2}{b^2} \leq 1$ es
la imagen del disco unidad por $(u, v) \mapsto (au, bv)$, de modo que
su \omterm{def:b2:multint:domain}{área} es $ab \cdot \pi$, sin cálculo alguno. Y para la integral de
$f(x + y)$ sobre el cuadrado $K = \intcc01^2$, la cizalla $\Phi(u, v)
= (u - v, v)$ (de \omterm{def:b2:linalg:det}{determinante} $1$) la convierte en una integral de
$f(u)$ sobre un paralelogramo, que Fubini rebana a $u$ constante: con
$f = \exp$,
\[
\iint_K \eu^{x+y}\,\dd x\,\dd y
= \Bigl(\int_0^1 \eu^x\,\dd x\Bigr)^2 = (\eu - 1)^2,
\]
como confirma la estructura de producto. Elegir coordenadas adaptadas
al integrando ---y no al dominio--- es la otra mitad del oficio.
\end{example}


\begin{example}[Coordenadas adaptadas a un dominio
curvilíneo]\label{ex:b2:multint:hyperbolicregion}
Sea $D$ la región del primer cuadrante limitada por las hipérbolas
$xy = 1$ y $xy = 3$ y las rectas $y = x$ e $y = 3x$. En las
coordenadas $u = xy$, $v = y/x$ el dominio se convierte en el
cuadrado $\intcc13 \times \intcc13$; invirtiendo,
\[
x = \sqrt{u/v}, \qquad y = \sqrt{uv},
\qquad
\det J = x_uy_v - x_vy_u = \frac{1}{2v}
\]
(un cálculo de dos líneas con $x = u^{1/2}v^{-1/2}$, $y =
u^{1/2}v^{1/2}$). De ahí,
\[
\operatorname{Area}(D)
= \int_1^3\!\!\int_1^3\frac{\dd u\,\dd v}{2v}
= 2\cdot\frac{\ln 3}{2} = \ln 3 \approx 1.10 .
\]
Intentar rebanar $D$ en coordenadas cartesianas obliga a cortarlo en
tres trozos con fronteras hiperbólicas y lineales: factible,
desagradable y propenso a errores. La moraleja repite el~\cref{ex:b2:multint:affine} con toda su fuerza: léanse las ecuaciones
de la frontera y déjese que sean \emph{ellas} las que elijan las
coordenadas; el jacobiano convierte entonces el \omterm{def:b2:multint:domain}{área} de la celda de
la malla curvilínea, exactamente como hacía $\rho$ para las
coordenadas polares.
\end{example}

\begin{example}[Valores medios]\label{ex:b2:multint:average}
El \emph{valor medio} de $f$ sobre un dominio $D$ es
$\frac1{\operatorname{Area}(D)}\iint_Df$. Muestra: la distancia media
al centro para un punto elegido uniformemente en el disco de radio
$R$ es
\[
\frac{1}{\pi R^2}\int_0^{2\pi}\!\!\int_0^R
\rho\cdot\rho\,\dd\rho\,\dd\alpha
= \frac{2\pi R^3/3}{\pi R^2} = \frac{2R}3 ,
\]
y no $R/2$: el \omterm{def:b2:multint:domain}{área} uniforme pone más masa en los radios grandes (la
corona de radio $\rho$ tiene peso proporcional a $\rho$), así que la
media queda más allá de la mitad. Acertar con este factor es
exactamente el jacobiano polar en acción, y la misma ponderación
explica que el centroide de la semibola calculado más adelante en el
capítulo sea $\bar z = 3R/8$ y no $R/2$.
\end{example}

\section{El teorema de Green--Riemann}

\begin{theorem}[Green--Riemann]\label{thm:b2:multint:green}
Sea $K \subseteq \R^2$ un dominio \omterm{def:b2:metric:compact}{compacto} elemental en ambas
direcciones (o una unión finita de tales, pegados a lo largo de
segmentos), con frontera $\partial K$ una curva cerrada
$\mathcal{C}^1$ a trozos orientada en sentido \emph{antihorario} (el
dominio queda a la izquierda). Para $P, Q$ de clase $\mathcal{C}^1$
en un entorno de $K$:
\[
\oint_{\partial K} P\,\dd x + Q\,\dd y
= \iint_K \Bigl(\frac{\partial Q}{\partial x}
- \frac{\partial P}{\partial y}\Bigr)\,\dd x\,\dd y .
\]
\end{theorem}

\begin{proof}
Primero, ambos miembros son aditivos al cortar $K$ a lo largo de un
segmento en dos trozos $K_1, K_2$: las integrales dobles se suman por
la aditividad de $\iint$; y en cuanto a las integrales de frontera,
las fronteras antihorarias de $K_1$ y de $K_2$ recorren cada una el
corte interior una vez, en sentidos \emph{opuestos}, de modo que en
la suma
\[
\oint_{\partial K_1} + \oint_{\partial K_2}
= \oint_{\partial K} + (\text{el corte, en ambos sentidos})
= \oint_{\partial K},
\]
los dos pasos por el corte se cancelan
(\cref{prop:b2:multint:lineinv}) y solo sobrevive la frontera
exterior. Iterando un número finito de cortes, basta tratar un
dominio elemental. Demostramos $\oint P\,\dd x = -\iint_K P_y$ sobre
un dominio $y$-elemental $D = \{a \leq x \leq b,\ \varphi_1(x) \leq y
\leq \varphi_2(x)\}$; la identidad $\oint Q\,\dd y = \iint_K Q_x$ es
\omterm{def:b2:quadratic:adjoint}{simétrica} ($x$-elemental), y el teorema es su suma.

Calculemos la integral doble por Fubini y el teorema fundamental de
una variable:
\[
\iint_D \frac{\partial P}{\partial y}\,\dd x\,\dd y
= \int_a^b \bigl(P(x, \varphi_2(x)) - P(x, \varphi_1(x))\bigr)
\,\dd x .
\]
Ahora bien, la frontera de $D$, en sentido antihorario, consta del
grafo inferior $y = \varphi_1(x)$ recorrido de izquierda a derecha,
del segmento vertical derecho $x = b$ (hacia arriba), del grafo
superior $y = \varphi_2(x)$ recorrido \emph{de derecha a izquierda} y
del segmento vertical izquierdo $x = a$ (hacia abajo). A lo largo de
los segmentos verticales, $x$ es constante, así que aportan $0$ a
$\oint P\,\dd x$; y los grafos, parametrizados por $x$, dan
\[
\oint_{\partial D} P\,\dd x
= \int_a^b P(x, \varphi_1(x))\,\dd x
- \int_a^b P(x, \varphi_2(x))\,\dd x
= -\iint_D \frac{\partial P}{\partial y}\,\dd x\,\dd y .
\qedhere
\]
\end{proof}

\begin{corollary}[Área por la frontera]\label{cor:b2:multint:area}
Bajo las hipótesis del~\cref{thm:b2:multint:green},
\[
\operatorname{Area}(K)
= \oint_{\partial K} x\,\dd y
= -\oint_{\partial K} y\,\dd x
= \frac12\oint_{\partial K} x\,\dd y - y\,\dd x .
\]
\end{corollary}

\begin{proof}
Aplíquese Green--Riemann a $(P, Q) = (0, x)$, $(-y, 0)$ y
$\frac12(-y, x)$: cada vez, $Q_x - P_y = 1$.
\end{proof}

\begin{remark}[Elegir entre las tres fórmulas del área]
Las tres fórmulas de frontera son iguales, pero no intercambiables en
la práctica. Úsese $\oint x\,\dd y$ cuando la parametrización haga
sencillo $\dd y$ (grafos sobre el eje $y$), $-\oint y\,\dd x$
simétricamente, y la semisuma \omterm{def:b2:quadratic:adjoint}{simétrica} cuando la parametrización
trate a $x$ y a $y$ por igual: para la elipse produjo un integrando
constante, sin ninguna linealización trigonométrica. Sobre fronteras
poligonales, la semisuma se convierte en la fórmula del cordón del~\cref{exo:b2:multint:12}, el algoritmo de los agrimensores. Y cuando
la parametrización dada recorre la frontera en sentido horario, las
tres fórmulas devuelven \emph{menos} el \omterm{def:b2:multint:domain}{área}: un resultado negativo
no es un error de cálculo, sino un informe de orientación; cámbiese
el signo, o la parametrización.
\end{remark}

\begin{example}[Área de la elipse]\label{ex:b2:multint:ellipsearea}
Para $x = a\cos t$, $y = b\sin t$, $t \in [0, 2\pi]$:
\[
\operatorname{Area}
= \frac12\int_0^{2\pi}\bigl(a\cos t \cdot b\cos t
- b\sin t\cdot(-a\sin t)\bigr)\,\dd t
= \frac{ab}{2}\int_0^{2\pi}\dd t = \pi ab .
\]
\end{example}

\begin{example}[Green--Riemann como
contraste]\label{ex:b2:multint:greencheck}
Tomemos $P = -y^3$, $Q = x^3$ sobre el disco unidad cerrado $D$. Lado
de la frontera, con $\gamma(t) = (\cos t, \sin t)$:
\[
\oint_{\partial D}P\,\dd x + Q\,\dd y
= \int_0^{2\pi}\bigl(\sin^4 t + \cos^4 t\bigr)\dd t
= 2\pi\cdot\Bigl(\frac38 + \frac38\Bigr) = \frac{3\pi}2 ,
\]
por linealización ($\sin^4 + \cos^4 = \tfrac34 + \tfrac14\cos4t$).
Lado del interior:
\[
\iint_D(Q_x - P_y)\,\dd x\,\dd y
= \iint_D 3(x^2 + y^2)\,\dd x\,\dd y
= 3\int_0^{2\pi}\!\!\int_0^1\rho^3\,\dd\rho\,\dd\alpha
= \frac{3\pi}2 .
\]
El mismo número, con dos cálculos muy distintos; y ese es el uso
práctico: el lado de la identidad de Green que resulte más fácil se
convierte en el cálculo, y el otro en la verificación. Para
circulaciones de campos polinómicos alrededor de curvas cerradas, la
integral doble es casi siempre el lado fácil.
\end{example}

\begin{remark}
Green--Riemann explica el~\cref{ex:b2:multint:angleform}: para una
\omterm{def:b2:multint:exact}{forma cerrada} ($Q_x = P_y$), la integral alrededor de la frontera de
cualquier dominio contenido en $U$ se anula. La forma ángulo deja de
ser \omterm{def:b2:multint:exact}{exacta} solo porque el pinchazo del origen impide que el disco
limitado por la circunferencia unidad esté dentro de $U$: las
\omterm{def:b2:multint:lineint}{integrales curvilíneas} de \omterm{def:b2:multint:exact}{formas cerradas} detectan los agujeros del
dominio. (Llevada más lejos, esta observación se convierte en la
cohomología de De Rham.)
\end{remark}

\section{Integrales triples}

La teoría se extiende a tres variables sin ninguna idea nueva: Fubini
reduce $\iiint$ a tres integrales de una variable (bien
\emph{rebanando}: $\iiint_K f = \int\bigl(\iint_{K_z} f\bigr)\dd z$
sobre las rebanadas horizontales $K_z$, bien \emph{apilando}:
integrando primero en $z$ a lo largo de barritas verticales), y la
fórmula del cambio de variables vale con el jacobiano $3 \times 3$.

\begin{example}[Coordenadas cilíndricas y esféricas]\label{ex:b2:multint:spherical}
\emph{Cilíndricas} $(x, y, z) = (\rho\cos\alpha, \rho\sin\alpha, z)$:
$\dd x\,\dd y\,\dd z = \rho\,\dd\rho\,\dd\alpha\,\dd z$.
\emph{Esféricas} $(x, y, z) = (r\cos\theta\cos\varphi,\
r\sin\theta\cos\varphi,\ r\sin\varphi)$ ($\theta$ la \omterm{def:b2:curves:length}{longitud},
$\varphi \in [-\frac\pi2, \frac\pi2]$ la latitud): desarrollando el
\omterm{def:b2:linalg:det}{determinante} $3 \times 3$ por la última fila,
\[
\det J = r^2\cos\varphi ,
\qquad
\dd x\,\dd y\,\dd z
= r^2\cos\varphi\;\dd r\,\dd\theta\,\dd\varphi .
\]
\omterm{pb:b2:multint:1}{Volumen de la bola} de radio $R$:
\[
V = \int_0^R\!\!\int_0^{2\pi}\!\!\int_{-\pi/2}^{\pi/2}
r^2\cos\varphi\;\dd\varphi\,\dd\theta\,\dd r
= \frac{R^3}{3}\cdot 2\pi \cdot 2
= \boxed{\frac43\pi R^3} ,
\]
saldando por fin la fórmula admitida en los capítulos de volúmenes de
los libros anteriores.
\end{example}

\begin{example}[El tetraedro, dos veces]
\label{ex:b2:multint:tetra}
El volumen de $T = \{x, y, z \geq 0,\ x + y + z \leq 1\}$, apilando:
para $(x, y)$ fijo en el triángulo $x + y \leq 1$, $z$ recorre
$\intcc0{1 - x - y}$, luego
\[
V = \int_0^1\!\!\int_0^{1-x}(1 - x - y)\,\dd y\,\dd x
= \int_0^1\frac{(1 - x)^2}{2}\,\dd x = \frac16 .
\]
Rebanando: la sección a la altura $z$ es el triángulo $\{x, y \geq
0,\ x + y \leq 1 - z\}$, de \omterm{def:b2:multint:domain}{área} $\frac{(1-z)^2}2$, y $V =
\int_0^1\frac{(1-z)^2}2\,\dd z = \frac16$ de nuevo; los dos cálculos
son las mismas integrales en otro orden, que es todo lo que afirma
Fubini. El valor $\frac16 = \frac13\cdot\frac12\cdot1$ es la fórmula
del cono (\cref{ex:b2:multint:cone}) con base triangular, y la
versión $n$-dimensional $1/n!$ se demuestra exactamente con este
rebanado en el problema de fin de semana.
\end{example}

\begin{example}[Centroide de una semibola]\label{ex:b2:multint:centroid}
Para la semibola superior $H$ de radio $R$ ($z \geq 0$), la altura
del centroide es $\bar z = \frac1{V}\iiint_H z$, con $V = \frac23\pi
R^3$. En coordenadas esféricas ($z = r\sin\varphi$, $\varphi \in
\intcc0{\pi/2}$):
\[
\iiint_H z
= \int_0^R r^3\,\dd r\int_0^{2\pi}\dd\theta
\int_0^{\pi/2}\sin\varphi\cos\varphi\,\dd\varphi
= \frac{R^4}4\cdot2\pi\cdot\frac12 = \frac{\pi R^4}4 ,
\]
de modo que
\[
\bar z = \frac{\pi R^4/4}{2\pi R^3/3} = \frac{3R}8 :
\]
el punto de equilibrio de un hemisferio macizo está a tres octavos
del radio por encima de la cara plana, por debajo de la altura media
$R/2$, como debe ser, ya que el sólido es más grueso cerca de la
base. Todo cálculo de centroide tiene esta forma: una integral de
momento, un volumen, un cociente y una comprobación de
verosimilitud contra la geometría.
\end{example}

\begin{example}[Volumen por rebanado: el cono]\label{ex:b2:multint:cone}
Un cono de \omterm{def:b2:multint:domain}{área} de base $A$ y altura $h$ (vértice arriba, base en $z
= 0$): la rebanada a la altura $z$ es la base escalada por el factor
$(1 - z/h)$, de \omterm{def:b2:multint:domain}{área} $A(1 - z/h)^2$. De ahí,
\[
V = \int_0^h A\Bigl(1 - \frac zh\Bigr)^2\dd z = \frac{Ah}{3} :
\]
el tercio de las fórmulas escolares, válido para \emph{cualquier}
forma de base; el rebanado lo convierte en la integral de un
cuadrado.
\end{example}

\begin{example}[Umbrales de integrabilidad en el
plano]\label{ex:b2:multint:threshold}
¿Para qué $\alpha > 0$ converge $\iint_{D}\rho^{-\alpha}\,\dd x\,\dd
y$ sobre el disco unidad pinchado $D$ (como límite sobre coronas
$\varepsilon \leq \rho \leq 1$)? En coordenadas polares,
\[
\int_0^{2\pi}\!\!\int_\varepsilon^1\rho^{-\alpha}\,
\rho\,\dd\rho\,\dd\alpha
= 2\pi\int_\varepsilon^1\rho^{1-\alpha}\,\dd\rho ,
\]
que converge cuando $\varepsilon \to 0$ si y solo si $1 - \alpha >
-1$, es decir, $\alpha < 2$: en dimensión $2$, el exponente crítico
de singularidad es la propia dimensión, pues el $\rho$ extra del
jacobiano suaviza la singularidad en una potencia. (Igualmente,
$\alpha < 3$ para una singularidad puntual en el espacio, vía $r^2$.)
Esta contabilidad radial es la manera de decidir la integrabilidad de
un vistazo en el marco de Lebesgue del tercer año, y es la razón de
que $\iiint 1/r$ convergiera sin esfuerzo en el~\cref{exo:b2:multint:7}.
\end{example}

\begin{remark}[Errores frecuentes]
(i) \emph{Orientación}: una \omterm{def:b2:multint:lineint}{integral curvilínea} cambia de signo con
el sentido del recorrido, y Green--Riemann exige la frontera en
sentido antihorario (con el dominio a la izquierda); para un dominio
con un agujero, la frontera interior se recorre en sentido
\emph{horario}. (ii) \emph{El jacobiano entra con valor absoluto}: un
cambio de variables nunca produce \omterm{def:b2:multint:domain}{área} negativa, y olvidar
$\abs{\det}$ suele invertir los signos justo cuando la aplicación
invierte la orientación. (iii) \emph{El factor polar $\rho$}: $\dd
x\,\dd y = \rho\,\dd\rho\,\dd\alpha$, no $\dd\rho\,\dd\alpha$; el
error más frecuente de todo el capítulo, que el análisis dimensional
caza, ya que $\dd\rho\,\dd\alpha$ tiene dimensión de longitud y no de
\omterm{def:b2:multint:domain}{área}. (iv) \emph{Integrales dobles impropias}: los límites sobre
discos crecientes y sobre cuadrados crecientes coinciden aquí porque
los integrandos son positivos (encaje); con integrandos de signo
variable el límite puede depender del agotamiento, y no se afirma
nada sin convergencia absoluta. (v) \emph{Dominios frente a
integrandos}: un integrando de tipo producto sobre un dominio que no
es producto \emph{no} factoriza la integral; la factorización
necesita ambas cosas, como en el cuadrado del~\cref{ex:b2:multint:polar}.
\end{remark}

\begin{remark}[Perspectivas dentro de este volumen]
La integral de Gauss calculada aquí está calladamente en todas partes
en los capítulos de probabilidad: la constante $\sqrt\pi$ que hay
dentro de la fórmula de Stirling
(\cref{thm:b2:comparison:stirling}) es esta integral del capítulo y,
a través de Stirling, fija la asintótica $1/\sqrt{\pi n}$ de las
probabilidades de retorno del paseo aleatorio del~\cref{ch:b2:proba}. Las \omterm{pb:b2:multint:1}{integrales de Wallis} del problema de fin de
semana reaparecen también allí, impulsando las mismas estimaciones
del coeficiente binomial central. En el otro sentido, los elementos
de \omterm{def:b2:multint:domain}{área} y de volumen de este capítulo completan la geometría del~\cref{ch:b2:surfaces}, y la fórmula de Green recalcula las \omterm{def:b2:multint:domain}{áreas} de
\omterm{pb:b2:curves:1}{envolventes} del~\cref{ch:b2:curves} (la \omterm{pb:b2:curves:1}{astroide}, en el~\cref{exo:b2:multint:5}). Un capítulo, tres servicios: medida para la
geometría, constantes para la probabilidad y la disciplina del cambio
de variables usada por ambas.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:multint:1}
Calcula $\int_\gamma y^2\,\dd x + x\,\dd y$ a lo largo de: (a) el
segmento de $(0,0)$ a $(1,1)$; (b) el arco de parábola $y = x^2$ de
$(0,0)$ a $(1,1)$. ¿Es \omterm{def:b2:multint:exact}{exacta} la forma?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:multint:2}
Prueba que $\omega = (2xy + y^3)\,\dd x + (x^2 + 3xy^2 + 1)\,\dd y$
es cerrada sobre $\R^2$, halla un \omterm{def:b2:multint:exact}{potencial} y calcula
$\int_\gamma\omega$ a lo largo de un arco cualquiera de $(0, 0)$ a
$(1, 2)$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:multint:3}
Calcula $\iint_D (x + y)\,\dd x\,\dd y$, donde $D$ es el dominio
limitado por $y = x^2$ e $y = x$ ($0 \leq x \leq 1$), en los dos
órdenes de integración.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:multint:4}
Usando coordenadas polares, calcula $\iint_D \frac{\dd x\,\dd y}{(1 +
x^2 + y^2)^2}$ sobre todo el plano (como límite sobre discos), y
$\iint_{D'} xy\,\dd x\,\dd y$ sobre el cuarto de disco $D' = \{x, y
\geq 0,\ x^2 + y^2 \leq 1\}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:multint:5}
Calcula el \omterm{def:b2:multint:domain}{área} encerrada por la \omterm{pb:b2:curves:1}{astroide} $x = \cos^3 t$, $y = \sin^3
t$, $t \in [0, 2\pi]$, usando el~\cref{cor:b2:multint:area}. \emph{(Linealiza $\sin^2 t\cos^2 t$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:multint:6}
Calcula el volumen del sólido limitado inferiormente por el
paraboloide $z = x^2 + y^2$ y superiormente por el plano $z = 1$, por
los dos métodos: apilando (intégrese $1 - x^2 - y^2$ sobre el disco
unidad, en coordenadas polares) y rebanando (las rebanadas
horizontales son discos de radio $\sqrt z$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:multint:7}
(Atracción gravitatoria de una bola: el teorema de Newton, caso
particular) Prueba que el volumen de la corona esférica $a \leq r
\leq b$ es $\frac43\pi(b^3 - a^3)$ y calcula $\iiint_{B} \frac{\dd
x\,\dd y\,\dd z}{r}$ sobre la bola $B$ de radio $R$ ($r$ la distancia
al origen). \emph{(Coordenadas esféricas.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:multint:8}
(Lema de Poincaré, caso estrellado) Sea $U$ estrellado respecto de
$0$ (es decir, $M \in U \Rightarrow [0, M] \subseteq U$) y sea
$\omega = P\dd x + Q\dd y$ una forma $\mathcal{C}^1$ cerrada sobre
$U$. Defínase
\[
f(x, y) = \int_0^1 \bigl(x\,P(tx, ty) + y\,Q(tx, ty)\bigr)\dd t .
\]
Usando la derivación bajo el signo integral
(\cref{ch:b2:integration}) y $P_y = Q_x$, prueba que $f_x = P$ y $f_y
= Q$: toda \omterm{def:b2:multint:exact}{forma cerrada} sobre un \omterm{def:b2:metric:topology}{abierto} estrellado es \omterm{def:b2:multint:exact}{exacta}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:multint:9}
(La integral de Dirichlet por integración doble) Justifica y explota
\[
\int_0^\infty\!\!\int_0^\infty e^{-xy}\sin x\;\dd y\,\dd x
\quad\text{frente a}\quad
\int_0^\infty\!\!\int_0^\infty e^{-xy}\sin x\;\dd x\,\dd y
\]
sobre $[0, A] \times [0, \infty)$: prueba que $\int_0^A \frac{\sin
x}{x}\dd x = \frac\pi2 - \int_0^\infty e^{-Ay}\frac{y\sin A + \cos
A}{1 + y^2}\dd y$ y recupera $\int_0^\infty \frac{\sin x}{x}\,\dd x =
\frac\pi2$, comparando con la demostración por \omterm{thm:b2:integration:continuity}{integrales con
parámetro} del~\cref{ch:b2:integration}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:multint:10}
(Desigualdad isoperimétrica vía Wirtinger) Sea $\gamma$ una curva
cerrada simple $\mathcal{C}^1$ de \omterm{def:b2:curves:length}{longitud} $2\pi$, parametrizada por
\omterm{def:b2:curves:length}{longitud de arco} sobre $[0, 2\pi]$, que encierra un \omterm{def:b2:multint:domain}{área} $A$. Usando
el~\cref{cor:b2:multint:area}, Parseval y la desigualdad de Wirtinger
(ejercicios del~\cref{ch:b2:fourier}), demuestra que $A \leq \pi$,
con igualdad para la circunferencia. \emph{(Normaliza
$\int_0^{2\pi} x(s)\dd s = 0$; escribe $2A = \oint x\,\dd y - y\,\dd
x$ y acota $2A \leq \int (x^2 + y'^2)$ con cuidado, vía $2A =
\int_0^{2\pi}(xy' - yx')\dd s$ y $x^2 + y'^2 \geq 2xy'$.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:multint:11}
(Momentos de la bola) Para la bola $B$ de radio $R$ en $\R^3$,
calcula $\iiint_B z^2\,\dd x\,\dd y\,\dd z$ en coordenadas esféricas
y deduce $\iiint_B (x^2 + y^2 + z^2)\,\dd x\,\dd y\,\dd z$ por
simetría. Contrasta esto último con el cálculo por coronas $\int_0^R
r^2\cdot4\pi r^2\,\dd r$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:multint:12}
(Fórmula del cordón) Sea $K$ un polígono de vértices $(x_1, y_1),
\dots, (x_m, y_m)$ en orden antihorario (índices módulo $m$). Deduce
del~\cref{cor:b2:multint:area} que
\[
\operatorname{Area}(K)
= \frac12\sum_{i=1}^m
\bigl(x_iy_{i+1} - x_{i+1}y_i\bigr) ,
\]
y comprueba la fórmula sobre el triángulo $(0,0)$, $(1,0)$, $(0,1)$.
\end{exercise}

\section{Problema: el volumen de la bola en dimensión $n$}

\begin{problem}[{Problema de fin de semana --- $V_n =
\pi^{n/2}/\Gamma(\frac n2 + 1)$, y la rareza de las dimensiones
altas}]\label{pb:b2:multint:1}
El disco tiene \omterm{def:b2:multint:domain}{área} $\pi$, la bola tiene volumen $\frac43\pi$ --- ¿y
después? Este problema calcula el \omterm{pb:b2:multint:1}{volumen de la bola} unidad de $\R^n$
para todo $n$, dos veces (mediante una recursión por rebanado
impulsada por las \omterm{pb:b2:multint:1}{integrales de Wallis}, y después mediante la
función $\Gamma$ y la integral de Gauss del~\cref{ex:b2:multint:polar}), y lee después la geometría: los
volúmenes alcanzan su máximo en la dimensión cinco y se precipitan a
cero, y casi toda una bola de dimensión alta se esconde en una fina
corona junto a su frontera.\index{volumen de la bola}\index{integrales
de Wallis} Para una función \omterm{def:b2:metric:continuity}{continua} sobre una bola de $\R^n$, la
integral se entiende como la integral iterada $n$ veces (rebanando
una coordenada cada vez, como en el capítulo para $n \leq 3$);
escribimos $B_n(R)$ para la bola cerrada de radio $R$ centrada en
$0$, $v_n(R)$ para su volumen y $V_n = v_n(1)$, con $V_0 = 1$ por
convenio.

\medskip
\noindent\textbf{Parte I --- La recursión por rebanado.}
\begin{enumerate}
  \item Sustituyendo $x_i = Ru_i$ en cada una de las $n$ integrales
        iteradas, prueba que $v_n(R) = V_nR^n$.
  \item Rebanando $B_n(1)$ a lo largo de su última coordenada, prueba
        que
        \[
        V_n = V_{n-1}\int_{-1}^{1}(1 -
        t^2)^{\frac{n-1}2}\,\dd t .
        \]
  \item Con $t = \sin\theta$, identifica la integral como una
        \omterm{pb:b2:multint:1}{integral de Wallis}: $\int_{-1}^1(1 - t^2)^{\frac{n-1}2}\dd t
        = 2W_n$, donde $W_n = \int_0^{\pi/2}\cos^n\theta\,\dd\theta =
        \int_0^{\pi/2}\sin^n\theta\,\dd\theta$.
  \item Demuestra las dos identidades de Wallis (intégrese por
        partes; después telescópese $nW_nW_{n-1}$):
        \[
        W_n = \frac{n-1}nW_{n-2} \quad (n \geq 2),
        \qquad
        W_nW_{n-1} = \frac{\pi}{2n} \quad (n \geq 1).
        \]
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La recursión resuelta.}
\begin{enumerate}[resume]
  \item Combina las preguntas 2--4 en la recursión de dos pasos
        \[
        V_n = \frac{2\pi}{n}\,V_{n-2}
        \qquad (n \geq 2).
        \]
  \item Deduce las \omterm{def:b2:multint:exact}{formas cerradas}, para $k \geq 0$:
        \[
        V_{2k} = \frac{\pi^k}{k!},
        \qquad
        V_{2k+1} = \frac{2^{k+1}\pi^k}{1\cdot3\cdot5\cdots
        (2k+1)} .
        \]
  \item Tabula numéricamente $V_1, \dots, V_7$. Usando el cociente
        $V_n/V_{n-2} = 2\pi/n$ y los valores de $2W_5$ y $2W_6$,
        demuestra que la sucesión $(V_n)$ crece hasta su máximo $V_5
        = \frac{8\pi^2}{15} \approx 5.26$ y decrece a partir de
        entonces.
  \item Prueba que $V_n \to 0$ más deprisa que toda sucesión
        geométrica, y que $\sum_{n\geq1} V_n$ converge: todas las
        bolas unidad juntas tienen volumen total finito.
  \item Demuestra la identidad generatriz
        \[
        \sum_{k\geq0} V_{2k}\,x^{2k} = \eu^{\pi x^2}
        \qquad (x \in \R),
        \]
        y deduce $\sum_{k\geq0}V_{2k} = \eu^\pi \approx 23.14$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Segunda vía: $\Gamma$ y la integral de
Gauss.}
\begin{enumerate}[resume]
  \item Prueba por Fubini (el integrando es un producto) que
        \[
        I_n = \int_{\R^n}\eu^{-\norm
        x^2}\dd x
        = \Bigl(\int_{-\infty}^{+\infty}
        \eu^{-t^2}\dd t\Bigr)^{\!n} = \pi^{n/2},
        \]
        la integral de Gauss $n$-dimensional, entendida como límite
        sobre cubos $\intcc{-R}{R}^n$.
  \item Recuérdese $\Gamma(s) = \int_0^\infty t^{s-1}\eu^{-t}\dd t$
        (\cref{def:b2:integration:gamma}). A partir de $\Gamma(s+1) =
        s\,\Gamma(s)$ (\cref{thm:b2:integration:gammaprops}) y
        $\Gamma(\tfrac12) = \sqrt\pi$ (sustitúyase $t = u^2$ e
        invóquese la integral de Gauss), calcula
        \[
        \Gamma(k + 1) = k!,
        \qquad
        \Gamma\Bigl(k + \frac32\Bigr) =
        \frac{1\cdot3\cdots(2k+1)}{2^{k+1}}\,\sqrt\pi .
        \]
  \item Demuestra, por inducción a través de la recursión de la
        pregunta 5, la fórmula única
        \[
        V_n = \frac{\pi^{n/2}}{\Gamma\bigl(\frac n2 +
        1\bigr)} \qquad (n \geq 1),
        \]
        y comprueba que reproduce las dos \omterm{def:b2:multint:exact}{formas cerradas} de la
        pregunta 6.
  \item Prueba que $\int_0^\infty \eu^{-r^2}r^{n-1}\dd r =
        \tfrac12\Gamma\bigl(\tfrac n2\bigr)$ y deduce la identidad
        \[
        I_n = n\,V_n\int_0^\infty \eu^{-r^2}\,r^{n-1}\,\dd r.
        \]
        Interprétala: la masa gaussiana de $\R^n$ se recoge a lo
        largo de coronas esféricas cuya ``\omterm{def:b2:multint:domain}{área}
        $(n-1)$-dimensional'' a radio $r$ es $nV_nr^{n-1}$; ambos
        miembros están ya demostrados de manera independiente, así
        que la interpretación no cuesta nada.
  \item Póngase $s_{n-1} = nV_n$ (el \omterm{def:b2:multint:domain}{área} de la esfera unidad
        $S^{n-1}$, coherentemente con $v_n(R) = \int_0^R
        s_{n-1}r^{n-1}\dd r$). Tabula $s_0, \dots, s_3$ y comprueba
        que $s_1 = 2\pi$, $s_2 = 4\pi$, $s_3 = 2\pi^2$.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Las dimensiones altas son extrañas.}
\begin{enumerate}[resume]
  \item A partir de la fórmula de Stirling
        (\cref{thm:b2:comparison:stirling}) aplicada a $k!$, prueba
        que, para $n = 2k$ par,
        \[
        V_n \sim \frac{1}{\sqrt{\pi n}}
        \Bigl(\frac{2\pi\eu}{n}\Bigr)^{n/2}
        \qquad (n \to \infty, \ n \text{ par}),
        \]
        y explica por qué la misma cota de decaimiento
        supergeométrico se extiende a los $n$ impares mediante la
        recursión.
  \item La bola unidad está dentro del cubo $\intcc{-1}1^n$, de
        volumen $2^n$. Calcula la razón de llenado $V_n/2^n$ para $n
        = 2, 3, 10$ y prueba que tiende a $0$: en dimensión alta,
        esencialmente todo el cubo está en sus esquinas.
  \item Prueba que la fracción de $v_n(1)$ situada a distancia
        $\varepsilon$ o menos de la esfera frontera es $1 - (1 -
        \varepsilon)^n \to 1$; numéricamente, ¿qué fracción de una
        bola de dimensión $100$ está en la corona exterior de grosor
        el $1\,\%$?
  \item Demuestra la asintótica de Wallis $W_n \sim
        \sqrt{\dfrac{\pi}{2n}}$ \emph{(monotonía de $(W_n)$, el
        cociente $W_n/W_{n-2} \to 1$ y $W_nW_{n-1} = \frac\pi{2n}$)},
        y la cota inferior $W_n \geq \sqrt{\dfrac{\pi}{2(n+1)}}$
        para todo $n$.
  \item (Concentración en una banda) La fracción de la bola unidad
        con primera coordenada más allá de $\delta$ es
        $\int_\delta^1(1 - x^2)^{\frac{n-1}2}\dd x \,\big/\, (2W_n)$.
        Usando $1 - u \leq \eu^{-u}$ y la cota de la cola
        $\int_\delta^\infty \eu^{-a x^2}\dd x \leq
        \frac{\eu^{-a\delta^2}}{2a\delta}$, prueba que esa fracción
        es a lo sumo
        \[
        \frac{\eu^{-(n-1)\delta^2/2}}{(n-1)\,\delta}
        \Big/ \sqrt{\frac{2\pi}{n+1}}
        \]
        y concluye: para $\delta = s/\sqrt{n-1}$, toda la bola salvo
        una fracción $O(\eu^{-s^2/2}/s)$ está en la banda
        $\abs{x_1} \leq s/\sqrt{n-1}$. Una bola de dimensión alta
        es, estadísticamente, una tortita fina en todas las
        direcciones a la vez.
  \item Ensambla las preguntas 16--19 en un párrafo: dónde está el
        volumen de $B_n(1)$ (junto a la esfera frontera y, a la vez,
        dentro de bandas de anchura $O(1/\sqrt n)$ de todo
        hiperplano por el centro), y por qué esas dos afirmaciones
        no se contradicen.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Otros cuerpos, y síntesis.}
\begin{enumerate}[resume]
  \item (Símplice) Sea $\Delta_n = \{x \in \R^n : x_i \geq 0,\ \sum
        x_i \leq 1\}$. Prueba por rebanado e inducción que
        $\operatorname{vol}(\Delta_n) = \frac1{n!}$.
  \item (Politopo cruzado) Deduce que $C_n = \{x : \sum\abs{x_i}
        \leq 1\}$ tiene volumen $\frac{2^n}{n!}$, y verifica el
        encaje $C_n \subseteq B_n(1) \subseteq \intcc{-1}1^n$ a
        nivel de volúmenes: $\frac{2^n}{n!} \leq V_n \leq 2^n$.
  \item Calcula $V_4$ de una tercera manera: rebana $\R^4 = \R^2
        \times \R^2$, integra el \omterm{def:b2:multint:domain}{área} del disco $(z, w)$ sobre el
        disco $(x, y)$ en coordenadas polares y recupera $V_4 =
        \frac{\pi^2}2$.
  \item (Monte Carlo en apuros) Se sortea un punto uniformemente en
        el cubo $\intcc{-1}1^{20}$. Prueba que la probabilidad de que
        caiga en la bola inscrita es $V_{20}/2^{20} \approx
        2.5\cdot10^{-8}$, de modo que hacen falta unos cuarenta
        millones de sorteos antes de esperar el primer acierto:
        estimar $V_n$ por muestreo de rechazo se derrumba en
        dimensión alta (la maldición de la dimensionalidad).
  \item Síntesis. Dos deducciones independientes se encontraron en
        $V_n = \pi^{n/2}/\Gamma(\frac n2 + 1)$: enumera qué teorema
        de este capítulo usó cada una (Fubini, cambio de variables,
        la integral de Gauss en polares) y qué insumos de una
        variable (Wallis, $\Gamma$, Stirling). ¿Dónde rehace el
        volumen del tercer año este cálculo con la teoría de
        Lebesgue, y qué añade?
\end{enumerate}
\end{problem}
