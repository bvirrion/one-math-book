\chapter{Ecuaciones diferenciales}\label{ch:b2:diffeq}

El primer año resolvió las ecuaciones lineales que admiten fórmula.
Este capítulo suministra lo que las fórmulas no pueden: el \emph{teorema
de Cauchy--Lipschitz} ---existencia y unicidad para $y' = f(t, y)$---
demostrado con el teorema del punto fijo de Banach, exactamente como se
prometió en el~\cref{ch:b2:metric}; y después la teoría completa de los
\emph{sistemas lineales} $X' = A(t)X + B(t)$, con la
\omterm{ex:b2:nvs:matrixexp}{exponencial de matrices} y el \omterm{def:b2:diffeq:wronskian}{wronskiano} como motores de cálculo.

\section{El teorema de Cauchy--Lipschitz}

\begin{theorem}[Cauchy--Lipschitz, versión lipschitziana global]\label{thm:b2:diffeq:cauchylipschitz}
Sean $I$ un segmento y $f \colon I \times \R^n \to \R^n$ \omterm{def:b2:metric:continuity}{continua} y
\emph{\omterm{def:b2:metric:continuity}{lipschitziana} en la segunda variable}, uniformemente en la
primera: $\norm{f(t, y) - f(t, z)} \leq k\,\norm{y - z}$ para todo $t
\in I$. Entonces, para todo $(t_0, y_0) \in I \times \R^n$, el
problema de Cauchy\index{teorema de Cauchy--Lipschitz}
\[
y' = f(t, y), \qquad y(t_0) = y_0
\]
tiene exactamente una solución $y \colon I \to \R^n$ de clase $C^1$.
\end{theorem}

\begin{proof}
\emph{Reformulación.} Una $y$ \omterm{def:b2:metric:continuity}{continua} resuelve el problema si y solo
si satisface la ecuación integral
\[
y(t) = y_0 + \int_{t_0}^{t} f\bigl(s, y(s)\bigr)\,\dd s
=: T(y)(t)
\]
(teorema fundamental del cálculo en ambos sentidos; una solución
\omterm{def:b2:metric:continuity}{continua} de la ecuación integral es automáticamente $C^1$).

\emph{Una contracción, tras renormar.} Sobre el \omterm{def:b2:nvs:banach}{espacio de Banach} $E =
C(I, \R^n)$ con la \omterm{def:b2:nvs:norm}{norma} \emph{con peso}
\[
N(y) = \sup_{t \in I}\; \eu^{-2k\abs{t - t_0}}\,\norm{y(t)} ,
\]
(equivalente a la \omterm{def:b2:nvs:norm}{norma} del supremo: el peso está acotado superior e
inferiormente sobre el segmento $I$, de modo que $E$ sigue siendo
\omterm{def:b2:metric:complete}{completo}), estimemos, para $y, z \in E$ y, digamos, $t \geq t_0$:
\[
\norm{T(y)(t) - T(z)(t)}
\leq \int_{t_0}^{t} k\,\norm{y(s) - z(s)}\,\dd s
\leq k\,N(y - z)\int_{t_0}^{t} \eu^{2k(s - t_0)}\dd s
\leq \frac{N(y-z)}{2}\,\eu^{2k(t - t_0)} .
\]
Multiplicando por $\eu^{-2k(t - t_0)}$ y tomando el supremo (el caso
$t < t_0$ es \omterm{def:b2:quadratic:adjoint}{simétrico}): $N\bigl(T(y) - T(z)\bigr) \leq \frac12 N(y -
z)$: $T$ es una contracción de razón $\frac12$ del espacio \omterm{def:b2:metric:complete}{completo}
$(E, N)$. El teorema del punto fijo de Banach
(\cref{thm:b2:metric:banach}) proporciona un único punto fijo: la
única solución.
\end{proof}

\begin{remark}
Para $f$ meramente $C^1$ (localmente \omterm{def:b2:metric:continuity}{lipschitziana}), el teorema vale
\emph{localmente}, con una solución maximal sobre un intervalo
\omterm{def:b2:metric:topology}{abierto} maximal; las soluciones pueden explotar en tiempo finito ($y'
= y^2$, $y(0) = 1$: $y(t) = \frac{1}{1-t}$, desaparecida en $t = 1$).
La hipótesis \omterm{def:b2:metric:continuity}{lipschitziana} global es lo que compra el segmento entero.
Dos consecuencias dignas de grabarse: las curvas solución de una EDO
con campo \omterm{def:b2:metric:continuity}{lipschitziano} \emph{nunca se cortan}; y la función nula es la
única solución de una ecuación lineal homogénea que se anule en algún
punto.
\end{remark}

\begin{example}[La unicidad es un teorema: un campo que se escapa]\label{ex:b2:diffeq:leaky}
Consideremos $y' = 2\sqrt{\abs y}$ con $y(0) = 0$. La función nula lo
resuelve; y también
\[
y(t) = \begin{cases} 0 & t \leq 0,\\ t^2 & t \geq 0,
\end{cases}
\]
que es $C^1$ (ambos trozos tienen derivada $0$ en el punto de
empalme) y cumple $y'(t) = 2t = 2\sqrt{t^2}$ para $t > 0$; de hecho,
retrasar el despegue da una solución para \emph{cada} instante de
salida $c \geq 0$: infinitas soluciones con el mismo dato inicial. No
hay contradicción con el~\cref{thm:b2:diffeq:cauchylipschitz}: cerca
de $y = 0$,
\[
\frac{\abs{2\sqrt y - 2\sqrt z}}{\abs{y - z}}
= \frac{2}{\sqrt y + \sqrt z} \longrightarrow +\infty ,
\]
el campo no es \omterm{def:b2:metric:continuity}{lipschitziano} en $y$, y el teorema calla. Moraleja: la
lectura física es la de un cubo que se vacía por gravedad, visto del
revés: desde el estado vacío no puede saberse cuándo empezó a
llenarse; el determinismo de las EDO es exactamente la condición
\omterm{def:b2:metric:continuity}{lipschitziana}, no una ley de la naturaleza.
\end{example}

\section{Sistemas lineales}

\begin{theorem}[Estructura de los sistemas lineales]\label{thm:b2:diffeq:linear}
Sean $A \colon I \to \mathcal{M}_n(\R)$ y $B \colon I \to \R^n$
\omterm{def:b2:metric:continuity}{continuas} sobre un intervalo $I$. Para todo $(t_0, X_0)$, el problema
\[
X' = A(t)X + B(t), \qquad X(t_0) = X_0
\]
tiene exactamente una solución sobre \emph{todo} $I$. Las soluciones
del sistema homogéneo ($B = 0$) forman un espacio vectorial
$\mathcal{S}_H$ de dimensión exactamente $n$, y la evaluación $X
\mapsto X(t_0)$ es un isomorfismo $\mathcal{S}_H \to \R^n$; solución
general $=$ particular $+$ homogénea.
\end{theorem}

\begin{proof}
Sobre todo segmento $J \subseteq I$ que contenga a $t_0$: $f(t, X) =
A(t)X + B(t)$ es \omterm{def:b2:metric:continuity}{continua} y \omterm{def:b2:metric:continuity}{lipschitziana} en $X$ con constante $k =
\sup_J \vertiii{A(t)}$ (finita: es \omterm{def:b2:metric:continuity}{continua} sobre un segmento), de
modo que el~\cref{thm:b2:diffeq:cauchylipschitz} se aplica sobre $J$;
haciendo que $J$ agote $I$, la unicidad pega las soluciones en una
sola sobre $I$. La linealidad del conjunto de soluciones y de la
aplicación evaluación son claras; la evaluación es biyectiva por la
existencia (sobreyectiva) y la unicidad (inyectiva): $\dim
\mathcal{S}_H = n$. La estructura afín es el argumento del primer año
palabra por palabra.
\end{proof}

\begin{example}[El isomorfismo de evaluación, en concreto]\label{ex:b2:diffeq:evaliso}
Para $y'' + y = 0$, visto como el sistema $X' = \begin{pmatrix} 0 &
1\\ -1 & 0\end{pmatrix}X$ con $X = (y, y')$: el teorema dice que el
espacio de soluciones es un plano y que $X \mapsto X(0) = (y(0),
y'(0))$ es un isomorfismo sobre $\R^2$. Las soluciones $\cos$ y $\sin$
se evalúan en $(1, 0)$ y $(0, 1)$ ---la base canónica de $\R^2$---,
así que forman una base del espacio de soluciones, y \emph{toda}
solución es
\[
y(t) = y(0)\cos t + y'(0)\sin t ,
\]
con los coeficientes leídos directamente en los datos iniciales, sin
ningún sistema lineal que resolver. Moraleja: elegir el
\omterm{def:b2:diffeq:wronskian}{sistema fundamental} cuyos valores iniciales sean la base canónica
(aquí, $\cos$ y $\sin$) es exactamente elegir las columnas de
$\eu^{tA}$; el isomorfismo de evaluación es la razón de que las
condiciones iniciales parametricen las trayectorias, el contenido
geométrico de la ``dinámica determinista'' para las ecuaciones
lineales.
\end{example}

\begin{definition}[Wronskiano]\label{def:b2:diffeq:wronskian}
Para soluciones $X_1, \dots, X_n$ del sistema homogéneo, el
\emph{wronskiano}\index{wronskiano} es $W(t) = \det\bigl(X_1(t),
\dots, X_n(t)\bigr)$. Por el isomorfismo anterior, o bien $W$ se
anula idénticamente (la familia es ligada) o bien no se anula nunca
(un \emph{sistema fundamental}\index{sistema fundamental});
cuantitativamente, $W' = \operatorname{tr}\bigl(A(t)\bigr) W$, de
modo que
\[
W(t) = W(t_0)\,\exp\Bigl(\int_{t_0}^{t}
\operatorname{tr} A(s)\,\dd s\Bigr)
\quad \text{(fórmula de Liouville)}.
\]
\end{definition}

\begin{example}[Liouville comprobado en una ecuación de Euler]\label{ex:b2:diffeq:eulerwronskian}
Sobre $\intoo{0}{\infty}$, la ecuación $t^2y'' + ty' - y = 0$ tiene
las soluciones $y_1(t) = t$ e $y_2(t) = \frac1t$ (sustitúyase). Su
\omterm{def:b2:diffeq:wronskian}{wronskiano}:
\[
W(t) = \det\begin{pmatrix} t & \tfrac1t\\[2pt]
1 & -\tfrac{1}{t^2}\end{pmatrix}
= -\frac1t - \frac1t = -\frac2t ,
\]
nunca nulo: un \omterm{def:b2:diffeq:wronskian}{sistema fundamental}. Comprobemos ahora Liouville: en
forma normalizada $y'' + \frac1t\,y' - \frac{1}{t^2}\,y = 0$, la
matriz compañera $A(t) = \begin{pmatrix} 0 & 1\\ \frac{1}{t^2} &
-\frac1t\end{pmatrix}$ tiene traza $-\frac1t$, luego
\[
W(t) = W(1)\exp\Bigl(-\int_1^t\frac{\dd s}{s}\Bigr)
= -2\,\eu^{-\ln t} = -\frac2t . \checkmark
\]
Moraleja: Liouville predice la \emph{forma} del \omterm{def:b2:diffeq:wronskian}{wronskiano} antes de
conocer ninguna solución ---aquí, que $W$ debe ser $\frac{c}{t}$---;
esto es lo que impulsa el método de reducción del orden
(\cref{prop:b2:diffeq:secondorder}), donde conocer $y_1$ y la forma
del \omterm{def:b2:diffeq:wronskian}{wronskiano} determina $y_2$ mediante una sola cuadratura.
\end{example}

\begin{proof}[Demostración de la fórmula de Liouville]
$W(t) = \det M(t)$ con $M' = AM$. Derivando el \omterm{def:b2:linalg:det}{determinante} como
función multilineal de las columnas,
\[
W'(t) = \sum_j \det(X_1, \dots, X_j', \dots, X_n)
= \sum_j \det(X_1, \dots, AX_j, \dots, X_n) .
\]
Ahora bien, la aplicación $(C_1, \dots, C_n) \mapsto \sum_j \det(C_1,
\dots, AC_j, \dots, C_n)$ es $n$-lineal y alternada (con dos columnas
iguales $C_i = C_k$, los términos $j \notin \{i, k\}$ se anulan de
entrada, y los términos $j = i$ y $j = k$ se cancelan por parejas tras
un intercambio de columnas): por el teorema de unicidad
(\cref{thm:b2:linalg:detspace}) vale $c \cdot \det$, con $c$ leído en
las columnas canónicas: $c = \sum_j \det(e_1, \dots, Ae_j, \dots,
e_n) = \sum_j a_{jj} = \operatorname{tr} A$. Por tanto, $W' =
\operatorname{tr}\bigl(A(t)\bigr)W$: una EDO lineal escalar, resuelta
por la fórmula del primer año.
\end{proof}

\section{Coeficientes constantes: la exponencial de matrices}

\begin{theorem}\label{thm:b2:diffeq:matrixexp}
Para $A \in \mathcal{M}_n(\R)$ (o $\C$), la exponencial $\eu^{tA} =
\sum_k \frac{(tA)^k}{k!}$ (\cref{ex:b2:nvs:matrixexp}) cumple: $t
\mapsto \eu^{tA}$ es $C^1$ (de hecho $C^\infty$)
con\index{exponencial de matrices}
\[
\frac{\dd}{\dd t}\,\eu^{tA} = A\,\eu^{tA} = \eu^{tA}A ,
\qquad
\eu^{(s+t)A} = \eu^{sA}\,\eu^{tA},
\qquad
(\eu^{A})^{-1} = \eu^{-A} ;
\]
y $\eu^{A + B} = \eu^A\eu^B$ \emph{cuando} $AB = BA$. El problema de
Cauchy $X' = AX$, $X(0) = X_0$ tiene la única solución $X(t) =
\eu^{tA}X_0$; con un término fuente vale la fórmula de
\emph{variación de las constantes}:
\[
X(t) = \eu^{(t - t_0)A}X_0 + \int_{t_0}^{t} \eu^{(t-s)A}B(s)\,\dd
s .
\]
\end{theorem}

\begin{proof}
\emph{\omterm{def:b2:diffcalc:differential}{Diferenciabilidad}:} la serie $\sum \frac{t^kA^k}{k!}$ y su
serie derivada término a término $\sum \frac{t^{k-1}A^k}{(k-1)!} =
A\sum \frac{(tA)^{k-1}}{(k-1)!}$ convergen \omterm{def:b2:funcseq:series}{normalmente} sobre todo
segmento (\omterm{def:b2:nvs:norm}{normas} $\leq \frac{(\abs t\,\vertiii A)^k}{k!}$): derívese
término a término (\cref{thm:b2:funcseq:seriestransfer}, con valores
vectoriales). Los dos órdenes $A\eu^{tA}$ y $\eu^{tA}A$ coinciden
porque toda suma parcial conmuta con $A$.

\emph{Ley de grupo:} para $A, B$ que conmutan, el
\omterm{thm:b2:series:fubini}{producto de Cauchy} de las dos series exponenciales se reorganiza por
el teorema del binomio exactamente como en el~\cref{ex:b2:series:exp} (la convergencia absoluta en el álgebra de
Banach lo justifica): $\eu^{A+B} = \eu^A\eu^B$; con $B = sA$ esto da
la ley de grupo uniparamétrica, y con $B = -A$, la inversa.

\emph{Problema de Cauchy:} $X(t) = \eu^{tA}X_0$ lo resuelve
(derívese); la unicidad, por el~\cref{thm:b2:diffeq:linear}.
Variación de las constantes: póngase $Y(t) = \eu^{-tA}X(t)$;
derivando, $Y' = \eu^{-tA}(X' - AX) = \eu^{-tA}B(t)$; intégrese de
$t_0$ a $t$ y multiplíquese de nuevo por $\eu^{tA}$.
\end{proof}

\begin{method}[Calcular $\eu^{tA}$]\label{met:b2:diffeq:computeexp}
Redúzcase $A$ (\cref{ch:b2:reduction}): si $A = PDP^{-1}$ es
diagonal, $\eu^{tA} = P\,\eu^{tD}P^{-1}$ con $\eu^{tD}$ diagonal de
entradas $\eu^{t\lambda_i}$; en general, úsese Dunford $A = D + N$
(que conmutan): $\eu^{tA} = \eu^{tD}\,\eu^{tN}$ con $\eu^{tN}$ un
\emph{polinomio} en $t$ (la nilpotencia trunca la serie). Los
\omterm{def:b2:reduction:eigen}{valores propios} complejos se emparejan en bloques de rotación por
exponencial (\cref{exo:b2:diffeq:5}).
\end{method}

\begin{remark}[Errores frecuentes]
\emph{(i) $\eu^{A+B} \neq \eu^A\eu^B$ sin conmutación:} tómense $A =
\begin{pmatrix} 0 & 1\\ 0 & 0\end{pmatrix}$ y $B = \begin{pmatrix} 0
& 0\\ 1 & 0\end{pmatrix}$. Entonces $\eu^A = I + A$ y $\eu^B = I + B$
(nilpotencia), de modo que
\[
\eu^A\eu^B = \begin{pmatrix} 2 & 1\\ 1 & 1\end{pmatrix},
\qquad\text{mientras que}\qquad
\eu^{A+B} = \cosh(1)\,I + \sinh(1)\,(A + B)
= \begin{pmatrix} \cosh 1 & \sinh 1\\ \sinh 1 & \cosh 1
\end{pmatrix},
\]
usando $(A+B)^2 = I$; y $\cosh 1 \approx 1.54 \neq 2$. La ley de
grupo del~\cref{thm:b2:diffeq:matrixexp} lleva una hipótesis genuina.
\emph{(ii) Intuición no lineal en terreno lineal:} las soluciones de
un sistema \emph{lineal} con coeficientes \omterm{def:b2:metric:continuity}{continuos} viven sobre todo
el intervalo (\cref{thm:b2:diffeq:linear}); si una candidata a
solución explota dentro de $I$, o la ecuación no era lineal o el
cálculo está mal. Recíprocamente, para las ecuaciones no lineales no
prometas nunca globalidad sin un argumento ($y' = y^2$).
\emph{(iii) Dividir por la incógnita:} separar variables en $y' =
y(1-y)$ descarta en silencio las soluciones constantes $0$ y $1$,
justo las que organizan la recta de fases
(\cref{exo:b2:diffeq:3}); enumérense primero las soluciones
constantes. \emph{(iv) Los datos iniciales fijan vectores, no
escalares:} una ecuación escalar de orden $n$ necesita $n$
condiciones ($y, y', \dots$ en $t_0$); ajustar solo $y(t_0)$ deja una
familia con $(n-1)$ parámetros, fuente clásica de constantes
``perdidas''.
\end{remark}

\begin{example}[Una exponencial $3\times3$ por Dunford]\label{ex:b2:diffeq:dunford3}
Resolvamos $X' = AX$ para $A = \begin{pmatrix} 2 & 1 & 0\\ 0 & 2 &
0\\ 0 & 0 & 3\end{pmatrix}$. Dunford por bloques: $A = D + N$ con $D
= \operatorname{diag}(2, 2, 3)$ y $N = E_{12}$, que conmutan ($N$
vive dentro del bloque del \omterm{def:b2:reduction:eigen}{valor propio} $2$) y con $N^2 = 0$:
\[
\eu^{tA} = \eu^{tD}\,\eu^{tN}
= \begin{pmatrix}
\eu^{2t} & t\,\eu^{2t} & 0\\
0 & \eu^{2t} & 0\\
0 & 0 & \eu^{3t}
\end{pmatrix} .
\]
La solución general se lee columna a columna: $X(t) =
\bigl(\eu^{2t}(x_0 + ty_0),\ \eu^{2t}y_0,\ \eu^{3t}z_0\bigr)$.
Comprobaciones de sensatez: en $t = 0$ la matriz es $I$; su
\omterm{def:b2:linalg:det}{determinante} es $\eu^{7t} = \eu^{t\operatorname{tr}A}$, como exige
Liouville; y el factor $t$ aparece justo donde el \omterm{def:b2:reduction:eigen}{valor propio} $2$ es
defectuoso. Moraleja: los polinomios por exponenciales no son una
conjetura que haya que memorizar: son la serie truncada $\eu^{tN}$, y
su grado está acotado por el índice de nilpotencia, nunca más.
\end{example}

\begin{method}[Resolver $X' = AX + B(t)$, de principio a fin]\label{met:b2:diffeq:solvelinear}
\begin{enumerate}
  \item \omterm{def:b2:reduction:eigen}{Espectro} de $A$; después $\eu^{tA}$ mediante el~\cref{met:b2:diffeq:computeexp} (diagonalizar, o Dunford como
        en el~\cref{ex:b2:diffeq:dunford3}, o un truco polinómico como
        $A^2 = -I$).
  \item Una solución particular: la variación de las constantes
        $\int_{t_0}^t\eu^{(t-s)A}B(s)\dd s$ funciona siempre; para
        $B$ exponencial-polinómica, un ansatz de la misma forma (con
        el grado elevado en caso de \omterm{pb:b2:diffeq:1}{resonancia},
        \cref{exo:b2:diffeq:10}) es más rápido.
  \item Solución general $= \eu^{(t-t_0)A}X_0 +$ particular; ajústense
        los datos iniciales \emph{al final}, sobre la fórmula
        completa.
  \item Comprobaciones de sensatez: que $X(t_0)$ sea correcto; que el
        crecimiento de la parte homogénea case con las partes reales
        de los \omterm{def:b2:reduction:eigen}{valores propios} (\cref{exo:b2:diffeq:8}); y que el
        $\det$ de una matriz fundamental obedezca a Liouville.
\end{enumerate}
\end{method}

\begin{example}[Un retrato de fases]\label{ex:b2:diffeq:phaseportrait}
$X' = AX$ con $A = \begin{pmatrix} 0 & 1\\ -1 & 0\end{pmatrix}$: $A^2
= -I$, luego la serie se escinde en
\[
\eu^{tA} = (\cos t)\,I + (\sin t)\,A
= \begin{pmatrix} \cos t & \sin t\\ -\sin t & \cos t
\end{pmatrix} :
\]
las trayectorias son circunferencias recorridas en el sentido de las
agujas del reloj; el oscilador armónico $x'' + x = 0$ vestido de
primer orden. \omterm{def:b2:reduction:eigen}{Valores propios} $\pm\iu$ sobre el eje imaginario: un
\emph{centro}. Más en general, las partes reales de los
\omterm{def:b2:reduction:eigen}{valores propios} de $A$ deciden el crecimiento o el decaimiento de
$\norm{X(t)}$ (\cref{exo:b2:diffeq:8}).
\end{example}

\begin{remark}[Dónde se usa]
Los sistemas lineales son el modelo local de todo lo no lineal: cerca
de un equilibrio, un campo vectorial regular se comporta (en los
casos hiperbólicos) como su linearización, cuyo retrato clasifica el
plano traza--determinante. El problema de fin de semana desarrolla
por completo la historia del oscilador ---amortiguamiento,
excitación, \omterm{pb:b2:diffeq:1}{resonancia} y teoremas de comparación de Sturm para
coeficientes variables---, la matemática que hay detrás de los
amortiguadores, de los circuitos de corriente alterna y de los huecos
espectrales por igual. El volumen del tercer año vuelve con la teoría
cualitativa (flujos, estabilidad, integrales primeras) sobre
variedades.
\end{remark}

\section{Segundo orden con coeficientes variables}

\begin{proposition}\label{prop:b2:diffeq:secondorder}
La ecuación $y'' + a(t)y' + b(t)y = c(t)$ ($a, b, c$ \omterm{def:b2:metric:continuity}{continuas} sobre
$I$) es el sistema $X' = A(t)X + B(t)$ para $X = (y, y')$: las
soluciones existen y son únicas sobre todo $I$ para cualesquiera
datos iniciales $(y(t_0), y'(t_0))$; las soluciones homogéneas forman
un plano. Si se conoce una solución homogénea $y_1$ que no se anule,
se halla otra independiente \emph{bajando el orden}: poniendo $y = y_1
z$, la ecuación homogénea se convierte en una ecuación de primer
orden para $z'$, resoluble por cuadraturas.
\end{proposition}

\begin{proof}
La forma de sistema y el~\cref{thm:b2:diffeq:linear} dan todo lo
estructural. Bajada del orden: sustituyendo $y = y_1z$,
\[
y_1 z'' + (2y_1' + a y_1)z' + \underbrace{(y_1'' + ay_1' +
by_1)}_{=\,0}\,z = 0 :
\]
una ecuación lineal de primer orden en $u = z'$, resoluble por la
fórmula del primer año; integrando $u$ se obtiene $z$, y de ahí $y_2 =
y_1 z$, independiente de $y_1$ siempre que $z$ no sea constante.
\end{proof}

\begin{example}\label{ex:b2:diffeq:lowering}
$t^2y'' - 2y = 0$ sobre $\intoo{0}{\infty}$: $y_1 = t^2$ es solución.
Sustitúyase $y = t^2z$: de $y' = t^2z' + 2tz$ e $y'' = t^2z'' + 4tz' +
2z$,
\[
t^2y'' - 2y = t^4 z'' + 4t^3z' = 0,
\qquad\text{es decir,}\qquad \frac{z''}{z'} = -\frac4t :
\]
$z' = t^{-4}$ (salvo constante), $z = -\frac{1}{3t^3}$ e $y_2 = t^2z
= -\frac{1}{3t}$. Solución general: $y = \alpha t^2 +
\frac{\beta}{t}$.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:diffeq:1}
Resuelve $X' = AX$, $X(0) = (1, 0)^{\mathsf T}$, para $A =
\begin{pmatrix} 1 & 1\\ 0 & 2\end{pmatrix}$ (diagonalizando) y para
$A = \begin{pmatrix} 2 & 1\\ 0 & 2 \end{pmatrix}$ (por Dunford).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:diffeq:2}
¿Qué problemas de Cauchy tienen solución global única sobre $\R$ por
el~\cref{thm:b2:diffeq:cauchylipschitz}? $y' = \sin(ty)$; $\;y' =
y^2$; $\;y' = \abs y$. Para el último, resuélvelo explícitamente con
$y(0) = 0$ y con $y(0) = 1$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:diffeq:3}
Demuestra que dos soluciones maximales distintas de $y' = f(t,y)$
($f$ \omterm{def:b2:metric:continuity}{lipschitziana} en $y$) nunca toman el mismo valor en el mismo
instante, y deduce que las soluciones de $y' = y(1 - y)$ que
arrancan en $\intoo{0}{1}$ permanecen en $\intoo{0}{1}$ para
siempre.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:4}
Calcula $\eu^{tA}$ para $A = \begin{pmatrix} 3 & 1\\ -1 &
1\end{pmatrix}$ \emph{(Dunford: $(A - 2I)^2 = 0$)} y resuelve $X' =
AX + \begin{pmatrix} \eu^{2t}\\ 0\end{pmatrix}$, $X(0) = 0$, por
variación de las constantes.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:5}
Para $A = \begin{pmatrix} \alpha & -\beta\\ \beta &
\alpha\end{pmatrix}$, demuestra que $\eu^{tA} = \eu^{\alpha
t}\begin{pmatrix} \cos\beta t & -\sin\beta t\\ \sin\beta t &
\cos\beta t\end{pmatrix}$ ---trayectorias en espiral--- de dos
maneras: mediante la serie (escribiendo $A = \alpha I + \beta J$ con
$J^2 = -I$) y mediante la identificación compleja $z' = (\alpha +
\iu\beta)z$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:6}
(Lema de Gronwall) Sea $u$ \omterm{def:b2:metric:continuity}{continua} y no negativa con $u(t) \leq C +
k\int_{t_0}^{t} u(s)\,\dd s$ sobre $\intco{t_0}{T}$. Demuestra que
$u(t) \leq C\,\eu^{k(t - t_0)}$ \emph{(deriva $v(t) =
\eu^{-kt}\int_{t_0}^t u$)}. Deduce de nuevo la unicidad en
Cauchy--Lipschitz y la dependencia \omterm{def:b2:metric:continuity}{continua} $\norm{y(t) - z(t)} \leq
\norm{y_0 - z_0}\,\eu^{k\abs{t - t_0}}$ para dos soluciones con
datos iniciales distintos.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:7}
Sabiendo que $y_1(t) = \frac{\sin t}{t}$ resuelve $ty'' + 2y' + ty =
0$ sobre $\intoo{0}{\pi}$, halla una segunda solución independiente
bajando el orden y da la solución general.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffeq:8}
Sea $A \in \mathcal{M}_n(\C)$ con todos sus \omterm{def:b2:reduction:eigen}{valores propios} de parte
real (estrictamente) negativa. Demuestra que toda solución de $X' =
AX$ tiende a $0$ cuando $t \to +\infty$, con ritmo exponencial:
$\norm{X(t)} \leq C\,\eu^{-\alpha t}$ para cierto $\alpha > 0$.
\emph{(Trigonaliza; trata el sistema triangular desde la última fila
hacia arriba, o usa Dunford: $\eu^{tA} = \eu^{tD}\eu^{tN}$ con
$\norm{\eu^{tD}} \leq \eu^{-\alpha' t}$ y $\eu^{tN}$ polinómica en
$t$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffeq:9}
(Sin escape en tiempo finito para crecimiento lineal) Supongamos que
$f$ es \omterm{def:b2:metric:continuity}{continua} con $\norm{f(t, y)} \leq a\norm y + b$ sobre
$\intco{0}{\infty} \times \R^n$ y localmente \omterm{def:b2:metric:continuity}{lipschitziana} en $y$.
Usando Gronwall (\cref{exo:b2:diffeq:6}) sobre la forma integral,
demuestra que las soluciones maximales son globales (definidas sobre
todo $\intco{0}{\infty}$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:diffeq:10}
Resuelve $y'' - 3y' + 2y = \eu^{t}$: soluciones homogéneas y después
una solución particular de la forma $\alpha t\,\eu^{t}$ \emph{(¿por
qué falla la conjetura ingenua $\alpha\eu^t$?)}; solución general y
la solución con $y(0) = y'(0) = 0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffeq:11}
Calcula $\eu^{tA}$ para el bloque de Jordan
\[
A = \begin{pmatrix} \lambda & 1 & 0\\ 0 & \lambda & 1\\
0 & 0 & \lambda\end{pmatrix},
\]
y describe todas las soluciones de $X' = AX$: exponenciales por
vectores polinómicos, con grados hasta $2$. ¿De dónde procede el
grado del polinomio?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffeq:12}
(Excitación periódica, respuesta periódica) Sean $A \in
\mathcal{M}_n(\R)$ y $B \colon \R \to \R^n$ \omterm{def:b2:metric:continuity}{continua} y
$T$-periódica.
\begin{enumerate}
  \item Prueba que una solución de $X' = AX + B(t)$ es $T$-periódica
        si y solo si $X(T) = X(0)$ \emph{(compara $X(\cdot + T)$ con
        $X$)}.
  \item Prueba que los \omterm{def:b2:reduction:eigen}{valores propios} de $\eu^{TA}$ son los
        $\eu^{T\lambda}$, $\lambda \in \operatorname{Sp}A$
        \emph{(trigonaliza sobre $\C$)}. Deduce que si ningún
        \omterm{def:b2:reduction:eigen}{valor propio} de $A$ está en $\frac{2\iu\pi}{T}\Z$, entonces
        $I - \eu^{TA}$ es invertible.
  \item Bajo esa hipótesis, demuestra que el sistema tiene
        exactamente una solución $T$-periódica, con
        \[
        X(0) = \bigl(I - \eu^{TA}\bigr)^{-1}
        \int_0^{T}\eu^{(T-s)A}B(s)\,\dd s .
        \]
        ¿A qué corresponde el caso excluido para el oscilador
        armónico? (El problema de fin de semana responde: la
        \omterm{pb:b2:diffeq:1}{resonancia}.)
\end{enumerate}
\end{exercise}

\section{Problema: oscilaciones, resonancia y los teoremas de
comparación de Sturm}

\begin{problem}\label{pb:b2:diffeq:1}
Una sola ecuación gobierna el mundo mecánico y el eléctrico:
\[
x'' + 2\zeta\omega\,x' + \omega^2 x = F(t),
\qquad \omega > 0,\ \zeta \geq 0 .
\]
Este problema la estudia por completo ---mediante la clasificación
traza--determinante de los sistemas lineales planos, los tres
regímenes de amortiguamiento y la respuesta estacionaria a una
excitación periódica con su pico de
\emph{resonancia}\index{resonancia} y la catástrofe de
resonancia--- y abandona después los coeficientes constantes en
favor de los \emph{teoremas de separación y comparación de
Sturm}\index{teorema de comparación de Sturm}, que controlan los
ceros de las soluciones de $y'' + q(t)y = 0$ sin fórmula alguna.

\medskip
\noindent\textbf{Parte I --- El plano traza--determinante.} Sean $A
\in \mathcal M_2(\R)$, $\tau = \operatorname{tr}A$, $\delta = \det
A$, $\Delta = \tau^2 - 4\delta$.
\begin{enumerate}
  \item Prueba que los \omterm{def:b2:reduction:eigen}{valores propios} de $A$ son
        $\frac{\tau\pm\sqrt\Delta}{2}$ y clasifica: dos
        \omterm{def:b2:reduction:eigen}{valores propios} reales de signos opuestos si y solo si
        $\delta < 0$; \omterm{def:b2:reduction:eigen}{valores propios} reales del mismo signo si y
        solo si $\delta > 0$ y $\Delta \geq 0$ (con el signo de
        $\tau$); y un par conjugado no real si y solo si $\Delta <
        0$ (de parte real $\frac\tau2$).
  \item (Silla, $\delta < 0$) Con \omterm{def:b2:reduction:eigen}{valores propios} $\mu < 0 <
        \lambda$ y \omterm{def:b2:reduction:eigen}{vectores propios} $v_\pm$, escribe la solución
        general y describe las trayectorias: dos semirrectas
        estables y dos inestables, y todas las demás órbitas
        asintóticas a ambas. ¿Por qué ninguna solución salvo $0$
        puede permanecer acotada sobre todo $\R$?
  \item (Nodos, $\delta > 0$, $\Delta > 0$) Para $\mu < \lambda <
        0$: prueba que toda solución no nula tiende a $0$ y que
        todas las órbitas, salvo las del eje rápido, llegan
        \emph{tangentes a la dirección propia lenta}
        \emph{(compara $\eu^{\mu t}$ con $\eu^{\lambda t}$)}.
  \item (Focos y centros, $\Delta < 0$) Escribiendo los
        \omterm{def:b2:reduction:eigen}{valores propios} como $\alpha \pm \iu\beta$, usa el~\cref{exo:b2:diffeq:5} (tras un cambio de base real,
        admitido en esa generalidad o demostrado para los sistemas
        de la parte II, que son los que se usan más abajo) para
        describir las órbitas: espirales convergentes para $\alpha
        = \frac\tau2 < 0$, divergentes para $\tau > 0$, y curvas
        cerradas (centro) para $\tau = 0$.
  \item (Casos frontera) Para un \omterm{def:b2:reduction:eigen}{valor propio} doble ($\Delta =
        0$): prueba que $\eu^{tA} = \eu^{\lambda t}(I + tN)$ con $N
        = A - \lambda I$ nilpotente, y distingue la estrella ($N =
        0$) del nodo impropio ($N \neq 0$). Resume la parte I sobre
        la imagen traza--determinante de la figura de este
        capítulo.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El oscilador amortiguado.} Ahora $F =
0$: $x'' + 2\zeta\omega x' + \omega^2x = 0$, es decir, $X' = AX$ con
$A = \begin{pmatrix} 0 & 1\\ -\omega^2 &
-2\zeta\omega\end{pmatrix}$.
\begin{enumerate}[resume]
  \item Calcula $\tau, \delta, \Delta$ y sitúa los tres regímenes en
        el plano traza--determinante: \emph{subamortiguado} $0 <
        \zeta < 1$ (foco estable), \emph{críticamente amortiguado}
        $\zeta = 1$ (\omterm{def:b2:reduction:eigen}{valor propio} doble) y \emph{sobreamortiguado}
        $\zeta > 1$ (nodo estable); $\zeta = 0$ es el centro.
  \item Resuelve explícitamente los tres regímenes:
        \[
        \zeta < 1:\ \eu^{-\zeta\omega t}\bigl(a\cos\omega_d t
        + b\sin\omega_dt\bigr),\ \omega_d =
        \omega\sqrt{1-\zeta^2};
        \qquad
        \zeta = 1:\ (a + bt)\,\eu^{-\omega t};
        \]
        y para $\zeta > 1$, dos exponenciales reales. Define el
        seudoperiodo $\frac{2\pi}{\omega_d}$ y prueba que el
        cociente entre máximos sucesivos de $\abs x$ es la constante
        $\eu^{-2\pi\zeta/\sqrt{1-\zeta^2}}$ (el decremento
        logarítmico).
  \item (El principio del cierrapuertas) Para $\zeta \geq 1$, el
        ritmo de decaimiento lo gobierna el \omterm{def:b2:reduction:eigen}{valor propio} más lento
        $\lambda_{\mathrm{lento}} = -\omega\bigl(\zeta -
        \sqrt{\zeta^2-1}\bigr)$. Prueba que
        $\abs{\lambda_{\mathrm{lento}}} = \frac{\omega}{\zeta +
        \sqrt{\zeta^2 - 1}}$ es una función \emph{decreciente} de
        $\zeta \geq 1$: el amortiguamiento crítico $\zeta = 1$ da el
        regreso al reposo más rápido sin oscilación.
  \item (Energía) Sea $E(t) = \frac12x'^2 + \frac12\omega^2x^2$.
        Demuestra que $E' = -2\zeta\omega\,x'^2 \leq 0$, y deduce
        que para $\zeta > 0$ la ecuación no tiene ninguna solución
        periódica no nula \emph{(un periodo forzaría a $E$ a ser
        constante, luego $x' \equiv 0$)}.
  \item Explica en dos frases por qué el centro $\zeta = 0$ es
        \emph{estructuralmente frágil}: cualquier $\zeta > 0$, por
        pequeño que sea, destruye la periodicidad; y dónde se ve eso
        en el plano traza--determinante (la recta de los centros
        tiene interior vacío).
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Oscilaciones forzadas y
\omterm{pb:b2:diffeq:1}{resonancia}.} Ahora $F(t) = F\cos(\gamma t)$ con $F, \gamma > 0$.
\begin{enumerate}[resume]
  \item ($\zeta > 0$: el régimen estacionario) Búsquese $x_p =
        \Re\bigl(z\,\eu^{\iu\gamma t}\bigr)$: prueba que
        \[
        z = \frac{F}{\omega^2 - \gamma^2 +
        2\iu\zeta\omega\gamma},
        \qquad
        A(\gamma) := \abs z = \frac{F}{\sqrt{(\omega^2 -
        \gamma^2)^2 + 4\zeta^2\omega^2\gamma^2}} ,
        \]
        y escribe $x_p = A(\gamma)\cos(\gamma t - \varphi)$ con
        $\tan\varphi = \frac{2\zeta\omega\gamma}{\omega^2-\gamma^2}$.
  \item Prueba que \emph{toda} solución es $x_p$ más un transitorio
        de la parte II, que tiende a $0$: sean cuales sean los datos
        iniciales, el sistema se engancha al régimen estacionario,
        de amplitud $A(\gamma)$ y desfase $\varphi$.
  \item (La curva de \omterm{pb:b2:diffeq:1}{resonancia}) Maximiza $A$: prueba que
        $A(\gamma)$ tiene un máximo interior si y solo si $\zeta <
        \frac{1}{\sqrt2}$, en
        \[
        \gamma_* = \omega\sqrt{1 - 2\zeta^2},
        \qquad
        A(\gamma_*) =
        \frac{F}{2\zeta\omega^2\sqrt{1-\zeta^2}} ,
        \]
        y que para $\zeta$ pequeño el pico amplifica la respuesta
        estática $A(0) = \frac F{\omega^2}$ por el factor $\approx
        \frac{1}{2\zeta}$.
  \item ($\zeta = 0$, fuera de \omterm{pb:b2:diffeq:1}{resonancia}) Para $\gamma \neq
        \omega$, prueba que la solución con $x(0) = x'(0) = 0$ es
        \[
        x(t) = \frac{F}{\omega^2 -
        \gamma^2}\bigl(\cos\gamma t - \cos\omega t\bigr)
        = \frac{2F}{\omega^2-\gamma^2}
        \sin\frac{(\omega-\gamma)t}{2}
        \sin\frac{(\omega+\gamma)t}{2} :
        \]
        acotada, y con \emph{batidos} ---una oscilación rápida bajo
        una envolvente lenta--- cuando $\gamma$ está cerca de
        $\omega$.
  \item ($\zeta = 0$, en \omterm{pb:b2:diffeq:1}{resonancia}) Para $\gamma = \omega$, prueba
        que $x_p(t) = \frac{F}{2\omega}\,t\sin(\omega t)$ es
        solución, y recupérala como límite de la pregunta 14 cuando
        $\gamma \to \omega$: la amplitud crece linealmente para
        siempre; la catástrofe de \omterm{pb:b2:diffeq:1}{resonancia}.
  \item (Enlace con Fourier) Una excitación periódica general se
        descompone en armónicos (capítulo de Fourier); por
        linealidad, el régimen estacionario es la suma de las
        respuestas armónicas. Para un oscilador sin amortiguamiento
        de frecuencia $\omega = 3$ excitado por una onda cuadrada
        del tipo del~\cref{exo:b2:fourier:1} (con armónicos en todos
        los enteros impares), ¿qué armónico entra en \omterm{pb:b2:diffeq:1}{resonancia}? Una
        frase sobre por qué los ingenieros temen las ondas
        cuadradas.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Los teoremas de Sturm.} Consideremos
$y'' + q(t)\,y = 0$ sobre un intervalo $I$, con $q$ \omterm{def:b2:metric:continuity}{continua}.
(Toda ecuación $y'' + ay' + by = 0$ se reduce a esta forma normal
mediante la sustitución $y = u\exp\bigl(-\frac12\int a\bigr)$; la
pregunta 21 muestra una variante del truco en acción.)
\begin{enumerate}[resume]
  \item Para dos soluciones $y_1, y_2$, prueba que el \omterm{def:b2:diffeq:wronskian}{wronskiano} $W
        = y_1y_2' - y_1'y_2$ es \emph{constante}, nulo si y solo si
        las soluciones son proporcionales; y que una solución no
        nula tiene solo ceros \emph{simples y aislados}.
  \item (Separación de Sturm) Sean $y_1, y_2$ soluciones
        independientes y $a < b$ dos ceros consecutivos de $y_1$.
        Demuestra que $y_2$ se anula exactamente una vez en
        $\intoo{a}{b}$ \emph{(evalúa la constante $W$ en $a$ y en
        $b$: allí $W = y_1'y_2$, y $y_1'(a)$ e $y_1'(b)$ tienen
        signos opuestos)}: los ceros de soluciones independientes se
        entrelazan.
  \item (Comparación de Sturm) Sean $q_1 \leq q_2$ sobre $I$, $y
        \neq 0$ con $y'' + q_1y = 0$, $z \neq 0$ con $z'' + q_2z =
        0$, y $a < b$ ceros consecutivos de $y$. Prueba que $z$ se
        anula en $\intcc{a}{b}$ ---estrictamente dentro si $q_1 <
        q_2$ en algún punto de $\intoo ab$--- \emph{(si $z \neq 0$
        sobre $\intoo ab$, estudia $(yz' - y'z)' = (q_1 - q_2)yz$
        con signos fijos para $y$ y $z$, y compara los valores en la
        frontera)}.
  \item Deduce las \emph{cotas del espaciado}: si $0 < m^2 \leq q(t)
        \leq M^2$ sobre $I$, entonces dos ceros consecutivos
        cualesquiera $a < b$ de una solución no nula de $y'' + qy =
        0$ cumplen
        \[
        \frac{\pi}{M} \;\leq\; b - a \;\leq\; \frac{\pi}{m}
        \]
        \emph{(compara con $u'' + M^2u = 0$ y $u'' + m^2u = 0$,
        cuyos ceros están espaciados $\frac\pi M$ y $\frac\pi
        m$)}. Compruébalo sobre el oscilador armónico.
  \item Transforma $ty'' + 2y' + ty = 0$ (\cref{exo:b2:diffeq:7})
        mediante $u = ty$ en $u'' + u = 0$, recupera al instante sus
        soluciones $\frac{\sin t}t$ y $\frac{\cos t}{t}$, y concluye
        que los ceros de toda solución no nula están espaciados
        exactamente $\pi$: la visión del mundo de Sturm, según la
        cual los ceros los controla el coeficiente $q$, haya
        fórmulas o no.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Duhamel y la frontera de la
acotación.}
\begin{enumerate}[resume]
  \item (Duhamel para el oscilador) Prueba que, para $F$ \omterm{def:b2:metric:continuity}{continua},
        la solución de $x'' + \omega^2x = F(t)$ con $x(0) = x'(0) =
        0$ es
        \[
        x(t) = \frac1\omega\int_0^t\sin\bigl(\omega(t -
        s)\bigr)F(s)\,\dd s ,
        \]
        y vuelve a deducir de ahí la solución resonante de la
        pregunta 15 con $F(s) = F\cos(\omega s)$ \emph{(fórmula de
        producto a suma)}.
  \item ($\zeta > 0$: entrada acotada, salida acotada) Prueba que
        para $\zeta > 0$ y \emph{cualquier} $F$ \omterm{def:b2:metric:continuity}{continua} y acotada,
        toda solución de la ecuación amortiguada está acotada sobre
        $\intco{0}{\infty}$ \emph{(variación de las constantes más
        el decaimiento exponencial $\vertiii{\eu^{tA}} \leq
        C\eu^{-\alpha t}$ del~\cref{exo:b2:diffeq:8})}.
  \item ($\zeta = 0$) Prueba que, sin amortiguamiento, una
        excitación periódica acotada mantiene acotadas todas las
        soluciones \emph{salvo} exactamente en \omterm{pb:b2:diffeq:1}{resonancia} ($\gamma =
        \omega$, pregunta 15 frente a pregunta 14): el
        amortiguamiento es lo que convierte la frontera de la
        acotación en estabilidad uniforme.
  \item Síntesis. Una frase para cada punto: (i) cómo organiza el
        plano traza--determinante las partes I--II y dónde lo
        abandona la excitación (parte III); (ii) el significado
        físico de $\gamma_*$, de $A(\gamma_*)$ y del factor
        $\frac1{2\zeta}$; (iii) qué dicen los teoremas de Sturm que
        las fórmulas explícitas no pueden decir; (iv) qué dos
        resultados de este problema reutilizará el resto del libro
        sin hacer ruido (los \omterm{def:b2:diffeq:wronskian}{wronskianos} de constante de Liouville;
        la estabilidad con entrada acotada).
\end{enumerate}
\end{problem}
