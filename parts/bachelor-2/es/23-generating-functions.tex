\chapter{Funciones generatrices de probabilidad}\label{ch:b2:genfun}

Las series de potencias del~\cref{ch:b2:powerseries} vuelven con una
misión probabilista: a una \omterm{def:b2:randomvar:law}{variable aleatoria} con valores en $\N$ le
asociamos la serie de potencias de coeficientes $\P(X = n)$. Esta
\emph{\omterm{ex:b2:powerseries:fibonacci}{función generatriz}} convierte las sumas de variables
\omterm{def:b2:proba:independence}{independientes} en productos, los momentos en derivadas en $1$ y las
identidades combinatorias difíciles en multiplicaciones de una línea.
El capítulo cierra el libro con dos piezas de lucimiento: la
aproximación de Poisson de los \omterm{def:b2:proba:space}{sucesos} raros y el criterio de extinción
de los \omterm{pb:b2:genfun:1}{procesos de ramificación}, un cálculo probabilista genuinamente
infinito resuelto por entero mediante la geometría de una curva
convexa.

\section{Definición y propiedades básicas}

\begin{definition}[Función generatriz de probabilidad]\label{def:b2:genfun:pgf}
Sea $X$ una \omterm{def:b2:randomvar:law}{variable aleatoria} con valores en $\N$, y $p_n = \P(X =
n)$. La \emph{función generatriz de probabilidad}\index{función
generatriz de probabilidad} de $X$ es la suma de la serie de
potencias
\[
G_X(t) = \E\bigl(t^X\bigr) = \sum_{n=0}^{\infty} p_n\,t^n .
\]
\end{definition}

\begin{example}[Primeros reflejos]\label{ex:b2:genfun:firstreflexes}
Una variable constante $X = c$ tiene $G_X(t) = t^c$; un
desplazamiento obedece a $G_{X+c}(t) = t^c\,G_X(t)$; y evaluar en
puntos especiales permite leer información sin desarrollo alguno:
$G_X(0) = \P(X = 0)$, $G_X(1) = 1$ y $G_X(-1) = \P(X\text{ par}) -
\P(X\text{ impar})$, el balance de paridades explotado en el~\cref{exo:b2:genfun:10}. Estas fórmulas de una línea se usan
calladamente en todo lo que sigue; y la evaluación $G_X(0)$ es
exactamente cómo se extraerán las probabilidades de extinción de las
\omterm{ex:b2:powerseries:fibonacci}{funciones generatrices} iteradas al final del capítulo.
\end{example}

\begin{proposition}[Radio y primeras propiedades]\label{prop:b2:genfun:radius}
La serie que define $G_X$ tiene \omterm{def:b2:powerseries:radius}{radio de convergencia} $\geq 1$; $G_X$
está definida y es \omterm{def:b2:metric:continuity}{continua} sobre $\intcc{-1}{1}$ y $\mathcal{C}^\infty$
sobre $\intoo{-1}{1}$, con $G_X(1) = 1$ y $\abs{G_X(t)} \leq 1$ ahí.
Además, $G_X$ determina la \omterm{def:b2:randomvar:law}{ley} de $X$:
\[
p_n = \frac{G_X^{(n)}(0)}{n!} .
\]
\end{proposition}

\begin{proof}
Como $\sum p_n = 1$ converge, los términos $p_n\,1^n$ están acotados,
de modo que el radio es $\geq 1$ (lema de Abel,
\cref{ch:b2:powerseries}); en $t = \pm1$ la serie converge
\omterm{def:b2:series:def}{absolutamente} ($\sum p_n = 1$ la domina); mejor aún, sobre todo el
intervalo $\intcc{-1}1$,
\[
\sup_{\abs t\leq1}\,\abs{p_nt^n} = p_n
\quad\text{con}\quad \sum_np_n < \infty :
\]
la serie converge \emph{\omterm{def:b2:funcseq:series}{normalmente}} sobre $\intcc{-1}1$, así que su
suma es \omterm{def:b2:metric:continuity}{continua} ahí
(\cref{thm:b2:funcseq:weierstrass,thm:b2:funcseq:continuity}). La
regularidad en el interior y la fórmula de los coeficientes son la
teoría general de las series de potencias; y como los coeficientes se
recuperan, dos variables con la misma \omterm{ex:b2:powerseries:fibonacci}{función generatriz} tienen la
misma \omterm{def:b2:randomvar:law}{ley}.
\end{proof}

\begin{example}[Las leyes clásicas]\label{ex:b2:genfun:classical}
\leavevmode
\begin{itemize}
\item Bernoulli $\mathcal{B}(p)$: $G(t) = 1 - p + pt$.
\item Binomial $\mathcal{B}(n, p)$: $G(t) = \sum_k \binom nk
(pt)^k(1-p)^{n-k} = (1 - p + pt)^n$ (teorema del binomio).
\item Geométrica $\mathcal{G}(p)$: $G(t) =
\sum_{k\geq1}(1-p)^{k-1}p\,t^k = \dfrac{pt}{1 - (1-p)t}$ (radio
$\frac{1}{1-p} > 1$).
\item Poisson $\mathcal{P}(\lambda)$: $G(t) = \sum_k
e^{-\lambda}\frac{(\lambda t)^k}{k!} = e^{\lambda(t - 1)}$ (radio
$\infty$).
\end{itemize}
\end{example}

\begin{example}[Integrar la función
generatriz]\label{ex:b2:genfun:integraltrick}
Las derivadas de $G_X$ en $1$ dan los momentos positivos; la
\emph{integral} da uno negativo. De $\int_0^1t^k\dd t = \frac1{k+1}$
y la integración término a término (\omterm{def:b2:funcseq:series}{convergencia normal} sobre
$\intcc01$):
\[
\int_0^1G_X(t)\,\dd t = \sum_{k\geq0}\frac{\P(X =
k)}{k+1} = \E\Bigl(\frac1{1+X}\Bigr).
\]
Para $X \sim \mathcal P(\lambda)$:
\[
\E\Bigl(\frac1{1+X}\Bigr) =
\int_0^1\eu^{\lambda(t-1)}\,\dd t = \frac{1 -
\eu^{-\lambda}}{\lambda},
\]
lo que recupera en una línea el cálculo con series del~\cref{ex:b2:randomvar:transferex}. La \omterm{ex:b2:powerseries:fibonacci}{función generatriz} es un
instrumento de doble sentido: derívese en $1$ para los momentos
$\E(X)$, $\E(X(X-1))$, e intégrese sobre $\intcc01$ para
$\E\bigl(\frac1{1+X}\bigr)$; un único objeto \omterm{def:b2:powerseries:analytic}{analítico}, interrogado
en la dirección que el problema necesite.
\end{example}

\begin{example}[Una ley de radio exactamente
uno]\label{ex:b2:genfun:heavytail}
Sea $\P(X = k) = \dfrac{6}{\pi^2k^2}$ para $k \geq 1$; una \omterm{def:b2:randomvar:law}{ley} de
probabilidad por la identidad de Basilea
(\cref{ex:b2:fourier:basel}). Su \omterm{ex:b2:powerseries:fibonacci}{función generatriz} $G(t) =
\frac6{\pi^2}\sum_{k\geq1}\frac{t^k}{k^2}$ tiene
\omterm{def:b2:powerseries:radius}{radio de convergencia} exactamente $1$: la cota general ``radio $\geq
1$'' de la~\cref{prop:b2:genfun:radius} no puede mejorarse. Y la
media es
\[
\sum_{k\geq1}k\,\P(X = k) =
\frac6{\pi^2}\sum_{k\geq1}\frac1k = \infty :
\]
$G$ es \omterm{def:b2:metric:continuity}{continua} sobre $\intcc{-1}1$ y regular en el interior, pero su
derivada estalla en $1^-$; la gráfica llega al punto $(1, 1)$ con
tangente vertical. Las colas pesadas son visibles
\emph{geométricamente} en la \omterm{ex:b2:powerseries:fibonacci}{función generatriz}, en el único punto $t
= 1$; el teorema de los momentos de más abajo hace exacta esa
correspondencia.
\end{example}

\begin{theorem}[Momentos a partir de la función
generatriz]\label{thm:b2:genfun:moments}
$X$ tiene \omterm{def:b2:randomvar:expectation}{esperanza} si y solo si $G_X$ es \omterm{def:b2:diffcalc:differential}{diferenciable} en $1^-$
(derivada por la izquierda, finita), y entonces $\E(X) = G_X'(1)$.
Análogamente, $X$ tiene momento de orden dos si y solo si $G_X$ es
dos veces \omterm{def:b2:diffcalc:differential}{diferenciable} en $1^-$, y entonces
\[
\E\bigl(X(X - 1)\bigr) = G_X''(1),
\qquad
V(X) = G_X''(1) + G_X'(1) - G_X'(1)^2 .
\]
\end{theorem}

\begin{proof}
Para $t \in \intoo{0}{1}$, la derivación término a término dentro del
disco da $G_X'(t) = \sum_{n\geq1} np_n t^{n-1}$, una serie de
coeficientes no negativos: $t \mapsto G_X'(t)$ es no decreciente
sobre $\intoo{0}{1}$ y, por la convergencia monótona de las sumas
parciales (o el teorema de Abel para coeficientes no negativos,
\cref{ch:b2:powerseries}),
\[
\lim_{t \to 1^-} G_X'(t)
= \sum_{n\geq1} n\,p_n \in \intcc{0}{+\infty} ,
\]
siendo cada miembro finito exactamente cuando lo es el otro. Cuando
es finito, el teorema del valor medio encaja los cocientes
incrementales $\frac{G_X(1) - G_X(t)}{1 - t}$ entre valores de
$G_X'$, de modo que $G_X$ es \omterm{def:b2:diffcalc:differential}{diferenciable} en $1^-$ con $G_X'(1) =
\sum np_n = \E(X)$ (por transferencia). El enunciado de segundo orden
repite el argumento un nivel más arriba: $G''_X(t) =
\sum_{n\geq2}n(n-1)p_nt^{n-2}$ es no decreciente sobre $\intoo01$ con
límite monótono $\sum_nn(n-1)p_n = \E(X(X-1))$, finito exactamente
cuando $X$ tiene momento de orden dos. La fórmula de la \omterm{def:b2:randomvar:variance}{varianza} se
sigue entonces de König--Huygens:
\[
V(X) = \E(X^2) - \E(X)^2 = \E\bigl(X(X-1)\bigr) + \E(X) -
\E(X)^2 = G''_X(1) + G'_X(1) - G'_X(1)^2 .
\]
\end{proof}

\begin{example}
Poisson: $G'(t) = \lambda e^{\lambda(t-1)}$, luego $\E(X) = \lambda$;
y $G''(1) = \lambda^2$, luego $V(X) = \lambda^2 + \lambda - \lambda^2
= \lambda$; los cálculos del~\cref{ch:b2:randomvar} en una línea cada
uno.
\end{example}

\begin{example}[La moda de una ley de Poisson]
\label{ex:b2:genfun:poissonmode}
¿Dónde es mayor $\P(X = k)$ para $X \sim \mathcal P(\lambda)$? Los
pesos consecutivos se comparan mediante el cociente
\[
\frac{\P(X = k+1)}{\P(X = k)} = \frac{\lambda}{k + 1} ,
\]
que supera $1$ mientras $k < \lambda - 1$ y cae por debajo de $1$ en
cuanto $k > \lambda - 1$: los pesos suben y luego bajan, con moda
$\floor\lambda$ (y un empate entre $\lambda - 1$ y $\lambda$ cuando
$\lambda$ es entero: para $\lambda = 3$, $\P(X = 2) = \P(X = 3) =
\frac92\eu^{-3} \approx 0.224$). Los criterios del cociente sobre los
coeficientes suelen ser la vía más rápida hacia los hechos
cualitativos de una \omterm{def:b2:randomvar:law}{ley} discreta; no hace falta ninguna
\omterm{ex:b2:powerseries:fibonacci}{función generatriz}, pero los coeficientes \emph{son} la
\omterm{ex:b2:powerseries:fibonacci}{función generatriz}, leída término a término.
\end{example}

\section{Sumas de variables independientes}

\begin{theorem}[Multiplicatividad]\label{thm:b2:genfun:product}
Si $X$ e $Y$ son \omterm{def:b2:randomvar:law}{variables aleatorias} \omterm{def:b2:proba:independence}{independientes} con valores en
$\N$, entonces
\[
G_{X + Y}(t) = G_X(t)\,G_Y(t)
\qquad (\abs t \leq 1),
\]
y por inducción, $G_{X_1 + \dots + X_n} = \prod_i G_{X_i}$ para $X_1,
\dots, X_n$ \omterm{def:b2:proba:independence}{independientes}.
\end{theorem}

\begin{proof}
Dos demostraciones, ambas instructivas. \emph{Vía \omterm{def:b2:randomvar:expectation}{esperanzas}:} $t^X$
y $t^Y$ son variables acotadas \omterm{def:b2:proba:independence}{independientes}, de modo que
(\cref{thm:b2:randomvar:product})
\[
G_{X+Y}(t) = \E\bigl(t^{X+Y}\bigr)
= \E\bigl(t^X t^Y\bigr)
= \E\bigl(t^X\bigr)\E\bigl(t^Y\bigr) .
\]
\emph{Vía \omterm{thm:b2:series:fubini}{productos de Cauchy}:} la \omterm{def:b2:randomvar:law}{ley} de $X + Y$ es la
convolución $\P(X + Y = n) = \sum_{k=0}^n \P(X = k)\P(Y = n - k)$, y
el teorema del \omterm{thm:b2:series:fubini}{producto de Cauchy} para series \omterm{def:b2:series:def}{absolutamente}
convergentes (\cref{ch:b2:series}) multiplica las dos series de
potencias exactamente a lo largo de esa convolución.
\end{proof}

\begin{example}[Estabilidad de las leyes clásicas]\label{ex:b2:genfun:stability}
Las binomiales \omterm{def:b2:proba:independence}{independientes} con el mismo $p$ se suman: $(1 - p +
pt)^m(1 - p + pt)^n = (1 - p + pt)^{m+n}$, luego $\mathcal{B}(m, p) +
\mathcal{B}(n, p) = \mathcal{B}(m + n, p)$; en particular, una suma
de $n$ variables de Bernoulli \omterm{def:b2:proba:independence}{independientes} es binomial, lo que vuelve
a demostrar la \omterm{def:b2:randomvar:law}{ley} del número de éxitos. Las de Poisson
\omterm{def:b2:proba:independence}{independientes} se suman: $e^{\lambda(t-1)}e^{\mu(t-1)} = e^{(\lambda +
\mu)(t-1)}$, luego $\mathcal{P}(\lambda) + \mathcal{P}(\mu) =
\mathcal{P}(\lambda + \mu)$; el cálculo de convolución del~\cref{exo:b2:randomvar:2}, ahora sin cálculo.
\end{example}

\begin{example}[Dos dados, un polinomio al
cuadrado]\label{ex:b2:genfun:twodice}
Para un dado equilibrado, $G(t) = \frac{t + t^2 + \dots + t^6}{6}$;
para la suma de dos,
\[
G(t)^2 = \frac{1}{36}\bigl(t^2 + 2t^3 + 3t^4 + 4t^5 + 5t^6 +
6t^7 + 5t^8 + 4t^9 + 3t^{10} + 2t^{11} + t^{12}\bigr) :
\]
la \omterm{def:b2:randomvar:law}{ley} triangular de las sumas de dados ($7$ es la moda, con
probabilidad $\frac6{36} = \frac16$), leída en un cuadrado de
polinomio que se desarrolla una vez en la vida. La fórmula de la
convolución habría exigido once argumentos de recuento por separado;
la \omterm{ex:b2:powerseries:fibonacci}{función generatriz} los hace todos a la vez, porque multiplicar
polinomios \emph{es} convolucionar coeficientes. Esta traducción
mecánica ---leyes a coeficientes, sumas a productos--- es todo el
modelo de negocio del capítulo, y el~\cref{exo:b2:genfun:11} lo lleva
hasta los sorprendentes dados de Sicherman.
\end{example}

\begin{example}[Tres dados y una extracción de
coeficiente]\label{ex:b2:genfun:threedice}
Para la suma $S$ de tres dados equilibrados, $\P(S = 10)$ es el
coeficiente de $t^{10}$ en $\bigl(\frac{t + \dots + t^6}6\bigr)^3$.
Factorícese y desarróllese con las series binomial y geométrica:
\[
\Bigl(\frac{t(1 - t^6)}{6(1 - t)}\Bigr)^{\!3}
= \frac{t^3}{216}\,\bigl(1 - 3t^6 + 3t^{12} -
t^{18}\bigr)\sum_{j\geq0}\binom{j+2}2t^j .
\]
El coeficiente de $t^{10}$ requiere un $t^7$ del producto: $j = 7$
con el término $1$, y $j = 1$ con el término $-3t^6$:
\[
\P(S = 10) = \frac{1}{216}\Bigl(\binom92 -
3\binom32\Bigr) = \frac{36 - 9}{216} = \frac{27}{216} =
\frac18 .
\]
Enumerar directamente las $27$ ternas es propenso a errores; el
álgebra es mecánica y escala a cualquier número de dados: la
inclusión-exclusión visible en $(1 - t^6)^3$ hace el análisis de
casos automáticamente.
\end{example}

\begin{example}[Leer una ley en su función
generatriz]\label{ex:b2:genfun:identify}
¿Qué \omterm{def:b2:randomvar:law}{ley} tiene $G(t) = \dfrac1{2 - t}$? Desarróllese en serie de
potencias:
\[
\frac{1}{2 - t} = \frac12\cdot\frac1{1 - t/2}
= \sum_{k\geq0}\frac{t^k}{2^{k+1}} :
\]
coeficientes no negativos que suman $G(1) = 1$, de modo que es una
\omterm{def:b2:randomvar:law}{ley} genuina, $\P(X = k) = 2^{-(k+1)}$ sobre $\N$; una \omterm{def:b2:randomvar:law}{ley}
geométrica que empieza en $0$. Por la unicidad
(\cref{prop:b2:genfun:radius}), ninguna otra \omterm{def:b2:randomvar:law}{ley} comparte esta $G$.
Reconocer \omterm{def:b2:randomvar:law}{leyes} a partir de sus \omterm{ex:b2:powerseries:fibonacci}{funciones generatrices} es una
destreza que conviene ejercitar: así se desenmascara el iterado
crítico de ramificación $G_n(t) = \frac{n - (n-1)t}{n+1 - nt}$ del
problema de fin de semana como una \omterm{def:b2:randomvar:law}{ley} geométrica condicionada a la
supervivencia.
\end{example}

\begin{remark}
La estabilidad va en un solo sentido: las sumas de variables de
Poisson \omterm{def:b2:proba:independence}{independientes} son de Poisson, pero las diferencias no; $X -
Y$ toma valores negativos, de modo que no tiene \omterm{ex:b2:powerseries:fibonacci}{función generatriz} en
absoluto, y su \omterm{def:b2:randomvar:law}{ley} (la \omterm{def:b2:randomvar:law}{distribución} de Skellam) queda fuera de la
caja de herramientas de este capítulo. Igualmente, $\mathcal B(m, p)
+ \mathcal B(n, p')$ con $p \neq p'$ \emph{no} es binomial: el
producto $(1 - p + pt)^m(1 - p' + p't)^n$ tiene dos localizaciones
distintas de raíces, mientras que toda \omterm{ex:b2:powerseries:fibonacci}{función generatriz} binomial
tiene una única raíz repetida. Leer la estabilidad en los patrones de
raíces es un pequeño anticipo de cuánta estructura codifica el
polinomio.
\end{remark}

\begin{remark}[El filtro de las raíces de la unidad]
Evaluar en $-1$ separa lo par de lo impar; evaluar en todas las
raíces $m$-ésimas de la unidad separa cada clase de restos: con
$\omega = \eu^{2\iu\pi/m}$,
\[
\P(X \equiv r \bmod m)
= \frac1m\sum_{j=0}^{m-1}\omega^{-jr}\,G_X(\omega^j),
\]
ya que promediar $\omega^{j(k-r)}$ sobre $j$ da $1$ si $k \equiv r$ y
$0$ en caso contrario. Dividendo de muestra: para la suma $S$ de dos
dados equilibrados, cada $G(\omega^j) = \frac16\sum_{k=1}^6
\omega^{jk} = -\frac16$ para $j \neq 0$ (las siete raíces séptimas de
la unidad suman cero), de modo que
\[
\P(7 \mid S) = \frac17\Bigl(1 +
6\cdot\frac1{36}\Bigr) = \frac16 ,
\]
lo que confirma el recuento del~\cref{ex:b2:genfun:twodice}; y el
método escala a preguntas donde el recuento directo no llega.
\end{remark}

\begin{theorem}[Sumas aleatorias: la identidad de Wald para funciones
generatrices]\label{thm:b2:genfun:compound}
Sean $(X_k)_{k\geq1}$ variables \omterm{def:b2:proba:independence}{independientes} con valores en $\N$, la
misma \omterm{def:b2:randomvar:law}{ley} y \omterm{ex:b2:powerseries:fibonacci}{función generatriz} $G_X$, y sea $N$ una variable con
valores en $\N$, \omterm{def:b2:proba:independence}{independiente} de las $X_k$ y de
\omterm{ex:b2:powerseries:fibonacci}{función generatriz} $G_N$. Entonces la suma aleatoria $S = X_1 + \dots +
X_N$ (con $S = 0$ cuando $N = 0$) tiene \omterm{ex:b2:powerseries:fibonacci}{función generatriz}
\[
G_S = G_N \circ G_X .
\]
En particular, si $N$ y $X_1$ tienen \omterm{def:b2:randomvar:expectation}{esperanza}, $\E(S) =
\E(N)\,\E(X_1)$.
\end{theorem}

\begin{proof}
Condiciónese a $N$ (probabilidad total,
\cref{thm:b2:proba:bayes}): para $\abs t \leq 1$,
\[
G_S(t) = \sum_{n=0}^\infty \P(N = n)\,
\E\bigl(t^{X_1 + \dots + X_n}\bigr)
= \sum_{n=0}^\infty \P(N = n)\,G_X(t)^n
= G_N\bigl(G_X(t)\bigr),
\]
usando la multiplicatividad para cada $n$ fijo y la \omterm{def:b2:series:summable}{sumabilidad} de
toda la familia doble ($\abs{G_X(t)} \leq 1$). El intercambio de
sumaciones es Fubini para \omterm{def:b2:series:summable}{familias sumables}
(\cref{ch:b2:series}). Derivando en $1^-$ por la regla de la cadena y
el~\cref{thm:b2:genfun:moments}: $\E(S) = G_N'(G_X(1))\,G_X'(1) =
G_N'(1)G_X'(1) = \E(N)\E(X_1)$.
\end{proof}

\begin{example}[Poisson compuesta: las pérdidas anuales de una
aseguradora]\label{ex:b2:genfun:compoundpoisson}
Una aseguradora recibe $N \sim \mathcal P(\lambda)$ siniestros en un
año, y cada siniestro cuesta $X_k$ (unidades enteras, i.i.d., con
\omterm{ex:b2:powerseries:fibonacci}{función generatriz} $G_X$, media $\mu$ e independiente de $N$). Por el~\cref{thm:b2:genfun:compound}, la pérdida total $S$ tiene
\[
G_S(t) = \eu^{\lambda(G_X(t) - 1)},
\qquad
\E(S) = \lambda\mu ,
\]
y derivando dos veces en $1^-$:
\[
V(S) = \lambda\,G_X''(1) + \lambda^2\mu^2 + \lambda\mu -
(\lambda\mu)^2 = \lambda\,\E(X^2) .
\]
La \omterm{def:b2:randomvar:variance}{varianza} involucra el momento \emph{segundo} de un siniestro
aislado, y no su \omterm{def:b2:randomvar:variance}{varianza}: una suma de Poisson compuesta siente dos
veces el siniestro grande ocasional, una por cuántos y otra por lo
grande. Para $\lambda = 10$ siniestros de \omterm{def:b2:randomvar:law}{ley} geométrica de media
$2$ ($\E X^2 = 6$): $\E S = 20$, $V(S) = 60$, y Chebyshev
(\cref{ch:b2:randomvar}) ya rinde márgenes de solvencia utilizables.
Este patrón de ``suma detenida al azar'' es el mismo que impulsará la
recursión de ramificación de la~\cref{prop:b2:genfun:branching}: la composición de
\omterm{ex:b2:powerseries:fibonacci}{funciones generatrices} es el álgebra de las poblaciones aleatorias.
\end{example}

\begin{remark}
La \omterm{def:b2:proba:independence}{independencia} de $N$ respecto de los sumandos no es decorativa.
Tómense $X_k \in \{0, 2\}$ con igual probabilidad y hágase $N = X_1$
(flagrantemente dependiente): entonces $S = X_1 + \dots + X_N$ vale
$0$ cuando $X_1 = 0$, y $2 + X_2$ cuando $X_1 = 2$, de modo que
$\E(S) = \frac12(2 + 1) = \frac32$, mientras que $\E(N)\E(X_1) =
1\cdot1 = 1$: la identidad de Wald falla. Cuando se permite que el
número de términos \emph{reaccione} a los propios términos, la
estructura limpia de producto se derrumba; la teoría completa de esas
reglas de ``parada'' es el capítulo de martingalas del volumen del
tercer año.
\end{remark}

\section{Aproximación de Poisson}

\begin{theorem}[Ley de los sucesos raros]\label{thm:b2:genfun:rareevents}
Sea $X_n \sim \mathcal{B}(n, p_n)$ con $n\,p_n \to \lambda > 0$.
Entonces, para todo $k \in \N$:
\[
\P(X_n = k)
\xrightarrow[n\to\infty]{}
e^{-\lambda}\frac{\lambda^k}{k!} :
\]
la \omterm{def:b2:randomvar:law}{ley} binomial de muchos \omterm{def:b2:proba:independence}{sucesos independientes} raros converge a la
\omterm{def:b2:randomvar:law}{ley} de Poisson de parámetro $\lambda$.
\end{theorem}

\begin{proof}
Cálculo directo con $p_n = \frac{\lambda_n}{n}$, $\lambda_n \to
\lambda$:
\[
\P(X_n = k)
= \binom nk p_n^k(1 - p_n)^{n-k}
= \frac{n(n-1)\cdots(n-k+1)}{n^k}\cdot
\frac{\lambda_n^k}{k!}\,
\bigl(1 - \tfrac{\lambda_n}{n}\bigr)^{n-k} .
\]
Cuando $n \to \infty$ con $k$ fijo: el primer factor tiende a $1$
(producto de $k$ factores $\to 1$); $\lambda_n^k \to \lambda^k$; y
$\bigl(1 - \frac{\lambda_n}{n}\bigr)^{n-k} = \exp\bigl((n-k)\ln(1 -
\frac{\lambda_n}{n})\bigr) \to e^{-\lambda}$, pues $(n - k)\ln\bigl(1
- \frac{\lambda_n}{n}\bigr) \sim -\lambda_n \to -\lambda$
(\cref{ch:b2:comparison}). Alternativamente, al nivel de las
\omterm{ex:b2:powerseries:fibonacci}{funciones generatrices}: $G_{X_n}(t) = \bigl(1 +
\frac{\lambda_n(t-1)}{n}\bigr)^n \to e^{\lambda(t - 1)} =
G_{\mathcal{P}(\lambda)}(t)$ para cada $t \in [0, 1]$ fijo; la
convergencia de las \omterm{ex:b2:powerseries:fibonacci}{funciones generatrices}, que (para variables con
valores en $\N$) equivale a la convergencia de cada $\P(X_n = k)$;
véase el~\cref{exo:b2:genfun:9}.
\end{proof}

\begin{remark}
Por eso las \omterm{def:b2:randomvar:law}{leyes} de Poisson modelan recuentos de \omterm{def:b2:proba:space}{sucesos} raros
---erratas por página, desintegraciones radiactivas por segundo,
accidentes diarios en un cruce---: cada oportunidad es casi
despreciable, las oportunidades son muchas y en el límite solo
sobrevive la tasa media $\lambda$.
\end{remark}

\begin{example}[Ver converger el límite de
Poisson]\label{ex:b2:genfun:poissonnumerics}
Fíjese $\lambda = 2$ y sea $X_n \sim \mathcal B(n, 2/n)$. La
probabilidad de ningún \omterm{def:b2:proba:space}{suceso} es exactamente $\P(X_n = 0) = (1 -
2/n)^n$:
\[
n = 10:\ 0.107, \qquad
n = 20:\ 0.122, \qquad
n = 50:\ 0.130, \qquad
n = 100:\ 0.133,
\]
frente al límite $\eu^{-2} \approx 0.135$. La convergencia es
monótona y de velocidad $O(1/n)$ ---desarrollando, $(1 - 2/n)^n =
\eu^{-2}\bigl(1 - \tfrac2n + O(n^{-2})\bigr)$---, de modo que para
$n$ en los cientos el modelo de Poisson ya es preciso hasta la
tercera cifra. Ese es el contenido práctico de la ley de los sucesos
raros: quien modela nunca conoce $n$ y $p$ por separado (¿cuántas
microoportunidades de errata alberga una página?), sino solo su
producto $\lambda$, y la \omterm{def:b2:randomvar:law}{ley} límite, misericordiosamente, no depende
de nada más.
\end{example}

\section{Procesos de ramificación}

Considérese una población que arranca de un único antepasado; cada
individuo, de manera \omterm{def:b2:proba:independence}{independiente}, tiene un número aleatorio de hijos
con \omterm{def:b2:randomvar:law}{ley} $(p_k)_{k \in \N}$ y \omterm{ex:b2:powerseries:fibonacci}{función generatriz} $G$ (la
\emph{\omterm{def:b2:randomvar:law}{distribución} de la descendencia}). Sea $Z_n$ el tamaño de la
generación $n$ ($Z_0 = 1$), y sea $m = G'(1) = \E(Z_1)$ el número
medio de hijos.

\begin{proposition}\label{prop:b2:genfun:branching}
La \omterm{ex:b2:powerseries:fibonacci}{función generatriz} de $Z_n$ es el $n$-ésimo iterado $G_{Z_n} = G
\circ G \circ \dots \circ G$ ($n$ veces), y las probabilidades de
extinción $q_n = \P(Z_n = 0)$ cumplen
\[
q_0 = 0, \qquad q_{n+1} = G(q_n),
\]
y crecen hasta la probabilidad $q$ de extinción final, que es un
punto fijo de $G$.
\end{proposition}

\begin{proof}
La generación $n + 1$ es la suma aleatoria de la descendencia de los
$Z_n$ miembros de la generación $n$, con recuentos \omterm{def:b2:proba:independence}{independientes}
entre sí y de $Z_n$: el~\cref{thm:b2:genfun:compound} da $G_{Z_{n+1}}
= G_{Z_n} \circ G$, y la inducción desde $G_{Z_0}(t) = t$ rinde el
iterado $n$-ésimo, que, por la asociatividad de la composición, puede
leerse igualmente como $G_{Z_{n+1}} = G \circ G_{Z_n}$. Evaluando
esta segunda forma en $0$: $q_{n+1} = G_{Z_{n+1}}(0) =
G\bigl(G_{Z_n}(0)\bigr) = G(q_n)$. Los \omterm{def:b2:proba:space}{sucesos} $\{Z_n = 0\}$ crecen
(las poblaciones extinguidas siguen extinguidas), de modo que $q_n
\uparrow q = \P\bigl(\bigcup_n\{Z_n = 0\}\bigr)$ por la
\omterm{def:b2:metric:continuity}{continuidad} monótona (\cref{thm:b2:proba:continuity}), y la
\omterm{def:b2:metric:continuity}{continuidad} de $G$ sobre $[0, 1]$ convierte $q_{n+1} = G(q_n)$ en $q =
G(q)$ al pasar al límite.
\end{proof}

\begin{example}[Ver converger la extinción]
\label{ex:b2:genfun:cobwebnumerics}
Para la \omterm{def:b2:randomvar:law}{ley} de descendencia $(p_0, p_1, p_2) = (\tfrac14, \tfrac14,
\tfrac12)$ del~\cref{ex:b2:genfun:branchingexample}, $G(t) = \tfrac14
+ \tfrac14t + \tfrac12t^2$ y la iteración $q_{n+1} = G(q_n)$ da
\[
q_1 = 0.25, \quad q_2 = 0.34375, \quad q_3 \approx 0.39502,
\quad q_4 \approx 0.42678, \quad q_5 \approx 0.44776,
\]
subiendo hacia la probabilidad de extinción $q = \tfrac12$. Las
diferencias $q - q_n$ valen $0.25$, $0.156$, $0.105$, $0.073$,
$0.052$: cada una es aproximadamente $\tfrac34$ de la anterior y, en
efecto, el teorema del valor medio da $q - q_{n+1} = G'(c_n)(q -
q_n)$ con $G'(q) = \tfrac14 + q = \tfrac34$. Dos moralejas: un linaje
todavía vivo en la generación $n$ tiene, incorporada en el mismo
cálculo, probabilidad $q - q_n$ de estar condenado más tarde; y el
ritmo de convergencia de la escalera de la figura de más abajo es la
derivada en el punto fijo. El problema de fin de semana convierte
ambas observaciones en teoremas.
\end{example}

\begin{theorem}[Criterio de extinción]\label{thm:b2:genfun:extinction}
Supongamos $p_1 \neq 1$. La probabilidad de extinción $q$ es el punto
fijo \emph{más pequeño} de $G$ en $\intcc{0}{1}$, y:
\begin{itemize}
\item si $m \leq 1$ (subcrítico o crítico), $q = 1$: la extinción es
cierta;
\item si $m > 1$ (supercrítico), $q < 1$: la población sobrevive para
siempre con probabilidad positiva $1 - q$.
\end{itemize}
\end{theorem}

\begin{proof}
$G$ es convexa sobre $\intcc{0}{1}$ (serie de potencias de
coeficientes no negativos: $G'' \geq 0$), no decreciente, y con $G(1)
= 1$.

\emph{Punto fijo más pequeño:} sea $r \in \intcc{0}{1}$ un punto fijo
cualquiera. Entonces $q_0 = 0 \leq r$ y, por inducción, $q_{n+1} =
G(q_n) \leq G(r) = r$ (monotonía): luego $q = \lim q_n \leq r$.

\emph{Caso $m \leq 1$:} supóngase que $r < 1$ es un punto fijo. Por
el teorema del valor medio sobre $[r, 1]$, hay un $c \in
\intoo{r}{1}$ con $G'(c) = \frac{G(1) - G(r)}{1 - r} = \frac{1 - r}{1
- r} = 1$. Pero $G'$ es no decreciente (por convexidad) con
$\lim_{t\to1^-}G'(t) = m \leq 1$, de modo que $G' \leq 1$ sobre
$\intoo{0}{1}$; la igualdad $G'(c) = 1$ fuerza entonces a $G'$ a ser
constante igual a $1$ sobre $\intco{c}{1}$, luego $G'' = \sum
n(n-1)p_nt^{n-2} \equiv 0$ ahí. Una serie de potencias de
coeficientes no negativos que se anula sobre un intervalo tiene todos
esos coeficientes nulos: $p_n = 0$ para $n \geq 2$, luego $G(t) = p_0
+ p_1t$ y $1 = G'(c) = p_1$, en contradicción con la hipótesis $p_1
\neq 1$. Así pues, $1$ es el único punto fijo: $q = 1$.

\emph{Caso $m > 1$:} cerca de $1$, $G(t) - t$ tiene derivada $G'(t) -
1 \to m - 1 > 0$ cuando $t \to 1^-$, de modo que $G(t) - t < G(1) - 1
= 0$ sobre algún intervalo $\intoo{1 - \delta}{1}$: la función
\omterm{def:b2:metric:continuity}{continua} $G(t) - t$ es $\geq 0$ en $t = 0$ ($G(0) = p_0 \geq 0$) y $<
0$ justo por debajo de $1$, así que se anula en algún $r < 1$
(teorema del valor intermedio). El punto fijo más pequeño cumple
entonces $q \leq r < 1$.
\end{proof}

\begin{remark}[Cómo leer la telaraña]
En la figura, un movimiento vertical aplica $G$ (de $(q_n, q_n)$ a
$(q_n, G(q_n))$) y un movimiento horizontal hasta la diagonal
convierte la salida en entrada: la escalera \emph{es} la recursión
$q_{n+1} = G(q_n)$. La convexidad de $G$ y $G(1) = 1$ dejan solo dos
geometrías. O bien la curva se mantiene por encima de la diagonal
sobre $\intco01$ (media $m \leq 1$): la escalera no tiene dónde
detenerse antes de $1$. O bien la curva cruza en algún $q < 1$ ($m >
1$): la escalera queda atrapada por debajo del cruce y converge a él,
al ritmo geométrico $G'(q) < 1$ cuantificado en el~\cref{ex:b2:genfun:cobwebnumerics}. Todo el análisis del teorema de
extinción es visible en esta única imagen; por eso vale la pena
dibujarla antes de calcular.
\end{remark}

\begin{example}\label{ex:b2:genfun:branchingexample}
\omterm{def:b2:randomvar:law}{Ley} de descendencia: ningún hijo, un hijo o dos hijos con
probabilidades $\frac14, \frac14, \frac12$. Entonces $m = \frac14 + 1
= \frac54 > 1$ y $G(t) = \frac14 + \frac14 t + \frac12 t^2$. Puntos
fijos: $\frac12 t^2 - \frac34 t + \frac14 = 0$, es decir, $2t^2 - 3t
+ 1 = (2t - 1)(t - 1) = 0$: $q = \frac12$. El linaje se extingue con
probabilidad $\frac12$ y, con probabilidad $\frac12$, vive para
siempre.
\end{example}

\begin{remark}[Perspectivas dentro de este volumen]
El capítulo es la encrucijada del libro, y cada ingrediente llegó de
un lugar con nombre: el álgebra de series, del~\cref{ch:b2:series} y
del~\cref{ch:b2:powerseries}; la probabilidad, del~\cref{ch:b2:proba} (la \omterm{def:b2:metric:continuity}{continuidad} monótona demuestra $q_n \uparrow
q$) y del~\cref{ch:b2:randomvar} ($G_X = \E(t^X)$ es una
\omterm{def:b2:randomvar:expectation}{esperanza}, y la multiplicatividad es el teorema del producto); y la
convexidad, del~\cref{ch:b2:realfun} a través del~\cref{ch:b2:affine}. Hasta las patologías de cola pesada conectan: la
variable de San Petersburgo del capítulo anterior tiene $G(t) =
\sum_k2^{-k}t^{2^k}$, una serie perfectamente convergente sobre
$\intcc01$ cuya derivada en $1^-$ diverge; media infinita, visible de
un vistazo. Un solo objeto, todas las herramientas del año: un último
capítulo apropiado.
\end{remark}

\begin{remark}[Errores frecuentes]
(i) Las \omterm{ex:b2:powerseries:fibonacci}{funciones generatrices} solo se aplican a variables con
valores en $\N$: para variables con signo o no enteras, el objeto
$\E(t^X)$ pierde su estructura de serie de potencias (el tercer año
lo sustituye por transformadas adaptadas a $\R$). (ii) La primera
comprobación de sensatez de toda $G$ calculada es $G(1) = 1$; la
segunda, que los coeficientes sean no negativos: un coeficiente
negativo significa un desliz algebraico, no una \omterm{def:b2:randomvar:law}{ley} nueva. (iii) En
las sumas aleatorias importa el orden de composición: $G_S = G_N
\circ G_X$, con la función \emph{exterior} contando los términos;
componer al revés carece de sentido ($G_X \circ G_N$ contaría
elementos de elementos). (iv) La multiplicatividad necesita
\omterm{def:b2:proba:independence}{independencia} y fuentes de aleatoriedad distintas: $G_{2X}(t) =
G_X(t^2)$, y no $G_X(t)^2$. (v) Derivar en $1$ es una operación de
frontera: cuando el radio es exactamente $1$, como en el~\cref{ex:b2:genfun:heavytail}, $G'(1^-)$ puede ser infinita, y la
formulación con límite monótono del teorema de los momentos no es una
sutileza pedante, sino el enunciado honesto.
\end{remark}

\section*{Cierre del volumen}

La \omterm{ex:b2:powerseries:fibonacci}{función generatriz} es un objeto final apropiado para este libro:
es simultáneamente una serie de potencias (\cref{ch:b2:powerseries}),
una herramienta de \omterm{def:b2:series:summable}{familias sumables} (\cref{ch:b2:series}), una
\omterm{def:b2:randomvar:expectation}{esperanza} (\cref{ch:b2:randomvar}), una función convexa cuya geometría
decide la extinción (\cref{ch:b2:realfun}) y una iteración de punto
fijo (\cref{ch:b2:metric}). La matemática del segundo año es una sola
materia. El volumen del tercer año abrirá las puertas deliberadamente
dejadas cerradas aquí: la integración de Lebesgue (saldando el teorema
de convergencia dominada del~\cref{ch:b2:integration}), la
probabilidad en el marco de la teoría de la medida sobre espacios no
\omterm{def:b2:structures:countable}{numerables} y la demostración completa del teorema de la función
inversa (\cref{ch:b2:diffcalc}) en el marco de la geometría
diferencial.

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:genfun:1}
Calcula la \omterm{ex:b2:powerseries:fibonacci}{función generatriz} de la \omterm{def:b2:randomvar:law}{ley} uniforme sobre $\{1, 2,
\dots, 6\}$ (un dado equilibrado). Prueba que la suma de dos dados
equilibrados \emph{no puede} ser uniforme sobre $\{2, \dots, 12\}$:
factoriza $G_{X+Y}$ y cuenta raíces. \emph{(Una suma uniforme
forzaría $G_X(t)G_Y(t) = \frac{t^2}{11}\sum_{k=0}^{10}t^k$, cuyas
raíces no nulas son las raíces $11$-ésimas de la unidad distintas de
$1$ ---ninguna de ellas real---, mientras que $G_X/t$ y $G_Y/t$ son
polinomios reales de grado $5$, cada uno con al menos una raíz
real.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:genfun:2}
Usando \omterm{ex:b2:powerseries:fibonacci}{funciones generatrices}, recupera $\E$ y $V$ para las \omterm{def:b2:randomvar:law}{leyes}
binomial y geométrica (\cref{thm:b2:genfun:moments}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:genfun:3}
Dos dados trucados: ¿es posible trucar dos dados (de manera
\omterm{def:b2:proba:independence}{independiente}, igual o no) para que su suma sea uniforme sobre $\{2,
\dots, 12\}$? \emph{(La misma obstrucción de factorización del~\cref{exo:b2:genfun:1}: la respuesta es no ni siquiera con trucajes
distintos, porque cada factor $G_X(t)/t$ tiene grado impar $5$ y, por
tanto, una raíz real, mientras que el objetivo no tiene ninguna.)}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:genfun:4}
Sean $X_1, X_2, \dots$ de Bernoulli $\mathcal{B}(p)$ \omterm{def:b2:proba:independence}{independientes} y
$N \sim \mathcal{P}(\lambda)$ \omterm{def:b2:proba:independence}{independiente} de ellas. Prueba, vía el~\cref{thm:b2:genfun:compound}, que $S = X_1 + \dots + X_N \sim
\mathcal{P}(\lambda p)$: un número de Poisson de elementos, cada uno
conservado con probabilidad $p$, deja un número de Poisson; el
\emph{adelgazamiento}. Calcula también la \omterm{def:b2:randomvar:law}{ley} del recuento
descartado y admira: es $\mathcal{P}(\lambda(1-p))$, y puede probarse
que es independiente de $S$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:genfun:5}
(Binomial negativa) Sea $T_r$ el número de lanzamientos para obtener
$r$ caras (con probabilidad de cara $p$). Escribe $T_r$ como suma de
$r$ variables geométricas \omterm{def:b2:proba:independence}{independientes}, deduce
\[
G_{T_r}(t) = \Bigl(\frac{pt}{1 - (1-p)t}\Bigr)^{r},
\qquad
\E(T_r) = \frac rp,
\qquad
V(T_r) = \frac{r(1-p)}{p^2},
\]
y desarrolla $G_{T_r}$ para hallar $\P(T_r = n) = \binom{n-1}{r-1}
p^r(1-p)^{n-r}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:genfun:6}
Para la \omterm{def:b2:randomvar:law}{ley} de descendencia $p_0 = \frac18$, $p_1 = \frac38$, $p_2 =
\frac38$, $p_3 = \frac18$: calcula $m$, decide la supercriticalidad y
calcula exactamente la probabilidad de extinción $q$. \emph{(Sáquese
factor la raíz $t = 1$ de $G(t) - t$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:7}
(Descendencia total) En un \omterm{pb:b2:genfun:1}{proceso de ramificación} subcrítico ($m <
1$), sea $Y = \sum_{n\geq0} Z_n$ el número total de individuos
nacidos alguna vez. Prueba que $\E(Y) = \sum_n m^n = \frac{1}{1 - m}$
(justifica el intercambio de sumaciones) y demuestra que la
\omterm{ex:b2:powerseries:fibonacci}{función generatriz} $H = G_Y$ satisface la ecuación funcional $H(t) =
t\,G(H(t))$. \emph{(El antepasado, más las descendencias totales de
cada uno de sus hijos, que son copias independientes de $Y$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:8}
Sea $X$ con \omterm{ex:b2:powerseries:fibonacci}{función generatriz} $G$ de \omterm{def:b2:powerseries:radius}{radio de convergencia} $> 1$.
Demuestra la \emph{cota exponencial de la cola}: existen $C > 0$ y
$\rho \in \intoo{0}{1}$ con $\P(X \geq n) \leq C\rho^n$. \emph{(Markov
aplicado a $t^X$ para un $t > 1$ fijo dentro del disco.)}
Recíprocamente, prueba que si $\P(X \geq n) \leq C\rho^n$ con $\rho <
1$, el radio de $G$ es $\geq 1/\rho > 1$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:9}
(Teorema de \omterm{def:b2:metric:continuity}{continuidad}, caso elemental) Sean $X, X_1, X_2, \dots$ con
valores en $\N$ y tales que $G_{X_n}(t) \to G_X(t)$ para todo $t \in
\intco{0}{1}$. Prueba que $\P(X_n = k) \to \P(X = k)$ para todo $k$.
\emph{(Inducción sobre $k$: para $k = 0$, hágase $t \to 0$, pero con
cuidado: fíjese $t$ pequeño y úsese $\abs{\P(X_n = 0) - G_{X_n}(t)}
\leq \frac{t}{1-t}$, válido porque la cola $\sum_{j \geq 1}p_jt^j \leq
\frac{t}{1 - t}$; después, diagonalícese. Para el paso de inducción,
considérese $\frac{G(t) - \P(X = 0)}{t}$, la \omterm{ex:b2:powerseries:fibonacci}{función generatriz} de
una \omterm{def:b2:randomvar:law}{ley} desplazada.)}
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:genfun:10}
(El truco de la paridad) Prueba que, para una variable $X$ con
valores en $\N$,
\[
\P(X \text{ par}) = \frac{1 + G_X(-1)}{2} ,
\]
y calcula esta probabilidad para $X \sim \mathcal P(\lambda)$ y para
$X \sim \mathcal B(n, p)$. ¿Qué significa probabilísticamente $G_X(-1)
\to 0$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:genfun:11}
(Dados de Sicherman) Verifica la factorización de la
\omterm{ex:b2:powerseries:fibonacci}{función generatriz} del dado equilibrado
\[
\frac{t + t^2 + \dots + t^6}{6}
= \frac{t\,(1 + t)(1 + t + t^2)(1 - t + t^2)}{6},
\]
y prueba que los dos dados de caras $\{1, 2, 2, 3, 3, 4\}$ y $\{1, 3,
4, 5, 6, 8\}$ tienen \omterm{ex:b2:powerseries:fibonacci}{funciones generatrices}
$\frac{t(1+t)(1+t+t^2)}6$ y $\frac{t(1+t)(1+t+t^2)(1-t+t^2)^2}6$,
cuyo producto es el de dos dados estándar: esos dados exóticos
producen cada total $2, \dots, 12$ exactamente con las probabilidades
estándar.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:genfun:12}
(Esperando dos caras seguidas) Se lanza una moneda con probabilidad
de cara $p$ hasta que aparecen dos caras consecutivas; sea $T$ el
número de lanzamientos (el juego del~\cref{exo:b2:proba:6}).
Condicionando a los primeros lanzamientos, deduce un sistema lineal
para las \omterm{ex:b2:powerseries:fibonacci}{funciones generatrices} desde los estados ``sin cara actual''
y ``con una cara actual'', y concluye que
\[
G_T(t) = \frac{p^2t^2}{1 - qt - pqt^2}
\qquad (q = 1 - p);
\]
comprueba $G_T(1) = 1$ y $\E(T) = \dfrac{1 + p}{p^2}$ ($= 6$ para una
moneda equilibrada).
\end{exercise}

\section{Problema: el proceso de Galton--Watson, resuelto}

\begin{problem}[{Problema de fin de semana --- ritmos de
crecimiento, soluciones exactas, descendencia total y la estimación
crítica de Kolmogórov}]\label{pb:b2:genfun:1}
El criterio de extinción (\cref{thm:b2:genfun:extinction}) reparte
los \omterm{pb:b2:genfun:1}{procesos de ramificación}\index{proceso de ramificación} en
subcríticos, críticos y supercríticos; pero no dice nada de los
\emph{ritmos}: cuán deprisa muere un linaje condenado, cuánto crece
uno que sobrevive. Este problema los calcula. Mantenemos la notación
del capítulo: \omterm{def:b2:randomvar:law}{ley} de descendencia $(p_k)$ con \omterm{ex:b2:powerseries:fibonacci}{función generatriz}
$G$, media $m = G'(1)$, tamaños de generación $Z_n$ ($Z_0 = 1$),
iterados $G_n = G_{Z_n}$ y probabilidades de extinción $q_n = \P(Z_n
= 0) \uparrow q$; suponemos siempre $p_1 \neq 1$ y, donde aparezcan
momentos de orden dos, $G''(1) < \infty$, y escribimos $\sigma^2 =
V(Z_1)$.

\medskip
\noindent\textbf{Parte I --- Momentos de las generaciones.}
\begin{enumerate}
  \item Prueba que $\E(Z_n) = m^n$ \emph{(regla de la cadena sobre
        $G_n = G \circ G_{n-1}$ en $1^-$, usando $G_{n-1}(1) = 1$ y
        el~\cref{thm:b2:genfun:moments})}.
  \item Establece la recursión $G_n''(1) = G''(1)\,m^{2(n-1)} +
        m\,G_{n-1}''(1)$ y resuélvela: $G_n''(1) =
        G''(1)\,m^{n-1}\dfrac{m^n - 1}{m - 1}$ para $m \neq 1$, y
        $G_n''(1) = n\,G''(1)$ para $m = 1$.
  \item Deduce que
        \[
        V(Z_n) = \sigma^2m^{n-1}\,\frac{m^n - 1}{m - 1}
        \quad (m \neq 1),
        \qquad
        V(Z_n) = n\,\sigma^2 \quad (m = 1).
        \]
  \item (Ritmo subcrítico, cota superior) Para $m < 1$, prueba que
        $\P(Z_n > 0) \leq m^n$ \emph{(Markov sobre la $Z_n$ de
        valores enteros)}: la extinción es cierta con ritmo
        geométrico; un refinamiento cuantitativo del criterio del
        capítulo.
  \item (Ritmo subcrítico, cota inferior) Usando Cauchy--Schwarz
        sobre $Z_n\mathbf 1_{Z_n > 0}$, prueba que
        \[
        \P(Z_n > 0) \geq \frac{\E(Z_n)^2}{\E(Z_n^2)}
        \geq c\,m^{n}
        \quad\text{con}\quad
        c = \Bigl(\frac{\sigma^2}{m(1-m)} + 1\Bigr)^{-1} :
        \]
        el ritmo geométrico $m^n$ es exacto salvo constantes.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La familia geométrica, resuelta
exactamente.} Sea la \omterm{def:b2:randomvar:law}{ley} de descendencia geométrica sobre $\N$: $p_k
= qp^k$ ($k \geq 0$), con $0 < p < 1$ y $q = 1 - p$.
\begin{enumerate}[resume]
  \item Calcula $G(t) = \dfrac{q}{1 - pt}$ y $m = \dfrac pq$; sitúa
        los tres regímenes en términos de $p$.
  \item Resuelve $G(t) = t$: prueba que los puntos fijos son $1$ y
        $q/p = 1/m$, y recupera la probabilidad de extinción
        $q_{\mathrm{ext}} = \min(1, 1/m)$.
  \item Demuestra por inducción las \omterm{def:b2:multint:exact}{formas cerradas}
        \[
        q_n = \frac{m^n - 1}{m^{n+1} - 1} \quad (m \neq 1),
        \qquad
        q_n = \frac{n}{n+1} \quad (m = 1).
        \]
  \item Deduce los ritmos exactos: $1 - q_n \sim (1 - m)\,m^n$ en el
        caso subcrítico, y $q_{\mathrm{ext}} - q_n \sim \dfrac{m -
        1}{m^{2}}\cdot m^{-n}$ en el supercrítico; comprueba que la
        razón de contracción supercrítica es $G'(q_{\mathrm{ext}}) =
        1/m$.
  \item Caso crítico ($p = \tfrac12$): calcula $\sigma^2 = 2$ y
        obsérvese que $1 - q_n = \frac1{n+1}$: la supervivencia decae
        como $\frac1n$; ni geométricamente ni de manera \omterm{def:b2:series:summable}{sumable}.
  \item Todavía en el caso crítico: demuestra por inducción el
        iterado \omterm{def:b2:metric:complete}{completo}
        \[
        G_n(t) = \frac{n - (n-1)t}{n + 1 - nt},
        \]
        y deduce que, condicionada a la supervivencia, $Z_n$ es
        geométrica sobre $\N^*$ de parámetro $\frac1{n+1}$:
        \[
        \P(Z_n = k \mid Z_n > 0) = \frac1{n+1}
        \Bigl(\frac{n}{n+1}\Bigr)^{k-1},
        \qquad
        \E(Z_n \mid Z_n > 0) = n + 1 .
        \]
        El linaje medio muere, pero los que sobreviven tienen tamaño
        de orden $n$.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Descendencia total.} Sea $Y =
\sum_{n\geq0}Z_n \in \N^* \cup \{\infty\}$ el número total de
individuos nacidos alguna vez, y $H(t) = \sum_{k\geq1}\P(Y =
k)t^k$.
\begin{enumerate}[resume]
  \item Justifica que $\P(Y < \infty) = q_{\mathrm{ext}}$, y recuerda
        del~\cref{exo:b2:genfun:7} la ecuación funcional $H(t) =
        t\,G(H(t))$ (cuya deducción no usaba $m < 1$).
  \item (Ramificación binaria) Para $p_0 = p_2 = \frac12$ (crítico),
        resuelve la ecuación funcional:
        \[
        H(t) = \frac{1 - \sqrt{1 - t^2}}{t},
        \]
        y desarróllala con el~\cref{ex:b2:powerseries:catalan} para
        obtener
        \[
        \P(Y = 2k + 1) = \frac{C_k}{2^{2k+1}},
        \qquad C_k = \frac1{k+1}\binom{2k}k ;
        \]
        comprueba los valores $\P(Y = 1) = \frac12$ y $\P(Y = 3) =
        \frac18$ por recuento directo.
  \item Derivando la ecuación funcional en $1^-$, prueba que $\E(Y) =
        \frac{1}{1-m}$ para $m < 1$, mientras que la criticidad
        fuerza $\E(Y) = \infty$: la descendencia total crítica es
        finita casi seguramente, con media infinita.
  \item Con las asintóticas del coeficiente binomial central
        (\cref{ex:b2:comparison:centralbinomial}), prueba que
        \[
        \P(Y = 2k+1) \sim \frac{1}{2\sqrt\pi\,k^{3/2}},
        \]
        una cola pesada $k^{-3/2}$, y deduce que $\P(Y > n) \asymp
        n^{-1/2}$ (bastan cotas superior e inferior de ese orden).
  \item Compara con el \omterm{pb:b2:proba:1}{paseo aleatorio} equilibrado (el problema de
        fin de semana del~\cref{ch:b2:proba}): allí, tiempos de
        retorno ciertos y de media infinita; aquí, descendencia total
        cierta y de media infinita, ambos con leyes locales
        $n^{-3/2}$. Un párrafo sobre por qué la criticidad produce
        esta firma.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La estimación de Kolmogórov en la
criticidad.} Supongamos $m = 1$ y $0 < \sigma^2 = G''(1) < \infty$.
\begin{enumerate}[resume]
  \item Prueba que $G''$ se extiende de manera \omterm{def:b2:metric:continuity}{continua} a $\intcc01$
        \emph{(no negativa, creciente y con límite finito)} y deduce
        el desarrollo de Taylor en $1$:
        \[
        G(t) = t + b\,(1-t)^2 + o\bigl((1-t)^2\bigr),
        \qquad b = \frac{G''(1)}2 = \frac{\sigma^2}2 .
        \]
  \item Para $t \in \intco01$, póngase $h(t) = \dfrac1{1 - G(t)} -
        \dfrac1{1 - t}$. Prueba que
        \[
        h(t) = \frac{G(t) - t}{(1 - G(t))(1 - t)}
        \xrightarrow[t\to1^-]{} b .
        \]
  \item Telescopa a lo largo de la iteración $q_{j+1} = G(q_j)$:
        \[
        \frac1{1 - q_n} = 1 + \sum_{j=0}^{n-1}h(q_j),
        \]
        y concluye con un argumento de Cesàro que
        \[
        \P(Z_n > 0) = 1 - q_n \sim \frac{2}{\sigma^2\,n}
        \]
        ---la \emph{estimación de Kolmogórov}: todo
        \omterm{pb:b2:genfun:1}{proceso de ramificación} crítico muere al ritmo universal $1/n$,
        y solo la constante recuerda la \omterm{def:b2:randomvar:law}{ley} de descendencia.
  \item Comprueba la estimación contra el caso geométrico crítico de
        la pregunta 10.
  \item Deduce que $\E(Z_n \mid Z_n > 0) = \dfrac{1}{1 - q_n} \sim
        \dfrac{\sigma^2 n}{2}$ \emph{(obsérvese que $\E(Z_n \mathbf
        1_{Z_n>0}) = \E(Z_n) = 1$)}, y contrástalo con la pregunta
        11: condicionada a la supervivencia, la población crece
        \emph{linealmente}; la cuerda floja crítica entre la muerte y
        la explosión.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Aplicaciones y síntesis.}
\begin{enumerate}[resume]
  \item (Epidemias, reacciones en cadena) Para una \omterm{def:b2:randomvar:law}{ley} de
        descendencia de Poisson $\mathcal P(\lambda)$ ---cada caso
        contagia a $\mathcal P(\lambda)$ casos nuevos---, escribe la
        ecuación de extinción $q = \eu^{\lambda(q-1)}$ y resuélvela
        numéricamente para $\lambda = 1.5$ ($q \approx 0.417$) y
        $\lambda = 2$ ($q \approx 0.203$): partiendo de un solo caso,
        un brote grande \emph{no} es seguro ni siquiera cuando
        $\lambda > 1$. Explica por qué la iteración $q_{n+1} =
        \eu^{\lambda(q_n - 1)}$ desde $q_0 = 0$ converge a la raíz
        correcta.
  \item Partiendo de $k$ antepasados en vez de uno, prueba que la
        probabilidad de extinción es $q^k$. Aplicación: con $\lambda
        = 1.5$, ¿cuántos casos iniciales hacen que un brote sea al
        menos un $99\,\%$ probable?
  \item (Condicionar a la extinción un proceso supercrítico) Para $m
        > 1$ con probabilidad de extinción $q \in \intoo01$:
        demuestra primero, por convexidad, que $G'(q) < 1$ en el
        punto fijo más pequeño, y deduce que $q_{\mathrm{ext}} - q_n
        = O\bigl(G'(q)^n\bigr)$ (convergencia geométrica, como
        ejemplificaba la pregunta 9). Prueba después que $\widehat
        G(t) = G(qt)/q$ es la \omterm{ex:b2:powerseries:fibonacci}{función generatriz} de una \omterm{def:b2:randomvar:law}{ley} de
        descendencia legítima, de media $\widehat m = G'(q) < 1$: un
        proceso compañero subcrítico. Verifícalo sobre la familia
        geométrica: condicionar a la extinción el proceso
        supercrítico $(p, q)$ intercambia $p$ y $q$. (El enunciado
        \omterm{def:b2:metric:complete}{completo} ---que el proceso condicionado \emph{es} el proceso
        compañero--- se demuestra en el volumen del tercer año; aquí
        has verificado su sombra en las \omterm{ex:b2:powerseries:fibonacci}{funciones generatrices}.)
  \item Síntesis: redacta la tabla de la tricotomía ---para $m < 1$,
        $m = 1$ y $m > 1$: valor de $q$; ritmo de $\P(Z_n > 0)$ o de
        $q - q_n$; $\E(Y)$; tamaño de una generación que sobrevive.
        Enuncia, en una frase por herramienta, cómo la composición de
        \omterm{ex:b2:powerseries:fibonacci}{funciones generatrices}, la convexidad, Taylor en $1^-$ y el
        promedio de Cesàro sostuvieron todo el problema, y qué añade
        el volumen del tercer año (la martingala $Z_n/m^n$ y la \omterm{def:b2:randomvar:law}{ley}
        límite exponencial de Yaglom).
\end{enumerate}
\end{problem}
