\chapter{Topología de los espacios métricos}\label{ch:b2:metric}

La topología de la recta real (volumen del primer año) se generaliza,
casi sin cambiar una palabra, a cualquier conjunto dotado de una
distancia. La ganancia es enorme: sucesiones de funciones, matrices,
curvas ---todas pasan a ser puntos de \omterm{def:b2:metric:def}{espacios métricos}, y los tres
pilares que aquí se demuestran ---la \omterm{def:b2:metric:complete}{completitud} con el teorema del
punto fijo de Banach, la \omterm{def:b2:metric:compact}{compacidad} y la conexidad--- se les
aplican de manera uniforme. Este capítulo es la columna vertebral de
toda la mitad de análisis del libro.

\section{Espacios métricos}

\begin{definition}\label{def:b2:metric:def}
Un \emph{espacio métrico}\index{espacio métrico} es un conjunto $X$
dotado de una aplicación $d \colon X \times X \to \R_+$ tal que, para
todos $x, y, z$:
\[
d(x,y) = 0 \iff x = y,
\qquad
d(x,y) = d(y,x),
\qquad
d(x,z) \leq d(x,y) + d(y,z).
\]
Bolas: $B(a, r) = \{x : d(a,x) < r\}$ (\omterm{def:b2:metric:topology}{abierta}), $\overline B(a,r) =
\{x : d(a,x) \leq r\}$ (cerrada). Un subconjunto $A \subseteq X$ pasa
a ser un espacio métrico con la distancia inducida.
\end{definition}

\begin{example}\label{ex:b2:metric:examples}
$\R$ con $\abs{x - y}$; $\R^n$ con cualquiera de las distancias
\[
d_1(x,y) = \sum_i \abs{x_i - y_i},
\quad
d_2(x,y) = \Bigl(\sum_i (x_i - y_i)^2\Bigr)^{1/2},
\quad
d_\infty(x,y) = \max_i \abs{x_i - y_i};
\]
el conjunto $C(\intcc{a}{b})$ de las funciones \omterm{def:b2:metric:continuity}{continuas} con la
\emph{distancia del supremo} $d_\infty(f, g) = \sup_{\intcc{a}{b}}
\abs{f - g}$ (finita, pues $f - g$ está acotada); cualquier conjunto
con la distancia \emph{discreta} ($d(x,y) = 1$ para $x \neq y$). Las
distancias procedentes de normas son el objeto del~\cref{ch:b2:nvs}.
\end{example}

\begin{definition}[Topología de un espacio métrico]\label{def:b2:metric:topology}
$U \subseteq X$ es \emph{abierto}\index{abierto!en un espacio métrico}
cuando todo punto de $U$ es centro de una bola contenida en $U$; $F$
es \emph{cerrado} cuando su complementario es abierto. Los entornos,
el interior, la clausura, la densidad y la frontera se definen
exactamente como en la recta real (volumen del primer año), con bolas
en lugar de intervalos, y los enunciados allí demostrados ---uniones e
intersecciones de abiertos, caracterizaciones del interior y de la
clausura, la clausura como el menor cerrado que contiene al
conjunto--- se trasladan con las mismas demostraciones. Las bolas
abiertas son abiertas y las cerradas, cerradas (desigualdad
triangular).
\end{definition}

\begin{example}[Interior, clausura y frontera de un mismo conjunto]\label{ex:b2:metric:intclosure}
En $\R$, sea $A = \intoc{0}{1} \cup \{2\}$. Interior:
$\intoo{0}{1}$ ---alrededor de cualquier $x \in \intoo01$ una bola
pequeña se queda dentro de $A$; alrededor de $1$, toda bola
$\intoo{1-r}{1+r}$ se sale de $A$ por la derecha, luego $1$ no es
interior; y el punto aislado $2$ tampoco lo es---. Clausura:
$\intcc{0}{1} \cup \{2\}$ (el punto $0$ es límite de puntos de $A$
y no se añade nada más). Frontera (clausura menos interior): $\{0,
1, 2\}$. Conviene recordar las asimetrías: un extremo puede
pertenecer al conjunto sin ser interior ($1$), puede ser adherente
sin pertenecer ($0$), y un punto aislado es su propia frontera
($2$). La misma contabilidad funciona literalmente en cualquier
\omterm{def:b2:metric:def}{espacio métrico}, con bolas en lugar de intervalos.
\end{example}

\begin{definition}[Límites, continuidad]\label{def:b2:metric:continuity}
$x_n \to x$ en $X$ cuando $d(x_n, x) \to 0$. Una aplicación $f \colon
X \to Y$ entre \omterm{def:b2:metric:def}{espacios métricos} es
\emph{continua}\index{continua} en $a$ cuando
\[
\forall \varepsilon > 0,\ \exists\delta > 0,\quad
d_X(x, a) \leq \delta \implies d_Y\bigl(f(x), f(a)\bigr) \leq
\varepsilon ;
\]
equivalentemente (misma demostración que en $\R$), $f(x_n) \to f(a)$
para toda sucesión $x_n \to a$. $f$ es
\emph{lipschitziana}\index{lipschitziana} de constante $k$ cuando
$d_Y(f(x), f(y)) \leq k\, d_X(x, y)$ siempre; entonces es
uniformemente continua y, en particular, continua.
\end{definition}

\begin{theorem}[Caracterización global de la continuidad]\label{thm:b2:metric:globalcontinuity}
$f \colon X \to Y$ es \omterm{def:b2:metric:continuity}{continua} (en todo punto) si y solo si la imagen
recíproca de todo \omterm{def:b2:metric:topology}{abierto} es \omterm{def:b2:metric:topology}{abierta}, si y solo si la imagen
recíproca de todo cerrado es cerrada.
\end{theorem}

\begin{proof}
($\Rightarrow$) Sea $V \subseteq Y$ \omterm{def:b2:metric:topology}{abierto} y $a \in f^{-1}(V)$:
alguna bola $B(f(a), \varepsilon) \subseteq V$; la \omterm{def:b2:metric:continuity}{continuidad} en $a$
proporciona un $\delta$ con $f(B(a, \delta)) \subseteq B(f(a),
\varepsilon)$, luego $B(a, \delta) \subseteq f^{-1}(V)$.

($\Leftarrow$) Dados $a$ y $\varepsilon$, el conjunto
$f^{-1}\bigl(B(f(a), \varepsilon)\bigr)$ es \omterm{def:b2:metric:topology}{abierto} y contiene a $a$,
luego contiene una bola $B(a, \delta)$: esa es la definición de
\omterm{def:b2:metric:continuity}{continuidad} en $a$. Los cerrados: por paso al complementario
(\cref{prop:b2:structures:images}).
\end{proof}

\section{Espacios completos}

\begin{definition}\label{def:b2:metric:complete}
Una sucesión $(x_n)$ es \emph{de Cauchy} cuando $\sup_{p, q \geq N}
d(x_p, x_q) \to 0$ al tender $N \to \infty$. Un \omterm{def:b2:metric:def}{espacio métrico} es
\emph{completo}\index{espacio métrico completo} cuando toda sucesión
de Cauchy converge. Convergente $\Rightarrow$ de Cauchy siempre; los
subconjuntos cerrados de un espacio completo son completos, y los
subconjuntos completos de cualquier espacio son cerrados (mismas
demostraciones que en $\R$: volumen del primer año).
\end{definition}

\begin{example}[De Cauchy pero sin límite]\label{ex:b2:metric:cauchynolimit}
En $X = \Q$ con la distancia usual, las truncaciones decimales de
$\sqrt2$,
\[
x_0 = 1,\quad x_1 = 1.4,\quad x_2 = 1.41,\quad x_3 = 1.414,
\quad\dots
\]
cumplen $\abs{x_p - x_q} \leq 10^{-\min(p,q)}$: son de Cauchy en
$\Q$. Un límite en $\Q$ sería también el límite en $\R$, a saber
$\sqrt2 \notin \Q$: no existe límite en $X$. La incompletitud es la
presencia de esos ``límites fantasma''; la \omterm{def:b2:metric:complete}{completitud} de $\R$ se
diseñó en el volumen del primer año precisamente para dar hogar a
toda sucesión de Cauchy.
\end{example}

\begin{theorem}\label{thm:b2:metric:rncomplete}
$\R^n$ (con cualquiera de las tres distancias del~\cref{ex:b2:metric:examples}) y $\bigl(C(\intcc{a}{b}),
d_\infty\bigr)$ son \omterm{def:b2:metric:complete}{completos}.
\end{theorem}

\begin{proof}
$\R^n$: una sucesión de Cauchy lo es en cada coordenada (pues
$\abs{x_i - y_i} \leq d(x,y)$ para las tres distancias), así que cada
coordenada converge (\omterm{def:b2:metric:complete}{completitud} de $\R$, volumen del primer año), y
la convergencia coordenada a coordenada implica la convergencia para
$d_\infty$ (hay un número finito de coordenadas) y por tanto para las
tres (las tres distancias se dominan mutuamente salvo constantes:
$d_\infty \leq d_2 \leq d_1 \leq n\,d_\infty$).

$C(\intcc{a}{b})$: sea $(f_n)$ de Cauchy para $d_\infty$. Para cada
$x$, la sucesión $(f_n(x))$ es de Cauchy en $\R$ (pues $\abs{f_p(x) -
f_q(x)} \leq d_\infty(f_p, f_q)$) y converge a cierto $f(x)$. Pasando
al límite en $\abs{f_p(x) - f_q(x)} \leq \varepsilon$ (válido para
$p, q \geq N_\varepsilon$ y todo $x$) cuando $q \to \infty$:
$\abs{f_p(x) - f(x)} \leq \varepsilon$ para todo $x$, es decir,
$d_\infty(f_p, f) \leq \varepsilon$: convergencia uniforme. El límite
es \omterm{def:b2:metric:continuity}{continuo}: dado $\varepsilon$, tómese $p$ con $\sup\abs{f_p - f}
\leq \varepsilon$ y úsese después la \omterm{def:b2:metric:continuity}{continuidad} de $f_p$ en $a$ junto
con la descomposición en tres términos
\[
\abs{f(x) - f(a)} \leq \abs{f(x) - f_p(x)} + \abs{f_p(x) - f_p(a)}
+ \abs{f_p(a) - f(a)} \leq 3\varepsilon
\]
para $x$ próximo a $a$. (Este ``argumento de los $3\varepsilon$''
reaparece como teorema del límite uniforme en el~\cref{ch:b2:funcseq}.)
\end{proof}

\begin{example}[Abiertos y cerrados reconocidos por continuidad]\label{ex:b2:metric:recognize}
La caracterización global (\cref{thm:b2:metric:globalcontinuity})
es la herramienta cotidiana para la contabilidad topológica. En
$\R^2$: el conjunto $\{(x, y) : x^2 + y^2 < 1,\ y > x^3\}$ es
\omterm{def:b2:metric:topology}{abierto}, pues es $g^{-1}(\intoo{-\infty}{1}) \cap
h^{-1}(\intoo{0}{+\infty})$ para las funciones \omterm{def:b2:metric:continuity}{continuas} $g(x,y) =
x^2 + y^2$ y $h(x, y) = y - x^3$: intersección de dos imágenes
recíprocas \omterm{def:b2:metric:topology}{abiertas}. En $\bigl(C(\intcc01), d_\infty\bigr)$: el
conjunto de las funciones con $f(0) = f(1)$ y $\int_0^1 f = 0$ es
cerrado, pues es la imagen recíproca de $\{(0,0)\}$ por la
aplicación \omterm{def:b2:metric:continuity}{continua} $f \mapsto \bigl(f(0) - f(1),\ \int_0^1
f\bigr)$ con valores en $\R^2$ (cada coordenada es
$1$-lipschitziana, como en el~\cref{exo:b2:metric:3}). El método no
dibuja nunca: se exhibe una aplicación \omterm{def:b2:metric:continuity}{continua}, se lee el conjunto
como imagen recíproca y se invoca el teorema.
\end{example}

\begin{example}[Un cerrado definido por infinitas condiciones]\label{ex:b2:metric:lipclosed}
En $\bigl(C(\intcc01), d_\infty\bigr)$, el conjunto
\[
L = \{f : \abs{f(x) - f(y)} \leq \abs{x - y}
\ \text{para todos } x, y\}
\]
de las funciones $1$-lipschitzianas es cerrado, pese a estar
recortado por una cantidad no \omterm{def:b2:structures:countable}{numerable} de condiciones: para cada
par fijo $(x, y)$, la aplicación $f \mapsto \abs{f(x) - f(y)} -
\abs{x - y}$ es \omterm{def:b2:metric:continuity}{continua} (las evaluaciones son
$1$-lipschitzianas), de modo que cada condición aislada define un
cerrado, y $L$ es la \emph{intersección} de esa familia; y una
intersección arbitraria de cerrados es cerrada. La misma plantilla
certifica el carácter cerrado del conjunto de funciones monótonas,
del de funciones convexas y del de funciones acotadas por una $g$
fija: los límites uniformes heredan toda propiedad expresable como
familia de restricciones puntuales cerradas. Lo que los límites
uniformes \emph{no} heredan automáticamente ---la derivabilidad,
por ejemplo--- es justamente lo que al~\cref{ch:b2:funcseq} le
costará trabajo.
\end{example}

\begin{theorem}[Teorema del punto fijo de Banach]\label{thm:b2:metric:banach}
Sea $X$ un \omterm{def:b2:metric:complete}{espacio métrico completo} no vacío y $f \colon X \to X$
una \emph{contracción}: \omterm{def:b2:metric:continuity}{lipschitziana} de constante $k < 1$. Entonces
$f$ tiene un único punto fijo $\ell$, y toda órbita $x_{n+1} =
f(x_n)$ converge a $\ell$, con\index{teorema del punto fijo de
Banach}
\[
d(x_n, \ell) \leq \frac{k^n}{1 - k}\, d(x_1, x_0) .
\]
\end{theorem}

\begin{proof}
Unicidad: dos puntos fijos están a distancia $\leq k$ veces esa misma
distancia. Existencia: $d(x_{n+1}, x_n) \leq k^n d(x_1, x_0)$ por
inducción, de modo que para $q > p$,
\[
d(x_q, x_p) \leq \sum_{j=p}^{q-1} d(x_{j+1}, x_j)
\leq d(x_1, x_0) \sum_{j \geq p} k^j
= \frac{k^p}{1-k}\, d(x_1, x_0) \xrightarrow[p\to\infty]{} 0 :
\]
la sucesión es de Cauchy y por tanto converge a cierto $\ell$; la
\omterm{def:b2:metric:continuity}{continuidad} de $f$ permite pasar $x_{n+1} = f(x_n)$ al límite: $\ell
= f(\ell)$. La cota del error es la estimación anterior con $q \to
\infty$.
\end{proof}

\begin{example}[Una ecuación integral]\label{ex:b2:metric:integralequation}
Sobre $X = C(\intcc{0}{1})$ (\omterm{def:b2:metric:complete}{completo},
\cref{thm:b2:metric:rncomplete}), consideremos $T(f)(x) = 1 +
\frac12 \int_0^x f(t)\,\dd t$. Para $f, g \in X$:
\[
\abs{T(f)(x) - T(g)(x)} \leq \frac12 \int_0^x \abs{f - g} \leq
\frac12\, d_\infty(f, g),
\]
así que $T$ es una contracción de razón $\frac12$: tiene un único
punto fijo \omterm{def:b2:metric:continuity}{continuo}, la solución de $f' = \frac f2$, $f(0) = 1$, es
decir, $\eu^{x/2}$. Este esquema, industrializado, se convierte en
el teorema de Cauchy--Lipschitz del~\cref{ch:b2:diffeq}.
\end{example}

\begin{example}[Un punto fijo numérico: $x = \cos x$]\label{ex:b2:metric:cosfix}
Sobre el \omterm{def:b2:metric:complete}{completo} $X = \intcc{0}{1}$, la aplicación $f = \cos$
envía $X$ en $\intcc{\cos 1}{1} \subseteq X$ y es una contracción:
por la desigualdad del valor medio,
\[
\abs{\cos x - \cos y} \leq \bigl(\sup_{\intcc01}\abs{\sin}\bigr)
\abs{x - y} = (\sin 1)\abs{x - y},
\qquad \sin 1 \approx 0.841 < 1 .
\]
Banach: hay una única solución de $x = \cos x$ en $\intcc01$ (y por
tanto en $\R$: todo punto fijo real está en $\intcc{-1}{1}$ y, tras
una aplicación, en $\intcc{\cos 1}{1}$), y la iteración $x_{n+1} =
\cos x_n$ converge a ella desde cualquier punto de partida:
$x_\infty \approx 0.739085$, el célebre número que se obtiene
aporreando la tecla del coseno de una calculadora. La cota del
error predice un decrecimiento en $(\sin1)^n/(1 - \sin1)$, es
decir, alrededor de una cifra por cada $13$ pulsaciones; la cota a
posteriori del problema de fin de semana de este capítulo (pregunta
14) certifica cada paso sobre la marcha.
\end{example}

\section{Compacidad}

\begin{definition}\label{def:b2:metric:compact}
Un \omterm{def:b2:metric:def}{espacio métrico} $X$ es \emph{compacto}\index{espacio métrico
compacto} cuando toda sucesión de $X$ admite una subsucesión
convergente \emph{en} $X$ (la propiedad de Bolzano--Weierstrass). Un
subconjunto es compacto cuando lo es con la distancia inducida.
\end{definition}

\begin{theorem}[Primeras propiedades]\label{thm:b2:metric:compactprops}
\begin{enumerate}
  \item Un subconjunto \omterm{def:b2:metric:compact}{compacto} es cerrado y acotado; un subconjunto
        cerrado de un espacio \omterm{def:b2:metric:compact}{compacto} es \omterm{def:b2:metric:compact}{compacto}.
  \item En $\R^n$ vale el recíproco: \omterm{def:b2:metric:compact}{compacto} $\iff$ cerrado y
        acotado.
  \item La imagen \omterm{def:b2:metric:continuity}{continua} de un espacio \omterm{def:b2:metric:compact}{compacto} es \omterm{def:b2:metric:compact}{compacta}; una
        función real \omterm{def:b2:metric:continuity}{continua} sobre un \omterm{def:b2:metric:compact}{compacto} no vacío está acotada
        y alcanza sus cotas.
  \item (Heine) Una aplicación \omterm{def:b2:metric:continuity}{continua} sobre un espacio \omterm{def:b2:metric:compact}{compacto} es
        uniformemente continua.
  \item Productos: si $X, Y$ son \omterm{def:b2:metric:compact}{compactos}, también lo es $X \times Y$
        (con $d\bigl((x,y),(x',y')\bigr) = d(x,x') + d(y,y')$).
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Los mismos argumentos que en la recta (volumen del primer año):
una sucesión que se escapa al infinito o que converge fuera no tiene
subsucesión convergente dentro; para la segunda afirmación,
extráigase en el \omterm{def:b2:metric:compact}{compacto} ambiente y úsese que el subconjunto es
cerrado.

(2) Las sucesiones acotadas de $\R^n$ tienen subsucesiones
convergentes componente a componente: extráigase sobre la primera
coordenada (Bolzano--Weierstrass en $\R$), después, de esa
subsucesión, sobre la segunda, y así sucesivamente ($n$ extracciones
sucesivas); el ser cerrado mantiene el límite dentro.

(3) Dada $(f(x_n))$, extráigase $x_{\varphi(n)} \to x \in X$; la
\omterm{def:b2:metric:continuity}{continuidad} da $f(x_{\varphi(n)}) \to f(x) \in f(X)$. Caso real: la
\omterm{def:b2:metric:compact}{compacidad} de $f(X) \subseteq \R$ lo hace cerrado y acotado, y $\sup
f(X) \in f(X)$ (el supremo de un conjunto es adherente a él, y $f(X)$
es cerrado).

(4) La demostración del primer año se traslada literalmente; hela
aquí, en versión métrica. Supongamos $f \colon X \to Y$ \omterm{def:b2:metric:continuity}{continua}
sobre el \omterm{def:b2:metric:compact}{compacto} $X$ pero no uniformemente continua: algún
$\varepsilon > 0$ admite, para todo $n$, puntos con
\[
d_X(x_n, y_n) \leq \frac{1}{n+1}
\qquad\text{y}\qquad
d_Y\bigl(f(x_n), f(y_n)\bigr) > \varepsilon .
\]
Extraigamos $x_{\varphi(n)} \to a \in X$; entonces también
$y_{\varphi(n)} \to a$ (las distancias mutuas tienden a $0$). La
\omterm{def:b2:metric:continuity}{continuidad} en $a$ envía ambas sucesiones imagen a $f(a)$, luego
$d_Y\bigl(f(x_{\varphi(n)}), f(y_{\varphi(n)})\bigr) \to 0$, en
contradicción con la separación uniforme $> \varepsilon$. La
\omterm{def:b2:metric:compact}{compacidad} aportó exactamente una cosa: el punto de acumulación
$a$ en el que aplicar la \omterm{def:b2:metric:continuity}{continuidad} ordinaria.

(5) Extráigase sobre las coordenadas en $X$ y después, de nuevo,
sobre las coordenadas en $Y$.
\end{proof}

\begin{example}[El teorema de Heine, con y sin compacidad]\label{ex:b2:metric:heinework}
Sobre $\intcc{0}{1}$, la función $x \mapsto x^2$ es uniformemente
continua ---Heine lo afirma sin cálculo alguno, pero la estimación
directa es instructiva---:
\[
\abs{x^2 - y^2} = \abs{x + y}\,\abs{x - y} \leq 2\abs{x - y},
\]
de modo que $\delta = \varepsilon/2$ sirve \emph{para todos los
puntos a la vez}. Sobre $\R$, esa misma función no es uniformemente
continua: con $x_n = n$ e $y_n = n + \frac1n$, la separación
$\abs{x_n - y_n} = \frac1n \to 0$ mientras que $\abs{x_n^2 -
y_n^2} = 2 + \frac1{n^2} \geq 2$: ningún $\delta$ único sirve para
$\varepsilon = 1$. El mecanismo queda a la vista: la constante de
Lipschitz local $\abs{x + y}$ está acotada sobre un \omterm{def:b2:metric:compact}{compacto} y no
lo está sobre $\R$; el teorema de Heine es exactamente la
afirmación de que la \omterm{def:b2:metric:compact}{compacidad} pone un techo uniforme a esas
constantes locales.
\end{example}

\begin{method}[Cómo demostrar que un conjunto es compacto]\label{met:b2:metric:compactness}
Tres vías, por orden de frecuencia. (1) \emph{Reconocimiento en el
ambiente:} en $\R^n$ (o en cualquier espacio normado de dimensión
finita, \cref{ch:b2:nvs}), compruébese que es cerrado ---por lo
general como imagen recíproca, \cref{ex:b2:metric:recognize}--- y
acotado. (2) \emph{Herencia:} un cerrado dentro de un \omterm{def:b2:metric:compact}{compacto}
conocido es \omterm{def:b2:metric:compact}{compacto}; una unión finita o un producto de
\omterm{def:b2:metric:compact}{compactos} es \omterm{def:b2:metric:compact}{compacto}; la imagen \omterm{def:b2:metric:continuity}{continua} de un \omterm{def:b2:metric:compact}{compacto} es
\omterm{def:b2:metric:compact}{compacta}. (3) \emph{A mano:} extráigase una subsucesión convergente
de una sucesión arbitraria, normalmente mediante extracciones
sucesivas coordenada a coordenada. Para demostrar la \emph{no}
\omterm{def:b2:metric:compact}{compacidad} basta un testigo: una sucesión sin subsucesión
convergente, casi siempre puntos a distancia mutua $\geq
\varepsilon$.
\end{method}

\begin{example}[Distancias entre conjuntos: la compacidad se gana el sueldo]\label{ex:b2:metric:setdistance}
Sean $K$ \omterm{def:b2:metric:compact}{compacto}, $F$ cerrado y $K \cap F = \emptyset$ en un
\omterm{def:b2:metric:def}{espacio métrico}. Entonces
\[
d(K, F) = \inf\,\{d(x, y) : x \in K,\ y \in F\} > 0 :
\]
la función $x \mapsto d(x, F)$ es \omterm{def:b2:metric:continuity}{continua}
(\cref{exo:b2:metric:11}) y positiva sobre $K$ (pues $d(x, F) = 0$
pondría $x \in \overline F = F$), luego alcanza un mínimo positivo
sobre el \omterm{def:b2:metric:compact}{compacto} $K$
(\cref{thm:b2:metric:compactprops}~(3)). La \omterm{def:b2:metric:compact}{compacidad} no es
decorativa: para dos \emph{cerrados} el ínfimo puede anularse sin
alcanzarse; en $\R^2$, la hipérbola $F_1 = \{xy = 1\}$ y el eje
$F_2 = \{y = 0\}$ son cerrados disjuntos con $d(F_1, F_2) = 0$
(los puntos $(n, \frac1n)$ se acercan al eje). La huida al
infinito es justamente lo que la \omterm{def:b2:metric:compact}{compacidad} prohíbe.
\end{example}

\begin{theorem}[Borel--Lebesgue]\label{thm:b2:metric:borellebesgue}
Un \omterm{def:b2:metric:def}{espacio métrico} $X$ es \omterm{def:b2:metric:compact}{compacto} si y solo si todo recubrimiento
de $X$ por \omterm{def:b2:metric:topology}{abiertos} admite un subrecubrimiento
\emph{finito}.\index{teorema de Borel--Lebesgue}
\end{theorem}

\begin{proof}
($\Leftarrow$) Supongamos que $(x_n)$ no tiene ninguna subsucesión
convergente. Afirmamos que todo $x \in X$ tiene una bola $B(x,
r_x)$ que contiene $x_n$ solo para un número finito de índices
$n$: en caso contrario, toda bola $B(x, \frac1{k+1})$ contendría
infinitos términos y, eligiendo índices
\[
\varphi(0) < \varphi(1) < \varphi(2) < \cdots
\quad\text{con}\quad
x_{\varphi(k)} \in B\Bigl(x, \frac{1}{k+1}\Bigr)
\]
(posible en cada paso precisamente porque quedan infinitos
candidatos), se construiría una subsucesión convergente a $x$.
Las bolas $B(x, r_x)$ recubren $X$; si un número finito de ellas
recubriera $X$, el conjunto de índices $\N$ sería unión finita de
conjuntos finitos: absurdo.

($\Rightarrow$) Dos pasos. \emph{Número de Lebesgue:} para un
recubrimiento \omterm{def:b2:metric:topology}{abierto} $(U_i)$ de un \omterm{def:b2:metric:compact}{compacto} $X$ existe $\rho > 0$
tal que toda bola de radio $\rho$ está contenida en algún $U_i$. En
caso contrario, para cada $n$ tómese $x_n$ tal que $B(x_n,
\frac{1}{n+1})$ no esté en ningún $U_i$; extráigase $x_{\varphi(n)}
\to x \in U_{i_0} \supseteq B(x, r)$; para $n$ grande,
$B(x_{\varphi(n)}, \frac{1}{\varphi(n)+1}) \subseteq B(x, r)
\subseteq U_{i_0}$: contradicción. \emph{Precompacidad:} para todo
$\rho > 0$, un número finito de bolas de radio $\rho$ recubre $X$.
En caso contrario, elíjase inductivamente $x_{n+1}$ fuera de $B(x_0,
\rho) \cup \dots \cup B(x_n, \rho)$: la sucesión tiene distancias
mutuas $\geq \rho$, luego no tiene ninguna subsucesión de Cauchy y,
por tanto, ninguna convergente: contradicción. Combinando ambos:
recúbrase $X$ con un número finito de bolas de radio $\rho$ (el
número de Lebesgue), cada una dentro de algún $U_i$; se obtiene un
subrecubrimiento finito.
\end{proof}

\begin{example}[Una $\varepsilon$-red, contada]\label{ex:b2:metric:epsnet}
La precompacidad (de la demostración del~\cref{thm:b2:metric:borellebesgue}) es muy concreta en
$\intcc{0}{1}$: para $\varepsilon > 0$, las $\lceil
\frac{1}{2\varepsilon}\rceil$ bolas de radio $\varepsilon$
centradas en $\varepsilon, 3\varepsilon, 5\varepsilon, \dots$ lo
recubren ---unas $\frac1{2\varepsilon}$ bolas---, y ningún
recubrimiento puede arreglárselas con menos de
$\frac{1}{2\varepsilon}$ (cada bola cubre longitud a lo sumo
$2\varepsilon$). En $\intcc01^2$ el recuento se eleva al cuadrado,
al orden $\varepsilon^{-2}$: los números de recubrimiento crecen
como $\varepsilon^{-d}$ en dimensión $d$, una cara cuantitativa de
la \omterm{def:b2:metric:compact}{compacidad} y la razón de que las bolas unidad de dimensión
infinita del~\cref{ch:b2:nvs} (donde no existe ninguna
$\frac13$-red finita) no puedan ser \omterm{def:b2:metric:compact}{compactas}.
\end{example}

\begin{example}[La compacidad leída en los recubrimientos]\label{ex:b2:metric:covers}
El intervalo semiabierto $\intoc{0}{1}$ está recubierto por los
\omterm{def:b2:metric:topology}{abiertos} $U_n = \intoo{\frac1n}{2}$, $n \geq 1$; toda subfamilia
finita tiene un índice máximo $N$ y deja fuera $\intoc{0}{\frac1N}$:
no hay subrecubrimiento finito, luego $\intoc{0}{1}$ no es
\omterm{def:b2:metric:compact}{compacto}, cosa que la definición sucesional ve mediante $x_n =
\frac1n$, cuyo límite $0$ se escapa. En cambio, añadir el único
punto $0$ repara ambos diagnósticos a la vez: sobre $\intcc{0}{1}$,
todo recubrimiento de ese tipo ha de contener un \omterm{def:b2:metric:topology}{abierto} que
contenga a $0$, el cual se traga todo un segmento inicial, y un
número finito de \omterm{def:b2:metric:topology}{abiertos} remata el resto. Los dos lenguajes del~\cref{thm:b2:metric:borellebesgue} fallan o triunfan siempre a la
vez: los recubrimientos detectan la huida exactamente allí donde lo
hacen las sucesiones.
\end{example}

\begin{remark}[Perspectivas dentro de este volumen]\label{rem:b2:metric:perspectives}
Este capítulo es el muro de carga del volumen; conviene ver dónde
soporta peso cada pilar. \emph{\omterm{def:b2:metric:complete}{Completitud}}: el criterio de Cauchy
pasa a ser el criterio de convergencia de series en espacios de
Banach (\cref{ch:b2:series}), la convergencia uniforme del~\cref{ch:b2:funcseq} es exactamente la convergencia en el
\omterm{def:b2:metric:complete}{completo} $\bigl(C, d_\infty\bigr)$, y Cauchy--Lipschitz
(\cref{ch:b2:diffeq}) es el teorema del punto fijo de Banach
disfrazado de ecuación integral. \emph{\omterm{def:b2:metric:compact}{Compacidad}}: demuestra la
equivalencia de normas (\cref{ch:b2:nvs}), la existencia de extremos
para la optimización del~\cref{ch:b2:diffcalc} y la existencia de
mejores aproximaciones (problema de fin de semana del~\cref{ch:b2:nvs}). \emph{\omterm{def:b2:metric:connected}{Conexidad}}: globaliza enunciados locales
---la unicidad de soluciones de ecuaciones diferenciales, el teorema
del valor intermedio sobre curvas (\cref{ch:b2:curves}) y las dos
componentes de $GL_n(\R)$ que la teoría de la orientación
(\cref{ch:b2:multint}) mantendrá separadas---.
\end{remark}

\begin{remark}[Errores frecuentes]\label{rem:b2:metric:pitfalls}
(i) ``Cerrado y acotado implica \omterm{def:b2:metric:compact}{compacto}'' es un teorema sobre
$\R^n$, no sobre \omterm{def:b2:metric:def}{espacios métricos}: un conjunto infinito con la
métrica discreta es cerrado y acotado en sí mismo y no es
\omterm{def:b2:metric:compact}{compacto} (\cref{exo:b2:metric:4}), y la bola unidad cerrada de
$C(\intcc01)$ también falla (\cref{ch:b2:nvs}). (ii) La
\omterm{def:b2:metric:complete}{completitud} es una propiedad de la \emph{distancia}, no de la
topología: $\R$ con $d(x,y) = \abs{\arctan x - \arctan y}$ tiene
las sucesiones convergentes de siempre y no es \omterm{def:b2:metric:complete}{completo}
(\cref{exo:b2:metric:1}). (iii) Una biyección \omterm{def:b2:metric:continuity}{continua} no tiene por
qué ser un homeomorfismo ---la parametrización de la circunferencia
del~\cref{exo:b2:metric:7}---; la \omterm{def:b2:metric:compact}{compacidad} del espacio de partida
lo repara. (iv) El teorema de Banach necesita $k < 1$ de manera
\emph{uniforme}: la condición $d(f(x), f(y)) < d(x,y)$ por sí sola
no garantiza nada en un espacio no \omterm{def:b2:metric:compact}{compacto}
(\cref{exo:b2:metric:5}). (v) \omterm{def:b2:metric:connected}{Conexo} no implica \omterm{def:b2:metric:connected}{conexo} por
caminos en general, aunque para los \omterm{def:b2:metric:topology}{abiertos} de espacios normados
que aparecen en este libro ambas nociones coinciden
(\cref{ch:b2:nvs}).
\end{remark}

\begin{remark}[Dónde se usa este capítulo]
En toda la mitad de análisis. La \omterm{def:b2:metric:complete}{completitud} de $C(\intcc{a}{b})$
alimenta los teoremas de convergencia del~\cref{ch:b2:funcseq} y la
teoría de Cauchy--Lipschitz del~\cref{ch:b2:diffeq} (el problema de
fin de semana de este capítulo ya demuestra el teorema local de
Picard--Lindelöf); la \omterm{def:b2:metric:compact}{compacidad} da la equivalencia de normas en
dimensión finita (\cref{ch:b2:nvs}) y la existencia de extremos en
el~\cref{ch:b2:diffcalc}; la \omterm{def:b2:metric:connected}{conexidad} sostiene los argumentos de
valor intermedio del~\cref{ch:b2:realfun} y la globalización de la
unicidad para ecuaciones diferenciales. En el volumen del tercer
año, la \omterm{def:b2:metric:compact}{compacidad} en espacios de funciones (el teorema de
Arzelà--Ascoli) y el teorema de Baire (aquí,
\cref{exo:b2:metric:12}) pasan a ser herramientas de uso diario.
\end{remark}

\section{Conexidad}

\begin{definition}\label{def:b2:metric:connected}
$X$ es \emph{conexo}\index{espacio conexo} cuando no admite ninguna
partición en dos \omterm{def:b2:metric:topology}{abiertos} no vacíos; equivalentemente, cuando sus
únicos subconjuntos a la vez \omterm{def:b2:metric:topology}{abiertos} y cerrados son $\emptyset$ y
$X$. $X$ es \emph{conexo por caminos} cuando dos puntos cualesquiera
se unen mediante una aplicación \omterm{def:b2:metric:continuity}{continua} $\gamma \colon \intcc{0}{1}
\to X$.
\end{definition}

\begin{theorem}\label{thm:b2:metric:connectedness}
\begin{enumerate}
  \item Los subconjuntos \omterm{def:b2:metric:connected}{conexos} de $\R$ son exactamente los
        intervalos.
  \item La imagen \omterm{def:b2:metric:continuity}{continua} de un \omterm{def:b2:metric:connected}{espacio conexo} es \omterm{def:b2:metric:connected}{conexa}, de donde
        el teorema del valor intermedio general: una función real
        \omterm{def:b2:metric:continuity}{continua} sobre un \omterm{def:b2:metric:connected}{espacio conexo} toma todos los valores
        comprendidos entre dos cualesquiera de sus valores.
  \item \omterm{def:b2:metric:connected}{Conexo} por caminos $\Rightarrow$ \omterm{def:b2:metric:connected}{conexo}. (El recíproco es
        falso en general; sí vale para los \omterm{def:b2:metric:topology}{abiertos} de espacios
        normados, \cref{ch:b2:nvs}.)
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Un conjunto $A$ que no sea intervalo se salta algún $z$
comprendido entre dos de sus puntos: $A = (A \cap
\intoo{-\infty}{z}) \cup (A \cap \intoo{z}{+\infty})$ lo parte en dos
piezas no vacías y \omterm{def:b2:metric:topology}{abiertas} en $A$. Recíprocamente, sea $I$ un
intervalo e $I = U \cup V$ una partición en \omterm{def:b2:metric:topology}{abiertos} relativos no
vacíos; tómense $a \in U$ y $b \in V$, digamos $a < b$, y póngase $s
= \sup\,(U \cap \intcc{a}{b})$, punto de $\intcc{a}{b} \subseteq I$.
Si $s \in U$: entonces $s \neq b$, y el ser $U$ \omterm{def:b2:metric:topology}{abierto} relativo
coloca todo un intervalo alrededor de $s$ (intersecado con $I$)
dentro de $U$, de modo que hay puntos de $U \cap \intcc{a}{b}$ que
superan a $s$, en contra del supremo. Si $s \in V$: el ser $V$
\omterm{def:b2:metric:topology}{abierto} relativo coloca un intervalo $\intoo{s - r}{s + r} \cap I$
dentro de $V$; pero el supremo es adherente a $U \cap \intcc{a}{b}$,
que ha de cortar a ese intervalo, en contradicción con $U \cap V =
\emptyset$. (Este es el argumento de abierto-y-cerrado del primer año
para $\R$, ejecutado dentro de $I$.)

(2) Si $f(X) = U' \cup V'$ se parte en \omterm{def:b2:metric:topology}{abiertos} relativos no vacíos,
entonces $X = f^{-1}(U') \cup f^{-1}(V')$ parte $X$
(\cref{thm:b2:metric:globalcontinuity}). Valor intermedio: $f(X)
\subseteq \R$ es \omterm{def:b2:metric:connected}{conexo} y por tanto un intervalo, por (1).

(3) Supongamos $X = U \cup V$ con ambos \omterm{def:b2:metric:topology}{abiertos} no vacíos, y unamos
$a \in U$ con $b \in V$ mediante un camino $\gamma$: entonces
$\gamma^{-1}(U)$ y $\gamma^{-1}(V)$ parten $\intcc{0}{1}$, en contra
de (1).
\end{proof}

\begin{example}\label{ex:b2:metric:glnr}
$GL_n(\R)$ no es \omterm{def:b2:metric:connected}{conexo}: $\det$ es \omterm{def:b2:metric:continuity}{continua} (un polinomio en las
entradas) sobre $\R^*$, que no es \omterm{def:b2:metric:connected}{conexo}; las imágenes recíprocas
de $\R_+^*$ y $\R_-^*$ parten $GL_n(\R)$. (Cada pieza es de hecho
\omterm{def:b2:metric:connected}{conexa} por caminos, un ejercicio agradable que va más allá de lo
que necesitamos.) En cambio, $GL_n(\C)$ \emph{sí} es \omterm{def:b2:metric:connected}{conexo} por
caminos: \cref{exo:b2:metric:10}.
\end{example}

\begin{example}[Un punto fijo que solo debe a la conexidad]\label{ex:b2:metric:ivtfixed}
Toda $f \colon \intcc01 \to \intcc01$ \omterm{def:b2:metric:continuity}{continua} tiene un punto fijo,
sin hipótesis de contracción ni iteración alguna. Consideremos
$g(x) = f(x) - x$, \omterm{def:b2:metric:continuity}{continua} sobre el \omterm{def:b2:metric:connected}{conexo} $\intcc01$:
\[
g(0) = f(0) \geq 0,
\qquad
g(1) = f(1) - 1 \leq 0 ,
\]
y el teorema del valor intermedio
(\cref{thm:b2:metric:connectedness}~(2)) proporciona un cero de
$g$, es decir, un punto fijo de $f$. Contrástese con Banach
(\cref{thm:b2:metric:banach}): aquí la existencia es topológica y
gratuita, pero se pierden la unicidad y el algoritmo ---$f =
\mathrm{id}$ tiene todos los puntos fijos, y la iteración de una $f$
que no contrae puede ciclar indefinidamente---. Los dos teoremas de
punto fijo de este capítulo responden a preguntas distintas y con
monedas distintas.
\end{example}

\begin{example}[$\R$ y $\R^2$ no son homeomorfos]\label{ex:b2:metric:rnotr2}
La \omterm{def:b2:metric:connected}{conexidad} es una huella dactilar topológica. Supongamos que $h
\colon \R^2 \to \R$ fuera un homeomorfismo (una biyección \omterm{def:b2:metric:continuity}{continua}
con inversa \omterm{def:b2:metric:continuity}{continua}). Quitemos un punto $a \in \R^2$: la
restricción $h \colon \R^2\setminus\{a\} \to \R\setminus\{h(a)\}$
sigue siendo un homeomorfismo. Pero $\R^2$ menos un punto es
\omterm{def:b2:metric:connected}{conexo} por caminos ---únanse dos puntos cualesquiera por un
segmento, dando un rodeo por un segundo segmento a través de un
punto auxiliar si $a$ bloquea el directo---, luego es \omterm{def:b2:metric:connected}{conexo}
(\cref{thm:b2:metric:connectedness}~(3)); mientras que $\R$ menos un
punto se parte en dos semirrectas \omterm{def:b2:metric:topology}{abiertas} no vacías: no es
\omterm{def:b2:metric:connected}{conexo}. Y la \omterm{def:b2:metric:connected}{conexidad} se conserva por aplicaciones \omterm{def:b2:metric:continuity}{continuas}:
contradicción. El plano y la recta son genuinamente distintos
\emph{como espacios topológicos}, hecho que el cardinal por sí solo
(¡las biyecciones al estilo del~\cref{exo:b2:structures:3} existen!)
es demasiado grosero para percibir.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:metric:1}
Sobre $\R$, comprueba que $\delta(x, y) = \min(1, \abs{x - y})$ y
$d(x,y) = \abs{\arctan x - \arctan y}$ son distancias. ¿Qué
sucesiones convergen para cada una? ¿Es $(\R, d)$ \omterm{def:b2:metric:complete}{completo}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:metric:2}
En un \omterm{def:b2:metric:def}{espacio métrico}, demuestra que toda sucesión convergente es de
Cauchy y está acotada, y que una sucesión de Cauchy con una
subsucesión convergente converge. Deduce de nuevo que los
\omterm{def:b2:metric:compact}{espacios métricos compactos} son \omterm{def:b2:metric:complete}{completos}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:metric:3}
En $\bigl(C(\intcc{0}{1}), d_\infty\bigr)$, calcula la distancia
entre $f(x) = x$ y $g(x) = x^2$; describe la bola cerrada $\overline
B(0, 1)$; y demuestra que el conjunto $\{f : f(0) = 0\}$ es cerrado
mientras que $\{f : f(0) > 0\}$ es \omterm{def:b2:metric:topology}{abierto}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:metric:4}
Demuestra que el \omterm{def:b2:metric:def}{espacio métrico} discreto $X$ (un conjunto
cualquiera) es \omterm{def:b2:metric:complete}{completo} y que es \omterm{def:b2:metric:compact}{compacto} si y solo si $X$ es
finito. ¿Qué subconjuntos son \omterm{def:b2:metric:connected}{conexos}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:metric:5}
Sea $X$ \omterm{def:b2:metric:compact}{compacto} y $f \colon X \to X$ tal que
\[
d\bigl(f(x), f(y)\bigr) < d(x, y) \quad \text{para todos } x \neq y .
\]
Demuestra que $f$ tiene un único punto fijo \emph{(minimiza $x
\mapsto d(x, f(x))$)}, y da un ejemplo sobre $X =
\intco{1}{+\infty}$ (no \omterm{def:b2:metric:compact}{compacto}) sin punto fijo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:metric:6}
(\omterm{def:b2:metric:compact}{Compactos} encajados) Sea $(K_n)$ una sucesión decreciente de
subconjuntos \omterm{def:b2:metric:compact}{compactos} no vacíos de un \omterm{def:b2:metric:def}{espacio métrico}. Demuestra
que $\bigcap_n K_n \neq \emptyset$ \emph{(toma $x_n \in K_n$ y
extrae)}. Muestra con un ejemplo que unos \emph{cerrados} encajados
no vacíos de $\R$ pueden tener intersección vacía.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:metric:7}
Sea $K$ \omterm{def:b2:metric:compact}{compacto} y $f \colon K \to Y$ \omterm{def:b2:metric:continuity}{continua} y biyectiva.
Demuestra que $f^{-1}$ es \omterm{def:b2:metric:continuity}{continua} \emph{(usa los cerrados:
\cref{thm:b2:metric:globalcontinuity} y
\cref{thm:b2:metric:compactprops})}. Da un contraejemplo sin
\omterm{def:b2:metric:compact}{compacidad} ($\gamma(t) = (\cos t, \sin t)$ sobre
$\intco{0}{2\pi}$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:metric:8}
El \emph{conjunto de Cantor} $C$ se obtiene de $\intcc{0}{1}$
suprimiendo reiteradamente los tercios centrales \omterm{def:b2:metric:topology}{abiertos}. Demuestra
que $C$ es \omterm{def:b2:metric:compact}{compacto}, tiene interior vacío y es infinito; de hecho,
\omterm{def:b2:structures:countable}{equipotente} a $\{0,1\}^{\N}$ \emph{(desarrollos en base $3$ con
cifras $0$ y $2$; \cref{exo:b2:structures:3})}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:metric:9}
Sea $X$ un \omterm{def:b2:metric:def}{espacio métrico} \emph{\omterm{def:b2:metric:compact}{compacto}} y $f \colon X \to X$ una
isometría: $d(f(x), f(y)) = d(x, y)$. Demuestra que $f$ es
sobreyectiva. \emph{Indicación: si $a \notin f(X)$, entonces
$\varepsilon = d(a, f(X)) > 0$ (¿por qué?); estudia la órbita $a,
f(a), f^2(a), \dots$ y prueba que sus puntos están a distancia mutua
$\geq \varepsilon$, en contradicción con la \omterm{def:b2:metric:compact}{compacidad}.}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:metric:10}
Demuestra que $GL_n(\C)$ es \omterm{def:b2:metric:connected}{conexo} por caminos. \emph{Indicación:
dadas $A, B$ invertibles, considera $p(z) = \det\bigl((1 - z)A +
zB\bigr)$ para $z \in \C$: es un polinomio en $z$ no idénticamente
nulo, luego tiene un número finito de raíces; une $0$ con $1$ en $\C$
mediante un camino que las evite.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:metric:11}
Para $\emptyset \neq A \subseteq X$, pongamos $d(x, A) = \inf_{a \in
A} d(x, a)$. Demuestra que $x \mapsto d(x, A)$ es $1$-lipschitziana,
que $d(x, A) = 0$ si y solo si $x \in \overline A$, y que para
\emph{cerrados} disjuntos no vacíos $A, B$ la función
\[
\varphi(x) = \frac{d(x, A)}{d(x, A) + d(x, B)}
\]
está bien definida, es \omterm{def:b2:metric:continuity}{continua} y vale $0$ exactamente sobre $A$ y
$1$ exactamente sobre $B$: un ``interruptor'' \omterm{def:b2:metric:continuity}{continuo} que separa dos
cerrados disjuntos cualesquiera.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:metric:12}
(Baire) Sea $X$ un \omterm{def:b2:metric:complete}{espacio métrico completo} y $(U_n)_{n\geq1}$ una
sucesión de \omterm{def:b2:metric:topology}{abiertos} densos. Demuestra que $\bigcap_n U_n$ es denso
en $X$ \emph{(dentro de cualquier bola, constrúyanse bolas cerradas
encajadas $\overline B(x_n, r_n) \subseteq U_n$ con $r_n \to 0$ y
úsese la \omterm{def:b2:metric:complete}{completitud})}. Deduce que $\R$ no es unión \omterm{def:b2:structures:countable}{numerable} de
cerrados de interior vacío y recupera ---una vez más--- que $\R$ no
es \omterm{def:b2:structures:countable}{numerable}.
\end{exercise}

\section{Problema: la iteración de Picard}

\omterm{def:b2:metric:complete}{Completitud} más contracción es una máquina de resolver: se le da
una ecuación escrita como problema de punto fijo y devuelve
existencia, unicidad, algoritmo y barras de error. Este problema de
fin de semana pone la máquina a plena potencia sobre la ecuación $y' =
f(t, y)$: demostramos el \emph{teorema local de Picard--Lindelöf} (el
corazón no lineal de la teoría de Cauchy--Lipschitz del~\cref{ch:b2:diffeq}), vemos cómo cada hipótesis se gana el sueldo a
través de contraejemplos y recogemos dividendos puramente métricos: la
dependencia \omterm{def:b2:metric:continuity}{continua} respecto de los datos, la ecuación de Kepler y la
autosemejanza del conjunto de Cantor.

\begin{problem}[{Problema de fin de semana --- el teorema de
Picard--Lindelöf}]
\label{pb:b2:metric:1}
En todo el problema, $t_0 \in \R$, $y_0 \in \R$, $a, b > 0$, y $f$
es una función \omterm{def:b2:metric:continuity}{continua} sobre el rectángulo $R = \intcc{t_0 - a}{t_0
+ a} \times \intcc{y_0 - b}{y_0 + b}$, acotada por $M = \sup_R\,\abs
f$ y \emph{$L$-lipschitziana en su segunda variable}: $\abs{f(t, y)
- f(t, z)} \leq L\abs{y - z}$ siempre que ambos puntos estén en $R$.
Pongamos
\[
h = \min\Bigl(a, \frac bM\Bigr) \quad (\text{con } h = a
\text{ si } M = 0), \qquad I = \intcc{t_0 - h}{t_0 + h}.
\]

\medskip
\noindent\textbf{Parte I --- El escenario completo.}
\begin{enumerate}
  \item Demuestra las dos afirmaciones citadas en la~\cref{def:b2:metric:complete}: un cerrado de un
        \omterm{def:b2:metric:complete}{espacio métrico completo} es \omterm{def:b2:metric:complete}{completo}, y un subconjunto
        \omterm{def:b2:metric:complete}{completo} de cualquier \omterm{def:b2:metric:def}{espacio métrico} es cerrado. Deduce
        que todo cerrado de $\bigl(C(I), d_\infty\bigr)$ es un
        \omterm{def:b2:metric:complete}{espacio métrico completo}.
  \item Prueba que la aplicación $C(I) \to C(I)$, $y \mapsto
        \bigl(t \mapsto \int_{t_0}^{t}y(s)\,\dd s\bigr)$, es
        $h$-lipschitziana para $d_\infty$.
  \item (Los puntos fijos se mueven menos que las aplicaciones)
        Sea $g \colon X \to X$ una contracción de razón $k$ de un
        \omterm{def:b2:metric:def}{espacio métrico}, con punto fijo $\ell_g$, y $\widetilde g
        \colon X \to X$ una aplicación \emph{cualquiera} con un
        punto fijo $\ell_{\widetilde g}$. Demuestra que
        \[
        d(\ell_g, \ell_{\widetilde g}) \leq
        \frac{d\bigl(g(\ell_{\widetilde g}),
        \widetilde g(\ell_{\widetilde g})\bigr)}{1 - k}
        \leq \frac{\sup_{x \in X} d\bigl(g(x), \widetilde
        g(x)\bigr)}{1 - k}.
        \]
  \item (El truco de la iterada) Sea $X$ \omterm{def:b2:metric:complete}{completo} y no vacío y $g
        \colon X \to X$ una aplicación ---sin suponerla
        continua--- tal que alguna iterada $g^m$ sea una
        contracción de razón $k$. Demuestra que $g$ tiene un único
        punto fijo $\ell$ y que \emph{toda} órbita $x_{n+1} =
        g(x_n)$ converge a $\ell$. \emph{(Los puntos fijos de $g$
        lo son de $g^m$; recíprocamente, $g(\ell)$ es punto fijo de
        $g^m$; sepárese la órbita según los restos módulo $m$.)}
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- El teorema de Picard--Lindelöf.}
\begin{enumerate}[resume]
  \item Prueba que una función $y \colon I \to \intcc{y_0 - b}{y_0
        + b}$ es de clase $C^1$ con $y(t_0) = y_0$ e $y' = f(t,
        y)$ sobre $I$ si y solo si es \omterm{def:b2:metric:continuity}{continua} y satisface la
        ecuación integral
        \[
        y(t) = y_0 + \int_{t_0}^{t} f\bigl(s, y(s)\bigr)\dd s
        \qquad (t \in I).
        \]
  \item Sea $X_h = \{y \in C(I) : \abs{y(t) - y_0} \leq b \text{
        sobre } I\}$ y sea $T$ definida por $T(y)(t) = y_0 +
        \int_{t_0}^{t}f(s, y(s))\dd s$. Prueba que $X_h$ es un
        cerrado no vacío de $C(I)$, y por tanto \omterm{def:b2:metric:complete}{completo}, y que $T$
        envía $X_h$ en $X_h$; aquí es donde trabaja $h \leq b/M$.
  \item Prueba que $d_\infty\bigl(T(y), T(z)\bigr) \leq
        Lh\,d_\infty(y, z)$ sobre $X_h$: si $Lh < 1$, el teorema de
        Banach ya concluye. A continuación eliminamos esa condición
        de pequeñez.
  \item Demuestra por inducción sobre $n$ que
        \[
        \abs{T^n(y)(t) - T^n(z)(t)} \leq
        \frac{\bigl(L\abs{t - t_0}\bigr)^n}{n!}\,
        d_\infty(y, z) \qquad (y, z \in X_h,\ t \in I),
        \]
        de modo que alguna iterada de $T$ es una contracción.
        Concluye con la pregunta 4 (\emph{el teorema de
        Picard--Lindelöf}): el problema de Cauchy $y' = f(t,y)$,
        $y(t_0) = y_0$ tiene exactamente una solución sobre $I =
        \intcc{t_0 - h}{t_0 + h}$ con valores en $\intcc{y_0 -
        b}{y_0 + b}$.
  \item Prueba que la restricción ``con valores en $\intcc{y_0 -
        b}{y_0 + b}$'' es automática: toda solución del problema de
        Cauchy definida sobre $I$ cuya gráfica arranque dentro de
        $R$ permanece en $\intcc{y_0 - b}{y_0 + b}$ \emph{(considera
        el primer instante de salida y acota $\abs{y(t) - y_0}$ por
        $M\abs{t - t_0}$)}. Por tanto, la unicidad vale entre todas
        las soluciones sobre $I$.
  \item Pon en marcha la máquina sobre $y' = y$, $y(0) = 1$,
        partiendo de la constante $y^{(0)} \equiv 1$: calcula las
        iteradas de Picard $y^{(n)}$, identifícalas y describe la
        convergencia.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Cada hipótesis se gana el sueldo.}
\begin{enumerate}[resume]
  \item (Sin Lipschitz no hay unicidad) Para $y' = 2\sqrt{\abs
        y}$, $y(0) = 0$: comprueba que $y \equiv 0$ y, para todo $c
        \geq 0$, la función $y_c(t) = 0$ si $t \leq c$ e $y_c(t) =
        (t - c)^2$ si $t > c$, son todas soluciones de clase $C^1$
        sobre $\R$. ¿Dónde exactamente deja $y \mapsto 2\sqrt{\abs
        y}$ de ser \omterm{def:b2:metric:continuity}{lipschitziana}?
  \item (La localidad es real) Para $y' = y^2$, $y(0) = 1$:
        resuelve explícitamente, da el intervalo maximal de
        existencia y calcula el mejor $h$ que el teorema puede
        certificar sobre todas las elecciones del rectángulo ($a$
        grande, $b$ libre): prueba que $h_{\max} = \sup_{b>0}
        \frac{b}{(1+b)^2} = \frac14$, mientras que la solución
        verdadera vive sobre $\intoo{-\infty}{1}$.
  \item (La \omterm{def:b2:metric:complete}{completitud} no es decoración) Sobre $X = \Q \cap
        \intcc{1}{2}$ con la distancia usual, sea $g(x) = \frac x2
        + \frac1x$. Prueba que $g(X) \subseteq X$, que $g$ es una
        contracción de razón $\frac12$ \emph{(desigualdad del valor
        medio)} y que $g$ no tiene ningún punto fijo en $X$. ¿Qué
        hipótesis del teorema de Banach falla y cuál es el punto
        fijo en el completado?
  \item (Barras de error) Para una contracción $g$ de razón $k$
        sobre un espacio \omterm{def:b2:metric:complete}{completo}, demuestra la estimación \emph{a
        posteriori} $d(x_n, \ell) \leq \frac{k}{1-k}\,d(x_n,
        x_{n-1})$. Para la aplicación de Herón $g(x) = \frac x2 +
        \frac 1x$ sobre $\intcc{1}{2}$ (con punto fijo $\sqrt2$),
        partiendo de $x_0 = \frac32$: ¿cuántos pasos exige la cota
        \emph{a priori} $\frac{k^n}{1-k}d(x_1, x_0)$ para una
        precisión de $10^{-6}$, y cuántos bastan en realidad?
        (Calcula $x_1, x_2, x_3$ y sus errores; la cota de la
        contracción es honesta pero pesimista, porque Herón converge
        cuadráticamente.)
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Dependencia \omterm{def:b2:metric:continuity}{continua}.} En esta parte
$Lh < 1$, de modo que el propio $T$ es una contracción sobre $X_h$
(pregunta 7); escribimos $y[\,y_0\,]$ para la solución de valor
inicial $y_0$.
\begin{enumerate}[resume]
  \item (Dependencia del valor inicial) Sea $z_0$ otro valor
        inicial con $\abs{z_0 - y_0}$ lo bastante pequeño para que
        ambos problemas quepan en el rectángulo. Usando la pregunta
        3, demuestra que
        \[
        d_\infty\bigl(y[y_0], y[z_0]\bigr) \leq
        \frac{\abs{y_0 - z_0}}{1 - Lh}.
        \]
  \item (Intervalos largos por encadenamiento) Supongamos que las
        soluciones existen sobre un segmento largo cortado en $m$
        trozos consecutivos en cada uno de los cuales se aplica la
        cota anterior con $Lh \leq \frac12$. Prueba que la
        desviación crece a lo sumo en un factor $2$ por trozo, de
        donde $d_\infty \leq 2^m\abs{y_0 - z_0}$ en total: una cota
        exponencial en la longitud, sombra discreta del
        $\eu^{L\abs{t - t_0}}$ del lema de Gronwall
        (\cref{ch:b2:diffeq}).
  \item (Dependencia del campo) Sea $g$ otro campo sobre $R$,
        también $L$-lipschitziano en $y$, con $\sup_R \abs{f - g}
        \leq \varepsilon$. Demuestra que las soluciones
        correspondientes cumplen $d_\infty \leq \frac{\varepsilon
        h}{1 - Lh}$: el error de modelización se propaga
        linealmente.
  \item (Parámetros) Si una familia $f_\lambda$ de campos es
        $L$-lipschitziana en $y$ de manera uniforme y
        $\sup_R\abs{f_\lambda - f_\mu} \leq C\abs{\lambda - \mu}$,
        deduce que $\lambda \mapsto y_\lambda$ es \omterm{def:b2:metric:continuity}{lipschitziana}
        del espacio de parámetros en $\bigl(C(I), d_\infty\bigr)$.
  \item (Los sistemas no cuestan nada) Explica por qué las partes
        I, II y IV valen literalmente para $y$ con valores en
        $\R^n$ (distancias del supremo construidas sobre
        cualquiera de las distancias del~\cref{ex:b2:metric:examples}), y calcula después todas las
        iteradas de Picard para el sistema $y' = Ay$, $y(0) = (c_1,
        c_2)$, $A = \begin{psmallmatrix} 0 & 1\\ 0 &
        0\end{psmallmatrix}$: prueba que la iteración se estaciona
        en la solución exacta después de un solo paso.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dividendos métricos y síntesis.}
\begin{enumerate}[resume]
  \item (Perturbación de la identidad) Sea $X$ un escenario
        completo de tipo espacio normado: tómese $X = C(I)$ o $X =
        \R^n$. Si $\eta \colon X \to X$ es $k$-lipschitziana con $k
        < 1$, demuestra que $x \mapsto x + \eta(x)$ es una
        biyección de $X$ cuya inversa es
        $\frac1{1-k}$-lipschitziana \emph{(para cada $y$, aplica
        Banach a $x \mapsto y - \eta(x)$)}. Este es el corazón
        métrico del teorema de la función inversa
        (\cref{ch:b2:diffcalc}).
  \item (La ecuación de Kepler) Para $0 \leq e < 1$ y $m \in \R$,
        demuestra que $x = m + e\sin x$ tiene exactamente una
        solución, que la iteración $x_{n+1} = m + e\sin x_n$
        converge a ella desde cualquier punto de partida, y estima:
        para $e = \frac12$ y $m = 1$, ¿cuántas iteraciones
        garantizan un error $\leq 10^{-3}$ según la cota a priori?
        (La solución es $x \approx 1.4987$.)
  \item (El conjunto de Cantor es un punto fijo) Sean $S_1(x) =
        \frac x3$ y $S_2(x) = \frac x3 + \frac23$ sobre $\R$, y sea
        $C$ el conjunto de Cantor del~\cref{exo:b2:metric:8}.
        Demuestra que $C = S_1(C) \cup S_2(C)$ y explica en una
        frase por qué ningún \emph{otro} \omterm{def:b2:metric:compact}{compacto} no vacío
        satisface esa ecuación (la aplicación $A \mapsto S_1(A)
        \cup S_2(A)$ es una contracción para una distancia entre
        \omterm{def:b2:metric:compact}{compactos}: la distancia de Hausdorff, formalizada en el
        volumen del tercer año).
  \item (La \omterm{def:b2:metric:connected}{conexidad} globaliza la unicidad) Sea $f$ localmente
        \omterm{def:b2:metric:continuity}{lipschitziana} en $y$ sobre un \omterm{def:b2:metric:topology}{abierto}, y sean $y, z$ dos
        soluciones de $y' = f(t, y)$ sobre un intervalo común $J$
        con $y(t_0) = z(t_0)$. Demuestra que $y = z$ sobre $J$:
        prueba que $\{t \in J : y(t) = z(t)\}$ es no vacío, cerrado
        en $J$ y \omterm{def:b2:metric:topology}{abierto} en $J$ (por unicidad local), y usa la
        \omterm{def:b2:metric:connected}{conexidad} de los intervalos
        (\cref{thm:b2:metric:connectedness}).
  \item (Sin pequeñez para las ecuaciones lineales) Para $y' =
        \alpha(t)y + \beta(t)$ con $\alpha, \beta$ \omterm{def:b2:metric:continuity}{continuas} sobre
        un segmento $\intcc{A}{B}$, adapta la pregunta 8 para
        probar que la cota factorial vale sobre el segmento
        \emph{entero}, de modo que allí la existencia y la unicidad
        son globales: el caso escalar del teorema de
        Cauchy--Lipschitz del~\cref{ch:b2:diffeq}, sin restricción
        alguna sobre la longitud $B - A$.
  \item (Síntesis) Una frase para cada punto: qué aportó la
        \omterm{def:b2:metric:complete}{completitud}; qué aportó la contracción; qué se ganó con el
        truco de la iterada frente al Banach simple; dónde
        intervino la \omterm{def:b2:metric:connected}{conexidad}; y qué contraejemplo de la parte III
        custodia cada hipótesis. Nombra el teorema cumbre y di qué
        sustituye a la contracción cuando $f$ es meramente
        \omterm{def:b2:metric:continuity}{continua} (el teorema de Peano, vía \omterm{def:b2:metric:compact}{compacidad} en espacios de
        funciones: el Arzelà--Ascoli del volumen del tercer año).
\end{enumerate}
\end{problem}
