\chapter{Integración}\label{ch:b2:integration}

El volumen del primer año construyó la integral sobre un segmento.
Este capítulo la extiende a intervalos arbitrarios (\omterm{def:b2:integration:improper}{integrales
impropias}, con toda la caja de herramientas de comparación) y estudia
después las integrales \emph{dependientes de un parámetro}
---continuidad y derivación bajo el signo integral---, movidas por el
teorema de convergencia dominada, el único resultado de este capítulo
que se acepta sin demostrar. La función $\Gamma$ hace de ejemplo
conductor y de puerta de entrada a la mitad de las funciones especiales
de las matemáticas.

\section{Integrales sobre un intervalo cualquiera}

\begin{definition}\label{def:b2:integration:improper}
Sea $f$ \omterm{def:b2:metric:continuity}{continua} a trozos sobre $\intco{a}{b}$ ($b \in \R$ o
$+\infty$). La integral \emph{converge}\index{integral impropia}
cuando existe $\lim_{x \to b^-} \int_a^x f$; se escribe entonces
$\int_a^b f$ para el límite. (Análogamente sobre $\intoc{a}{b}$, y
sobre $\intoo{a}{b}$ separando en un punto interior; la elección no
importa, por Chasles.) La integral \emph{converge \omterm{def:b2:series:def}{absolutamente}}
cuando converge $\int_a^b \abs f$; la convergencia absoluta implica
la convergencia, por el criterio de Cauchy:
\[
\Bigl| \int_x^{y} f \Bigr| \leq \int_x^{y} \abs f
\]
y la \omterm{def:b2:metric:complete}{completitud} de $\R$ (la primitiva tiene la propiedad de
Cauchy). En detalle: sean $F(x) = \int_a^x f$ y $G(x) = \int_a^x
\abs f$. Si $\int^b\abs f$ converge, $G$ tiene límite en $b^-$, de
modo que para todo $\varepsilon > 0$ existe $c < b$ con $G(y) - G(x)
\leq \varepsilon$ siempre que $c \leq x \leq y < b$; la fórmula
anterior transfiere esa propiedad de Cauchy a $F$. Para cualquier
sucesión $x_n \to b^-$, los valores $F(x_n)$ forman entonces una
sucesión de Cauchy de reales, convergente por \omterm{def:b2:metric:complete}{completitud}, e
intercalar dos sucesiones así muestra que el límite es el mismo para
todas: $F$ tiene límite en $b^-$.
\end{definition}

\begin{theorem}[Herramientas de comparación en el caso positivo]\label{thm:b2:integration:comparison}
Para $f, g \geq 0$ \omterm{def:b2:metric:continuity}{continuas} a trozos sobre $\intco{a}{b}$:
\begin{enumerate}
  \item $\int_a^b f$ \omterm{def:b2:integration:improper}{converge} si y solo si la primitiva $x \mapsto
        \int_a^x f$ está acotada;
  \item si $f \leq g$: la convergencia de $\int g$ fuerza la de
        $\int f$, y la divergencia se transfiere en sentido
        contrario;
  \item si $f \sim g$ en $b$: las dos integrales tienen el mismo
        carácter;
  \item escalas de referencia: en $+\infty$, $\int^{\infty}
        \frac{\dd t}{t^\alpha}$ \omterm{def:b2:integration:improper}{converge} si y solo si $\alpha > 1$, y
        $\int^\infty \frac{\dd t}{t(\ln t)^\beta}$ si y solo si
        $\beta > 1$; en un extremo finito $b$, $\int^b \frac{\dd
        t}{(b - t)^\alpha}$ \omterm{def:b2:integration:improper}{converge} si y solo si $\alpha < 1$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) La primitiva $F(x) = \int_a^x f$ es no decreciente ($f \geq 0$).
Si está acotada, $\ell = \sup_{x < b}F$ es finito y $F(x) \to \ell$:
dado $\varepsilon > 0$, algún $F(x_0) > \ell - \varepsilon$, y la
monotonía atrapa $F(x) \in \intoc{\ell - \varepsilon}{\ell}$ para
$x_0 \leq x < b$. Si no está acotada, $F \to +\infty$: divergencia.

(2) De $f \leq g$: $\int_a^x f \leq \int_a^x g$ para todo $x$; si
$\int^b g$ \omterm{def:b2:integration:improper}{converge}, el miembro derecho está acotado y también el
izquierdo, y (1) concluye. La contraposición transfiere la
divergencia en sentido contrario.

(3) $f \sim g$ en $b$ proporciona un $c < b$ con
\[
\tfrac12\,g(t) \;\leq\; f(t) \;\leq\; 2\,g(t)
\qquad (c \leq t < b) :
\]
por (2), aplicado en ambos sentidos sobre $\intco{c}{b}$, las dos
integrales tienen el mismo carácter; y el trozo inicial
$\intcc{a}{c}$ es una integral propia y no cambia nada.

(4) Primitivas explícitas: para $\alpha \neq 1$ y $\beta \neq 1$,
\[
\int_c^x \frac{\dd t}{t^\alpha}
= \frac{x^{1-\alpha} - c^{1-\alpha}}{1 - \alpha},
\qquad
\int_c^x \frac{\dd t}{t(\ln t)^\beta}
= \frac{(\ln x)^{1-\beta} - (\ln c)^{1-\beta}}{1 - \beta},
\]
con logaritmos en los casos excluidos: acotadas cuando $x \to
+\infty$ exactamente si $\alpha > 1$, o si $\beta > 1$,
respectivamente. En un extremo finito, la sustitución $u = b - t$
reduce a la escala $\int_0 u^{-\alpha}\,\dd u$, acotada si y solo si
$\alpha < 1$. Aplíquese (1) en cada caso.
\end{proof}

\begin{example}[Dos calentamientos, hasta el final]\label{ex:b2:integration:warmups}
\emph{(a)} $\displaystyle\int_0^1 \ln t\,\dd t$: el integrando
explota en $0^+$, pero allí $\abs{\ln t} = o\bigl(t^{-1/2}\bigr)$
(los logaritmos pierden frente a las potencias) y $\int_0 t^{-1/2}$
\omterm{def:b2:integration:improper}{converge}: convergencia absoluta. El valor, integrando por partes
sobre $\intcc{\varepsilon}{1}$:
\[
\int_\varepsilon^1 \ln t\,\dd t
= \bigl[t\ln t - t\bigr]_\varepsilon^1
= -1 - \varepsilon\ln\varepsilon + \varepsilon
\xrightarrow[\varepsilon\to0^+]{} -1 .
\]
\emph{(b)} $\displaystyle\int_0^\infty \frac{\ln t}{1 +
t^2}\,\dd t$: hay problemas en ambos extremos, así que se separa en
$1$. Cerca de $0$: $\abs{\ln t}$ es integrable como en (a); cerca de
$\infty$: $\frac{\ln t}{1+t^2} = o(t^{-3/2})$: \omterm{def:b2:series:def}{absolutamente}
convergente. La sustitución $t = \frac1u$ envía $\intoo{0}{1}$ sobre
$\intoo{1}{\infty}$ y
\[
\int_0^1 \frac{\ln t}{1+t^2}\,\dd t
= \int_1^{\infty} \frac{-\ln u}{1 + u^{-2}}\cdot
\frac{\dd u}{u^2}
= -\int_1^\infty \frac{\ln u}{1+u^2}\,\dd u :
\]
las dos mitades se cancelan y la integral vale $0$. Moraleja: la
simetría bajo $t \mapsto \frac1t$ vale por una página de cálculo; el
mismo truco ya movía el~\cref{exo:b2:integration:3}.
\end{example}

\begin{example}[Un valor, tres integrales]\label{ex:b2:integration:onecos}
Estudiemos $I = \displaystyle\int_0^{\infty} \frac{1 - \cos
t}{t^2}\,\dd t$. En $0$: $1 - \cos t \sim \frac{t^2}2$, de modo que
el integrando se extiende continuamente con el valor $\frac12$: no
hay singularidad alguna. En $\infty$: $0 \leq \frac{1 - \cos t}{t^2}
\leq \frac{2}{t^2}$: convergencia absoluta
(\cref{thm:b2:integration:comparison}). Valor: integremos por partes
sobre $\intcc{\varepsilon}{M}$ con $u = 1 - \cos t$ y $v' = t^{-2}$:
\[
\int_\varepsilon^M \frac{1 - \cos t}{t^2}\,\dd t
= \Bigl[-\frac{1 - \cos t}{t}\Bigr]_\varepsilon^M
+ \int_\varepsilon^M \frac{\sin t}{t}\,\dd t .
\]
El corchete se anula en ambos extremos ($\frac{1 -
\cos\varepsilon}{\varepsilon} \sim \frac\varepsilon2$; numerador
acotado en $M$) y la integral tiende al valor de Dirichlet
$\frac\pi2$ (\cref{exo:b2:integration:10}): $I = \frac\pi2$.
Moraleja: con $1 - \cos t = 2\sin^2\frac t2$ y $u = \frac t2$,
\[
I = \int_0^\infty \frac{2\sin^2 u}{(2u)^2}\;2\,\dd u
= \int_0^\infty \Bigl(\frac{\sin u}{u}\Bigr)^{\!2}\dd u :
\]
los tres clásicos $\int_0^\infty\frac{\sin t}{t}\dd t$,
$\int_0^\infty\bigl(\frac{\sin t}{t}\bigr)^2\dd t$
(\cref{exo:b2:integration:11}) e $I$ comparten el valor $\frac\pi2$,
que se pasan unos a otros por partes y por sustitución; y solo el
primero es semiconvergente: la integración por partes cambió la
convergencia absoluta por un integrando más sencillo.
\end{example}

\begin{example}[Una integral semiconvergente]\label{ex:b2:integration:sint}
$\displaystyle\int_1^{\infty} \frac{\sin t}{t}\,\dd t$ \omterm{def:b2:integration:improper}{converge}:
integrando por partes,
\[
\int_1^x \frac{\sin t}{t}\dd t
= \Bigl[\frac{-\cos t}{t}\Bigr]_1^x - \int_1^x \frac{\cos
t}{t^2}\dd t ,
\]
donde el corchete tiene límite y la última integral \omterm{def:b2:integration:improper}{converge}
\omterm{def:b2:series:def}{absolutamente} ($\abs{\cos t}/t^2 \leq t^{-2}$). Pero no
\omterm{def:b2:series:def}{absolutamente}: de $\abs{\sin t} \geq \sin^2 t$,
\[
\int_1^x \frac{\abs{\sin t}}{t}\,\dd t
\;\geq\; \int_1^x \frac{\sin^2t}{t}\,\dd t
= \underbrace{\int_1^x \frac{\dd t}{2t}}_{=\ \frac12\ln x
\ \to\ \infty}
\;-\; \underbrace{\int_1^x \frac{\cos 2t}{2t}\,\dd
t}_{\text{convergente}} ,
\]
donde la última integral \omterm{def:b2:integration:improper}{converge} por la \emph{misma} integración
por partes de antes (con $\sin 2t$ en el corchete): una pieza
divergente menos una convergente diverge. Así pues,
$\int_1^\infty\frac{\sin t}{t}\dd t$ \omterm{def:b2:integration:improper}{converge} sin converger
\omterm{def:b2:series:def}{absolutamente}: el análogo integral de la serie alternada, con la
integración por partes en el papel del criterio de las alternadas.
\end{example}

\section{El teorema de convergencia}

\begin{theorem}[Convergencia dominada]\label{thm:b2:integration:dominated}
Sea $(f_n)$ una sucesión de funciones \omterm{def:b2:metric:continuity}{continuas} a trozos sobre un
intervalo $I$, convergente puntualmente a una $f$ \omterm{def:b2:metric:continuity}{continua} a
trozos, y supongamos que existe una $\varphi \geq 0$ integrable fija
($\int_I \varphi < \infty$) con\index{teorema de convergencia
dominada}
\[
\abs{f_n(t)} \leq \varphi(t) \qquad (t \in I,\ n \in \N).
\]
Entonces todas las $\int_I f_n$ e $\int_I f$ convergen
\omterm{def:b2:series:def}{absolutamente} y
\[
\int_I f_n \xrightarrow[n \to \infty]{} \int_I f .
\]
\end{theorem}
\admitted

\begin{remark}
La demostración honesta pertenece a la teoría de la integración de
Lebesgue del tercer año; el enunciado, en cambio, se usa
constantemente a partir de ahora. La hipótesis de \emph{dominación}
es todo el asunto: la convergencia puntual sola no basta ($f_n =
n\,\mathbf{1}_{\intoo{0}{1/n}}$, jorobas deslizantes: $\int f_n = 1
\not\to 0 = \int f$). El teorema vale también para un parámetro
\omterm{def:b2:metric:continuity}{continuo} ($f_\lambda$, $\lambda \to \lambda_0$), por la
caracterización sucesional de los límites.
\end{remark}

\begin{example}[Un límite gaussiano, por dominación]\label{ex:b2:integration:wallislimit}
Calculemos $\displaystyle\lim_{n\to\infty} I_n$, donde $I_n =
\int_0^\infty \Bigl(1 + \frac{t^2}{n}\Bigr)^{\!-n}\dd t$.
Puntualmente, $(1 + t^2/n)^n \to \eu^{t^2}$ (límite del interés
compuesto), de modo que los integrandos tienden a $\eu^{-t^2}$.
Dominación: la sucesión $n \mapsto (1 + u/n)^n$ es no decreciente
para $u \geq 0$ (la desigualdad entre medias aritmética y geométrica
sobre los $n + 1$ factores $1, 1 + \frac un, \dots, 1 + \frac un$ da
$(1 + \frac u{n+1})^{n+1} \geq (1 + \frac un)^n$), luego para $n
\geq 2$:
\[
\Bigl(1 + \frac{t^2}{n}\Bigr)^{\!-n}
\leq \Bigl(1 + \frac{t^2}{2}\Bigr)^{\!-2},
\]
un dominador integrable ($\sim 4t^{-4}$ en el infinito). Por
convergencia dominada:
\[
I_n \xrightarrow[n\to\infty]{} \int_0^\infty \eu^{-t^2}\dd t
= \frac{\sqrt\pi}{2}
\]
(la integral de Gauss del~\cref{exo:b2:integration:8}).
Comprobación final: la sustitución $t = \sqrt n\tan\theta$ calcula
$I_n$ exactamente, $I_n = \sqrt
n\int_0^{\pi/2}\cos^{2n-2}\theta\,\dd\theta = \sqrt n\,W_{2n-2}$, y
la asintótica de Wallis $W_m \sim \sqrt{\pi/(2m)}$
(\cref{lem:b2:comparison:wallis}) vuelve a dar $\sqrt n\,W_{2n-2}
\to \frac{\sqrt\pi}2$: los dos pilares de este capítulo y del
anterior concuerdan.
\end{example}

\begin{example}[Convergencia dominada con parámetro continuo]
\label{ex:b2:integration:arctanlimit}
Calculemos
\[
\lim_{x\to+\infty}\int_0^\infty
\frac{\arctan(xt)}{1+t^2}\,\dd t .
\]
Para cada $t > 0$, $\arctan(xt) \to \frac\pi2$ cuando $x \to
\infty$; y la dominación
\[
\Bigl|\frac{\arctan(xt)}{1+t^2}\Bigr|
\leq \frac{\pi/2}{1+t^2},
\qquad\text{integrable e independiente de } x,
\]
vale para todo $x$. Por la forma con parámetro \omterm{def:b2:metric:continuity}{continuo} del~\cref{thm:b2:integration:dominated} (caracterización sucesional:
compruébese a lo largo de toda $x_n \to \infty$),
\[
\int_0^\infty\frac{\arctan(xt)}{1+t^2}\,\dd t
\xrightarrow[x\to+\infty]{}
\frac\pi2\int_0^\infty\frac{\dd t}{1+t^2}
= \frac{\pi^2}{4} .
\]
Moraleja: el único punto $t = 0$, donde el límite puntual vale $0$ y
no $\frac\pi2$, no cambia nada; la función límite solo interviene a
través de su integral, una de las mercedes silenciosas del teorema.
\end{example}

\section{Integrales con parámetro}

\begin{theorem}[Continuidad bajo el signo integral]\label{thm:b2:integration:continuity}
Sea $f \colon A \times I \to \R$ ($A$ un \omterm{def:b2:metric:def}{espacio métrico} e $I$ un
intervalo) tal que: $t \mapsto f(x, t)$ es \omterm{def:b2:metric:continuity}{continua} a trozos para
cada $x$; $x \mapsto f(x, t)$ es \omterm{def:b2:metric:continuity}{continua} para cada $t$; y existe una
\emph{dominación} $\abs{f(x,t)} \leq \varphi(t)$ (con $\varphi$
integrable sobre $I$ e independiente de $x$). Entonces
\[
F(x) = \int_I f(x, t)\,\dd t
\]
está definida y es \omterm{def:b2:metric:continuity}{continua} sobre $A$.\index{integral con parámetro}
\end{theorem}

\begin{proof}
Buena definición: la dominación da la convergencia absoluta.
\omterm{def:b2:metric:continuity}{Continuidad} en $x_0$: para toda sucesión $x_n \to x_0$, las
funciones $g_n(t) = f(x_n, t)$ convergen puntualmente a $f(x_0, t)$
(por la \omterm{def:b2:metric:continuity}{continuidad} en $x$) bajo la dominación fija $\varphi$: la
convergencia dominada da $F(x_n) \to F(x_0)$; y se concluye por la
caracterización sucesional de la \omterm{def:b2:metric:continuity}{continuidad}
(\cref{def:b2:metric:continuity}).
\end{proof}

\begin{theorem}[Derivación bajo el signo integral]\label{thm:b2:integration:leibnizrule}
Sea $f \colon J \times I \to \R$ ($J$ un intervalo de parámetros)
tal que: $t \mapsto f(x,t)$ es integrable sobre $I$ para cada $x$; $x
\mapsto f(x,t)$ es de clase $C^1$ para cada $t$, con derivada
parcial $\frac{\partial f}{\partial x}$ \omterm{def:b2:metric:continuity}{continua} a trozos en $t$ y
dominada: $\bigl|\frac{\partial f}{\partial x}(x,t)\bigr| \leq
\psi(t)$ con $\psi$ integrable. Entonces $F(x) = \int_I f(x,t)\dd t$
es de clase $C^1$ sobre $J$ y
\[
F'(x) = \int_I \frac{\partial f}{\partial x}(x, t)\,\dd t .
\]
\end{theorem}

\begin{proof}
Fijemos $x$ y $h_n \to 0$. Los cocientes incrementales
\[
\frac{F(x + h_n) - F(x)}{h_n}
= \int_I \frac{f(x + h_n, t) - f(x, t)}{h_n}\,\dd t
\]
tienen integrandos que convergen puntualmente a $\frac{\partial
f}{\partial x}(x, t)$ y están dominados por $\psi(t)$: por la
desigualdad del valor medio aplicada en $x$ con $t$ fijo,
\[
\Bigl|\frac{f(x + h_n, t) - f(x,t)}{h_n}\Bigr|
\leq \sup_{\xi} \Bigl|\frac{\partial f}{\partial x}(\xi, t)\Bigr|
\leq \psi(t) .
\]
La convergencia dominada da el límite $\int_I \frac{\partial
f}{\partial x}(x,t)\dd t$ de los cocientes: $F$ es derivable con la
derivada anunciada, que es \omterm{def:b2:metric:continuity}{continua} por el~\cref{thm:b2:integration:continuity} aplicado a $\frac{\partial
f}{\partial x}$.
\end{proof}

\begin{example}[Una integral con parámetro contrastada con una fórmula]\label{ex:b2:integration:arctanparam}
Sea $F(x) = \displaystyle\int_0^\infty \frac{\dd t}{t^2 + x}$ para
$x > 0$. Sobre todo $\intcc{a}{b} \subset \intoo{0}{\infty}$, el
integrando está dominado por $\frac{1}{t^2 + a}$, integrable e
independiente de $x$: $F$ es \omterm{def:b2:metric:continuity}{continua}
(\cref{thm:b2:integration:continuity}). Aquí el teorema puede
contrastarse con un valor explícito:
\[
F(x) = \Bigl[\frac{1}{\sqrt x}\arctan\frac{t}{\sqrt
x}\Bigr]_0^\infty = \frac{\pi}{2\sqrt x} ,
\]
visiblemente \omterm{def:b2:metric:continuity}{continua}. Derivemos ahora bajo la integral: la
derivada en $x$, $-\frac{1}{(t^2+x)^2}$, está dominada sobre
$\intcc ab$ por $\frac{1}{(t^2+a)^2}$, integrable, y el~\cref{thm:b2:integration:leibnizrule} da
\[
F'(x) = -\int_0^\infty \frac{\dd t}{(t^2 + x)^2}
\qquad\text{mientras que}\qquad
F'(x) = -\frac{\pi}{4}\,x^{-3/2} ,
\]
de modo que hemos \emph{calculado} gratis una integral nueva:
$\int_0^\infty\frac{\dd t}{(t^2+x)^2} = \frac{\pi}{4x^{3/2}}$.
Moraleja: derivar una \omterm{thm:b2:integration:continuity}{integral con parámetro} conocida es una
fábrica de fórmulas nuevas; iterando se obtiene
$\int_0^\infty\frac{\dd t}{(t^2+1)^n}$ para todo $n$, sin
sustituciones trigonométricas.
\end{example}

\begin{method}[Cómo estudiar una integral impropia]\label{met:b2:integration:study}
Dada $\int_a^b f$:
\begin{enumerate}
  \item Localícese el problema: enumérense los extremos (o los
        puntos interiores) donde $f$ no está acotada o el intervalo
        es infinito, y sepárese de modo que cada trozo tenga
        exactamente un extremo problemático.
  \item Si $f$ tiene signo constante cerca de ese extremo, hállese
        un equivalente y compárese con las escalas de referencia del~\cref{thm:b2:integration:comparison}.
  \item Si $f$ oscila, pruébese primero con $\abs f$ (convergencia
        absoluta). Si $\int\abs f$ diverge, intégrese por partes
        para cambiar la oscilación por decrecimiento, como en el~\cref{ex:b2:integration:sint}; minoraciones del tipo
        $\abs{\sin t} \geq \sin^2t$ detectan la semiconvergencia
        auténtica.
  \item Para obtener un valor, y no solo el carácter: partes,
        sustitución o un parámetro (derívese una integral más
        sencilla, como en el~\cref{ex:b2:integration:arctanparam} y
        el~\cref{ex:b2:integration:laplace}).
  \item Comprobaciones de sensatez sobre todo valor calculado: el
        signo y el tamaño aproximado contra una cota tosca
        ($\int_0^\infty \eu^{-t^2}\dd t \in \intoo{0}{1 +
        \int_1^\infty \eu^{-t}}$, de modo que $\frac{\sqrt\pi}{2}
        \approx 0.886$ es plausible); y la coherencia dimensional
        por escalado ($t \mapsto \lambda t$ debe reescalar ambos
        miembros del mismo modo: el detector más rápido de un factor
        perdido).
\end{enumerate}
\end{method}

\begin{remark}[Errores frecuentes]
Tres errores recurrentes. \emph{(i) Dominadores que dependen del
parámetro:} la dominación $\abs{f(x,t)} \leq \varphi(t)$ ha de ser
uniforme en $x$ sobre el conjunto considerado; suele valer sobre
segmentos $\intcc ab$ pero no globalmente. Para
$\int_0^\infty\eu^{-xt}\dd t$ no hay ningún dominador integrable
válido para todo $x > 0$, y sin embargo dominar sobre $x \geq a > 0$
basta para trabajar en toda la semirrecta \omterm{def:b2:metric:topology}{abierta}, ya que la
\omterm{def:b2:metric:continuity}{continuidad} y las derivadas son nociones locales. \emph{(ii)
Comparar integrandos con signo:} la caja de herramientas de
comparación es para funciones \emph{no negativas}; de $\abs f \leq
g$ con $\int g$ divergente no puede concluirse nada, pues
$\int_1^\infty\frac{\sin t}t\,\dd t$ \omterm{def:b2:integration:improper}{converge} aunque toda comparación
con $\frac1t$ fracase. \emph{(iii) Olvidar la mitad del problema:}
sobre $\intoo{0}{\infty}$ hay que estudiar siempre los dos extremos
por separado; $\int_0^\infty\frac{\dd t}{t}$ diverge en
\emph{ambos}, y una separación de aspecto convergente puede cancelar
en silencio dos infinitos. El reflejo seguro es la lista del~\cref{met:b2:integration:study}.
\end{remark}

\begin{example}[Un caso frontera de Bertrand, hasta la cifra]\label{ex:b2:integration:bertrandboundary}
La escala $\int^\infty\frac{\dd t}{t(\ln t)^\beta}$ del~\cref{thm:b2:integration:comparison} se sitúa exactamente en el
borde de las escalas de potencias; sus casos frontera merecen un
cálculo \omterm{def:b2:metric:complete}{completo}. Para $\beta = 2$:
\[
\int_\eu^{\infty}\frac{\dd t}{t(\ln t)^2}
= \Bigl[-\frac{1}{\ln t}\Bigr]_\eu^{\infty}
= 0 - (-1) = 1 ,
\]
una integral convergente con un valor agradablemente exacto;
mientras que para $\beta = 1$,
\[
\int_\eu^{x}\frac{\dd t}{t\ln t}
= \bigl[\ln\ln t\bigr]_\eu^{x} = \ln\ln x
\longrightarrow \infty ,
\]
divergente, pero tan despacio que alcanzar $\ln\ln x = 10$ exige $x
= \eu^{\eu^{10}} \approx 10^{9566}$. Moraleja: entre ``toda potencia
$t^{-1-\varepsilon}$ \omterm{def:b2:integration:improper}{converge}'' y ``$t^{-1}$ diverge'' vive una
escalera infinita de escalas logarítmicas, cada una afinando la
anterior; la sustitución $u = \ln t$ derrumba cada peldaño sobre el
precedente, y por eso los criterios de Bertrand son el eco de los de
Riemann un nivel más arriba.
\end{example}

\begin{remark}[Perspectivas dentro de este volumen]
Las herramientas de este capítulo están a punto de aparecer por
todas partes. La convergencia dominada es el motor de las
identidades aproximadas del capítulo siguiente (núcleos
deslizantes, tanto los de Bernstein como los de Fejér); la
\omterm{def:b2:metric:continuity}{continuidad} y la derivación bajo el signo integral producen el
cálculo con los coeficientes de Fourier en el capítulo
correspondiente, donde cada $c_n(f)$ es una \omterm{thm:b2:integration:continuity}{integral con parámetro}
disfrazada. La función $\Gamma$ regresa dos veces: en el capítulo de
integrales múltiples, donde una integral doble demuestra por fin la
fórmula Beta--Gamma de Euler en toda su generalidad, y en los
capítulos de probabilidad, donde las integrales de tipo $\Gamma$
normalizan las densidades estándar y calculan sus momentos. Y la
semiconvergente $\int\frac{\sin t}{t}$ resurge como la constante de
Gibbs del capítulo de Fourier: la misma integral, midiendo el
sobrepaso de las sumas parciales en un salto.
\end{remark}

\begin{definition}[La función $\Gamma$]\label{def:b2:integration:gamma}
Para $x > 0$:\index{función Gamma}
\[
\Gamma(x) = \int_0^{\infty} t^{x-1}\,\eu^{-t}\,\dd t ,
\]
convergente en ambos extremos ($t^{x-1}$ es integrable en $0^+$ para
$x > 0$; y hay decrecimiento exponencial en $\infty$).
\end{definition}

\begin{theorem}\label{thm:b2:integration:gammaprops}
$\Gamma$ es \omterm{def:b2:metric:continuity}{continua} sobre $\intoo{0}{+\infty}$, satisface la
ecuación funcional
\[
\Gamma(x + 1) = x\,\Gamma(x),
\qquad \Gamma(1) = 1,
\qquad\text{de donde}\qquad \Gamma(n + 1) = n! ,
\]
y es de clase $C^1$ (de hecho, $C^\infty$) con $\Gamma'(x) =
\int_0^\infty t^{x-1}\eu^{-t}\ln t\,\dd t$.
\end{theorem}

\begin{proof}
Ecuación funcional: intégrese por partes sobre
$\intcc{\varepsilon}{M}$ y háganse tender los extremos: $\int
t^{x}\eu^{-t} = [-t^x\eu^{-t}] + x\int t^{x-1}\eu^{-t}$, con
términos de frontera que se anulan; en efecto,
$\varepsilon^x\eu^{-\varepsilon} \to 0$ cuando $\varepsilon \to 0^+$
porque $x > 0$, y $M^x\eu^{-M} \to 0$ cuando $M \to \infty$ porque
la exponencial gana a toda potencia; y ambas integrales truncadas
convergen a sus valores impropios por la convergencia establecida en
la~\cref{def:b2:integration:gamma}. Además $\Gamma(1) = \int
\eu^{-t} = 1$, y la inducción da el factorial.

\omterm{def:b2:metric:continuity}{Continuidad} sobre $\intcc{a}{b} \subset \intoo{0}{\infty}$:
domínese $t^{x-1}\eu^{-t}$ por $\varphi(t) = (t^{a-1} +
t^{b-1})\eu^{-t}$, integrable e independiente de $x \in
\intcc{a}{b}$: el~\cref{thm:b2:integration:continuity} se aplica en
todo segmento así y, por tanto, en toda la semirrecta.
Derivabilidad: la derivada en $x$, $t^{x-1}\eu^{-t}\ln t$, está
dominada sobre $\intcc{a}{b}$ por $(t^{a-1} +
t^{b-1})\eu^{-t}\,\abs{\ln t}$, todavía integrable: aplíquese el~\cref{thm:b2:integration:leibnizrule}; iterando se obtienen todas
las derivadas (cada una añade una potencia de $\ln t$, inofensiva).
\end{proof}

\begin{example}[Factoriales de semienteros]\label{ex:b2:integration:halffactorial}
La ecuación funcional y $\Gamma\bigl(\frac12\bigr) = \sqrt\pi$ (a
una sustitución del~\cref{exo:b2:integration:8}: póngase $t = u^2$
en la integral que la define) generan todos los valores
semienteros:
\[
\Gamma\Bigl(\frac32\Bigr) = \frac12\,\Gamma\Bigl(\frac12\Bigr)
= \frac{\sqrt\pi}{2},
\qquad
\Gamma\Bigl(\frac52\Bigr) = \frac32\cdot\frac{\sqrt\pi}{2}
= \frac{3\sqrt\pi}{4},
\qquad
\Gamma\Bigl(\frac72\Bigr) = \frac{15\sqrt\pi}{8} .
\]
Como $\Gamma(n+1) = n!$, es lícito decir que ``$\frac12! =
\frac{\sqrt\pi}{2} \approx 0.886$'': el factorial ha quedado
interpolado, y la curva interpoladora baja por debajo de $1$ entre
$0! = 1$ y $1! = 1$ (su mínimo $\approx 0.8856$ en $x \approx
1.4616$ concuerda con la imagen de convexidad de la parte I del
problema de fin de semana). Moraleja: nada en la integral
$\int_0^\infty t^{x-1}\eu^{-t}\dd t$ privilegia a los enteros; el
carácter discreto del factorial fue un accidente del recuento, y
$\sqrt\pi$ es lo que vive entre $1$ y $1$.
\end{example}

\begin{remark}[Adónde va $\Gamma$ a partir de aquí]
El problema de fin de semana de este capítulo construye todo el
cálculo de Euler en torno a $\Gamma$: la \omterm{pb:b2:integration:1}{función Beta}, sus
recurrencias por integración por partes, las integrales de Wallis
como valores de Beta y la fórmula límite de Gauss. El capítulo de
integrales múltiples demuestra la fórmula Beta--Gamma de Euler para
todos los argumentos mediante una integral doble; los capítulos de
probabilidad reencuentran $\Gamma$ en la normalización de las
densidades más frecuentes y en los momentos de los tiempos de
espera. El volumen del tercer año reconstruye $\Gamma$ sobre
cimientos de Lebesgue, demuestra el teorema de unicidad de
Bohr--Mollerup y extiende la fórmula de Stirling de los enteros a la
semirrecta real por convergencia dominada.
\end{remark}

\begin{example}[Un cálculo clásico por derivación]\label{ex:b2:integration:laplace}
Para $x \in \R$, sea $F(x) = \int_0^{\infty} \eu^{-t^2}\cos(xt)\,\dd
t$ (\omterm{def:b2:series:def}{absolutamente} convergente, dominada por $\eu^{-t^2}$). Por el~\cref{thm:b2:integration:leibnizrule} (dominación de la derivada en
$x$ por $t\,\eu^{-t^2}$, integrable):
\[
F'(x) = -\int_0^\infty t\,\eu^{-t^2}\sin(xt)\,\dd t
= \Bigl[\frac{\eu^{-t^2}}{2}\sin(xt)\Bigr]_0^\infty
- \frac x2\int_0^\infty \eu^{-t^2}\cos(xt)\,\dd t
= -\frac x2\,F(x),
\]
(por partes con $u' = t\eu^{-t^2}$). La ecuación diferencial $F' =
-\frac x2 F$ se integra hasta $F(x) = F(0)\,\eu^{-x^2/4}$: la
integral de tipo gaussiano se reproduce a sí misma. La constante
$F(0) = \int_0^\infty \eu^{-t^2}\dd t = \frac{\sqrt\pi}{2}$ se
calcula en el~\cref{exo:b2:integration:8} y, de nuevo, por
integración doble, en el~\cref{ch:b2:multint}.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:integration:1}
Carácter de $\displaystyle\int_0^1 \frac{\dd t}{\sqrt{t(1-t)}}$;
$\;\displaystyle\int_1^\infty \frac{\ln t}{t^2}\dd t$;
$\;\displaystyle\int_0^\infty \frac{\dd t}{1 + t^2\sin^2 t}$
\emph{(compara con el comportamiento divergente de tipo armónico
cerca de $t = n\pi$)}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:integration:2}
Calcula $\displaystyle\int_0^\infty t^n \eu^{-\lambda t}\,\dd t$
($\lambda > 0$) mediante $\Gamma$, y $\displaystyle\int_0^1 (\ln
t)^n \dd t$ mediante la sustitución $t = \eu^{-u}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:integration:3}
Demuestra que $\displaystyle\int_0^{\infty} \frac{\dd t}{(1 +
t^2)(1 + t^x)}$ está bien definida para todo $x \in \R$ y no depende
de $x$. \emph{(Sustituye $t \mapsto \frac1t$ y promedia las dos
expresiones.)} ¿Cuál es su valor?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:4}
(Integrales de Bertrand en un extremo finito) ¿Para qué $(\alpha,
\beta)$ \omterm{def:b2:integration:improper}{converge} $\displaystyle\int_0^{1/2}
\frac{\dd t}{t^\alpha\,\abs{\ln t}^\beta}$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:5}
Sea $F(x) = \displaystyle\int_0^{\infty} \frac{\eu^{-xt}}{1 +
t^2}\,\dd t$ para $x \geq 0$. Demuestra que $F$ es \omterm{def:b2:metric:continuity}{continua} sobre
$\intco{0}{\infty}$, de clase $C^2$ sobre $\intoo{0}{\infty}$, que
allí cumple $F'' + F = \frac1x$ y que $F(x) \to 0$ cuando $x \to
+\infty$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:6}
(Frullani) Sea $f$ \omterm{def:b2:metric:continuity}{continua} sobre $\intco{0}{+\infty}$ con límite
finito $f(\infty)$ en $+\infty$. Demuestra que para $a, b > 0$:
\[
\int_0^{\infty} \frac{f(at) - f(bt)}{t}\,\dd t
= \bigl(f(0) - f(\infty)\bigr)\,\ln\frac ba .
\]
\emph{(Sobre $\intcc{\varepsilon}{M}$, sustituye en cada pieza y
reagrupa hasta $\int_{a\varepsilon}^{b\varepsilon} - \int_{aM}^{bM}$
de $\frac{f(u)}u\,\dd u$; encaja usando la \omterm{def:b2:metric:continuity}{continuidad} en $0$ y el
límite en $\infty$.)} Calcula $\int_0^\infty \frac{\eu^{-t} -
\eu^{-2t}}{t}\dd t$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:7}
Justifica y calcula $\lim_{n\to\infty} \displaystyle\int_0^n \Bigl(1
- \frac tn\Bigr)^{\!n} t^{x-1}\,\dd t$ para $x > 0$
\emph{(convergencia dominada con $\varphi(t) = \eu^{-t}t^{x-1}$,
usando $(1 - t/n)^n \leq \eu^{-t}$; el límite es $\Gamma(x)$)}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:8}
(La integral de Gauss mediante un truco con parámetro) Para $x \geq
0$ pongamos
\[
G(x) = \Bigl(\int_0^x \eu^{-t^2}\dd t\Bigr)^{\!2},
\qquad
H(x) = \int_0^1 \frac{\eu^{-x^2(1+t^2)}}{1 + t^2}\,\dd t .
\]
Demuestra que $G' + H' = 0$ (deriva $H$ bajo la integral y sustituye
$u = xt$ en la integral resultante), deduce $G(x) + H(x) =
\frac\pi4$ para todo $x$, y concluye
\[
\int_0^{\infty} \eu^{-t^2}\,\dd t = \frac{\sqrt\pi}{2} .
\]
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:9}
Demuestra que $\Gamma$ es \emph{log-convexa}: $\ln\Gamma$ es convexa
sobre $\intoo{0}{\infty}$. \emph{(La desigualdad de Cauchy--Schwarz
para integrales aplicada a $t^{(x+y)/2 - 1}\eu^{-t} =
\bigl(t^{x-1}\eu^{-t}\bigr)^{1/2} \bigl(t^{y-1}\eu^{-t}\bigr)^{1/2}$
da $\Gamma\bigl(\frac{x+y}{2}\bigr)^2 \leq \Gamma(x)\Gamma(y)$;
combina con la \omterm{def:b2:metric:continuity}{continuidad} y el~\cref{exo:b2:realfun:8}.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:10}
(Integral de Dirichlet) Pongamos $F(x) = \displaystyle\int_0^{\infty}
\frac{\sin t}{t}\,\eu^{-xt}\,\dd t$ para $x > 0$.
\begin{enumerate}
  \item Justifica que $F'(x) = -\frac{1}{1 + x^2}$ (deriva bajo la
        integral; calcula $\int_0^\infty \eu^{-xt}\sin t\,\dd t$ con
        dos integraciones por partes).
  \item Demuestra que $F(x) \to 0$ cuando $x \to +\infty$ y deduce
        $F(x) = \frac\pi2 - \arctan x$.
  \item Admitiendo la \omterm{def:b2:metric:continuity}{continuidad} de $F$ en $0^+$ (un teorema de
        tipo Abel), concluye el valor de la integral
        semiconvergente:
        \[
        \int_0^{\infty} \frac{\sin t}{t}\,\dd t = \frac{\pi}{2}.
        \]
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:integration:11}
Justifica la convergencia de $\displaystyle\int_0^\infty
\Bigl(\frac{\sin t}{t}\Bigr)^{\!2}\dd t$ y calcúlala después con una
integración por partes y el~\cref{exo:b2:integration:10}:
\[
\int_0^\infty \Bigl(\frac{\sin t}{t}\Bigr)^{\!2}\dd t
= \frac{\pi}{2} .
\]
(El mismo valor que $\int_0^\infty \frac{\sin t}{t}\dd t$, pero esta
vez la convergencia es absoluta.)
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:integration:12}
(La cola gaussiana) Para $x > 0$ pongamos $T(x) = \displaystyle
\int_x^\infty \eu^{-t^2}\dd t$.
\begin{enumerate}
  \item Escribiendo $\eu^{-t^2} = \frac{1}{-2t}\cdot(-2t\,\eu^{-t^2})$,
        integra por partes dos veces para obtener
        \[
        T(x) = \eu^{-x^2}\Bigl(\frac{1}{2x} -
        \frac{1}{4x^3}\Bigr) +
        \frac34\int_x^\infty \frac{\eu^{-t^2}}{t^4}\,\dd t .
        \]
  \item Acota el resto: $0 \leq \frac34\int_x^\infty
        t^{-4}\eu^{-t^2}\dd t \leq \frac{3}{8x^5}\,\eu^{-x^2}$, y
        deduce el encaje
        \[
        \eu^{-x^2}\Bigl(\frac{1}{2x} - \frac{1}{4x^3}\Bigr)
        \leq T(x) \leq \frac{\eu^{-x^2}}{2x},
        \qquad\text{de donde}\qquad
        T(x) \sim \frac{\eu^{-x^2}}{2x} \quad (x \to +\infty).
        \]
  \item ¿Por qué la serie alternada completa que se obtiene
        iterando la integración por partes no puede converger nunca
        para $x$ fijo? \emph{(Compara el crecimiento de los
        coeficientes $1\cdot3\cdots(2k-1)$ con el de las potencias
        $(2x^2)^k$.)}
\end{enumerate}
\end{exercise}

\section{Problema: las integrales de Euler ---Beta, Gamma y la
fórmula límite de Gauss}

\begin{problem}\label{pb:b2:integration:1}
La función $\Gamma$ de la~\cref{def:b2:integration:gamma} es una
mitad del cálculo de integrales de Euler; la otra mitad es la
\emph{función Beta}\index{función Beta}
\[
B(x, y) = \int_0^1 t^{x-1}(1 - t)^{y-1}\,\dd t .
\]
Este problema desarrolla el par $(\Gamma, B)$ solo con las
herramientas de este capítulo ---integración por partes,
sustitución y convergencia dominada--- y culmina en la
\emph{fórmula Beta--Gamma de Euler} $B(x,y) =
\frac{\Gamma(x)\Gamma(y)} {\Gamma(x+y)}$ sobre los semienteros y en
la \emph{fórmula límite de Gauss}\index{fórmula límite de Gauss}
para $\Gamma$. Por el camino, las integrales de Wallis del~\cref{lem:b2:comparison:wallis} reaparecen como valores de Beta, y
la fórmula de duplicación de Legendre cae por su propio peso.

\medskip
\noindent\textbf{Parte I --- Estructura fina de $\Gamma$.}
\begin{enumerate}
  \item Recuerda por qué $\Gamma(x) = \int_0^\infty
        t^{x-1}\eu^{-t}\dd t$ \omterm{def:b2:integration:improper}{converge} exactamente para $x > 0$, y
        prueba que
        \[
        \Gamma(x) \sim \frac1x \qquad (x \to 0^+)
        \]
        \emph{(ecuación funcional más \omterm{def:b2:metric:continuity}{continuidad} de $\Gamma$ en
        $1$)}.
  \item Demuestra que $\Gamma\bigl(\tfrac12\bigr) = \sqrt\pi$
        \emph{(sustituye $t = u^2$ e invoca el~\cref{exo:b2:integration:8})}, y deduce $\int_\R
        \eu^{-u^2/2}\dd u = \sqrt{2\pi}$.
  \item Prueba por inducción que, para $n \in \N$,
        \[
        \Gamma\Bigl(n + \frac12\Bigr)
        = \frac{(2n)!}{4^n\,n!}\,\sqrt\pi .
        \]
  \item Justifica que $\Gamma''(x) = \int_0^\infty
        t^{x-1}\eu^{-t}(\ln t)^2\dd t > 0$ y deduce que $\Gamma$ es
        estrictamente convexa, alcanza un único mínimo en cierto
        $x_0 \in \intoo{1}{2}$ \emph{($\Gamma(1) = \Gamma(2) = 1$ y
        Rolle)}, decrece sobre $\intoo{0}{x_0}$ y crece sobre
        $\intoo{x_0}{\infty}$.
  \item Prueba que $\Gamma$ gana a toda potencia: para cada $k \in
        \N$, $x^k = o\bigl(\Gamma(x)\bigr)$ cuando $x \to +\infty$
        \emph{(encaja $x$ entre enteros y usa $\Gamma(n+1) = n!$ con
        la monotonía de la pregunta 4)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- La \omterm{pb:b2:integration:1}{función Beta}, por partes.}
\begin{enumerate}[resume]
  \item Prueba que $B(x,y)$ \omterm{def:b2:integration:improper}{converge} exactamente para $x > 0$ e $y >
        0$, y que $B(x,y) = B(y,x)$.
  \item Calcula $B(x, 1) = \frac1x$ y demuestra por integración por
        partes que, para $x, y > 0$,
        \[
        B(x, y+1) = \frac{y}{x}\,B(x+1, y) .
        \]
  \item A partir de la descomposición $t^{x-1}(1-t)^{y-1} =
        t^{x}(1-t)^{y-1} + t^{x-1}(1-t)^{y}$, deduce $B(x,y) =
        B(x+1,y) + B(x,y+1)$ y combínalo con la pregunta 7 hasta las
        \emph{relaciones de descenso}
        \[
        B(x, y+1) = \frac{y}{x+y}\,B(x,y),
        \qquad
        B(x+1, y) = \frac{x}{x+y}\,B(x,y) .
        \]
  \item Deduce que, para $m, n \geq 1$ enteros,
        \[
        B(m, n) = \frac{(m-1)!\,(n-1)!}{(m+n-1)!}
        = \frac{1}{(m+n-1)\binom{m+n-2}{m-1}} .
        \]
  \item Demuestra la \emph{fórmula de Euler con un argumento
        entero}: para todo $x > 0$ y todo $n \in \N^*$,
        \[
        B(x, n) = \frac{\Gamma(x)\,\Gamma(n)}{\Gamma(x + n)}
        \]
        \emph{(inducción sobre $n$: ambos miembros valen $\frac1x$
        en $n = 1$ y obedecen la misma relación de descenso)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Las integrales de Wallis como
valores de Beta.}
\begin{enumerate}[resume]
  \item Sustituye $t = \sin^2\theta$ para obtener la forma
        trigonométrica
        \[
        B(x, y) = 2\int_0^{\pi/2}
        \sin^{2x-1}\theta\,\cos^{2y-1}\theta\,\dd\theta .
        \]
  \item Deduce $W_n = \frac12\,B\bigl(\frac{n+1}2, \frac12\bigr)$
        para la integral de Wallis $W_n = \int_0^{\pi/2}\sin^n
        \theta\,\dd\theta$, y recupera la recurrencia $W_n =
        \frac{n-1}{n}W_{n-2}$ del~\cref{lem:b2:comparison:wallis}
        únicamente a partir de las relaciones de descenso de la
        pregunta 8.
  \item Calcula $B\bigl(\frac12, \frac12\bigr) = 2W_0 = \pi$ y
        contrástalo con $\Gamma\bigl(\frac12\bigr)^2/\Gamma(1)$: la
        fórmula de Euler se cumple en $\bigl(\frac12,
        \frac12\bigr)$.
  \item Deduce de la recurrencia la forma cerrada $W_{2n} =
        \frac\pi2\, \frac{(2n)!}{4^n(n!)^2}$ y verifica que
        \[
        B\Bigl(n + \frac12, \frac12\Bigr)
        = \frac{\Gamma\bigl(n + \frac12\bigr)\Gamma\bigl(
        \frac12\bigr)}{\Gamma(n+1)} .
        \]
        Concluye, por inducción con las relaciones de descenso, que
        la fórmula de Euler $B(x,y) =
        \frac{\Gamma(x)\Gamma(y)}{\Gamma(x+y)}$ vale siempre que
        $2x$ y $2y$ sean enteros positivos.
  \item Sustituye $u = \frac{t}{1-t}$ para obtener la tercera forma
        clásica
        \[
        B(x,y) = \int_0^\infty
        \frac{u^{x-1}}{(1+u)^{x+y}}\,\dd u ,
        \]
        y comprueba directamente el caso $x = y = \frac12$ ($u =
        v^2$ lo reduce a $\int_0^\infty\frac{2\,\dd v}{1+v^2}$).
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- La fórmula límite de Gauss.}
\begin{enumerate}[resume]
  \item Para $x > 0$ y $n \in \N^*$, demuestra mediante $n$
        integraciones por partes sucesivas que
        \[
        \int_0^n \Bigl(1 - \frac tn\Bigr)^{\!n} t^{x-1}\,\dd t
        = \frac{n!\;n^x}{x(x+1)\cdots(x+n)} .
        \]
  \item Concluye con el~\cref{exo:b2:integration:7} (convergencia
        dominada) la \emph{fórmula límite de Gauss}:
        \[
        \Gamma(x) = \lim_{n\to\infty}
        \frac{n!\;n^x}{x(x+1)\cdots(x+n)}
        \qquad (x > 0).
        \]
  \item Tomando logaritmos, prueba que para $x > 0$
        \[
        \ln\Gamma(x) = -\ln x - \gamma x +
        \sum_{k=1}^{\infty}\Bigl(\frac xk -
        \ln\Bigl(1 + \frac xk\Bigr)\Bigr),
        \]
        donde $\gamma$ es la constante de Euler
        (\cref{ex:b2:comparison:harmonic}); justifica la
        convergencia de la serie \emph{(el término general es $\sim
        \frac{x^2}{2k^2}$)}.
  \item Usa la fórmula de Gauss en $x = \frac12$ y la asintótica del
        binomial central $\binom{2n}{n} \sim
        \frac{4^n}{\sqrt{\pi n}}$
        (\cref{ex:b2:comparison:centralbinomial}) para volver a
        calcular $\Gamma\bigl(\frac12\bigr) = \sqrt\pi$: la
        constante de Stirling y la integral de Gauss son el mismo
        número con dos disfraces.
  \item Comprueba que la fórmula de Gauss vuelve a demostrar la
        ecuación funcional: a partir de la identidad exacta
        \[
        \frac{n!\,n^{x+1}}{(x+1)\cdots(x+n+1)}
        = \frac{n!\,n^{x}}{x(x+1)\cdots(x+n)}\cdot
        \frac{n\,x}{x+n+1},
        \]
        concluye de nuevo $\Gamma(x+1) = x\,\Gamma(x)$. (La fórmula
        de Gauss determina $\Gamma$ por completo; el volumen del
        tercer año demuestra el teorema más fino de
        Bohr--Mollerup: la ecuación funcional más la log-convexidad
        ya fijan $\Gamma$.)
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dividendos.}
\begin{enumerate}[resume]
  \item Para $a > 0$, prueba que $\int_0^\infty \eu^{-t^a}\dd t =
        \Gamma\bigl(1 + \frac1a\bigr)$ y calcula el límite cuando $a
        \to +\infty$ por convergencia dominada \emph{(límite puntual
        $\mathbf 1_{t < 1}$; domínese por $1$ sobre $\intoc{0}{1}$ y
        por $\eu^{-t^2}$ más allá, para $a \geq 2$)}. Contrasta la
        respuesta con la \omterm{def:b2:metric:continuity}{continuidad} de $\Gamma$.
  \item Para $n \geq 1$, prueba que
        \[
        \int_0^1 \frac{\dd t}{\sqrt{1 - t^n}}
        = \frac1n\,B\Bigl(\frac1n, \frac12\Bigr),
        \]
        y recupera los valores $2$ ($n = 1$) y $\frac\pi2$ ($n =
        2$). (Para $n = 4$ esta es la \emph{constante de la
        lemniscata}, que no tiene forma cerrada elemental; su
        historia pertenece a la teoría de las integrales
        elípticas.)
  \item (Momentos) Para $x > 0$ y $k \in \N$, prueba que
        \[
        \frac{1}{\Gamma(x)}\int_0^\infty
        t^{k}\,t^{x-1}\eu^{-t}\,\dd t
        = \frac{\Gamma(x+k)}{\Gamma(x)}
        = x(x+1)\cdots(x+k-1),
        \]
        el factorial ascendente; comprueba que $x = 1$ da $k!$. (En
        los capítulos de probabilidad, este es el momento $k$-ésimo
        de una densidad estándar de tiempo de espera.)
  \item Demuestra la identidad de Beta, válida para \emph{todo} $x >
        0$:
        \[
        B(x, x) = 2^{1-2x}\,B\Bigl(x, \frac12\Bigr)
        \]
        \emph{(sustituye $t = \frac{1+s}2$, aprovecha la simetría en
        $s$ y pon después $s = \sqrt v$)}. Deduce, para $2x \in
        \N^*$, la \emph{fórmula de duplicación de Legendre}
        \[
        \Gamma(x)\,\Gamma\Bigl(x + \frac12\Bigr)
        = 2^{1-2x}\,\sqrt\pi\;\Gamma(2x),
        \]
        y verifícala directamente en $x = n$ mediante la pregunta 3.
        (Para $x$ general se sigue de la misma identidad una vez
        conocida la fórmula de Euler para todos los argumentos: la
        demostración por integral doble del capítulo de integrales
        múltiples.)
  \item Síntesis. Una frase para cada punto: (i) dónde soportó la
        integración por partes toda la parte II; (ii) dónde
        intervino la convergencia dominada en las partes IV y V;
        (iii) qué insumos asintóticos se importaron del capítulo de
        comparación; (iv) qué queda ya demostrado de la fórmula de
        Euler $B(x,y) = \Gamma(x)\Gamma(y)/\Gamma(x+y)$ y qué le
        queda por zanjar a la integral doble.
\end{enumerate}
\end{problem}
