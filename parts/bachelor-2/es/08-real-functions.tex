\chapter{Funciones de una variable real}\label{ch:b2:realfun}

Antes de que el análisis pase a las funciones de funciones
(\cref{ch:b2:funcseq}) conviene conocer el paisaje de una variable con
más detalle del que exigía el primer año: cuán discontinua puede ser
una función monótona, cuán regular ha de ser una función convexa y de
qué propiedades especiales gozan las derivadas (Darboux). Estos
resultados estructurales son breves, afilados y muy queridos por los
examinadores.

\section{Funciones monótonas}

\begin{theorem}[Regularidad de las funciones monótonas]\label{thm:b2:realfun:monotone}
Sea $f \colon I \to \R$ creciente sobre un intervalo.
\begin{enumerate}
  \item En todo punto interior $a$ existen los límites laterales:
        \[
        f(a^-) = \sup_{x < a} f(x) \;\leq\; f(a) \;\leq\;
        f(a^+) = \inf_{x > a} f(x) ;
        \]
        toda discontinuidad es un \emph{salto}.
  \item El conjunto de discontinuidades de $f$ es a lo sumo
        \omterm{def:b2:structures:countable}{numerable}.\index{función monótona!discontinuidades}
\end{enumerate}
\end{theorem}

\begin{proof}
(1) El conjunto $\{f(x) : x < a\}$ es no vacío y está acotado
superiormente por $f(a)$: su supremo $s$ cumple $f(x) \to s$ cuando
$x \to a^-$ (dado $\varepsilon$, algún $f(x_0) > s - \varepsilon$, y
la monotonía atrapa $f(x) \in \intoc{s - \varepsilon}{s}$ para $x
\in \intoo{x_0}{a}$). Simétricamente por la derecha.

(2) A cada discontinuidad $a$ asociémosle el intervalo \omterm{def:b2:metric:topology}{abierto} no
vacío $J_a = \intoo{f(a^-)}{f(a^+)}$ (un salto auténtico). Para
discontinuidades $a < b$, los intervalos $J_a$ y $J_b$ son
disjuntos: $f(a^+) \leq f(c) \leq f(b^-)$ para cualquier $c$
intermedio. Cada $J_a$ contiene un racional, y discontinuidades
distintas reciben racionales distintos: se obtiene una inyección del
conjunto de discontinuidades en $\Q$, que es \omterm{def:b2:structures:countable}{numerable}
(\cref{prop:b2:structures:countablestable}).
\end{proof}

\begin{example}
La cota es óptima: fijemos una enumeración $(r_n)$ de $\Q \cap
\intoo{0}{1}$ y pongamos $f(x) = \sum_{n : r_n \leq x} 2^{-n}$ (una
definición mediante \omterm{def:b2:series:summable}{familias sumables},
\cref{def:b2:series:summable}). Entonces $f$ es creciente sobre
$\intcc{0}{1}$ y discontinua exactamente en cada racional de
$\intoo{0}{1}$ (con salto $2^{-n}$ en $r_n$): una función monótona
\emph{puede} ser discontinua sobre un \omterm{def:b2:structures:countable}{conjunto numerable} denso.
\end{example}

\begin{example}[Los saltos no pueden superar el ascenso]\label{ex:b2:realfun:jumpbudget}
Para $f$ creciente sobre $\intcc{a}{b}$, los saltos tienen
presupuesto: si $a < c_1 < \dots < c_m < b$ son discontinuidades
con saltos $s_i = f(c_i^+) - f(c_i^-) > 0$, entonces, eligiendo
puntos intercalados $a < c_1 < t_1 < c_2 < \dots$ y usando la
monotonía en cada trozo,
\[
\sum_{i=1}^{m} s_i \;\leq\; f(b) - f(a) :
\]
el ascenso total acota el salto total. Consecuencia: para cada $k$,
a lo sumo $k\,\bigl(f(b) - f(a)\bigr)$ discontinuidades tienen salto
$\geq \frac1k$, refinamiento cuantitativo del~\cref{thm:b2:realfun:monotone}~(2), ya que el conjunto de
discontinuidades es la unión \omterm{def:b2:structures:countable}{numerable} sobre $k$ de esos conjuntos
finitos. En la función de saltos racionales de más arriba, el
presupuesto se gasta exactamente: los saltos $2^{-n}$ suman $1 =
f(1^+) - f(0^-)$ en el sentido extendido evidente. Las funciones
monótonas pueden saltar densamente, pero solo con una asignación
estricta.
\end{example}

\section{Funciones convexas}

\begin{lemma}[Desigualdad de las pendientes]\label{lem:b2:realfun:slopes}
Sea $f$ convexa sobre $I$ y $x < y < z$ en $I$. Entonces
\[
\frac{f(y) - f(x)}{y - x}
\;\leq\; \frac{f(z) - f(x)}{z - x}
\;\leq\; \frac{f(z) - f(y)}{z - y} :
\]
las pendientes de las cuerdas crecen en ambos extremos.\index{función
convexa!desigualdad de las pendientes}
\end{lemma}

\begin{proof}
Escribamos $y = \frac{z - y}{z - x}\,x + \frac{y - x}{z - x}\,z$:
una combinación convexa, pues los dos coeficientes son positivos y
suman $1$. La convexidad da
\[
f(y) \;\leq\; \frac{z-y}{z-x}\,f(x) + \frac{y-x}{z-x}\,f(z).
\]
Para la desigualdad de la izquierda, réstese $f(x)$ de ambos
miembros, usando $\frac{z-y}{z-x} - 1 = -\frac{y-x}{z-x}$:
\[
f(y) - f(x) \leq \frac{y - x}{z - x}\bigl(f(z) - f(x)\bigr),
\]
y divídase por $y - x > 0$. Para la de la derecha, réstese en cambio
de $f(z)$:
\[
f(z) - f(y) \geq f(z) - \frac{z-y}{z-x}f(x) -
\frac{y-x}{z-x}f(z)
= \frac{z - y}{z - x}\bigl(f(z) - f(x)\bigr),
\]
y divídase por $z - y > 0$. Ambos pasos son la misma identidad
baricéntrica leída contra un extremo distinto.
\end{proof}

\begin{theorem}[Regularidad de las funciones convexas]\label{thm:b2:realfun:convexreg}
Sea $f$ convexa sobre un intervalo $I$.
\begin{enumerate}
  \item En todo punto interior, $f$ tiene derivadas laterales finitas
        $f'_g \leq f'_d$; ambas son funciones crecientes del punto; en
        particular, $f$ es \omterm{def:b2:metric:continuity}{continua} en el interior de $I$ (aunque
        quizá no en los extremos).
  \item $f$ queda por encima de cada \emph{recta de apoyo}: para $a$
        interior y todo $m \in \intcc{f'_g(a)}{f'_d(a)}$,
        \[
        f(x) \geq f(a) + m(x - a) \qquad (x \in I).
        \]
  \item (Jensen, con pesos) Para $x_i \in I$ y pesos $\lambda_i \geq
        0$ con $\sum\lambda_i = 1$:
        \[
        f\Bigl(\sum_i \lambda_i x_i\Bigr) \leq \sum_i \lambda_i
        f(x_i) .\index{desigualdad de Jensen}
        \]
\end{enumerate}
\end{theorem}

\begin{proof}
(1) Fijemos $a$ interior. Por el~\cref{lem:b2:realfun:slopes}, la
pendiente $\tau(h) = \frac{f(a + h) - f(a)}{h}$ es una función
creciente de $h$ (en ambos lados, y $\tau(h_-) \leq \tau(h_+)$ para
$h_- < 0 < h_+$). Por tanto $\tau$ tiene límite finito cuando $h \to
0^-$ (creciente y acotada superiormente por cualquier pendiente por
la derecha) ---ese es $f'_g(a)$--- y cuando $h \to 0^+$ ($f'_d(a)$),
con $f'_g(a) \leq f'_d(a)$. La existencia de derivadas laterales
finitas fuerza la \omterm{def:b2:metric:continuity}{continuidad} en $a$. Monotonía respecto del punto:
para $a < b$ interiores, $f'_d(a) \leq \frac{f(b) - f(a)}{b - a}
\leq f'_g(b)$, de nuevo por la desigualdad de las pendientes.

(2) Para $x > a$: $\frac{f(x) - f(a)}{x - a} \geq f'_d(a) \geq m$;
para $x < a$: $\frac{f(a) - f(x)}{a - x} \leq f'_g(a) \leq m$. Ambas
se reordenan hasta la afirmación.

(3) Inducción sobre el número de puntos, exactamente como en el
volumen del primer año (el caso de dos puntos es la definición); o
de un golpe: aplíquese (2) en $a = \sum\lambda_i x_i$ y promédiense
las desigualdades de la recta de apoyo en los puntos $x_i$ con pesos
$\lambda_i$: $\sum_i \lambda_i f(x_i) \geq f(a) +
m\sum_i\lambda_i(x_i - a) = f(a)$.
\end{proof}

\begin{example}[Discontinuidad en un extremo]
Sobre $\intcc{0}{1}$, la función $f(0) = 1$, $f(x) = 0$ para $x >
0$, es convexa y discontinua en el extremo $0$: el enunciado (1) es
óptimo.
\end{example}

\begin{example}[Esquinas y el haz de rectas de apoyo]\label{ex:b2:realfun:corner}
Para $f(x) = \abs x$ en $a = 0$: las derivadas laterales son
$f'_g(0) = -1$ y $f'_d(0) = +1$, y el~\cref{thm:b2:realfun:convexreg}~(2) reparte una recta de apoyo para
\emph{cada} pendiente $m \in \intcc{-1}{1}$:
\[
\abs x \geq m\,x \qquad (x \in \R,\ -1 \leq m \leq 1),
\]
cada una con igualdad exactamente sobre una semirrecta o en $0$.
Una función convexa es derivable en $a$ precisamente cuando el haz
se reduce a una única recta ($f'_g(a) = f'_d(a)$); las esquinas
llevan todo un intervalo de tangentes. Este haz es el germen en
dimensión finita del \emph{subdiferencial} de la optimización
convexa, y la razón de que las funciones convexas sean tan robustas:
aun donde falla la derivada sobrevive la geometría de apoyo, que es
todo lo que usó la demostración de Jensen.
\end{example}

\begin{example}[Desigualdad de las medias potenciales]\label{ex:b2:realfun:powermeans}
Para $0 < p < q$ y $x_i$ positivos con pesos $\lambda_i$ que suman
$1$, aplicando Jensen a la función convexa $t \mapsto t^{q/p}$ en
los puntos $x_i^p$:
\[
\Bigl(\sum \lambda_i x_i^{p}\Bigr)^{1/p}
\leq \Bigl(\sum \lambda_i x_i^{q}\Bigr)^{1/q} :
\]
las medias potenciales crecen con el exponente, lo que contiene la
desigualdad entre medias aritmética y cuadrática y, en el límite $p
\to 0$ (\cref{exo:b2:realfun:6}), una vez más la desigualdad entre
las medias aritmética y geométrica.
\end{example}

\begin{example}[Entropía máxima]\label{ex:b2:realfun:entropy}
Para un vector de probabilidad $(p_1, \dots, p_n)$ (positivo y de
suma $1$), la entropía $H(p) = -\sum_i p_i\ln p_i$ cumple
\[
H(p) \leq \ln n ,
\qquad\text{con igualdad si y solo si } p_i = \frac1n
\text{ para todo } i .
\]
Demostración por Jensen (\cref{thm:b2:realfun:convexreg}~(3))
aplicado a la función \emph{cóncava} $\ln$ con pesos $p_i$ en los
puntos $\frac{1}{p_i}$:
\[
H(p) = \sum_i p_i\ln\frac{1}{p_i}
\leq \ln\Bigl(\sum_i p_i\,\frac1{p_i}\Bigr) = \ln n ,
\]
donde la igualdad obliga a que todos los puntos $\frac1{p_i}$ sean
iguales (concavidad estricta), es decir, a que $p$ sea uniforme.
Equivalentemente, esto es el~\cref{exo:b2:realfun:7} con $q$
uniforme. La incertidumbre se maximiza con la ignorancia repartida
uniformemente: el principio variacional que hay detrás de la
codificación, la mecánica estadística y las apariciones de la
entropía en el~\cref{ch:b2:randomvar}.
\end{example}

\begin{method}[Cómo hallar la función convexa que hay tras una desigualdad]\label{met:b2:realfun:spotting}
La mayoría de las desigualdades clásicas son Jensen disfrazado; para
desvestir una: (1) normalícese de modo que aparezca una \emph{media
ponderada} (pesos positivos que sumen $1$; divídase por una masa
total si hace falta); (2) obsérvese qué función se aplica dentro y
qué función fuera de la media: la afirmación
``$f(\text{media}) \leq$ media de $f$'' nombra la $f$ convexa; (3)
certifíquese la convexidad con la segunda derivada y trátese la
igualdad mediante la estricticidad; (4) si no se ve ninguna media,
tómense antes logaritmos: los productos y las potencias pasan a ser
medias, y la concavidad de $\ln$ arrastra las desigualdades entre
medias, la de Young y sus parientes (el problema de fin de semana de
este capítulo ejecuta los pasos 1--4 sobre cada una de ellas). Si ni
siquiera los logaritmos revelan una media, pruébese a leer la
desigualdad como monotonía de pendientes
(\cref{lem:b2:realfun:slopes}); allí viven los enunciados de
superaditividad como el del~\cref{exo:b2:realfun:9}.
\end{method}

\begin{remark}[Errores frecuentes]\label{rem:b2:realfun:pitfalls}
(i) La convexidad no se conserva por productos: $x$ y $(x - 1)^2$
son convexas sobre $\intcc{0}{2}$, pero su producto $x(x-1)^2$ tiene
segunda derivada $6x - 4$, negativa sobre $\intco{0}{\frac23}$: no
es convexa; tampoco se conserva por composición sin monotonía
(\cref{exo:b2:realfun:10}). (ii) Jensen se invierte para funciones
cóncavas: la mitad de las desigualdades clásicas son la versión
cóncava con $\ln$; aplicar la forma convexa a $\ln$ es la manera más
rápida de demostrar \emph{al revés} la desigualdad entre las medias
aritmética y geométrica. (iii) La convexidad en el punto medio por
sí sola no implica convexidad: hace falta la \omterm{def:b2:metric:continuity}{continuidad} (o
simplemente la acotación) (\cref{exo:b2:realfun:8}); los
contraejemplos patológicos viven más allá de los axiomas de este
libro. (iv) Una función convexa sobre un intervalo \emph{\omterm{def:b2:metric:topology}{abierto}}
es \omterm{def:b2:metric:continuity}{continua}, incluso localmente \omterm{def:b2:metric:continuity}{lipschitziana}
(\cref{exo:b2:realfun:12}); en los extremos no hay nada gratis. (v)
Las derivadas cumplen Darboux pero no tienen por qué ser \omterm{def:b2:metric:continuity}{continuas}
(\cref{ex:b2:realfun:oscillation}): ``$f'$ no tiene saltos'' nunca
significa ``$f'$ es \omterm{def:b2:metric:continuity}{continua}''.
\end{remark}

\section{La propiedad de Darboux}

\begin{theorem}[Darboux]\label{thm:b2:realfun:darboux}
Sea $f$ derivable sobre un intervalo $I$. Entonces $f'$ toma todos
los valores comprendidos entre dos cualesquiera de sus valores, aun
cuando $f'$ no tiene por qué ser
\omterm{def:b2:metric:continuity}{continua}.\index{teorema de Darboux}
\end{theorem}

\begin{proof}
Sean $a < b$ en $I$ y $v$ estrictamente comprendido entre $f'(a)$ y
$f'(b)$, digamos $f'(a) < v < f'(b)$. La función $g(x) = f(x) - vx$
es derivable con $g'(a) < 0 < g'(b)$: su mínimo sobre $\intcc{a}{b}$
(alcanzado, por \omterm{def:b2:metric:continuity}{continuidad} sobre un \omterm{def:b2:metric:compact}{compacto}) no está en $a$
(justo después de $a$, $g$ decrece por debajo de $g(a)$) ni en $b$
(justo antes de $b$, $g$ está por debajo de $g(b)$): es interior, y
allí $g'(c) = 0$, es decir, $f'(c) = v$. (Esto fue un ejercicio con
estrella del primer año; su lugar en la teoría es este.)
\end{proof}

\begin{example}[¿Qué funciones son derivadas?]\label{ex:b2:realfun:whichderivatives}
El teorema de Darboux es una máquina de no existencia. La función
parte entera $\lfloor x\rfloor$ no es la derivada de ninguna función
sobre $\R$: toma los valores $0$ y $1$ pero se salta $\frac12$ sobre
$\intcc{0}{1}$, cosa que el~\cref{thm:b2:realfun:darboux} prohíbe a
las derivadas. El mismo veredicto alcanza a toda función con un
salto ---el signo, la de Heaviside, todas las funciones escalonadas,
por inocentes que parezcan---; sus ``primitivas'' ($\abs x$ para el
signo, etc.) solo existen lejos del salto y allí se anudan con una
esquina. En cambio, la $f'$ salvajemente discontinua del~\cref{ex:b2:realfun:oscillation} \emph{sí} es una derivada: su
discontinuidad es una oscilación, que Darboux tolera. La frontera
entre ambos comportamientos es exactamente el corolario de ausencia
de saltos de más abajo.
\end{example}

\begin{corollary}\label{cor:b2:realfun:nojumps}
Una derivada no tiene discontinuidades de salto: si existen
$f'(a^-)$ y $f'(a^+)$, ambos valen $f'(a)$. Las discontinuidades de
una derivada son siempre de tipo oscilatorio (la derivada de
$x^2\sin\frac1x$ en $0$, volumen del primer año).
\end{corollary}

\begin{proof}
Si $f'(a^+) = \lim_{x\to a^+} f'(x)$ existe y difiere de $f'(a)$,
los valores estrictamente comprendidos entre ambos quedarían
saltados por $f'$ en un entorno por la derecha, en contra de Darboux
sobre los intervalos $\intcc{a}{a + h}$. (Alternativamente: el
teorema del valor medio fuerza $f'(a) = \lim_{h\to0^+}
\frac{f(a+h)-f(a)}{h} = f'(a^+)$, siendo el cociente incremental un
valor de $f'$ en un punto intermedio.) Lo mismo por la izquierda.
\end{proof}

\begin{example}[La derivada oscilante canónica]\label{ex:b2:realfun:oscillation}
Sea $f(x) = x^2\sin\frac1x$ para $x \neq 0$ y $f(0) = 0$. En $0$:
$\bigl|\frac{f(h) - f(0)}{h}\bigr| = \abs{h\sin\frac1h} \leq \abs h
\to 0$, luego $f'(0) = 0$ existe. Fuera de $0$,
\[
f'(x) = 2x\sin\frac1x - \cos\frac1x ,
\]
cuyo primer término tiende a $0$ mientras que $\cos\frac1x$ oscila
por todo $\intcc{-1}{1}$ en cada intervalo $\intoo{0}{\delta}$: el
límite $f'(0^+)$ no existe. Así pues, $f'$ está definida en todas
partes y es discontinua en $0$ y, exactamente como predice el~\cref{cor:b2:realfun:nojumps}, la discontinuidad es una oscilación y
no un salto: en cada $\intoo{0}{\delta}$, $f'$ sigue barriendo todo
un intervalo alrededor de $0$. Las derivadas pueden ser salvajes,
pero solo de la manera compatible con Darboux.
\end{example}

\begin{remark}[Dónde se usa este capítulo]
La convexidad es el motor de la industria de las desigualdades: el
problema de fin de semana de este capítulo fabrica a partir de ella
las de Young, Hölder y Minkowski y la cadena de medias potenciales,
que consumen la teoría de \omterm{def:b2:nvs:norm}{normas} del~\cref{ch:b2:nvs} y las
estimaciones integrales del~\cref{ch:b2:integration}; Jensen
reaparece en probabilidad como las desigualdades sobre momentos del~\cref{ch:b2:randomvar}. La regularidad de las monótonas vuelve en el~\cref{ch:b2:integration} (las funciones monótonas son integrables)
y, en el volumen del tercer año, como la derivabilidad en casi todo
punto de las funciones monótonas, donde el ``\omterm{def:b2:structures:countable}{numerables} saltos'' pasa
a ser el primer paso de la teoría de Lebesgue.
\end{remark}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:realfun:1}
Determina los conjuntos de discontinuidades y los tamaños de los
saltos de $\lfloor x \rfloor$; $\;x - \lfloor x\rfloor$;
$\;\lfloor x \rfloor + \sqrt{x - \lfloor x\rfloor}$; y de la función
del ejemplo que sigue al~\cref{thm:b2:realfun:monotone},
restringida a los racionales diádicos $r_n$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:realfun:2}
Demuestra que una función creciente $f \colon I \to \R$ con la
propiedad del valor intermedio (la imagen de cualquier subintervalo
es un intervalo) es \omterm{def:b2:metric:continuity}{continua}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:realfun:3}
¿Cuáles de las siguientes son convexas en su dominio? $x \mapsto
x\ln x$ ($x > 0$); $\;x \mapsto \ln(1 + \eu^x)$; $\;x \mapsto
\sqrt{1 + x^2}$; $\;x \mapsto x^3$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:4}
Sea $f$ convexa sobre $\R$ y acotada superiormente. Demuestra que
$f$ es constante. \emph{(Si $f(a) \neq f(b)$, la desigualdad de las
pendientes propaga la pendiente no nula de la cuerda: más allá del
punto con el valor mayor, $f$ crece al menos linealmente, en contra
de la acotación. Trátense ambos signos de la pendiente.)} Deduce que
una función convexa sobre $\R$ con asíntota en ambos extremos es
afín.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:5}
Sea $f$ derivable sobre $I$ con $f'$ monótona. Demuestra que $f'$ es
\omterm{def:b2:metric:continuity}{continua} \emph{(combina el~\cref{thm:b2:realfun:monotone} y el~\cref{cor:b2:realfun:nojumps})}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:6}
(La media geométrica como límite) Para $x_i$ positivos y pesos
$\lambda_i$ que suman $1$, demuestra que
\[
\lim_{p \to 0^+} \Bigl(\sum_i \lambda_i x_i^p\Bigr)^{1/p}
= \prod_i x_i^{\lambda_i} ,
\]
mediante $x_i^p = \eu^{p\ln x_i} = 1 + p\ln x_i + O(p^2)$, y deduce
la desigualdad ponderada entre las medias aritmética y geométrica a
partir del~\cref{ex:b2:realfun:powermeans}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:7}
(Desigualdad de la entropía) Usando la convexidad estricta de $t
\mapsto t\ln t$, demuestra que para $p_i, q_i$ positivos con $\sum
p_i = \sum q_i = 1$:
\[
\sum_i p_i \ln\frac{p_i}{q_i} \geq 0 ,
\]
con igualdad si y solo si $p = q$. \emph{(Escribe el miembro
izquierdo como $\sum q_i\, \varphi\bigl(\frac{p_i}{q_i}\bigr)$ con
$\varphi(t) = t\ln t$ y aplica Jensen con pesos $q_i$.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:realfun:8}
(Convexidad en el punto medio) $f \colon I \to \R$ es
\emph{convexa en el punto medio} cuando
$f\bigl(\frac{x+y}{2}\bigr) \leq \frac{f(x) + f(y)}{2}$ siempre.
Demuestra que una función \emph{\omterm{def:b2:metric:continuity}{continua}} convexa en el punto medio
es convexa. \emph{(Establece la desigualdad de convexidad para pesos
diádicos $\frac{k}{2^m}$ por inducción sobre $m$, y pasa después al
límite usando la densidad y la \omterm{def:b2:metric:continuity}{continuidad}.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:realfun:9}
Sea $f$ convexa sobre $\intco{0}{+\infty}$ con $f(0) \leq 0$.
Demuestra que $x \mapsto \frac{f(x)}{x}$ es creciente sobre
$\intoo{0}{+\infty}$, y deduce que, para $f$ convexa con $f(0) = 0$,
se cumple $f(x + y) \geq f(x) + f(y)$ para $x, y \geq 0$
(superaditividad).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:realfun:10}
Sea $f$ convexa sobre $I$ y $g$ convexa \emph{creciente} sobre un
intervalo que contenga a $f(I)$. Demuestra que $g \circ f$ es
convexa, y muestra con un contraejemplo que no puede prescindirse de
la monotonía de $g$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:realfun:11}
(Hermite--Hadamard) Sea $f$ convexa y \omterm{def:b2:metric:continuity}{continua} sobre
$\intcc{a}{b}$. Demuestra que
\[
f\Bigl(\frac{a+b}{2}\Bigr) \;\leq\; \frac{1}{b -
a}\int_a^b f(t)\,\dd t \;\leq\; \frac{f(a) + f(b)}{2} .
\]
\emph{(Izquierda: intégrese una recta de apoyo en el punto medio.
Derecha: acótese $f$ por la cuerda.)}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:realfun:12}
Demuestra que una función convexa sobre un intervalo
\emph{\omterm{def:b2:metric:topology}{abierto}} $I$ es localmente \omterm{def:b2:metric:continuity}{lipschitziana}: para todo
segmento $\intcc{a}{b} \subseteq I$ y todo margen $\delta > 0$ con
$\intcc{a - \delta}{b + \delta} \subseteq I$, la restricción de $f$
a $\intcc{a}{b}$ es \omterm{def:b2:metric:continuity}{lipschitziana} con constante
$\max\Bigl(\bigl|\frac{f(a) - f(a - \delta)}{\delta}\bigr|,
\bigl|\frac{f(b + \delta) - f(b)}{\delta}\bigr|\Bigr)$
\emph{(atrapa toda pendiente de cuerda entre esas dos mediante la
desigualdad de las pendientes)}.
\end{exercise}

\section{Problema: la caja de herramientas de la convexidad}

Una sola definición ---la cuerda por encima de la gráfica--- genera
toda la caja de herramientas de las desigualdades clásicas. Este
problema de fin de semana la construye en orden lógico: criterios de
convexidad y Jensen estricto, después Young, Hölder y Minkowski (las
partidas de nacimiento de las $p$-normas), la cadena completa de
medias potenciales desde el mínimo hasta el máximo, y dos dividendos
de corona: la desigualdad de Carleman y Hölder leída como dualidad.
Todo se demuestra; nada se importa.

\begin{problem}[{Problema de fin de semana --- Young, Hölder,
Minkowski y la cadena de medias potenciales}]
\label{pb:b2:realfun:1}
En todo el problema, $p, q > 1$ son \emph{exponentes conjugados}:
$\frac1p + \frac1q = 1$; los vectores son $a = (a_1, \dots, a_n) \in
\R^n$; y los pesos $\lambda_i > 0$ cumplen $\sum_i\lambda_i = 1$.

\medskip
\noindent\textbf{Parte I --- Criterios y Jensen estricto.}
\begin{enumerate}
  \item Sea $f$ derivable sobre un intervalo $I$. Demuestra que $f$
        es convexa si y solo si $f'$ es creciente \emph{(una
        implicación pasando al límite en la desigualdad de las
        pendientes, \cref{lem:b2:realfun:slopes}; la otra por el
        teorema del valor medio)}. Deduce el criterio $C^2$: $f''
        \geq 0$.
  \item Supongamos $f'' > 0$ sobre $I$. Demuestra que $f$ es
        \emph{estrictamente} convexa (desigualdad estricta para $x
        \neq y$ y $\lambda \in \intoo01$) y que una función
        estrictamente convexa cumple la desigualdad de Jensen
        (\cref{thm:b2:realfun:convexreg}~(3)) con igualdad
        \emph{únicamente} cuando todos los $x_i$ coinciden.
  \item Certifica las materias primas de la caja: $-\ln$ es
        estrictamente convexa sobre $\intoo{0}{+\infty}$; $t \mapsto
        t^r$ es allí estrictamente convexa para $r > 1$ y
        estrictamente cóncava para $0 < r < 1$; y $\exp$ es
        estrictamente convexa sobre $\R$.
  \item (Desigualdad de Young) Para $a, b \geq 0$, demuestra que
        \[
        ab \;\leq\; \frac{a^p}{p} + \frac{b^q}{q},
        \]
        con igualdad si y solo si $a^p = b^q$ \emph{(aplica la
        concavidad de $\ln$ a los dos puntos $a^p, b^q$ con pesos
        $\frac1p, \frac1q$)}.
  \item Vuelve a deducir en una línea, a partir de la concavidad de
        $\ln$, la desigualdad ponderada entre las medias aritmética
        y geométrica:
        \[
        \prod_i x_i^{\lambda_i} \leq \sum_i\lambda_ix_i \qquad
        (x_i > 0),
        \]
        con su caso de igualdad; compárala con la vía del límite del~\cref{exo:b2:realfun:6}.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Hölder y Minkowski.} Escribamos
$\norm{a}_p = \bigl(\sum_i \abs{a_i}^p\bigr)^{1/p}$ y
$\norm{a}_\infty = \max_i\abs{a_i}$.
\begin{enumerate}[resume]
  \item (Hölder) Demuestra que
        \[
        \sum_{i=1}^{n}\abs{a_ib_i} \;\leq\;
        \norm a_p\,\norm b_q ,
        \]
        con igualdad si y solo si los vectores $(\abs{a_i}^p)$ y
        $(\abs{b_i}^q)$ son proporcionales \emph{(normaliza $\norm
        a_p = \norm b_q = 1$ y aplica Young término a término)}.
  \item Identifica los casos particulares: $p = q = 2$
        (Cauchy--Schwarz) y el par extremo $(p, q) = (1, \infty)$:
        enuncia y demuestra $\sum\abs{a_ib_i} \leq \norm
        a_1\norm b_\infty$.
  \item (Minkowski) Para $p \geq 1$, demuestra que
        \[
        \norm{a + b}_p \leq \norm a_p + \norm b_p
        \]
        \emph{(escribe $\abs{a_i + b_i}^p \leq \abs{a_i +
        b_i}^{p-1}(\abs{a_i} + \abs{b_i})$ y aplica Hölder a cada
        producto)}. Concluye: $\norm\cdot_p$ es una \omterm{def:b2:nvs:norm}{norma} sobre
        $\R^n$ para todo $p \in \intco{1}{+\infty}$, completando el
        panorama del~\cref{ch:b2:nvs}.
  \item Versiones integrales: para $f, g$ \omterm{def:b2:metric:continuity}{continuas} sobre
        $\intcc{a}{b}$, enuncia y demuestra Hölder y Minkowski para
        $\norm f_p = \bigl(\int_a^b\abs f^p\bigr)^{1/p}$ \emph{(las
        mismas demostraciones, con la positividad estricta de la
        integral para la discusión de la igualdad)}.
  \item Demuestra la monotonía $\norm a_q \leq \norm a_p$ para $1
        \leq p \leq q$, el límite $\norm a_p \to \norm a_\infty$
        cuando $p \to \infty$, y la comparación inversa con la
        constante óptima:
        \[
        \norm a_p \leq n^{\frac1p - \frac1q}\,\norm a_q
        \]
        \emph{(Hölder contra el vector constante)}. Identifica los
        vectores que alcanzan cada igualdad.
  \item (Interpolación) Para $1 \leq p < r < q$ y $\theta \in
        \intoo01$ con $\frac1r = \frac\theta p + \frac{1-\theta}q$,
        demuestra que
        \[
        \norm a_r \leq \norm a_p^{\theta}\,
        \norm a_q^{1-\theta}
        \]
        \emph{(aplica Hölder con exponentes $\frac{p}{\theta r}$ y
        $\frac{q}{(1-\theta)r}$ a $\abs{a_i}^{\theta
        r}\abs{a_i}^{(1-\theta)r}$)}.
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- La cadena de medias potenciales, al
\omterm{def:b2:metric:complete}{completo}.} Para $p \neq 0$ pongamos $M_p = \bigl(\sum_i\lambda_i
x_i^p\bigr)^{1/p}$ (con $x_i > 0$), y $M_0 = \prod_i
x_i^{\lambda_i}$.
\begin{enumerate}[resume]
  \item Demuestra que $p \mapsto M_p$ es creciente en todo $\R^*$:
        trata el caso $p < q < 0$ mediante la identidad recíproca
        $M_{-p}(x) = M_p(1/x)^{-1}$, y salva el paso por $0$
        probando que $M_p \leq M_0 \leq M_q$ para $p < 0 < q$
        \emph{(aplica la concavidad de $\ln$ a $x_i^q$ y la
        desigualdad invertida para exponentes negativos)}.
  \item Demuestra los límites $M_p \to \max_i x_i$ cuando $p \to
        +\infty$ y $M_p \to \min_i x_i$ cuando $p \to -\infty$.
  \item Escribe la cadena $\min \leq \mathrm{MH} \leq \mathrm{MG}
        \leq \mathrm{MA} \leq \mathrm{MC} \leq \max$ para pesos
        iguales, y demuestra la consecuencia clásica: para $a_1,
        \dots, a_n$ positivos,
        \[
        \Bigl(\sum_i a_i\Bigr)\Bigl(\sum_i\frac1{a_i}\Bigr)
        \geq n^2 .
        \]
  \item Relaciona medias y \omterm{def:b2:nvs:norm}{normas}: para pesos iguales $\lambda_i =
        \frac1n$ se tiene $M_p(x) = n^{-1/p}\norm x_p$. Reconcilia
        en una frase las dos monotonías ---las medias
        \emph{crecen} con $p$ mientras que las \omterm{def:b2:nvs:norm}{normas}
        \emph{decrecen} (pregunta 10)--- mediante el factor
        $n^{-1/p}$.
  \item Determina los casos de igualdad a lo largo de toda la cadena
        de la pregunta 14 (con pesos positivos): la igualdad en
        cualquier punto obliga a que todos los $x_i$ sean iguales;
        la convexidad estricta rinde sus frutos.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Dividendos.}
\begin{enumerate}[resume]
  \item (Young con mando) Para $a, b \geq 0$ y $\varepsilon > 0$,
        demuestra que
        \[
        ab \leq \varepsilon\,\frac{a^p}{p} +
        \varepsilon^{-q/p}\,\frac{b^q}{q},
        \]
        y el caso caballo de batalla $ab \leq \varepsilon a^2 +
        \frac{b^2}{4\varepsilon}$: el truco de absorción usado por
        todo el análisis.
  \item (Camino a Carleman) Sea $c_k = \frac{(k+1)^k}{k^{k-1}}$.
        Demuestra la identidad telescópica $\prod_{k=1}^{n}c_k =
        (n+1)^n$ y deduce, aplicando la desigualdad entre las medias
        aritmética y geométrica a los números $c_ka_k$,
        \[
        (a_1a_2\cdots a_n)^{1/n} \leq
        \frac{1}{n(n+1)}\sum_{k=1}^{n} c_k a_k
        \qquad (a_k > 0).
        \]
  \item (Desigualdad de Carleman) Suma sobre $n$, intercambia el
        orden de sumación (\omterm{def:b2:series:summable}{familias sumables} positivas,
        \cref{thm:b2:series:fubini}) y usa $\sum_{n \geq
        k}\frac{1}{n(n+1)} = \frac1k$ y $c_k/k = \bigl(1 +
        \frac1k\bigr)^k < \eu$ para concluir: para toda serie
        convergente $\sum a_k$ de términos positivos,
        \[
        \sum_{n=1}^{\infty}(a_1a_2\cdots a_n)^{1/n}
        \;\leq\; \eu\sum_{k=1}^{\infty}a_k .
        \]
  \item Para $f$ \omterm{def:b2:metric:continuity}{continua} y positiva sobre $\intcc{0}{1}$,
        demuestra que
        \[
        \Bigl(\int_0^1 f\Bigr)\Bigl(\int_0^1\frac1f\Bigr)
        \geq 1,
        \]
        con igualdad si y solo si $f$ es constante
        \emph{(Cauchy--Schwarz sobre $\sqrt f\cdot\frac1{\sqrt
        f}$)}.
  \item (Geometría de las bolas) Usando el caso de igualdad de
        Minkowski, prueba que para $1 < p < \infty$ la esfera
        unidad de $\norm\cdot_p$ no contiene ningún segmento (la
        \omterm{def:b2:nvs:norm}{norma} es estrictamente convexa en el sentido del~\cref{pb:b2:nvs:1}), mientras que para $p = 1$ y $p =
        \infty$ sí lo contiene: exhibe las piezas planas.
\end{enumerate}

\medskip
\noindent\textbf{Parte V --- Dualidad y síntesis.}
\begin{enumerate}[resume]
  \item (Hölder como dualidad) Demuestra que para todo $a \in \R^n$
        \[
        \norm a_p = \max_{\norm b_q \leq 1}\ \sum_i a_ib_i ,
        \]
        exhibiendo explícitamente un $b$ maximizante. (La $p$-norma
        es la \omterm{def:b2:linalg:dual}{dual} de la $q$-norma: el germen en dimensión finita
        de la dualidad $L^p$.)
  \item (Momentos) Sea $X$ una variable aleatoria que toma un número
        finito de valores positivos $x_i$ con probabilidades
        $\lambda_i$. Reformula la pregunta 12 así: $r \mapsto
        \E[X^r]^{1/r}$ es creciente; es la desigualdad de momentos
        (de Liapunov), que se reutilizará en el~\cref{ch:b2:randomvar}.
  \item Resuelve con herramientas nombradas, en dos líneas cada uno:
        (i) para $a, b, c$ positivos, $a^3 + b^3 + c^3 \geq
        \frac{(a+b+c)^3}{9}$; (ii) para $x_1, \dots, x_n$ positivos,
        $\bigl(\sum_i\sqrt{x_i}\bigr)^2 \leq n\sum_i x_i$.
  \item (Síntesis) Traza la genealogía en cinco frases: de la
        definición por cuerdas al lema de las pendientes; de las
        pendientes a las rectas de apoyo y a Jensen; de la
        concavidad de $\ln$ a Young, a Hölder, a Minkowski y a las
        $p$-normas; de Jensen a la cadena de medias potenciales y a
        los momentos; de la desigualdad entre medias aritmética y
        geométrica a Carleman. Nombra las cumbres
        (Hölder--Minkowski; Carleman) y di adónde se dirige la caja
        de herramientas: a los espacios $L^p$ del volumen del tercer
        año, cuyos axiomas son exactamente las preguntas 6 y 8.
\end{enumerate}
\end{problem}
