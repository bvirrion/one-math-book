\chapter{الأعداد حتى 1000}\label{ch:g2:numbers}

في السنة الماضية جمعنا الآحاد في عشرات (\cref{ch:g1:tens})؛ والآن
تجتمع عشر عشرات في \emph{مئة}، وتمتد الأعداد حتى
$1000$. ثلاثة أرقام، وثلاث وظائف: مئات، وعشرات، وآحاد.

\section{المئات والعشرات والآحاد}

\begin{definition}[المراتب الثلاث]\label{def:g2:numbers:places}
\emph{المئة} عشر عشرات، أي عشر حزم من عشرة. والعدد
$346$ فيه ثلاثة أجزاء:
\[
346 = 3 \text{ مئات} + 4 \text{ عشرات} + 6 \text{ آحاد}
= 300 + 40 + 6 .
\]
\end{definition}

\begin{example}[الصفر يحفظ المرتبة]\label{ex:g2:numbers:zero}
«ثلاثمئة وخمسة» هو $305$: ثلاث مئات، و\emph{لا} عشرات، وخمسة
آحاد. فالرقم $0$ يُبقي الرقم $5$ في مرتبة الآحاد. ولولاه لكان
$35$ عددًا مختلفًا تمامًا.
\end{example}

\begin{example}[التبديل]\label{ex:g2:numbers:exchange}
عشرة آحاد تصنع عشرة، وعشر عشرات تصنع مئة. إذن $10$ عشرات
$= 100$، و $2$ من المئات $= 20$ عشرة. والعدد $250$ يحتوي
$25$ عشرة، لا الرقم $5$ وحده.
\end{example}

\section{المقارنة والترتيب}

\begin{method}[مقارنة الأعداد ذات الأرقام الثلاثة]\label{met:g2:numbers:compare}
\begin{enumerate}
  \item صاحب الأرقام الأكثر يفوز: $103 > 98$؛
  \item وعند تساوي عدد الأرقام: قارن من اليسار --- المئات أولًا،
        ثم العشرات، ثم الآحاد. وأول اختلاف هو الذي يقرر:
        $472 > 468$ لأن في العشرات $7 > 6$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g2:numbers:order}
رتّب $530$ و $305$ و $503$: المئات أولًا ($5$ و $3$ و $5$)، فيتقدم
$305$؛ وبين $530$ و $503$ تقرر العشرات ($3 > 0$):
\[
305 < 503 < 530 .
\]
\end{example}

\section{القفزات على مستقيم الأعداد}

\begin{method}[إضافة العشرات والمئات]\label{met:g2:numbers:jumps}
على المستقيم:
\begin{enumerate}
  \item $+1$ / $-1$: يتغير رقم الآحاد؛
  \item $+10$ / $-10$: يتغير رقم العشرات ($362 + 10 = 372$)؛
  \item $+100$ / $-100$: يتغير رقم المئات ($362 + 100 = 462$).
\end{enumerate}
ولا يتحرك إلا رقم واحد في كل مرة --- إلا إذا «امتلأت» مرتبة
فحملت إلى ما بعدها، كما في $195 + 10 = 205$.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g2:numbers:1}
كم مئة وكم عشرة وكم واحدًا في: $258$؛ \ $470$؛ \ $600$؛ \
$909$؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:2}
اُكتب العدد: $4$ مئات و $0$ عشرات و $7$ آحاد؛ \ $6$ مئات
و $3$ عشرات؛ \ عشر مئات.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:3}
اُكتب بالأرقام: مئتان وستة عشر؛ \ خمسمئة وأربعة؛ \
سبعمئة وتسعون.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:4}
فكّك كما في \cref{def:g2:numbers:places}: $438$؛ \ $702$؛ \
$560$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:5}
اُنقل وأكمل بالإشارة $<$ أو $>$:
\[
327 \;?\; 372, \qquad
580 \;?\; 508, \qquad
199 \;?\; 200, \qquad
640 \;?\; 604 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:6}
رتّب من الأصغر إلى الأكبر: $430$؛ \ $304$؛ \ $340$؛ \ $403$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:7}
احسب بالقفز على المستقيم: $253 + 10$؛ \quad $253 + 100$؛
\quad $480 - 10$؛ \quad $716 - 100$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:8}
أكمل العدّ: $195, 205, 215, ?, ?$. \quad
$300, 400, 500, ?, ?$. \quad واِنتبه إلى الأولى!
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:9}
في متجر $6$ علب في كل واحدة $100$ قلم، و $5$ أقلام مفردة. فكم
قلمًا في المجموع؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:numbers:10}
مستعملًا الأرقام $2$ و $5$ و $8$ مرة واحدة لكل منها، اُكتب أكبر
عدد من ثلاثة أرقام وأصغره. وكم يبعد أحدهما عن الآخر؟
(ويمكنك أن تصعد بينهما على المستقيم بالقفزات.)
\end{exercise}
