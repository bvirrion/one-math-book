\chapter{النقود والقياسات}\label{ch:g2:measure}

مساطر القياس، وموازين المطبخ، والأباريق، والنقود، والساعات،
والتقاويم: هذا الفصل عن أعداد الحياة اليومية --- كم الطول، وكم
الوزن، وكم يسع الإناء، وكم يكلّف، وكم الساعة
الآن.

\section{الأطوال والكتل والسعات}

\begin{definition}[وحدات الحياة اليومية]\label{def:g2:measure:units}
\begin{itemize}
  \item \emph{الطول}: السنتيمتر (سم) للأشياء الصغيرة، والمتر (م)
        للكبيرة، مع $1$ م $= 100$ سم؛
  \item و\emph{الكتلة}\index{كتلة}: الغرام (غ) للأشياء الخفيفة،
        والكيلوغرام (كغ) للثقيلة، مع $1$ كغ $= 1000$ غ؛
  \item و\emph{السعة}\index{سعة} (ما يسعه الإناء): اللتر (ل).
\end{itemize}
والوحدات الأصغر وقواعدها الدقيقة تأتي في \cref{ch:g3:measure}.
\end{definition}

\begin{example}[أي وحدة؟]\label{ex:g2:measure:choose}
قلم تلوين: نحو $9$ سم. وباب: نحو $2$ م. وتفاحة: نحو $150$
غ. وكيس دقيق كبير: $1$ كغ. وقنينة حليب: $1$ ل. خمِّن الوحدة قبل
القياس --- فذلك يمنع أجوبة سخيفة مثل «الباب $2$
سم».
\end{example}

\section{النقود}

\begin{method}[تكوين مبلغ]\label{met:g2:measure:money}
بقطع وأوراق من فئات $1$ و $2$ و $5$ و $10$ و $20$:
\begin{enumerate}
  \item اِبدأ بالأكبر الذي يدخل في المبلغ، ثم أكمل الباقي؛
  \item لتكوين $27$: ورقة $20$، وقطعة $5$، وقطعة $2$؛
  \item وغالبًا ما تكون هناك طرق عدة --- فالعدد $27$ هو أيضًا
        $10 + 10 + 5 + 2$.
\end{enumerate}
\end{method}

\begin{example}[الباقي]\label{ex:g2:measure:change}
يكلّف كتاب $14$؛ وتدفع بورقة $20$. فالباقي هو $20 - 14 =
6$، ويُوجد بالعدّ إلى الأمام: من $14 \to 20$ مقداره $6$
(\cref{met:g2:subtraction:countup}).
\end{example}

\section{الزمن}

\begin{method}[أنصاف الساعات وأرباعها]\label{met:g2:measure:clock}
العقرب الطويل يدلّ على الدقائق بعد الساعة:
\begin{enumerate}
  \item مستقيمًا إلى الأعلى ($12$): على رأس الساعة --- $3{:}00$؛
  \item ربع دورة نحو اليمين ($3$): والربع --- $3{:}15$؛
  \item مستقيمًا إلى الأسفل ($6$): والنصف --- $3{:}30$؛
  \item ثلاثة أرباع الدورة ($9$): إلا ربعًا من الساعة التالية ---
        $3{:}45$، أي «الرابعة إلا ربعًا».
\end{enumerate}
\end{method}

\begin{example}[التقويم]\label{ex:g2:measure:calendar}
في السنة $12$ شهرًا، بالترتيب: يناير، فبراير، مارس، أبريل،
مايو، يونيو، يوليو، أغسطس، سبتمبر، أكتوبر، نوفمبر، ديسمبر.
وأكثر الشهور فيها $30$ أو $31$ يومًا؛ أما فبراير ففيه $28$ (أو $29$).
والتاريخ يسمّي اليوم والشهر: «$5$ مارس».
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g2:measure:1}
اختر الوحدة (سم، أو م، أو غ، أو كغ، أو ل): قلم؛ وطول
الصف؛ وريشة؛ وحقيبة مدرسية؛ وقنينة ماء.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:2}
حوِّل: $2$ م إلى سم؛ و $300$ سم إلى م؛ و $1$ كغ إلى غ.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:3}
كوِّن هذه المبالغ بقطع وأوراق: $16$؛ \quad $23$؛ \quad
$35$. وأعطِ طريقة واحدة لكل مبلغ.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:4}
احسب الباقي: ادفع $10$ ثمنًا للعبة قيمتها $6$؛ وادفع $20$ ثمنًا
لكتاب قيمته $17$؛ وادفع $50$ ثمنًا للعبة قيمتها $32$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:5}
اقرأ الساعة: العقرب الطويل على $6$ والقصير بين $4$ و $5$؛
والعقرب الطويل على $3$ والقصير بعد $8$ بقليل.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:6}
اُرسم ساعة تشير إلى الثانية والنصف. وإلى أين يشير كل عقرب؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:7}
سمِّ الشهر: بعد مايو؛ وقبل يناير؛ وشهر عيد
ميلادك.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:8}
تشتري لينا شرابًا قيمته $3$ وكعكة قيمتها $4$. وتدفع بورقة $10$.
فكم الباقي؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:9}
أيهما أثقل: كيس سكر كتلته $1$ كغ أم $800$ غ من التفاح؟ وبكم
غرامًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measure:10}
كوِّن $40$ بأربع قطع أو أوراق بالضبط (مختارة من $1$ و $2$ و
$5$ و $10$ و $20$). وجِد طريقتين مختلفتين.
\end{exercise}
