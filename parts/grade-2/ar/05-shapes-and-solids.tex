\chapter{الأشكال والمجسمات}\label{ch:g2:shapes}

الأشكال المستوية تعيش على الورق، والمجسمات تملأ يديك. يصنّف هذا
الفصل الأشكال المستوية بعدّ الأضلاع والرؤوس، ويختبر الزاوية
القائمة بركن ورقة مطويّة، ويلتقي بأولى المجسمات ---
التي أوجهها هي الأشكال المستوية التي نعرفها.

\section{تصنيف الأشكال المستوية}

\begin{definition}[ذوات الأضلاع المستقيمة والمستديرة]\label{def:g2:shapes:sort}
الشكل الذي تتكوّن حدوده من أضلاع مستقيمة فقط هو
\emph{مضلع}\index{مضلع}: مثلث ($3$ أضلاع)، ورباعي ($4$ أضلاع:
فالمربعات والمستطيلات رباعيات)، وخماسي ($5$)،
وسداسي ($6$). أما الشكل ذو الحدّ المنحني، كالدائرة، فليس
\emph{مضلعًا}.
\end{definition}

\begin{method}[اختبار الزاوية القائمة]\label{met:g2:shapes:rightangle}
اِطوِ ورقة مرتين لتصنع ركنًا حادًّا --- وذلك الركن زاوية قائمة.
ضعه في رأس الشكل:
\begin{enumerate}
  \item إن انطبق ضلعا الشكل تمامًا على الركن المطويّ، فالزاوية
        قائمة؛
  \item وإن كان الرأس أوسع أو أضيق، فليست قائمة.
\end{enumerate}
وللمربع وللمستطيل أربع زوايا قائمة.
\end{method}

\section{النقل على الشبكة}

\begin{method}[نقل شكل]\label{met:g2:shapes:copy}
على الورق المربّعات، اِنقل الشكل رأسًا رأسًا:
\begin{enumerate}
  \item اختر رأسًا وعلّمه على الشبكة الفارغة؛
  \item عُدّ المربعات إلى الرأس التالي («$3$ يمينًا و $2$
        فوق») وعلّمه؛
  \item صِل الرؤوس بالترتيب؛ ثم قارن بالنموذج.
\end{enumerate}
\end{method}

\section{المجسمات}

\begin{definition}[أولى المجسمات]\label{def:g2:shapes:solids}
المجسمات تشغل حيزًا. وجوانبها المستوية هي \emph{أوجه}\index{وجه}:
\begin{itemize}
  \item \emph{المكعب}\index{مكعب}: $6$ أوجه مربعة (حجر النرد)؛
  \item و\emph{الصندوق} (متوازي المستطيلات): $6$ أوجه مستطيلة
        (علبة الحبوب)؛
  \item و\emph{الأسطوانة}\index{أسطوانة}: وجهان مستديران وجانب
        ملفوف (علبة معدنية)؛
  \item و\emph{الكرة}: مستديرة تمامًا، بلا وجه مستوٍ.
\end{itemize}
وتعود المجسمات وأوجهها في \cref{ch:g5:solids}.
\end{definition}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g2:shapes:1}
لكل شكل، عُدّ الأضلاع والرؤوس ثم سمِّه: شكل له $5$
أضلاع؛ وشكل له $3$ رؤوس؛ وشكل مستدير.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:2}
صنّف إلى «\omterm{def:g2:shapes:sort}{مضلع}» أو «ليس مضلعًا»: مربع؛ ودائرة؛
ومثلث؛ وسداسي.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:3}
اِصنع مختبرًا للزاوية القائمة بطيّ ورقة. ثم جِد ثلاث زوايا قائمة
في الغرفة، وزاوية واحدة ليست قائمة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:4}
كم زاوية قائمة للمربع؟ وللمستطيل؟ وكم زاوية قائمة يمكن أن تكون
في مثلث على الأكثر؟ (حاول أن ترسم مثلثًا فيه زاويتان قائمتان.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:5}
على الورق المربّعات، اِنقل مستطيلًا عرضه $4$ مربعات وارتفاعه $2$،
ثم مثلثًا قاعدته $3$ مربعات.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:6}
سمِّ المجسم: حجر النرد؛ وعلبة الحساء؛ وكرة القدم؛ وعلبة الحذاء.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:7}
كم وجهًا \omterm{def:g2:shapes:solids}{للمكعب}؟ وما شكل كل \omterm{def:g2:shapes:solids}{وجه}؟ وكم وجهًا
للصندوق؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:8}
أي المجسمات يمكن أن يتدحرج: \omterm{def:g2:shapes:solids}{المكعب}، أم الصندوق، أم الكرة، أم \omterm{def:g2:shapes:solids}{الأسطوانة}؟
ولماذا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:9}
شكل له $6$ أضلاع. فما اسمه؟ وكم رأسًا له؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:shapes:10}
على الورق المربّعات، اُرسم بيتًا: مربعًا للجدار
($4 \times 4$) وفوقه مثلث للسقف. فكم ضلعًا وكم رأسًا لحدّ البيت
كله؟
\end{exercise}
