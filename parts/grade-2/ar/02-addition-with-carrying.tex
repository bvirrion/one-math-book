\chapter{الجمع مع الاحتفاظ}\label{ch:g2:addition}

حين تتجاوز آحاد \omterm{def:g1:addition:def}{المجموع} تسعة، تجتمع عشرة منها في عشرة جديدة ---
وتلك الحزمة الصغيرة، حين تنتقل إلى العمود التالي، هي
\emph{الاحتفاظ} المشهور. يشرح هذا الفصل الاحتفاظ بالأعواد، ثم
يدرّب على الطريقة المكتوبة.

\section{لماذا نحتفظ}

\begin{example}[جمع عشرة آحاد في حزمة]\label{ex:g2:addition:bundle}
$27 + 15$: اِجمع الآحاد، $7 + 5 = 12$ واحدًا. لكن عشرة
آحاد تصنع عشرة! فاِربط عشرة منها: يبقى $2$ من الآحاد المفردة
وعشرة جديدة نحتفظ بها. ثم العشرات: $2 + 1 + 1 = 4$ عشرات. إذن
$27 + 15 = 42$.
\end{example}

\section{الطريقة المكتوبة}

\begin{method}[الجمع بالأعمدة]\label{met:g2:addition:column}
\begin{enumerate}
  \item اُكتب العددين أحدهما تحت الآخر، الآحاد تحت الآحاد،
        والعشرات تحت العشرات؛
  \item اِجمع الآحاد؛ فإن تجاوز الناتج $9$، اُكتب رقم الآحاد
        واحتفظ برقم $1$ صغير فوق العشرات؛
  \item اِجمع العشرات (مع المحتفظ به)، ثم المئات؛
  \item تحقّق بالتقدير: قرِّب كل عدد واجمع تقريبًا.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g2:addition:worked}
احسب $268 + 156$:
\[
\begin{array}{r}
{\scriptstyle 1}\ {\scriptstyle 1}\ \phantom{0} \\[-3pt]
2\,6\,8 \\
+\ 1\,5\,6 \\
\hline
4\,2\,4
\end{array}
\]
الآحاد: $8 + 6 = 14$: اُكتب $4$ واحتفظ برقم $1$. والعشرات:
$6 + 5 + 1 = 12$: اُكتب $2$ واحتفظ برقم $1$. والمئات:
$2 + 1 + 1 = 4$. والتحقّق بالتقدير: نحو $270 + 160 = 430$، وهو قريب
من $424$. \checkmark
\end{example}

\section{الجمع الذهني}

\begin{example}[اِصنع عشرة]\label{ex:g2:addition:mental}
في المجاميع الصغيرة، اقفز إلى عدد مستدير أولًا:
\[
48 + 6 = 48 + 2 + 4 = 50 + 4 = 54 .
\]
واِجمع العشرات والآحاد كلًّا على حدة:
$35 + 23 = (30 + 20) + (5 + 3) = 50 + 8 = 58$. وحيلة القريب من
المئة: $99 + 46 = 100 + 46 - 1 = 145$.
\end{example}

\begin{example}[مسألة صغيرة]\label{ex:g2:addition:problem}
«في المكتبة $185$ كتابًا، ووصل إليها $57$ كتابًا جديدًا. فكم
كتابًا فيها الآن؟» بالجمع: $185 + 57 = 242$ (الآحاد $5 + 7 = 12$،
ثم الاحتفاظ\,\dots). في المكتبة $242$ كتابًا.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:g2:addition:1}
احسب (بلا احتفاظ): $34 + 25$؛ \quad $126 + 43$؛ \quad
$507 + 82$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:2}
احسب مع الاحتفاظ: $47 + 8$؛ \quad $56 + 27$؛ \quad $65 + 19$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:3}
احسب بالأعمدة: $268 + 145$؛ \quad $374 + 158$؛ \quad
$429 + 293$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:4}
احسب ذهنيًّا بصنع عشرة: $38 + 5$؛ \quad $67 + 6$؛
\quad $54 + 8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:5}
احسب بجمع العشرات والآحاد كلًّا على حدة: $42 + 36$؛ \quad
$53 + 24$؛ \quad $61 + 28$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:6}
قدِّر أولًا (بالتقريب إلى أقرب عشرة أو مئة)، ثم احسب
بالضبط: $58 + 34$؛ \quad $296 + 187$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:7}
«عند آية $128$ كرة، وعند عمر $95$. فكم عندهما معًا؟» اُكتب
عملية الجمع وجملة الجواب.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:8}
أكمل فراغ الآحاد: $3\,?\, + 27 = 64$ (جِد رقم الآحاد الناقص
في العدد الأول).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:9}
صفّان يذهبان إلى العرض: $27$ و $28$ طفلًا، ومعهم $6$
كبار. فكم شخصًا في المجموع؟ (اِجمع الصفين أولًا.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:addition:10}
جِد ثلاثة أعداد من القائمة $27$ و $33$ و $40$ و $65$ مجموعها
$100$. (اِبحث أولًا عن عددين آحادهما صديقان في العشرة.)
\end{exercise}
