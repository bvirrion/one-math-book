\chapter{Números até 1000}\label{ch:g2:numbers}

No Ano 1 agrupamos unidades em dezenas (\cref{ch:g1:tens}); agora dez
dezenas se agrupam numa \emph{centena}, e os números se esticam até
$1000$. Três algarismos, três funções: centenas, dezenas, unidades.

\section{Centenas, dezenas, unidades}

\begin{definition}[As três casas]\label{def:g2:numbers:places}
Uma \emph{centena} é dez dezenas, ou seja, dez pacotes de dez. O
número $346$ tem três partes:
\[
346 = 3 \text{ centenas} + 4 \text{ dezenas} + 6 \text{ unidades}
= 300 + 40 + 6 .
\]
\end{definition}

\begin{example}[O zero guarda a casa]\label{ex:g2:numbers:zero}
``Trezentos e cinco'' é $305$: três centenas, \emph{nenhuma} dezena,
cinco unidades. O $0$ segura o $5$ na casa das unidades. Sem ele,
$35$ seria um número completamente diferente.
\end{example}

\begin{example}[Trocar]\label{ex:g2:numbers:exchange}
Dez unidades formam uma dezena, dez dezenas formam uma centena.
Então $10$ dezenas $= 100$, e $2$ centenas $= 20$ dezenas. O número
$250$ contém $25$ dezenas, e não apenas o algarismo $5$.
\end{example}

\section{Comparar e ordenar}

\begin{method}[Comparar números de três algarismos]\label{met:g2:numbers:compare}
\begin{enumerate}
  \item Ganha quem tem mais algarismos: $103 > 98$;
  \item com a mesma quantidade de algarismos: compare da esquerda
        para a direita --- primeiro as centenas, depois as dezenas,
        depois as unidades. A primeira \omterm{ex:g1:subtraction:difference}{diferença} decide:
        $472 > 468$ porque, nas dezenas, $7 > 6$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g2:numbers:order}
Ordene $530$, $305$, $503$: primeiro as centenas ($5$, $3$, $5$),
então $305$ vem na frente; entre $530$ e $503$, decidem as dezenas
($3 > 0$):
\[
305 < 503 < 530 .
\]
\end{example}

\section{Saltos na reta numérica}

\begin{method}[Somar dezenas e centenas]\label{met:g2:numbers:jumps}
Na reta:
\begin{enumerate}
  \item $+1$ / $-1$: mudam as unidades;
  \item $+10$ / $-10$: mudam as dezenas ($362 + 10 = 372$);
  \item $+100$ / $-100$: mudam as centenas ($362 + 100 = 462$).
\end{enumerate}
Só um algarismo se mexe de cada vez --- a não ser que uma casa
``encha'' e passe para a seguinte, como em $195 + 10 = 205$.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g2:numbers:1}
Quantas centenas, dezenas e unidades há em: $258$; \ $470$; \
$600$; \ $909$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:2}
Escreva o número: $4$ centenas, $0$ dezenas, $7$ unidades; \ $6$
centenas e $3$ dezenas; \ dez centenas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:3}
Escreva em algarismos: duzentos e dezesseis; \ quinhentos e quatro;
\ setecentos e noventa.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:4}
Decomponha como na \cref{def:g2:numbers:places}: $438$; \ $702$; \
$560$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:5}
Copie e complete com $<$ ou $>$:
\[
327 \;?\; 372, \qquad
580 \;?\; 508, \qquad
199 \;?\; 200, \qquad
640 \;?\; 604 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:6}
Ordene do menor para o maior: $430$; \ $304$; \ $340$; \ $403$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:7}
Calcule saltando na reta: $253 + 10$; \quad $253 + 100$;
\quad $480 - 10$; \quad $716 - 100$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:8}
Continue a contagem: $195, 205, 215, ?, ?$. \quad
$300, 400, 500, ?, ?$. \quad Cuidado com a primeira!
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:9}
Uma loja tem $6$ caixas de $100$ lápis e $5$ lápis soltos. Quantos
lápis há ao todo?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:numbers:10}
Usando os algarismos $2$, $5$ e $8$ uma vez cada, escreva o maior
número de três algarismos e o menor. Qual é a distância entre eles?
(Você pode fazer a conta na reta, por saltos.)
\end{exercise}
