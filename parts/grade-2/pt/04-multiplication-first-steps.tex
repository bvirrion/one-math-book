\chapter{Multiplicação: primeiros passos}\label{ch:g2:mult}

Quatro caixas de três bolinhos: em vez de $3 + 3 + 3 + 3$,
escrevemos $4 \times 3$. A multiplicação é o atalho para somar
várias vezes o mesmo número --- e o seu melhor desenho é um
quadriculado arrumado de linhas e colunas.

\section{Somar sempre o mesmo número}

\begin{definition}[Multiplicação]\label{def:g2:mult:def}
$4 \times 3$ quer dizer ``quatro vezes três'':
\[
4 \times 3 = 3 + 3 + 3 + 3 = 12 .
\]
O sinal é $\times$, que se lê ``vezes''. O resultado é o
\emph{produto}\index{produto} (a multiplicação cresce no \cref{ch:g3:mult}).
\end{definition}

\begin{omfigure}
\begin{tikzpicture}[scale=0.5]
  \foreach \r in {0,1,2,3}
    \foreach \c in {0,1,2}
      \fill[omDef] (\c,-\r) circle (0.28);
  \node[right, font=\small] at (3.0,-1.5)
    {$4$ linhas de $3$ pontos: $4 \times 3 = 12$};
\end{tikzpicture}

{\small Uma \emph{disposição retangular}: $4$ linhas de $3$. Contar
por linhas dá $4 \times 3$; os mesmos pontos contados por colunas dão
$3 \times 4$ --- por isso a ordem não importa.}
\end{omfigure}

\begin{example}[A ordem não importa]\label{ex:g2:mult:order}
$4 \times 3$ e $3 \times 4$ contam os mesmos $12$ pontos, uma vez
por linhas e outra por colunas. Girar o desenho de lado é a prova.
Assim, aprender um \omterm{def:g2:mult:def}{produto} dá o gêmeo dele de graça.
\end{example}

\section{Contar de tantos em tantos e as tabuadas fáceis}

\begin{method}[Montar uma tabuada contando de tantos em tantos]\label{met:g2:mult:skip}
Para achar $5 \times 4$, conte de quatro em quatro cinco vezes:
$4, 8, 12, 16, 20$. Contar de tantos em tantos na reta numérica
\emph{é} a tabuada.
\begin{itemize}
  \item Tabuada do $2$: $2, 4, 6, 8, 10, \dots$ (os dobros);
  \item tabuada do $5$: $5, 10, 15, 20, 25, \dots$ (termina em $5$ ou
        $0$);
  \item tabuada do $10$: $10, 20, 30, \dots$ (basta acrescentar um
        \omterm{def:g1:counting:zero}{zero}).
\end{itemize}
\end{method}

\begin{omfigure}
\begin{tikzpicture}[scale=0.6]
  \draw[-Stealth] (-0.4,0) -- (11.4,0);
  \foreach \i in {0,...,11} \draw (\i,-0.1) -- (\i,0.1);
  \foreach \i in {0,2,4,6,8,10}
    \node[below, font=\footnotesize] at (\i,-0.12) {\i};
  \foreach \i in {0,1,2,3,4}
    \draw[-Stealth, omDef, thick] (2*\i,0.28) .. controls
      (2*\i+1,0.85) .. (2*\i+2,0.28);
  \node[omDef, font=\small] at (5.5,1.1) {pulos de $2$};
\end{tikzpicture}

{\small A tabuada do $2$ como pulos na reta: $2, 4, 6, 8, 10$. Cinco
pulos de $2$ chegam ao $10$, então $5 \times 2 = 10$.}
\end{omfigure}

\begin{example}[Dobros e metades]\label{ex:g2:mult:double}
Multiplicar por $2$ é dobrar: $2 \times 7 = 14$. O caminho de volta,
tirar a metade, desfaz isso: a metade de $14$ é $7$. Saber os dobros
de cor deixa a tabuada do $2$ instantânea --- e prepara a divisão
(\cref{ch:g3:division}).
\end{example}

\section{Multiplicar em problemas}

\begin{example}\label{ex:g2:mult:problem}
``Uma caixa leva $5$ ovos. Quantos ovos há em $4$ caixas?'' Quatro
grupos de cinco: $4 \times 5 = 20$ (contando de cinco em cinco:
$5, 10, 15, 20$). Há $20$ ovos.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g2:mult:1}
Escreva como multiplicação e depois calcule: $2 + 2 + 2$; \quad
$5 + 5 + 5 + 5$; \quad $10 + 10 + 10$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:2}
Desenhe $3$ linhas de $5$ pontos. Que multiplicação isso mostra? E
girando o desenho de lado?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:3}
Contando de tantos em tantos, complete a tabuada do $2$ até
$2 \times 6$ e a tabuada do $5$ até $5 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:4}
Calcule: $2 \times 7$; \quad $5 \times 3$; \quad $10 \times 4$;
\quad $2 \times 9$; \quad $5 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:5}
Calcule os dobros: o dobro de $6$; o dobro de $8$; o dobro de $10$.
Depois as metades: a metade de $12$; a metade de $20$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:6}
Monte a tabuada do $3$ contando de três em três: $3, 6, 9, ?, ?, ?$.
Quanto é $3 \times 5$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:7}
``Há $4$ mesas com $5$ cadeiras cada uma. Quantas cadeiras?''
Escreva a multiplicação e responda.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:8}
Uma aranha tem $8$ patas. Quantas patas têm $3$ aranhas? (Conte de
oito em oito: $8, 16, \dots$)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:9}
Duas maneiras de arrumar $12$ pontos em linhas iguais: desenhe $2$
linhas de $6$ e $3$ linhas de $4$. Escreva a multiplicação de cada
uma.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:mult:10}
Os bolinhos vêm em caixas de $5$. Mia precisa de $18$ bolinhos para
uma festa. Quantas caixas ela tem de comprar? (Conte de cinco em
cinco até passar de $18$.)
\end{exercise}
