\chapter{Resolução de problemas}\label{ch:g2:problems}

Uma história com números esconde uma pergunta: somar? tirar?
multiplicar? Este capítulo dá um jeito de decidir, usa desenhos de
barras simples para \emph{enxergar} a operação e lê informações em
tabelas e pictogramas.

\section{Escolher a operação}

\begin{method}[Que operação?]\label{met:g2:problems:choose}
\begin{enumerate}
  \item \emph{juntar, ao todo}: some ($+$);
  \item \emph{tirar, quantos restam, quantos a mais}: subtraia
        ($-$);
  \item \emph{tantos grupos iguais}: multiplique ($\times$);
  \item na dúvida, faça um desenho rápido, e responda sempre com uma
        frase.
\end{enumerate}
\end{method}

  \draw[thick, fill=omThm!20] (3,0) rectangle (4.6,0.8);

{\small Desenhos de barras: duas barras emendadas querem dizer uma
adição; duas barras comparadas querem dizer uma subtração.}
\end{omfigure}

\begin{example}[Um problema de duas etapas]\label{ex:g2:problems:twostep}
``Samuel compra $3$ pacotes de $6$ adesivos e depois dá $5$ de
presente. Com quantos ele fica?''
\begin{enumerate}
  \item Grupos: $3 \times 6 = 18$ adesivos;
  \item dar de presente: $18 - 5 = 13$.
\end{enumerate}
Samuel fica com $13$ adesivos. Releia a pergunta: a história acabou
mesmo? Aqui, sim.
\end{example}

\begin{example}[Um número escondido a mais]\label{ex:g2:problems:extra}
``Um livro de $210$ páginas\,\dots'' --- cuidado, nem todo número de
uma história é necessário. ``Bento tem $8$ bolinhas vermelhas e $6$
azuis e tem $7$ anos. Quantas bolinhas ele tem?'' A idade é uma
isca: $8 + 6 = 14$ bolinhas. Separe os números antes de calcular.
\end{example}

\section{Tabelas e pictogramas}

\begin{example}[Uma tabela]\label{ex:g2:problems:table}
Doces vendidos numa barraquinha:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|ccc}
 & sábado & domingo & total \\
\hline
bolinhos & $12$ & $9$ & $21$ \\
tortinhas & $7$ & $10$ & $17$ \\
\end{tabular}
\end{center}
Leitura: tortinhas no domingo, $10$. Cálculo: bolinhos nos dois
dias, $12 + 9 = 21$; no domingo venderam-se $10 - 7 = 3$ tortinhas a
mais que no sábado.
\end{example}

\begin{example}[Um pictograma]\label{ex:g2:problems:pictogram}
Leia a legenda primeiro: aqui, uma bolinha vale $2$ frutas.

\begin{omfigure}
\begin{tikzpicture}[scale=0.62]
  \node[left, font=\small] at (0,2) {vermelhas};
  \foreach \i in {0,1,2,3} \fill[omThm] (0.5+\i*0.75,2) circle (0.24);
  \node[left, font=\small] at (0,1) {verdes};
  \foreach \i in {0,1} \fill[omMeth] (0.5+\i*0.75,1) circle (0.24);
  \node[left, font=\small] at (0,0) {amarelas};
  \foreach \i in {0,1,2} \fill[omProp] (0.5+\i*0.75,0) circle (0.24);
  \node[right, font=\small] at (4.0,1) {legenda: uma bolinha $= 2$ frutas};
\end{tikzpicture}

{\small Um pictograma: as vermelhas têm $4$ bolinhas, então
$4 \times 2 = 8$ frutas vermelhas; as verdes, $2$ bolinhas, $4$
frutas; as amarelas, $3$ bolinhas, $6$ frutas.}
\end{omfigure}
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g2:problems:1}
Diga a operação (não calcule): ``$34$ e $28$ bolinhas juntas''; ``$50$
balas, $12$ comidas''; ``$4$ caixas de $5$ maçãs''; ``Ana tem $30$ e
Léo tem $18$: quantas Ana tem a mais?''.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:2}
Resolva: ``Um estacionamento tem $156$ carros; $48$ vão embora.
Quantos ficam?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:3}
Resolva: ``$5$ crianças colhem $4$ flores cada uma. Quantas flores ao
todo?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:4}
Faça um desenho de barras e depois resolva: ``Tomás tem $45$
figurinhas e Mia tem $17$ a menos. Quantas Mia tem?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:5}
Duas etapas: ``Um engradado leva $6$ garrafas. Uma loja recebe $4$
engradados e vende $9$ garrafas. Quantas garrafas sobram?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:6}
Encontre e risque o número inútil e depois resolva: ``Bento tem $8$
anos e $23$ bolinhas. Ele ganha mais $15$. Com quantas bolinhas
fica?''
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:7}
Usando a tabela do \cref{ex:g2:problems:table}: quantas tortinhas
foram vendidas nos dois dias? E quantos doces de todos os tipos no
sábado?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:8}
Usando o pictograma do \cref{ex:g2:problems:pictogram}: quantas
frutas ao todo? Qual cor tem mais?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:9}
Resolva: ``Uma fita tem $80$ cm de comprimento. Lia corta $35$ cm.
Que comprimento tem o pedaço que sobra?''
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:problems:10}
``Um padeiro faz $5$ bandejas de $4$ bolos de manhã e $3$ bandejas de
$4$ bolos à tarde. Quantos bolos naquele dia?'' (Duas
multiplicações e depois uma \omterm{def:g1:addition:def}{soma} --- ou conte primeiro as
bandejas!)
\end{exercise}
