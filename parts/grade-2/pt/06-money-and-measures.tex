\chapter{Dinheiro e medidas}\label{ch:g2:measure}

Réguas, balanças de cozinha, jarras, moedas, relógios e
calendários: este capítulo é sobre os números do dia a dia --- que
comprimento tem, quanto pesa, quanto cabe, quanto custa e que horas
são.

\section{Comprimentos, massas, capacidades}

\begin{definition}[Unidades do dia a dia]\label{def:g2:measure:units}
\begin{itemize}
  \item \emph{Comprimento}: centímetro (cm) para as coisas pequenas,
        metro (m) para as grandes, com $1$ m $= 100$ cm;
  \item \emph{massa}\index{massa}: grama (g) para as coisas leves, quilograma
        (kg) para as pesadas, com $1$ kg $= 1000$ g;
  \item \emph{capacidade}\index{capacidade} (quanto cabe num recipiente): litro (L).
\end{itemize}
As unidades menores e as suas regras exatas chegam no
\cref{ch:g3:measure}.
\end{definition}

\begin{example}[Que unidade?]\label{ex:g2:measure:choose}
Um giz de cera: cerca de $9$ cm. Uma porta: cerca de $2$ m. Uma
maçã: cerca de $150$ g. Um pacote grande de farinha: $1$ kg. Uma
garrafa de leite: $1$ L. Adivinhe a unidade antes de medir --- isso
evita respostas bobas como ``a porta tem $2$ cm''.
\end{example}

\section{Dinheiro}

\begin{method}[Formar uma quantia]\label{met:g2:measure:money}
Com moedas e notas de $1$, $2$, $5$, $10$, $20$:
\begin{enumerate}
  \item comece pela maior que couber e depois complete o resto;
  \item para formar $27$: uma de $20$, uma de $5$ e uma de $2$;
  \item quase sempre há várias maneiras --- $27$ também é
        $10 + 10 + 5 + 2$.
\end{enumerate}
\end{method}

\begin{example}[Troco]\label{ex:g2:measure:change}
Um livro custa $14$; você paga com $20$. O troco é $20 - 14 = 6$,
encontrado contando para cima: de $14 \to 20$ vão $6$
(\cref{met:g2:subtraction:countup}).
\end{example}

\section{Tempo}

\begin{method}[Meias horas e quartos de hora]\label{met:g2:measure:clock}
O ponteiro grande diz os minutos passados da hora:
\begin{enumerate}
  \item apontando para cima ($12$): hora cheia --- $3{:}00$;
  \item um quarto de volta para a direita ($3$): três e quinze ---
        $3{:}15$;
  \item apontando para baixo ($6$): três e meia --- $3{:}30$;
  \item três quartos de volta ($9$): quinze para a hora seguinte ---
        $3{:}45$, ``quinze para as quatro''.
\end{enumerate}
\end{method}

\begin{example}[O calendário]\label{ex:g2:measure:calendar}
O ano tem $12$ meses, nesta ordem: janeiro, fevereiro, março, abril,
maio, junho, julho, agosto, setembro, outubro, novembro, dezembro.
Quase todos os meses têm $30$ ou $31$ dias; fevereiro tem $28$ (ou
$29$). Uma data diz o dia e o mês: ``$5$ de março''.
\end{example}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:g2:measure:1}
Escolha a unidade (cm, m, g, kg ou L): um lápis; o comprimento da
sala de aula; uma pena; uma mochila; uma garrafa de água.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:2}
Converta: $2$ m em cm; $300$ cm em m; $1$ kg em g.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:3}
Forme estas quantias com moedas e notas: $16$; \quad $23$; \quad
$35$. Dê uma maneira para cada uma.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:4}
Calcule o troco: pague $10$ por um brinquedo de $6$; pague $20$ por
um livro de $17$; pague $50$ por um jogo de $32$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:5}
Leia a hora: ponteiro grande no $6$, ponteiro pequeno entre $4$ e
$5$; ponteiro grande no $3$, ponteiro pequeno logo depois do $8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:6}
Desenhe um relógio marcando duas e meia. Para onde aponta cada
ponteiro?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:7}
Diga o mês: depois de maio; antes de janeiro; o mês do seu
aniversário.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:8}
Lia compra um suco por $3$ e um bolo por $4$. Ela paga com uma nota
de $10$. Quanto recebe de troco?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:9}
O que é mais pesado: um pacote de $1$ kg de açúcar ou $800$ g de
maçãs? Por quantos gramas?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measure:10}
Forme $40$ com exatamente quatro moedas ou notas (escolhidas entre
$1$, $2$, $5$, $10$, $20$). Encontre duas maneiras diferentes.
\end{exercise}
