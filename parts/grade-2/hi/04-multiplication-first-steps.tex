\chapter{गुणा: पहले कदम}\label{ch:g2:mult}

तीन-तीन केक के चार डिब्बे: $3 + 3 + 3 + 3$ लिखने के बजाय हम
$4 \times 3$ लिखते हैं। एक ही संख्या को बार-बार जोड़ने का छोटा
रास्ता ही गुणा है --- और इसका सबसे अच्छा चित्र पंक्तियों और
स्तंभों की सुथरी सरणी है।

\section{एक ही संख्या को जोड़ना}

\begin{definition}[गुणा]\label{def:g2:mult:def}
$4 \times 3$ का मतलब है ``चार गुना तीन'':
\[
4 \times 3 = 3 + 3 + 3 + 3 = 12 .
\]
चिह्न $\times$ है, जिसे ``गुना'' पढ़ते हैं। उत्तर को
\emph{गुणनफल}\index{गुणनफल} कहते हैं (गुणा \cref{ch:g3:mult} में बड़ा होता है)।
\end{definition}

\begin{omfigure}
\begin{tikzpicture}[scale=0.5]
  \foreach \r in {0,1,2,3}
    \foreach \c in {0,1,2}
      \fill[omDef] (\c,-\r) circle (0.28);
  \node[right, font=\small] at (3.0,-1.5)
    {$3$ बिंदुओं की $4$ पंक्तियाँ: $4 \times 3 = 12$};
\end{tikzpicture}

{\small एक \emph{सरणी}: $3$-$3$ की $4$ पंक्तियाँ। पंक्तियों से
गिनने पर $4 \times 3$ मिलता है; वही बिंदु स्तंभों से गिनने पर
$3 \times 4$ --- यानी क्रम से फ़र्क़ नहीं पड़ता।}
\end{omfigure}

\begin{example}[क्रम से फ़र्क़ नहीं पड़ता]\label{ex:g2:mult:order}
$4 \times 3$ और $3 \times 4$ उन्हीं $12$ बिंदुओं को गिनते हैं,
एक बार पंक्तियों से, एक बार स्तंभों से। सरणी को घुमाकर देख लो।
इसलिए एक \omterm{def:g2:mult:def}{गुणनफल} याद करने पर उसका जुड़वाँ मुफ़्त में मिल जाता है।
\end{example}

\section{छलाँग गिनती और आसान पहाड़े}

\begin{method}[छलाँग गिनकर पहाड़ा बनाना]\label{met:g2:mult:skip}
$5 \times 4$ निकालने के लिए चार-चार करके पाँच बार गिनो:
$4, 8, 12, 16, 20$। संख्या रेखा पर छलाँग गिनती \emph{ही}
पहाड़ा है।
\begin{itemize}
  \item $2$ का पहाड़ा: $2, 4, 6, 8, 10, \dots$ (दुगुने);
  \item $5$ का पहाड़ा: $5, 10, 15, 20, 25, \dots$ ($5$ या $0$
        पर ख़त्म);
  \item $10$ का पहाड़ा: $10, 20, 30, \dots$ (बस एक \omterm{def:g1:counting:zero}{शून्य} लगा दो)।
\end{itemize}
\end{method}

\begin{omfigure}
\begin{tikzpicture}[scale=0.6]
  \draw[-Stealth] (-0.4,0) -- (11.4,0);
  \foreach \i in {0,...,11} \draw (\i,-0.1) -- (\i,0.1);
  \foreach \i in {0,2,4,6,8,10}
    \node[below, font=\footnotesize] at (\i,-0.12) {\i};
  \foreach \i in {0,1,2,3,4}
    \draw[-Stealth, omDef, thick] (2*\i,0.28) .. controls
      (2*\i+1,0.85) .. (2*\i+2,0.28);
  \node[omDef, font=\small] at (5.5,1.1) {$2$-$2$ की छलाँगें};
\end{tikzpicture}

{\small रेखा पर छलाँगों के रूप में $2$ का पहाड़ा: $2, 4, 6, 8, 10$।
$2$ की पाँच छलाँगें $10$ तक पहुँचती हैं, इसलिए $5 \times 2 = 10$।}
\end{omfigure}

\begin{example}[दुगुना और आधा]\label{ex:g2:mult:double}
$2$ से गुणा करना यानी दुगुना करना: $2 \times 7 = 14$। इसका उलटा,
आधा करना, इसे मिटा देता है: $14$ का आधा $7$ है। दुगुने याद हों
तो $2$ का पूरा पहाड़ा तुरंत आ जाता है --- और भाग की तैयारी भी
हो जाती है (\cref{ch:g3:division})।
\end{example}

\section{समस्याओं में गुणा}

\begin{example}\label{ex:g2:mult:problem}
``एक डिब्बे में $5$ अंडे आते हैं। $4$ डिब्बों में कितने अंडे?''
पाँच-पाँच के चार समूह: $4 \times 5 = 20$ (छलाँग गिनती:
$5, 10, 15, 20$)। कुल $20$ अंडे हैं।
\end{example}

\section{अभ्यास}

\begin{exercise}[$\star$]\label{exo:g2:mult:1}
गुणा के रूप में लिखो, फिर निकालो: $2 + 2 + 2$; \quad
$5 + 5 + 5 + 5$; \quad $10 + 10 + 10$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:2}
$5$-$5$ बिंदुओं की $3$ पंक्तियों वाली सरणी बनाओ। यह कौन-सा गुणा
दिखाती है? और घुमाने पर?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:3}
छलाँग गिनकर $2$ का पहाड़ा $2 \times 6$ तक भरो; $5$ का पहाड़ा
$5 \times 6$ तक।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:4}
निकालो: $2 \times 7$; \quad $5 \times 3$; \quad $10 \times 4$;
\quad $2 \times 9$; \quad $5 \times 6$।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:5}
दुगुने निकालो: $6$ का दुगुना; $8$ का दुगुना; $10$ का दुगुना।
फिर आधे: $12$ का आधा; $20$ का आधा।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:6}
छलाँग गिनकर $3$ का पहाड़ा निकालो: $3, 6, 9, ?, ?, ?$।
$3 \times 5$ कितना है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:7}
``$5$-$5$ कुर्सियों वाली $4$ मेज़ें हैं। कुल कितनी कुर्सियाँ?''
गुणा और उत्तर लिखो।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:8}
एक मकड़ी के $8$ पैर होते हैं। $3$ मकड़ियों के कितने पैर? ($8$-$8$
करके गिनो: $8, 16, \dots$)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:9}
$12$ बिंदुओं को बराबर पंक्तियों में सजाने के दो तरीक़े: $6$-$6$ की
$2$ पंक्तियाँ बनाओ, और $4$-$4$ की $3$। हर एक के लिए गुणा लिखो।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:mult:10}
केक $5$-$5$ के डिब्बों में आते हैं। मिया को दावत के लिए $18$ केक
चाहिए। उसे कितने डिब्बे लेने होंगे? ($18$ पार होने तक $5$-$5$ गिनो।)
\end{exercise}
