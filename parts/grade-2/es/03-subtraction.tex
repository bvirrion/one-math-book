\chapter{La resta}\label{ch:g2:subtraction}

Cuando arriba hay pocas unidades para poder quitar, abrimos una
decena --- el \emph{préstamo}, el espejo de la llevada. Este capítulo lo
explica con palillos y luego entrena la resta escrita, con
el contar hacia delante como atajo amable.

\section{Por qué se pide prestado}

\begin{example}[Deshacer una decena]\label{ex:g2:subtraction:unbundle}
$42 - 17$: queremos quitar $7$ unidades de $2$ unidades --- ¡son muy pocas!
Así que desatamos una de las $4$ decenas en diez unidades sueltas: arriba quedan
$3$ decenas y $12$ unidades. Quitamos: $12 - 7 = 5$ unidades y
$3 - 1 = 2$ decenas. Por tanto, $42 - 17 = 25$.
\end{example}

\section{El método escrito}

\begin{method}[La resta en columna]\label{met:g2:subtraction:column}
\begin{enumerate}
  \item escribe arriba el número mayor, alineado por la derecha;
  \item resta las unidades; si la cifra de arriba es demasiado pequeña, pide
        prestada una decena (que pasa a ser diez unidades en esa columna);
  \item resta las decenas y después las centenas;
  \item \emph{comprueba}: el resultado más el número que has quitado debe
        devolverte el número de arriba.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g2:subtraction:worked}
Calcula $503 - 168$:
\[
\begin{array}{r}
5\,0\,3 \\
-\ 1\,6\,8 \\
\hline
3\,3\,5
\end{array}
\]
Unidades: $3 - 8$ no se puede; pide prestado a las decenas (no hay ninguna,
así que primero pide a las centenas): arriba quedan $4$ centenas,
$9$ decenas y $13$ unidades. Entonces $13 - 8 = 5$; $9 - 6 = 3$;
$4 - 1 = 3$. Comprobación: $335 + 168 = 503$. \checkmark
\end{example}

\section{Contar hacia delante}

\begin{method}[Restar contando hacia delante]\label{met:g2:subtraction:countup}
Cuando los números están cerca, sube del pequeño al
grande y \omterm{def:g1:addition:def}{suma} los saltos:
\[
83 - 78: \quad 78 \to 80 \text{ son } 2, \ 80 \to 83 \text{ son } 3,
\text{ y por tanto } 83 - 78 = 5 .
\]
Contar hacia delante es más rápido que pedir prestado cuando los dos números están
cerca uno del otro --- y es exactamente como da el cambio un tendero.
\end{method}

\begin{omfigure}
\begin{tikzpicture}[scale=0.85]
  \draw[-Stealth] (76.6,0) -- (84.4,0);
  \foreach \i in {77,...,84} \draw (\i,-0.1) -- (\i,0.1);
  \foreach \i in {78,80,83}
    \node[below, font=\footnotesize] at (\i,-0.12) {\i};
  \draw[-Stealth, omDef, thick] (78,0.3) .. controls (79,0.9) ..
    (80,0.3);
  \node[omDef, font=\small] at (79,1.0) {$+2$};
  \draw[-Stealth, omThm, thick] (80,0.3) .. controls (81.5,1.05) ..
    (83,0.3);
  \node[omThm, font=\small] at (81.5,1.1) {$+3$};
\end{tikzpicture}

{\small $83 - 78$ contando hacia delante: dos saltos, $2 + 3 = 5$.}
\end{omfigure}

\begin{example}[Dar el cambio]\label{ex:g2:subtraction:change}
Un juguete cuesta $34$ y pagas con un $50$. El cambio: cuenta hacia delante,
del $34$ al $40$ hay $6$ y del $40$ al $50$ hay $10$, así que el cambio es $16$.
La resta $50 - 34 = 16$, hecha a la manera del tendero.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:1}
Calcula (sin préstamo): $58 - 23$; \quad $176 - 45$; \quad
$389 - 164$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:2}
Calcula con préstamo: $52 - 17$; \quad $73 - 28$; \quad
$60 - 24$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:3}
Calcula en columna y comprueba cada una: $324 - 167$; \quad
$500 - 236$; \quad $703 - 158$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:4}
Calcula contando hacia delante: $62 - 58$; \quad $91 - 85$; \quad
$100 - 96$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:5}
Calcula el cambio (contando hacia delante): pagas $50$ por un libro de $42$; pagas
$20$ por un juguete de $13$; pagas $100$ por un bolso de $75$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:6}
«Un árbol mide $156$ cm de alto. Este año ha crecido $28$ cm. ¿Cuánto medía
el año pasado?» Escribe la resta y la respuesta.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:7}
Comprueba con la \omterm{def:g1:addition:def}{suma} gemela: $85 - 39 = 46$, ¿está bien?
$214 - 68 = 156$, ¿está bien?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:8}
Para cada historia, elige $+$ o $-$: «$47$ pájaros y $19$ salen volando»;
«$47$ manzanas y se recogen $19$ más»; «Samuel tiene $47$ y Lea tiene $19$:
¿cuántas tiene Samuel de más?».
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:9}
Completa el hueco: $63 - \;?\; = 45$. (Cuenta hacia delante del $45$ al $63$.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:subtraction:10}
Un juego: piensa un número, súmale $30$ y después quítale $18$. El
resultado es $58$. ¿Qué número era? (¡Trabaja hacia atrás!)
\end{exercise}
