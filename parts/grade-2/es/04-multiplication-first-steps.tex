\chapter{La multiplicación: primeros pasos}\label{ch:g2:mult}

Cuatro cajas de tres pasteles: en lugar de $3 + 3 + 3 + 3$, escribimos
$4 \times 3$. La multiplicación es el atajo para sumar muchas veces el mismo
número --- y su mejor dibujo es una disposición ordenada de filas
y columnas.

\section{Sumar el mismo número}

\begin{definition}[La multiplicación]\label{def:g2:mult:def}
$4 \times 3$ significa «cuatro veces tres»:
\[
4 \times 3 = 3 + 3 + 3 + 3 = 12 .
\]
El signo es $\times$ y se lee «por». El resultado es el
\emph{producto}\index{producto} (la multiplicación crece en el \cref{ch:g3:mult}).
\end{definition}

\begin{omfigure}
\begin{tikzpicture}[scale=0.5]
  \foreach \r in {0,1,2,3}
    \foreach \c in {0,1,2}
      \fill[omDef] (\c,-\r) circle (0.28);
  \node[right, font=\small] at (3.0,-1.5)
    {$4$ filas de $3$ puntos: $4 \times 3 = 12$};
\end{tikzpicture}

{\small Una \emph{disposición rectangular}: $4$ filas de $3$. Contar por filas da
$4 \times 3$; los mismos puntos contados por columnas dan $3 \times 4$ --- así que
el orden no importa.}
\end{omfigure}

\begin{example}[El orden no importa]\label{ex:g2:mult:order}
$4 \times 3$ y $3 \times 4$ cuentan los mismos $12$ puntos, unos por
filas y otros por columnas. Girar la disposición de lado lo demuestra. Así,
aprenderse un \omterm{def:g2:mult:def}{producto} regala su gemelo.
\end{example}

\section{Contar a saltos y las tablas fáciles}

\begin{method}[Construir una tabla contando a saltos]\label{met:g2:mult:skip}
Para hallar $5 \times 4$, cuenta de cuatro en cuatro cinco veces:
$4, 8, 12, 16, 20$. Contar a saltos en la recta numérica \emph{es}
la tabla.
\begin{itemize}
  \item tabla del $2$: $2, 4, 6, 8, 10, \dots$ (los dobles);
  \item tabla del $5$: $5, 10, 15, 20, 25, \dots$ (termina en $5$ o en
        $0$);
  \item tabla del $10$: $10, 20, 30, \dots$ (basta con añadir un \omterm{def:g1:counting:zero}{cero}).
\end{itemize}
\end{method}

\begin{tikzpicture}[scale=0.6]
  \draw[-Stealth] (-0.4,0) -- (11.4,0);
  \foreach \i in {0,...,11} \draw (\i,-0.1) -- (\i,0.1);
  \foreach \i in {0,2,4,6,8,10}
    \node[below, font=\footnotesize] at (\i,-0.12) {\i};
  \foreach \i in {0,1,2,3,4}
    \draw[-Stealth, omDef, thick] (2*\i,0.28) .. controls
      (2*\i+1,0.85) .. (2*\i+2,0.28);
  \node[omDef, font=\small] at (5.5,1.1) {saltos de $2$};
\end{tikzpicture}

{\small La tabla del $2$ como saltos en la recta: $2, 4, 6, 8, 10$. Cinco
saltos de $2$ llegan a $10$, así que $5 \times 2 = 10$.}
\end{omfigure}

\begin{example}[Dobles y mitades]\label{ex:g2:mult:double}
Multiplicar por $2$ es doblar: $2 \times 7 = 14$. Lo contrario,
hallar la mitad, lo deshace: la mitad de $14$ es $7$. Saberse los dobles de
memoria hace instantánea toda la tabla del $2$ --- y prepara la división
(\cref{ch:g3:division}).
\end{example}

\section{Multiplicar en los problemas}

\begin{example}\label{ex:g2:mult:problem}
«Una caja tiene $5$ huevos. ¿Cuántos huevos hay en $4$ cajas?» Cuatro grupos de
cinco: $4 \times 5 = 20$ (contando a saltos: $5, 10, 15, 20$). Hay
$20$ huevos.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g2:mult:1}
Escribe como multiplicación y calcula: $2 + 2 + 2$; \quad
$5 + 5 + 5 + 5$; \quad $10 + 10 + 10$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:2}
Dibuja una disposición de $3$ filas de $5$ puntos. ¿Qué multiplicación
representa? ¿Y girada de lado?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:3}
Cuenta a saltos para completar la tabla del $2$ hasta $2 \times 6$; y la tabla
del $5$ hasta $5 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:4}
Calcula: $2 \times 7$; \quad $5 \times 3$; \quad $10 \times 4$;
\quad $2 \times 9$; \quad $5 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:5}
Calcula los dobles: el doble de $6$; el doble de $8$; el doble de $10$.
Después las mitades: la mitad de $12$; la mitad de $20$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:6}
Calcula la tabla del $3$ contando a saltos: $3, 6, 9, ?, ?, ?$. ¿Cuánto
es $3 \times 5$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:7}
«Hay $4$ mesas con $5$ sillas cada una. ¿Cuántas sillas hay?»
Escribe la multiplicación y la respuesta.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:8}
Una araña tiene $8$ patas. ¿Cuántas patas tienen $3$ arañas? (Cuenta a saltos
de $8$ en $8$: $8, 16, \dots$)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:9}
Dos maneras de colocar $12$ puntos en filas iguales: dibuja $2$ filas de $6$
y $3$ filas de $4$. Escribe la multiplicación de cada una.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:mult:10}
Los pasteles vienen en cajas de $5$. Mía necesita $18$ pasteles para una fiesta.
¿Cuántas cajas tiene que comprar? (Cuenta de $5$ en $5$ hasta pasar de $18$.)
\end{exercise}
