\chapter{Resolver problemas}\label{ch:g2:problems}

Una historia con números esconde una pregunta: ¿sumar? ¿quitar? ¿multiplicar?
Este capítulo da una manera de decidirlo, usa esquemas de barras sencillos para
\emph{ver} la operación y lee información en tablas y
pictogramas.

\section{Elegir la operación}

\begin{method}[¿Qué operación?]\label{met:g2:problems:choose}
\begin{enumerate}
  \item \emph{juntar, en total}: sumar ($+$);
  \item \emph{quitar, cuántos quedan, cuántos más}: restar
        ($-$);
  \item \emph{tantos grupos iguales}: multiplicar ($\times$);
  \item haz un dibujo rápido si dudas y responde siempre con una
        frase.
\end{enumerate}
\end{method}

\begin{example}[Un problema de dos pasos]\label{ex:g2:problems:twostep}
«Samuel compra $3$ paquetes de $6$ pegatinas y luego regala $5$. ¿Cuántas
se queda?»
\begin{enumerate}
  \item los grupos: $3 \times 6 = 18$ pegatinas;
  \item regalar: $18 - 5 = 13$.
\end{enumerate}
Samuel se queda con $13$ pegatinas. Vuelve a leer la pregunta: ¿de verdad está
terminada la historia? Aquí sí.
\end{example}

\begin{example}[Un número de más escondido]\label{ex:g2:problems:extra}
«Un libro de $210$ páginas\,\dots» --- cuidado, no todos los números de una
historia hacen falta. «Ben tiene $8$ canicas rojas y $6$ azules y tiene $7$
años. ¿Cuántas canicas tiene?» La edad es un señuelo: $8 + 6 = 14$
canicas. Ordena los números antes de calcular.
\end{example}

\section{Tablas y pictogramas}

\begin{example}[Una tabla]\label{ex:g2:problems:table}
Pasteles vendidos en un puesto:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{l|ccc}
 & sábado & domingo & total \\
\hline
magdalenas & $12$ & $9$ & $21$ \\
tartaletas & $7$ & $10$ & $17$ \\
\end{tabular}
\end{center}
Lectura: tartaletas del domingo, $10$. Cálculo: magdalenas de los dos días,
$12 + 9 = 21$; el domingo se vendieron $10 - 7 = 3$ tartaletas más que el sábado.
\end{example}

\begin{example}[Un pictograma]\label{ex:g2:problems:pictogram}
Lee primero la clave: aquí una manzana representa $2$ piezas de fruta.

\begin{omfigure}
\begin{tikzpicture}[scale=0.62]

  \foreach \i in {0,1,2,3} \fill[omThm] (0.5+\i*0.75,2) circle (0.24);
  \node[left, font=\small] at (0,1) {verdes};
  \foreach \i in {0,1} \fill[omMeth] (0.5+\i*0.75,1) circle (0.24);
  \node[left, font=\small] at (0,0) {amarillas};
  \foreach \i in {0,1,2} \fill[omProp] (0.5+\i*0.75,0) circle (0.24);
  \node[right, font=\small] at (4.0,1) {clave: un punto $= 2$ frutas};
\end{tikzpicture}

{\small Un pictograma: las rojas tienen $4$ puntos, o sea, $4 \times 2 = 8$ frutas
rojas; las verdes $2$ puntos, $4$ frutas; las amarillas $3$ puntos, $6$ frutas.}
\end{omfigure}
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g2:problems:1}
Di la operación (sin calcular): «$34$ y $28$ canicas puestas
juntas»; «$50$ caramelos y se comen $12$»; «$4$ cajas de $5$
manzanas»; «Ana tiene $30$ y Leo tiene $18$: ¿cuántas tiene Ana de más?».
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:2}
Resuelve: «Un aparcamiento tiene $156$ coches y se van $48$. ¿Cuántos quedan?»
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:3}
Resuelve: «$5$ niños cogen $4$ flores cada uno. ¿Cuántas flores hay en
total?»
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:4}
Dibuja un esquema de barras y resuelve: «Tomás tiene $45$ cromos y Mía tiene $17$
menos. ¿Cuántos tiene Mía?»
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:5}
Dos pasos: «Una caja lleva $6$ botellas. Una tienda recibe $4$ cajas y
vende $9$ botellas. ¿Cuántas botellas le quedan?»
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:6}
Busca y tacha el número que sobra y después resuelve: «Ben tiene $8$
años y $23$ canicas. Gana $15$ más. ¿Cuántas canicas tiene
ahora?»
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:7}
Con la tabla del \cref{ex:g2:problems:table}: ¿cuántas tartaletas se vendieron
en los dos días? ¿Cuántos pasteles de todo tipo el sábado?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:8}
Con el pictograma del \cref{ex:g2:problems:pictogram}: ¿cuántas
frutas hay en total? ¿De qué color hay más?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:problems:9}
Resuelve: «Una cinta mide $80$ cm de largo. Lea corta $35$ cm. ¿Cuánto mide
el trozo que queda?»
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:problems:10}
«Un panadero hace $5$ bandejas de $4$ pasteles por la mañana y $3$ bandejas
de $4$ pasteles por la tarde. ¿Cuántos pasteles hace ese día?» (Dos
multiplicaciones y luego una \omterm{def:g1:addition:def}{suma} --- ¡o cuenta primero las bandejas!)
\end{exercise}
