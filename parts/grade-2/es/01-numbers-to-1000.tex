\chapter{Los números hasta 1000}\label{ch:g2:numbers}

El año pasado agrupamos las unidades en decenas (\cref{ch:g1:tens}); ahora diez
decenas se agrupan en una \emph{centena} y los números se estiran hasta
$1000$. Tres cifras, tres tareas: centenas, decenas y unidades.

\section{Centenas, decenas y unidades}

\begin{definition}[Tres lugares]\label{def:g2:numbers:places}
Una \emph{centena} es diez decenas, es decir, diez paquetes de diez. El
número $346$ tiene tres partes:
\[
346 = 3 \text{ centenas} + 4 \text{ decenas} + 6 \text{ unidades}
= 300 + 40 + 6 .
\]
\end{definition}

\begin{example}[El cero guarda el sitio]\label{ex:g2:numbers:zero}
«Trescientos cinco» es $305$: tres centenas, \emph{ninguna}
decena y cinco unidades. El $0$ mantiene al $5$ en el lugar de las unidades. Sin
él, $35$ sería un número completamente distinto.
\end{example}

\begin{example}[Cambiar]\label{ex:g2:numbers:exchange}
Diez unidades hacen una decena y diez decenas hacen una centena. Así, $10$ decenas
$= 100$ y $2$ centenas $= 20$ decenas. El número $250$ contiene
$25$ decenas, no solo la cifra $5$.
\end{example}

\section{Comparar y ordenar}

\begin{method}[Comparar números de tres cifras]\label{met:g2:numbers:compare}
\begin{enumerate}
  \item gana quien tiene más cifras: $103 > 98$;
  \item con el mismo número de cifras: compara desde la izquierda --- primero las
        centenas, luego las decenas y luego las unidades. La primera \omterm{ex:g1:subtraction:difference}{diferencia} decide:
        $472 > 468$ porque, en las decenas, $7 > 6$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g2:numbers:order}
Ordena $530$, $305$ y $503$: primero las centenas ($5$, $3$, $5$), así que
$305$ va delante; entre $530$ y $503$ deciden las decenas ($3 > 0$):
\[
305 < 503 < 530 .
\]
\end{example}

\section{Saltos en la recta numérica}

\begin{method}[Sumar decenas y centenas]\label{met:g2:numbers:jumps}
En la recta:
\begin{enumerate}
  \item $+1$ / $-1$: cambian las unidades;
  \item $+10$ / $-10$: cambian las decenas ($362 + 10 = 372$);
  \item $+100$ / $-100$: cambian las centenas ($362 + 100 = 462$).
\end{enumerate}
Cada vez se mueve una sola cifra --- salvo que un lugar «se llene» y
haya que llevar, como en $195 + 10 = 205$.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g2:numbers:1}
¿Cuántas centenas, decenas y unidades hay en: $258$; \ $470$; \ $600$; \
$909$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:2}
Escribe el número: $4$ centenas, $0$ decenas y $7$ unidades; \ $6$ centenas
y $3$ decenas; \ diez centenas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:3}
Escribe con cifras: doscientos dieciséis; \ quinientos cuatro; \
setecientos noventa.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:4}
Descompón como en la \cref{def:g2:numbers:places}: $438$; \ $702$; \
$560$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:5}
Copia y completa con $<$ o $>$:
\[
327 \;?\; 372, \qquad
580 \;?\; 508, \qquad
199 \;?\; 200, \qquad
640 \;?\; 604 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:6}
Ordena de menor a mayor: $430$; \ $304$; \ $340$; \ $403$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:7}
Calcula dando saltos en la recta: $253 + 10$; \quad $253 + 100$;
\quad $480 - 10$; \quad $716 - 100$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:8}
Sigue contando: $195, 205, 215, ?, ?$. \quad
$300, 400, 500, ?, ?$. \quad ¡Cuidado con la primera!
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:9}
Una tienda tiene $6$ cajas de $100$ lápices y $5$ lápices sueltos. ¿Cuántos
lápices hay en total?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:numbers:10}
Usando las cifras $2$, $5$ y $8$ una vez cada una, escribe el mayor
número de tres cifras y el menor. ¿Qué distancia los separa?
(Puedes calcularla en la recta dando saltos.)
\end{exercise}
