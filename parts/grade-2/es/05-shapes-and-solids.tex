\chapter{Figuras y cuerpos}\label{ch:g2:shapes}

Las figuras planas viven en el papel; los cuerpos llenan tus manos. Este capítulo
clasifica las figuras planas contando lados y esquinas, comprueba el ángulo
recto con una esquina de papel doblado y presenta los primeros cuerpos ---
cuyas \omterm{def:g2:shapes:solids}{caras} son las figuras planas que ya conocemos.

\section{Clasificar figuras planas}

\begin{definition}[De lados rectos y redondas]\label{def:g2:shapes:sort}
Una figura cuyo borde está hecho solo de lados rectos es un
\emph{polígono}\index{polígono}: triángulo ($3$ lados), cuadrilátero ($4$ lados:
los cuadrados y los rectángulos son cuadriláteros), pentágono ($5$) y
hexágono ($6$). Una figura con el borde curvo, como el círculo,
\emph{no} es un polígono.
\end{definition}

\begin{method}[Comprobar un ángulo recto]\label{met:g2:shapes:rightangle}
Dobla dos veces una hoja de papel para formar una esquina afilada --- esa esquina
es un ángulo recto. Colócala en la esquina de una figura:
\begin{enumerate}
  \item si los dos lados de la figura siguen exactamente la esquina
        doblada, el ángulo es recto;
  \item si la esquina es más abierta o más cerrada, no es recto.
\end{enumerate}
Un cuadrado y un rectángulo tienen cuatro ángulos rectos.
\end{method}

\section{Reproducir en la cuadrícula}

\begin{method}[Copiar una figura]\label{met:g2:shapes:copy}
En papel cuadriculado, copia una figura vértice a vértice:
\begin{enumerate}
  \item elige una esquina y márcala en la cuadrícula vacía;
  \item cuenta las casillas hasta la esquina siguiente («$3$ a la derecha, $2$
        arriba») y márcala;
  \item une las esquinas en orden y compara con el modelo.
\end{enumerate}
\end{method}

\section{Los cuerpos}

\begin{definition}[Los primeros cuerpos]\label{def:g2:shapes:solids}
Los cuerpos ocupan espacio. Sus lados planos son las \emph{caras}\index{cara}:
\begin{itemize}
  \item el \emph{cubo}\index{cubo}: $6$ caras cuadradas (un dado);
  \item la \emph{caja} (ortoedro): $6$ caras rectangulares (un
        paquete de cereales);
  \item el \emph{cilindro}\index{cilindro}: dos caras redondas y un costado enrollado
        (una lata);
  \item la \emph{bola}: perfectamente redonda, sin ninguna cara plana.
\end{itemize}
Los cuerpos y sus caras vuelven en el \cref{ch:g5:solids}.
\end{definition}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g2:shapes:1}
Para cada una, cuenta lados y esquinas y di su nombre: una figura con $5$
lados; una figura con $3$ esquinas; una figura redonda.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:2}
Clasifica en «\omterm{def:g2:shapes:sort}{polígono}» o «no es \omterm{def:g2:shapes:sort}{polígono}»: un cuadrado; un círculo; un
triángulo; un hexágono.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:3}
Fabrica un comprobador de ángulos rectos doblando papel. Busca tres ángulos rectos
en la clase y un ángulo que no sea recto.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:4}
¿Cuántos ángulos rectos tiene un cuadrado? ¿Y un rectángulo? ¿Cuántos, como mucho,
puede tener un triángulo? (Intenta dibujar uno con dos.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:5}
En papel cuadriculado, copia un rectángulo de $4$ casillas de ancho y $2$
de alto, y después un triángulo con una base de $3$ casillas.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:6}
Nombra el cuerpo: un dado; una lata de sopa; un balón; una caja de zapatos.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:7}
¿Cuántas \omterm{def:g2:shapes:solids}{caras} tiene un \omterm{def:g2:shapes:solids}{cubo}? ¿Qué forma tiene cada \omterm{def:g2:shapes:solids}{cara}? ¿Cuántas \omterm{def:g2:shapes:solids}{caras}
tiene una caja?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:8}
¿Qué cuerpos pueden rodar: el \omterm{def:g2:shapes:solids}{cubo}, la caja, la bola o el \omterm{def:g2:shapes:solids}{cilindro}?
¿Por qué?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:9}
Una figura tiene $6$ lados. ¿Cómo se llama? ¿Cuántas esquinas tiene?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:shapes:10}
En papel cuadriculado, dibuja una casa: un cuadrado para la pared
($4 \times 4$) con un tejado triangular encima. ¿Cuántos lados y cuántas
esquinas tiene el contorno completo de la casa?
\end{exercise}
