\chapter{La suma con llevadas}\label{ch:g2:addition}

Cuando las unidades de una \omterm{def:g1:addition:def}{suma} pasan de nueve, diez de ellas se agrupan
en una decena nueva --- y ese paquetito, trasladado a la columna siguiente, es
la famosa \emph{llevada}. Este capítulo explica la llevada con palillos
y luego entrena el método escrito.

\section{Por qué se lleva}

\begin{example}[Agrupar diez unidades]\label{ex:g2:addition:bundle}
$27 + 15$: junta las unidades, $7 + 5 = 12$ unidades. ¡Pero diez
unidades hacen una decena! Agrupa diez de ellas: quedan $2$ unidades sueltas
y una decena nueva que se lleva. Ahora las decenas: $2 + 1 + 1 = 4$ decenas. Así,
$27 + 15 = 42$.
\end{example}

\section{El método escrito}

\begin{method}[La suma en columna]\label{met:g2:addition:column}
\begin{enumerate}
  \item escribe los números uno debajo del otro, unidades bajo unidades
        y decenas bajo decenas;
  \item \omterm{def:g1:addition:def}{suma} las unidades; si el resultado pasa de $9$, escribe la cifra de las
        unidades y lleva un $1$ pequeño encima de las decenas;
  \item \omterm{def:g1:addition:def}{suma} las decenas (con la llevada) y después las centenas;
  \item comprueba con una estimación: redondea cada número y \omterm{def:g1:addition:def}{suma} a ojo.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g2:addition:worked}
Calcula $268 + 156$:
\[
\begin{array}{r}
{\scriptstyle 1}\ {\scriptstyle 1}\ \phantom{0} \\[-3pt]
2\,6\,8 \\
+\ 1\,5\,6 \\
\hline
4\,2\,4
\end{array}
\]
Unidades: $8 + 6 = 14$: escribe $4$ y lleva $1$. Decenas:
$6 + 5 + 1 = 12$: escribe $2$ y lleva $1$. Centenas:
$2 + 1 + 1 = 4$. Comprobación estimada: unos $270 + 160 = 430$, cerca de
$424$. \checkmark
\end{example}

\section{Sumar de cabeza}

\begin{example}[Formar una decena]\label{ex:g2:addition:mental}
Para \omterm{def:g1:addition:def}{sumas} pequeñas, salta primero a un número redondo:
\[
48 + 6 = 48 + 2 + 4 = 50 + 4 = 54 .
\]
\omterm{def:g1:addition:def}{Suma} por separado las decenas y las unidades:
$35 + 23 = (30 + 20) + (5 + 3) = 50 + 8 = 58$. Y un truco para los números
cercanos a cien: $99 + 46 = 100 + 46 - 1 = 145$.
\end{example}

\begin{example}[Un problema pequeño]\label{ex:g2:addition:problem}
«La biblioteca tiene $185$ libros y llegan $57$ nuevos. ¿Cuántos libros
hay ahora?» Sumamos: $185 + 57 = 242$ (unidades $5 + 7 = 12$, llevada\,\dots).
La biblioteca tiene $242$ libros.
\end{example}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:g2:addition:1}
Calcula (sin llevadas): $34 + 25$; \quad $126 + 43$; \quad
$507 + 82$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:2}
Calcula con llevada: $47 + 8$; \quad $56 + 27$; \quad $65 + 19$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:3}
Calcula en columna: $268 + 145$; \quad $374 + 158$; \quad
$429 + 293$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:4}
Calcula de cabeza formando una decena: $38 + 5$; \quad $67 + 6$;
\quad $54 + 8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:5}
Calcula sumando por separado decenas y unidades: $42 + 36$; \quad
$53 + 24$; \quad $61 + 28$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:6}
Estima primero (redondeando a la decena o a la centena más cercana) y después
calcula con exactitud: $58 + 34$; \quad $296 + 187$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:7}
«Ana tiene $128$ canicas y Tomás tiene $95$. ¿Cuántas tienen entre los dos?»
Escribe la \omterm{def:g1:addition:def}{suma} y la frase de respuesta.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:8}
Completa el hueco de las unidades: $3\,?\, + 27 = 64$ (halla la cifra de
las unidades que falta en el primer número).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:9}
Dos clases van al espectáculo: $27$ y $28$ niños, más $6$
adultos. ¿Cuántas personas hay en total? (¡\omterm{def:g1:addition:def}{Suma} primero las dos clases!)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:addition:10}
Busca en la lista $27$, $33$, $40$, $65$ tres números que sumen
$100$. (Busca primero dos cuyas unidades sean amigas del diez.)
\end{exercise}
