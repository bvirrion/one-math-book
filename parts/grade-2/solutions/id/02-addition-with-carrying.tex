\section*{Bab \ref{ch:g2:addition} --- Penjumlahan dengan Menyimpan}

\begin{solution}{exo:g2:addition:1}
$34 + 25 = 59$; \quad $126 + 43 = 169$; \quad $507 + 82 = 589$.
\end{solution}

\begin{solution}{exo:g2:addition:2}
$47 + 8 = 55$; \quad $56 + 27 = 83$; \quad $65 + 19 = 84$.
\end{solution}

\begin{solution}{exo:g2:addition:3}
$268 + 145 = 413$; \quad $374 + 158 = 532$; \quad
$429 + 293 = 722$.
\end{solution}

\begin{solution}{exo:g2:addition:4}
$38 + 5 = 38 + 2 + 3 = 43$; \quad $67 + 6 = 67 + 3 + 3 = 73$; \quad
$54 + 8 = 54 + 6 + 2 = 62$.
\end{solution}

\begin{solution}{exo:g2:addition:5}
$42 + 36 = 70 + 8 = 78$; \quad $53 + 24 = 70 + 7 = 77$; \quad
$61 + 28 = 80 + 9 = 89$.
\end{solution}

\begin{solution}{exo:g2:addition:6}
$58 + 34$: taksiran $60 + 30 = 90$; tepatnya $92$.
$296 + 187$: taksiran $300 + 200 = 500$; tepatnya $483$.
\end{solution}

\begin{solution}{exo:g2:addition:7}
$128 + 95 = 223$. Kalau digabung mereka punya $223$ kelereng.
\end{solution}

\begin{solution}{exo:g2:addition:8}
Bilangan pertama ditambah $27$ sama dengan $64$, jadi bilangan itu
$64 - 27 = 37$: angka satuan yang hilang adalah $7$ (bilangannya $37$).
\end{solution}

\begin{solution}{exo:g2:addition:9}
Anak-anak: $27 + 28 = 55$; bersama orang dewasa, $55 + 6 = 61$
orang.
\end{solution}

\begin{solution}{exo:g2:addition:10}
$27$ dan $33$ punya satuan $7$ dan $3$ (sahabat sepuluh):
$27 + 33 = 60$, lalu $60 + 40 = 100$. Ketiga bilangan itu adalah
$27$, $33$ dan $40$.

\end{solution}
