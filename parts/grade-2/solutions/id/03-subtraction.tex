\section*{Bab \ref{ch:g2:subtraction} --- Pengurangan}

\begin{solution}{exo:g2:subtraction:1}
$58 - 23 = 35$; \quad $176 - 45 = 131$; \quad $389 - 164 = 225$.
\end{solution}

\begin{solution}{exo:g2:subtraction:2}
$52 - 17 = 35$; \quad $73 - 28 = 45$; \quad $60 - 24 = 36$.
\end{solution}

\begin{solution}{exo:g2:subtraction:3}
$324 - 167 = 157$ (periksa $157 + 167 = 324$); \quad
$500 - 236 = 264$ (periksa $264 + 236 = 500$); \quad
$703 - 158 = 545$ (periksa $545 + 158 = 703$).
\end{solution}

\begin{solution}{exo:g2:subtraction:4}
$62 - 58 = 4$; \quad $91 - 85 = 6$; \quad $100 - 96 = 4$.
\end{solution}

\begin{solution}{exo:g2:subtraction:5}
$50 - 42 = 8$; \quad $20 - 13 = 7$; \quad $100 - 75 = 25$.
\end{solution}

\begin{solution}{exo:g2:subtraction:6}
Tahun lalu: $156 - 28 = 128$ cm.
\end{solution}

\begin{solution}{exo:g2:subtraction:7}
$85 - 39 = 46$: periksa $46 + 39 = 85$ --- benar.
$214 - 68 = 156$: periksa $156 + 68 = 224 \neq 214$ --- salah
(jawaban yang benar adalah $146$).
\end{solution}

\begin{solution}{exo:g2:subtraction:8}
``terbang pergi'': $47 - 19 = 28$. ``dipetik lagi'': $47 + 19 = 66$.
``berapa lebihnya'': $47 - 19 = 28$.
\end{solution}

\begin{solution}{exo:g2:subtraction:9}
Dari $45$ naik ke $63$ ada $18$, jadi $63 - 18 = 45$: yang kosong adalah $18$.
\end{solution}

\begin{solution}{exo:g2:subtraction:10}
Kerjakan mundur dari $58$: sebelum dikurangi $18$, bilangannya adalah
$58 + 18 = 76$; sebelum ditambah $30$, bilangannya $76 - 30 = 46$.
Bilangan awalnya $46$ (periksa: $46 + 30 - 18 = 58$).

\end{solution}
