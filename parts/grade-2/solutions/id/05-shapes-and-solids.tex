\section*{Bab \ref{ch:g2:shapes} --- Bangun Datar dan Bangun Ruang}

\begin{solution}{exo:g2:shapes:1}
$5$ sisi, $5$ titik sudut: segi lima. $3$ titik sudut (dan $3$ sisi):
segitiga. Bundar: lingkaran.
\end{solution}

\begin{solution}{exo:g2:shapes:2}
\omterm{def:g2:shapes:sort}{Segi banyak}: persegi, segitiga, segi enam (bersisi lurus). Bukan
\omterm{def:g2:shapes:sort}{segi banyak}: lingkaran (tepinya melengkung).
\end{solution}

\begin{solution}{exo:g2:shapes:3}
Sudut siku-siku, misalnya, pada pojok papan tulis, buku, dan jendela;
sudut yang tidak siku-siku pada tutup kotak yang terbuka atau pada
jarum jam pukul $2$.
\end{solution}

\begin{solution}{exo:g2:shapes:4}
Persegi: $4$ sudut siku-siku. Persegi panjang: $4$. Segitiga paling
banyak punya $1$ sudut siku-siku --- dua sudut siku-siku akan membuat
kedua sisi lainnya tidak pernah bertemu.
\end{solution}

\begin{solution}{exo:g2:shapes:5}
Salinan di kertas berpetak dibuat dengan membilang petak di antara
titik sudut; membandingkannya dengan model memeriksa hasilnya.
\end{solution}

\begin{solution}{exo:g2:shapes:6}
Dadu: \omterm{def:g2:shapes:solids}{kubus}. Kaleng sup: \omterm{def:g2:shapes:solids}{tabung}. Bola sepak: bola. Kotak sepatu: balok.
\end{solution}

\begin{solution}{exo:g2:shapes:7}
Sebuah \omterm{def:g2:shapes:solids}{kubus} punya $6$ sisi, masing-masing berbentuk persegi. Balok
juga punya $6$ sisi (persegi panjang).
\end{solution}

\begin{solution}{exo:g2:shapes:8}
Bola menggelinding (bundar di segala arah) dan \omterm{def:g2:shapes:solids}{tabung} menggelinding
(pada sisi lengkungnya). \omterm{def:g2:shapes:solids}{Kubus} dan balok tidak: keduanya hanya punya
sisi datar dan berhenti di atasnya.
\end{solution}

\begin{solution}{exo:g2:shapes:9}
$6$ sisi: segi enam, dengan $6$ titik sudut.
\end{solution}

\begin{solution}{exo:g2:shapes:10}
Tepi rumah itu (dinding persegi dengan atap segitiga, alas atapnya
duduk pada sisi atas persegi) punya $5$ sisi dan $5$ titik sudut ---
bentuknya segi lima (alas, dua dinding, dan dua sisi miring atap;
sisi atas dinding tersembunyi di bawah atap).

\end{solution}
