\section*{अध्याय \ref{ch:g2:measure} --- पैसा और माप}

\begin{solution}{exo:g2:measure:1}
पेंसिल: cm। कक्षा की लंबाई: m। पंख: g। बस्ता: kg। पानी की
बोतल: L।
\end{solution}

\begin{solution}{exo:g2:measure:2}
$2$ m $= 200$ cm; \quad $300$ cm $= 3$ m; \quad $1$ kg $= 1000$ g।
\end{solution}

\begin{solution}{exo:g2:measure:3}
जैसे: $16 = 10 + 5 + 1$; \ $23 = 20 + 2 + 1$; \
$35 = 20 + 10 + 5$। (दूसरे तरीक़े भी हो सकते हैं।)
\end{solution}

\begin{solution}{exo:g2:measure:4}
$10 - 6 = 4$; \quad $20 - 17 = 3$; \quad $50 - 32 = 18$।
\end{solution}

\begin{solution}{exo:g2:measure:5}
लंबी सुई $6$ पर, छोटी $4$ और $5$ के बीच: साढ़े चार
($4{:}30$)। लंबी सुई $3$ पर, छोटी $8$ से थोड़ा आगे: सवा
आठ ($8{:}15$)।
\end{solution}

\begin{solution}{exo:g2:measure:6}
साढ़े दो: लंबी सुई नीचे $6$ की ओर, छोटी सुई $2$ और $3$ के
बीच।
\end{solution}

\begin{solution}{exo:g2:measure:7}
मई के बाद: जून। जनवरी से पहले: दिसंबर। (जन्मदिन का महीना अलग-अलग।)
\end{solution}

\begin{solution}{exo:g2:measure:8}
ख़र्च: $3 + 4 = 7$। बाक़ी: $10 - 7 = 3$।
\end{solution}

\begin{solution}{exo:g2:measure:9}
$1$ kg $= 1000$ g, जो $800$ g से ज़्यादा है: चीनी भारी है,
$1000 - 800 = 200$ g से।
\end{solution}

\begin{solution}{exo:g2:measure:10}
जैसे $20 + 10 + 5 + 5 = 40$ और $10 + 10 + 10 + 10 = 40$
(ठीक चार सिक्कों या नोटों वाले दो तरीक़े)।
\end{solution}
