\section*{अध्याय \ref{ch:g2:mult} --- गुणा: पहले कदम}

\begin{solution}{exo:g2:mult:1}
$2 + 2 + 2 = 3 \times 2 = 6$; \quad
$5 + 5 + 5 + 5 = 4 \times 5 = 20$; \quad
$10 + 10 + 10 = 3 \times 10 = 30$।
\end{solution}

\begin{solution}{exo:g2:mult:2}
$5$-$5$ की $3$ पंक्तियाँ: $3 \times 5 = 15$। घुमाने पर $3$-$3$ की
$5$ पंक्तियाँ: $5 \times 3 = 15$ --- वही $15$।
\end{solution}

\begin{solution}{exo:g2:mult:3}
$2$ का पहाड़ा: $2, 4, 6, 8, 10, 12$। $5$ का पहाड़ा:
$5, 10, 15, 20, 25, 30$।
\end{solution}

\begin{solution}{exo:g2:mult:4}
$2 \times 7 = 14$; \quad $5 \times 3 = 15$; \quad
$10 \times 4 = 40$; \quad $2 \times 9 = 18$; \quad
$5 \times 6 = 30$।
\end{solution}

\begin{solution}{exo:g2:mult:5}
दुगुने: $12$; $16$; $20$। आधे: $12$ का आधा $6$; $20$ का
आधा $10$।
\end{solution}

\begin{solution}{exo:g2:mult:6}
$3, 6, 9, 12, 15, 18$। तो $3 \times 5 = 15$।
\end{solution}

\begin{solution}{exo:g2:mult:7}
$4 \times 5 = 20$ कुर्सियाँ।
\end{solution}

\begin{solution}{exo:g2:mult:8}
$3 \times 8 = 24$ पैर ($8, 16, 24$)।
\end{solution}

\begin{solution}{exo:g2:mult:9}
$6$-$6$ की $2$ पंक्तियाँ: $2 \times 6 = 12$। $4$-$4$ की $3$ पंक्तियाँ:
$3 \times 4 = 12$।
\end{solution}

\begin{solution}{exo:g2:mult:10}
$5$-$5$ गिनो: $5, 10, 15, 20$ --- तीन डिब्बों से केवल $15$
मिलते हैं (कम पड़ेंगे), चार से $20$। मिया को $4$ डिब्बे लेने होंगे।

\end{solution}
