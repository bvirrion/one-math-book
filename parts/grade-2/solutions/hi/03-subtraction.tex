\section*{अध्याय \ref{ch:g2:subtraction} --- घटाव}

\begin{solution}{exo:g2:subtraction:1}
$58 - 23 = 35$; \quad $176 - 45 = 131$; \quad $389 - 164 = 225$।
\end{solution}

\begin{solution}{exo:g2:subtraction:2}
$52 - 17 = 35$; \quad $73 - 28 = 45$; \quad $60 - 24 = 36$।
\end{solution}

\begin{solution}{exo:g2:subtraction:3}
$324 - 167 = 157$ (जाँच $157 + 167 = 324$); \quad
$500 - 236 = 264$ (जाँच $264 + 236 = 500$); \quad
$703 - 158 = 545$ (जाँच $545 + 158 = 703$)।
\end{solution}

\begin{solution}{exo:g2:subtraction:4}

\end{solution}

\begin{solution}{exo:g2:subtraction:5}

\begin{solution}{exo:g2:subtraction:6}

\end{solution}

\begin{solution}{exo:g2:subtraction:7}
$85 - 39 = 46$: जाँच $46 + 39 = 85$ --- सही।
$214 - 68 = 156$: जाँच $156 + 68 = 224 \neq 214$ --- ग़लत (सही
उत्तर $146$ है)।
\end{solution}

\begin{solution}{exo:g2:subtraction:8}
``उड़ जाती हैं'': $47 - 19 = 28$। ``और तोड़े गए'': $47 + 19 = 66$।
``कितने ज़्यादा'': $47 - 19 = 28$।
\end{solution}

\begin{solution}{exo:g2:subtraction:9}
$45$ से $63$ तक $18$ है, इसलिए $63 - 18 = 45$: ख़ाली जगह में $18$।
\end{solution}

\begin{solution}{exo:g2:subtraction:10}
$58$ से उलटा चलो: $18$ घटाने से पहले संख्या
$58 + 18 = 76$ थी; $30$ जोड़ने से पहले वह $76 - 30 = 46$ थी।
शुरू की संख्या $46$ थी (जाँच: $46 + 30 - 18 = 58$)।
\end{solution}
