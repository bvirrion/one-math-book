\section*{अध्याय \ref{ch:g2:problems} --- समस्याएँ हल करना}

\begin{solution}{exo:g2:problems:1}
मिलाना: जोड़। $50$ मिठाइयाँ, $12$ खाई गईं: घटाव। $5$-$5$ के $4$
डिब्बे: गुणा। कितने ज़्यादा: घटाव।
\end{solution}

\begin{solution}{exo:g2:problems:2}
$156 - 48 = 108$। $108$ कारें रह जाती हैं।
\end{solution}

\begin{solution}{exo:g2:problems:3}
$5 \times 4 = 20$ फूल।
\end{solution}

\begin{solution}{exo:g2:problems:4}
तुलना वाली पट्टी: टॉम $45$, मिया की पट्टी $17$ छोटी:
$45 - 17 = 28$ कार्ड।
\end{solution}

\begin{solution}{exo:g2:problems:5}
मिली बोतलें: $4 \times 6 = 24$; बचीं:
$24 - 9 = 15$ बोतलें।
\end{solution}

\begin{solution}{exo:g2:problems:6}
बेकार संख्या उम्र ($8$) है। कंचे:
$23 + 15 = 38$।
\end{solution}

\begin{solution}{exo:g2:problems:7}
टार्ट: $7 + 10 = 17$। शनिवार को सब केक:
$12 + 7 = 19$।
\end{solution}

\begin{solution}{exo:g2:problems:8}
लाल $8$, हरे $4$, पीले $6$: कुल $8 + 4 + 6 = 18$ फल।
लाल सबसे ज़्यादा हैं।
\end{solution}

\begin{solution}{exo:g2:problems:9}
$80 - 35 = 45$ cm बचते हैं।
\end{solution}

\begin{solution}{exo:g2:problems:10}
सुबह: $5 \times 4 = 20$; दोपहर: $3 \times 4 = 12$; कुल
$20 + 12 = 32$ केक। (या: $5 + 3 = 8$ ट्रे, $4$-$4$ की, और
$8 \times 4 = 32$।)

\end{solution}
