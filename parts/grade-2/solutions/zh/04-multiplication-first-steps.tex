\section*{第 \ref{ch:g2:mult} 章——乘法：初步认识}

\begin{solution}{exo:g2:mult:1}
$2 + 2 + 2 = 3 \times 2 = 6$; \quad
$5 + 5 + 5 + 5 = 4 \times 5 = 20$; \quad
$10 + 10 + 10 = 3 \times 10 = 30$.
\end{solution}

\begin{solution}{exo:g2:mult:2}
$3$ 行，每行 $5$ 个：$3 \times 5 = 15$。转过来看，$5$ 行，
每行 $3$ 个：$5 \times 3 = 15$——同样是 $15$。
\end{solution}

\begin{solution}{exo:g2:mult:3}
$2$ 的乘法表：$2, 4, 6, 8, 10, 12$。$5$ 的乘法表：
$5, 10, 15, 20, 25, 30$。
\end{solution}

\begin{solution}{exo:g2:mult:4}
$2 \times 7 = 14$; \quad $5 \times 3 = 15$; \quad
$10 \times 4 = 40$; \quad $2 \times 9 = 18$; \quad
$5 \times 6 = 30$.
\end{solution}

\begin{solution}{exo:g2:mult:5}
两倍：$12$；$16$；$20$。一半：$12$ 的一半是 $6$；$20$ 的
一半是 $10$。
\end{solution}

\begin{solution}{exo:g2:mult:6}
$3, 6, 9, 12, 15, 18$。所以 $3 \times 5 = 15$。
\end{solution}

\begin{solution}{exo:g2:mult:7}
$4 \times 5 = 20$ 把椅子。
\end{solution}

\begin{solution}{exo:g2:mult:8}
$3 \times 8 = 24$ 条腿（$8, 16, 24$）。
\end{solution}

\begin{solution}{exo:g2:mult:9}
$2$ 行，每行 $6$ 个：$2 \times 6 = 12$。$3$ 行，每行 $4$ 个：
$3 \times 4 = 12$。
\end{solution}

\begin{solution}{exo:g2:mult:10}
按 $5$ 数数：$5, 10, 15, 20$——三盒只能得 $15$
（不够），四盒得 $20$。米娅必须买 $4$ 盒。
\end{solution}
