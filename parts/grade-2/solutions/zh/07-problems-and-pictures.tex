\section*{第 \ref{ch:g2:problems} 章——解决问题}

\begin{solution}{exo:g2:problems:1}
合在一起：加法。$50$ 颗糖果，吃掉 $12$ 颗：减法。
$4$ 盒，每盒 $5$ 个：乘法。多出多少：减法。
\end{solution}

\begin{solution}{exo:g2:problems:2}
$156 - 48 = 108$。剩下 $108$ 辆车。
\end{solution}

\begin{solution}{exo:g2:problems:3}
$5 \times 4 = 20$ 朵花。
\end{solution}

\begin{solution}{exo:g2:problems:4}
线段图：汤姆有 $45$ 张卡片，米娅比他少 $17$ 张：
$45 - 17 = 28$ 张。
\end{solution}

\begin{solution}{exo:g2:problems:5}
收到的瓶子：$4 \times 6 = 24$ 个；
剩下：$24 - 9 = 15$ 个瓶子。
\end{solution}

\begin{solution}{exo:g2:problems:6}
年龄（$8$）是无用的数字。弹珠：
$23 + 15 = 38$ 颗。
\end{solution}

\begin{solution}{exo:g2:problems:7}
蛋挞：$7 + 10 = 17$ 个。周六，蛋糕一共：
$12 + 7 = 19$ 个。
\end{solution}

\begin{solution}{exo:g2:problems:8}
红色 $8$ 个，绿色 $4$ 个，黄色 $6$ 个：共 $8 + 4 + 6 = 18$ 个水果。
红色最多。
\end{solution}

\begin{solution}{exo:g2:problems:9}
剩下 $80 - 35 = 45$ 厘米。
\end{solution}

\begin{solution}{exo:g2:problems:10}
上午：$5 \times 4 = 20$；下午：$3 \times 4 = 12$；合计
$20 + 12 = 32$ 个蛋糕。（或：$5 + 3 = 8$ 盘，每盘 $4$ 个，
则 $8 \times 4 = 32$ 个。）
\end{solution}
