\section*{Capítulo \ref{ch:g2:problems} --- Resolver problemas}

\begin{solution}{exo:g2:problems:1}
Juntar: \omterm{def:g1:addition:def}{suma}. $50$ caramelos y se comen $12$: resta. $4$
cajas de $5$: multiplicación. Cuántas de más: resta.
\end{solution}

\begin{solution}{exo:g2:problems:2}
$156 - 48 = 108$. Quedan $108$ coches.
\end{solution}

\begin{solution}{exo:g2:problems:3}
$5 \times 4 = 20$ flores.
\end{solution}

\begin{solution}{exo:g2:problems:4}
Esquema de comparación: Tomás $45$, Mía $17$ menos:
$45 - 17 = 28$ cromos.
\end{solution}

\begin{solution}{exo:g2:problems:5}
Botellas recibidas: $4 \times 6 = 24$; quedan:
$24 - 9 = 15$ botellas.
\end{solution}

\begin{solution}{exo:g2:problems:6}
El número que sobra es la edad ($8$). Canicas:
$23 + 15 = 38$.
\end{solution}

\begin{solution}{exo:g2:problems:7}
Tartaletas: $7 + 10 = 17$. El sábado, todos los pasteles:
$12 + 7 = 19$.
\end{solution}

\begin{solution}{exo:g2:problems:8}
Rojas $8$, verdes $4$, amarillas $6$: en total $8 + 4 + 6 = 18$ frutas.
De las que más hay es de rojas.
\end{solution}

\begin{solution}{exo:g2:problems:9}
Quedan $80 - 35 = 45$ cm.
\end{solution}

\begin{solution}{exo:g2:problems:10}
Mañana: $5 \times 4 = 20$; tarde: $3 \times 4 = 12$; total
$20 + 12 = 32$ pasteles. (O bien: $5 + 3 = 8$ bandejas de $4$, y
$8 \times 4 = 32$.)
\end{solution}
