\section*{Capítulo \ref{ch:g2:shapes} --- Figuras y cuerpos}

\begin{solution}{exo:g2:shapes:1}
$5$ lados y $5$ esquinas: un pentágono. $3$ esquinas (y $3$ lados): un
triángulo. Redonda: un círculo.
\end{solution}

\begin{solution}{exo:g2:shapes:2}
\omterm{def:g2:shapes:sort}{Polígonos}: cuadrado, triángulo y hexágono (lados rectos). No es
\omterm{def:g2:shapes:sort}{polígono}: el círculo (borde curvo).
\end{solution}

\begin{solution}{exo:g2:shapes:3}
Ángulos rectos, por ejemplo, en las esquinas de la pizarra, de un libro o de una
ventana; un ángulo que no es recto, en la tapa abierta de una caja o en las agujas
de un reloj a las $2$ en punto.
\end{solution}

\begin{solution}{exo:g2:shapes:4}
Cuadrado: $4$ ángulos rectos. Rectángulo: $4$. Un triángulo puede tener
como mucho $1$ ángulo recto --- con dos, los otros dos lados no se
cortarían nunca.
\end{solution}

\begin{solution}{exo:g2:shapes:5}
Copias en la cuadrícula hechas contando las casillas entre esquinas; comparar
con el modelo permite comprobar el trabajo.
\end{solution}

\begin{solution}{exo:g2:shapes:6}
Dado: \omterm{def:g2:shapes:solids}{cubo}. Lata de sopa: \omterm{def:g2:shapes:solids}{cilindro}. Balón: bola. Caja de zapatos: caja.
\end{solution}

\begin{solution}{exo:g2:shapes:7}
Un \omterm{def:g2:shapes:solids}{cubo} tiene $6$ \omterm{def:g2:shapes:solids}{caras}, cada una un cuadrado. Una caja también tiene $6$ \omterm{def:g2:shapes:solids}{caras}
(rectángulos).
\end{solution}

\begin{solution}{exo:g2:shapes:8}
Ruedan la bola (redonda por todas partes) y el \omterm{def:g2:shapes:solids}{cilindro} (sobre su
costado curvo). El \omterm{def:g2:shapes:solids}{cubo} y la caja no: solo tienen \omterm{def:g2:shapes:solids}{caras}
planas y se paran sobre ellas.
\end{solution}

\begin{solution}{exo:g2:shapes:9}
$6$ lados: un hexágono, con $6$ esquinas.
\end{solution}

\begin{solution}{exo:g2:shapes:10}
El contorno de la casa (pared cuadrada con tejado triangular, con la base del
tejado apoyada en el lado superior del cuadrado) tiene $5$ lados y $5$ esquinas ---
es un pentágono (el suelo, dos paredes y dos vertientes del tejado; la parte
alta de la pared queda escondida bajo el tejado).
\end{solution}
