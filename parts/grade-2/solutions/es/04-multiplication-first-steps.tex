\section*{Capítulo \ref{ch:g2:mult} --- La multiplicación: primeros pasos}

\begin{solution}{exo:g2:mult:1}
$2 + 2 + 2 = 3 \times 2 = 6$; \quad
$5 + 5 + 5 + 5 = 4 \times 5 = 20$; \quad
$10 + 10 + 10 = 3 \times 10 = 30$.
\end{solution}

\begin{solution}{exo:g2:mult:2}
$3$ filas de $5$: $3 \times 5 = 15$. Girada de lado, $5$ filas de
$3$: $5 \times 3 = 15$ --- los mismos $15$.
\end{solution}

\begin{solution}{exo:g2:mult:3}
Tabla del $2$: $2, 4, 6, 8, 10, 12$. Tabla del $5$:
$5, 10, 15, 20, 25, 30$.
\end{solution}

\begin{solution}{exo:g2:mult:4}
$2 \times 7 = 14$; \quad $5 \times 3 = 15$; \quad
$10 \times 4 = 40$; \quad $2 \times 9 = 18$; \quad
$5 \times 6 = 30$.
\end{solution}

\begin{solution}{exo:g2:mult:5}
Dobles: $12$; $16$; $20$. Mitades: la mitad de $12$ es $6$; la mitad de
$20$ es $10$.
\end{solution}

\begin{solution}{exo:g2:mult:6}
$3, 6, 9, 12, 15, 18$. Así que $3 \times 5 = 15$.
\end{solution}

\begin{solution}{exo:g2:mult:7}
$4 \times 5 = 20$ sillas.
\end{solution}

\begin{solution}{exo:g2:mult:8}
$3 \times 8 = 24$ patas ($8, 16, 24$).
\end{solution}

\begin{solution}{exo:g2:mult:9}
$2$ filas de $6$: $2 \times 6 = 12$. $3$ filas de $4$:
$3 \times 4 = 12$.
\end{solution}

\begin{solution}{exo:g2:mult:10}
Contando de $5$ en $5$: $5, 10, 15, 20$ --- con tres cajas solo hay $15$
(muy pocos) y con cuatro cajas hay $20$. Mía tiene que comprar $4$ cajas.
\end{solution}
