\section*{Capítulo \ref{ch:g2:problems} --- Resolução de problemas}

\begin{solution}{exo:g2:problems:1}
Juntar: adição. $50$ balas, $12$ comidas: subtração. $4$ caixas de
$5$: multiplicação. Quantas a mais: subtração.
\end{solution}

\begin{solution}{exo:g2:problems:2}
$156 - 48 = 108$. Ficam $108$ carros.
\end{solution}

\begin{solution}{exo:g2:problems:3}
$5 \times 4 = 20$ flores.
\end{solution}

\begin{solution}{exo:g2:problems:4}
Barra de comparação: Tomás $45$, a de Mia é $17$ mais curta:
$45 - 17 = 28$ figurinhas.
\end{solution}

\begin{solution}{exo:g2:problems:5}
Garrafas recebidas: $4 \times 6 = 24$; sobram: $24 - 9 = 15$
garrafas.
\end{solution}

\begin{solution}{exo:g2:problems:6}
O número inútil é a idade ($8$). Bolinhas: $23 + 15 = 38$.
\end{solution}

\begin{solution}{exo:g2:problems:7}
Tortinhas: $7 + 10 = 17$. No sábado, todos os doces:
$12 + 7 = 19$.
\end{solution}

\begin{solution}{exo:g2:problems:8}
Vermelhas $8$, verdes $4$, amarelas $6$: ao todo $8 + 4 + 6 = 18$
frutas. As vermelhas são as mais numerosas.
\end{solution}

\begin{solution}{exo:g2:problems:9}
$80 - 35 = 45$ cm sobram.
\end{solution}

\begin{solution}{exo:g2:problems:10}
De manhã: $5 \times 4 = 20$; à tarde: $3 \times 4 = 12$; total
$20 + 12 = 32$ bolos. (Ou: $5 + 3 = 8$ bandejas de $4$, e
$8 \times 4 = 32$.)
\end{solution}
