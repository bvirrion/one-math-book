\section*{الفصل \ref{ch:g2:subtraction} --- الطرح}

\begin{solution}{exo:g2:subtraction:1}
$58 - 23 = 35$؛ \quad $176 - 45 = 131$؛ \quad $389 - 164 = 225$.
\end{solution}

\begin{solution}{exo:g2:subtraction:2}
$52 - 17 = 35$؛ \quad $73 - 28 = 45$؛ \quad $60 - 24 = 36$.
\end{solution}

\begin{solution}{exo:g2:subtraction:3}
$324 - 167 = 157$ (التحقّق $157 + 167 = 324$)؛ \quad
$500 - 236 = 264$ (التحقّق $264 + 236 = 500$)؛ \quad
$703 - 158 = 545$ (التحقّق $545 + 158 = 703$).
\end{solution}

\begin{solution}{exo:g2:subtraction:4}

\end{solution}

\begin{solution}{exo:g2:subtraction:5}

\end{solution}

\begin{solution}{exo:g2:subtraction:6}
السنة الماضية: $156 - 28 = 128$ سم.
\end{solution}

\begin{solution}{exo:g2:subtraction:7}
$85 - 39 = 46$: التحقّق $46 + 39 = 85$ --- صحيح.
$214 - 68 = 156$: التحقّق $156 + 68 = 224 \neq 214$ --- خطأ (والجواب
الصحيح هو $146$).
\end{solution}

\begin{solution}{exo:g2:subtraction:8}
«طار منها»: $47 - 19 = 28$. و«قُطف غيرها»: $47 + 19 = 66$.
و«بكم يزيد»: $47 - 19 = 28$.
\end{solution}

\begin{solution}{exo:g2:subtraction:9}
من $45$ صعودًا إلى $63$ مقداره $18$، إذن $63 - 18 = 45$: فالفراغ هو $18$.
\end{solution}

\begin{solution}{exo:g2:subtraction:10}
اِعمل بالعكس من $58$: قبل طرح $18$ كان العدد
$58 + 18 = 76$؛ وقبل إضافة $30$ كان $76 - 30 = 46$. فالعدد
الأول هو $46$ (والتحقّق: $46 + 30 - 18 = 58$).
\end{solution}
