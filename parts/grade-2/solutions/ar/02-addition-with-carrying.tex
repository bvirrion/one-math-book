\section*{الفصل \ref{ch:g2:addition} --- الجمع مع الاحتفاظ}

\begin{solution}{exo:g2:addition:1}
$34 + 25 = 59$؛ \quad $126 + 43 = 169$؛ \quad $507 + 82 = 589$.
\end{solution}

\begin{solution}{exo:g2:addition:2}
$47 + 8 = 55$؛ \quad $56 + 27 = 83$؛ \quad $65 + 19 = 84$.
\end{solution}

\begin{solution}{exo:g2:addition:3}
$268 + 145 = 413$؛ \quad $374 + 158 = 532$؛ \quad
$429 + 293 = 722$.
\end{solution}

\begin{solution}{exo:g2:addition:4}
$38 + 5 = 38 + 2 + 3 = 43$؛ \quad $67 + 6 = 67 + 3 + 3 = 73$؛ \quad
$54 + 8 = 54 + 6 + 2 = 62$.
\end{solution}

\begin{solution}{exo:g2:addition:5}
$42 + 36 = 70 + 8 = 78$؛ \quad $53 + 24 = 70 + 7 = 77$؛ \quad
$61 + 28 = 80 + 9 = 89$.
\end{solution}

\begin{solution}{exo:g2:addition:6}
$58 + 34$: التقدير $60 + 30 = 90$؛ وبالضبط $92$.
$296 + 187$: التقدير $300 + 200 = 500$؛ وبالضبط $483$.
\end{solution}

\begin{solution}{exo:g2:addition:7}
$128 + 95 = 223$. عندهما معًا $223$ كرة.
\end{solution}

\begin{solution}{exo:g2:addition:8}
العدد الأول زائد $27$ يساوي $64$، إذن هو $64 - 27 = 37$: فرقم
الآحاد الناقص هو $7$ (والعدد هو $37$).
\end{solution}

\begin{solution}{exo:g2:addition:9}
الأطفال: $27 + 28 = 55$؛ ومع الكبار، $55 + 6 = 61$
شخصًا.
\end{solution}

\begin{solution}{exo:g2:addition:10}
آحاد $27$ و $33$ هي $7$ و $3$ (صديقان في العشرة):
$27 + 33 = 60$، ثم $60 + 40 = 100$. والأعداد الثلاثة هي
$27$ و $33$ و $40$.
\end{solution}
