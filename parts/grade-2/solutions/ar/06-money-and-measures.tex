\section*{الفصل \ref{ch:g2:measure} --- النقود والقياسات}

\begin{solution}{exo:g2:measure:1}
القلم: سم. وطول الصف: م. والريشة: غ. والحقيبة المدرسية: كغ.
وقنينة الماء: ل.
\end{solution}

\begin{solution}{exo:g2:measure:2}
$2$ م $= 200$ سم؛ \quad $300$ سم $= 3$ م؛ \quad $1$ كغ $= 1000$ غ.
\end{solution}

\begin{solution}{exo:g2:measure:3}
مثلًا: $16 = 10 + 5 + 1$؛ \ $23 = 20 + 2 + 1$؛ \
$35 = 20 + 10 + 5$. (وهناك طرق أخرى ممكنة.)
\end{solution}

\begin{solution}{exo:g2:measure:4}
$10 - 6 = 4$؛ \quad $20 - 17 = 3$؛ \quad $50 - 32 = 18$.
\end{solution}

\begin{solution}{exo:g2:measure:5}
العقرب الطويل على $6$ والقصير بين $4$ و $5$: الرابعة والنصف
($4{:}30$). والعقرب الطويل على $3$ والقصير بعد $8$ بقليل: الثامنة
والربع ($8{:}15$).
\end{solution}

\begin{solution}{exo:g2:measure:6}
الثانية والنصف: العقرب الطويل يشير إلى الأسفل نحو $6$، والعقرب
القصير يقع بين $2$ و $3$.
\end{solution}

\begin{solution}{exo:g2:measure:7}
بعد مايو: يونيو. وقبل يناير: ديسمبر. (وشهر عيد الميلاد يختلف.)
\end{solution}

\begin{solution}{exo:g2:measure:8}
المصروف: $3 + 4 = 7$. والباقي: $10 - 7 = 3$.
\end{solution}

\begin{solution}{exo:g2:measure:9}
$1$ كغ $= 1000$ غ، وهي أكثر من $800$ غ: فالسكر أثقل،
بمقدار $1000 - 800 = 200$ غ.
\end{solution}

\begin{solution}{exo:g2:measure:10}
مثلًا $20 + 10 + 5 + 5 = 40$ و $10 + 10 + 10 + 10 = 40$
(طريقتان بأربع قطع أو أوراق بالضبط).
\end{solution}
