\section*{الفصل \ref{ch:g2:problems} --- حلّ المسائل}

\begin{solution}{exo:g2:problems:1}
الضمّ معًا: جمع. و«$50$ حلوى، أُكل منها $12$»: طرح. و«$4$
علب من $5$»: ضرب. و«بكم تزيد»: طرح.
\end{solution}

\begin{solution}{exo:g2:problems:2}
$156 - 48 = 108$. تبقى $108$ سيارة.
\end{solution}

\begin{solution}{exo:g2:problems:3}
$5 \times 4 = 20$ زهرة.
\end{solution}

\begin{solution}{exo:g2:problems:4}
شريط مقارنة: عمر $45$، ومريم أقلّ منه بمقدار $17$:
$45 - 17 = 28$ بطاقة.
\end{solution}

\begin{solution}{exo:g2:problems:5}
القنينات المستلمة: $4 \times 6 = 24$؛ والباقية:
$24 - 9 = 15$ قنينة.
\end{solution}

\begin{solution}{exo:g2:problems:6}
العدد غير المفيد هو العمر ($8$). والكرات:
$23 + 15 = 38$.
\end{solution}

\begin{solution}{exo:g2:problems:7}
الفطائر: $7 + 10 = 17$. ويوم السبت، كل الأنواع:
$12 + 7 = 19$.
\end{solution}

\begin{solution}{exo:g2:problems:8}
الأحمر $8$، والأخضر $4$، والأصفر $6$: في المجموع $8 + 4 + 6 = 18$ فاكهة.
والأحمر هو الأكثر.
\end{solution}

\begin{solution}{exo:g2:problems:9}
$80 - 35 = 45$ سم تبقى.
\end{solution}

\begin{solution}{exo:g2:problems:10}
صباحًا: $5 \times 4 = 20$؛ وبعد الظهر: $3 \times 4 = 12$؛ \omterm{def:g1:addition:def}{والمجموع}
$20 + 12 = 32$ كعكة. (أو: $5 + 3 = 8$ صوانٍ من $4$، و
$8 \times 4 = 32$.)
\end{solution}
