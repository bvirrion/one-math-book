\chapter{Vermenigvuldigen: eerste stappen}\label{ch:g2:mult}

Vier bakjes met drie koekjes: in plaats van $3 + 3 + 3 + 3$ schrijven
we $4 \times 3$. Vermenigvuldigen is de korte weg voor hetzelfde getal
vaak optellen --- en het mooiste plaatje ervan is een net rooster van
rijen en kolommen.

\section{Hetzelfde getal optellen}

\begin{definition}[Vermenigvuldigen]\label{def:g2:mult:def}
$4 \times 3$ betekent ``vier keer drie'':
\[
4 \times 3 = 3 + 3 + 3 + 3 = 12 .
\]
Het teken is $\times$, gelezen als ``keer''. De uitkomst is het
\emph{product}\index{product} (vermenigvuldigen wordt groot in
\cref{ch:g3:mult}).
\end{definition}

\begin{omfigure}
\begin{tikzpicture}[scale=0.5]
  \foreach \r in {0,1,2,3}
    \foreach \c in {0,1,2}
      \fill[omDef] (\c,-\r) circle (0.28);
  \node[right, font=\small] at (3.0,-1.5)
    {$4$ rijen van $3$ stippen: $4 \times 3 = 12$};
\end{tikzpicture}

{\small Een \emph{rooster}: $4$ rijen van $3$. Per rij tellen geeft
$4 \times 3$; dezelfde stippen per kolom geven $3 \times 4$ --- de
volgorde maakt dus niets uit.}
\end{omfigure}

\begin{example}[De volgorde maakt niets uit]\label{ex:g2:mult:order}
$4 \times 3$ en $3 \times 4$ tellen dezelfde $12$ stippen, de ene keer
per rij, de andere keer per kolom. Draai het rooster een kwartslag en je
ziet het. Wie één \omterm{def:g2:mult:def}{product} kent, krijgt zijn tweeling er dus gratis bij.
\end{example}

\section{Sprongen tellen en de makkelijke tafels}

\begin{method}[Een tafel opbouwen met sprongen]\label{met:g2:mult:skip}
Om $5 \times 4$ te vinden, tel je vijf keer met vier verder:
$4, 8, 12, 16, 20$. Met sprongen tellen op de getallenlijn \emph{is} de
tafel.
\begin{itemize}
  \item Tafel van $2$: $2, 4, 6, 8, 10, \dots$ (de dubbels);
  \item tafel van $5$: $5, 10, 15, 20, 25, \dots$ (eindigt op $5$ of
        $0$);
  \item tafel van $10$: $10, 20, 30, \dots$ (zet er gewoon een \omterm{def:g1:counting:zero}{nul}
        achter).
\end{itemize}
\end{method}

  \node[omDef, font=\small] at (5.5,1.1) {sprongen van $2$};
\end{tikzpicture}

{\small De tafel van $2$ als sprongen op de lijn: $2, 4, 6, 8, 10$.
Vijf sprongen van $2$ komen op $10$ uit, dus $5 \times 2 = 10$.}
\end{omfigure}

\begin{example}[Dubbels en helften]\label{ex:g2:mult:double}
Met $2$ vermenigvuldigen is verdubbelen: $2 \times 7 = 14$. Halveren
draait het terug: de helft van $14$ is $7$. Wie de dubbels uit het hoofd
kent, kent de hele tafel van $2$ in één klap --- en is voorbereid op het
delen (\cref{ch:g3:division}).
\end{example}

\section{Vermenigvuldigen in verhaaltjes}

\begin{example}\label{ex:g2:mult:problem}
``In een doosje zitten $5$ eieren. Hoeveel eieren zitten er in $4$
doosjes?'' Vier groepjes van vijf: $4 \times 5 = 20$ (met sprongen
tellen: $5, 10, 15, 20$). Er zijn $20$ eieren.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g2:mult:1}
Schrijf als vermenigvuldiging en reken uit: $2 + 2 + 2$; \quad
$5 + 5 + 5 + 5$; \quad $10 + 10 + 10$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:2}
Teken een rooster van $3$ rijen van $5$ stippen. Welke
vermenigvuldiging laat het zien? En een kwartslag gedraaid?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:3}
Vul met sprongen tellen de tafel van $2$ tot $2 \times 6$ in; en de
tafel van $5$ tot $5 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:4}
Reken uit: $2 \times 7$; \quad $5 \times 3$; \quad $10 \times 4$;
\quad $2 \times 9$; \quad $5 \times 6$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:5}
Reken de dubbels uit: het dubbele van $6$; van $8$; van $10$. En dan de
helften: de helft van $12$; de helft van $20$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:6}
Reken de tafel van $3$ uit met sprongen: $3, 6, 9, ?, ?, ?$. Hoeveel is
$3 \times 5$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:7}
``Er staan $4$ tafels met elk $5$ stoelen. Hoeveel stoelen zijn dat?''
Schrijf de vermenigvuldiging en het antwoord.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:8}
Een spin heeft $8$ poten. Hoeveel poten hebben $3$ spinnen? (Tel met
sprongen van $8$: $8, 16, \dots$)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:mult:9}
Twee manieren om $12$ stippen in gelijke rijen te leggen: teken $2$
rijen van $6$, en $3$ rijen van $4$. Schrijf bij elk de
vermenigvuldiging.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:mult:10}
Koekjes zitten in pakjes van $5$. Mia heeft $18$ koekjes nodig voor een
feest. Hoeveel pakjes moet ze kopen? (Tel met sprongen van $5$ tot je
voorbij $18$ bent.)
\end{exercise}
