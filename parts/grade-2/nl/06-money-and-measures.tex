\chapter{Geld en maten}\label{ch:g2:measure}

Meetlatten, keukenweegschalen, maatbekers, muntjes, klokken en
kalenders: dit hoofdstuk gaat over de getallen van het dagelijks leven
--- hoe lang, hoe zwaar, hoeveel er in gaat, hoeveel het kost, en hoe
laat het is.

\section{Lengtes, massa's, inhouden}

\begin{definition}[Alledaagse eenheden]\label{def:g2:measure:units}
\begin{itemize}
  \item \emph{Lengte}: centimeter (cm) voor kleine dingen, meter (m)
        voor grote, met $1$ m $= 100$ cm;
  \item \emph{massa}\index{massa}: gram (g) voor lichte dingen,
        kilogram (kg) voor zware, met $1$ kg $= 1000$ g;
  \item \emph{inhoud}\index{inhoud} (hoeveel er in een bak gaat): liter (L).
\end{itemize}
Kleinere eenheden en de precieze regels volgen in
\cref{ch:g3:measure}.
\end{definition}

\begin{example}[Welke eenheid?]\label{ex:g2:measure:choose}
Een kleurkrijtje: ongeveer $9$ cm. Een deur: ongeveer $2$ m. Een appel:
ongeveer $150$ g. Een grote zak bloem: $1$ kg. Een fles melk: $1$ L.
Raad de eenheid voordat je meet --- dat voorkomt onzinnige antwoorden
als ``de deur is $2$ cm''.
\end{example}

\section{Geld}

\begin{method}[Een bedrag maken]\label{met:g2:measure:money}
Met muntjes en biljetten van $1$, $2$, $5$, $10$, $20$:
\begin{enumerate}
  \item begin met het grootste dat past en vul de rest aan;
  \item om $27$ te maken: een $20$, een $5$ en een $2$;
  \item vaak kan het op meer manieren --- $27$ is ook
        $10 + 10 + 5 + 2$.
\end{enumerate}
\end{method}

\begin{example}[Wisselgeld]\label{ex:g2:measure:change}
Een boek kost $14$; je betaalt met $20$. Het wisselgeld is
$20 - 14 = 6$, gevonden door door te tellen: $14 \to 20$ is $6$
(\cref{met:g2:subtraction:countup}).
\end{example}

\section{Tijd}

\begin{method}[Halve uren en kwartieren]\label{met:g2:measure:clock}
De lange wijzer geeft de minuten na het hele uur:
\begin{enumerate}
  \item recht naar boven ($12$): op het hele uur --- $3{:}00$, ``drie
        uur'';
  \item een kwartslag naar rechts ($3$): ``kwart over drie'' ---
        $3{:}15$;
  \item recht naar beneden ($6$): $3{:}30$ --- in het Nederlands
        ``half vier'', want we noemen het \emph{volgende} uur;
  \item driekwart slag ($9$): ``kwart voor vier'' --- $3{:}45$.
\end{enumerate}
\end{method}

\begin{example}[De kalender]\label{ex:g2:measure:calendar}
Een jaar heeft $12$ maanden, op volgorde: januari, februari, maart,
april, mei, juni, juli, augustus, september, oktober, november,
december. De meeste maanden hebben $30$ of $31$ dagen; februari heeft
$28$ (of $29$). Een datum noemt de dag en de maand: ``$5$ maart''.
\end{example}

\section{Oefeningen}

\begin{exercise}[$\star$]\label{exo:g2:measure:1}
Kies de eenheid (cm, m, g, kg of L): een potlood; de lengte van het
klaslokaal; een veertje; een schooltas; een fles water.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:2}
Zet om: $2$ m in cm; $300$ cm in m; $1$ kg in g.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:3}
Maak deze bedragen met muntjes en biljetten: $16$; \quad $23$; \quad
$35$. Geef bij elk één manier.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:4}
Reken het wisselgeld uit: je betaalt $10$ voor speelgoed van $6$; $20$
voor een boek van $17$; $50$ voor een spel van $32$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:5}
Lees de tijd: lange wijzer op $6$, korte wijzer tussen $4$ en $5$;
lange wijzer op $3$, korte wijzer net na $8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:6}
Teken een klok die half drie aanwijst. Waar wijst elke wijzer naartoe?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:7}
Noem de maand: na mei; voor januari; de maand van je verjaardag.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:8}
Lea koopt een drankje van $3$ en een gebakje van $4$. Ze betaalt met
een biljet van $10$. Hoeveel wisselgeld krijgt ze?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:measure:9}
Wat is zwaarder: een zak suiker van $1$ kg of $800$ g appels? Hoeveel
gram scheelt het?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:measure:10}
Maak $40$ met precies vier muntjes of biljetten (gekozen uit $1$, $2$,
$5$, $10$, $20$). Zoek twee verschillende manieren.
\end{exercise}
