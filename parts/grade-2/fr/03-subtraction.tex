\chapter{Soustraction}\label{ch:g2:subtraction}

Quand les unités du haut sont trop peu nombreuses pour enlever,
on défait une dizaine --- c'est l'\emph{emprunt}, le miroir de la
retenue. Ce chapitre l'explique avec des bâtonnets, puis
entraîne la soustraction écrite, avec le comptage en montant
comme raccourci amical.

\section{Pourquoi on emprunte}

\begin{example}[Défaire une dizaine]\label{ex:g2:subtraction:unbundle}
$42 - 17$~: on veut enlever $7$ unités de $2$ unités --- trop
peu~! Alors délier une des $4$ dizaines en dix unités isolées~:
maintenant le haut a $3$ dizaines et $12$ unités. Enlever~:
$12 - 7 = 5$ unités, et $3 - 1 = 2$ dizaines. Donc
$42 - 17 = 25$.
\end{example}

\section{La technique écrite}

\begin{method}[Soustraction en colonnes]\label{met:g2:subtraction:column}
\begin{enumerate}
  \item Écrire le plus grand nombre en haut, aligné à droite~;
  \item soustraire les unités~; si le chiffre du haut est trop
        petit, emprunter une dizaine (elle devient dix unités dans
        la colonne en cours)~;
  \item soustraire les dizaines, puis les centaines~;
  \item \emph{vérifier}~: le résultat plus le nombre enlevé
        doit redonner le nombre du haut.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g2:subtraction:worked}
Calculer $503 - 168$~:
\[
\begin{array}{r}
5\,0\,3 \\
-\ 1\,6\,8 \\
\hline
3\,3\,5
\end{array}
\]
Unités~: $3 - 8$ trop petit~; emprunter à travers les dizaines
(il n'y en a pas, donc emprunter d'abord aux centaines)~: le haut
devient $4$ centaines, $9$ dizaines, $13$ unités. Puis
$13 - 8 = 5$~; $9 - 6 = 3$~; $4 - 1 = 3$. Vérification~:
$335 + 168 = 503$. \checkmark
\end{example}

\section{Compter en montant}

\begin{method}[Soustraire en comptant en montant]\label{met:g2:subtraction:countup}
Quand les nombres sont proches, monter du plus petit au plus grand
et additionner les sauts~:
\[
83 - 78: \quad 78 \to 80 \text{ c'est } 2, \ 80 \to 83 \text{ c'est } 3,
\text{ donc } 83 - 78 = 5 .
\]
Compter en montant est plus rapide qu'emprunter quand les deux
nombres sont voisins --- et c'est exactement comment un commerçant
rend la monnaie.
\end{method}

{\small $83 - 78$ en comptant en montant~: deux sauts, $2 + 3 = 5$.}
\end{omfigure}

\begin{example}[Rendre la monnaie]\label{ex:g2:subtraction:change}
Un jouet coûte $34$~; on paie avec un billet de $50$. Monnaie~:
compter en montant, $34 \to 40$ c'est $6$, $40 \to 50$ c'est $10$,
donc la monnaie est $16$. La soustraction $50 - 34 = 16$, faite à
la manière du commerçant.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:1}
Calcule (sans emprunt)~: $58 - 23$~; \quad $176 - 45$~; \quad
$389 - 164$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:2}
Calcule avec un emprunt~: $52 - 17$~; \quad $73 - 28$~; \quad
$60 - 24$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:3}
Calcule en colonnes et vérifie chaque résultat~: $324 - 167$~;
\quad $500 - 236$~; \quad $703 - 158$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:4}
Calcule en comptant en montant~: $62 - 58$~; \quad $91 - 85$~; \quad
$100 - 96$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:5}
Calcule la monnaie (compte en montant)~: payer $50$ pour un livre à
$42$~; payer $20$ pour un jouet à $13$~; payer $100$ pour un sac à
$75$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:6}
\guillemotleft{} Un arbre mesure $156$~cm de haut. Il a grandi de
$28$~cm cette année. Quelle était sa hauteur l'an dernier~?
\guillemotright{} Écris la soustraction et la réponse.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:7}
Complète avec l'addition jumelle pour vérifier~:
$85 - 39 = 46$~; est-ce juste~? $214 - 68 = 156$~; est-ce juste~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:8}
Pour chaque histoire, choisis $+$ ou $-$~:
\guillemotleft{} $47$ oiseaux, $19$ s'envolent \guillemotright{}~;
\guillemotleft{} $47$ pommes, $19$ de plus cueillies
\guillemotright{}~; \guillemotleft{} Sam a $47$, Léa a $19$~:
combien de plus a Sam~? \guillemotright{}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:subtraction:9}
Remplis le trou~: $63 - \;?\; = 45$. (Compte de $45$ à $63$ en
montant.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:subtraction:10}
Un jeu~: pense à un nombre, ajoute $30$, puis enlève $18$. Le
résultat est $58$. Quel était le nombre~? (Remonte en arrière~!)
\end{exercise}
