\chapter{Les nombres jusqu'à 1000}\label{ch:g2:numbers}

L'année~1 a regroupé les unités en dizaines
(\cref{ch:g1:tens})~; maintenant dix dizaines se regroupent en une
\emph{centaine}, et les nombres s'étendent jusqu'à $1000$. Trois
chiffres, trois rôles~: centaines, dizaines, unités.

\section{Centaines, dizaines, unités}

\begin{definition}[Trois places]\label{def:g2:numbers:places}
Une \emph{centaine} c'est dix dizaines, c'est-à-dire dix paquets de
dix. Le nombre $346$ a trois parties~:
\[
346 = 3 \text{ centaines} + 4 \text{ dizaines} + 6 \text{ unités}
= 300 + 40 + 6 .
\]
\end{definition}

\begin{example}[Le zéro garde la place]\label{ex:g2:numbers:zero}
\guillemotleft{} Trois cent cinq \guillemotright{} s'écrit $305$~:
trois centaines, \emph{pas} de dizaine, cinq unités. Le $0$ garde
le $5$ à la place des unités. Sans lui, $35$ serait un nombre
complètement différent.
\end{example}

\begin{example}[Échanges]\label{ex:g2:numbers:exchange}
Dix unités font une dizaine, dix dizaines font une centaine. Donc
$10$ dizaines $= 100$, et $2$ centaines $= 20$ dizaines. Le nombre
$250$ contient $25$ dizaines, pas seulement le chiffre $5$.
\end{example}

\section{Comparer et ranger}

\begin{method}[Comparer des nombres à trois chiffres]\label{met:g2:numbers:compare}
\begin{enumerate}
  \item Plus de chiffres l'emporte~: $103 > 98$~;
  \item même longueur~: comparer de gauche à droite --- d'abord
        les centaines, puis les dizaines, puis les unités. La
        première \omterm{ex:g1:subtraction:difference}{différence} décide~: $472 > 468$ car aux
        dizaines, $7 > 6$.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g2:numbers:order}
Ranger $530$, $305$, $503$~: d'abord les centaines ($5$, $3$, $5$),
donc $305$ est le plus petit~; entre $530$ et $503$, les dizaines
décident ($3 > 0$)~:
\[
305 < 503 < 530 .
\]
\end{example}

\section{Sauts sur la droite numérique}

\begin{method}[Ajouter des dizaines et des centaines]\label{met:g2:numbers:jumps}
Sur la droite~:
\begin{enumerate}
  \item $+1$ / $-1$~: changer les unités~;
  \item $+10$ / $-10$~: changer les dizaines ($362 + 10 = 372$)~;
  \item $+100$ / $-100$~: changer les centaines
        ($362 + 100 = 462$).
\end{enumerate}
Un seul chiffre bouge à chaque fois --- sauf si une place
\guillemotleft{} se remplit \guillemotright{} et déborde, comme
$195 + 10 = 205$.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g2:numbers:1}
Combien de centaines, de dizaines et d'unités dans~: $258$~;
\ $470$~; \ $600$~; \ $909$~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:2}
Écris le nombre~: $4$ centaines, $0$ dizaine, $7$ unités~; \
$6$ centaines et $3$ dizaines~; \ dix centaines.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:3}
Écris en chiffres~: deux cent seize~; \ cinq cent quatre~; \
sept cent quatre-vingt-dix.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:4}
Décompose comme dans \cref{def:g2:numbers:places}~: $438$~; \
$702$~; \ $560$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:5}
Recopie et complète avec $<$ ou $>$~:
\[
327 \;?\; 372, \qquad
580 \;?\; 508, \qquad
199 \;?\; 200, \qquad
640 \;?\; 604 .
\]
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:6}
Range du plus petit au plus grand~: $430$~; \ $304$~; \ $340$~; \
$403$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:7}
Calcule en sautant sur la droite~: $253 + 10$~; \quad
$253 + 100$~; \quad $480 - 10$~; \quad $716 - 100$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:8}
Continue~: $195, 205, 215, ?, ?$. \quad
$300, 400, 500, ?, ?$. \quad Attention au premier~!
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:numbers:9}
Un magasin a $6$ boîtes de $100$ crayons et $5$ crayons isolés.
Combien de crayons en tout~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:numbers:10}
Avec les chiffres $2$, $5$ et $8$ utilisés une seule fois chacun,
écris le plus grand nombre à trois chiffres et le plus petit.
Quelle est la distance entre eux~? (Tu peux les soustraire sur la
droite par sauts.)
\end{exercise}
