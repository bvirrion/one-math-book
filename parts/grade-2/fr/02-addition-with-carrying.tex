\chapter{Addition avec retenue}\label{ch:g2:addition}

Quand les unités d'une \omterm{def:g1:addition:def}{somme} dépassent neuf, dix d'entre elles se
regroupent en une nouvelle dizaine --- et ce petit paquet, déplacé
dans la colonne suivante, c'est la célèbre \emph{retenue}. Ce
chapitre explique la retenue avec des bâtonnets, puis entraîne
la technique écrite.

\section{Pourquoi on retient}

\begin{example}[Regrouper dix unités]\label{ex:g2:addition:bundle}
$27 + 15$~: rassembler les unités, $7 + 5 = 12$ unités. Mais
dix unités font une dizaine~! En regrouper dix~: il reste $2$
unités isolées et une nouvelle dizaine à retenir. Maintenant
les dizaines~: $2 + 1 + 1 = 4$ dizaines. Donc $27 + 15 = 42$.
\end{example}

\begin{tikzpicture}[scale=0.75]

  \foreach \i in {0,...,9} \draw[omThm, very thick]
    (0.24*\i,0) -- (0.24*\i,1.7);
  \draw[omExo, thick, rounded corners=3pt]

  \node[below, font=\small] at (1.08,-0.25) {$10$ regroupées};

    (4.0+0.4*\i,0) -- (4.0+0.4*\i,1.7);

  \node[right, font=\small, align=left] at (5.5,0.85)

\end{tikzpicture}

{\small La retenue est un paquet~: dix unités isolées deviennent
une dizaine, déplacée dans la colonne des dizaines.}
\end{omfigure}

\section{La technique écrite}

\begin{method}[Addition en colonnes]\label{met:g2:addition:column}
\begin{enumerate}
  \item Écrire les nombres l'un sous l'autre, unités sous
        unités, dizaines sous dizaines~;
  \item additionner les unités~; si le résultat dépasse $9$,
        écrire le chiffre des unités et retenir un petit $1$
        au-dessus des dizaines~;
  \item additionner les dizaines (avec la retenue), puis les
        centaines~;
  \item vérifier avec une estimation~: arrondir chaque nombre et
        additionner approximativement.
\end{enumerate}
\end{method}

\begin{example}\label{ex:g2:addition:worked}
Calculer $268 + 156$~:
\[
\begin{array}{r}
{\scriptstyle 1}\ {\scriptstyle 1}\ \phantom{0} \\[-3pt]
2\,6\,8 \\
+\ 1\,5\,6 \\
\hline
4\,2\,4
\end{array}
\]
Unités~: $8 + 6 = 14$~: écrire $4$, retenir $1$. Dizaines~:
$6 + 5 + 1 = 12$~: écrire $2$, retenir $1$. Centaines~:
$2 + 1 + 1 = 4$. Estimation de contrôle~: environ
$270 + 160 = 430$, proche de $424$. \checkmark
\end{example}

\section{Additionner de tête}

\begin{example}[Faire une dizaine]\label{ex:g2:addition:mental}
Pour les petites \omterm{def:g1:addition:def}{sommes}, sauter d'abord vers un nombre rond~:
\[
48 + 6 = 48 + 2 + 4 = 50 + 4 = 54 .
\]
Ajouter séparément dizaines et unités~:
$35 + 23 = (30 + 20) + (5 + 3) = 50 + 8 = 58$. Et l'astuce du
presque-cent~: $99 + 46 = 100 + 46 - 1 = 145$.
\end{example}

\begin{example}[Un petit problème]\label{ex:g2:addition:problem}
\guillemotleft{} La bibliothèque a $185$ livres~; $57$ nouveaux
arrivent. Combien de livres maintenant~? \guillemotright{}
Addition~: $185 + 57 = 242$ (unités $5 + 7 = 12$,
retenue\,\dots). La bibliothèque a $242$ livres.
\end{example}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g2:addition:1}
Calcule (sans retenue)~: $34 + 25$~; \quad $126 + 43$~; \quad
$507 + 82$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:2}
Calcule avec une retenue~: $47 + 8$~; \quad $56 + 27$~; \quad
$65 + 19$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:3}
Calcule en colonnes~: $268 + 145$~; \quad $374 + 158$~; \quad
$429 + 293$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:4}
Calcule de tête en faisant une dizaine~: $38 + 5$~; \quad
$67 + 6$~; \quad $54 + 8$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:5}
Calcule en ajoutant séparément dizaines et unités~:
$42 + 36$~; \quad $53 + 24$~; \quad $61 + 28$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:6}
Estime d'abord (arrondis à la dizaine ou à la centaine la plus
proche), puis calcule exactement~: $58 + 34$~; \quad $296 + 187$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:7}
\guillemotleft{} Ana a $128$ billes, Tom en a $95$. Combien en
tout~? \guillemotright{} Écris l'addition et la phrase réponse.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:8}
Complète le trou des unités~: $3\,?\, + 27 = 64$ (trouve le
chiffre des unités manquant du premier nombre).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:addition:9}
Deux classes vont au spectacle~: $27$ et $28$ enfants, plus $6$
adultes. Combien de personnes en tout~? (Ajoute d'abord les deux
classes.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:addition:10}
Trouve trois nombres de la liste $27$, $33$, $40$, $65$ dont la
\omterm{def:g1:addition:def}{somme} fait $100$. (Cherche d'abord deux dont les unités sont des
amis de $10$.)
\end{exercise}
