\chapter{Figures et solides}\label{ch:g2:shapes}

Les figures plates vivent sur le papier~; les solides remplissent les
mains. Ce chapitre trie les figures plates en comptant côtés et
sommets, teste l'angle droit avec un coin de papier plié, et
rencontre les premiers solides --- dont les \omterm{def:g2:shapes:solids}{faces} sont les figures
plates déjà connues.

\section{Trier les figures plates}

\begin{definition}[Côtés droits et ronds]\label{def:g2:shapes:sort}
Une figure dont le bord n'est fait que de côtés droits est un
\emph{polygone}\index{polygone}~: triangle ($3$ côtés), quadrilatère ($4$
côtés~: les carrés et les rectangles sont des
quadrilatères), pentagone ($5$), hexagone ($6$). Une figure au
bord courbé, comme le cercle, n'est \emph{pas} un polygone.
\end{definition}

\begin{method}[Tester un angle droit]\label{met:g2:shapes:rightangle}
Plier une feuille de papier deux fois pour faire un coin net --- ce
coin est un angle droit. Le poser dans le coin d'une figure~:
\begin{enumerate}
  \item si les deux côtés de la figure suivent exactement le coin
        plié, l'angle est droit~;
  \item si le coin est plus ouvert ou plus fermé, ce n'est pas un
        angle droit.
\end{enumerate}
Un carré et un rectangle ont quatre angles droits.
\end{method}

\section{Reproduire sur un quadrillage}

\begin{method}[Copier une figure]\label{met:g2:shapes:copy}
Sur du papier quadrillé, copier une figure sommet par sommet~:
\begin{enumerate}
  \item choisir un sommet et le marquer sur le quadrillage vide~;
  \item compter les cases jusqu'au sommet suivant
        (\guillemotleft{} $3$ à droite, $2$ vers le haut
        \guillemotright{}) et le marquer~;
  \item relier les sommets dans l'ordre~; comparer avec le modèle.
\end{enumerate}
\end{method}

\section{Les solides}

\begin{definition}[Premiers solides]\label{def:g2:shapes:solids}
Les solides occupent de l'espace. Leurs côtés plats sont des
\emph{faces}\index{face}~:
\begin{itemize}
  \item le \emph{cube}\index{cube}~: $6$ faces carrées (un dé)~;
  \item le \emph{pavé} (parallélépipède rectangle)~: $6$ faces
        rectangulaires (une boîte de céréales)~;
  \item le \emph{cylindre}\index{cylindre}~: deux faces rondes et un côté enroulé
        (une boîte de conserve)~;
  \item la \emph{boule}~: parfaitement ronde, pas de face plate.
\end{itemize}
Les solides et leurs faces reviennent dans \cref{ch:g5:solids}.
\end{definition}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:g2:shapes:1}
Pour chacune, compte côtés et sommets et nomme-la~: une figure
avec $5$ côtés~; une figure avec $3$ sommets~; une figure ronde.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:2}
Classe en \guillemotleft{} \omterm{def:g2:shapes:sort}{polygone} \guillemotright{} ou
\guillemotleft{} pas un \omterm{def:g2:shapes:sort}{polygone} \guillemotright{}~: un carré~; un
cercle~; un triangle~; un hexagone.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:3}
Fabrique un testeur d'angle droit en pliant du papier. Trouve trois
angles droits dans la pièce, et un angle qui n'est pas droit.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:4}
Combien d'angles droits a un carré~? Un rectangle~? Un triangle
peut en avoir au plus combien~? (Essaie d'en dessiner un avec
deux.)
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:5}
Sur du papier quadrillé, copie un rectangle de $4$ cases de large
et $2$ de haut, puis un triangle avec une base de $3$ cases.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:6}
Nomme le solide~: un dé~; une boîte de conserve de soupe~; un
ballon de football~; une boîte à chaussures.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:7}
Combien de \omterm{def:g2:shapes:solids}{faces} a un \omterm{def:g2:shapes:solids}{cube}~? Quelle figure est chaque \omterm{def:g2:shapes:solids}{face}~? Combien
de \omterm{def:g2:shapes:solids}{faces} a un pavé~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:8}
Quels solides peuvent rouler~: le \omterm{def:g2:shapes:solids}{cube}, le pavé, la boule, le
\omterm{def:g2:shapes:solids}{cylindre}~? Pourquoi~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:g2:shapes:9}
Une figure a $6$ côtés. Comment s'appelle-t-elle~? Combien de
sommets a-t-elle~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:g2:shapes:10}
Sur du papier quadrillé, dessine une maison~: un carré pour le
mur ($4 \times 4$) avec un toit triangulaire au-dessus. Combien de
côtés et de sommets a le contour entier de la maison~?
\end{exercise}
