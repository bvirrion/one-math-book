\section*{第 \ref{ch:g5:division} 章 --- 除法与倍数}

\begin{solution}{exo:g5:division:1}
$851 \div 23$：$23$ 的倍数为 $23, 46, 69, 92, 115, 138, 161,
184, 207$。再算 $85 \div 23$：商 $3$（$69$），余数 $16$；
落下 $1$：$161 \div 23 = 7$ 恰好整除。商为 $37$，
余数为 $0$（$23 \times 37 = 851$）。

$704 \div 32$：$70 \div 32$ 商 $2$（$64$），余数 $6$；落
下 $4$：$64 \div 32 = 2$。商为 $22$，余数为 $0$。
\end{solution}

\begin{solution}{exo:g5:division:2}
$9 \div 2 = 4.5$; \quad $27 \div 4 = 6.75$; \quad
$33 \div 5 = 6.6$; \quad $21 \div 8 = 2.625$.
\end{solution}

\begin{solution}{exo:g5:division:3}
$54 \div 4 = 13.5$：每人付 $13.50$。
\end{solution}

\begin{solution}{exo:g5:division:4}
$10 \div 3 = 3.333\dots$：数字 $3$ 无限重复，
除法永不结束。精确到百分位：$3.33$。
\end{solution}

\begin{solution}{exo:g5:division:5}
$7$ 的倍数（不超过 $70$）有：$7, 14, 21, 28, 35, 42, 49, 56, 63,
70$。$18$ 的因数有：$1, 2, 3, 6, 9, 18$。$24$ 的因数有：
$1, 2, 3, 4, 6, 8, 12, 24$。
\end{solution}

\begin{solution}{exo:g5:division:6}
$470$：能被 $2$ 整除（末位是 $0$），能被 $5$、$10$ 整除；数字和为 $11$，
不能被 $3$ 整除。

$735$：末位是 $5$，能被 $5$ 整除，不能被 $2$ 和 $10$ 整除；数字和为 $15$，
能被 $3$ 整除。

$8\,001$：是奇数；数字和为 $9$，只能被 $3$ 整除。

$1\,314$：是偶数；数字和为 $9$，能被 $2$ 和 $3$ 整除（因而也能被 $6$ 整除），
不能被 $5$ 整除。
\end{solution}

\begin{solution}{exo:g5:division:7}
同时能被 $3$ 和 $5$ 整除，即能被 $15$ 整除：在 $60$ 与
$80$ 之间的数为 $60$、$75$。
\end{solution}

\begin{solution}{exo:g5:division:8}
$7.2 \div 6 = 1.2$：每段长 $1.2$ m（检验：
$1.2 \times 6 = 7.2$）。
\end{solution}

\begin{solution}{exo:g5:division:9}
$1\,000 = 24 \times 41 + 16$：装满 41 袋，
还剩 $16$ 颗弹珠。
\end{solution}

\begin{solution}{exo:g5:division:10}
$7 \div 5 = 1.4$：$5$ 位朋友每人分得 $1.4$ 块 ---
可行，把每块切成十分之一。对于 $5 \div 7$：
商为 $0.714285\dots$，永不结束 --- 因此不存在精确的小数份额，
只有分数 $\frac57$ 能精确地表示它。
\end{solution}

\begin{solution}{exo:g5:division:11}
数字和为 $5 + 2 + d = 7 + d$。要能被 $3$ 整除，$7 + d$ 须为
$9$、$12$ 或 $15$：$d = 2$、$5$ 或 $8$。最小的是 $d = 2$
（得 $522 = 3 \times 174$）。
\end{solution}
