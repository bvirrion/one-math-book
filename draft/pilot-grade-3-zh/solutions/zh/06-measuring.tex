\section*{第 \ref{ch:g3:measure} 章 --- 测量}

\begin{solution}{exo:g3:measure:1}
钉子的宽度：mm。同学的身高：cm（或 m）。操场：m。
两镇之间的距离：km。
\end{solution}

\begin{solution}{exo:g3:measure:2}
$5$ m $= 500$ cm; \quad $70$ mm $= 7$ cm; \quad
$4$ km $= 4\,000$ m; \quad $600$ cm $= 6$ m.
\end{solution}

\begin{solution}{exo:g3:measure:3}
答案取决于所画的三角形；边界的总长度就是
三条所量边长的和（这个总长称为
\emph{周长}，在\cref{ch:g4:measure}中学习）。
\end{solution}

\begin{solution}{exo:g3:measure:4}
$2$ m $+ 45$ cm $= 200 + 45 = 245$ cm.

$1$ km $+ 250$ m $= 1\,000 + 250 = 1\,250$ m.

$3$ cm $+ 8$ mm $= 30 + 8 = 38$ mm.
\end{solution}

\begin{solution}{exo:g3:measure:5}
$3$ kg $= 3\,000$ g；\quad $2\,500$ g $= 2$ kg 和 $500$ g；\quad
$4$ L $= 400$ cL；\quad $350$ cL $= 3$ L 和 $50$ cL。
\end{solution}

\begin{solution}{exo:g3:measure:6}
$1$ kg $= 1\,000$ g $> 750$ g：是，她够了，并且
还剩 $1\,000 - 750 = 250$ g。
\end{solution}

\begin{solution}{exo:g3:measure:7}
时针在 $2$ 和 $3$ 之间，分针指着 $6$：$2{:}30$。
时针指着 $9$，分针指着 $12$：$9{:}00$。时针接近
$5$，分针指着 $10$：$4{:}50$（“差十分五点” --- 时针
还没有到达 $5$）。
\end{solution}

\begin{solution}{exo:g3:measure:8}
从 $8{:}30$ 到 $9{:}00$：$30$ 分钟；从 $9{:}00$ 到 $11{:}45$：
$2$ 小时 $45$ 分钟。共计：$3$ 小时 $15$ 分钟。
\end{solution}

\begin{solution}{exo:g3:measure:9}
从 $9{:}50$ 到 $10{:}00$：$10$ 分钟，还剩
$45 - 10 = 35$ 分钟（在 $10{:}00$ 之后）。到达时间：$10{:}35$。
\end{solution}

\begin{solution}{exo:g3:measure:10}
$1$ L $= 100$ cL，且 $100 = 20 \times 5$：保罗可以倒满
$5$ 杯。
\end{solution}

\begin{solution}{exo:g3:measure:11}
每天：$600 + 600 = 1\,200$ m。五天共：
$1\,200 \times 5 = 6\,000$ m $= 6$ km.
\end{solution}
