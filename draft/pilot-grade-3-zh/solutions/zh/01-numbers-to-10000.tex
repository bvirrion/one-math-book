\section*{第 \ref{ch:g3:numbers} 章 --- 10\,000 以内的数}

\begin{solution}{exo:g3:numbers:1}
$2\,345$; \quad $6\,050$; \quad $9\,001$.
\end{solution}

\begin{solution}{exo:g3:numbers:2}
$1\,208$：一千二百零八。\quad
$4\,070$：四千零七十。\quad
$9\,999$：九千九百九十九。
\end{solution}

\begin{solution}{exo:g3:numbers:3}
十位数字是 $8$。千位数字是 $7$。共含有
$73$ 个百（因为 $7\,382 = 73$ 个百 $+ 82$，即
$73 \times 100 + 82$）。
\end{solution}

\begin{solution}{exo:g3:numbers:4}
$5\,631 = 5$ 个千 $+ 6$ 个百 $+ 3$ 个十 $+ 1$ 个一。

$2\,046 = 2$ 个千 $+ 0$ 个百 $+ 4$ 个十 $+ 6$ 个一。

$8\,500 = 8$ 个千 $+ 5$ 个百。
\end{solution}

\begin{solution}{exo:g3:numbers:5}
$456 < 465$（十位：$5 < 6$）；\quad
$1\,203 > 989$（四位数比三位数大）；\quad
$6\,090 > 6\,009$（十位：$9 > 0$）；\quad
$9\,999 < 10\,000$.
\end{solution}

\begin{solution}{exo:g3:numbers:6}
$353 < 503 < 530 < 3\,500 < 5\,033$.
\end{solution}

\begin{solution}{exo:g3:numbers:7}
$150$ 恰在 $100$ 与 $200$ 中间；$480$ 恰在
$500$ 之前；$820$ 恰在 $800$ 之后；$990$ 接近 $1\,000$。
\end{solution}

\begin{solution}{exo:g3:numbers:8}
偶数：$34$、$70$、$6\,548$（个位数字分别是 $4$、$0$、$8$）。奇数：$57$、
$2\,481$（个位数字分别是 $7$、$1$）。
\end{solution}

\begin{solution}{exo:g3:numbers:9}
从 $2\,970$ 出发：$2\,980$、$2\,990$、$3\,000$、$3\,010$、$3\,020$。

从 $860$ 出发：$960$、$1\,060$、$1\,160$、$1\,260$。

从 $9\,996$ 出发：$9\,997$、$9\,998$、$9\,999$、$10\,000$ --- 最后一步
跨入了五位数，这正是本章的上限！
\end{solution}

\begin{solution}{exo:g3:numbers:10}
最大的数是 $7\,530$（数字从大到小排列）。最小的数是 $3\,057$
（先放最小的非零数字，其余数字从小到大排列 --- $0$
紧跟在 $3$ 之后）。
\end{solution}

\begin{solution}{exo:g3:numbers:11}
大于 $800$：百位数字是 $8$。偶数：个位数字
必须是 $2$（$2$、$5$、$8$ 各用一次）。所以这个数是
$852$。
\end{solution}
