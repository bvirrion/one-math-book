\section*{第 \ref{ch:g2:measure} 章 --- 钱和度量}

\begin{solution}{exo:g2:measure:1}
铅笔：厘米。教室的长度：米。羽毛：克。书包：千克。水
瓶：升。
\end{solution}

\begin{solution}{exo:g2:measure:2}
$2$ m $= 200$ cm; \quad $300$ cm $= 3$ m; \quad $1$ kg $= 1000$ g.
\end{solution}

\begin{solution}{exo:g2:measure:3}
例如：$16 = 10 + 5 + 1$；\ $23 = 20 + 2 + 1$；\
$35 = 20 + 10 + 5$。（其他方法也可以。）
\end{solution}

\begin{solution}{exo:g2:measure:4}
$10 - 6 = 4$; \quad $20 - 17 = 3$; \quad $50 - 32 = 18$.
\end{solution}

\begin{solution}{exo:g2:measure:5}
分针指向 $6$，时针在 $4$ 和 $5$ 之间：
四点半（$4{:}30$）。分针指向 $3$，时针刚过 $8$：
八点一刻（$8{:}15$）。
\end{solution}

\begin{solution}{exo:g2:measure:6}
两点半：分针指向下方的 $6$，
时针停在 $2$ 和 $3$ 之间。
\end{solution}

\begin{solution}{exo:g2:measure:7}
五月的后一个月是六月。一月的前一个月是十二月。（生日所在的月份因人而异。）
\end{solution}

\begin{solution}{exo:g2:measure:8}
花费：$3 + 4 = 7$。找零：$10 - 7 = 3$。
\end{solution}

\begin{solution}{exo:g2:measure:9}
$1$ kg $= 1000$ g，比 $800$ g 多，糖更
重，多出 $1000 - 800 = 200$ g。
\end{solution}

\begin{solution}{exo:g2:measure:10}
例如 $20 + 10 + 5 + 5 = 40$ 和 $10 + 10 + 10 + 10 = 40$
（恰好四枚硬币或纸币的两种方法。）
\end{solution}
