\section*{第 \ref{ch:g1:counting} 章 --- 数到 20}

\begin{solution}{exo:g1:counting:1}
答案因名字而异；两只手有 $10$ 根手指；两只猫
有 $8$ 条腿。
\end{solution}

\begin{solution}{exo:g1:counting:2}
$7$；\ $12$；\ $20$；\ $0$。用文字写：四；十五；十八。
\end{solution}

\begin{solution}{exo:g1:counting:3}
$8, 9, 10, 11, 12, 13$。倒着数：$15, 14, 13, 12, 11, 10$。
\end{solution}

\begin{solution}{exo:g1:counting:4}
$10$ 的前一个数是 $9$。$16$ 的后一个数是 $17$。$12$ 和
$14$ 之间的数是 $13$。
\end{solution}

\begin{solution}{exo:g1:counting:5}
$6 < 9$；\ $14 > 8$；\ $11 < 13$；\ $20 > 19$。
\end{solution}

\begin{solution}{exo:g1:counting:6}
$6$ 个圆与 $8$ 个叉一一配对后，剩下 $2$ 个叉
没有配对，叉比圆多 $2$ 个。
\end{solution}

\begin{solution}{exo:g1:counting:7}
$2 < 7 < 10 < 12 < 17$。
\end{solution}

\begin{solution}{exo:g1:counting:8}
第二个是鲍勃。第四个是丹，丹站在第 $4$ 个位置。
\end{solution}

\begin{solution}{exo:g1:counting:9}
$13$ 后面数三步是 $16$。$13$ 前面数三步是 $10$。
\end{solution}
